"""
Hybrid optimization combining LP alternating, peaked Newton steps, FISTA proximal
gradient, and simulated annealing to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Improves upon previous LP-only approach by adding spectral-shaped Newton steps
and multi-scale exploration.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def project_simplex(v):
    """Project v onto the non-negative simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - sv / rho > 0
    rho_val = np.max(rho[cond]) if np.any(cond) else 1
    theta = (sv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def peaked_newton_step(a, soft_beta=50.0, step_size=0.1):
    """
    Newton-like step targeting the peak(s) of autocorrelation.
    Uses soft-max weighting for near-max peaks.
    Returns new a (projected onto simplex).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    
    # Soft-max weights over peaks
    if soft_beta > 0:
        weights = np.exp((conv - max_conv) * soft_beta)
        weights = weights / np.sum(weights)
    else:
        # Hard: only the max peak
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0

    # Compute gradient: grad[i] = 2 * sum_k weights[k] * a[k-i]
    grad = np.zeros(n)
    for k in range(len(weights)):
        if weights[k] < 1e-10:
            continue
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * weights[k] * a[j]

    # Newton direction: the Hessian is approximately a weighted sum of permutation matrices
    # Newton step: d[i] = grad[i] - mean(grad) to maintain sum constraint
    d = grad - np.mean(grad)
    
    a_new = a - step_size * d
    a_new = project_simplex(a_new)
    return a_new


def fista_optimize(a, n_iters=500, time_budget=100.0, soft_beta=30.0):
    """
    FISTA (accelerated proximal gradient) for min max(conv(a,a)) s.t. a in simplex.
    """
    n = len(a)
    a = a / np.sum(a)
    a_curr = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Initial step size
    step_size = 1.0 / (2.0 * np.max(a)) if np.max(a) > 0 else 0.1
    t_k = 1.0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient at a_curr
        conv = compute_conv(a_curr)
        max_conv = np.max(conv)
        
        # Soft-max weights
        weights = np.exp((conv - max_conv) * soft_beta)
        w_sum = np.sum(weights)
        if w_sum > 0:
            weights = weights / w_sum
        else:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        
        # Gradient: grad[i] = 2 * sum_k weights[k] * a_curr[k-i]
        grad = np.zeros(n)
        for k in range(len(weights)):
            if weights[k] < 1e-12:
                continue
            w = weights[k]
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * w * a_curr[j]
        
        # FISTA update
        a_trial = a_curr - step_size * grad
        a_new = project_simplex(a_trial)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        a_curr = a_new + ((t_k - 1.0) / t_new) * (a_new - a_prev)
        a_curr = np.maximum(a_curr, 0)
        s = np.sum(a_curr)
        if s > 1e-10:
            a_curr = a_curr / s
        else:
            a_curr = a_new.copy()
        
        a_prev = a_new
        t_k = t_new
        
        # Adaptive step size
        obj_curr = compute_obj(a_new)
        if obj_curr < best_obj:
            best_obj = obj_curr
            best_a = a_new.copy()
            step_size *= 1.05
        else:
            step_size *= 0.9

    return best_a

def simulated_annealing(a, time_budget=60.0, rng=None):
    """
    Simulated annealing: perturb the sequence and accept based on energy.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    n = len(a)
    a = a.copy()
    s = float(np.sum(a))

    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    start = time.monotonic()
    
    T = 0.002
    T_min = T * 0.05
    cooling = 0.997
    
    while time.monotonic() - start < time_budget:
        # Generate perturbation: random pair swap + noise
        perturbation_type = rng.integers(0, 3)
        a_trial = a.copy()
        
        if perturbation_type == 0:
            # Pair move: move mass from one position to another
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = a[i] * rng.uniform(0.01, 0.15)
                a_trial[i] -= delta
                a_trial[j] += delta
        elif perturbation_type == 1:
            # Gaussian noise on random subset
            idx = rng.choice(n, size=max(1, n // 10), replace=False)
            a_trial[idx] += rng.normal(0, 0.01, len(idx))
        else:
            # Local smoothing: average with neighbors
            i = rng.integers(1, n - 1)
            w = rng.uniform(0.1, 0.4)
            a_trial[i] = (1 - w) * a_trial[i] + w * 0.5 * (a_trial[i-1] + a_trial[i+1])
        
        a_trial = np.maximum(a_trial, 0)
        s_trial = float(np.sum(a_trial))
        if s_trial < 1e-10:
            continue
        a_trial = a_trial / s_trial
        
        obj_trial = compute_obj(a_trial)
        delta_obj = obj_trial - cur_obj
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            a = a_trial
            cur_obj = obj_trial
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        
        T = max(T * cooling, T_min)
    
    return best_a

                    break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def nested_optimize(a, time_budget=100.0):
    """
    Run multiple optimization methods in sequence, keeping the best result.
    """
    n = len(a)
    a = a.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Step 1: LP alternating
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        alloc = min(remaining * 0.4, 40.0)
        a_lp = lp_alternating(a, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
        a = best_a
    
    # Step 2: Newton polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_newt = newton_polish(a, n_iters=300, time_budget=alloc)
        obj_newt = compute_obj(a_newt)
        if obj_newt < best_obj:
            best_obj = obj_newt
            best_a = a_newt.copy()
    
    # Step 3: FISTA
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        alloc = min(remaining * 0.2, 25.0)
        a_fista = fista_optimize(a, n_iters=1000, time_budget=alloc, soft_beta=25.0)
        obj_fista = compute_obj(a_fista)
        if obj_fista < best_obj:
            best_obj = obj_fista
            best_a = a_fista.copy()
    
    # Step 4: Final LP round
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        alloc = min(remaining * 0.3, 25.0)
        a_lp2 = lp_alternating(a, time_budget=alloc)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
    
    # Step 5: Final Newton
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.4, 20.0)
        a_final_newt = newton_polish(a, n_iters=500, time_budget=alloc)
        obj_final = compute_obj(a_final_newt)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final_newt.copy()
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, peaked Newton, FISTA,
    and simulated annealing for minimizing the autocorrelation peak objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    n_parent = len(best_seq)
    
    # === Phase 1: Intensive LP alternating on parent ===
    print(f"\n=== Phase 1: LP Alternating (n={n_parent}) ===")
    phase1_budget = min(350.0, (deadline - start_time) * 0.3)
    a = best_seq.copy()
    
    for round_idx in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(phase1_budget / (3 - round_idx), remaining * 0.5)
        a_new = lp_alternating(a, time_budget=alloc, verbose=(round_idx == 0))
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP Round {round_idx}: obj={best_score:.8f} ***")
        else:
            print(f"  LP Round {round_idx}: obj={obj:.8f}")
    
    # === Phase 2: Newton + FISTA polishing ===
    print(f"\n=== Phase 2: Newton + FISTA Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 100.0)
        a_poly = newton_polish(a, n_iters=500, time_budget=alloc * 0.5)
        obj_newt = compute_obj(a_poly)
        if obj_newt < best_score:
            best_score = obj_newt
            best_seq = a_poly.copy()
            a = a_poly.copy()
            print(f"  Newton: obj={best_score:.8f} ***")
        else:
            print(f"  Newton: obj={obj_newt:.8f}")
        
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.15, 50.0)
            a_fista = fista_optimize(a, n_iters=2000, time_budget=alloc, soft_beta=20.0)
            obj_fista = compute_obj(a_fista)
            if obj_fista < best_score:
                best_score = obj_fista
                best_seq = a_fista.copy()
                a = a_fista.copy()
                print(f"  FISTA: obj={best_score:.8f} ***")
            else:
                print(f"  FISTA: obj={obj_fista:.8f}")
    
    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_candidates = sorted(set([
        n_parent - 20, n_parent - 10, n_parent,
        n_parent + 10, n_parent + 20, n_parent + 40
    ]))
    n_candidates = [n for n in n_candidates if 500 <= n <= 1200]
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        alloc = min(remaining * 0.2, 60.0)
        a_opt = nested_optimize(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} ***")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 4: Simulated annealing for exploration ===
    print(f"\n=== Phase 4: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Different perturbation strategies
        init_type = r % 2
        if init_type == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.05, n_best))
        else:
            # Structured perturbation
            a_r = best_seq.copy()
            n_perturb = n_best // 5
            idx = rng.choice(n_best, size=n_perturb, replace=False)
            a_r[idx] += rng.normal(0, 0.005, n_perturb)
        
        a_r = np.maximum(a_r, 0)
        s = float(np.sum(a_r))
        if s > 1e-10:
            a_r = a_r / s
        
        alloc = min(remaining * 0.15, 40.0)
        a_sa = simulated_annealing(a_r, time_budget=alloc, rng=rng)
        obj_sa = compute_obj(a_sa)
        
        # Follow up with local optimization if SA found something good
        if obj_sa < best_score + 0.005:
            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc2 = min(remaining * 0.1, 30.0)
                a_follow = nested_optimize(a_sa, time_budget=alloc2)
                obj_follow = compute_obj(a_follow)
                if obj_follow < obj_sa:
                    a_sa = a_follow
                    obj_sa = obj_follow
        
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA restart {r}: obj={best_score:.8f} ***")
        else:
            print(f"  SA restart {r}: obj={obj_sa:.8f}")
    
    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_final = best_seq.copy()
        
        # Multiple LP rounds with decreasing budget
        for lp_round in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.25, 40.0)
            a_final = lp_alternating(a_final, time_budget=alloc, verbose=True)
            obj = compute_obj(a_final)
            if obj < best_score:
                best_score = obj
                best_seq = a_final.copy()
                print(f"  Final LP {lp_round}: obj={best_score:.8f} ***")
            else:
                print(f"  Final LP {lp_round}: obj={obj:.8f}")
        
        # Newton polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.2, 30.0)
            a_newt = newton_polish(a_final, n_iters=500, time_budget=alloc)
            obj_newt = compute_obj(a_newt)
            if obj_newt < best_score:
                best_score = obj_newt
                best_seq = a_newt.copy()
                print(f"  Final Newton: obj={best_score:.8f} ***")
            else:
                print(f"  Final Newton: obj={obj_newt:.8f}")
        
        # FISTA with different beta
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.3, 25.0)
            a_fista = fista_optimize(a_final, n_iters=2000, time_budget=alloc, soft_beta=10.0)
            obj_fista = compute_obj(a_fista)
            if obj_fista < best_score:
                best_score = obj_fista
                best_seq = a_fista.copy()
                print(f"  Final FISTA: obj={best_score:.8f} ***")
            else:
                print(f"  Final FISTA: obj={obj_fista:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    # Ensure valid output format
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
