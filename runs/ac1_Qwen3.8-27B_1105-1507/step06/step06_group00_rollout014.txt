```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Enhanced optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses PDLP-accelerated LP alternating, bisection line search, multi-peak LP,
subgradient polishing, and coordinate descent. Explores multiple n values
and uses dual-guided restarts to escape local minima.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        idx = k_end - k_start + 1
        A[k_start:k_end + 1, j] = a[:idx]
    return A


def solve_lp_minpeak(a, threshold_frac=0.0, use_multi_peak=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k (or above threshold),
    sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    conv_a = compute_conv(a_norm)
    max_conv = float(np.max(conv_a))

    if use_multi_peak and threshold_frac > 0:
        threshold = max_conv * threshold_frac
        active = conv_a >= threshold
        A_sub = A[active, :]
    else:
        A_sub = A

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sub @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.PDLP, cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            kwargs = {}
            if solver == cp.PDLP:
                kwargs = {'eps_abs': 1e-9, 'eps_rel': 1e-9, 'max_iter': 1000, 'verbose': False}
            prob.solve(solver=solver, **kwargs)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        m = A_sub.shape[0]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A_sub
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def bisection_line_search(a, g, best_obj, max_iters=30):
    """
    Find optimal t in [0,1] via bisection for a_new = (1-t)*a + t*g.
    Returns (best_t, best_a, best_obj).
    """
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_a = a.copy()
    best_obj_local = best_obj

    # First evaluate at t=1
    a_at_1 = np.maximum(g, 0)
    s1 = float(np.sum(a_at_1))
    if s1 > 1e-10:
        a_at_1 = a_at_1 / s1
        obj_at_1 = compute_obj(a_at_1)
        if obj_at_1 < best_obj_local:
            best_t = 1.0
            best_a = a_at_1.copy()
            best_obj_local = obj_at_1

    for _ in range(max_iters):
        mid = (lo + hi) / 2.0
        a_mid = (1.0 - mid) * a + mid * g
        a_mid = np.maximum(a_mid, 0)
        sm = float(np.sum(a_mid))
        if sm < 1e-10:
            hi = mid
            continue
        a_mid = a_mid / sm
        obj_mid = compute_obj(a_mid)

        if obj_mid < best_obj_local:
            best_t = mid
            best_a = a_mid.copy()
            best_obj_local = obj_mid

        # Determine which half to keep
        a_lo = np.maximum((1.0 - lo) * a + lo * g, 0)
        slo = float(np.sum(a_lo))
        if slo > 1e-10:
            a_lo = a_lo / slo
            obj_lo = compute_obj(a_lo)
        else:
            obj_lo = np.inf

        if obj_mid < obj_lo:
            lo = mid
        else:
            hi = mid

        if hi - lo < 1e-6:
            break

    return best_t, best_a, best_obj_local


def accelerated_subgradient(a, time_budget=60.0, eta0=0.05):
    """
    Averaged subgradient method with Nesterov acceleration.
    Minimizes max_k (a*a)[k] s.t. sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    a_prev = a.copy()
    start = time.monotonic()
    t_acc = 1.0

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov momentum
        if it > 0:
            y = a + (t_acc - 1) / (t_acc + 2) * (a - a_prev)
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy > 1e-10:
                y = y / sy
            else:
                y = a.copy()
        else:
            y = a.copy()

        # Compute subgradient at y
        conv_y = compute_conv(y)
        max_val = float(np.max(conv_y))
        # Active set: peaks within 1% of max
        threshold = max_val * 0.99
        active = np.where(conv_y >= threshold)[0]

        # Subgradient: average over active peaks
        subgrad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    subgrad[i] += y[j]
        subgrad /= len(active)

        # Step size
        eta = eta0 / (1.0 + 0.1 * t_acc)

        # Update
        a_new = y - eta * subgrad
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn > 1e-10:
            a_new = a_new / sn
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a_prev = a.copy()
            a = a_new
        else:
            a = a_prev.copy()

        t_acc += 1

    return best_a


def subgradient_method(a, time_budget=60.0, eta0=0.02):
    """Standard subgradient method (no acceleration) for robustness."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Averaging sequence
    a_avg = a.copy()
    T = 1

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        conv_a = compute_conv(a)
        k_star = int(np.argmax(conv_a))

        # Subgradient at a for peak k_star
        subgrad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                subgrad[i] = a[j]

        # Step
        eta = eta0 / np.sqrt(1.0 + it * 0.01)
        a_new = a - eta * subgrad
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn > 1e-10:
            a_new = a_new / sn
            a = a_new
            a_avg = a_avg * (T / (T + 1)) + a_new / (T + 1)
            T += 1

            obj_check = compute_obj(a_avg)
            if obj_check < best_obj:
                best_obj = obj_check
                best_a = a_avg.copy()

    return best_a

                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

            if not improved:
                break

    return best_a


def pdlp_alternating_optimize(a, time_budget=120.0, verbose=False, multi_peak=False):
    """
    Main LP alternating optimization using PDLP.
    With bisection line search and adaptive multi-peak handling.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_obj = best_obj

    max_iters = 2000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a, threshold_frac=0.95, use_multi_peak=multi_peak)
        if g_lp is None:
            g_lp, M_lp = solve_lp_minpeak(a, threshold_frac=0.0, use_multi_peak=False)
        if g_lp is None:
            break

        # Bisection line search
        t_best, a_best, obj_best = bisection_line_search(a, g_lp, best_obj, max_iters=25)

        if obj_best < best_obj - 1e-12:
            best_obj = obj_best
            best_a = a_best.copy()
            a = a_best.copy()
            stall = 0
            if verbose and it % 50 == 0:
                print(f"          it{it}: obj={best_obj:.8f}, t={t_best:.4f}")
        else:
            stall += 1
            if stall > 30:
                # Perturb to escape
                rng = np.random.default_rng(hash((it, n, stall)) % (2**32))
                noise = rng.normal(0, 0.03 + 0.01 * min(stall // 30, 5), n)
                a_pert = best_a * np.exp(noise)
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj * 1.02:  # Allow small degradation
                        a = a_pert.copy()
                        best_a = a_pert.copy()
                        best_obj = obj_pert
                stall = 0

        # Check convergence
        if it > 0 and abs(last_obj - best_obj) < 1e-10:
            if stall > 50:
                break
        last_obj = best_obj

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, mode=0, rng=None):
    """Generate structured initial sequences."""
    if rng is None:
        rng = np.random.default_rng(0)

    if mode == 0:
        # Uniform
        a = np.ones(n) / n
    elif mode == 1:
        # Triangular (tent)
        mid = n / 2.0
        a = 1.0 - np.abs(np.arange(n) - mid) / mid
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    elif mode == 2:
        # Gaussian bell
        mid = n / 2.0
        sigma = n / 4.0
        a = np.exp(-((np.arange(n) - mid) ** 2) / (2 * sigma ** 2))
        a = a / np.sum(a)
    elif mode == 3:
        # U-shaped (high at edges)
        x = np.linspace(-1, 1, n)
        a = 0.5 + 0.5 * x ** 2
        a = a / np.sum(a)
    elif mode == 4:
        # Random smooth
        freqs = rng.integers(1, 5, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        a = np.ones(n)
        x = np.arange(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x / n + p)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
    elif mode == 5:
        # Log-uniform random
        a = rng.lognormal(0, 0.5, n)
        a = a / np.sum(a)
    else:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / np.sum(a)

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating (PDLP), subgradient methods,
    coordinate descent, multi-n exploration, and structured restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = structured_init(n_init, 0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Intensive LP Alternating on parent ===
    print(f"\n=== Phase 1: LP Alternating (PDLP) on n={n_parent} ===")
    remaining = deadline - time.monotonic()
    phase1_budget = min(500.0, remaining * 0.4)

    a = best_seq.copy()
    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(phase1_budget / (4 - round_idx), remaining * 0.7)
        mp = (round_idx >= 2)  # Use multi-peak in later rounds

        a_new = pdlp_alternating_optimize(a, time_budget=alloc, verbose=True, multi_peak=mp)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Round {round_idx} (mp={mp}): obj={best_score:.8f}")
        else:
            print(f"  Round {round_idx} (mp={mp}): obj={obj:.8f} (no improvement)")

    # Subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_sg = accelerated_subgradient(a, time_budget=min(remaining * 0.1, 30.0))
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  Subgradient: obj={best_score:.8f}")

    # === Phase 2: Multi-n Exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    n_candidates = sorted(set([
        max(200, n_parent - 100), max(200, n_parent - 50),
        n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
        400, 500, 600, 700, 900, 1000, 1200, 1500
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 2000]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.2, min(60.0, max(20.0, 2000.0 / n_try * 5)))
        a_opt = pdlp_alternating_optimize(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_parent = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Structured Restarts ===
    print(f"\n=== Phase 3: Structured Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    restart_configs = [
        (n_best, 1), (n_best, 2), (n_best, 4), (n_best, 5),
        (max(200, n_best - 50), 0), (n_best + 50, 3),
        (n_best, 6), (max(200, n_best - 100), 2),
    ]

    for idx, (n_r, mode) in enumerate(restart_configs):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_r = structured_init(n_r, mode, rng)
        alloc = min(remaining * 0.15, 40.0)
        a_opt = pdlp_alternating_optimize(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            n_best = len(a_opt)
            print(f"  Restart {idx} (n={n_r}, mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {idx} (n={n_r}, mode={mode}): obj={obj_r:.8f}")

    # === Phase 4: Fine Polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP alternating one more time
    if remaining > 30:
        a_final = best_seq.copy()
        alloc_lp = min(remaining * 0.3, 80.0)
        a_lp = pdlp_alternating_optimize(a_final, time_budget=alloc_lp, verbose=True, multi_peak=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = coordinate_descent_polish(a_final, time_budget=min(remaining * 0.3, 40.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a_final = a_cd.copy()
            print(f"  Coord Desc: obj={best_score:.8f}")

    # Another LP round after CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = pdlp_alternating_optimize(a_final, time_budget=min(remaining * 0.4, 50.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  LP after CD: obj={best_score:.8f}")

    # Final coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cd2 = coordinate_descent_polish(best_seq, time_budget=min(remaining * 0.5, 30.0))
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Final CD: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
