"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with multiple step strategies, spectral equalization,

Key improvements: diverse power-law/exponential initializations, variable-M LP,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate_v2(a, max_iters=40, time_budget=80.0):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Try multiple M values for diversity
        g_candidates = []
        for M_factor in [1.0, 1.001, 1.005]:
            M_try = M * M_factor
            tl = max(3, int(remaining * 0.25))
            g = solve_lp(a, M_try, time_limit=tl)
            if g is not None:
                g_candidates.append(g)

        if not g_candidates:
            break

        # Try multiple blending strategies
        best_candidate = None
        best_candidate_obj = best_obj

        for g in g_candidates[:2]:
            # Try multiple t values
            for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
                a_try = normalize((1.0 - t) * a + t * g)
                obj_try = compute_obj(a_try)
                if obj_try < best_candidate_obj:
                    best_candidate_obj = obj_try
                    best_candidate = a_try

        if best_candidate is not None:
            a = best_candidate
            best_a = best_candidate.copy()
            best_obj = best_candidate_obj
        else:
            # Perturb to escape
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03 + 0.01 * (it % 7), n)))

    return best_a


def subgradient_method(a, max_steps=300, time_budget=40.0):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Initial step size
    step_size = 0.05
    decay = 0.98
    momentum = 0.7
    vel = np.zeros(n)

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Find all peaks above threshold (top few)
        threshold = cmax - 0.1 * cmax
        peak_indices = np.where(conv >= threshold)[0]

        # Compute weighted subgradient
        grad = np.zeros(n)
        for k in peak_indices:
            weight = (conv[k] - threshold + 1e-12) / cmax
            for j in range(n):
                k2 = k - j
                if 0 <= k2 < n:
                    grad[j] += weight * a[k2]
                if k2 != j and 0 <= k2 < n:
                    pass  # already counted
            # Handle the diagonal term for k=2j
            for j in range(n):
                if 2 * j == k:
                    grad[j] += weight * a[j]

        # Project to sum=const
        grad = grad - np.mean(grad)

        # Momentum update
        vel = momentum * vel - step_size * grad
        a_new = normalize(a + vel)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= decay
            vel *= 0.5
            if step_size < 1e-8:
                break

    return best_a


def spectral_equalize_v2(a, max_iters=60, time_budget=40.0):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_mean = float(np.mean(P[1:])) if n > 1 else 1.0
        p_max = float(np.max(P))
        if p_max < 1e-30 or p_mean < 1e-30:
            break

        # Boost weak frequency components, suppress strong ones
        # Target: make spectrum more uniform
        ratio = p_mean / (P + 1e-30)
        gamma = 0.1 + 0.2 * (it / float(max_iters))
        P_new = P + gamma * p_mean * (1.0 - P / p_max)

        # Also try: P_new = P ** alpha (compression)
        alpha = 0.7 + 0.2 * (it / float(max_iters))
        P_new2 = (P / p_max) ** alpha * p_max

        # Use whichever gives better result after projection
        phase = np.angle(A)
        candidates = []
        for P_use in [P_new, P_new2]:
            A_new = np.sqrt(np.maximum(P_use, 0)) * np.exp(1j * phase)
            a_try = np.real(ifft(A_new, size))[:n]
            a_try = np.maximum(a_try, 0.0)
            s = float(np.sum(a_try))
            if s > 1e-10:
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                candidates.append((obj_try, a_try))

        if candidates:
            candidates.sort(key=lambda x: x[0])
            best_candidate_obj, best_candidate_a = candidates[0]
            if best_candidate_obj < best_obj:
                best_obj = best_candidate_obj
                best_a = best_candidate_a.copy()
                a = best_candidate_a
            else:
                # Blend
                a = normalize(0.6 * a + 0.4 * best_candidate_a)

    return best_a


def basin_hopping(a, n_restarts=6, time_budget=100.0):
    n = len(a)
    best_a = normalize(a.copy().astype(np.float64))
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    per_restart = time_budget / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - start > time_budget - per_restart * 0.5:
            break

        # Perturb
        strategy = r % 4
        if strategy == 0:
            a_try = best_a * np.exp(rng.normal(0, 0.06, n))
        elif strategy == 1:
            freq = rng.uniform(1.0, 8.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            a_try = best_a * mod
        elif strategy == 2:
            a_rand = rng.dirichlet(np.ones(n) * 0.3)
            mix = rng.uniform(0.15, 0.4)
            a_try = (1 - mix) * best_a + mix * a_rand
        else:
            a_try = best_a.copy()
            # Random block reversal
            blen = int(rng.integers(n // 5, n // 3))
            s_idx = int(rng.integers(0, n - blen))
            a_try[s_idx:s_idx + blen] = a_try[s_idx:s_idx + blen][::-1]

        a_try = normalize(a_try)

        # Optimize
        alloc = min(per_restart * 0.6, 25.0)
        if time.monotonic() + alloc < start + time_budget:
            a_try = lp_alternate_v2(a_try, max_iters=12, time_budget=alloc)

        alloc2 = min(per_restart * 0.4, 15.0)
        if time.monotonic() + alloc2 < start + time_budget:
            a_try = subgradient_method(a_try, max_steps=80, time_budget=alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def make_power_law(n, p):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return normalize(a)


def make_exponential(n, beta, direction=1):
    x = np.arange(n, dtype=np.float64)
    if direction == 1:
        a = np.exp(-beta * x / n)
    else:
        a = np.exp(-beta * (n - 1 - x) / n)
    return normalize(a)


def make_two_bump(n, center1, center2, width):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-((x - center1) / width) ** 2) + np.exp(-((x - center2) / width) ** 2)
    return normalize(a)


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # =========================================================
    # Phase 1: Try diverse initializations at parent n
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Diverse initializations (n={n_parent})")
    print(f"{'='*60}")

    n = n_parent
    initializations = []

    # Parent
    if best_seq is not None:
        initializations.append(('parent', normalize(best_seq.copy())))

    # Power laws
    for p in [-0.3, -0.5, -0.7, 0.3, 0.5, 1.0, -1.0]:
        initializations.append((f'power_{p}', make_power_law(n, p)))

    # Exponentials
    for beta in [1.0, 2.0, 3.0, 5.0]:
        initializations.append((f'exp_{beta}_fwd', make_exponential(n, beta, 1)))
        initializations.append((f'exp_{beta}_rev', make_exponential(n, beta, -1)))

    # Two-bump
    initializations.append(('bump_1', make_two_bump(n, n * 0.2, n * 0.8, n * 0.15)))
    initializations.append(('bump_2', make_two_bump(n, n * 0.3, n * 0.7, n * 0.1)))

    # Random
    rng_init = np.random.default_rng(seed)
    initializations.append(('random', normalize(rng_init.dirichlet(np.ones(n) * 0.5))))

    # Screen initializations: compute objective and keep top ones
    init_objs = []
    for name, a_init in initializations:
        obj_i = compute_obj(a_init)
        init_objs.append((obj_i, name, a_init))

    init_objs.sort(key=lambda x: x[0])
    print("  Top initializations:")
    for obj_i, name, _ in init_objs[:5]:
        print(f"    {name}: {obj_i:.6f}")

    # Optimize top-6 initializations
    top_k = min(6, len(init_objs))
    per_init = min(40.0, (deadline - time.monotonic()) / (top_k + 4))

    for idx in range(top_k):
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        obj_i, name, a_init = init_objs[idx]
        alloc = min(per_init, remaining * 0.15)
        if alloc < 10:
            break

        print(f"  Optimizing {name} (init obj={obj_i:.6f}, alloc={alloc:.1f}s)")
        a_opt = lp_alternate_v2(a_init, max_iters=20, time_budget=alloc * 0.6)
        obj_opt = compute_obj(a_opt)

        if time.monotonic() < deadline - 100:
            a_opt = subgradient_method(a_opt, max_steps=100, time_budget=min(15, alloc * 0.3))
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"    ** New best from {name}: {best_score:.8f} **")
        else:
            print(f"    {name} -> {obj_opt:.8f}")

    # =========================================================
    # Phase 2: Intensive LP on best found
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Intensive LP (n={len(best_seq)})")
    print(f"{'='*60}")

    a = normalize(best_seq.copy())
    for cycle in range(6):
        if time.monotonic() > deadline - 250:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.15, 50.0)

        a_new = lp_alternate_v2(a, max_iters=30, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP cycle {cycle}: {best_score:.8f} **")
        else:
            a = a_new
            print(f"  LP cycle {cycle}: {obj_new:.8f}")

    # =========================================================
    # Phase 3: Spectral + subgradient on best
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Spectral + subgradient polish")
    print(f"{'='*60}")

    a = normalize(best_seq.copy())

    if time.monotonic() < deadline - 200:
        a_spec = spectral_equalize_v2(a, max_iters=50, time_budget=35.0)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: {best_score:.8f} **")
        else:
            print(f"  Spectral: {obj_spec:.8f}")
            a = normalize(0.7 * a + 0.3 * a_spec)

    if time.monotonic() < deadline - 180:
        a_sub = subgradient_method(a, max_steps=200, time_budget=30.0)
        obj_sub = compute_obj(a_sub)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  Subgradient: {best_score:.8f} **")
        else:
            print(f"  Subgradient: {obj_sub:.8f}")

    # Follow-up LP
    if time.monotonic() < deadline - 150:
        a = normalize(best_seq.copy())
        a_new = lp_alternate_v2(a, max_iters=25, time_budget=min(40, deadline - time.monotonic() - 100))
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Post Phase 3 LP: {best_score:.8f} **")
        else:
            print(f"  Post Phase 3 LP: {obj_new:.8f}")

    # =========================================================
    # Phase 4: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_cur = len(best_seq)
    n_options = sorted(set([
        n_cur - 100, n_cur - 50, n_cur - 25,
        n_cur, n_cur + 25, n_cur + 50, n_cur + 100,
        n_cur + 150, n_cur + 200, n_cur + 300
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1200]

    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if n_try == n_cur:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample_sequence(best_seq, n_try))

        alloc = min(remaining * 0.12, 30.0)
        if n_try <= 900 and alloc > 15:
            a_try = lp_alternate_v2(a_try, max_iters=10, time_budget=alloc * 0.5)
        if time.monotonic() < deadline - 30:
            a_try = subgradient_method(a_try, max_steps=60, time_budget=min(10, alloc * 0.4))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 5: Basin hopping
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Basin hopping")
    print(f"{'='*60}")

    a = normalize(best_seq.copy())
    if time.monotonic() < deadline - 60:
        a_bh = basin_hopping(a, n_restarts=6, time_budget=min(80, deadline - time.monotonic() - 40))
        obj_bh = compute_obj(a_bh)
        if obj_bh < best_score:
            best_score = obj_bh
            best_seq = a_bh.copy()
            print(f"  Basin hopping: {best_score:.8f} **")
        else:
            print(f"  Basin hopping: {obj_bh:.8f}")

    # =========================================================
    # Phase 6: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    a_final = normalize(best_seq.copy())
    n_final = len(a_final)

    for polish_round in range(4):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * 0.3, 30.0)

        # LP (most effective)
        if n_final <= 900 and alloc > 12:
            a_pol = lp_alternate_v2(a_final, max_iters=15, time_budget=alloc * 0.5)
        else:
            a_pol = a_final.copy()

        # Subgradient
        if time.monotonic() < deadline - 5:
            a_pol = subgradient_method(a_pol, max_steps=80, time_budget=min(10, alloc * 0.3))

        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            a_final = a_pol
            print(f"  Polish {polish_round}: {best_score:.8f} **")
        else:
            a_final = a_pol
            print(f"  Polish {polish_round}: {obj_pol:.8f}")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
