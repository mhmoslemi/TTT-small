1. Gerchberg-Saxton spectral initialization
2. Nesterov accelerated smooth gradient descent

Key idea: The LP best-response map provides a powerful fixed-point iteration

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    """FFT-based full convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: iteratively flatten the power spectrum."""
    a = np.maximum(init.astype(np.float64), 1e-10)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress dynamic range
        mag_norm = mag / (mag.max() + 1e-30)
        # Power law compression
        power = 0.6
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_alt
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def smooth_max_grad(a, tau):
    """Compute smooth approximation of the gradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.3, lr_end=0.005):
    """Nesterov accelerated smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = beta * (y - a)
        a_new = np.maximum(y - lr * (grad + momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def build_conv_matrix(n, a):
    """Build the sparse convolution matrix where (a*g)(k) = (A @ g)_k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0."""
    n = len(a)
    rows, cols, vals = build_conv_matrix(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

    if step_sizes is None:

    a = a / a.sum()
    n = len(a)

    no_imp = 0

        if time.monotonic() > deadline:

        M = float(b.max())
        
        if M < best_M * (1 - 1e-12):

            no_imp = 0

            stall_count = 0

def resample_seq(src, n):
    """Resample a sequence to length n using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    if src.size < 2:
        return np.full(n, src.mean())
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def fast_conv(a):
    """FFT-based full autocorrelation of a."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton spectral flattening."""
    a = np.maximum(init.astype(np.float64), 1e-10)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        power = 0.5 + 0.3 * beta
        mag_p = mag ** power
        mag_p = mag_p / (mag_p.max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_p
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def smooth_max_grad(a, tau):
    """Smooth gradient of max(conv(a,a)) via softmax."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.3, lr_end=0.005):

    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()

    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0

        a_new = np.maximum(y - lr * (grad + momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:

        a = a_new

def build_conv_matrix(n, a):
    """Build sparse matrix A where (a*g)(k) = (A @ g)_k."""
    rows = np.empty(2 * n * n, dtype=np.int32)
    cols = np.empty(2 * n * n, dtype=np.int32)
    vals = np.empty(2 * n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    return rows[:idx], cols[:idx], vals[:idx]


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0."""
    n = len(a)
    rows, cols, vals = build_conv_matrix(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9, 'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def find_max_step(aa, ag, gg, M, max_bisect=40):
    """Find largest t in [0,1] s.t. max((1-t)^2*aa + 2t(1-t)*ag + t^2*gg) <= M.
    Uses bisection since the objective is convex in t."""
    def peak_at(t):
        val = (1.0 - t) ** 2 * aa + 2.0 * t * (1.0 - t) * ag + t ** 2 * gg
        return float(val.max())

    if peak_at(1.0) <= M * (1 + 1e-12):
        return 1.0

    lo, hi = 0.0, 1.0
    for _ in range(max_bisect):
        mid = (lo + hi) * 0.5
        if peak_at(mid) <= M * (1 + 1e-12):
            lo = mid
        else:
            hi = mid
    return lo

    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        aa = fast_conv(a)
        M = float(aa.max())

        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
            stall = 0
            if verbose and (it % 20 == 0 or it < 10):
                print(f"    LP it={it}: M={M:.10f} J={2*n*M:.8f} t={time.monotonic()-a_t0:.0f}s", flush=True)
        else:
            no_imp += 1

        g = lp_best_response(a, M)
        if g is None:
            if verbose:
                print(f"    LP failed at it={it}", flush=True)
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        ag = cross_conv(a, g)
        gg = fast_conv(g)

        t_best = find_max_step(aa, ag, gg, M)

        if t_best < 1e-10:
            stall += 1
            if stall > 10:
                # Perturbation restart
                pert_scale = 0.01 * (1 + stall * 0.5)
                a_pert = a + pert_scale * rng.standard_normal(n)
                a_pert = np.maximum(a_pert, 1e-6)
                a_pert /= a_pert.sum()
                M_pert = fast_conv(a_pert).max()
                if M_pert < M * 1.01:
                    a = a_pert
                    stall = 0
                    no_imp = 0
                elif stall > 30:
                    break
            continue

        a = (1.0 - t_best) * a + t_best * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        if no_imp > 80:
            break

    return best_a


def run_lp_mirror_aggressive(a, deadline, max_iters=500):
    """LP mirror descent that also tries t=1 directly and uses dual info for restarts."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    stall = 0
    rng = np.random.default_rng(999)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        aa = fast_conv(a)
        M = float(aa.max())

        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
            stall = 0
        else:
            no_imp += 1

        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Try t=1 first (most aggressive)
        a_full = np.maximum(g, 0)
        s_full = a_full.sum()
        if s_full > 1e-10:
            a_full = a_full / s_full
            M_full = float(fast_conv(a_full).max())
            if M_full < M * (1 - 1e-12):
                a = a_full
                continue

        # Convex bisection
        ag = cross_conv(a, g)
        gg = fast_conv(g)
        t_best = find_max_step(aa, ag, gg, M)

        if t_best < 1e-10:
            stall += 1
            if stall > 5:
                # Use a blend of random and structured perturbation
                a_pert = a.copy()
                # Reduce elements that contribute most to peaks
                peak_k = np.argmax(aa)
                j_min = max(0, int(peak_k) - n + 1)
                j_max = min(n - 1, int(peak_k))
                a_pert[j_min:j_max+1] *= 0.95
                a_pert += 0.005 * rng.standard_normal(n)
                a_pert = np.maximum(a_pert, 1e-6)
                a_pert /= a_pert.sum()
                if fast_conv(a_pert).max() < M * 1.001:
                    a = a_pert
                    stall = 0
                elif stall > 25:
                    break
            continue

        a = (1.0 - t_best) * a + t_best * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        if no_imp > 100:
            break

    return best_a


def resample_seq(src, n):
    """Resample sequence to length n."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    if src.size < 2:
        return np.full(n, float(src.mean()))
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    """Main search: minimize J(a) using LP mirror descent with convex line search."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 25.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(1024)
        best.update(parent)

    n_parent = parent.size
    print(f"\nStarting: n={n_parent}", flush=True)

    # === PHASE 1: Quick initialization (60s) ===
    p1_dl = time.monotonic() + 60.0
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Initialization (60s)", flush=True)
    print(f"{'='*50}", flush=True)

    a = parent.copy()
    a = np.maximum(a, 1e-8)
    a = a / a.sum()

    # GS from parent
    a_gs = gs_flat_spectral(a, n_iters=150, beta=0.85)
    best.update(a_gs)

    # Nesterov polish
    a_nest = run_nesterov_smooth(a_gs, p1_dl, max_iters=2000,
                                  tau_start=2.0, tau_end=100.0,
                                  lr_init=0.25, lr_end=0.01)
    best.update(a_nest)

    best.report("Phase 1")

    # === PHASE 2: Main LP mirror descent (bulk of time) ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (main)", flush=True)
    print(f"{'='*50}", flush=True)

    a = best.seq.copy()
    n = a.size

    # Measure LP time
    t0 = time.monotonic()
    aa_test = fast_conv(a)
    M_test = float(aa_test.max())
    g_test = lp_best_response(a, M_test)
    lp_time = time.monotonic() - t0
    print(f"  n={n}, LP time={lp_time:.2f}s", flush=True)

    # Budget: use most of remaining time for LP
    remaining = deadline - time.monotonic()
    reserve = 120.0  # Reserve for other phases
    lp_budget = max(60.0, remaining - reserve)
    max_iters = max(30, int(lp_budget / max(0.5, lp_time * 1.5)))
    print(f"  Budget={lp_budget:.0f}s, max_iters={max_iters}", flush=True)

    # Run main LP
    a_t0 = time.monotonic()
    a_lp = run_lp_mirror(a, time.monotonic() + lp_budget * 0.55,
                         max_iters=max_iters, verbose=True)
    best.update(a_lp)

    # Second LP pass (finer convergence)
    if time.monotonic() < deadline - 80:
        a_lp2 = run_lp_mirror_aggressive(a_lp, min(time.monotonic() + lp_budget * 0.35, deadline - 60),
                                          max_iters=max_iters // 2)
        best.update(a_lp2)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Multi-length exploration ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Multi-length exploration", flush=True)
    print(f"{'='*50}", flush=True)

    n_best = best.seq.size
    for delta in [-128, 128, -256, 256, -64, 64]:
        if time.monotonic() > deadline - 60:
            break
        n_try = n_best + delta
        if n_try < 256 or n_try > 3000:
            continue
        a_r = resample_seq(best.seq, n_try)
        a_r = np.maximum(a_r, 1e-6)
        a_r = a_r / a_r.sum()
        # Quick Nesterov
        a_r = run_nesterov_smooth(a_r, min(time.monotonic() + 20, deadline - 50),
                                   max_iters=800, tau_start=3.0, tau_end=80.0,
                                   lr_init=0.2, lr_end=0.01)
        best.update(a_r)
        # LP polish
        if time.monotonic() < deadline - 40:
            a_lp3 = run_lp_mirror(a_r, min(time.monotonic() + 25, deadline - 20),
                                   max_iters=25, verbose=False)
            best.update(a_lp3)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Final LP polish ===
    if best.seq is not None and time.monotonic() < deadline - 5:
        a_final = run_lp_mirror(best.seq, deadline, max_iters=30, verbose=False)
        best.update(a_final)

    best.report("Phase 4")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
