def fast_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def spectral_init(n, rng, target_flat=True):
    """Initialize a non-negative sequence with approximately flat conv(a,a).
    
    In the frequency domain, F(a)^2 = F(conv(a,a)).
    To make conv(a,a) approximately flat (= constant), we want F(a)^2 ≈ constant.
    So |F(a)| ≈ sqrt(constant) = constant, and phase of F(a) can be random.
    But F(a) must be conjugate symmetric (for real a) and the inverse must be non-negative.
    """
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    # Create flat spectrum with random phase
    # F(a) has conjugate symmetry: F[k] = conj(F[n-k])
    phase = rng.uniform(0, 2*np.pi, size // 2 + 1)
    # Ensure conjugate symmetry: F[0] and F[n] (if n even) are real
    mag = np.ones(size // 2 + 1)
    F = mag * np.exp(1j * phase)
    # Make F[0] real and positive
    F[0] = np.abs(F[0])
    # If size is even, make F[size//2] real
    if size % 2 == 0:
        F[size // 2] = np.abs(F[size // 2])
    
    # Inverse FFT
    a = np.fft.irfft(F, size)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton to flatten conv(a,a)."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(a.size) / a.size
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Compress dynamic range
        mag_norm = mag / (mag.max() + 1e-30)
        # Target: make mag more uniform
        target_mag = np.full_like(mag, mag.mean())
        mag_new = (1 - beta) * mag + beta * target_mag
        
        F_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    
    return best_a


def smooth_max_grad(a, tau):
    """Compute gradient of smooth approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, p, b, b_max


def run_smooth_descent(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                       lr_init=0.3, lr_end=0.005):
    """Nesterov-accelerated smooth max gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new
    
    return best_a

def coord_descent_sweep(a, order=None, verbose=False):
    """One sweep of coordinate descent. Returns improved a and improvement."""
    n = len(a)
    if order is None:
        order = range(n)
    
    a_work = a.copy()
    b = fast_conv(a_work)
    current_max = float(b.max())
    
    for i in order:
        if i >= n:
            break
        # Compute the 1D objective: g(x) = max_k f_k(x)
        # where f_k(x) = conv_k(a) - 2*a_i*a_{k-i} + 2*a_{k-i}*x (k != 2i)
        #              = conv_k(a) - a_i^2 + x^2 (k = 2i)
        
        a_i = a_work[i]
        k_min = i
        k_max = min(i + n - 1, 2 * n - 2)
        
        # For all k in [k_min, k_max]:
        ks = np.arange(k_min, k_max + 1)
        
        # coeff_linear[k] = 2 * a_work[k - i] for k != 2i, 0 for k = 2i
        # coeff_quad[k] = 1 for k = 2i, 0 for k != 2i
        # constant[k] = conv_k(a_work) - 2*a_i*a_{k-i} for k != 2i
        #              = conv_k(a_work) - a_i^2 for k = 2i

        conv_vals = b[k_min:k_max + 1]
        
        # Get a_{k-i} for k in [k_min, k_max]
        # k-i ranges from 0 to min(n-1, 2n-2-i)
        ki_idx = ks - i  # = 0, 1, ..., (k_max - k_min)
        a_ki = a_work[ki_idx]
        
        # Build the 1D functions
        # f_k(x) = c0_k + c1_k * x + c2_k * x^2
        c2 = np.zeros(len(ks))
        c1 = 2.0 * a_ki
        c0 = conv_vals - 2.0 * a_i * a_ki

def find_optimal_coord(c0, c1, c2, current_max, x_init):

        # Vectorized: compute all f_k(x) and take max
        vals = c0 + c1 * x + c2 * x * x
        return np.max(vals)
    
    # Bounds for x
    # Lower: 0
    # Upper: we can bound by current_max / min(c1[k] for c1[k]>0) approximately
    # Or just use a reasonable upper bound
    x_lo = 0.0
    # Simple upper bound: if all c1 are positive, the max grows with x,
    # so the optimum is bounded. Use 2*x_init + 0.5 as a start.
    x_hi = max(2.0 * x_init + 0.1, 0.5)
    # But don't go too far
    x_hi = min(x_hi, 10.0)

    g_hi = g(x_hi)

        g_hi = g(x_hi)

    b = x_hi
    c = b - golden * (b - a)

    fc = g(c)
    fd = g(d)
    
    for _ in range(50):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - golden * (b - a)
            fc = g(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + golden * (b - a)
            fd = g(d)
    
    x_opt = (a + b) / 2
    return x_opt, g(x_opt)


def coord_descent_sweep_fast(a, order=None):
    """Fast coordinate descent sweep using golden section search."""
    n = len(a)
    if order is None:
        order = np.arange(n)
    
    a_work = a.copy()
    b = fast_conv(a_work)
    current_max = float(b.max())
    best_max = current_max
    
    for i in order:
        a_i = a_work[i]
        k_min = i
        k_max = min(i + n - 1, 2 * n - 2)
        ks = np.arange(k_min, k_max + 1)
        
        # a_{k-i} for k in [k_min, k_max]
        ki_idx = ks - i  # 0 to (k_max - k_min)
        a_ki = a_work[ki_idx]
        
        conv_vals = b[k_min:k_max + 1]
        
        # Build 1D function coefficients
        c0 = conv_vals - 2.0 * a_i * a_ki
        c1 = 2.0 * a_ki
        c2 = np.zeros(len(ks))
        
        # Handle k = 2i case
        idx_2i = (ks == 2 * i)
        if np.any(idx_2i):
            c0[idx_2i] = conv_vals[idx_2i] - a_i * a_i
            c1[idx_2i] = 0.0
            c2[idx_2i] = 1.0
        
        # Find optimal x
        x_lo = 0.0
        x_hi = max(2.0 * a_i + 0.1, 0.1)
        
        # Quick check: evaluate at a few points
        # Try x = 0, x = a_i, x = a_i/2, x = 2*a_i
        candidates_x = [0.0, a_i, max(0.0, a_i - 0.1 * np.max(a_ki)), min(2*a_i, x_hi)]
        candidates_x = [x for x in candidates_x if 0 <= x <= max(x_hi, 0.01)]
        if not candidates_x:
            candidates_x = [0.0, 0.01]
        
        best_x = a_i
        best_g = float('inf')
        for x in candidates_x:
            vals = c0 + c1 * x + c2 * x * x
            g_val = np.max(vals)
            if g_val < best_g:
                best_g = g_val
                best_x = x
        
        # Refine with golden section if we found improvement
        if best_g < current_max - 1e-12:
            # Golden section around the best candidate
            sa = max(0.0, best_x - 0.5 * a_ki.max())
            sb = min(2 * a_ki.max() + 0.1, best_x + 0.5 * a_ki.max())
            sa = max(sa, 0.0)
            
            golden = (np.sqrt(5) - 1) / 2
            aa, bb = sa, sb
            cc = bb - golden * (bb - aa)
            dd = aa + golden * (bb - aa)
            
            def g_func(x):
                return np.max(c0 + c1 * x + c2 * x * x)
            
            fc = g_func(cc)
            fd = g_func(dd)
            
            for _ in range(30):
                if bb - aa < 1e-10:
                    break
                if fc < fd:
                    bb = dd
                    dd = cc
                    fd = fc
                    cc = bb - golden * (bb - aa)
                    fc = g_func(cc)
                else:
                    aa = cc
                    cc = dd
                    fc = fd
                    dd = aa + golden * (bb - aa)
                    fd = g_func(dd)
            
            x_opt = (aa + bb) / 2
            g_opt = g_func(x_opt)
            
            if g_opt < best_g:
                best_x = x_opt
                best_g = g_opt
        
        if best_g < current_max - 1e-12:
            # Update a_work and b
            delta = best_x - a_i
            a_work[i] = best_x
            
            # Update b
            if np.any(idx_2i):
                b[2 * i] += delta * (best_x + a_i)
            mask = ~idx_2i
            if np.any(mask):
                b[ks[mask]] += 2.0 * delta * a_ki[mask]
            
            new_max = float(np.max(b))
            if new_max < best_max:
                best_max = new_max
            current_max = max(current_max, 0)  # keep for reference
    
    # Renormalize
    s = a_work.sum()
    if s > 1e-15:
        a_work = a_work / s
    
    return a_work, best_max


def run_coord_descent(a, deadline, max_sweeps=5000, restart_every=100,
                      perturb_scale=0.01):
    """Run coordinate descent for the given time budget."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    no_imp = 0
    sweep_count = 0
    order = np.arange(n)
    
    while time.monotonic() < deadline:
        # Shuffle order periodically
        if sweep_count % 10 == 0:
            rng = np.random.default_rng(sweep_count)
            order = rng.permutation(n)
        
        a_new, M_new = coord_descent_sweep_fast(a, order)
        sweep_count += 1
        
        if M_new < best_M - 1e-10:
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        a = a_new
        
        # Periodic perturbation to escape local minima
        if no_imp > restart_every:
            # Add random perturbation
            rng = np.random.default_rng(sweep_count * 7919)
            a_pert = np.maximum(a + rng.normal(0, perturb_scale * np.mean(a), n), 0)
            s = a_pert.sum()
            if s > 1e-15:
                a_pert /= s
            M_pert = fast_conv(a_pert).max()
            if M_pert < best_M:
                M_pert = best_M
            a = a_pert
            no_imp = 0
            perturb_scale *= 0.8  # reduce perturbation over time
        
        # Check time
        if sweep_count > max_sweeps:
            break
    
    return best_a


def build_A_values(n, a):
    """Build sparse matrix for LP constraint (a*g)(k) <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    """LP: maximize sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    rows, cols, vals = build_A_values(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=100, t_base=0.2):
    """LP mirror descent for final polish."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        improved = False
        for t in [t_base, t_base*0.5, t_base*0.25, t_base*0.1, t_base*0.05]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break
    
    return best_a


def resample_seq(src, n):
    """Resample a sequence to length n."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
