"""
Hybrid optimizer for low-autocorrelation non-negative sequences.
Uses smoothed gradient (L-BFGS-B with log param), LP alternating, exponentiated gradient,
and systematic n-exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time


def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    """Convolution a*b via FFT."""
    n, m = len(a), len(b)
    size = next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_and_grad(x, tau):
    """
    Compute smoothed objective and gradient for a = exp(x).
    Objective: logsumexp(conv(a,a)/tau) / (sum(a))^2
    Gradient computed via FFT.
    """
    n = len(x)
    # Normalize x for numerical stability (objective is scale-invariant)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    
    # Convolution
    conv = compute_conv(a)
    
    # Softmax weights
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    
    # Objective value: logsumexp(c/tau) / s^2
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    
    # Gradient: dF/dx_i = 2*corr[i]/(tau*s^2) - 2*F*a_i/s
    # where corr[i] = sum_k w[k]*a[k-i] = cross-correlation
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    """
    Optimize using L-BFGS-B on smoothed objective with adaptive temperature.
    a_init: initial non-negative sequence.
    Returns best a found.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Temperature schedule: start large, decrease
    taus = [10.0 / n, 5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n]
    
    x = np.log(a) - np.log(a).max()
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        
        # Define objective and gradient for scipy
        def fun(xv):
            xv = xv - np.max(xv)  # normalize for stability
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g
        
        # Run L-BFGS-B
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxls': 50})
        
        if res.success or res.fun < fun(x):
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    grad tau={tau:.4f}: obj={best_obj:.8f}")
            else:
                # Keep old x
                if verbose:
                    print(f"    grad tau={tau:.4f}: obj={obj_new:.8f} (no improve)")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, use_cvxpy=True):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    if use_cvxpy:
        A = build_conv_matrix(a_norm)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    
    # Fallback: scipy linprog
    try:
        A = build_conv_matrix(a_norm)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_alternate_one_step(a, time_budget=10.0):
    """One step of LP alternating: solve LP, line search, return improved a."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    g_lp, M_lp = solve_lp_minpeak(a_norm)
    if g_lp is None:
        return a, compute_obj(a)
    
    # Line search
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for t in [0.3, 0.5, 0.7, 0.9, 1.0, 0.1, 0.2, 0.4, 0.6, 0.8]:
        a_try = (1.0 - t) * a_norm + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
    
    return best_a, best_obj


def lp_alternate_optimize(a, time_budget=60.0, verbose=False):
    """
    LP alternating optimization with adaptive blending and restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        a_new, obj_new = lp_alternate_one_step(a, time_budget=min(5.0, time_budget - (time.monotonic() - start)))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                scale = 0.01 + 0.01 * min(stall // 10, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
    
    return best_a


def exp_gradient_optimize(a_init, time_budget=30.0, verbose=False):
    """
    Exponentiated gradient (multiplicative weights) for the peak objective.
    Very fast per iteration, good for diversity.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 1.0 / n
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Gradient of max conv w.r.t. a[i] = 2*a[k_star - i] (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Exponentiated gradient update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        s = float(np.sum(a_new))
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Slowly decrease eta
            eta = max(eta * 0.999, 0.1 / n)
        else:
            # Try smaller step
            eta *= 0.9
            if eta < 0.01 / n:
                # Reset with perturbation
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                eta = 1.0 / n
    
    return best_a


def symmetric_refine(a, time_budget=30.0):
    """Try enforcing palindromic symmetry and optimize."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    a_sym = a_sym / np.sum(a_sym)
    
    obj_sym = compute_obj(a_sym)
    obj_orig = compute_obj(a)
    
    if obj_sym < obj_orig:
        # Optimize the symmetric version
        a_opt = gradient_optimize(a_sym, time_budget=time_budget * 0.5)
        # Re-enforce symmetry
        a_opt = (a_opt + a_opt[::-1]) / 2.0
        a_opt = np.maximum(a_opt, 0)
        a_opt = a_opt / np.sum(a_opt)
        return a_opt
    return a


def resample_sequence(a, new_n):
    """Resample to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Smooth random (low-pass filtered)
        a = rng.uniform(0, 1, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = a / np.sum(a)
    elif mode == 3:
        # Triangular
        a = 1.0 - np.abs(np.arange(n) - n / 2) / (n / 2)
        a = a + 0.1 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 4:
        # Two peaks
        c1, c2 = n // 3, 2 * n // 3
        a = np.exp(-((np.arange(n) - c1) ** 2) / (n * n / 20)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (n * n / 20))
        a = a / np.sum(a)
    else:
        # Flat with noise
        a = np.ones(n) + 0.3 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid of smoothed gradient, LP alternating, exponentiated gradient,
    multi-n exploration, and symmetric refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")
    
    n_parent = len(best_seq)
    
    # === Phase 1: Gradient optimization on parent ===
    print(f"\n=== Phase 1: Gradient Optimization (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_grad = gradient_optimize(best_seq, time_budget=min(remaining * 0.25, 120.0), verbose=True)
        obj = compute_obj(a_grad)
        if obj < best_score:
            best_score = obj
            best_seq = a_grad.copy()
            print(f"  After gradient: obj={best_score:.10f}")
    
    # === Phase 2: Exponentiated gradient ===
    print(f"\n=== Phase 2: Exponentiated Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_eg = exp_gradient_optimize(best_seq, time_budget=min(remaining * 0.1, 40.0), verbose=True)
        obj = compute_obj(a_eg)
        if obj < best_score:
            best_score = obj
            best_seq = a_eg.copy()
            print(f"  After exp-grad: obj={best_score:.10f}")
    
    # === Phase 3: LP Alternating polish ===
    print(f"\n=== Phase 3: LP Alternating Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_lp = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.2, 120.0), verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  After LP: obj={best_score:.10f}")
    
    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_candidates = sorted(set([
        n_parent - 100, n_parent - 50, n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
        600, 700, 900, 1000, 1200
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 1500]
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        
        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        # Quick gradient optimization
        alloc = min(remaining * 0.15, 40.0)
        a_opt = gradient_optimize(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_parent = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")
    
    # === Phase 5: Intensify on best n ===
    print(f"\n=== Phase 5: Intensify (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # More gradient with finer temperature
        a_g = gradient_optimize(best_seq, time_budget=min(remaining * 0.2, 80.0), verbose=True)
        obj = compute_obj(a_g)
        if obj < best_score:
            best_score = obj
            best_seq = a_g.copy()
            print(f"  Gradient2: obj={best_score:.10f}")
        
        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_lp2 = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.2, 60.0))
            obj = compute_obj(a_lp2)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp2.copy()
                print(f"  LP2: obj={best_score:.10f}")
    
    # === Phase 6: Random restarts with gradient ===
    print(f"\n=== Phase 6: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    restart_budget = min(30.0, remaining * 0.08)
    
    n_restarts = max(3, int(remaining / (restart_budget + 5)))
    n_restarts = min(n_restarts, 10)
    
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < restart_budget + 5:
            break
        
        if r == 0:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.05, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 1:
            # Different n
            n_r = n_best + int(rng.integers(-100, 100))
            n_r = max(400, min(1400, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.03, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 2:
            # Symmetric perturbation
            a_r = best_seq + 0.02 * rng.normal(0, 1, n_best)
            a_r = (a_r + a_r[::-1]) / 2.0
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 3:
            # Cosine window
            x = np.linspace(0, np.pi, n_best)
            a_r = np.cos(x) ** 2 + 0.3 + 0.05 * rng.normal(0, 1, n_best)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            a_r = random_init(n_best, rng)
        
        a_opt = gradient_optimize(a_r, time_budget=restart_budget)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r}: obj={best_score:.10f} **")
        else:
            print(f"  Restart {r}: obj={obj:.10f}")
    
    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Gradient
        alloc = min(remaining * 0.3, 60.0)
        a_f = gradient_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final grad: obj={best_score:.10f}")
        
        # LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_lp = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.3, 40.0))
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  Final LP: obj={best_score:.10f}")
        
        # Symmetric
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sym = symmetric_refine(best_seq, time_budget=min(remaining * 0.3, 20.0))
            obj = compute_obj(a_sym)
            if obj < best_score:
                best_score = obj
                best_seq = a_sym.copy()
                print(f"  Final sym: obj={best_score:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
