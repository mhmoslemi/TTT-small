def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def build_lp_matrix(a, active_indices):

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            m = k - j
            rows.append(idx)
            cols.append(j)
            vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LP-based minimax optimization
with constraint generation, adaptive step sizes, multi-start exploration, and coordinate descent polish.
Key improvements: full constraint LP via HiGHS, smart active set management, perturbation restarts,
and efficient FFT-based convolution.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv_xy(a, x):
    """FFT-based linear convolution of a with x."""
    n = len(a)
    m = len(x)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fx = np.fft.rfft(x, size)
    return np.fft.irfft(fa * fx, size)[:n + m - 1]


def compute_M(a):
    return fast_conv(a).max()


def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2 * len(a) * compute_M(a) / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def build_lp_matrix(a, active_indices, n):
    """Build sparse LP matrix for given active constraint indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def solve_lp(a, active_indices, M, n, timeout=5.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    A = build_lp_matrix(a, active_indices, n)
    if A.nnz == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': timeout, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_with_constraint_generation(a, M, n, n_active_start, n_active_max, deadline, timeout=4.0):
    """LP with iterative constraint generation."""
    b = fast_conv(a)
    
    n_act = min(n_active_start, 2 * n - 1)
    active = list(np.argsort(b)[::-1][:n_act])
    
    g = solve_lp(a, active, M, n, timeout=timeout)
    if g is None:
        return None
    
    # Constraint generation: check violations and add
    for _ in range(15):
        if time.monotonic() > deadline:
            break
        ag = fast_conv_xy(a, g)
        violated = np.where(ag > M * (1 + 1e-5))[0]
        if len(violated) == 0:
            break
        # Add the most violated constraint
        worst = int(violated[np.argmax(ag[violated])])
        if worst not in active:
            active.append(worst)
            if len(active) > n_active_max:
                # Remove the least violated from the old set
                old_ag = ag[np.array(active)]
                remove_idx = int(np.argmin(old_ag))
                active = [x for i, x in enumerate(active) if i != remove_idx]
            g = solve_lp(a, active, M, n, timeout=timeout)
            if g is None:
                return None
    return g


def run_lp_minimax(a_init, deadline, n_active_start=40, n_active_max=250,
                   t_init=0.3, timeout=4.0, seed=0):
    """LP-based minimax optimization with adaptive step size."""
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    t = t_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        b = fast_conv(a)
        M_cur = b.max()
        
        g = lp_with_constraint_generation(a, M_cur, n, n_active_start, n_active_max,
                                          deadline, timeout=timeout)
        if g is None:
            no_improve += 2
            if no_improve > 15:
                break
            continue
        
        # Adaptive line search
        improved = False
        for factor in [t, t * 0.6, t * 0.3, t * 0.15, 0.05, 0.02]:
            a_trial = (1.0 - factor) * a + factor * g
            a_trial = np.maximum(a_trial, 0.0)
            s_tr = a_trial.sum()
            if s_tr < 0.01:
                continue
            a_trial = a_trial / s_tr
            M_trial = compute_M(a_trial)
            if M_trial < best_M * (1.0 - 1e-12):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
            t = min(0.5, t * 1.05)
        else:
            no_improve += 1
            t = max(0.01, t * 0.7)
            if no_improve > 25:
                # Try a perturbation to escape
                if time.monotonic() < deadline - 5:
                    a_pert = a + rng.normal(0, 0.01, n)
                    a_pert = np.maximum(a_pert, 0)
                    a_pert = a_pert / a_pert.sum()
                    if compute_M(a_pert) < best_M:
                        best_M = compute_M(a_pert)
                        best_a = a_pert.copy()
                        a = a_pert
                        no_improve = 0
                        t = 0.2
                    else:
                        break
        
        it += 1
    
    return best_a, best_M


def coordinate_descent_polish(a_init, deadline, seed=0):
    """Coordinate descent with exact 1D optimization."""
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    while time.monotonic() < deadline and no_improve < 10:
        b = fast_conv(a)
        sweep_improved = False
        order = np.arange(n)
        rng.shuffle(order)
        
        for i in order:
            if time.monotonic() > deadline:
                    break
            # Find optimal a[i] given all others
            x_cur = a[i]
            # Affected k range: [i, min(2n-2, i+n-1)]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Compute base values and slopes
            c_vals = []
            slopes = []
            has_quad = False
            quad_pos = -1
            
            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    c = b[k] - x_cur * x_cur
                    c_vals.append(c)
                    slopes.append(0.0)
                    has_quad = True
                    quad_pos = len(c_vals) - 1
                else:
                    j = k - i
                    if 0 <= j < n:
                        c = b[k] - 2.0 * x_cur * a[j]
                        c_vals.append(c)
                        slopes.append(2.0 * a[j])
                    else:
                        c_vals.append(b[k])
                        slopes.append(0.0)
            
            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            
            c_arr = np.array(c_vals)
            s_arr = np.array(slopes)
            
            def obj_at(x_test):
                vals = c_arr + s_arr * x_test
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if has_quad:
                    qv = c_arr[quad_pos] + x_test * x_test
                    if qv > m:
                        m = qv
                return m
            
            # Golden section search
            lo = 0.0
            hi = max(2.0 * x_cur, 0.5, np.sqrt(max(b.max(), 0)) + 0.1)
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_at(a_pt)
            fb = obj_at(b_pt)
            
            for _ in range(30):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_at(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)
            
            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - x_cur) > 1e-12:
                # Update b
                delta = x_opt - x_cur
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += 2 * x_cur * delta + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2 * a[j] * delta
                a[i] = x_opt
                sweep_improved = True
        
        a = a / a.sum()
        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        # Resample parent to n
        t_old = np.linspace(0, 1, parent.size, endpoint=False)
        p = np.maximum(np.interp(t, t_old, parent), 0.0)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed parent
        p_pert = np.maximum(p + rng.normal(0, 0.02, n), 1e-6)
        inits.append(p_pert / p_pert.sum())
        p_pert2 = np.maximum(p + rng.normal(0, 0.05, n), 1e-6)
        inits.append(p_pert2 / p_pert2.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine windows
    for freq in [1, 2, 3]:
        cw = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cw = np.maximum(cw, 0.01)
        inits.append(cw / cw.sum())
    
    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Parabolic (concave)
    para = (t * (1 - t)) * 4 + 0.1
    inits.append(para / para.sum())
    
    # Multi-lobe
    multi = 0.3 + 0.7 * np.abs(np.cos(np.pi * t * 3))
    inits.append(multi / multi.sum())
    
    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_optimization_for_length(n, deadline, rng, parent=None, timeout=4.0):
    """Run full optimization pipeline for a given length."""
    inits = generate_inits(n, rng, parent)
    
    best_a = None
    best_M = float('inf')
    
    # Allocate time per init
    n_inits = len(inits)
    time_per_init = max(10.0, (deadline - time.monotonic()) / n_inits)
    
    for ii, a0 in enumerate(inits):
        sub_deadline = min(deadline, time.monotonic() + time_per_init)
        if time.monotonic() >= sub_deadline:
            break
        
        a0 = np.maximum(a0, 1e-8)
        a0 = a0 / a0.sum()
        
        # LP minimax
        a_lp, M_lp = run_lp_minimax(a0, sub_deadline,
                                     n_active_start=40, n_active_max=200,
                                     t_init=0.3, timeout=timeout,
                                     seed=1000 * n + ii)
        
        if M_lp < best_M:
            best_M = M_lp
            best_a = a_lp.copy()
            J_val = compute_J(best_a)
            print(f"    init {ii}/{n_inits}: M={M_lp:.10f} J={J_val:.6f} n={n}", flush=True)
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_M = compute_M(best_a)
    
    return best_a, best_M

        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
