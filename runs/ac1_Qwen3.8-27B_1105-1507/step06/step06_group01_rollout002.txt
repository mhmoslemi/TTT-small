    cols = []

        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_full_lp_matrix(a):
    """Build sparse matrix for full LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            m = k - j
            if a[m] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(a, M, time_limit=5):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g or None."""
    n = len(a)
    A = build_full_lp_matrix(a)
    if A.nnz == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section_min(f, lo, hi, tol=1e-10, max_iter=60):
    """Golden section search for minimum of unimodal function on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = f(a_pt)
    fb = f(b_pt)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = f(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = f(b_pt)
    return (lo + hi) / 2.0


def line_search_t(a, g, M_cur):
    """Find optimal t in [0,1] minimizing max(conv((1-t)a + t*g, ...))."""
    n = len(a)

    def M_of_t(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-10:
            return float('inf')
        a_t = a_t / s
        return fast_conv(a_t).max()

    # Check endpoints and a few points
    M0 = M_of_t(0.0)
    M1 = M_of_t(1.0)
    M05 = M_of_t(0.5)

    if M0 <= M05 and M0 <= M1:
        return 0.0
    if M1 <= M05 and M1 <= M0:
        return 1.0

    # Golden section on [0, 1]
    t_opt = golden_section_min(M_of_t, 0.0, 1.0, tol=1e-8, max_iter=40)

    # Verify improvement
    M_opt = M_of_t(t_opt)
    if M_opt >= M_cur:
        # Try a few discrete values as fallback
        best_t = 0.0
        best_M = M_cur
        for t_test in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            mt = M_of_t(t_test)
            if mt < best_M:
                best_M = mt
                best_t = t_test
        return best_t
    return t_opt


def coordinate_optimize_fast(a, i, b):
    """Find optimal a[i] given all others. O(n) using vectorized ops."""
    n = len(a)
    x_cur = a[i]

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Build linear and quadratic terms
    c_vals = []
    s_vals = []
    quad_c = None

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            quad_c = b[k] - x_cur * x_cur
        else:
            j = k - i
            if 0 <= j < n:
                c_vals.append(b[k] - 2.0 * x_cur * a[j])
                s_vals.append(2.0 * a[j])

    # Base max (unaffected)
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    if not c_vals and quad_c is None:
        return x_cur

    c_arr = np.array(c_vals) if c_vals else np.array([])
    s_arr = np.array(s_vals) if s_vals else np.array([])

    def obj(x_test):
        m = base_max
        if c_arr.size > 0:
            vm = np.max(c_arr + s_arr * x_test)
            if vm > m:
                m = vm
        if quad_c is not None:
            qv = quad_c + x_test * x_test
            if qv > m:
                m = qv
        return m

    # Bounds for search
    hi_bound = max(2.0 * x_cur + 0.5, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.5)

    obj0 = obj(0.0)
    if obj0 <= obj(x_cur):
        return 0.0

    t_opt = golden_section_min(obj, 0.0, hi_bound, tol=1e-10, max_iter=35)
    x_opt = t_opt
    if obj(0.0) <= obj(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_cd_sweeps(a, max_sweeps, deadline, seed=0):
    """Run coordinate descent sweeps with incremental b updates."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    b = fast_conv(a)
    best_a = a.copy()
    best_M = b.max()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = np.arange(n)
        rng.shuffle(order)
        improved = False
        for i in order:
            x_opt = coordinate_optimize_fast(a, i, b)
            if abs(x_opt - a[i]) > 1e-12:
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += x_opt * x_opt - a[i] * a[i]
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - a[i])
                a[i] = x_opt
                improved = True

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        b = fast_conv(a)  # recompute to avoid drift
        M_cur = b.max()
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()
        if not improved:
            break

    return best_a


def run_alt_lp_cd(a_init, max_rounds, deadline, lp_time_limit=5):
    """Alternating LP best-response + coordinate descent."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # LP step
        M_cur = best_M
        g = solve_lp(a, M_cur, time_limit=lp_time_limit)

        if g is not None:
            # Line search
            t_opt = line_search_t(a, g, M_cur)
            if t_opt > 1e-4:
                a_new = (1.0 - t_opt) * a + t_opt * g
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    M_new = fast_conv(a_new).max()
                    if M_new < best_M:
                        a = a_new
                        best_M = M_new
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        # CD step (1 sweep)
        if time.monotonic() < deadline:
            a_cd = run_cd_sweeps(a, 1, min(deadline, time.monotonic() + 2.0), seed=rnd)
            M_cd = fast_conv(a_cd).max()
            if M_cd < best_M:
                a = a_cd
                best_M = M_cd
                best_a = a.copy()
                no_improve = 0

        if no_improve > 30:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def perturb_and_reoptimize(a, rng, deadline):
    """Perturb a and re-run optimization."""
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for noise_scale in [0.05, 0.02, 0.01]:
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, noise_scale, n)
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()

        sub_dl = min(deadline, time.monotonic() + 30)
        a_opt = run_alt_lp_cd(a_pert, 30, sub_dl, lp_time_limit=3)
        M_opt = fast_conv(a_opt).max()
        if M_opt < best_M:
            best_M = M_opt
            best_a = a_opt.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: full-constraint LP alternating + CD + perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={time.monotonic()-t_start:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        update_best(parent)

    n_parent = parent.size
    a_cur = parent.copy()
    a_cur = a_cur / a_cur.sum()

    # === Phase 1: Main LP+CD on parent length (50% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Alt LP+CD on n={n_parent}", flush=True)
    print(f"{'='*60}", flush=True)
    t1_deadline = t_start + 0.5 * (budget_s - 30)
    a1 = run_alt_lp_cd(a_cur, 500, t1_deadline, lp_time_limit=5)
    update_best(a1)
    print(f"  Phase 1 done: M={fast_conv(a1).max():.10f}", flush=True)

    # === Phase 2: Length exploration (20% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    t2_end = t_start + 0.7 * (budget_s - 30)
    candidate_ns = []
    for delta in [-64, 64, -32, 32, 128, -128, 256, -256]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl != n_parent:
            candidate_ns.append(nl)

    if candidate_ns:
        time_per_len = max(15, (t2_end - time.monotonic()) / len(candidate_ns))
        for nl in candidate_ns:
            if time.monotonic() > t2_end:
                break
            a_res = resample_seq(a1, nl)
            a_res = np.maximum(a_res, 1e-8)
            a_res = a_res / a_res.sum()
            sub_dl = min(t2_end, time.monotonic() + time_per_len)
            a_opt = run_alt_lp_cd(a_res, 50, sub_dl, lp_time_limit=3)
            update_best(a_opt)

    # === Phase 3: Intensive refinement on best (15% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: Intensive refinement", flush=True)
    print(f"{'='*60}", flush=True)
    if best_seq is not None and time.monotonic() < t_start + 0.85 * (budget_s - 30):
        t3_end = t_start + 0.85 * (budget_s - 30)
        a3 = run_alt_lp_cd(best_seq, 200, t3_end, lp_time_limit=3)
        update_best(a3)
        # CD polish
        if time.monotonic() < t3_end:
            a3_cd = run_cd_sweeps(a3, 5, t3_end, seed=12345)
            update_best(a3_cd)

    # === Phase 4: Perturbation restarts (10% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Perturbation restarts", flush=True)
    print(f"{'='*60}", flush=True)
    if best_seq is not None and time.monotonic() < t_start + 0.95 * (budget_s - 30):
        t4_end = t_start + 0.95 * (budget_s - 30)
        a4 = perturb_and_reoptimize(best_seq, rng, t4_end)
        update_best(a4)

    # === Phase 5: Final CD polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Final CD polish", flush=True)
    print(f"{'='*60}", flush=True)
    if best_seq is not None and time.monotonic() < deadline:
        a_final = run_cd_sweeps(best_seq, 10, deadline, seed=99999)
        update_best(a_final)
        # One more quick LP pass
        if time.monotonic() < deadline:
            a_lp = run_alt_lp_cd(a_final, 20, deadline, lp_time_limit=3)
            update_best(a_lp)

    if best_seq is None:
        best_seq = np.ones(512)
        best_score = compute_J(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
