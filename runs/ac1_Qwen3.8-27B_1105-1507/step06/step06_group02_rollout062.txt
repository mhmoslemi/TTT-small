import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """Compute smooth max approximation and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # Smooth max value
    smax = b_max + np.log(exp_b.sum()) / tau
    # Gradient: grad[a[j]] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2 * p[k] * a[k - (lo:hi+1)]
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * p[k] * a[k - idx][::-1]
    return smax, grad, b, b_max, p

            break
        t += 1
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        smax, grad, b, b_max, p = smooth_max_and_grad(a, tau)
        
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = np.maximum(a - lr * m_hat / (np.sqrt(v_hat) + eps), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    return best_a


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (A @ g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'time_limit': 5.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def kkt_best_response(a, active_set):
    """Solve KKT system: C @ x = 1 where C[j,m] = 1_{j+m in active_set}.
    Returns x / (1^T x) if all positive, else None."""
    n = len(a)
    S = set(active_set)
    # Build C matrix: C[j,m] = 1 if (j+m) in S
    C = np.zeros((n, n), dtype=np.float64)
    for j in range(n):
        for m in range(n):
            if (j + m) in S:
                C[j, m] = 1.0
    try:
        x = np.linalg.solve(C, np.ones(n))
    except np.linalg.LinAlgError:
        return None
    if np.any(x < -1e-10):
        return None
    x = np.maximum(x, 0)
    sx = x.sum()
    if sx < 1e-10:
        return None
    return x / sx


def kkt_growing_active_set(a, deadline, max_iters=200):
    """Iteratively grow the active set and solve KKT system."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Determine active set: taps within 1% of max
        threshold = M * 0.999
        active = set(np.where(b >= threshold)[0].tolist())
        if not active:
            active = set([int(np.argmax(b))])
        
        # Solve KKT
        x = kkt_best_response(a, active)
        if x is None:
            # Try with a slightly larger active set
            threshold = M * 0.99
            active = set(np.where(b >= threshold)[0].tolist())
            x = kkt_best_response(a, active)
            if x is None:
                break
        
        # Check if this is actually better
        b_new = fast_conv(x)
        M_new = float(b_new.max())
        if M_new < best_M * (1 - 1e-10):
            best_M = M_new
            best_a = x.copy()
            a = x
        else:
            # Try interpolation
            for t in [0.5, 0.3, 0.1]:
                a_try = (1 - t) * a + t * x
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try /= s
                M_try = fast_conv(a_try).max()
                if M_try < best_M * (1 - 1e-12):
                    best_M = M_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                break
    return best_a


def anderson_mix(history_x, history_Fx, m):
    """Anderson mixing: find optimal combination of history."""
    if m < 2:
        return history_Fx[-1].copy()
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = float(ones @ G_inv_ones)
        if abs(denom) < 1e-14:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -2.0, 2.0)
    s = alpha.sum()
    if abs(s) < 1e-14:
        alpha = ones / m
        s = 1.0
    alpha = alpha / s
    mixed = np.zeros_like(history_Fx[0])
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def lp_mirror_anderson_v2(a, deadline, max_iters=300, window=6, t_base=0.3, min_t=0.003):
    """LP mirror descent with Anderson acceleration (v2 with backtracking)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    hist_x = []
    hist_Fx = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Determine step size with backtracking
        if it < 3:
            t = t_base
        else:
            t = max(min_t, t_base / (1 + 0.03 * it))
        
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
            M_and = fast_conv(a_and).max()
            M_new = fast_conv(a_new).max()
            if M_and <= M_new * 1.02:
                a = a_and
            else:
                a = a_new
        else:
            a = a_new
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent=None):
    """Create diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.01))
    
    inits.append(np.ones(n))
    
    # Power law shapes
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * t**alpha, 0.01))
        inits.append(np.maximum(0.1 + 0.9 * (1 - t)**alpha, 0.01))
    
    # Cosine modulations
    for f in [1, 2, 3, 4, 6]:
        inits.append(np.maximum(0.5 + 0.45 * np.cos(2 * np.pi * f * t), 0.01))
    
    # Bumps
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.05, 0.1, 0.15, 0.25]:
            inits.append(np.exp(-((t - center) / width) ** 2) + 0.02)
    
    # Double bump
    inits.append(np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.02)
    inits.append(np.exp(-4 * (t - 0.2) ** 2) + np.exp(-4 * (t - 0.8) ** 2) + 0.05)
    
    # Random smooth
    k = max(5, n // 16)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    
    # Symmetric
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.2, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    
    # Anti-symmetric tendency (monotone)
    inits.append(np.linspace(0.1, 1.0, n))
    inits.append(np.linspace(1.0, 0.1, n))
    
    # Random with structure
    inits.append(np.maximum(0.3 + 0.5 * np.sin(np.pi * t) + 0.2 * rng.normal(0, 1, n), 0.01))
    
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
