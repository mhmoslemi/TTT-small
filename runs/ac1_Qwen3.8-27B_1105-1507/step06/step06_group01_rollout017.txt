def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_full_lp_matrix(n, a):
    """Build the (2n-1) x n matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_best_response(a, M, n):
    """Solve the full LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    
    Returns (g, dual_y, lp_status) where g is the optimal solution,
    dual_y are the dual variables, and lp_status is 'optimal' or 'failed'.
    """
    A = build_full_lp_matrix(n, a)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 5})
        if r.success:
            g = np.maximum(r.x, 0.0)
            # Extract dual variables from unevaluated bounds (lambda_ineq_u)
            dual_y = r.ineqlin.marginals if hasattr(r.ineqlin, 'marginals') else None
            # For HiGHS, duals are in r.ineqlin.marginals (negated)
            if dual_y is None:
                # Try alternate attribute
                if hasattr(r, 'ineqlin') and hasattr(r.ineqlin, 'marginals'):
                    dual_y = -r.ineqlin.marginals
                else:
                    dual_y = np.zeros(2 * n - 1)
            else:
                dual_y = np.maximum(-dual_y, 0.0)  # duals should be non-negative for max problem
            return g, dual_y, 'optimal'
    except Exception as e:
        pass
    return None, None, 'failed'


def lp_game_iteration(a, max_iters, deadline, verbose=True):
    """Run the LP game iteration until convergence or deadline.
    
    Returns the best a found.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = 2 * n * best_M  # since sum(a) = 1
    
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Solve full LP
        g, dual_y, status = solve_lp_best_response(a, M, n)
        if status != 'optimal' or g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        # Duality gap: M * sum(dual_y) - g_sum (should be >= 0 at optimum)
        # Actually, the dual objective is M * sum(y), primal is g_sum
        # Gap = M * sum(y) - g_sum
        if dual_y is not None:
            dual_obj = M * dual_y.sum()
            gap = dual_obj - g_sum
        else:
            gap = abs(g_sum - 1.0)  # fallback
        
        # Normalize g to probability vector
        g_norm = g / g_sum
        
        # Adaptive step size based on gap
        # If gap is large, take larger steps; if small, take smaller steps
        t = min(1.0, max(0.05, 0.5 * (1.0 - min(gap / M, 1.0))))
        # Alternative: use the improvement ratio
        # t = min(0.5, g_sum / (M * (2*n-1)))  # rough
        
        # Line search: try a few step sizes
        a_trial = None
        M_trial = None
        for t_candidate in [t, t * 0.5, t * 0.25, 0.3, 0.2, 0.1, 0.05]:
            a_t = (1.0 - t_candidate) * a + t_candidate * g_norm
            a_t = np.maximum(a_t, 0.0)
            s_t = a_t.sum()
            if s_t < 1e-10:
                continue
            a_t = a_t / s_t
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-15:
                a_trial = a_t
                M_trial = M_t
                break
        
        if a_trial is not None:
            a = a_trial
            best_M = M_trial
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                J_cur = 2 * n * best_M
                print(f"    LP iter {it}: M={best_M:.10f} J={J_cur:.8f} gap={gap:.2e}", flush=True)
        else:
            no_improve += 1
            if no_improve > 30:
                break
    
    return best_a


def coordinate_descent_polish(a, deadline, verbose=True):
    """Polish with coordinate descent (exact 1D optimization per coordinate)."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    b = fast_conv(a)
    no_improve = 0
    sweep = 0
    
    while time.monotonic() < deadline and no_improve < 15:
        sweep += 1
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            # Compute optimal a[i] given all others
            x_cur = a[i]
            # Affected convolution indices: k in [i, i+n-1] intersect [0, 2n-2]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Base values (convolution without a[i]'s contribution)
            # (a*a)[k] = base[k] + slope[k]*x + (1 if k==2i else 0)*x^2
            c_vals = np.zeros(k_hi - k_lo + 1)
            s_vals = np.zeros(k_hi - k_lo + 1)
            has_quad = np.zeros(k_hi - k_lo + 1, dtype=bool)
            
            for idx, k in enumerate(range(k_lo, k_hi + 1)):
                j = k - i
                if 0 <= j < n:
                    if j == i:
                        # This is the a[i]*a[i] term
                        c_vals[idx] = b[k] - x_cur * x_cur
                        s_vals[idx] = 0.0
                        has_quad[idx] = True
                    else:
                        c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                        s_vals[idx] = 2.0 * a[j]
                # else: no contribution from a[i] at this k
            
            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max()
            else:
                base_max = 0.0
            
            # Golden section search for min of max(base_max, c+s*x, c_quad+x^2)
            lo = 0.0
            hi = max(2.0 * x_cur, 0.02)
            
            def obj_at(x_test):
                vals = c_vals + s_vals * x_test
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                quad_mask = has_quad
                if quad_mask.any():
                    qv = c_vals[quad_mask] + x_test * x_test
                    qv_max = qv.max()
                    if qv_max > m:
                        m = qv_max
                return m
            
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_at(a_pt)
            fb = obj_at(b_pt)
            
            for _ in range(50):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_at(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)
            
            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - a[i]) > 1e-12:
                # Update b for affected indices
                for idx, k in enumerate(range(k_lo, k_hi + 1)):
                    j = k - i
                    if 0 <= j < n:
                        if j == i:
                            b[k] = b[k] - x_cur * x_cur + x_opt * x_opt
                        else:
                            b[k] = b[k] - 2.0 * x_cur * a[j] + 2.0 * x_opt * a[j]
                a[i] = x_opt
                improved = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        b = fast_conv(a)  # Recompute to avoid drift
        
        M = b.max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
            if verbose and sweep % 5 == 0:
                print(f"    CD sweep {sweep}: M={best_M:.10f} J={2*n*best_M:.8f}", flush=True)
        else:
            no_improve += 1
    
    return best_a


def spectral_init(n, rng, iterations=50):
    """Initialize by trying to flatten the power spectrum."""
    a = np.ones(n) / n
    for _ in range(iterations):
        A = np.fft.rfft(a, 2 * n - 1)
        power = np.abs(A) ** 2
        # We want power to be flat. Gradient of max(power) w.r.t. a:
        # d/d a_i of |A(k)|^2 = 2 * Re(A(k) * e^{-iω_k i})
        # Use the peak frequency to update
        k_peak = np.argmax(power)
        # Gradient at peak frequency
        omega = 2 * np.pi * k_peak / (2 * n - 1)
        grad = 2.0 * np.real(A[k_peak] * np.exp(-1j * omega * np.arange(n)))
        # Project gradient to sum=0 (tangent plane)
        grad -= grad.mean()
        # Update
        a = a - 0.02 * grad
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
    return a


def cosine_init(n, alpha=0.5):
    """Cosine-tapered initialization."""
    t = np.linspace(0, 1, n)
    # Generalized cosine window
    a = 1.0 - alpha * np.cos(2 * np.pi * t)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    
    # Resample parent
    if parent is not None and parent.size > 0:
        p = parent.copy()
        if p.size != n:
            t_old = np.linspace(0, 1, p.size)
            t_new = np.linspace(0, 1, n)
            p = np.maximum(np.interp(t_new, t_old, p), 0.0)
        p = p / p.sum()
        inits.append(p)
        
        # Perturbed parent
        p_pert = np.maximum(p + rng.normal(0, 0.05, n), 1e-6)
        inits.append(p_pert / p_pert.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine
    inits.append(cosine_init(n, 0.3))
    inits.append(cosine_init(n, 0.7))

    cos2 = np.cos(np.pi * (t - 0.5)) ** 2 + 0.05
    inits.append(cos2 / cos2.sum())
    
    return inits

Key improvements over previous approach:
1. Full LP (all 2n-1 constraints) for exact best-response
2. LP-based stopping criterion (sum(g*) close to 1 indicates optimality)

4. Multi-length search with adaptive time allocation

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    return np.maximum(result, 0.0)


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n, a):
    """Build sparse (2n-1) x n matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    A = csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(2 * n - 1, n))
    return A


def solve_lp_full(a, M, n):
    """Solve full LP: max sum(g) s.t. A*g <= M*1, g >= 0.
    Returns (g, g_sum) or (None, 0) on failure.
    """
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30, 'mip_rel_gap': 1e-8})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            return g, g.sum()
    except Exception:
        pass
    return None, 0.0
