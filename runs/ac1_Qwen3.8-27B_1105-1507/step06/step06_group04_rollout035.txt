import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a

def subgradient_optimize(a, time_budget=60.0, tau_init=0.01, tau_min=0.0005, 
                         step_init=0.05, mu=0.9, num_restarts=3):
    """
    Subgradient method with adaptive softmax temperature and Nesterov momentum.
    Multiple restarts with different temperatures.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    for restart in range(num_restarts):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Reset for this restart
        a = best_a * np.exp(np.random.default_rng(restart * 12345 + 7).normal(0, 0.05, n))
        a = normalize_simplex(a)
        v = np.zeros(n)
        step = step_init
        tau = tau_init * (1 + 0.5 * restart)
        
        it = 0
        while time.monotonic() - start < time_budget - 5:
            it += 1
            conv = compute_conv(a)
            cmax = float(np.max(conv))
            
            # Softmax weights over peaks
            shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
            weights = np.exp(shifted)
            w_sum = np.sum(weights)
            if w_sum < 1e-300:
                idx = int(np.argmax(conv))
                weights = np.zeros(len(conv))
                weights[idx] = 1.0
            else:
                weights /= w_sum
            
            # Compute gradient via FFT
            # grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
            # = 2 * cross_correlation(w, a)[i]
            w_ext = np.zeros(fft_size)
            w_ext[:len(weights)] = weights
            W_ext = rfft(w_ext, fft_size)
            A_ext = rfft(a, fft_size)
            corr = irfft(A_ext * np.conj(W_ext), fft_size)
            grad = 2.0 * corr[:n]
            
            # Center gradient
            grad_centered = grad - np.mean(grad)
            grad_norm = np.linalg.norm(grad_centered)
            if grad_norm < 1e-15:
                break
            
            # Nesterov momentum
            v = mu * v - step * grad_centered / grad_norm
            a_new = a - v
            a_new = normalize_simplex(a_new)
            a = a_new
            obj = compute_obj(a)
            
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            
            # Adapt step and temperature
            step *= 0.999
            tau *= 0.9997
            if tau < tau_min:
                tau = tau_min
            
            # Periodic step reduction
            if it % 200 == 199:
                step = max(0.01, step * 0.8)
    
    return best_a, best_obj

def coordinate_descent(a, time_budget=30.0):
    """
    Coordinate descent: for each coordinate, find the optimal value
    using 1D search.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for epoch in range(50):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                    break
            # Try different values of a[i]
            # Current sum of others
            s_others = a.sum() - a[i]
            if s_others < 1e-15:
                continue
            
            # Try: reduce a[i] and distribute to others, or increase a[i]
            # Parameterize: a_new[i] = t, others scaled to sum to 1-t
            # Try a few values of t
            best_t = a[i]
            best_local_obj = best_obj
            
            for t in [a[i] * 0.5, a[i] * 0.7, a[i] * 1.3, a[i] * 1.5, 
                      a[i] + 0.01, a[i] - 0.01, 0.0]:
                if t < 0 or t > 0.99:
                    continue
                a_try = a.copy()
                a_try[i] = t
                s_others_try = 1.0 - t
                if s_others_try <= 0:
                    continue
                # Scale others to sum to s_others_try
                others_sum = a_try.sum() - t
                if others_sum > 1e-15:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    a_try[mask] *= s_others_try / others_sum
                obj_try = compute_obj(a_try)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_t = t
            
            if best_t != a[i]:
                a[i] = best_t
                s_others_new = 1.0 - best_t
                others_sum = a.sum() - best_t
                if others_sum > 1e-15:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    a[mask] *= s_others_new / others_sum
                a = normalize_simplex(a)
                if best_local_obj < best_obj:
                    best_obj = best_local_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def chase_top_peaks(a, num_peaks=5, num_steps=200, time_budget=30.0):
    """
    Chase the top K peaks using weighted subgradient.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    step = 0.03
    for it in range(num_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Get top K peak positions with softmax weights
        top_k = np.argsort(conv)[-num_peaks:]
        weights = np.zeros(len(conv))
        for k in top_k:
            weights[k] = 1.0 / num_peaks
        
        # Compute gradient
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]
        grad_centered = grad - np.mean(grad)
        gn = np.linalg.norm(grad_centered)
        if gn < 1e-15:
            break
        
        a_new = a - step * grad_centered / gn
        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.95
    
    return best_a, best_obj
