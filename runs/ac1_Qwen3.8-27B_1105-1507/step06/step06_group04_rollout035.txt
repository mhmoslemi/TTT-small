import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def get_fft_size(n):
    """Get next power of 2 >= 2*n-1."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv_from_fft(fa, fft_size, n):
    """Compute conv(a,a) from precomputed FFT fa."""
    conv = irfft(fa * fa, fft_size)
    return conv[:2 * n - 1]


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=20.0):
    """
    Solve: min t subject to (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g_normalized, t_opt) or (None, None).
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(fixed_seq, M, time_limit=20.0):
    """
    Solve: max sum(g) subject to (fixed_seq * g)[k] <= M for all k, g >= 0.
    Returns g (normalized to sum=1) or None.
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def line_search_fft(a, g, fa, fg, fft_size, n, sum_a=1.0, sum_g=1.0):
    """
    Efficient line search using precomputed FFTs.
    Finds best t in [0,1] minimizing obj of (1-t)*a + t*g.
    Returns (best_t, best_obj, best_a).
    """
    best_t = 0.0
    best_obj = compute_obj(a)
    best_a = a.copy()

    # Coarse search
    ts_coarse = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]
    for t in ts_coarse:
        f_blend = (1.0 - t) * fa + t * fg
        conv = irfft(f_blend * f_blend, fft_size)[:2 * n - 1]
        s = (1.0 - t) * sum_a + t * sum_g
        obj = 2.0 * n * np.max(conv) / (s * s)
        if obj < best_obj:
            best_obj = obj
            best_t = t
            a_try = (1.0 - t) * a + t * g
            best_a = normalize_simplex(a_try)

    # Fine search around best_t
    if 0.0 < best_t < 1.0:
        lo = max(0.0, best_t - 0.1)
        hi = min(1.0, best_t + 0.1)
        ts_fine = np.linspace(lo, hi, 11)
        for t in ts_fine:
            if abs(t - best_t) < 0.005:
                continue
            f_blend = (1.0 - t) * fa + t * fg
            conv = irfft(f_blend * f_blend, fft_size)[:2 * n - 1]
            s = (1.0 - t) * sum_a + t * sum_g
            obj = 2.0 * n * np.max(conv) / (s * s)
            if obj < best_obj:
                best_obj = obj
                best_t = t
                a_try = (1.0 - t) * a + t * g
                best_a = normalize_simplex(a_try)

    return best_t, best_obj, best_a


def subgradient_fft_step(a, tau, step_size, v, mu):
    """
    Single FFT-accelerated subgradient step with momentum.
    Returns (new_a, new_v, conv, max_b).
    """
    n = len(a)
    fft_size = get_fft_size(n)

    conv = compute_conv_from_fft(rfft(a, fft_size), fft_size, n)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # Compute gradient via FFT correlation
    w_ext = np.zeros(fft_size)
    w_ext[:len(weights)] = weights
    W_ext = rfft(w_ext, fft_size)
    A_ext = rfft(a, fft_size)
    corr = irfft(A_ext * np.conj(W_ext), fft_size)
    grad = 2.0 * corr[:n]
    grad_centered = grad - np.mean(grad)
    grad_norm = np.linalg.norm(grad_centered)
    if grad_norm < 1e-15:
        return a, v, conv, cmax

    v = mu * v - step_size * grad_centered / grad_norm
    a_new = a - v
    a_new = normalize_simplex(a_new)
    return a_new, v, conv, cmax


def quick_subgradient(a, time_budget=10.0):

def structured_perturbation(a, rng, freq_range=(1, 12), amp_range=(0.01, 0.08)):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)

        print(f"  Candidate {ci} (n={n_cand}, obj={obj_cand:.8f}): LP budget={alloc:.0f}s")

        a_pol, obj_pol = lp_optimize(a_cand, time_budget=alloc, max_rounds=40)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"    ** UPDATE: {obj_pol:.8f} **")
        else:
            print(f"    No improvement (got {obj_pol:.8f})")

        seen_seqs += 1

    # Phase 3: Subgradient + LP hybrid polish (300s)
    print(f"\n=== Phase 3: Hybrid polish (best: {best_score:.8f}) ===")
    phase3_end = start_time + 1300.0

    if best_seq is not None:
        n_best = len(best_seq)
        a_work = best_seq.copy()
        a_work = normalize_simplex(a_work)
        obj_work = compute_obj(a_work)
        v = np.zeros(n_best)
        mu = 0.85
        step = 0.04
        tau = 0.01
        stall = 0

        while time.monotonic() < phase3_end - 10:
            remaining = phase3_end - time.monotonic()
            if remaining < 10:
                break

            # Alternating: subgradient for 2 sec, LP for 8 sec
            # Subgradient burst
            sg_budget = min(2.0, remaining * 0.1)
            for _ in range(50):
                if time.monotonic() >= phase3_end - 15:
                    break
                a_work, v, conv, cmax = subgradient_fft_step(a_work, tau, step, v, mu)
                obj_new = compute_obj(a_work)
                if obj_new < obj_work - 1e-12:
                    obj_work = obj_new
                    if obj_new < best_score:
                        best_score = obj_new
                        best_seq = a_work.copy()
                        stall = 0
                step *= 0.999
                tau *= 0.9998
                if step < 0.003:
                    break

            # LP burst
            if time.monotonic() < phase3_end - 20:
                lp_budget = min(10.0, remaining * 0.3)
                a_lp, obj_lp, improved = lp_alternate_round(a_work, time_budget=lp_budget)
                if improved:
                    a_work = a_lp
                    obj_work = obj_lp
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = a_work.copy()
                        stall = 0
                        print(f"    LP improvement: {obj_lp:.8f}")
                else:
                    stall += 1
                    if stall > 3:
                        # Structured perturbation
                        rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000003)
                        a_work = structured_perturbation(best_seq, rng)
                        a_work, obj_rec = quick_subgradient(a_work, time_budget=3.0)
                        stall = 0
                        step = 0.04
                        tau = 0.01
                        v = np.zeros(n_best)

    # Phase 4: Length exploration (100s)
    print(f"\n=== Phase 4: Length exploration (best: {best_score:.8f}) ===")
    phase4_end = start_time + 1400.0

    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = [n_best - 50, n_best + 50, n_best - 100, n_best + 100,
                        n_best - 200, n_best + 200, n_best - 150, n_best + 150]
        test_lengths = sorted(set([x for x in test_lengths if 400 <= x <= 1200]))

        for n_test in test_lengths:
            if time.monotonic() > phase4_end:
                break
            remaining = phase4_end - time.monotonic()
            if remaining < 15:
                break

            a_res = resample_sequence(best_seq, n_test)
            a_res, obj_res = quick_subgradient(a_res, time_budget=min(5.0, remaining * 0.2))

            if remaining > 25:
                a_res, obj_res2 = lp_optimize(a_res, time_budget=min(remaining * 0.3, 30.0),
                                               max_rounds=5)
                if obj_res2 < obj_res:
                    a_res, obj_res = a_res, obj_res2

            if obj_res < best_score:
                best_score = obj_res
                best_seq = a_res.copy()
                print(f"  n={n_test}: ** UPDATE to {best_score:.8f} **")
            else:
                print(f"  n={n_test}: {obj_res:.8f}")

    # Phase 5: Final intensive LP on the absolute best (70s)
    print(f"\n=== Phase 5: Final LP (best: {best_score:.8f}) ===")

    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        n_final = len(best_seq)
        if remaining > 20 and n_final <= 1000:
            a_final, obj_final = lp_optimize(best_seq, time_budget=min(remaining * 0.6, 50.0),
                                             max_rounds=10)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: ** {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
