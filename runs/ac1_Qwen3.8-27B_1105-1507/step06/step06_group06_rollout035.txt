    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf

    a = np.maximum(a, 0)

    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

"""
Minimax QP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: iterative minimax framework where we alternate between
(1) computing softmax weights on convolution peaks, and
(2) solving a QP (via proximal projected gradient) to minimize the weighted
quadratic a^T Q a. Combined with frequency-domain smoothing and multi-length
exploration.
"""
import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def Q_matvec(n, p, a):
    """Compute Q @ a where Q = sum_k p[k] C_k, C_k[i,j]=1 if i+j=k."""
    result = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-15:
            continue
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        j_lo = k - i_max
        j_hi = k - i_min
        result[i_min:i_max + 1] += w * a[j_lo:j_hi + 1][::-1]
    return result


def estimate_spectral_norm(n, p, vec, iters=30):
    """Power iteration to estimate spectral norm of Q."""
    v = vec.copy()
    v = v / (np.linalg.norm(v) + 1e-15)
    for _ in range(iters):
        v_new = Q_matvec(n, p, v)
        nv = np.linalg.norm(v_new)
        if nv < 1e-15:
            return 0.0
        v = v_new / nv
    return abs(v @ Q_matvec(n, p, v))


def solve_qp_prox(n, p, a_prev, mu, max_inner=300, eig_est=None):
    """
    Solve min a^T Q a + (1/(2*mu))||a - a_prev||^2 s.t. a >= 0, sum a = 1.
    Uses projected gradient with backtracking.
    """
    a = project_simplex(a_prev.copy())
    inv_mu = 1.0 / mu

    if eig_est is None:
        eig_est = estimate_spectral_norm(n, p, a)
    L = 2.0 * max(eig_est, 1e-8) + inv_mu
    step = 1.0 / L

    best_a = a.copy()
    best_val = a @ Q_matvec(n, p, a) + inv_mu * 0.5 * np.sum((a - a_prev) ** 2)

    for it in range(max_inner):
        grad = 2.0 * Q_matvec(n, p, a) + inv_mu * (a - a_prev)
        a_new = a - step * grad
        a_new = project_simplex(a_new)
        val_new = a_new @ Q_matvec(n, p, a_new) + inv_mu * 0.5 * np.sum((a_new - a_prev) ** 2)

        if val_new < best_val - 1e-14:
            # Accept, maybe increase step
            if val_new < val_new:
                pass
            a = a_new
            if it % 20 == 19:
                step = min(step * 1.1, 2.0 / L)
        else:
            step *= 0.5
            if step < 1e-18:
                break

        if it > 5 and np.max(np.abs(a_new - a)) < 1e-12:
            break

    return a


def minimax_optimize(a_init, max_outer=80, time_budget=300.0, verbose=False):
    """
    Minimax iteration: alternate between weight update and QP solve.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute spectral norm estimate (will update periodically)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    tau = 0.1 * conv_max
    p = np.exp(np.clip(conv, 0, None) / max(tau, 1e-10))
    p /= p.sum()
    eig_est = estimate_spectral_norm(n, p, a)

    stall = 0
    mu = 1.0 / max(eig_est, 0.1)  # proximal parameter

    for outer in range(max_outer):
        if time.monotonic() - start > time_budget:
            break

        # Solve proximal QP
        inner_iters = 200
        a_qp = solve_qp_prox(n, p, a, mu, max_inner=inner_iters, eig_est=eig_est)

        # Line search
        best_t = 0.3
        best_blend_obj = np.inf
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = project_simplex((1 - t) * a + t * a_qp)
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t

        if best_blend_obj < best_obj - 1e-12:
            a = project_simplex((1 - best_t) * a + best_t * a_qp)
            best_obj = best_blend_obj
            best_a = a.copy()
            stall = 0
            if verbose:
                print(f"    MM iter {outer}: obj={best_obj:.8f}, t={best_t}")
        else:
            stall += 1
            # Try different tau values
            improved = False
            for tau_frac in [0.05, 0.2, 0.5, 1.0, 2.0]:
                if time.monotonic() - start > time_budget:
                    break
                tau_new = tau_frac * conv_max
                p_new = np.exp(np.clip(conv, 0, None) / max(tau_new, 1e-10))
                p_new /= p_new.sum()
                eig_new = estimate_spectral_norm(n, p_new, a, iters=20)
                mu_new = 1.0 / max(eig_new, 0.1)
                a_qp2 = solve_qp_prox(n, p_new, a, mu_new, max_inner=150, eig_est=eig_new)
                for t in [0.3, 0.5, 0.7, 1.0]:
                    a_try = project_simplex((1 - t) * a + t * a_qp2)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        a = a_try
                        best_obj = obj_try
                        best_a = a.copy()
                        p = p_new
                        eig_est = eig_new
                        mu = mu_new
                        improved = True
                        break
                if improved:
                    break

            if not improved:
                # Perturb and try again
                if stall < 15:
                    rng = np.random.default_rng(outer)
                    a_pert = a * np.exp(rng.normal(0, 0.05, n))
                    a_pert = project_simplex(a_pert)
                    conv = compute_conv(a_pert)
                    conv_max = np.max(conv)
                    tau = 0.1 * conv_max
                    p = np.exp(np.clip(conv, 0, None) / max(tau, 1e-10))
                    p /= p.sum()
                    eig_est = estimate_spectral_norm(n, p, a_pert, iters=20)
                    mu = 1.0 / max(eig_est, 0.1)
                    a = a_pert
                else:
                    break

        # Update weights every few iterations
        if outer % 5 == 4:
            conv = compute_conv(a)
            conv_max = np.max(conv)
            tau = 0.1 * conv_max
            p = np.exp(np.clip(conv, 0, None) / max(tau, 1e-10))
            p /= p.sum()
            eig_est = estimate_spectral_norm(n, p, a, iters=20)
            mu = 1.0 / max(eig_est, 0.1)

    return best_a, best_obj


def freq_domain_optimize(a_init, max_iters=200, time_budget=60.0):
    """
    Frequency domain optimization: iteratively compress the power spectrum
    to flatten the autocorrelation.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        ps_max = np.max(power)
        if ps_max < 1e-15:
            break

        # Compress power spectrum: reduce dynamic range
        ps_norm = power / ps_max
        alpha = 0.6 + 0.3 * (it / max_iters)  # gradually reduce compression
        ps_comp = np.maximum(ps_norm, 1e-12) ** alpha

        # Also apply a floor to prevent zeros
        ps_comp = np.maximum(ps_comp, 1e-6)

        # Reconstruct with same phase
        phase = np.angle(fa)
        mag = np.sqrt(ps_comp * ps_max)
        fa_new = mag * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def convexify_iterate(a_init, max_iters=100, time_budget=60.0):
    """
    Directly minimize the max using iterative peak reduction.
    At each step, identify the peak lag and solve a small optimization
    to reduce it while maintaining flatness.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        c_max = conv[k_star]

        # Subgradient direction: reduce a[i] where a[k_star - i] is large
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        # Adaptive step size
        for step in [0.05, 0.02, 0.01, 0.005, 0.001]:
            a_try = a - step * grad_centered
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

        # Also try reducing the second peak
        conv2 = compute_conv(a)
        k_star2 = int(np.argmax(conv2))
        if k_star2 != k_star:
            grad2 = np.zeros(n)
            for i in range(n):
                j = k_star2 - i
                if 0 <= j < n:
                    grad2[i] = a[j]
            grad2_centered = grad2 - np.mean(grad2)
            for step in [0.05, 0.02, 0.01]:
                a_try = a - step * grad2_centered
                a_try = project_simplex(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(a_new)


def smoothed_subgradient(a_init, max_iters=500, time_budget=60.0, tau_init=0.02):
    """
    Subgradient method on the smoothed (log-sum-exp) approximation of max.
    With Nesterov momentum.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    momentum = 0.8
    step = 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        c_max = np.max(conv)
        tau = tau_init * c_max + 0.1 * it / max_iters * c_max
        shifted = np.clip((conv - c_max) / max(tau, 1e-10), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient: grad[i] = sum_k weights[k] * a[k-i] (for valid k-i)
        grad = np.zeros(n)
        for k in range(len(weights)):
            if weights[k] < 1e-12:
                continue
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            j_lo = k - i_max
            j_hi = k - i_min
            grad[i_min:i_max + 1] += weights[k] * a[j_lo:j_hi + 1][::-1]

        grad_centered = grad - np.mean(grad)
        v = momentum * v - step * grad_centered
        a_new = project_simplex(a - v)
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        step *= 0.997
        if it % 50 == 49:
            momentum = max(0.5, momentum * 0.95)

    return best_a, best_obj


def edge_aware_init(n, rng):
    """
    Generate initialization with the correct edge decay pattern.
    The edges should decay as ~1/sqrt(2n) for flat autocorrelation.
    """
    a = np.ones(n)
    # Edge decay: a[i] should be ~ 1/sqrt(2n-1) * sin(pi*(i+1)/(n+1)) roughly
    x = np.linspace(0, 1, n)
    # Use a raised cosine or similar
    a = 0.5 * (1 + np.cos(np.pi * (2 * x - 1))) ** 0.5
    a = a + 0.01 * rng.random(n)
    return project_simplex(a)


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: minimax QP + frequency domain + multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}")

    # === Phase 1: Minimax QP on parent length (40% budget) ===
    print("\n=== Phase 1: Minimax QP Optimization ===")
    phase1_budget = min(600.0, (deadline - start_time) * 0.40)
    a1 = project_simplex(best_seq.copy())
    a1, obj1 = minimax_optimize(a1, max_outer=100, time_budget=phase1_budget, verbose=True)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        print(f"  Minimax improved: {best_score:.8f}")
    else:
        print(f"  Minimax: {obj1:.8f} (no improvement)")

    # === Phase 2: Frequency domain + smoothed subgradient (20%) ===
    print(f"\n=== Phase 2: Freq Domain + Subgradient ===")
    print(f"Best so far: {best_score:.8f}")
    phase2_budget = min(300.0, (deadline - start_time) * 0.20)

    a2 = project_simplex(best_seq.copy())
    a2, obj2 = freq_domain_optimize(a2, max_iters=300, time_budget=phase2_budget * 0.5)
    if obj2 < best_score:
        best_score = obj2
        best_seq = a2.copy()
        print(f"  Freq domain improved: {best_score:.8f}")

    remaining2 = phase2_budget - (time.monotonic() - start_time) if False else min(phase2_budget, 200) - (time.monotonic() - (start_time + phase1_budget))
    # Just use remaining time
    elapsed2 = time.monotonic() - start_time - phase1_budget
    rem2 = phase2_budget - elapsed2
    if rem2 > 10:
        a2b = project_simplex(best_seq.copy())
        a2b, obj2b = smoothed_subgradient(a2b, max_iters=500, time_budget=rem2)
        if obj2b < best_score:
            best_score = obj2b
            best_seq = a2b.copy()
            print(f"  Subgradient improved: {best_score:.8f}")

    # === Phase 3: Multi-length exploration (20%) ===
    print(f"\n=== Phase 3: Multi-Length Exploration ===")
    print(f"Best so far: {best_score:.8f}")
    phase3_budget = min(300.0, (deadline - start_time) * 0.20)
    phase3_start = time.monotonic()
    n_best = len(best_seq)

    # Try nearby lengths
    n_tries = sorted(set([
        max(100, n_best - 50), n_best - 20, n_best,
        n_best + 20, n_best + 50, n_best + 100,
        max(100, n_best - 100), n_best + 200
    ]))
    n_tries = [n for n in n_tries if 100 <= n <= 2000]

    per_n_budget = max(5.0, phase3_budget / len(n_tries))

    for n_try in n_tries:
        if time.monotonic() > phase3_start + phase3_budget:
            break
        if n_try == n_best:
            continue

        a_res = resample_sequence(best_seq, n_try)

        # Quick minimax
        a_opt, obj_opt = minimax_optimize(a_res, max_outer=30, time_budget=per_n_budget * 0.6)

        # Quick freq domain
        rem_n = per_n_budget - (time.monotonic() - (start_time + phase1_budget + phase2_budget) if False else 0)
        a_opt2, obj_opt2 = freq_domain_optimize(a_opt, max_iters=100, time_budget=per_n_budget * 0.3)
        if obj_opt2 < obj_opt:
            a_opt, obj_opt = a_opt2, obj_opt2

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 4: Intensive minimax on best (15%) ===
    print(f"\n=== Phase 4: Intensive Minimax Polish ===")
    print(f"Best so far: {best_score:.8f}")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 30:
        phase4_budget = min(remaining4 * 0.6, 300.0)
        a4 = project_simplex(best_seq.copy())

        # Multiple minimax runs with different initializations
        for run in range(3):
            if time.monotonic() > deadline - 10:
                break
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            alloc = min(rem * 0.4, 100.0)

            if run == 0:
                a_cur = a4
            elif run == 1:
                a_cur = a4 * np.exp(rng.normal(0, 0.05, len(a4)))
                a_cur = project_simplex(a_cur)
            else:
                a_cur = edge_aware_init(len(a4), rng)
                # Blend with best
                a_cur = project_simplex(0.5 * a_cur + 0.5 * a4)

            a4, obj4 = minimax_optimize(a_cur, max_outer=50, time_budget=alloc)
            if obj4 < best_score:
                best_score = obj4
                best_seq = a4.copy()
                print(f"  Run {run}: {best_score:.8f}")

    # === Phase 5: Final convexify + small perturbations (remaining) ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 10 and best_seq is not None:
        n_final = len(best_seq)
        a_final = project_simplex(best_seq.copy())

        # Convexify (peak chasing)
        a_p, obj_p = convexify_iterate(a_final, max_iters=200, time_budget=min(remaining5 * 0.3, 60.0))
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            a_final = a_p.copy()
            print(f"  Convexify: {best_score:.8f}")

        # Small random perturbations
        rem5 = deadline - time.monotonic()
        n_pert = min(20, int(rem5 / 5))
        for pidx in range(n_pert):
            if time.monotonic() > deadline - 5:
                break
            a_pert = a_final * np.exp(rng.normal(0, 0.02, n_final))
            a_pert = project_simplex(a_pert)
            a_pert, obj_pert = convexify_iterate(a_pert, max_iters=50, time_budget=3.0)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                a_final = a_pert.copy()

        # One more minimax round
        rem5 = deadline - time.monotonic()
        if rem5 > 20:
            a_final2 = project_simplex(a_final.copy())
            a_final3, obj_final = minimax_optimize(a_final2, max_outer=30, time_budget=min(rem5 - 5, 60.0))
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final3.copy()
                print(f"  Final minimax: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    print(f"\nEvaluated: {val:.8f}")
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
