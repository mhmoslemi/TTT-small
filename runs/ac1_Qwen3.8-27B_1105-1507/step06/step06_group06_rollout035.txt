"""

Uses: (1) LP alternating with warm starts, (2) log-sum-exp smoothed gradient
with FFT-accelerated cross-correlation, (3) frequency-domain spectrum shaping,
(4) transfer-move coordinate descent, (5) simulated annealing for global

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv_fft(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def build_toeplitz(a):
    """Build (2n-1) x n matrix T where (T@g)[k] = conv(a,g)[k]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        T[k_start:k_end, j] = a[:k_end - k_start]
    return T


def solve_lp_minpeak(a, warm_start=None, time_limit=15):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if n > 3000:
        return None, None
    T = build_toeplitz(a)
    # Variables: g[0..n-1], M (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        opts = {'presolve': True, 'time_limit': time_limit}
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options=opts)
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = g.sum()
            if s > 1e-12:
                g = g / s
                M_val = float(res.x[n])
                return g, M_val
    except Exception:
        pass
    return None, None

def lse_smoothed_gradient(a, tau=0.05):
    """
    Compute gradient of smoothed objective f_tau(a) = 2*n*max_smooth/(sum a)^2
    where max_smooth = tau * logsumexp(conv/tau).
    Returns (gradient, objective_value).
    """
    n = len(a)
    conv = compute_conv_fft(a)
    sum_a = a.sum()

    cmax = np.max(conv)
    shifted = (conv - cmax) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    w = exp_shifted / sum_exp  # softmax weights

    # C = cmax + tau * log(sum_exp)
    C_val = cmax + tau * np.log(sum_exp)

    # Cross-correlation via FFT: corr[i] = sum_j w[i+j] * a[j]
    N = next_pow2(3 * n)
    W = rfft(w, N)
    A = rfft(a, N)
    cross = irfft(W * np.conj(A), N).real
    corr = cross[:n]

    # Gradient: df/da[i] = 2n/S^2 * 2*corr[i] - 4n*C/S^3
    f_val = 2.0 * n * C_val / (sum_a * sum_a)
    grad = (4.0 * n / (sum_a * sum_a)) * corr - (4.0 * n * C_val / (sum_a ** 3))

    return grad, f_val


def smoothed_gradient_descent(a, max_iters=2000, time_budget=30.0,
                               tau_init=0.03, lr_init=0.1, beta1=0.9, beta2=0.999):
    """Adam gradient descent on the log-sum-exp smoothed objective."""
    n = len(a)
    a = project(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        t_count += 1

        # Anneal tau
        tau = tau_init * (1.0 + it / max_iters)
        lr = lr_init / (1.0 + 0.01 * it)

        grad, _ = lse_smoothed_gradient(a, tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project(a_new)

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def freq_shape(a, max_iters=100, time_budget=20.0, alpha=0.8):
    """
    Frequency domain: compress the power spectrum to reduce time-domain peaks.
    |A(f)|^2 flat => flat circular autocorrelation.
    """
    n = len(a)
    a = project(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = next_pow2(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2

        # Compress: raise to alpha power (reduces dynamic range)
        pmax = power.max()
        if pmax < 1e-15:
            break
        pnorm = power / pmax
        pcomp = np.power(np.maximum(pnorm, 1e-12), alpha)

        # Reconstruct with modified magnitude, same phase
        phase = np.angle(fa)
        new_mag = np.sqrt(pcomp * pmax)
        fa_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj

                a[j] -= best_delta
                a = project(a)
                if best_local_obj < best_obj:
                    best_obj = best_local_obj
                    best_a = a.copy()

    return best_a, best_obj


def simulated_annealing(a, rng, max_iters=200, T_init=0.1, T_min=0.001, time_budget=60.0):
    """SA with multiple move types."""
    n = len(a)
    a = project(a.astype(np.float64))
    cur_a = a.copy()
    cur_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = cur_obj
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        T = T_init * (T_min / T_init) ** (it / max_iters)
        a_trial = cur_a.copy()
        move = rng.integers(0, 4)

        if move == 0:
            # Window noise
            w = max(5, n // 20)
            s = rng.integers(0, n - w)
            noise = rng.normal(0, 0.2, w) * (cur_a[s:s+w].mean() + 0.005)
            a_trial[s:s+w] += noise
        elif move == 1:
            # Window swap
            w = max(5, n // 25)
            s1 = rng.integers(0, n - w)
            s2 = rng.integers(0, n - w)
            if abs(s1 - s2) >= w:
                a_trial[s1:s1+w], a_trial[s2:s2+w] = a_trial[s2:s2+w].copy(), a_trial[s1:s1+w].copy()
        elif move == 2:
            # Spike redistribution
            w = max(10, n // 10)
            s = rng.integers(0, n - w)
            mass = a_trial[s:s+w].sum()
            a_trial[s:s+w] = 0
            targets = rng.choice(n, size=min(w, n), replace=False)
            a_trial[targets] += mass / len(targets)
        else:
            # Random pair swap
            i, j = rng.integers(0, n, 2)
            a_trial[i], a_trial[j] = a_trial[j], a_trial[i]

        a_trial = project(a_trial)
        trial_obj = compute_obj(a_trial)

        if trial_obj < cur_obj or np.log(rng.random()) < (cur_obj - trial_obj) / T:
            cur_a = a_trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()

    return best_a, best_obj


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] / new_n * n if n > 0 else 1.0/new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    for freq in [1, 2, 3, 5]:
        s = 0.7 + 0.3 * np.cos(2 * np.pi * freq * x)
        inits.append(s / s.sum())

    for center in [0.3, 0.5, 0.7]:
        g = np.exp(-((x - center) ** 2) / 0.02)
        inits.append(g / g.sum())

    for bw in [0.01, 0.05, 0.1]:
        bi = np.exp(-((x - 0.3) ** 2) / bw) + np.exp(-((x - 0.7) ** 2) / bw)
        inits.append(bi / bi.sum())

    inits.append((rng.random(n) + 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()

    if parent is not None:
        inits.append(resample(parent, n))
        pert = resample(parent, n) * np.exp(rng.normal(0, 0.1, n))
        inits.append(project(pert))

    return inits


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization combining LP alternating,
    smoothed gradient descent, frequency shaping, transfer moves, and SA.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None and len(parent) > 0:
        parent_norm = project(parent)
        p_obj = compute_obj(parent_norm)
        print(f"Parent: n={len(parent)}, obj={p_obj:.8f}")
        best_score = p_obj
        best_seq = parent_norm.copy()
    else:
        best_seq = np.ones(500) / 500
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"No parent, starting uniform n=500, obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Initial best: {best_score:.8f} (n={n_parent})")

    # === Phase 1: Quick diverse screening (8% time) ===
    p1_budget = min(80.0, (deadline - start_time) * 0.08)
    print(f"\n=== Phase 1: Diverse Screening (budget={p1_budget:.0f}s) ===")

    n_options = sorted(set([
        n_parent,
        max(200, n_parent - 100),
        max(200, n_parent - 50),
        n_parent + 50,
        n_parent + 100,
        min(2000, n_parent + 200),
    ]))
    n_options = [n for n in n_options if 100 <= n <= 3000]

    top_candidates = []
    per_screen = max(2.0, p1_budget / (len(n_options) * 5))

    for n_try in n_options:
        if time.monotonic() - start_time > p1_budget:
            break
        inits = generate_inits(n_try, rng, parent)
        for init in inits[:6]:
            if time.monotonic() - start_time > p1_budget:
                break
            a = project(init)
            # Quick: 100 gradient steps
            a_q, obj_q = smoothed_gradient_descent(a, max_iters=100, time_budget=per_screen)
            top_candidates.append((obj_q, a_q, n_try))

    top_candidates.sort(key=lambda x: x[0])
    for obj_c, a_c, n_c in top_candidates[:3]:
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  Screening improvement: n={n_c}, obj={best_score:.8f}")
    print(f"  Best after screening: {best_score:.8f}")

    # === Phase 2: Intensive optimization (55% time) ===
    p2_budget = min(800.0, (deadline - start_time) * 0.55)
    print(f"\n=== Phase 2: Intensive Optimization (budget={p2_budget:.0f}s) ===")
    p2_start = time.monotonic()

    # Focus on the best candidate
    a_work = project(best_seq.copy())
    n_work = len(a_work)
    local_best = a_work.copy()
    local_obj = best_score

    strat_cycle = 0
    while time.monotonic() < p2_start + p2_budget:
        remaining = p2_start + p2_budget - time.monotonic()
        if remaining < 5:
            break
        strat = strat_cycle % 5
        strat_cycle += 1

        if strat == 0 and n_work <= 1500:
            # LP alternating
            a_new, obj_new = lp_alternate(a_work, max_iters=15, time_budget=min(remaining * 0.3, 60.0))
        elif strat == 1:
            # Smoothed gradient
            a_new, obj_new = smoothed_gradient_descent(
                a_work, max_iters=500, time_budget=min(remaining * 0.2, 25.0),
                tau_init=0.02 + 0.01 * (strat_cycle % 3), lr_init=0.05)
        elif strat == 2:
            # Frequency shaping
            a_new, obj_new = freq_shape(a_work, max_iters=100,
                                        time_budget=min(remaining * 0.15, 15.0),
                                        alpha=0.7 + 0.1 * (strat_cycle % 3))
        elif strat == 3:
            # Transfer moves
            a_new, obj_new = transfer_moves(a_work, n_rounds=30,
                                            time_budget=min(remaining * 0.15, 15.0))
        else:
            # Combined: gradient then short LP
            a_tmp, obj_tmp = smoothed_gradient_descent(
                a_work, max_iters=300, time_budget=min(remaining * 0.1, 10.0))
            if n_work <= 1200:
                a_new, obj_new = lp_alternate(a_tmp, max_iters=8,
                                              time_budget=min(remaining * 0.15, 20.0))
            else:
                a_new, obj_new = a_tmp, obj_tmp

        if obj_new < local_obj - 1e-10:
            local_obj = obj_new
            local_best = a_new.copy()
            a_work = a_new.copy()
            if local_obj < best_score:
                best_score = local_obj
                best_seq = local_best.copy()
                elapsed = time.monotonic() - start_time
                print(f"  [t={elapsed:.0f}s] NEW BEST: n={n_work}, obj={best_score:.8f} (strat={strat})")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Global exploration (20% time) ===
    p3_budget = min(300.0, (deadline - time.monotonic()) * 0.25)
    print(f"\n=== Phase 3: Global Exploration (budget={p3_budget:.0f}s) ===")
    p3_start = time.monotonic()

    # SA on best
    a_sa, obj_sa = simulated_annealing(best_seq, rng, max_iters=200,
                                       T_init=0.05, T_min=0.001,
                                       time_budget=min(p3_budget * 0.3, 60.0))
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = a_sa.copy()
        print(f"  SA: {best_score:.8f}")

    # Multi-length exploration
    remaining3 = p3_budget - (time.monotonic() - p3_start)
    if remaining3 > 20:
        n_cur = len(best_seq)
        n_tries = [n_cur - 30, n_cur - 60, n_cur + 30, n_cur + 60, n_cur + 100]
        n_tries = sorted(set([n for n in n_tries if 150 <= n <= 2500]))
        per_n = max(5.0, remaining3 * 0.7 / len(n_tries))

        for n_try in n_tries:
            if time.monotonic() - p3_start > p3_budget:
                break
            a_r = resample(best_seq, n_try)
            a_r = project(a_r)
            a_opt, obj_opt = smoothed_gradient_descent(a_r, max_iters=200, time_budget=per_n * 0.5)
            if n_try <= 1200:
                a_opt2, obj_opt2 = lp_alternate(a_opt, max_iters=10, time_budget=per_n * 0.4)
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")

    # Second SA round with different params
    remaining3 = p3_budget - (time.monotonic() - p3_start)
    if remaining3 > 20:
        a_sa2, obj_sa2 = simulated_annealing(best_seq, rng, max_iters=150,
                                             T_init=0.02, T_min=0.0005,
                                             time_budget=min(remaining3 * 0.5, 40.0))
        if obj_sa2 < best_score:
            best_score = obj_sa2
            best_seq = a_sa2.copy()
            print(f"  SA2: {best_score:.8f}")

    # === Phase 4: Final polish (remaining time) ===
    remaining4 = deadline - time.monotonic()
    print(f"\n=== Phase 4: Final Polish (budget={remaining4:.0f}s) ===")

    if remaining4 > 10 and best_seq is not None:
        a_fin = project(best_seq.copy())
        n_fin = len(a_fin)

        polish_start = time.monotonic()
        while time.monotonic() < deadline - 5:
            rem = deadline - 5 - time.monotonic()
            if rem < 5:
                break

            # LP (most precise)
            if n_fin <= 1500:
                a_p, obj_p = lp_alternate(a_fin, max_iters=10, time_budget=min(rem * 0.25, 30.0))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_fin = a_p.copy()
                    print(f"  Polish LP: {best_score:.8f}")

            # Gradient
            rem = deadline - 5 - time.monotonic()
            if rem > 3:
                a_g, obj_g = smoothed_gradient_descent(a_fin, max_iters=300,
                                                       time_budget=min(rem * 0.15, 15.0),
                                                       tau_init=0.01, lr_init=0.02)
                if obj_g < best_score:
                    best_score = obj_g
                    best_seq = a_g.copy()
                    a_fin = a_g.copy()
                    print(f"  Polish Grad: {best_score:.8f}")

            # Freq
            rem = deadline - 5 - time.monotonic()
            if rem > 3:
                a_f, obj_f = freq_shape(a_fin, max_iters=50,
                                        time_budget=min(rem * 0.1, 10.0), alpha=0.9)
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    a_fin = a_f.copy()
                    print(f"  Polish Freq: {best_score:.8f}")

            # Small perturbation + re-optimize
            rem = deadline - 5 - time.monotonic()
            if rem > 10:
                a_pert = a_fin * np.exp(rng.normal(0, 0.01, n_fin))
                a_pert = project(a_pert)
                a_pert, obj_pert = smoothed_gradient_descent(a_pert, max_iters=200,
                                                             time_budget=min(rem * 0.2, 10.0))
                if n_fin <= 1200:
                    a_pert2, obj_pert2 = lp_alternate(a_pert, max_iters=8,
                                                      time_budget=min(rem * 0.2, 15.0))
                    if obj_pert2 < obj_pert:
                        a_pert, obj_pert = a_pert2, obj_pert2
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_fin = a_pert.copy()
                    print(f"  Polish Pert: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
