"""
Optimized search for minimum autocorrelation peak sequences.
Uses fast LP alternating (scipy linprog), asymmetric initialization,
gradient polishing with smooth max, and multi-scale exploration.
Key insight: asymmetric sequences can beat the symmetric bound of 2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_min = max(0, j - n + 1)  # k-j >= 0 -> k >= j... wait
        # Actually A[k,j] = a[k-j] if 0 <= k-j < n, i.e., j <= k < j+n
        # So for column j, non-zero entries are at rows k = j, j+1, ..., j+n-1
        # but clamped to [0, 2n-2]
        k_start = j
        k_end = min(2 * n - 1, j + n)
        if k_start < k_end:
            length = k_end - k_start
            A[k_start:k_end, j] = a[:length]
    return A


def solve_minpeak_lp(a):
    """
    Solve LP: min M s.t. A @ g <= M, sum(g) = 1, g >= 0.
    where A is the convolution matrix for fixed a.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Variables: g[0], ..., g[n-1], M (total n+1 variables)
    # Minimize: M
    # s.t.: A @ g - M <= 0 (2n-1 constraints)
    #       sum(g) = 1 (1 equality)
    #       g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@g - M <= 0
    
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    
    return None, None


def smooth_max_obj_grad(a, beta=50.0):
    """
    Compute smooth max objective and its gradient.
    Returns (obj, grad) where grad is d obj / d a (before normalization).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return 1e10, np.zeros(n)
    a_norm = a / s
    conv = compute_conv(a_norm)
    
    # Smooth max: (1/beta) * log(sum(exp(beta * conv_k)))
    max_c = np.max(conv)
    shifted = beta * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    logsum = max_c + np.log(sum_exp) / beta
    obj = 2.0 * n * logsum
    
    # Weights for the softmax
    weights = exp_shifted / sum_exp
    
    # Gradient of smooth_max wrt a_norm:
    # d smax / d a_norm[m] = sum_k weights[k] * 2 * a_norm[k-m] (with bounds)
    grad_an = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            grad_an[indices] += 2.0 * w * a_norm[k - indices]
    
    # Now we need d obj / d a (where a = a_raw, and a_norm = a_raw / s)
    # obj = 2n * smax(a_norm)
    # d obj / d a_raw[m] = 2n * sum_i (d smax / d a_norm[i]) * (d a_norm[i] / d a_raw[m])
    # a_norm[i] = a_raw[i] / s
    # d a_norm[i] / d a_raw[m] = delta_{im}/s - a_raw[i]/s^2 = (delta_{im} - a_norm[i])/s
    # So d obj / d a_raw[m] = 2n/s * (grad_an[m] - sum_i grad_an[i] * a_norm[i])
    
    correction = np.sum(grad_an * a_norm)
    grad_a = (2.0 * n / s) * (grad_an - correction * a_norm)
    
    return obj, grad_a


def gradient_descent_polish(a, time_budget=30.0, beta_start=50.0, beta_end=200.0, max_iters=200):
    """
    Polish using gradient descent with smooth max, annealing beta.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adaptive step size
    step = 0.1
    momentum = np.zeros(n)
    mu = 0.9
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal beta
        progress = it / max_iters
        beta = beta_start + (beta_end - beta_start) * progress
        
        obj, grad = smooth_max_obj_grad(a, beta)
        
        # Project gradient onto tangent space of simplex (sum = const)
        grad = grad - np.mean(grad)
        
        # Momentum
        momentum = mu * momentum - step * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                step = min(step * 1.05, 0.5)
            else:
                step *= 0.7
        else:
            step *= 0.5
    
    return best_a


def lp_alternate(a, time_budget=60.0, verbose=False, max_iters=100):

    """
    n = len(a)

def init_lopsided(n, alpha=0.3, rng=None):
    """
    Create a lopsided initialization: mass concentrated toward one end.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    mode = rng.integers(0, 5)
    
    if mode == 0:
        # Exponential decay
        a = np.exp(-alpha * np.arange(n))
    elif mode == 1:
        # Polynomial bias
        a = np.linspace(1, 1 + alpha * (n-1), n) ** 2
    elif mode == 2:
        # Two bumps (asymmetric)
        center1 = n // 4
        center2 = 3 * n // 4
        width = n // 8
        a = np.exp(-((np.arange(n) - center1)**2) / (2*width**2)) + \
            0.7 * np.exp(-((np.arange(n) - center2)**2) / (2*(width*1.5)**2))
    elif mode == 3:
        # Sigmoid shift
        x = np.arange(n) / n
        a = 1.0 / (1.0 + np.exp(-5.0 * (x - 0.3)))
    else:
        # Random
        a = rng.uniform(0.1, 1.0, n)
    
    a = np.maximum(a, 1e-6)
    s = float(np.sum(a))
    return a / s


def init_smooth_asymmetric(n, rng=None):
    """
    Create a smooth asymmetric initialization using random Fourier features.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    x = np.linspace(0, 1, n)
    n_freq = rng.integers(3, 8)
    a = np.ones(n)
    for _ in range(n_freq):
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    
    a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via spline interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    
    from scipy.interpolate import PchipInterpolator
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        interp = PchipInterpolator(x_old, a)
        a_new = interp(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_targeted_refine(a, time_budget=30.0):
    """
    Target the current peak specifically: redistribute mass from peak-contributing
    positions to peak-reducing positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    max_rounds = 50
    for _ in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        
        # Find positions that contribute most to the peak
        # Contribution of a[i] to conv[k_star] is a[k_star - i] (times 2 for the quadratic term)
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        for i in range(lo, hi + 1):
            contrib[i] = 2.0 * a[k_star - i]
            # Also the symmetric contribution
            j = k_star - i
            if 0 <= j < n and j != i:
                contrib[j] += 2.0 * a[i]
        
        # Find positions that contribute least to OTHER convolution values
        # (i.e., positions where adding mass won't create new peaks)
        # Approximation: find positions with low total contribution across all k
        total_contrib = np.zeros(n)
        for k in range(2 * n - 1):
            lo_k = max(0, k - n + 1)
            hi_k = min(k, n - 1)
            for i in range(lo_k, hi_k + 1):
                total_contrib[i] += a[k - i]
        
        # Source: high contrib to peak, low total contrib to others
        # Target: low contrib to peak, low total contrib to others
        peak_ratio = contrib / (total_contrib + 1e-10)
        
        # Move mass from high peak_ratio to low peak_ratio
        src_candidates = np.argsort(peak_ratio)[-5:]  # top 5 contributors
        tgt_candidates = np.argsort(peak_ratio)[:5]   # bottom 5
        
        improved = False
        for si in src_candidates:
            for ti in tgt_candidates:
                if si == ti or a[si] < 1e-8:
                    continue
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * a[si]
                    a[si] -= delta
                    a[ti] += delta
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[si] += delta
                        a[ti] -= delta
                        break
        
        if not improved:
            break
    
    return best_a
