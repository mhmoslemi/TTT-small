def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):

    s = float(np.sum(a))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)

def build_sparse_toeplitz(a):
    """Build sparse Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    # For each column j, the non-zero entries are at rows j to j+n-1
    # with values a[0], a[1], ..., a[n-1]
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        length = k_end - k_start
        rows.extend(range(k_start, k_end))
        cols.extend([j] * length)
        vals.extend(a[:length])
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    return A

def solve_lp_sparse(a, M, time_limit=30):

def spectral_step(a, alpha=0.3):
    """Single spectral equalization step."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1

    P = np.abs(A) ** 2
    p_max = float(np.max(P))
    if p_max < 1e-30:
        return a.copy()
    
    P_norm = P / p_max
    P_new = (P_norm ** alpha) * p_max
    
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, size))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_sparse_toeplitz(a):
    """Build sparse Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for j in range(n):
        k_end = min(j + n, m)
        length = k_end - j
        rows[idx:idx + length] = range(j, k_end)
        cols[idx:idx + length] = j
        vals[idx:idx + length] = a[:length]
        idx += length
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(m, n))
    return A


def solve_lp_sparse(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0 using sparse matrix."""
    n = len(a)
    try:
        A = build_sparse_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def compute_multi_peak_grad(a, top_k=10, tau=0.005):
    """Compute gradient weighted by top-k autocorrelation peaks."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    size = 1
    while size < 3 * n:
        size <<= 1

    w_ext = np.zeros(2 * n - 1)
    w_ext[:] = weights
    a_flip = a[::-1]

    Wf = rfft(w_ext, size)
    Af = rfft(a_flip, size)
    corr = irfft(Wf * Af, size)
    grad = 2.0 * corr[n - 1:2 * n - 1]

    return grad, conv, cmax

def spectral_step(a, alpha=0.3):
    """Single spectral equalization step (POCS-like)."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1

    A = fft(a, size)
    P = np.abs(A) ** 2
    p_max = float(np.max(P))
    if p_max < 1e-30:
        return a.copy()

    P_norm = P / p_max
    P_new = np.power(P_norm, alpha) * p_max

    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, size))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

    # ============ Phase 2: Momentum proximal gradient (multiple configs) ============
    print(f"\n{'='*60}")
    print(f"Phase 2: Momentum PG (multi-config)")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        configs = [
            (5, 0.003, 0.04, 0.7),
            (10, 0.005, 0.05, 0.7),
            (15, 0.008, 0.05, 0.75),
            (30, 0.012, 0.06, 0.8),
            (50, 0.02, 0.07, 0.8),
            (100, 0.03, 0.08, 0.85),
        ]
        for (tk, ti, si, mom) in configs:
            if time.monotonic() > deadline - 350:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.1, 35.0)

            a_pg = proximal_grad_optimize(a, max_iters=250, time_budget=alloc,
                                           top_k=tk, tau_init=ti, step_init=si, momentum=mom)
            obj_pg = compute_obj(a_pg)
            if obj_pg < best_score:
                best_score = obj_pg
                best_seq = a_pg.copy()
                a = a_pg
                print(f"  PG(k={tk},tau={ti}): {best_score:.8f} **")
            else:
                print(f"  PG(k={tk},tau={ti}): {obj_pg:.8f}")
                a = normalize(0.7 * a + 0.3 * a_pg)

        # Follow-up LP
        if n_parent <= 800 and time.monotonic() < deadline - 250:
            a_new = lp_alternate(a, max_iters=15, time_budget=45.0)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Post-PG LP: {best_score:.8f}")

    # ============ Phase 3: Systematic n-exploration ============
    print(f"\n{'='*60}")
    print(f"Phase 3: N-exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700
    n_options = sorted(set([n_best + dn for dn in
                            [-300, -200, -100, -50, -25, 0, 25, 50, 100, 200, 300, 500]
                            if 80 <= n_best + dn <= 2500]))

    for n_try in n_options:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break

        if best_seq is not None:
            if n_try == n_best:
                a_try = normalize(best_seq.copy())
            else:
                a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try

        alloc = min(remaining * 0.07, 20.0)

        # Quick optimize
        a_try = proximal_grad_optimize(a_try, max_iters=80, time_budget=alloc * 0.5,
                                         top_k=10, tau_init=0.005, step_init=0.04, momentum=0.7)
        if n_try <= 600:
            a_try2 = lp_alternate(a_try, max_iters=6, time_budget=alloc * 0.4)
            if compute_obj(a_try2) < compute_obj(a_try):
                a_try = a_try2

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # ============ Phase 4: Population-based diversity ============
    print(f"\n{'='*60}")
    print(f"Phase 4: Population diversity")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed)
        population = [best_seq.copy()]

        # Generate diverse population members
        for i in range(4):
            if i == 0:
                member = best_seq * np.exp(rng.normal(0, 0.12, n_cur))
            elif i == 1:
                A = fft(best_seq, 2 * n_cur)
                noise = rng.normal(0, 0.15, len(A))
                A_pert = A * (1.0 + noise)
                member = np.real(ifft(A_pert, 2 * n_cur))[:n_cur]
            elif i == 2:
                a_rand = rng.dirichlet(np.ones(n_cur) * 0.3)
                member = 0.6 * normalize(best_seq) + 0.4 * a_rand
            else:
                member = best_seq.copy()
                BL = int(rng.integers(n_cur // 4, n_cur // 2))
                si = int(rng.integers(0, n_cur - BL))
                member[si:si + BL] *= rng.uniform(0.4, 1.6)
                member = member + 0.05 * best_seq[::-1]
            member = normalize(member)
            population.append(member)

        # Optimize each population member
        for pi in range(len(population)):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.12, 30.0)

            a_mem = population[pi]
            a_mem, obj_mem = diversified_optimize(a_mem, time.monotonic() + alloc)

            if obj_mem < best_score:
                best_score = obj_mem
                best_seq = a_mem.copy()
                print(f"  Pop member {pi}: {best_score:.8f} **")
            else:
                print(f"  Pop member {pi}: {obj_mem:.8f}")

    # ============ Phase 5: Intensive final polish ============
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        n_final = len(best_seq)
        a_final = normalize(best_seq.copy())

        for r in range(6):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            alloc = min(remaining * 0.18, 25.0)

            # LP if possible
            if n_final <= 700 and alloc > 12:
                a_pol = lp_alternate(a_final, max_iters=10, time_budget=alloc * 0.45)
            else:
                a_pol = a_final.copy()

            # Fine gradient
            a_pol = proximal_grad_optimize(a_pol, max_iters=100, time_budget=alloc * 0.35,
                                             top_k=5, tau_init=0.002, step_init=0.03, momentum=0.8)

            # Spectral touch
            a_pol = spectral_step(a_pol, alpha=0.15)
            obj_spec = compute_obj(a_pol)

            # Ultra-fine gradient
            a_pol2 = proximal_grad_optimize(a_pol, max_iters=50, time_budget=min(5.0, alloc * 0.15),
                                              top_k=3, tau_init=0.001, step_init=0.02, momentum=0.85)
            if compute_obj(a_pol2) < obj_spec:
                a_pol = a_pol2

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {r}: {best_score:.8f} **")
            else:
                a_final = a_pol
                print(f"  Polish {r}: {obj_pol:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
