import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Bisection on lambda
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > 1.0:
            lo = mid

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} via bisection on lambda."""
    n = len(v)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - (lo + hi) * 0.5, 0.0)


def soft_subgradient(a, conv, tau):
    """Compute soft subgradient using exponential weighting over peaks."""
    n = len(a)
    m = 2 * n - 1
    max_c = float(np.max(conv))
    if tau < 1e-12:
        idx = int(np.argmax(conv))
        weights = np.zeros(m)
        weights[idx] = 1.0
    else:
        shifted = (conv - max_c) / tau
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(m)
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
    sg = 2.0 * np.correlate(weights, a, mode='valid')
    return sg


def hard_subgradient_topk(a, conv, k=3):
    """Compute subgradient using top-k peaks as active set."""
    n = len(a)
    m = 2 * n - 1
    max_c = float(np.max(conv))
    eps = max_c * 1e-6
    active = np.where(conv >= max_c - eps)[0]
    if len(active) < k:
        idx = np.argsort(conv)[::-1][:k]
        active = idx
    else:
        active = active[:k] if len(active) <= k * 3 else active
    weights = np.zeros(m)
    weights[active] = 1.0 / len(active)
    sg = 2.0 * np.correlate(weights, a, mode='valid')
    return sg


def subgradient_optimize(a, time_budget=60.0, tau_init=0.005, tau_decay=0.995,
                         step_init=0.05, momentum=0.9, max_steps=5000):
    """
    Subgradient optimization with soft active set, backtracking line search,
    and Nesterov-like acceleration.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_init
    step = step_init
    v = np.zeros(n)  # momentum
    a_prev = a.copy()

    for it in range(max_steps):
        if time.monotonic() - start > time_budget - 0.5:
            break

        conv = compute_conv(a)
        f_curr = float(np.max(conv))

        # Compute subgradient
        sg = soft_subgradient(a, conv, tau)
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-15:
            break

        # Nesterov-like: use momentum
        v = momentum * v + sg
        v_centered = v - np.mean(v)

        # Backtracking line search
        accepted = False
        for bt in range(20):
            trial = a - step * v_centered
            trial = project_simplex(trial)
            conv_trial = compute_conv(trial)
            f_trial = float(np.max(conv_trial))

            if f_trial < f_curr:
                a = trial
                step = min(step * 1.1, 0.5)
                accepted = True
                break
            else:
                step *= 0.5

        if not accepted:
            # Reset momentum and try smaller step
            v = np.zeros(n)
            step = step_init * 0.1
            a = project_simplex(a - step * soft_subgradient(a, conv, tau))
            f_curr = float(np.max(compute_conv(a)))

        obj = 2.0 * n * f_curr
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        # Cool temperature
        tau *= tau_decay
        a_prev = a.copy()

    return best_a


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_direction(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def simulated_annealing(a, time_budget=300.0, T_init=0.005, T_final=1e-7,
                        cool_rate=0.999, subgrad_budget=2.0):
    """
    Simulated annealing using subgradient steps as local moves.
    Accepts worse solutions with probability exp(-delta/T).
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    T = T_init
    steps_per_iter = 10

    while time.monotonic() - start < time_budget - 1:
        # Perform a few subgradient steps
        a_cur = a.copy()
        for _ in range(steps_per_iter):
            if time.monotonic() - start > time_budget - 1:
                break
            conv = compute_conv(a_cur)
            tau = max(0.001, T * n * 0.1)
            sg = soft_subgradient(a_cur, conv, tau)
            sg_centered = sg - np.mean(sg)
            sg_norm = np.linalg.norm(sg_centered)
            if sg_norm < 1e-15:
                break
            step = 0.02 * n / sg_norm
            a_new = project_simplex(a_cur - step * sg_centered)
            a_cur = a_new

        obj_cur = compute_obj(a_cur)

        # Accept or reject
        if obj_cur < best_obj:
            best_obj = obj_cur
            best_a = a_cur.copy()
            a = a_cur
        else:
            delta = obj_cur - best_obj
            if delta < T * 10:
                prob = np.exp(-delta / max(T, 1e-15))
                if rng.random() < prob:
                    a = a_cur
                else:
                    a = best_a.copy()
                    # Small perturbation
                    a = normalize(a * np.exp(rng.normal(0, 0.02 * (T / T_init + 0.1), n)))

        T = max(T * cool_rate, T_final)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def structured_starts(n, rng):
    """Generate diverse structured initial sequences of length n."""
    starts = []
    t = np.linspace(0, 1, n)

    # Triangular
    starts.append(np.sin(np.pi * t))

    # Exponential decay
    starts.append(np.exp(-3.0 * t))

    # Bimodal
    starts.append(np.exp(-50 * (t - 0.3)**2) + np.exp(-50 * (t - 0.7)**2))

    # Cosine window
    starts.append(0.5 + 0.5 * np.cos(2 * np.pi * t))

    # Random smooth (spline-like)
    coeff = rng.normal(0, 1, 5)
    starts.append(np.abs(np.polyval(coeff, t)))

    # Flat with edges reduced
    flat = np.ones(n)
    edge = int(n * 0.1)
    flat[:edge] = np.linspace(0.3, 1.0, edge)
    flat[-edge:] = np.linspace(1.0, 0.3, edge)
    starts.append(flat)

    # Power law
    starts.append((1.0 - t + 0.01) ** 0.5)

    return [normalize(s) for s in starts]


def coordinate_descent(a, time_budget=15.0):
    """Targeted coordinate descent on the convolution peak."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contrib[:4]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            # Move mass from contributor pair to random location
            for _ in range(2):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue
                a_try = a.copy()
                delta = 0.03 * a_try[j1]
                a_try[j1] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and rng.random() < 0.15:
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase strategy:
    Phase 1: Subgradient optimization from diverse structured starts
    Phase 2: LP polishing on best result
    Phase 3: Simulated annealing for global exploration
    Phase 4: Multi-n exploration
    Phase 5: Final intensive polish (LP + subgradient + coordinate descent)
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    rng = np.random.default_rng(seed)

    # =========================================================
    # Phase 1: Subgradient optimization from multiple starts
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Subgradient optimization")
    print(f"{'='*60}")

    # Start from parent
    if best_seq is not None:
        a = normalize(best_seq.copy())
        a = subgradient_optimize(a, time_budget=80.0, tau_init=0.005,
                                 tau_decay=0.995, step_init=0.05, momentum=0.85)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Parent subgrad: {best_score:.8f}")

    # Try structured starts at parent n
    structured = structured_starts(n_parent, rng)
    for i, s in enumerate(structured[:4]):
        if time.monotonic() > deadline - 400:
            break
        a = subgradient_optimize(s, time_budget=30.0, tau_init=0.01,
                                 tau_decay=0.99, step_init=0.03, momentum=0.8)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Structured {i}: {best_score:.8f} **")
        else:
            print(f"  Structured {i}: {obj:.8f}")

    # =========================================================
    # Phase 2: LP polishing
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP polishing")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        a = lp_polish(a, time_budget=min(180, (deadline - time.monotonic()) * 0.25),
                      max_iters=35)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  LP polish 1: {best_score:.8f}")

        # Second LP pass with perturbation
        if time.monotonic() < deadline - 300:
            a = normalize(best_seq * np.exp(rng.normal(0, 0.02, len(best_seq))))
            a = lp_polish(a, time_budget=min(120, (deadline - time.monotonic()) * 0.2),
                          max_iters=25)
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP polish 2: {best_score:.8f}")

    # =========================================================
    # Phase 3: Simulated annealing
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Simulated annealing")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 100:
        sa_budget = min(250, (deadline - time.monotonic()) * 0.3)
        a = normalize(best_seq.copy())
        a = simulated_annealing(a, time_budget=sa_budget,
                                T_init=0.003, T_final=1e-7,
                                cool_rate=0.9995)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  SA: {best_score:.8f} **")
        else:
            print(f"  SA: {obj:.8f} (no improvement)")

        # Follow up with subgradient + LP
        if time.monotonic() < deadline - 80:
            a = subgradient_optimize(normalize(best_seq.copy()),
                                     time_budget=30.0, tau_init=0.002,
                                     tau_decay=0.998, step_init=0.02, momentum=0.9)
            a = lp_polish(a, time_budget=40.0, max_iters=15)
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Post-SA polish: {best_score:.8f}")

    # =========================================================
    # Phase 4: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700
    n_options = sorted(set([
        max(200, n_best - 200), max(150, n_best - 100),
        n_best,
        n_best + 100, n_best + 200, min(1500, n_best + 400)
    ]))

    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if best_seq is not None:
            if n_try == n_best:
                a_try = normalize(best_seq.copy())
            else:
                a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try

        # Quick subgradient optimization
        budget = min(remaining * 0.15, 35.0)
        a_try = subgradient_optimize(a_try, time_budget=budget * 0.6,
                                     tau_init=0.005, tau_decay=0.995,
                                     step_init=0.03, momentum=0.85)
        # Brief LP
        if budget > 10 and n_try <= 1000:
            a_try = lp_polish(a_try, time_budget=budget * 0.35, max_iters=8)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 5: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        n_final = len(best_seq)
        a_final = normalize(best_seq.copy())

        for polish_round in range(6):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # Subgradient
            a_pol = subgradient_optimize(a_final, time_budget=min(20, remaining * 0.25),
                                         tau_init=0.001, tau_decay=0.999,
                                         step_init=0.015, momentum=0.92)

            # LP
            if remaining > 25 and n_final <= 1000:
                a_pol = lp_polish(a_pol, time_budget=min(30, remaining * 0.3),
                                  max_iters=12)

            # Coordinate descent
            if remaining > 15:
                a_pol = coordinate_descent(a_pol, time_budget=min(8, remaining * 0.15))

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish_round}: {best_score:.8f} **")
            else:
                a_final = a_pol
                print(f"  Polish {polish_round}: {obj_pol:.8f}")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
