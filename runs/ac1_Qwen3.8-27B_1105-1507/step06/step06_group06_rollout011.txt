Also, for "spread out" sequences, the peak is typically at the center. For "concentrated" sequences, the peak might be at the edges (k=0 or k=2n-2). Since we want to minimize the peak, we'll be looking at "spread out" sequences, so the peak will be near the center.

"""
Optimized sequence design for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Combines LP-based alternating minimization, soft-max subgradient
descent with adaptive step sizes, frequency-domain smoothing, and systematic
multi-length exploration. Key improvements: (1) more efficient LP solves with

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Fast self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute full objective."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def build_footprint(a):
    """Build the (2n-1) x n matrix A where (A@b)[k] = conv(a,b)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = a[:end - start]
    return A


def solve_lp_for_g(a, M, time_limit=10):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    if n > 3000:
        return None
    A = build_footprint(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=10):
    """Solve: min t s.t. conv(a,g) <= t for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 3000:
        return None, None
    A = build_footprint(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def soft_max_subgradient(a, max_iters=200, time_budget=30.0, tau_init=0.01, k_top=5):
    """
    Subgradient descent using top-k peak weighting for smoother gradients.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Use top-k peaks with soft weighting
        tau = tau_init * (1.0 + 0.5 * it / max_iters)
        # Weight by exp((conv - cmax) / tau)
        shifted = (conv - cmax) / tau
        weights = np.exp(np.clip(shifted, -50, 0))
        w_sum = np.sum(weights)
        if w_sum < 1e-100:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient: grad[i] = sum_k weights[k] * 2*a[k-i] (valid k-i)
        grad = np.zeros(n)
        # Use FFT for efficiency
        # grad[i] = sum_k w[k] * 2*a[k-i] where 0<=k-i<n
        # = 2 * (w_rev * a) shifted... let's use direct computation via FFT
        N_fft = 1
        while N_fft < 2 * n - 1 + n:
            N_fft <<= 1
        fw = rfft(weights, N_fft)
        fa = rfft(a, N_fft)
        # Cross-correlation: sum_k w[k] * a[k-i]
        # This is ifft(fft(a) * conj(fft(w)))... 
        # Actually, let me think. sum_k w[k] * a[k-i] is the cross-correlation of w and a.
        # In terms of convolution: sum_k w[k] * a[k-i] = (w_rev * a)[n-1-i + ...]
        # Simpler: just use the direct formula
        # For each k with significant weight, add w[k] * 2 * a[k-i] to grad[i] for valid i
        significant = weights > 1e-10
        k_indices = np.where(significant)[0]
        for k in k_indices:
            wk = weights[k]
            if wk < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_end > i_start:
                i_vals = np.arange(i_start, i_end)
                a_vals = a[k - i_vals]
                grad[i_vals] += 2.0 * wk * a_vals

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        # Adaptive step size
        # Use a fixed step that decreases over time
        step = 0.1 * (1.0 + it / max_iters) ** (-0.5)
        
        a_new = a - step * grad_centered
        a_new = project_simplex(a_new)
        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj

def freq_domain_smooth(a, max_iters=100, time_budget=20.0):
    """Frequency domain smoothing of the power spectrum."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        ps_max = np.max(power)
        if ps_max < 1e-15:
            break

        ps_norm = np.maximum(power / ps_max, 1e-10)
        alpha = 0.5 + 0.3 * (it / max_iters)
        ps_compressed = ps_norm ** alpha

        phase = np.angle(fa)
        magnitude = np.sqrt(ps_compressed * ps_max)
        fa_new = magnitude * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_simplex(a_new)

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def multiplicative_update(a, max_iters=100, time_budget=20.0):
    """
    Multiplicative update inspired by matrix factorization algorithms.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Find the peak lag
        k_star = int(np.argmax(conv))
        
        # Compute: for each i, the "pressure" from the peak
        # gradient[i] = 2*a[k_star - i] (valid)
        pressure = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pressure[i] = 2.0 * a[j]
        
        # Multiplicative update: a[i] *= (1 - eta * pressure[i] / max(pressure))
        # But ensure non-negativity and sum=1
        max_p = np.max(pressure)
        if max_p < 1e-15:
            break
        
        eta = 0.1 * (1.0 + it / max_iters) ** (-1.0)
        a_new = a * (1.0 - eta * pressure / max_p)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangular
    x = np.abs(np.arange(n) - n/2) / (n/2)
    tri = (1 - x)
    inits.append(tri / tri.sum())
    
    # Sine
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2
    inits.append(sine / sine.sum())
    
    # Sine^0.5 (wider)
    sine_wide = np.sin(np.pi * x) ** 0.5
    inits.append(sine_wide / sine_wide.sum())
    
    # Cosine bump
    cos_bump = np.cos(np.pi * (x - 0.5)) ** 2
    cos_bump = np.maximum(cos_bump, 0)
    if cos_bump.sum() > 0:
        inits.append(cos_bump / cos_bump.sum())
    
    # Gaussian
    xg = np.linspace(-3, 3, n)
    gauss = np.exp(-xg**2 / 3)
    inits.append(gauss / gauss.sum())
    
    # Bimodal
    bi = np.exp(-((x - 0.3)**2) / 0.01) + np.exp(-((x - 0.7)**2) / 0.01)
    inits.append(bi / bi.sum())
    
    # Random smooth
    rough = rng.random(n)
    # Smooth with a kernel
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rough, kernel, mode='same')
    inits.append(project_simplex(smooth))
    
    # Parent resampled
    if parent is not None:
        inits.append(project_simplex(resample(parent, n)))
        # Perturbed parent
        p_res = resample(parent, n)
        pert = p_res * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_simplex(pert))
        # More perturbed
        pert2 = p_res * np.exp(rng.normal(0, 0.15, n))
        inits.append(project_simplex(pert2))
    
    return inits

Optimized sequence design for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Combines LP-based alternating minimization, soft-max subgradient
descent with adaptive step sizes, frequency-domain smoothing, and systematic
multi-length exploration. Key improvements: (1) more efficient LP solves with
multiple blend factors, (2) improved subgradient using top-k peak weighting
with adaptive temperature, (3) better time management with progressive
refinement, (4) systematic length sweep, (5) stall detection with escape.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Fast self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute full objective."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def build_footprint(a):
    """Build the (2n-1) x n matrix A where (A@b)[k] = conv(a,b)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = a[:end - start]
    return A


def solve_lp_minpeak(a, time_limit=8):
    """Solve: min t s.t. conv(a,g) <= t for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 2500:
        return None, None
    A = build_footprint(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def soft_max_subgradient(a, max_iters=300, time_budget=30.0, tau_init=0.01, step_init=0.15):
    """
    Subgradient descent using top-k peak weighting for smoother gradients.
    Adaptive tau and step size with stall detection.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        tau = tau_init * (1.0 + 0.3 * it / max_iters)
        shifted = (conv - cmax) / tau
        weights = np.exp(np.clip(shifted, -80, 0))
        w_sum = np.sum(weights)
        if w_sum < 1e-100:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient using only significant weights
        grad = np.zeros(n)
        sig_thresh = weights.max() * 1e-6
        k_indices = np.where(weights > sig_thresh)[0]
        for k in k_indices:
            wk = weights[k]
            if wk < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_end > i_start:
                i_vals = np.arange(i_start, i_end)
                a_vals = a[k - i_vals]
                grad[i_vals] += 2.0 * wk * a_vals

        grad_centered = grad - np.mean(grad)

        # Adaptive step: decrease over time, increase if stalled

        obj_new = compute_obj_normalized(a_new)

            best_a = a_new.copy()
            stall_count = 0
        else:

            # If long stall, revert to best and use larger step
            if stall_count % 20 == 0:
                a = best_a.copy()

        a = a_new

def freq_domain_smooth(a, max_iters=150, time_budget=20.0):

        if time.monotonic() - start > time_budget:

        power = np.abs(fa) ** 2

            best_a = a_new.copy()

    return best_a, best_obj
