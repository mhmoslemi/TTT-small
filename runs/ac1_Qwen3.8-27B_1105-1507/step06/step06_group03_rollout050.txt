def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s) * (s * s)
                    # Actually just compute raw
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                # Scale to compare: we want to compare 2n*max/s^2
                # Let's just track the full J
                new_j = 2.0 * n * conv.max() / (new_s * new_s)
                if new_j < best_local / (s * s):
                    best_local / (s * s)  # placeholder
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            # Simpler: just track J directly
            best_val = old
            best_j = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                new_s = s - old + x
                if new_s < 1e-10:
                    new_j = float('inf')
                else:
                    new_j = 2.0 * n * conv.max() / (new_s * new_s)
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

                if new_j < best_j:
                    best_j = new_j
                    best_val = x

            if abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s + delta
                improved = True

        current_j = 2.0 * n * conv.max() / (s * s)
        if current_j < best_obj / (1.0) - 1e-13:
            # best_obj was stored as 2n*max (without dividing by s^2 since s=1 initially)
            # Let's just track J consistently
            pass

        # Track properly
        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj / (s * s) if s != 1 else current_obj < best_obj:
            pass

        # Simplified: just recompute
        obj_now = 2.0 * n * conv.max() / (s * s)
        # We compare against the running best
        if obj_now < best_obj - 1e-13:
            best_obj = obj_now
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def sca_lp_improved(a, max_iters, deadline, verbose=False):
    """
    Improved SCA-LP: At each step, linearize the max constraint and solve LP.
    Key improvement: use a tighter target M and solve a cleaner LP.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()
        # Use slightly relaxed M to allow more flexibility
        M_target = M * 1.0  # tight

        # LP: minimize t
        # s.t. 2*(a*g)[k] <= c[k] + t  for all k
        #      sum(g) = 1
        #      g >= 0
        # Variables: g[0..n-1], t (index n)

        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]

        A_eq = coo_matrix((np.ones(n), (list(range(n)), list(range(n)))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    J_g = compute_J(g)
                    if J_g < best_obj - 1e-14:
                        a = g.copy()
                        best_obj = J_g
                        best_a = g.copy()
                        no_improve = 0
                    else:
                        found = False
                        for frac in [0.5, 0.3, 0.2, 0.1]:
                            a_try = (1 - frac) * a + frac * g
                            a_try = np.maximum(a_try, 0)
                            if a_try.sum() < 1e-10:
                                continue
                            a_try /= a_try.sum()
                            J_try = compute_J(a_try)
                            if J_try < best_obj - 1e-14:
                                a = a_try
                                best_obj = J_try
                                best_a = a_try.copy()
                                no_improve = 0
                                found = True
                                break
                        if not found:
                            no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 10 == 0:
            print(f"      SCA it {it}: obj={best_obj:.8f}", flush=True)

        if no_improve > 30:
            break

    return best_a, best_obj


def mirror_descent(a, max_iters, deadline, lr_init=0.1, verbose=False):
    """
    Mirror descent with entropy regularization on the simplex.
    Minimizes max(conv(a,a)) by moving mass away from high-contribution coordinates.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()
        # Find which bins are at or near the max
        threshold = M - 0.01 * M
        hot_bins = np.where(c >= threshold)[0]

        # Compute subgradient: g_i = sum_{k in hot} w_k * (a[(k-i)%n-ish])
        # For each hot bin k, the gradient of conv[k] w.r.t. a_i is:
        #   a[k-i] if 0 <= k-i < n, plus a[i-k+n-1]... 
        # Actually: conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/da[i] = a[k-i] (if valid) + a[i-k] (if valid, i.e., k-j=i means j=k-i)
        # Wait: d/d a_i of sum_j a_j * a_{k-j}
        # = a_{k-i} (from j=i term) + a_{i} * 0... 
        # No: = a_{k-i} (j=i) + sum over j where k-j=i, i.e., j=k-i: a_j * 1 = a_{k-i}
        # So if k != 2i: gradient contribution is 2*a_{k-i}
        # If k == 2i: gradient contribution is a_i (since a_{k-i} = a_i, and it's a_i^2)
        # Wait no. conv[k] = sum_j a_j * a_{k-j}. 
        # d/da_i = a_{k-i} (from j=i) + (if there exists j such that k-j=i, i.e., j=k-i) a_j
        # The second term: j = k-i, so a_{k-i} * 1 = a_{k-i}
        # So total: a_{k-i} + a_{k-i} = 2*a_{k-i} (when i != k-i, i.e., 2i != k)
        # When 2i == k: the term a_i * a_i gives derivative 2*a_i, so it's still 2*a_i.
        # Wait, d(a_i^2)/da_i = 2*a_i. And the cross terms: for j where k-j = i and j != i, 
        # we get a_j. But if k = 2i, then j = k-i = i, so it's the same term.
        # So d(conv[k])/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} when k-i is in [0,n-1]
        # and = a_{k-i} when only one of i or k-i is valid.
        # More precisely: 
        #   term1: j=i contributes a_i * a_{k-i} -> derivative a_{k-i} (if 0<=k-i<n)
        #   term2: j=k-i contributes a_{k-i} * a_i -> derivative a_i * 1 = a_i... 
        #   Wait no. d/da_i of a_{k-i} * a_i = a_{k-i} (treating a_{k-i} as constant w.r.t. a_i)
        #   But if k-i = i (i.e., k=2i), then it's a_i * a_i = a_i^2, derivative = 2*a_i.
        #   If k-i != i: two separate terms, each contributing a_{k-i}, total 2*a_{k-i}.
        # So: d(conv[k])/da_i = 2*a_{k-i} if 0<=k-i<n and k-i != i
        #                           a_i (=2*a_i/2... no) 
        # Let me just be explicit:
        # d(conv[k])/da_i = [1 if 0<=k-i<n else 0] * a_{k-i} + [1 if 0<=k-i<n else 0] * a_{k-i}
        #                  = 2*a_{k-i} if 0<=k-i<n
        # But when k=2i: a_{k-i} = a_i, and the derivative of a_i^2 is 2*a_i, which matches 2*a_{k-i}=2*a_i. Good.
        # When i is valid but k-i is not: only one term, gradient = a_{k-i}... but k-i not valid means a_{k-i}=0.
        # So gradient is 0 in that case from that bin.
        # Actually I need to be more careful. Let me just compute it directly.
        
        grad = np.zeros(n)
        for k in hot_bins:
            for i in range(n):
                # d(conv[k])/da_i
                # conv[k] = sum_j a[j] * a[k-j] (a[m]=0 for m outside [0,n-1])
                # d/da_i: 
                #   from j=i: a[k-i] if 0<=k-i<n
                #   from j=k-i: a[i] * 1 if 0<=k-i<n (i.e., j=k-i is valid)
                #   These are the same when i = k-i i.e. 2i=k
                gk = 0.0
                if 0 <= k - i < n:
                    gk += a[k - i]
                if 0 <= k - i < n:
                    gk += a[i] if (k - i) != i else a[i]
                    # Hmm this is getting confusing. Let me just do:
                # d(conv[k])/da_i = a[k-i] * 1{i valid and k-i valid} + a[k-i] * 1{k-i valid and i valid}
                # = 2*a[k-i] if both i and k-i are in [0,n-1]
                # = 0 otherwise (since if k-i is out of range, a[k-i]=0)
                # But wait, that's only if i != k-i. If i == k-i, it's a[i]^2 and d/d a_i = 2*a[i].
                # And 2*a[k-i] = 2*a[i] when i=k-i. So the formula 2*a[k-i] works in all cases.
                # EXCEPT when we need to check validity: a[k-i] is 0 if k-i not in [0,n-1].
                if 0 <= k - i < n:
                    gk = 2.0 * a[k - i]
                else:
                    gk = 0.0
                grad[i] += gk

        # Normalize gradient (optional, for stability)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm

        # Project to simplex tangent: subtract mean
        grad = grad - np.mean(grad)

        # Mirror descent step: a_new_i = a_i * exp(-lr * grad_i) / Z
        scaled = a * np.exp(-lr * grad)
        Z = scaled.sum()
        if Z > 1e-15:
            a_new = scaled / Z
        else:
            a_new = a.copy()

        J_new = compute_J(a_new)
        if J_new < best_obj - 1e-14:
            best_obj = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Adaptive lr increase
            lr = min(lr * 1.05, 2.0)
        else:
            no_improve += 1
            lr = max(lr * 0.8, 0.01)
            a = a  # stay at current

        if no_improve > 50:
            break

    return best_a, best_obj


def lp_best_response(a, max_iters, deadline, verbose=False):
    """
    LP-based best response: find g maximizing sum(g) subject to (a*g)[k] <= M.
    Then blend toward g.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # LP: maximize sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [0.5, 0.3, 0.2, 0.1, 0.05]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-14:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 25:
            break

    return best_a, best_obj


def frank_wolfe(a, max_iters, deadline, verbose=False):
    """
    Frank-Wolfe (conditional gradient) on the min-max problem.
    At each step, find the simplex vertex that most reduces the max autocorrelation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()
        # Hot bins
        threshold = M - 0.05 * M
        hot = np.where(c >= threshold)[0]

        # Subgradient: g_i = sum_{k in hot} d(conv[k])/da_i
        grad = np.zeros(n)
        for k in hot:
            # d(conv[k])/da_i = 2*a[k-i] if 0<=k-i<n
            valid_ks = np.arange(n)
            ki = k - valid_ks  # k - i for i in 0..n-1
            valid = (ki >= 0) & (ki < n)
            grad[valid] += 2.0 * a[ki[valid]]

        # FW: find vertex e_j that minimizes <grad, e_j> = grad_j
        j_star = np.argmin(grad)

        # Step size: use 2/(it+2) or line search
        t_step = 2.0 / (it + 2)
        a_new = (1 - t_step) * a + t_step * np.eye(n)[j_star]
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < best_obj - 1e-14:
            best_obj = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            # Try smaller step
            for t_try in [t_step * 0.5, t_step * 0.1, t_step * 0.01]:
                a_try = (1 - t_try) * a + t_try * np.eye(n)[j_star]
                a_try = np.maximum(a_try, 0)
                a_try /= a_try.sum()
                J_try = compute_J(a_try)
                if J_try < best_obj - 1e-14:
                    best_obj = J_try
                    best_a = a_try.copy()
                    a = a_try
                    no_improve = 0
                    break
            else:
                no_improve += 1

        if no_improve > 100:
            break

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns
    for freq in [1, 2, 3, 5, 7]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))

    # Gaussian
    for center in [0.25, 0.5, 0.75]:
        for width in [0.1, 0.2, 0.3]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.05))

    # Power law
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    # Random smooth
    ksize = max(7, n // 12)
    kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) / (ksize * 0.5)) ** 2)
    kernel /= kernel.sum()
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    # Triangular / tent
    inits.append(project_simplex(1.0 - 2.0 * np.abs(t - 0.5)))
    inits.append(project_simplex(np.abs(np.sin(np.pi * t)) + 0.1))

    # Bimodal
    inits.append(project_simplex(np.exp(-((t - 0.3) / 0.15) ** 2) + np.exp(-((t - 0.7) / 0.15) ** 2) + 0.05))

    return inits


def optimize_sequence(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 1: Fast CD (20%)
    t1 = time.monotonic() + 0.20 * remaining
    a = cd_fast(a, max_sweeps=50000, deadline=t1, random_order=True, seed=42)
    a = try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 2: SCA-LP (25%)
    t2 = time.monotonic() + 0.25 * remaining
    a_sca, obj_sca = sca_lp_improved(a, max_iters=100, deadline=t2)
    if obj_sca < best_obj:
        best_a = a_sca.copy()
        best_obj = obj_sca
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 3: LP Best Response (20%)
    t3 = time.monotonic() + 0.20 * remaining
    a_lp, obj_lp = lp_best_response(a, max_iters=80, deadline=t3)
    if obj_lp < best_obj:
        best_a = a_lp.copy()
        best_obj = obj_lp
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 4: Mirror Descent (15%)
    t4 = time.monotonic() + 0.15 * remaining
    a_md, obj_md = mirror_descent(a, max_iters=2000, deadline=t4, lr_init=0.05)
    if obj_md < best_obj:
        best_a = a_md.copy()
        best_obj = obj_md
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 5: Final CD polish (15%)
    t5 = deadline - 2
    a = cd_fast(a, max_sweeps=50000, deadline=t5, random_order=True, seed=123)
    a = try_update(a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses: fast CD, SCA-LP, LP best response, mirror descent, multi-restart, multi-n.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Intensive optimization at parent length ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Intensive optimization at n={n_parent}", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase1_deadline = t_start + min(500.0, 0.35 * (deadline - t_start))

    # Multiple diverse initializations
    inits = generate_inits(n_parent, rng, parent)
    # Limit to top diverse ones
    max_inits = 8
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, max_inits) - 5

    for idx in range(min(len(inits), max_inits)):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 30:
            break
        alloc = min(per_init_time, remaining_check - 5)
        if alloc < 20:
            break

        a0 = inits[idx]
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt, obj_opt = optimize_sequence(a0, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.8f} n={n_parent} (budget {alloc:.0f}s)", flush=True)

    # Also do a longer optimization from the parent directly
    remaining_check = phase1_deadline - time.monotonic()
    if remaining_check > 60:
        a_parent = parent.copy()
        a_parent = a_parent / a_parent.sum()
        a_opt, obj_opt = optimize_sequence(a_parent, time.monotonic() + remaining_check - 5)
        update_best(a_opt)
        print(f"  Parent direct: J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Multi-length exploration ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase2_deadline = t_start + min(900.0, 0.65 * (deadline - t_start))

    candidate_ns = [256, 320, 384, 448, 512, 576, 640, 704, 768, 832, 896, 1024, 1152, 1280]
    candidate_ns = [n for n in candidate_ns if n != n_parent]
    # Sort by distance to parent, prefer nearby
    candidate_ns.sort(key=lambda n: abs(n - n_parent))

    for new_n in candidate_ns[:6]:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 60:
            break
        alloc = min(80.0, remaining_check - 15)
        if alloc < 40:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = optimize_sequence(a_ext, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive local optimization on best with multiple methods ===
    phase3_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive local optimization", flush=True)
    print(f"  Budget: {phase3_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase3_deadline - time.monotonic()
        a_curr = best_seq.copy()

        # Pass 1: Long SCA-LP
        t_pass1 = time.monotonic() + 0.25 * remaining
        print(f"  SCA-LP pass ({t_pass1 - time.monotonic():.0f}s)...", flush=True)
        a_sca, obj_sca = sca_lp_improved(a_curr, max_iters=200, deadline=t_pass1)
        update_best(a_sca)
        a_curr = best_seq.copy()

        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            # Pass 2: LP Best Response
            t_pass2 = time.monotonic() + 0.25 * remaining
            print(f"  LP-BR pass ({t_pass2 - time.monotonic():.0f}s)...", flush=True)
            a_lp, obj_lp = lp_best_response(a_curr, max_iters=100, deadline=t_pass2)
            update_best(a_lp)
            a_curr = best_seq.copy()

        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            # Pass 3: Mirror Descent
            t_pass3 = time.monotonic() + 0.20 * remaining
            print(f"  Mirror Descent pass ({t_pass3 - time.monotonic():.0f}s)...", flush=True)
            a_md, obj_md = mirror_descent(a_curr, max_iters=5000, deadline=t_pass3, lr_init=0.03)
            update_best(a_md)
            a_curr = best_seq.copy()

        remaining = phase3_deadline - time.monotonic()
        if remaining > 20:
            # Pass 4: CD polish
            t_pass4 = time.monotonic() + 0.20 * remaining
            print(f"  CD polish ({t_pass4 - time.monotonic():.0f}s)...", flush=True)
            a_cd = cd_fast(a_curr, max_sweeps=100000, deadline=t_pass4, random_order=True, seed=777)
            update_best(a_cd)
            a_curr = best_seq.copy()

        remaining = phase3_deadline - time.monotonic()
        if remaining > 20:
            # Pass 5: Another SCA-LP round
            t_pass5 = deadline - 5
            print(f"  Final SCA-LP ({t_pass5 - time.monotonic():.0f}s)...", flush=True)
            a_sca2, obj_sca2 = sca_lp_improved(a_curr, max_iters=200, deadline=t_pass5)
            update_best(a_sca2)

            # Final CD
            if time.monotonic() < deadline - 10:
                a_final = cd_fast(best_seq, max_sweeps=50000,
                                  deadline=deadline - 5, random_order=True, seed=999)
                update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
