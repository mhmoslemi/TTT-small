```python
"""
Smoothed L-BFGS optimization with tau annealing for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FFT-based gradient computation and multi-start strategy.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_true_obj(a):
    """Compute the true objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smoothed_obj_and_grad(a, tau):
    """
    Compute smoothed objective and gradient.
    obj = 2n * logsumexp(conv/tau) / (sum(a))^2
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    conv = compute_conv(a)
    
    # log-sum-exp with shift for numerical stability
    conv_scaled = conv / tau
    conv_max = np.max(conv_scaled)
    exp_shifted = np.exp(conv_scaled - conv_max)
    log_sum = np.sum(exp_shifted)
    L = conv_max + np.log(log_sum)  # = logsumexp(conv/tau)
    
    # Weights
    w = exp_shifted / log_sum
    
    # Normalize w to be non-negative (it already is)
    
    # Cross-correlation: corr[i] = sum_j a[j] * w[i+j]
    # Using FFT
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    w_ext = np.zeros(fft_size)
    w_ext[:2*n-1] = w
    
    A = rfft(a, fft_size)
    W = rfft(w_ext, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    
    # dL/da[i] = (2/tau) * corr[i]
    dL = (2.0 / tau) * corr[:n]
    
    # Objective
    obj = 2.0 * n * L / (s * s)
    
    # Gradient
    grad = (2.0 * n / (s * s)) * dL - (4.0 * n * L) / (s ** 3)
    
    return obj, grad


def optimize_smoothed(a_init, tau_schedule, max_iter_per_tau=100, time_budget=120.0):
    """
    Run L-BFGS-B with tau annealing.
    Returns (best_a, best_true_obj).
    """
    start_time = time.monotonic()
    best_a = a_init.copy()
    best_true_obj = compute_true_obj(a_init)
    current_a = a_init.copy()
    
    for tau in tau_schedule:
        if time.monotonic() - start_time > time_budget:
            break
        
        n = len(current_a)
        
        def func_and_grad(x):
            x = np.maximum(x, 0)  # enforce non-negativity (L-BFGS-B should handle bounds)
            obj, grad = smoothed_obj_and_grad(x, tau)
            return obj, grad
        
        result = minimize(
            func_and_grad,
            current_a,
            method='L-BFGS-B',
            jac=True,
            bounds=[(0, None)] * n,
            options={'maxiter': max_iter_per_tau, 'ftol': 1e-12, 'gtol': 1e-8,
                     'maxfun': max_iter_per_tau * 3}
        )
        
        if result.x is not None:
            new_a = np.maximum(result.x, 0)
            s = np.sum(new_a)
            if s > 1e-10:
                new_a = new_a / s  # normalize (doesn't change objective)
                true_obj = compute_true_obj(new_a)
                if true_obj < best_true_obj:
                    best_true_obj = true_obj
                    best_a = new_a.copy()
                current_a = new_a.copy()
    
    return best_a, best_true_obj


def generate_starts(n, num_starts=10, seed=42):
    """Generate diverse starting sequences."""
    rng = np.random.default_rng(seed)
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    starts = []
    
    # 1. Uniform
    starts.append(np.ones(n) / n)
    
    # 2. Hann window
    hann = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
    starts.append(hann / np.sum(hann))
    
    # 3. Gaussian (narrow)
    g1 = np.exp(-((i - center) ** 2) / (2 * (n / 4) ** 2))
    starts.append(g1 / np.sum(g1))
    
    # 4. Gaussian (medium)
    g2 = np.exp(-((i - center) ** 2) / (2 * (n / 3) ** 2))
    starts.append(g2 / np.sum(g2))
    
    # 5. Gaussian (wide)
    g3 = np.exp(-((i - center) ** 2) / (2 * (n / 2.5) ** 2))
    starts.append(g3 / np.sum(g3))
    
    # 6. Raised cosine (Tukey with r=0.5)
    r = 0.5
    tukey = np.ones(n)
    m = int(r * n / 2)
    for j in range(m):
        tukey[j] = 0.5 * (1 + np.cos(np.pi * (2 * j / (r * n) - 1)))
        tukey[n-1-j] = tukey[j]
    starts.append(tukey / np.sum(tukey))
    
    # 7-10: Random smooth sequences
    for _ in range(num_starts - 6):
        # Random Fourier series
        coeffs = rng.normal(0, 1, n // 2 + 1)
        freqs = np.arange(n // 2 + 1)
        x = np.zeros(n)
        for k, (c, f) in enumerate(zip(coeffs, freqs)):
            if k == 0:
                x += c
            else:
                x += c * np.cos(2 * np.pi * f * i / n)
        x = np.abs(x) + 0.1  # ensure positive
        starts.append(x / np.sum(x))
    
    return starts


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def exact_subgrad_polish(a, time_budget=30.0, num_peaks=5):
    """
    Final polish using exact (non-smoothed) subgradient with multi-peak tracking.
    """
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    best_a = a.copy()
    best_obj = compute_true_obj(a)
    start = time.monotonic()
    
    step = 0.05
    v = np.zeros(n)
    mu = 0.9
    
    # FFT setup
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    A = None  # will be recomputed
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Get top K peaks
        k = min(num_peaks, len(conv))
        top_idx = np.argsort(conv)[-k:]
        
        # Equal weights on top K
        w = np.zeros(2 * n - 1)
        for idx in top_idx:
            w[idx] = 1.0 / k
        
        # Compute subgradient
        w_ext = np.zeros(fft_size)
        w_ext[:2*n-1] = w
        A = rfft(a, fft_size)
        W = rfft(w_ext, fft_size)
        corr = irfft(A * np.conj(W), fft_size)
        
        # d(conv_max)/da[i] ≈ (2) * sum_j a[j]*w[i+j] (using the same formula)
        grad = 2.0 * corr[:n]
        grad -= np.mean(grad)  # center (project onto sum=const)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        v = mu * v - step * grad / gn
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        obj_new = compute_true_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.95
            if step < 0.001:
                break
    
    return best_a, best_obj
```
