        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_obj_and_grad(a, beta):
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj

def adam_log_opt(a, max_iters=5000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995):
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)
        
        obj_sm, grad_a = smooth_obj_and_grad(a, beta)
        
        grad_x = grad_a * a
        
        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 200 == 0:
            x -= np.mean(x)
        
        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a

def lbfgs_smooth_opt(a, time_budget=60.0, beta=50.0):
    """Use L-BFGS on the smoothed objective in log-space."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x0 = np.log(a)
    x0 -= np.mean(x0)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    def objective(x):
        if time.monotonic() - start > time_budget:
            return np.inf
        x_c = x - np.max(x)
        a = np.exp(x_c)
        a = a / np.sum(a)
        obj, _ = smooth_obj_and_grad(a, beta)
        return obj
    
    def gradient(x):
        if time.monotonic() - start > time_budget:
            return np.zeros(n)
        x_c = x - np.max(x)
        a = np.exp(x_c)
        a = a / np.sum(a)
        _, grad_a = smooth_obj_and_grad(a, beta)
        grad_x = grad_a * a
        # Remove the mean to maintain the centering constraint
        grad_x -= np.mean(grad_x)
        return grad_x
    
    try:
        res = minimize(objective, x0, jac=gradient, method='L-BFGS-B',
                      options={'maxiter': 5000, 'maxfun': 5000,
                              'maxls': 50, 'ftol': 1e-15, 'gtol': 1e-12})
        
        if time.monotonic() - start < time_budget:
            x_c = res.x - np.max(res.x)
            a_new = np.exp(x_c)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
    except Exception as e:
        pass
    
    return best_a

def peak_shave(a, time_budget=30.0, seed=0):
    """Iteratively reduce the peak of the autocorrelation by redistributing mass."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy().astype(np.float64)
    s = np.sum(best_a)
    if s < 1e-10:
        return best_a
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(seed)
    
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(best_a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]
        mean_val = np.mean(conv)
        
        if peak_val / max(mean_val, 1e-15) < 1.01:
            break
        
        # Find contributing pairs
        i_min = max(0, peak_idx - (n - 1))
        i_max = min(peak_idx, n - 1)
        
        # Get the products for contributing pairs
        products = []
        for i in range(i_min, i_max + 1):
            j = peak_idx - i
            if 0 <= j < n:
                products.append((i, j, best_a[i] * best_a[j]))
        
        if not products:
            break
        
        # Sort by product size (largest first)
        products.sort(key=lambda x: -x[2])
        
        # Try to reduce the largest products
        improved = False
        for i, j, prod in products[:5]:
            if time.monotonic() - start > time_budget:
                break
            if i == j:
                continue
            
            # Try moving mass from the larger to a random other position
            if best_a[i] > best_a[j]:
                src = i
                dst_pool = peak_idx - j  # don't move to position that also contributes
            else:
                src = j
                dst_pool = peak_idx - i  # hmm this doesn't make sense
            
            # Better: try reducing src and increasing a random position far from the peak
            for _ in range(10):
                dst = int(rng.integers(0, n))
                if dst == i or dst == j:
                    continue
                # Check if dst would create a new peak
                # (skip this check for speed)
                
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    delta = frac * best_a[src]
                    if delta > best_a[src] * 0.5:
                        continue
                    a_try = best_a.copy()
                    a_try[src] -= delta
                    a_try[dst] += delta
                    if a_try[src] < 0:
                        continue
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved and it > 100:
            break
    
    return best_a

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=20):
    n = len(a)
    if n > 900:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def lp_refine(a, time_budget=30.0):
    n = len(a)
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    
    conv = compute_conv(a_norm)
    M = float(np.max(conv))
    
    # Try slightly different M values
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for M_scale in [0.95, 1.0, 1.05]:
        if time.monotonic() - start > time_budget:
            break
        g = solve_lp(a_norm, M * M_scale, time_limit=min(time_budget * 0.3, 15))
        if g is None:
            continue
        
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            if time.monotonic() - start > time_budget:
                break
            a_try = (1 - t) * a_norm + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new
