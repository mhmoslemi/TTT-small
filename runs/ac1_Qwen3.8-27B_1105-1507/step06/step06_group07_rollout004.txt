        if j_max >= j_min:
            # a[k-j] for j in [j_min, j_max]
            # When j=j_min: a[k-j_min], when j=j_max: a[k-j_max]
            # So we need a[k-j_max : k-j_min+1] reversed
            idx_end = k - j_min + 1
            idx_start = k - j_max
            segment = a[idx_start:idx_end][::-1]
            grad[j_min:j_max+1] += segment
        
        # Term 2: a[j-k] for j where 0 <= j-k <= n-1 => k <= j <= k+n-1
        j_min2 = max(0, k)
        j_max2 = min(k + n - 1, n - 1)
        if j_max2 >= j_min2:
            # a[j-k] for j in [j_min2, j_max2]
            # When j=j_min2: a[j_min2-k], when j=j_max2: a[j_max2-k]
            idx_start2 = j_min2 - k
            idx_end2 = j_max2 - k + 1
            segment2 = a[idx_start2:idx_end2]
            grad[j_min2:j_max2+1] += segment2
    
    return grad


def mirror_descent_subgrad(a_init, max_iters=5000, time_budget=60.0,
                           top_k_start=20, top_k_end=3,
                           eta_start=0.01, eta_end=0.001,
                           weight=0.7):
    """
    Mirror descent (exponentiated gradient) with adaptive subgradient.
    a_init: initial non-negative vector (will be normalized)
    weight: moving average weight for the iterate (1.0 = no averaging)
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    a = a / s
    
    # Also track running average for faster convergence
    a_avg = a.copy()
    count = 0
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    m = np.zeros(n)  # momentum
    beta_mom = 0.9
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal parameters
        progress = it / max(max_iters - 1, 1)
        top_k = max(3, int(top_k_start - (top_k_start - top_k_end) * progress))
        eta = eta_start * (eta_end / eta_start) ** progress
        
        # Compute subgradient
        g = compute_subgradient(a, top_k=top_k)
        
        # Momentum
        m = beta_mom * m + (1.0 - beta_mom) * g
        
        # Normalize gradient for step size
        g_norm = np.linalg.norm(m)
        if g_norm < 1e-15:
            break
        
        # Mirror descent: a_new = a * exp(-eta * m / g_norm_normalized)
        # Scale eta by the mean gradient magnitude for stability
        g_mean = np.mean(np.abs(m))
        if g_mean < 1e-15:
            break
        scaled_eta = eta * g_norm / (g_mean * n)
        
        # Multiplicative update (log-space for stability)
        log_a = np.log(a)
        log_a -= scaled_eta * m / g_mean
        # Center for numerical stability
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        a_new_sum = np.sum(a_new)
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum
        
        # Update running average
        count += 1
        a_avg = weight * a_avg + (1.0 - weight) * a_new
        a_avg = a_avg / np.sum(a_avg)
        
        a = a_new
        
        # Check objective periodically
        if it % 50 == 0 or it == max_iters - 1:
            obj_cur = compute_obj(a)
            obj_avg = compute_obj(a_avg)
            if obj_cur < best_obj:
                best_obj = obj_cur
                best_a = a.copy()
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = a_avg.copy()
    
    # Final check
    obj_cur = compute_obj(a)
    obj_avg = compute_obj(a_avg)
    if obj_cur < best_obj:
        best_obj = obj_cur
        best_a = a.copy()
    if obj_avg < best_obj:
        best_obj = obj_avg
        best_a = a_avg.copy()
    
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 800:
        return None
    try:
        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, time_budget=20.0):
    """LP-based refinement."""
    n = len(a)
    if n > 800:
        return a
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    
    conv = compute_conv(a_norm)
    M = float(np.max(conv))
    
    g = solve_lp_max_sum(a_norm, M, time_limit=min(time_budget * 0.4, 12))
    if g is None:
        return a
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        if time.monotonic() - start > time_budget:
            break
        a_try = (1 - t) * a_norm + t * g
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
    
    return best_a


def coordinate_refine(a, time_budget=20.0, seed=0):
    """Coordinate descent: adjust pairs of coordinates to reduce peaks."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(best_a)
        max_idx = np.argmax(conv)
        
        # Find positions contributing to the peak
        i_min = max(0, max_idx - (n - 1))
        i_max = min(max_idx, n - 1)
        
        improved = False
        # Try moving mass from high contributors to random other positions
        contributors = list(range(i_min, i_max + 1))
        rng.shuffle(contributors)
        
        for i in contributors[:10]:
            if time.monotonic() - start > time_budget:
                break
            if best_a[i] < 1e-10:
                continue
            # Move some mass to a random position far from the peak
            for _ in range(3):
                j = rng.integers(0, n)
                if j == i:
                    continue
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * best_a[i]
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    s_try = np.sum(a_try)
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
        if not improved and it > 30:
            break
    
    return best_a


def smooth_obj_grad(a, beta):
    """Compute smoothed objective (logsumexp) and gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    # Gradient via cross-correlation
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj


def adam_log_opt(a, max_iters=5000, time_budget=50.0, beta_start=20.0,
                 beta_end=500.0, lr_start=0.03, lr_decay=0.99998):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_centered = x - np.max(x)
        a_cur = np.exp(x_centered)
        a_cur = a_cur / np.sum(a_cur)
        
        obj_sm, grad_a = smooth_obj_grad(a_cur, beta)
        grad_x = grad_a * a_cur
        
        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 200 == 0:
            x -= np.mean(x)
        
        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)

    return None, np.inf


def diverse_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    
    # 1. Resampled parent
    if parent is not None:
        p = resample_sequence(parent, n)
        p = p / np.sum(p)
        inits.append(p)
    
    # 2. Uniform
    inits.append(np.ones(n) / n)
    
    # 3. Gaussian window
    x = np.arange(n) / (n - 1)
    g = np.exp(-20 * (x - 0.5) ** 2)
    g = g / np.sum(g)
    inits.append(g)
    
    # 4. Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    rc = rc / np.sum(rc)
    inits.append(rc)
    
    # 5. Power of parent (flatter)
    if parent is not None:
        p = resample_sequence(parent, n)
        p = p / np.sum(p)
        p2 = p ** 1.5
        p2 = p2 / np.sum(p2)
        inits.append(p2)
    
    # 6. Random
    r = rng.random(n)
    r = r / np.sum(r)
    inits.append(r)
    
    # 7. Bimodal
    x = np.arange(n) / (n - 1)
    b = np.exp(-50 * (x - 0.3) ** 2) + np.exp(-50 * (x - 0.7) ** 2)
    b = b / np.sum(b)
    inits.append(b)
    
    # 8. Inverse cosine (centered peak)
    ic = 1.0 / (0.1 + np.abs(x - 0.5))
    ic = ic / np.sum(ic)
    inits.append(ic)
    
    return inits

        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            a_hb = adam_log_opt(best_seq, max_iters=8000,
                                time_budget=min(remaining * 0.3, 15.0),
                                beta_start=200.0, beta_end=1000.0,
                                lr_start=0.005, lr_decay=0.99999)
            obj_hb = compute_obj(a_hb)
            if obj_hb < best_score:
                best_score = obj_hb
                best_seq = a_hb.copy()
                print(f"  High-beta Adam: {best_score:.8f} ***")
    
    # === Phase 6: Random restarts (if time remains) ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Random restarts")
    print(f"{'='*60}")
    
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        
        for restart in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            
            # Perturb
            strategy = restart % 3
            if strategy == 0:
                # Multiplicative noise
                scale = 0.05 + 0.03 * restart
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            elif strategy == 1:
                # Blend with uniform
                alpha = 0.1 + 0.05 * restart
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * (1.0/n_best)
            else:
                # Blend with gaussian
                x = np.arange(n_best) / (n_best - 1)
                g = np.exp(-10 * (x - 0.5)**2)
                g = g / np.sum(g)
                alpha = 0.1 + 0.05 * restart
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * g
            
            a_pert = np.maximum(a_pert, 1e-12)
            a_pert = a_pert / np.sum(a_pert)
            
            alloc = min(remaining * 0.1, 12.0)
            a_opt = mirror_descent_subgrad(
                a_pert, max_iters=4000, time_budget=alloc,
                top_k_start=10, top_k_end=3,
                eta_start=0.015, eta_end=0.003, weight=0.85
            )
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart}: {obj:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    # Ensure sum is not too small
    if np.sum(best_seq) < 0.01:
        best_seq = best_seq * (1.0 / np.sum(best_seq))
    
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif np.isinf(val):
        # Fallback
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")
    
    return result
