"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP alternating for coarse optimization
- Smooth gradient descent (log-sum-exp) for fine optimization
- Spectral equalization (POCS) for diversification
- Multi-n exploration and coordinate descent polish

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def project_simplex(x):
    """Project x onto {y >= 0, sum(y) = 1}."""
    n = len(x)
    # Sort in descending order
    u = np.sort(x)[::-1]
    cssv = np.concatenate(([0.0], np.cumsum(u)))
    rho = np.arange(n, 0, -1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_bar = np.argmax(cond)
    theta = (cssv[rho_bar] - 1.0) / (rho_bar + 1.0)
    return np.maximum(x - theta, 0.0)

def next_power_of_2(n):
    return 1 << (n - 1).bit_length()

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_max_sum(a, M, time_limit=30):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()

        if time.monotonic() - start > time_budget - 5:

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp_max_sum(a, M, time_limit=max(5, int(remaining * 0.5)))
        if g is None:
            break
        
        # Try several step sizes

def smooth_gradient_descent(a, time_budget=200.0, tau_init=0.005, tau_min=0.0001):
    """
    Gradient descent on log-sum-exp approximation of max(conv(a,a)).
    Uses FFT for efficient gradient computation and simplex projection.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    tau = tau_init
    step = 0.05
    stall_count = 0
    
    # Precompute FFT size
    size = next_power_of_2(3 * n)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Compute soft weights
        tau_eff = max(tau, cmax * 1e-6)
        shifted = (conv - cmax) / tau_eff
        shifted = np.clip(shifted, -100.0, 100.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Compute gradient via cross-correlation
        # grad[i] = 2 * sum_j weights[i+j] * a[j]
        Wf = rfft(weights, size)
        Af = rfft(a, size)
        corr = irfft(Wf * np.conj(Af), size)
        grad = 2.0 * corr[:n]
        
        # Center gradient (to maintain sum ≈ constant before projection)
        grad_centered = grad - np.mean(grad)
        
        # Update
        a_new = a - step * grad_centered
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            step = min(step * 1.05, 0.3)
        else:
            step *= 0.8
            stall_count += 1
            if stall_count > 20:
                # Perturb and reset
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                a = normalize(best_a * np.exp(rng.normal(0, 0.02, n)))
                step = 0.05
                stall_count = 0
        
        # Decrease temperature
        tau = max(tau * 0.995, tau_min)
    
    return best_a

def spectral_equalization_v2(a, time_budget=60.0, alpha_init=0.3):
    """
    Improved spectral equalization: flatten the power spectrum while
    maintaining non-negativity.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = next_power_of_2(2 * n)
    alpha = alpha_init
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: make magnitudes more uniform (but DC is fixed)
        # Use geometric mean of non-DC magnitudes as target
        non_dc_mag = mag[1:]
        # Include negative frequencies too (for real signals, they're conjugate)
        target_mag = np.zeros(size)
        target_mag[0] = mag[0]  # DC fixed
        
        # For the rest, aim for a level between current and uniform
        if np.min(non_dc_mag) > 0:
            target_level = np.exp(np.mean(np.log(np.maximum(non_dc_mag, 1e-10))))
        else:
            target_level = np.mean(non_dc_mag)
        
        # Animate: move toward target by factor alpha
        for k in range(1, size):
            target_mag[k] = (1 - alpha) * mag[k] + alpha * target_level
        
        # Reconstruct
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        
        # Project onto non-negative simplex
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend to maintain progress
            a = normalize(0.3 * a + 0.7 * a_new)
        
        # Increase alpha over time (more aggressive flattening)
        alpha = min(alpha * 1.01, 0.7)
    
    return best_a

def coordinate_peel(a, time_budget=30.0):
    """
    Coordinate descent that peels mass from peak contributors.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find contributors to peak at k_star
        # conv[k_star] = sum_j a[j] * a[k_star - j]
        contribs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)
        
        improved = False
        for contrib_val, j1, j2 in contribs[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                # Try reducing a[j1] and spreading to random positions
                for _ in range(2):
                    j_t = int(rng.integers(0, n))
                    if j_t == j1:
                        continue
                    a_try = a.copy()
                    delta = 0.03 * a_try[j1]
                    a_try[j1] -= delta
                    a_try[j_t] += delta
                    a_try = project_simplex(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
            else:
                # Try reducing one of the pair
                for j_src in [j1, j2]:
                    for _ in range(2):
                        j_t = int(rng.integers(0, n))
                        if j_t == j1 or j_t == j2:
                            continue
                        a_try = a.copy()
                        delta = 0.03 * a_try[j_src]
                        a_try[j_src] -= delta
                        a_try[j_t] += delta
                        a_try = project_simplex(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                    if improved:
                        break
        
        if not improved:
            # Random perturbation
            a = normalize(best_a * np.exp(rng.normal(0, 0.015, n)))
            if compute_obj(a) > best_obj * 1.01:
                a = best_a.copy()
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(x):
    """Project x onto {y >= 0, sum(y) = 1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.concatenate(([0.0], np.cumsum(u)))
    rho = np.arange(n, 0, -1, dtype=np.float64)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.argmax(cond)
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(x - theta, 0.0)


def next_power_of_2(n):

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def compute_smooth_gradient(a, weights, size):
    """Compute gradient of weighted sum of convolution peaks using FFT cross-correlation.
    grad[i] = 2 * sum_j weights[i+j] * a[j]

    Wf = rfft(weights, size)
    Af = rfft(a, size)
    corr = irfft(Wf * np.conj(Af), size)
    return 2.0 * corr[:len(a)]

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)
