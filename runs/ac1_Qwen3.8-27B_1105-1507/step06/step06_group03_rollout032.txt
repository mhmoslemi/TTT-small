def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def cd_1d_search(a, i, conv, s, n, old_val):
    """
    1D line search for coordinate i.
    Returns (best_val, best_obj, best_conv_slice_change)
    """
    lo = max(0, i)
    hi = min(2 * n - 2, i + n - 1)
    kq = 2 * i
    valid_kq = kq <= 2 * n - 2
    a_slice = a[lo:hi + 1].copy()

    def eval_obj(x):
        delta = x - old_val
        if abs(delta) < 1e-15:
            return 2.0 * n * conv.max() / (s * s)
        # Save and update conv
        saved = conv[lo:hi + 1].copy()
        conv[lo:hi + 1] += 2.0 * delta * a_slice
        if valid_kq:
            conv[kq] += delta * delta
        new_s = s - old_val + x
        if new_s < 1e-10:
            obj = float('inf')
        else:
            obj = 2.0 * n * conv.max() / (new_s * new_s)
        # Restore
        conv[lo:hi + 1] = saved
        return obj

    # Candidate values
    if old_val < 1e-8:
        candidates = [0.0, 0.5/n, 1.0/n, 2.0/n, 5.0/n]
    else:
        candidates = [0.0, old_val * 0.1, old_val * 0.3, old_val * 0.5,
                      old_val * 0.7, old_val, old_val * 1.3, old_val * 1.7,
                      old_val * 2.5, old_val * 4.0]

    # Evaluate all candidates
    best_x = old_val
    best_obj = eval_obj(old_val)
    for x in candidates:
        x = max(0.0, x)
        obj = eval_obj(x)
        if obj < best_obj:
            best_obj = obj
            best_x = x

    # Refine with local search around best
    if best_x > 0 and best_x < old_val * 4:
        step = max(0.05 * old_val, 0.02 / n)
        for _ in range(3):
            x_left = max(0.0, best_x - step)
            x_right = best_x + step
            obj_left = eval_obj(x_left)
            obj_right = eval_obj(x_right)
            if obj_left < best_obj:
                best_x = x_left
                best_obj = obj_left
                step *= 0.5
            elif obj_right < best_obj:
                best_x = x_right
                best_obj = obj_right
                step *= 0.5
            else:
                break
    elif best_x == 0.0:
        # Check if we should increase from zero
        step = 0.05 / n
        for _ in range(3):
            x_try = best_x + step
            obj_try = eval_obj(x_try)
            if obj_try < best_obj:
                best_x = x_try
                best_obj = obj_try
                step *= 0.5
            else:
                break

    return best_x, best_obj


def coord_descent(a, max_sweeps, deadline, random_order=False, seed=0,
                  refine_every=5, verbose=False):
    """Fast coordinate descent with incremental convolution and line search."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 200

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved = False

        for i in order:
            old = a[i]
            best_x, best_local = cd_1d_search(a, i, conv, s, n, old)

            if best_x != old and abs(best_x - old) > 1e-14:
                delta = best_x - old
                lo = max(0, i)
                hi = min(2 * n - 2, i + n - 1)
                kq = 2 * i
                valid_kq = kq <= 2 * n - 2
                a_slice = a[lo:hi + 1]
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_x
                s = s - old + best_x
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        # Periodic perturbation to escape local minima
        if refine_every > 0 and sweep % refine_every == 0 and sweep > 0:
            if time.monotonic() > deadline:
                break
            # Small random perturbation on a random subset
            n_perturb = max(1, n // 20)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a[idx] *= (1.0 + rng.uniform(-0.1, 0.1, n_perturb))
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a /= s
                s = 1.0
            conv = fast_conv(a)

        if verbose and sweep % 100 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a, compute_J(a)


def sca_lp_step(a, max_iters, deadline, verbose=False):
    """
    SCA-LP: Successive Convex Approximation with LP subproblems.
    Linearizes the quadratic constraint at current point and solves LP.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    last_lp_time = 0.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # LP: minimize t
        # s.t. 2*(a*g)[k] - c[k] <= t  for k=0,...,2n-2
        #      sum(g) = 1, g >= 0
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]

        A_eq = coo_matrix((np.ones(n), (list(range(n)), list(range(n)))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        t0 = time.monotonic()
        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 30, 'presolve': True})
            last_lp_time = time.monotonic() - t0

            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    # Try multiple step sizes
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        ts = a_try.sum()
                        if ts < 1e-10:
                            continue
                        a_try /= ts
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-13:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 5 == 0:
            print(f"    SCA it {it}: obj={best_obj:.8f} lp_time={last_lp_time:.1f}s", flush=True)

        if no_improve > 30:
            break

    return best_a, best_obj

    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_count = 0
    no_improve = 0

    # Precompute FFT size
    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_init + (tau_max - tau_init) * frac

        # Compute smooth gradient
        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(np.minimum(tau * c_shifted, 500))  # clip for stability
        w = exp_c / exp_c.sum()

        # Gradient: grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) / 2 * 2
        # = sum_k w[k] * (a[(k-i) mod n_circ] + ...)
        # Use FFT: grad = ifft(fft(w) * fft(a_padded_shifted))
        # More precisely: grad[i] = sum_k w[k] * d/d a[i] conv[k]
        # d/d a[i] conv[k] = a[k-i] + a[i-k] (with zero padding)
        # = (a * w_rev)[i] + (a_rev * w)[i] where a_rev[j] = a[n-1-j]

        a_padded = np.zeros(2 * n - 1)
        a_padded[:n] = a
        w_padded = np.zeros(2 * n - 1)
        w_padded[:] = w

        # conv1[i] = sum_k w[k] * a_padded[k + i - (n-1)] ... 
        # Let me use direct FFT approach
        fa = np.fft.rfft(a, size)
        fw = np.fft.rfft(w, size)
        fw_rev = np.fft.rfft(w[::-1], size)

        # grad[i] = sum_k w[k] * (a[k-i] if valid + a[i-k] if valid)
        # = (w * a)[i] (circular, with proper indexing)
        # Using FFT: ifft(fft(w) * fft(a))[i] gives circular cross-correlation

        conv_wa = np.fft.irfft(fw * fa, size)  # circular cross-corr of w and a
        # Linear version: conv_wa[i] = sum_k w[k] * a[(i-k) mod size]
        # We want: sum_k w[k] * a[k-i] for valid k-i
        # = correlation of w and a at lag i

        # Actually let me compute directly using the relation:
        # grad[i] = sum_k w[k] * d(conv[k])/d(a[i])
        # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (0 if index out of range)
        # = (a indexed by k-i) + (a indexed by i-k)
        # First term: sum_k w[k] * a[k-i] = (w * a_shifted)[?] 
        # Let's just do it with two convolutions

        # term1[i] = sum_k w[k] * a[k-i] (valid when 0<=k-i<n)
        # = sum_j w[j+i] * a[j] (valid when 0<=j<n and 0<=j+i<2n-1)
        # This is the cross-correlation of w and a at lag i

        # term2[i] = sum_k w[k] * a[i-k] (valid when 0<=i-k<n)
        # = sum_j w[i-j] * a[j]
        # This is the cross-correlation of w and a at lag -i = correlation at lag i (by symmetry)

        # By symmetry of the problem (conv is symmetric), term1 = term2
        # So grad[i] = 2 * sum_j w[j+i] * a[j]

        # Using FFT: cross_correlation(w, a)[i] = ifft(fft(w) * conj(fft(a)))[-i mod N]
        # But since a is real: conj(fft(a)) = fft(a[::-1] padded)
        
        # Simpler: use np.correlate or direct FFT
        # cross_corr[i] = sum_j w[j+i] * a[j] for j=0..n-1
        # = (w_shifted * a) where w_shifted[i] = w[i+i] ... 
        
        # Let me just use the FFT cross-correlation properly
        # cross-correlation of w (length 2n-1) and a (length n):
        # cc[m] = sum_j w[j+m] * a[j]
        # We want cc[i] for i = 0, ..., n-1

        # Using FFT: cc = ifft(fft(w, N) * conj(fft(a, N)))
        # where N >= 2n-1+n-1 = 3n-2
        
        N = 1
        while N < 3 * n:
            N <<= 1
        fw_full = np.fft.rfft(w, N)
        fa_full = np.fft.rfft(a, N)
        cc = np.fft.irfft(fw_full * np.conj(fa_full), N)
        # cc[m] = sum_j conj(a[j]) * w[j+m] = sum_j a[j] * w[j+m] (a is real)
        # So cc[i] = sum_j a[j] * w[j+i] which is what we want for term1

        grad = 2.0 * cc[:n]

        # Project to simplex tangent space
        grad = grad - np.mean(grad)

        # Adam update
        t_count += 1
        lr = lr_init / (1.0 + 0.05 * t_count ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                # Restart with perturbation
                rng_local = np.random.default_rng(it)
                a = best_a + rng_local.normal(0, 0.03, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

        if verbose and it % 200 == 0:
            print(f"    Grad it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def active_set_lp(a, max_iters, deadline, n_top=5, verbose=False):
    """
    LP that only constrains the top n_top peaks of the convolution.
    This is much faster than the full LP and focuses on the binding constraints.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Find top peaks
        top_indices = np.argsort(c)[::-1][:n_top]
        top_indices = sorted(top_indices.tolist())

        # LP: minimize t
        # s.t. 2*(a*g)[k] - c[k] <= t for k in top_indices
        #      sum(g) = 1, g >= 0
        rows = []
        cols = []
        vals = []
        b_ub = []

        for idx, k in enumerate(top_indices):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(idx)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(n_top, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0, None)] * n + [(None, None)]

        A_eq = coo_matrix((np.ones(n), (list(range(n)), list(range(n)))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [1.0, 0.6, 0.4, 0.25, 0.15, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        ts = a_try.sum()
                        if ts < 1e-10:
                            continue
                        a_try /= ts
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-13:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 10 == 0:
            print(f"    AS-LP it {it}: obj={best_obj:.8f}", flush=True)

        if no_improve > 40:
            break

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences for optimization."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    inits.append(project_simplex(1.0 + 0.5 * np.cos(2 * np.pi * t)))
    inits.append(project_simplex(1.0 - 0.5 * np.cos(2 * np.pi * t)))

    # Two-level (binary-like)
    for freq in [1, 2, 3]:
        binary = (np.sin(2 * np.pi * freq * t) > 0).astype(float)
        inits.append(project_simplex(binary + 0.1))

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    # Tapered (Battard/Blackman-like window)
    win = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    inits.append(project_simplex(np.maximum(win, 0) + 0.1))

    # Power law
    inits.append(project_simplex((t + 0.01) ** 1.5))
    inits.append(project_simplex(((1 - t) + 0.01) ** 1.5))
    inits.append(project_simplex((1.0 + (t - 0.5) ** 2) ** 1.0))

    return inits


def solve_pipeline(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            if verbose:
                print(f"      Updated: J={obj:.8f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 1: Coordinate descent (35% of time)
    t1 = time.monotonic() + 0.35 * remaining
    if verbose:
        print(f"    Stage 1: CD (budget {t1 - time.monotonic():.0f}s)", flush=True)
    a = coord_descent(a, max_sweeps=50000, deadline=t1, random_order=True, seed=42)
    try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 2: Active-set LP (25% of time) - faster than full LP
    t2 = time.monotonic() + 0.25 * remaining
    if verbose:
        print(f"    Stage 2: AS-LP (budget {t2 - time.monotonic():.0f}s)", flush=True)
    a_as, obj_as = active_set_lp(best_a, max_iters=500, deadline=t2)
    try_update(a_as)

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 3: Gradient method (20% of time)
    t3 = time.monotonic() + 0.20 * remaining
    if verbose:
        print(f"    Stage 3: Grad (budget {t3 - time.monotonic():.0f}s)", flush=True)
    a_grad, obj_grad = gradient_step(best_a, max_iters=5000, deadline=t3,
                                      tau_init=30.0, tau_max=200.0, lr_init=0.04)
    try_update(a_grad)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 4: Full SCA-LP (10% of time)
    t4 = time.monotonic() + 0.10 * remaining
    if verbose:
        print(f"    Stage 4: Full SCA-LP (budget {t4 - time.monotonic():.0f}s)", flush=True)
    a_sca, obj_sca = sca_lp_step(best_a, max_iters=100, deadline=t4)
    try_update(a_sca)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 5: Final CD polish (10% of time)
    t5 = deadline - 5
    if verbose:
        print(f"    Stage 5: Final CD (budget {t5 - time.monotonic():.0f}s)", flush=True)
    a = coord_descent(best_a, max_sweeps=100000, deadline=t5,
                      random_order=True, seed=777)
    try_update(a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses diverse initializations, improved coordinate descent with line search,
    active-set LP, gradient methods, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Optimize at parent length with diverse inits ===
    phase1_deadline = t_start + min(400.0, 0.30 * (deadline - t_start))
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-init optimization at n={n_parent}", flush=True)
    print(f"  Budget: {phase1_deadline - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    # Prioritize parent-based inits (first 4) then diverse ones
    n_init = min(len(inits), 10)
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, n_init) - 3

    for idx in range(n_init):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 30:
            break
        alloc = min(per_init_time, remaining_check - 5)
        if alloc < 20:
            break

        a0 = inits[idx].copy()
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt, obj_opt = solve_pipeline(a0, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  Init {idx} (n={n_parent}): J={obj_opt:.8f} t={time.monotonic()-t_start:.0f}s", flush=True)

    # === PHASE 2: Try different lengths ===
    phase2_deadline = t_start + min(900.0, 0.60 * (deadline - t_start))
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"  Budget: {phase2_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Try lengths around parent
    candidate_ns = []
    for delta in [0, -96, 96, -192, 192, -64, 64, 128, -128]:
        nn = n_parent + delta
        if 128 <= nn <= 3000 and nn not in candidate_ns:
            candidate_ns.append(nn)
    # Also try some round numbers
    for nn in [256, 384, 512, 640, 768, 896, 1024, 1152, 1280, 1536]:
        if 128 <= nn <= 3000 and nn not in candidate_ns:
            candidate_ns.append(nn)

    # Sort by priority: prefer near parent
    candidate_ns.sort(key=lambda x: (abs(x - n_parent), x))

    tried_ns = {n_parent}
    for new_n in candidate_ns:
        if new_n in tried_ns:
            continue
        tried_ns.add(new_n)
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 60:
            break
        alloc = min(120.0, remaining_check - 20)
        if alloc < 40:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = solve_pipeline(a_ext, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f} t={time.monotonic()-t_start:.0f}s", flush=True)

    # === PHASE 3: Intensive optimization on best ===
    phase3_deadline = deadline
    remaining = phase3_deadline - time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive optimization (budget {remaining:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and remaining > 30:
        a_curr = best_seq.astype(np.float64).copy()
        a_curr = np.maximum(a_curr, 0)
        a_curr = a_curr / a_curr.sum()

        # Pass 1: Long CD
        t_pass1 = time.monotonic() + 0.25 * remaining
        print(f"  CD pass (budget {t_pass1 - time.monotonic():.0f}s)...", flush=True)
        a_curr, obj_cd = coord_descent(a_curr, max_sweeps=100000, deadline=t_pass1,
                                       random_order=True, seed=1234)
        update_best(a_curr)

        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            # Pass 2: Active-set LP
            t_pass2 = time.monotonic() + 0.25 * remaining
            print(f"  AS-LP pass (budget {t_pass2 - time.monotonic():.0f}s)...", flush=True)
            a_as, obj_as = active_set_lp(best_seq, max_iters=1000, deadline=t_pass2)
            update_best(a_as)

        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            # Pass 3: Gradient
            t_pass3 = time.monotonic() + 0.20 * remaining
            print(f"  Grad pass (budget {t_pass3 - time.monotonic():.0f}s)...", flush=True)
            a_grad, obj_grad = gradient_step(best_seq, max_iters=10000,
                                             deadline=t_pass3, tau_init=20.0,
                                             tau_max=150.0, lr_init=0.03)
            update_best(a_grad)

        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            # Pass 4: Full SCA-LP
            t_pass4 = time.monotonic() + 0.15 * remaining
            print(f"  Full SCA-LP pass (budget {t_pass4 - time.monotonic():.0f}s)...", flush=True)
            a_sca, obj_sca = sca_lp_step(best_seq, max_iters=200, deadline=t_pass4)
            update_best(a_sca)

        remaining = phase3_deadline - time.monotonic()
        if remaining > 20:
            # Final CD on best
            t_final = phase3_deadline - 5
            print(f"  Final CD (budget {t_final - time.monotonic():.0f}s)...", flush=True)
            a_final, obj_final = coord_descent(best_seq, max_sweeps=200000,
                                               deadline=t_final, random_order=True,
                                               seed=4567)
            update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
