def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ConvState:
    """Maintains convolution state with incremental updates."""
    
    def __init__(self, a):
        self.a = np.maximum(a.astype(np.float64), 0.0)
        self.n = len(self.a)
        self.conv = fast_conv(self.a)
        self.s = self.a.sum()
        self._recompute_max()
    
    def _recompute_max(self):
        self.max_val = self.conv.max()
        self.max_idx = np.argmax(self.conv)
    
    def update_coord(self, i, new_val):
        """Update coordinate i to new_val, returning the change in max."""
        old = self.a[i]
        new_val = max(0.0, new_val)
        delta = new_val - old
        if abs(delta) < 1e-15:
            return
        
        n = self.n
        # Update convolution
        # conv[k] changes for k in [max(0, 2*i-(n-1)), min(2*n-2, 2*i+(n-1))]
        # Actually, conv[k] = sum_j a[j]*a[k-j], terms involving a[i]:
        # j=i: a[i]*a[k-i] valid when 0<=k-i<n, i.e., i <= k <= i+n-1
        # j=k-i: a[k-i]*a[i] valid when 0<=k-i<n, i.e., i <= k <= i+n-1 (same range)
        # So k ranges from i to i+n-1 (but also need k >= 0 and k <= 2n-2)
        
        k_lo = max(0, i)
        k_hi = min(2 * n - 2, i + n - 1)
        
        for k in range(k_lo, k_hi + 1):
            if k == 2 * i:
                self.conv[k] += 2 * old * delta + delta * delta
            else:
                self.conv[k] += 2 * delta * self.a[k - i]
        
        self.a[i] = new_val
        self.s += delta
        self._recompute_max()
    
    @property
    def J(self):
        if self.s < 1e-10:
            return float('inf')
        return 2.0 * self.n * self.max_val / (self.s * self.s)

def frank_wolfe(a, max_iters, deadline, verbose=False):
    """Frank-Wolfe (conditional gradient) method with exact active set."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = fast_conv(a)
        M = conv.max()
        
        # Find active set (indices where conv is close to max)
        threshold = M * (1 - 1e-6)
        active = np.where(conv >= threshold)[0]
        
        # Compute gradient of average active quadratic form
        # g[i] = mean_{k in active} (a[k-i] + a[i-k]) with bounds checking
        g = np.zeros(n)
        for k in active:
            for i in range(n):
                val = 0.0
                if 0 <= k - i < n:
                    val += a[k - i]
                if 0 <= i - k < n:
                    val += a[i - k]
                g[i] += val
        g /= len(active)
        
        # Frank-Wolfe: linear minimization oracle for simplex
        # min g[i] over i (since we minimize g^T a over simplex)
        i_min = np.argmin(g)
        v = np.zeros(n)
        v[i_min] = 1.0
        
        # Step size rule
        # FW step size: gamma = 2/(it+2)
        gamma = 2.0 / (it + 2)
        
        a_new = (1 - gamma) * a + gamma * v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 50 == 0:
            print(f"  FW it {it}: J={best_J:.8f}, |active|={len(active)}", flush=True)
    
    return best_a, best_J

"""
Hybrid optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Primary method: SCA-LP (Successive Convex Approximation with LP direction finding).
Secondary: Fine-grid coordinate descent with incremental convolution updates.
Tertiary: Multi-length exploration and evolutionary restarts.
Key improvements over previous: (1) more aggressive LP iterations with adaptive step size,
(2) active-set-aware coordinate descent focusing on coordinates affecting max entries,
(3) dual-variable tracking for smarter search, (4) balanced time allocation between methods.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def compute_gradient_active(a, active_indices):
    """Compute subgradient of max active conv entries using FFT."""
    n = len(a)
    conv_len = 2 * n - 1
    w = np.zeros(conv_len)
    for k in active_indices:
        w[k] = 1.0
    w /= len(active_indices)
    
    # Gradient: g[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # = cross_correlation(w, a) computed via FFT
    size = 1
    while size < conv_len + n - 1:
        size <<= 1
    
    a_fft = np.fft.rfft(a, size)
    w_fft = np.fft.rfft(w, size)
    
    # First term: sum_k w[k]*a[k-i] for valid k-i in [0,n-1]
    # This is (w * a_rev)[n-1+i] where a_rev = a[::-1]
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, size)
    
    conv1 = np.fft.irfft(w_fft * a_rev_fft, size)
    # conv1[j] = sum_k w[k] * a_rev[j-k] = sum_k w[k] * a[n-1-(j-k)]
    # We want: for each i, sum_k w[k]*a[k-i]
    # a[k-i] = a_rev[n-1-(k-i)] = a_rev[n-1-k+i]
    # sum_k w[k]*a_rev[n-1-k+i] = (w * a_rev)[n-1+i] ... 
    # Actually conv1[m] = sum_k w[k]*a_rev[m-k]
    # We want sum_k w[k]*a[k-i] = sum_k w[k]*a_rev[n-1-k+i]
    # Let m = n-1+i-k, so k = n-1+i-m: sum_m a_rev[m]*w[n-1+i-m]
    # = (a_rev * w)[n-1+i] ... this is getting confusing.
    # Let me just do it directly for small active sets.
    
    if len(active_indices) <= 20:
        g = np.zeros(n)
        for k in active_indices:
            # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (with bounds)
            for i in range(n):
                val = 0.0
                if 0 <= k - i < n:
                    val += a[k - i]
                if 0 <= i - k < n:
                    val += a[i - k]
                g[i] += val
        g /= len(active_indices)
        return g
    else:
        # Use FFT for large active sets
        g = np.zeros(n)
        for k in active_indices:
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            for i in range(lo1, hi1 + 1):
                g[i] += a[k - i]
            for i in range(lo2, hi2 + 1):
                g[i] += a[i - k]
        g /= len(active_indices)
        return g


def coord_descent_fine(a, max_sweeps, deadline, focus_active=False, verbose=False):
    """
    Coordinate descent with fine grid and optional focus on active-set-relevant coords.
    Uses incremental convolution updates for speed.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0
    
    conv = fast_conv(a).copy()
    best_a = a.copy()
    best_obj = 2.0 * n * conv.max() / (s * s)
    no_improve = 0
    
    rng = np.random.default_rng(42)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Determine which coordinates to focus on
        if focus_active:
            M = conv.max()
            threshold = M * 0.95
            active_k = np.where(conv >= threshold)[0]
            # Coordinates that affect active entries
            relevant = set()
            for k in active_k:
                for i in range(max(0, k - n + 1), min(n, k + 1)):
                    relevant.add(i)
            relevant = list(relevant)
            if len(relevant) < n // 2:
                order = rng.permutation(n)
                # Put relevant first
                rel_in_order = [i for i in order if i in relevant]
                other_in_order = [i for i in order if i not in relevant]
                order = rel_in_order + other_in_order
            else:
                order = rng.permutation(n)
        else:
            order = rng.permutation(n)
        
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            old = a[i]
            
            # Determine candidate values
            if old < 1e-12:
                candidates = np.array([0.0, 1e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2])
            else:
                # Fine grid around current value
                ratios = np.array([0.0, 0.1, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 1.7, 2.0])
                candidates = ratios * old
            
            # Compute which conv entries are affected
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            # The slice of other a values involved
            other_vals = np.zeros(k_hi - k_lo + 1)
            for idx, k in enumerate(range(k_lo, k_hi + 1)):
                j = k - i
                if 0 <= j < n and j != i:
                    other_vals[idx] = a[j]
                elif j == i:
                    other_vals[idx] = old  # self term handled separately
            
            best_x = old
            best_local_obj = 2.0 * n * conv.max() / (s * s)
            
            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                
                # Update conv incrementally
                for idx, k in enumerate(range(k_lo, k_hi + 1)):
                    if k == 2 * i:
                        conv[k] += 2.0 * old * delta + delta * delta
                    else:
                        conv[k] += 2.0 * delta * other_vals[idx]
                
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                
                if new_obj < best_local_obj:
                    best_local_obj = new_obj
                    best_x = x
                
                # Revert
                for idx, k in enumerate(range(k_lo, k_hi + 1)):
                    if k == 2 * i:
                        conv[k] -= 2.0 * old * delta + delta * delta
                    else:
                        conv[k] -= 2.0 * delta * other_vals[idx]
            
            if best_x != old and abs(best_x - old) > 1e-14:
                delta = best_x - old
                for idx, k in enumerate(range(k_lo, k_hi + 1)):
                    if k == 2 * i:
                        conv[k] += 2.0 * old * delta + delta * delta
                    else:
                        conv[k] += 2.0 * delta * other_vals[idx]
                a[i] = best_x
                s = s - old + best_x
                improved = True

        if current_obj < best_obj - 1e-13:

        else:
            no_improve += 1

            break
        
        if verbose and sweep % 200 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}", flush=True)

def sca_lp_extended(a, max_iters, deadline, verbose=False):

    Solves LP to find best response direction, uses backtracking line search.
    """
    n = len(a)

    a = a / a.sum()

            break

        # Build and solve LP
        rows = []
        cols = []
        vals = []
        
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        
        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 20, 'presolve': True})
            if r.success and r.x is not None:

                gs = g.sum()

                            continue
                        a_try = a_try / s_try

                            best_obj = J_try
                            best_a = a.try = a.copy()

                    if not found:

                                best_a = a.copy()
                                no_improve = 0
                                found = True
                        if not found:
                            no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1
        
        if no_improve > 40:
            break
        
        if verbose and it % 10 == 0:
            print(f"    SCA it {it}: obj={best_obj:.8f}, |active|={len(active)}", flush=True)

def block_cd(a, block_size, max_sweeps, deadline, verbose=False):
    """Block coordinate descent: optimize small blocks simultaneously."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()

        a = np.ones(n) / n

    best_a = a.copy()

    no_improve = 0
    rng = np.random.default_rng(99)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Random blocks
        perm = rng.permutation(n)
        improved = False
        
        for start in range(0, n, block_size):
            if time.monotonic() > deadline:
                break
            block = perm[start:start + block_size]
            if len(block) < 2:
                continue
            
            # For each block, try small perturbations
            best_block_vals = a[block].copy()
            best_block_obj = 2.0 * n * conv.max() / (s * s)
            
            # Try random perturbations of the block
            for trial in range(5):
                perturbation = rng.normal(0, 0.1 * np.mean(a[block]) + 0.01, len(block))
                new_block = np.maximum(a[block] + perturbation, 0)
                delta_sum = new_block.sum() - a[block].sum()
                
                # Update conv for the block
                old_block = a[block].copy()
                deltas = new_block - old_block
                
                for idx, i in enumerate(block):
                    delta = deltas[idx]
                    if abs(delta) < 1e-15:
                        continue
                    k_lo = max(0, i)
                    k_hi = min(2 * n - 2, i + n - 1)
                    for k in range(k_lo, k_hi + 1):
                        if k == 2 * i:
                            conv[k] += 2.0 * old_block[idx] * delta + delta * delta
                        else:
                            j = k - i
                            if 0 <= j < n:
                                conv[k] += 2.0 * delta * a[j]
                
                new_s = s + delta_sum
                if new_s > 1e-10:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                    if new_obj < best_block_obj:
                        best_block_obj = new_obj
                        best_block_vals = new_block.copy()
                
                # Revert
                for idx, i in enumerate(block):
                    delta = deltas[idx]
                    if abs(delta) < 1e-15:
                        continue
                    k_lo = max(0, i)
                    k_hi = min(2 * n - 2, i + n - 1)

                            j = k - i
                            if 0 <= j < n:

                # Revert a for next trial
                a[block] = old_block
            
            if best_block_obj < best_obj - 1e-13:
                # Apply the best block update
                old_block = a[block].copy()
                deltas = best_block_vals - old_block
                for idx, i in enumerate(block):

def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences for optimization."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various structured inits
    for freq in [1, 2, 3, 5, 8]:
        inits.append(project_simplex(1.0 + 0.5 * np.cos(2 * np.pi * freq * t)))
    
    # Triangular
    inits.append(project_simplex(np.abs(1 - 2 * t) + 0.1))
    
    # Gaussian
    for center in [0.3, 0.5, 0.7]:
        inits.append(project_simplex(np.exp(-((t - center) / 0.2) ** 2) + 0.05))
    
    # Power law
    for alpha in [0.5, 1.5, 2.5]:
        inits.append(project_simplex((t + 0.05) ** alpha))
    
    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))
    
    return inits


def solve_sequence(a, deadline, verbose=False):

    best_a = a.copy()

    def try_update(a_new, label=""):

            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:

            best_obj = obj

    remaining = deadline - time.monotonic()

    # Phase 1: SCA-LP (40% of time)
    t1 = time.monotonic() + 0.40 * remaining
    a, obj = sca_lp_extended(a, max_iters=500, deadline=t1, verbose=verbose)

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj
    
    # Phase 3: Another SCA-LP on the CD result (20% of time)

    # Phase 4: Final CD polish (15% of time)

                                         focus_active=True, verbose=verbose)

    # === PHASE 2: Diverse inits at parent length (250s) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Diverse initializations at n={n_parent}", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    p2_deadline = t_start + 600.0
    inits = diverse_inits(n_parent, rng, parent)
    # Skip the parent init (already done in phase 1)
    inits = inits[1:]  # Remove the first one (parent itself)
    
    per_init_time = (p2_deadline - time.monotonic()) / max(1, len(inits)) - 10
    per_init_time = max(30.0, min(per_init_time, 60.0))
    
    for idx, a0 in enumerate(inits):
        remaining_check = p2_deadline - time.monotonic()
        if remaining_check < 40:
            break
        alloc = min(per_init_time, remaining_check - 10)
        if alloc < 25:
            break
        
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()
        
        a_opt, obj_opt = solve_sequence(a0, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        if (idx + 1) % 3 == 0:
            print(f"  Init {idx+1}/{len(inits)}: best so far J={best_score:.8f}", flush=True)
    
    # === PHASE 3: Multi-length exploration (250s) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    p3_deadline = t_start + 850.0
    candidate_ns = [n for n in [512, 640, 768, 896, 1024, 1152] if n != n_parent and n <= 1200]
    candidate_ns.sort(key=lambda x: abs(x - n_parent))
    
    for new_n in candidate_ns[:4]:
        remaining_check = p3_deadline - time.monotonic()
        if remaining_check < 60:
            break
        alloc = min(80.0, remaining_check - 20)
        if alloc < 40:
            break
        
        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()
        
        a_opt, obj_opt = solve_sequence(a_ext, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"    n={new_n}: J={best_score:.8f}", flush=True)
    
    # === PHASE 4: Final intensive optimization on best (remaining time) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Final intensive optimization (budget {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 100:
            # Long SCA-LP
            t_step1 = time.monotonic() + 0.4 * remaining
            a_sca, obj_sca = sca_lp_extended(best_seq, max_iters=1000, deadline=t_step1, verbose=True)
            update_best(a_sca)
            
            remaining = deadline - time.monotonic()
            if remaining > 50:
                # Long CD
                t_step2 = time.monotonic() + 0.3 * remaining
                a_cd, obj_cd = coord_descent_fine(best_seq, max_sweeps=10000, deadline=t_step2,
                                                   focus_active=True, verbose=True)
                update_best(a_cd)
            
            remaining = deadline - time.monotonic()
            if remaining > 30:
                # Final SCA
                a_sca2, obj_sca2 = sca_lp_extended(best_seq, max_iters=200, deadline=deadline - 10)
                update_best(a_sca2)
            
            remaining = deadline - time.monotonic()
            if remaining > 20:
                # Final CD
                a_cd2, obj_cd2 = coord_descent_fine(best_seq, max_sweeps=5000,
                                                     deadline=deadline - 5, focus_active=True)
                update_best(a_cd2)
        
        # Try a few more perturbations of the best
        remaining = deadline - time.monotonic()
        if remaining > 60:
            for sigma in [0.01, 0.03, 0.05]:
                if remaining < 30:
                    break
                a_pert = best_seq + rng.normal(0, sigma, len(best_seq))
                a_pert = project_simplex(a_pert)
                alloc = min(25.0, remaining - 10)
                if alloc < 15:
                    break
                a_opt, obj_opt = solve_sequence(a_pert, time.monotonic() + alloc)
                update_best(a_opt)
                remaining = deadline - time.monotonic()
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)
    
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    return [float(x) for x in best_seq]
