For the specific objective 2n * max_conv / (sum)^2, with sum=1:
- We want to minimize max_k (a*a)[k]
- (a*a)[k] is the autocorrelation at lag k
- We want the autocorrelation to be as flat as possible (low peaks)

The "flat" autocorrelation problem with non-negative sequences is related to:
- Ideal impulse response sequences
- Low-autocorrelation binary sequences (for the binary case)
- But here we have continuous non-negative values

One approach that might work well: start from a sequence that has good autocorrelation properties (like a Zadoff-Chu sequence made real and non-negative) and then optimize.

Zadoff-Chu sequences: z[k] = exp(-j * pi * k * (k+d) / N) for N odd. These have perfectly flat autocorrelation. The real part would be cos(pi * k * (k+d) / N), which is non-negative only for some k.

Alternatively, we could use the absolute value: |z[k]| = 1, which is just a constant. Not helpful.

What about using |cos(pi * k * (k+d) / N)|? This would be non-negative and have some structure.

Actually, let me think about this differently. The optimal solution for the problem "minimize max autocorrelation subject to a >= 0, sum(a) = 1" is known in the signal processing literature. It's related to the "minimum peak-to-average power ratio" (PAPR) problem for the autocorrelation.

For the continuous case (a can take any non-negative values), I believe the optimal solution has a specific structure related to the roots of certain polynomials or to the eigenfunctions of a certain integral operator.

But for practical purposes, iterative methods (like the one I described) should work well.

"""
Optimization using multiplicative weight (mirror descent) method with LP refinement
for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import next_fast_len
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = next_fast_len(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_gradient(p, a, N_fft):
    """
    Compute gradient: grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i.
    p: weights over convolution indices (length 2n-1)
    a: sequence (length n)
    N_fft: FFT size (>= 3n)
    """
    n = len(a)
    fp = np.fft.fft(p, N_fft)
    fa = np.fft.fft(a, N_fft)
    # Cross-correlation: corr[m] = sum_k p[k] * a[k-m]
    corr = np.fft.ifft(fp * np.conj(fa)).real
    grad = 2.0 * corr[:n]
    return grad


def multiplicative_update(a, grad, eta):
    """Multiplicative weight update: a_new[i] ∝ a[i] * exp(-eta * grad[i])."""
    n = len(a)
    # For numerical stability, work in log space
    log_ratio = -eta * grad
    log_ratio = log_ratio - np.max(log_ratio)  # shift for stability
    weights = np.exp(np.maximum(log_ratio, -700))
    a_new = a * weights
    s = np.sum(a_new)
    if s < 1e-30:
        return a.copy()
    return a_new / s


def softmax_weights(conv, tau):
    """Compute softmax weights from convolution values."""
    cmax = np.max(conv)
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= s
    return w


def mirror_descent(a, max_iters=5000, tau_init=None, tau_min=None, lr=1.0, time_budget=60.0):
    """
    Mirror descent (multiplicative weights) for minimizing max autocorrelation.
    
    Parameters:
    - a: initial sequence (will be normalized to sum=1)
    - max_iters: maximum number of iterations
    - tau_init: initial temperature for softmax (default: max_conv)
    - tau_min: minimum temperature (default: 0.05 * max_conv_initial)
    - lr: learning rate (step size factor, eta = lr * n)
    - time_budget: time limit in seconds
    """
    n = len(a)
    a = normalize_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    
    conv = compute_conv(a)
    cmax_init = float(np.max(conv))
    best_obj = 2.0 * n * cmax_init  # since sum=1
    
    if tau_init is None:
        tau_init = cmax_init
    if tau_min is None:
        tau_min = 0.01 * cmax_init
    
    N_fft = next_fast_len(3 * n)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature
        tau = tau_init * (tau_min / tau_init) ** (it / max_iters)
        tau = max(tau, 1e-15)
        
        # Compute weights
        p = softmax_weights(conv, tau)
        
        # Compute gradient
        grad = compute_gradient(p, a, N_fft)
        
        # Step size: eta = lr * n (since grad is O(1/n))
        eta = lr * n
        # Adaptive: reduce step if gradient is large
        grad_scale = np.max(np.abs(grad))
        if grad_scale > 1e-10:
            eta = min(eta, 1.0 / grad_scale)
        
        # Multiplicative update
        a_new = multiplicative_update(a, grad, eta)
        
        # Evaluate
        conv = compute_conv(a_new)
        cmax = float(np.max(conv))
        obj = 2.0 * n * cmax
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj / (2.0 * n)  # Return max_conv (since sum=1)


def solve_lp_best_response(a, M, time_limit=30):
    """
    Solve LP: max sum(g) subject to (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    if n > 3000:
        return None
    
    # Build constraint matrix: (a*g)[k] = sum_i a[i] * g[k-i] <= M
    # For each k (0 to 2n-2), the constraint is sum_i a[i] * g[k-i] <= M
    # where g[j] = 0 for j < 0 or j >= n
    # So for constraint k: sum_{i: 0<=k-i<n} a[i] * g[k-i] <= M
    # Let j = k-i: sum_{j: 0<=j<n, 0<=k-j<n} a[k-j] * g[j] <= M
    
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            A[k, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate_step(a, max_iters=20, time_budget=30.0):
    """One round of LP alternating."""
    n = len(a)
    a = normalize_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        M = float(np.max(compute_conv(a)))
        
        g = solve_lp_best_response(a, M, time_limit=min(10, (time_budget - (time.monotonic() - start)) * 0.5))
        if g is None:
            break
        
        # Try different blend factors
        best_t_obj = best_obj
        best_t_val = 0.5
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = normalize_sum1(a_try)
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
        
        if best_t_obj < best_obj - 1e-15:
            a = ((1.0 - best_t_val) * a + best_t_val * g)
            a = normalize_sum1(a)
            best_obj = best_t_obj
            best_a = a.copy()
        else:
            break
    
    return best_a, best_obj / (2.0 * n)


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0/new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize_sum1(a_new)


def freq_domain_flat(a, max_iters=200, time_budget=20.0):
    """
    Frequency domain approach: try to flatten the power spectrum.
    """
    n = len(a)
    a = normalize_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    
    N = next_fast_len(2 * n - 1)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        pmax = np.max(power)
        if pmax < 1e-30:
            break
        
        # Compress the power spectrum
        p_norm = power / pmax
        p_norm = np.maximum(p_norm, 1e-10)
        alpha = 0.5 + 0.3 * (it / max_iters)  # Start with more compression
        p_compressed = p_norm ** alpha
        
        # Reconstruct with compressed magnitude
        phase = np.angle(fa)
        mag = np.sqrt(p_compressed) * np.sqrt(pmax)
        fa_new = mag * np.exp(1j * phase)
        
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = normalize_sum1(a_new)
        
        conv = compute_conv(a_new)
        obj = 2.0 * n * float(np.max(conv))
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a, best_obj / (2.0 * n)

def generate_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangular
    x = np.linspace(0, 1, n)
    tri = np.sin(np.pi * x)
    inits.append(tri / tri.sum())
    
    # Gaussian
    x = np.linspace(-3, 3, n)
    gauss = np.exp(-x**2 / 2)
    inits.append(gauss / gauss.sum())
    
    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())
    
    # Cosine-squared
    x = np.linspace(0, 1, n)
    cos_sq = np.cos(np.pi * (x - 0.5)) ** 2
    inits.append(cos_sq / cos_sq.sum())
    
    # Bimodal
    x = np.linspace(0, 1, n)
    bimodal = np.exp(-((x-0.3)/0.1)**2) + np.exp(-((x-0.7)/0.1)**2)
    inits.append(bimodal / bimodal.sum())
    
    # Ramp
    ramp = np.arange(1, n+1, dtype=float)
    inits.append(ramp / ramp.sum())
    
    return inits

"""
Mirror descent (multiplicative weights) optimization for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines O(n log n) mirror descent iterations with LP best responses,
frequency-domain flattening, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import next_fast_len
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = next_fast_len(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_weights(conv, tau):
    """Compute softmax weights from convolution values with temperature tau."""
    cmax = float(np.max(conv))
    if cmax < 1e-30:
        return np.ones(len(conv)) / len(conv)
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500.0, 500.0)
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= s
    return w


def compute_gradient_fft(p, a, N_fft):
    """
    Compute gradient: grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i.
    Uses FFT cross-correlation for O(n log n) efficiency.
    """
    n = len(a)
    fp = np.fft.fft(p, N_fft)
    fa = np.fft.fft(a, N_fft)
    corr = np.fft.ifft(fp * np.conj(fa)).real
    grad = 2.0 * corr[:n]
    return grad


def multiplicative_update(a, grad, eta):
    """Multiplicative weight update: a_new[i] proportional to a[i] * exp(-eta * grad[i])."""
    n = len(a)
    log_ratio = -eta * grad
    log_ratio = log_ratio - float(np.max(log_ratio))
    log_ratio = np.maximum(log_ratio, -700.0)
    weights = np.exp(log_ratio)
    a_new = a * weights
    s = float(np.sum(a_new))
    if s < 1e-30:
        return a.copy()
    return a_new / s


def run_mirror_descent(a, n_iters, lr, tau_init, tau_min, N_fft, time_budget):
    """
    Run mirror descent iterations.
    Returns (best_a, best_maxconv).
    """
    n = len(a)
    a = normalize_sum1(a.copy().astype(np.float64))
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    best_a = a.copy()
    best_cmax = cmax
    start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, n_iters)
        tau = tau_init * (tau_min / max(tau_init, 1e-30)) ** frac
        tau = max(tau, 1e-18)

        p = softmax_weights(conv, tau)
        grad = compute_gradient_fft(p, a, N_fft)

        eta = lr * n
        gs = float(np.max(np.abs(grad)))
        if gs > 1e-12:
            eta = min(eta, 2.0 / gs)
        eta = max(eta, 1e-6)

        a_new = multiplicative_update(a, grad, eta)
        conv = compute_conv(a_new)
        cmax_new = float(np.max(conv))

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_cmax


def solve_lp_best_response(a, M, time_limit=20):
    """
    Solve LP: max sum(g) subject to (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    if n > 2500:
        return None

    # Build Toeplitz constraint matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            # A[k, j] = a[k-j] for valid j
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]

    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

    n = len(a)

    conv = compute_conv(a)
    best_a = a.copy()
    best_cmax = float(np.max(conv))

    stall = 0

            break

        M = float(np.max(compute_conv(a)))
        g = solve_lp_best_response(a, M, time_limit=min(15, (time_budget - (time.monotonic() - start)) * 0.4))
        if g is None:
            break

        best_t_cmax = best_cmax
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = normalize_sum1(a_try)
            cmax_try = float(np.max(compute_conv(a_try)))
            if cmax_try < best_t_cmax:
                best_t_cmax = cmax_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_cmax < best_cmax - 1e-15:
            best_cmax = best_t_cmax
            best_a = best_a_local.copy()
            a = best_a_local
            stall = 0
        else:
            stall += 1
            if stall > 5:
                break

    return best_a, best_cmax


def freq_domain_flat(a, max_iters=200, time_budget=20.0):
    """Frequency domain flattening. Returns (best_a, best_maxconv)."""
    n = len(a)
    a = normalize_sum1(a.copy().astype(np.float64))
    conv = compute_conv(a)
    best_a = a.copy()
    best_cmax = float(np.max(conv))
    start = time.monotonic()
    N = next_fast_len(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        pmax = float(np.max(power))
        if pmax < 1e-30:
            break

        p_norm = power / pmax
        p_norm = np.maximum(p_norm, 1e-12)
        alpha = 0.4 + 0.4 * (it / max(max_iters, 1))
        p_compressed = p_norm ** alpha

        phase = np.angle(fa)
        mag = np.sqrt(p_compressed) * np.sqrt(pmax)
        fa_new = mag * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = normalize_sum1(a_new)

        cmax_new = float(np.max(compute_conv(a_new)))
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

    return best_a, best_cmax


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize_sum1(a_new)

        tau_min = 0.005 * cmax_cur

        a_new, cmax_new = run_mirror_descent(
            a_cur, n_iters=n_iters, lr=lr,
            tau_init=tau_init, tau_min=tau_min,
            N_fft=N_fft, time_budget=alloc
        )
        obj_new = 2.0 * n * cmax_new

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            local_best_a = a_new.copy()
            local_best_cmax = cmax_new
            print(f"  MD cfg{ci} (lr={lr},tau_f={tau_f}): obj={best_score:.8f}")
        elif cmax_new < local_best_cmax:
            local_best_a = a_new.copy()
            local_best_cmax = cmax_new

        # Keep the best local for next round
        if cmax_new < float(np.max(compute_conv(local_best_a))):
            local_best_a = a_new.copy()
            local_best_cmax = cmax_new

    a = normalize_sum1(local_best_a.copy())

    # === Phase 3: LP refinement ===
    print(f"\n=== Phase 3: LP Refinement ===")
    print(f"Best so far: {best_score:.8f}")
    phase3_budget = min(300.0, (deadline - time.monotonic()) * 0.25)
    phase3_start = time.monotonic()

    if n <= 2000:
        lp_a, lp_cmax = lp_alternate(a, max_iters=30, time_budget=phase3_budget * 0.5)
        lp_obj = 2.0 * n * lp_cmax
        if lp_obj < best_score:
            best_score = lp_obj
            best_seq = lp_a.copy()
            a = lp_a.copy()
            print(f"  LP improved: obj={best_score:.8f}")
        elif lp_cmax < local_best_cmax:
            local_best_a = lp_a.copy()
            local_best_cmax = lp_cmax
            a = lp_a.copy()

    # Perturbed + LP
    remaining3 = phase3_budget - (time.monotonic() - phase3_start)
    if remaining3 > 15 and n <= 1500:
        for pi in range(min(4, int(remaining3 / 12))):
            if time.monotonic() > phase3_start + phase3_budget:
                break
            a_pert = a * np.exp(rng.normal(0, 0.02 + 0.02 * pi, n))
            a_pert = normalize_sum1(a_pert)

            # Quick MD warmup
            conv_p = compute_conv(a_pert)
            tc = float(np.max(conv_p))
            a_pert, _ = run_mirror_descent(a_pert, 500, 1.0, tc, 0.01 * tc,
                                           N_fft, time_budget=min(5, remaining3 * 0.2))

            # LP
            ap, mp = lp_alternate(a_pert, max_iters=8, time_budget=min(8, remaining3 * 0.3))
            op = 2.0 * n * mp
            if op < best_score:
                best_score = op
                best_seq = ap.copy()
                a = ap.copy()
                print(f"  LP pert {pi}: obj={best_score:.8f}")

    # === Phase 4: Frequency domain + length adjustment ===
    print(f"\n=== Phase 4: Freq Domain + Length Search ===")
    print(f"Best so far: {best_score:.8f}")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 30:
        phase4_budget = min(200.0, remaining4 * 0.35)
        phase4_start = time.monotonic()

        # Freq domain on current best
        a_fd, fd_cmax = freq_domain_flat(a, max_iters=300, time_budget=min(30, phase4_budget * 0.25))
        fd_obj = 2.0 * n * fd_cmax
        if fd_obj < best_score:
            best_score = fd_obj
            best_seq = a_fd.copy()
            a = a_fd.copy()
            print(f"  Freq domain: obj={best_score:.8f}")

        # Try nearby lengths
        n_cur = len(a)
        n_tries = sorted(set([n_cur - 30, n_cur - 60, n_cur + 30, n_cur + 60,
                               n_cur + 100, n_cur - 100]))
        n_tries = [nt for nt in n_tries if 50 <= nt <= 3000]
        n_tries = list(dict.fromkeys(n_tries))

        per_n = max(5.0, (phase4_budget - (time.monotonic() - phase4_start)) / max(1, len(n_tries)))

        for nt in n_tries:
            if time.monotonic() > phase4_start + phase4_budget:
                break
            a_rs = resample_seq(a, nt)
            nt_n = len(a_rs)
            N_fft_rs = next_fast_len(3 * nt_n)
            conv_rs = compute_conv(a_rs)
            tc_rs = float(np.max(conv_rs))

            a_rs, cmax_rs = run_mirror_descent(
                a_rs, n_iters=2000, lr=1.0,
                tau_init=tc_rs, tau_min=0.01 * tc_rs,
                N_fft=N_fft_rs, time_budget=per_n
            )
            obj_rs = 2.0 * nt_n * cmax_rs
            if obj_rs < best_score:
                best_score = obj_rs
                best_seq = a_rs.copy()
                print(f"  n={nt}: NEW BEST obj={best_score:.8f}")
            else:
                print(f"  n={nt}: obj={obj_rs:.8f}")

    # === Phase 5: Final intensive polish with low temperature ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()

    if remaining5 > 15 and best_seq is not None:
        a_final = normalize_sum1(best_seq.copy())
        n_f = len(a_final)
        N_fft_f = next_fast_len(3 * n_f)
        conv_f = compute_conv(a_final)
        cmax_f = float(np.max(conv_f))

        # Very fine-grained mirror descent with low temperature
        # Multiple passes with decreasing temperature
        for pass_i in range(5):
            remain = deadline - time.monotonic()
            if remain < 10:
                break
            tau_start = (0.1 - 0.02 * pass_i) * cmax_f
            tau_end = max(0.001 * cmax_f, 1e-15)
            lr_f = 0.3 + 0.2 * pass_i

            a_final, cmax_f = run_mirror_descent(
                a_final, n_iters=20000, lr=lr_f,
                tau_init=tau_start, tau_min=tau_end,
                N_fft=N_fft_f, time_budget=min(remain * 0.8, 120.0)
            )
            obj_f = 2.0 * n_f * cmax_f
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_final.copy()
                print(f"  Polish pass {pass_i}: obj={best_score:.8f}")

        # Final LP pass
        remain = deadline - time.monotonic()
        if remain > 20 and n_f <= 2000:
            a_lp_f, lpf_cmax = lp_alternate(a_final, max_iters=15, time_budget=min(40, remain * 0.4))
            obj_lp = 2.0 * n_f * lpf_cmax
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp_f.copy()
                print(f"  Final LP: obj={best_score:.8f}")

        # Tiny perturbation + MD
        remain = deadline - time.monotonic()
        if remain > 15 and best_seq is not None:
            a_tiny = normalize_sum1(best_seq * np.exp(rng.normal(0, 0.005, len(best_seq))))
            nt = len(a_tiny)
            Nf = next_fast_len(3 * nt)
            conv_t = compute_conv(a_tiny)
            a_tiny, cmax_t = run_mirror_descent(
                a_tiny, n_iters=10000, lr=0.5,
                tau_init=0.05 * float(np.max(conv_t)), tau_min=0.001 * float(np.max(conv_t)),
                N_fft=Nf, time_budget=min(remain * 0.5, 60.0)
            )
            obj_t = 2.0 * nt * cmax_t
            if obj_t < best_score:
                best_score = obj_t
                best_seq = a_tiny.copy()
                print(f"  Tiny pert: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
