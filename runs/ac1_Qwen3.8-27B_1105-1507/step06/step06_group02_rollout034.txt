def smooth_gradient_fast(a, tau):
    """Faster smooth gradient using FFT."""

    p = exp_b / (exp_b.sum() + 1e-30)

    conv_full = np.fft.irfft(faf * fp, size)
    grad = 2.0 * conv_full[n-1:2*n-1]
    
    return grad, p, b, b_max


def run_smooth_descent(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                       lr_init=0.4, lr_end=0.003):

    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_gradient_fast(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        
        # Nesterov momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * 0.81)) / 2.0  # beta=0.9
        a_new = np.maximum(y - lr * (grad + 0.9 * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a

def conv_efficient(a, g):
    """Compute conv(a, g) efficiently using FFT."""
    n = max(len(a), len(g))
    size = 1
    while size < len(a) + len(g) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    return np.fft.irfft(fa * fg, size)[:len(a) + len(g) - 1]

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast smooth gradient descent with Nesterov acceleration for initial convergence
2. Mirror descent with working-set (constraint generation) LP for the inner
   best-response computation - reducing each LP from O((2n)^3) to O(|W|^3)
3. Anderson acceleration to speed up mirror descent convergence
4. Backtracking line search for robust step size selection
5. Multi-resolution and multi-length strategy for global search
Key innovation: Working-set LP makes ~100x more mirror descent iterations
feasible within the time budget, enabling convergence to tighter optima.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a, b=None):
    """Compute full convolution of a (and b if provided) using FFT."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def conv_aa(a):
    """Compute conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute objective J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = conv_aa(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad(a, tau):
    """Compute smooth gradient of max(conv(a,a)) using soft-max approximation.
    
    Returns (gradient, boltzmann_weights, convolution, max_value).
    """
    b = conv_aa(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / (exp_b.sum() + 1e-30)
    
    n = len(a)
    # grad[m] = 2 * sum_k p[k] * a[k-m] (if 0 <= k-m < n)
    # = 2 * conv(p, a_flipped)[m + n - 1]
    a_flip = a[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    fp = np.fft.rfft(p, size)
    faf = np.fft.rfft(a_flip, size)
    conv_result = np.fft.irfft(fp * faf, size)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    
    return grad, p, b, b_max


def run_smooth_descent(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                       lr_init=0.4, lr_end=0.002):
    """Nesterov-accelerated smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(conv_aa(a).max())
    y = a.copy()
    beta = 0.9
    t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        momentum = 0.9 * (y - a)
        a_new = np.maximum(y - lr * (grad + momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a


def solve_lp_working_set(a, M, g_init=None, max_inner_iter=50, max_ws_size=150, tol=1e-12):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Uses constraint generation (working set) for speed.
    """
    n = len(a)
    
    if g_init is not None and g_init.sum() > 1e-15:
        g = np.maximum(g_init.astype(np.float64), 0)
    else:
        g = np.ones(n) / n
    
    # Determine initial working set from current constraint values
    ag = fast_conv(a, g)
    sort_idx = np.argsort(-ag)[:8]
    ws = set(sort_idx.tolist())
    
    for ws_iter in range(max_inner_iter):
        ws_list = sorted(ws)
        num_c = len(ws_list)
        
        # Build constraint matrix A_ws where A_ws[k_idx, i] = a[k-i] if valid
        rows = []
        cols = []
        vals = []
        for k_idx, k in enumerate(ws_list):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                rows.append(k_idx)
                cols.append(i)
                vals.append(a[k - i])
        
        A = csr_matrix((vals, (rows, cols)), shape=(num_c, n))
        c = -np.ones(n)
        b_ub = np.full(num_c, M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 2.0})
            if not r.success or r.x is None:
                break
            g_new = np.maximum(r.x, 0)
        except Exception:
            break
        
        if g_new.sum() < 1e-15:
            break
        
        g = g_new
        
        # Check all constraints for violations
        ag = fast_conv(a, g)
        violations = ag - M
        violated = np.where(violations > max(tol, 1e-10 * M))[0]
        
        if len(violated) == 0:
            break
        
        # Add violated constraints to working set
        ws.update(violated.tolist())
        
        # Cap working set size - keep most violated
        if len(ws) > max_ws_size:
            ws_list_all = sorted(ws)
            viol_vals = ag[ws_list_all]
            sorted_by_viol = sorted(ws_list_all, key=lambda k: -ag[k])
            ws = set(sorted_by_viol[:max_ws_size])
    
    if g.sum() > 1e-15:
        return g
    return None


def solve_lp_full(a, M):
    """Solve the full LP (expensive but exact)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 5.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(history_x, history_Fx, m):
    """Anderson mixing: find alpha minimizing ||sum alpha_i*(x_i - Fx_i)||^2 s.t. sum alpha_i = 1."""
    residuals = []
    for i in range(m):
        residuals.append(history_x[i] - history_Fx[i])
    residuals = np.array(residuals)
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -2.0, 2.0)
    s_alpha = alpha.sum()
    if abs(s_alpha) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s_alpha
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def run_mirror_descent(a, deadline, max_iters=500, t_init=0.4, use_anderson=True,
                       anderson_window=5, ws_max_size=120, full_lp_every=200):
    """Mirror descent with working-set LP, Anderson acceleration, and backtracking."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(conv_aa(a).max())
    g_prev = None
    no_imp = 0
    
    hist_x = []
    hist_Fx = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = conv_aa(a)
        M = float(b.max())
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        
        # Solve LP for best response
        if it % full_lp_every == full_lp_every - 1 and n <= 2048:
            g = solve_lp_full(a, M)
        else:
            g = solve_lp_working_set(a, M, g_init=g_prev, max_inner_iter=40,
                                    max_ws_size=ws_max_size)
        
        if g is None:
            break
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        g_prev = g.copy()
        
        # Backtracking line search
        best_t = t_init / (1 + 0.03 * it)
        best_M_new = M
        candidates = [best_t, best_t * 1.5, best_t * 0.7, best_t * 0.4, 0.05, 0.02]
        for t in candidates:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = float(conv_aa(a_t).max())
            if M_t < best_M_new:
                best_M_new = M_t
                best_t = t
        
        a_candidate = np.maximum((1.0 - best_t) * a + best_t * g, 0.0)
        s = a_candidate.sum()
        if s < 1e-10:
            a_candidate = a.copy()
        else:
            a_candidate /= s
        
        # Anderson acceleration
        if use_anderson:
            hist_x.append(a.copy())
            hist_Fx.append(a_candidate.copy())
            if len(hist_x) > anderson_window:
                hist_x.pop(0)
                hist_Fx.pop(0)
            if len(hist_x) >= 3:
                a_and = anderson_mix(hist_x, hist_Fx, len(hist_x))
                sa = a_and.sum()
                if sa > 1e-10:
                    a_and /= sa
                    M_and = float(conv_aa(a_and).max())
                    M_cand = float(conv_aa(a_candidate).max())
                    if M_and <= M_cand * 1.02:
                        a_candidate = a_and
        
        a = a_candidate
    
    return best_a


def resample_seq(src, n_target):
    """Resample a sequence to a new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n_target:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_target)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    
    # Triangular
    inits.append(np.maximum(1.0 - np.abs(2 * t - 1), 0.01))
    
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """Main search function. Returns the best sequence found within budget_s seconds."""
    rng = np.random.default_rng(seed)
    absolute_deadline = time.monotonic() + budget_s - 10.0
    best = BestTracker()

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    # Phase 1: Smooth gradient descent from parent for quick improvement
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Gradient Descent", flush=True)
    print(f"{'='*60}", flush=True)
    
    p1_deadline = time.monotonic() + 120.0
    a_p1 = run_smooth_descent(parent, p1_deadline, max_iters=3000,
                              tau_start=1.0, tau_end=200.0,
                              lr_init=0.35, lr_end=0.002)
    best.update(a_p1)
    
    # Phase 2: Mirror descent with working-set LP on the best sequence
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Mirror Descent + Working-Set LP (n={best.seq.size})", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_deadline = min(time.monotonic() + 600.0, absolute_deadline - 200.0)
    n_best = best.seq.size
    
    # Estimate LP time for working set
    t0_lp = time.monotonic()
    b = conv_aa(best.seq)
    M = float(b.max())
    g_test = solve_lp_working_set(best.seq, M, max_inner_iter=20, max_ws_size=80)
    lp_time = time.monotonic() - t0_lp
    print(f"  Working-set LP time: {lp_time*1000:.1f}ms", flush=True)
    
    # Run mirror descent
    max_iters_est = max(50, int(500.0 / max(lp_time, 0.005)))
    a_p2 = run_mirror_descent(best.seq, p2_deadline, max_iters=max_iters_est,
                              t_init=0.4, use_anderson=True, anderson_window=5,
                              ws_max_size=100, full_lp_every=100)
    best.update(a_p2)
    print(f"  Phase 2 iterations completed", flush=True)
    
    # Phase 3: Try multiple lengths
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    p3_start = time.monotonic()
    p3_deadline = min(p3_start + 300.0, absolute_deadline - 100.0)
    
    lengths_to_try = []
    base_n = best.seq.size
    for delta in [-128, -64, 64, 128, 256, 512]:
        n_try = base_n + delta
        if 128 <= n_try <= 4096:
            lengths_to_try.append(n_try)
    # Also try some "round" lengths
    for n_round in [512, 768, 1024, 1536, 2048, 3072]:
        if n_round not in lengths_to_try and abs(n_round - base_n) < base_n:
            lengths_to_try.append(n_round)
    
    lengths_to_try = sorted(set(lengths_to_try))
    print(f"  Lengths to try: {lengths_to_try}", flush=True)
    
    for n_try in lengths_to_try:
        if time.monotonic() > p3_deadline:
            break
        
        remaining = p3_deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Check if this length is already the best
        if n_try == best.seq.size:
            continue
        
        # Resample from current best
        a_resample = resample_seq(best.seq, n_try)
        a_resample = np.maximum(a_resample, 1e-6)
        a_resample /= a_resample.sum()
        
        # Smooth descent
        sub_dl = time.monotonic() + min(60.0, remaining * 0.4)
        a_smooth = run_smooth_descent(a_resample, sub_dl, max_iters=1000,
                                      tau_start=3.0, tau_end=100.0,
                                      lr_init=0.2, lr_end=0.005)
        
        J_smooth = compute_J(a_smooth)
        print(f"  n={n_try}: smooth J={J_smooth:.6f}", flush=True)
        best.update(a_smooth)
        
        # Mirror descent if time permits
        remaining = p3_deadline - time.monotonic()
        if remaining > 60:
            est_lp = max(0.005, lp_time * (n_try / n_best) ** 2)
            max_it = max(20, int(min(remaining * 0.6, 120.0) / max(est_lp, 0.005)))
            a_md = run_mirror_descent(a_smooth, time.monotonic() + min(remaining * 0.6, 120.0),
                                      max_iters=max_it, t_init=0.35,
                                      use_anderson=True, anderson_window=4,
                                      ws_max_size=80, full_lp_every=50)
            best.update(a_md)
    
    # Phase 4: Intensive optimization on the best length
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Intensive optimization (n={best.seq.size})", flush=True)
    print(f"{'='*60}", flush=True)
    
    p4_deadline = min(time.monotonic() + 400.0, absolute_deadline - 20.0)
    
    # Multiple mirror descent passes with different parameters
    for pass_idx, params in enumerate([
        {'t_init': 0.3, 'ws_max_size': 120, 'anderson_window': 5},
        {'t_init': 0.2, 'ws_max_size': 80, 'anderson_window': 4},
        {'t_init': 0.4, 'ws_max_size': 150, 'anderson_window': 6},
    ]):
        if time.monotonic() > p4_deadline:
            break
        remaining = p4_deadline - time.monotonic()
        if remaining < 30:
            break
        sub_dl = time.monotonic() + min(remaining * 0.7, 150.0)
        a_md = run_mirror_descent(best.seq, sub_dl, max_iters=500,
                                  t_init=params['t_init'], use_anderson=True,
                                  anderson_window=params['anderson_window'],
                                  ws_max_size=params['ws_max_size'],
                                  full_lp_every=50)
        best.update(a_md)
        print(f"  Pass {pass_idx+1}: J={best.score:.8f}", flush=True)
    
    # Phase 5: Final polish with full LP
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if time.monotonic() < absolute_deadline - 5:
        # Short smooth polish
        a_pp = run_smooth_descent(best.seq, min(time.monotonic() + 30, absolute_deadline - 10),
                                  max_iters=500, tau_start=10, tau_end=50,
                                  lr_init=0.1, lr_end=0.005)
        best.update(a_pp)
        
        # Mirror descent with tighter parameters
        if time.monotonic() < absolute_deadline - 3:
            a_final = run_mirror_descent(best.seq, absolute_deadline, max_iters=200,
                                         t_init=0.15, use_anderson=True,
                                         anderson_window=5, ws_max_size=100,
                                         full_lp_every=50)
            best.update(a_final)
    
    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best.seq]
