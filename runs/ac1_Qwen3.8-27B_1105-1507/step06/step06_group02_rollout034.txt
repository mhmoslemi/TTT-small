```python
    b_shifted = b - b.max()
    exp_b = np.exp(tau * b_shifted)
    log_sum_exp = b.max() + np.log(exp_b.sum()) / tau
    p = exp_b / exp_b.sum()
    # Gradient: sum_k p_k * 2 * a[k-j] for valid j
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
    return log_sum_exp, grad
```

Hmm, the gradient computation with the loop over active peaks could be slow if there are many active peaks. Let me think of a faster way.

Actually, for large tau, only a few peaks will be active (p_k will be concentrated on the max). So the loop should be fast. For small tau, many peaks are active, but then the gradient is smoother and we can afford the computation.

grad = 2 * conv(p, a) (but with the right indexing)

Actually, grad[j] / 2 = sum_k p_k * a[k-j] = (p * a)[j] in the "cross-correlation" sense.

Let me define: (p ⋆ a)[j] = sum_k p[k] * a[k-j] (with a out of range = 0).

This is the cross-correlation of p and a, which can be computed as:
(p ⋆ a) = IFFT(FFT(p) * conj(FFT(a)))

But p has length 2n-1 and a has length n. We need to be careful with padding.

Actually, let me just use the direct computation for clarity and rely on the fact that for large tau, few peaks are active.

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with adaptive soft-max (fast convergence)
2. L-BFGS with continuation on logsumexp approximation (precise refinement)
3. LP-based best response for final polishing
4. Multi-start with asymmetric initializations and length exploration
Key improvements over previous: L-BFGS second-order convergence,
adaptive temperature schedule, asymmetric ramp initializations,
and targeted LP with reduced active set.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_obj_grad(a, tau):
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    val = b_max + np.log(np.sum(np.exp(tau * (b - b_max)))) / tau
    n = len(a)
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
    return val, grad


def run_smooth_nesterov(a, deadline, tau_start=2.0, tau_end=500.0,
                        lr_init=0.15, lr_end=0.001, max_iters=3000):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()
    beta = 0.85
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = (it - 1) / max(iter(max_iters - 1, 1), 1)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        val, grad = smooth_obj_grad(y, tau)
        b = fast_conv(y)
        b_max = float(b.max())
        if b_max < best_M - 1e-12:
            best_M = b_max
            best_a = a.copy()
        t_k = 1.0 + (1.0 + 8.0 * beta * beta) ** 0.5
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def lbfgs_smooth_objective(a, tau):
    val, grad = smooth_obj_grad(a, tau)
    return val, grad


def run_lbfgs_continuation(a, deadline, tau_vals=None, max_iter_per_stage=300):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    if tau_vals is None:
        tau_vals = [5.0, 20.0, 50.0, 100.0, 200.0, 400.0, 800.0, 1500.0]
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    for tau in tau_vals:
        if time.monotonic() > deadline:
            break
        try:
            res = minimize(
                lambda x: lbfgs_smooth_objective(x, tau)[0],
                a.copy(),
                jac=lambda x: lbfgs_smooth_objective(x, tau)[1],
                method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': max_iter_per_stage, 'gtol': 1e-8,
                         'ftol': 1e-12, 'iprint': -1}
            )
            a_new = np.maximum(res.x, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new /= s
            M_new = float(fast_conv(a_new).max())
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
            a = a_new
        except Exception:
            break
    return best_a


def build_conv_matrix(n, a, top_k=50):
    b = fast_conv(a)
    sorted_idx = np.argsort(b)[::-1][:top_k]
    rows = []
    cols = []
    vals = []
    for k in sorted_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * a[k - j])
    M = float(b.max())
    return rows, cols, vals, sorted_idx, M


def lp_best_response(a, M, top_k=50):
    n = len(a)
    rows, cols, vals, sorted_idx, M_actual = build_conv_matrix(n, a, top_k)
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(sorted_idx), n))
    c = -np.ones(n)
    b_ub = np.full(len(sorted_idx), M_actual)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_refine(a, deadline, max_iters=30, top_k=50):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 10:
                break
        g = lp_best_response(a, M, top_k)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        improved = False
        for t in [0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = float(fast_conv(a_t).max())
            if M_t < best_M - 1e-15:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if not improved:
            no_imp += 1
            if no_imp > 10:
                break
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        # Asymmetric perturbation
        ramp = np.linspace(-1, 1, n)
        inits.append(np.maximum(p * (1.0 + 0.05 * ramp), 0.005))
        inits.append(np.maximum(p * (1.0 + 0.1 * ramp), 0.005))
        inits.append(np.maximum(p * (1.0 + 0.15 * ramp), 0.005))
        # Quadratic asymmetry
        quad = (t - 0.5) ** 2 - 0.25
        inits.append(np.maximum(p * (1.0 + 0.2 * quad), 0.005))
    # Uniform
    inits.append(np.ones(n, dtype=np.float64))
    # Slight ramps (asymmetric)
    for eps in [0.1, 0.2, 0.3, 0.5]:
        ramp = np.linspace(-eps, eps, n)
        inits.append(np.maximum(1.0 + ramp, 0.01))
    # Exponential tilt
    for alpha in [0.5, 1.0, 2.0]:
        inits.append(np.maximum(np.exp(alpha * (t - 0.5)), 0.01))
    # Bump in center
    for width in [0.2, 0.3, 0.5]:
        bump = np.exp(-(t - 0.5) ** 2 / (2 * width ** 2))
        inits.append(0.5 + 0.5 * bump)
    # Two bumps
    inits.append(0.3 + 0.4 * np.exp(-((t-0.3)**2)/0.02) + 0.4 * np.exp(-((t-0.7)**2)/0.02))
    # Cosine variations
    for f in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t), 0.05))
    # Random smooth
    k = max(5, n // 10)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    return inits
```
