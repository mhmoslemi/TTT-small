"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def fft_cross_correlation(a, w):
    """Compute cross-correlation: corr[i] = sum_j a[j] * w[i+j] for i=0..n-1."""
    n = len(a)
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    return corr[:n]


def subgradient_with_backtracking(a, max_iters=3000, time_budget=60.0,
                                   tau_init=0.01, tau_decay=0.9997,
                                   momentum=0.85, init_step=0.1,
                                   rho=0.5, c1=1e-4):

    Uses soft-max weighting of peaks with adaptive temperature.

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    tau = tau_init
    step = init_step

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient: grad[i] = 2 * sum_j a[j] * w[i+j]
        grad = 2.0 * fft_cross_correlation(a, weights)
        # Center (project onto sum=const)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        # Nesterov: look-ahead step
        a_look = a + (momentum / (1.0 - momentum)) * v if momentum > 0 else a
        # Recompute gradient at look-ahead (approximate: use same direction)

        accepted = False

def constraint_generation_lp(a_init, time_budget=120.0, max_iters=100):
    """
    Constraint generation (cutting plane) approach:
    Start with a small active set, solve LP, check all constraints,
    add violated ones, repeat.
    """
    n = len(a_init)
    a = normalize_simplex(a_init.copy().astype(np.float64))
    conv = compute_conv(a)
    max_val = float(np.max(conv))

    best_a = a.copy()
    best_peak = max_val
    start = time.monotonic()

    # Initial active set: lags at or very near the peak
    active = np.where(conv >= max_val * 0.999)[0]
    if len(active) < 2:
        active = np.argsort(conv)[-5:]

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Build LP: min t s.t. sum_i g[i]*a[k-i] <= t for k in active, sum(g)=1, g>=0
        n_act = len(active)
        c = np.zeros(n + 1)
        c[n] = 1.0

        A_ub = np.zeros((n_act, n + 1))
        for idx, k in enumerate(active):
            k_int = int(k)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n - 1, k_int)
            for i in range(i_lo, i_hi + 1):
                A_ub[idx, i] = a[k_int - i]
            A_ub[idx, n] = -1.0
        b_ub = np.zeros(n_act)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0.0, None)] * (n + 1)

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(3, remaining - 5)})
        except Exception:
            break

        if res.status != 0:
            break

        g = np.maximum(res.x[:n], 0)
        s = g.sum()
        if s < 1e-10:
            break
        g = g / s

        # Check ALL constraints
        conv_g = compute_conv(g)
        peak_g = float(np.max(conv_g))

        if peak_g < best_peak - 1e-14:
            best_peak = peak_g
            best_a = g.copy()
            a = g

        # Check if active set needs updating
        new_active = np.where(conv_g >= peak_g * 0.995)[0]
        # Also add any previously violated constraints
        if len(new_active) != len(active):
            if len(new_active) > 200:
                new_active = np.argsort(conv_g)[-200:]
            active = new_active
        else:
            # Check if actually the same set
            if not np.array_equal(sorted(active.tolist()), sorted(new_active.tolist())):
                active = new_active
            else:
                # Converged (active set stable)
                break

    return best_a, 2.0 * n * best_peak

            mid2 = (mid + hi) / 2.0
            a_try2 = (1.0 - mid2) * a + mid2 * g
            a_try2 = normalize_simplex(a_try2)
            obj_try2 = compute_obj(a_try2)
            if obj_try2 < obj_try:
                lo = mid
            else:
                hi = mid

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local
            a = best_a_local
            last_t = best_t
        else:
            # No improvement, perturb
            rng = np.random.default_rng(it * 7919 + 3)
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = normalize_simplex(a)
            last_t = 1.0

    return best_a, best_obj


def structured_perturb_and_optimize(a, time_budget=30.0, seed=None):
    """
    Apply structured perturbations (sinusoidal, ramp, block) and optimize each.
    Returns the best result.
    """
    n = len(a)
    a = normalize_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if seed is None:
        seed = int(time.monotonic() * 10000) % 1000000
    rng = np.random.default_rng(seed)

    perturbations = []

    # Sinusoidal perturbations
    for freq in [1, 2, 3, 5, 8]:
        phase = 2 * np.pi * rng.random()
        amp = 0.05 + 0.1 * rng.random()
        i = np.arange(n)
        pert = 1.0 + amp * np.sin(2 * np.pi * freq * i / n + phase)
        perturbations.append(a * pert)

    # Block perturbations
    for _ in range(3):
        block_size = n // (4 + int(rng.integers(0, 4)))
        start_blk = int(rng.integers(0, max(1, n - block_size)))
        pert = np.ones(n)
        pert[start_blk:start_blk + block_size] *= (0.5 + rng.random())
        perturbations.append(a * pert)

    # Ramp perturbations
    for _ in range(2):
        ramp = np.linspace(1.0 - 0.1 * rng.random(), 1.0 + 0.1 * rng.random(), n)
        if rng.random() > 0.5:
            ramp = ramp[::-1]
        perturbations.append(a * ramp)

    # Gaussian smoothing
    for sigma in [0.5, 1.0, 2.0]:
        kernel = np.exp(-0.5 * (np.arange(-10, 11) / sigma) ** 2)
        kernel /= kernel.sum()
        smoothed = np.convolve(a, kernel, mode='same')
        perturbations.append(smoothed)

    time_per_pert = max(2.0, (time_budget - 2) / len(perturbations))

    for i, pert in enumerate(perturbations):
        if time.monotonic() - start > time_budget - 3:
            break
        pert = normalize_simplex(pert)
        # Quick subgradient
        a_opt, obj_opt = subgradient_with_backtracking(
            pert, max_iters=200, time_budget=time_per_pert,
            tau_init=0.005, momentum=0.8)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def generate_diverse_starts(n, num_starts=10, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    starts.append(np.ones(n) / n)

    # Triangular
    tri = 1.0 - np.abs(i - center) / (center + 1)
    tri = np.maximum(tri, 0.01)
    starts.append(tri / tri.sum())

    # Gaussian (narrow)
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 8.0) ** 2))
    starts.append(bell / bell.sum())

    # Gaussian (medium)
    bell_m = np.exp(-((i - center) ** 2) / (2 * (n / 4.0) ** 2))
    starts.append(bell_m / bell_m.sum())

    # Gaussian (wide)
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 2.5) ** 2))
    starts.append(bell_w / bell_w.sum())

    # Cosine window
    cos_win = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
    starts.append(cos_win / cos_win.sum())

    # Random parametric
    for _ in range(num_starts - 6):
        sigma = n * (0.1 + 0.4 * rng.random())
        freq = int(1 + rng.integers(0, 6))
        amp = 0.1 + 0.4 * rng.random()
        phase = 2 * np.pi * rng.random()
        bell_r = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
        a = np.maximum(bell_r * mod, 0)
        s = a.sum()
        if s > 1e-10:
            starts.append(a / s)

    return starts


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize_simplex(a_new)


def polish_sequence(a, time_budget=120.0):
    """
    Full polishing pipeline: constraint gen LP -> alternating LP ->
    subgradient -> structured perturbations -> repeat if time allows.
    """
    n = len(a)
    start = time.monotonic()
    best_a = normalize_simplex(a.copy().astype(np.float64))
    best_obj = compute_obj(best_a)
    print(f"    Polish start: n={n}, obj={best_obj:.8f}, budget={time_budget:.0f}s")

    # Round 1: Constraint generation LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 30 and n <= 1000:
        t_alloc = min(remaining * 0.35, 100.0)
        a_cg, obj_cg = constraint_generation_lp(best_a, time_budget=t_alloc)
        if obj_cg < best_obj:
            best_obj = obj_cg
            best_a = a_cg
            print(f"    CG-LP: {best_obj:.8f}")

    # Round 2: Subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        t_alloc = min(remaining * 0.25, 60.0)
        a_sg, obj_sg = subgradient_with_backtracking(
            best_a, max_iters=int(t_alloc * 30), time_budget=t_alloc,
            tau_init=0.005, momentum=0.85, init_step=0.08)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"    Subgrad: {best_obj:.8f}")

    # Round 3: Alternating LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 30 and n <= 800:
        t_alloc = min(remaining * 0.3, 80.0)
        a_alp, obj_alp = alternating_lp_with_linesearch(best_a, time_budget=t_alloc)
        if obj_alp < best_obj:
            best_obj = obj_alp
            best_a = a_alp
            print(f"    Alt-LP: {best_obj:.8f}")

    # Round 4: Structured perturbations
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t_alloc = min(remaining * 0.3, 40.0)
        a_sp, obj_sp = structured_perturb_and_optimize(best_a, time_budget=t_alloc)
        if obj_sp < best_obj:
            best_obj = obj_sp
            best_a = a_sp
            print(f"    Perturb: {best_obj:.8f}")

    # Round 5: Final subgradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t_alloc = remaining * 0.5
        a_final, obj_final = subgradient_with_backtracking(
            best_a, max_iters=int(t_alloc * 40), time_budget=t_alloc,
            tau_init=0.001, momentum=0.9, init_step=0.03)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"    Final SG: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses constraint generation LP, backtracking
    subgradient, alternating LP, and structured perturbations to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Phase 0: Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: {best_score:.8f}, n={n_parent}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Phase 1: Quick exploration of different lengths
    print("\n=== Phase 1: Length exploration ===")

    n_values = sorted(set([n_parent, n_parent - 100, n_parent + 100,
                           n_parent - 200, n_parent + 200,
                           n_parent - 50, n_parent + 50]))
    n_values = [x for x in n_values if 400 <= x <= 1200]

    candidates = []  # (obj, seq)

    for n in n_values:
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"  Exploring n={n} ({remaining:.0f}s remaining)")

        if n == n_parent and best_seq is not None:
            starts = [normalize_simplex(best_seq.copy())]
        else:
            starts = [resample_sequence(best_seq, n)] if best_seq is not None else [np.ones(n) / n]
            starts.extend(generate_diverse_starts(n, num_starts=5, seed=seed + n))

        # Quick screen: subgradient on each start
        time_per_start = max(5.0, min(30.0, remaining / (len(starts) + 1) * 0.5))

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 250:
                break
            t_alloc = min(time_per_start, 20.0)
            if t_alloc < 3:
                break

            a_opt, obj_opt = subgradient_with_backtracking(
                s, max_iters=int(t_alloc * 20), time_budget=t_alloc,
                tau_init=0.01, momentum=0.8, init_step=0.1)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"    n={n}, start {si}: {obj_opt:.8f} ** NEW BEST **")

            candidates.append((obj_opt, a_opt.copy(), n))

    # Sort candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop 5 candidates: {[f'{c[0]:.6f}' for c in candidates[:5]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 2: Intensive optimization of top candidates
    print("\n=== Phase 2: Intensive optimization ===")

    top_k = min(4, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_cand = candidates[ci][1].copy()
        obj_cand = candidates[ci][0]
        n_cand = len(a_cand)
        alloc = min(remaining * 0.35, 150.0)

        print(f"  Candidate {ci} (n={n_cand}, obj={obj_cand:.8f}): budget={alloc:.0f}s")

        a_pol, obj_pol = polish_sequence(a_cand, time_budget=alloc)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {best_score:.8f} **")
        else:
            print(f"  No improvement (got {obj_pol:.8f}, best={best_score:.8f})")

    # Phase 3: Further length refinement around best
    print(f"\n=== Phase 3: Length refinement (best: {best_score:.8f}, n={len(best_seq)}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        fine_lengths = [x for x in range(n_best - 30, n_best + 31, 10)
                        if 400 <= x <= 1200 and x != n_best]
        fine_lengths = sorted(set(fine_lengths))[:6]  # limit to 6

        for n_test in fine_lengths:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            a_res = resample_sequence(best_seq, n_test)
            alloc = min(remaining * 0.2, 50.0)
            a_pol, obj_pol = polish_sequence(a_res, time_budget=alloc)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** UPDATE to {best_score:.8f} **")
            else:
                print(f"  n={n_test}: {obj_pol:.6f} (no improvement)")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.5, 120.0)
            a_final, obj_final = polish_sequence(best_seq, time_budget=alloc)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final polish: {best_score:.8f}")
            else:
                print(f"  Final polish: {obj_final:.8f} (no improvement)")

        # Additional subgradient with very low temperature
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_alloc = min(remaining * 0.4, 60.0)
            a_sg2, obj_sg2 = subgradient_with_backtracking(
                best_seq, max_iters=int(t_alloc * 50), time_budget=t_alloc,
                tau_init=0.001, tau_decay=0.9999, momentum=0.9, init_step=0.02)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                print(f"  Low-temp SG: {best_score:.8f}")

    # Phase 5: More constraint generation LP if time allows
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        if n_best <= 900 and remaining > 30:
            print(f"\n=== Phase 5: Final CG-LP (n={n_best}) ===")
            t_alloc = min(remaining * 0.6, 80.0)
            a_cg, obj_cg = constraint_generation_lp(best_seq, time_budget=t_alloc, max_iters=80)
            if obj_cg < best_score:
                best_score = obj_cg
                best_seq = a_cg.copy()
                print(f"  CG-LP final: {best_score:.8f}")
            else:
                print(f"  CG-LP final: {obj_cg:.8f} (no improvement)")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        # Fallback
        result = [1.0] * 100
        best_score = 2.0
        val = evaluate_sequence(result)
        if np.isfinite(val):
            best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
