Optimized step function sequence minimizer using LP-based direction finding,
subgradient methods with adaptive step sizes, and diverse starting points

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz_conv(a):
    """Build matrix T such that T @ g = conv(a, g) (linear convolution, length 2n-1)."""
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        # a[j] * g[k] contributes to conv[k+j]
        # conv[m] = sum_j a[j] * g[m-j], so for fixed j,
        # g[j] contributes a[j]*g[j] to conv[2j], a[j]*g[j-1] to conv[2j-1], etc.
        # Actually: conv[m] = sum_{j: 0<=j<n, 0<=m-j<n} a[j]*g[m-j]
        # So for fixed j, the contribution to conv[m] is a[j]*g[m-j] where m-j is in [0,n)
        # i.e., m is in [j, j+n-1]
        # T[m, j] = a[j] when 0 <= m-j < n, i.e., j <= m <= j+n-1
        lo = j
        hi = min(j + n, m)
        T[lo:hi, j] = a[:hi - lo]
    return T

        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,

            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))

    except Exception:
        pass
    return None

def spectral_flatten(a, time_budget=20.0):
    """Iterative spectral flattening via IFFT of compressed |A|^2."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, size)
        power = fa * np.conj(fa).real  # |fa|^2
        p_max = float(np.max(power))
        if p_max < 1e-30:
            break
        
        # Compress: power_new = power^alpha (normalized)
        alpha = 0.5 + 0.3 * (it / 200.0)
        power_norm = power / p_max
        power_new = (power_norm ** alpha) * p_max
        
        # Reconstruct: need fa_new such that |fa_new|^2 = power_new
        # Keep phase of fa
        phase = np.angle(fa)
        fa_new = np.sqrt(power_new) * np.exp(1j * phase)
        
        a_new = np.real(irfft(fa_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.7 * a + 0.3 * a_new
            a = normalize(a)
    
    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast multi-peak subgradient (additive + multiplicative), LP direction finding,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale-invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def build_toeplitz_conv(a):
    """Build Toeplitz matrix T: (T@g)[k] = (conv(a,g))[k], shape (2n-1, n)."""
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, m)
        T[lo:hi, j] = a[:hi - lo]
    return T


def solve_lp_direction(a, M, time_limit=20):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    T = build_toeplitz_conv(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def subgradient_additive(a, step_size, n_peaks=1, temperature=0.0):
    """
    Multi-peak subgradient step (additive).
    grad[i] for peak k: 2*a[k-i] if 0<=k-i<n, else 0.
    Uses softmax weighting over top peaks if n_peaks > 1.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    if n_peaks == 1:
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
    else:
        # Softmax weights over top peaks
        top_idx = np.argsort(conv)[::-1][:n_peaks]
        if temperature > 0:
            weights = np.exp((conv[top_idx] - cmax) / temperature)
            weights /= weights.sum()
        else:
            weights = np.ones(n_peaks) / n_peaks

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += w * 2.0 * a[j]

    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return a.copy()
    return a_new / s


def subgradient_multiplicative(a, step_size, n_peaks=1, temperature=0.0):
    """
    Multiplicative (mirror descent) subgradient step.
    a_new[i] = a[i] * exp(-eta * grad[i] / max(a[i], eps))
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    eps = 1e-15

    if n_peaks == 1:
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
    else:
        top_idx = np.argsort(conv)[::-1][:n_peaks]
        if temperature > 0:
            weights = np.exp((conv[top_idx] - cmax) / temperature)
            weights /= weights.sum()
        else:
            weights = np.ones(n_peaks) / n_peaks
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += w * 2.0 * a[j]

    # Multiplicative update
    safe_a = np.maximum(a, eps)
    log_update = -step_size * grad / safe_a
    log_update = np.clip(log_update, -5.0, 5.0)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return a.copy()
    return a_new / s


def subgradient_optimize(a, time_budget=30.0, initial_step=0.02, mode='additive', n_peaks=1):
    """Subgradient optimization with backtracking and adaptive step size."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = initial_step
    stuck = 0
    best_step = initial_step

    while time.monotonic() - start < time_budget:
        if mode == 'additive':
            a_new = subgradient_additive(a, step, n_peaks=n_peaks)
        else:
            a_new = subgradient_multiplicative(a, step, n_peaks=n_peaks)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stuck = 0
            best_step = step
            step = min(step * 1.03, 0.08)
        else:
            stuck += 1
            step *= 0.7
            if stuck > 30:
                # Perturb and reset
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))
                step = best_step
                stuck = 0

        if step < 1e-9:
            break

    return best_a


def lp_optimize(a, max_iters=20, time_budget=50.0):
    """LP-based direction finding with adaptive step size."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 3:
            break
        remaining = time_budget - elapsed
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        lp_time = min(remaining * 0.5, 20.0)
        g = solve_lp_direction(a, M, time_limit=max(2, int(lp_time)))
        if g is None:
            break

        # Try decreasing step sizes
        best_t = 0.0
        best_trial_obj = best_obj
        for t in [1.0, 0.7, 0.5, 0.3, 0.15]:
            a_try = normalize((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_trial_obj:
                best_trial_obj = obj_try
                best_t = t

        if best_t > 0:
            a_new = normalize((1.0 - best_t) * a + best_t * g)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
        else:
            # Perturb to escape
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            a = normalize(a * np.exp(rng.normal(0, 0.04, n)))

    return best_a


def spectral_flatten(a, time_budget=20.0):
    """Iterative spectral flattening: compress |FFT(a)|^2 and project back."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, size)
        power = (fa * np.conj(fa)).real
        p_max = float(np.max(power))
        if p_max < 1e-30:
            break

        alpha = 0.4 + 0.4 * min(it / 100.0, 1.0)
        power_norm = power / p_max
        power_new = (np.maximum(power_norm, 0) ** alpha) * p_max

        phase = np.angle(fa)
        fa_new = np.sqrt(np.maximum(power_new, 0)) * np.exp(1j * phase)

        a_new = np.real(irfft(fa_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Partial update
            a = normalize(0.6 * a + 0.4 * a_new)

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def make_inv_sqrt(n):
    """a[k] proportional to 1/sqrt(k+1)."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    return a / np.sum(a)


def make_uniform(n):
    return np.ones(n, dtype=np.float64) / n


def try_load_parent():
    """Load parent sequence from global variable if available."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP direction finding, multi-peak subgradient
    (additive and multiplicative), spectral flattening, diverse starting points,
    multi-n exploration, and final polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 35.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    rng = np.random.default_rng(seed)

    # =========================================================
    # Phase 1: Core LP + subgradient alternating at parent n
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Core optimization (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())

        for cycle in range(8):
            if time.monotonic() > deadline - 250:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.12, 70.0)

            # LP step
            if n_parent <= 900 and alloc > 15:
                a_lp = lp_optimize(a, max_iters=12, time_budget=alloc * 0.45)
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a = a_lp
                    print(f"  Cycle {cycle} LP: {best_score:.8f}")
                else:
                    a = normalize(0.6 * normalize(a) + 0.4 * a_lp)
                    print(f"  Cycle {cycle} LP: {obj_lp:.8f}")
            else:
                a = a.copy()
                print(f"  Cycle {cycle}: skipping LP (n too large or time low)")

            # Subgradient step (additive)
            if time.monotonic() < deadline - 250:
                a_sg = subgradient_optimize(a, time_budget=min(alloc * 0.3, 15.0),
                                            initial_step=0.015, mode='additive', n_peaks=1)
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    a = a_sg
                    print(f"  Cycle {cycle} SG+: {best_score:.8f}")
                else:
                    a = normalize(0.5 * normalize(a) + 0.5 * a_sg)
                    print(f"  Cycle {cycle} SG+: {obj_sg:.8f}")

            # Subgradient step (multiplicative, multi-peak)
            if time.monotonic() < deadline - 250:
                a_sg2 = subgradient_optimize(a, time_budget=min(alloc * 0.25, 12.0),
                                             initial_step=0.01, mode='multiplicative', n_peaks=3)
                obj_sg2 = compute_obj(a_sg2)
                if obj_sg2 < best_score:
                    best_score = obj_sg2
                    best_seq = a_sg2.copy()
                    a = a_sg2
                    print(f"  Cycle {cycle} SG*: {best_score:.8f}")
                else:
                    a = normalize(0.5 * normalize(a) + 0.5 * a_sg2)
                    print(f"  Cycle {cycle} SG*: {obj_sg2:.8f}")

    # =========================================================
    # Phase 2: Spectral flattening
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Spectral flattening")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        if time.monotonic() < deadline - 150:
            a_spec = spectral_flatten(a, time_budget=25.0)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                print(f"  Spectral: {best_score:.8f}")
            else:
                print(f"  Spectral: {obj_spec:.8f}")

            # Follow up with subgradient after spectral
            if time.monotonic() < deadline - 150:
                a_follow = subgradient_optimize(a_spec, time_budget=15.0,
                                                initial_step=0.01, mode='additive', n_peaks=2)
                obj_follow = compute_obj(a_follow)
                if obj_follow < best_score:
                    best_score = obj_follow
                    best_seq = a_follow.copy()
                    print(f"  Post-spectral SG: {best_score:.8f}")

    # =========================================================
    # Phase 3: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700

    n_options = []
    for delta in [-200, -150, -100, -75, -50, -25, 0, 25, 50, 75, 100, 150, 200, 300]:
        n_try = n_best + delta
        if 200 <= n_try <= 1500:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if n_try == n_best:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample(best_seq, n_try))

        alloc = min(remaining * 0.07, 18.0)

        # Quick subgradient
        a_try = subgradient_optimize(a_try, time_budget=min(alloc * 0.6, 10.0),
                                     initial_step=0.012, mode='additive', n_peaks=1)

        # If n is small enough, also try LP
        if n_try <= 800 and alloc > 12:
            a_try = lp_optimize(a_try, max_iters=6, time_budget=min(alloc * 0.35, 8.0))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 4: Diverse restarts with different initializations
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Diverse restarts")
    print(f"{'='*60}")

    n_cur = len(best_seq) if best_seq is not None else 700

    restarts_info = []
    # Perturbation strategies
    for restart in range(12):
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.1, 22.0)

        base = normalize(best_seq.copy())
        strategy = restart % 6

        if strategy == 0:
            # Gaussian multiplier
            a_try = base * np.exp(rng.normal(0, 0.07, n_cur))
        elif strategy == 1:
            # Sinusoidal modulation (multiple frequencies)
            freq1 = rng.uniform(1.0, 6.0)
            freq2 = rng.uniform(6.0, 15.0)
            mod = (1.0 + 0.15 * np.sin(2 * np.pi * freq1 * np.arange(n_cur) / n_cur)
                   + 0.1 * np.sin(2 * np.pi * freq2 * np.arange(n_cur) / n_cur))
            a_try = base * mod
        elif strategy == 2:
            # Blend with inverse-sqrt
            a_inv = make_inv_sqrt(n_cur)
            mix = rng.uniform(0.2, 0.5)
            a_try = (1 - mix) * base + mix * a_inv
        elif strategy == 3:
            # Dirichlet blend
            concentration = rng.uniform(0.2, 1.0)
            a_rand = rng.dirichlet(np.full(n_cur, concentration))
            mix = rng.uniform(0.1, 0.35)
            a_try = (1 - mix) * base + mix * a_rand
        elif strategy == 4:
            # Block reversal
            a_try = base.copy()
            n_blocks = int(rng.integers(3, min(10, n_cur // 50 + 1)))
            block_size = n_cur // n_blocks
            indices = list(range(n_blocks))
            rng.shuffle(indices)
            new_a = np.zeros(n_cur)
            for new_idx, old_idx in enumerate(indices):
                lo = new_idx * block_size
                hi = (new_idx + 1) * block_size if new_idx < n_blocks - 1 else n_cur
                olo = old_idx * block_size
                ohi = (old_idx + 1) * block_size if old_idx < n_blocks - 1 else n_cur
                new_a[lo:hi] = a_try[olo:ohi]
            a_try = new_a
        else:
            # Mirror + blend
            a_mirror = base[::-1].copy()
            mix = rng.uniform(0.3, 0.7)
            a_try = mix * base + (1 - mix) * a_mirror

        a_try = normalize(a_try)

        # Optimize
        a_try = subgradient_optimize(a_try, time_budget=min(alloc * 0.5, 10.0),
                                     initial_step=0.015, mode='additive', n_peaks=1)
        if time.monotonic() < deadline - 50:
            a_try = subgradient_optimize(a_try, time_budget=min(alloc * 0.25, 6.0),
                                         initial_step=0.008, mode='multiplicative', n_peaks=2)
        if n_cur <= 800 and alloc > 15:
            a_try = lp_optimize(a_try, max_iters=5, time_budget=min(alloc * 0.2, 6.0))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = len(a_try)
            print(f"  Restart {restart} (strat {strategy}): {best_score:.8f} **")
        else:
            print(f"  Restart {restart} (strat {strategy}): {obj_try:.8f}")

    # =========================================================
    # Phase 5: Try different starting points from scratch
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Fresh starts")
    print(f"{'='*60}")

    fresh_starts = [
        ("inv_sqrt", make_inv_sqrt(n_cur)),
        ("uniform", make_uniform(n_cur)),
    ]
    # Also try a few intermediate lengths
    for n_fresh in [n_cur - 50, n_cur + 50]:
        if 200 <= n_fresh <= 1500:
            fresh_starts.append((f"inv_sqrt_{n_fresh}", make_inv_sqrt(n_fresh)))

    for name, a_fresh in fresh_starts:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * 0.15, 20.0)

        a_fresh = subgradient_optimize(a_fresh, time_budget=min(alloc * 0.5, 10.0),
                                       initial_step=0.02, mode='additive', n_peaks=1)
        a_fresh = subgradient_optimize(a_fresh, time_budget=min(alloc * 0.3, 6.0),
                                       initial_step=0.01, mode='multiplicative', n_peaks=3)

        obj_fresh = compute_obj(a_fresh)
        if obj_fresh < best_score:
            best_score = obj_fresh
            best_seq = a_fresh.copy()
            print(f"  Fresh {name}: {best_score:.8f} **")
        else:
            print(f"  Fresh {name}: {obj_fresh:.8f}")

    # =========================================================
    # Phase 6: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        n_final = len(a_final)

        # Alternate between different subgradient modes and LP
        for polish in range(8):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.2, 25.0)

            # Additive subgradient with fine step
            if alloc > 5:
                a_pol = subgradient_optimize(a_final, time_budget=alloc * 0.3,
                                             initial_step=0.006, mode='additive', n_peaks=1)
                obj_pol = compute_obj(a_pol)
                if obj_pol < best_score:
                    best_score = obj_pol
                    best_seq = a_pol.copy()
                    a_final = a_pol
                    print(f"  Polish {polish} SG+: {best_score:.8f}")

            # Multiplicative with multi-peak
            if time.monotonic() < deadline - 5 and alloc > 8:
                a_pol2 = subgradient_optimize(a_final, time_budget=alloc * 0.25,
                                              initial_step=0.005, mode='multiplicative', n_peaks=3)
                obj_pol2 = compute_obj(a_pol2)
                if obj_pol2 < best_score:
                    best_score = obj_pol2
                    best_seq = a_pol2.copy()
                    a_final = a_pol2
                    print(f"  Polish {polish} SG*: {best_score:.8f}")

            # LP if sizes allow
            if n_final <= 800 and alloc > 15 and time.monotonic() < deadline - 5:
                a_pol3 = lp_optimize(a_final, max_iters=8, time_budget=min(alloc * 0.3, 12.0))
                obj_pol3 = compute_obj(a_pol3)
                if obj_pol3 < best_score:
                    best_score = obj_pol3
                    best_seq = a_pol3.copy()
                    a_final = a_pol3
                    print(f"  Polish {polish} LP: {best_score:.8f}")

            # Very fine subgradient
            if time.monotonic() < deadline - 5:
                a_pol4 = subgradient_optimize(a_final, time_budget=min(alloc * 0.15, 5.0),
                                              initial_step=0.003, mode='additive', n_peaks=2)
                obj_pol4 = compute_obj(a_pol4)
                if obj_pol4 < best_score:
                    best_score = obj_pol4
                    best_seq = a_pol4.copy()
                    a_final = a_pol4
                    print(f"  Polish {polish} fine: {best_score:.8f}")

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200, dtype=np.float64)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Ensure sum constraint
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate with actual evaluation
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"Target: 1.5027")
    print(f"{'='*60}")

    return result
