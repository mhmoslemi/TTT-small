Optimized step function search using FFT-based gradient descent with log-sum-exp,
LP alternating, peak deflation, and multi-n exploration.
Key improvements over previous: faster gradient via FFT, Armijo backtracking,
structured initialization exploiting non-palindromic asymmetry, and peak deflation.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv_fast(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n <= 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv_fast(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def fft_gradient(a, beta):
    """Compute gradient of log-sum-exp(beta * conv(a,a)) using FFT."""
    n = len(a)
    conv = compute_conv_fast(a)
    max_c = float(np.max(conv))
    shifted = beta * (conv - max_c)
    shifted = np.clip(shifted, -300, 300)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        weights = np.zeros(2 * n - 1)
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum

    a_rev = a[::-1]

    size = 1
    while size < 3 * n + 1:
        size <<= 1

    Wf = rfft(weights, size)
    Af = rfft(a, size)
    Arf = rfft(a_rev, size)

    conv_wa = irfft(Wf * Af, size)
    conv_war = irfft(Wf * Arf, size)

    R2 = conv_wa[0:n]
    R1 = conv_war[n - 1:2 * n - 1]

    grad = R1 + R2
    return grad


def gradient_descent_lse(a, beta_init=30.0, max_steps=500, time_budget=60.0):
    """Log-sum-exp gradient descent with Armijo backtracking."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = beta_init
    step_size = 0.5 / n

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        grad = fft_gradient(a, beta)
        s = float(np.sum(a))
        grad = grad / (s * s)
        grad = grad - np.mean(grad)

        # Armijo backtracking
        alpha = step_size
        improved = False
        for _ in range(20):
            a_try = a - alpha * grad
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj + 1e-15:
                a = a_try
                best_obj = obj_try
                best_a = a_try.copy()
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try small random perturbation to escape
            rng = np.random.default_rng((step * 31 + 7) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.02, n)))
            if step > 50:
                break

        # Slowly decrease beta (converge to true max)
        if step % 50 == 0 and step > 0:
            beta *= 1.3

    return best_a


def peak_deflation(a, time_budget=30.0):
    """Greedy peak deflation: move mass away from peak contributors."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv_fast(a)
        k_star = int(np.argmax(conv))
        c_max = float(conv[k_star])

        # Find top contributors to the peak
        contribs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contribs[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            # Try moving mass from j1 to a position that doesn't help the peak
            for attempt in range(5):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue

                # Check if target position would increase the peak
                k2_t = k_star - j_target
                if 0 <= k2_t < n and a[k2_t] > 0.5 * a[j2]:
                    continue

                delta = 0.03 * a[j1]
                a_try = a.copy()
                a_try[j1] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)

                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            # Try more aggressive move or perturb
            if contribs and contribs[0][0] > 0:
                j1 = contribs[0][1]
                j2 = contribs[0][2]
                # Find position with minimum contribution to peak
                min_contrib = float('inf')
                j_best = 0
                for j in range(n):
                    k2 = k_star - j
                    val = 0.0
                    if 0 <= k2 < n:
                        val = a[k2]
                    if val < min_contrib:
                        min_contrib = val
                        j_best = j
                if j_best != j1 and j_best != j2:
                    delta = 0.1 * a[j1]
                    a_try = a.copy()
                    a_try[j1] -= delta
                    a_try[j_best] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True

            if not improved:
                a = normalize(a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def lp_alternate(a, max_iters=30, time_budget=80.0):
    """LP alternating with golden section line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        conv = compute_conv_fast(a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break

        # Golden section for blend
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        a_c = normalize((1.0 - c_pt) * a + c_pt * g)
        a_d = normalize((1.0 - d_pt) * a + d_pt * g)
        fc = compute_obj(a_c)
        fd = compute_obj(a_d)

        for _ in range(12):
            if hi - lo < 0.02:
                break
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                a_c = normalize((1.0 - c_pt) * a + c_pt * g)
                fc = compute_obj(a_c)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                a_d = normalize((1.0 - d_pt) * a + d_pt * g)
                fd = compute_obj(a_d)

        t_opt = (lo + hi) / 2.0
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            rng = np.random.default_rng((it * 7919 + 12345) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03 + 0.02 * (it % 5), n)))

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def structured_init(n, variant=0):
    """Create diverse structured initializations."""
    if variant == 0:
        # Asymmetric Gaussian (centered at 1/4)
        x = np.arange(n) / n
        a = np.exp(-8.0 * (x - 0.25) ** 2)
    elif variant == 1:
        # Asymmetric Gaussian (centered at 3/4)
        x = np.arange(n) / n
        a = np.exp(-8.0 * (x - 0.75) ** 2)
    elif variant == 2:
        # Sinusoidal modulation
        x = np.arange(n) / n
        a = 0.5 + 0.5 * np.sin(2 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif variant == 3:
        # Exponential decay
        a = np.exp(-3.0 * np.arange(n) / n)
    elif variant == 4:
        # Two bumps
        x = np.arange(n) / n
        a = np.exp(-12.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-12.0 * (x - 0.7) ** 2)
    else:
        # Linear ramp
        a = np.linspace(1.0, 0.1, n)
    return normalize(a)


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    Phase 1: Multi-n exploration with gradient + LP
    Phase 2: Intensive optimization at best n
    Phase 3: Peak deflation + fine LP
    Phase 4: Diversification and final polish
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # ================================================================
    # Phase 1: Multi-n exploration (200s)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-n exploration")
    print(f"{'='*60}")

    n_options = sorted(set([
        200, 300, 400, 500, 600, 700, 800, 900, 1000, 1100, 1200
    ]))
    # Add neighbors of parent
    if best_seq is not None:
        n_options = sorted(set(list(n_options) + [
            n_parent - 100, n_parent - 50, n_parent,
            n_parent + 50, n_parent + 100
        ]))
    n_options = [n for n in n_options if 150 <= n <= 1300]

    phase1_deadline = start_time + 200.0

    for n_try in n_options:
        if time.monotonic() > min(phase1_deadline, deadline - 300):
            break
        remaining = min(phase1_deadline, deadline - 300) - time.monotonic()
        if remaining < 15:
            break

        if best_seq is not None:
            a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = structured_init(n_try, variant=0)

        # Quick gradient optimization
        alloc = min(remaining * 0.15, 25.0)
        a_try = gradient_descent_lse(a_try, beta_init=20.0, max_steps=200, time_budget=alloc * 0.6)
        obj_try = compute_obj(a_try)

        # Quick LP if n is small enough
        if n_try <= 900 and alloc > 8:
            a_try = lp_alternate(a_try, max_iters=8, time_budget=alloc * 0.35)
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # ================================================================
    # Phase 2: Intensive optimization at best n (500s)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Intensive optimization (n={len(best_seq)})")
    print(f"{'='*60}")

    phase2_deadline = min(start_time + 700.0, deadline - 200.0)
    n_cur = len(best_seq)

    for cycle in range(8):
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.14, 70.0)

        a_cur = normalize(best_seq.copy())

        # LP step
        if n_cur <= 1000 and alloc > 20:
            a_cur = lp_alternate(a_cur, max_iters=15, time_budget=alloc * 0.4)
        obj_cur = compute_obj(a_cur)
        if obj_cur > best_score:
            a_cur = normalize(best_seq.copy())

        # Gradient step
        a_cur = gradient_descent_lse(a_cur, beta_init=30.0, max_steps=300, time_budget=alloc * 0.35)
        obj_cur = compute_obj(a_cur)

        # Peak deflation
        a_cur = peak_deflation(a_cur, time_budget=min(alloc * 0.2, 15.0))
        obj_cur = compute_obj(a_cur)

        if obj_cur < best_score:
            best_score = obj_cur
            best_seq = a_cur.copy()
            print(f"  Cycle {cycle}: {best_score:.8f} **")
        else:
            print(f"  Cycle {cycle}: {obj_cur:.8f}")
            # Perturb for next cycle
            rng = np.random.default_rng(seed + cycle * 100)
            best_seq = normalize(best_seq * np.exp(rng.normal(0, 0.04, n_cur)))

    # ================================================================
    # Phase 3: Diversification (300s)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Diversification")
    print(f"{'='*60}")

    phase3_deadline = min(start_time + 1050.0, deadline - 100.0)
    n_cur = len(best_seq)

    for restart in range(12):
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.09, 30.0)

        rng = np.random.default_rng(seed + 1000 + restart * 7)

        # Diverse perturbation
        r = restart % 6
        if r == 0:
            a_try = best_seq * np.exp(rng.normal(0, 0.1, n_cur))
        elif r == 1:
            # Anti-symmetric modulation
            x = np.arange(n_cur) / n_cur
            mod = 1.0 + 0.2 * np.sin(2 * np.pi * rng.uniform(1, 4) * x)
            a_try = best_seq * np.maximum(mod, 0.1)
        elif r == 2:
            # Block reversal
            a_try = best_seq.copy()
            blk = int(rng.integers(n_cur // 3, 2 * n_cur // 3))
            st = int(rng.integers(0, n_cur - blk))
            a_try[st:st + blk] = a_try[st:st + blk][::-1]
        elif r == 3:
            # Blend with structured init
            a_struct = structured_init(n_cur, variant=restart % 5)
            mix = rng.uniform(0.1, 0.4)
            a_try = (1 - mix) * normalize(best_seq) + mix * a_struct
        elif r == 4:
            # Random mask (zero out random 20%)
            a_try = best_seq.copy()
            mask = rng.random(n_cur) > 0.2
            a_try[~mask] *= 0.1
        else:
            # Frequency-domain perturbation
            size = 1
            while size < 2 * n_cur:
                size <<= 1
            F = rfft(a_try if 'a_try' in dir() else best_seq, size)
            F *= np.exp(rng.normal(0, 0.05, size // 2 + 1) + 1j * rng.normal(0, 0.05, size // 2 + 1))
            a_try = np.real(irfft(F, size))[:n_cur]

        a_try = normalize(a_try)

        # Optimize
        a_try = gradient_descent_lse(a_try, beta_init=25.0, max_steps=150, time_budget=alloc * 0.5)
        if n_cur <= 900:
            a_try = lp_alternate(a_try, max_iters=6, time_budget=alloc * 0.3)
        a_try = peak_deflation(a_try, time_budget=min(alloc * 0.2, 10.0))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  Restart {restart}: {best_score:.8f} **")
        else:
            print(f"  Restart {restart}: {obj_try:.8f}")

    # ================================================================
    # Phase 4: Final polish (remaining time)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final polish")
    print(f"{'='*60}")

    n_final = len(best_seq)

    for polish in range(6):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.2, 35.0)

        a_pol = normalize(best_seq.copy())

        # Alternating: gradient -> LP -> deflation
        a_pol = gradient_descent_lse(a_pol, beta_init=50.0, max_steps=200, time_budget=alloc * 0.4)
        if n_final <= 1000:
            a_pol = lp_alternate(a_pol, max_iters=10, time_budget=alloc * 0.3)
        a_pol = peak_deflation(a_pol, time_budget=min(alloc * 0.25, 12.0))

        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Polish {polish}: {best_score:.8f} **")
        else:
            print(f"  Polish {polish}: {obj_pol:.8f}")

    # ================================================================
    # Final validation
    # ================================================================
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
