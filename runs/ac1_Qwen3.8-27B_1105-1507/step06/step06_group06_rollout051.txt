"""
Optimized search using FISTA (accelerated projected subgradient) for the convex
problem min max_k (a*a)[k] s.t. a>=0, sum(a)=1, combined with LP alternating,
spectral peak reduction, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.signal import correlate
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def proj_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1} using bisection."""
    n = len(v)
    if n == 0:
        return v
    # Find tau such that sum(max(v - tau, 0)) = 1
    lo, hi = float(np.min(v)), float(np.max(v))
    for _ in range(80):
        tau = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - tau, 0.0))
        if s > 1.0:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.maximum(v - tau, 0.0)


def compute_gradient_smoothed(a, conv, temperature):
    """Compute smoothed gradient of max_k (a*a)[k]."""
    n = len(a)
    cmax = np.max(conv)
    shifted = (conv - cmax) / temperature
    weights = np.exp(np.clip(shifted, -500.0, 500.0))
    wsum = np.sum(weights)
    if wsum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= wsum
    
    # g[i] = 2 * sum_m weights[i+m] * a[m] = 2 * correlate(weights, a, 'full')[i]
    g = 2.0 * correlate(weights, a, mode='full')[:n]
    return g, cmax


def fista_optimize(a0, max_iters=5000, L=4.0, tau_init=0.05, tau_end=0.005,
                   time_budget=60.0, verbose=False):
    """
    FISTA for minimize max_k (a*a)[k] s.t. a>=0, sum(a)=1.
    """
    n = len(a0)
    a0 = proj_simplex(a0.astype(np.float64).copy())
    x = a0.copy()
    x_prev = a0.copy()
    t = 1.0
    best_x = a0.copy()
    best_obj = compute_obj(a0)
    start = time.monotonic()
    
    for k in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # FISTA momentum
        t_prev = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + (t_prev - 1.0) / t * (x - x_prev)
        
        # Compute gradient at y
        conv_y = compute_conv(y)
        # Animate temperature
        tau = tau_init + (tau_end - tau_init) * (k / max_iters)
        grad, cmax = compute_gradient_smoothed(y, conv_y, tau)
        
        # Gradient step
        step = 1.0 / L
        z = y - step * grad
        
        # Project
        x_new = proj_simplex(z)
        
        # Evaluate
        obj_new = compute_obj(x_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_x = x_new.copy()
            if verbose and k % 100 == 0:
                print(f"    FISTA it{k}: obj={best_obj:.8f}, cmax={cmax:.6f}")
        
        x_prev = x
        x = x_new
    
    return best_x, best_obj


def fista_adaptive(a0, max_iters=10000, time_budget=120.0, verbose=False):
    """FISTA with adaptive step size (backtracking)."""
    n = len(a0)
    a0 = proj_simplex(a0.astype(np.float64).copy())
    x = a0.copy()
    x_prev = a0.copy()
    t = 1.0
    best_x = a0.copy()
    best_obj = compute_obj(a0)
    start = time.monotonic()
    
    step = 0.5  # initial step size
    tau = 0.05  # initial temperature
    
    for k in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Decrease temperature over time
        tau = max(0.001, 0.05 * (1.0 - k / max_iters))
        
        # FISTA momentum
        t_prev = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + (t_prev - 1.0) / t * (x - x_prev)
        
        # Compute gradient at y
        conv_y = compute_conv(y)
        grad, cmax = compute_gradient_smoothed(y, conv_y, tau)
        
        # Backtracking line search
        z = y - step * grad
        x_new = proj_simplex(z)
        
        # Check if function decreased (sufficient decrease condition)
        # f(x_new) <= f(y) + grad'(x_new - y) + L/2 * ||x_new - y||^2
        # Simplified: just check if objective improved
        obj_new = compute_obj(x_new)
        
        if obj_new > compute_obj(y) + 0.1:
            # Step too large, reduce
            step *= 0.5
            # Recompute with smaller step
            z = y - step * grad
            x_new = proj_simplex(z)
        else:
            step = min(step * 1.1, 2.0)  # cautiously increase
        
        obj_new = compute_obj(x_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_x = x_new.copy()
        
        x_prev = x
        x = x_new
    
    return best_x, best_obj

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) momentum-based gradient descent with adaptive smoothing,
(2) LP alternating, (3) spectral peak reduction, (4) multi-length exploration.
Key improvement: proper momentum (Nesterov-style) acceleration and adaptive
temperature schedule for the softmax-smoothed max operator.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute a*a (full linear convolution) via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def proj_simplex(v):
    """Euclidean projection onto {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    # Sort-based method for speed
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_smoothed(a, conv, tau):
    """Smoothed gradient of max_k (a*a)[k] using softmax with temperature tau."""
    n = len(a)
    cmax = float(np.max(conv))
    shifted = (conv - cmax) / tau
    weights = np.exp(np.clip(shifted, -400.0, 400.0))
    ws = weights.sum()
    if ws < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= ws
    # g[i] = 2 * sum_m weights[i+m] * a[m] for m = 0..n-1, i+m < 2n-1
    # = 2 * np.correlate(weights, a, 'full')[i]
    corr = np.correlate(weights, a, mode='full')
    g = 2.0 * corr[:n]
    return g, cmax


def momentum_grad_descent(a0, max_iters=20000, time_budget=120.0,
                          L=3.0, tau_start=0.1, tau_end=0.003,
                          verbose=False):
    """
    Nesterov-style accelerated gradient descent on the smoothed objective,
    projected onto the simplex.
    """
    n = len(a0)
    x = proj_simplex(a0.astype(np.float64).copy())
    x_prev = x.copy()
    v = np.zeros(n)  # momentum
    t = 1.0
    best_x = x.copy()
    best_obj = compute_obj(x)
    start = time.monotonic()
    
    for k in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive temperature
        frac = k / max(max_iters, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        tau = max(tau, 0.001)
        
        # Nesterov lookahead
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (1.0 - t) / t_new
        y = x + beta * (x - x_prev)
        
        # Gradient at y
        conv_y = compute_conv(y)
        grad, cmax = grad_smoothed(y, conv_y, tau)
        
        # Step
        step = 1.0 / L
        z = y - step * grad
        x_new = proj_simplex(z)
        
        # Evaluate
        obj_new = compute_obj(x_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_x = x_new.copy()
            if verbose and (k % 500 == 0):
                print(f"      MGD it{k}: obj={best_obj:.8f}")
        
        x_prev = x
        x = x_new
        t = t_new
    
    return best_x, best_obj
