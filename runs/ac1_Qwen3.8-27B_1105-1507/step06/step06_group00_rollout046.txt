"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using
smooth gradient descent, LP alternating, multi-peak BLISS, and simulated annealing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)[:2 * n - 1]
    return np.maximum(result, 0)  # Should be non-negative but numerical errors


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def compute_full_obj(a):
    """Compute the full objective: 2*n*max_b/sum_a^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)


def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al. (2011)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def smooth_gradient(a, beta):
    """
    Compute gradient of 2n * (1/beta) * log(sum exp(beta * conv(a,a)))
    w.r.t. a, using FFT-based cross-correlation.
    Returns (gradient, smooth_objective).
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Compute softmax weights
    beta_conv = beta * conv
    max_bc = np.max(beta_conv)
    exp_bc = np.exp(beta_conv - max_bc)
    w = exp_bc / np.sum(exp_bc)
    
    # Smooth objective
    smooth_obj = 2.0 * n * (max_bc + np.log(np.sum(exp_bc))) / beta
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j]
    # This is 2 * cross_correlate(w, a)[j] for j = 0..n-1
    # Use FFT: ifft(fft(w, N) * conj(fft(a, N)))
    N = 1
    while N < len(w) + n - 1:
        N <<= 1
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * A.conj(), N)[:n]
    
    grad = 2.0 * corr
    return grad, smooth_obj

def build_conv_matrix(A, a):
    """Build convolution matrix: A @ g = conv(a, g). Pre-allocate A to save memory."""
    n = len(a)
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_obj = compute_obj_normalized(a)
                best_a = a.copy()
                stall_count = 0
    
    return best_a


def multi_peak_bliss(a, time_budget=30.0, K=5):
    """
    Multi-peak BLISS: chase top K peaks simultaneously.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    step_size = 0.02
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Get top K peak positions
        top_k = np.argsort(conv)[::-1][:K]
        
        # Weighted gradient
        weights = conv[top_k] - np.median(conv) + 1e-10
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for k_idx in range(K):
            k = top_k[k_idx]
            w = weights[k_idx]
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * w * a[j]
        
        # Project to tangent space
        grad_proj = grad - np.mean(grad)
        
        a_new = a - step_size * grad_proj
        a_new = simplex_project(a_new)
        obj_new = compute_obj_normalized(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.8
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 0:
        a_new = a_new / s
    return a_new


def simulated_annealing_refine(a, time_budget=30.0):
    """
    Simulated annealing: random perturbations with Metropolis acceptance.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    T = 0.01
    T_min = 0.0001
    cooling = 0.995
    
    max_iters = int(time_budget * 50)
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Random perturbation
        scale = 0.02 + 0.03 * (T / 0.01)
        a_pert = a * np.exp(rng.normal(0, scale, n))
        a_pert = np.maximum(a_pert, 0)
        s_pert = float(np.sum(a_pert))
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert
        
        obj_pert = compute_obj_normalized(a_pert)
        delta = obj_pert - current_obj
        
        if delta < 0 or rng.random() < math.exp(-delta / T):
            a = a_pert
            current_obj = obj_pert
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()
        
        T = max(T_min, T * cooling)
    
    return best_a


def diverse_init(n, rng, mode):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        # Bimodal
        x = np.arange(n) / n
        a = np.exp(-20 * (x - 0.15)**2) + np.exp(-20 * (x - 0.85)**2) + 0.1
        return a / a.sum()
    elif mode == 3:
        # Edge-heavy
        x = np.arange(n)
        center = n / 2
        a = 1.0 / (1.0 + 0.5 * ((x - center) / (n/4))**2)
        return a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x/2)**2 + 0.05
        return a / a.sum()
    elif mode == 5:
        # Random smooth
        freq = rng.integers(1, 5)
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 6:
        # Triangular
        x = np.arange(n) / n
        a = 1.0 - np.abs(2*x - 1)
        a = a**2 + 0.02
        return a / a.sum()
    else:
        # Inverted triangular (U-shaped)
        x = np.arange(n) / n
        a = np.abs(2*x - 1)**2 + 0.02
        return a / a.sum()
