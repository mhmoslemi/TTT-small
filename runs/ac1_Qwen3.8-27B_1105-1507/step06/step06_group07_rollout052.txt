"""
Multi-strategy optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses subgradient descent with line search, joint-peak handling, LP refinement,
spectrally-motivated initializations, and coordinate descent.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def get_peaks(conv, threshold_frac=0.999):
    """Get all lags where conv is close to the max (within threshold_frac)."""
    max_val = np.max(conv)
    if max_val <= 0:
        return [0]
    threshold = max_val * threshold_frac
    peaks = np.where(conv >= threshold)[0]
    return peaks.tolist()


def subgradient_peak(a, peaks):
    """
    Compute subgradient of max_k conv(a,a)[k] w.r.t. a.
    If peaks is a set of lags achieving (near) the max, the subgradient is the
    average of the individual subgradients.
    
    d(conv(a,a)[k])/d(a[m]) = 2*a[k-m] if 0 <= k-m < n
    """
    n = len(a)
    subgrad = np.zeros(n)
    for k in peaks:
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                subgrad[m] += 2.0 * a[j]
    subgrad /= len(peaks)
    return subgrad


def subgradient_linesearch(a, max_iters=3000, time_budget=60.0, tol=1e-10):
    """
    Subgradient method with Polyak-style step size for minimizing
    max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    max_b = float(np.max(conv))
    # Polyak step: need a lower bound on optimal value
    # Lower bound: 1/n (from Cauchy-Schwarz on conv[0])
    lb = 1.0 / n
    gap = max_b - lb
    step = gap / (np.sum(subgradient_peak(a, [int(np.argmax(conv))] ** 1) ** 2) + 1e-15)
    step = min(step, 0.5)  # Cap step size

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_b = float(np.max(conv))
        peaks = get_peaks(conv, 0.999)
        
        # Compute subgradient
        sg = subgradient_peak(a, peaks)
        
        # Polyak step size: (f(x) - f*) / ||g||^2
        gap = max_b - lb
        sg_norm_sq = np.sum(sg ** 2)
        if sg_norm_sq < 1e-20:
            break
        step = gap / sg_norm_sq
        step = min(step, 0.5)  # Cap
        
        # Take subgradient step
        a_new = a - step * sg
        
        # Project onto {a >= 0, sum(a) = 1}
        # First clip to non-negative
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new

        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

        # Early stopping
        if obj_new - lb < tol:
            break

    return best_a


def adam_log_opt_v2(a, max_iters=10000, time_budget=60.0, beta_start=10.0,
                    beta_end=500.0, lr_start=0.03, lr_decay=0.99999):
    """
    Improved Adam in log-space with LSE smoothing and joint-peak weighting.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start

    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)

        conv = compute_conv(a)
        sum_a = float(np.sum(a))

        cmax = float(np.max(conv))
        shifted = beta * (conv - cmax)
        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        e_sum = float(np.sum(e))
        w = e / e_sum
        ls = cmax + np.log(e_sum) / beta

        obj = 2.0 * n * ls / (sum_a * sum_a)

        # Gradient via cross-correlation
        size = 1
        while size < 2 * len(conv):
            size <<= 1
        W = rfft(w, size)
        A = rfft(a, size)
        cc = irfft(W * np.conj(A), size)
        grad_ls = 2.0 * cc[:n]

        grad_a = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
        grad_x = grad_a * a

        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 100 == 0:
            x -= np.mean(x)

        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_for_direction(a, M, time_limit=15):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns normalized g.
    """
    n = len(a)
    if n > 800:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def peak_shaving(a, time_budget=30.0):
    """
    Iterative peak shaving: identify peak lag, reduce its contribution
    by redistributing mass.
    """

        # conv[peak_lag] = sum_i a[i]*a[peak_lag-i]
        # Elements: pairs (i, peak_lag-i) where both in [0, n-1]
        i_min = max(0, peak_lag - (n - 1))
        i_max = min(peak_lag, n - 1)

        # Strategy: reduce the product by moving mass from larger to smaller
        # Try reducing a[i] for the i that has the largest a[i]*a[peak_lag-i]
        products = []
        for i in range(i_min, i_max + 1):
            j = peak_lag - i
            if 0 <= j < n and i != j:

        improved = False
        for prod, i, j in products[:5]:
            if time.monotonic() - start > time_budget:

                a_try[i] -= delta

                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
                # Also try reverse
                a_try2 = a.copy()
                a_try2[j] -= delta
                a_try2[i] += delta
                if a_try2[j] < -1e-15:
                    continue
                a_try2 = np.maximum(a_try2, 0)

def spectral_init(n, method='flat', seed=42):
    """
    Generate initial sequences motivated by spectral flatness.
    The idea: if we want flat autocorrelation, we want flat power spectrum.
    """
    rng = np.random.default_rng(seed)

    if method == 'flat':
        # Approximate a flat-top window (which has较低 sidelobes)
        a = np.sinc(np.arange(n) - n / 2)
        a = np.abs(a)
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    elif method == 'random_phase':
        # Random phase in frequency domain
        N = 1 << (n - 1).bit_length()
        magnitude = np.ones(N) / np.sqrt(N)
        phase = rng.uniform(0, 2 * np.pi, N)
        # Make it Hermitian for real signal
        phase[0] = 0
        if N % 2 == 0:
            phase[N // 2] = 0
        spectrum = magnitude * np.exp(1j * phase)
        a = np.real(ifft(spectrum))[:n]
        a = np.maximum(a, 0)
        if np.sum(a) < 1e-10:
            a = np.ones(n)
        return a / np.sum(a)
    elif method == 'hann':
        a = np.hanning(n) if n > 1 else np.ones(n)
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    elif method == 'triangle':
        a = 1 - np.abs(np.arange(n) - n / 2) / (n / 2)
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    elif method == 'raised_cosine':
        t = np.linspace(0, 2 * np.pi, n)
        a = 0.5 * (1 + np.cos(t))
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    else:
        return np.ones(n) / n

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

        a_cd = coordinate_descent(best_seq, time_budget=alloc, seed=seed)
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  Coord descent: {best_score:.8f} ***")
        else:
            print(f"  Coord descent: {obj:.8f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n{'='*50}")
    print(f"Phase 6: Multi-n exploration")
    print(f"{'='*50}")

    n_best = len(best_seq)
    n_options = sorted(set([
        max(100, n_best - 100),
        max(100, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 300,
        500,
        800,
        1000,
        1200,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 2000]
    n_options = list(dict.fromkeys(n_options))  # Remove duplicates, preserve order

    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        s_try = np.sum(a_try)
        if s_try > 1e-10:
            a_try = a_try / s_try

        alloc = min(remaining * 0.1, 30.0)
        a_opt = adam_log_opt_v2(a_try, max_iters=10000, time_budget=alloc,
                                beta_start=10.0, beta_end=300.0,
                                lr_start=0.03, lr_decay=0.99998)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  n={n_try}: {obj:.8f}")

    # === Phase 7: Spectral initialization (new!) ===
    print(f"\n{'='*50}")
    print(f"Phase 7: Spectral initialization exploration")
    print(f"{'='*50}")

    n_use = len(best_seq)
    methods = ['flat', 'random_phase', 'hann', 'triangle', 'raised_cosine']
    
    for m_idx, method in enumerate(methods):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        a_spec = spectral_init(n_use, method=method, seed=seed + m_idx)
        # Blend with current best
        for alpha in [0.3, 0.5, 0.7]:
            if time.monotonic() > deadline - 30:
                break
            a_blend = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * a_spec
            a_blend = np.maximum(a_blend, 1e-10)
            a_blend = a_blend / np.sum(a_blend)

            alloc = min((deadline - time.monotonic()) * 0.05, 10.0)
            if alloc < 3:
                break
            a_opt = adam_log_opt_v2(a_blend, max_iters=5000, time_budget=alloc,
                                    beta_start=10.0, beta_end=200.0,
                                    lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Spectral {method} (alpha={alpha}): {best_score:.8f} *** UPDATE")

    # === Phase 8: Random restarts ===
    print(f"\n{'='*50}")
    print(f"Phase 8: Random restarts")
    print(f"{'='*50}")

    n_best = len(best_seq)
    rng = np.random.default_rng(seed + 1000)

    for restart in range(6):
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break

        if restart % 3 == 0:
            scale = 0.02 + 0.01 * restart
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
        elif restart % 3 == 1:
            a_pert = best_seq.copy()
            num_spikes = max(1, n_best // 30)
            for _ in range(num_spikes):
                idx = rng.integers(0, n_best)
                a_pert[idx] += rng.uniform(0.01, 0.05) * np.mean(best_seq)
        else:
            # Blend with uniform
            alpha = rng.uniform(0.1, 0.4)
            a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * np.ones(n_best) / n_best

        a_pert = np.maximum(a_pert, 1e-10)
        s_p = np.sum(a_pert)
        if s_p < 1e-10:
            continue
        a_pert = a_pert / s_p

        alloc = min(remaining * 0.1, 12.0)
        if alloc < 3:
            break
        a_opt = adam_log_opt_v2(a_pert, max_iters=8000, time_budget=alloc,
                                beta_start=10.0, beta_end=300.0,
                                lr_start=0.03, lr_decay=0.99998)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  Restart {restart}: {obj:.8f}")

    # === Phase 9: Final intensive polish ===
    print(f"\n{'='*50}")
    print(f"Phase 9: Final polish")
    print(f"{'='*50}")

    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        # High-beta Adam for fine-tuning
        a_final = adam_log_opt_v2(best_seq, max_iters=10000,
                                  time_budget=min(remaining * 0.4, 20.0),
                                  beta_start=200.0, beta_end=1000.0,
                                  lr_start=0.005, lr_decay=0.99999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final polish: {best_score:.8f} ***")

        # One more subgradient pass
        if time.monotonic() < deadline - 3:
            a_sg2 = subgradient_linesearch(best_seq, max_iters=2000,
                                           time_budget=min(deadline - time.monotonic() - 2, 10.0))
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                print(f"  Final subgradient: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0 / 100] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
