"""
Nesterov-accelerated smoothed optimization with adaptive temperature,
peak-directed coordinate descent, asymmetric perturbations, and multi-n

Key improvements: Nesterov momentum, adaptive beta, true-objective

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """Compute smoothed logsumexp objective and gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -600, 600)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    if e_sum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        ls = cmax
    else:
        w = e / e_sum
        ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)

    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj, w


def nesterov_smooth_opt(a, max_iters=50000, time_budget=100.0,
                        beta_start=10.0, beta_end=500.0,
                        lr_start=0.03, lr_end=0.003, momentum=0.9):
    """
    Nesterov-accelerated gradient descent on smoothed logsumexp objective
    in log-space with adaptive temperature and learning rate.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    v = np.zeros(n)  # Nesterov velocity
    m = np.zeros(n)  # Adam first moment
    vq = np.zeros(n)  # Adam second moment
    t_count = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_true_obj = compute_obj(a)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # Nesterov lookahead
        x_look = x - momentum * v
        x_c = x_look - np.max(x_look)
        a = np.exp(x_c)
        a = a / np.sum(a)

        obj_sm, grad_a, w = smooth_obj_grad(a, beta)

        # Gradient in log-space
        grad_x = grad_a * a
        # Subtract mean for invariance under translation of x
        grad_x -= np.mean(grad_x)

        # Adam update with Nesterov momentum
        t_count += 1
        m = beta1 * m + (1 - beta1) * grad_x
        vq = beta2 * vq + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t_count)
        vq_hat = vq / (1 - beta2 ** t_count)
        adam_step = m_hat / (np.sqrt(vq_hat) + eps)

        # Nesterov update
        v = momentum * v + lr * adam_step
        x = x - v

        # Periodic centering
        if it % 100 == 0:
            x -= np.mean(x)

        # Check true objective periodically
        if it % 200 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

    return best_a

def build_toeplitz(a):
    """Build matrix for (a*g) convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_refine(a, time_budget=20.0):
    """LP-based direction finding with line search."""
    n = len(a)
    if n > 1000:
        return a
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s

    conv = compute_conv(a_norm)
    M = float(np.max(conv))

    A = build_toeplitz(a_norm)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        time_limit = min(time_budget * 0.5, 15)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status != 0:
            return a
        g = np.maximum(res.x, 0)
        sg = float(np.sum(g))
        if sg < 1e-10:
            return a
        g = g / sg
    except Exception:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Fine line search
    t_vals = np.linspace(0.05, 1.0, 20)
    for t in t_vals:
        if time.monotonic() - start > time_budget - 2:
            break
        a_try = (1 - t) * a_norm + t * g
        a_try = np.maximum(a_try, 1e-15)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

                    best_seq = a_post.copy()
                    a = a_post.copy()
                    print(f"  LP+Nest {cyc}: {best_score:.8f} ***")

    # === Phase 4: Peak-directed coordinate descent (150s) ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Peak coordinate descent")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        for cd_round in range(3):
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.2, 40.0)
            a_cd = peak_coordinate_descent(a, time_budget=alloc, seed=seed + cd_round)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                a = a_cd.copy()
                print(f"  CD round {cd_round}: {best_score:.8f} ***")
            else:
                print(f"  CD round {cd_round}: {obj_cd:.8f} (no imp)")

            # Short Nesterov after CD
            if time.monotonic() < deadline - 40:
                remaining = deadline - time.monotonic()
                alloc = min(remaining * 0.1, 15.0)
                a_post = nesterov_smooth_opt(a, max_iters=20000, time_budget=alloc,
                                              beta_start=500.0, beta_end=5000.0,
                                              lr_start=0.005, lr_end=0.0002,
                                              momentum=0.95)
                obj_post = compute_obj(a_post)
                if obj_post < best_score:
                    best_score = obj_post
                    best_seq = a_post.copy()
                    a = a_post.copy()
                    print(f"  CD+Nest {cd_round}: {best_score:.8f} ***")

    # === Phase 5: Asymmetric restarts (150s) ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Asymmetric restarts")
    print(f"{'='*60}")

    rng = np.random.default_rng(seed + 100)
    n_best = len(best_seq) if best_seq is not None else 700

    for restart in range(10):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if best_seq is None:
            a_pert = np.ones(n_best)
        else:
            a_base = best_seq.copy()
            s = np.sum(a_base)
            a_base = a_base / s

            # Different asymmetric perturbation strategies
            strat = restart % 4
            if strat == 0:
                # Strong asymmetric multiplicative
                t = np.linspace(0, 1, n_best)
                bias = np.exp(0.5 * (t - 0.5) * rng.uniform(2, 5))
                a_pert = a_base * bias
            elif strat == 1:
                # Block perturbation (shift mass to one side)
                a_pert = a_base.copy()
                split = n_best // 2 + rng.integers(-n_best // 4, n_best // 4)
                scale = rng.uniform(0.5, 1.5)
                a_pert[:split] *= scale
                a_pert[split:] *= (1.0 / scale)
            elif strat == 2:
                # Add asymmetric linear trend
                trend = np.linspace(1 - rng.uniform(0.2, 0.6), 1 + rng.uniform(0.2, 0.6), n_best)
                a_pert = a_base * trend
            else:
                # Mixed: add a shifted Gaussian bump
                a_pert = a_base.copy()
                center = rng.integers(n_best // 4, 3 * n_best // 4)
                width = rng.integers(n_best // 10, n_best // 4)
                x = np.arange(n_best)
                bump = np.exp(-0.5 * ((x - center) / width) ** 2)
                a_pert += 0.1 * bump

        a_pert = np.maximum(a_pert, 1e-12)
        s_p = np.sum(a_pert)
        if s_p < 1e-10:
            continue
        a_pert = a_pert / s_p

        alloc = min(remaining * 0.1, 15.0)
        a_opt = nesterov_smooth_opt(a_pert, max_iters=20000, time_budget=alloc,
                                     beta_start=50.0, beta_end=500.0,
                                     lr_start=0.02, lr_end=0.002, momentum=0.92)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(a_opt)
            print(f"  Restart {restart} (strat {strat}): {best_score:.8f} *** UPDATE")
        else:
            print(f"  Restart {restart} (strat {strat}): {obj:.8f}")

    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()

        # Very high beta Nesterov
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.4, 60.0)
            a_final = nesterov_smooth_opt(a, max_iters=50000, time_budget=alloc,
                                          beta_start=2000.0, beta_end=20000.0,
                                          lr_start=0.002, lr_end=0.0001,
                                          momentum=0.95)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                a = a_final.copy()
                print(f"  High-beta Nesterov: {best_score:.8f} ***")

            # Final coordinate descent
            if time.monotonic() < deadline - 10:
                remaining = deadline - time.monotonic()
                alloc = min(remaining * 0.3, 20.0)
                a_cd = peak_coordinate_descent(a, time_budget=alloc, seed=seed + 999)
                obj_cd = compute_obj(a_cd)
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    print(f"  Final CD: {best_score:.8f} ***")

            # One more micro LP
            if time.monotonic() < deadline - 8 and len(a_cd if best_seq is a_cd else a) <= 1000:
                remaining = deadline - time.monotonic()
                a_cur = best_seq.copy()
                alloc = min(remaining * 0.3, 10.0)
                a_lp = lp_refine(a_cur, time_budget=alloc)
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  Final LP: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
