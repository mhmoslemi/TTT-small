"""
Multi-phase optimization combining LP refinement, subgradient peak-shaving,
multi-n exploration, and high-precision Adam polish to minimize
2*n*max(conv(a,a))/(sum(a))^2. Key improvements over previous: more aggressive
LP iterations with decreasing M, novel subgradient peak-shaving that targets
multiple peaks simultaneously, systematic n sweeping, and mirrored descent
with multiplicative updates.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """Compute smoothed objective (logsumexp of beta*conv) and its gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)

    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def adam_log_opt(a, max_iters=5000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start

    best_a = a.copy()
    best_true_obj = compute_obj(a)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)

        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a

        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 200 == 0:
            x -= np.mean(x)

        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine_iterative(a, num_iters=8, time_budget=60.0):

        M = float(np.max(conv))

        time_limit = max(3.0, min(remaining * 0.3, 12.0))
        g = solve_lp_max_sum(a_norm, M, time_limit=time_limit)
        if g is None:

            obj_try = compute_obj(a_try)

def peak_shaving_subgradient(a, time_budget=30.0, num_peaks=5, eta_start=0.01, eta_end=0.001):
    """
    Subgradient peak-shaving: identify top peaks, compute sensitivity vectors,
    and take steps that reduce the weighted combination of top peaks.
    Uses multiplicative (mirror) descent for natural non-negativity handling.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    a_work = a.copy() / np.sum(a)

    num_steps = 500

    for step in range(num_steps):
        if time.monotonic() - start > time_budget:
            break

        # Anneal step size
        progress = step / max(num_steps - 1, 1)
        eta = eta_start * (eta_end / eta_start) ** progress

        conv = compute_conv(a_work)
        sorted_idx = np.argsort(conv)[::-1]
        top_k = sorted_idx[:num_peaks]

        # Compute sensitivity: for each top peak, the gradient is 2*a shifted
        # For peak at position k, grad[i] = 2*a[k-i] (if valid), a[k-i] if k==2i
        sens = np.zeros(n)
        weights = np.zeros(len(top_k))
        for idx, k in enumerate(top_k):
            # Weight by how much this peak exceeds the mean
            mean_conv = np.mean(conv)
            weights[idx] = max(conv[k] - mean_conv, 0.1)
            # Compute gradient contribution
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    if i == j:
                        sens[i] += weights[idx] * a_work[j]
                    else:
                        sens[i] += weights[idx] * 2.0 * a_work[j]

        # Normalize sensitivity
        s_max = np.max(sens)
        if s_max < 1e-15:
            break
        sens = sens / s_max

        # Mirror descent step: reduce high-sensitivity elements
        # a_new[i] = a[i] * (1 - eta * sens[i])
        a_try = a_work * (1.0 - eta * sens)
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / np.sum(a_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a_work = a_try.copy()
        else:
            # Try smaller step
            a_try2 = a_work * (1.0 - eta * 0.1 * sens)
            a_try2 = np.maximum(a_try2, 1e-15)
            a_try2 = a_try2 / np.sum(a_try2)
            obj_try2 = compute_obj(a_try2)
            if obj_try2 < best_obj:
                best_obj = obj_try2
                best_a = a_try2.copy()
                a_work = a_try2.copy()
            else:
                # Reduce number of peaks and retry
                num_peaks = max(1, num_peaks - 1)

    return best_a, best_obj


def multiplicative_weight_descent(a, time_budget=30.0, num_iters=300):
    """
    Multiplicative weight update: at each step, compute the peak and
    multiplicatively reduce elements contributing to it.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    a_work = a.copy() / np.sum(a)

    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a_work)
        k_star = int(np.argmax(conv))

        # Sensitivity: elements contributing to peak k_star
        sens = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    sens[i] = 1.0
                else:
                    sens[i] = 2.0

        # Anneal
        eta = 0.005 * (0.998 ** it)

        # Multiplicative update
        a_try = a_work * (1.0 - eta * sens)
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / np.sum(a_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a_work = a_try.copy()

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def generate_structured_init(n, method='cosine'):
    """Generate structured initial sequences with potential for flat convolution."""
    if method == 'cosine':
        # Cosine window - high at edges
        t = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
        a = np.maximum(a, 0.01)
    elif method == 'power':
        # Power law decay from edges
        t = np.arange(n)
        mid = (n - 1) / 2.0
        a = 1.0 / (1.0 + 0.1 * np.abs(t - mid) ** 1.5)
    elif method == 'two_bump':
        # Two bumps at 1/4 and 3/4
        t = np.arange(n)
        a = np.exp(-((t - n * 0.25) ** 2) / (n * 0.05)) + np.exp(-((t - n * 0.75) ** 2) / (n * 0.05))
        a = np.maximum(a, 0.01)
    elif method == 'asym_conf':
        # Asymmetric concentration
        t = np.arange(n)
        a = np.exp(-((t - n * 0.3) ** 2) / (n * 0.1)) + 0.3 * np.exp(-((t - n * 0.7) ** 2) / (n * 0.05))
    elif method == 'ramp':
        # Linear ramp
        a = np.linspace(0.1, 1.0, n)
    else:
        a = np.ones(n)

    a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining:
    1. Multi-n exploration with structured initializations
    2. Intensive iterative LP refinement with decreasing M
    3. Subgradient peak-shaving (novel)
    4. Multiplicative weight descent (novel)
    5. Final high-precision Adam polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, budget={budget_s}s")

    # === Phase 1: Multi-n exploration with diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-n exploration")
    print(f"{'='*60}")

    n_options = sorted(set([
        n_parent - 100, n_parent - 50, n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
        n_parent + 400, n_parent + 600, n_parent + 800,
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500]

    for n_try in n_options:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Determine initial sequence
        if n_try == len(best_seq):
            a_try = best_seq.copy()
            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s
        else:
            # Resample from best, also try structured inits
            a_resampled = resample_sequence(best_seq, n_try)
            a_resampled = a_resampled / np.sum(a_resampled)

            # Quick Adam on resampled
            alloc = min(remaining * 0.08, 15.0)
            a_opt = adam_log_opt(a_resampled, max_iters=8000, time_budget=alloc,
                                 beta_start=10.0, beta_end=200.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)

            # Also try a structured init
            a_struct = generate_structured_init(n_try, 'cosine')
            a_struct2 = generate_structured_init(n_try, 'asym_conf')
            for a_init in [a_opt, a_struct, a_struct2]:
                alloc2 = min(remaining * 0.05, 8.0)
                a_opt2 = adam_log_opt(a_init, max_iters=5000, time_budget=alloc2,
                                      beta_start=20.0, beta_end=300.0,
                                      lr_start=0.03, lr_decay=0.99998)
                obj2 = compute_obj(a_opt2)
                if obj2 < obj:
                    a_opt = a_opt2
                    obj = obj2

            a_try = a_opt

        if compute_obj(a_try) < best_score:
            best_score = compute_obj(a_try)
            best_seq = a_try.copy()
            print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  n={n_try}: {compute_obj(a_try):.8f}")

    # === Phase 2: Intensive iterative LP refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Iterative LP refinement (n={len(best_seq)})")
    print(f"{'='*60}")

    n_best = len(best_seq)
    if n_best <= 1200:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.25, 120.0)
            a_lp, obj_lp = lp_refine_iterative(best_seq, num_iters=10, time_budget=alloc)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP refined: {best_score:.8f} ***")

    # Adam after LP
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        a_post = adam_log_opt(best_seq, max_iters=10000,
                              time_budget=min(remaining * 0.1, 20.0),
                              beta_start=50.0, beta_end=400.0,
                              lr_start=0.02, lr_decay=0.99999)
        obj_post = compute_obj(a_post)
        if obj_post < best_score:
            best_score = obj_post
            best_seq = a_post.copy()
            print(f"  Post-LP Adam: {best_score:.8f} ***")

    # === Phase 3: Subgradient peak-shaving (novel) ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Peak shaving subgradient")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.1, 30.0)
        a_ps, obj_ps = peak_shaving_subgradient(best_seq, time_budget=alloc,
                                                num_peaks=5, eta_start=0.008,
                                                eta_end=0.0005)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Peak shaving: {best_score:.8f} ***")
        else:
            print(f"  Peak shaving: {obj_ps:.8f} (no imp)")

    # === Phase 4: Multiplicative weight descent (novel) ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multiplicative weight descent")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.08, 20.0)
        a_mwd, obj_mwd = multiplicative_weight_descent(best_seq, time_budget=alloc)
        if obj_mwd < best_score:
            best_score = obj_mwd
            best_seq = a_mwd.copy()
            print(f"  MWD: {best_score:.8f} ***")
        else:
            print(f"  MWD: {obj_mwd:.8f} (no imp)")

    # === Phase 5: More LP iterations ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Additional LP refinement")
    print(f"{'='*60}")

    n_best = len(best_seq)
    if n_best <= 1200:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.15, 60.0)
            a_lp2, obj_lp2 = lp_refine_iterative(best_seq, num_iters=6, time_budget=alloc)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  LP round 2: {best_score:.8f} ***")

    # === Phase 6: Final intensive Adam polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Adam polish")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # High-beta Adam (nearly exact)
        alloc = min(remaining * 0.3, 40.0)
        a_final = adam_log_opt(best_seq, max_iters=20000,
                               time_budget=alloc,
                               beta_start=200.0, beta_end=1500.0,
                               lr_start=0.008, lr_decay=0.99999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Adam: {best_score:.8f} ***")

    # One more very fine Adam
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 25.0)
        a_fine = adam_log_opt(best_seq, max_iters=15000,
                              time_budget=alloc,
                              beta_start=500.0, beta_end=2000.0,
                              lr_start=0.004, lr_decay=0.999995)
        obj_fine = compute_obj(a_fine)
        if obj_fine < best_score:
            best_score = obj_fine
            best_seq = a_fine.copy()
            print(f"  Fine Adam: {best_score:.8f} ***")

    # === Phase 7: Random restarts with diverse perturbations ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Random restarts")
    print(f"{'='*60}")

    n_best = len(best_seq)
    rng = np.random.default_rng(seed)

    for restart in range(6):
        if time.monotonic() > deadline - 15:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        if restart % 3 == 0:
            scale = 0.02 + 0.01 * restart
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
        elif restart % 3 == 1:
            # Blend with structured
            alpha = rng.uniform(0.05, 0.2)
            a_struct = generate_structured_init(n_best, method=['cosine', 'power', 'two_bump'][restart % 3])
            a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * a_struct
        else:
            # Local perturbation on peak contributors
            a_pert = best_seq.copy()
            conv = compute_conv(a_pert)
            k_peak = int(np.argmax(conv))
            # Perturb elements near the peak
            i_min = max(0, k_peak - (n_best - 1))
            i_max = min(k_peak, n_best - 1)
            n_perturb = max(5, (i_max - i_min + 1) // 3)
            perturb_idx = rng.choice(range(i_min, i_max + 1), size=min(n_perturb, i_max - i_min + 1), replace=False)
            for i in perturb_idx:
                a_pert[i] *= rng.uniform(0.7, 1.3)

        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / np.sum(a_pert)

        alloc = min(remaining * 0.15, 12.0)
        a_opt = adam_log_opt(a_pert, max_iters=8000, time_budget=alloc,
                             beta_start=50.0, beta_end=500.0,
                             lr_start=0.02, lr_decay=0.99999)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  Restart {restart}: {obj:.8f}")

    # === Phase 8: Final LP if time permits ===
    if best_seq is not None and len(best_seq) <= 1200 and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.4, 40.0)
        a_final_lp, obj_final_lp = lp_refine_iterative(best_seq, num_iters=5, time_budget=alloc)
        if obj_final_lp < best_score:
            best_score = obj_final_lp
            best_seq = a_final_lp.copy()
            print(f"  Final LP: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
