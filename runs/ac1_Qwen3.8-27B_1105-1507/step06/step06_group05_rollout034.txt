import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end-j]
    return A

def solve_lp(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except:
        pass
    return None

def compute_grad_smoothed(a, tau):
    """Compute gradient of tau * log(sum(exp(conv/tau))) w.r.t. a"""
    n = len(a)
    conv = compute_conv(a)
    # Log-sum-exp for numerical stability
    cmax = np.max(conv)
    shifted = (conv - cmax) / tau
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    weights /= w_sum  # Now weights sum to 1
    
    # grad[i] = 2 * sum_k weights[k] * a[k-i] (valid range)
    # This is a cross-correlation
    grad = np.zeros(n)
    for i in range(n):
        k_max = min(i + n, 2*n - 1)
        if k_max > i:
            grad[i] = 2.0 * np.dot(weights[i:k_max], a[:k_max-i])
    return grad, conv

def nesterov_gradient_descent(a, max_iters=500, tau_init=1e-4, time_budget=60.0):

    v = np.zeros(n)  # momentum
    beta = 0.9  # momentum coefficient
    best_a = a.copy()
    best_obj = compute_obj(a)

    lr = 0.1  # initial learning rate

    min_lr = 1e-8
    no_improve = 0

        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov lookahead

        # Compute gradient at y
        y = np.maximum(y, 0.0)
        sy = np.sum(y)
        if sy < 1e-15:

        y = y / sy
        
        grad, conv = compute_grad_smoothed(y, tau)

        grad_centered = grad - np.mean(grad)

"""
Hybrid Nesterov-gradient + LP + spectral optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 with non-negative a.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time

# Precompute FFT sizes globally for speed

def compute_conv(a):
    n = len(a)
    size = 1 << (2*n-2).bit_length()
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0: return np.inf
    s = np.sum(a)
    if s < 0.01: return np.inf
    return 2.0*n*np.max(compute_conv(a)) / (s*s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15: return np.ones(len(a))/len(a)
    return a/s

# Gradient via FFT
def smooth_grad(a, tau, fft_size):
    conv = compute_conv(a)
    cmax = conv.max()
    w = np.exp((conv - cmax)/tau)
    w /= w.sum()
    # grad[i] = 2 * sum_j w[i+j] * a[j], computed via cross-correlation
    W = fft(w.astype(complex), fft_size)
    A = fft(a.astype(complex), fft_size)
    corr = ifft(W * A.conj()).real
    return 2.0 * corr[:len(a)]

# Nesterov GD
def nesterov_opt(a, max_iters, tau, lr, beta, time_budget):
    n = len(a)
    fft_size = 1 << (3*n-3).bit_length()
    a = normalize(a.copy())
    v = np.zeros(n)
    best_a, best_obj = a.copy(), compute_obj(a)
    t0 = time.monotonic()
    stagnation = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget: break
        
        y = a + beta * v
        y = normalize(y)
        g = smooth_grad(y, tau, fft_size)
        g -= g.mean()  # preserve sum
        v = beta*v - lr*g
        a_new = normalize(a + v)
        obj = compute_obj(a_new)
        
        if obj < best_obj - 1e-12:
            best_obj, best_a = obj, a_new.copy()
            a, stagnation = a_new, 0
        else:
            stagnation += 1
            if stagnation > 20:
                lr *= 0.5
                v *= 0.3
                if lr < 1e-10: break
            a = best_a.copy()
    
    return best_a

# LP refinement
def lp_direction(a, M, tl):

    # Build sparse-ish Toeplitz
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a[:min(n, 2*n-1-j)]
    c = -np.ones(n)
    try:
        r = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                    bounds=[(0,None)]*n, method='highs',
                    options={'time_limit': tl})
        if r.status == 0:
            g = np.maximum(r.x, 0)
            s = g.sum()
            if s > 1e-10: return g/s
    except: pass
    return None

def lp_refine(a, max_iters, time_budget):

            else: lo, c, fc = c, d, fd

# Spectral equalization
def spectral_opt(a, max_iters, time_budget):

"""
Hybrid Nesterov-gradient + LP + spectral optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 with non-negative a.
Key improvements over previous: FFT-accelerated gradients, Nesterov momentum,
adaptive temperature, fine multi-n search, and subgradient polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def get_fft_size(n, factor=2):
    """Get next power of 2 >= factor * n."""
    target = factor * n
    size = 1
    while size < target:
        size <<= 1
    return size

def smooth_gradient(a, tau, fft_size):
    """
    Compute gradient of smoothed max-conv objective.
    Returns (grad, conv) where grad[i] = 2 * sum_k w[k] * a[k-i].
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    # Log-sum-exp weights
    shifted = (conv - cmax) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500.0, 500.0)
    w = np.exp(shifted)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    
    # Cross-correlation via FFT: corr[i] = sum_j w[i+j] * a[j]
    W = fft(w.astype(np.complex128), fft_size)
    A = fft(a.astype(np.complex128), fft_size)
    corr = ifft(W * A.conj()).real
    grad = 2.0 * corr[:n]
    return grad, conv

def nesterov_optimize(a, tau_init, lr_init, beta_init, max_iters, time_budget):
    """
    Nesterov-accelerated gradient descent on smoothed objective.
    Adaptive parameters with backtracking.
    """
    n = len(a)
    fft_size = get_fft_size(n, 3)
    a = normalize(np.array(a, dtype=np.float64).copy())
    v = np.zeros(n)
    beta = beta_init
    lr = lr_init
    tau = tau_init
    best_a = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 50
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Nesterov lookahead
        y = a + beta * v
        y = normalize(y)
        
        # Gradient at y
        grad, conv = smooth_gradient(y, tau, fft_size)
        
        # Center gradient (remove component along all-ones)
        grad = grad - np.mean(grad)
        
        # Normalize gradient direction
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad_dir = grad / g_norm
        
        # Update velocity
        v = beta * v - lr * grad_dir
        
        # Limit step size
        v_norm = np.linalg.norm(v)
        max_step = 0.3
        if v_norm > max_step:
            v = v * max_step / v_norm
        
        # Update position
        a_new = normalize(a + v)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slightly increase momentum as we converge
            beta = min(0.99, beta + 0.001)
        else:
            no_improve += 1
            a = best_a.copy()
            if no_improve > 10:
                lr *= 0.7
                v *= 0.5
                beta = max(0.8, beta - 0.05)
            if no_improve > max_no_improve:
                # Reduce temperature
                tau *= 0.5
                lr = lr_init * 0.1
                v = np.zeros(n)
                no_improve = 0
                if tau < 1e-12:
                    break
    
    return best_a

            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95
            a = best_a.copy()
            if no_improve > 200 or lr < 1e-10:
                break
    
    return best_a

def build_toeplitz(a):
    """Build Toeplitz matrix where (T@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, m)
        T[j:k_end, j] = a[:k_end - j]
    return T

def lp_refine(a, max_iters, time_budget):
    """LP-based alternating optimization with golden section line search."""
    n = len(a)
    if n > 1000:
        return normalize(a.copy())
    
    a = normalize(np.array(a, dtype=np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0
        T = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        tl = max(3, int(min(remaining * 0.5, 20.0)))
        
        try:
            res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'time_limit': tl,
                                  'dual_feasibility_tolerance': 1e-9,
                                  'primal_feasibility_tolerance': 1e-9})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            g_sum = float(np.sum(g))
            if g_sum < 1e-10:
                break
            g = g / g_sum
        except Exception:
            break
        
        # Golden section search for optimal blend
        def obj_at_t(t):
            a_try = normalize((1.0 - t) * a + t * g)
            return compute_obj(a_try)
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c_point = hi - gr * (hi - lo)
        d_point = lo + gr * (hi - lo)
        fc = obj_at_t(c_point)
        fd = obj_at_t(d_point)
        for _ in range(15):
            if hi - lo < 0.01:
                break
            if fc < fd:
                hi = d_point
                d_point = c_point
                fd = fc
                c_point = hi - gr * (hi - lo)
                fc = obj_at_t(c_point)
            else:
                lo = c_point
                c_point = d_point
                fc = fd
                d_point = lo + gr * (hi - lo)
                fd = obj_at_t(d_point)
        
        t_opt = (lo + hi) / 2.0
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Perturb to escape
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))
            # Reset to best after perturbation
            a = best_a.copy()
    
    return best_a

def spectral_equalize(a, max_iters, time_budget):
    """POCS-style spectral equalization with adaptive compression."""
    n = len(a)
    size = get_fft_size(n, 2)
    a = normalize(np.array(a, dtype=np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        A = fft(a.astype(np.complex128), size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        
        # Adaptive alpha: start gentle, become more aggressive
        alpha = 0.4 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend to maintain progress
            a = normalize(0.6 * a + 0.4 * a_new)
    
    return best_a

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)
