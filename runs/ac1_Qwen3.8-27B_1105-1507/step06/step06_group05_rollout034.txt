- If 0 <= k-i < n and i ≠ k-i: d(conv[k])/d(a[i]) = 2*a[k-i]
- If i = k-i (i.e., 2i = k) and 0 <= i < n: d(conv[k])/d(a[i]) = a[i]
- Otherwise: 0

This is important! The subgradient of the peak conv[k*] with respect to a[i] is 2*a[k*-i] (or a[i] if 2i = k*).

So the "direction of steepest increase" for conv[k*] is given by the vector v where v[i] = 2*a[k*-i] (for valid indices). To DECREASE conv[k*], we should move in the direction -v, i.e., decrease a[i] where a[k*-i] is large.

This gives a very concrete descent direction: to reduce the peak at lag k*, we should decrease a[i] for those i where a[k*-i] is large. In other words, we should reduce the "alignment" of a with its shift by k*.

**Concrete algorithm:**

1. Compute conv = a * a.
2. Find top-m peaks: k_1, k_2, ..., k_m (by height).
3. For each peak k_j, compute the "descent direction" d_j where d_j[i] = a[k_j - i] (for valid i).
4. Average: d = (1/m) * sum_j d_j.
5. The subgradient of the objective (scaled) is proportional to d.
6. To decrease the objective, we want to decrease a[i] where d[i] is large, and increase a[i] where d[i] is small.
7. Concrete step: a_new[i] = a[i] - step * (d[i] - mean(d)) [removing the mean to preserve the sum]
8. Project: a_new = max(0, a_new), then normalize to sum = 1.

This is essentially a projected subgradient method with a specific structure.

**Even better: Use the fact that the subgradient has a specific form.**

The subgradient of max_k conv[k] at a point where k* is the unique max is:
sg[i] = 2*n * a[k*-i] / (sum(a))^2 (for valid i, with the factor of 2 as discussed)

The steepest descent direction (in the L2 norm) on the constraint manifold {a >= 0, sum(a) = 1} is:
- Project -sg onto the tangent space of the manifold.
- Tangent space: vectors v with sum(v) = 0 and v[i] >= 0 where a[i] > 0.

So: v = -sg - mean(-sg) * 1 (to make sum = 0), then set v[i] = 0 where a[i] = 0.

Then: a_new = a + step * v, project to a >= 0, normalize.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix T where (T@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        T[k_start:k_end, j] = a[:k_end - k_start]
    return T


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    T = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def compute_subgradient(a, top_m=5):
    """
    Compute averaged subgradient of max_k conv[k](a) using top-m peaks.
    Returns (subgradient, peak_values).
    """
    n = len(a)
    conv = compute_conv(a)
    m = 2 * n - 1
    
    # Find top-m peaks
    sorted_idx = np.argsort(conv)[::-1][:top_m]
    weights = np.zeros(m)
    # Exponential weighting over top-m
    for rank, idx in enumerate(sorted_idx):
        weights[idx] = np.exp(-0.3 * rank)
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        weights = np.zeros(m)
        weights[sorted_idx[0]] = 1.0
    else:
        weights /= w_sum
    
    # Compute subgradient: sg[i] = sum_k weights[k] * d(conv[k])/d(a[i])
    # d(conv[k])/d(a[i]) = a[k-i] (if valid) + a[k-i] (if valid and k-i != i)
    sg = np.zeros(n)
    for k in range(m):
        if weights[k] < 1e-15:
            continue
        # For this k, the subgradient contribution to position i is:
        # a[k-i] if 0 <= k-i < n, plus another a[k-i] if k-i != i
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                if i == j:
                    sg[i] += weights[k] * a[j]
                else:
                    sg[i] += weights[k] * 2.0 * a[j]
    
    return sg, conv[sorted_idx[0]]

def pairwise_local_search(a, time_budget=60.0, top_m=3):

def resample_sequence(a, new_n):
    """Resample via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def compute_subgradient_fft(a, w):
    """
    Compute the weighted subgradient of max_k conv[k](a) using FFT.
    w[k] are the weights (normalized to sum to 1).
    Returns the subgradient vector of length n.
    """
    n = len(a)
    m = 2 * n - 1
    
    # crosscorr(w, a)[i] = conv(w, a[::-1])[i + n - 1]
    a_flip = a[::-1]
    size = 1
    while size < m + n:
        size <<= 1
    
    Wf = rfft(w, size)
    Af = rfft(a_flip, size)
