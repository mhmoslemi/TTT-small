Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse initialization (parent resampling, spectral methods, random)
2. Smooth Nesterov warm-up for quick convergence

4. Aggressive diversification to escape local optima
5. Multi-length exploration

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2 * len(a) * fast_conv(a).max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def build_dual_matrix(n, a):
    """Build the dual constraint matrix.
    Dual: min M * sum(y) s.t. for each i: sum_j a[j]*y[i+j] >= 1, y >= 0
    Matrix D has shape (n, 2n-1), D[i, i+j] = a[j] for j=0..n-1 (if i+j < 2n-1)
    """
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for j in range(n):
            k = i + j
            if k < 2 * n - 1:
                rows.append(i)
                cols.append(k)
                vals.append(a[j])
    return csr_matrix((vals, (rows, cols)), shape=(n, 2 * n - 1))


def solve_lp_dual(a, M):
    """Solve the dual LP and recover primal solution.
    Dual: min M * sum(y) s.t. D @ y >= 1, y >= 0
    where D[i, i+j] = a[j]
    
    Actually, let me use the primal since it's more straightforward for recovery.
    Primal: max sum(g) s.t. conv(a,g) <= M, g >= 0
    """
    n = len(a)
    # Build primal constraint matrix: A has shape (2n-1, n)
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(a[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def line_search(a, g, n, max_evals=20):
    """Find t in [0,1] minimizing max(conv((1-t)a+tg, (1-t)a+tg)) using ternary search."""
    lo, hi = 0.0, 1.0
    for _ in range(max_evals):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        u1 = (1 - m1) * a + m1 * g
        u2 = (1 - m2) * a + m2 * g
        M1 = compute_M(u1)
        M2 = compute_M(u2)
        if M1 < M2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2


def smooth_max_grad(a, tau):
    """Compute gradient of smooth approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad_i = 2 * sum_k p[k] * a[k-i]
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * p[k] * a[k - i]
    return grad


def run_nesterov(a_init, deadline, max_iters=3000, tau_start=5.0, tau_end=500.0,
                 lr_init=0.3, lr_end=0.001):
    """Nesterov-accelerated smooth max gradient descent."""
    a = normalize(a_init.astype(np.float64) + 1e-8)
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad = smooth_max_grad(y, tau)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
        M = compute_M(a)
        if M < best_M:
            best_M = M
            best_a = a.copy()
    return best_a


def anderson_step(history_x, history_Fx, m):
    """Anderson acceleration mixing."""
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T + 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 1.5)
    if alpha.sum() < 1e-10:
        alpha = ones / m
    else:
        alpha = alpha / alpha.sum()
    mixed = sum(alpha[i] * history_Fx[i] for i in range(m))
    return np.maximum(mixed, 0)


def run_lp_bcd(a_init, deadline, max_iters=500, use_line_search=True,
               anderson_window=4, no_imp_limit=40):
    """LP-based Block Coordinate Descent with line search and Anderson acceleration."""
    a = normalize(a_init.astype(np.float64) + 1e-8)
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_imp = 0
    hist_x = []
    hist_Fx = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = compute_M(a)
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > no_imp_limit:
                break
        
        g = solve_lp_dual(a, M)
        if g is None:
            break
        g = normalize(g)
        
        if use_line_search:
            t = line_search(a, g, n, max_evals=15)
        else:
            # Simple step size decay
            t = max(0.05, 0.4 / (1 + 0.1 * it))
        
        a_new = (1 - t) * a + t * g
        a_new = normalize(a_new)
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 3:
            a_and = anderson_step(hist_x, hist_Fx, len(hist_x))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
            M_and = compute_M(a_and)
            M_new = compute_M(a_new)
            if M_and <= M_new:
                a = a_and
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def make_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        # Resample parent
        if len(parent) != n:
            t_old = np.linspace(0, 1, len(parent))
            p = np.interp(t, t_old, parent)
        else:
            p = parent.copy()
        inits.append(np.maximum(p, 1e-6))
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 1e-6))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 1e-6))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Raised cosine variants
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        inits.append(np.maximum(0.3 + 0.7 * (t**alpha + (1-t)**alpha) / 2, 0.01))
    
    # Cosine bumps
    for f, phase in [(1, 0), (2, 0), (3, np.pi/2), (2, np.pi/4)]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2*np.pi*f*t + phase), 0.01))
    
    # Gaussian
    inits.append(np.exp(-4*(t-0.5)**2) + 0.1)
    inits.append(np.exp(-6*(t-0.3)**2) + np.exp(-6*(t-0.7)**2) + 0.05)
    
    # Random smooth
    k = max(3, n // 16)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.zeros(n)
    full[:half_n] = half
    full[half_n:] = half[-2::-1][:n-half_n]
    inits.append(full)
    
    return inits

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse initialization with spectral and random methods

4. Anderson acceleration

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Auto-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    return float(2 * n * fast_conv(a).max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def solve_lp(a, M):
    """Solve primal LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def line_search_fast(A, B, C, n_iter=25):
    """Ternary search for t minimizing max((1-t)^2*A + 2t(1-t)*B + t^2*C)."""
    lo, hi = 0.0, 1.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = (1-m1)*(1-m1) * A + 2.0*m1*(1-m1) * B + m1*m1 * C
        f2 = (1-m2)*(1-m2) * A + 2.0*m2*(1-m2) * B + m2*m2 * C
        if f1.max() < f2.max():
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def smooth_grad(a, tau):
    """Gradient of smooth max approximation."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(len(b)):
        if p[k] < 1e-12:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            grad[i] += 2.0 * p[k] * a[k - i]
    return grad


def run_nesterov(a_init, deadline, max_iters=3000):
    """Nesterov smooth max descent."""
    a = normalize(a_init + 1e-8)
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    y = a.copy()
    beta = 0.9
    tau_start = 5.0
    tau_end = 500.0
    lr_start = 0.3
    lr_end = 0.001
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_start * (lr_end / lr_start) ** frac
        grad = smooth_grad(y, tau)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
        M = compute_M(a)
        if M < best_M:
            best_M = M
            best_a = a.copy()
    return best_a


def anderson_mix(hist_x, hist_Fx):
    """Anderson acceleration."""
    m = len(hist_x)
    if m < 2:
        return hist_Fx[-1].copy()
    res = np.array([hist_x[i] - hist_Fx[i] for i in range(m)])
    G = res @ res.T + 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        alpha = np.linalg.solve(G, ones)
        s = alpha.sum()
        if abs(s) < 1e-15:
            alpha = ones / m
        else:
            alpha = alpha / s
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 1.5)
    s = alpha.sum()
    if s < 1e-10:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(hist_Fx[0]))
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    return np.maximum(mixed, 0)


def run_bcd(a_init, deadline, max_iters=500, anderson_win=5, no_imp_limit=50):
    """LP BCD with fast line search and Anderson acceleration."""
    a = normalize(a_init + 1e-8)
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_imp = 0
    hx = []
    hf = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        A = fast_conv(a)
        M = float(A.max())
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > no_imp_limit:
                break
        
        g = solve_lp(a, M)
        if g is None:
            break
        g = normalize(g)
        
        # Fast line search using precomputed convolutions
        B = cross_conv(a, g)
        C = fast_conv(g)
        t = line_search_fast(A, B, C)
        
        a_new = (1 - t) * a + t * g
        a_new = normalize(a_new)
        
        hx.append(a.copy())
        hf.append(a_new.copy())
        if len(hx) > anderson_win:
            hx.pop(0)
            hf.pop(0)
        
        if len(hx) >= 3:
            a_and = anderson_mix(hx, hf)
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
            if compute_M(a_and) <= compute_M(a_new):
                a = a_and
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample(parent, n)
        inits.append(np.maximum(p, 1e-6))
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 1e-6))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 1e-6))
    
    inits.append(np.ones(n))
    
    for alpha in [0.4, 0.7, 1.0, 1.5, 2.5]:
        inits.append(np.maximum(0.2 + 0.8 * (t**alpha + (1-t)**alpha) / 2, 0.01))
    
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.05)
    inits.append(np.exp(-6 * (t - 0.25)**2) + np.exp(-6 * (t - 0.75)**2) + 0.02)
    
    k = max(3, n // 16)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0.2, 1.0, n), kern, 'same'), 0.01))
    
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.zeros(n)
    full[:half_n] = half
    full[half_n:] = half[-2::-1][:n - half_n]
    inits.append(full)
    
    return inits
