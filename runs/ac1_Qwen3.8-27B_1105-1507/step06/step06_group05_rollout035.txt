"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating direction, multiplicative weight updates, 
spectral equalization, and coordinate descent with adaptive parameters.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    k_idx = np.arange(m)[:, None]
    i_idx = np.arange(n)[None, :]
    j_val = k_idx - i_idx
    mask = (j_val >= 0) & (j_val < n)
    A = np.where(mask, a[np.clip(j_val, 0, n - 1)], 0.0)
    return A


def solve_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. A@G <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def parabolic_line_search(f, x0, x1, x2, max_iter=20):
    """Parabolic (3-point quadratic) line search. f is the objective."""
    fx0, fx1, fx2 = f(x0), f(x1), f(x2)
    # Start with the best of the three
    if fx0 <= fx1 and fx0 <= fx2:
        best_x, best_f = x0, fx0
    elif fx1 <= fx0 and fx1 <= fx2:
        best_x, best_f = x1, fx1
    else:
        best_x, best_f = x2, fx2
    
    for _ in range(max_iter):
        # Fit quadratic through 3 points
        d1 = (fx0 - fx1) / (x0 - x1) if x0 != x1 else 0
        d2 = (fx0 - fx2) / (x0 - x2) if x0 != x2 else 0
        if abs(x1 - x2) < 1e-10:
            break
        denom = (x1 - x2) * (d1 + d2)
        if abs(denom) < 1e-15:
            break
        x_new = x0 - 0.5 * (x1 - x2) * (d1 - d2) / abs(denom)
        
        # Clamp to [min(x0,x1,x2), max(x0,x1,x2)]
        lo = min(x0, x1, x2)
        hi = max(x0, x1, x2)
        x_new = max(lo, min(hi, x_new))
        
        fx_new = f(x_new)
        if fx_new < best_f:
            best_x, best_f = x_new, fx_new
        
        # Replace the worst point
        if fx0 >= fx1 and fx0 >= fx2:
            x0, fx0 = x_new, fx_new
        elif fx1 >= fx0 and fx1 >= fx2:
            x1, fx1 = x_new, fx_new
        else:
            x2, fx2 = x_new, fx_new
        
        if max(x0, x1, x2) - min(x0, x1, x2) < 0.001:
            break
    
    return best_x


def lp_optimize(a, max_iters=50, time_budget=100.0):
    """LP alternating with parabolic line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        g = solve_lp(a, M, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break
        
        # Line search for blend parameter t
        def obj_at_t(t):
            a_try = normalize((1.0 - t) * a + t * g)
            return compute_obj(a_try)
        
        t_opt = parabolic_line_search(obj_at_t, 0.0, 0.3, 0.7, max_iter=15)
        
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Perturb to escape
            rng = np.random.default_rng((it * 7919 + 12345) % (2**31))
            sigma = 0.03 + 0.02 * (it % 5)
            a = normalize(best_a * np.exp(rng.normal(0, sigma, n)))
    
    return best_a


def multiplicative_weight_update(a, max_iters=300, time_budget=60.0):
    """Mirror descent with multiplicative weight updates."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.5
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of max conv at k_star
        j = k_star - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad = 2.0 * np.where(mask, a[np.clip(j, 0, n - 1)], 0.0)
        
        # Mirror descent step
        a_new = a * np.exp(-eta * grad)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Slightly increase eta as we converge
            eta = min(eta * 1.01, 2.0)
        else:
            eta *= 0.7
            if eta < 1e-6:
                break
    
    return best_a


def spectral_equalization(a, max_iters=80, time_budget=40.0):
    """POCS-style spectral equalization."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        
        # Adaptive compression
        alpha = 0.4 + 0.4 * (it / float(max_iters))
        P_new = (P / p_max) ** alpha * p_max
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.5 * a + 0.5 * a_new)
    
    return best_a


def coordinate_descent(a, time_budget=30.0):
    """Pairwise mass-transfer targeting convolution peaks."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find top contributors to the peak
        j = np.arange(n)
        k2 = k_star - j
        valid = (k2 >= 0) & (k2 < n)
        contrib = np.where(valid, a[j] * a[np.clip(k2, 0, n-1)], 0.0)
        
        top_idx = np.argsort(contrib)[::-1][:5]
        improved = False
        
        for idx in top_idx:
            j1 = int(idx)
            j2 = int(k_star - j1)
            if j1 == j2 or not valid[j1]:
                continue
            
            for _ in range(5):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue
                
                a_try = a.copy()
                delta = 0.03 * a_try[j1]
                a_try[j1] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved and rng.random() < 0.15:
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating direction with parabolic line search, multiplicative
weight (mirror descent) updates, spectral equalization, multi-n exploration,
adaptive M-relaxation for LP diversity, and coordinate descent polishing.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix A (m x n) where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    k_idx = np.arange(m)[:, None]
    i_idx = np.arange(n)[None, :]
    j_val = k_idx - i_idx
    mask = (j_val >= 0) & (j_val < n)
    A = np.where(mask, a[np.clip(j_val, 0, n - 1)], 0.0)
    return A


def solve_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def line_search_1d(f, lo, hi, n_coarse=8, n_fine=5):
    """Two-stage line search: coarse grid then refine."""
    xs = np.linspace(lo, hi, n_coarse)
    fs = np.array([f(x) for x in xs])
    best_idx = int(np.argmin(fs))
    best_x = xs[best_idx]
    best_f = fs[best_idx]

    if 0 < best_idx < n_coarse - 1:
        new_lo = xs[best_idx - 1]
        new_hi = xs[best_idx + 1]
        xs2 = np.linspace(new_lo, new_hi, n_fine)
        fs2 = np.array([f(x) for x in xs2])
        best_idx2 = int(np.argmin(fs2))
        if fs2[best_idx2] < best_f:
            best_x = xs2[best_idx2]
            best_f = fs2[best_idx2]

    return best_x


def lp_optimize(a, max_iters=50, time_budget=100.0, m_factors=None):
    """LP alternating with adaptive M and line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

            break

        conv = compute_conv(a)
        M_base = float(np.max(conv))

        # Cycle through different M factors for diversity
        factor = m_factors[it % len(m_factors)]
        M = M_base * factor

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

            break

            return compute_obj(a_try)

def multiplicative_weight_update(a, max_iters=300, time_budget=60.0):
    """Mirror descent with multiplicative weight updates targeting worst lag."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.5
    patience = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Use top-k lags for smoother gradient
        top_k = max(1, min(10, len(conv) // 20))
        top_indices = np.argsort(conv)[::-1][:top_k]
        weights = np.zeros(len(conv))
        for rank, idx in enumerate(top_indices):
            weights[idx] = 1.0 / (rank + 1)
        weights /= weights.sum()

        # Weighted subgradient
        grad = np.zeros(n)
        for idx in top_indices:
            k = int(idx)
            j = k - np.arange(n)
            mask = (j >= 0) & (j < n)
            grad += weights[idx] * 2.0 * np.where(mask, a[np.clip(j, 0, n - 1)], 0.0)

        # Mirror descent step with annealing
        eta_current = eta * (1.0 + 0.5 * np.cos(np.pi * it / max_iters))
        a_new = a * np.exp(-eta_current * grad)
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            eta *= 0.85
            if patience > 50 or eta < 1e-7:
                break

    return best_a


def spectral_equalization(a, max_iters=80, time_budget=40.0):
    """POCS-style spectral equalization with adaptive compression."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Adaptive alpha: start gentle, become more aggressive
        alpha = 0.3 + 0.5 * (it / float(max_iters))
        P_new = (P / p_max) ** alpha * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.4 * a + 0.6 * a_new)

    return best_a


def coordinate_descent(a, time_budget=30.0):
    """Pairwise mass-transfer targeting convolution peaks."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Target multiple peaks
        top_peaks = np.argsort(conv)[::-1][:3]

        improved = False
        for k_star in top_peaks:
            k_star = int(k_star)
            j = np.arange(n)
            k2 = k_star - j
            valid = (k2 >= 0) & (k2 < n)
            contrib = np.where(valid, a[j] * a[np.clip(k2, 0, n - 1)], 0.0)

            top_contrib = np.argsort(contrib)[::-1][:8]
            for idx in top_contrib:
                j1 = int(idx)
                j2 = int(k_star - j1)
                if j1 == j2 or not valid[j1]:
                    continue

                for _ in range(4):
                    j_target = int(rng.integers(0, n))
                    if j_target == j1 or j_target == j2:
                        continue

                    a_try = a.copy()
                    delta = 0.02 * a_try[j1]
                    a_try[j1] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)

                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break

        if not improved and rng.random() < 0.2:
            a = normalize(a * np.exp(rng.normal(0, 0.015, n)))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:

    if n == 1:

    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

                best_score = obj_spec
                best_seq = a_spec.copy()
                a = a_spec
                print(f"  Spectral: {best_score:.8f}")
            else:
                print(f"  Spectral: {obj_spec:.8f}")

        # Follow-up LP
        if time.monotonic() < deadline - 180:
            a_new = lp_optimize(a, max_iters=35, time_budget=80.0,
                                m_factors=[1.0, 1.0, 1.03, 0.97])
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Post-Phase2 LP: {best_score:.8f}")

    # === Phase 3: Multi-n exploration (10% of time) ===
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Multi-n exploration")
    print(f"{'=' * 60}")

    n_best = len(best_seq) if best_seq is not None else 700
    n_options = []
    for delta in range(-300, 500, 50):
        n_try = n_best + delta
        if 250 <= n_try <= 1600:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    # Sort by proximity to current best n (explore nearby first)
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if best_seq is not None:
            if n_try == n_best:
                a_try = normalize(best_seq.copy())
            else:
                a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try

        alloc = min(remaining * 0.10, 35.0)
        if n_try <= 900:
            a_try = lp_optimize(a_try, max_iters=15, time_budget=alloc * 0.5,
                                m_factors=[1.0, 1.0, 1.05])
        a_try = multiplicative_weight_update(a_try, max_iters=100, time_budget=alloc * 0.3)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 4: Random restarts (8% of time) ===
    print(f"\n{'=' * 60}")
    print(f"Phase 4: Random restarts")
    print(f"{'=' * 60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed + 200)

        for restart in range(8):
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.10, 35.0)

            # Diverse perturbation strategies
            r = restart % 4
            if r == 0:
                a_try = best_seq * np.exp(rng.normal(0, 0.08, n_cur))
            elif r == 1:
                freq = rng.uniform(0.3, 5.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
                a_try = best_seq * mod
            elif r == 2:
                a_rand = rng.dirichlet(np.ones(n_cur) * 0.3)
                mix = rng.uniform(0.25, 0.5)
                a_try = (1 - mix) * normalize(best_seq) + mix * a_rand
            else:
                # Block reversal
                a_try = best_seq.copy()
                block_len = int(rng.integers(n_cur // 5, n_cur // 2))
                start_idx = int(rng.integers(0, max(1, n_cur - block_len)))
                a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]

            a_try = normalize(a_try)

            if n_cur <= 900:
                a_try = lp_optimize(a_try, max_iters=12, time_budget=alloc * 0.5,
                                    m_factors=[1.0, 1.05, 0.95])
            a_try = multiplicative_weight_update(a_try, max_iters=80, time_budget=alloc * 0.35)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart}: {best_score:.8f} **")
            else:
                print(f"  Restart {restart}: {obj_try:.8f}")

    # === Phase 5: Final intensive polish (7% of time) ===
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Final polish")
    print(f"{'=' * 60}")

    if best_seq is not None:
        n_final = len(best_seq)
        a_final = normalize(best_seq.copy())

        for round_num in range(4):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.28, 35.0)

            # LP polish
            if n_final <= 900 and alloc > 15:
                a_pol = lp_optimize(a_final, max_iters=25, time_budget=alloc * 0.4,
                                    m_factors=[1.0, 1.0, 1.0])
            else:
                a_pol = a_final.copy()

            # MWU fine-tuning
            a_pol = multiplicative_weight_update(a_pol, max_iters=150, time_budget=alloc * 0.25)

            # Coordinate descent
            if alloc > 12:
                a_pol = coordinate_descent(a_pol, time_budget=min(12, alloc * 0.2))

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {round_num}: {best_score:.8f}")
            else:
                print(f"  Polish {round_num}: {obj_pol:.8f} (no improvement)")

    # Final validation and return
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'=' * 60}")

    return result
