import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient(a, tau=0.01):

    Compute soft subgradient of max_k conv(a,a)[k].
    tau: temperature for soft active set (0 = hard).
    Returns (gradient, max_conv_value).

    cmax = float(np.max(conv))

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Duchi et al. 2008
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def spectral_equalization_v2(a, max_iters=100, time_budget=50.0):

    Improved spectral equalization with adaptive compression.
    """

    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

            break
        
        # Compress the spectrum: P_new = P^alpha (normalized to same mean)
        P_norm = P / p_max

        # Renormalize to keep mean power the same
        P_new *= p_mean / float(np.mean(P_new))
        
        # Reconstruct

        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)

def lp_step(a, M, time_limit=20):

    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        ks = min(j + n, m)
        A[j:ks, j] = a[:ks - j]
    
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:

def resample_sequence(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated subgradient with soft active sets, multiplicative
weight updates, spectral equalization, LP polishing, and multi-n exploration.
Key differences from prior approach: avoids expensive LP solves in the main
loop, uses faster O(n^2) subgradient computations, and employs multiplicative
updates that naturally maintain non-negativity.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient(a, tau=0.01):
    """
    Compute soft subgradient of g(a) = max_k conv(a,a)[k].
    Returns (gradient of length n, max_conv_value).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    if tau > 1e-12:
        shifted = np.clip((conv - cmax) / tau, -500.0, 0.0)
        weights = np.exp(shifted)
        wsum = float(np.sum(weights))
        if wsum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= wsum
    else:
        active = np.where(conv >= cmax - 1e-12)[0]
        weights = np.zeros(len(conv))
        weights[active] = 1.0 / len(active)

    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * weights[m+j]
    # Computed via cross-correlation
    grad = 2.0 * np.correlate(weights, a, mode='full')[:n]
    return grad, cmax


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_subgradient(a, max_iters=10000, time_budget=100.0, tau_init=0.01,
                         step_init=0.1, decay_rate=0.01):
    """
    Nesterov-accelerated subgradient with soft active set.
    Minimizes max conv on the simplex.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = a.copy()
    tau = tau_init
    no_improve_count = 0

    for t in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov extrapolation
        if t > 0:
            beta = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * (t - 1) ** 2))
            v = a + ((t - 1) / beta) * (a - v)
        else:
            v = a.copy()

        # Compute subgradient at v
        grad, cmax = compute_subgradient(v, tau)

        # Center gradient for sum constraint
        grad_centered = grad - np.mean(grad)

        # Adaptive step size with periodic restart
        step_t = step_init / (1.0 + decay_rate * t)

        # Update
        a_new = v - step_t * grad_centered
        a_new = simplex_project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        a = a_new

        # Cool temperature
        tau *= 0.997

        # Periodic reheat to escape plateaus
        if no_improve_count > 500 and t > 200:
            tau = cmax / (3.0 * n)
            no_improve_count = 0

    return best_a


def multiplicative_update(a, max_iters=5000, time_budget=80.0, tau_init=0.005,
                          step_init=0.1):
    """
    Multiplicative weight update: a[i] *= exp(-eta * s[i]).
    Naturally maintains non-negativity.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_init
    step = step_init
    no_improve = 0

    for t in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, cmax = compute_subgradient(a, tau)
        grad_centered = grad - np.mean(grad)

        step_t = step_init / (1.0 + 0.02 * t)

        # Multiplicative update with clipping
        log_update = -step_t * grad_centered
        log_update = np.clip(log_update, -40.0, 40.0)
        a_new = a * np.exp(log_update)
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Simulated annealing acceptance
            delta = obj_new - best_obj
            accept_prob = np.exp(-delta / max(0.0005, 0.005 / (1.0 + t / 200.0)))
            if np.random.random() < accept_prob:
                a = a_new
            else:
                step *= 0.97

        tau *= 0.996

        if no_improve > 800 and t > 300:
            tau = cmax / (4.0 * n)
            step = step_init / 3.0
            no_improve = 0

    return best_a


def spectral_equalization_v2(a, max_iters=100, time_budget=50.0):
    """
    POCS-style spectral equalization with adaptive compression exponent.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    alpha = 0.6
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_mean = float(np.mean(P))
        p_max = float(np.max(P))

        if p_max < 1e-30:
            break

        # Compress spectrum toward flatness
        P_norm = P / p_max
        P_new = P_norm ** alpha * p_max
        P_new *= p_mean / (float(np.mean(P_new)) + 1e-30)

        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            alpha = max(0.2, alpha * 0.97)
        else:
            no_improve += 1
            a = 0.7 * a + 0.3 * a_new
            a = normalize(a)
            alpha = min(0.95, alpha + 0.03)

        if no_improve > 30:
            alpha = 0.9
            no_improve = 0

    return best_a


def lp_polish(a, max_iters=10, time_budget=60.0):
    """LP-based polishing: alternate with LP projection steps."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        m = 2 * n - 1

        # Build Toeplitz constraint matrix
        A_mat = np.zeros((m, n), dtype=np.float64)
        for j in range(n):
            k_end = min(j + n, m)
            A_mat[j:k_end, j] = a[:k_end - j]

        c_vec = -np.ones(n)
        b_ub = np.full(m, M)
        bounds = [(0.0, None)] * n

        remaining = time_budget - (time.monotonic() - start)
        try:
            res = linprog(c_vec, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': max(2, int(remaining * 0.4))})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g
        except Exception:
            break

        # Try multiple blend ratios
        best_blend_obj = best_obj
        best_blend_a = a.copy()
        for t_blend in [0.1, 0.2, 0.3, 0.5, 0.7]:
            a_try = normalize((1.0 - t_blend) * a + t_blend * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_blend_a = a_try

        if best_blend_obj < best_obj - 1e-12:
            best_obj = best_blend_obj
            best_a = best_blend_a.copy()
            a = best_blend_a
        else:
            break

    return best_a


def coordinate_local_search(a, time_budget=20.0):
    """
    Local coordinate search: try transferring mass from high-contribution
    positions to low-contribution positions.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            for _ in range(5):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue

                a_try = a.copy()
                for frac in [0.03, 0.06, 0.1]:
                    delta = frac * a_try[j1]
                    if delta < 1e-12:
                        break
                    a_try = a.copy()
                    a_try[j1] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved and rng.random() < 0.15:
            a = normalize(a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining Nesterov subgradient,
    multiplicative updates, spectral equalization, LP polish, coordinate search,
    multi-n exploration, and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: Nesterov subgradient (main workhorse) ===
    print(f"\nPhase 1: Nesterov subgradient (n={n_parent})")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        configs = [
            {'tau_init': 0.02, 'step_init': 0.2, 'decay': 0.01, 'budget': 70},
            {'tau_init': 0.01, 'step_init': 0.12, 'decay': 0.008, 'budget': 70},
            {'tau_init': 0.005, 'step_init': 0.08, 'decay': 0.006, 'budget': 70},
            {'tau_init': 0.002, 'step_init': 0.05, 'decay': 0.004, 'budget': 70},
            {'tau_init': 0.001, 'step_init': 0.03, 'decay': 0.003, 'budget': 60},
        ]
        for cfg in configs:
            if time.monotonic() > deadline - 250:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(cfg['budget'], remaining * 0.18)
            a_new = nesterov_subgradient(a, max_iters=20000, time_budget=alloc,
                                          tau_init=cfg['tau_init'],
                                          step_init=cfg['step_init'],
                                          decay_rate=cfg['decay'])
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  Nesterov tau={cfg['tau_init']:g}: {best_score:.8f} **")
            else:
                a = a_new
                print(f"  Nesterov tau={cfg['tau_init']:g}: {obj_new:.8f}")

    # === Phase 2: Multiplicative updates ===
    print(f"\nPhase 2: Multiplicative updates")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        configs = [
            {'tau_init': 0.01, 'step_init': 0.15, 'budget': 60},
            {'tau_init': 0.005, 'step_init': 0.1, 'budget': 60},
            {'tau_init': 0.002, 'step_init': 0.06, 'budget': 60},
            {'tau_init': 0.001, 'step_init': 0.04, 'budget': 50},
        ]
        for cfg in configs:
            if time.monotonic() > deadline - 200:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            alloc = min(cfg['budget'], remaining * 0.18)
            a_new = multiplicative_update(a, max_iters=15000, time_budget=alloc,
                                           tau_init=cfg['tau_init'],
                                           step_init=cfg['step_init'])
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  MWU tau={cfg['tau_init']:g}: {best_score:.8f} **")
            else:
                a = a_new
                print(f"  MWU tau={cfg['tau_init']:g}: {obj_new:.8f}")

    # === Phase 3: Spectral equalization ===
    print(f"\nPhase 3: Spectral equalization")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        for _ in range(3):
            if time.monotonic() > deadline - 120:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(40.0, remaining * 0.15)
            a_new = spectral_equalization_v2(a, max_iters=100, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  Spectral: {best_score:.8f} **")
            else:
                a = normalize(0.7 * a + 0.3 * a_new)
                print(f"  Spectral: {obj_new:.8f}")

    # === Phase 4: Follow-up Nesterov after spectral ===
    print(f"\nPhase 4: Follow-up optimization")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        if time.monotonic() < deadline - 100:
            alloc = min(50.0, (deadline - time.monotonic()) * 0.25)
            a = nesterov_subgradient(a, max_iters=10000, time_budget=alloc * 0.5,
                                      tau_init=0.003, step_init=0.04)
            a = multiplicative_update(a, max_iters=8000, time_budget=alloc * 0.5,
                                       tau_init=0.002, step_init=0.03)
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
            print(f"  Follow-up: {best_score:.8f}")

    # === Phase 5: LP polishing ===
    print(f"\nPhase 5: LP polishing")

    if best_seq is not None and len(best_seq) <= 1200:
        a = normalize(best_seq.copy())
        if time.monotonic() < deadline - 80:
            alloc = min(50.0, (deadline - time.monotonic()) * 0.25)
            a_new = lp_polish(a, max_iters=8, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  LP polish: {best_score:.8f} **")
            else:
                print(f"  LP polish: {obj_new:.8f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\nPhase 6: Multi-n exploration")

    n_best = len(best_seq) if best_seq is not None else n_parent
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best - 25,
        n_best + 25, n_best + 50, n_best + 100,
        n_best + 150, n_best + 250, n_best + 400
    ]))
    n_options = [n for n in n_options if 100 <= n <= 2500]

    for n_try in n_options:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if best_seq is not None:
            a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try

        alloc = min(30.0, remaining * 0.2)
        a_try = nesterov_subgradient(a_try, max_iters=5000, time_budget=alloc * 0.5,
                                      tau_init=0.005, step_init=0.08)
        a_try = multiplicative_update(a_try, max_iters=3000, time_budget=alloc * 0.5,
                                       tau_init=0.003, step_init=0.05)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 7: Coordinate local search ===
    print(f"\nPhase 7: Coordinate local search")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        if time.monotonic() < deadline - 20:
            alloc = min(20.0, (deadline - time.monotonic()) * 0.2)
            a = coordinate_local_search(a, time_budget=alloc)
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
            print(f"  Coord: {best_score:.8f}")

    # === Phase 8: Random restarts ===
    print(f"\nPhase 8: Random restarts")

    if best_seq is not None and time.monotonic() < deadline - 20:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed)

        for restart in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(25.0, remaining * 0.3)

            r = rng.random()
            if r < 0.25:
                a_try = best_seq * np.exp(rng.normal(0, 0.06, n_cur))
            elif r < 0.5:
                freq = rng.uniform(1.0, 6.0)
                ph = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.12 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + ph)
                a_try = best_seq * mod
            elif r < 0.75:
                a_rand = rng.dirichlet(np.ones(n_cur) * 0.4)
                mix = rng.uniform(0.15, 0.35)
                a_try = (1 - mix) * normalize(best_seq) + mix * a_rand
            else:
                # Block rearrangement
                a_try = best_seq.copy()
                blk = int(rng.integers(n_cur // 5, n_cur // 3))
                st = int(rng.integers(0, n_cur - blk))
                tmp = a_try[st:st + blk].copy()
                a_try[st:st + blk] = tmp[::-1]

            a_try = normalize(a_try)
            a_try = nesterov_subgradient(a_try, max_iters=5000, time_budget=alloc * 0.5,
                                          tau_init=0.005, step_init=0.06)
            a_try = multiplicative_update(a_try, max_iters=3000, time_budget=alloc * 0.5,
                                           tau_init=0.003, step_init=0.04)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart}: {best_score:.8f} **")
            else:
                print(f"  Restart {restart}: {obj_try:.8f}")

    # === Phase 9: Final intensive polish ===
    print(f"\nPhase 9: Final polish")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        for rnd in range(3):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            alloc = min(20.0, remaining * 0.3)

            a = nesterov_subgradient(a, max_iters=5000, time_budget=alloc * 0.35,
                                      tau_init=0.001, step_init=0.02)
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()

            a = multiplicative_update(a, max_iters=4000, time_budget=alloc * 0.35,
                                       tau_init=0.001, step_init=0.015)
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()

            a = spectral_equalization_v2(a, max_iters=30, time_budget=alloc * 0.3)
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()

            print(f"  Polish {rnd}: {best_score:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")

    return result
