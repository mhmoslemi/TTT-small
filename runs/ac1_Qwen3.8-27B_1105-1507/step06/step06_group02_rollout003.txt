"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast gradient via np.correlate (O(n log n) per step)
2. Multi-scale smooth optimization (tau from 1 to 50000) with Nesterov momentum
3. Precomputed-structure LP mirror descent for final polish
4. Multi-length exploration with adaptive time allocation
5. Asymmetric perturbations to escape symmetric local minima
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad_fast(a, tau):
    """Fast gradient of smooth max using np.correlate."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    # Numerical stability: subtract max of shifted (which is 0)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    # grad[i] = 2 * sum_j p[i+j] * a[j] = 2 * correlate(p, a, 'valid')[i]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, b, b_max


def exact_subgrad(a):
    """Exact subgradient at the max elements."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    S = np.where(b >= b_max - 1e-14)[0]
    p = np.zeros(2 * n - 1)
    p[S] = 1.0 / len(S)
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, b, b_max


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s


def run_nesterov_multiscale(a, deadline, tau_list=None, max_iters_per_tau=3000):
    """Multi-scale Nesterov optimization with adaptive learning rates."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = project_simplex(a)
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    if tau_list is None:
        tau_list = [1, 3, 10, 30, 100, 300, 1000, 3000, 10000, 30000, 50000]

    for tau in tau_list:
        if time.monotonic() > deadline:
            break
        # Learning rate schedule: start moderate, decay
        lr_init = min(0.5, 2.0 / np.sqrt(n))
        lr_end = lr_init * 0.01
        y = a.copy()
        beta = 0.9
        for it in range(1, max_iters_per_tau + 1):
            if time.monotonic() > deadline:
                break
            frac = min(1.0, it / max_iters_per_tau)
            lr = lr_init * (lr_end / lr_init) ** frac
            # Nesterov lookahead
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
            y = a + ((t_k - 1.0) / (t_k + 1.0)) * (y - a)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-10:
                y = a.copy()
            else:
                y = y / sy

            grad, b, b_max = smooth_grad_fast(y, tau)
            if b_max < best_M * (1 - 1e-12):
                best_M = b_max
                best_a = a.copy()

            # Gradient step with momentum
            a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
            s = a_new.sum()
            if s < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s
            a = a_new

            # Early stopping if converged
            if it > 100 and b_max < best_M * (1 + 1e-12):
                break

    return best_a


def precompute_lp_structure(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


class LPSolver:
    """LP solver with precomputed structure for repeated solves."""

    def __init__(self, n):
        self.n = n
        self.rows, self.cols = precompute_lp_structure(n)
        self.c = -np.ones(n)
        self.bounds = [(0, None)] * n

    def solve(self, a, M):
        """Solve max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
        vals = np.empty(len(self.rows), dtype=np.float64)
        r = self.rows
        c = self.cols
        vals[c] = a[r - c]  # a[k-j] where row=k, col=j
        A = csr_matrix((vals, (r, c)), shape=(2 * self.n - 1, self.n))
        b_ub = np.full(2 * self.n - 1, M)
        try:
            result = linprog(self.c, A_ub=A, b_ub=b_ub, bounds=self.bounds,
                             method='highs',
                             options={'presolve': True, 'time_limit': 10})
            if result.success and result.x is not None and result.x.sum() > 1e-15:
                return np.maximum(result.x, 0)
        except Exception:
            pass
        return None


def run_lp_mirror(a, deadline, max_iters=200, t_base=0.3):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = project_simplex(a)
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    solver = LPSolver(n)
    no_improve = 0
    t = t_base

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = solver.solve(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Backtracking line search
        improved = False
        for trial_t in [t, t * 0.5, t * 0.25, t * 0.1, 0.05, 0.01]:
            a_t = np.maximum((1.0 - trial_t) * a + trial_t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = fast_conv(a_t).max()
            if M_t < best_M * (1 - 1e-14):
                a = a_t
                best_M = M_t
                best_a = a.copy()
                t = trial_t
                improved = True
                break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            t *= 0.5
            if t < 0.003:
                break
            if no_improve > 40:
                break

    return best_a


def run_subgrad_descent(a, deadline, max_iters=5000, lr_init=0.1):
    """Subgradient descent with Nesterov acceleration (exact subgradient)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = project_simplex(a)
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        lr = lr_init / (1.0 + 0.01 * it)

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = a + ((t_k - 1.0) / (t_k + 1.0)) * (y - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y = y / sy

        grad, b, b_max = exact_subgrad(y)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent):
    """Generate diverse initializations including asymmetric ones."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        # Asymmetric perturbations
        for amp in [0.01, 0.03, 0.05, 0.1]:
            inits.append(np.maximum(p + amp * rng.normal(0, 1, n), 0.001))
        # Linear tilt (asymmetry)
        for slope in [0.1, 0.2, -0.1, -0.2]:
            tilt = 1.0 + slope * (t - 0.5)
            inits.append(np.maximum(p * tilt, 0.001))
        # Quadratic asymmetry
        inits.append(np.maximum(p * (1.0 + 0.15 * (t - 0.3)), 0.001))
        inits.append(np.maximum(p * (1.0 + 0.15 * (0.7 - t)), 0.001))

    # Standard inits
    inits.append(np.ones(n))
    for alpha in [0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t**alpha + (1 - t)**alpha) / 2, 0.01))
    for f in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t + rng.uniform(0, 2*np.pi)), 0.01))
    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        inits.append(np.exp(-5 * (t - center)**2) + 0.05)
    # Double bump
    inits.append(np.exp(-8 * (t - 0.3)**2) + np.exp(-8 * (t - 0.7)**2) + 0.03)
    # Random smooth
    k = max(3, n // 15)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # Power law
    inits.append(np.maximum(t**0.3 + (1-t)**0.3 + 0.1, 0.01))
    # Sawtooth (asymmetric)
    saw = np.abs(t * 4 - np.floor(t * 4 + 0.5))
    inits.append(np.maximum(0.3 + 0.7 * saw, 0.01))

    return inits
