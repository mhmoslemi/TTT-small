"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

2. LP best-response with tolerance-based active sets for polishing

Key improvements: direct O(n) coordinate updates (no FFT per step),

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J_raw(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2. a should be non-negative."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def cd_one_sweep(a, b, n, order):
    """One GS sweep of coordinate descent. Updates a and b in place.
    
    Returns (improved_count, reduction_in_max).
    """
    improved = 0
    max_before = b.max()
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for i in order:
        old = a[i]
        if old < 0:
            old = 0.0

        # Affected k range: k-i in [0, n-1] => k in [i, i+n-1], clipped to [0, 2n-2]
        k_start = i
        k_end = min(2 * n - 2, i + n - 1)

        # Base max from unaffected bins
        base_max = 0.0
        if k_start > 0:
            base_max = max(base_max, np.max(b[:k_start]))
        if k_end < 2 * n - 2:
            base_max = max(base_max, np.max(b[k_end + 1:]))

        # Quadratic term (k = 2i)
        has_quad = (2 * i <= 2 * n - 2)
        quad_c = b[2 * i] - old * old if has_quad else 0.0

        # Linear terms (k != 2i in affected range)
        ks = np.arange(k_start, k_end + 1)
        mask = ks != 2 * i
        ks_lin = ks[mask]

        if len(ks_lin) > 0:
            c_lin = b[ks_lin] - 2.0 * old * a[ks_lin - i]
            s_lin = 2.0 * a[ks_lin - i]
        else:
            c_lin = np.array([])
            s_lin = np.array([])

        # Golden section search for x that minimizes f(x) = max(base_max, max(c_lin+s_lin*x), quad_c+x^2)
        # Bounds: x in [0, hi]
        hi_bound = max(2.0 * old + 0.5, np.sqrt(b.max() + 1.0))
        lo = 0.0
        hi = hi_bound

        # Expand hi if needed
        for _ in range(5):
            if hi < 50.0:
                f_hi = max(base_max, quad_c + hi * hi)
                if len(c_lin) > 0:
                    f_hi = max(f_hi, np.max(c_lin + s_lin * hi))
                f_mid = max(base_max, quad_c + (hi * 0.5) ** 2)
                if len(c_lin) > 0:
                    f_mid = max(f_mid, np.max(c_lin + s_lin * hi * 0.5))
                if f_hi > f_mid:
                    hi *= 2.0
                else:
                    break
            else:
                break

        # Golden section
        a_pt = hi - gr * (hi - lo)
        b_pt = lo + gr * (hi - lo)

        # Evaluate f at a_pt
        fa = base_max
        if has_quad:
            fa = max(fa, quad_c + a_pt * a_pt)
        if len(c_lin) > 0:
            fa = max(fa, np.max(c_lin + s_lin * a_pt))

        fb = base_max
        if has_quad:
            fb = max(fb, quad_c + b_pt * b_pt)
        if len(c_lin) > 0:
            fb = max(fb, np.max(c_lin + s_lin * b_pt))

        for _ in range(45):
            if fa < fb:
                hi = b_pt
                b_pt = a_pt
                fb = fa
                a_pt = hi - gr * (hi - lo)
                fa = base_max
                if has_quad:
                    fa = max(fa, quad_c + a_pt * a_pt)
                if len(c_lin) > 0:
                    fa = max(fa, np.max(c_lin + s_lin * a_pt))
            else:
                lo = a_pt
                a_pt = b_pt
                fa = fb
                b_pt = lo + gr * (hi - lo)
                fb = base_max
                if has_quad:
                    fb = max(fb, quad_c + b_pt * b_pt)
                if len(c_lin) > 0:
                    fb = max(fb, np.max(c_lin + s_lin * b_pt))

        x_opt = (lo + hi) * 0.5
        # Check boundary x=0
        f0 = base_max
        if has_quad:
            f0 = max(f0, quad_c)  # quad_c + 0 = b[2i] - old^2
        if len(c_lin) > 0:
            f0 = max(f0, np.max(c_lin))
        if f0 <= fa + 1e-14:
            x_opt = 0.0

        x_opt = max(0.0, x_opt)

        if abs(x_opt - old) > 1e-12:
            # Update b
            delta = x_opt - old
            if len(ks_lin) > 0:
                b[ks_lin] += 2.0 * delta * a[ks_lin - i]
            if has_quad:
                b[2 * i] += x_opt * x_opt - old * old
            a[i] = x_opt
            improved += 1

    max_after = b.max()
    return improved, max_before - max_after


def run_cd(a_init, max_sweeps, deadline, seed=0):

    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fast_conv(a)
    best_a = a.copy()
    best_M = b.max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)

        imp, red = cd_one_sweep(a, b, n, order)

        # Re-normalize
        s = a.sum()
        if s < 1e-10:
            break
        if abs(s - 1.0) > 1e-6:
            a = a / s
            b = fast_conv(a)  # recompute after rescale

        M = b.max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if imp == 0:
            no_improve += 5
        if no_improve > 30:
            break

    return best_a, best_M

def lp_best_response(a, M, tol=0.01):
    """Solve LP best response using tolerance-based active set.
    
    Active set: all k where (a*a)[k] >= M - tol*M
    Returns best response g (normalized to sum=1) or None.
    """
    n = len(a)
    b = fast_conv(a)
    active_mask = b >= M - tol * M
    active_indices = np.where(active_mask)[0]

    if len(active_indices) == 0:
        return None
    if len(active_indices) > 2 * n:  # safety cap
        # Fall back to top-k
        top_k = np.argsort(b)[::-1][:2 * n]
        active_indices = top_k

    # Build constraint matrix: (a*g)[k] = sum_j a[k-j] * g[j] for k in active
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(active_indices):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            m = k - j
            if 0 <= m < n and a[m] > 1e-15:
                rows.append(row_idx)
                cols.append(j)
                vals.append(a[m])

    if len(rows) == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_iteration(a, M, max_iters, deadline, tol=0.01):
    """Run LP mirror descent iterations. Returns best a (sum=1)."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_M = M
    no_improve = 0

        if time.monotonic() > deadline:
            break

        g = lp_best_response(a, best_M, tol=tol)
        if g is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue

        improved = False

        if improved:
            no_improve = 0
            tol = max(0.001, tol * 0.8)  # tighten tolerance
        else:
            no_improve += 1
            tol = min(0.1, tol * 1.5)  # loosen
            if no_improve > 10:
                break

    return best_a, best_M


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n, each normalized to sum=1."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p2 = np.maximum(p + rng.normal(0, 0.05, n), 1e-4)
        inits.append(p2 / p2.sum())
        p3 = np.maximum(p + rng.normal(0, 0.15, n), 1e-4)
        inits.append(p3 / p3.sum())

    inits.append(np.ones(n) / n)

    # Parabolic (concave)
    poly = (t * (1 - t)) * 4 + 0.1
    inits.append(poly / poly.sum())

    # U-shaped (convex)
    poly2 = ((2 * t - 1) ** 2) + 0.1
    inits.append(poly2 / poly2.sum())

    # Cosine windows
    for freq in [1, 2, 3]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())

    # Gaussian
    gauss = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # Spectral init: flatten circular autocorrelation
    a_spec = np.ones(n) / n
    for _ in range(30):
        N = 1
        while N < 2 * n:
            N <<= 1
        A = np.fft.fft(a_spec, N)
        ac = np.fft.ifft(A * A).real[:2 * n - 1]
        # Reduce peaks by adjusting a_spec
        M_s = ac.max()
        k_star = int(np.argmax(ac))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
            elif n <= j < 2 * n:
                corr[i] = a_spec[j - n]
        a_spec = a_spec - 0.15 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())

    # Bimodal
    bim = np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.05
    inits.append(bim / bim.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def refine_sequence(a0, n, deadline, seed=0):
    """Full refinement pipeline: CD -> LP -> CD -> LP -> CD."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    t_start = time.monotonic()
    total_time = max(1.0, deadline - t_start)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    # Stage 1: CD (40% of time)
    t1 = t_start + 0.4 * total_time
    if time.monotonic() < t1:
        a_cd, M_cd = run_cd(a, 20000, t1, seed=seed)
        if M_cd < best_M:
            best_M = M_cd
            best_a = a_cd.copy()

    # Stage 2: LP (30% of time)
    t2 = t_start + 0.7 * total_time
    if time.monotonic() < t2 and best_M > 0:
        a_lp, M_lp = run_lp_iteration(best_a, best_M, 50, t2, tol=0.02)
        if M_lp < best_M:
            best_M = M_lp
            best_a = a_lp.copy()

    # Stage 3: CD again (20% of time)
    t3 = t_start + 0.9 * total_time
    if time.monotonic() < t3:
        a_cd2, M_cd2 = run_cd(best_a, 20000, t3, seed=seed + 1000)
        if M_cd2 < best_M:
            best_M = M_cd2
            best_a = a_cd2.copy()

    # Stage 4: Final LP (10% of time)
    if time.monotonic() < deadline and best_M > 0:
        a_lp2, M_lp2 = run_lp_iteration(best_a, best_M, 20, deadline, tol=0.005)
        if M_lp2 < best_M:
            best_M = M_lp2
            best_a = a_lp2.copy()
        # Quick CD after LP
        if time.monotonic() < deadline:
            a_final, M_final = run_cd(best_a, 5000, deadline, seed=seed + 2000)
            if M_final < best_M:
                best_M = M_final
                best_a = a_final.copy()

    return best_a, best_M


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: diverse inits x multiple lengths, each refined with
    CD+LP pipeline, then global best gets extra polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size

    # Choose lengths to try
    lengths = []
    if n_parent > 0:
        lengths.append(n_parent)
    for nl in [384, 512, 640, 768, 896, 1024, 1280]:
        if nl not in lengths and 256 <= nl <= 1536:
            lengths.append(nl)
    # Sort by priority
    lengths.sort(key=lambda x: (abs(x - n_parent)))
    lengths = lengths[:5]  # top 5 lengths

    print(f"\n{'='*60}", flush=True)
    print(f"Lengths to try: {lengths}", flush=True)
    print(f"{'='*60}", flush=True)

    # Reserve time for final polish
    reserve_time = 200
    search_deadline = deadline - reserve_time

    # Phase 1: Multi-length, multi-init search
    n_lengths = len(lengths)
    time_per_length = (search_deadline - time.monotonic()) / n_lengths
    time_per_length = max(time_per_length, 30)

    for li, n in enumerate(lengths):
        if time.monotonic() > search_deadline - 30:
            break
        t0 = time.monotonic()
        remaining = search_deadline - time.monotonic()
        if remaining < 30:
            break

        time_alloc = min(time_per_length, remaining)
        print(f"\n  Length n={n}, alloc={time_alloc:.0f}s", flush=True)

        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)

        # Time per init
        n_inits = len(inits)
        time_per_init = (time_alloc - 5) / n_inits
        time_per_init = max(time_per_init, 10)

        best_for_n = None
        best_M_for_n = float('inf')

        for ii, a0 in enumerate(inits):
            if time.monotonic() > search_deadline - 30:
                break
            sub_deadline = min(t0 + time_alloc, time.monotonic() + time_per_init)
            if time.monotonic() >= sub_deadline:
                break

            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            a_opt, M_opt = refine_sequence(a0, n, sub_deadline, seed=seed + li * 100 + ii)

            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J_raw(best_for_n)
            print(f"  n={n}: best M={best_M_for_n:.10f} J={J_val:.6f} "
                  f"(elapsed={time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("After Phase 1 (multi-length search)")

    # Phase 2: Intensive polish on global best
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 2: Intensive polish on best (n={tracker.best.size})", flush=True)
        print(f"{'='*60}", flush=True)

        n_best = tracker.best.size
        a_polish = tracker.best.copy()
        a_polish = a_polish / a_polish.sum()

        # Long CD
        t_cd = min(deadline - 60, time.monotonic() + 120)
        a_cd, M_cd = run_cd_jitter(a_polish, 50000, t_cd, seed=99999, n_jitters=20, jitter_size=20)
        tracker.update(a_cd)

        # LP polish
        if time.monotonic() < deadline - 30:
            M_cur = fast_conv(tracker.best).max() if tracker.best.sum() > 0 else 0
            if M_cur > 0:
                a_lp, M_lp = run_lp_iteration(tracker.best, M_cur, 30,
                                              min(deadline - 20, time.monotonic() + 60), tol=0.01)
                tracker.update(a_lp)

        # Final CD
        if time.monotonic() < deadline - 10:
            a_final, M_final = run_cd(tracker.best, 20000, deadline - 5, seed=77777)
            tracker.update(a_final)

    tracker.report("After Phase 2 (polish)")

    # Phase 3: Try perturbation-based escape
    if tracker.best is not None and time.monotonic() < deadline - 20:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 3: Perturbation escape", flush=True)
        print(f"{'='*60}", flush=True)

        n_esc = tracker.best.size
        base = tracker.best.copy() / tracker.best.sum()
        t_esc = deadline - 15

        for attempt in range(5):
            if time.monotonic() > t_esc - 20:
                break
            # Random perturbation
            pert = base + rng.normal(0, 0.08, n_esc)
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            sub_dl = min(t_esc, time.monotonic() + 40)
            a_esc, M_esc = refine_sequence(pert, n_esc, sub_dl, seed=seed + 5000 + attempt)
            tracker.update(a_esc)

    tracker.report("After Phase 3 (escape)")

    # Final fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J_raw(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
