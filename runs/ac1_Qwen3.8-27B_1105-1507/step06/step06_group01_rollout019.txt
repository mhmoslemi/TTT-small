"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frank-Wolfe (conditional gradient) for coarse optimization - O(n) per iter
2. Mirror descent (multiplicative weights) for medium refinement
3. LP best-response with adaptive active sets and tight M for fine polish
4. Coordinate descent with exact 1D minimization
5. Multi-length, multi-init search with perturbation-based restarts
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = np.max(np.where(cond)[0])
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

    3. Update a <- (1-t)*a + t*v

    n = len(a)

    best_M = fast_conv(a).max()

        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))

        # The valid range for k_star - i is [0, n-1], so i is in [k_star-(n-1), k_star] ∩ [0, n-1]

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Frank-Wolfe coarse optimization (fast, O(n) per step)
2. LP best-response with cyclical active sets and adaptive M
3. Mirror descent (multiplicative weights) for medium refinement
4. Exact 1D coordinate descent with golden section
5. Multi-start perturbation-based restarts to escape local optima
6. Multi-length exploration (512-1280)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = np.max(np.where(cond)[0])
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def fw_optimize(a_init, max_iters, deadline):
    """Frank-Wolfe with adaptive step on simplex."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))

        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        if i_lo <= i_hi:
            j_lo = k_star - i_hi
            j_hi = k_star - i_lo
            j_min = j_lo + np.argmin(a[j_lo:j_hi + 1])
            i_best = k_star - j_min
        else:
            i_best = np.argmin(a)

        t = 2.0 / (it + 2)
        v = np.zeros(n)
        v[i_best] = 1.0
        a_new = (1.0 - t) * a + t * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        a = a_new

        if it % 200 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    return best_a


def mirror_descent(a_init, max_iters, deadline, step_init=0.15):
    """Mirror descent with KL geometry."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))

        grad = np.zeros(n)
        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        for i in range(i_lo, i_hi + 1):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        grad_tan = grad - grad.mean()
        step = step_init / (1.0 + 0.01 * it)
        log_w = -step * grad_tan
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new

        if M < best_M:
            best_M = M
            best_a = a.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    return best_a


def build_lp_matrix(a, active_indices):
    """Build sparse LP constraint matrix for given active lag indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response(a, M, active_indices, lp_time_limit=10.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a, active_indices)
    if A.nnz == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': lp_time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def get_cyclical_active_set(b, n_active, offset=0):
    """Get a cyclical active set: top values with rotation."""
    conv_len = len(b)
    sorted_idx = np.argsort(b)[::-1]
    # Take top n_active but rotate by offset for diversity
    pool = sorted_idx[:min(2 * n_active, conv_len)]
    if len(pool) <= n_active:
        return pool.tolist()
    # Cyclical: take a window of size n_active starting at offset
    start = offset % len(pool)
    selected = []
    for i in range(n_active):
        selected.append(pool[(start + i) % len(pool)])
    return selected


def run_lp_optimize(a_init, max_iters, deadline, n_active_start=60, n_active_end=15,
                    m_tighten=0.0, lp_time=8.0):
    """LP mirror descent with cyclical active sets and adaptive M."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    offset = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive active set size: start large, shrink over time
        frac = min(1.0, it / max(1, max_iters))
        n_active_cur = int(n_active_start + (n_active_end - n_active_start) * frac)
        n_active_cur = max(10, n_active_cur)

        b = fast_conv(a)
        M_cur = b.max()

        # Adaptive M: slightly tighten with iteration
        M_lp = M_cur * (1.0 + m_tighten * (1.0 - frac))
        # Ensure M_lp >= sum(a)*sum(a)/(2n-1) (the theoretical minimum)
        M_floor = 1.0 / (2 * n - 1)
        M_lp = max(M_lp, M_floor * 1.001)

        active = get_cyclical_active_set(b, n_active_cur, offset)
        offset += 1

        g = lp_best_response(a, M_lp, active, lp_time)
        if g is None:
            no_improve += 2
            if no_improve > 20:
                break
            continue

        # Line search: geometric
        improved = False
        base_t = 0.35
        for t in [base_t, base_t * 0.6, base_t * 0.3, base_t * 0.15, 0.03, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    return best_a

def coord_descent_1d_full(a, b, i, n):
    """Find optimal a[i] given current b and a."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = np.zeros(k_hi - k_lo + 1)
    s_arr = np.zeros(k_hi - k_lo + 1)
    has_quad = False
    quad_pos = -1

    for idx, k in enumerate(range(k_lo, k_hi + 1)):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            s_arr[idx] = 0.0
            has_quad = True
            quad_pos = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                s_arr[idx] = 2.0 * a[j]
            else:
                c_vals[idx] = b[k]
                s_arr[idx] = 0.0

    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    def obj_at(x):
        vals = c_vals + s_arr * x
        m = base_max
        vm = vals.max()
        if vm > m:
            m = vm
        if has_quad:
            qv = c_vals[quad_pos] + x * x
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(3.0 * x_cur, np.sqrt(max(b.max(), 0.0)) * 1.5, 0.1)
    for _ in range(8):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 50.0:
            hi *= 2.0
        else:
            break

    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(30):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coord_descent(a_init, max_sweeps, deadline, seed=0):
    """Randomized coordinate descent."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = np.arange(n)
        rng.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False

        for i in order:
            x_opt = coord_descent_1d_full(a, b, i, n)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                new = x_opt
                a[i] = new
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += new * new - old * old
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (new - old)
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_improved:
            no_improve += 3
        if no_improve > 15:
            break

    return best_a


def resample_seq(src, dest_n):
    """Linear interpolation resampling."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.02, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-4)
            inits.append(p_n / p_n.sum())

    inits.append(np.ones(n) / n)

    # U-shaped (high at edges)
    u_shape = 0.1 + ((2 * t - 1) ** 2)
    inits.append(u_shape / u_shape.sum())

    # Inverted U (peak in middle)
    inv_u = 0.1 + (1.0 - (2 * t - 1) ** 2)
    inits.append(inv_u / inv_u.sum())

    # Cosine windows
    for freq in [1, 2, 3]:
        cw = 0.3 + 0.7 * np.abs(np.cos(np.pi * freq * t))
        inits.append(cw / cw.sum())

    # Gaussian
    g = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(g / g.sum())

    # Two-peak
    two_peak = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
    inits.append(two_peak / two_peak.sum())

    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits

        if M_md < best_M:
            best_M = M_md
            best = a_md.copy()

    # Stage 3: LP (50% time)
    t3 = t_start + 0.80 * phase_duration
    if time.monotonic() < t3:
        a_lp = run_lp_optimize(best, 500, t3, n_active_start=50, n_active_end=20,
                               m_tighten=0.0, lp_time=6.0)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()

    # Stage 4: CD (20% time)
    if time.monotonic() < deadline:
        max_sw = max(3, int((deadline - time.monotonic()) / max(0.3, n / 80.0)))
        a_cd = run_coord_descent(best, max_sw, deadline, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()

    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: diverse inits -> FW -> Mirror -> LP -> CD -> perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size

    # Determine lengths to try
    lengths = [n_parent]
    for delta in [-128, -64, 64, 128, 256, 512]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl not in lengths:
            lengths.append(nl)
    if 768 not in lengths:
        lengths.append(768)
    if 1024 not in lengths:
        lengths.append(1024)

    # Sort: parent first, then by closeness
    lengths.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    lengths = lengths[:5]  # Limit to 5 lengths

    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-length search on {lengths}", flush=True)
    print(f"{'='*60}", flush=True)

    # Phase 1: Multi-length, multi-init search
    time_per_length = (deadline - time.monotonic() - 300) / len(lengths)
    time_per_length = max(time_per_length, 40)

    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - 300:
            break
        t0 = time.monotonic()
        remaining = deadline - 300 - time.monotonic()
        if remaining < 20:
            break
        time_alloc = min(time_per_length, remaining / max(1, len(lengths) - li))
        print(f"\n  n={n}, alloc={time_alloc:.0f}s", flush=True)

        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_M_for_n = float('inf')

        # Limit number of inits based on time
        n_inits = min(len(inits), max(2, int(time_alloc / 15)))

        for ii in range(n_inits):
            if time.monotonic() > deadline - 300:
                break
            sub_dl = min(t0 + time_alloc, deadline - 300)
            if time.monotonic() >= sub_dl:
                break
            a0 = np.maximum(inits[ii], 1e-6)
            a0 = a0 / a0.sum()
            a_opt = run_optimization_pipeline(a0, n, sub_dl, seed=seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best J={J_val:.6f} (t={time.monotonic()-t0:.0f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("After Phase 1")

    # Phase 2: Intensive LP polish on best
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Intensive LP polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 60:
        t_lp = min(deadline - 30, time.monotonic() + 200)
        a_lp = run_lp_optimize(tracker.best, 300, t_lp, n_active_start=40, n_active_end=15,
                               m_tighten=0.0, lp_time=5.0)
        tracker.update(a_lp)

        if time.monotonic() < deadline - 30:
            a_lp2 = run_lp_optimize(a_lp, 150, min(deadline - 20, time.monotonic() + 80),
                                    n_active_start=25, n_active_end=10,
                                    m_tighten=0.0, lp_time=4.0)
            tracker.update(a_lp2)

    tracker.report("After Phase 2")

    # Phase 3: Perturbation + re-optimize
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: Perturbation restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_best = tracker.best.size
        n_perturb = max(3, int((deadline - 30 - time.monotonic()) / 40))
        for pi in range(n_perturb):
            if time.monotonic() > deadline - 30:
                break
            sigma = 0.01 * (1 + pi * 0.5)
            a_pert = tracker.best + rng.normal(0, sigma, n_best)
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert = a_pert / a_pert.sum()
            sub_dl = min(deadline - 20, time.monotonic() + 35)
            a_opt = run_optimization_pipeline(a_pert, n_best, sub_dl, seed=seed + 5000 + pi)
            tracker.update(a_opt)

    tracker.report("After Phase 3")

    # Phase 4: Coordinate descent final polish
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Final CD + LP polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        max_sw = max(5, int((deadline - 10 - time.monotonic()) / max(0.5, tracker.best.size / 60.0)))
        a_cd = run_coord_descent(tracker.best, max_sw, deadline - 5, seed=99999)
        tracker.update(a_cd)

        if time.monotonic() < deadline - 5:
            a_final = run_lp_optimize(a_cd, 30, deadline, n_active_start=20, n_active_end=10,
                                      m_tighten=0.0, lp_time=3.0)
            tracker.update(a_final)

    tracker.report("After Phase 4")

    # Phase 5: Try one more length from best
    if tracker.best is not None and time.monotonic() < deadline - 15:
        n_cur = tracker.best.size
        for mult in [1.2, 0.8]:
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 2048)
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 5, time.monotonic() + 25)
            a_opt = run_optimization_pipeline(a_ext, new_n, sub_dl, seed=seed + 777)
            tracker.update(a_opt)

    tracker.report("FINAL")

    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
