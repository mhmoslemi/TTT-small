    size = 1

    fa = rfft(a, size)
    conv = irfft(fa * fa, size)

    m = 2 * n - 1
    A_ub = np.hstack([A, -np.eye(m)])
    b_ub = np.zeros(m)
    
    # Equality: sum(q) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(None, None)]  # t is free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit,
                              'dual_feasibility_tolerance': 1e-10,
                              'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            q = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(q))
            if s > 1e-10:
                return q / s
    except Exception:
        pass
    return None

def solve_lp_max_sum(a, M, time_limit=30):
    """Original LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def golden_section_search(f, lo, hi, tol=0.005, max_iter=30):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0

def lp_alternate(a, max_iters=50, time_budget=80.0, use_min_peak=True):
    """LP alternating with golden section line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        if use_min_peak:
            g = solve_lp_min_peak(a, time_limit=max(5, int(remaining * 0.5)))
        else:
            g = solve_lp_max_sum(a, M, time_limit=max(5, int(remaining * 0.5)))
        
        if g is None:
            break

        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g
            a_try = normalize(a_try)
            return compute_obj(a_try)

        t_opt = golden_section_search(obj_at_t, 0.0, 1.0, tol=0.005, max_iter=25)
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            rng = np.random.default_rng((it * 7919 + 12345) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.05 + 0.02 * (it % 5), n)))

    return best_a

def peak_shaving(a, max_iters=100, time_budget=60.0):
    """
    Iterative peak shaving: identify top peaks in convolution,
    reduce contributions from high-impact positions.
    Uses multiplicative updates.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # FFT size for gradient-like computations
    size = 1
    while size < 2 * n:
        size <<= 1

    eta = 0.1  # learning rate
    top_k = min(10, n)  # number of peaks to target

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Select top-k peaks
        if top_k >= len(conv):
            active = np.ones(len(conv))
        else:
            # Soft selection: weight peaks by their excess over median
            threshold = np.median(conv) + 0.5 * (cmax - np.median(conv))
            active = np.maximum(0, (conv - threshold) / (cmax - threshold + 1e-15))
            active /= np.sum(active) + 1e-30

        # Compute "pressure" on each position i:
        # pressure_i = sum_k active_k * a_{k-i} (contribution of i to active peaks)
        # Use FFT for efficiency
        a_flip = a[::-1]
        Gf = rfft(active, size)
        Af = rfft(a_flip, size)
        corr = irfft(Gf * Af, size)
        pressure = corr[n-1:2*n-1]  # (2n-1) or n elements?
        
        # Actually pressure should be length n:
        # pressure_i = sum_k active_k * a_{k-i} for valid k-i
        # This is (active * a_flipped) convolved, taking the middle n
        pressure = corr[:n]  # need to verify indexing
        
        # Normalize pressure
        p_mean = np.mean(pressure)
        p_std = np.std(pressure) + 1e-30
        
        # Multiplicative update: reduce high-pressure positions
        update = np.exp(-eta * (pressure - p_mean) / p_std)
        a_new = normalize(a * update)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.1, 0.5)
        else:
            eta *= 0.7
            if eta < 0.001:
                break

    return best_a

def spectral_equalization_v2(a, max_iters=80, time_budget=50.0):
    """
    Improved spectral equalization with adaptive alpha and multiple projections.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 3 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Adaptive alpha: start aggressive, ease off
        progress = it / float(max_iters)
        alpha = 0.3 + 0.5 * (1.0 - progress)
        
        P_norm = P / p_max
        # Slight floor to avoid zeroing
        P_norm = np.maximum(P_norm, 0.01)
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        # Non-negativity + normalization
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blended update
            blend = 0.3 + 0.4 * (1.0 - progress)
            a = normalize((1.0 - blend) * a + blend * a_new)

    return best_a

def soft_max_gradient(a, tau_init=0.001, step_size=0.04, max_steps=300, time_budget=50.0):
    """Soft-max gradient descent targeting multiple peaks."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 3 * n:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Adaptive temperature
        tau = tau_init + 0.3 * cmax / (n * (step + 1))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient via FFT
        w_ext = np.zeros(2 * n - 1)
        w_ext[:] = weights
        a_flip = a[::-1]
        Wf = rfft(w_ext, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Project onto sum=const hyperplane
        grad_centered = grad - np.mean(grad)

        # Backtracking line search
        ls = step_size
        for _ in range(10):
            a_new = normalize(a - ls * grad_centered)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                break
            ls *= 0.5
        else:
            if ls < 1e-10:
                break
            a = normalize(a - ls * grad_centered)

        tau_init *= 0.98

    return best_a

def coordinate_descent(a, time_budget=20.0):
    """Targeted coordinate descent on peak contributors."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        contrib = []
        for j in range(min(n, 200)):  # Limit for speed
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            for _ in range(3):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue

                for frac in [0.03, 0.05, 0.08]:
                    a_try = a.copy()
                    delta = frac * a_try[j1]
                    if a_try[j1] - delta < 0:
                        continue
                    a_try[j1] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)

                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved and rng.random() < 0.15:
            a = normalize(a * np.exp(rng.normal(0, 0.04, n)))

    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def structural_init(n, rng):
    """Create structurally diverse initial sequences."""
    method = rng.integers(0, 6)
    x = np.arange(n) / max(n - 1, 1)
    
    if method == 0:
        # Uniform
        a = np.ones(n)
    elif method == 1:
        # Linear ramp (increasing)
        a = np.linspace(0.5, 1.5, n)
    elif method == 2:
        # Quadratic (center-heavy)
        a = np.exp(-2.0 * (x - 0.5)**2 / (0.25 * 0.25))
    elif method == 3:
        # Bimodal (two peaks)
        a = np.exp(-5.0 * (x - 0.25)**2) + np.exp(-5.0 * (x - 0.75)**2)
    elif method == 4:
        # Cosine window
        a = np.cos(np.pi * x) ** 2 + 0.1
    elif method == 5:
        # Noise
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0.01)
    return a / np.sum(a)

def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf

def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    Phase 1: LP optimization at parent and nearby n values
    Phase 2: Peak shaving + spectral equalization
    Phase 3: Aggressive n-sweep with structural initialization
    Phase 4: Random restarts with diverse strategies
    Phase 5: Final intensive polish with all methods
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    rng = np.random.default_rng(seed)

    # =========================================================
    # Phase 1: Core LP optimization (most time here)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: LP optimization")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        n_cur = len(a)

        # Multiple LP cycles with alternating formulations
        for cycle in range(8):
            if time.monotonic() > deadline - 400:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.12, 80.0)

            # Alternate between min-peak and max-sum LP
            use_min = (cycle % 2 == 0)
            a_new = lp_alternate(a, max_iters=30, time_budget=alloc, use_min_peak=use_min)
            obj_new = compute_obj(a_new)
            
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  LP cycle {cycle} ({'min_peak' if use_min else 'max_sum'}): {best_score:.8f}")
            else:
                a = a_new
                print(f"  LP cycle {cycle} ({'min_peak' if use_min else 'max_sum'}): {obj_new:.8f} (no imp)")

    # =========================================================
    # Phase 2: Peak shaving + spectral + gradient
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Peak shaving + spectral + gradient")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())

        # Peak shaving
        if time.monotonic() < deadline - 250:
            a_ps = peak_shaving(a, max_iters=100, time_budget=50.0)
            obj_ps = compute_obj(a_ps)
            if obj_ps < best_score:
                best_score = obj_ps
                best_seq = a_ps.copy()
                a = a_ps
                print(f"  Peak shaving: {best_score:.8f}")
            else:
                print(f"  Peak shaving: {obj_ps:.8f} (no imp)")
                a = normalize(0.7 * a + 0.3 * a_ps)

        # Spectral equalization
        if time.monotonic() < deadline - 200:
            a_spec = spectral_equalization_v2(a, max_iters=60, time_budget=40.0)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                a = a_spec
                print(f"  Spectral: {best_score:.8f}")
            else:
                print(f"  Spectral: {obj_spec:.8f} (no imp)")

        # Soft-max gradient
        if time.monotonic() < deadline - 150:
            a_soft = soft_max_gradient(a, tau_init=0.001, step_size=0.04,
                                       max_steps=200, time_budget=40.0)
            obj_soft = compute_obj(a_soft)
            if obj_soft < best_score:
                best_score = obj_soft
                best_seq = a_soft.copy()
                a = a_soft
                print(f"  SoftMax grad: {best_score:.8f}")
            else:
                print(f"  SoftMax grad: {obj_soft:.8f} (no imp)")

        # Follow-up LP
        if time.monotonic() < deadline - 100:
            a_new = lp_alternate(a, max_iters=20, time_budget=50.0, use_min_peak=True)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Post-Phase2 LP: {best_score:.8f}")

    # =========================================================
    # Phase 3: Aggressive n-sweep
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: n-sweep")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700
    
    # Try a range of n values
    n_candidates = []
    for n_try in range(max(100, n_best - 200), min(1500, n_best + 300), 25):
        n_candidates.append(n_try)
    
    # Also try some specific values
    for n_sp in [200, 300, 400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400]:
        if 100 <= n_sp <= 1400:
            n_candidates.append(n_sp)
    
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if n_try == n_best and best_seq is not None:
            a_try = normalize(best_seq.copy())
        elif best_seq is not None:
            a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = structural_init(n_try, rng)

        # Quick optimization
        alloc = min(remaining * 0.08, 30.0)
        if n_try <= 1000:
            a_try = lp_alternate(a_try, max_iters=8, time_budget=alloc * 0.4, use_min_peak=True)
        a_try = soft_max_gradient(a_try, tau_init=0.001, step_size=0.03,
                                  max_steps=50, time_budget=alloc * 0.4)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")

    # =========================================================
    # Phase 4: Random restarts with structural diversity
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Random restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)

        for restart in range(10):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.1, 30.0)

            # Diverse perturbation
            method = restart % 5
            if method == 0:
                a_try = best_seq * np.exp(rng.normal(0, 0.08, n_cur))
            elif method == 1:
                freq = rng.uniform(0.5, 5.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
                a_try = best_seq * mod
            elif method == 2:
                a_rand = structural_init(n_cur, rng)
                mix = rng.uniform(0.2, 0.5)
                a_try = (1 - mix) * normalize(best_seq) + mix * a_rand
            elif method == 3:
                # Frequency domain perturbation
                size = 1
                while size < 2 * n_cur:
                    size <<= 1
                A = fft(best_seq, size)
                A *= np.exp(1j * rng.normal(0, 0.1, size))
                a_try = np.real(ifft(A, size))[:n_cur]
            else:
                # Block reversal
                a_try = best_seq.copy()
                blen = int(rng.integers(n_cur // 4, n_cur // 2))
                si = int(rng.integers(0, n_cur - blen))
                a_try[si:si+blen] = a_try[si:si+blen][::-1]

            a_try = normalize(a_try)

            # Optimize
            if n_cur <= 1000:
                a_try = lp_alternate(a_try, max_iters=8, time_budget=alloc * 0.4, use_min_peak=True)
            a_try = soft_max_gradient(a_try, tau_init=0.001, step_size=0.03,
                                      max_steps=40, time_budget=alloc * 0.35)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart}: {best_score:.8f} **")
            else:
                print(f"  Restart {restart}: {obj_try:.8f}")

    # =========================================================
    # Phase 5: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        n_final = len(a_final)

        for polish_round in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.2, 40.0)

            # Cycle through methods
            method = polish_round % 4
            if method == 0 and n_final <= 1000:
                a_pol = lp_alternate(a_final, max_iters=15, time_budget=alloc * 0.5, use_min_peak=True)
            elif method == 1:
                a_pol = peak_shaving(a_final, max_iters=80, time_budget=alloc * 0.5)
            elif method == 2:
                a_pol = spectral_equalization_v2(a_final, max_iters=50, time_budget=alloc * 0.5)
            else:
                a_pol = soft_max_gradient(a_final, tau_init=0.0005, step_size=0.02,
                                          max_steps=100, time_budget=alloc * 0.5)

            # Coordinate descent finish
            if remaining > 15:
                a_pol = coordinate_descent(a_pol, time_budget=min(15, remaining * 0.2))

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish_round} (method {method}): {best_score:.8f}")
            else:
                a_final = a_pol
                print(f"  Polish {polish_round} (method {method}): {obj_pol:.8f}")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale-invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_min_peak(a, time_limit=30):
    """Solve LP: min t s.t. conv(a,q) <= t for all k, sum(q)=1, q>=0.
    Returns the optimal q (normalized to sum=1) or None."""
    n = len(a)
    A = build_toeplitz(a)
    m = 2 * n - 1
    # Variables: q_0..q_{n-1}, t (total n+1)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # A_ub: [A | -I_m] @ [q; t] <= 0
    A_ub = np.hstack([A, -np.eye(m)])
    b_ub = np.zeros(m)
    # Equality: sum(q) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            q = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(q))
            if s > 1e-10:
                return q / s
    except Exception:
        pass
    return None


def solve_lp_max_sum(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:

def golden_section_search(f, lo, hi, tol=0.005, max_iter=30):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def lp_alternate(a, max_iters=50, time_budget=80.0, use_min_peak=True):
    """LP alternating with golden section line search for step size."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        if use_min_peak:
            g = solve_lp_min_peak(a, time_limit=max(3, int(remaining * 0.5)))
        else:
            g = solve_lp_max_sum(a, M, time_limit=max(3, int(remaining * 0.5)))

        if g is None:
            break

        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g
            a_try = normalize(a_try)
            return compute_obj(a_try)

        t_opt = golden_section_search(obj_at_t, 0.0, 1.0, tol=0.005, max_iter=25)
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

def peak_shaving(a, max_iters=100, time_budget=60.0):
    """
    Iterative peak shaving: identify top peaks in convolution,
    reduce contributions from high-impact positions via multiplicative updates.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 3 * n:
        size <<= 1

    eta = 0.1
    a_flip = None

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Select top peaks with soft weighting
        sorted_idx = np.argsort(conv)[::-1]
        top_k = min(15, len(conv))
        threshold = conv[sorted_idx[top_k]] if top_k < len(conv) else 0.0
        active = np.maximum(0.0, (conv - threshold) / (cmax - threshold + 1e-30))
        w_sum = float(np.sum(active))
        if w_sum < 1e-30:
            active = np.zeros(len(conv))
            active[int(np.argmax(conv))] = 1.0
        else:
            active /= w_sum

        # Compute pressure via FFT
        if a_flip is None or len(a_flip) != n:
            a_flip = a[::-1]
        else:
            a_flip = a[::-1].copy()

        w_ext = np.zeros(2 * n - 1)
        w_ext[:] = active
        Wf = rfft(w_ext, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        pressure = corr[n - 1:2 * n - 1]

        # Normalize pressure
        p_mean = np.mean(pressure)
        p_std = np.std(pressure) + 1e-30

        # Multiplicative update: reduce high-pressure positions
        update = np.exp(-eta * (pressure - p_mean) / p_std)
        a_new = normalize(a * update)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.05, 0.4)
        else:
            eta *= 0.7
            if eta < 0.001:
                break

    return best_a


def spectral_equalization_v2(a, max_iters=80, time_budget=50.0):
    """
    Improved spectral equalization with adaptive alpha.
    Alternates between frequency-domain power flattening and time-domain projection.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 3 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        progress = it / float(max_iters)
        alpha = 0.3 + 0.6 * (1.0 - progress)

        P_norm = P / p_max
        P_norm = np.maximum(P_norm, 0.005)
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.3 + 0.4 * (1.0 - progress)
            a = normalize((1.0 - blend) * a + blend * a_new)

    return best_a


def soft_max_gradient(a, tau_init=0.001, step_size=0.04, max_steps=300, time_budget=50.0):
    """Soft-max gradient descent with backtracking line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 3 * n:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        tau = tau_init + 0.3 * cmax / (n * (step + 1))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient via FFT
        w_ext = np.zeros(2 * n - 1)
        w_ext[:] = weights
        a_flip = a[::-1]
        Wf = rfft(w_ext, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        grad_centered = grad - np.mean(grad)

        # Backtracking line search
        ls = step_size
        found = False
        for _ in range(12):
            a_new = normalize(a - ls * grad_centered)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                found = True
                break
            ls *= 0.5
        if not found:
            if ls < 1e-12:
                break
            a = normalize(a - ls * grad_centered)

        tau_init *= 0.98

    return best_a

    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        contrib = []
        j_max = min(n, 300)
        for j in range(j_max):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contrib[:10]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            for _ in range(4):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue

                for frac in [0.03, 0.05, 0.08, 0.12]:
                    a_try = a.copy()
                    delta = frac * a_try[j1]
                    if a_try[j1] - delta < 0:
                        continue
                    a_try[j1] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)

                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved:
            if rng.random() < 0.2:
                a = normalize(a * np.exp(rng.normal(0, 0.04, n)))
            else:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def structural_init(n, rng):
    """Create structurally diverse initial sequences."""
    method = int(rng.integers(0, 7))
    x = np.arange(n) / max(n - 1, 1)

    if method == 0:
        a = np.ones(n)
    elif method == 1:
        a = np.linspace(0.5, 1.5, n)
    elif method == 2:
        sigma = 0.2 + 0.3 * rng.random()
        a = np.exp(-2.0 * (x - 0.5) ** 2 / (sigma * sigma))
    elif method == 3:
        c1 = rng.uniform(0.1, 0.4)
        c2 = 1.0 - c1
        w1 = 0.1 + 0.2 * rng.random()
        a = np.exp(-5.0 * (x - c1) ** 2 / w1) + np.exp(-5.0 * (x - c2) ** 2 / w1)
    elif method == 4:
        a = np.cos(np.pi * x) ** 2 + 0.1
    elif method == 5:
        a = rng.random(n) + 0.1
    else:
        freq = rng.uniform(1.0, 4.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x) + 0.1 * np.cos(4 * np.pi * freq * x)

    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    Phase 1: Core LP optimization (alternating min-peak and max-sum)
    Phase 2: Peak shaving + spectral equalization + soft-max gradient
    Phase 3: Aggressive n-sweep with structural initialization
    Phase 4: Random restarts with diverse strategies
    Phase 5: Final intensive polish cycling through all methods
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    rng = np.random.default_rng(seed)

    # =========================================================
    # Phase 1: Core LP optimization
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: LP optimization")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        n_cur = len(a)

        for cycle in range(10):
            if time.monotonic() > deadline - 350:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.11, 70.0)

            use_min = (cycle % 2 == 0)
            a_new = lp_alternate(a, max_iters=30, time_budget=alloc, use_min_peak=use_min)
            obj_new = compute_obj(a_new)

            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  LP cycle {cycle} ({'min_peak' if use_min else 'max_sum'}): {best_score:.8f} **")
            else:
                a = a_new
                print(f"  LP cycle {cycle} ({'min_peak' if use_min else 'max_sum'}): {obj_new:.8f}")

    # =========================================================
    # Phase 2: Peak shaving + spectral + gradient
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Peak shaving + spectral + gradient")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())

        if time.monotonic() < deadline - 250:
            a_ps = peak_shaving(a, max_iters=100, time_budget=45.0)
            obj_ps = compute_obj(a_ps)
            if obj_ps < best_score:
                best_score = obj_ps
                best_seq = a_ps.copy()
                a = a_ps
                print(f"  Peak shaving: {best_score:.8f} **")
            else:
                print(f"  Peak shaving: {obj_ps:.8f}")
                a = normalize(0.7 * a + 0.3 * a_ps)

        if time.monotonic() < deadline - 200:
            a_spec = spectral_equalization_v2(a, max_iters=60, time_budget=35.0)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                a = a_spec
                print(f"  Spectral: {best_score:.8f} **")
            else:
                print(f"  Spectral: {obj_spec:.8f}")

        if time.monotonic() < deadline - 150:
            a_soft = soft_max_gradient(a, tau_init=0.001, step_size=0.04,
                                       max_steps=200, time_budget=35.0)
            obj_soft = compute_obj(a_soft)
            if obj_soft < best_score:
                best_score = obj_soft
                best_seq = a_soft.copy()
                a = a_soft
                print(f"  SoftMax grad: {best_score:.8f} **")
            else:
                print(f"  SoftMax grad: {obj_soft:.8f}")

        if time.monotonic() < deadline - 100:
            a_new = lp_alternate(a, max_iters=20, time_budget=45.0, use_min_peak=True)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Post-Phase2 LP: {best_score:.8f} **")
            else:
                print(f"  Post-Phase2 LP: {obj_new:.8f}")

    # =========================================================
    # Phase 3: Aggressive n-sweep
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: n-sweep")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700

    n_candidates = sorted(set([
        n_best - 200, n_best - 150, n_best - 100, n_best - 75,
        n_best - 50, n_best - 25, n_best, n_best + 25,
        n_best + 50, n_best + 75, n_best + 100, n_best + 150,
        n_best + 200, n_best + 300,
        200, 300, 400, 500, 600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_candidates = [n for n in n_candidates if 150 <= n <= 1400]
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break

        if n_try == n_best and best_seq is not None:
            a_try = normalize(best_seq.copy())
        elif best_seq is not None:
            a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = structural_init(n_try, rng)

        alloc = min(remaining * 0.08, 25.0)
        if n_try <= 1000:
            a_try = lp_alternate(a_try, max_iters=8, time_budget=alloc * 0.4, use_min_peak=True)
        a_try = soft_max_gradient(a_try, tau_init=0.001, step_size=0.03,
                                  max_steps=50, time_budget=alloc * 0.4)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 4: Random restarts with diverse strategies
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Random restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)

        for restart in range(12):
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            alloc = min(remaining * 0.1, 25.0)

            method = restart % 6
            if method == 0:
                a_try = best_seq * np.exp(rng.normal(0, 0.08, n_cur))
            elif method == 1:
                freq = rng.uniform(0.5, 5.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
                a_try = best_seq * mod
            elif method == 2:
                a_rand = structural_init(n_cur, rng)
                mix = rng.uniform(0.2, 0.5)
                a_try = (1 - mix) * normalize(best_seq) + mix * a_rand
            elif method == 3:
                size = 1
                while size < 2 * n_cur:
                    size <<= 1
                A = fft(best_seq.astype(np.float64), size)
                A *= np.exp(1j * rng.normal(0, 0.15, size))
                a_try = np.real(ifft(A, size))[:n_cur]
            elif method == 4:
                a_try = best_seq.copy()
                blen = int(rng.integers(n_cur // 4, n_cur // 2))
                si = int(rng.integers(0, n_cur - blen))
                a_try[si:si + blen] = a_try[si:si + blen][::-1]
            else:
                # Invert locally
                a_try = best_seq.copy()
                for _ in range(3):
                    si = int(rng.integers(0, n_cur - 20))
                    a_try[si:si + 20] = a_try[si + 20 - 1:si - 1:-1] if si + 20 <= n_cur else a_try[si:si+20]

            a_try = normalize(a_try)

            if n_cur <= 1000:
                a_try = lp_alternate(a_try, max_iters=8, time_budget=alloc * 0.4, use_min_peak=True)
            a_try = soft_max_gradient(a_try, tau_init=0.001, step_size=0.03,
                                      max_steps=40, time_budget=alloc * 0.35)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart} (m={method}): {best_score:.8f} **")
            else:
                print(f"  Restart {restart} (m={method}): {obj_try:.8f}")

    # =========================================================
    # Phase 5: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        n_final = len(a_final)

        for polish_round in range(8):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.2, 35.0)

            method = polish_round % 5
            if method == 0 and n_final <= 1000:
                a_pol = lp_alternate(a_final, max_iters=15, time_budget=alloc * 0.5, use_min_peak=True)
            elif method == 1:
                a_pol = peak_shaving(a_final, max_iters=80, time_budget=alloc * 0.5)
            elif method == 2:
                a_pol = spectral_equalization_v2(a_final, max_iters=50, time_budget=alloc * 0.5)
            elif method == 3:
                a_pol = soft_max_gradient(a_final, tau_init=0.0005, step_size=0.02,
                                          max_steps=100, time_budget=alloc * 0.5)
            else:
                if n_final <= 1000:
                    a_pol = lp_alternate(a_final, max_iters=10, time_budget=alloc * 0.3, use_min_peak=False)
                    a_pol = peak_shaving(a_pol, max_iters=40, time_budget=alloc * 0.3)
                    a_pol = soft_max_gradient(a_pol, tau_init=0.0005, step_size=0.02,
                                              max_steps=50, time_budget=alloc * 0.25)
                else:
                    a_pol = soft_max_gradient(a_final, tau_init=0.0005, step_size=0.02,
                                              max_steps=100, time_budget=alloc * 0.7)

            if remaining > 15:
                a_pol = coordinate_descent(a_pol, time_budget=min(12, remaining * 0.15))

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish_round} (m={method}): {best_score:.8f} **")
            else:
                a_final = a_pol
                print(f"  Polish {polish_round} (m={method}): {obj_pol:.8f}")

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
