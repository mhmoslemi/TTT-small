        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def lp_with_constraint_generation(a, M, max_rounds=50, time_limit=None):
    n = len(a)
    b = fast_conv(a)
    active = [int(np.argmax(b))]
    for _ in active[:]:
        for k in range(len(b)):
            if k not in active and b[k] >= M - 1e-10:
                active.append(k)
    best_g = None
    for _round in range(max_rounds):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        g = solve_lp_best_response(a, active, M)
        if g is None:
            break
        ag_conv = cross_conv(a, g)
        violated = []
        for k in range(len(b)):
            if k not in active and ag_conv[k] > M + 1e-10:
                violated.append(k)
        if not violated:
            best_g = g
            break
        for k in violated:
            active.append(k)
        if len(active) > 2 * n:
            active = sorted(active, key=lambda k: -b[k])[:2 * n]
            break
        best_g = g
    return best_g


def run_lp_optimization(a_init, deadline, max_iters=200, initial_active=10):
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        top_k = min(initial_active + it, 2 * n - 1)
        g = lp_with_constraint_generation(a, M, max_rounds=20,
                                          time_limit=min(deadline, time.monotonic() + 5))
        if g is None:
            no_improve += 2
            if no_improve > 10:
                break
            continue
        improved = False
        best_trial_M = best_M
        best_trial_a = a.copy()
        for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize(a_trial)
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_trial_a = a_trial.copy()
            if M_trial < best_M - 1e-12:
                a = best_trial_a
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
    return best_a


def run_fista(a_init, deadline, tau_start=50, tau_end=2000, max_iters=5000):
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    a_prev = a.copy()
    t_mom = 1.0
    step = 0.3
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        b_y = fast_conv(y)
        _, p = smooth_max_grad(b_y, tau)
        grad = grad_from_p(p, y)
        grad_tan = grad - grad.mean()
        a_new = proj_simplex(y - step * grad_tan)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = proj_simplex(a_new + alpha * (a_new - a_prev))
        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        if it % 10 == 0:
            M_a = fast_conv(a).max()
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
            step *= 0.999
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    return best_a


def lbfgs_smooth_max(a_init, tau, deadline, max_iter=2000):
    n = a_init.size
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_M = fast_conv(a).max()

    def objective_and_grad(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(n)
        x = x / s
        b = fast_conv(x)
        _, p = smooth_max_grad(b, tau)
        grad = grad_from_p(p, x)
        grad_tan = grad - grad.mean()
        return fast_conv(x).max(), grad_tan

    T0 = (1.0, 1.0)
    state = [a.copy()]

    def fun(x):
        val, _ = objective_and_grad(x)
        return val

    def jac(x):
        _, g = objective_and_grad(x)
        return g

    try:
        res = minimize(fun, a, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                'ftol': 1e-14, 'gtol': 1e-8, 'iprint': -1})
        if res.x is not None:
            a_opt = normalize(res.x)
            M_opt = fast_conv(a_opt).max()
            if M_opt < best_M:
                best_M = M_opt
                best_a = a_opt.copy()
    except Exception:
        pass
    return best_a


def coordinate_descent(a_init, deadline, max_sweeps=50, seed=42):
    rng_cd = np.random.default_rng(seed)
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False
        for i in order:
            if time.monotonic() > deadline:
                break
            x_cur = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            c_vals = []
            slopes = []
            has_quad = False
            quad_pos = -1
            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    c_vals.append(b[k] - x_cur * x_cur)
                    slopes.append(0.0)
                    has_quad = True
                    quad_pos = len(c_vals) - 1
                else:
                    j = k - i
                    if 0 <= j < n:
                        c_vals.append(b[k] - 2.0 * x_cur * a[j])
                        slopes.append(2.0 * a[j])
                    else:
                        c_vals.append(b[k])
                        slopes.append(0.0)
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            c_arr = np.array(c_vals)
            s_arr = np.array(slopes)

            def obj_at(x_test):
                vals = c_arr + s_arr * x_test
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if has_quad:
                    qv = c_arr[quad_pos] + x_test * x_test
                    if qv > m:
                        m = qv
                return m

            lo = 0.0
            hi = max(2.0 * x_cur + 0.1, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.5)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_at(a_pt)
            fb = obj_at(b_pt)
            for _ in range(35):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_at(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)
            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0
            x_opt = max(0.0, x_opt)
            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
                k_lo2 = i
                k_hi2 = min(2 * n - 2, i + n - 1)
                for k in range(k_lo2, k_hi2 + 1):
                    if k == 2 * i:
                        b[k] = x_opt * x_opt
                    else:
                        j2 = k - i
                        if 0 <= j2 < n:
                            b[k] = b[k] - (2.0 * x_cur * a[j2] if j2 != i else 0)
                            if j2 == i:
                                pass
                            b[k] = 2.0 * x_opt * a[j2] if j2 != i else x_opt * x_opt
                            if j2 == i and k == 2 * i:
                                b[k] = x_opt * x_opt
                            elif j2 != i:
                                b[k] = 2.0 * x_opt * a[j2]
                sweep_improved = True
        a = normalize(a)
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_improved:
            no_improve += 5
        if no_improve > 30:
            break
    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for noise_scale in [0.02, 0.05, 0.1, 0.2]:
            p_n = np.maximum(p + rng.normal(0, noise_scale, n), 1e-6)
            inits.append(p_n / p_n.sum())
    inits.append(np.ones(n) / n)
    poly = (4.0 * t * (1 - t)) + 0.05
    inits.append(poly / poly.sum())
    for freq in [1, 2, 3, 4]:
        cos_w = 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * np.pi * freq * t))
        inits.append(cos_w / cos_w.sum())
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    a_spec = np.ones(n) / n
    for _ in range(50):
        bs = fast_conv(a_spec)
        M_s = bs.max()
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.05 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    ksize = max(5, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    return inits


def run_combined_opt(a0, deadline, seed=0):
    a = normalize(a0.copy())
    best = a.copy()
    best_M = fast_conv(a).max()
    elapsed_func = lambda: time.monotonic()
    total_time = max(1.0, deadline - elapsed_func())

    t_lp1 = elapsed_func() + 0.35 * total_time
    if time.monotonic() < t_lp1:
        a_lp = run_lp_optimization(best, t_lp1, max_iters=100, initial_active=8)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
        print(f"    After LP1: M={best_M:.12f}", flush=True)

    t_fista = elapsed_func() + 0.15 * (deadline - elapsed_func())
    if time.monotonic() < t_fista:
        a_f = run_fista(best, t_fista, tau_start=100, tau_end=3000, max_iters=3000)
        M_f = fast_conv(a_f).max()
        if M_f < best_M:
            best_M = M_f
            best = a_f.copy()
        print(f"    After FISTA: M={best_M:.12f}", flush=True)

    t_lp2 = elapsed_func() + 0.25 * (deadline - elapsed_func())
    if time.monotonic() < t_lp2:
        a_lp2 = run_lp_optimization(best, t_lp2, max_iters=80, initial_active=15)
        M_lp2 = fast_conv(a_lp2).max()
        if M_lp2 < best_M:
            best_M = M_lp2
            best = a_lp2.copy()
        print(f"    After LP2: M={best_M:.12f}", flush=True)

    t_cd = deadline
    if time.monotonic() < t_cd:
        a_cd = coordinate_descent(best, t_cd, max_sweeps=100, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
        print(f"    After CD: M={best_M:.12f}", flush=True)

    return best


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)
