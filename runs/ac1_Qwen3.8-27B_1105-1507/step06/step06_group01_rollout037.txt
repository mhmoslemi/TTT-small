"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated FISTA on log-sum-exp smooth approximation (O(n log n) per step)
2. Dual-guided active-set LP best-response with cutting planes
3. Fast coordinate descent on critical coordinates
4. Multi-scale and multi-length search strategy

Key innovation: O(n log n) gradient via FFT enables 10-50x more iterations,
plus cutting-plane LP for final polish.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """FFT-based linear cross-convolution of a and b."""
    na, nb = len(a), len(b)
    if na == 0 or nb == 0:
        return np.array([])
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv
    if cond.all():
        theta = (cv[-1] - 1.0) / n
    else:
        idx = np.where(~cond)[0]
        if len(idx) == 0:
            theta = 0.0
        else:
            rho_idx = int(idx[0])
            theta = (cv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad_fft(a, tau):
    """Compute log-sum-exp smooth max of conv(a,a) and its gradient using FFT.
    
    Returns (smooth_value, gradient of length n).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    scaled = tau * (b - b_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    smooth_val = b_max + np.log(exp_scaled.sum()) / tau
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] * I(0<=k-i<n)
    # = 2 * cross_correlation(a, p)[lag=-i]
    # Compute via FFT: ifft(fft(a) * conj(fft(p))) gives cross-correlation
    
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    
    fa = np.fft.rfft(a, L)
    fp = np.fft.rfft(p, L)
    
    cross_corr = np.fft.irfft(fa * np.conj(fp), L)
    
    # term[i] = cross_corr at lag -i = cross_corr[(-i) % L]
    # For i=0: cross_corr[0]
    # For i=1: cross_corr[L-1]
    # For i=2: cross_corr[L-2]
    # ...
    term = np.empty(n)
    term[0] = cross_corr[0]
    if n > 1:
        term[1:] = cross_corr[L - 1:L - n:-1]
    
    grad = 2.0 * term
    return smooth_val, grad


def run_fista_fft(a_init, deadline, tau_start=30.0, tau_end=500.0, max_iters=200000):
    """FISTA with FFT-based gradient on log-sum-exp approximation.
    
    Uses adaptive tau (annealing) and backtracking step size.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    # Adaptive step size
    step = 0.3
    
    it = 0
    stagnation = 0
    
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters // 2))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_grad_fft(y, tau)
        
        # Project gradient to tangent space of simplex
        grad_tan = grad_y - grad_y.mean()
        
        # Backtracking line search
        a_trial = y - step * grad_tan
        a_trial = proj_simplex(a_trial)
        
        # Check descent (simple check every 100 iters to save time)
        if it % 200 == 0:
            M_trial = fast_conv(a_trial).max()
            M_curr = fast_conv(y).max()
            if M_trial > M_curr * 1.01:
                step *= 0.5
                a_trial = y - step * grad_tan
                a_trial = proj_simplex(a_trial)
            elif M_trial < M_curr * 0.99:
                step = min(step * 1.2, 2.0)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = a_trial + alpha * (a_trial - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_trial
        y = y_new
        t_mom = t_new
        it += 1
        
        # Track best
        if it % 50 == 0 or it == max_iters - 1:
            M_a = fast_conv(a).max()
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()
                stagnation = 0
            else:
                stagnation += 1
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
                stagnation = 0
            
            # Early stop if no improvement for a long time
            if stagnation > 200 and tau > 200:
                break
    
    # Final check
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def build_active_lp(n, a, active_indices):
    """Build sparse constraint matrix for LP: conv(a,g)[k] <= M for k in active."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response_cutting(a, M, max_rounds=30, time_limit=10.0):
    """Solve LP best response using cutting plane method.
    
    max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0
    """
    n = len(a)
    t_start = time.monotonic()
    
    # Initial active set: top convolution values
    b = fast_conv(a)
    top_k = min(20, len(b))
    active = list(np.argsort(b)[::-1][:top_k])
    
    g_best = None
    
    for round_idx in range(max_rounds):
        if time.monotonic() - t_start > time_limit:
            break
        
        A = build_active_lp(n, a, active)
        if A.nnz == 0:
            break
        
        c_obj = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0.0, None)] * n
        
        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs',
                        options={'time_limit': 5, 'presolve': True})
            if not r.success or r.x is None:
                break
            
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            
            # Check if g violates any non-active constraint
            ag = fast_cross_conv(a, g)
            violations = np.where(ag > M * (1 + 1e-8))[0]
            
            if len(violations) == 0:
                g_best = g / g_sum
                break
            
            # Add violated constraints (up to 10 per round)
            new_constraints = violations[:10].tolist()
            existing = set(active)
            for vc in new_constraints:
                if vc not in existing:
                    active.append(vc)
                    existing.add(vc)
            g_best = g / g_sum
            
        except Exception:
            break
    
    return g_best


def run_lp_polish(a_init, max_iters, deadline, initial_active=30):
    """LP-based polishing with cutting planes."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:

    n = len(a)

        g = lp_best_response_cutting(a, best_M, max_rounds=20, time_limit=8.0)

        # Geometric line search
        improved = False

def coordinate_descent_fast(a_init, max_sweeps, deadline, seed=0):
    """Fast coordinate descent using precomputed structures."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()

        a = a / s

    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        
        b = fast_conv(a)
        sweep_changed = False
        
        for i in order:
            x_cur = a[i]
            
            # Precompute affected convolution values
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # For each affected k, c[k] = base[k] + slope[k]*x + quad (if k==2i)
            c_arr = np.zeros(k_hi - k_lo + 1)
            s_arr = np.zeros(k_hi - k_lo + 1)
            has_quad = False
            quad_idx = -1
            quad_base = 0.0
            
            for kk in range(k_lo, k_hi + 1):
                if kk == 2 * i:
                    c_arr[kk - k_lo] = b[kk] - x_cur * x_cur
                    s_arr[kk - k_lo] = 0.0
                    has_quad = True
                    quad_idx = kk - k_lo
                else:
                    j = kk - i
                    if 0 <= j < n:
                        c_arr[kk - k_lo] = b[kk] - 2.0 * x_cur * a[j]
                        s_arr[kk - k_lo] = 2.0 * a[j]
                    else:
                        c_arr[kk - k_lo] = b[kk]
                        s_arr[kk - k_lo] = 0.0
            
            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            
            def obj(x):
                vals = c_arr + s_arr * x
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if has_quad:
                    qv = c_arr[quad_idx] + x * x
                    if qv > m:
                        m = qv
                return m
            
            # Golden section search
            lo = 0.0
            hi = max(2.0 * x_cur, 0.5, np.sqrt(max(b.max(), 0.0)) * 1.5)
            for _ in range(5):
                if obj(hi) > obj(hi * 0.5) and hi < 5.0:
                    hi *= 2.0
                else:
                    break
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)

            fa = obj(a_pt)
            fb = obj(b_pt)
            
            for _ in range(30):
                if fa < fb:

                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj(b_pt)
            
            x_opt = (lo + hi) / 2.0
            if obj(0.0) <= obj(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - x_cur) > 1e-10:
                a[i] = max(0.0, x_opt)
                sweep_changed = True
        
        s = a.sum()
        if s < 1e-10:

        M = fast_conv(a).max()

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.02, n), 1e-6)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.08, n), 1e-6)
        inits.append(p_n2 / p_n2.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine (various widths)
    for alpha in [1.5, 2.0, 2.5, 3.0, 4.0]:
        rc = (1 + np.cos(2 * np.pi * t)) ** alpha
        rc = np.maximum(rc, 1e-8)
        inits.append(rc / rc.sum())
    
    # Asymmetric windows
    for skew in [0.3, 0.7, 0.2, 0.8]:
        asym = np.exp(-((t - skew) ** 2) / 0.05)
        asym = np.maximum(asym, 1e-8)
        inits.append(asym / asym.sum())
    
    # Polynomial windows
    poly = (t * (1 - t)) ** 0.7 + 0.05
    inits.append(poly / poly.sum())
    
    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Spectral init: reduce autocorrelation peak
    a_spec = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_spec)
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.15 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    
    return inits

def run_optimization_pipeline(a0, n, deadline, seed=0):
    """Run combined optimization: FISTA -> LP -> coordinate descent."""
    best = a0.copy()
    best_M = fast_conv(best).max()
    t_start = time.monotonic()
    
    # Stage 1: FISTA with multiple tau ranges (50% of time)
    t1 = t_start + 0.5 * (deadline - t_start)
    if time.monotonic() < t1:
        a_fISTA = run_fista_fft(best, t1, tau_start=20.0, tau_end=300.0, max_iters=500000)
        M_f = fast_conv(a_fISTA).max()
        if M_f < best_M:
            best_M = M_f
            best = a_fISTA.copy()
        # Second FISTA pass with higher tau
        if time.monotonic() < t1:
            a_f2 = run_fista_fft(best, t1, tau_start=200.0, tau_end=1000.0, max_iters=200000)
            M_f2 = fast_conv(a_f2).max()
            if M_f2 < best_M:
                best_M = M_f2
                best = a_f2.copy()
        print(f"    FISTA: M={best_M:.10f}", flush=True)
    
    # Stage 2: LP polish (30% of time)
    t2 = t_start + 0.8 * (deadline - t_start)
    if time.monotonic() < t2:
        a_lp = run_lp_polish(best, 100, t2)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
        print(f"    LP: M={best_M:.10f}", flush=True)
    
    # Stage 3: Coordinate descent (20% of time)
    if time.monotonic() < deadline:
        max_sw = max(10, int((deadline - time.monotonic()) / max(0.05, n / 200.0)))
        a_cd = coordinate_descent_fast(best, max_sw, deadline, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
        print(f"    CD: M={best_M:.10f}", flush=True)
    
    return best

            break
        
        time_alloc = min(time_per_length, remaining / max(1, n_lengths - li))
        print(f"\n  Length n={n}, time={time_alloc:.0f}s", flush=True)
        
        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        
        # Limit inits to fit time
        time_per_init = time_alloc / max(1, len(inits))
        time_per_init = max(time_per_init, 10.0)
        max_inits = max(2, int(time_alloc / max(20.0, time_per_init)))
        inits = inits[:max_inits]
        
        best_for_n = None
        best_M_for_n = float('inf')
        
        for ii, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline - 10:
                break
            
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()
            
            sub_deadline = min(time.monotonic() + time_per_init, phase1_deadline - 5)
            a_opt = run_optimization_pipeline(a0, n, sub_deadline, seed=seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()
            
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best M={best_M_for_n:.10f} J={J_val:.6f}", flush=True)
            tracker.update(best_for_n)
    
    tracker.report("After Phase 1")
    
    # === Phase 2: Intensive LP polish on global best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive LP polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    phase2_deadline = time.monotonic() + 0.20 * (deadline - time.monotonic())
    phase2_deadline = min(phase2_deadline, deadline - 120.0)
    
    if tracker.best is not None and time.monotonic() < phase2_deadline:
        a_lp = run_lp_polish(tracker.best, 200, phase2_deadline)
        tracker.update(a_lp)
        print(f"  After LP polish: J={tracker.best_score:.8f}", flush=True)
    
    tracker.report("After Phase 2")
    
    # === Phase 3: High-tau FISTA refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: High-tau FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    phase3_deadline = time.monotonic() + 0.12 * (deadline - time.monotonic())
    phase3_deadline = min(phase3_deadline, deadline - 60.0)
    
    if tracker.best is not None and time.monotonic() < phase3_deadline:
        for tau_s, tau_e in [(200, 800), (500, 2000)]:
            if time.monotonic() > phase3_deadline:
                break
            sub_dl = min(phase3_deadline, time.monotonic() + (phase3_deadline - time.monotonic()) / 2)
            a_f = run_fista_fft(tracker.best, sub_dl, tau_start=tau_s, tau_end=tau_e, max_iters=300000)
            tracker.update(a_f)
    
    tracker.report("After Phase 3")
    
    # === Phase 4: Coordinate descent + LP alternating ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: CD + LP alternating polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    phase4_deadline = deadline - 30.0
    
    if tracker.best is not None and time.monotonic() < phase4_deadline:
        current = tracker.best.copy()
        
        for round_idx in range(5):
            if time.monotonic() > phase4_deadline - 5:
                break
            
            # Coordinate descent
            sub_dl1 = min(phase4_deadline - 5, time.monotonic() + 60)
            a_cd = coordinate_descent_fast(current, 50, sub_dl1, seed=1000 + round_idx)
            if fast_conv(a_cd).max() < fast_conv(current).max():
                current = a_cd
                tracker.update(current)
            
            # LP pass
            sub_dl2 = min(phase4_deadline - 5, time.monotonic() + 40)
            a_lp = run_lp_polish(current, 30, sub_dl2)
            if fast_conv(a_lp).max() < fast_conv(current).max():
                current = a_lp
                tracker.update(current)
            
            print(f"  Round {round_idx}: J={tracker.best_score:.8f}", flush=True)
    
    tracker.report("After Phase 4")
    
    # === Phase 5: Try nearby lengths from best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 20:
        n_cur = tracker.best.size
        for mult in [1.2, 0.8, 1.5, 0.67]:
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 2048)
            if abs(new_n - n_cur) < 32:
                continue
            if time.monotonic() > deadline - 15:
                break
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 10, time.monotonic() + 40)
            a_opt = run_fista_fft(a_ext, sub_dl, tau_start=100, tau_end=500, max_iters=100000)
            # Quick LP
            if time.monotonic() < deadline - 10:
                a_opt = run_lp_polish(a_opt, 20, min(deadline - 5, time.monotonic() + 20))
            tracker.update(a_opt)
    
    tracker.report("After Phase 5")
    
    # === Final: ensure valid return ===
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Verify
    final_score = float(evaluate_sequence([float(x) for x in tracker.best]))
    if not np.isfinite(final_score) or final_score > tracker.best_score:
        print(f"  WARNING: verification score {final_score} != tracked {tracker.best_score}", flush=True)
        # Fall back to a simple valid sequence
        fallback = np.ones(512).tolist()
        return fallback
    
    return [float(x) for x in tracker.best]
