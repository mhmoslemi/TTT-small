def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def build_lp_matrices(a):
    """Build sparse LP matrices for best response problem.
    
    min t s.t. (a*g)[k] <= t for k=0..2n-2, sum(g)=1, g>=0
    Variables: x = [g_0, ..., g_{n-1}, t]
    """
    n = len(a)
    num_k = 2 * n - 1
    
    rows = []
    cols = []
    vals = []
    
    for k in range(num_k):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
        # -t term
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(num_k, n + 1))
    b_ub = np.zeros(num_k)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    return c, A_ub, b_ub, A_eq, b_eq, bounds


def solve_best_response_lp(a, max_time=3.0):
    """Solve LP: min t s.t. (a*g)[k] <= t, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None)."""
    n = len(a)
    try:
        c, A_ub, b_ub, A_eq, b_eq, bounds = build_lp_matrices(a)
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': max_time, 'presolve': True,
                                 'dual_feasibility_tolerance': 1e-10,
                                 'primal_feasibility_tolerance': 1e-10})
        if result.success and result.x is not None:
            g = np.maximum(result.x[:n], 0.0)
            t_opt = result.x[n]
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
            return g, t_opt
    except Exception:
        pass
    return None, None


def solve_active_lp(a, n_active=40, max_time=2.0):
    """Solve LP using only top n_active constraints (faster).
    Returns (g, t_opt) or (None, None)."""
    n = len(a)
    b = fast_conv(a)
    num_k = 2 * n - 1
    
    # Get top n_active indices by convolution value
    top_indices = np.argsort(b)[::-1][:min(n_active, num_k)]
    
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(top_indices):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(idx)
            cols.append(j)
            vals.append(a[i])
        rows.append(idx)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(len(top_indices), n + 1))
    b_ub = np.zeros(len(top_indices))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': max_time, 'presolve': True})
        if result.success and result.x is not None:
            g = np.maximum(result.x[:n], 0.0)
            t_opt = result.x[n]
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
            # Verify: check if solution satisfies ALL constraints
            check_b = np.array([np.dot(a, np.roll(g, k % n)[:n]) for k in range(num_k)])
            # Actually need proper linear conv check
            full_b = fast_conv(g)
            cross = np.convolve(a, g)
            if cross.max() <= t_opt + 1e-8:
                return g, t_opt
            else:
                # Solution violates some constraints; return with actual max
                return g, cross.max()
    except Exception:
        pass
    return None, None


def lp_optimize(a_init, max_iters, deadline, use_full_lp=True, n_active=50):
    """Main LP-based optimization loop.
    
    Iterates: solve LP for best response, take step with line search.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    print(f"    LP start: M={best_M:.12f}", flush=True)
    
    no_improve = 0
    it = 0
    
    while it < max_iters and time.monotonic() < deadline:
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            break
        
        # Solve LP
        if use_full_lp and n <= 800:
            g, t_opt = solve_best_response_lp(a, max_time=3.0)
        else:
            g, t_opt = solve_active_lp(a, n_active=n_active, max_time=2.0)
        
        if g is None:
            break
        
        # Check if improvement is possible
        if t_opt >= M * (1.0 - 1e-9):
            no_improve += 10
            if no_improve > 60:
                break
            # Perturb to escape
            rng = np.random.default_rng(it * 7919)
            perturbation = rng.normal(0, 0.001, n)
            a = a + perturbation * a.sum() / n
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue
        
        # Line search
        best_t = 0
        best_M_trial = M
        step_candidates = [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02, 0.01]
        
        for step in step_candidates:
            a_trial = (1.0 - step) * a + step * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_t = step
        
        if best_t == 0:
            no_improve += 5
            if no_improve > 30:
                break
            # Try smaller steps
            for step in [0.005, 0.001]:
                a_trial = (1.0 - step) * a + step * g
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M_trial:
                    best_M_trial = M_trial
                    best_t = step
        
        if best_t == 0:
            break
        
        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        it += 1
        if it % 20 == 0:
            print(f"    LP iter {it}: M={fast_conv(a).max():.12f}, best={best_M:.12f}, t_opt/M={t_opt/M:.6f}", flush=True)
    
    # Final check
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    print(f"    LP end: M={best_M:.12f} after {it} iters", flush=True)
    return best_a


def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and gradient using log-sum-exp.
    Uses FFT for efficiency."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    # Log-sum-exp
    exp_b = np.exp(tau * (b - b_max))
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum  # softmax weights
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (with proper boundaries)
    # This is 2 * (p * a_flipped) convolved appropriately
    # grad[i] = 2 * sum_{k} p[k] * a[k-i] where a[m]=0 if m not in [0,n-1]
    # = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]  (when k = i+j)
    
    # Use correlation: grad[i] = 2 * sum_j p[i+j] * a[j]
    # This is like a cross-correlation of p and a
    # grad = 2 * np.correlate(p, a, mode='valid') if len(p) >= len(a)
    # But p has length 2n-1 and a has length n
    # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j] for i = 0, ..., n-1
    
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(p[i:i+n], a)
    
    smooth_val = b_max + np.log(exp_sum) / tau
    return smooth_val, grad


def fista_optimize(a_init, deadline, tau_start=100, tau_end=500, max_iters=5000, lr=0.5):
    """FISTA on smooth max approximation."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        step = lr / (1.0 + 0.01 * tau)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        it += 1
        
        if it % 10 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def coordinate_descent_1d(a, i, b, n):
    """Find optimal value for a[i] given all others (unnormalized)."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals) if c_vals else np.array([0.0])
    s_arr = np.array(slopes) if slopes else np.array([0.0])
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) * 1.5, 0.1)
    
    for _ in range(5):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 50.0:
            hi *= 2.0
        else:
            break
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Run coordinate descent with random order."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        
        b = fast_conv(a)
        sweep_improved = False
        max_b_sweep = b.max()
        
        for i in order:
            if time.monotonic() > deadline:
                break
            x_opt = coordinate_descent_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                old_val = a[i]
                a[i] = x_opt
                # Update affected b values
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - old_val * old_val)
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old_val)
                if b[k_lo:k_hi+1].max() > max_b_sweep:
                    max_b_sweep = b.max()
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-14):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 5
        if no_improve > 30:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.01, 0.03, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various smooth profiles
    inits.append(((t * (1 - t)) + 0.05) / (t * (1 - t) + 0.05).sum())
    inits.append(((1 - 6*t*(1-t)) ** 2 + 0.05) / ((1 - 6*t*(1-t)) ** 2 + 0.05).sum())
    
    # Cosine windows
    for freq in [1, 2, 3, 4]:
        cos_w = 0.5 + 0.3 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.02
    inits.append(gauss / gauss.sum())
    
    # Two-peaked
    two_peak = np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2) + 0.02
    inits.append(two_peak / two_peak.sum())
    
    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.10f} n={self.best.size}", flush=True)

    print(f"PHASE 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    t_phase5_deadline = time.monotonic() + 0.10 * (budget_s - 30)
    if tracker.best is not None:
        n_cur = tracker.best.size
        for mult in [1.25, 0.8, 1.5]:
            if time.monotonic() > t_phase5_deadline:
                break
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 512), 2048)
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 256:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(t_phase5_deadline, time.monotonic() + 20)
            a_opt = lp_optimize(a_ext, max_iters=500, deadline=sub_dl, use_full_lp=False, n_active=40)
            tracker.update(a_opt)
    
    # Phase 6: Final coordinate descent + LP (remaining time)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 10:
        # Coordinate descent
        remaining = deadline - time.monotonic() - 10
        n_final_cd = max(5, int(remaining * 0.5 / max(0.5, tracker.best.size / 40.0)))
        a_final_cd = run_coordinate_descent(tracker.best, n_final_cd, time.monotonic() + remaining * 0.5, seed=99999)
        tracker.update(a_final_cd)
        
        # One more LP pass
        if time.monotonic() < deadline - 10:
            a_final_lp = lp_optimize(tracker.best, max_iters=500, deadline=deadline - 10, use_full_lp=True)
            tracker.update(a_final_lp)
    
    # Final coordinate descent squeeze
    if tracker.best is not None and time.monotonic() < deadline - 5:
        n_squeeze = max(3, int((deadline - 5 - time.monotonic()) / max(0.3, tracker.best.size / 50.0)))
        a_squeeze = run_coordinate_descent(tracker.best, n_squeeze, deadline - 3, seed=12345)
        tracker.update(a_squeeze)
    
    # Ensure we return something valid
    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.10f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
