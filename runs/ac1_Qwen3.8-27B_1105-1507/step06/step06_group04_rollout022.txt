    """
    Design a sequence by specifying the target autocorrelation shape.
    Target: b[k] = exp(-beta * (k-(n-1))^2 / (2*(n/2)^2)) normalized
    """
    # Target autocorrelation
    k = np.arange(2*n-1)
    center = n - 1
    b_target = np.exp(-beta * (k - center)**2 / (2 * (n/2)**2))
    b_target /= np.sum(b_target)
    
    # Compute FFT of target
    size = 1
    while size < 4*n:
        size <<= 1
    B = np.fft.fft(b_target, size)

    # B is real and non-negative (it's the FFT of a non-negative sequence)
    # Actually, B is the FFT of b_target which is real, so B is Hermitian
    # B[k] = B[-k]* (conjugate symmetry)
    # For A^2 = B, we need A[k] = sqrt(B[k]) with appropriate phase
    # The simplest choice: A[k] = sqrt(|B[k]|) * exp(i*angle(B[k])/2)
    
    A = np.sqrt(np.abs(B)) * np.exp(1j * np.angle(B) / 2)

    a = np.maximum(a, 0)
    
    # Normalize
    s = np.sum(a)
    if s < 1e-15:

    else:
        a = a / s
    
    return a

def subgradient_optimize(a, time_budget=60.0, seed=42):
    """Adaptive softmax subgradient with Nesterov momentum."""
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.82
    step = 0.06
    tau = 0.012
    tau_min = 0.0005
    tau_decay = 0.9997

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        step *= 0.9995
        tau = max(tau_min, tau * tau_decay)
        it += 1

        if it % 500 == 499:
            if step > 0.015:
                step *= 0.85
            tau = max(tau_min, tau * 0.8)

    return best_a, best_obj


def swap_sa(a, time_budget=30.0, seed=42):
    """Swap-based simulated annealing for arrangement optimization."""
    n = len(a)
    if n < 4:
        return a.copy(), compute_obj(a)

    a = a.copy()
    if n <= 3000:
        conv = compute_conv(a)
    current_peak = float(np.max(conv))
    best_a = a.copy()
    best_peak = current_peak
    S = float(np.sum(a))

    rng = np.random.default_rng(seed)
    T = max(1e-8, 0.05 * current_peak / n)
    T_min = max(1e-12, 0.001 * current_peak / n)

    start = time.monotonic()
    total_iters = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        ai = a[i]
        aj = a[j]
        if ai == aj:
            continue

        a[i] = aj
        a[j] = ai

        if n <= 3000:
            conv = compute_conv(a)
            new_peak = float(np.max(conv))
        else:
            conv = None
            new_peak = compute_obj(a)

        delta = new_peak - current_peak

        if delta <= 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            current_peak = new_peak
            total_iters += 1
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            a[i] = ai
            a[j] = aj
            total_iters += 1
            no_improve += 1

        T = max(T_min, T * 0.99995)
        if no_improve > 50000:
            if T > T_min:
                T = max(T_min, T * 0.99)
            no_improve = 0

    return best_a, 2.0 * n * best_peak / (S * S)


def two_coord_descent(a, time_budget=30.0, seed=42):
    """Two-coordinate simultaneous descent respecting sum constraint."""
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:

        ai = a[i]
        aj = a[j]
        S = float(np.sum(a))

        b_vals = compute_conv(a)
        current_max = float(np.max(b_vals))

        best_local_t = 0.0
        best_local_max = current_max
        for t in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2]:
            if t >= ai:
                continue
            a_try = a.copy()
            a_try[i] = ai - t

            if a_try[j] < 0:

            b_try = compute_conv(a_try)

            a[j] += best_local_t

def kaiser_sequence(n, alpha):
    """Kaiser-Bessel window."""
    n_half = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64) - n_half
    r = x / max(n_half, 1.0)
    r = np.clip(r, -1.0, 1.0)
    a = i0(alpha * np.sqrt(np.maximum(0.0, 1.0 - r * r)))
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def inverse_design(n, beta, seed=42):
    """Design sequence by specifying target autocorrelation shape."""
    k = np.arange(2 * n - 1, dtype=np.float64)
    center = n - 1
    sigma = n / 2.0
    b_target = np.exp(-beta * (k - center) ** 2 / (2.0 * sigma * sigma))
    b_target = np.maximum(b_target, 0.0)
    s = np.sum(b_target)
    if s < 1e-15:
        return np.ones(n) / n
    b_target /= s

    fft_size = 1
    while fft_size < 4 * n:
        fft_size <<= 1
    B = np.fft.fft(b_target, fft_size)
    B_abs = np.abs(B)
    B_angle = np.angle(B)
    A = np.sqrt(B_abs) * np.exp(0.5j * B_angle)
    a = np.fft.ifft(A).real[:n]
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def build_toeplitz_convolve_mat(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=20):

    A_ub[:, :n] = A
    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

def resample_sequence(a, new_n):

    for alpha in [2.0, 4.0, 6.0, 8.0]:
        starts.append(kaiser_sequence(n, alpha))

    for beta in [0.5, 1.0, 2.0, 4.0, 6.0]:
        starts.append(inverse_design(n, beta))

    for _ in range(5):
        sigma = n * (0.15 + 0.35 * rng.random())
        freq = int(1 + rng.integers(0, 6))
        amp = 0.15 + 0.4 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        bell = np.exp(-((i - center) ** 2) / (2.0 * sigma * sigma))
        mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
        a = np.maximum(bell * mod, 0.0)
        s = np.sum(a)
        if s > 1e-15:
            starts.append(a / s)

    noisy = np.abs(1.0 + rng.normal(0, 0.3, n))
    starts.append(noisy / np.sum(noisy))

    unique = [starts[0]]
    for s in starts[1:]:
        if all(np.linalg.norm(s - u) > 0.03 for u in unique):
            unique.append(s)

    return unique


def multi_scale_optimize(target_n, time_budget, seed=42):
    """Coarse-to-fine multigrid optimization."""
    coarse_n = max(50, target_n // 4)
    scales = sorted(set([coarse_n, max(100, target_n // 2), target_n]))

    a = kaiser_sequence(scales[0], alpha=4.0)
    a, obj = subgradient_optimize(a, time_budget * 0.15, seed)

    for si in range(1, len(scales)):
        new_n = scales[si]
        a = resample_sequence(a, new_n)
        a = normalize_simplex(a)
        remaining = time_budget * (1.0 - si * 0.15)
        t_alloc = min(remaining, time_budget * 0.35)
        a, obj = subgradient_optimize(a, t_alloc, seed + si)

        if new_n <= 2000 and time_budget > 60:
            a, obj2 = swap_sa(a, min(10.0, time_budget * 0.05), seed + si + 100)
            if obj2 < obj:
                obj = obj2
                a = normalize_simplex(a)

    return a, obj
