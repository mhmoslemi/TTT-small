def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

def build_conv_matrix(a):

    return A

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):

                print(f"  Gradient: internal={obj_grad:.8f}, evaluated={ev_grad:.8f}")
                if ev_grad < best_score:
                    best_score = ev_grad
                    best_seq = a_grad.copy()

    # Phase 2: optimize the current best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 2: Refine best (n={len(best_seq)}) ===")
        a_best = best_seq.copy()
        s = float(np.sum(a_best))
        if s > 0:
            a_best = a_best / s

        # More LP iterations
        alloc = min(180.0, remaining * 0.4)
        a_lp2 = alternating_optimize(a_best, time_budget=alloc, verbose=True)
        ev_lp2 = evaluate_sequence([float(x) for x in a_lp2])
        print(f"  LP refine: evaluated={ev_lp2:.8f}")
        if ev_lp2 < best_score:
            best_score = ev_lp2
            best_seq = a_lp2.copy()

        # Gradient on best
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_g2 = gradient_descent_optimize(a_best, time_budget=min(60.0, remaining * 0.3))
            ev_g2 = evaluate_sequence([float(x) for x in a_g2])
            print(f"  Gradient refine: evaluated={ev_g2:.8f}")
            if ev_g2 < best_score:
                best_score = ev_g2
                best_seq = a_g2.copy()

    # Phase 3: Multi-n exploration
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: Multi-n exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(100, n_best - 200), max(100, n_best - 100),
            n_best,
            n_best + 100, n_best + 200,
            200, 300, 500, 700, 900, 1100, 1500, 2000
        ]))

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            if n_try == n_best:
                a_try = best_seq.copy()
            elif n_try < len(best_seq):
                a_try = resample_sequence(best_seq, n_try)
            else:
                a_try = resample_sequence(best_seq, n_try)

            s_try = float(np.sum(a_try))
            if s_try > 0:
                a_try = a_try / s_try

            # Quick optimization
            alloc = min(40.0, remaining * 0.15)
            a_opt = alternating_optimize(a_try, time_budget=alloc)
            ev_try = evaluate_sequence([float(x) for x in a_opt])
            if ev_try < best_score:
                best_score = ev_try
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  ** n={n_try}: score={best_score:.8f} **")
            else:
                print(f"  n={n_try}: score={ev_try:.8f}")

    # Phase 4: Random restarts
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Random restarts ===")
        n_best = len(best_seq)
        restart_beach = min(20.0, remaining * 0.08)

        for r in range(12):
            remaining = deadline - time.monotonic()
            if remaining < restart_beach + 10:
                break

            init_type = r % 4
            if init_type == 0:
                a_r = random_init(n_best, rng)
            elif init_type == 1:
                # Perturbed best
                a_r = best_seq * np.exp(rng.normal(0, 0.1, n_best))
                a_r = np.maximum(a_r, 0)
                sr = float(np.sum(a_r))
                a_r = a_r / sr if sr > 0 else np.ones(n_best) / n_best
            elif init_type == 2:
                n_r = max(200, min(2000, n_best + int(rng.integers(-300, 300))))
                a_r = resample_sequence(best_seq, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
                a_r = np.maximum(a_r, 0)
                sr = float(np.sum(a_r))
                a_r = a_r / sr if sr > 0 else np.ones(n_r) / n_r
            else:
                # Near-constant with small noise
                n_r = n_best
                a_r = np.ones(n_r) + rng.normal(0, 0.01, n_r)
                a_r = np.maximum(a_r, 0)
                a_r = a_r / np.sum(a_r)

            a_opt = alternating_optimize(a_r, time_budget=restart_beach)
            ev_r = evaluate_sequence([float(x) for x in a_opt])
            if ev_r < best_score:
                best_score = ev_r
                best_seq = a_opt.copy()
                print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): score={best_score:.8f} **")
            else:
                print(f"  Restart {r} (type={init_type}): score={ev_r:.8f}")

    # Phase 5: Final intensive polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 5: Final polish (remaining={remaining:.0f}s) ===")
        a_final = best_seq.copy()
        s = float(np.sum(a_final))
        if s > 0:
            a_final = a_final / s

        # Multiple short LP rounds
        for round_idx in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.3, 50.0)
            a_pol = alternating_optimize(a_final, time_budget=alloc)
            ev_pol = evaluate_sequence([float(x) for x in a_pol])
            if ev_pol < best_score:
                best_score = ev_pol
                best_seq = a_pol.copy()
                a_final = a_pol.copy()
                print(f"  Polish round {round_idx}: score={best_score:.8f} **")
            else:
                print(f"  Polish round {round_idx}: score={ev_pol:.8f}")

        # Gradient polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_gp = gradient_descent_optimize(a_final, time_budget=min(remaining * 0.3, 30.0))
            ev_gp = evaluate_sequence([float(x) for x in a_gp])
            if ev_gp < best_score:
                best_score = ev_gp
                best_seq = a_gp.copy()
                print(f"  Gradient polish: score={best_score:.8f} **")
            else:
                print(f"  Gradient polish: score={ev_gp:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
