def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Improved LP-alternating optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: warm-started LP, adaptive golden-section line search,
KKT-informed perturbations, multi-n exploration, and momentum-based peak chasing.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    if warm_start is not None:
        g.value = np.clip(warm_start, 0, None)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def golden_section_line_search(a, g_lp, obj_current, n_points=12):
    """
    Adaptive line search using golden section + local refinement.
    Returns best (a_new, obj, t).
    """
    # First: coarse scan
    ts_coarse = np.linspace(0.05, 1.0, n_points)
    best_obj = obj_current
    best_t = 0.0
    best_a = a.copy()

    for t in ts_coarse:
        a_try = (1.0 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
            best_a = a_try.copy()

    if best_t == 0.0:
        return best_a, best_obj, 0.0

    # Refined: golden section around the best coarse point
    idx = int(np.argmin(np.abs(ts_coarse - best_t)))
    t_lo = ts_coarse[max(0, idx - 1)]
    t_hi = ts_coarse[min(len(ts_coarse) - 1, idx + 1)]

    if t_hi - t_lo < 1e-6:
        return best_a, best_obj, best_t

    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    for _ in range(8):
        t1 = t_hi - gr * (t_hi - t_lo)
        t2 = t_lo + gr * (t_hi - t_lo)

        a1 = (1.0 - t1) * a + t1 * g_lp
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        obj1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf

        a2 = (1.0 - t2) * a + t2 * g_lp
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        obj2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

        if obj1 < obj2:
            t_hi = t2
            if obj1 < best_obj:
                best_obj = obj1
                best_t = t1
                best_a = a1 / s1
        else:
            t_lo = t1
            if obj2 < best_obj:
                best_obj = obj2
                best_t = t2
                best_a = a2 / s2

    return best_a, best_obj, best_t


def kkt_informed_perturb(a, g_lp, n, rng, scale=0.05):
    """
    Perturb based on KKT structure: positions where g_lp has high values
    are "important" directions. Add noise weighted by |a - g_lp|.
    """
    diff = np.abs(a - g_lp)
    weights = 1.0 + 10.0 * diff * n  # More noise where a and g_lp differ
    noise = rng.normal(0, 1, n)
    a_new = a + scale * noise * weights / np.mean(weights)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def chase_peak_refine(a, n_steps=200, time_budget=25.0):
    """
    Chase-the-peak with momentum, targeting top-2 peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.04
    momentum = np.zeros(n)
    beta = 0.85

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Find top-2 peaks
        k_star = int(np.argmax(conv))
        conv_masked = conv.copy()
        conv_masked[k_star] = -np.inf
        k_star2 = int(np.argmax(conv_masked))

        # Subgradient from both peaks
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
            j2 = k_star2 - i
            if 0 <= j2 < n:
                grad[i] += 0.7 * 2.0 * a[j2]

        grad_centered = grad - np.mean(grad)
        momentum = beta * momentum + (1 - beta) * grad_centered

        a_new = a - step_size * momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                step_size *= 0.92
                momentum *= 0.7
        else:
            step_size *= 0.5

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
