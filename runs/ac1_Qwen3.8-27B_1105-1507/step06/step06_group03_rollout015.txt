import numpy as np

from scipy.optimize import minimize


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_obj_grad(b, tau):
    """
    Objective and gradient for log-softmax parameterization.
    a = softmax(b), objective = 2*n*(1/tau)*log(sum exp(tau*c_k))
    where c_k = (a*a)[k].
    Returns (obj, grad).
    """
    a = np.exp(b - np.max(b))
    a = a / a.sum()
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    w = exp_c / exp_c.sum()
    obj = float(2.0 * n * (c_max + np.log(exp_c.sum()) / tau))

    # Gradient: grad_b[i] = 2*a[i] * (sum_k w[k]*a[k-i] - C)
    # where C = sum_k w[k]*c[k]
    C = float(np.dot(w, c))

    # Compute sum_k w[k] * a[k-i] for all i via FFT
    # This is the cross-correlation of w and a
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    W_fft = np.fft.rfft(w, size)
    A_fft = np.fft.rfft(a, size)
    # w flipped: w_flip[k] = w[-k mod (2n-2)]
    w_flip = w[::-1]
    Wf_fft = np.fft.rfft(w_flip, size)
    # conv(w_flip, a)[i] = sum_k w_flip[k]*a[i-k] = sum_k w[2n-2-k]*a[i-k]
    # = sum_j w[j]*a[i-(2n-2-j)] = sum_j w[j]*a[i-2n+2+j]
    # We want sum_k w[k]*a[k-i]
    # Let's use: correlate(w, a)[i] = sum_k w[k]*a[k+i]
    # So sum_k w[k]*a[k-i] = correlate(w, a)[-i] = correlate(w, a)[(2n-2)-i]... 
    
    # Simpler: just compute directly using the convolution
    # sum_k w[k] * a[k-i] where a[m]=0 outside [0,n-1]
    # = (w_conj * a)[i] in some sense
    # Let's use: result[i] = sum_k w[k] * a[k-i]
    # = sum_j w[j+i] * a[j]  (substituting j = k-i)
    # This is the cross-correlation <shift_i(w), a>
    
    # Using FFT: let's just do direct convolution
    # result[i] = sum_k w[k] * a_pad[k-i]
    # where a_pad is a padded to [0, 2n-2]
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    # result[i] = sum_k w[k] * a_pad[k-i] for k=0..2n-2
    # = (w * a_pad_flipped)[i + (2n-2)] ... let me just use correlate
    # np.correlate(w, a_pad, mode='full')[2n-2+i] ... 
    
    # Actually the simplest: result[i] = sum_k w[k]*a_pad[k-i]
    # This is the same as sum_k w[k]*a_pad[(k-i)]
    # Let d = k-i, so k = d+i: sum_d w[d+i]*a_pad[d]
    # This is the cross-correlation of w and a_pad at lag i
    
    w_pad = np.zeros(2 * n - 1)
    w_pad[:2*n-1] = w  # w already has length 2n-1
    # cross_corr[i] = sum_d w[d+i]*a_pad[d]
    # = (w_rev * a_pad)[i + (2n-2)] where w_rev = w[::-1]
    w_rev = w[::-1]
    size2 = 1
    while size2 < 4 * n - 1:
        size2 <<= 1
    Wr_fft = np.fft.rfft(w_rev, size2)
    Ap_fft = np.fft.rfft(a_pad, size2)
    cross = np.fft.irfft(Wr_fft * Ap_fft, size2)
    # cross[m] = sum_k w_rev[k]*a_pad[m-k] = sum_k w[2n-2-k]*a_pad[m-k]
    # = sum_j w[j]*a_pad[m-2n+2+j]  (j = 2n-2-k)
    # We want sum_d w[d+i]*a_pad[d] = sum_j w[j]*a_pad[j-i]  (j=d+i, d=j-i)
    # = cross[j-i + 2n-2]... 
    
    # Let me just do it differently. 
    # I want: g[i] = sum_{k=0}^{2n-2} w[k] * a_ext[k-i]
    # where a_ext[m] = a[m] if 0<=m<n else 0
    # g[i] = sum_{k=i}^{i+n-1} w[k] * a[k-i]  (since a[k-i] nonzero only for 0<=k-i<n)
    
    # This is a valid convolution/correlation. Let me use:
    # g = np.convolve(w, a[::-1], mode='full')  -- no this isn't right either
    
    # Most direct approach using FFT:
    # g[i] = sum_k w[k] * a[k-i]
    # Define a_ext of length 2n-1: a_ext[m] = a[m] for 0<=m<n, 0 for n<=m<2n-1
    # g[i] = sum_{k=0}^{2n-2} w[k] * a_ext[k-i]
    #      = sum_{j=-i}^{2n-2-i} w[j+i] * a_ext[j]   (j = k-i)
    #      = sum_j w[j+i] * a_ext[j]
    # This is the cross-correlation R_wa[i] = sum_j w[j+i]*a_ext[j]
    
    # Using FFT: R_wa[i] = (w_rev * a_ext)[i + (2n-2)]
    # where * denotes full convolution and w_rev = w[::-1]
    
    # cross_full[m] = sum_k w_rev[k] * a_ext[m-k] = sum_k w[2n-1-1-k] * a_ext[m-k]
    # Let j = 2n-2-k: = sum_j w[j] * a_ext[m-2n+2+j]
    # We want R_wa[i] = sum_j w[j+i] * a_ext[j]
    # = sum_j w[j'] * a_ext[j'-i]  (j' = j+i)
    # = cross_full[j'-i + 2n-2]  ... 
    
    # OK I'm overcomplicating this. Let me just use np.dot for each i (slow but correct)
    # or better, use a single convolution correctly.
    
    # The correct formula: g[i] = sum_k w[k] * a_ext[k-i]
    # where a_ext has length 2n-1 (padded with zeros from n to 2n-2)
    # This equals: (w * a_ext_flipped)[i + (2n-2)]  NO
    
    # Let's use: g = np.correlate(w, a_ext, mode='full')
    # np.correlate(a, b, 'full')[k] = sum_n a[n+k] * b[n]
    # So np.correlate(w, a_ext, 'full')[k] = sum_n w[n+k]*a_ext[n]
    # We want g[i] = sum_j w[j]*a_ext[j-i] = sum_n w[n]*a_ext[n-i]
    # = np.correlate(a_ext, w, 'full')[i]  ... let me verify
    # np.correlate(a_ext, w, 'full')[i] = sum_n a_ext[n+i]*w[n]
    # Hmm that's not quite right either.
    
    # Let me just use the simplest correct method:
    # g[i] = sum_{k=0}^{2n-2} w[k] * (a[k-i] if 0<=k-i<n else 0)
    #      = sum_{j=0}^{n-1} w[j+i] * a[j]  (where j = k-i, so k=j+i, and we need j+i <= 2n-2)
    
    # This is: g[i] = sum_j a[j] * w[j+i]
    # This is the cross-correlation of a and w at lag i.
    # In numpy: np.correlate(a, w, 'full') gives:
    #   result[k] = sum_n a[n] * w[n+k-len(w)+1]  ... this is confusing.
    
    # Let me just use FFT directly:
    # g[i] = sum_j a[j] * w[j+i]
    # Define: a_ext2 of length 4n-3: a_ext2[j] = a[j] for 0<=j<n, 0 otherwise
    # w_ext of length 4n-3: w_ext[j] = w[j] for 0<=j<2n-1, 0 otherwise
    # g[i] = sum_j a_ext2[j] * w_ext[j+i]
    # = (a_ext2 * w_ext_flipped)[i + (4n-4)]  ... 
    
    # I'll just compute it using a single FFT-based cross-correlation:
    # g = ifft(fft(a_ext2) * conj(fft(w_ext)))  gives cross-correlation
    # cross[i] = sum_j a_ext2[j] * w_ext[j-i]
    # We want sum_j a_ext2[j] * w_ext[j+i] = cross[-i] = cross[(len)-i]
    
    a_ext2 = np.zeros(4 * n - 3)
    a_ext2[:n] = a
    w_ext = np.zeros(4 * n - 3)
    w_ext[:2*n-1] = w
    
    size3 = 1
    while size3 < 2 * (4 * n - 3):
        size3 <<= 1
    A2_fft = np.fft.rfft(a_ext2, size3)
    We_fft = np.fft.rfft(w_ext, size3)
    cross = np.fft.irfft(A2_fft * np.conj(We_fft), size3)
    # cross[i] = sum_j a_ext2[j] * w_ext[j-i]
    # g[i] = sum_j a[j] * w[j+i] = cross[-i] = cross[(4n-3)*2 - i] ... 
    # Actually cross has length size3, and cross[m] = sum_j a_ext2[j]*w_ext[j-m]
    # g[i] = sum_j a[j]*w[j+i] = sum_j a_ext2[j]*w_ext[j-(-i)] = cross[-i mod size3]
    # For i=0: cross[0] = sum_j a[j]*w[j] = sum_j a[j]*w[j] (correct, g[0] = sum a[j]*w[j])
    # For i=1: g[1] = sum_j a[j]*w[j+1] = cross[-1] = cross[size3-1]
    
    g = np.zeros(n)
    for i in range(n):
        g[i] = cross[(-i) % size3]
    
    grad = 2.0 * a * (g - C)
    return obj, grad


def lbfgs_optimize(a_init, deadline, tau_schedule=None, verbose=False):
    """
    Optimize using L-BFGS on log-softmax parameterization.
    a_init: initial sequence (will be normalized to simplex)
    """
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = compute_J(a)
    
    # Log parameterization
    b0 = np.log(a + 1e-30)
    
    if tau_schedule is None:
        tau_schedule = [50, 100, 200, 400, 800, 1500]
    
    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break
        
        def fun(b):
            obj, grad = smooth_obj_grad(b, tau)
            return obj, grad
        
        try:
            res = minimize(fun, b0, jac=True, method='L-BFGS-B',
                          options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-8,
                                   'maxfun': 2000})
            if res.x is not None:
                a_new = np.exp(res.x - np.max(res.x))
                a_new = a_new / a_new.sum()
                a_new = np.maximum(a_new, 0)
                obj_new = compute_J(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    b0 = res.x.copy()
                else:
                    # Try from current best
                    b0 = np.log(best_a + 1e-30)
        except Exception:
            pass
    
    return best_a, best_obj


def coord_descent_continuous(a, max_sweeps, deadline, random_order=False, seed=0, verbose=False):
    """
    Coordinate descent with golden-section 1D line search.
    Much more effective than discrete candidate search.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 50

    # Golden section ratio
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order and sweep % 2 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved = False

        for i in order:
            old = a[i]
            lo_idx = max(0, i)
            hi_idx = min(2 * n - 2, i + n - 1)
            kq = 2 * i
            a_slice = a[lo_idx:hi_idx + 1].copy()

            def f_1d(x):
                delta = x - old
                conv_new_max = conv.max()
                # Update affected convolution entries
                affected = conv[lo_idx:hi_idx + 1].copy()
                affected += 2.0 * delta * a_slice
                new_local_max = max(affected.max(), np.delete(conv, slice(lo_idx, hi_idx+1)).max() if len(conv) > (hi_idx - lo_idx + 1) else 0)
                if lo_idx <= kq <= hi_idx:
                    new_local_max = max(new_local_max, conv[kq] + 2.0 * delta * a[kq - lo_idx] + delta * delta)
                else:
                    new_local_max = max(new_local_max, conv[kq] + delta * delta)
                new_s = s - old + x
                if new_s < 1e-10:
                    return float('inf')
                return 2.0 * n * new_local_max / (new_s * new_s)

            # For efficiency, use a simpler objective (since s=1, it's just 2n*max)
            # Quick scan to find good bracket
            candidates_x = np.array([0.0, old * 0.1, old * 0.5, old, old * 1.5, old * 2.0, 
                                     old * 3.0 + 0.01, 0.1, 0.01])
            candidates_x = np.clip(candidates_x, 0, 1.0)
            f_vals = np.array([f_1d(x) for x in candidates_x])
            min_idx = np.argmin(f_vals)
            
            # Golden section around the best candidate
            x_best = candidates_x[min_idx]
            f_best = f_vals[min_idx]
            
            # Set bracket
            x_lo = max(0.0, candidates_x[min_idx] * 0.1)
            x_hi = min(1.0, candidates_x[min_idx] * 5.0 + 0.1)
            
            if x_best <= x_lo:
                x_lo = 0.0
            if x_best >= x_hi:
                x_hi = 1.0
            
            # Golden section search
            c = x_hi - gr * (x_hi - x_lo)
            d = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c)
            fd = f_1d(d)
            
            for _ in range(20):
                if fc < fd:
                    x_hi = d
                    d = c
                    fd = fc
                    c = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c)
                else:
                    x_lo = c
                    c = d
                    fc = fd
                    d = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d)
            
            x_gold = (x_lo + x_hi) / 2.0
            f_gold = f_1d(x_gold)
            
            if f_gold < f_best:
                x_best = x_gold
                f_best = f_gold
            
            if x_best != old and abs(x_best - old) > 1e-12:
                delta = x_best - old
                conv[lo_idx:hi_idx + 1] += 2.0 * delta * a_slice
                if lo_idx <= kq <= hi_idx:
                    conv[kq] += delta * delta + 2.0 * (x_best - old - delta) * a[kq - lo_idx]
                    # Already handled by the slice update above if kq is in range
                    # Actually the slice update already includes the 2*delta*a_slice term
                    # We just need to add delta^2 for the kq position
                    # But wait, if kq is in [lo_idx, hi_idx], the slice update already
                    # added 2*delta*a[kq-lo_idx] to conv[kq]. We need to add delta^2 too.
                    conv[kq] += delta * delta
                else:
                    conv[kq] += delta * delta
                a[i] = x_best
                s = s - old + x_best
                # Re-normalize to keep s=1
                a = a / a.sum()
                s = 1.0
                conv = fast_conv(a)
                improved = True

        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    return best_a


def coord_descent_fast(a, max_sweeps, deadline, random_order=False, seed=0, verbose=False):
    """
    Fast coordinate descent with a few continuous line search steps.
    Uses incremental conv updates for speed.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 100

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n) if random_order else range(n)
        improved = False

        for i in order:
            old = a[i]
            if old < 1e-14:
                continue

            lo_idx = max(0, i)
            hi_idx = min(2 * n - 2, i + n - 1)
            kq = 2 * i
            a_slice = a[lo_idx:hi_idx + 1].copy()
            
            # Save conv for restoration
            conv_save = conv[lo_idx:hi_idx + 1].copy()
            kq_in_range = (lo_idx <= kq <= hi_idx)
            conv_kq_save = conv[kq].copy()

            # 1D function: f(x) = 2n * max(conv_modified) / (s - old + x)^2
            # Since s=1: f(x) = 2n * max(conv_mod) / (1 - old + x)^2
            
            def eval_x(x):
                delta = x - old
                conv[lo_idx:hi_idx + 1] = conv_save + 2.0 * delta * a_slice
                if kq_in_range:
                    conv[kq] = conv_kq_save + 2.0 * delta * a[kq - lo_idx] + delta * delta
                else:
                    conv[kq] = conv_kq_save + delta * delta
                new_s = 1.0 - old + x
                if new_s < 1e-10:
                    return float('inf')
                return 2.0 * n * conv.max() / (new_s * new_s)

            # Bracket: try a few points
            f_old = eval_x(old)
            if f_old >= best_obj - 1e-12:
                # Restore
                conv[lo_idx:hi_idx + 1] = conv_save
                conv[kq] = conv_kq_save
                continue

            # Try to find better x via a few evaluations
            ratios = [0.2, 0.5, 0.8, 1.2, 1.5, 2.0]
            best_x = old
            best_f = f_old
            for r in ratios:
                x = old * r
                if x < 1e-14 or x > 1.0:
                    continue
                f_x = eval_x(x)
                if f_x < best_f:
                    best_f = f_x
                    best_x = x

            if best_x != old:
                # Refine with golden section around best_x
                xl = max(0.0, best_x * 0.3)
                xh = min(1.0, best_x * 3.0 + 0.05)
                if best_x <= xl:
                    xl = 0.0
                if best_x >= xh:
                    xh = min(1.0, best_x + 0.1)
                
                c = xh - gr * (xh - xl)
                d = xl + gr * (xh - xl)
                fc = eval_x(c)
                fd = eval_x(d)
                for _ in range(12):
                    if fc < fd:
                        xh = d
                        d = c
                        fd = fc
                        c = xh - gr * (xh - xl)
                        fc = eval_x(c)
                    else:
                        xl = c
                        c = d
                        fc = fd
                        d = xl + gr * (xh - xl)
                        fd = eval_x(d)
                
                x_gold = (xl + xh) / 2.0
                f_gold = eval_x(x_gold)
                if f_gold < best_f:
                    best_x = x_gold
                    best_f = f_gold

                if abs(best_x - old) > 1e-13:
                    # Apply the change
                    conv[lo_idx:hi_idx + 1] = conv_save + 2.0 * (best_x - old) * a_slice
                    if kq_in_range:
                        conv[kq] = conv_kq_save + 2.0 * (best_x - old) * a[kq - lo_idx] + (best_x - old)**2
                    else:
                        conv[kq] = conv_kq_save + (best_x - old)**2
                    a[i] = best_x
                    # Renormalize
                    s_new = a.sum()
                    a = a / s_new
                    conv = fast_conv(a)
                    improved = True

            # Restore conv
            conv[lo_idx:hi_idx + 1] = conv_save
            conv[kq] = conv_kq_save
            if improved:
                # conv was already updated above
                pass
            else:
                conv[lo_idx:hi_idx + 1] = conv_save
                conv[kq] = conv_kq_save

        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences on the simplex."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window (Hann)
    inits.append(project_simplex(0.5 * (1 + np.cos(2 * np.pi * t))))

    # Raised cosine variants
    for beta in [0.5, 0.7, 0.9]:
        inits.append(project_simplex(0.5 * (1 + beta * np.cos(2 * np.pi * t))))

    # Flat-top
    inits.append(project_simplex(np.sinc(2 * (t - 0.5))))

    # Gaussian
    for sigma in [0.1, 0.15, 0.2, 0.25, 0.3]:
        inits.append(project_simplex(np.exp(-((t - 0.5) / sigma) ** 2 * 2)))

    # Power law
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        inits.append(project_simplex(1.0 + alpha * (t - 0.5) ** 2))
        inits.append(project_simplex(1.0 - alpha * (t - 0.5) ** 2 + 0.1))

    # Multi-band
    for f1, f2 in [(2, 5), (3, 7), (1, 4)]:
        inits.append(project_simplex(1.0 + 0.3 * np.cos(2*np.pi*f1*t) + 0.2 * np.cos(2*np.pi*f2*t)))

    # Smooth random
    ksize = max(7, n // 12)
    kernel = np.exp(-0.5 * (np.arange(-ksize, ksize+1) / (ksize*0.5))**2)
    kernel /= kernel.sum()
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.1))

    # Alternating (for odd/even structure)
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits.append(project_simplex(alt))

    return inits


def solve_pipeline(a_init, deadline, verbose=False):
    """
    Full optimization pipeline:
    1. Fast coordinate descent
    2. L-BFGS with multiple tau
    3. Continuous coordinate descent (polish)
    4. L-BFGS at fine tau (final polish)
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new, label=""):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj - 1e-12:
            best_a = a_new.copy()
            best_obj = obj
            if verbose:
                print(f"      {label}: J={obj:.8f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 1: Fast coordinate descent (35% time)
    t1 = time.monotonic() + 0.35 * remaining
    if verbose:
        print(f"    CD-fast ({t1 - time.monotonic():.0f}s)", flush=True)
    a_cd = coord_descent_fast(a, max_sweeps=50000, deadline=t1, random_order=True, seed=42)
    try_update(a_cd, "CD-fast")
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 2: L-BFGS (30% time)
    t2 = time.monotonic() + 0.30 * remaining
    if verbose:
        print(f"    L-BFGS ({t2 - time.monotonic():.0f}s)", flush=True)
    a_lb, obj_lb = lbfgs_optimize(a, deadline=t2, tau_schedule=[30, 60, 120, 250, 500])
    try_update(a_lb, "L-BFGS")
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 3: Continuous coordinate descent (20% time)
    t3 = time.monotonic() + 0.20 * remaining
    if verbose:
        print(f"    CD-cont ({t3 - time.monotonic():.0f}s)", flush=True)
    a_cc = coord_descent_continuous(a, max_sweeps=5000, deadline=t3, random_order=True, seed=77)
    try_update(a_cc, "CD-cont")
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 4: Final L-BFGS polish (15% time)
    t4 = time.monotonic() + 0.15 * remaining
    if verbose:
        print(f"    L-BFGS fine ({t4 - time.monotonic():.0f}s)", flush=True)
    a_lbf, obj_lbf = lbfgs_optimize(a, deadline=t4, tau_schedule=[100, 300, 800])
    try_update(a_lbf, "L-BFGS-fine")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 5: Final CD polish
    t5 = deadline - 5
    a_final = coord_descent_fast(best_a, max_sweeps=10000, deadline=t5, random_order=True, seed=999)
    try_update(a_final, "Final-CD")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses L-BFGS on log-softmax parameterization, continuous coordinate descent,
    diverse initializations, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Optimize at parent length with diverse inits ===
    phase1_deadline = t_start + 400.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-init at n={n_parent} (budget {phase1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    n_init = min(len(inits), 10)
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, n_init) - 5

    for idx in range(n_init):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 30:
            break
        alloc = min(per_init_time, remaining_check - 5)
        if alloc < 20:
            break

        a0 = inits[idx].copy()
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt, obj_opt = solve_pipeline(a0, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.8f} n={n_parent}", flush=True)

    # === PHASE 2: Try different lengths ===
    phase2_deadline = min(t_start + 900.0, deadline - 200.0)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration (budget {phase2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        candidate_ns = [512, 640, 768, 896, 1024, 1152, 1280]
        candidate_ns = [n for n in candidate_ns if n != n_parent and n <= 2048 and n >= 256]
        candidate_ns.sort(key=lambda x: abs(x - n_parent))

        for new_n in candidate_ns[:4]:
            remaining_check = phase2_deadline - time.monotonic()
            if remaining_check < 60:
                break
            alloc = min(120.0, remaining_check - 20)
            if alloc < 40:
                break

            print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
            a_ext = resample_seq(best_seq, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            a_opt, obj_opt = solve_pipeline(a_ext, time.monotonic() + alloc, verbose=False)
            update_best(a_opt)
            print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive local optimization on best ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive refinement (budget {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            a_curr = best_seq.copy()

            # Multiple L-BFGS passes with different tau schedules
            lbfgs_schedules = [
                [20, 40, 80, 160, 320],
                [50, 100, 200, 400],
                [15, 30, 60, 120],
            ]
            
            lbfgs_deadline = time.monotonic() + 0.4 * remaining
            per_lbfgs = (lbfgs_deadline - time.monotonic()) / len(lbfgs_schedules)
            
            for si, sched in enumerate(lbfgs_schedules):
                if time.monotonic() > lbfgs_deadline:
                    break
                alloc = min(per_lbfgs, lbfgs_deadline - time.monotonic() - 5)
                if alloc < 15:
                    break
                a_lb, obj_lb = lbfgs_optimize(best_seq, deadline=time.monotonic() + alloc, tau_schedule=sched)
                update_best(a_lb)
                print(f"  L-BFGS pass {si}: J={obj_lb:.8f}", flush=True)

            remaining = deadline - time.monotonic()
            if remaining > 30:
                # Final coordinate descent polish
                cd_deadline = time.monotonic() + 0.4 * remaining
                a_cd = coord_descent_fast(best_seq, max_sweeps=100000, deadline=cd_deadline,
                                          random_order=True, seed=12345)
                update_best(a_cd)
                print(f"  Final CD: J={compute_J(a_cd):.8f}", flush=True)

            remaining = deadline - time.monotonic()
            if remaining > 20:
                # One more fine L-BFGS
                a_final_lb, obj_final_lb = lbfgs_optimize(best_seq, deadline=time.monotonic() + remaining - 5,
                                                          tau_schedule=[500, 1000, 2000])
                update_best(a_final_lb)
                print(f"  Final L-BFGS: J={obj_final_lb:.8f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
