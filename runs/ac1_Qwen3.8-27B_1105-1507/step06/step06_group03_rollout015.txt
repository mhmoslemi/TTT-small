import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

Multi-strategy optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Combines: (1) multiplicative mirror descent with exact max subgradient,
(2) Top-K SCA-LP with warm-started HiGHS solver,
(3) coordinate descent with adaptive candidates,
(4) spectral flatness iteration, (5) multi-length exploration.
Key improvements over previous: top-K constraints in LP, multiplicative updates,

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def mirror_descent(a, max_iters, deadline, lr_init=0.1, verbose=False):
    """
    Multiplicative mirror descent using the subgradient of the exact max.
    Update: a_new[i] = a[i] * exp(-lr * g[i]) / Z
    where g[i] = 2*a_reversed_shifted[i] at the argmax of conv.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = np.argmax(c)

        # Subgradient: grad[i] = 2*a[k_star - i] (interior) or a[k_star-i] (boundary)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    grad[i] = a[i]  # boundary case
                else:
                    grad[i] = 2.0 * a[j]

        # Normalize gradient to have unit mean (for scale invariance)
        g_mean = grad.mean()
        grad_centered = grad - g_mean

        # Multiplicative update with adaptive step
        expo = -lr * grad_centered
        expo -= expo.max()  # numerical stability
        weights = np.exp(expo)
        a_new = a * weights
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            a_new = a.copy()
        else:
            a_new /= a_new.sum()

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slowly decay learning rate
            lr = max(0.001, lr * 0.999)
        else:
            no_improve += 1
            # Reduce learning rate on failure
            lr *= 0.8
            if no_improve > 100:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                a = best_a + rng.normal(0, 0.01, n)
                a = project_simplex(a)
                lr = lr_init * 0.5
                no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    Mirror it {it}: lr={lr:.5f} obj={best_obj:.8f}", flush=True)

        if no_improve > 200:

    return best_a, best_obj

        else:
            # Try blending
            found = False
            for frac in [0.3, 0.2, 0.1, 0.05]:
                a_try = (1 - frac) * a + frac * a_new
                a_try = np.maximum(a_try, 0)
                if a_try.sum() > 1e-10:
                    a_try /= a_try.sum()
                    if compute_J(a_try) < best_obj - 1e-14:
                        a = a_try
                        best_a = a.copy()
                        no_improve = 0
                        found = True
                        break
            if not found:
                no_improve += 1

        if verbose and it % 20 == 0:
            print(f"    Spectral it {it}: beta={beta:.3f} obj={best_obj:.8f}", flush=True)

        if no_improve > 40:
            break

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    inits.append(np.ones(n) / n)

    # Cosine families
    for freq in [1, 2, 3]:
        inits.append(project_simplex(1.0 + 0.5 * np.cos(2 * np.pi * freq * t)))
        inits.append(project_simplex(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + 0.5)))

    # Gaussian
    for center in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.25, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.02))

    # Power law
    for alpha in [0.5, 1.5, 2.5]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    # Bump (high in middle)
    inits.append(project_simplex(1.0 - 0.5 * (2 * t - 1) ** 2))
    inits.append(project_simplex(np.cos(np.pi * (t - 0.5)) ** 2 + 0.01))

    return inits


def solve_one(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence:
    1. Coordinate descent
    2. Mirror descent
    3. Top-K SCA-LP
    4. Spectral iteration
    5. Final polish
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return a
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 1: Quick coordinate descent (20%)
    t1 = time.monotonic() + 0.20 * remaining
    a = coord_descent(a, max_sweeps=5000, deadline=t1)
    a = try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 2: Mirror descent (20%)
    t2 = time.monotonic() + 0.20 * remaining
    a_md, obj_md = mirror_descent(a, max_iters=3000, deadline=t2, lr_init=0.05)
    if obj_md < best_obj:
        best_a = a_md.copy()
        best_obj = obj_md
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 3: Top-K SCA-LP (35%)
    t3 = time.monotonic() + 0.35 * remaining
    # Try multiple K values
    for K in [10, 20, 50]:
        if time.monotonic() > t3 - 5:
            break
        a_lp, obj_lp = topk_sca_lp(a, K=K, max_iters=100, deadline=t3)
        if obj_lp < best_obj:
            best_a = a_lp.copy()
            best_obj = obj_lp
        a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 4: Spectral iteration (15%)
    t4 = time.monotonic() + 0.15 * remaining
    a_sp, obj_sp = spectral_iteration(a, max_iters=500, deadline=t4)
    if obj_sp < best_obj:
        best_a = a_sp.copy()
        best_obj = obj_sp
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 5: Final coordinate descent polish (10%)
    t5 = deadline - 3
    a = coord_descent(a, max_sweeps=10000, deadline=t5)
    a = try_update(a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a).
    Uses: coordinate descent, mirror descent, top-K SCA-LP, spectral iteration.
    Explores multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        update_best(parent)

    # === PHASE 1: Optimize parent length (500s) ===
    n_parent = len(parent)
    phase1_deadline = t_start + 500.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Optimizing at n={n_parent} (budget {phase1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    n_init = min(len(inits), 8)
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, n_init) - 5

    for idx in range(n_init):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 40:
            break
        alloc = min(per_init_time, remaining_check - 10)
        if alloc < 25:
            break

        a0 = inits[idx].copy()
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt, obj_opt = solve_one(a0, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Multi-length exploration (400s) ===
    phase2_deadline = t_start + 950.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Try lengths that might be better
    candidate_ns = [512, 640, 896, 1024, 1152, 1280]
    candidate_ns = [n for n in candidate_ns if n != n_parent]
    candidate_ns.sort()

    for new_n in candidate_ns:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 80:
            break
        alloc = min(120.0, remaining_check - 30)
        if alloc < 50:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = solve_one(a_ext, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive polish on best (remaining time) ===
    phase3_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive polish (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            a_curr = best_seq.copy()

            # Pass 1: Long CD
            t_p1 = time.monotonic() + 0.20 * remaining
            a_curr = coord_descent(a_curr, max_sweeps=30000, deadline=t_p1, seed=111)
            update_best(a_curr)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass 2: Mirror descent extended
                t_p2 = time.monotonic() + 0.20 * remaining
                a_md, obj_md = mirror_descent(a_curr, max_iters=5000, deadline=t_p2, lr_init=0.03)
                update_best(a_md)
                a_curr = best_seq.copy()

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass 3: Top-K LP with various K
                t_p3 = time.monotonic() + 0.25 * remaining
                for K in [5, 10, 20, 30]:
                    if time.monotonic() > t_p3 - 10:
                        break
                    a_lp, obj_lp = topk_sca_lp(a_curr, K=K, max_iters=150, deadline=t_p3)
                    update_best(a_lp)
                    a_curr = best_seq.copy()

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass 4: Spectral
                t_p4 = time.monotonic() + 0.15 * remaining
                a_sp, obj_sp = spectral_iteration(a_curr, max_iters=1000, deadline=t_p4)
                update_best(a_sp)
                a_curr = best_seq.copy()

            remaining = phase3_deadline - time.monotonic()
            if remaining > 20:
                # Pass 5: Final CD
                t_p5 = phase3_deadline - 5
                a_curr = coord_descent(a_curr, max_sweeps=20000, deadline=t_p5, seed=222)
                update_best(a_curr)

                # One more LP at the end
                if time.monotonic() < phase3_deadline - 15:
                    a_lp, obj_lp = topk_sca_lp(best_seq, K=15, max_iters=50,
                                                deadline=phase3_deadline - 3)
                    update_best(a_lp)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        update_best(best_seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
