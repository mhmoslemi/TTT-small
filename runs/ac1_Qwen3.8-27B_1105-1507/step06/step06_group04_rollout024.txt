But A(f) = sum_j a_j e^{-2πifj/n}, and A(0) = sum_j a_j = 1. So we want the sequence to be close to a DC signal (uniform), but with small ripple. However, since a_j >= 0, the most "DC-like" non-negative sequence with sum 1 is the uniform one, which gives the triangular convolution.

The issue is that the uniform sequence gives a TRIANGULAR convolution (not flat). The triangle has a peak of 1/n at the center. To make it flatter, we need to "notch out" the center of the convolution.

The convolution at the center (k = n-1) is:
conv[n-1] = sum_j a_j * a[n-1-j]

This is the "middle" coefficient. To reduce it while keeping the sum constant, we want to make a_j * a_{n-1-j} small in aggregate. This suggests making the sequence "anti-symmetric" in some sense, but since all values are non-negative, we can't do true antisymmetry.

Instead, we want a_j to be large where a_{n-1-j} is small, and vice versa. This suggests a sequence that is "lopsided" - concentrated on one side.

But if we concentrate on one side, the convolution peak moves to a different location and might be even higher (since the convolution of a concentrated signal with itself is more concentrated).

The optimal solution is somewhere in between: spread out enough to avoid a high peak, but with enough structure to cancel out the convolution at the peak locations.

This is essentially the problem of finding a "flat" convolution, which is related to:
- Complementary sequences (Golay sequences)
- Low-autocorrelation sequences
- Fats and zero-convolution sequences

For the non-negative constraint, this is more restrictive than the general case.

Let me think about a specific approach: **iterative water-filling**.

The idea:
1. Start with a uniform sequence.
2. Compute the convolution.
3. Find the peak location k*.
4. The gradient tells us which a_j contribute most to conv[k*]: those with large a_{k*-j}.
5. Reduce those a_j and increase others (those that don't contribute to peaks).
6. Repeat.

This is essentially what the subgradient method does, but we can make it more effective by:
- Using the exact gradient (log-sum-exp)
- Using a smart step size (backtracking line search)
- Proactively identifying and reducing contributions to ALL peaks, not just the maximum

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """FFT-based linear convolution a*a."""
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute 2n*max(conv)/(sum)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (s * s)

def normalize(a):
    """Project onto probability simplex."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lse_gradient(a, beta):
    """Compute exact gradient of log-sum-exp(beta * conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    shifted = beta * (conv - np.max(conv))
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        idx = np.argmax(conv)
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= wsum
    
    # Gradient: grad[i] = 2 * sum_j a[j] * weights[i+j]
    # = 2 * cross_correlation(a, weights) at lag i
    fft_size = 1
    while fft_size < 2*n-1:
        fft_size <<= 1
    w_padded = np.zeros(fft_size)
    w_padded[:len(weights)] = weights
    W = rfft(w_padded, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    grad = 2.0 * corr[:n]
    return grad, np.max(conv), weights

                  lr_start=0.1, momentum=0.9, time_budget=60):
    """
    Nesterov accelerated gradient on log-sum-exp smoothed objective.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    mu = momentum
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive beta
        frac = it / max(max_iters, 1)
        beta = beta_start * (beta_end / beta_start) ** frac

def backtracking_lse(a, max_iters=3000, beta=1000, lr0=0.05, 
                     c=0.5, sigma=0.5, time_budget=60):
    """
    Gradient descent with backtracking line search on LSE-smoothed objective.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        grad, cmax, _ = lse_gradient(a, beta)
        grad_c = grad - np.mean(grad)
        gn = np.linalg.norm(grad_c)
        if gn < 1e-15:
            break
        
        direction = -grad_c / gn
        
        # Backtracking line search
        t = lr
        f0 = cmax  # Use max(conv) as surrogate (scales with objective)
        improved = False
        for _ in range(20):
            a_try = a + t * direction
            a_try = normalize(a_try)
            f_try, _, _ = lse_gradient(a_try, beta)
            f_try_val = np.max(compute_conv(a_try))
            if f_try_val <= f0 - c * t * np.dot(grad, direction) / gn:
                a = a_try
                improved = True
                break
            t *= sigma
        
        if not improved:
            a = a + t * direction
            a = normalize(a)
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        lr = t  # Warm start
    
    return best_a, best_obj

    return inits

import numpy as np

def compute_conv(a):
    """FFT-based linear convolution a*a."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_exact_gradient(a, beta):
    """
    Compute exact gradient of h(a) = (1/beta)*log(sum_k exp(beta*conv_k(a))).
    Returns (gradient, max_conv, softmax_weights).
    Gradient of h w.r.t. a_i = 2 * sum_j a_j * p_{i+j} where p = softmax(beta*conv).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -600.0, 600.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= wsum

    # Gradient: grad[i] = 2 * sum_j a[j] * weights[i+j]
    # = 2 * cross_correlation(a, weights) at lag i
    # Using FFT: ifft(fft(a, N) * conj(fft(weights_padded, N)))[i] = sum_j a[j]*weights[i+j]
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    w_pad = np.zeros(fft_size)
    w_pad[:len(weights)] = weights
    W = rfft(w_pad, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    grad = 2.0 * corr[:n]
    return grad, cmax, weights


def multi_peak_gradient(a, num_peaks, beta):
    """
    Compute gradient using top-K peak weighting instead of full softmax.
    More aggressive peak reduction.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    # Get top-K peaks with exponential weighting
    sorted_idx = np.argsort(conv)[::-1]
    top_k = sorted_idx[:min(num_peaks, len(conv))]
    weights = np.zeros(len(conv))
    for rank, k in enumerate(top_k):
        weights[k] = np.exp(-0.5 * rank)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= wsum

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    w_pad = np.zeros(fft_size)
    w_pad[:len(weights)] = weights
    W = rfft(w_pad, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    grad = 2.0 * corr[:n]
    return grad, cmax


def nesterov_lse(a, max_iters=5000, beta_start=100.0, beta_end=5000.0,
                  lr_start=0.08, momentum=0.92, time_budget=30.0):
    """
    Nesterov accelerated gradient descent on log-sum-exp smoothed objective.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = momentum

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(max_iters, 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        grad, cmax, _ = lse_exact_gradient(a, beta)
        grad_c = grad - np.mean(grad)
        gn = np.linalg.norm(grad_c)

            break

        a = a_new

def backtracking_descent(a, max_iters=3000, beta=1000.0, lr0=0.04,
                         c=0.3, sigma=0.6, time_budget=30.0):
    """
    Gradient descent with backtracking line search on LSE-smoothed objective.
    Uses max(conv) as the surrogate for the non-smooth part.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, cmax, _ = lse_exact_gradient(a, beta)
        grad_c = grad - np.mean(grad)
        gn = np.linalg.norm(grad_c)
        if gn < 1e-15:
            break

        direction = -grad_c / gn
        f0 = cmax

        # Backtracking
        t = lr
        accepted = False
        for _ in range(15):
            a_try = a + t * direction
            a_try = normalize_simplex(a_try)
            f_try = float(np.max(compute_conv(a_try)))
            if f_try <= f0 - c * t * np.dot(grad, direction) / gn:
                a = a_try
                accepted = True
                break
            t *= sigma

        if not accepted:
            a = a + t * direction
            a = normalize_simplex(a)

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        lr = max(t, 0.001)

    return best_a, best_obj

def minpeak_lp_one_round(a, time_limit=15):
    """One round of min-peak LP: fix a, find g minimizing max(a*g)."""
    n = len(a)
    if n > 600:
        return None

    # Build Toeplitz matrix for a*g
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A_mat[lo:hi, j] = a[:hi - lo]

    # min t s.t. A_mat @ g <= t * 1, sum(g) = 1, g >= 0
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_mat
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)

    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            gs = np.sum(g)
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

                if t_alloc < 3:
                    break
                a_p, obj_p = nesterov_lse(
                    a_pert, max_iters=int(t_alloc * 15),
                    beta_start=200, beta_end=5000,
                    lr_start=0.06, momentum=0.92, time_budget=t_alloc
                )
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"    Perturb {pi}: ** {obj_p:.8f} **")

    # Phase 3: Final intensive on the best
    print(f"\n=== Phase 3: Final intensive (best: {best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Very long Nesterov with fine-tuned parameters
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc = min(remaining * 0.35, 80.0)
            print(f"  Long Nesterov LSE (alloc={alloc:.0f}s)")
            a_f, obj_f = nesterov_lse(
                best_seq, max_iters=int(alloc * 30),
                beta_start=500, beta_end=30000,
                lr_start=0.04, momentum=0.94, time_budget=alloc
            )
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"    ** {obj_f:.8f} **")

        # Multi-peak with more peaks
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.25, 50.0)
            print(f"  Multi-peak chase (alloc={alloc:.0f}s)")
            a_f, obj_f = nesterov_multi_peak(
                best_seq, max_iters=int(alloc * 20), num_peaks=8,
                beta_start=500, beta_end=20000,
                lr_start=0.04, momentum=0.94, time_budget=alloc
            )
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"    ** {obj_f:.8f} **")

        # Backtracking on the best
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.2, 30.0)
            print(f"  Backtracking (alloc={alloc:.0f}s)")
            a_f, obj_f = backtracking_descent(
                best_seq, max_iters=int(alloc * 15), beta=5000,
                lr0=0.02, time_budget=alloc
            )
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"    ** {obj_f:.8f} **")

        # LP polish if n is small enough
        remaining = deadline - time.monotonic()
        if remaining > 20 and n_best <= 550:
            alloc = min(remaining * 0.3, 30.0)
            print(f"  LP polish (alloc={alloc:.0f}s)")
            a_f, obj_f = lp_polish(best_seq, time_budget=alloc)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"    ** {obj_f:.8f} **")

        # Try resampling to slightly different lengths
        remaining = deadline - time.monotonic()
        if remaining > 30:
            for dn in [-30, 30, -60, 60]:
                if time.monotonic() > deadline - 20:
                    break
                n_try = n_best + dn
                if n_try < 300 or n_try > 1300:
                    continue
                a_res = normalize_simplex(resample_to_length(best_seq, n_try))
                t_alloc = min(remaining * 0.15, 15.0)
                if t_alloc < 3:
                    break
                a_r, obj_r = nesterov_lse(
                    a_res, max_iters=int(t_alloc * 15),
                    beta_start=300, beta_end=10000,
                    lr_start=0.05, momentum=0.93, time_budget=t_alloc
                )
                if obj_r < best_score:
                    best_score = obj_r
                    best_seq = a_r.copy()
                    print(f"  Resample n={n_try}: ** {obj_r:.8f} **")

        # Final burst of perturbations
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_f = len(best_seq)
            rng = np.random.default_rng(seed + 77777)
            for pi in range(8):
                if time.monotonic() > deadline - 10:
                    break
                scale = 0.01 + 0.005 * pi
                a_p = best_seq * np.exp(rng.normal(0, scale, n_f))
                a_p = normalize_simplex(a_p)
                t_alloc = min(remaining * 0.08, 8.0)
                if t_alloc < 2:
                    break
                a_p, obj_p = nesterov_lse(
                    a_p, max_iters=int(t_alloc * 20),
                    beta_start=500, beta_end=8000,
                    lr_start=0.05, momentum=0.92, time_budget=t_alloc
                )
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Final perturb {pi} (s={scale:.3f}): ** {obj_p:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(500)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
