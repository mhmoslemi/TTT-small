import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz_matrix(a):
    """Build sparse Toeplitz A where (A@g)[k] = (conv(a,g))[k]."""
    n_a = len(a)
    n_g = n_a
    m = 2 * n_a - 1
    rows = []
    cols = []
    vals = []
    for j in range(n_g):
        for i in range(j, min(j + n_a, m)):
            rows.append(i)
            cols.append(j)
            vals.append(a[i - j])
    return csr_matrix((vals, (rows, cols)), shape=(m, n_g))


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def quadratic_step_size(obj_func, t0=0.0, t1=1.0):
    """Find optimal step using quadratic interpolation from 3 points."""
    f0 = obj_func(t0)
    f_mid = obj_func(0.5 * (t0 + t1))
    f1 = obj_func(t1)
    # Fit quadratic: f(t) = at^2 + bt + c
    # Use the three points to solve
    h = t1 - t0
    # Parabolic interpolation
    if abs(f1 - f0) < 1e-12:
        return 0.5 * (t0 + t1)
    # Standard parabolic interpolation formula
    denom = (f0 - 2.0 * f_mid + f1)
    if abs(denom) < 1e-12:
        return 0.5 * (t0 + t1)
    t_opt = t0 + h * 0.5 * (f0 - f1) / denom
    t_opt = max(t0, min(t1, t_opt))
    return t_opt

def nesterov_subgradient(a, max_steps=300, time_budget=50.0):
    """
    Nesterov-accelerated subgradient descent on the normalized objective.
    f(a) = 2n * max_k (a*a)[k] / (sum(a))^2
    Subgradient at peak k*: grad[i] = 4n*(a[k*-i]*S - M) / S^3 (where a[k*-i]=0 if out of range)
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)  # momentum accumulator
    y = a.copy()     # Nesterov lookahead point
    beta = 0.9      # momentum parameter
    step_size = 0.02

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        # Compute subgradient at y
        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))
        M = float(conv_y[k_star])
        S = float(np.sum(y))

        # Multi-peak subgradient: weight all peaks above 0.95*max
        threshold = 0.95 * M
        peak_indices = np.where(conv_y >= threshold)[0]
        peak_weights = (conv_y[peak_indices] - threshold + 0.01)  # positive weights

        grad = np.zeros(n)
        for k2 in peak_indices:
            w = peak_weights[k2 - peak_indices[0]] if len(peak_indices) > 1 else 1.0
            # Subgradient contribution from peak at k2
            sg = np.zeros(n)
            for j in range(n):
                idx = k2 - j
                if 0 <= idx < n:
                    sg[j] += 2.0 * y[idx]
            grad += w * sg

        # Normalize subgradient
        w_sum = sum(peak_weights) if len(peak_weights) > 0 else 1.0
        grad /= max(w_sum, 1e-10)

        # Full subgradient of f(a) = 2n*M/S^2
        # df/da[i] = 2n * (sg[i]*S - 2*a[i]*M) / S^3
        grad_full = 2.0 * n * (grad * S - 2.0 * y * M) / (S ** 3)

        # Project onto sum=1 hyperplane (remove mean component)
        grad_centered = grad_full - np.mean(grad_full)

        # Nesterov update
        t_k = 2.0 / (step + 2.0 + beta)
        x_old = a.copy()
        v = beta * v + (1.0 - beta) * grad_centered
        a_new = normalize(y - step_size * v)

        # Update y for next Nesterov iteration
        y = a_new + (t_k / (t_k_prev if step > 0 else 1.0) - 1.0) * (a_new - x_old) if step > 0 else a_new
        t_k_prev = t_k

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.7
            y = a.copy()
            if step_size < 1e-8:
                break

    return best_a


def spectral_equalization_aug(a, max_iters=80, time_budget=40.0):
    """
    Augmented spectral equalization: flatten spectrum with adaptive
    compression and careful non-negativity handling via soft thresholding.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Adaptive compression: start aggressive, ease off
        progress = it / float(max_iters)
        alpha = 0.3 + 0.5 * progress  # from 0.3 to 0.8

        # Compress spectrum (bring high peaks down, low peaks up)
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        # Preserve DC component exactly
        P_new[0] = P[0]

        # Reconstruct
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        # Soft non-negativity: use exp(softplus) projection
        # Instead of hard clip, use a_n = max(a_n, eps) with small eps, then renormalize
        eps = 1e-8
        a_new = np.maximum(a_new, eps)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)

        # Adaptive blending based on improvement
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            # Take a larger step when improving
            blend = 0.7 + 0.3 * progress
            a = blend * a_new + (1.0 - blend) * a
            a = normalize(a)
        else:
            # Blend conservatively
            a = 0.6 * a + 0.4 * a_new
            a = normalize(a)

    return best_a


def direct_peak_reduction(a, time_budget=30.0):

        # Find contributors to the peak

        for j in range(n):

            if 0 <= k2 < n:

        contrib.sort(reverse=True)

        # Find low-autocorrelation positions for redistribution
        # Positions where adding mass won't create new peaks
        lag_contrib = np.zeros(n)
        for j in range(n):
            for k in range(max(0, j - (n - 1)), min(j + n, 2 * n)):

                if 0 <= k2 < n:
                    lag_contrib[j] += a[k2]

        # Try moving mass from top contributors to low-impact positions
        improved = False
        for peak_val_c, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            # Target: positions with low contribution to any autocorrelation
            candidates = np.argsort(lag_contrib)[:n // 4]
            for j_target in candidates[:5]:

                    a_try = normalize(a_try)

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    # Use spline-like interpolation for smoother results
    from scipy.interpolate import PchipInterpolator
    try:
        interp = PchipInterpolator(x_old, a)
        result = interp(x_new)
        result = np.maximum(result, 0)
    except Exception:
        result = np.interp(x_new, x_old, a)
    return result

        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def quick_optimize(a, time_budget=20.0):
    """Quick optimization using a mix of methods."""
    n = len(a)
    start = time.monotonic()

    # Quick LP
    if n <= 900 and time_budget > 10:
        a = lp_alternate_improved(a, max_iters=8, time_budget=min(time_budget * 0.4, 15.0))

    # Nesterov subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a = nesterov_subgradient(a, max_steps=100, time_budget=min(remaining * 0.5, 15.0))

    # Spectral
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_spec = spectral_equalization_aug(a, max_iters=30, time_budget=min(remaining * 0.4, 12.0))
        obj_a = compute_obj(a)
        obj_spec = compute_obj(a_spec)
        if obj_spec < obj_a:
            a = a_spec

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function with multi-phase strategy focused on:
    1. Intensive LP optimization at the parent length
    2. Nesterov subgradient for fast progress
    3. Systematic n-exploration
    4. Diverse restarts with peak-aware perturbations
    5. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"\nInitial: n={n_parent}, score={best_score:.8f}")

    # =========================================================
    # Phase 1: Intensive LP optimization at parent length
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Intensive LP optimization (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        for cycle in range(6):
            if time.monotonic() > deadline - 250:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.15, 70.0)

            a_new = lp_alternate_improved(a, max_iters=30, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  LP cycle {cycle}: {best_score:.8f}")
            else:
                a = a_new
                print(f"  LP cycle {cycle}: {obj_new:.8f} (no improvement)")

    # =========================================================
    # Phase 2: Nesterov + Spectral optimization
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Nesterov subgradient + Spectral equalization")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())

        if time.monotonic() < deadline - 200:
            a_nes = nesterov_subgradient(a, max_steps=500, time_budget=40.0)
            obj_nes = compute_obj(a_nes)
            if obj_nes < best_score:
                best_score = obj_nes
                best_seq = a_nes.copy()
                a = a_nes
                print(f"  Nesterov: {best_score:.8f}")
            else:
                print(f"  Nesterov: {obj_nes:.8f} (no improvement)")

        if time.monotonic() < deadline - 180:
            a_spec = spectral_equalization_aug(a, max_iters=100, time_budget=35.0)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                a = a_spec
                print(f"  Spectral: {best_score:.8f}")
            else:
                print(f"  Spectral: {obj_spec:.8f} (no improvement)")
                a = 0.7 * normalize(a) + 0.3 * normalize(a_spec)

        # Follow-up LP after spectral
        if time.monotonic() < deadline - 130:
            a_new = lp_alternate_improved(a, max_iters=25, time_budget=50.0)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  Post-spectral LP: {best_score:.8f}")

    # =========================================================
    # Phase 3: Systematic n-exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Systematic n-exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700
    # Explore a range around the best n
    n_options = []
    for dn in [-200, -150, -100, -75, -50, -25, 0, 25, 50, 75, 100, 150, 200, 300]:
        n_try = n_best + dn
        if 200 <= n_try <= 1200:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if best_seq is not None:
            if n_try == n_best:
                a_try = normalize(best_seq.copy())
            else:
                a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try

        alloc = min(remaining * 0.1, 25.0)
        a_try = quick_optimize(a_try, time_budget=alloc)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 4: Diverse restarts with peak-aware perturbations
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Diverse restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed)

        for restart in range(10):
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.1, 25.0)

            # Diverse perturbation strategies
            r = restart % 5
            if r == 0:
                # Gaussian multiplier
                a_try = best_seq * np.exp(rng.normal(0, 0.1, n_cur))
            elif r == 1:
                # Sinusoidal modulation
                freq = rng.uniform(1.0, 8.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
                a_try = best_seq * mod
            elif r == 2:
                # Blend with structured random
                a_rand = np.abs(np.sin(np.pi * rng.uniform(0.1, 3.0) * np.arange(n_cur) / n_cur + rng.uniform(0, np.pi)))
                mix = rng.uniform(0.1, 0.3)
                a_try = (1 - mix) * normalize(best_seq) + mix * normalize(a_rand)
            elif r == 3:
                # Block reversal at random position
                a_try = best_seq.copy()
                blen = int(rng.integers(n_cur // 5, n_cur // 3))
                bstart = int(rng.integers(0, n_cur - blen))
                a_try[bstart:bstart + blen] = a_try[bstart:bstart + blen][::-1]
            else:
                # Targeted: reduce mass at peak contributors
                conv_cur = compute_conv(normalize(best_seq))
                k_star = int(np.argmax(conv_cur))
                a_try = normalize(best_seq.copy())
                # Reduce top contributors to peak
                contrib_idx = []
                for j in range(n_cur):
                    k2 = k_star - j
                    if 0 <= k2 < n_cur:
                        contrib_idx.append(j)
                contrib_idx = np.array(contrib_idx)
                top = contrib_idx[np.argsort([a_try[j] * a_try[k_star - j] for j in contrib_idx])[-5:]]
                a_try[top] *= 0.7  # reduce
                # Add to random positions
                targets = rng.choice(n_cur, size=5, replace=False)
                total_reduced = float(np.sum(best_seq[top]) * 0.3)
                a_try[targets] += total_reduced / 5.0
                a_try = normalize(a_try)

            a_try = normalize(a_try)
            a_try = quick_optimize(a_try, time_budget=alloc)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_cur = len(a_try)
                print(f"  Restart {restart}: {best_score:.8f} **")
            else:
                print(f"  Restart {restart}: {obj_try:.8f}")

    # =========================================================
    # Phase 5: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final intensive polish")
    print(f"{'='*60}")

    if best_seq is not None:
        n_final = len(best_seq)
        a_final = normalize(best_seq.copy())

        for polish_round in range(6):
            if time.monotonic() > deadline - 8:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            alloc = min(remaining * 0.2, 30.0)

            # LP (most effective)
            if n_final <= 900 and alloc > 10:
                a_pol = lp_alternate_improved(a_final, max_iters=20, time_budget=alloc * 0.4)
            else:
                a_pol = a_final.copy()

            # Nesterov
            remaining2 = time_budget - (time.monotonic() - start_time)
            rem = alloc - (time.monotonic() - (deadline - remaining))
            a_pol = nesterov_subgradient(a_pol, max_steps=200, time_budget=min(alloc * 0.3, 15.0))

            # Peak reduction
            a_pol = direct_peak_reduction(a_pol, time_budget=min(alloc * 0.2, 10.0))

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish_round}: {best_score:.8f}")
            else:
                a_final = a_pol
                print(f"  Polish {polish_round}: {obj_pol:.8f} (no improvement)")

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate with the actual evaluation function
    try:
        val = evaluate_sequence(result)
        if val < np.inf and val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
