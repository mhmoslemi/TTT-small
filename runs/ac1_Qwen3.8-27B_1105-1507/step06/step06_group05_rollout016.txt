"""
Subgradient + simulated annealing + peak equalization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def gradient_of_peak(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for p in range(n):
        kp = k - p
        if 0 <= kp < n:
            grad[p] = a[p] + a[kp]
    return grad

def multi_peak_gradient(a, top_k=5):
    """Compute weighted subgradient using top-k peaks."""
    n = len(a)
    conv = compute_conv(a)
    # Get top k peaks
    sorted_idx = np.argsort(conv)[::-1][:top_k]
    weights = conv[sorted_idx]
    weights = weights / np.sum(weights)
    grad = np.zeros(n)
    for w, k in zip(weights, sorted_idx):
        grad += w * gradient_of_peak(a, int(k))
    return grad, conv

def subgradient_descent(a, max_iters=200, lr_init=0.1, time_budget=30.0):
    """Subgradient descent with adaptive step size."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    lr = lr_init
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        grad, conv = multi_peak_gradient(a, top_k=5)
        
        # Center the gradient (remove mean to preserve sum)
        grad = grad - np.mean(grad)
        
        a_new = project_simplex(a - lr * grad)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.05, 0.3)  # grow step size slowly
        else:
            lr *= 0.7
        
        # Periodic re-normalization check
        if it % 20 == 0 and lr < 1e-6:
            break
    
    return best_a

def simulated_annealing(a, time_budget=50.0, T0=0.1, T1=0.001, cooling=0.995):
    """Simulated annealing with structured moves."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    T = T0
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget and T > T1:
        # Choose a move
        move_type = rng.random()
        
        if move_type < 0.3:
            # Gaussian perturbation
            a_try = a * np.exp(rng.normal(0, T, n))
        elif move_type < 0.5:
            # Pair swap: move mass from one position to another
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                a_try = a.copy()
                delta = T * a[i]
                a_try[i] -= delta
                a_try[j] += delta
            else:
                a_try = a
        elif move_type < 0.7:
            # Block swap
            blen = int(rng.integers(max(2, n//20), max(3, n//5)))
            start_idx = int(rng.integers(0, n - blen))
            a_try = a.copy()
            a_try[start_idx:start_idx+blen], a_try[n-blen:n] = a_try[n-blen:n].copy(), a_try[start_idx:start_idx+blen].copy()
        elif move_type < 0.85:
            # Sinusoidal modulation
            freq = rng.uniform(1.0, 10.0)
            amp = T
            mod = 1.0 + amp * np.sin(2 * np.pi * freq * np.arange(n) / n)
            a_try = a * mod
        else:
            # Copy neighborhood
            i = int(rng.integers(0, n))
            a_try = a.copy()
            if i > 0:
                a_try[i] = (a_try[i] + a_try[i-1]) / 2
            if i < n-1:
                a_try[i] = (a_try[i] + a_try[i+1]) / 2
        
        a_try = normalize(a_try)
        obj_try = compute_obj(a_try)
        
        # Accept or reject
        delta = obj_try - compute_obj(a)
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()
        
        T *= cooling
    
    return best_a

def peak_equalization(a, max_iters=50, time_budget=30.0):

    Try to equalize the top peaks of the convolution using 
    targeted mass redistribution.
    """
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        cmean = float(np.mean(conv))
        
        # Find positions above threshold
        threshold = cmax - 0.05 * (cmax - cmean)
        high_peaks = np.where(conv > threshold)[0]
        
        if len(high_peaks) == 0:
            break
        
        # For each high peak, try to reduce it by moving mass
        improved = False
        for k in high_peaks[:3]:
            k = int(k)
            # Identify contributors to this peak
            contribs = []
            for j in range(n):
                kp = k - j
                if 0 <= kp < n:
                    contribs.append((a[j] * a[kp], j, kp))
            contribs.sort(reverse=True)
            
            for _, j1, j2 in contribs[:5]:
                if j1 == j2:
                    delta = 0.02 * a[j1]
                    # Move half from j1 to a random far position
                    target = int(rng.integers(0, n))
                    if target != j1:
                        a_try = a.copy()
                        a_try[j1] -= delta
                        a_try[target] += delta
                        a_try = normalize(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                else:
                    # Move mass from the larger of j1, j2
                    src = j1 if a[j1] > a[j2] else j2
                    delta = 0.02 * a[src]
                    target = int(rng.integers(0, n))
                    if target != src:
                        a_try = a.copy()
                        a_try[src] -= delta
                        a_try[target] += delta
                        a_try = normalize(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
            if improved:
                break
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

This is the multiplicative update rule, which naturally preserves non-negativity and the simplex constraint.

"""
LSE-smoothed L-BFGS + multiplicative weights + simulated annealing
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lse_value_and_weights(conv, tau):
    """Compute LSE value and softmax weights over convolution values."""
    cmax = np.max(conv)
    shifted = tau * (conv - cmax)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
        return cmax + 1.0/tau, w
    w = exp_vals / total
    lse_val = cmax + np.log(total) / tau
    return lse_val, w

def lse_gradient(a, tau, fft_size=None):
    """Compute gradient of LSE(conv(a,a), tau) with respect to a."""
    n = len(a)
    if fft_size is None:
        fft_size = 1
        while fft_size < 2 * n - 1:
            fft_size <<= 1
    
    conv = compute_conv(a)
    lse_val, w = lse_value_and_weights(conv, tau)
    
    # grad[p] = sum_k w[k] * (a[p] + a[k-p]) for valid k-p
    # = a[p] * S[p] + C[p]
    # S[p] = sum of w[k] for k in [p, p+n-1] (intersected with [0, 2n-2])
    # C[p] = sum_j a[j] * w[j+p]
    
    # Compute S using cumulative sum
    w_ext = np.zeros(2 * n)
    w_ext[:2*n-1] = w
    # S[p] = sum(w_ext[p:p+n]) for p = 0, ..., n-1
    cumsum_w = np.zeros(2*n + 1)
    cumsum_w[1:] = np.cumsum(w_ext)
    S = np.zeros(n)
    for p in range(n):
        S[p] = cumsum_w[p+n] - cumsum_w[p]
    
    # Compute C using cross-correlation via FFT
    # C[p] = sum_j a[j] * w[j+p] for j = 0, ..., n-1
    # This is the cross-correlation of a and w at lag p
    fa = rfft(a, fft_size)
    fw = rfft(w[:2*n-1], fft_size)
    corr = irfft(np.conj(fa) * fw, fft_size)
    C = corr[:n]
    
    grad = a * S + C
    return lse_val * 2 * n, grad * 2 * n  # multiplied by 2n for the full objective

def softmax_obj_grad(x, tau, n):
    """
    Objective and gradient in the softmax parameterization.
    a = softmax(x), minimize 2n * LSE(conv(a,a), tau)
    """
    # Compute a from x (numerically stable softmax)
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    a /= np.sum(a)
    
    # Compute LSE and gradient wrt a
    lse_val, grad_a = lse_gradient(a, tau)
    
    # Gradient wrt x: grad_x[i] = a[i] * (grad_a[i] - sum_j a[j]*grad_a[j])
    mean_grad = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_grad)
    
    return lse_val, grad_x

def lse_lbfgs(a_init, tau=10.0, max_iter=200, time_budget=60.0):
    """Optimize using LSE-smoothed objective with L-BFGS."""

    # Convert back to a

    a_opt /= np.sum(a_opt)
    a_opt = np.maximum(a_opt, 0)

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)
