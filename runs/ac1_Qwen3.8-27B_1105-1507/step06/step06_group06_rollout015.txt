    """Optimize coordinate i while keeping others fixed."""
    n = len(a)
    a_i = a[i]
    
    def obj_of_t(t):
        a_t = a.copy()
        a_t[i] = t
        # Renormalize
        s = a_t.sum()
        if s < 1e-15:
            return np.inf
        a_t = a_t / s
        conv = compute_conv(a_t)
        return 2.0 * n * np.max(conv)  # sum is 1 so denominator is 1
    
    # Golden section search over [0, 1]
    golden = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - golden * (hi - lo)
    d = lo + golden * (hi - lo)
    fc = obj_of_t(c)
    fd = obj_of_t(d)
    
    for _ in range(50):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - golden * (hi - lo)
            fc = obj_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + golden * (hi - lo)
            fd = obj_of_t(d)

    a_new = a.copy()
    a_new[i] = t_opt

from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def compute_subgradient(a, K=20, tau=0.01):
    """
    Compute the subgradient of max(conv) using top-K peaks with softmax weighting.
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    
    # Top-K peaks
    sorted_idx = np.argsort(conv)[::-1][:K]
    weights = conv[sorted_idx] - cmax
    weights = np.exp(weights / tau)
    weights /= np.sum(weights)
    
    # Gradient: grad[i] = 2 * sum_k w_k * a[k-i] (for valid k-i)
    # = 2 * sum_j a[j] * w_{i+j} (for valid i+j in [0, 2n-2])
    grad = np.zeros(n)
    for idx, k in enumerate(sorted_idx):
        w = weights[idx] * 2.0
        # conv[k] = sum_j a[j]*a[k-j], d/d a[i] = a[k-i] + a[i*(k-i inverses)]
        # d conv[k]/d a[i] = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n, i.e., j=k-i gives a[j]*a[i])
        # = 2*a[k-i] if 0<=k-i<n
        j = k - np.arange(n)  # j[i] = k-i for i=0..n-1
        valid = (j >= 0) & (j < n)
        grad[valid] += w * a[j[valid]]
    
    return grad, cmax, conv

    """
    Accelerated subgradient descent with Nesterov momentum.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    step = 0.1
    mu = momentum
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov step
        a_prev = a.copy()
        a = y - mu * v
        
        # Compute gradient at current a
        a_clipped = np.maximum(a, 0)
        s = a_clipped.sum()
        if s < 1e-15:
            a = a_prev
            continue
        a_work = a_clipped / s
        
        grad, cmax, conv = compute_subgradient(a_work, K=K, tau=tau_init)
        
        # Center the gradient (since we're constrained to sum=1)
        grad_centered = grad - np.mean(grad)
        
        # Update
        v = grad_centered
        obj = 2.0 * n * cmax  # since sum=1
        if obj < best_obj:
            best_obj = obj
            best_a = a_work.copy()
        
        # Adaptive step size
        step = 1.0 / (np.sqrt(it + 1) * 10.0)
        
        # Move in negative gradient direction
        a_new = a_work - step * grad_centered
        a_new = project_nonneg_sum1(a_new)
        a = a_new
        y = a + mu * (a - a_prev)
    
    return best_a, best_obj

    """
    Subgradient descent with fixed step and momentum.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Softmax weights
        shifted = np.clip((conv - cmax) / tau, -100.0, 0.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            idx = np.argmax(conv)
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= wsum
        
        # Gradient
        grad = np.zeros(n)
        # Only iterate over significant weights
        for k in range(len(conv)):
            if weights[k] < 1e-8:
                continue
            wk = 2.0 * weights[k]
            j = k - np.arange(n)
            valid = (j >= 0) & (j < n)
            grad[valid] += wk * a[j[valid]]
        
        grad_centered = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v - step * grad_centered
        a_new = a + v
        a_new = project_nonneg_sum1(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_obj

"""
Step function optimization combining multiplicative updates, accelerated subgradient descent,
LP min-peak alternating, and coordinate descent to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: (1) multiplicative (exponential) updates for natural non-negativity,
(2) adaptive-temperature softmax subgradient with FFT-accelerated gradient,
(3) LP direction with line search rather than direct replacement,
(4) coordinate descent on peak-contributing coordinates,
(5) multi-length exploration with careful resampling.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def topk_gradient(a, K=20, tau=0.03):
    """
    Compute weighted subgradient of max(conv) using top-K peaks with softmax.
    Returns (gradient, conv, max_val).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    sorted_idx = np.argsort(conv)[::-1][:K]
    diffs = conv[sorted_idx] - cmax
    weights = np.exp(diffs / tau)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros(K)
        weights[0] = 1.0
    else:
        weights /= wsum

    grad = np.zeros(n)
    arange = np.arange(n)
    for idx in range(K):
        k = sorted_idx[idx]
        wk = 2.0 * weights[idx]
        j = k - arange
        valid = (j >= 0) & (j < n)
        grad[valid] += wk * a[j[valid]]

    return grad, conv, cmax


def multiplicative_subgradient(a, max_iters=200, time_budget=40.0, K=20, tau=0.03, eta_init=0.3):
    """
    Multiplicative update: a[i] *= exp(-eta * grad[i]), then renormalize.
    This naturally maintains non-negativity and often converges faster.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = eta_init
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, conv, cmax = topk_gradient(a, K=K, tau=tau)
        grad_centered = grad - np.mean(grad)

        # Multiplicative update
        log_update = -eta * grad_centered
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        a_new = project_nonneg_sum1(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stagnant = 0
            eta = min(eta * 1.05, 1.0)
        else:
            stagnant += 1
            if stagnant > 5:
                eta = eta * 0.7
                stagnant = 0

        a = a_new

    return best_a, best_obj


def additive_subgradient(a, max_iters=150, time_budget=30.0, K=15, tau=0.05, step_init=0.05, momentum=0.85):
    """
    Additive subgradient descent with momentum.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    step = step_init
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad_centered = grad - np.mean(grad)

        v = momentum * v - step * grad_centered
        a_new = a + v
        a_new = project_nonneg_sum1(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 3:
                step *= 0.8
                v *= 0.5
                stagnant = 0

        a = a_new

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix for cross-convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    if n > 2000:
        return None, None
    try:
        A = build_toeplitz(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0.0, None)] * n + [(0.0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def lp_direction_step(a, time_budget=30.0):
    """
    Use LP to get a direction, then line search along it.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    g, M_opt = solve_lp_minpeak(a, time_limit=min(15, time_budget * 0.4))
    if g is None:
        return best_a, best_obj

    # Line search along direction (1-t)*a + t*g
    candidate_ts = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
    for t in candidate_ts:
        if time.monotonic() - start > time_budget:
            break
        a_try = (1.0 - t) * a + t * g
        a_try = project_nonneg_sum1(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a, best_obj


def coordinate_descent_peaks(a, n_coords=10, time_budget=20.0):
    """
    Coordinate descent: for each of the top n_coords most important coordinates,
    do a 1D optimization (ternary search).
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Find which coordinates most affect the peak
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    # Coordinates that contribute to the peak: i where a[i]*a[k_star-i] is large
    contributions = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            contributions[i] = a[i] * a[j]
        else:
            contributions[i] = a[i] * a[i]  # approx for self-term

    top_coords = np.argsort(contributions)[::-1][:n_coords]

    for coord in top_coords:
        if time.monotonic() - start > time_budget:
            break

        # Ternary search for optimal value of a[coord]
        lo = 0.0
        hi = min(1.0, 3.0 * a[coord] + 0.01)
        a_save = a.copy()
        a_save_sum_others = a_save.sum() - a_save[coord]

        for _ in range(25):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0

            a_test1 = a_save.copy()
            a_test1[coord] = m1
            a_test1 = np.maximum(a_test1, 0)
            s1 = a_test1.sum()
            if s1 > 1e-15:
                a_test1 = a_test1 / s1
                obj1 = compute_obj(a_test1)
            else:
                obj1 = np.inf

            a_test2 = a_save.copy()
            a_test2[coord] = m2
            a_test2 = np.maximum(a_test2, 0)
            s2 = a_test2.sum()
            if s2 > 1e-15:
                a_test2 = a_test2 / s2
                obj2 = compute_obj(a_test2)
            else:
                obj2 = np.inf

            if obj1 < obj2:
                hi = m2
            else:
                lo = m1

        t_opt = (lo + hi) / 2.0
        a_try = a_save.copy()
        a_try[coord] = t_opt
        a_try = project_nonneg_sum1(a_try)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()

    return best_a, best_obj


def spectral_flatten(a, n_iters=50, time_budget=20.0):
    """
    Spectral approach: modify the amplitude spectrum to reduce time-domain peaks.
    Uses iterative amplitude compression in frequency domain.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n:
        N <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = fft(a, N)
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Compress the amplitude spectrum
        mag_max = np.max(mag)
        if mag_max < 1e-15:
            break
        mag_norm = mag / mag_max
        # Apply power compression (reduces dynamic range)
        alpha = 0.7 + 0.3 * (it / n_iters)
        mag_compressed = mag_norm ** alpha * mag_max

        # Reconstruct
        fa_new = mag_compressed * np.exp(1j * phase)
        a_new = ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def generate_diverse_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular
    tri = np.arange(1, n + 1, dtype=float)
    tri = np.concatenate([tri, tri[:max(1, n // 2)][::-1]])[:n]
    inits.append(tri / tri.sum())

    # Sine
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    # Gaussian (centered)
    x = np.linspace(-2.5, 2.5, n)
    gauss = np.exp(-x ** 2 / 3.0)
    inits.append(gauss / gauss.sum())

    # Two-lobe
    x = np.linspace(0, 1, n)
    two_lobe = np.exp(-((x - 0.25) ** 2) / 0.015) + np.exp(-((x - 0.75) ** 2) / 0.015)
    inits.append(two_lobe / two_lobe.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(cos_w / cos_w.sum())

    if parent is not None:
        inits.append(resample_sequence(parent, n))
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_nonneg_sum1(pert))

    return inits


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def quick_optimize(a, time_budget=15.0):
    """Quick optimization: a few rounds of multiplicative subgradient + chase."""
    a = project_nonneg_sum1(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)

    a2, obj2 = multiplicative_subgradient(a, max_iters=30, time_budget=time_budget * 0.5, K=15, tau=0.05)
    if obj2 < best_obj:
        best_a = a2
        best_obj = obj2

    a3, obj3 = additive_subgradient(best_a, max_iters=30, time_budget=time_budget * 0.3, K=10, tau=0.05, step_init=0.03)
    if obj3 < best_obj:
        best_a = a3
        best_obj = obj3

    return best_a, best_obj


def main_optimize_loop(a, rng, time_budget=900.0):
    """
    Main optimization loop cycling through methods.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    cycle = 0

    while True:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Choose method based on cycle and remaining time
        method = cycle % 5

        if method == 0:
            # Multiplicative subgradient (fast, reliable)
            a_new, obj_new = multiplicative_subgradient(
                best_a, max_iters=100, time_budget=min(remaining * 0.2, 40.0),
                K=20, tau=0.03, eta_init=0.2)

        elif method == 1:
            # LP direction (strong but slow)
            if n <= 1200:
                a_new, obj_new = lp_direction_step(best_a, time_budget=min(remaining * 0.2, 45.0))
            else:
                a_new, obj_new = additive_subgradient(
                    best_a, max_iters=80, time_budget=min(remaining * 0.2, 30.0),
                    K=15, tau=0.04)

        elif method == 2:
            # Additive subgradient with momentum
            a_new, obj_new = additive_subgradient(
                best_a, max_iters=100, time_budget=min(remaining * 0.2, 35.0),
                K=15, tau=0.05, step_init=0.04, momentum=0.85)

        elif method == 3:
            # Coordinate descent on peak coordinates
            a_new, obj_new = coordinate_descent_peaks(
                best_a, n_coords=10, time_budget=min(remaining * 0.2, 25.0))

        else:
            # Spectral flatten
            a_new, obj_new = spectral_flatten(
                best_a, n_iters=40, time_budget=min(remaining * 0.2, 20.0))

        # Accept if improved
        if obj_new < best_obj - 1e-12:
            improvement = best_obj - obj_new
            best_obj = obj_new
            best_a = a_new.copy()
            if improvement > 1e-5:
                print(f"    cycle {cycle}: obj {best_obj:.8f} (improved by {improvement:.2e})")

        # Occasionally perturb to escape local minima
        if cycle % 10 == 7:
            pert = best_a * np.exp(rng.normal(0, 0.01, n))
            pert = project_nonneg_sum1(pert)
            a_p, obj_p = multiplicative_subgradient(pert, max_iters=30, time_budget=10.0, K=15, tau=0.05)
            if obj_p < best_obj:
                best_obj = obj_p
                best_a = a_p.copy()
                print(f"    cycle {cycle} (perturb): obj {best_obj:.8f}")

        cycle += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function using multiplicative updates, subgradient, LP, and
    coordinate descent to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}")

    # === Phase 1: Diverse initialization screening (80s) ===
    print("\n=== Phase 1: Initialization Screening ===")
    phase1_budget = min(80.0, (deadline - start_time) * 0.06)
    phase1_start = time.monotonic()

    n_options = [n_parent, max(200, n_parent - 50), max(200, n_parent + 50),
                 min(2000, n_parent + 100)]
    n_options = list(dict.fromkeys([n for n in n_options if 200 <= n <= 2500]))

    for n_try in n_options:
        if time.monotonic() - phase1_start > phase1_budget:
            break
        inits = generate_diverse_initializations(n_try, rng, parent)
        per_init_time = max(2.0, phase1_budget / len(n_options) / len(inits))

        for init in inits:
            if time.monotonic() - phase1_start > phase1_budget:
                break
            a_q, obj_q = quick_optimize(init, time_budget=per_init_time)
            if obj_q < best_score:
                best_score = obj_q
                best_seq = a_q.copy()
                print(f"  n={n_try}: new best {best_score:.8f}")

    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: Intensive optimization (60% of time) ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    phase2_budget = min(800.0, (deadline - start_time) * 0.55)
    a_main = project_nonneg_sum1(best_seq.copy())
    a_main, obj_main = main_optimize_loop(a_main, rng, time_budget=phase2_budget)
    if obj_main < best_score:
        best_score = obj_main
        best_seq = a_main.copy()
        print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Multi-length exploration (20%) ===
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    phase3_budget = min(250.0, (deadline - time.monotonic()) * 0.25)
    phase3_start = time.monotonic()
    n_current = len(best_seq)

    n_tries = [n_current - 20, n_current - 40, n_current + 20, n_current + 40,
               n_current + 80, n_current - 80, n_current + 100, n_current - 100]
    n_tries = [n for n in n_tries if 200 <= n <= 2500 and n != n_current]
    n_tries = list(dict.fromkeys(n_tries))

    per_n_budget = max(5.0, phase3_budget / max(1, len(n_tries)))

    for n_try in n_tries:
        if time.monotonic() - phase3_start > phase3_budget:
            break
        a_res = resample_sequence(best_seq, n_try)
        a_res = project_nonneg_sum1(a_res)

        # Quick optimize at new length
        a_opt, obj_opt = quick_optimize(a_res, time_budget=min(per_n_budget * 0.5, 15.0))

        # LP if feasible
        if n_try <= 1000 and time.monotonic() - phase3_start < phase3_budget - 10:
            a_lp, obj_lp = lp_direction_step(a_opt, time_budget=min(per_n_budget * 0.3, 15.0))
            if obj_lp < obj_opt:
                a_opt, obj_opt = a_lp, obj_lp

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 4: Final polish (remaining time) ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        n_final = len(best_seq)
        a_final = project_nonneg_sum1(best_seq.copy())

        # Multiple polishing rounds
        for pr in range(min(15, int(remaining / 8))):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 5:
                break

            # Alternate between methods
            if pr % 3 == 0:
                a_p, obj_p = multiplicative_subgradient(a_final, max_iters=50,
                                                        time_budget=min(rem * 0.3, 20.0), K=20, tau=0.02)
            elif pr % 3 == 1 and n_final <= 1200:
                a_p, obj_p = lp_direction_step(a_final, time_budget=min(rem * 0.3, 30.0))
            else:
                a_p, obj_p = coordinate_descent_peaks(a_final, n_coords=8,
                                                      time_budget=min(rem * 0.3, 15.0))

            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  Polish r{pr}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
