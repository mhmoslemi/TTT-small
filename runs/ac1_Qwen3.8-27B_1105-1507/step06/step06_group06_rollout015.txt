import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def simplex_project(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def smoothed_max_gradient(a, tau):
    """
    Compute gradient of tau * log(sum(exp(conv[k]/tau))) w.r.t. a.
    Returns (gradient, smoothed_value).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    # Shift for numerical stability
    shifted = (conv - cmax) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum
    
    # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] for 0 <= k-i < n
    # = 2 * sum_j a[j] * weights[i+j]
    # This is a cross-correlation
    # Let's compute via FFT
    # grad[i]/2 = sum_j a[j] * weights[i+j]
    # Let q = weights (length 2n-1), then grad[i]/2 = sum_j a[j] * q[i+j]
    # This is the "anti-convolution" or cross-correlation
    # cross_corr[i] = sum_j a[j] * q[i+j]
    # = (a reversed) convolved with q, shifted
    # Let a_rev[j] = a[n-1-j]
    # Then (a_rev * q)[n-1+i] = sum_j a_rev[j] * q[n-1+i-j] = sum_j a[n-1-j] * q[n-1+i-j]
    # Let k = n-1-j: sum_k a[k] * q[i+k] = grad[i]/2 ✓
    
    N = 1
    while N < n + len(weights) - 1:
        N <<= 1
    a_rev = a[::-1]
    fa_rev = rfft(a_rev, N)
    fq = rfft(weights, N)
    conv_result = irfft(fa_rev * fq, N)
    grad = 2.0 * conv_result[n-1:n-1+n]

    best_a = cur_a.copy()
    best_obj = compute_obj(cur_a)
    
    a_avg = cur_a.copy()
    start = time.monotonic()
    stall_count = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(cur_a)
        M = float(np.max(conv))
        
        g = lp_solve(cur_a, M, time_limit=min(20, (time_budget - (time.monotonic() - start)) * 0.3))
        if g is None:
            break
        
        a_new, obj_new = lp_blend_line_search(cur_a, g, best_obj)
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            cur_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 15:
                break
        
        # Update average
        a_avg = 0.9 * a_avg + 0.1 * cur_a
    
    a_avg_check = simplex_project(a_avg)
    obj_avg = compute_obj(a_avg_check)
    if obj_avg < best_obj:
        best_obj = obj_avg
        best_a = a_avg_check.copy()
    
    return best_a, best_obj


def sparse_initialization(n, rng, sparsity=0.1):
    num_active = max(10, int(n * sparsity))
    a = np.zeros(n)
    indices = rng.choice(n, size=num_active, replace=False)
    weights = rng.dirichlet(np.ones(num_active) * 0.5)
    a[indices] = weights
    return simplex_project(a)


def diverse_initializations(n, rng, parent=None):
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(sparse_initialization(n, rng, 0.1))
    inits.append(sparse_initialization(n, rng, 0.05))
    inits.append(sparse_initialization(n, rng, 0.2))
    
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())
    
    gauss = np.exp(-((x - 0.5) ** 2) / 0.05)
    inits.append(gauss / gauss.sum())
    
    bimodal = np.exp(-((x - 0.25) ** 2) / 0.02) + np.exp(-((x - 0.75) ** 2) / 0.02)
    inits.append(bimodal / bimodal.sum())
    
    rnd = rng.random(n) ** 3 + 0.01
    inits.append(rnd / rnd.sum())
    
    if parent is not None:
        pn = len(parent)
        if pn != n:
            x_old = np.linspace(0, 1, pn)
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, parent)
        else:
            interp = parent.copy()
        inits.append(simplex_project(interp))
        perturbed = simplex_project(interp * np.exp(rng.normal(0, 0.05, n)))
        inits.append(perturbed)
    
    return inits


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)
    
    parent = try_load_parent()
    
    if parent is not None:
        parent = np.clip(parent, 0, 1000.0)
        best_a = simplex_project(parent.astype(np.float64))
        best_obj = compute_obj(best_a)
        print(f"Parent loaded: n={len(best_a)}, obj={best_obj:.8f}")
    else:
        best_a = np.ones(500) / 500
        best_obj = compute_obj(best_a)
        print(f"No parent, using uniform n=500: obj={best_obj:.8f}")
    
    n_best = len(best_a)
    
    # === Phase 1: Multi-start screening (10% time) ===
    print("\n=== Phase 1: Multi-start Screening ===")
    p1_budget = min(120.0, (deadline - start_time) * 0.10)
    p1_start = time.monotonic()
    
    n_options = sorted(set([
        n_best,
        max(100, n_best - 100),
        max(100, n_best - 200),
        n_best + 50,
        n_best + 100,
        min(1200, n_best + 200),
        400, 600, 800, 1000
    ]))
    n_options = [n for n in n_options if 100 <= n <= 1500]
    
    per_init_budget = max(3.0, p1_budget / (len(n_options) * 3))
    candidates = []
    
    for n_try in n_options:
        if time.monotonic() - p1_start > p1_budget:
            break
        inits = diverse_initializations(n_try, rng, parent)
        for init in inits[:4]:
            if time.monotonic() - p1_start > p1_budget:
                break
            a_scr, obj_scr = mirror_descent_phase(
                init, tau_init=0.5, tau_final=0.1,
                n_steps=500, time_budget=min(per_init_budget, p1_budget - (time.monotonic() - p1_start))
            )
            candidates.append((obj_scr, a_scr))
    
    candidates.sort(key=lambda x: x[0])
    if candidates:
        print(f"  Best from screening: obj={candidates[0][0]:.8f}, n={len(candidates[0][1])}")
        if candidates[0][0] < best_obj:
            best_obj = candidates[0][0]
            best_a = candidates[0][1].copy()
            n_best = len(best_a)
    
    # === Phase 2: Main mirror descent (40% time) ===
    print(f"\n=== Phase 2: Main Mirror Descent (n={len(best_a)}) ===")
    p2_budget = min(600.0, (deadline - time.monotonic()) * 0.40)
    p2_start = time.monotonic()
    
    # Multiple rounds with cooling
    remaining = p2_budget
    tau_start = 1.0
    round_num = 0
    while remaining > 10:
        round_num += 1
        alloc = min(150.0, remaining)
        tau_i = max(0.1, tau_start * (0.6 ** round_num))
        tau_f = max(0.05, tau_i * 0.3)
        
        a_new, obj_new = mirror_descent_phase(
            best_a, tau_init=tau_i, tau_final=tau_f,
            n_steps=20000, time_budget=alloc, eta_init=0.5
        )
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"  Round {round_num}: obj={best_obj:.8f} (tau={tau_i:.3f}->{tau_f:.3f})")
        else:
            print(f"  Round {round_num}: no improvement (best={best_obj:.8f})")
        
        remaining = p2_budget - (time.monotonic() - p2_start)
    
    # === Phase 3: LP Alternating (25% time) ===
    print(f"\n=== Phase 3: LP Alternating (n={len(best_a)}) ===")
    print(f"  Before: obj={best_obj:.8f}")
    p3_budget = min(400.0, (deadline - time.monotonic()) * 0.25)
    p3_start = time.monotonic()
    
    if len(best_a) <= 1500:
        a_new, obj_new = lp_alternating_phase(best_a, n_iters=50, time_budget=p3_budget)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"  LP alternating improved: obj={best_obj:.8f}")
        else:
            print(f"  LP alternating: no improvement (best={best_obj:.8f})")
    else:
        print("  Skipping LP (n too large)")
    
    # Small perturbation + LP restart
    remaining_p3 = p3_budget - (time.monotonic() - p3_start)
    if remaining_p3 > 20 and len(best_a) <= 1200:
        n = len(best_a)
        for pert_idx in range(min(5, int(remaining_p3 // 15))):
            if time.monotonic() - p3_start > p3_budget:
                break
            a_pert = simplex_project(best_a * np.exp(rng.normal(0, 0.03, n)))
            a_pert2, obj_pert2 = lp_alternating_phase(a_pert, n_iters=15,
                                                       time_budget=min(20, remaining_p3 - pert_idx * 15))
            if obj_pert2 < best_obj:
                best_obj = obj_pert2
                best_a = a_pert2.copy()
                print(f"  Perturbation {pert_idx}: obj={best_obj:.8f}")
    
    # === Phase 4: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 30:
        p4_budget = min(150.0, remaining4 * 0.15)
        p4_start = time.monotonic()
        n_current = len(best_a)
        
        n_tries = sorted(set([
            n_current - 50, n_current - 100,
            n_current + 50, n_current + 100,
            max(200, n_current - 200),
            min(1200, n_current + 200),
        ]))
        n_tries = [n for n in n_tries if 150 <= n <= 1200]
        
        per_n = max(5.0, p4_budget / max(len(n_tries), 1))
        
        for n_try in n_tries:
            if time.monotonic() - p4_start > p4_budget:
                break
            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, n_try)
            a_res = np.interp(x_new, x_old, best_a)
            a_res = simplex_project(a_res)
            
            a_opt, obj_opt = mirror_descent_phase(
                a_res, tau_init=0.3, tau_final=0.05,
                n_steps=5000, time_budget=per_n * 0.6
            )
            if n_try <= 1000:
                a_opt2, obj_opt2 = lp_alternating_phase(
                    a_opt, n_iters=5,
                    time_budget=min(per_n * 0.4, 20)
                )
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2
            
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                print(f"  n={n_try}: NEW BEST obj={best_obj:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining5 = deadline - time.monotonic()
    print(f"  Time remaining: {remaining5:.1f}s, best={best_obj:.8f}")
    
    if remaining5 > 10:
        n_final = len(best_a)
        
        # LP polish if possible
        if n_final <= 1200 and remaining5 > 30:
            a_pol, obj_pol = lp_alternating_phase(best_a, n_iters=30, time_budget=min(remaining5 * 0.5, 120))
            if obj_pol < best_obj:
                best_obj = obj_pol
                best_a = a_pol.copy()
                print(f"  LP polish: obj={best_obj:.8f}")
        
        remaining5 = deadline - time.monotonic()
        
        # Mirror descent polish with very low temperature
        if remaining5 > 10:
            a_pol2, obj_pol2 = mirror_descent_phase(
                best_a, tau_init=0.1, tau_final=0.02,
                n_steps=10000, time_budget=min(remaining5 * 0.5, 60)
            )
            if obj_pol2 < best_obj:
                best_obj = obj_pol2
                best_a = a_pol2.copy()
                print(f"  MD polish: obj={best_obj:.8f}")
        
        # Final small perturbations
        remaining5 = deadline - time.monotonic()
        if remaining5 > 15 and n_final <= 1000:
            for pf in range(min(3, int(remaining5 // 10))):
                if deadline - time.monotonic() < 8:
                    break
                a_p = simplex_project(best_a * np.exp(rng.normal(0, 0.01, n_final)))
                a_p, obj_p = mirror_descent_phase(a_p, tau_init=0.05, tau_final=0.02,
                                                   n_steps=3000, time_budget=min(8, deadline - time.monotonic() - 5))
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_p.copy()
                    print(f"  Perturb {pf}: obj={best_obj:.8f}")
    
    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    print(f"\nFinal: n={len(result)}, computed_obj={best_obj:.8f}, evaluated={val:.8f}")
    
    if np.isfinite(val) and val < best_obj:
        best_obj = val
    elif not np.isfinite(val):
        result = [1.0] * 100
    
    elapsed = time.monotonic() - start_time
    print(f"Time used: {elapsed:.1f}s / {budget_s}s")
    
    return result
