"""
Symmetric Adam optimization with cosine annealing, multi-length exploration,
and LP refinement for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: enforce approximate symmetry to halve effective parameters.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad_sym(a_half, n, beta):
    """
    Compute smoothed objective and gradient for symmetric sequence.
    a_half: first half of the sequence (length ceil(n/2))
    Returns (obj, grad_half) where grad_half has same shape as a_half.
    """
    # Reconstruct full symmetric sequence
    n = int(n)
    a = np.zeros(n)
    half_len = (n + 1) // 2
    a[:half_len] = a_half
    a[half_len:] = a_half[:n - half_len][::-1]
    
    sum_a = float(np.sum(a))
    conv = compute_conv(a)
    
    # Logsumexp smoothing
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    # Gradient of soft_max w.r.t. a[i]
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    
    grad_obj_full = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    # Fold gradient back to half
    grad_half = grad_obj_full[:half_len].copy()
    if n - half_len > 0:
        grad_half += grad_obj_full[half_len:][::-1]
    
    return obj, grad_half, a


def sym_adam_opt(a_init, max_iters=20000, time_budget=100.0,
                 beta_start=5.0, beta_end=500.0, lr_start=0.05,
                 lr_end=0.001, use_cosine=True):
    """
    Adam optimization on symmetric sequence with cosine LR annealing.
    a_init: full sequence (will be symmetrized internally)
    """
    n = len(a_init)
    a_full = np.maximum(a_init.copy(), 1e-12)
    s = float(np.sum(a_full))
    if s < 1e-10:
        return a_init
    
    # Symmetrize
    a_full = (a_full + a_full[::-1]) / 2.0
    a_full = a_full / np.sum(a_full)
    
    half_len = (n + 1) // 2
    # Work in log-space for the half
    x = np.log(np.maximum(a_full[:half_len], 1e-12))
    x -= np.mean(x)
    
    # Adam state
    m = np.zeros(half_len)
    v = np.zeros(half_len)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a_full.copy()
    best_obj = compute_obj(a_full)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
            
        # Temperature schedule (geometric)
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Learning rate schedule (cosine)
        if use_cosine:
            lr = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * progress))
        else:
            lr = lr_start * (0.5) ** (progress * 10)
        
        # Reconstruct a from x
        x_c = x - np.max(x)
        a_half = np.exp(x_c)
        
        a = np.zeros(n)
        a[:half_len] = a_half
        a[half_len:] = a_half[:n - half_len][::-1]
        a = a / np.sum(a)
        
        # Compute smoothed objective and gradient
        obj_sm, grad_half, a_recon = smooth_obj_grad_sym(a_half, n, beta)
        
        # Chain rule: grad w.r.t. x = grad_a * a_half
        grad_x = grad_half * a_half
        
        # Clip gradient
        grad_norm = np.linalg.norm(grad_x)
        if grad_norm > 100:
            grad_x = grad_x * 100 / grad_norm
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic centering
        if it % 100 == 0:
            x -= np.mean(x)
        
        # Check true objective periodically
        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_half_check = np.exp(x_c)
            a_check = np.zeros(n)
            a_check[:half_len] = a_half_check
            a_check[half_len:] = a_half_check[:n - half_len][::-1]
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()
    
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 800:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine_sym(a, time_budget=30.0):

    n = len(a)
    start = time.monotonic()

    if g is not None:

        g_sym = g_sym / np.sum(g_sym)
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            if time.monotonic() - start > time_budget:

            a_try = (1 - t) * a_sym + t * g_sym

def resample_sym(a, new_n):
    """Resample symmetric sequence to new length, maintaining symmetry."""
    n = len(a)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    # Enforce symmetry
    a_new = (a_new + a_new[::-1]) / 2.0
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def structured_init(n, shape='raised_cosine'):
    """Create a structured initial sequence."""
    if shape == 'raised_cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif shape == 'parabola':
        x = np.linspace(0, 1, n)
        a = 4 * x * (1 - x)
    elif shape == 'sin_sq':
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif shape == 'flat_top':
        x = np.linspace(0, 1, n)
        # Flat top with smooth edges
        edge = 0.1
        a = np.ones(n)
        a[:int(edge*n)] = np.linspace(0, 1, int(edge*n))
        a[-int(edge*n):] = np.linspace(1, 0, int(edge*n))
    elif shape == 'gaussian':
        x = np.linspace(0, 1, n)
        sig = 0.25
        a = np.exp(-0.5 * ((x - 0.5) / sig) ** 2)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    # Enforce symmetry
    a = (a + a[::-1]) / 2.0
    a = a / np.sum(a)
    return a
