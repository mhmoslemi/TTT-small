"""
Hybrid optimizer combining log-space Adam with L-BFGS refinement, multi-start
exploration, and simulated annealing for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach: L-BFGS quasi-Newton polish, better
annealing schedules, Frank-Wolfe direction search, and adaptive multi-n strategy.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def true_obj(a):
    """Compute true objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def smooth_obj_grad(x, beta):
    """
    Smoothed objective and gradient in log-space.
    x: log-space variables, a = exp(x)/sum(exp(x)) has sum=1.
    Returns (objective, gradient w.r.t. x).
    """
    xc = x - np.max(x)
    a = np.exp(xc)
    a = a / np.sum(a)
    n = len(a)

    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip(beta * (conv - cmax), -500, 500)
    e = np.exp(shifted)
    es = float(np.sum(e))
    w = e / es
    ls = cmax + np.log(es) / beta

    obj = 2.0 * n * ls

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_a = 2.0 * cc[:n]

    dot = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - dot)

    return obj, grad_x


def adam_logspace(a_init, beta_start, beta_end, lr, time_budget, max_iters=100000,
                  lr_decay=0.99999, center_every=100):
    """Adam optimization in log-space with temperature annealing."""
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    s = np.sum(a)
    if s < 1e-12:
        return a_init
    a = a / s

    x = np.log(a)
    x -= x.mean()
    n = len(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    best_a = a.copy()
    best_obj = true_obj(a)
    start = time.monotonic()
    current_lr = lr

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        f, g = smooth_obj_grad(x, beta)

        t += 1
        current_lr = lr * (lr_decay ** it)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        x = x - current_lr * mh / (np.sqrt(vh) + eps)

        if it % center_every == 0:
            x -= x.mean()

        if it % 50 == 0 or it == max_iters - 1:
            xc = x - x.max()
            ac = np.exp(xc)
            ac = ac / ac.sum()
            o = true_obj(ac)
            if o < best_obj:
                best_obj = o
                best_a = ac.copy()

    return best_a


def lbfgs_refine(a_init, beta, time_budget=60.0, maxiter=3000):
    """L-BFGS-B refinement in log-space for final polish."""
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    s = np.sum(a)
    if s < 1e-12:
        return a_init
    a = a / s
    x0 = np.log(a)
    x0 -= x0.mean()

    def fun(x):
        f, g = smooth_obj_grad(x, beta)
        return f, g

    best_a = a.copy()
    best_obj = true_obj(a)
    start = time.monotonic()

    try:
        res = minimize(fun, x0, method='L-BFGS-B',

            xr = res.x - res.x.max()
            ar = np.exp(xr)
            ar = ar / ar.sum()
            orr = true_obj(ar)
            if orr < best_obj:

                best_obj = orr
    except Exception:
        pass

    return best_a


def frank_wolfe_step(a, beta, time_budget=30.0, max_iters=1000):
    """
    Frank-Wolfe (conditional gradient) on the smoothed objective in the simplex.
    At each step, move towards the coordinate with smallest gradient.
    """
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-12:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = true_obj(a)
    start = time.monotonic()
    step_size = 2.0 / 3.0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        f, gx = smooth_obj_grad(np.log(np.maximum(a, 1e-15)), beta)
        # Convert gradient from log-space to a-space
        # Actually, let's work directly with a-space gradient
        conv = compute_conv(a)
        cmax = np.max(conv)
        shifted = np.clip(beta * (conv - cmax), -500, 500)
        e = np.exp(shifted)
        es = e.sum()
        w = e / es
        ls = cmax + np.log(es) / beta

        size = 1
        while size < 2 * len(conv):
            size <<= 1
        W = rfft(w, size)
        A = rfft(a, size)
        cc = irfft(W * np.conj(A), size)
        grad_a = 2.0 * cc[:n]

        # Frank-Wolfe: pick coordinate with minimum gradient
        j = int(np.argmin(grad_a))
        e_j = np.zeros(n)
        e_j[j] = 1.0

        # Line search along a -> e_j
        best_t = 0.0
        best_f = f
        for t in [0.01, 0.02, 0.05, 0.1, 0.2, step_size, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * e_j
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 1e-12:
                continue
            a_try = a_try / st
            # Quick objective check
            o = true_obj(a_try)
            if o < best_obj:
                best_obj = o
                best_a = a_try.copy()
                best_t = t

        if best_t == 0.0:
            # Try smaller steps
            for t in [0.005, 0.001, 0.0005]:
                a_try = (1 - t) * a + t * e_j
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 1e-12:
                    continue
                a_try = a_try / st
                o = true_obj(a_try)
                if o < best_obj:
                    best_obj = o
                    best_a = a_try.copy()
                    best_t = t
                    break

        # Update a
        a = (1 - best_t) * a + best_t * e_j
        a = np.maximum(a, 0)
        st = a.sum()
        if st > 1e-12:
            a = a / st

        # Decay step size
        step_size = 2.0 / (it + 3.0)

    return best_a


def simulated_annealing(a_init, time_budget=30.0, T_start=0.01, T_end=0.001,
                        max_iters=5000, rng=None):
    """Simulated annealing to escape local minima."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-12:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = true_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T_start * (T_end / T_start) ** (it / max(max_iters - 1, 1))
        a_try = a.copy()
        # Perturb random subset of coordinates
        n_pert = max(1, n // 20)
        idx = rng.integers(0, n, size=n_pert)
        scale = 0.05 + 0.1 * T / T_start
        a_try[idx] *= np.exp(rng.normal(0, scale, n_pert))
        a_try = np.maximum(a_try, 0)
        st = a_try.sum()
        if st < 1e-12:
            continue
        a_try = a_try / st

        o_try = true_obj(a_try)
        o_cur = true_obj(a)

        if o_try < o_cur or rng.random() < np.exp(-(o_try - o_cur) / (T * 0.1)):
            a = a_try
            if o_try < best_obj:
                best_obj = o_try
                best_a = a.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def generate_structured_init(n, seed=0):
    """Generate various structured initial sequences."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian
    x = np.linspace(0, 1, n)
    g = np.exp(-8 * (x - 0.5) ** 2)
    inits.append(g / g.sum())

    # Triangular
    t = np.minimum(np.arange(n), n - np.arange(n))
    inits.append(t / t.sum())

    # Cosine window
    c = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    inits.append(c / c.sum())

    # Random
    r = rng.random(n)
    inits.append(r / r.sum())

    # Bimodal
    b = np.exp(-4 * (x - 0.3) ** 2) + np.exp(-4 * (x - 0.7) ** 2)
    inits.append(b / b.sum())

    # Sine-modulated
    s = 0.5 + 0.5 * np.sin(np.pi * x)
    inits.append(s / s.sum())

    return inits

                                  lr=0.04, time_budget=alloc * 0.6, max_iters=30000)
            # L-BFGS polish
            if time_budget_remaining() > 10:
                a_opt = lbfgs_refine(a_opt, beta=500,
                                     time_budget=min(time_budget_remaining() * 0.1, 15.0),
                                     maxiter=1000)
            obj = true_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 5: Simulated annealing + restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Simulated annealing & restarts")
    print(f"{'='*60}")

    if best_seq is not None and time_budget_remaining() > 40:
        rng = np.random.default_rng(seed)
        n_best = len(best_seq)

        for restart in range(10):
            if time_budget_remaining() < 15:
                break

            # Different perturbation strategies
            r = restart % 5
            if r == 0:
                # Multiplicative noise
                a_pert = best_seq * np.exp(rng.normal(0, 0.05, n_best))
            elif r == 1:
                # Spike injection
                a_pert = best_seq.copy()
                for _ in range(max(1, n_best // 30)):
                    a_pert[rng.integers(0, n_best)] += rng.uniform(0.02, 0.1) * np.mean(best_seq)
            elif r == 2:
                # Blend with structured
                structured = generate_structured_init(n_best, seed=restart)
                alpha = rng.uniform(0.1, 0.3)
                a_pert = (1 - alpha) * (best_seq / best_seq.sum()) + alpha * structured[restart % len(structured)]
            elif r == 3:
                # Zeros injection (create sparsity)
                a_pert = best_seq.copy()
                n_zeros = max(1, n_best // 20)
                zero_idx = rng.choice(n_best, size=n_zeros, replace=False)
                a_pert[zero_idx] = 0
            else:
                # Large perturbation
                a_pert = best_seq * np.exp(rng.normal(0, 0.15, n_best))

            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp < 1e-10:
                continue
            a_pert = a_pert / sp

            # SA then Adam
            alloc_sa = min(time_budget_remaining() * 0.04, 15.0)
            a_sa = simulated_annealing(a_pert, time_budget=alloc_sa,
                                       T_start=0.05, T_end=0.005,
                                       max_iters=3000, rng=rng)
            alloc_adam = min(time_budget_remaining() * 0.06, 20.0)
            a_opt = adam_logspace(a_sa, beta_start=20.0, beta_end=1000.0,
                                  lr=0.04, time_budget=alloc_adam, max_iters=20000)
            # L-BFGS if time
            if time_budget_remaining() > 10:
                a_opt = lbfgs_refine(a_opt, beta=1000,
                                     time_budget=min(time_budget_remaining() * 0.04, 15.0))
            obj = true_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart} (type {r}): {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart} (type {r}): {obj:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if best_seq is not None and time_budget_remaining() > 20:
        a = best_seq.copy() / np.sum(best_seq)

        # Multiple L-BFGS passes with increasing beta
        for final_beta in [500, 2000, 5000, 10000, 20000]:
            if time_budget_remaining() < 10:
                break
            alloc = min(time_budget_remaining() * 0.06, 25.0)
            a_f = lbfgs_refine(a, beta=final_beta, time_budget=alloc, maxiter=2000)
            obj = true_obj(a_f)
            if obj < best_score:
                best_score = obj
                best_seq = a_f.copy()
                a = a_f.copy()
                print(f"  Final L-BFGS beta={final_beta}: {best_score:.8f} ***")
            else:
                print(f"  Final L-BFGS beta={final_beta}: {obj:.8f}")

        # One more Adam at very high beta
        if time_budget_remaining() > 15:
            alloc = min(time_budget_remaining() * 0.3, 30.0)
            a_final = adam_logspace(a, beta_start=1000, beta_end=50000, lr=0.005,
                                    time_budget=alloc, max_iters=50000, lr_decay=0.99999)
            obj = true_obj(a_final)
            if obj < best_score:
                best_score = obj
                best_seq = a_final.copy()
                print(f"  Final high-beta Adam: {best_score:.8f} ***")
            else:
                print(f"  Final high-beta Adam: {obj:.8f}")

        # Ultimate L-BFGS
        if time_budget_remaining() > 10:
            a_ult = lbfgs_refine(best_seq, beta=10000,
                                  time_budget=min(time_budget_remaining() * 0.5, 20.0),
                                  maxiter=5000)
            obj = true_obj(a_ult)
            if obj < best_score:
                best_score = obj
                best_seq = a_ult.copy()
                print(f"  Ultimate L-BFGS: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = true_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - _start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
