def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):

    a = np.maximum(a, 0.0)

    if s < 1e-15:

def solve_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:

                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def spectral_pocs(a, max_iters=80, time_budget=40.0):
    """POCS spectral flattening with adaptive compression."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_mean = float(np.mean(P))
        p_max = float(np.max(P))
        
        if p_max < 1e-30:
            break
        
        # Adaptive compression: more aggressive early, less later
        progress = it / max_iters
        alpha = 0.3 + 0.7 * progress  # Start with strong compression
        
        # Compress toward mean
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max
        
        # Actually, we want to flatten, so compress high values and boost low values
        # P_new[k] = p_mean + (P[k] - p_mean) * (1 - beta)
        beta = 0.3 + 0.5 * progress
        P_new = p_mean + (P - p_mean) * (1 - beta)
        P_new = np.maximum(P_new, 0.0)
        
        # Reconstruct
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        
        # Project
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            blend = 0.3
            a = normalize((1 - blend) * a + blend * a_new)
    
    return best_a, best_obj

def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a[:n]
    return A


def solve_lp(a, M, time_limit=30):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:

        grad_centered = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad_centered))

        if grad_norm < 1e-15:
            break

        t = step_size
        found = False
        for _ in range(12):
            a_try = normalize(a - t * grad_centered / grad_norm)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                found = True
                break
            t *= 0.5

        if found:
            step_size = min(0.12, t * 1.4)
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        else:
            step_size *= 0.5
            if step_size < 1e-8:
                rng = np.random.default_rng(step * 31 + 17)
                a = normalize(a * np.exp(rng.normal(0, 0.025, n)))
                step_size = 0.04

    return best_a, best_obj


def spectral_pocs(a, max_iters=60, time_budget=35.0):
    """POCS spectral flattening."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_mean = float(np.mean(P))
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        beta = 0.2 + 0.6 * (it / max_iters)
        P_new = p_mean + (P - p_mean) * (1.0 - beta)
        P_new = np.maximum(P_new, 0.0)

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.6 * a + 0.4 * a_new)

    return best_a, best_obj


def targeted_local_search(a, time_budget=25.0):
    """Move mass away from peak autocorrelation contributors."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        pairs = []
        for j in range(n):
            j2 = k_star - j
            if 0 <= j2 < n:
                pairs.append((a[j] * a[j2], j, j2))
        pairs.sort(reverse=True)

        improved = False
        for val, j1, j2 in pairs[:10]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            for _ in range(4):
                j_t = int(rng.integers(0, n))
                if j_t == j1 or j_t == j2:
                    continue
                for frac in [0.04, 0.07, 0.1, 0.15]:
                    delta = frac * a[j1]
                    if delta > a[j1]:
                        continue
                    a_try = a.copy()
                    a_try[j1] -= delta
                    a_try[j_t] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved:
            if rng.random() < 0.2:
                a = normalize(a * np.exp(rng.normal(0, 0.015, n)))
            else:
                break

    return best_a, best_obj


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(budget_s=1500):
    """
    Multi-phase optimization:
    Phase 1: Intensive LP at parent length
    Phase 2: Spectral + gradient refinement
    Phase 3: Broad n exploration (400-1200)
    Phase 4: Diverse restarts from best found
    Phase 5: Final intensive polish with all methods
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 35.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_init = len(best_seq) if best_seq is not None else 700
    print(f"Starting: n={n_init}, score={best_score:.8f}")

    # Phase 1: Core LP optimization
    print(f"\n{'='*50}")
    print(f"Phase 1: LP at n={n_init}")
    print(f"{'='*50}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        for cyc in range(4):
            if time.monotonic() > deadline - 500:
                break
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            alloc = min(rem * 0.15, 100.0)
            a, obj = lp_optimize(a, max_iters=25, time_budget=alloc)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP cycle {cyc}: {best_score:.8f} **")
            else:
                print(f"  LP cycle {cyc}: {obj:.8f}")

    # Phase 2: Spectral + gradient
    print(f"\n{'='*50}")
    print(f"Phase 2: Spectral + Gradient")
    print(f"{'='*50}")

    if best_seq is not None:
        a = normalize(best_seq.copy())

        if time.monotonic() < deadline - 400:
            a, obj = spectral_pocs(a, max_iters=50, time_budget=40.0)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Spectral: {best_score:.8f} **")
            else:
                print(f"  Spectral: {obj:.8f}")

        if time.monotonic() < deadline - 350:
            a, obj = gradient_descent(normalize(best_seq), time_budget=45.0)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Gradient: {best_score:.8f} **")
            else:
                print(f"  Gradient: {obj:.8f}")

        # LP follow-up
        if time.monotonic() < deadline - 300:
            a, obj = lp_optimize(normalize(best_seq), max_iters=20, time_budget=80.0)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP polish: {best_score:.8f} **")
            else:
                print(f"  LP polish: {obj:.8f}")

    # Phase 3: Multi-n exploration
    print(f"\n{'='*50}")
    print(f"Phase 3: Multi-n exploration")
    print(f"{'='*50}")

    n_cur = len(best_seq) if best_seq is not None else 700
    n_options = sorted(set([n_cur + d for d in [-300, -250, -200, -150, -100,
                                                  -75, -50, -25, 0, 25, 50, 75,
                                                  100, 150, 200, 250, 300, 400]]) & set(range(300, 1401)))

    for n_try in n_options:
        if time.monotonic() > deadline - 120:
            break
        rem = deadline - time.monotonic()
        if rem < 25:
            break

        if best_seq is not None:
            a_try = normalize(resample(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try

        alloc = min(rem * 0.08, 35.0)

        # Quick optimization pipeline
        a_try, obj_try = gradient_descent(a_try, time_budget=alloc * 0.3)
        if n_try <= 900:
            a_try, obj_try = lp_optimize(a_try, max_iters=8, time_budget=alloc * 0.4)
        a_try, obj_try = spectral_pocs(a_try, max_iters=15, time_budget=alloc * 0.3)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # Phase 4: Diverse restarts
    print(f"\n{'='*50}")
    print(f"Phase 4: Diverse restarts")
    print(f"{'='*50}")

    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(777)

        for rst in range(12):
            if time.monotonic() > deadline - 70:
                break
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            alloc = min(rem * 0.08, 28.0)

            r = rst % 6
            if r == 0:
                a_try = best_seq * np.exp(rng.normal(0, 0.08, n_cur))
            elif r == 1:
                freq = rng.uniform(1.0, 6.0)
                mod = 1.0 + 0.15 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur)
                a_try = best_seq * mod
            elif r == 2:
                a_r = rng.dirichlet(np.ones(n_cur) * 0.4)
                a_try = 0.65 * normalize(best_seq) + 0.35 * a_r
            elif r == 3:
                a_try = best_seq.copy()
                bl = int(rng.integers(n_cur // 5, n_cur // 3))
                st = int(rng.integers(0, n_cur - bl))
                a_try[st:st+bl] = a_try[st:st+bl][::-1]
            elif r == 4:
                # Smooth perturbation using Gaussian kernel
                kernel = np.exp(-0.5 * (np.arange(n_cur) - n_cur // 2) ** 2 / (n_cur * 0.1))
                a_try = best_seq * (1.0 + 0.1 * kernel * rng.normal(0, 1, n_cur))
            else:
                # Sinusoidal wrapping
                a_try = best_seq.copy()
                freq = rng.uniform(2, 10)
                shift = np.roll(a_try, int(rng.integers(-5, 5)))
                a_try = 0.5 * a_try + 0.5 * shift

            a_try = normalize(a_try)

            if n_cur <= 900:
                a_try, obj_try = lp_optimize(a_try, max_iters=8, time_budget=alloc * 0.4)
            else:
                a_try, obj_try = gradient_descent(a_try, time_budget=alloc * 0.35)
            a_try, obj_try = gradient_descent(a_try, time_budget=alloc * 0.3)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {rst}: {best_score:.8f} **")
            else:
                print(f"  Restart {rst}: {obj_try:.8f}")

    # Phase 5: Final intensive polish
    print(f"\n{'='*50}")
    print(f"Phase 5: Final polish")
    print(f"{'='*50}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        n_fin = len(a)

        for rnd in range(6):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            alloc = min(rem * 0.25, 35.0)

            # Cycle through methods
            method = rnd % 3
            if method == 0 and n_fin <= 900 and alloc > 20:
                a, obj = lp_optimize(a, max_iters=12, time_budget=alloc * 0.5)
            elif method == 1:
                a, obj = gradient_descent(a, time_budget=alloc * 0.5)
            else:
                a, obj = spectral_pocs(a, max_iters=30, time_budget=alloc * 0.4)

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                a = a
                print(f"  Polish {rnd} ({['LP','grad','spec'][method]}): {best_score:.8f} **")
            else:
                print(f"  Polish {rnd} ({['LP','grad','spec'][method]}): {obj:.8f}")
                a = normalize(best_seq)

            # Local search
            if time.monotonic() < deadline - 5:
                a2, obj2 = targeted_local_search(normalize(best_seq), time_budget=min(8, rem * 0.15))
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a2.copy()
                    a = a2
                    print(f"  Local {rnd}: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 200

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = min(best_score, val)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
