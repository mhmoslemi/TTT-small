    na = len(a)

    ag = conv_cross(a, g)
    gg = fast_conv(g)
    
    def f(t):
        val = (1.0 - t) ** 2 * aa + 2.0 * t * (1.0 - t) * ag + t * t * gg
        return np.max(val)
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    a1 = hi - gr * (hi - lo)
    b1 = lo + gr * (hi - lo)
    fa1 = f(a1)
    fb1 = f(b1)
    
    for _ in range(50):
        if fa1 < fb1:
            hi = b1
            b1 = a1
            fb1 = fa1
            a1 = hi - gr * (hi - lo)
            fa1 = f(a1)
        else:
            lo = a1
            a1 = b1
            fa1 = fb1
            b1 = lo + gr * (hi - lo)
            fb1 = f(b1)
    
    t_opt = (lo + hi) / 2.0
    # Check endpoints
    f_end = f(1.0)
    if f_end < fa1 and f_end < f(t_opt):
        t_opt = 1.0
    
    return t_opt


def run_lp_iterations(a_init, deadline, max_iters=2000, verbose=True):
    """Main LP-based fictitious play with exact line search."""
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    if verbose:
        print(f"    LP iter start: M={best_M:.10f}, n={n}", flush=True)
    
    it = 0
    no_improve = 0
    
    while it < max_iters and time.monotonic() < deadline:
        # Check time for LP
        lp_time = min(15.0, max(2.0, (deadline - time.monotonic()) * 0.05))
        g, t_opt = lp_best_response_cutting_plane(a, time_budget=lp_time)
        
        if g is None:
            no_improve += 5
            if no_improve > 10:
                break
            it += 1
            continue
        
        # Exact line search
        step = line_search_convex(a, g)
        
        # Update
        a_new = (1.0 - step) * a + step * g
        a_new = normalize(a_new)
        
        M_new = fast_conv(a_new).max()
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 20 == 0:
                print(f"    iter {it}: M={best_M:.10f}, step={step:.4f}", flush=True)
        else:
            no_improve += 1
        
        a = a_new
        it += 1
        
        if no_improve > 30:
            if verbose:
                print(f"    Stagnation at iter {it}, M={best_M:.10f}", flush=True)
            break
    
    if verbose:
        print(f"    LP iter done: M={best_M:.10f}, iters={it}", flush=True)
    
    return best_a


def coordinate_descent_polish(a_init, deadline, verbose=True):
    """Coordinate descent with exact 1D sub-optimization."""
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    if verbose:
        print(f"    CD start: M={best_M:.10f}", flush=True)
    
    sweep = 0
    max_sweeps = 100
    
    while sweep < max_sweeps and time.monotonic() < deadline:
        b = fast_conv(a)
        any_improved = False
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            x = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Build piecewise objective
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[k_lo:k_hi + 1] = False
            base_max = b[mask].max() if mask.any() else 0.0
            
            c_list = []
            s_list = []
            has_quad = False
            quad_c = 0.0
            
            for k in range(k_lo, k_hi + 1):
                j = k - i
                if 0 <= j < n:
                    if k == 2 * i:
                        has_quad = True
                        quad_c = b[k] - x * x
                    else:
                        c_list.append(b[k] - 2.0 * x * a[j])
                        s_list.append(2.0 * a[j])
            
            c_arr = np.array(c_list) if c_list else np.array([0.0])
            s_arr = np.array(s_list) if s_list else np.array([0.0])
            
            # Find minimum of max(base_max, max(c + s*y), quad_c + y^2) over y >= 0
            # This is convex in y
            hi = max(2.0 * x + 0.5, np.sqrt(max(b.max(), 0.0) + 1.0), 1.0)
            # Ensure hi is large enough
            for _ in range(5):
                val_lo = _obj_cd(c_arr, s_arr, base_max, has_quad, quad_c, 0.0)
                val_mid = _obj_cd(c_arr, s_arr, base_max, has_quad, quad_c, hi * 0.5)
                if val_mid < val_lo and hi < 10.0:
                    hi *= 2.0
                else:
                    break
            
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo_s = 0.0
            a1 = hi - gr * (hi - lo_s)
            b1 = lo_s + gr * (hi - lo_s)
            fa1 = _obj_cd(c_arr, s_arr, base_max, has_quad, quad_c, a1)
            fb1 = _obj_cd(c_arr, s_arr, base_max, has_quad, quad_c, b1)
            
            for _ in range(35):
                if fa1 < fb1:
                    hi = b1
                    b1 = a1
                    fb1 = fa1
                    a1 = hi - gr * (hi - lo_s)
                    fa1 = _obj_cd(c_arr, s_arr, base_max, has_quad, quad_c, a1)
                else:
                    lo_s = a1
                    a1 = b1
                    fa1 = fb1
                    b1 = lo_s + gr * (hi - lo_s)
                    fb1 = _obj_cd(c_arr, s_arr, base_max, has_quad, quad_c, b1)
            
            y_opt = (lo_s + hi) / 2.0
            if _obj_cd(c_arr, s_arr, base_max, has_quad, quad_c, 0.0) <= _obj_cd(c_arr, s_arr, base_max, has_quad, quad_c, y_opt):
                y_opt = 0.0
            
            if abs(y_opt - x) > 1e-10:
                a[i] = y_opt
                any_improved = True
        
        a = normalize(a)
        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-11):
            best_M = M
            best_a = a.copy()
            if verbose and sweep % 10 == 0:
                print(f"    CD sweep {sweep}: M={best_M:.10f}", flush=True)
        elif not any_improved:
            break
        
        sweep += 1
    
    if verbose:
        print(f"    CD done: M={best_M:.10f}, sweeps={sweep}", flush=True)
    
    return best_a


def _obj_cd(c_arr, s_arr, base_max, has_quad, quad_c, y):
    """Objective for coordinate descent at value y."""
    vals = c_arr + s_arr * y
    m = base_max
    if vals.size > 0:
        vm = np.max(vals)
        if vm > m:
            m = vm
    if has_quad:
        qv = quad_c + y * y
        if qv > m:
            m = qv
    return m


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.01, 0.05, 0.1, 0.2]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(p_n / p_n.sum())
    
    inits.append(np.ones(n) / n)
    
    # Parabolic
    poly = t * (1 - t) + 0.05
    inits.append(poly / poly.sum())
    
    # U-shaped
    poly2 = (2 * t - 1) ** 2 + 0.05
    inits.append(poly2 / poly2.sum())
    
    # Cosine windows
    for freq in [1, 2, 3, 5]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    # Gaussian
    gauss = np.exp(-6.0 * (t - 0.5) ** 2) + 0.03
    inits.append(gauss / gauss.sum())
    
    # Spectral: try to concentrate energy at few frequencies
    a_spec = np.ones(n) / n
    target_freqs = rng.choice(n, size=max(2, n // 64), replace=False)
    for _ in range(30):
        F = np.fft.rfft(a_spec)
        F_mag = np.abs(F)
        # Boost target frequencies, suppress others
        target_mask = np.zeros_like(F_mag)
        valid_targets = target_freqs[target_freqs < len(F_mag)]
        target_mask[valid_targets] = 1.0
        boost = 0.1 * (target_mask - F_mag / F_mag.sum() * len(F_mag) / max(1, len(valid_targets)))
        F_new = F + boost * F
        a_spec = np.real(np.fft.irfft(F_new, n))
        a_spec = np.maximum(a_spec, 1e-6)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    
    # Smoothed random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: LP best-response with cutting planes + exact line search,
    coordinate descent polishing, multi-start, multi-length.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)
    
    n_parent = parent.size
    
    # === Phase 1: LP optimization on parent length ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP optimization on n={n_parent}", flush=True)
    print(f"{'='*60}", flush=True)
    
    t_phase1_deadline = time.monotonic() + 0.5 * (deadline - time.monotonic())
    
    a_lp = run_lp_iterations(parent, t_phase1_deadline, max_iters=1000, verbose=True)
    tracker.update(a_lp)
    
    # Coordinate descent polish
    if time.monotonic() < deadline - 100:
        a_cd = coordinate_descent_polish(a_lp, min(deadline - 80, time.monotonic() + 200), verbose=True)
        tracker.update(a_cd)
    
    tracker.report("After Phase 1")
    
    # === Phase 2: LP on resampled/modified length ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    remaining = deadline - time.monotonic() - 100
    if remaining > 50:
        # Try a few nearby lengths
        candidates_n = []
        for delta in [128, -128, 256, -256, 64, -64]:
            nn = n_parent + delta
            if 256 <= nn <= 2048:
                candidates_n.append(nn)
        # Also try rounding to "nice" lengths
        for nice_n in [512, 640, 768, 896, 1024, 1280]:
            if nice_n not in candidates_n and abs(nice_n - n_parent) <= 512:
                candidates_n.append(nice_n)
        
        candidates_n = list(set(candidates_n))[:3]
        
        time_per_len = remaining / max(1, len(candidates_n))
        
        for nn in candidates_n:
            if time.monotonic() > deadline - 60:
                break
            print(f"\n  Trying n={nn}, time={time_per_len:.0f}s", flush=True)
            a_resample = resample_seq(tracker.best if tracker.best is not None else parent, nn)
            a_resample = normalize(a_resample)
            
            sub_deadline = min(deadline - 40, time.monotonic() + time_per_len * 0.7)
            a_opt = run_lp_iterations(a_resample, sub_deadline, max_iters=500, verbose=False)
            
            # Quick CD polish
            if time.monotonic() < deadline - 30:
                a_opt = coordinate_descent_polish(a_opt, min(deadline - 20, time.monotonic() + time_per_len * 0.3), verbose=False)
            
            tracker.update(a_opt)
    
    tracker.report("After Phase 2")
    
    # === Phase 3: Intensive LP polish on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Intensive LP polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t_polish = min(deadline - 10, time.monotonic() + (deadline - time.monotonic()) * 0.5)
        a_polish = run_lp_iterations(tracker.best, t_polish, max_iters=2000, verbose=True)
        tracker.update(a_polish)
        
        # Final CD
        if time.monotonic() < deadline - 5:
            a_final_cd = coordinate_descent_polish(a_polish, deadline - 3, verbose=False)
            tracker.update(a_final_cd)
    
    tracker.report("After Phase 3")
    
    # === Phase 4: Multi-start diverse inits ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-start diverse inits", flush=True)
    print(f"{'='*60}", flush=True)
    
    if time.monotonic() < deadline - 30:
        n_best = tracker.best.size if tracker.best is not None else n_parent
        remaining = deadline - time.monotonic() - 20
        inits = generate_diverse_inits(n_best, rng, tracker.best)
        
        # Sort by initial quality
        init_scores = []
        for a_init in inits:
            M_init = fast_conv(a_init).max()
            init_scores.append((M_init, a_init))
        init_scores.sort()
        
        # Try top diverse inits
        time_per_init = remaining / max(1, min(5, len(init_scores)))
        tried = 0
        
        for M_init, a_init in init_scores[:min(5, len(init_scores))]:
            if time.monotonic() > deadline - 10:
                break
            if tried >= 3:
                break
            sub_dl = min(deadline - 5, time.monotonic() + time_per_init)
            a_opt = run_lp_iterations(a_init, sub_dl, max_iters=300, verbose=False)
            if time.monotonic() < deadline - 5:
                a_opt = coordinate_descent_polish(a_opt, min(deadline - 3, time.monotonic() + 20), verbose=False)
            tracker.update(a_opt)
            tried += 1
    
    tracker.report("After Phase 4")
    
    # === Final ===
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
