    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    """Compute subgradient of (conv(a,a))_{k_star} w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    if k_star == 0:
        grad[:] = 2.0 * a
    else:
        # d/da_i [sum_j a_j a_{j+k}] = a_{i+k} + a_{i-k}
        for i in range(n):
            if i + k_star < n:
                grad[i] += a[i + k_star]
            if i - k_star >= 0:
                grad[i] += a[i - k_star]
    return grad

    """Compute weighted subgradient over top-k peaks."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    
    # Get top-k peaks
    top_indices = np.argsort(conv)[::-1][:top_k]
    
    # Softmax weights
    peaks = conv[top_indices]
    weights = np.exp(-(peaks - cmax) / (temperature * cmax + 1e-15))
    weights /= np.sum(weights)
    
    # Weighted sum of subgradients
    grad = np.zeros(n)
    for w, k in zip(weights, top_indices):
        grad += w * subgradient_at_peak(a, int(k))
    
    return grad, cmax

        accepted = False
        for _ in range(20):

            log_a = np.log(a + 1e-30) - eta_try * grad_centered

            a_new = np.exp(log_a)
            a_new = a_new / np.sum(a_new)
            a_new = np.maximum(a_new, 0)  # safety
            if np.sum(a_new) < 1e-10:

            a_new = a_new / np.sum(a_new)

            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
        
        # Decrease top_k over time for more precise targeting
        # (handled by temperature)
    
    return best_a
