def compute_conv(a):

        else:
            weights /= w_sum

        # Gradient via FFT
        w_ext = np.zeros(2 * n - 1)
        w_ext[:] = weights
        a_flip = a[::-1]
        Wf = rfft(w_ext, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Center gradient (remove mean to preserve sum)
        grad_centered = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad_centered))
        if grad_norm < 1e-15:
            break

        # Backtracking line search with Armijo condition
        alpha = 0.01  # Armijo parameter
        c discounts = 0.5
        best_step = 0.0
        best_trial_obj = obj_current

        for backtrack in range(15):
            a_trial = a + step_size * grad_norm * (-grad_centered / grad_norm)
            # Simplified: a_trial = a - step_size * grad_centered
            a_trial = a - step_size * grad_centered
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = float(np.sum(a_trial))
            if s_trial < 1e-10:
                break
            a_trial = a_trial / s_trial

            conv_trial = compute_conv(a_trial)
            obj_trial = float(np.max(conv_trial))  # sum is 1

            # Armijo: obj_trial <= obj_current - alpha * step_size * ||grad||^2
            if obj_trial <= obj_current - alpha * step_size * grad_norm * grad_norm:
                best_step = step_size
                best_trial_obj = obj_trial
                break

            step_size *= 0.5

        if best_step > 0:
            a = a_trial
            new_obj = compute_obj(a)
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                no_improve_count = 0
            else:
                # Keep best but continue from current
                no_improve_count += 1
                a = best_a.copy()
        else:
            no_improve_count += 1

        # Adapt temperature
        tau *= 0.96
        tau = max(tau, cmax * 1e-6)

        # Reheat if stuck
        if no_improve_count >= max_no_improve:
            tau = cmax / n * 0.5
            no_improve_count = 0
            # Perturb
            rng = np.random.default_rng(step_count * 104729 + 7)
            a = normalize(a * np.exp(rng.normal(0, 0.05, n)))
            step_size = base_step

        # Adapt step size
        if no_improve_count > 10:
            step_size = max(base_step * 0.5, 0.005)
        elif no_improve_count == 0:
            step_size = min(base_step * 1.1, 0.15)

        # Periodic LP boost
        if step_count % 200 == 0 and n <= 1000:
            remaining = time_budget - (time.monotonic() - start)
            if remaining > 5:
                conv_cur = compute_conv(a)
                M_cur = float(np.max(conv_cur))
                g = solve_lp_max_sum(a, M_cur, time_limit=3)
                if g is not None:
                    a_blend = 0.5 * a + 0.5 * g
                    a = normalize(a_blend)
                    obj_blend = compute_obj(a)
                    if obj_blend < best_obj:
                        best_obj = obj_blend
                        best_a = a.copy()

    return best_a


def subgradient_polish(a, time_budget=30.0):
    """
    Use exact subgradient at the convolution maximum for final polishing.
    When tau -> 0, the soft-max gradient approaches the subgradient.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.02

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        # Find all positions achieving (near) the maximum
        threshold = cmax * 0.999
        active = np.where(conv >= threshold)[0]

        if len(active) == 0:
            break

        # Subgradient: average over active maxima
        size = 1
        while size < 2 * n - 1:
            size <<= 1

        grad = np.zeros(n)
        for k in active:
            # d(conv[k])/da[i] = 2*a[k-i] if 0 <= k-i <= n-1
            for i in range(n):
                k_minus_i = k - i
                if 0 <= k_minus_i < n:
                    grad[i] += 2.0 * a[k_minus_i]
        grad /= len(active)

        grad_centered = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad_centered))
        if grad_norm < 1e-12:
            break

        # Backtracking
        s = step_size
        improved = False
        for _ in range(20):
            a_trial = a - s * grad_centered
            a_trial = np.maximum(a_trial, 0.0)
            tsum = float(np.sum(a_trial))
            if tsum < 1e-10:
                break
            a_trial = a_trial / tsum
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                break
            s *= 0.5

        if not improved:
            step_size *= 0.7
            if step_size < 1e-6:
                break

    return best_a


def lp_alternate_enhanced(a, max_iters=40, time_budget=80.0):
    """
    LP alternating with golden section search, enhanced with
    adaptive M and multiple LP starts.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_max_sum(a, M, time_limit=max(5, int(remaining * 0.5)))
        if g is None:
            break

        # Golden section search for optimal blend
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)

        def obj_at_blend(t):
            a_try = (1.0 - t) * a + t * g
            a_try = normalize(a_try)
            return compute_obj(a_try)

        fc = obj_at_blend(c)
        fd = obj_at_blend(d)
        for _ in range(12):
            if hi - lo < 0.01:
                break
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = obj_at_blend(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at_blend(d)

        t_opt = (lo + hi) / 2.0
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller blend
            for t_try in [0.1, 0.05, 0.02]:
                a_try = normalize((1.0 - t_try) * a + t_try * g)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # Perturb to escape
                rng = np.random.default_rng((it * 6151 + 999) % (2**31))
                a = normalize(best_a * np.exp(rng.normal(0, 0.03 + 0.01 * (it % 7), n)))

    return best_a


def spectral_equalization_v2(a, max_iters=80, time_budget=45.0):
    """
    Enhanced spectral equalization with adaptive compression and
    better non-negativity handling.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Adaptive compression: start aggressive, ease off
        progress = it / float(max_iters)
        alpha = 0.3 + 0.5 * progress

        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        # Preserve DC component (sum)
        P_new[0] = P[0]

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        # Soft non-negativity: squash negatives smoothly
        a_new = a_new + np.maximum(-a_new, 0.0) * 0.5  # partial reflection
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend with previous
            a = 0.6 * a + 0.4 * a_new
            a = normalize(a)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation with normalization."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    return normalize(a_new)


def diverse_resample(a, new_n, method='interp', seed=42):
    """Resample with different strategies."""
    n = len(a)
    if n == new_n:
        return a.copy()
    rng = np.random.default_rng(seed)

    if method == 'interp':
        return resample_sequence(a, new_n)
    elif method == 'zero_pad':
        if new_n > n:
            a_new = np.zeros(new_n)
            a_new[:n] = a
            # Smooth the boundary
            pad = new_n - n
            for i in range(min(5, pad)):
                a_new[n - 1 + i] *= (1.0 - i / (pad + 1))
            return normalize(a_new)
        else:
            # Truncate keeping the largest values
            keep = new_n
            a_sorted_idx = np.argsort(a)[::-1]
            a_keep = np.zeros(new_n)
            for i, idx in enumerate(a_sorted_idx[:keep]):
                a_keep[int(idx)] = a[int(idx)]
            return normalize(a_keep)
    elif method == 'repeat':
        if new_n >= n:
            full_repeats = new_n // n
            remainder = new_n % n
            a_new = np.zeros(new_n)
            for rep in range(full_repeats):
                a_new[rep * n:(rep + 1) * n] = a
            a_new[full_repeats * n:full_repeats * n + remainder] = a[:remainder]
            # Add small noise to break exact repetition
            a_new *= np.exp(rng.normal(0, 0.02, new_n))
            return normalize(a_new)
        else:
            return resample_sequence(a, new_n)
    else:
        return resample_sequence(a, new_n)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def escape_local_minima(a, n_restarts=3, time_budget=30.0):
    """Try multiple perturbations to escape local minima."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    alloc_per_restart = time_budget / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - start > time_budget:
            break

        # Different perturbation strategies
        strategy = r % 4
        if strategy == 0:
            # Gaussian multiplier
            a_try = best_a * np.exp(rng.normal(0, 0.1, n))
        elif strategy == 1:
            # Sinusoidal modulation
            freq = rng.uniform(0.3, 3.0)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.arange(n) / n
            mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * x + phase)
            a_try = best_a * mod
        elif strategy == 2:
            # Block reversal
            a_try = best_a.copy()
            block_len = int(rng.integers(n // 5, n // 2))
            start_idx = int(rng.integers(0, n - block_len))
            a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]
        else:
            # Blend with structured pattern
            x = np.arange(n) / n
            patterns = [
                np.cos(np.pi * x) ** 2,
                1.0 + np.cos(2 * np.pi * x),
                np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2)),
                x ** 0.5 * (1 - x) ** 0.5,
            ]
            pattern = patterns[int(rng.integers(0, len(patterns)))]
            pattern = normalize(pattern)
            a_try = 0.7 * best_a + 0.3 * pattern

        a_try = normalize(a_try)

        # Quick optimize
        a_try = soft_max_gradient_descent(a_try, time_budget=alloc_per_restart * 0.6,
                                          base_step=0.04)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function with multi-phase strategy:
    Phase 1: Intensive soft-max gradient with momentum at parent length
    Phase 2: LP alternating for structural improvements
    Phase 3: Spectral equalization for diverse optimization direction
    Phase 4: Multi-n exploration with diverse resampling
    Phase 5: Subgradient polishing and final refinement
    Phase 6: Escape attempts and diverse restarts
    Phase 7: Final validation
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # =========================================================
    # Phase 1: Intensive soft-max gradient (primary optimizer)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Soft-max gradient descent (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())

        # Multiple gradient passes with decreasing step sizes
        for pass_i in range(4):
            if time.monotonic() > deadline - 400:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.2, 100.0)
            base_step = 0.06 / (1 + 0.3 * pass_i)

            a_new = soft_max_gradient_descent(a, time_budget=alloc,
                                              base_step=base_step,
                                              momentum_init=0.9)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  Grad pass {pass_i}: {best_score:.8f} **")
            else:
                print(f"  Grad pass {pass_i}: {obj_new:.8f}")
                # Blend
                a = 0.7 * normalize(a) + 0.3 * normalize(a_new)

    # =========================================================
    # Phase 2: LP alternating for structural improvements
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP alternating (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None and n_parent <= 1000:
        a = normalize(best_seq.copy())
        for cycle in range(3):
            if time.monotonic() > deadline - 300:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.2, 80.0)

            a_new = lp_alternate_enhanced(a, max_iters=25, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  LP cycle {cycle}: {best_score:.8f} **")
            else:
                print(f"  LP cycle {cycle}: {obj_new:.8f}")
                a = 0.8 * normalize(a) + 0.2 * normalize(a_new)

    # =========================================================
    # Phase 3: Spectral equalization
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Spectral equalization")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        if time.monotonic() < deadline - 250:
            a_spec = spectral_equalization_v2(a, max_iters=60, time_budget=50.0)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                a = a_spec
                print(f"  Spectral: {best_score:.8f} **")
            else:
                print(f"  Spectral: {obj_spec:.8f}")
                # Blend and continue
                a = 0.6 * normalize(a) + 0.4 * normalize(a_spec)
                a = normalize(a)

                # Quick gradient after spectral
                if time.monotonic() < deadline - 200:
                    a_grad = soft_max_gradient_descent(a, time_budget=30, base_step=0.03)
                    obj_grad = compute_obj(a_grad)
                    if obj_grad < best_score:
                        best_score = obj_grad
                        best_seq = a_grad.copy()
                        a = a_grad
                        print(f"  Post-spectral grad: {best_score:.8f} **")

    # =========================================================
    # Phase 4: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        # Explore a range of n values
        n_options = sorted(set([
            n_cur - 200, n_cur - 150, n_cur - 100, n_cur - 50,
            n_cur - 25, n_cur + 25, n_cur + 50, n_cur + 100,
            n_cur + 150, n_cur + 200, n_cur + 300
        ]))
        n_options = [n for n in n_options if 200 <= n <= 1500]

        for n_try in n_options:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # Use different resampling methods
            if n_try < n_cur:
                a_try = diverse_resample(best_seq, n_try, method='interp', seed=n_try)
            elif n_try > n_cur:
                a_try = diverse_resample(best_seq, n_try, method='zero_pad', seed=n_try)
            else:
                a_try = normalize(best_seq.copy())

            # Quick optimization
            alloc = min(remaining * 0.15, 30.0)
            a_try = soft_max_gradient_descent(a_try, time_budget=alloc, base_step=0.04)
            obj_try = compute_obj(a_try)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_try:.8f}")

    # Phase 5: Subgradient polishing
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Subgradient polishing")

        if time.monotonic() < deadline - 60:
            a_pol = subgradient_polish(a, time_budget=40.0)
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a = a_pol
                print(f"  Subgradient: {best_score:.8f} **")
            else:
                print(f"  Subgradient: {obj_pol:.8f}")

            # Follow up with soft-max at very low temperature
            if time.monotonic() < deadline - 30:
                conv_cur = compute_conv(a)
                cmax_cur = float(np.max(conv_cur))
                a_pol2 = soft_max_gradient_descent(a, time_budget=25,
                                                   tau_init=cmax_cur / 100.0,
                                                   base_step=0.02)
                obj_pol2 = compute_obj(a_pol2)
                if obj_pol2 < best_score:
                    best_score = obj_pol2
                    best_seq = a_pol2.copy()
                    print(f"  Low-temp grad: {best_score:.8f} **")
                else:
                    print(f"  Low-temp grad: {obj_pol2:.8f}")

    # Phase 6: Escape attempts and diverse restarts
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Diverse restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        if time.monotonic() < deadline - 40:
            remaining = deadline - time.monotonic()
            n_escapes = max(2, int(remaining / 30))
            a_esc = escape_local_minima(best_seq, n_restarts=n_escapes,
                                        time_budget=min(remaining * 0.8, 120.0))
            obj_esc = compute_obj(a_esc)
            if obj_esc < best_score:
                best_score = obj_esc
                best_seq = a_esc.copy()
                print(f"  Escape: {best_score:.8f} **")
            else:
                print(f"  Escape: {obj_esc:.8f}")

    # =========================================================
    # Phase 7: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        n_final = len(a_final)

        # LP polish if feasible
        if n_final <= 900 and time.monotonic() < deadline - 20:
            alloc_lp = min((deadline - time.monotonic()) * 0.5, 30.0)
            if alloc_lp > 10:
                a_lp = lp_alternate_enhanced(a_final, max_iters=15, time_budget=alloc_lp)
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_final = a_lp
                    print(f"  Final LP: {best_score:.8f} **")
                else:
                    print(f"  Final LP: {obj_lp:.8f}")

        # Final soft-max at very low temperature
        if time.monotonic() < deadline - 10:
            conv_f = compute_conv(a_final)
            cmax_f = float(np.max(conv_f))
            alloc_sm = min((deadline - time.monotonic()) * 0.7, 20.0)
            if alloc_sm > 5:
                a_sm = soft_max_gradient_descent(a_final, time_budget=alloc_sm,
                                                 tau_init=cmax_f / 200.0,
                                                 base_step=0.01)
                obj_sm = compute_obj(a_sm)
                if obj_sm < best_score:
                    best_score = obj_sm
                    best_seq = a_sm.copy()
                    print(f"  Final soft-max: {best_score:.8f} **")
                else:
                    print(f"  Final soft-max: {obj_sm:.8f}")

    # =========================================================
    # Final validation and return

    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Ensure sum constraint
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100
        s = 100.0

    # Validate with the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
