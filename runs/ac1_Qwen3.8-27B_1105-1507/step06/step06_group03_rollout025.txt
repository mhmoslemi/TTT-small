Optimized search for minimum J(a) = 2n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        w_rev = w[::-1]
        wr_fft = rfft(w_rev, size)
        conv_awr = irfft(a_fft * wr_fft, size)[:3 * n - 2]

        grad = np.zeros(n)
        for i in range(n):
            idx2 = 2 * n - 2 - i
            if 0 <= idx2 < len(conv_awr):
                grad[i] = conv_wa[i] + conv_awr[idx2]
            else:
                grad[i] = conv_wa[i]

        # Project gradient to tangent space of simplex
        grad = grad - np.mean(grad)

        # Adam update
        t_count += 1
        lr = lr_init / (1.0 + 0.05 * t_count ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        # Compute objective
        s = a_new.sum()
        if s < 1e-10:
            a_new = best_a.copy()
            no_improve += 1
        else:
            a_norm = a_new / s
            conv_new = fast_conv(a_norm)
            obj_new = float(2 * n * conv_new.max())

            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_norm.copy()
                no_improve = 0
                a = a_norm.copy()
                conv = conv_new.copy()
            else:
                no_improve += 1
                a = a_norm.copy()
                conv = conv_new.copy()

        # Adaptive tau
        if no_improve > 50:
            tau = min(tau * tau_growth, tau_max)
        if no_improve > max_no_improve:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            a = project_simplex(best_a + rng.normal(0, 0.03, n))
            conv = fast_conv(a)
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0
            no_improve = 0
            tau = max(tau_init, tau * 0.5)

        if verbose and it % 500 == 0:
            print(f"      AGD it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def coord_descent_refined(a, deadline, max_sweeps=100000, seed=42, verbose=False):
    """
    Refined coordinate descent with adaptive candidate values.
    Uses incremental convolution updates.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 500

    # Precompute which conv indices are affected by each coordinate
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n) if sweep % 3 == 0 else range(n)
        improved = False

        for i in order:
            old = a[i]
            # Candidate values
            if old < 1e-10:
                candidates = [0.0, 1e-5, 1e-4, 5e-4, 1e-3, 5e-3, 0.01]
            else:
                # Adaptive candidates around current value
                candidates = [0.0,
                              old * 0.1, old * 0.3, old * 0.5, old * 0.7,
                              old, old * 1.3, old * 1.5, old * 1.7, old * 2.0]

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            a_slice = a[lo - i:hi - i + 1].copy()
            conv_base = conv[lo:hi + 1].copy()

            best_val = old
            best_local = 2.0 * n * conv.max()

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue

                conv[lo:hi + 1] = conv_base + 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

            # Restore conv
            conv[lo:hi + 1] = conv_base
            if valid_kq and best_val != old:
                pass  # will be set below

            if best_val != old:
                delta = best_val - old
                conv[lo:hi + 1] = conv_base + 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 200 == 0:
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def golden_section_polish(a, deadline, max_coors_per_iter=None, verbose=False):
    """
    Golden-section line search on the most impactful coordinates.
    Focuses on coordinates that contribute most to the max conv value.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    # Identify active positions (where conv is close to max)
    conv_max = conv.max()
    active_thresh = conv_max * 0.95
    active_positions = np.where(conv >= active_thresh)[0]

    # Identify coordinates that affect active positions most
    coord_impact = np.zeros(n)
    for k in active_positions:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            coord_impact[j] += conv[k]

    # Sort by impact
    if max_coors_per_iter is not None:
        top_coors = np.argsort(coord_impact)[::-1][:max_coors_per_iter]
    else:
        top_coors = np.argsort(coord_impact)[::-1]

    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    no_improve = 0

    for i in top_coors:
        if time.monotonic() > deadline:
            break

        lo = max(0, i)
        hi = min(2 * n - 2, i + n - 1)
        kq = 2 * i
        valid_kq = kq <= 2 * n - 2
        a_slice = a[lo - i:hi - i + 1].copy()
        conv_base = conv[lo:hi + 1].copy()

        # Golden section search for optimal a[i]
        # Objective as function of x: 2n * max(conv_updated) / (s - old + x)^2
        old = a[i]
        lo_x = 0.0
        hi_x = 5.0 * max(old, 0.01) if old > 0 else 0.05

        def obj_at_x(x):
            delta = x - old
            conv[lo:hi + 1] = conv_base + 2.0 * delta * a_slice
            if valid_kq:
                conv[kq] += delta * delta
            new_s = s - old + x
            if new_s < 1e-10:
                res = float('inf')
            else:
                res = 2.0 * n * conv.max() / (new_s * new_s)
            conv[lo:hi + 1] = conv_base
            return res

        # Golden section
        c_x = (1 - phi) * lo_x + phi * hi_x
        d_x = phi * lo_x + (1 - phi) * hi_x
        f_c = obj_at_x(c_x)
        f_d = obj_at_x(d_x)

        for _ in range(20):
            if f_c < f_d:
                hi_x = d_x
                d_x = c_x
                f_d = f_c
                c_x = (1 - phi) * lo_x + phi * hi_x
                f_c = obj_at_x(c_x)
            else:
                lo_x = c_x
                c_x = d_x
                f_c = f_d
                d_x = phi * lo_x + (1 - phi) * hi_x
                f_d = obj_at_x(d_x)

        opt_x = (lo_x + hi_x) / 2.0
        opt_val = obj_at_x(opt_x)

        if opt_val < best_obj - 1e-14:
            conv[lo:hi + 1] = conv_base + 2.0 * (opt_x - old) * a_slice
            if valid_kq:
                conv[kq] += (opt_x - old) ** 2
            a[i] = opt_x
            s = s - old + opt_x
            best_obj = 2.0 * n * conv.max() / (s * s)
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))
        # Symmetric projection
        sym_p = 0.5 * (p + p[::-1])
        sym_p = project_simplex(sym_p)
        inits.append(sym_p)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns
    for freq in [1, 2, 3, 4, 5, 7]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    # Gaussian bumps
    for center in [0.2, 0.35, 0.5, 0.65, 0.8]:
        for width in [0.08, 0.15, 0.25, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.05))

    # Product of cosines (more structured)
    for f1 in [2, 3, 5]:
        for f2 in [3, 4, 7]:
            inits.append(project_simplex(1.0 + 0.4 * np.cos(2 * np.pi * f1 * t) *
                                              np.cos(2 * np.pi * f2 * t)))

    # Power-law
    for alpha in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    # Smooth random
    ksize = max(5, n // 12)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    # Window functions
    inits.append(project_simplex(np.hanning(n)))
    inits.append(project_simplex(np.hamming(n)))
    inits.append(project_simplex(np.blackman(n)))

    # Alternating with damping
    alt = 1.0 + 0.3 * np.exp(-t * 3) * np.cos(2 * np.pi * 5 * t)
    inits.append(project_simplex(alt))

    return inits


def solve_sequence(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj - 1e-14:
            best_a = a_new.copy()
            best_obj = obj
        return best_obj

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 1: Quick coordinate descent (20%)
    t1 = time.monotonic() + 0.20 * remaining
    a_cd = coord_descent_refined(a, deadline=t1, max_sweeps=50000, seed=42)
    try_update(a_cd)

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 2: Adaptive gradient descent (40%)
    t2 = time.monotonic() + 0.40 * remaining
    a_grad, obj_grad = adaptive_grad_descent(best_a, deadline=t2,
                                             tau_init=3.0, tau_max=150.0,
                                             lr_init=0.08)
    try_update(a_grad)

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 3: Golden section polish (20%)
    t3 = time.monotonic() + 0.20 * remaining
    a_gold = golden_section_polish(best_a, deadline=t3)
    try_update(a_gold)

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 4: More coordinate descent + gradient (20%)
    t4a = time.monotonic() + 0.12 * remaining
    a_cd2 = coord_descent_refined(best_a, deadline=t4a, max_sweeps=50000, seed=123)
    try_update(a_cd2)

    remaining = deadline - time.monotonic()
    if remaining > 15:
        t4b = time.monotonic() + remaining - 5
        a_grad2, _ = adaptive_grad_descent(best_a, deadline=t4b,
                                           tau_init=10.0, tau_max=100.0,
                                           lr_init=0.03)
        try_update(a_grad2)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses adaptive gradient descent, refined coordinate descent, golden-section polish,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Multi-init at parent length (first 400s) ===
    phase1_deadline = t_start + 400.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-init optimization at n={n_parent}", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_diverse_inits(n_parent, rng, parent)
    # Limit to top inits based on quick evaluation
    valid_inits = []
    for init in inits:
        init = np.maximum(init, 0)
        if init.sum() < 1e-10:
            continue
        init = init / init.sum()
        obj = compute_J(init)
        valid_inits.append((obj, init))
    valid_inits.sort(key=lambda x: x[0])
    valid_inits = [v[1] for v in valid_inits[:15]]

    per_init_time = (phase1_deadline - time.monotonic()) / max(1, len(valid_inits)) - 5

    for idx in range(len(valid_inits)):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 30:
            break
        alloc = min(per_init_time, remaining_check - 5)
        if alloc < 15:
            break

        a0 = valid_inits[idx]
        a_opt, obj_opt = solve_sequence(a0, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.8f} n={n_parent}", flush=True)

    # === PHASE 2: Multi-length exploration (next 500s) ===
    phase2_deadline = t_start + 900.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Try lengths near parent and some others
    candidate_ns = set()
    for delta in [-5, -3, -2, -1, 1, 2, 3, 5, 8, 13, 21, 32, 50]:
        new_n = n_parent + delta
        if 64 <= new_n <= 3000:
            candidate_ns.add(new_n)
    # Also try some "round" lengths
    for n_try in [256, 384, 512, 640, 768, 896, 1024, 1152, 1280, 1536, 2048]:
        if 64 <= n_try <= 3000:
            candidate_ns.add(n_try)

    candidate_ns = sorted(candidate_ns - {n_parent}, key=lambda n: abs(n - n_parent))

    for new_n in candidate_ns[:8]:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 60:
            break
        alloc = min(80.0, remaining_check - 20)
        if alloc < 40:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = solve_sequence(a_ext, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive local optimization on best (remaining time) ===
    phase3_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive local optimization "
          f"(budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            a_curr = best_seq.copy()

            # Pass 1: Long coordinate descent
            t_pass1 = time.monotonic() + 0.20 * remaining
            print(f"  CD pass (budget {t_pass1 - time.monotonic():.0f}s)...", flush=True)
            a_curr = coord_descent_refined(a_curr, max_sweeps=100000,
                                           deadline=t_pass1, seed=777)
            update_best(a_curr)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass 2: Gradient descent with higher tau
                t_pass2 = time.monotonic() + 0.25 * remaining
                print(f"  Grad descent pass (budget {t_pass2 - time.monotonic():.0f}s)...",
                      flush=True)
                a_grad, _ = adaptive_grad_descent(best_seq, deadline=t_pass2,
                                                  tau_init=5.0, tau_max=120.0,
                                                  lr_init=0.05)
                update_best(a_grad)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass 3: Golden section polish
                t_pass3 = time.monotonic() + 0.20 * remaining
                print(f"  Golden section pass (budget {t_pass3 - time.monotonic():.0f}s)...",
                      flush=True)
                a_gold = golden_section_polish(best_seq, deadline=t_pass3)
                update_best(a_gold)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass 4: Another round of CD with different seed
                t_pass4 = time.monotonic() + 0.20 * remaining
                print(f"  CD pass 2 (budget {t_pass4 - time.monotonic():.0f}s)...", flush=True)
                a_curr = coord_descent_refined(best_seq, max_sweeps=100000,
                                               deadline=t_pass4, seed=999)
                update_best(a_curr)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 20:
                # Pass 5: Final gradient with very high tau
                t_pass5 = time.monotonic() + remaining - 5
                print(f"  Final grad (budget {t_pass5 - time.monotonic():.0f}s)...", flush=True)
                a_grad2, _ = adaptive_grad_descent(best_seq, deadline=t_pass5,
                                                   tau_init=20.0, tau_max=200.0,
                                                   lr_init=0.02)
                update_best(a_grad2)

                # Final CD
                if time.monotonic() < phase3_deadline - 10:
                    a_final = coord_descent_refined(best_seq, max_sweeps=50000,
                                                    deadline=phase3_deadline - 5, seed=1111)
                    update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
