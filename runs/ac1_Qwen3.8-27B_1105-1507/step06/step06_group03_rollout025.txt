def compute_upper_envelope(lines_m, lines_c):
    """Compute upper envelope of lines y = m*x + c for x >= 0.
    Returns breakpoints and line indices for binary search evaluation."""
    # Sort by slope
    order = np.argsort(lines_m)
    m = lines_m[order]
    c = lines_c[order]
    
    # Build upper envelope
    envelope = []  # list of (x_start, m, c)
    for idx in range(len(m)):
        mi, ci = m[idx], c[idx]
        while envelope:
            # Check if the last line in envelope is dominated
            # Intersection of new line with last envelope line:
            # m_last * x + c_last = mi * x + ci
            # x = (ci - c_last) / (m_last - mi)
            x_break = (ci - envelope[-1][2]) / (envelope[-1][1] - mi) if mi != envelope[-1][1] else 0
            if x_break <= envelope[-1][0]:
                envelope.pop()
            else:
                break
        if not envelope:
            envelope.append((0.0, mi, ci))
        else:
            x_break = (ci - envelope[-1][2]) / (envelope[-1][1] - mi) if mi != envelope[-1][1] else 0
            if x_break > envelope[-1][0]:
                envelope.append((x_break, mi, ci))
            # else: this line is never optimal
    
    return envelope

def eval_envelope(envelope, x):
    """Evaluate upper envelope at point x."""
    if not envelope:
        return -np.inf
    # Binary search for the segment
    lo, hi = 0, len(envelope) - 1
    while lo < hi:
        mid = (lo + hi + 1) // 2
        if envelope[mid][0] <= x:
            lo = mid
        else:
            hi = mid - 1
    return envelope[lo][1] * x + envelope[lo][2]

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)

    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

Exact 1D coordinate descent for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))².
Uses vectorized inner loops for speed, enabling thousands of sweeps in the time budget.
Combines exact coordinate optimization with multi-n exploration and random restarts.

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent_exact(a, deadline, x_samples=150, seed=0, verbose=False):
    """
    Coordinate descent with exact 1D optimization via vectorized evaluation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    conv = np.maximum(conv, 0)
    current_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    best_obj = current_obj
    rng = np.random.default_rng(seed)
    no_improve_sweeps = 0
    max_no_improve = 40

    X = np.linspace(0.0, 1.0, x_samples)
    sweep_count = 0

    while time.monotonic() < deadline:
        order = rng.permutation(n)
        sweep_improved = 0

        for i in order:
            old = a[i]
            S = s - old

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            other_idx = np.arange(0, k_hi - k_lo + 1)
            other_vals = a[other_idx]

            # Vectorized conv_new: shape (num_k, x_samples)
            conv_new = conv[k_lo:k_hi+1, None] + 2.0 * (X[None, :] - old) * other_vals[:, None]
            if k_lo <= 2 * i <= k_hi:
                idx_2i = 2 * i - k_lo
                conv_new[idx_2i, :] = conv[2 * i] - old * old + X * X

            # Unaffected max
            max_unaffected = 0.0
            if k_lo > 0:
                mv = conv[:k_lo].max()
                if mv > max_unaffected:
                    max_unaffected = mv
            if k_hi < 2 * n - 2:
                mv = conv[k_hi+1:].max()
                if mv > max_unaffected:
                    max_unaffected = mv

            max_affected = conv_new.max(axis=0)
            full_max = np.maximum(max_affected, max_unaffected)

            denom = (S + X) ** 2
            denom[denom < 1e-20] = 1e-20
            obj_vals = 2.0 * n * full_max / denom

            idx_best = np.argmin(obj_vals)
            x_best = X[idx_best]
            obj_best = obj_vals[idx_best]

            if obj_best < current_obj * (1 - 1e-10):
                delta = x_best - old
                if abs(delta) < 1e-15:
                    continue
                a[i] = x_best
                s = S + x_best
                conv[k_lo:k_hi+1] += 2.0 * delta * other_vals
                if k_lo <= 2 * i <= k_hi:
                    conv[2 * i] += delta * delta
                conv = np.maximum(conv, 0)  # Clean up numerical issues
                new_max = conv.max()
                current_obj = 2.0 * n * new_max
                sweep_improved += 1

        sweep_count += 1
        if sweep_improved == 0:
            no_improve_sweeps += 1
        else:
            no_improve_sweeps = 0

        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()

        if verbose and sweep_count % 100 == 0:
            print(f"    CD sweep {sweep_count}: obj={best_obj:.8f}", flush=True)

        if no_improve_sweeps > max_no_improve:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.01, 0.03, 0.05, 0.1]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns
    for freq in [1, 2, 3, 4, 5]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))

    # Gaussian
    for center in [0.2, 0.5, 0.8]:
        for width in [0.05, 0.1, 0.2, 0.3]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.01))

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    return inits

"""
Exact 1D coordinate descent for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))².
Uses vectorized inner loops enabling thousands of sweeps in the time budget.
Combines exact coordinate optimization with adaptive grids, multi-n exploration,
and perturbation-based restarts to escape local minima.
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent_exact(a, deadline, x_samples=200, seed=0, verbose=False, max_no_improve=35):
    """
    Coordinate descent with exact 1D optimization via vectorized evaluation.
    For each coordinate, evaluates the objective at x_samples grid points and picks the best.
    Uses adaptive grid centered on current value for better resolution.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    conv = np.maximum(conv, 0)
    current_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    best_obj = current_obj
    rng = np.random.default_rng(seed)
    no_improve_sweeps = 0
    sweep_count = 0

    # Base grid
    X_base = np.linspace(0.0, 1.0, x_samples)

    while time.monotonic() < deadline:
        order = rng.permutation(n)
        sweep_improved = 0

        for i in order:
            old = a[i]
            if old < 1e-14:
                # Try small values
                X = np.linspace(0.0, 0.05, x_samples)
            else:
                # Adaptive grid: fine around current value + coarse over [0,1]
                lo = max(0.0, old * 0.2)
                hi = min(1.0, old * 3.0 + 0.01)
                X_fine = np.linspace(lo, hi, x_samples // 2)
                X_coarse = np.linspace(0.0, 1.0, x_samples // 2)
                X = np.unique(np.concatenate([np.array([0.0]), X_fine, X_coarse]))

            S = s - old
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            num_k = k_hi - k_lo + 1
            other_vals = a[:num_k]

            # Vectorized conv_new: shape (num_k, len(X))
            conv_new = conv[k_lo:k_hi + 1, None] + 2.0 * (X[None, :] - old) * other_vals[:, None]
            if k_lo <= 2 * i <= k_hi:
                idx_2i = 2 * i - k_lo
                conv_new[idx_2i, :] = conv[2 * i] - old * old + X * X

            # Unaffected max
            max_unaffected = 0.0
            if k_lo > 0:
                mv = conv[:k_lo].max()
                if mv > max_unaffected:
                    max_unaffected = mv
            if k_hi < 2 * n - 2:
                mv = conv[k_hi + 1:].max()
                if mv > max_unaffected:
                    max_unaffected = mv

            max_affected = conv_new.max(axis=0)
            full_max = np.maximum(max_affected, max_unaffected)

            denom = (S + X) ** 2
            denom[denom < 1e-20] = 1e-20
            obj_vals = 2.0 * n * full_max / denom

            idx_best = np.argmin(obj_vals)
            x_best = X[idx_best]
            obj_best = obj_vals[idx_best]

            if obj_best < current_obj * (1.0 - 1e-10):
                delta = x_best - old
                if abs(delta) < 1e-15:
                    continue
                a[i] = x_best
                s = S + x_best
                conv[k_lo:k_hi + 1] += 2.0 * delta * other_vals
                if k_lo <= 2 * i <= k_hi:
                    conv[2 * i] += delta * delta
                # Clean up small negatives from accumulated FP error
                conv = np.maximum(conv, 0)
                current_obj = 2.0 * n * conv.max()
                sweep_improved += 1

        sweep_count += 1
        if sweep_improved == 0:
            no_improve_sweeps += 1
        else:
            no_improve_sweeps = 0

        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()

        if verbose and sweep_count % 50 == 0:
            print(f"    CD sweep {sweep_count}: obj={best_obj:.8f}", flush=True)

        if no_improve_sweeps > max_no_improve:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.005, 0.01, 0.02, 0.05, 0.1]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns (key for low autocorrelation)
    for freq in [1, 2, 3, 4, 5, 7]:
        for amp in [0.1, 0.2, 0.3, 0.5, 0.7]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))

    # Two-frequency
    for f1, f2 in [(1, 3), (2, 5), (3, 7)]:
        inits.append(project_simplex(1.0 + 0.3 * np.cos(2 * np.pi * f1 * t) + 0.2 * np.cos(2 * np.pi * f2 * t)))

    # Gaussian mixtures
    for center in [0.2, 0.5, 0.8]:
        for width in [0.05, 0.1, 0.15, 0.25]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.01))

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    return inits
