"""
Spectral flattening optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Novel approach: Gerchberg-Saxton-style spectral magnitude flattening in Fourier domain,
combined with adaptive beta annealing, symmetry exploitation, and multi-phase polishing
via smoothed gradient descent and LP mirror descent.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def next_pow2(m):
    """Next power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def spectral_flatten(a, beta_start, beta_end, n_iters, deadline,
                     enforce_symmetry=False, gamma_mix=1.0, verbose=False):
    """
    Gerchberg-Saxton-style spectral magnitude flattening.
    Iteratively flattens |FFT(a)| and projects back to {a >= 0, sum=1}.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    N = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for t in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive beta schedule: ease-in
        frac = t / max(1, n_iters)
        beta = beta_start + (beta_end - beta_start) * (frac ** 0.5)

        # FFT
        F = np.fft.fft(a, N)
        mag = np.abs(F)
        mean_mag = mag.mean()

        if mean_mag < 1e-15:
            break

        # Flatten magnitude: mag_new = mean * (mag/mean)^(1-beta)
        # This shrinks peaks and boosts valleys
        ratio = mag / mean_mag
        # Avoid log(0)
        ratio = np.maximum(ratio, 1e-15)
        mag_new = mean_mag * np.power(ratio, 1.0 - beta)

        # Optional: also slightly adjust phase towards 0 or pi (for real F^2)
        # This helps make conv(a,a) more "even"
        phase = np.angle(F)
        # Soft phase correction: pull phase towards 0 or pi
        # cos(2*phase) measures how "real" F^2 is
        # We nudge phase towards the nearest multiple of pi
        phase_correction = 0.1 * gamma_mix
        # sin(phase) * cos(phase) = 0.5*sin(2*phase); we want to reduce this
        # d(phase)/d(t) = -0.5 * sin(2*phase) * correction
        phase_new = phase - phase_correction * 0.5 * np.sin(2.0 * phase)

        # Reconstruct
        F_new = mag_new * np.exp(1j * phase_new)
        a_new = np.fft.ifft(F_new, N).real[:n]

        # Project to {a >= 0, sum = 1}
        a_new = np.maximum(a_new, 0.0)
        if enforce_symmetry:
            a_new = 0.5 * (a_new + a_new[::-1])
            a_new = np.maximum(a_new, 0.0)

        sn = a_new.sum()
        if sn < 1e-10:
            a_new = best_a.copy()
            no_improve += 10
        else:
            a_new = a_new / sn
            obj_new = compute_J(a_new)
            if obj_new < best_obj - 1e-12:
                best_a = a_new.copy()
                best_obj = obj_new
                no_improve = 0
            else:
                no_improve += 1

        a = a_new

        if no_improve > 500:
            break

        if verbose and t % 200 == 0:
            print(f"    SpecFlatten it {t}: beta={beta:.3f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def spectral_flatten_hybrid(a, deadline, verbose=False):

    - Multiplicative magnitude flattening
    - Additive (gamma) shrinkage
    - Phase correction
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()

    else:

    # Phase 1: Gentle flattening (beta 0 to 0.3)

                                enforce_symmetry=False, verbose=False)
    if obj1 < best_obj:
        best_a = a.copy()
        best_obj = obj1

    remaining = deadline - time.monotonic()
    if remaining < 1:
        return best_a, best_obj

    # Phase 2: Moderate flattening (beta 0.2 to 0.6)
    a, obj2 = spectral_flatten(a, 0.2, 0.6, 3000, min(deadline, time.monotonic() + remaining * 0.4),

    if obj2 < best_obj:
        best_a = a.copy()
        best_obj = obj2

    remaining = deadline - time.monotonic()
    if remaining < 1:
        return best_a, best_obj

    # Phase 3: With symmetry
    a_sym = 0.5 * (best_a + best_a[::-1])
    a_sym = np.maximum(a_sym, 0)
    a_sym /= a_sym.sum()
    a, obj3 = spectral_flatten(a_sym, 0.1, 0.5, 3000, deadline,
                                enforce_symmetry=True, verbose=False)
    if obj3 < best_obj:
        best_a = a.copy()
        best_obj = obj3

    return best_a, best_obj

        # For i != k-i (i.e., k != 2i): both terms present if valid
        # For i == k-i (k=2i): single term a[i]*a[i], derivative = 2*a[i]
        # Simplified: d(conv[k])/da[i] = a[k-i] if 0<=k-i<n, plus a[i] if 0<=k-i<n and k-i!=i
        # = a[k-i] * I(0<=k-i<n) + a[i] * I(0<=k-i<n) * I(k-i != i)
        # Hmm, let me just use: d(conv[k])/da[i] = a[k-i] + a[k-i] (if k != 2i) or 2*a[i] (if k=2i)
        # which is just: 2*a[k-i] (if 0<=k-i<n and k!=2i) or 2*a[i] (if k=2i and 0<=k-i<n=i)
        # = 2*a[k-i] in all valid cases! (Since when k=2i, k-i=i, so a[k-i]=a[i])
        # Wait: when k=2i, the term in the sum is a[i]*a[i], and d/da[i] = 2*a[i] = 2*a[k-i]. Correct!
        # When k!=2i but k-i valid: terms are a[j]*a[k-j] with j=i OR j=k-i. These are different j values.
        # j=i gives a[i]*a[k-i], d/da[i] = a[k-i].
        # j=k-i gives a[k-i]*a[i], d/da[i] = a[k-i].
        # Total: 2*a[k-i]. But wait, these are the same product a[i]*a[k-i], counted once in the sum!
        # conv[k] = sum_j a[j]*a[k-j]. The term j=i is a[i]*a[k-i]. The term j=k-i is a[k-i]*a[i].
        # These are the same term only if i = k-i, i.e., k = 2i.
        # If i != k-i, they are two separate terms in the sum, each contributing a[k-i] to the derivative.
        # So: d(conv[k])/da[i] = a[k-i] + a[k-i] = 2*a[k-i] (when i != k-i and both valid)
        #                            = 2*a[i] (when i = k-i, i.e., the squared term)
        # In both cases: 2*a[k-i] (since when k=2i, k-i=i).
        # BUT: we need k-i to be a valid index [0, n-1] for the term to exist.
        # And also i must be valid [0, n-1] (which it is since we're computing grad[i]).
        # Additionally, when j=k-i, we need 0<=k-i<n for that j to be valid.
        # And when j=i, we need 0<=i<n (always true) and 0<=k-i<n (same condition).
        # So: d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n, else 0.

        # So: grad[i] = 2 * sum_k w[k]*a[k-i] * I(0<=k-i<n)
        # = 2 * (w * a)[i] where (w*a) is the valid cross-correlation

        # Using FFT for efficiency:
        # Let r[i] = sum_k w[k]*a[k-i] for k-i in [0,n-1]
        # This is the "cross-correlation" of w and a, specifically:
        # r[i] = sum_{j=0}^{n-1} w[i+j]*a[j]  (letting j=k-i, so k=i+j)
        #      = sum_{j=0}^{n-1} a[j]*w[i+j]
        # This is the cross-correlation (a cross-corr w) at lag -i, or equivalently
        # the convolution (w_flipped * a) at position i... let me just use FFT directly.
        #
        # r[i] = sum_{j=0}^{min(n-1, 2n-2-i)} w[i+j] * a[j]
        # This is the cross-correlation of sequences w (length 2n-1) and a (length n).
        # In FFT: r = IFFT(FFT(w, N3) * conj(FFT(a, N3))) ... need to be careful about alignment.
        #
        # Actually, the simplest correct approach for moderate n:
        # r[i] = sum_j a[j] * w[i+j] where j in [0,n-1] and i+j in [0, 2n-2]
        # This is: r = correlate(w, a, mode='valid') ... no, it's a bit different.
        #
        # Let me just use the PSD approach:
        # r[i] = sum_j a[j]*w[i+j]
        # If I define w_ext of length N with w_ext[k] = w[k] for k in [0,2n-2], 0 otherwise
        # Then r[i] = (a * w_ext_flipped)[i + n - 1] ... this is getting too convoluted.
        #
        # For n up to ~2000, a direct O(n^2) computation is fine (4M ops per gradient step).
        # Let me use a slightly optimized direct approach:

        # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j] (where i+j < 2n-1)
        # Using the convolution: let c2 = convolve(a, w, mode='same' or 'full')
        # Actually: sum_j a[j]*w[i+j] is the cross-correlation.
        # np.correlate(w, a, mode='full')[i + n - 1] = sum_j w[j+i+n-1-n+1]*a[j] = sum_j w[j+i]*a[j]
        # Hmm, let me just use FFT:
        # sum_j a[j]*w[i+j] = IFFT(FFT(a, Sz) * conj(FFT(w, Sz)))[i + offset]

        Sz = next_pow2(n + 2 * n - 1)
        A_s = np.fft.rfft(a, Sz)
        W_s = np.fft.rfft(w, Sz)
        # correlate(w, a)[i] = sum_j w[j] * conj(a[j-i]) ... for real: sum_j w[j]*a[j-i]
        # We want sum_j a[j]*w[i+j] = sum_j a[j]*w[j+i]
        # = (a * w_rev)[i + 2n - 2 - ?] ...
        # Let me define: b[k] = w[2n-2-k] (reversed w)
        # Then (a * b)[i + 2n - 2] = sum_j a[j]*b[i+2n-2-j] = sum_j a[j]*w[2n-2-i-2n+2+j] = sum_j a[j]*w[j-i]
        # That gives w[j-i], not w[j+i].
        #
        # Alternative: (a * w)[i] = sum_j a[j]*w[i-j] (standard convolution)
        # We want sum_j a[j]*w[j+i] = sum_j a[j]*w[i+j]
        # = (a_rev * w)[i] where a_rev[j] = a[n-1-j]
        # (a_rev * w)[i] = sum_j a_rev[j]*w[i-j] = sum_j a[n-1-j]*w[i-j]
        # Let k = n-1-j: = sum_k a[k]*w[i-n+1+k]
        # = sum_k a[k]*w[k+(i-n+1)]
        # This gives w[k+(i-n+1)], so we need i-n+1 = i' for the shift we want.
        # We want w[k+i], so i-n+1 = i, meaning i = -1... that doesn't work.
        #
        # OK let me just do it the simple way. For n <= 2000, O(n^2) gradient is fine.
        # grad[i] = 2 * sum_{j=max(0,-i)}^{min(n-1, 2n-2-i)} a[j] * w[i+j]

        # Vectorized: for each i, grad[i] = 2 * dot(a[j], w[i+j] for valid j)
        # This is a "diagonal" computation. We can use:
        # Create a Toeplitz-like matrix and multiply... or just use a loop for clarity.

        # Efficient: use the fact that this is a convolution
        # grad[i]/2 = sum_j a[j]*w[i+j]
        # Let's define: for fixed i, the valid j range is [0, n-1] intersected with [0, 2n-2-i]
        # So j in [0, min(n-1, 2n-2-i)]
        # And we compute dot(a[0:min(n, 2n-1-i)], w[i:i+min(n, 2n-1-i)])

        # This is the "valid" cross-correlation of a and w at various lags.
        # Specifically: (a * w_flipped) gives us what we need at the right indices.
        # Let w_f = w[::-1] (length 2n-1)
        # full_conv = convolve(a, w_f, mode='full') (length n + 2n-2 - 1 = 3n-3)
        # full_conv[k] = sum_j a[j]*w_f[k-j] = sum_j a[j]*w[2n-2-k+j]
        # We want sum_j a[j]*w[i+j] for each i.
        # Set 2n-2-k+j = i+j => 2n-2-k = i => k = 2n-2-i
        # So: grad[i]/2 = full_conv[2n-2-i] = full_conv[2n-2-i]

        w_f = w[::-1]
        Nc = next_pow2(n + 2 * n - 1)
        A_c = np.fft.rfft(a, Nc)
        Wf_c = np.fft.rfft(w_f, Nc)
        full_conv = np.fft.irfft(A_c * Wf_c, Nc)

        # full_conv has length Nc, but the meaningful part is [0, n+2n-3]
        # grad[i]/2 = full_conv[2n-2-i] for i in [0, n-1]
        indices = 2 * n - 2 - np.arange(n)
        grad = 2.0 * full_conv[indices]

        # Project gradient to simplex tangent space
        grad = grad - np.mean(grad)

        # Adam update
        t_count += 1
        lr_t = lr / (1.0 + 0.05 * t_count ** 0.4)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-13:
            best_a = a_new.copy()
            best_obj = obj_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                a = best_a.copy() + rng.normal(0, 0.02, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

        if verbose and it % 200 == 0:
            print(f"    GradDesc it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def coord_descent(a, max_sweeps, deadline, random_order=False, seed=0):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 500

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            if old < 1e-12:
                candidates = [0.0, 1e-4, 1e-3, 1e-2, 5e-3]
            else:
                candidates = np.array([0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7]) * old

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    a = np.maximum(best_a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent: solve LP for best response direction at each step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [0.5, 0.3, 0.2, 0.15, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))
        # Symmetrized parent
        p_sym = 0.5 * (p + p[::-1])
        inits.append(np.maximum(p_sym, 0))
        inits[-1] = inits[-1] / inits[-1].sum()

    inits.append(np.ones(n) / n)

    # Cosine patterns
    for freq in [1, 2, 3]:
        inits.append(project_simplex(1.0 + 0.5 * np.cos(2 * np.pi * freq * t)))

    # Gaussian
    inits.append(project_simplex(np.exp(-((t - 0.5) / 0.3) ** 2) + 0.05))

    # Smooth random
    ksize = max(5, n // 10)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase spectral optimization for minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses: spectral flattening (Gerchberg-Saxton), smoothed gradient descent,
    coordinate descent, and LP mirror descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Spectral flattening at parent length (500s) ===
    phase1_deadline = t_start + 500.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Spectral flattening at n={n_parent} (budget {phase1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    n_init = min(len(inits), 6)
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, n_init) - 5

    for idx in range(n_init):
        remaining = phase1_deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_init_time, remaining - 5)
        if alloc < 15:
            break

        a0 = inits[idx].copy()
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        # Spectral flattening
        a_opt, obj_opt = spectral_flatten_hybrid(a0, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  SpecFlatten init {idx}: J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Smoothed gradient descent on best (300s) ===
    phase2_deadline = t_start + 800.0
    remaining = phase2_deadline - time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Gradient descent (budget {remaining:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None and remaining > 30:
        a_curr = best_seq.copy()
        t_sub1 = time.monotonic() + remaining * 0.5
        a_grad, obj_grad = smoothed_gradient_descent(a_curr, max_iters=20000, deadline=t_sub1,
                                                      tau_init=5.0, tau_end=300.0, lr=0.015)
        update_best(a_grad)

        # Second pass with different params
        remaining = phase2_deadline - time.monotonic()
        if remaining > 20:
            a_grad2, obj_grad2 = smoothed_gradient_descent(best_seq, max_iters=20000, deadline=phase2_deadline - 2,
                                                            tau_init=20.0, tau_end=1000.0, lr=0.01)
            update_best(a_grad2)

    # === PHASE 3: Multi-length exploration (250s) ===
    phase3_deadline = t_start + 1050.0
    remaining = phase3_deadline - time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length exploration (budget {remaining:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    candidate_ns = [384, 512, 640, 768, 896, 1024, 1280]
    candidate_ns = [n for n in candidate_ns if n != n_parent]
    candidate_ns.sort(key=lambda n: abs(n - n_parent))

    for new_n in candidate_ns[:4]:
        remaining = phase3_deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(80.0, remaining - 10)
        if alloc < 25:
            break

        print(f"  Trying n={new_n}...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        # Spectral flatten at new length
        a_opt, obj_opt = spectral_flatten_hybrid(a_ext, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 4: Intensive polish (remaining time) ===
    phase4_deadline = deadline
    remaining = phase4_deadline - time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Intensive polish (budget {remaining:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None and remaining > 20:
        # Coordinate descent
        t_cd = time.monotonic() + remaining * 0.30
        print(f"  CD pass...", flush=True)
        a_cd = coord_descent(best_seq, max_sweeps=50000, deadline=t_cd, random_order=True, seed=111)
        update_best(a_cd)

        remaining = phase4_deadline - time.monotonic()
        if remaining > 20:
            # LP
            t_lp = time.monotonic() + remaining * 0.40
            print(f"  LP pass...", flush=True)
            a_lp, obj_lp = lp_mirror_descent(best_seq, max_iters=200, deadline=t_lp)
            update_best(a_lp)

        remaining = phase4_deadline - time.monotonic()
        if remaining > 20:
            # Another spectral pass on best
            t_spec = phase4_deadline - 5
            print(f"  Final spectral pass...", flush=True)
            a_final_spec, obj_final = spectral_flatten_hybrid(best_seq, t_spec)
            update_best(a_final_spec)

        # Final CD
        if time.monotonic() < phase4_deadline - 15:
            a_final = coord_descent(best_seq, max_sweeps=30000,
                                    deadline=phase4_deadline - 5, random_order=True, seed=444)
            update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
