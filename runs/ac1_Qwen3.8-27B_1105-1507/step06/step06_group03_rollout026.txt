import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        # For T2: T2[i] = sum_j a[j]*w[i-j] = (a * w_rev)[n-1+i]
        # where w_rev = w reversed (padded to same length as needed)
        w_rev2 = np.zeros(2 * n - 1)
        w_rev2[:] = w[::-1]
        w_rev2_fft = np.fft.rfft(w_rev2, size)
        conv2 = np.fft.irfft(a_fft * w_rev2_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i  # for T1
            idx2 = n - 1 + i  # for T2
            if idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if idx2 < len(conv2):
                grad[i] += conv2[idx2]

        # Project gradient to simplex tangent
        grad = grad - np.mean(grad)

        # Adam update
        t_count += 1
        lr_t = lr / (1.0 + 0.05 * t_count ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                # Perturb and restart
                rng = np.random.default_rng(it * 31 + 7)
                a = project_simplex(best_a + rng.normal(0, 0.03, n))
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

        if verbose and it % 200 == 0:
            print(f"    Adam it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def sparse_lp_best_response(a, deadline, n_constraints_frac=0.3, verbose=False):
    """
    LP best-response with sparse constraints (only top fraction of conv positions).
    Solve: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0, sum(g)=1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    s = a.sum()
    a_norm = a / s

    c = fast_conv(a_norm)
    M = c.max()

    # Select top constraints (positions with high conv values)
    n_total = 2 * n - 1
    n_active = max(n, int(n_constraints_frac * n_total))
    top_indices = np.argsort(c)[::-1][:n_active]

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    b_ub = []

    for k in top_indices:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(len(b_ub))
            cols.append(j)
            vals.append(a_norm[k - j])
        b_ub.append(M)

    if len(rows) == 0:
        return a, compute_J(a)

    A_ub = coo_matrix((vals, (rows, cols)), shape=(len(b_ub), n)).tocsr()
    c_obj = -np.ones(n)

    bounds = [(0, None)] * n
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])

    try:
        remaining = deadline - time.monotonic()
        t_limit = max(5, min(30, remaining * 0.3))
        r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': t_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                g /= gs
                return g, compute_J(g)
    except Exception:
        pass

    return a, compute_J(a)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(('parent', p.copy()))
        inits.append(('parent_noise1', project_simplex(p + rng.normal(0, 0.02, n))))
        inits.append(('parent_noise2', project_simplex(p + rng.normal(0, 0.05, n))))
        inits.append(('parent_noise3', project_simplex(p + rng.normal(0, 0.1, n))))
        # Peak-targeting perturbation
        conv_p = fast_conv(p)
        peak_pos = conv_p.argmax()
        pert = p.copy()
        for i in range(n):
            if 0 <= peak_pos - i < n or 0 <= i - peak_pos < n:
                pert[i] *= 0.7
        inits.append(('peak_shave', project_simplex(pert)))

    # Structured inits
    inits.append(('uniform', np.ones(n) / n))
    for freq in [1, 2, 3]:
        inits.append((f'cos_{freq}', project_simplex(1.0 + 0.5 * np.cos(2 * np.pi * freq * t))))
    for center in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.25]:
            inits.append((f'gauss_{center}_{width}',
                          project_simplex(np.exp(-((t - center) / width) ** 2) + 0.1)))
    # Parabolic
    inits.append(('para', project_simplex(4 * t * (1 - t) + 0.1)))
    # Smooth random
    ksize = max(3, n // 20)
    for _ in range(3):
        kernel = np.ones(ksize) / ksize
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(('rand_smooth', project_simplex(smooth + 0.05)))

    return inits


def solve_single(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return best_obj

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 1: Coordinate descent (35%)
    t1 = time.monotonic() + 0.35 * remaining
    a = coord_descent_grad(a, max_sweeps=20000, deadline=t1, seed=42, tau=40.0)
    try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 2: Smooth Adam (30%)
    t2 = time.monotonic() + 0.30 * remaining
    a_adam, obj_adam = smooth_adam(best_a, max_iters=5000, deadline=t2,
                                   tau_init=15.0, tau_end=120.0, lr=0.03)
    try_update(a_adam)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 3: Second CD pass (15%)
    t3 = time.monotonic() + 0.15 * remaining
    a2 = coord_descent_grad(best_a, max_sweeps=10000, deadline=t3, seed=123, tau=60.0)
    try_update(a2)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 4: Sparse LP (15%)
    t4 = time.monotonic() + 0.15 * remaining
    if n <= 1200:
        a_lp, obj_lp = sparse_lp_best_response(best_a, deadline=t4, n_constraints_frac=0.25)
        if obj_lp < best_obj:
            # Line search
            for frac in [0.5, 0.3, 0.7, 1.0]:
                a_try = (1 - frac) * best_a + frac * a_lp
                a_try = np.maximum(a_try, 0)
                if a_try.sum() > 1e-10:
                    a_try /= a_try.sum()
                    obj_try = compute_J(a_try)
                    if obj_try < best_obj:
                        best_a = a_try
                        best_obj = obj_try
                        break

    remaining = deadline - time.monotonic()
    if remaining < 3:
        return best_a, best_obj

    # Stage 5: Final CD polish (remaining)
    a_final = coord_descent_grad(best_a, max_sweeps=10000, deadline=deadline - 1,
                                 seed=999, tau=80.0)
    try_update(a_final)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Phase 1: Multi-init optimization at parent length.
    Phase 2: Multi-length exploration.
    Phase 3: Intensive polish on best found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Multi-init at parent length (40% budget) ===
    phase1_deadline = t_start + 0.40 * (budget_s - 15.0)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-init at n={n_parent} (deadline in {phase1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    # Limit to fit in budget
    n_inits_to_try = min(len(inits), 6)
    per_init_budget = (phase1_deadline - time.monotonic()) / max(1, n_inits_to_try) - 3

    # Sort inits by priority (parent-based first)
    inits_sorted = sorted(inits, key=lambda x: 0 if x[0].startswith('parent') or x[0] == 'peak_shave' else 1)

    for idx in range(n_inits_to_try):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 20:
            break
        alloc = min(per_init_budget, remaining_check - 3)
        if alloc < 10:
            break

        name, a0 = inits_sorted[idx]
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        t_alloc_end = time.monotonic() + alloc
        a_opt, obj_opt = solve_single(a0, t_alloc_end, verbose=False)
        update_best(a_opt)
        print(f"  [{name}] J={obj_opt:.8f} n={n_parent}", flush=True)

    # === PHASE 2: Multi-length exploration (25% budget) ===
    phase2_deadline = t_start + 0.65 * (budget_s - 15.0)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration (deadline in {phase2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidate_ns = sorted(set([256, 384, 512, 640, 768, 896, 1024, 1152, 1280, 1536]),
                          key=lambda x: abs(x - n_parent))
    candidate_ns = [n for n in candidate_ns if n != n_parent and 128 <= n <= 2048]

    for new_n in candidate_ns[:5]:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 50:
            break
        alloc = min(80.0, remaining_check - 10)
        if alloc < 30:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        t_alloc_end = time.monotonic() + alloc
        a_opt, obj_opt = solve_single(a_ext, t_alloc_end, verbose=False)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive polish on best (35% budget) ===
    phase3_deadline = deadline
    remaining = phase3_deadline - time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive polish (budget {remaining:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and remaining > 30:
        n_best = len(best_seq)

        # Pass 1: Long coordinate descent
        t_p1 = time.monotonic() + 0.25 * remaining
        print(f"  CD pass 1 (budget {t_p1 - time.monotonic():.0f}s)...", flush=True)
        a_curr = coord_descent_grad(best_seq, max_sweeps=30000, deadline=t_p1,
                                    seed=555, tau=50.0)
        update_best(a_curr)

        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            # Pass 2: Adam with different params
            t_p2 = time.monotonic() + 0.20 * remaining
            print(f"  Adam pass (budget {t_p2 - time.monotonic():.0f}s)...", flush=True)
            a_ad, obj_ad = smooth_adam(best_seq, max_iters=10000, deadline=t_p2,
                                       tau_init=10.0, tau_end=100.0, lr=0.02)
            update_best(a_ad)

        remaining = phase3_deadline - time.monotonic()
        if remaining > 20:
            # Pass 3: CD with different seed/tau
            t_p3 = time.monotonic() + 0.20 * remaining
            print(f"  CD pass 2 (budget {t_p3 - time.monotonic():.0f}s)...", flush=True)
            a_c2 = coord_descent_grad(best_seq, max_sweeps=20000, deadline=t_p3,
                                      seed=777, tau=100.0)
            update_best(a_c2)

        remaining = phase3_deadline - time.monotonic()
        if remaining > 15 and n_best <= 1200:
            # Pass 4: Sparse LP
            t_p4 = time.monotonic() + 0.15 * remaining
            print(f"  LP pass (budget {t_p4 - time.monotonic():.0f}s)...", flush=True)
            a_lp, obj_lp = sparse_lp_best_response(best_seq, deadline=t_p4, n_constraints_frac=0.2)
            if obj_lp < best_score:
                for frac in [1.0, 0.5, 0.3]:
                    a_try = (1 - frac) * best_seq + frac * a_lp
                    a_try = np.maximum(a_try, 0)
                    if a_try.sum() > 1e-10:
                        a_try /= a_try.sum()
                        update_best(a_try)
                        break

        remaining = phase3_deadline - time.monotonic()
        if remaining > 10:
            # Pass 5: Final CD
            print(f"  Final CD (budget {remaining - 2:.0f}s)...", flush=True)
            a_fin = coord_descent_grad(best_seq, max_sweeps=20000,
                                       deadline=phase3_deadline - 2, seed=999, tau=80.0)
            update_best(a_fin)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
