import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    if len(conv) == 0:
        return np.inf
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def topk_smoothed_conv(a, k=5, beta=50.0):
    """
    Compute a smoothed version of max(conv) using top-k averaging.
    Returns (smoothed_max, weights_on_conv_positions).
    """
    conv = compute_conv(a)
    # Use logsumexp for smooth max
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    return ls, w


def smoothed_obj_and_grad(a, beta=50.0):
    """
    Compute smoothed objective and gradient using logsumexp smoothing.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    # Gradient: d(ls)/d(a[i]) = 2 * sum_j w[j] * a[j-i] (cross-correlation)
    size = 1

    W = rfft(w, size)
    A = rfft(a, size)

    grad_ls = 2.0 * cc[:n]

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_find_direction(a, M, time_limit=10):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

            break

    return best_a


def fw_update(a, k_peaks=3, time_budget=20.0):
    """
    Frank-Wolfe inspired update: identify peak-contributing elements
    and redistribute mass to reduce peaks.
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        # Find top peaks
        peak_indices = np.argsort(conv)[::-1][:k_peaks]
        
        # Compute subgradient at the peaks
        # d(max(conv))/d(a[i]) = 2 * sum_{p in peaks} a[p-i] (for valid p-i)
        subgrad = np.zeros(n)
        for p in peak_indices:
            for i in range(n):
                j = p - i
                if 0 <= j < n:
                    subgrad[i] += a[j]
        subgrad *= 2.0

        # FW step: move mass to the element with minimum subgradient
        # (this reduces the peak the most)
        min_idx = np.argmin(subgrad)
        
        # Move a small amount of mass
        lam = 0.1  # step size
        a_new = (1 - lam) * best_a
        a_new[min_idx] += lam / n  # spread uniformly? No, concentrate on min
        # Actually, FW says: s = e_{min_idx} (the vertex of the simplex)
        # But that's too aggressive. Let's use a soft version.
        
        # Better: shift mass from peak-contributing elements to low-subgradient elements
        # Find elements with highest subgradient (they contribute most to peaks)
        top_contrib = np.argsort(subgrad)[::-1][:n // 10]
        
        a_try = best_a.copy()
        total_shift = 0.0
        for i in top_contrib:
            shift = 0.01 * a_try[i]
            a_try[i] -= shift
            total_shift += shift
        
        # Distribute to low-contributing elements
        low_contrib = np.argsort(subgrad)[:n // 10]
        for i in low_contrib:
            a_try[i] += total_shift / len(low_contrib)
        
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def adam_momentum_opt(a, max_iters=10000, time_budget=60.0, beta_start=20.0,
                      beta_end=500.0, lr_start=0.03, momentum=0.9):
    """
    Momentum-based gradient descent in log-space with adaptive beta.
    Different from Adam: uses heavy ball momentum, not Adam's adaptive LR.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_true_obj = compute_obj(a)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        x_centered = x - np.max(x)
        a_curr = np.exp(x_centered)
        a_curr = a_curr / np.sum(a_curr)

        obj_sm, grad_a = smoothed_obj_and_grad(a_curr, beta)
        grad_x = grad_a * a_curr

        # Heavy ball momentum
        v = momentum * v - lr_start * grad_x
        x = x + v

        # Centering
        if it % 100 == 0:
            x -= np.mean(x)

        # Check true objective
        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

    return best_a

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses heavy-ball momentum in log-space, aggressive LP refinement with line search,
Frank-Wolfe inspired peak redistribution, coordinate descent on active peaks,
and multi-n exploration with structured random restarts.
Differs from previous approach by using momentum (not Adam), top-k peak targeting,
aggressive LP line search, and FW-style mass redistribution.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    if len(conv) == 0:
        return np.inf
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smoothed_obj_and_grad(a, beta=50.0):
    """
    Compute smoothed objective (logsumexp of beta*conv) and its gradient.
    Returns (objective_value, gradient_array).
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    # Gradient via cross-correlation
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def momentum_log_opt(a, max_iters=10000, time_budget=60.0,
                     beta_start=20.0, beta_end=500.0,
                     lr_start=0.03, lr_end=0.005,
                     momentum_start=0.85, momentum_end=0.95):
    """
    Heavy-ball momentum gradient descent in log-space with annealed beta and LR.
    Different from Adam: uses simple momentum, not adaptive per-coordinate LR.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    v = np.zeros(n)
    best_a = a.copy()
    best_true_obj = compute_obj(a)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start + (lr_end - lr_start) * progress
        mom = momentum_start + (momentum_end - momentum_start) * progress

        x_centered = x - np.max(x)
        a_curr = np.exp(x_centered)
        a_curr = a_curr / np.sum(a_curr)

        obj_sm, grad_a = smoothed_obj_and_grad(a_curr, beta)
        grad_x = grad_a * a_curr

        # Heavy ball momentum update
        v = mom * v - lr * grad_x
        x = x + v

        if it % 100 == 0:
            x -= np.mean(x)

        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_find_direction(a, M, time_limit=10):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine_aggressive(a, time_budget=35.0):
    """
    Aggressive LP refinement: iteratively reduce peak target,
    with detailed line search over blending parameter.
    """
    n = len(a)
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)

    for round in range(12):
        if time.monotonic() - start > time_budget:
            break

        best_norm = best_a / np.sum(best_a)
        conv = compute_conv(best_norm)
        M = float(np.max(conv))

        # Try progressively tighter constraints
        found_direction = False
        for frac in [0.99, 0.97, 0.95, 0.90, 0.85]:
            if time.monotonic() - start > time_budget:
                break
            M_try = M * frac
            g = lp_find_direction(best_norm, M_try,
                                  time_limit=min(5, time_budget * 0.08))
            if g is not None:
                found_direction = True
                break

        if not found_direction:
            break

        # Detailed line search
        best_t = 0.0
        best_obj_t = best_obj
        best_a_t = best_a.copy()

        for t in np.linspace(0.02, 1.0, 20):
            if time.monotonic() - start > time_budget:
                break
            a_try = (1 - t) * best_norm + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t
                best_a_t = a_try.copy()

        if best_t > 0 and best_obj_t < best_obj:
            best_obj = best_obj_t
            best_a = best_a_t
        else:
            break

    return best_a


def fw_peak_redistribution(a, k_peaks=3, time_budget=25.0):
    """
    Frank-Wolfe inspired: identify elements contributing to top peaks,
    shift their mass to low-contributing elements.
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        peak_indices = np.argsort(conv)[::-1][:k_peaks]

        # Subgradient: contribution of each a[i] to the peaks
        subgrad = np.zeros(n)
        for p in peak_indices:
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            for i in range(i_min, i_max + 1):
                j = p - i
                if 0 <= j < n:
                    subgrad[i] += best_a[j]
        subgrad *= 2.0

        # Identify top contributors (highest subgradient) and bottom (lowest)
        n_shift = max(3, n // 15)
        top_idx = np.argsort(subgrad)[::-1][:n_shift]
        bot_idx = np.argsort(subgrad)[:n_shift]

        # Shift mass from top to bottom
        a_try = best_a.copy()
        total_shift = 0.0
        for i in top_idx:
            shift = 0.02 * a_try[i]
            a_try[i] -= shift
            total_shift += shift

        for i in bot_idx:
            a_try[i] += total_shift / n_shift

        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def coordinate_peak_descent(a, time_budget=25.0, seed=0):
    """
    Coordinate descent focused on elements contributing to top convolution peaks.
    For each peak, try reducing the contributing elements and compensating elsewhere.
    """
    n = len(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(best_a)
    top_peaks = np.argsort(conv)[::-1][:5]

    stuck_count = 0

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        if stuck_count > 200:
            break

        improved = False

        for p in top_peaks:
            if time.monotonic() - start > time_budget:
                break

            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)

            # Try modifying pairs that contribute to this peak
            candidates = list(range(i_min, i_max + 1))
            if len(candidates) > 10:
                candidates = list(rng.choice(candidates, 10, replace=False))

            for i in candidates:
                if time.monotonic() - start > time_budget:
                    break
                j = p - i
                if j < 0 or j >= n:
                    continue

                # Try reducing a[i] and adding to a random other position
                for delta_frac in [0.01, 0.02, 0.05, -0.01]:
                    delta = delta_frac * best_a[i]
                    if delta >= best_a[i] * 0.5:
                        continue

                    # Choose target: prefer low values
                    target = rng.integers(0, n)
                    if target == i or target == j:
                        target = (target + 1) % n

                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[target] += delta

                    if a_try[i] < 0:
                        continue

                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try

                    # Quick check: does this reduce the peak at p?
                    conv_try = compute_conv(a_try)
                    if conv_try[p] < conv[p] - 1e-10:
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            conv = conv_try
                            top_peaks = np.argsort(conv)[::-1][:5]
                            improved = True
                            stuck_count = 0
                            break

        if not improved:
            stuck_count += 1

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Loaded parent: n={parent.size}, obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf

            method = round % 4
            alloc = min(remaining * 0.18, 30.0)

            else:
                no_improve += 1
                print(f"  R{round} (m{method}): {obj:.8f}")

            # If stuck, perturb
            if no_improve >= 3:
                scale = 0.02 * (1 + no_improve * 0.3)
                a = a * np.exp(rng.normal(0, scale, len(a)))
                a = np.maximum(a, 1e-10)

    print(f"{'='*60}")

                block_size = max(1, n_best // n_blocks)
                for b in range(n_blocks):

                    e_idx = min(s_idx + block_size, n_best)

               _struct_val = restart % 3

                elif _struct_val == 1:
                    structured = np.exp(-((x - n_best / 2) / (n_best / 3)) ** 2)
                else:
                    structured = 1.0 + 0.3 * np.sin(4 * np.pi * x / max(n_best - 1, 1))
                structured = np.maximum(structured, 0.01)
                structured = structured / np.sum(structured)
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * structured
            else:
                # Sparse multiplicative perturbation
                a_pert = best_seq.copy()
                num_mod = max(10, n_best // 15)
                indices = rng.choice(n_best, num_mod, replace=False)
                factors = rng.uniform(0.3, 3.0, num_mod)
                a_pert[indices] *= factors

            a_pert = np.maximum(a_pert, 1e-10)
            s_p = np.sum(a_pert)
            if s_p < 1e-10:
                continue
            a_pert = a_pert / s_p

            alloc = min(remaining * 0.15, 25.0)
            a_opt = momentum_log_opt(a_pert, max_iters=8000, time_budget=alloc,
                                      beta_start=50.0, beta_end=800.0,
                                      lr_start=0.025, lr_end=0.008,
                                      momentum_start=0.88, momentum_end=0.95)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart} (s{strat}): {best_score:.8f} ***")
            else:
                print(f"  Restart {restart} (s{strat}): {obj:.8f}")

    # === Final polish ===
    if best_seq is not None and time.monotonic() < deadline - 10:

        if obj_final < best_score:

            best_seq = a_final.copy()
            print(f"  Final momentum: {best_score:.8f} ***")

        # Quick LP if applicable
        if len(best_seq) <= 800 and time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic()
            a_lp = lp_refine_aggressive(best_seq, time_budget=min(remaining * 0.6, 15.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.8f} ***")

    # Validation and return
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
