import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def build_toeplitz(a):
    """Build Toeplitz matrix for conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_dual(a, M):
    """
    Solve the dual LP: min M * sum(y) s.t. A^T y >= 1, y >= 0
    where A is the Toeplitz matrix for conv(a, .)
    Returns y (the dual variables)
    """
    n = len(a)
    A = build_toeplitz(a)
    # Dual: min M * sum(y) s.t. A^T y >= 1, y >= 0
    # Rewrite as: min M * sum(y) s.t. -A^T y <= -1, y >= 0
    c = M * np.ones(2 * n - 1)
    A_ub = -A.T  # shape (n, 2n-1)
    b_ub = -np.ones(n)
    bounds = [(0.0, None)] * (2 * n - 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.status == 0:
            y = np.maximum(res.x, 0)
            s = np.sum(y)
            if s > 1e-10:
                return y / s
    except Exception:
        pass
    return None

def multiplicative_descent(a, max_iters=200, time_budget=30.0, alpha_init=0.5):
    """
    Multiplicative (mirror) descent using LP dual weights.
    a_new[i] = a[i] / c[i]^alpha where c[i] = weighted gradient at i
    """
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = alpha_init
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_b = float(np.max(conv))
        
        # Get dual weights (subgradient)
        y = solve_lp_dual(a, max_b)
        if y is None:
            # Fallback: softmax weights
            tau = 0.05 * max_b
            shifted = (conv - max_b) / tau
            weights = np.exp(np.clip(shifted, -100, 100))
            w_sum = np.sum(weights)
            if w_sum < 1e-300:
                y = np.zeros(len(conv))
                y[np.argmax(conv)] = 1.0
            else:
                y = weights / w_sum
        
        # Compute cost vector: c[i] = sum_k y[k] * 2*a[k-i] (for valid k-i)
        c = np.zeros(n)
        for k in range(len(y)):
            if y[k] < 1e-12:
                continue
            # c[i] += y[k] * 2 * a[k-i] for 0 <= k-i < n
            i_min = max(0, k - n + 1)
            i_max = min(n, k + 1)
            for i in range(i_min, i_max):
                j = k - i
                if 0 <= j < n:
                    c[i] += y[k] * 2.0 * a[j]
        
        c = np.maximum(c, 1e-10)
        
        # Multiplicative update
        a_new = a / (c ** alpha)
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            alpha = max(0.01, alpha * 0.95)  # decay step size
        else:
            alpha = max(0.01, alpha * 0.7)  # reduce step size more aggressively
            # Try a smaller step
            a_try = a / (c ** (alpha * 0.5))
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
    
    return best_a, best_obj
