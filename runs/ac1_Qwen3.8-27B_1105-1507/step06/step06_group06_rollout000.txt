def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

    sum_a = float(np.sum(a))

            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def lp_alternate_extended(a, max_iters=80, time_budget=60.0, verbose=False):
    """
    Extended LP alternating with adaptive line search and dual-based step sizing.
    Alternates between max-sum LP and min-peak LP for richer search directions.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Primary: max-sum LP
        g = solve_lp_max_sum(a, M, time_limit=min(10, max(3, time_budget / (max_iters - it))))
        
        # Secondary: min-peak LP for larger n
        g2 = None
        if n <= 1000 and it % 3 == 1:
            g2, M2 = solve_lp_minpeak(a, time_limit=min(10, max(3, time_budget / (max_iters - it))))

        if g is None and g2 is None:
            break

        # Build candidate directions
        directions = []
        if g is not None:
            directions.append(g)
        if g2 is not None:
            directions.append(g2)

        best_step_obj = best_obj
        best_step_a = best_a.copy()

        for d in directions:
            # Adaptive line search
            candidate_ts = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
            for t in candidate_ts:
                a_try = (1.0 - t) * a + t * d
                a_try = project_nonneg_sum1(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_step_obj:
                    best_step_obj = obj_try
                    best_step_a = a_try.copy()

        if best_step_obj < best_obj - 1e-13:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = best_step_a
            stall = 0
            if verbose and it % 10 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 15:
                break

    return best_a, best_obj


def spectral_flatten(a, max_iters=200, time_budget=30.0):
    """
    Frequency-domain spectral flattening.
    Iteratively reduces the dynamic range of the power spectrum while
    maintaining non-negativity in time domain.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    alpha = 0.6
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2

        # Normalize and compress
        p_max = np.max(power)
        if p_max < 1e-30:
            break
        p_norm = power / p_max
        p_norm = np.maximum(p_norm, 1e-12)

        # Apply compression: reduce peaks in power spectrum
        p_comp = p_norm ** (1.0 / (1.0 + alpha))
        # This maps [0,1] to [0,1] with compression toward 1 (flattening)

        # Reconstruct with modified magnitude
        phase = np.angle(fa)
        new_mag = np.sqrt(p_comp) * np.sqrt(p_max)
        fa_new = new_mag * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Reduce compression strength
            alpha *= 0.95

    return best_a, best_obj


def gibbs_coordinate_descent(a, max_iters=500, time_budget=30.0, step_scale=0.1):
    """
    Gibbs-style coordinate descent: iteratively adjust each coordinate
    to minimize the peak of the convolution.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for i in range(n):
            # Compute current peak
            conv = compute_conv(a)
            M = float(np.max(conv))

            # Try modifying a[i]
            # The contribution of a[i] to conv[k] is a[i]*a[k-i] + a[i]*a[i-(k-n+1)]...
            # Simplified: try a[i] * (1+delta) and a[i] * (1-delta)
            orig = a[i]
            if orig < 1e-12:
                a[i] = step_scale / n
            else:
                for factor in [0.7, 0.85, 1.0, 1.15, 1.3]:
                    a[i] = orig * factor
                    a_proj = project_nonneg_sum1(a)
                    obj = compute_obj(a_proj)
                    if obj < M and obj < best_obj:
                        a = a_proj.copy()
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                else:
                    a[i] = orig

        if not improved and it > 50:
            break

    return best_a, best_obj


def chase_peak(a, n_steps=80, step_size=0.05):
    """Chase-the-peak: iteratively reduce the maximum convolution peak via subgradient."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for step in range(n_steps):
        conv = compute_conv(a)
        # Find all peaks above 95% of max
        M = float(np.max(conv))
        peak_mask = conv >= 0.95 * M
        peak_indices = np.where(peak_mask)[0]

        # Compute weighted gradient
        weights = conv[peak_mask] - M + 0.01 * M  # positive weights for peaks
        weights = np.maximum(weights, 0)
        if np.sum(weights) < 1e-15:
            # Single peak
            k_star = int(np.argmax(conv))
            grad = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] += a[j]
                j2 = i - (k_star - n + 1)
                if 0 <= j2 < n:
                    grad[i] += a[j2]
        else:
            weights = weights / weights.sum()
            grad = np.zeros(n)
            for k_idx in peak_indices:
                w = weights[list(peak_indices).index(k_idx)]
                if w < 1e-15:
                    continue
                i_start = max(0, k_idx - n + 1)
                i_end = min(n, k_idx + 1)
                indices = np.arange(i_start, i_end)
                a_indices = k_idx - indices
                grad[indices] += w * a[a_indices]

        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        # Adaptive step
        ss = step_size * (best_obj - 1.0) / max(grad_norm, 0.01)
        ss = np.clip(ss, 0.005, 0.2)

        a_try = a - ss * grad_centered
        a_try = project_nonneg_sum1(a_try)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()

    return best_a, best_obj


def gradient_flow_nesterov(a, max_iters=500, time_budget=40.0, lr_init=0.05, momentum=0.9):
    """
    Gradient flow with Nesterov momentum on the smoothed max objective.
    Uses softmax weighting over convolution values for a smooth approximation.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        tau = 0.05 * cmax + 0.001

        # Softmax weights (smooth max approximation)
        shifted = np.clip((conv - cmax) / tau, -100.0, 100.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient via efficient method
        # grad[i] = sum_k weights[k] * (d conv[k]/d a[i])
        # d conv[k]/d a[i] = a[k-i] + a[i] (if both indices valid)
        # Actually: conv[k] = sum_j a[j]*a[k-j], so d/d a[i] = a[k-i] (if 0<=k-i<n)
        #   + a[i] (if k-i = i, i.e., k=2i, and 0<=i<n) -- wait no
        # d/d a[i] of sum_j a[j]*a[k-j]:
        #   = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
        #   = a[k-i] (if 0<=k-i<n) + a[i] (if k-i >= 0 and k-i < n, i.e., same condition)
        # Wait, the second term: delta(k-j, i) means j = k-i, so we get a[k-i] when j=k-i is valid
        # So d conv[k]/d a[i] = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n?
        # No wait: first term is a[k-i] (from j=i), second term is a[j] where j=k-i, so a[k-i]
        # So total is 2*a[k-i] if 0 <= k-i < n.
        # Hmm but that would mean d/d a[0] of conv[0] = 2*a[0], which means d(a[0]^2)/d a[0] = 2*a[0]. Correct!
        # And d/d a[0] of conv[1] = 2*a[1] (since k=1, i=0, k-i=1). conv[1] = a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1]. d/d a[0] = 2*a[1]. Correct!

        grad = np.zeros(n)
        for k_idx in range(len(weights)):
            if weights[k_idx] < 1e-10:
                continue
            i_start = max(0, k_idx - n + 1)
            i_end = min(n, k_idx + 1)
            indices = np.arange(i_start, i_end)
            a_indices = k_idx - indices
            grad[indices] += 2.0 * weights[k_idx] * a[a_indices]

        # Center gradient (for sum=1 constraint)
        grad_centered = grad - np.mean(grad)

        # Nesterov step
        v = momentum * v - lr * grad_centered
        a_new = a - v
        a_new = project_nonneg_sum1(a_new)

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Decay learning rate
        lr *= 0.997

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def generate_spectral_init(n, rng):
    """Generate a sequence with nearly flat power spectrum (good starting point)."""
    # Use a chirp-modulated random sequence
    freqs = np.arange(n)
    phase = rng.uniform(0, 2 * np.pi, n)
    # Create a sequence with flat spectrum: inverse FFT of flat magnitude
    # |A(k)| = 1 for all k, phase random
    A = np.exp(1j * phase)
    # Ensure A is Hermitian-symmetric for real output
    A_real = np.zeros(2 * n, dtype=complex)
    A_real[:n] = A
    A_real[n:] = np.conj(A[:0:-1]) if n > 1 else np.array([0.0])
    
    # Just use inverse FFT of a random phase sequence
    a = np.fft.ifft(np.exp(1j * rng.uniform(0, 2 * np.pi, n))).real
    # Shift to make positive
    a = a - np.min(a) + 0.1
    a = project_nonneg_sum1(a)
    return a


def generate_block_init(n, rng, n_blocks=3):
    """Generate a block-structured initialization."""
    a = np.zeros(n)
    block_width = n // (2 * n_blocks)
    for b in range(n_blocks):
        center = n * (b + 0.5) / n_blocks + rng.normal(0, n * 0.02)
        center = int(np.clip(center, block_width, n - block_width))
        spread = block_width + rng.integers(0, block_width // 2)
        start = max(0, center - spread)
        end = min(n, center + spread)
        a[start:end] += rng.uniform(0.5, 2.0)
    return project_nonneg_sum1(a)


def simulated_annealing_structured(a, rng, max_iters=400, T_init=0.5, T_min=0.01, time_budget=80.0):
    """
    Simulated annealing with structured moves:
    - Window perturbation with Gaussian noise
    - Block rearrangement
    - Mirror reflection of sub-segments
    - Frequency-domain perturbation
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    cur_a = a.copy()
    cur_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = cur_obj
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T_init * (T_min / T_init) ** (it / max_iters)
        T = max(T, 1e-6)

        move_type = rng.integers(0, 4)
        a_trial = cur_a.copy()

        if move_type == 0:
            # Gaussian window perturbation
            wsize = max(5, n // 20)
            wstart = int(rng.integers(0, n - wsize))
            sigma = 0.1 + 0.3 * (it / max_iters)
            a_trial[wstart:wstart + wsize] += rng.normal(0, sigma * a.mean(), wsize)

        elif move_type == 1:
            # Mirror a sub-segment
            length = max(5, n // 10)
            start = int(rng.integers(0, n - length))
            seg = a_trial[start:start + length].copy()
            a_trial[start:start + length] = seg[::-1]

        elif move_type == 2:
            # Frequency-domain perturbation
            N_fft = max(1, 1 << int(np.log2(2 * n)))
            fa = np.fft.fft(a_trial, N_fft)
            fa *= np.exp(1j * rng.normal(0, 0.05, N_fft))
            a_trial = np.fft.ifft(fa).real[:n]

        else:
            # Block swap
            wsize = max(10, n // 8)
            w1 = int(rng.integers(0, n - wsize))
            w2 = int(rng.integers(0, n - wsize))
            if abs(w1 - w2) >= wsize:
                a_trial[w1:w1 + wsize], a_trial[w2:w2 + wsize] = \
                    a_trial[w2:w2 + wsize].copy(), a_trial[w1:w1 + wsize].copy()

        a_trial = project_nonneg_sum1(a_trial)
        trial_obj = compute_obj(a_trial)

        delta = trial_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_a = a_trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()

    return best_a, best_obj

            best_seq, rng, max_iters=300, T_init=0.3, T_min=0.005,
            time_budget=min(phase3_budget * 0.5, 100.0))
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA improved: {best_score:.8f}")

        # Multiple perturbation + local optimization
        remaining3_sub = phase3_start + phase3_budget - time.monotonic()
        n_perturb = min(8, int(remaining3_sub / 15))
        for pi in range(n_perturb):
            if time.monotonic() > phase3_start + phase3_budget:
                break

            # Different perturbation strengths
            sigma = 0.01 + 0.02 * (pi % 3)
            a_pert = best_seq * np.exp(rng.normal(0, sigma, len(best_seq)))
            a_pert = project_nonneg_sum1(a_pert)

            # Local optimization
            a_local, obj_local = chase_peak(a_pert, n_steps=40, step_size=0.03)
            n_cur = len(a_local)
            if n_cur <= 1000:
                a_local2, obj_local2 = lp_alternate_extended(
                    a_local, max_iters=10, time_budget=min(10, remaining3_sub * 0.2))
                if obj_local2 < obj_local:
                    a_local, obj_local = a_local2, obj_local2

            if obj_local < best_score:
                best_score = obj_local
                best_seq = a_local.copy()
                print(f"  Perturb {pi} improved: {best_score:.8f}")

    # === Phase 4: Multi-length refinement (10% budget) ===
    print(f"\n=== Phase 4: Multi-length Refinement ===")
    print(f"Best before: {best_score:.8f}")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 40:
        phase4_budget = min(remaining4 * 0.15, 150.0)
        phase4_start = time.monotonic()

        n_cur = len(best_seq)
        n_tries = list(dict.fromkeys([
            max(400, n_cur - 30),
            n_cur - 10,
            n_cur,
            n_cur + 10,
            n_cur + 30,
            min(1500, n_cur + 50),
        ]))
        n_tries = [n for n in n_tries if 400 <= n <= 1500]

        per_n_budget = max(5.0, phase4_budget / len(n_tries))

        for n_try in n_tries:
            if time.monotonic() > phase4_start + phase4_budget:
                break
            rem4 = phase4_start + phase4_budget - time.monotonic()
            if rem4 < 5:
                break

            a_res = resample_sequence(best_seq, n_try)
            a_res = project_nonneg_sum1(a_res)

            # Quick optimization cycle
            a_opt, obj_opt = chase_peak(a_res, n_steps=30, step_size=0.03)
            if n_try <= 1000:
                a_opt2, obj_opt2 = lp_alternate_extended(a_opt, max_iters=8,
                                                         time_budget=min(per_n_budget * 0.4, 15.0))
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 15 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        n_final = len(a_final)

        polish_rounds = min(15, int(remaining5 / 8))
        for pr in range(polish_rounds):
            if time.monotonic() > deadline - 10:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            strategy = pr % 4

            if strategy == 0 and n_final <= 1200:
                a_p, obj_p = lp_alternate_extended(a_final, max_iters=min(20, int(rem / 2)),
                                                   time_budget=min(rem * 0.3, 25.0))
            elif strategy == 1:
                a_p, obj_p = gradient_flow_nesterov(a_final, max_iters=100,
                                                    time_budget=min(rem * 0.25, 20.0),
                                                    lr_init=0.02, momentum=0.92)
            elif strategy == 2:
                a_p, obj_p = spectral_flatten(a_final, max_iters=80, time_budget=min(rem * 0.25, 20.0))
            else:
                # Tiny perturbation + chase
                a_pert = a_final * np.exp(rng.normal(0, 0.005, n_final))
                a_pert = project_nonneg_sum1(a_pert)
                a_p, obj_p = chase_peak(a_pert, n_steps=30, step_size=0.02)

            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  Polish r{pr} (strat {strategy}): {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
