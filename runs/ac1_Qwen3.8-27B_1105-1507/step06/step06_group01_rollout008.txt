def fast_conv(a):

        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(a):
    """Build the sparse matrix A where (A @ g)[k] = (a*g)[k] for linear convolution."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-constraint LP best-response iteration (all 2n-1 constraints, solved via HiGHS)
2. Dual-guided gradient steps using LP dual variables for richer direction information
3. Dense line search with refinement for optimal interpolation
4. Coordinate descent polish using golden-section on critical coordinates
5. Multi-length, multi-init search with time-sliced portfolio approach
Key improvements: full LP (not top-k), dual-variable exploitation, adaptive line search.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ g)[k] = sum_j a[k-j]*g[j] for linear conv."""
    n = len(a)
    total_k = 2 * n - 1
    rows = np.empty(total_k * n, dtype=np.int32)
    cols = np.empty(total_k * n, dtype=np.int32)
    vals = np.empty(total_k * n, dtype=np.float64)
    idx = 0
    for k in range(total_k):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        count = j_hi - j_lo + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(j_lo, j_hi + 1)
        vals[idx:idx + count] = a[k - j_hi:j_lo - n + 1:-1] if j_lo > 0 else a[0:k - j_lo + 1][::-1]
        # More carefully:
        js = np.arange(j_lo, j_hi + 1)
        vals[idx:idx + count] = a[k - js]
        idx += count
    # Trim to actual size
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(total_k, n))
    return A


def solve_full_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g_normalized, dual_variables) or (None, None)."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 60})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                dual = r.ineqlin.marginals if hasattr(r.ineqlin, 'marginals') else None
                return g / g_sum, dual
            else:
                return None, None
    except Exception as e:
        pass
    return None, None


def get_dual_from_lp(a, M):
    """Solve LP and extract dual variables (shadow prices of constraints)."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 60})
        if r.success:
            # Dual variables for inequality constraints
            marginal = np.zeros(2 * n - 1)
            if r.ineqlin is not None and hasattr(r.ineqlin, 'marginals'):
                marginal = np.asarray(r.ineqlin.marginals, dtype=np.float64)
            # For scipy linprog with method='highs', marginals are on r.ineqlin.marginals
            # but sign convention may differ. Let's just use the residual to identify active set.
            return marginal
    except Exception:
        pass
    return None


def line_search_dense(a, g, current_M, n_grid=80):
    """Find optimal t in [0,1] minimizing max(conv((1-t)a+tg, (1-t)a+tg)).
    Uses dense grid then refinement."""
    d = g - a
    best_t = 0.0
    best_val = current_M

    # Dense grid
    ts = np.linspace(0.0, 1.0, n_grid)
    objs = np.full(n_grid, np.inf)
    for i, t in enumerate(ts):
        x_t = a + t * d
        objs[i] = fast_conv(x_t).max()

    i_min = int(np.argmin(objs))
    best_val = objs[i_min]
    best_t = ts[i_min]

    # Refine: local density
    if 0 < i_min < n_grid - 1:
        t_lo = ts[max(0, i_min - 2)]
        t_hi = ts[min(n_grid - 1, i_min + 2)]
        ts2 = np.linspace(t_lo, t_hi, 30)
        objs2 = np.full(len(ts2), np.inf)
        for i, t in enumerate(ts2):
            x_t = a + t * d
            objs2[i] = fast_conv(x_t).max()
        i_min2 = int(np.argmin(objs2))
        if objs2[i_min2] < best_val:
            best_val = objs2[i_min2]
            best_t = ts2[i_min2]

    if best_val < current_M:
        return best_t, best_val
    return 0.0, current_M


def dual_guided_gradient(a, M, n_samples=5):
    """Compute a gradient-like direction using the top convolution peaks.
    Returns a direction vector (not normalized)."""
    n = len(a)
    b = fast_conv(a)

    # Get top peaks
    top_k = np.argsort(b)[::-1][:min(n_samples, 2 * n - 1)]

    # Gradient: sum over top peaks of (shifted a)
    grad = np.zeros(n)
    for k in top_k:
        w = 1.0 / (1.0 + (b.max() - b[k]) / max(M, 1e-10))
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                grad[j] += w * a[m]

    # Project to tangent of simplex (remove mean)
    grad = grad - grad.mean()
    return grad


def run_lp_iterations(a_init, max_iters, deadline, verbose=True):
    """Main LP best-response iteration loop."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    t_start = time.monotonic()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M_cur = fast_conv(a).max()
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()

        # Solve full LP
        g, _ = solve_full_lp(a, M_cur)
        if g is None:
            no_improve += 3
            if no_improve > 30:
                break
            continue

        # Line search for best response direction
        t_opt, M_new = line_search_dense(a, g, M_cur, n_grid=60)

        # Also try dual-guided direction
        grad_dir = dual_guided_gradient(a, M_cur)
        grad_dir = -grad_dir  # descent direction
        grad_norm = np.linalg.norm(grad_dir)
        if grad_norm > 1e-10:
            grad_dir = grad_dir / grad_norm
            # Try small steps in gradient direction
            best_grad_M = M_cur
            best_grad_t = 0.0
            for step in [0.05, 0.1, 0.15, 0.2]:
                x_test = a + step * grad_dir * a.sum()  # scale by sum for consistent magnitude
                x_test = np.maximum(x_test, 0.0)
                if x_test.sum() < 1e-10:
                    continue
                x_test = x_test / x_test.sum()
                M_test = fast_conv(x_test).max()
                if M_test < best_grad_M:
                    best_grad_M = M_test
                    best_grad_t = step

        # Choose the better direction
        if M_new < best_grad_M:
            # Use LP best response
            if t_opt > 1e-6 and M_new < M_cur - 1e-12:
                a = a + t_opt * (g - a)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Use gradient direction
            if best_grad_t > 0 and best_grad_M < M_cur - 1e-12:
                grad_dir2 = dual_guided_gradient(a, M_cur)
                grad_dir2 = -grad_dir2
                gn = np.linalg.norm(grad_dir2)
                if gn > 1e-10:
                    grad_dir2 = grad_dir2 / gn
                    a = a + best_grad_t * grad_dir2 * a.sum()
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 40:
            break

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP iter {it}: M={M_cur:.10f} best_M={best_M:.10f} "
                  f"t={t_opt:.4f} no_imp={no_improve} ({elapsed:.0f}s)", flush=True)

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent: optimize one coordinate at a time using golden section."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = np.random.default_rng(seed + sweep).permutation(n)
        b = fast_conv(a)
        sweep_improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            x_cur = a[i]
            # Affected k range: [i, i+n-1] intersect [0, 2n-2]
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            # Precompute for each affected k: b[k] = base + slope*x + quad*x^2
            # (a*a)[k] = sum_j a[j]*a[k-j]
            # When we change a[i] to x:
            #   If k-j == i (j = k-i): term a[j]*x
            #   If j == i: term x*a[k-j]
            #   If j == i and k-j == i (k=2i): term x^2
            # So: new_b[k] = (b[k] - contribution_of_old_x_i) + new_contribution
            # Old contribution of a[i] to b[k]:
            #   a[i]*a[k-i] (if 0<=k-i<n and k-i != i) + a[k-i]*a[i] (if 0<=k-i<n and k-i != i)
            #   = 2*a[i]*a[k-i] if k-i != i
            #   = a[i]^2 if k-i == i (i.e., k=2i)
            # New contribution with x:
            #   2*x*a[k-i] if k-i != i, x^2 if k-i == i

            # Build linear/quadratic model
            qs = 0.0  # quadratic coefficient
            ls = 0.0  # linear coefficient
            base = 0.0

            for k in range(k_lo, k_hi + 1):
                j = k - i
                if 0 <= j < n:
                    if j == i:  # k = 2i
                        qs += 1.0
                        base += b[k] - x_cur * x_cur
                    else:
                        ls += 2.0 * a[j]
                        base += b[k] - 2.0 * x_cur * a[j]
                # else: a[i] doesn't contribute to b[k]

            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0

            def obj_at(x):
                val = base + ls * x + qs * x * x
                return max(val, base_max)

            # Golden section on [0, hi]
            lo = 0.0
            hi = max(3.0 * x_cur, np.sqrt(max(fast_conv(a).max(), 1e-10)) + 0.5, 0.1)

            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = obj_at(c_pt)
            fd = obj_at(d_pt)

            for _ in range(35):
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = obj_at(c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = obj_at(d_pt)

            x_opt = max(0.0, (lo + hi) / 2.0)

            if abs(x_opt - x_cur) > 1e-10 * max(1.0, x_cur):
                a[i] = x_opt
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_improved:
            no_improve += 3
        if no_improve > 15:
            break

    return best_a


def run_smooth_gradient(a_init, deadline, tau=100, lr=0.3, max_iters=5000):
    """FISTA on log-sum-exp smooth approximation of max(conv)."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    t = 1.0
    y = a.copy()
    a_prev = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # Gradient
        grad = np.zeros(n)
        for i in range(n):
            grad[i] = 2.0 * np.dot(p[i:i + n], a)

        grad_tan = grad - grad.mean()
        step = lr / (1.0 + 0.01 * it)
        a_new = y - step * grad_tan

        # Project to simplex
        u = np.sort(a_new)[::-1]
        cv = np.cumsum(u)
        rho_arr = np.arange(1, n + 1, dtype=np.float64)
        cond = (1.0 + u * rho_arr) > cv
        if cond.any():
            rho = int(np.argmax(cond))
            theta = (cv[rho - 1] - 1.0) / rho if rho > 0 else 0.0
        else:
            theta = 0.0
        a_new = np.maximum(a_new - theta, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / a_new_sum

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = np.maximum(y_new, 0.0)
        y_new_sum = y_new.sum()
        if y_new_sum < 1e-15:
            y_new = np.ones(n) / n
        else:
            y_new = y_new / y_new_sum

        a_prev = a
        a = a_new
        y = y_new
        t = t_new

        if it % 10 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for noise_level in [0.02, 0.05, 0.1, 0.15]:
            p_n = np.maximum(p + rng.normal(0, noise_level, n), 1e-4)
            inits.append(p_n / p_n.sum())

    inits.append(np.ones(n) / n)

    # Cosine windows
    for freq in [1, 2, 3, 4]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())

    # Gaussian
    gauss = np.exp(-6.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    # Parabolic (concave)
    poly = (t * (1 - t)) * 4 + 0.1
    inits.append(poly / poly.sum())

    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # Triangular
    tri = 1.0 - np.abs(2 * t - 1)
    tri = np.maximum(tri, 0.05)
    inits.append(tri / tri.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')

    def update(self, a):

            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()

            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)

            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768) / 768
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)

    # Determine lengths to try
    n_primary = parent.size if parent.size > 0 else 768
    lengths = [n_primary]
    for nl in [512, 768, 1024, 1536]:
        if nl not in lengths and 256 <= nl <= 2048:
            lengths.append(nl)
    lengths = [l for l in lengths if l >= 256]

    # Sort by priority
    lengths.sort(key=lambda x: (x != n_primary, abs(x - n_primary)))
    n_lengths = min(len(lengths), 4)
    lengths = lengths[:n_lengths]

    print(f"\n{'='*60}", flush=True)
    print(f"Lengths to try: {lengths}", flush=True)
    print(f"Total time budget: {budget_s}s, deadline at t={deadline - time.monotonic():.0f}s remaining", flush=True)
    print(f"{'='*60}", flush=True)

    # Phase 1: Primary optimization on best length (60% of time)
    t_phase1_end = time.monotonic() + 0.6 * (deadline - time.monotonic() - 100)
    print(f"\nPHASE 1: Primary optimization (deadline t={t_phase1_end - time.monotonic():.0f}s)", flush=True)

    # Focus on the primary length with multiple inits
    n_best = n_primary
    inits = generate_inits(n_best, rng, tracker.best)
    time_per_init = (t_phase1_end - time.monotonic() - 5) / max(1, len(inits))
    time_per_init = max(time_per_init, 30)  # minimum 30s per init

    for ii, a0 in enumerate(inits):
        if time.monotonic() > t_phase1_end - 5:
            break
        sub_dl = min(t_phase1_end - 2, time.monotonic() + time_per_init)
        print(f"\n  Init {ii}/{len(inits)-1} (n={n_best}, ~{time_per_init:.0f}s):", flush=True)
        a_opt = optimize_sequence(a0, n_best, sub_dl, seed=seed + ii)
        tracker.update(a_opt)

    # Try secondary length if time remains
    if time.monotonic() < t_phase1_end - 60 and len(lengths) > 1:
        n_sec = lengths[1]
        if n_sec != n_best:
            print(f"\n  Secondary length n={n_sec}:", flush=True)
            a0 = resample_seq(tracker.best, n_sec) if tracker.best is not None else np.ones(n_sec) / n_sec
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            a_opt = optimize_sequence(a0, n_sec, t_phase1_end - 5, seed=seed + 100)
            tracker.update(a_opt)

    tracker.report("After Phase 1")

    # Phase 2: Intensive LP polish on best (20% of time)
    t_phase2_end = time.monotonic() + 0.2 * (deadline - time.monotonic())
    print(f"\nPHASE 2: Intensive LP polish (deadline t={t_phase2_end - time.monotonic():.0f}s)", flush=True)

    if tracker.best is not None:
        n_pol = tracker.best.size
        # Multiple LP passes with increasing aggressiveness
        a_cur = tracker.best.copy()
        for polish_round in range(3):
            if time.monotonic() > t_phase2_end - 10:
                break
            round_dl = t_phase2_end - 5
            a_pol = run_lp_iterations(a_cur, 100, round_dl, verbose=(polish_round == 0))
            M_pol = fast_conv(a_pol).max()
            print(f"  Polish round {polish_round}: M={M_pol:.10f} J={compute_J(a_pol):.6f}", flush=True)
            tracker.update(a_pol)
            if M_pol >= fast_conv(a_cur).max():
                break
            a_cur = a_pol

    tracker.report("After Phase 2")

    # Phase 3: Coordinate descent + gradient polish (10% of time)
    t_phase3_end = time.monotonic() + 0.1 * (deadline - time.monotonic())
    print(f"\nPHASE 3: CD + gradient polish (deadline t={t_phase3_end - time.monotonic():.0f}s)", flush=True)

    if tracker.best is not None:
        # Coordinate descent
        a_cd = run_coordinate_descent(tracker.best, 50, t_phase3_end - 10, seed=42)
        tracker.update(a_cd)

        # Smooth gradient from CD result
        if time.monotonic() < t_phase3_end - 10:
            a_sg = run_smooth_gradient(a_cd, t_phase3_end - 3, tau=200, lr=0.1, max_iters=3000)
            tracker.update(a_sg)

        # Final LP pass
        if time.monotonic() < t_phase3_end - 3:
            a_final = run_lp_iterations(tracker.best, 50, t_phase3_end, verbose=False)
            tracker.update(a_final)

    tracker.report("After Phase 3")

    # Phase 4: Final coordinate descent squeeze
    if tracker.best is not None and time.monotonic() < deadline - 10:
        print(f"\nPHASE 4: Final CD squeeze", flush=True)
        a_final = run_coordinate_descent(tracker.best, 30, deadline - 3, seed=999)
        tracker.update(a_final)

    tracker.report("FINAL")

    # Ensure we return something valid
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
