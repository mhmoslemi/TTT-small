        # Use optimal 1D step
        t_opt = optimal_step_1d(a, g, best_M, n)

        if t_opt < 1e-6:
            no_improve += 1
            n_active = min(n_active + 5, n_active_max)
            if no_improve > 20:
                break
            continue

        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 0.01:
            no_improve += 1
            continue
        a_new = a_new / s_new
        M_new = fast_conv(a_new).max()

        if M_new < best_M - 1e-15:
            a = a_new
            best_M = M_new
            best_a = a.copy()
            no_improve = 0
            # Slightly reduce active set on success
            n_active = max(20, int(n_active * 0.95))
        else:
            no_improve += 1
            n_active = min(n_active + 10, n_active_max)
            if no_improve > 25:
                break

    return best_a


def run_lp_optimize_v2(a_init, max_iters, deadline, n_active=60):
    """LP optimization with multiple candidate step sizes (more robust)."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g, _ = lp_best_response(a, best_M, n_active)
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        improved = False
        # Try optimal step first, then a range of alternatives
        t_candidates = [optimal_step_1d(a, g, best_M, n)]
        t_candidates.extend([0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.05, 0.08])
        # Deduplicate and sort
        t_candidates = sorted(set([max(0.0, min(1.0, t)) for t in t_candidates]))

        best_trial_M = best_M
        best_trial_a = None
        for t in t_candidates:
            if t < 1e-6:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 0.01:
                continue
            a_trial = a_trial / st
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_trial_a = a_trial.copy()

        if best_trial_a is not None and best_trial_M < best_M - 1e-15:
            a = best_trial_a
            best_M = best_trial_M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    return best_a


def spectral_init(n, rng, max_iters=2000):
    """Generate initialization using AMOS-like spectral flattening.
    Alternates between: (1) project to make |FFT|^2 flat, (2) project to non-negative + sum=1."""
    a = rng.uniform(0.5, 1.5, n)
    a = a / a.sum()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(max_iters):
        # Compute current spectrum
        fa = np.fft.rfft(a, size)
        spec = np.abs(fa) ** 2

        # Target: flat spectrum (all values equal to the mean)
        target = spec.mean()

        # Adjust: scale magnitudes to flatten
        # New magnitude = sqrt(target) for all frequencies
        # But keep phases
        phase = np.exp(1j * np.angle(fa))
        new_mag = np.sqrt(np.maximum(spec, 1e-20))
        # Blend: partially flatten
        target_mag = np.sqrt(target)
        blended_mag = 0.7 * new_mag + 0.3 * target_mag
        fa_new = blended_mag * phase

        # Inverse FFT and keep real part
        a_new = np.fft.irfft(fa_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new

    return a


def golay_like_init(n, rng):
    """Generate a Golay-like initialization using recursive structure."""
    # Start with a small Barker-like sequence and expand
    base = np.array([1, 1, 1, -1, 1, 1, -1], dtype=np.float64)
    # Make non-negative
    base = np.abs(base) + 0.3
    base = base / base.sum()

    # Upsample to length n using interpolation
    t_old = np.linspace(0, 1, len(base))
    t_new = np.linspace(0, 1, n)
    a = np.interp(t_new, t_old, base)

    # Add structured perturbation
    freqs = rng.integers(1, 5, size=3)
    for f in freqs:
        a = a + 0.05 * np.cos(2 * np.pi * f * np.arange(n) / n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def structured_init(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        # Resample parent
        t_old = np.linspace(0, 1, len(parent))
        p = np.interp(t, t_old, parent)
        p = np.maximum(p, 1e-4)
        inits.append(p / p.sum())
        # Noisy version
        p2 = np.maximum(p + rng.normal(0, 0.02, n), 0.01)
        inits.append(p2 / p2.sum())
        # More noise
        p3 = np.maximum(p + rng.normal(0, 0.08, n), 0.01)
        inits.append(p3 / p3.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Parabolic (concave)
    par = (1 - 4 * (t - 0.5) ** 2) + 0.1
    inits.append(par / par.sum())

    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())

    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    # Spectral init
    inits.append(spectral_init(n, rng, max_iters=500))

    # Golay-like
    inits.append(golay_like_init(n, rng))

    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


def coordinate_descent_exact(a, i, n, b):
    """Exact 1D coordinate update for a[i]."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 1.0, 0.5)
    for _ in range(5):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 50.0:
            hi *= 2.0
        else:
            break

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with exact 1D optimization."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False

        for i in order:
            x_opt = coordinate_descent_exact(a, i, n, b)
            if abs(x_opt - a[i]) > 1e-12:
                old_val = a[i]
                a[i] = x_opt
                # Update b
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - old_val * old_val)
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old_val)
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        b = fast_conv(a)  # Renormalize b
        M = b.max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_improved:
            no_improve += 2
        if no_improve > 15:
            break

    return best_a


def run_combined_optimize(a0, n, deadline, seed):
    """Run combined optimization: LP -> coordinate descent -> LP again."""
    best = a0.copy()
    best_M = fast_conv(best).max()
    t_start = time.monotonic()
    remaining = deadline - t_start
    if remaining < 5:
        return best

    # Stage 1: LP optimization (60% of time)
    t1 = t_start + 0.6 * remaining
    a_lp = run_lp_optimize_v2(best, 1000, t1, n_active=60)
    M_lp = fast_conv(a_lp).max()
    if M_lp < best_M:
        best_M = M_lp
        best = a_lp.copy()
    print(f"    After LP v2: M={best_M:.12f}", flush=True)

    # Stage 2: Coordinate descent (20% of time)
    t2 = t_start + 0.8 * remaining
    a_cd = run_coordinate_descent(best, 500, t2, seed=seed)
    M_cd = fast_conv(a_cd).max()
    if M_cd < best_M:
        best_M = M_cd
        best = a_cd.copy()
    print(f"    After CD: M={best_M:.12f}", flush=True)

    # Stage 3: Final LP polish (20% of time)
    a_final = run_lp_optimize(best, 500, deadline, n_active_init=80, n_active_max=150)
    M_final = fast_conv(a_final).max()
    if M_final < best_M:
        best_M = M_final
        best = a_final.copy()
    print(f"    After final LP: M={best_M:.12f}", flush=True)

    return best


def peak_redistribute(a, n, rng, n_trials=20):
    """Try to reduce the peak by redistributing mass from contributors to non-contributors."""
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    best_a = a.copy()
    best_M = b.max()

    for trial in range(n_trials):
        # Identify contributors to the peak
        contrib_mask = np.zeros(n, dtype=bool)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib_mask[i] = True

        # Randomly select a contributor to reduce and a non-contributor to increase
        contributors = np.where(contrib_mask)[0]
        non_contributors = np.where(~contrib_mask)[0]

        if len(contributors) < 2 or len(non_contributors) < 1:
            continue

        i_src = contributors[rng.integers(len(contributors))]
        i_dst = non_contributors[rng.integers(len(non_contributors))]

        # Try several amounts of redistribution
        for frac in [0.01, 0.02, 0.05, 0.08, 0.12]:
            delta = frac * a[i_src]
            if delta < 1e-12:
                continue
            a_trial = a.copy()
            a_trial[i_src] -= delta
            a_trial[i_dst] += delta
            if a_trial[i_src] < 0:
                continue
            s = a_trial.sum()
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M:
                best_M = M_trial
                best_a = a_trial.copy()

    return best_a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)

                parent = parent / parent.sum()
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)

        pass

            M_opt = fast_conv(a_opt).max()
