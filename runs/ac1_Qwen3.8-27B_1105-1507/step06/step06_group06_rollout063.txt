    a = project_nonneg_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    N = 1
    while N < 3*n:  # need N >= (2n-1) + n - 1 = 3n-2 for the conv of w and a_rev
        N <<= 1
    
    a_rev = a[::-1].copy()
    
    tau = 0.005  # temperature for softmax
    eta = 0.3  # step size
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute convolution
        A = np.fft.fft(a, N)
        conv = np.fft.ifft(A * A).real[:2*n-1]
        conv_max = conv.max()
        
        # Compute softmax weights
        shifted = (conv - conv_max) / tau
        weights = np.exp(np.clip(shifted, -30, 30))
        w_sum = weights.sum()
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(2*n-1)
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Compute gradient via FFT
        W = np.fft.fft(weights, N)
        Ar = np.fft.fft(a_rev, N)
        corr_full = np.fft.ifft(W * Ar).real
        # grad[i] = corr_full[n-1+i]
        grad = corr_full[n-1:n-1+n]
        
        # Also account for the factor of 2 in d(conv[k])/da[i] = 2*a[k-i]
        # Wait, I derived grad[i] = sum_k weights[k]*a[k-i], but the actual gradient is
        # grad[i] = 2 * sum_k weights[k]*a[k-i] for the non-diagonal terms
        # and grad[i] = sum_k weights[k]*2*a[k-i] for diagonal terms (k=2i)

        # So grad[i] = 2 * sum_k weights[k]*a[k-i]
        grad *= 2.0
        
        # Mirror descent update
        log_update = -eta * grad
        # For numerical stability
        log_update -= log_update.max()
        mult = np.exp(log_update)
        a_new = a * mult
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.05, 2.0)  # increase step size slightly
        else:
            eta *= 0.9  # decrease step size
            a = a_new  # still move (for exploration)
            # Or: a stays the same (strict descent)
        
        # Cool down tau slowly
        tau *= 0.9999
    
    return best_a

"""
Mirror descent optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated subgradient computation with adaptive temperature and step size.
Combined with LP alternation, multi-length exploration, and simulated annealing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 1500:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 1500:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def mirror_descent(a, max_iters=20000, time_budget=60.0, tau_init=0.01, tau_min=0.001,
                   eta_init=0.5, verbose=False):
    """
    Mirror descent (exponential weights) for minimizing max(conv(a,a))
    subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64)).copy()
    # Ensure strictly positive
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 3 * n:  # Need N >= (2n-1) + n - 1 = 3n - 2
        N <<= 1
    
    a_rev = a[::-1].copy()
    tau = tau_init
    eta = eta_init
    stall_count = 0
    best_ever_obj = best_obj
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute convolution
        A = np.fft.rfft(a, N)
        conv = np.fft.irfft(A * A, N)[:2 * n - 1]
        conv_max = float(np.max(conv))
        
        # Compute softmax weights
        shifted = (conv - conv_max) / tau
        # Clip for numerical stability

        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))

            weights = np.zeros(2 * n - 1)
            weights[idx] = 1.0

        a_new = project_nonneg_sum1(a_new)

"""
Mirror descent with FFT-accelerated subgradient for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Primary optimizer is multiplicative weights (exponential weights) in log-space with adaptive
temperature annealing. Combined with LP alternation, multi-length exploration, and simulated
annealing for global diversity. Key improvement over prior approach: O(n log n) per iteration
enables millions of steps within time budget, achieving tight convergence to local optima.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv_fast(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fast(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, time_limit=15):
    n = len(a)
    if n > 1200:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_max_sum(a, M, time_limit=15):
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def mirror_descent_core(a, time_budget, tau_init=0.008, tau_min=0.0005,
                        eta_init=0.4, max_iters=500000, verbose=False,
                        cool_rate=0.99997, eta_grow=1.0005, eta_shrink=0.96):
    """
    Fast mirror descent (multiplicative weights) for min max(conv(a,a)) s.t. sum(a)=1, a>=0.
    Each iteration: O(n log n) via FFT.
    """
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64)).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_conv_max = float(np.max(compute_conv_fast(a)))
    best_obj = 2.0 * n * best_conv_max  # sum=1 so obj = 2n*max_conv
    start = time.monotonic()

    N = 1
    while N < 3 * n:
        N <<= 1

    tau = tau_init
    eta = eta_init
    stall = 0
    accept_threshold = 0.02

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute convolution via FFT
        A = rfft(a, N)
        conv = irfft(A * A, N)[:2 * n - 1]
        conv_max = float(np.max(conv))
        cur_obj = 2.0 * n * conv_max

        # Softmax weights over conv values
        shifted = (conv - conv_max) / tau
        shifted = np.clip(shifted, -25.0, 0.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-200:
            weights = np.zeros(2 * n - 1)
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum

        # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i]
        # Computed as FFT correlation
        a_rev = a[::-1]
        W = rfft(weights, N)
        Ar = rfft(a_rev, N)
        corr = irfft(W * Ar, N)
        grad = corr[n - 1:n - 1 + n] * 2.0

        # Multiplicative update
        log_mult = -eta * grad
        log_mult_max = float(np.max(log_mult))
        if log_mult_max > -700:
            log_mult -= log_mult_max
            mult = np.exp(log_mult)
            a_new = a * mult
        else:
            # Fallback: rank-based update
            mult = np.exp(-eta * (grad - grad.min()))
            a_new = a * mult

        a_new = project_nonneg_sum1(a_new)

        # Evaluate
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * eta_grow, 3.0)
        else:
            stall += 1
            if stall % 50 == 0:
                eta = max(eta * eta_shrink, 0.01)
            if stall > 2000:
                # Accept slightly worse moves for exploration
                if obj_new < best_obj * (1.0 + accept_threshold):
                    a = a_new.copy()
                    stall = 0
                    eta = max(eta * 0.5, 0.01)
                else:
                    break

        a = a_new.copy()
        tau = max(tau * cool_rate, tau_min)

        if verbose and it % 50000 == 0:
            print(f"      MD it={it}: obj={cur_obj:.8f}, best={best_obj:.8f}, tau={tau:.6f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_obj


def lp_alternate(a, time_budget, max_iters=30):
    """LP alternating with adaptive blending."""
    n = len(a)
    if n > 1200:
        return a, compute_obj(a)

    a = project_nonneg_sum1(a.astype(np.float64)).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv_fast(a)
        M = float(np.max(conv))

        g = solve_lp_max_sum(a, M, time_limit=8)
        if g is None:
            break

        # Try multiple blending ratios
        best_t_obj = best_obj
        best_t_a = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break

    return best_a, best_obj


def freq_domain_flatten(a, time_budget, iters=2000):
    """
    Frequency domain approach: iteratively flatten the power spectrum to reduce conv peak.
    """
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64)).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    alpha = 0.9  # compression strength
    for it in range(iters):
        if time.monotonic() - start > time_budget:
            break

        A = np.fft.fft(a, N)
        power = np.abs(A) ** 2
        power_max = float(np.max(power))
        if power_max < 1e-30:
            break

        # Compress the power spectrum (reduce dynamic range)
        power_norm = power / power_max
        power_compressed = power_norm ** alpha
        power_compressed[0] = power[0] / power_max  # preserve DC

        # Reconstruct with compressed magnitude, same phase
        phase = np.angle(A)
        new_mag = np.sqrt(power_compressed * power_max)
        A_new = new_mag * np.exp(1j * phase)

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            alpha = max(alpha * 0.999, 0.3)
        else:
            alpha = min(alpha * 1.01, 0.99)

    return best_a, best_obj


def simulated_annealing(a, rng, time_budget, T_init=0.3, T_min=0.003):
    """SA with window moves for global exploration."""
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64)).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    total_iters = 50000
    T = T_init

    for it in range(total_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T_init * (T_min / T_init) ** (it / total_iters)

        move_type = rng.integers(0, 4)
        a_trial = a.copy()

        if move_type == 0:
            # Window noise
            wsize = max(5, n // 20)
            wstart = int(rng.integers(0, n - wsize))
            noise = rng.normal(0, 0.5, wsize)
            a_trial[wstart:wstart + wsize] += noise * (a.mean() + 0.01)
        elif move_type == 1:
            # Window swap
            wsize = max(5, n // 25)
            w1 = int(rng.integers(0, n - wsize))
            w2 = int(rng.integers(0, n - wsize))
            if abs(w1 - w2) >= wsize:
                a_trial[w1:w1 + wsize], a_trial[w2:w2 + wsize] = \
                    a_trial[w2:w2 + wsize].copy(), a_trial[w1:w1 + wsize].copy()
        elif move_type == 2:
            # Spike redistribution
            wsize = max(10, n // 15)
            wstart = int(rng.integers(0, n - wsize))
            mass = float(a_trial[wstart:wstart + wsize].sum())
            a_trial[wstart:wstart + wsize] = 0
            targets = rng.choice(n, size=min(wsize, n // 2), replace=False)
            a_trial[targets] += mass / len(targets)
        else:
            # Gaussian perturbation
            a_trial = a_trial * np.exp(rng.normal(0, 0.1, n))

        a_trial = project_nonneg_sum1(a_trial)
        trial_obj = compute_obj(a_trial)

        delta = trial_obj - compute_obj(a)
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(T, 1e-10)):
            a = a_trial.copy()
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()

    return best_a, best_obj


def chase_peak(a, n_steps=100, step_size=0.05):
    """Simple chase-the-peak with subgradient."""
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64)).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)

    N = 1
    while N < 3 * n:
        N <<= 1

    for _ in range(n_steps):
        A = rfft(a, N)
        conv = irfft(A * A, N)[:2 * n - 1]
        k_star = int(np.argmax(conv))

        # Subgradient at the peak
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
            j2 = i + n - 1 - k_star
            if 0 <= j2 < n and j2 != j:
                grad[i] += a[j2]

        grad_centered = grad - float(np.mean(grad))

        for ss in [step_size, step_size * 0.5, step_size * 2.0]:
            a_try = a - ss * grad_centered
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a, best_obj

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian
    x = np.linspace(-2.5, 2.5, n)
    g = np.exp(-x ** 2 / 2)
    inits.append(g / g.sum())

    # Sinc-like (for flat convolution)
    x = np.linspace(-3, 3, n)
    sinc = np.sinc(x / 3)
    sinc = np.maximum(sinc, 0.01)
    inits.append(sinc / sinc.sum())

    # Raised cosine
    x = np.linspace(0, 1, n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # Bimodal
    x = np.linspace(0, 1, n)
    bim = np.exp(-((x - 0.25) ** 2) / 0.01) + np.exp(-((x - 0.75) ** 2) / 0.01)
    inits.append(bim / bim.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=float)
    inits.append(ramp / ramp.sum())

    if parent is not None:
        resampled = resample_sequence(parent, n)
        inits.append(resampled)
        pert = resampled * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_nonneg_sum1(pert))

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining mirror descent, LP alternation,
    frequency domain flattening, and simulated annealing across multiple sequence lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    print(f"Starting best: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time budget: {budget_s}s, deadline at t={deadline - start_time:.1f}s")

    # === Phase 1: Multi-length screening with mirror descent (25% time) ===
    print("\n=== Phase 1: Multi-length Screening ===")
    phase1_budget = min((deadline - start_time) * 0.25, 300.0)
    n_parent = len(best_seq)
    n_options = sorted(set([
        max(200, n_parent - 200),
        max(200, n_parent - 100),
        n_parent,
        n_parent + 100,
        n_parent + 200,
        n_parent + 400,
        min(3000, n_parent + 600),
        min(3000, n_parent + 1000),
    ]))
    n_options = [n for n in n_options if 100 <= n <= 3000]
    n_options = list(dict.fromkeys(n_options))

    per_n_budget = max(5.0, phase1_budget / len(n_options))
    phase1_start = time.monotonic()

    phase1_results = []
    for n_try in n_options:
        if time.monotonic() > phase1_start + phase1_budget:
            break
        remaining_n = phase1_start + phase1_budget - time.monotonic()
        if remaining_n < 5:
            break

        # Try a few initializations
        inits = generate_inits(n_try, rng, parent)
        best_local = None
        best_local_obj = np.inf

        per_init_budget = max(1.0, remaining_n / (len(inits) + 1))
        for init in inits[:5]:
            if time.monotonic() > phase1_start + phase1_budget:
                break
            rem = phase1_start + phase1_budget - time.monotonic()
            if rem < 1:
                break
            # Quick mirror descent
            a_md, obj_md = mirror_descent_core(init, time_budget=min(per_init_budget, 15.0),
                                               tau_init=0.01, eta_init=0.3,
                                               cool_rate=0.9999)
            if obj_md < best_local_obj:
                best_local_obj = obj_md
                best_local = a_md.copy()

        if best_local is not None:
            phase1_results.append((best_local_obj, best_local, n_try))
            if best_local_obj < best_score:
                best_score = best_local_obj
                best_seq = best_local.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={best_local_obj:.8f}")

    # Sort results
    phase1_results.sort(key=lambda x: x[0])
    print(f"  Phase 1 complete. Best: {phase1_results[0][0]:.8f}" if phase1_results else "  Phase 1: no results")

    # === Phase 2: Intensive optimization on top candidates (45% time) ===
    print(f"\n=== Phase 2: Intensive Optimization (best={best_score:.8f}) ===")
    phase2_budget = min((deadline - time.monotonic()) * 0.45, 600.0)
    phase2_start = time.monotonic()

    # Focus on the global best and top candidates from phase 1
    top_candidates = [(best_score, best_seq.copy(), len(best_seq))]
    for obj_c, a_c, n_c in phase1_results[:3]:
        if abs(n_c - len(top_candidates[0][2])) > 10:
            top_candidates.append((obj_c, a_c.copy(), n_c))

    for ci, (obj_init, a_init, n_init) in enumerate(top_candidates[:3]):
        rem = phase2_start + phase2_budget - time.monotonic()
        if rem < 10:
            break
        alloc = min(rem / (len(top_candidates) - ci), 200.0)
        if alloc < 10:
            break

        print(f"  Candidate {ci} (n={n_init}): obj={obj_init:.8f}, budget={alloc:.0f}s")
        a = project_nonneg_sum1(a_init)
        local_best = a.copy()
        local_obj = compute_obj(a)

        # Long mirror descent
        md_budget = alloc * 0.5
        a_md, obj_md = mirror_descent_core(a, time_budget=md_budget,
                                           tau_init=0.006, tau_min=0.0003,
                                           eta_init=0.3, cool_rate=0.99998,
                                           eta_grow=1.0003, eta_shrink=0.97)
        if obj_md < local_obj:
            local_obj = obj_md
            local_best = a_md.copy()
            print(f"    After MD: {local_obj:.8f}")

        # LP alternation (if n small enough)
        rem2 = phase2_start + phase2_budget - time.monotonic()
        if n_init <= 1200 and rem2 > 10:
            lp_budget = min(alloc * 0.25, rem2 * 0.5, 60.0)
            a_lp, obj_lp = lp_alternate(local_best, time_budget=lp_budget, max_iters=20)
            if obj_lp < local_obj:
                local_obj = obj_lp
                local_best = a_lp.copy()
                print(f"    After LP: {local_obj:.8f}")

        # Frequency domain flattening
        rem3 = phase2_start + phase2_budget - time.monotonic()
        if rem3 > 5:
            fd_budget = min(alloc * 0.1, rem3 * 0.5, 20.0)
            a_fd, obj_fd = freq_domain_flatten(local_best, time_budget=fd_budget)
            if obj_fd < local_obj:
                local_obj = obj_fd
                local_best = a_fd.copy()
                print(f"    After FD: {local_obj:.8f}")

        # Another round of mirror descent (finer tuning)
        rem4 = phase2_start + phase2_budget - time.monotonic()
        if rem4 > 10:
            md2_budget = min(rem4 * 0.6, 50.0)
            a_md2, obj_md2 = mirror_descent_core(local_best, time_budget=md2_budget,
                                                 tau_init=0.003, tau_min=0.0002,
                                                 eta_init=0.15, cool_rate=0.99999,
                                                 eta_grow=1.0002, eta_shrink=0.98)
            if obj_md2 < local_obj:
                local_obj = obj_md2
                local_best = a_md2.copy()
                print(f"    After MD2: {local_obj:.8f}")

        if local_obj < best_score:
            best_score = local_obj
            best_seq = local_best.copy()
            print(f"  ** GLOBAL UPDATE: n={n_init}, obj={best_score:.8f} **")

    # === Phase 3: Simulated Annealing + Crossover (20% time) ===
    print(f"\n=== Phase 3: Diversity Search (best={best_score:.8f}) ===")
    phase3_budget = min((deadline - time.monotonic()) * 0.20, 250.0)
    phase3_start = time.monotonic()
    n_best = len(best_seq)

    # SA on the best
    sa_budget = phase3_budget * 0.4
    a_sa, obj_sa = simulated_annealing(best_seq, rng, time_budget=sa_budget,
                                       T_init=0.2, T_min=0.002)
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = a_sa.copy()
        print(f"  SA: NEW BEST {best_score:.8f}")

    # Crossover between top candidates
    rem3 = phase3_start + phase3_budget - time.monotonic()
    if rem3 > 20 and len(phase1_results) >= 2:
        # Try crossover
        a1 = best_seq
        a2 = resample_sequence(phase1_results[1][1], n_best)
        for t in [0.3, 0.5, 0.7]:
            a_cross = (1 - t) * a1 + t * a2
            a_cross = project_nonneg_sum1(a_cross)
            a_cross, obj_cross = mirror_descent_core(a_cross, time_budget=min(15.0, rem3 * 0.2),
                                                     tau_init=0.005, eta_init=0.2)
            if obj_cross < best_score:
                best_score = obj_cross
                best_seq = a_cross.copy()
                print(f"  Crossover t={t}: NEW BEST {best_score:.8f}")
            rem3 = phase3_start + phase3_budget - time.monotonic()
            if rem3 < 5:
                break

    # Destroy and repair
    rem3 = phase3_start + phase3_budget - time.monotonic()
    for dr in range(min(5, int(rem3 / 15))):
        if time.monotonic() > phase3_start + phase3_budget:
            break
        # Destroy: zero out random 10-20%
        n_destroy = int(n_best * (0.1 + 0.03 * dr))
        destroy_idx = rng.choice(n_best, size=n_destroy, replace=False)
        a_dest = best_seq.copy()
        a_dest[destroy_idx] = 0
        a_dest = project_nonneg_sum1(a_dest)

        # Repair with mirror descent
        a_rep, obj_rep = mirror_descent_core(a_dest, time_budget=min(10.0, rem3 * 0.3),
                                             tau_init=0.008, eta_init=0.3)
        if obj_rep < best_score:
            best_score = obj_rep
            best_seq = a_rep.copy()
            print(f"  D&R {dr}: NEW BEST {best_score:.8f}")

    # === Phase 4: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.8f}) ===")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 10:
        n_final = len(best_seq)

        # Long, fine-tuned mirror descent
        md_final_budget = remaining4 * 0.4
        a_f, obj_f = mirror_descent_core(best_seq, time_budget=md_final_budget,
                                         tau_init=0.002, tau_min=0.0001,
                                         eta_init=0.1, cool_rate=0.99999,
                                         eta_grow=1.0001, eta_shrink=0.98)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final MD: {best_score:.8f}")

        # LP polish
        rem = deadline - time.monotonic()
        if n_final <= 1200 and rem > 10:
            a_lp2, obj_lp2 = lp_alternate(best_seq, time_budget=min(rem * 0.3, 40.0),
                                          max_iters=15)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  Final LP: {best_score:.8f}")

        # One more MD round
        rem = deadline - time.monotonic()
        if rem > 10:
            a_f2, obj_f2 = mirror_descent_core(best_seq, time_budget=rem * 0.5,
                                               tau_init=0.001, tau_min=0.0001,
                                               eta_init=0.05, cool_rate=0.99999,
                                               eta_grow=1.0001, eta_shrink=0.99)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final MD2: {best_score:.8f}")

        # Try a slightly different length
        rem = deadline - time.monotonic()
        if rem > 15:
            for n_adj in [n_final + 20, n_final - 20, n_final + 50]:
                if n_adj < 50 or n_adj > 3000:
                    continue
                a_res = resample_sequence(best_seq, n_adj)
                a_res, obj_res = mirror_descent_core(a_res, time_budget=min(rem * 0.3, 20.0),
                                                     tau_init=0.003, eta_init=0.2)
                if obj_res < best_score:
                    best_score = obj_res
                    best_seq = a_res.copy()
                    print(f"  Length {n_adj}: NEW BEST {best_score:.8f}")
                rem = deadline - time.monotonic()
                if rem < 5:
                    break

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    # Ensure all values are valid for evaluate_sequence
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        print(f"\n=== FINAL: n={len(result)}, score={val:.8f}, time={time.monotonic()-start_time:.1f}s ===")
    else:
        print(f"Validation failed, using fallback")
        result = [1.0] * 100

    return result
