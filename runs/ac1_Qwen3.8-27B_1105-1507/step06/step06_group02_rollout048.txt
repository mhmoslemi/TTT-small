"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive log-sum-exp gradient descent with Nesterov acceleration and FFT-based gradients
2. LP mirror descent with Anderson acceleration (window=10) and warm-start heuristics
3. Multi-scale length exploration (n=768 to 2048)
4. Population-based diversity with random perturbations to escape local minima
5. Coordinate descent refinement for final polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Fast full convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize <<= 1
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_J(a):
    """Compute objective J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def grad_max_conv(a, tau):
    """
    Compute gradient of smooth max approximation via FFT.
    Returns (grad, b, b_max) where grad = 2 * sum_k p_k * a_{k-j} (aligned).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    max_shifted = shifted.max()
    if max_shifted > 500:
        shifted = shifted - max_shifted + 500
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    # grad_j = 2 * sum_m a_m * p_{j+m}
    # Using FFT: this is 2 * (a * flip(p))_{2n-2-j}
    p_rev = p[::-1]
    fsize = 1
    while fsize < 2 * n - 1:
        fsize <<= 1
    fa = np.fft.rfft(a, fsize)
    fp = np.fft.rfft(p_rev, fsize)
    conv_ar = np.fft.irfft(fa * fp, fsize)
    grad = 2.0 * conv_ar[2 * n - 2: n - 1: -1]
    return grad, b, b_max


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP: (conv(a,g))_k <= M for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            v = a[k - j]
            if v > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, cached_A=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if cached_A is None:
        A = build_lp_matrix(n, a)
    else:
        A = cached_A
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(history_x, history_Fx, m, extrapolate=True):
    """Anderson mixing: find optimal convex combination of history_Fx."""
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-10 * np.eye(m)
    ones = np.ones(m)
    try:
        alpha = np.linalg.solve(G, ones)
        denom = alpha.sum()
        if abs(denom) < 1e-14:
            alpha = ones / m
        else:
            alpha = alpha / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    if extrapolate:
        alpha = np.clip(alpha, -0.5, 1.5)
        s = alpha.sum()
        if abs(s) > 1e-14:
            alpha = alpha / s
        else:
            alpha = ones / m
    else:
        alpha = np.maximum(alpha, 0)
        s = alpha.sum()
        if s > 1e-14:
            alpha = alpha / s
        else:
            alpha = ones / m
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def nesterov_optimize(a, deadline, max_iters=5000, tau_start=2.0, tau_end=500.0,
                      lr_init=0.3, lr_end=0.001, momentum_start=0.85, momentum_end=0.98):
    """Nesterov-accelerated gradient descent with adaptive log-sum-exp smoothing."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    t_k = 1.0
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        # Adaptive temperature: increase gradually
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        # Cosine annealing learning rate
        lr = 0.5 * (lr_end + (lr_init - lr_end) * np.cos(np.pi * frac))
        # Adaptive momentum
        beta = momentum_start + (momentum_end - momentum_start) * frac
        grad, b, b_max = grad_max_conv(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        theta = (t_new - 1.0) / (t_new + 1.0)
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + theta * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y = y / sy
        a = a_new
        t_k = t_new
    return best_a


def lp_mirror(a, deadline, max_iters=300, anderson_window=10, t_base=0.3,
              min_t=0.002, explore_every=20, explore_scale=0.05):
    """
    LP mirror descent with Anderson acceleration and periodic random exploration.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    hist_x = []
    hist_Fx = []
    cached_A = None
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break
        # Rebuild LP matrix (a has changed)
        cached_A = build_lp_matrix(n, a)
        g = lp_best_response(a, M, cached_A)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        # Step size with backtracking
        if it < 3:
            t = t_base
        else:
            t = max(min_t, t_base / (1 + 0.03 * it))
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        if len(hist_x) >= 4:
            a_anderson = anderson_mix(hist_x, hist_Fx, len(hist_x), extrapolate=True)
            sa = a_anderson.sum()
            if sa > 1e-10:
                a_anderson = a_anderson / sa
            M_and = float(fast_conv(a_anderson).max())
            M_new = float(fast_conv(a_new).max())
            if M_and <= M_new * 1.02:
                a = a_anderson
            else:
                a = a_new
        else:
            a = a_new
        # Periodic random exploration
        if explore_every > 0 and (it + 1) % explore_every == 0:
            if time.monotonic() < deadline - 5:
                noise = np.random.randn(n) * explore_scale
                a_explore = np.maximum(a + noise, 0.0)
                se = a_explore.sum()
                if se > 1e-10:
                    a_explore = a_explore / se
                    M_exp = float(fast_conv(a_explore).max())
                    if M_exp < best_M:
                        best_M = M_exp
                        best_a = a_explore.copy()
                        a = a_explore.copy()
                        no_imp = 0
                        # Clear history after jump
                        hist_x = []
                        hist_Fx = []
    return best_a


def coordinate_descent(a, deadline, n_coords=50, n_rounds=5):
    """Simple coordinate descent: adjust one coordinate at a time."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_M = float(fast_conv(a).max())
    best_a = a.copy()
    rng = np.random.default_rng(12345)
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        for _ in range(n_coords):
            if time.monotonic() > deadline:
                break
            j = rng.integers(0, n)
            # Try small perturbations of coordinate j
            base = a[j]
            for delta in [0.01 * base, -0.01 * base, 0.05 * base, -0.05 * base,
                          0.1 * base, -0.1 * base]:
                a[j] = max(0.0, base + delta)
                s = a.sum()
                if s < 1e-10:
                    a[j] = base
                    continue
                a_norm = a / s
                M = float(fast_conv(a_norm).max())
                if M < best_M - 1e-14:
                    best_M = M
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                else:
                    a[j] = base
                    break
    return best_a


def resample_seq(src, n):
    """Resample a sequence to length n using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.01))
    inits.append(np.ones(n))
    # Power-law shapes
    for alpha in [0.3, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t**alpha + (1 - t)**alpha) / 2, 0.01))
    # Gaussian bumps
    for center, width in [(0.5, 0.1), (0.3, 0.15), (0.7, 0.15), (0.5, 0.3),
                          (0.2, 0.1), (0.8, 0.1)]:
        inits.append(np.exp(-((t - center) / width)**2) + 0.05)
    # Double bump
    inits.append(np.exp(-6 * (t - 0.25)**2) + np.exp(-6 * (t - 0.75)**2) + 0.05)
    # Cosine
    for f in [1, 2, 3]:
        inits.append(0.5 + 0.4 * np.cos(2 * np.pi * f * t))
    # Random smooth
    k = max(3, n // 16)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # Linear ramp
    inits.append(0.1 + 0.9 * t)
    inits.append(0.1 + 0.9 * (1 - t))
    # Triangular
    inits.append(1.0 - np.abs(2 * t - 1))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None or a.size == 0:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        try:
            s = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** NEW BEST: J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def gs_spectral_init(n, rng, n_iters=200):
    """Gerchberg-Saxton style initialization for flat autocorrelation."""
    a = np.maximum(rng.normal(0.5, 0.2, n), 0.01)
    a = a / a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    beta = 0.85
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress magnitude spectrum
        mag_norm = mag / (mag.max() + 1e-30)
        mag_new = (1 - beta) * mag + beta * mag.mean()
        # Also try power compression
        power = 0.3 + 0.3 * beta
        mag_p = (mag ** power)
        mag_p = mag_p / (mag_p.max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_p
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        M = fast_conv(a_new).max()
        if M < best_M:
            best_M = M
            best_a = a_new.copy()
        a = a_new
    return best_a


def propose_candidate(seed=42, budget_s=1500, **kw):
    """Main search function: finds the best sequence minimizing J(a)."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent loaded: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)
        print(f"No parent found, using ones(256), J={best.score:.6f}", flush=True)

    # Lengths to explore
    base_n = parent.size
    lengths_to_try = sorted(set([
        512, 640, 768, 896, 1024, 1152, 1280, 1408, 1536, 1792, 2048
    ]))
    # Filter to reasonable range
    lengths_to_try = [n for n in lengths_to_try if 256 <= n <= 2048]
    print(f"Lengths to explore: {lengths_to_try}", flush=True)

    # =========================================
    # PHASE 1: Fast gradient optimization (250s)
    # =========================================
    p1_deadline = time.monotonic() + 250.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Gradient Optimization (250s)", flush=True)
    print(f"{'='*60}", flush=True)

    best_per_length = {}
    t_per_length = 250.0 / len(lengths_to_try)

    for n in lengths_to_try:
        if time.monotonic() > p1_deadline - 10:
            break
        t0 = time.monotonic()
        sub_dl = min(time.monotonic() + t_per_length, p1_deadline - 5)

        inits = make_initializations(n, rng, best.seq if best.seq is not None else parent)
        # Add GS spectral init
        inits.append(gs_spectral_init(n, rng, n_iters=min(200, int(30 * n / 256))))

        best_n = None
        best_M = float('inf')
        n_iters_est = max(300, int(15000 * 512.0 / n))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > sub_dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 = a0 / a0.sum()
            # Allocate decreasing time to later inits
            remaining = sub_dl - time.monotonic()
            if remaining < 2:
                break
            frac = max(0.2, 1.0 / (1.0 + 0.4 * idx))
            a_sub_dl = time.monotonic() + remaining * frac
            a_opt = nesterov_optimize(a0, a_sub_dl, max_iters=n_iters_est,
                                      tau_start=2.0, tau_end=400.0,
                                      lr_init=0.25, lr_end=0.001,
                                      momentum_start=0.8, momentum_end=0.97)
            M = float(fast_conv(a_opt).max())
            if M < best_M:
                best_M = M
                best_n = a_opt.copy()

        if best_n is not None:
            J = compute_J(best_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M:.8f} J={J:.6f} ({elapsed:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1 complete")

    # =========================================
    # PHASE 2: LP Mirror Descent (800s)
    # =========================================
    p2_start = time.monotonic()
    p2_deadline = p2_start + 800.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (800s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        print(f"  Starting LP: n={n_best}", flush=True)

        # Estimate LP time
        t_lp_test = time.monotonic()
        b_test = fast_conv(best.seq)
        M_test = float(b_test.max())
        A_test = build_lp_matrix(n_best, best.seq)
        g_test = lp_best_response(best.seq, M_test, A_test)
        lp_time = time.monotonic() - t_lp_test
        print(f"  LP call time: {lp_time*1000:.1f}ms", flush=True)

        p2_dp = p2_deadline - time.monotonic()
        if p2_dp > 30:
            # First LP pass with larger steps
            a_lp1 = lp_mirror(best.seq, min(time.monotonic() + p2_dp * 0.5, p2_deadline - 60),
                              max_iters=2000, anderson_window=10,
                              t_base=0.35, min_t=0.005,
                              explore_every=30, explore_scale=0.04)
            best.update(a_lp1)

            # Second LP pass with smaller steps
            remaining = p2_deadline - time.monotonic()
            if remaining > 30:
                a_lp2 = lp_mirror(a_lp1, time.monotonic() + remaining * 0.7,
                                  max_iters=2000, anderson_window=12,
                                  t_base=0.15, min_t=0.002,
                                  explore_every=40, explore_scale=0.02)
                best.update(a_lp2)

            # Third pass: very fine
            remaining = p2_deadline - time.monotonic()
            if remaining > 20:
                a_lp3 = lp_mirror(a_lp2, time.monotonic() + remaining * 0.8,
                                  max_iters=2000, anderson_window=12,
                                  t_base=0.08, min_t=0.001,
                                  explore_every=50, explore_scale=0.01)
                best.update(a_lp3)

    best.report("Phase 2 complete")

    # =========================================
    # PHASE 3: Length exploration (300s)
    # =========================================
    p3_start = time.monotonic()
    p3_deadline = p3_start + 300.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Length Exploration (300s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_cur = best.seq.size
        # Try nearby lengths
        delta_options = [128, 256, -128, -256, 512, 64, -64]
        for delta in delta_options:
            if time.monotonic() > p3_deadline - 30:
                break
            n_try = n_cur + delta
            if n_try < 256 or n_try > 2048 or n_try == n_cur:
                continue
            t0 = time.monotonic()
            a_res = resample_seq(best.seq, n_try)
            a_res = np.maximum(a_res, 1e-8)
            a_res = a_res / a_res.sum()
            # Quick gradient optimization
            a_grad = nesterov_optimize(a_res, min(time.monotonic() + 30, p3_deadline - 25),
                                       max_iters=1000, tau_start=5, tau_end=200,
                                       lr_init=0.15, lr_end=0.005,
                                       momentum_start=0.85, momentum_end=0.96)
            best.update(a_grad)
            # LP polish
            remaining = p3_deadline - time.monotonic()
            if remaining > 30:
                a_lpn = lp_mirror(a_grad, time.monotonic() + min(remaining * 0.6, 120),
                                  max_iters=500, anderson_window=8,
                                  t_base=0.2, min_t=0.005,
                                  explore_every=30, explore_scale=0.03)
                best.update(a_lpn)
            print(f"  n={n_try}: J={best.score:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Also try +512 and +1024 from parent if not tried
    if parent is not None:
        for delta in [512, 1024]:
            if time.monotonic() > p3_deadline - 20:
                break
            n_try = parent.size + delta
            if n_try > 2048 or n_try < 256:
                continue
            if n_try in [x + best.seq.size for x in [128, 256, -128, -256, 512, 64, -64]]:
                continue
            a_res = resample_seq(best.seq, n_try)
            a_res = np.maximum(a_res, 1e-8)
            a_res = a_res / a_res.sum()
            a_grad = nesterov_optimize(a_res, min(time.monotonic() + 40, p3_deadline - 15),
                                       max_iters=1500, tau_start=3, tau_end=300,
                                       lr_init=0.2, lr_end=0.003,
                                       momentum_start=0.82, momentum_end=0.97)
            best.update(a_grad)
            remaining = p3_deadline - time.monotonic()
            if remaining > 30:
                a_lpn = lp_mirror(a_grad, time.monotonic() + min(remaining * 0.7, 150),
                                  max_iters=500, anderson_window=8,
                                  t_base=0.25, min_t=0.005,
                                  explore_every=30, explore_scale=0.03)
                best.update(a_lpn)
            print(f"  n={n_try} (from parent+{delta}): J={best.score:.6f}", flush=True)

    best.report("Phase 3 complete")

    # =========================================
    # PHASE 4: Coordinate descent + fine polish (100s)
    # =========================================
    p4_deadline = time.monotonic() + 100.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate Descent + Fine Polish (100s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        # Coordinate descent
        a_cd = coordinate_descent(best.seq, min(time.monotonic() + 40, p4_deadline - 20),
                                  n_coords=100, n_rounds=10)
        best.update(a_cd)

        # Very fine LP
        remaining = p4_deadline - time.monotonic()
        if remaining > 15:
            a_fine = lp_mirror(a_cd, time.monotonic() + remaining * 0.8,
                               max_iters=1000, anderson_window=12,
                               t_base=0.05, min_t=0.001,
                               explore_every=60, explore_scale=0.005)
            best.update(a_fine)

    best.report("Phase 4 complete")

    # =========================================
    # PHASE 5: Final exploration with random restarts
    # =========================================
    if time.monotonic() < deadline - 10 and best.seq is not None:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 5: Random Restart Exploration", flush=True)
        print(f"{'='*60}", flush=True)

        n_best = best.seq.size
        for restart in range(5):
            if time.monotonic() > deadline - 10:
                break
            # Perturb the best solution
            scale = 0.1 * (1 - restart * 0.15)
            a_pert = np.maximum(best.seq + rng.normal(0, scale, n_best), 0.01)
            a_pert = a_pert / a_pert.sum()
            # Quick optimization
            sub_dl = min(time.monotonic() + 20, deadline - 10)
            a_opt = nesterov_optimize(a_pert, sub_dl, max_iters=500,
                                      tau_start=10, tau_end=200,
                                      lr_init=0.1, lr_end=0.005,
                                      momentum_start=0.9, momentum_end=0.98)
            best.update(a_opt)

    best.report("Phase 5 complete")

    # Final verification
    if best.seq is not None:
        final_J = compute_J(best.seq)
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL RESULT: J={best.score:.8f} (computed J={final_J:.8f}) n={best.seq.size}", flush=True)
        print(f"Total time: {time.monotonic() - best.t0:.0f}s", flush=True)
        print(f"{'='*60}", flush=True)
    else:
        # Fallback
        best.seq = np.ones(256)
        best.score = compute_J(best.seq)
        print(f"Fallback: J={best.score:.6f}", flush=True)

    return [float(x) for x in best.seq]
