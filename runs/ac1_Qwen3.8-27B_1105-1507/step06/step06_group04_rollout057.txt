"""
Advanced peak-convolution minimizer combining:
1. Log-sum-exp smoothed L-BFGS with temperature continuation
2. p-norm continuation for smooth-to-sharp transition
3. Active-set LP alternating (reduced constraint LPs for speed)
4. Full LP alternating for precision
5. Structured peak-targeting perturbations
6. Multi-start diverse search with length exploration
Targets improvement below 1.5027 from parent's 1.506547.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def softmax(x):
    """Numerically stable softmax."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def cross_corr_fft(a, w, n):
    """Compute cross[i] = sum_j a[j]*w[i+j] for i=0..n-1 using FFT.
    w should have length >= 2n-1 (zero-padded for the rest)."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    w_ext = np.zeros(size)
    w_ext[:min(len(w), size)] = w[:min(len(w), size)]
    Af = rfft(a, size)
    Wf = rfft(w_ext, size)
    return irfft(Af * np.conj(Wf), size)[:n]


def lse_smooth_obj_grad(x, T, n):
    """Log-sum-exp smoothed objective and gradient with softmax parameterization."""
    a = softmax(x)
    conv = compute_conv(a)
    Tc = T * conv
    max_Tc = np.max(Tc)
    exp_Tc = np.exp(Tc - max_Tc)
    sum_exp = np.sum(exp_Tc)
    if sum_exp < 1e-300:
        idx = int(np.argmax(Tc))
        exp_Tc = np.zeros(len(Tc))
        exp_Tc[idx] = 1.0
        sum_exp = 1.0
    M_T = (max_Tc + np.log(sum_exp)) / T
    obj = 2.0 * n * M_T

    p = exp_Tc / sum_exp
    cross = cross_corr_fft(a, p, n)
    grad_a = 2.0 * n * (cross + a)
    dot_ga = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot_ga)
    return obj, grad_x


def pnorm_obj_grad(x, p, n):
    """p-norm objective and gradient with softmax parameterization."""
    a = softmax(x)
    conv = compute_conv(a)
    max_c = np.max(conv)
    if max_c < 1e-300:
        return 2.0 * n * max_c, np.zeros_like(x)
    cn = conv / max_c
    cn_p = cn ** p
    sum_cp = np.sum(cn_p)
    if sum_cp < 1e-300:
        idx = int(np.argmax(cn))
        w_cn = np.zeros(len(cn))
        w_cn[idx] = 1.0
        Cn_p = 1.0
    else:
        Cn_p = sum_cp ** (1.0 / p)
        w_cn = cn ** (p - 1)
        sw = np.sum(w_cn)
        if sw < 1e-300:
            idx = int(np.argmax(cn))
            w_cn = np.zeros(len(cn))
            w_cn[idx] = 1.0
            sw = 1.0

    obj = 2.0 * n * max_c * Cn_p
    cross_cn = cross_corr_fft(a, w_cn, n)
    sum_w_cn = np.sum(w_cn)
    grad_a = 2.0 * n / (Cn_p ** (p - 1)) * (cross_cn + a * sum_w_cn)
    dot_ga = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot_ga)
    return obj, grad_x


def optimize_smoothed(a_init, method='lse', T_start=5.0, T_end=500.0,
                      num_temps=8, max_iters=200, time_budget=60.0):
    """Optimize using smoothed objective with temperature/norm continuation."""
    n = len(a_init)
    a = normalize_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if method == 'lse':
        temps = np.geomspace(T_start, T_end, num_temps)
    else:
        temps = np.geomspace(2.0, 64.0, num_temps)

    x = np.log(a + 1e-15)

    for ti, T in enumerate(temps):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        iters = max(20, int(max_iters * remaining / time_budget))

        def fun(x_in):
            if method == 'lse':
                return lse_smooth_obj_grad(x_in, T, n)
            else:
                return pnorm_obj_grad(x_in, T, n)

        try:
            res = minimize(fun, x, jac=True, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-12,
                                    'gtol': 1e-8, 'maxfun': iters * 3})
            if res.success or res.fun < np.inf:
                a_try = softmax(res.x)
                a_try = normalize_simplex(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    x = res.x.copy()
        except Exception:
            pass

    return best_a, best_obj


def build_toeplitz_convolve_mat(fixed_seq):
    """Build matrix A where (A @ g)[k] = (convolve(fixed_seq, g))[k]."""
    n = len(fixed_seq)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = fixed_seq[:hi - lo]
    return A


def active_set_lp(fixed_seq, num_active=10, time_limit=10):
    """
    Solve min t s.t. (fixed_seq * g)[k] <= t for active k, sum(g)=1, g>=0.
    Uses only the top num_active convolution positions as constraints.
    """
    n = len(fixed_seq)
    conv = compute_conv(fixed_seq)
    # Get top positions
    top_idx = np.argsort(conv)[-num_active:]
    top_set = set(top_idx.tolist())

    A_full = build_toeplitz_convolve_mat(fixed_seq)
    # Select active rows
    rows = [k for k in range(2 * n - 1) if k in top_set]
    A_act = A_full[rows, :]

    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((len(rows), n + 1))
    A_ub[:, :n] = A_act
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(rows))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None

def subgradient_active(a_init, max_iters=500, time_budget=60.0,
                       num_peaks=5, tau_init=0.01, momentum=0.8):
    """Subgradient with active peak tracking and momentum."""
    n = len(a_init)
    a = normalize_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    step = 0.05
    tau = tau_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Soft weights over top peaks
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
            w_sum = 1.0
        weights /= w_sum

        cross = cross_corr_fft(a, weights, n)
        grad = 2.0 * (cross + a)
        grad_centered = grad - np.mean(grad)
        gn = np.linalg.norm(grad_centered)
        if gn < 1e-15:
            break

        v = momentum * v - step * grad_centered / gn
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        step *= 0.998
        tau *= 0.999

    return best_a, best_obj


def structured_perturb(a, num_peaks=5, scale=0.05, seed=0):
    """Perturb by moving mass away from peak-contributing positions."""
    n = len(a)
    rng = np.random.default_rng(seed)
    conv = compute_conv(a)
    top_k = np.argsort(conv)[-num_peaks:]

    # For each peak, find contributing positions
    contrib_score = np.zeros(n)
    for k in top_k:
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                contrib_score[j] += a[j] * a[ki]

    # Reduce mass at high-contribution positions, add to low-contribution
    a_pert = a.copy()
    top_contrib = np.argsort(contrib_score)[-n // 4:]
    bottom_contrib = np.argsort(contrib_score)[:n // 4]

    mass_to_move = a[top_contrib] * scale
    total_mass = np.sum(mass_to_move)
    if total_mass < 1e-15:
        return a_pert

    a_pert[top_contrib] -= mass_to_move
    # Distribute to bottom positions proportionally
    bottom_weights = a[bottom_contrib] + 1e-10
    bottom_weights /= bottom_weights.sum()
    a_pert[bottom_contrib] += total_mass * bottom_weights
    a_pert = np.maximum(a_pert, 0)
    a_pert = normalize_simplex(a_pert)
    return a_pert


def generate_starts(n, num_starts=10, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    starts.append(np.ones(n) / n)

    # Gaussian (various widths)
    for sigma_frac in [0.15, 0.25, 0.4, 0.6]:
        bell = np.exp(-((i - center) ** 2) / (2 * (n * sigma_frac) ** 2))
        starts.append(bell / bell.sum())

    # Triangular
    tri = 1.0 - np.abs(i - center) / center
    tri = np.maximum(tri, 0.01)
    starts.append(tri / tri.sum())

    # V-shaped (dip in middle)
    v_shape = np.abs(i - center) / center + 0.1
    starts.append(v_shape / v_shape.sum())

    # Cosine modulation
    for freq in [2, 3, 5]:
        cos_mod = 0.5 + 0.5 * np.cos(2 * np.pi * freq * i / n)
        bell = np.exp(-((i - center) ** 2) / (2 * (n * 0.3) ** 2))
        seq = bell * cos_mod
        seq = np.maximum(seq, 0)
        if seq.sum() > 1e-10:
            starts.append(seq / seq.sum())

    return starts


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                print(f"  LSE smooth: {obj_smooth:.8f} (no improvement)")

            # Try p-norm
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_pn, obj_pn = optimize_smoothed(
                    best_seq, method='pnorm', T_start=2.0, T_end=64.0,
                    num_temps=6, max_iters=150, time_budget=min(remaining * 0.2, 80.0))
                if obj_pn < best_score:
                    best_score = obj_pn
                    best_seq = a_pn.copy()
                    print(f"  p-norm: {best_score:.8f} ** IMPROVED **")
                else:
                    print(f"  p-norm: {obj_pn:.8f} (no improvement)")

    # Phase 2: LP alternating on best
    print(f"\n=== Phase 2: LP alternating (best={best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_lp, obj_lp = lp_alternate(best_seq, max_iters=50,
                                        time_budget=min(remaining * 0.3, 180.0),
                                        use_active=True, num_active=10)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Active LP: {best_score:.8f} ** IMPROVED **")
            else:
                print(f"  Active LP: {obj_lp:.8f} (no improvement)")

            # Full LP for precision
            remaining = deadline - time.monotonic()
            if remaining > 30 and len(best_seq) <= 900:
                a_lpf, obj_lpf = lp_alternate(best_seq, max_iters=30,
                                              time_budget=min(remaining * 0.25, 100.0),
                                              use_active=False)
                if obj_lpf < best_score:
                    best_score = obj_lpf
                    best_seq = a_lpf.copy()
                    print(f"  Full LP: {best_score:.8f} ** IMPROVED **")
                else:
                    print(f"  Full LP: {obj_lpf:.8f} (no improvement)")

    # Phase 3: Structured perturbations + local optimization
    print(f"\n=== Phase 3: Structured perturbations (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed)

        for pert_round in range(12):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            # Generate perturbation
            if pert_round < 4:
                a_pert = structured_perturb(best_seq, num_peaks=5,
                                            scale=0.03 + 0.02 * pert_round,
                                            seed=seed + pert_round * 1000)
            elif pert_round < 8:
                # Random multiplicative noise
                scale = 0.02 + 0.01 * (pert_round - 4)
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
                a_pert = normalize_simplex(a_pert)
            else:
                # Mix with a different structure
                alt_starts = generate_starts(n_best, num_starts=3, seed=seed + pert_round * 777)
                mix_frac = 0.1 + 0.05 * (pert_round - 8)
                a_pert = (1 - mix_frac) * best_seq + mix_frac * alt_starts[pert_round % len(alt_starts)]
                a_pert = normalize_simplex(a_pert)

            # Quick polish
            alloc = min(remaining * 0.15, 40.0)
            a_pol, obj_pol = polish_pipeline(a_pert, time_budget=alloc,
                                             seed=seed + pert_round * 31)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Pert {pert_round}: {best_score:.8f} ** IMPROVED **")
            else:
                print(f"  Pert {pert_round}: {obj_pol:.8f}")

    # Phase 4: Multi-start exploration with different lengths
    print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = []
        for dn in [-100, -50, 50, 100, -200, 200, -150, 150]:
            nv = n_best + dn
            if 200 <= nv <= 1200:
                test_lengths.append(nv)
        test_lengths = sorted(set(test_lengths))

        for n_test in test_lengths:
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            a_res = resample_sequence(best_seq, n_test)
            a_res = normalize_simplex(a_res)
            obj_init = compute_obj(a_res)

            alloc = min(remaining * 0.2, 50.0)
            a_pol, obj_pol = polish_pipeline(a_res, time_budget=alloc,
                                             seed=seed + n_test)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: {best_score:.8f} ** IMPROVED **")
            else:
                print(f"  n={n_test}: {obj_pol:.8f} (init={obj_init:.6f})")

    # Phase 5: Diverse starts at best length (or nearby)
    print(f"\n=== Phase 5: Diverse starts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        if remaining > 40:
            starts = generate_starts(n_best, num_starts=6, seed=seed + 999)
            # Include current best
            starts.append(normalize_simplex(best_seq.copy()))

            for si, s in enumerate(starts):
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                alloc = min(remaining * 0.15, 40.0)
                if alloc < 10:
                    break
                a_pol, obj_pol = polish_pipeline(s, time_budget=alloc,
                                                 seed=seed + si * 555)
                if obj_pol < best_score:
                    best_score = obj_pol
                    best_seq = a_pol.copy()
                    print(f"  Start {si}: {best_score:.8f} ** IMPROVED **")
                else:
                    print(f"  Start {si}: {obj_pol:.8f}")

    # Phase 6: Final intensive polish
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_final, obj_final = polish_pipeline(best_seq, time_budget=min(remaining * 0.7, 200.0),
                                                 seed=seed + 77777)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: {best_score:.8f} ** IMPROVED **")
            else:
                print(f"  Final: {obj_final:.8f}")

            # One more LP round
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_lp2, obj_lp2 = lp_alternate(best_seq, max_iters=30,
                                              time_budget=min(remaining * 0.5, 60.0),
                                              use_active=True, num_active=6)
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  Final LP: {best_score:.8f} ** IMPROVED **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
