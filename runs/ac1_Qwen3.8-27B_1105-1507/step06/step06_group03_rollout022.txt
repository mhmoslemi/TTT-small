import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def project_symmetric_simplex(v):
    """Project onto symmetric simplex: a[i]=a[n-1-i], sum=1, a>=0."""
    n = len(v)
    half = (n + 1) // 2
    # Average with reversed
    sym = 0.5 * (v + v[::-1])
    # Project the first half onto simplex (with appropriate weights)
    # a[0..half-1] are free (and a[n-1..n-half] are determined)
    # sum(a) = 2*sum(a[0..half-1]) - (a[n/2] if n even else 0) = 1
    # This is tricky, let's just project the full thing and then symmetrize
    proj = project_simplex(sym)
    result = 0.5 * (proj + proj[::-1])
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def active_set_sca(a, max_iters, deadline, top_k=50, verbose=False):

    Active-set SCA: Only use the top-k most binding constraints in the LP.
    This makes the LP much faster for large n.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()
        
        # Select top-k active constraints
        if top_k < 2 * n - 1:

            top_indices = np.arange(2 * n - 1)

        cols = []
        vals = []
        b_ub = []

            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)  # t variable at index n
            vals.append(-1.0)
            b_ub.append(c[k])

        A_ub = coo_matrix((vals, (rows, cols)), shape=(len(top_indices), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]

        b_eq = np.array([1.0])

        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=np.array(b_ub), A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:

                if gs > 1e-10:

                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:

                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if found:

                    else:
                        no_improve += 1
                else:

            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 10 == 0:
            print(f"    AS-SCA it {it}: obj={best_obj:.8f}", flush=True)

        if no_improve > 30:

"""
Multi-phase optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Improvements over previous:
1. Active-set SCA (only top-k binding constraints in LP) for dramatic speedup
2. Exact 1D golden-section line search in coordinate descent
3. Symmetric constraint exploitation (halves search space)
4. Spectral shaping for diverse initializations
5. Subgradient method with momentum on the simplex
6. Aggressive multi-restart strategy with perturbation
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def golden_section_1d(f, lo, hi, tol=1e-10, max_iter=60):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return 0.5 * (lo + hi)


def exact_coord_descent(a, deadline, seed=42, max_sweeps=50000, verbose=False):
    """
    Coordinate descent with exact golden-section line search per coordinate.
    Uses incremental convolution updates for speed.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            old = a[i]
            lo_idx = max(0, i)
            hi_idx = min(2 * n - 2, i + n - 1)
            offset = i
            a_slice = a[lo_idx - offset:hi_idx - offset + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            # Define objective as function of new value x
            def obj_of_x(x):
                x = max(0.0, x)
                delta = x - old
                if abs(delta) < 1e-15:
                    return 2.0 * n * conv.max()
                # Preview: update conv
                conv_preview = conv.copy()
                conv_preview[lo_idx:hi_idx + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv_preview[kq] += delta * delta
                new_s = s - old + x
                if new_s < 1e-10:
                    return float('inf')
                return 2.0 * n * conv_preview.max() / (new_s * new_s)

            # Search range: [0, max(2*old, 1/n * 3)]
            search_hi = max(2.0 * old + 1e-6, 3.0 / n)
            # Quick check at boundaries and current
            f_lo = obj_of_x(0.0)
            f_mid = obj_of_x(old)
            f_hi = obj_of_x(search_hi)

            # Find rough minimum bracket
            x_best = old
            f_best = f_mid
            if f_lo < f_best:
                f_best = f_lo
                x_best = 0.0
            if f_hi < f_best:
                f_best = f_hi
                x_best = search_hi

            # Golden section in the most promising region
            if x_best == 0.0:
                search_lo, search_hi_gs = 0.0, min(old * 0.5 + 1e-8, search_hi)
            elif x_best == search_hi:
                search_lo, search_hi_gs = max(old * 1.5 - 1e-8, 0.0), search_hi
            else:
                search_lo, search_hi_gs = max(0.0, old * 0.5 - 1e-8), min(old * 1.5 + 1e-8, search_hi)

            if search_hi_gs > search_lo + 1e-12:
                x_opt = golden_section_1d(obj_of_x, search_lo, search_hi_gs,
                                          tol=1e-12, max_iter=40)
                f_opt = obj_of_x(x_opt)
                if f_opt < f_best - 1e-13:
                    f_best = f_opt
                    x_best = x_opt

            if abs(x_best - old) > 1e-12 and f_best < 2.0 * n * conv.max() - 1e-12:
                delta = x_best - old
                conv[lo_idx:hi_idx + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = x_best
                s = s - old + x_best
                improved = True

        if s > 1e-10:
            current_obj = 2.0 * n * conv.max() / (s * s)
        else:
            current_obj = float('inf')
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            break

        if verbose and sweep % 500 == 0:
            print(f"    ECD sweep {sweep}: obj={best_obj:.8f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def active_set_sca(a, deadline, top_k=100, max_iters=500, verbose=False):
    """
    Active-set SCA: Only use top-k most binding constraints.
    Much faster than full SCA for large n.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Select top-k active constraints (where c is closest to max)
        if top_k < 2 * n - 1:
            threshold = np.sort(c)[::-1][top_k - 1]
            active = np.where(c >= threshold)[0]
        else:
            active = np.arange(2 * n - 1)

        # Build LP: min t
        # s.t. 2*(a*g)[k] - t <= c[k] for k in active
        #      sum(g) = 1, g >= 0
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A_ub = coo_matrix((vals, (rows, cols)), shape=(len(active), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = np.array([1.0])

        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=np.array(b_ub), A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
                        a_try = (1.0 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 5 == 0:
            print(f"    AS-SCA it {it}: obj={best_obj:.8f}", flush=True)

        if no_improve > 40:
            break

    return best_a, best_obj


def subgradient_momentum(a, deadline, lr_init=0.02, momentum=0.9,
                         max_iters=50000, verbose=False):
    """
    Subgradient method with momentum on the simplex.
    Uses exact max (not smooth approximation) with averaging of active lags.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    v = np.zeros(n)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Find active lags (within 5% of max)
        threshold = M * 0.95
        active = np.where(c >= threshold)[0]

        # Compute subgradient: average of gradients of active (a*a)[k]
        # d(a*a)[k]/da[i] = a[k-i] + a[i-k] (with bounds)
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j1 = k - i
                if 0 <= j1 < n:
                    grad[i] += a[j1]
                j2 = i - k
                if 0 <= j2 < n:
                    grad[i] += a[j2]
        grad /= len(active)

        # Project gradient to simplex tangent (remove mean)
        grad = grad - np.mean(grad)

        # Momentum update
        lr = lr_init / (1.0 + 0.5 * np.sqrt(it + 1))
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Restart with perturbation
                rng = np.random.default_rng(it + 42)
                a = best_a + rng.normal(0, 0.01, n)
                a = project_simplex(a)
                v = np.zeros(n)
                no_improve = 0

        a = a_new

        if verbose and it % 500 == 0:
            print(f"    SubGrad it {it}: obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def spectral_init(n, rng, mode=0):
    """Generate initial sequences using spectral shaping ideas."""
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []

    if mode == 0:
        # Uniform
        inits.append(np.ones(n) / n)
    elif mode == 1:
        # Cosine window (raised cosine)
        inits.append(project_simplex(0.5 + 0.5 * np.cos(2 * np.pi * t)))
    elif mode == 2:
        # Sine squared
        inits.append(project_simplex(np.sin(np.pi * t) ** 2 + 0.05))
    elif mode == 3:
        # Gaussian center
        inits.append(project_simplex(np.exp(-((t - 0.5) / 0.25) ** 2) + 0.02))
    elif mode == 4:
        # Two Gaussians
        inits.append(project_simplex(
            np.exp(-((t - 0.3) / 0.15) ** 2) + np.exp(-((t - 0.7) / 0.15) ** 2) + 0.02))
    elif mode == 5:
        # Edge-concentrated
        inits.append(project_simplex(np.exp(-((t - 0.0) / 0.2) ** 2) +
                                     np.exp(-((t - 1.0) / 0.2) ** 2) + 0.02))
    elif mode == 6:
        # Smooth random
        ksize = max(5, n // 10)
        kernel = np.exp(-((np.arange(ksize) - ksize // 2) / (ksize // 4.0)) ** 2)
        kernel /= kernel.sum()
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))
    elif mode == 7:
        # Power spectrum flattening: start with flat spectrum, inverse FFT
        # Take random phases, unit magnitude, IFFT, take abs, normalize
        size = 1
        while size < n:
            size <<= 1
        phases = rng.uniform(0, 2 * np.pi, size // 2 + 1)
        mag = np.ones(size // 2 + 1)
        mag[0] = 0  # Remove DC
        spec = mag * np.exp(1j * phases)
        # Make Hermitian symmetric
        full_spec = np.zeros(size, dtype=complex)
        full_spec[:size // 2 + 1] = spec
        for i in range(1, size // 2):
            full_spec[size - i] = np.conj(spec[i])
        sig = np.fft.ifft(full_spec).real
        sig = np.abs(sig[:n])
        inits.append(project_simplex(sig + 0.01))
    elif mode == 8:
        # Welch-like: orthogonal to low-frequency components
        base = np.ones(n) / n
        # Subtract low-frequency cosine components
        for f in [1, 2, 3]:
            cos_f = np.cos(2 * np.pi * f * t)
            base -= np.mean(base * cos_f) * cos_f
        inits.append(project_simplex(base + 0.1))
    elif mode == 9:
        # Alternating with decay
        alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.3)
        inits.append(project_simplex(alt + 0.1))
    elif mode == 10:
        # Chebyshev-like
        x = 2 * t - 1
        for deg in [2, 4, 6]:
            cheb = np.polynomial.chebyshev.chebval(x, np.eye(deg + 1)[0])
            inits.append(project_simplex(1.0 + 0.3 * cheb + 0.1))
            break

    return inits

        b_eq = np.array([1.0])

        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=np.array(b_ub), A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 12, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    # Symmetrize
                    g = 0.5 * (g + g[::-1])
                    g = np.maximum(g, 0)
                    g /= g.sum()
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2]:
                        a_try = (1.0 - frac) * full + frac * g
                        a_try = 0.5 * (a_try + a_try[::-1])  # enforce symmetry
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            full = a_try
                            best_obj = J_try
                            best_full = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 10 == 0:
            print(f"    SymSCA it {it}: obj={best_obj:.8f}", flush=True)

        if no_improve > 30:
            break

    return best_full, best_obj


def solve_single(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return best_obj

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 1: Active-set SCA (40% of time)
    t1 = time.monotonic() + 0.40 * remaining
    a_sca, obj_sca = active_set_sca(a, deadline=t1, top_k=min(100, n), max_iters=200)
    try_update(a_sca)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 2: Exact coordinate descent (30% of time)
    t2 = time.monotonic() + 0.30 * remaining
    a_cd = exact_coord_descent(a, deadline=t2, seed=42)
    try_update(a_cd)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 3: Symmetric SCA (20% of time)
    t3 = time.monotonic() + 0.20 * remaining
    a_sym, obj_sym = symmetric_sca(a, deadline=t3, top_k=min(60, n // 2))
    try_update(a_sym)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 4: Final CD polish (10% of time)
    t4 = deadline - 5
    a_final = exact_coord_descent(a, deadline=t4, seed=99)
    try_update(a_final)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses active-set SCA, exact coordinate descent, symmetric optimization,
    spectral initializations, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    n_parent = parent.size if parent is not None else 768

    if parent is None:
        parent = np.ones(n_parent) / n_parent
        update_best(parent)

    # === PHASE 1: Multi-start optimization at parent length (45% of time) ===
    phase1_deadline = t_start + 0.45 * (budget_s - 25.0)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-start at n={n_parent} (budget {phase1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Generate diverse initializations
    inits = []
    # Parent-based
    inits.append(parent.copy() / parent.sum())
    p_sym = 0.5 * (parent + parent[::-1])
    p_sym = np.maximum(p_sym, 0)
    if p_sym.sum() > 1e-10:
        inits.append(p_sym / p_sym.sum())
    # Perturbed parent
    for sigma in [0.02, 0.05, 0.10, 0.20]:
        pert = parent / parent.sum() + rng.normal(0, sigma, n_parent)
        pert = project_simplex(pert)
        inits.append(pert)
    # Spectral/structured inits
    for mode in range(11):
        inits.extend(spectral_init(n_parent, rng, mode))

    # Remove duplicates (by similarity)
    inits_dedup = [inits[0]]
    for idx in range(1, len(inits)):
        is_dup = False
        for ref in inits_dedup:
            cos_sim = np.dot(inits[idx], ref) / (np.linalg.norm(inits[idx]) * np.linalg.norm(ref) + 1e-15)
            if cos_sim > 0.999:
                is_dup = True
                break
        if not is_dup:
            inits_dedup.append(inits[idx])
    inits = inits_dedup

    # Limit to top 15 most diverse
    inits = inits[:15]

    n_init = len(inits)
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, n_init)

    for idx in range(n_init):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 20:
            break
        alloc = min(per_init_time, remaining_check - 5)
        if alloc < 10:
            break

        a0 = inits[idx].copy()
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        t_init_start = time.monotonic()
        print(f"  Init {idx}/{n_init-1} (budget {alloc:.0f}s)...", end=" ", flush=True)
        a_opt, obj_opt = solve_single(a0, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Multi-length exploration (30% of time) ===
    phase2_deadline = t_start + 0.75 * (budget_s - 25.0)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration (budget {phase2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidate_ns = [384, 512, 640, 896, 1024, 1280]
    candidate_ns = [n for n in candidate_ns if n != n_parent]
    candidate_ns.sort(key=lambda n: abs(n - n_parent))

    for new_n in candidate_ns[:4]:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 50:
            break
        alloc = min(120.0, remaining_check - 20)
        if alloc < 30:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", end=" ", flush=True)
        a_ext = resample_seq(best_seq, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = solve_single(a_ext, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive final optimization (25% of time) ===
    phase3_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive final (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            a_curr = best_seq.copy()

            # Pass 1: Long active-set SCA
            t_p1 = time.monotonic() + 0.30 * remaining
            print(f"  AS-SCA intensive (budget {t_p1 - time.monotonic():.0f}s)...", end=" ", flush=True)
            a_sca, obj_sca = active_set_sca(a_curr, deadline=t_p1, top_k=150, max_iters=500)
            update_best(a_sca)
            print(f"J={obj_sca:.8f}", flush=True)
            a_curr = best_seq.copy()

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass 2: Exact CD
                t_p2 = time.monotonic() + 0.30 * remaining
                print(f"  Exact CD (budget {t_p2 - time.monotonic():.0f}s)...", end=" ", flush=True)
                a_cd = exact_coord_descent(a_curr, deadline=t_p2, seed=777)
                update_best(a_cd)
                print(f"J={compute_J(best_seq):.8f}", flush=True)
                a_curr = best_seq.copy()

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass 3: Symmetric SCA
                t_p3 = time.monotonic() + 0.25 * remaining
                print(f"  Symmetric SCA (budget {t_p3 - time.monotonic():.0f}s)...", end=" ", flush=True)
                a_sym, obj_sym = symmetric_sca(a_curr, deadline=t_p3, top_k=80)
                update_best(a_sym)
                print(f"J={obj_sym:.8f}", flush=True)
                a_curr = best_seq.copy()

            remaining = phase3_deadline - time.monotonic()
            if remaining > 20:
                # Pass 4: Final exact CD polish
                t_p4 = phase3_deadline - 5
                print(f"  Final CD polish (budget {t_p4 - time.monotonic():.0f}s)...", end=" ", flush=True)
                a_final = exact_coord_descent(a_curr, deadline=t_p4, seed=1234)
                update_best(a_final)
                print(f"J={compute_J(best_seq):.8f}", flush=True)

            # Pass 5: Subgradient with momentum (if time remains)
            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                t_p5 = phase3_deadline - 5
                print(f"  SubGrad momentum (budget {t_p5 - time.monotonic():.0f}s)...", end=" ", flush=True)
                a_sg, obj_sg = subgradient_momentum(a_curr, deadline=t_p5, lr_init=0.03, momentum=0.9)
                update_best(a_sg)
                print(f"J={obj_sg:.8f}", flush=True)

                # One final CD
                if time.monotonic() < phase3_deadline - 10:
                    a_last = exact_coord_descent(best_seq, deadline=phase3_deadline - 5, seed=555)
                    update_best(a_last)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
