"""
Iterative peak-reduction optimizer with exact line search, adaptive
temperature annealing, FFT-accelerated subgradients, and coordinate
descent polishing. Designed to beat 1.506547 (n=700) and approach
the 1.5027 target.
"""

import numpy as np
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    b = compute_conv(a)
    return 2.0 * n * np.max(b) / (s * s)


def normalize_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient(a, b, beta):
    """
    Compute subgradient of smoothed max via softmax weights.
    grad[j] = 2 * sum_k w[k] * a[k-j] (with bounds)
    Computed via FFT.
    """
    n = len(a)
    conv_len = 2 * n - 1
    size = 1
    while size < conv_len:
        size <<= 1
    
    # Softmax weights
    b_shifted = b - np.max(b)
    b_scaled = beta * b_shifted
    b_scaled = np.clip(b_scaled, -500, 500)
    w = np.exp(b_scaled)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(conv_len)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    
    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid indices
    # = 2 * cross_correlation of a and w
    # ifft(fft(a, N) * conj(fft(w, N)))[j] = sum_m a[m] * w[(m+j) mod N]
    # For linear (N >= 2n-1): = sum_m a[m] * w[m+j]
    # We want sum_k w[k] * a[k-j] = sum_m a[m] * w[m+j] (substituting k=m+j, m=k-j)
    # Yes! That's exactly what we have.
    
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    grad = np.fft.irfft(A * np.conj(W), size)[:n]
    grad *= 2.0
    
    return grad


def line_search(a, grad, f_current, max_evals=30):
    """
    Backtracking + golden section line search.
    Minimize f(a - t * grad) for t >= 0.
    Returns optimal t and the new point.
    """
    n = len(a)
    
    # First, find a step that gives improvement (backtracking)
    t = 1.0
    f_t = f_current
    for _ in range(20):
        a_try = normalize_simplex(a - t * grad)
        f_try = compute_obj(a_try)
        if f_try < f_current - 1e-12:
            f_t = f_try
            break
        t *= 0.5
        if t < 1e-8:
            break
    
    if f_t >= f_current - 1e-12:
        return 0.0, a  # No improvement found
    
    # Golden section to refine
    # We know f(t) < f(0) and f(t) is being minimized
    # Search in [t/2, t*2] (or [0, t] if we haven't checked beyond t)
    lo = 0.0
    hi = t * 2.0
    # Check if hi gives improvement
    a_hi = normalize_simplex(a - hi * grad)
    f_hi = compute_obj(a_hi)
    if f_hi >= f_t:
        hi = t
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    
    a_c = normalize_simplex(a - c * grad)
    f_c = compute_obj(a_c)
    a_d = normalize_simplex(a - d * grad)
    f_d = compute_obj(a_d)
    
    evals = 3
    for _ in range(15):
        if hi - lo < 1e-8 * max(1, t):
            break
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = hi - gr * (hi - lo)
            a_c = normalize_simplex(a - c * grad)
            f_c = compute_obj(a_c)
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + gr * (hi - lo)
            a_d = normalize_simplex(a - d * grad)
            f_d = compute_obj(a_d)
        evals += 1
        if evals >= max_evals:
            break
    
    t_opt = (lo + hi) / 2.0
    a_opt = normalize_simplex(a - t_opt * grad)
    f_opt = compute_obj(a_opt)
    
    # Make sure we got improvement
    if f_opt >= f_current:
        return 0.0, a
    
    return t_opt, a_opt


def coordinate_descent_polish(a, time_budget=30.0):
    """
    Coordinate descent: for each coordinate i, find the optimal value
    that minimizes the objective (keeping others fixed).
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute FFT size
    conv_len = 2 * n - 1
    size = 1
    while size < conv_len:
        size <<= 1
    
    for sweep in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            a_old = a[i]
            # Try a few candidate values
            # The optimal a[i] given others is found by 1D minimization
            # f(a[i]) = 2n * max_k b[k](a[i]) / (sum)^2
            # where b[k](a[i]) = b[k] + (change due to a[i])
            
            # Quick approach: try current value ± perturbations
            # and also try setting to 0
            candidates = [a_old, 0.0, a_old * 0.5, a_old * 1.5, a_old * 2.0]
            # Also try to reduce a[i] if it's contributing to the peak
            best_val = a_old
            best_f = best_obj
            
            for val in candidates:
                a[i] = val
                s = np.sum(a)
                if s < 1e-10:
                    a[i] = a_old

                f_val = compute_obj(a)
                if f_val < best_f:
                    best_f = f_val
                    best_val = val
            
            if best_f < best_obj - 1e-13:
                a[i] = best_val
                best_obj = best_f
                improved = True
            else:
                a[i] = a_old
        
        if not improved:
            break
    
    return normalize_simplex(a), best_obj

            scale = 0.01 + 0.01 * min(stall, 10)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = normalize_simplex(a)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a, best_obj


def aggressive_peak_chase(a, time_budget=30.0, K=1):
    """
    Chase the top K peaks with exact line search.
    Much more aggressive than softmax subgradient.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        b = compute_conv(a)
        
        # Get top K peak lags
        top_k = np.argsort(b)[-K:]
        
        # Equal weight on top K
        w = np.zeros(len(b))
        for k in top_k:
            w[k] = 1.0 / K
        
        # Compute gradient (same as subgradient but with discrete weights)
        conv_len = 2 * n - 1
        size = 1
        while size < conv_len:
            size <<= 1
        
        A = np.fft.rfft(a, size)
        W = np.fft.rfft(w, size)
        grad = np.fft.irfft(A * np.conj(W), size)[:n] * 2.0
        
        # Center
        grad -= np.mean(grad)
        if np.linalg.norm(grad) < 1e-15:
            break
        
        # Line search
        t_opt, a_new = line_search(a, grad, best_obj, max_evals=25)
        
        if t_opt > 0:
            a = a_new
            obj = compute_obj(a)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
            else:
                a = best_a.copy()
        else:
            # Try with different K
            K = K + 1 if K < 5 else 1
            if K == 1:
                break
    
    return best_a, best_obj


def generate_random_start(n, seed=42):
    """Generate a random non-negative sequence."""
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, n)
    return a / np.sum(a)


def generate_parametric_start(n, seed=42):
    """Generate a parametric start: bell with sinusoidal modulation."""
    rng = np.random.default_rng(seed)
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n * (0.2 + 0.3 * rng.random())
    bell = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
    freq = int(1 + rng.integers(0, 6))
    amp = 0.3 * rng.random()
    phase = 2 * np.pi * rng.random()
    mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
    a = bell * mod
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def try_load_parent():
    """Try to load the parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except (NameError, Exception):
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses aggressive peak chasing with exact line search,
    adaptive temperature subgradient, and coordinate descent polishing to
    minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
    
    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n={n_parent}")
    
    # Phase 1: Optimize the parent with aggressive peak chasing
    print("\n=== Phase 1: Aggressive peak chasing on parent ===")
    if best_seq is not None:
        t_alloc = min(200.0, (deadline - time.monotonic()) * 0.3)
        a_opt, obj_opt = aggressive_peak_chase(best_seq, time_budget=t_alloc, K=1)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Peak chase: {best_score:.8f} ** IMPROVED **")
        else:
            print(f"  Peak chase: {obj_opt:.8f} (no improvement)")
        
        # Try with K=2,3
        for K in [2, 3]:
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(100.0, (deadline - time.monotonic()) * 0.2)
            a_opt, obj_opt = aggressive_peak_chase(best_seq, time_budget=t_alloc, K=K)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Peak chase K={K}: {best_score:.8f} ** IMPROVED **")
    
    # Phase 2: Software subgradient with line search
    print("\n=== Phase 2: Softmax subgradient with line search ===")
    if best_seq is not None:
        t_alloc = min(200.0, (deadline - time.monotonic()) * 0.3)
        a_opt, obj_opt = softmax_subgradient_optimize(
            best_seq, time_budget=t_alloc, beta_init=30.0, beta_end=500.0)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Subgradient: {best_score:.8f} ** IMPROVED **")
        else:
            print(f"  Subgradient: {obj_opt:.8f} (no improvement)")
    
    # Phase 3: Try different n values
    print(f"\n=== Phase 3: Length exploration (best: {best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = sorted(set([
            n_best - 100, n_best - 50, n_best + 50, n_best + 100,
            n_best + 200, n_best - 200
        ]))
        test_lengths = [x for x in test_lengths if 400 <= x <= 1500]
        
        for n_test in test_lengths:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            
            # Resample
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_test)
            a_resample = np.interp(x_new, x_old, best_seq)
            a_resample = normalize_simplex(a_resample)
            
            t_alloc = min(60.0, remaining * 0.3)
            a_opt, obj_opt = aggressive_peak_chase(a_resample, time_budget=t_alloc, K=1)
            
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_test}: {best_score:.8f} ** IMPROVED **")
            else:
                print(f"  n={n_test}: {obj_opt:.8f}")
    
    # Phase 4: Multi-start from different initializations
    print(f"\n=== Phase 4: Multi-start exploration (best: {best_score:.8f}) ===")
    n_use = len(best_seq) if best_seq is not None else 700
    
    starts = []
    # Random starts
    for s in range(5):
        starts.append(generate_random_start(n_use, seed=seed + 1000 + s))
    # Parametric starts
    for s in range(5):
        starts.append(generate_parametric_start(n_use, seed=seed + 2000 + s))
    # If we have a best, include it
    if best_seq is not None:
        starts.insert(0, normalize_simplex(best_seq.copy()))
    
    for si, s in enumerate(starts):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        
        t_alloc = min(30.0, remaining * 0.4)
        a_opt, obj_opt = softmax_subgradient_optimize(
            s, time_budget=t_alloc * 0.6, beta_init=20.0, beta_end=200.0)
        
        if obj_opt < best_score - 0.001:
            # Promising, do more
            t_alloc2 = min(30.0, remaining * 0.3)
            a_opt2, obj_opt2 = aggressive_peak_chase(
                a_opt, time_budget=t_alloc2, K=1)
            if obj_opt2 < best_score:
                best_score = obj_opt2
                best_seq = a_opt2.copy()
                print(f"  Start {si}: {best_score:.8f} ** IMPROVED **")
        elif obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Start {si}: {best_score:.8f} ** IMPROVED **")
    
    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final polish (best: {best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # More aggressive peak chasing
            t_alloc = min(remaining * 0.4, 120.0)
            a_opt, obj_opt = aggressive_peak_chase(best_seq, time_budget=t_alloc, K=1)

                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  After polish: {best_score:.8f}")

            if remaining > 15:

                a_opt, obj_opt = softmax_subgradient_optimize(
                    best_seq, time_budget=t_alloc, beta_init=100.0, beta_end=1000.0)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  After subgradient: {best_score:.8f}")
            
            # Coordinate descent polish
            remaining = deadline - time.monotonic()
            if remaining > 10:

                    best_score = obj_opt

                    print(f"  After coord descent: {best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
