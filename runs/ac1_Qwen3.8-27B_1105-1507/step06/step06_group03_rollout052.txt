def fast_conv(a):

    while size < 2*n - 1:
        size <<= 1

    if s < 0.01:

    b = fast_conv(a)
    return float(2*len(a)*b.max()/(s*s))

    if n == 0:
        return v

    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n)/n
    lam = cssv[k[-1]]/(k[-1]+1)
    return np.maximum(v - lam, 0.0)
