Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines enhanced LP alternating, Nesterov-accelerated mirror descent on the simplex,
peak-targeted coordinate descent, and multi-n exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz_a(a):
    """Build matrix A where A @ g = conv(a, g). Shape: (2n-1, n)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end - j]
    return A

def lp_alt_step(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Return normalized g or None."""
    n = len(a)
    A = build_toeplitz_a(a)

            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))

    return None

def peak_shaving(a, max_steps=50, time_budget=30.0):
    """
    Targeted peak shaving: identify the top peaks in the convolution
    and try to reduce them by adjusting contributing elements.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv_fft(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]

        # Find top contributors to this peak
        # (a*a)[k] = sum_j a[j] * a[k-j]
        contribs = []
        for j in range(n):
            k2 = k_peak - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)

        improved = False
        for cval, j1, j2 in contribs[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            # Try to reduce this pair's contribution
            # Move mass from the larger element to a random position
            if a[j1] >= a[j2]:
                src = j1
                other = j2
            else:
                src = j2
                other = j1

            for _ in range(4):
                tgt = int(rng.integers(0, n))
                if tgt == src or tgt == other:
                    continue

                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[src]
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break

        if not improved and rng.random() < 0.05:
            a = normalize(best_a * np.exp(rng.normal(0, 0.02, n)))

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)
