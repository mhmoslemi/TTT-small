import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a * a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def grad_logsumexp_conv(a, tau, size_cache=None):
    """
    Compute gradient of (1/tau)*log(sum(exp(tau*c[k]))) where c = conv(a,a).
    Returns gradient vector of length n.

    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = tau * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
        w_sum = 1.0
    else:

    # Efficiently: convolve weights with a (cross-correlation)
    size = 1
    while size < 3 * n:

    w_ext = np.zeros(2 * n - 1)
    w_ext[:] = weights

    Af = rfft(a_flip, size)
    corr = irfft(Wf * Af, size)

    n = len(a)
    if n == 0:
        return a
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0)


def fista_smoothed(a_init, tau_start=5.0, tau_end=50.0, max_iters=500,
                   time_budget=60.0, print_info=False):
    """
    FISTA with adaptive log-sum-exp smoothing for minimizing 2*n*max(conv(a,a))/S^2.
    Works on normalized (sum=1) sequences.
    """
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    z = a.copy()
    t = 1.0
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adaptive step size (Lipschitz estimate)
    step = 1.0
    n_active = max(n // 10, 10)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive tau: start smooth, get sharper
        progress = it / float(max_iters)
        tau = tau_start * (tau_end / tau_start) ** progress

        # Compute gradient at z
        grad = grad_logsumexp_conv(z, tau)

        # Center gradient (remove mean to stay on sum=1 manifold approximately)
        grad = grad - np.mean(grad)

        # Backtracking line search
        a_new = z - step * grad
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)

        # If objective increased too much, reduce step
        if obj_new > best_obj * 1.5:
            step *= 0.5
            a_new = z - step * grad
            a_new = project_simplex(a_new)

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z_new = a_new + ((t - 1.0) / t_new) * (a_new - a)
        z_new = project_simplex(z_new)

        a = a_new
        z = z_new
        t = t_new

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            step = min(step * 1.05, 2.0)  # cautious increase
        else:
            # Try smaller step
            if step > 0.01:
                step *= 0.8

        if print_info and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    FISTA it={it}, tau={tau:.1f}, step={step:.4f}, obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a, best_obj


def spectral_flatten(a, n_iters=30, time_budget=20.0):
    """
    POCS-style spectral flattening: flatten |FFT(a)|^2 while maintaining a>=0.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Progressive flattening
        alpha = 0.3 + 0.4 * (it / float(n_iters))
        P_norm = P / p_max
        P_new = np.power(P_norm, alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.6 * a + 0.4 * a_new)

    return best_a, best_obj


def lp_refine(a, M_mult=1.0, time_budget=30.0):
    """
    LP refinement: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Then blend and re-optimize.
    """
    n = len(a)
    if n > 800:
        return a, compute_obj(a)

    a = normalize(a.copy().astype(np.float64))
    conv = compute_conv(a)
    M = float(np.max(conv)) * M_mult

    # Build Toeplitz matrix for conv(a, g)
    m = 2 * n - 1
    A_mat = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]

    c_vec = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_vec, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_budget, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s_g = float(np.sum(g))
            if s_g > 1e-10:
                g = g / s_g
                best_a = a.copy()
                best_obj = compute_obj(a)
                # Try multiple blend ratios
                for t in [0.3, 0.5, 0.7, 1.0]:
                    a_try = normalize((1 - t) * a + t * g)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try
                return best_a, best_obj
    except Exception:
        pass
    return a, compute_obj(a)


def coord_descent(a, time_budget=15.0):
    """
    Targeted coordinate descent: move mass from peak contributors to reduce max conv.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    stagnation = 0
    while time.monotonic() - start < time_budget:
        if stagnation > 50:
            break
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to peak
        contrib = []
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        for j in range(j_min, j_max + 1):
            k2 = k_star - j
            contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget - 1:
                break
            if j1 == j2:
                continue
            # Try moving mass from j1 to random distant position
            for _ in range(2):
                j_target = int(rng.integers(0, n))
                if j_target in (j1, j2):
                    continue
                for frac in [0.1, 0.3]:
                    delta = frac * a[j1]
                    a_try = a.copy()
                    a_try[j1] = max(0, a_try[j1] - delta)
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = opt_try if False else compute_obj(a_try)
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        stagnation = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stagnation += 1
            # Try random perturbation
            if rng.random() < 0.3:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def proposal_run(a_init, n, time_budget=120.0):
    """
    Run full optimization pipeline on a sequence of length n.
    Returns (best_sequence, best_obj).
    """
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: FISTA with graduated smoothing
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t1 = min(remaining * 0.4, 45.0)
        a_new, obj_new = fista_smoothed(a, tau_start=3.0, tau_end=30.0,
                                        max_iters=200, time_budget=t1)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new
        else:
            a_new, obj_new = fista_smoothed(a, tau_start=8.0, tau_end=60.0,
                                            max_iters=300, time_budget=t1)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new

    # Phase 2: Spectral flattening
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_new, obj_new = spectral_flatten(best_a, n_iters=25,
                                          time_budget=min(remaining * 0.2, 20.0))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    # Phase 3: FISTA again (sharper)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t3 = min(remaining * 0.3, 30.0)
        a_new, obj_new = fista_smoothed(best_a, tau_start=20.0, tau_end=100.0,
                                        max_iters=200, time_budget=t3)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    # Phase 4: LP refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15 and n <= 800:
        a_new, obj_new = lp_refine(best_a, M_mult=1.0,
                                   time_budget=min(remaining * 0.3, 25.0))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    # Phase 5: Coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_new, obj_new = coord_descent(best_a, time_budget=min(remaining * 0.4, 12.0))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    # Phase 6: Final FISTA polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_new, obj_new = fista_smoothed(best_a, tau_start=50.0, tau_end=200.0,
                                        max_iters=150, time_budget=min(remaining * 0.5, 15.0))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FISTA with adaptive smoothing, spectral equalization, LP refinement,
    coordinate descent, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf
    best_n = 700

    # --- Load parent sequence ---
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            if parent_obj < np.inf:
                best_seq = parent.copy()
                best_score = parent_obj
                best_n = len(parent)
                print(f"Parent loaded: n={best_n}, obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        # No parent: use uniform initialization
        best_n = 700
        best_seq = np.ones(best_n) / best_n
        best_score = compute_obj(best_seq)
        print(f"No parent, using uniform init: n={best_n}, obj={best_score:.8f}")

    rng = np.random.default_rng(seed)
    a_normalized = normalize(best_seq.copy())

    # --- Main Optimization at parent n ---
    print(f"\n{'='*60}")
    print(f"Main optimization: n={best_n}")
    print(f"{'='*60}")

    # Run 1: Standard pipeline
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_budget = min(remaining * 0.25, 180.0)
        a1, obj1 = proposal_run(a_normalized, best_n, time_budget=t_budget)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Run 1: {best_score:.8f}")
        else:
            print(f"  Run 1: {obj1:.8f} (no improvement)")

    # Run 2: Perturbed start
    remaining = deadline - time.monotonic()
    if remaining > 30:
        pert = normalize(best_seq * np.exp(rng.normal(0, 0.05, best_n)))
        t_budget = min(remaining * 0.2, 150.0)
        a2, obj2 = proposal_run(pert, best_n, time_budget=t_budget)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Run 2 (perturbed): {best_score:.8f}")
        else:
            print(f"  Run 2 (perturbed): {obj2:.8f} (no improvement)")

    # Run 3: Different smoothing schedule
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a3, obj3 = fista_smoothed(best_seq, tau_start=1.0, tau_end=50.0,
                                  max_iters=400, time_budget=min(remaining * 0.15, 90.0))
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Run 3 (FISTA long): {best_score:.8f}")
        else:
            print(f"  Run 3 (FISTA long): {obj3:.8f} (no improvement)")

    # --- Multi-n exploration ---
    print(f"\n{'='*60}")
    print(f"Multi-n exploration around n={best_n}")
    print(f"{'='*60}")

    n_candidates = []
    for delta in [-100, -75, -50, -25, 25, 50, 75, 100, 150, 200]:
        n_try = best_n + delta
        if 200 <= n_try <= 1500:
            n_candidates.append(n_try)
    # Remove duplicates and sort
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_try = normalize(resample_sequence(best_seq, n_try))
        t_budget = min(remaining * 0.12, 60.0)
        a_res, obj_res = proposal_run(a_try, n_try, time_budget=t_budget)

        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            best_n = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_res:.8f}")

    # --- Final intensive polish at best n ---
    print(f"\n{'='*60}")
    print(f"Final polish: n={best_n}")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Multiple short FISTA rounds with different parameters
        for round_i in range(4):
            if time.monotonic() > deadline - 10:
                break
            t_alloc = min(remaining * 0.15, 40.0)
            tau_s = [5, 15, 30, 80][round_i % 4]
            tau_e = [30, 60, 120, 300][round_i % 4]
            a_pol, obj_pol = fista_smoothed(best_seq, tau_start=tau_s, tau_end=tau_e,
                                            max_iters=150, time_budget=t_alloc)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Polish {round_i}: {best_score:.8f}")
            else:
                print(f"  Polish {round_i}: {obj_pol:.8f} (no improvement)")
                # Try spectral
                if time.monotonic() < deadline - 15:
                    a_sp, obj_sp = spectral_flatten(best_seq, n_iters=20, time_budget=min(t_alloc*0.5, 15.0))
                    if obj_sp < best_score:
                        best_score = obj_sp
                        best_seq = a_sp.copy()
                        print(f"  Polish {round_i} spectral: {best_score:.8f}")

        # Coordinate descent
        if time.monotonic() < deadline - 10:
            a_cd, obj_cd = coord_descent(best_seq, time_budget=min(15.0, (deadline - time.monotonic()) * 0.5))
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Coord descent: {best_score:.8f}")

        # LP final
        if time.monotonic() < deadline - 10 and best_n <= 800:
            a_lp, obj_lp = lp_refine(best_seq, M_mult=1.0,
                                     time_budget=min(20.0, (deadline - time.monotonic()) * 0.5))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP final: {best_score:.8f}")

    # --- Validate and return ---
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
