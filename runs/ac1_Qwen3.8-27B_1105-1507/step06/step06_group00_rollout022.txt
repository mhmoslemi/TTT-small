Exploits the structure of the autocorrelation minimization problem with multiple escape mechanisms.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g, float(res.x[n])
    except Exception:
        pass

    if g_val is not None:
        return g_val, M_val
    return None, None


def lp_alternate(a, time_budget, verbose=False, max_iters=300):
    """
    LP alternating optimization with adaptive line search and perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Adaptive line search: try geometrically spaced t values
        t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose and it % 10 == 0:
                print(f"    LP alt it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, elapsed={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 15:
                # Perturb to escape
                if stall < 25:
                    scale = 0.02
                elif stall < 35:
                    scale = 0.05
                elif stall < 50:
                    scale = 0.1
                else:
                    scale = 0.15
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_p = float(np.sum(a))
                if s_p > 1e-10:
                    a = a / s_p
                stall = 0

    return best_a


def nesterov_subgrad(a, time_budget, verbose=False, max_iters=2000):
    """
    Nesterov-accelerated subgradient descent on the (non-smooth) peak objective.
    Operates on normalized a (sum=1).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum
    y = a.copy()
    momentum = 0.9
    step_size = 0.01
    decay = 0.999

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        conv = compute_conv(y)
        max_val = float(np.max(conv))
        k_star = int(np.argmax(conv))

        # Subgradient of max_k (a*a)[k] at a
        # (a*a)[k] = sum_i a[i] * a[k-i]
        # d/d a_j of (a*a)[k_star] = a[k_star - j] (if 0 <= k_star-j < n) + a[j-k_star] (if 0 <= j-k_star < n, j != k_star/2)
        grad = np.zeros(n)
        for j in range(n):
            v1 = k_star - j
            v2 = j - k_star
            if 0 <= v1 < n:
                grad[j] += y[v1]
            if 0 <= v2 < n and v2 != v1:
                grad[j] += y[v2]

        # Project gradient onto sum=1 hyperplane
        grad_centered = grad - np.mean(grad)

        # Nesterov update
        t_k = 2.0 / (it + 1 + 2.0)
        y_new = y - step_size * t_k * grad_centered
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new > 1e-10:
            y_new = y_new / s_new
            obj_new = compute_obj(y_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = y_new.copy()
            y = y_new
        step_size *= decay

    return best_a


def peak_shave(a, time_budget, verbose=False):
    """
    Peak shaving: iteratively reduce the highest convolution bin by
    redistributing mass from peak-contributing positions to low-contribution positions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    for it in range(500):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = float(conv[k_star])

        # Find top contributors to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
            else:
                contrib[i] = 0.0

        # Sort by contribution
        top_idx = np.argsort(-contrib)[:max(5, n // 10)]
        bottom_idx = np.argsort(contrib)[:max(5, n // 10)]

        # Try moving small amounts from top to bottom
        improved = False
        for _ in range(20):
            if improved:
                break
            src = top_idx[rng.integers(len(top_idx))]
            dst = bottom_idx[rng.integers(len(bottom_idx))]
            if src == dst or a[src] < 1e-8:
                continue
            for frac in [0.005, 0.01, 0.02, 0.05]:
                delta = frac * a[src]
                a[src] -= delta
                a[dst] += delta
                s_try = float(np.sum(a))
                if s_try > 0:
                    a_try = a / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                a[src] += delta
                a[dst] -= delta

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence with various structures."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # U-shaped (high at edges)
        x = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * (2 * x - 1) ** 2
        a = a / np.sum(a)
    elif mode == 2:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = 1.0 + 1.5 * np.cos(4 * np.pi * x) ** 2
        a = np.maximum(a, 0.1)
        a = a / np.sum(a)
    elif mode == 3:
        # Cosine window variants
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x) ** 2
        a = a / np.sum(a)
    elif mode == 4:
        # Random smooth
        a = np.cumsum(rng.uniform(0, 1, n))
        a = a - a[0] + 0.5
        a = a / np.sum(a)
    else:
        # Pure random
        a = rng.uniform(0.1, 1.0, n)
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating + Nesterov subgradient + peak shaving +
    multi-n exploration + diversified restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    # === Phase 1: Intensive LP alternating on parent ===
    print(f"\n=== Phase 1: LP Alternating (n={len(best_seq)}) ===")
    phase1_end = start_time + (deadline - start_time) * 0.35
    a = best_seq.copy()

    for round_i in range(4):
        remaining = phase1_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * 0.5, 120.0)
        a_new = lp_alternate(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP Round {round_i}: obj={best_score:.8f}")
        else:
            print(f"  LP Round {round_i}: obj={obj:.8f} (no improvement)")

    # Peak shaving
    remaining = phase1_end - time.monotonic()
    if remaining > 15:
        a_ps = peak_shave(a, time_budget=min(remaining * 0.3, 40.0))
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            a = a_ps.copy()
            print(f"  Peak Shave: obj={best_score:.8f}")

    # Nesterov subgradient
    remaining = phase1_end - time.monotonic()
    if remaining > 15:
        a_ns = nesterov_subgrad(a, time_budget=min(remaining * 0.3, 30.0))
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            a = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f}")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    phase2_end = start_time + (deadline - start_time) * 0.65
    n_parent = len(best_seq)
    n_options = sorted(set([n_parent - 100, n_parent - 50, n_parent - 25,
                            n_parent, n_parent + 25, n_parent + 50,
                            n_parent + 100, 600, 700, 900, 1000, 1100]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    # Remove n_parent (already optimized) to save time, but keep it for now
    # Actually let's keep it but deprioritize

    for n_try in n_options:
        remaining = phase2_end - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP alternating
        alloc = min(remaining * 0.2, 60.0)
        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Diversified restarts ===
    print(f"\n=== Phase 3: Diversified Restarts ===")
    phase3_end = start_time + (deadline - start_time) * 0.85
    n_best = len(best_seq)

    for r in range(15):
        remaining = phase3_end - time.monotonic()
        if remaining < 20:
            break

        mode = r % 5
        n_r = n_best + int(rng.integers(-50, 50))
        n_r = max(400, min(1200, n_r))

        if mode == 0:
            a_r = random_init(n_r, rng, mode=r % 5)
        elif mode == 1:
            # Perturbed best
            a_r = best_seq[:n_r] if n_r <= len(best_seq) else resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.08, len(a_r)))
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 0:
                a_r = a_r / s_r
        elif mode == 2:
            # U-shaped
            x = np.linspace(0, 1, n_r)
            a_r = 1.0 + 1.0 * (2 * x - 1) ** 2
            a_r += rng.normal(0, 0.05, n_r)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif mode == 3:
            # Two-peak
            c1, c2 = n_r // 4, 3 * n_r // 4
            w = n_r // 8
            a_r = np.exp(-((np.arange(n_r) - c1) ** 2) / (2 * w ** 2)) + \
                  np.exp(-((np.arange(n_r) - c2) ** 2) / (2 * w ** 2))
            a_r += 0.1
            a_r = a_r / np.sum(a_r)
        else:
            # Best with structural perturbation
            a_r = resample_sequence(best_seq, n_r)
            # Flip a random block
            blen = n_r // 5
            bstart = int(rng.integers(0, max(1, n_r - blen)))
            a_r[bstart:bstart + blen] *= 0.5
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        alloc = min(remaining * 0.15, 40.0)
        a_opt = lp_alternate(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}, n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}, n={n_r}): obj={obj_r:.8f}")

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()

        # LP alternating
        alloc = min(remaining * 0.4, 80.0)
        a_lp = lp_alternate(a_final, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")

        # Peak shave
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_ps = peak_shave(a_final, time_budget=min(remaining * 0.3, 30.0))
            obj_ps = compute_obj(a_ps)
            if obj_ps < best_score:
                best_score = obj_ps
                best_seq = a_ps.copy()
                a_final = a_ps.copy()
                print(f"  Final Peak Shave: obj={best_score:.8f}")

        # Nesterov
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_ns = nesterov_subgrad(a_final, time_budget=min(remaining * 0.3, 25.0))
            obj_ns = compute_obj(a_ns)
            if obj_ns < best_score:
                best_score = obj_ns
                best_seq = a_ns.copy()
                print(f"  Final Nesterov: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < np.inf and val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
