def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)

    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 4)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
    elif mode == 2:
        # Smooth random
        a = np.cumsum(rng.uniform(0, 1, n))
        a = a - a[0]
        a[0] = 1.0
        a = a + 0.1
        a = a / np.sum(a)
    else:
        # Bimodal
        center = n // 2
        width = n // 4
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
        a = a + 0.3 * np.exp(-((np.arange(n) - center * 0.5) ** 2) / (2 * (width // 2) ** 2))
        a = a / np.sum(a)
    return a


def coordinate_refine(a, time_budget=20.0):
    """
    Coordinate descent: adjust pairs of coordinates to reduce peak convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from peak-contributing positions
        for i in range(0, n, 3):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i and a[i] > a[j]:
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv)) / (float(np.sum(a)) ** 2)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        if not improved:
            break

    return best_a
