Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Improved version with better LP alternating, peak shaving, n-sweeping,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=30):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

    best_a = a.copy()

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating, accelerated subgradient, spectral equalization,
n-sweeping, and diverse random restarts.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def spectral_equalize(a, max_iters=40, time_budget=30.0):

    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        
        alpha = 0.3 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.6 * a + 0.4 * a_new)
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

    for cycle in range(6):
        if time.monotonic() > deadline - 400:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.15, 80.0)
        a = lp_alternate(a, max_iters=30, time_budget=alloc)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  LP cycle {cycle}: {best_score:.8f}")
        else:
            print(f"  LP cycle {cycle}: {obj:.8f}")
    
    # Phase 2: Subgradient
    print("\n--- Phase 2: Subgradient ---")
    a = best_seq.copy()
    for cycle in range(3):
        if time.monotonic() > deadline - 350:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.12, 40.0)
        a = subgradient_optimize(a, max_steps=200, time_budget=alloc)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Subgrad cycle {cycle}: {best_score:.8f}")
        else:
            print(f"  Subgrad cycle {cycle}: {obj:.8f}")
    
    # Phase 3: Spectral
    print("\n--- Phase 3: Spectral equalization ---")
    a = best_seq.copy()
    spec = spectral_equalize(a, max_iters=30, time_budget=30.0)
    obj_spec = compute_obj(spec)
    if obj_spec < best_score:
        best_score = obj_spec
        best_seq = spec.copy()
    
    # Follow up with LP
    if time.monotonic() < deadline - 250:
        a = best_seq.copy()
        a = lp_alternate(a, max_iters=20, time_budget=50.0)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Post-spectral LP: {best_score:.8f}")
    
    # Phase 4: n-sweep
    print("\n--- Phase 4: n-sweep ---")
    n_best = len(best_seq)
    n_options = sorted(set([max(200, n_best + d) for d in range(-300, 400, 50)]))
    n_options = [n for n in n_options if 200 <= n <= 1500]
    
    for n_try in n_options:
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        
        a_try = normalize(resample(best_seq, n_try))
        alloc = min(remaining * 0.08, 25.0)
        
        a_try = lp_alternate(a_try, max_iters=10, time_budget=alloc * 0.5)
        a_try = subgradient_optimize(a_try, max_steps=50, time_budget=alloc * 0.4)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")
    
    # Phase 5: Random restarts
    print("\n--- Phase 5: Random restarts ---")
    n_cur = len(best_seq)
    rng = np.random.default_rng(seed)
    
    for restart in range(10):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.1, 30.0)
        
        if restart % 5 == 0:
            a_try = best_seq * np.exp(rng.normal(0, 0.06, n_cur))
        elif restart % 5 == 1:
            freq = rng.uniform(1.0, 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.12 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
            a_try = best_seq * mod
        elif restart % 5 == 2:
            a_rand = rng.dirichlet(np.ones(n_cur) * 0.3)
            mix = rng.uniform(0.3, 0.6)
            a_try = (1 - mix) * normalize(best_seq) + mix * a_rand
        elif restart % 5 == 3:
            # Block reversal
            a_try = best_seq.copy()
            blen = int(rng.integers(n_cur // 4, n_cur // 2))
            s_idx = int(rng.integers(0, n_cur - blen))
