import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """A[k,j] = a[k-j] if 0 <= k-j < n, else 0. Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k,j] = a[k-j] for k such that 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - j]
    return A

"""
LP-alternating optimization with subgradient polishing for minimizing 
2*n*max(conv(a,a))/(sum(a))^2. Uses PDLP solver, adaptive steps, 
multi-n exploration, and iterative improvement from parent sequence.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - j]
    return A

def solve_minpeak_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.PDLP, cp.CLARABEL, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
        b_ub = np.zeros(2*n-1)
        A_eq = np.hstack([np.ones((1, n)), np.array([[0.0]])])
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_minpeak_lp(a)
        if g_lp is None:
            break
        
        # Line search
        best_t_obj = best_obj
        best_t = 0.5
        best_a_try = best_a.copy()
        
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_try = a_try
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_try
            a = best_a
            stall = 0
            if verbose and it % 50 == 0:
                print(f"    it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 30:

                a = best_a * np.exp(rng.normal(0, 0.05, n))

                s = float(np.sum(a))
                if s > 1e-10:

                stall = 0

def subgradient_polish(a, time_budget=30.0):
    """Subgradient-based polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.01
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: grad[j] = 2*a[k_star - j] if 0 <= k_star-j < n
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
        
        # Center the gradient (to maintain sum)
        grad_centered = grad - np.mean(grad)
        
        # Step
        a_new = a - eta * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new
                a = a_new
            else:
                eta *= 0.9
        else:
            eta *= 0.5
    
    return best_a

    """Resample sequence to new length."""

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

"""
LP-alternating optimization with PDLP solver, adaptive line search, subgradient polishing,
and multi-n exploration for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Builds on parent sequence with intensive LP iterations and periodic perturbations.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - j]
    return A


def solve_minpeak_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    if warm_start is not None:
        g.value = warm_start
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack((A, -np.ones((2 * n - 1, 1))))
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.hstack((np.ones((1, n)), np.array([[0.0]])))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget=100.0, verbose=False):
    """
    LP alternating: iteratively solve min-peak LP and blend.
    Uses adaptive step size and periodic perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    prev_g = None

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_minpeak_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp.copy()

        # Adaptive line search
        candidate_ts = [0.3, 0.5, 0.7, 0.9, 1.0]
        if 0.1 < last_t < 1.0:
            candidate_ts.append(max(0.1, last_t * 0.6))
            candidate_ts.append(min(1.0, last_t * 1.4))
            candidate_ts = sorted(set(candidate_ts))

        best_t_obj = best_obj
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and it % 100 == 0:
                print(f"      it{it}: obj={best_obj:.9f}, t={best_t_val:.3f}, M_lp={M_lp:.6f}")
        else:
            stall += 1
            if stall % 30 == 0:
                scale = 0.03 + 0.01 * min(stall // 30, 5)
                rng = np.random.default_rng(it * 31 + n * 7)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                obj_a = compute_obj(a)
                if obj_a < best_obj + 0.01:
                    best_a = a.copy()
                    best_obj = obj_a
                stall = 0

    return best_a


def subgradient_polish(a, time_budget=30.0):
    """
    Subgradient-based polishing targeting the peak convolution bin.
    Uses the analytic subgradient: grad[j] = 2*a[k*-j] for valid indices.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.005

    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: grad[j] = 2*a[k_star - j] if 0 <= k_star-j < n
        grad = np.zeros(n)
        valid = np.arange(n)
        idx = k_star - valid
        mask = (idx >= 0) & (idx < n)
        grad[mask] = 2.0 * a[idx[mask]]

        # Center gradient to preserve sum
        grad_c = grad - np.mean(grad)

        # Multiplicative update (mirror descent)
        a_new = a * np.exp(-eta * grad_c)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                eta *= 0.95
        else:
            eta *= 0.5

    return best_a


def multiplicative_weights_polish(a, time_budget=30.0):
    """
    Multiplicative weights / exponential tilting polish.
    Adjusts a[j] proportionally to reduce peak convolution contribution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.02

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # For each j, the contribution to conv[k_star] is a[j]*a[k_star-j]
        # We want to reduce a[j] where a[k_star-j] is large
        contrib = np.zeros(n)
        idx = k_star - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib[mask] = a[idx[mask]]

        # Proportional reduction: a[j] *= (1 - eta * contrib[j] / max_contribution)
        max_c = float(np.max(contrib))
        if max_c > 1e-15:
            a_new = a * (1.0 - eta * contrib / max_c)
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                eta *= 0.9
        else:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via piecewise linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
