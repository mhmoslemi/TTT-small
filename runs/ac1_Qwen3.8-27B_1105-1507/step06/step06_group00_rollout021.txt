    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt, dual_vars) or (None, None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    C = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [C @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    best_g = None
    best_M = np.inf
    best_dual = None

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if M_val < best_M:
                        best_M = M_val
                        best_g = g_val
                        # Get dual variables if available
                        try:
                            if constraints[0].dual_value is not None:
                                best_dual = np.array(constraints[0].dual_value)
                        except:
                            best_dual = None
        except Exception:
            continue

    if best_g is None:
        # Fallback to scipy
        try:
            from scipy.optimize import linprog
            c_vec = np.zeros(n + 1)
            c_vec[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = C
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(0, None)]
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    best_g = g_val / s
                    best_M = float(res.x[n])
        except Exception:
            pass

    if best_g is None:
        return None, None, None
    return best_g, best_M, best_dual


def smooth_max(conv, beta):
    """Smooth approximation of max using log-sum-exp."""
    m = float(np.max(conv))
    if m < 1e-300:
        return m, np.zeros_like(conv)
    shifted = beta * (conv - m)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    lse = m + np.log(sum_exp) / beta
    weights = exp_vals / sum_exp
    return lse, weights

import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a * a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:

    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix C (2n-1 x n) such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        C[lo:hi, j] = a[:hi - lo]
    return C

def smooth_lse_max(conv, beta):
    """Compute log-sum-exp approximation of max and the softmax weights."""
    m = float(np.max(conv))
    if m < 1e-300:
        return m, np.full(len(conv), 1.0 / len(conv))
    shifted = beta * (conv - m)
    # Clamp to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        return m, np.full(len(conv), 1.0 / len(conv))
    lse = m + np.log(sum_exp) / beta
    weights = exp_vals / sum_exp
    return lse, weights


def smooth_grad_obj(a, beta):
    """
    Compute smooth objective and gradient for f(a) = 2n * LSE(beta * conv(a/s, a/s)).
    Returns (objective, gradient) where gradient is w.r.t. a (unnormalized).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(n)
    h = a / s
    conv = compute_conv(h)
    lse, weights = smooth_lse_max(conv, beta)
    obj = 2.0 * n * lse

    # grad[k] = 4n/s^2 * (cross_corr[k] - s * dot(weights, conv))
    # cross_corr[k] = sum_{i=0}^{n-1-k} a[i] * weights[i+k]
    # Use np.correlate: np.correlate(weights, a, 'valid')[k] = sum_n weights[n+k]*a[n]
    # weights has length 2n-1, a has length n
    # 'valid' mode: output length = (2n-1) - n + 1 = n
    cross = np.correlate(weights, a, mode='valid')
    dot_wc = float(np.dot(weights, conv))
    grad = (4.0 * n / (s * s)) * (cross - s * dot_wc)

    return obj, grad


def lp_alternating(a, time_budget=60.0, use_t1=True, verbose=False):
    """
    LP alternating optimization.
    If use_t1, directly replaces a with g (t=1) and uses averaging.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    # For t=1 mode: maintain a running average
    avg_a = a.copy()
    avg_count = 0.0

    max_iters = 300
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        if use_t1:
            # Direct replacement with averaging
            avg_a = (avg_a * avg_count + g_lp) / (avg_count + 1.0)
            avg_count += 1.0
            a_try = avg_a / np.sum(avg_a)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                stall = 0
                if verbose and it % 10 == 0:
                    print(f"      LP-t1 it{it}: obj={best_obj:.8f}")
            else:
                # Also try the raw g_lp
                obj_raw = compute_obj(g_lp)
                if obj_raw < best_obj - 1e-12:
                    best_obj = obj_raw
                    best_a = g_lp.copy()
                    a = g_lp.copy()
                    stall = 0
                else:
                    stall += 1
                    if stall > 15:
                        # Perturb to escape
                        rng = np.random.default_rng(it * 7919 + n)
                        scale = 0.01 + 0.01 * min(stall // 15, 5)
                        a = best_a * np.exp(rng.normal(0, scale, n))
                        a = np.maximum(a, 0)
                        s = float(np.sum(a))
                        if s > 1e-10:
                            a = a / s
                            avg_a = a.copy()
                            avg_count = 1.0
                        stall = 0
        else:
            # Blending mode
            candidate_ts = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
            best_t_obj = np.inf
            best_t_val = 0.5
            best_a_local = best_a.copy()
            for t in candidate_ts:
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t_val = t
                    best_a_local = a_try.copy()

            if best_t_obj < best_obj - 1e-12:
                best_obj = best_t_obj
                best_a = best_a_local.copy()
                a = best_a
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    rng = np.random.default_rng(it * 104729 + n)
                    scale = 0.02
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    stall = 0

    return best_a


def smooth_refine(a, time_budget=30.0, beta_start=30.0):
    """
    Smooth gradient refinement using log-sum-exp approximation.
    Uses adaptive step size with backtracking.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = beta_start
    lr = 0.02

    max_iters = 500
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        obj_s, grad = smooth_grad_obj(a, beta)

        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                # Slowly increase step size
                lr = min(lr * 1.02, 0.1)
            else:
                lr *= 0.7
        else:
            lr *= 0.5

        # Increase beta to sharpen
        beta = min(beta * 1.05, 500.0)

    return best_a


def chase_peak(a, time_budget=20.0):
    """
    Chase-the-peak: iterative subgradient targeting the max convolution bin.
    With adaptive step size and multi-peak awareness.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05
    conv = compute_conv(a)
    top_k = 5  # Track top K peaks

    max_iters = 1000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Find top K peaks
        peak_indices = np.argsort(conv)[-top_k:][::-1]
        peak_vals = conv[peak_indices]
        # Weight peaks by their values (weighted subgradient)
        weights = peak_vals / (peak_vals.sum() + 1e-30)

        # Compute weighted subgradient
        grad = np.zeros(n)
        for k_star in peak_indices:
            w = weights[list(peak_indices).index(k_star)]
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] += 2.0 * w * a[j]

        # Project: subtract mean to preserve sum, clip to non-negative
        grad_centered = grad - np.mean(grad)
        a_new = a - step_size * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                conv = compute_conv(a)
                step_size = min(step_size * 1.01, 0.15)
            else:
                step_size *= 0.85
        else:
            step_size *= 0.5

    return best_a


def peak_shave(a, time_budget=30.0):
    """
    Peak shaving: identify active peaks and try to reduce them by
    moving mass from contributing positions to non-contributing ones.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        # Find the peak
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Identify positions that contribute to this peak
        # conv[k_star] = sum_i a[i]*a[k_star-i] for valid i
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((i, j, a[i] * a[j]))

        if not contributors:
            break

        # Sort by contribution
        contributors.sort(key=lambda x: -x[2])

        improved = False
        # Try moving mass from high-contributing positions
        for i, j, contrib in contributors[:10]:
            if a[i] < 1e-6:
                continue
            # Try moving some mass from position i to a position that doesn't
            # contribute to the peak
            for target in range(n):
                # Check if target contributes to the peak
                j_t = k_star - target
                if 0 <= j_t < n:
                    continue  # This position also contributes to the peak

                # Try moving a small amount
                for frac in [0.01, 0.03, 0.05]:
                    delta = frac * a[i]
                    if delta < 1e-8:
                        break
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_peak = float(np.max(new_conv))
                    new_obj = 2.0 * n * new_peak  # sum is still 1
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break

            if improved:
                break

        if not improved:
            # Try splitting mass between two non-contributing positions
            broken = False
            for i, j, contrib in contributors[:5]:
                if a[i] < 1e-6:
                    continue
                for target in range(n):
                    j_t = k_star - target
                    if 0 <= j_t < n:
                        continue
                    delta = 0.02 * a[i]
                    if delta < 1e-8:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        broken = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating + smooth gradient + peak shaving + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # Phase 1: Intensive LP alternating with t=1
    print(f"\n=== Phase 1: LP Alternating t=1 (n={n_parent}) ===")
    phase1_budget = min(400.0, (deadline - start_time) * 0.35)

    a = best_seq.copy()
    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(phase1_budget / (4 - round_idx), remaining * 0.5)
        a_new = lp_alternating(a, time_budget=alloc, use_t1=True, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP-t1 Round {round_idx}: obj={best_score:.8f}")
        else:
            print(f"  LP-t1 Round {round_idx}: obj={obj:.8f} (no improvement)")

    # Phase 2: LP alternating with blending (different dynamics)
    print(f"\n=== Phase 2: LP Alternating blended ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 80.0)
        a_blend = lp_alternating(a, time_budget=alloc, use_t1=False, verbose=True)
        obj = compute_obj(a_blend)
        if obj < best_score:
            best_score = obj
            best_seq = a_blend.copy()
            a = a_blend.copy()
            print(f"  LP-blend: obj={best_score:.8f}")
        else:
            print(f"  LP-blend: obj={obj:.8f}")

    # Phase 3: Smooth gradient refinement
    print(f"\n=== Phase 3: Smooth Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.1, 40.0)
        a_smooth = smooth_refine(a, time_budget=alloc, beta_start=20.0)
        obj = compute_obj(a_smooth)
        if obj < best_score:
            best_score = obj
            best_seq = a_smooth.copy()
            a = a_smooth.copy()
            print(f"  Smooth: obj={best_score:.8f}")
        else:
            print(f"  Smooth: obj={obj:.8f}")

    # Phase 4: Chase peak
    print(f"\n=== Phase 4: Chase Peak ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.1, 25.0)
        a_cp = chase_peak(a, time_budget=alloc)
        obj = compute_obj(a_cp)
        if obj < best_score:
            best_score = obj
            best_seq = a_cp.copy()
            a = a_cp.copy()
            print(f"  Chase: obj={best_score:.8f}")
        else:
            print(f"  Chase: obj={obj:.8f}")

    # Phase 5: Peak shaving
    print(f"\n=== Phase 5: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.1, 30.0)
        a_shave = peak_shave(a, time_budget=alloc)
        obj = compute_obj(a_shave)
        if obj < best_score:
            best_score = obj
            best_seq = a_shave.copy()
            a = a_shave.copy()
            print(f"  Shave: obj={best_score:.8f}")
        else:
            print(f"  Shave: obj={obj:.8f}")

    # Phase 6: Multi-n exploration
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        n_parent - 100, n_parent - 50, n_parent - 25,
        n_parent, n_parent + 25, n_parent + 50, n_parent + 100,
        n_parent + 200,
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP alternating
        alloc = min(remaining * 0.15, 40.0)
        a_opt = lp_alternating(a_try, time_budget=alloc, use_t1=True)

    # Phase 7: Random restarts with structured perturbations
    print(f"\n=== Phase 7: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    restart_budget = min(25.0, remaining * 0.08)

    for r in range(10):

            a_r = a_r + 0.01 * rng.random(n_r)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            # Different n with perturbation
            n_r = int(n_best + rng.integers(-150, 150))
            n_r = max(300, min(1500, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.03, n_r))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 3:
            # Bimodal
            n_r = n_best
            c1, c2 = n_r // 3, 2 * n_r // 3
            w = n_r // 8
            a_r = np.exp(-((np.arange(n_r) - c1) ** 2) / (2 * w ** 2)) + \

            a_r = a_r + 0.01 * rng.random(n_r)
            a_r = a_r / np.sum(a_r)
        else:
            # Sparse random
            n_r = n_best
            a_r = np.zeros(n_r)
            n_nonzero = n_r // 3
            indices = rng.choice(n_r, size=n_nonzero, replace=False)
            a_r[indices] = rng.uniform(0.5, 1.0, n_nonzero)
            if np.sum(a_r) > 0:
                a_r = a_r / np.sum(a_r)
            else:
                a_r = np.ones(n_r) / n_r

        a_opt = lp_alternating(a_r, time_budget=restart_budget, use_t1=True)
        # Quick smooth refine
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_opt = smooth_refine(a_opt, time_budget=min(5.0, remaining * 0.3))
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_r:.8f}")

    # Phase 8: Final intensive polish on best found
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()
        n_final = len(a_final)

        # LP alternating
        alloc = min(remaining * 0.3, 60.0)
        a_lp = lp_alternating(a_final, time_budget=alloc, use_t1=True, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")

        # Smooth
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sm = smooth_refine(a_final, time_budget=min(remaining * 0.2, 20.0), beta_start=50.0)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                a_final = a_sm.copy()
                print(f"  Final Smooth: obj={best_score:.8f}")

        # Chase peak
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_cp = chase_peak(a_final, time_budget=min(remaining * 0.2, 20.0))
            obj_cp = compute_obj(a_cp)
            if obj_cp < best_score:
                best_score = obj_cp
                best_seq = a_cp.copy()
                a_final = a_cp.copy()
                print(f"  Final Chase: obj={best_score:.8f}")

        # Peak shave
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sh = peak_shave(a_final, time_budget=min(remaining * 0.2, 15.0))
            obj_sh = compute_obj(a_sh)
            if obj_sh < best_score:
                best_score = obj_sh
                best_seq = a_sh.copy()
                a_final = a_sh.copy()
                print(f"  Final Shave: obj={best_score:.8f}")

        # One more LP

        if remaining > 10:
            a_last = lp_alternating(a_final, time_budget=min(remaining * 0.4, 20.0), use_t1=True)
            obj_last = compute_obj(a_last)

                best_seq = a_last.copy()

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
