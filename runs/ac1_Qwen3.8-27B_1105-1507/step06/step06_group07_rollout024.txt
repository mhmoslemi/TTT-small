import numpy as np

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_obj_and_grad(a, beta):
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj

    beta2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):

            break

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def build_chirp(n, chirp_rate=1.0):
    """Build a chirp (quadratic phase) sequence."""
    t = np.arange(n) / n
    phase = chirp_rate * np.pi * t * t
    # Use absolute value of complex chirp as real sequence
    a = np.abs(np.cos(2 * np.pi * phase))
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a

def build_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped sequence."""
    t = np.arange(n) / n - 0.5
    a = np.exp(-t * t / (2 * sigma_frac * sigma_frac))
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a

def build_hann(n):
    """Hann window."""
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a

"""
Multi-phase optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse structured initializations (ramp, quadratic, exponential, parent)
2. Log-space Adam with cosine LR and adaptive beta
3. Peak-targeted subgradient descent
4. Pairwise coordinate descent
5. Multi-n exploration with resampling
6. Final high-precision polish with very high beta

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_and_grad(a, beta):
    """Compute smoothed objective and gradient via logsumexp softmax."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def adam_log_optimize(a, time_budget, beta_start, beta_end, lr_start, lr_end, max_iters=50000):
    """Adam in log-space with cosine LR schedule and centering."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-12:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * progress))

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a_arr_sum = np.sum(a)
        if a_arr_sum < 1e-30:
            break
        a = a / a_arr_sum

        _, grad_a = smooth_obj_and_grad(a, beta)
        grad_x = grad_a * a

        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 200 == 0:
            x -= np.mean(x)

        if it % 150 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            asc = np.sum(a_check)
            if asc < 1e-30:
                break
            a_check = a_check / asc
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()

    return best_a


def peak_targeted_descent(a, time_budget, max_steps=3000):
    """Direct subgradient descent targeting the maximum of the autocorrelation."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    s = np.sum(best_a)
    if s < 1e-12:
        return best_a
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    stagnant = 0

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        _, grad_a = smooth_obj_and_grad(best_a, beta=800.0)
        grad_norm = np.linalg.norm(grad_a)
        if grad_norm < 1e-15:
            break

        found = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            a_try = best_a - alpha * grad_a
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-12:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_a = a_try.copy()
                best_obj = obj_try
                found = True
                stagnant = 0
                break

        if not found:
            stagnant += 1
            if stagnant > 200:
                break
            if stagnant % 50 == 0:
                rng = np.random.default_rng(1000 + step)
                a_try = best_a + rng.normal(0, 0.002, n)
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try > 1e-12:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_a = a_try.copy()
                        best_obj = obj_try
                        stagnant = 0

    return best_a


def coordinate_descent(a, time_budget, max_rounds=200):
    """Pairwise coordinate descent focusing on top peaks."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    s = np.sum(best_a)
    best_a = best_a / s
    best_obj = compute_obj(best_a)

    conv = compute_conv(best_a)

    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        peaks = np.argsort(conv)[::-1][:min(5, len(conv))]

        for p in peaks:
            if time.monotonic() - start > time_budget:
                break
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            span = i_max - i_min + 1
            step_size = max(1, span // 15)

            for i in range(i_min, i_max + 1, step_size):
                if time.monotonic() - start > time_budget:
                    break
                j = p - i
                if j < 0 or j >= n or j == i:
                    continue

                for frac in [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01]:
                    delta = frac * max(best_a[i], best_a[j], 1e-8)
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    s_try = np.sum(a_try)
                    if s_try < 1e-12:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        conv = compute_conv(best_a)
                        peaks = np.argsort(conv)[::-1][:min(5, len(conv))]
                        improved = True
                        break

        if not improved:
            break

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def build_ramp(n):
    """Linearly increasing ramp."""
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / np.sum(a)


def build_quadratic(n):
    """Quadratic ramp."""
    t = np.arange(1, n + 1, dtype=np.float64)
    a = t * t
    return a / np.sum(a)


def build_exp_decay(n, rate=0.5):
    """Exponential decay from left."""
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * t / n * 3)
    return a / np.sum(a)


def build_bimodal(n):
    """Two peaks: one at 1/4, one at 3/4."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.25) ** 2) / 0.01) + np.exp(-((t - 0.75) ** 2) / 0.01)
    return a / np.sum(a)


def build_sinusoidal(n, freq=2.0):
    """Sinusoidal modulation of a base envelope."""
    t = np.linspace(0, 1, n)
    envelope = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = 0.5 + 0.5 * np.abs(np.sin(np.pi * freq * t))
    a = a * (1 - 0.3 * envelope) + 0.3 * envelope
    a = np.maximum(a, 0.01)
    return a / np.sum(a)
