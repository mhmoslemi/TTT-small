import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):

def gs_flat_spectral(init, n_iters=300, beta=0.8):

        return a

def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-11)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, p, b, b_max

def build_A_values(n, a):
    """Build sparse matrix values for the LP constraint matrix A where (a*g)(k) = (A g)_k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    n = len(a)
    rows, cols, vals = build_A_values(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_solve(history_x, history_Fx, m):
    """Solve the Anderson mixing QP: find alpha minimizing ||sum alpha_i * r_i||^2 s.t. sum alpha = 1.
    Returns the mixed point sum(alpha_i * Fx_i)."""
    residuals = []
    for i in range(m):
        residuals.append(history_x[i] - history_Fx[i])
    residuals = np.array(residuals)  # shape (m, n)
    # G_{ij} = r_i . r_j
    G = residuals @ residuals.T  # (m, m)
    # Add small regularization for stability
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    # Clamp alpha to be in a reasonable range (allow mild extrapolation)
    alpha = np.clip(alpha, -1.0, 1.0)
    alpha = alpha / alpha.sum()
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def run_lp_anderson(a, deadline, max_iters=200, anderson_window=5,

            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-10):

            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        # Determine step size
        if it < 5:
            t = t_base
        else:
            t = max(min_t, t_base / (1 + 0.05 * it))
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        if len(hist_x) >= 3:

            M_new = fast_conv(a_new).max()
            if M_and <= M_new * 1.05:
                a = a_anderson
            else:
                a = a_new
        else:
            a = a_new
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):

    inits.append(np.ones(n))
    for alpha in [0.5, 0.8, 1.0, 1.5, 2.5]:
        inits.append(np.maximum(0.2 + 0.8 * (t**alpha + (1 - t)**alpha) / 2, 0.01))
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.05)
    inits.append(np.exp(-8 * (t - 0.3)**2) + np.exp(-8 * (t - 0.7)**2) + 0.03)
    # Random smooth
    k = max(3, n // 20)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    # GS from random

    inits.append(gs_flat_spectral(a_rand, n_iters=100, beta=0.85))
    # GS from parent (if available)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(gs_flat_spectral(p, n_iters=150, beta=0.9))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
