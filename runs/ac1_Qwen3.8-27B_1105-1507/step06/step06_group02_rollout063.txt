def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def gs_flat_spectral(init, n_iters=500, beta_schedule=None):
    """Aggressive Gerchberg-Saxton with adaptive compression."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    
    for it in range(n_iters):
        frac = it / max(n_iters, 1)
        if beta_schedule is not None:
            beta = beta_schedule(frac)
        else:
            beta = 0.7 + 0.3 * frac
        
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Power-law compression: mag -> mag^p / max(mag^p) * max_mag
        p = 0.3 + 0.7 * (1 - beta)  # Start with strong compression, end with mild
        mag_p = mag ** p
        mag_new = mag_p / (mag_p.max() + 1e-30) * mag.max()
        
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = float(fast_conv(a_candidate).max())
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    
    return best_a

def smooth_objective(a, tau):
    """Compute smooth approximation of J and its gradient."""
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / (exp_b.sum() + 1e-30)
    
    # Objective: (1/tau) * log(sum(exp(tau*b))) / sum(a)^2 * 2n
    # But since we normalize sum(a)=1, it's just 2n * (1/tau) * log(sum(exp(tau*b)))
    obj = 2 * n * (b_max + np.log(p.sum() + 1e-30) / tau)
    
    # Gradient with respect to a (before normalization)
    grad_b = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad_b[lo:hi + 1] += 2.0 * p[k] * seg
    
    # Account for normalization: a_norm = a / sum(a)
    s = a.sum()
    grad = (grad_b - grad_b * s / s * 0)  # Simplified since we work with normalized a
    # Actually, for normalized a, the gradient of f(a/sum(a)) is:
    # (grad_f - mean(grad_f)) * s (approximately, for large s)
    # More precisely: grad_obj = (1/s) * (grad_b - a * (grad_b @ a / s^2))
    # Let's just use the unnormalized gradient and let the optimizer handle it
    return obj, 2 * n * grad_b / (s * s)

            # Gradient of the normalized objective
            s = x.sum()
            # d/da [2n * smoothmax(a*a) / s^2] where s = sum(a)

def build_A_sparse(n, a):
    """Build sparse matrix for LP constraint (a*g)(k) for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals

def lp_best_response(a, M, verbose=False):
    """Solve the LP: max sum(g) s.t. (a*g)(k) <= M, g >= 0."""
    n = len(a)
    rows, cols, vals = build_A_sparse(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'time_limit': 30.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception as e:
        if verbose:
            print(f"  LP error: {e}")
    return None

def lp_mirror_backtrack(a, deadline, max_iters=100, t_init=0.3):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()

    no_imp = 0
    t = t_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Backtracking line search
        improved = False
        for t_try in [t, t*0.7, t*0.5, t*0.3, t*0.1, 0.05, 0.02, 0.01]:
            a_try = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            M_try = float(fast_conv(a_try).max())
            J_try = compute_J(a_try)
            if J_try < best_J - 1e-12:
                a = a_try
                best_M = M_try
                best_J = J_try
                best_a = a.copy()
                t = t_try  # Keep this step size
                improved = True
                break
            elif M_try < best_M - 1e-12:
                a = a_try
                best_M = M_try
                best_a = a.copy()
                t = t_try
                improved = True
                break
        
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            t *= 0.5
            if no_imp > 20:
                break
    
    return best_a

def lp_mirror_anderson(a, deadline, max_iters=150, anderson_window=5, t_base=0.3, min_t=0.005):

        else:
            a_new /= s
        
        # Anderson acceleration

        if len(hist_x) > anderson_window:
            hist_x.pop(0)

        if len(hist_x) >= 3:

            ones = np.ones(len(hist_x))
            try:

                    alpha = G_inv_ones / denom
                    alpha = np.clip(alpha, -1.5, 1.5)

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        for sigma in [0.01, 0.03, 0.05, 0.1]:
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 0.005))
    
    # All ones
    inits.append(np.ones(n))
    
    # Power law shapes
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t**alpha), 0.01))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive spectral flattening (GS with adaptive power-law compression)
2. FFT-accelerated L-BFGS-B with adaptive smoothing (fast intermediate optimization)
3. LP mirror descent with Anderson acceleration and backtracking line search
4. Multi-start with diverse initializations and extended length exploration
5. Perturbation-based restarts to escape local minima
Key improvements: FFT-based gradient (O(n log n) vs O(n^2)), L-BFGS quasi-Newton
convergence, backtracking LP step sizes, and longer sequence exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def fft_cross_corr(p, a, n):
    """Compute (p circ_conv a)[0:n] where p has length 2n-1, using FFT.
    Returns C[i] = sum_{m=0}^{n-1} p[m+i] * a[m] for i=0..n-1."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    p_padded = np.zeros(size)
    p_padded[:len(p)] = p
    a_padded = np.zeros(size)
    a_padded[:n] = a
    corr = np.fft.ifft(np.conj(np.fft.fft(a_padded)) * np.fft.fft(p_padded))
    return np.real(corr[:n])


def gs_flat_spectral(init, n_iters, mode):
    """Gerchberg-Saxton with different compression strategies."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())

    for it in range(n_iters):
        frac = it / max(n_iters, 1)
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_max = mag.max() + 1e-30

        if mode == 0:
            # Power-law compression
            p = 0.3 + 0.7 * (1.0 - frac)
            mag_p = mag ** p
            mag_new = mag_p / (mag_p.max() + 1e-30) * mag_max
        elif mode == 1:
            # Log compression
            mag_log = 1.0 + np.log(mag + 1e-30)
            beta = 0.5 + 0.5 * frac
            mag_new = (1.0 - beta) * mag + beta * (mag_log / (mag_log.max() + 1e-30) * mag_max)
        else:
            # Quantile-based: keep top fraction, compress rest
            threshold = np.percentile(mag, 10)
            mag_new = np.where(mag > threshold, mag, threshold * 0.5)
            mag_new = mag_new / (mag_new.max() + 1e-30) * mag_max

        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = float(fast_conv(a_candidate).max())
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate

    return best_a


def lbfgs_func_grad(x, tau, n):
    """Objective and gradient for L-BFGS: f(a) = 2n * smoothmax(a*a) / (sum(a))^2."""
    x = np.maximum(x, 1e-15)
    s = x.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(x)
    b_max = float(b.max())
    # Log-sum-exp for numerical stability
    shifted = tau * (b - b_max)
    log_sum = b_max * tau + np.log(np.sum(np.exp(shifted)))
    S = log_sum / tau  # smoothmax value
    exp_shifted = np.exp(shifted)
    p = exp_shifted / (exp_shifted.sum() + 1e-30)

    obj = 2.0 * n * S / (s * s)

    # Gradient: 4n/s^2 * (p cross_corr x) - 4n*S/s^3
    corr = fft_cross_corr(p, x, n)
    grad = 4.0 * n * (corr / (s * s) - S / (s * s * s))
    return obj, grad


def lbfgs_optimize(a0, deadline, max_iter):
    """L-BFGS-B optimization with adaptive smoothing."""
    a0 = np.maximum(a0.astype(np.float64), 1e-10)
    a0 = a0 / a0.sum()
    n = len(a0)
    best_a = a0.copy()
    best_J = compute_J(a0)

    tau_values = [5.0, 10.0, 25.0, 50.0, 100.0, 200.0, 500.0]
    iters_per_tau = max(30, int(3000 / n) + 20)

    for tau in tau_values:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(lbfgs_func_grad, a0.copy(), args=(tau, n),
                             method='L-BFGS-B', jac=True,
                             bounds=[(0, None)] * n,
                             options={'maxiter': iters_per_tau, 'gtol': 1e-5,
                                      'maxfun': iters_per_tau * 3})
            if result.x is not None:
                a_new = np.maximum(result.x, 0)
                sn = a_new.sum()
                if sn > 1e-10:
                    a_new /= sn
                    J_new = compute_J(a_new)
                    if J_new < best_J:
                        best_J = J_new
                        best_a = a_new.copy()
                    a0 = a_new
        except Exception:
            pass

    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix for LP constraint (a*g)(k) for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0."""
    n = len(a)
    rows, cols, vals = build_A_sparse(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'time_limit': 60.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(hist_x, hist_Fx):
    """Anderson mixing: find alpha minimizing ||sum alpha_i * r_i||^2 s.t. sum alpha = 1."""
    m = len(hist_x)
    residuals = np.array([hx - hf for hx, hf in zip(hist_x, hist_Fx)])
    G = residuals @ residuals.T + 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
            alpha = np.clip(alpha, -2.0, 2.0)
            alpha_sum = alpha.sum()
            if abs(alpha_sum) < 1e-15:
                alpha = ones / m
            else:
                alpha = alpha / alpha_sum
    except np.linalg.LinAlgError:
        alpha = ones / m
    n = len(hist_x[0])
    mixed = np.zeros(n)
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    return np.maximum(mixed, 0)


def lp_mirror_anderson(a, deadline, max_iters, anderson_window, t_base, min_t):
    """LP mirror descent with Anderson acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    hist_x = []
    hist_Fx = []

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break

        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        t = max(min_t, t_base / (1.0 + 0.02 * it))
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s

        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)

        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_Fx)
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
                M_and = float(fast_conv(a_and).max())
                M_new = float(fast_conv(a_new).max())
                if M_and <= M_new * 1.01:
                    a = a_and
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new

    return best_a


def lp_mirror_backtrack(a, deadline, max_iters, t_init):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    best_M = float(fast_conv(a).max())
    no_imp = 0
    t = t_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        for t_try in [t, t * 0.7, t * 0.5, t * 0.3, 0.1, 0.05, 0.02, 0.01]:
            a_try = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            J_try = compute_J(a_try)
            if J_try < best_J - 1e-13:
                a = a_try
                best_J = J_try
                best_M = float(fast_conv(a).max())
                best_a = a.copy()
                t = t_try
                improved = True
                break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            t *= 0.5
            if no_imp > 25:
                break

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        for sigma in [0.01, 0.03, 0.05, 0.08, 0.12]:
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 0.005))

    inits.append(np.ones(n))

    # Power law shapes
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        inits.append(np.maximum(0.05 + 0.95 * (t ** alpha), 0.01))

    # Cosine shapes
    for f in [1, 2, 3, 5, 8]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            inits.append(np.maximum(1.0 - amp * np.cos(2 * np.pi * f * t), 0.01))

    # Gaussian bumps
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.08, 0.15, 0.25, 0.4]:
            inits.append(np.exp(-((t - center) / width) ** 2) + 0.03)

    # Double/triple bumps
    inits.append(np.exp(-12 * (t - 0.25) ** 2) + np.exp(-12 * (t - 0.75) ** 2) + 0.03)
    inits.append(np.exp(-15 * (t - 0.2) ** 2) + np.exp(-15 * (t - 0.5) ** 2) +
                 np.exp(-15 * (t - 0.8) ** 2) + 0.02)

    # Random smooth (various kernel sizes)
    for k_factor in [15, 8, 4, 2]:
        k = max(3, n // k_factor)
        kern = np.ones(k) / k
        inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))

    # Random Fourier (flat spectrum)
    for _ in range(5):
        phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
        F = np.zeros(n // 2 + 1, dtype=complex)
        F[0] = 2.0
        F[1:] = np.exp(1j * phases[1:])
        a_rand = np.fft.irfft(F, n)[:n]
        inits.append(np.maximum(a_rand, 0.01))

    # Inverse spectrum (concentrated energy at low freq)
    for decay in [0.5, 1.0, 2.0]:
        freqs = np.arange(n // 2 + 1, dtype=float)
        mags = np.exp(-decay * freqs / (n // 4))
        phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
        F = mags * np.exp(1j * phases)
        F[0] = mags[0]
        a_inv = np.fft.irfft(F, n)[:n]
        inits.append(np.maximum(a_inv, 0.01))

    # Triangular
    inits.append(np.maximum(1.0 - 2.0 * np.abs(t - 0.5), 0.01))

    # Sinc-like
    x = 15.0 * (t - 0.5)
    inits.append(np.maximum(np.sinc(x / np.pi) ** 2 + 0.05, 0.01))

    # Sorted random (monotone)
    sorted_rand = np.sort(rng.uniform(0.1, 1.0, n))
    inits.append(sorted_rand)
    inits.append(sorted_rand[::-1])

    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print("  ** J={:.8f} n={} t={:.0f}s".format(s, a.size, el), flush=True)

    def report(self, label):
        if self.seq is not None:
            print("  {}: J={:.8f} n={}".format(label, self.score, self.seq.size), flush=True)


def smooth_grad_descent(a, deadline, max_iters, tau_start, tau_end, lr_init, lr_end):
    """Nesterov smooth gradient descent with FFT-accelerated gradients."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        b = fast_conv(y)
        b_max = float(b.max())
        shifted = tau * (b - b_max)
        exp_b = np.exp(shifted)
        p = exp_b / (exp_b.sum() + 1e-30)

        corr = fft_cross_corr(p, y, n)
        grad = 2.0 * corr

        s_y = y.sum()
        if s_y < 1e-10:
            break
        grad_norm = grad - np.mean(grad)

        a_new = np.maximum(y - lr * grad_norm / s_y, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

        M = float(fast_conv(a).max())
        if M < best_M:
            best_M = M
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print("Parent: n={}, J={:.6f}".format(parent.size, best.score), flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(512)
        best.update(parent)

    # Determine lengths to explore
    lengths = [256, 384, 512, 640, 768, 896, 1024, 1152, 1280, 1408, 1536, 1792, 2048]
    if parent is not None:
        pn = parent.size
        for delta in [-128, -64, -32, 0, 32, 64, 128, 256, 512]:
            n_try = pn + delta
            if 192 <= n_try <= 2560:
                lengths.append(n_try)
        lengths = sorted(set(lengths))

    print("Lengths to explore: {}".format(lengths), flush=True)

    # === PHASE 1: Spectral initialization + smooth optimization ===
    p1_start = time.monotonic()
    print("\n" + "=" * 50, flush=True)
    print("PHASE 1: Spectral Init + Smooth Opt ({:.0f}s budget)".format(300.0), flush=True)
    print("=" * 50, flush=True)

    p1_budget = 300.0
    best_per_length = {}

    # Allocate time: favor longer sequences
    weights = []
    for n in lengths:
        w = (n / 512.0) ** 0.5 / np.sqrt(n) * min(1.0, n / 384.0)
        weights.append(max(w, 0.3))
    total_w = sum(weights)

    for n in lengths:
        t0 = time.monotonic()
        remaining = deadline - 250 - time.monotonic()
        if remaining < 20:
            break
        allot = min(p1_budget * weights[lengths.index(n)] / total_w, remaining * 0.35)
        dl = time.monotonic() + allot
        if time.monotonic() >= dl:
            continue

        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_J_n = float('inf')

        # GS initialization with multiple modes
        for mode in [0, 1, 2]:
            if time.monotonic() > dl - 5:
                break
            a_gs = gs_flat_spectral(inits[0], n_iters=max(80, min(400, int(30000 // n))), mode=mode)
            J_gs = compute_J(a_gs)
            if J_gs < best_J_n:
                best_J_n = J_gs
                best_n = a_gs.copy()

        # Smooth gradient descent from best GS result
        if best_n is not None and time.monotonic() < dl:
            sub_dl = min(time.monotonic() + allot * 0.4, dl)
            max_it = max(100, int(15000 // n))
            a_smooth = smooth_grad_descent(best_n, sub_dl, max_iters=max_it,
                                           tau_start=2.0, tau_end=150.0,
                                           lr_init=0.3, lr_end=0.005)
            J_s = compute_J(a_smooth)
            if J_s < best_J_n:
                best_J_n = J_s
                best_n = a_smooth.copy()

        # L-BFGS refinement
        if best_n is not None and time.monotonic() < dl:
            sub_dl2 = min(time.monotonic() + allot * 0.3, dl)
            a_lbfgs = lbfgs_optimize(best_n, sub_dl2, max_iter=max(50, int(5000 // n)))
            J_l = compute_J(a_lbfgs)
            if J_l < best_J_n:
                best_J_n = J_l
                best_n = a_lbfgs.copy()

        if best_n is not None:
            print("  n={:4d}: J={:.6f} ({:.1f}s)".format(n, best_J_n, time.monotonic() - t0), flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print("Phase 1 time: {:.1f}s".format(time.monotonic() - p1_start), flush=True)

    # === PHASE 2: LP Mirror Descent on best sequence ===
    p2_start = time.monotonic()
    print("\n" + "=" * 50, flush=True)
    print("PHASE 2: LP Mirror Descent (Anderson + Backtrack)", flush=True)
    print("=" * 50, flush=True)

    if best.seq is not None:
        n = best.seq.size
        est_lp_ms = 4.0 * (n / 128.0) ** 2
        print("  n={}, est LP time={:.0f}ms/iter".format(n, est_lp_ms), flush=True)

        remaining_lp = deadline - 100 - time.monotonic()
        max_iters_lp = max(20, int(remaining_lp / max(0.005, est_lp_ms / 1000.0)))
        print("  LP budget={:.0f}s, max_iters={}".format(remaining_lp, max_iters_lp), flush=True)

        # Anderson-accelerated LP (main)
        a_and = lp_mirror_anderson(best.seq,
                                   min(time.monotonic() + remaining_lp * 0.45, deadline - 80),
                                   max_iters=max_iters_lp, anderson_window=6,
                                   t_base=0.35, min_t=0.003)
        best.update(a_and)

        # Second Anderson pass with different params
        if time.monotonic() < deadline - 50:
            a_and2 = lp_mirror_anderson(a_and,
                                        min(time.monotonic() + remaining_lp * 0.25, deadline - 50),
                                        max_iters=max(20, int(max_iters_lp * 0.5)),
                                        anderson_window=5, t_base=0.2, min_t=0.005)
            best.update(a_and2)

        # Backtracking LP (different step size strategy)
        if time.monotonic() < deadline - 30:
            a_bt = lp_mirror_backtrack(a_and2,
                                       min(time.monotonic() + remaining_lp * 0.15, deadline - 30),
                                       max_iters=max(15, int(max_iters_lp * 0.3)), t_init=0.25)
            best.update(a_bt)

    best.report("Phase 2")
    print("Phase 2 time: {:.1f}s".format(time.monotonic() - p2_start), flush=True)

    # === PHASE 3: Polish other promising lengths ===
    p3_start = time.monotonic()
    print("\n" + "=" * 50, flush=True)
    print("PHASE 3: Multi-length LP Polish", flush=True)
    print("=" * 50, flush=True)

    if best_per_length and time.monotonic() < deadline - 40:
        sorted_lengths = sorted(best_per_length.keys(),
                                key=lambda nn: compute_J(best_per_length[nn]))
        for n_try in sorted_lengths[:3]:
            if time.monotonic() > deadline - 30:
                break
            if n_try == best.seq.size and len(sorted_lengths) > 1:
                continue
            a_cand = best_per_length[n_try].copy()
            est_t = 4.0 * (n_try / 128.0) ** 2
            max_it = max(10, int(min(60.0, deadline - 25 - time.monotonic()) / max(0.005, est_t / 1000.0)))
            if time.monotonic() < deadline - 25:
                a_lp = lp_mirror_anderson(a_cand,
                                          min(time.monotonic() + min(50.0, max_it * est_t / 1000.0),
                                              deadline - 25),
                                          max_iters=max_it, anderson_window=5,
                                          t_base=0.3, min_t=0.01)
                best.update(a_lp)

    best.report("Phase 3")
    print("Phase 3 time: {:.1f}s".format(time.monotonic() - p3_start), flush=True)

    # === PHASE 4: Final intensification ===
    p4_start = time.monotonic()
    print("\n" + "=" * 50, flush=True)
    print("PHASE 4: Final Intensification", flush=True)
    print("=" * 50, flush=True)

    if best.seq is not None and time.monotonic() < deadline - 5:
        # Multiple LP passes with decreasing step sizes
        lp_params = [
            (0.2, 0.008, 5, 25),
            (0.12, 0.004, 7, 35),
            (0.06, 0.002, 5, 45),
            (0.03, 0.001, 6, 55),
        ]
        for t_base, min_t, window, iters in lp_params:
            if time.monotonic() > deadline - 4:
                break
            remaining = deadline - 4 - time.monotonic()
            a_final = lp_mirror_anderson(best.seq, time.monotonic() + min(remaining, 40.0),
                                         max_iters=iters, anderson_window=window,
                                         t_base=t_base, min_t=min_t)
            best.update(a_final)

    best.report("Phase 4")
    print("Phase 4 time: {:.1f}s".format(time.monotonic() - p4_start), flush=True)

    # === PHASE 5: Perturbation and length exploration ===
    p5_start = time.monotonic()
    if best.seq is not None and time.monotonic() < deadline - 10:
        print("\n" + "=" * 50, flush=True)
        print("PHASE 5: Perturbation + Length Exploration", flush=True)
        print("=" * 50, flush=True)
        n_best = best.seq.size

        # Perturbation restarts
        for sigma in [0.01, 0.03, 0.05]:
            if time.monotonic() > deadline - 8:
                break
            for trial in range(3):
                a_pert = np.maximum(best.seq + rng.normal(0, sigma, n_best), 0.001)
                a_pert /= a_pert.sum()
                a_ref = lbfgs_optimize(a_pert, min(time.monotonic() + 15, deadline - 5),
                                       max_iter=max(30, int(3000 // n_best)))
                best.update(a_ref)
                if time.monotonic() < deadline - 5:
                    a_lp = lp_mirror_anderson(a_ref, deadline - 3, max_iters=10,
                                              anderson_window=4, t_base=0.2, min_t=0.01)
                    best.update(a_lp)

        # Length exploration around best
        for delta in [32, 64, -32, -64, 128, -128, 256]:
            if time.monotonic() > deadline - 8:
                break
            n_try = n_best + delta
            if n_try < 192 or n_try > 2560:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = smooth_grad_descent(a_ext, min(time.monotonic() + 12, deadline - 5),
                                        max_iters=150, tau_start=5, tau_end=80,
                                        lr_init=0.15, lr_end=0.01)
            best.update(a_ext)
            if time.monotonic() < deadline - 5:
                a_lp = lp_mirror_anderson(a_ext, deadline - 3, max_iters=15,
                                          anderson_window=4, t_base=0.25, min_t=0.01)
                best.update(a_lp)

    best.report("Phase 5")
    print("Phase 5 time: {:.1f}s".format(time.monotonic() - p5_start), flush=True)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(256)
        best.score = compute_J(best.seq)

    print("\n" + "=" * 50, flush=True)
    print("FINAL: J={:.8f} n={} time={:.0f}s".format(best.score, best.seq.size,
                                                      time.monotonic() - best.t0), flush=True)
    print("=" * 50, flush=True)
    return [float(x) for x in best.seq]
