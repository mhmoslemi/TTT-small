"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP fixed-point iteration, active-set subgradient, direct peak reduction,
and high-beta smoothed gradient descent. Explores multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    return a / s

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def lp_fixed_point_iterate(a, num_iters=50, time_budget=60.0, step_sizes=None):
    """Iterate the LP fixed-point: a -> blend(a, LP_best_response(a))."""
    start = time.monotonic()
    n = len(a)
    a = a.copy()
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    if step_sizes is None:
        step_sizes = [0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01, 0.005]
    
    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Solve LP for best response
        g = solve_lp(a, M, time_limit=min(5, time_budget / (num_iters - it)))
        if g is None:
            break
        
        # Try different step sizes
        improved = False
        for t in step_sizes[:min(5, len(step_sizes))]:
            t_eff = t * (1.0 - 0.5 * it / max(num_iters, 1))  # decay step size
            a_try = (1 - t_eff) * a + t_eff * g
            a_try = np.maximum(a_try, 0)
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
        
        if not improved:

            for t in [0.001, 0.002, 0.003]:
                a_try = (1 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break
    
    return best_a

def active_set_subgradient(a, K=5, lr=0.1, num_iters=200, time_budget=30.0):
    """Subgradient method focusing on top K active lags."""
    start = time.monotonic()
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Get top K lags
        top_k = np.argsort(conv)[::-1][:K]
        M = conv[top_k[0]]
        
        # Compute subgradient: average of gradients at top K lags
        # grad[m] = (2/K) * sum_{k in top_k} a[k-m] (if valid)
        grad = np.zeros(n)
        for k in top_k:
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    grad[m] += a[j]
        grad *= 2.0 / K
        
        # With sum=1 constraint, projected gradient:
        # a_new = a - lr * (grad - mean(grad))
        grad_centered = grad - np.mean(grad)
        
        # Adaptive learning rate
        lr_eff = lr / (1 + it * 0.01)
        a_try = a - lr_eff * grad_centered
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-10:
            lr_eff *= 0.1
            a_try = a - lr_eff * grad_centered
            a_try = np.maximum(a_try, 0)
            s = np.sum(a_try)
            if s < 1e-10:
                continue
        a_try = a_try / s
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
    
    return best_a

def direct_peak_reduction(a, num_iters=100, time_budget=30.0):
    """Directly try to reduce the peak by moving mass from peak-contributing indices."""
    start = time.monotonic()
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        M = conv[peak_idx]
        
        # Find indices contributing to the peak
        i_min = max(0, peak_idx - (n - 1))
        i_max = min(peak_idx, n - 1)
        
        # For each pair (i, j) with i+j = peak_idx, try to reduce contribution
        # by moving mass from a[i] or a[j] to other positions
        improved = False
        for i in range(i_min, i_max + 1):
            j = peak_idx - i
            if j < 0 or j >= n:
                continue
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue
            
            # Try moving a small fraction of a[i] to a random other position
            for target in [0, n-1, (n//2)]:
                if target == i or target == j:
                    continue
                for frac in [0.001, 0.005, 0.01, 0.02]:
                    delta = frac * a[i]
                    a_try = a.copy()
                    a_try[i] -= delta
                    a_try[target] += delta
                    if a_try[i] < 0:
                        continue
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try the opposite: move mass TO the peak-contributing indices
            # to create destructive interference
            for i in range(i_min, i_max + 1):
                j = peak_idx - i
                if j < 0 or j >= n:
                    continue
                # Find a non-contributing index with high value
                for source in range(n):
                    if source == i or source == j:
                        continue
                    # Check if source contributes to the peak
                    if (source + (peak_idx - source)) == peak_idx and 0 <= peak_idx - source < n:
                        # source also contributes, skip
                        pass
                    for frac in [0.001, 0.005, 0.01]:
                        delta = frac * a[source]
                        a_try = a.copy()
                        a_try[source] -= delta
                        a_try[i] += delta
                        if a_try[source] < 0:
                            continue
                        a_try = normalize(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved and it > 30:
            break
    
    return best_a

def smooth_obj_grad(a, beta):
    """Smoothed objective using logsumexp."""
    n = len(a)
    conv = compute_conv(a)
    s = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (s * s)
    
    # Gradient via cross-correlation
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    
    grad_obj = (2.0 * n / (s * s)) * (grad_ls - ls / s)
    return obj, grad_obj

def adam_log_opt(a, max_iters=5000, time_budget=50.0, beta_start=50.0,
                 beta_end=500.0, lr_start=0.02, lr_decay=0.99995):
    """Adam in log-space with high beta (sharp focus on peak)."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    a = np.maximum(a, 1e-15)
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_c = x - np.max(x)
        a = np.exp(x_c)
        a = a / np.sum(a)
        
        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a
        
        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 100 == 0:
            x -= np.mean(x)
        
        if it % 50 == 0:
            a_check = normalize(np.exp(x - np.max(x)))
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()
    
    a_check = normalize(np.exp(x - np.max(x)))
    true_obj = compute_obj(a_check)
    if true_obj < best_obj:
        best_obj = true_obj
        best_a = a_check.copy()
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(x_new, x_old, a)
    return a_new

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP fixed-point iteration, active-set subgradient on top peaks,
high-beta smoothed Adam, direct peak reduction, spectral reweighting,
multi-n exploration, and random restarts.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    return a / s


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (convolve(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=8):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_fixed_point_iterate(a, num_iters=60, time_budget=60.0):
    """Iterate LP fixed-point with adaptive step sizes."""
    start = time.monotonic()
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        remaining = time_budget - (time.monotonic() - start)
        lt = max(1.0, min(6.0, remaining / max(num_iters - it, 1)))
        g = solve_lp(a, M, time_limit=lt)
        if g is None:
            break

        # Adaptive step sizes: start larger, shrink over iterations
        base_t = 0.4 * (0.85 ** it)
        candidates = [base_t, base_t * 0.5, base_t * 0.25,
                      base_t * 0.1, 0.01, 0.005, 0.002, 0.001]

        improved = False
        for t in candidates:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved and it > 20:
            break

    return best_a


def active_set_subgradient(a, K=3, lr=0.05, num_iters=500, time_budget=30.0):
    """Subgradient method focusing on top K active convolution lags."""
    start = time.monotonic()
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:K]

        # Compute subgradient: for each top lag k, grad_k[m] = 2*a[k-m] if valid
        grad = np.zeros(n)
        for k in top_k:
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    grad[m] += a[j]
        grad *= (2.0 / K)

        # Projected gradient with sum=1 constraint
        grad_centered = grad - np.mean(grad)
        lr_eff = lr * (0.995 ** it)
        a_try = a - lr_eff * grad_centered
        a_try = np.maximum(a_try, 1e-15)
        a_try = normalize(a_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()

    return best_a


def direct_peak_reduction(a, num_iters=200, time_budget=30.0, rng_seed=0):

                        best_a = a_try.copy()
                        a = a_try.copy()

def spectral_flatten(a, num_iters=100, time_budget=20.0, alpha_start=0.8, alpha_end=0.3):
    """Iterative spectral flattening: compress |FFT(a)| dynamic range."""
    start = time.monotonic()
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(num_iters - 1, 1)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        # Compute FFT
        F = fft(a)
        magnitude = np.abs(F)
        phase = np.angle(F)

        # Compress dynamic range
        mag_new = np.power(magnitude + 1e-10, alpha)

        # Reconstruct
        F_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(F_new))
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a


def smooth_obj_grad(a, beta):
    """Smoothed objective (logsumexp) and gradient."""
    n = len(a)
    conv = compute_conv(a)
    s = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (s * s)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]

    grad_obj = (2.0 * n / (s * s)) * (grad_ls - ls / s)
    return obj, grad_obj


def adam_log_opt(a, max_iters=5000, time_budget=50.0, beta_start=50.0,
                 beta_end=500.0, lr_start=0.02, lr_decay=0.99995):
    """Adam in log-space with temperature annealing."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    a = np.maximum(a, 1e-15)

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        x_c = x - np.max(x)
        a = np.exp(x_c)
        a = a / np.sum(a)

        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a

        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m = b1 * m + (1 - b1) * grad_x
        v = b2 * v + (1 - b2) * (grad_x * grad_x)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 200 == 0:
            x -= np.mean(x)

        if it % 50 == 0:
            a_check = normalize(np.exp(x - np.max(x)))
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()

    a_check = normalize(np.exp(x - np.max(x)))
    true_obj = compute_obj(a_check)
    if true_obj < best_obj:
        best_obj = true_obj
        best_a = a_check.copy()

    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, a)

                                  beta_start=50.0, beta_end=500.0,
                                  lr_start=0.02, lr_decay=0.99998)
            obj = compute_obj(a_opt)

            if len(a_opt) <= 1000 and time.monotonic() < deadline - 8:
                a_lp = lp_fixed_point_iterate(a_opt, num_iters=10, time_budget=8.0)
                obj_lp = compute_obj(a_lp)
                if obj_lp < obj:
                    obj = obj_lp
                    a_opt = a_lp

            if obj < best_score - 1e-10:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.10f} *** UPDATE")
            else:
                print(f"  Restart {restart}: {obj:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
