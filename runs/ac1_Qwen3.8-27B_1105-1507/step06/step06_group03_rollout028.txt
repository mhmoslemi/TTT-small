            kq = 2 * i
            lo_k = max(0, i)
            hi_k = min(2 * n - 2, i + n - 1)
            sl = a[lo_k - i:hi_k - i + 1]

            # Generate candidate values
            if old < 1e-8:
                cands = np.array([0.0, 1e-5, 1e-4, 3e-4, 1e-3, 3e-3])
            else:
                # Mix of multiplicative and additive perturbation
                mults = np.array([0.2, 0.5, 0.8, 1.0, 1.25, 1.6, 2.0, 0.05])
                cands = np.clip(old * mults, 0.0, 1.0)
                # Add a few golden-section style points around old
                hw = old * 0.5
                cands = np.unique(np.concatenate([cands,
                                                  [old - hw, old - hw * 0.5, old + hw * 0.5, old + hw]]))
                cands = np.clip(cands, 0.0, 1.0)

            best_x = old
            best_local = best_obj

            for x in cands:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-16:
                    local_obj = best_obj
                else:
                    # Apply incremental update
                    conv[lo_k:hi_k + 1] += 2.0 * delta * sl
                    if kq <= 2 * n - 2:
                        conv[kq] += delta * delta

                    new_s = s - old + x
                    if new_s < 1e-10:
                        local_obj = float('inf')
                    else:
                        local_obj = 2.0 * n * conv.max() / (new_s * new_s)

                    # Revert
                    conv[lo_k:hi_k + 1] -= 2.0 * delta * sl
                    if kq <= 2 * n - 2:
                        conv[kq] -= delta * delta

                if local_obj < best_local - 1e-15:
                    best_local = local_obj
                    best_x = x

            if best_x != old and abs(best_x - old) > 1e-14:
                delta = best_x - old
                conv[lo_k:hi_k + 1] += 2.0 * delta * sl
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = best_x
                s = s - old + best_x
                sweep_improved = True

        cur_s = a.sum()
        if cur_s < 1e-10:
            break
        cur_obj = 2.0 * n * conv.max() / (cur_s * cur_s)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Perturb to escape
                a = best_a + rng.normal(0, 0.01, n)
                a = np.maximum(a, 0)
                a /= a.sum()
                conv = fast_conv(a)
                s = 1.0
                best_obj = 2.0 * n * conv.max()
                no_improve = 0

        if verbose:
            print(f"    CD: obj={best_obj:.10f} no_improve={no_improve}", flush=True)

    best_a = np.maximum(best_a, 0)
    bs = best_a.sum()
    if bs > 1e-10:
        best_a /= bs
    final_obj = compute_J(best_a)
    return best_a, final_obj


def coord_descent_fine(a, deadline, seed=123, verbose=False):
    """
    Finer-grained coordinate descent using binary search on the 1D objective.
    For each coordinate, finds the optimal value via bisection on the derivative.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a).copy()
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    def eval_coord(x, old, i):
        """Evaluate J with a[i] changed from old to x."""
        delta = x - old
        if abs(delta) < 1e-16:
            return 2.0 * n * conv.max() / (s * s)
        kq = 2 * i
        lo_k = max(0, i)
        hi_k = min(2 * n - 2, i + n - 1)
        sl = a[lo_k - i:hi_k - i + 1]

        conv[lo_k:hi_k + 1] += 2.0 * delta * sl
        if kq <= 2 * n - 2:
            conv[kq] += delta * delta
        new_s = s - old + x
        if new_s < 1e-10:
            obj = float('inf')
        else:
            obj = 2.0 * n * conv.max() / (new_s * new_s)
        conv[lo_k:hi_k + 1] -= 2.0 * delta * sl
        if kq <= 2 * n - 2:
            conv[kq] -= delta * delta
        return obj

    while time.monotonic() < deadline:
        order = rng.permutation(n)
        improved = False

        for i in order:
            old = a[i]
            if old < 1e-12:
                # Just try setting to a small value
                for x in [0.0, 5e-5, 2e-4, 1e-3]:
                    obj_x = eval_coord(x, old, i)
                    if obj_x < best_obj - 1e-14:
                        # Accept
                        delta = x - old
                        kq = 2 * i
                        lo_k = max(0, i)
                        hi_k = min(2 * n - 2, i + n - 1)
                        sl = a[lo_k - i:hi_k - i + 1]
                        conv[lo_k:hi_k + 1] += 2.0 * delta * sl
                        if kq <= 2 * n - 2:
                            conv[kq] += delta * delta
                        a[i] = x
                        s = s - old + x
                        best_obj = obj_x
                        improved = True
                        break
                continue

            # Golden section search
            hw = max(old * 1.5, 0.02)
            lo = max(0.0, old - hw)
            hi = min(1.0, old + hw)

            # Evaluate at golden section points
            for _gs in range(12):
                d = hi - lo
                if d < 1e-10:
                    break
                x1 = lo + 0.382 * d
                x2 = lo + 0.618 * d
                f1 = eval_coord(x1, old, i)
                f2 = eval_coord(x2, old, i)
                if f1 < f2:
                    hi = x2
                else:
                    lo = x1

            x_opt = (lo + hi) / 2.0
            f_opt = eval_coord(x_opt, old, i)

            if f_opt < best_obj - 1e-13 and abs(x_opt - old) > 1e-10:
                delta = x_opt - old
                kq = 2 * i
                lo_k = max(0, i)
                hi_k = min(2 * n - 2, i + n - 1)
                sl = a[lo_k - i:hi_k - i + 1]
                conv[lo_k:hi_k + 1] += 2.0 * delta * sl
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = x_opt
                s = s - old + x_opt
                best_obj = f_opt
                improved = True

        # Check for stall
        cur_s = a.sum()
        if cur_s < 1e-10:
            break
        cur_obj = 2.0 * n * conv.max() / (cur_s * cur_s)
        if cur_obj < best_obj:
            best_a = a.copy()
            best_obj = cur_obj
        if not improved:
            no_improve += 1
            if no_improve > 80:
                # Add perturbation
                a = best_a + rng.normal(0, 0.02, n)
                a = np.maximum(a, 0)
                a /= a.sum()
                conv = fast_conv(a)
                s = 1.0
                no_improve = 0
        else:
            no_improve = 0

    best_a = np.maximum(best_a, 0)
    bs = best_a.sum()
    if bs > 1e-10:
        best_a /= bs
    return best_a, compute_J(best_a)


def sca_lp_step(a, deadline, max_iters=200, verbose=False):
    """
    SCA-LP: At each step, linearize and solve LP for best direction.
    LP: min t s.t. 2*(a*g)[k] - (a*a)[k] <= t, sum(g)=1, g>=0
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv_a = fast_conv(a)
        M = conv_a.max()

        # Build LP constraint matrix
        # For each k in 0..2n-2: sum_j 2*a[k-j]*g[j] <= t + conv_a[k]
        # i.e., sum_j 2*a_ext[k-j]*g[j] - t <= conv_a[k]
        rows = []
        cols = []
        vals = []
        rhs = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(len(rhs))
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(len(rhs))
            cols.append(n)  # t variable at index n
            vals.append(-1.0)
            rhs.append(conv_a[k])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t

        bounds = [(0, None)] * n + [(-np.inf, np.inf)]
        A_eq = np.zeros((1, n + 1))
        for j in range(n):
            A_eq[0, j] = 1.0
        b_eq = np.array([1.0])

        try:
            r = linprog(c_obj, A_ub=A, b_ub=rhs, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    # Line search
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        ts = a_try.sum()
                        if ts < 1e-10:
                            continue
                        a_try /= ts
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-13:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 10 == 0:
            print(f"    SCA-LP it {it}: obj={best_obj:.10f}", flush=True)

        if no_improve > 30:
            break

    return best_a, best_obj


def subgrad_redistribute(a, n_steps, deadline, verbose=False):
    """
    Subgradient-guided mass redistribution.
    Moves mass from positions contributing to the max convolution
    to positions that don't, using subgradient information.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    rng = np.random.default_rng(999)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        max_val = conv.max()
        max_idx = np.where(conv >= max_val - 1e-10)[0]

        # Compute subgradient: for each coordinate i,
        # d(conv[k])/da[i] = a[k-i] (if k-i in [0,n-1] and 2i != k)
        #                       2*a[i] (if 2i == k)
        # Subgradient of max is average over achieving k's
        subgrad = np.zeros(n)
        for k in max_idx:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    subgrad[i] += a[ki]
                if 2 * i == k:
                    subgrad[i] += a[i]  # extra from the quadratic term
        subgrad /= len(max_idx)

        # Direction: decrease coordinates with high subgrad, increase those with low
        # Move from high-subgrad to low-subgrad positions
        sg_mean = subgrad.mean()
        # Shifted gradient (zero mean for simplex)
        direction = subgrad - sg_mean

        # Small step
        alpha = 0.01 / (step + 1)
        a_new = a - alpha * direction
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < best_obj - 1e-14:
            a = a_new
            best_obj = J_new
            best_a = a_new.copy()
        elif J_new < best_obj:
            a = a_new
        else:
            # Try smaller step
            a_new2 = a - alpha * 0.1 * direction
            a_new2 = np.maximum(a_new2, 0)
            a_new2 = a_new2 / a_new2.sum()
            J_new2 = compute_J(a_new2)
            if J_new2 < best_obj:
                a = a_new2
                best_obj = J_new2
                best_a = a_new2.copy()

        if verbose and step % 20 == 0:
            print(f"    Subgrad step {step}: obj={best_obj:.10f}", flush=True)

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns
    for freq in [1, 2, 3, 4, 6]:
        for amp in [0.3, 0.5, 0.8]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    # Gaussian bumps
    for center in [0.25, 0.5, 0.75]:
        for width in [0.08, 0.15, 0.25, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.03))

    # Power law
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(5):
        raw = rng.uniform(0, 1, n)
        smooth = np.convolve(raw, kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    # Weyl-like: fractional parts
    for base in [np.sqrt(2), np.sqrt(3), (1 + np.sqrt(5)) / 2]:
        vals = np.abs(np.sin(2 * np.pi * base * t))
        inits.append(project_simplex(vals))

    return inits


def solve_pipeline(a, deadline, verbose=False):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return best_a

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 1: Fast CD (35%)
    t1 = time.monotonic() + 0.35 * remaining
    a_cd, obj_cd = coord_descent(a, deadline=t1, seed=42)
    try_update(a_cd)
    if verbose:
        print(f"      After CD: {obj_cd:.10f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 2: Fine CD with golden section (25%)
    t2 = time.monotonic() + 0.25 * remaining
    a_fine, obj_fine = coord_descent_fine(best_a, deadline=t2, seed=77)
    try_update(a_fine)
    if verbose:
        print(f"      After Fine CD: {obj_fine:.10f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 3: SCA-LP (25%)
    t3 = time.monotonic() + 0.25 * remaining
    a_lp, obj_lp = sca_lp_step(best_a, deadline=t3, max_iters=300)
    try_update(a_lp)
    if verbose:
        print(f"      After SCA-LP: {obj_lp:.10f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 4: Subgradient + final CD (15%)
    t4a = time.monotonic() + 0.08 * remaining
    a_sg, obj_sg = subgrad_redistribute(best_a, n_steps=50, deadline=t4a)
    try_update(a_sg)

    t4b = deadline - 5
    a_final, obj_final = coord_descent(best_a, deadline=t4b, seed=314)
    try_update(a_final)

    return best_a, best_obj
