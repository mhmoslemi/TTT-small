"""
Improved optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Golden-section coordinate descent, (2) Active-set LP with top-m peaks,
(3) Frank-Wolfe with away steps, (4) Multi-length search with resampling.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix
import math

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def golden_section_1d(a, i, n, conv, s, x_lo, x_hi, tol=1e-10, max_iter=50):
    """
    Golden section search to find optimal a[i] = x that minimizes J.
    Uses incremental convolution updates.
    Returns (best_x, best_j, updated_conv, updated_s).
    """
    # Precompute the slice of a that affects conv when a[i] changes
    lo = max(0, i)
    hi = min(2 * n - 2, i + n - 1)
    a_slice = a[lo - i:hi - i + 1].copy()
    kq = 2 * i
    valid_kq = kq <= 2 * n - 2
    
    # Remove current contribution
    old = a[i]
    conv[lo:hi+1] -= 2.0 * old * a_slice
    if valid_kq:
        conv[kq] -= old * old
    
    # Now conv has a[i] = 0 contribution
    # For new value x: conv[lo:hi+1] += 2*x*a_slice, conv[kq] += x*x
    # s_new = s - old + x
    
    gr = (math.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    
    c = x_hi - gr * (x_hi - x_lo)
    d = x_lo + gr * (x_hi - x_lo)
    
    def eval_at(x):
        if valid_kq:
            conv[lo:hi+1] += 2.0 * x * a_slice
            conv[kq] += x * x
            s_new = s - old + x
            j = 2.0 * n * conv.max() / (s_new * s_new) if s_new > 1e-10 else float('inf')
            # Undo
            conv[lo:hi+1] -= 2.0 * x * a_slice
            conv[kq] -= x * x
        else:
            conv[lo:hi+1] += 2.0 * x * a_slice
            s_new = s - old + x
            j = 2.0 * n * conv.max() / (s_new * s_new) if s_new > 1e-10 else float('inf')
            conv[lo:hi+1] -= 2.0 * x * a_slice
        return j
    
    fc = eval_at(c)
    fd = eval_at(d)
    
    best_x = c if fc < fd else d
    best_j = min(fc, fd)
    
    for _ in range(max_iter):
        if x_hi - x_lo < tol:
            break
        if fc < fd:
            x_hi = d
            d = c
            fd = fc
            c = x_hi - gr * (x_hi - x_lo)
            fc = eval_at(c)
        else:
            x_lo = c
            c = d
            fc = fd
            d = x_lo + gr * (x_hi - x_lo)
            fd = eval_at(d)
        
        if fc < best_j:
            best_j = fc
            best_x = c
        if fd < best_j:
            best_j = fd
            best_x = d
    
    # Restore conv with best_x
    delta = best_x - old  # But we already removed old, so we add best_x
    # Actually conv currently has a[i]=0, so we add best_x
    conv[lo:hi+1] += 2.0 * best_x * a_slice
    if valid_kq:
        conv[kq] += best_x * best_x
    
    s_new = s - old + best_x
    return best_x, best_j, conv, s_new

def coord_descent_golden(a, max_sweeps, deadline, seed=42, verbose=False):
    """Coordinate descent with golden section 1D optimization."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 200
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            old = a[i]
            # Search range: [0, 2*old + 0.01] or [max(0, old-0.05*old), old*2+0.01]
            x_lo = 0.0
            x_hi = max(2.0 * old + 0.01, 0.02)
            
            new_x, new_j, conv, s = golden_section_1d(a, i, n, conv, s, x_lo, x_hi)
            
            if abs(new_x - old) > 1e-12:
                a[i] = new_x
                improved = True
        
        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        if verbose and sweep % 100 == 0:
            print(f"    CD-G sweep {sweep}: obj={best_obj:.10f}", flush=True)
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a

def active_set_lp(a, top_m, max_iters, deadline, verbose=False):
    """
    LP that only constrains the top-m autocorrelation lags.
    Minimize t s.t. 2*(a*g)[k] - c[k] <= t for k in active set, sum(g)=1, g>=0.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = c.max()
        
        # Find top-m lags
        top_indices = np.argsort(c)[-top_m:]
        
        rows = []
        cols = []
        vals = []
        b_ub = []
        
        for k in top_indices:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])
        
        A = coo_matrix((vals, (rows, cols)), shape=(len(top_indices), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), ([0]*n, list(range(n)))), shape=(1, n + 1)).tocsr()
        b_eq = [1.0]
        
        try:
            r = linprog(c_obj, A_ub=A, b_ub=np.array(b_ub), A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs', options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                if g.sum() > 1e-10:
                    g /= g.sum()
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try.copy()
                            best_obj = J_try
                            best_a = a.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1
        
        if no_improve > 15:
            break
        
        if verbose and it % 5 == 0:
            print(f"    AS-LP it {it}: obj={best_obj:.10f}", flush=True)
    
    return best_a, best_obj

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))
    
    inits.append(np.ones(n) / n)
    
    # Various smooth patterns
    for freq in [1, 2, 3, 4, 5, 7]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            if amp <= 0.6:
                inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))
    
    # Gaussian
    for center in [0.1, 0.3, 0.5, 0.7, 0.9]:
        for width in [0.05, 0.15, 0.3, 0.5]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.02))
    
    # Power law
    for alpha in [0.3, 0.5, 1.0, 1.5, 2.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))
    
    # Smooth random
    ksize = max(3, n // 8)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.02))
    
    # Banded (nonzero in a band)
    for band_w in [0.3, 0.5, 0.7]:
        mask = np.zeros(n)
        start = int(n * (1 - band_w) / 2)
        end = int(n * (1 + band_w) / 2)
        mask[start:end] = 1
        inits.append(project_simplex(mask * (0.5 + 0.5 * rng.random(n))))
    
    # Random with structure
    for _ in range(4):
        base = project_simplex(1 + 0.5 * np.cos(2 * np.pi * rng.integers(1, 5) * t))
        inits.append(project_simplex(base + rng.normal(0, 0.1, n)))
    
    return inits

Improved optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Combines: (1) Golden-section coordinate descent with incremental conv,
(2) Active-set LP constraining top-m autocorrelation peaks,
(3) Frank-Wolfe with away/pairwise steps,
(4) Smooth Adam with log-sum-exp, (5) Multi-length exploration.

import numpy as np
import time
import math
from scipy.optimize import linprog

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def golden_section_1d(a, i, n, conv, s, x_lo, x_hi, max_iter=35):
    """Golden section search for optimal a[i]. Returns (best_x, conv, s_new)."""
    lo = max(0, i)
    hi = min(2 * n - 2, i + n - 1)
    a_slice = a[lo - i:hi - i + 1].copy()
    kq = 2 * i
    valid_kq = kq <= 2 * n - 2

    old = a[i]
    # Remove old contribution
    conv[lo:hi + 1] -= 2.0 * old * a_slice
    if valid_kq:
        conv[kq] -= old * old

    gr = (math.sqrt(5.0) - 1.0) / 2.0

    def eval_at(x):
        if valid_kq:
            conv[lo:hi + 1] += 2.0 * x * a_slice
            conv[kq] += x * x
            s_new = s - old + x
            if s_new < 1e-10:
                result = float('inf')
            else:
                result = 2.0 * n * conv.max() / (s_new * s_new)
            conv[lo:hi + 1] -= 2.0 * x * a_slice
            conv[kq] -= x * x
        else:
            conv[lo:hi + 1] += 2.0 * x * a_slice
            s_new = s - old + x
            if s_new < 1e-10:
                result = float('inf')
            else:
                result = 2.0 * n * conv.max() / (s_new * s_new)
            conv[lo:hi + 1] -= 2.0 * x * a_slice
        return result

    c = x_hi - gr * (x_hi - x_lo)
    d = x_lo + gr * (x_hi - x_lo)
    fc = eval_at(c)
    fd = eval_at(d)

    for _ in range(max_iter):
        if x_hi - x_lo < 1e-12:
            break
        if fc < fd:
            x_hi = d
            d = c
            fd = fc
            c = x_hi - gr * (x_hi - x_lo)
            fc = eval_at(c)
        else:
            x_lo = c
            c = d
            fc = fd
            d = x_lo + gr * (x_hi - x_lo)
            fd = eval_at(d)

    best_x = c if fc < fd else d

    # Apply best
    conv[lo:hi + 1] += 2.0 * best_x * a_slice
    if valid_kq:
        conv[kq] += best_x * best_x
    s_new = s - old + best_x
    return best_x, conv, s_new


def coord_descent_golden(a, max_sweeps, deadline, seed=42, verbose=False):
    """Coordinate descent with golden section 1D optimization."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 150

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)

        for i in order:
            old = a[i]
            x_lo = 0.0
            # Adaptive upper bound
            x_hi = max(3.0 * old + 0.005, 0.01)

            best_x, conv, s = golden_section_1d(a, i, n, conv, s, x_lo, x_hi)

            if abs(best_x - old) > 1e-12:
                a[i] = best_x

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 200 == 0:
            print(f"    CD-G sweep {sweep}: obj={best_obj:.10f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def active_set_lp(a, top_m, max_iters, deadline, verbose=False):
    """LP constraining only top-m autocorrelation lags."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Find top-m lags
        if top_m >= 2 * n - 1:
            top_indices = np.arange(2 * n - 1)
        else:
            top_indices = np.argsort(c)[-top_m:]

        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in top_indices:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub))
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(len(b_ub))
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(len(b_ub), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), ([0] * n, list(range(n)))), shape=(1, n + 1)).tocsr()
        b_eq = np.array([1.0])

        try:
            r = linprog(c_obj, A_ub=A, b_ub=np.array(b_ub), A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs', options={'time_limit': 20, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                if g.sum() > 1e-10:
                    g /= g.sum()
                    found = False
                    for frac in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15]:
                        a_try = (1.0 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-13:
                            a = a_try.copy()
                            best_obj = J_try
                            best_a = a.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 12:
            break

        if verbose and it % 5 == 0:
            print(f"    AS-LP it {it}: obj={best_obj:.10f}", flush=True)

    return best_a, best_obj


def smooth_adam_v2(a, max_iters, deadline, tau_init=15.0, tau_max=80.0,
                   lr_init=0.04, verbose=False):
    """Adam on simplex with log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_init + (tau_max - tau_init) * frac

        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()

        # Gradient via FFT
        a_rev = a[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        conv_full = np.fft.irfft(A_fft * W_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx = n - 1 + i
            if 0 <= idx < len(conv_full):
                grad[i] = conv_full[idx]

        grad = grad - grad.mean()

        t_count += 1
        lr = lr_init / (1.0 + 0.05 * t_count ** 0.5)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 60:
                rng = np.random.default_rng(it * 13 + 7)
                a = best_a + rng.normal(0, 0.015, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

        if verbose and it % 300 == 0:
            print(f"    Adam2 it {it}: tau={tau:.1f} obj={best_obj:.10f}", flush=True)

    return best_a, best_obj


def frank_wolfe_away(a, max_iters, deadline, tau=40.0, verbose=False):
    """Frank-Wolfe with away steps."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    support = [i for i in range(n) if a[i] > 1e-6]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()

        a_rev = a[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        conv_full = np.fft.irfft(A_fft * W_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx = n - 1 + i
            if 0 <= idx < len(conv_full):
                grad[i] = conv_full[idx]
        grad = grad - grad.mean()

        step_taken = False

        # Away step
        if len(support) > 1:
            support_grads = np.array([grad[s] for s in support])
            away_idx = support[np.argmax(support_grads)]
            if grad[away_idx] > 0 and a[away_idx] > 1e-5:
                sigma = a[away_idx]
                a_new = a.copy()
                a_new[away_idx] = 0.0
                denom = 1.0 - sigma
                if denom > 1e-8:
                    a_new = a_new / denom
                    a_new = np.maximum(a_new, 0)
                    if a_new.sum() > 1e-10:
                        a_new /= a_new.sum()
                        J_new = compute_J(a_new)
                        if J_new < best_obj - 1e-13:
                            a = a_new
                            best_obj = J_new
                            best_a = a.copy()
                            no_improve = 0
                            step_taken = True
                            if away_idx in support:
                                support.remove(away_idx)

        # FW step
        if not step_taken:
            fw_idx = np.argmin(grad)
            for t in [0.1, 0.2, 0.35, 0.5, 0.7, 1.0]:
                a_new = (1.0 - t) * a
                a_new[fw_idx] += t
                a_new = np.maximum(a_new, 0)
                if a_new.sum() > 1e-10:
                    a_new /= a_new.sum()
                    J_new = compute_J(a_new)
                    if J_new < best_obj - 1e-13:
                        a = a_new
                        best_obj = J_new
                        best_a = a.copy()
                        no_improve = 0
                        if fw_idx not in support:
                            support.append(fw_idx)
                        step_taken = True
                        break
            if not step_taken:
                no_improve += 1
        else:
            no_improve = 0

        if no_improve > 40:
            break

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.01, 0.03, 0.06, 0.12, 0.25]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))

    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3, 4, 5, 7, 10]:
        for amp in [0.15, 0.3, 0.5, 0.7]:
            inits.append(project_simplex(1.0 + amp * np.cos(2.0 * np.pi * freq * t)))
            if amp <= 0.4:
                inits.append(project_simplex(1.0 + amp * np.sin(2.0 * np.pi * freq * t)))

    for center in [0.1, 0.3, 0.5, 0.7, 0.9]:
        for width in [0.05, 0.1, 0.2, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.01))

    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1.0 - t) + 0.01) ** alpha))

    ksize = max(3, n // 6)
    kernel = np.ones(ksize) / ksize
    for _ in range(8):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    # Banded
    for bw in [0.3, 0.5, 0.7, 0.9]:
        mask = np.zeros(n)
        s0 = int(n * (1.0 - bw) / 2.0)
        s1 = int(n * (1.0 + bw) / 2.0)
        mask[s0:s1] = 1.0
        inits.append(project_simplex(mask * (0.5 + 0.5 * rng.random(n))))

    # Two-peak
    for c1 in [0.2, 0.4]:
        for c2 in [0.6, 0.8]:
            inits.append(project_simplex(
                np.exp(-((t - c1) / 0.15) ** 2) + np.exp(-((t - c2) / 0.15) ** 2) + 0.01))

    # Alternating smooth
    inits.append(project_simplex(1.0 + 0.3 * np.cos(2.0 * np.pi * 4.0 * t) *
                                  np.cos(2.0 * np.pi * 1.0 * t)))
    inits.append(project_simplex(1.0 + 0.2 * np.sin(4.0 * np.pi * t) +
                                  0.1 * np.sin(8.0 * np.pi * t)))

    return inits


def full_optimize(a, deadline, verbose=False):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return a_new
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 1: Golden section CD
    t1 = time.monotonic() + 0.35 * remaining
    a = coord_descent_golden(a, max_sweeps=8000, deadline=t1, seed=42)
    a = try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 2: Active-set LP
    t2 = time.monotonic() + 0.25 * remaining
    a_lp, obj_lp = active_set_lp(a, top_m=25, max_iters=120, deadline=t2)
    if obj_lp < best_obj:
        best_a = a_lp.copy()
        best_obj = obj_lp
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 3: Adam
    t3 = time.monotonic() + 0.25 * remaining
    a_adam, obj_adam = smooth_adam_v2(a, max_iters=4000, deadline=t3,
                                       tau_init=12.0, tau_max=70.0, lr_init=0.03)
    if obj_adam < best_obj:
        best_a = a_adam.copy()
        best_obj = obj_adam
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 4: FW + final CD
    t4a = time.monotonic() + 0.15 * remaining
    a_fw, obj_fw = frank_wolfe_away(a, max_iters=300, deadline=t4a, tau=30.0)
    if obj_fw < best_obj:
        best_a = a_fw.copy()
        best_obj = obj_fw

    t4b = deadline - 3
    a = coord_descent_golden(best_a, max_sweeps=5000, deadline=t4b, seed=77)
    a = try_update(a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: diverse inits, golden-section CD, active-set LP,
    FW-away, Adam, multi-length exploration. Targets J < 1.5027.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        update_best(parent)

    # === PHASE 1: Optimize at parent length with diverse inits ===
    n_parent = len(parent)
    phase1_deadline = t_start + min(550.0, 0.38 * budget_s)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-init at n={n_parent} (budget {phase1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    init_scores = [(compute_J(init), idx) for idx, init in enumerate(inits)]
    init_scores.sort()
    n_inits_to_try = min(len(inits), 14)
    prioritized_inits = [inits[idx] for _, idx in init_scores[:n_inits_to_try]]

    per_init_time = max(20.0, (phase1_deadline - time.monotonic()) / max(1, n_inits_to_try) - 3)

    for idx, a0 in enumerate(prioritized_inits):
        rem = phase1_deadline - time.monotonic()
        if rem < 25:
            break
        alloc = min(per_init_time, rem - 5)
        if alloc < 20:
            break

        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt, obj_opt = full_optimize(a0, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.10f} n={n_parent}", flush=True)

    # === PHASE 2: Multi-length exploration ===
    phase2_deadline = t_start + min(1050.0, 0.70 * budget_s)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length (remaining {phase2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidate_ns = []
    base_n = n_parent
    for delta in [-128, -96, -64, -32, -16, 16, 32, 64, 96, 128, 192, 256, 384]:
        nn = base_n + delta
        if 192 <= nn <= 2048 and nn % 32 == 0 and nn not in candidate_ns:
            candidate_ns.append(nn)
    for nn in [512, 640, 768, 896, 1024, 1152, 1280, 1408, 1536]:
        if nn not in candidate_ns and 192 <= nn <= 2048:
            candidate_ns.append(nn)

    candidate_ns = [nn for nn in candidate_ns if nn != n_parent]
    candidate_ns.sort(key=lambda nn: abs(nn - base_n))

    for new_n in candidate_ns[:6]:
        rem = phase2_deadline - time.monotonic()
        if rem < 55:
            break
        alloc = min(90.0, rem - 20)
        if alloc < 35:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = full_optimize(a_ext, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.10f}", flush=True)

    # === PHASE 3: Intensive exploitation ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive exploitation (remaining {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        n_best = len(best_seq)
        for round_num in range(5):
            rem = deadline - time.monotonic()
            if rem < 25:
                break

            # CD pass
            t_cd = time.monotonic() + 0.25 * rem
            a_cd = coord_descent_golden(best_seq, max_sweeps=15000, deadline=t_cd,
                                        seed=200 + round_num * 50)
            update_best(a_cd)

            rem = deadline - time.monotonic()
            if rem < 25:
                break

            # LP pass with increasing constraints
            top_m = min(20 + round_num * 10, 100)
            t_lp = time.monotonic() + 0.20 * rem
            a_lp, obj_lp = active_set_lp(best_seq, top_m=top_m, max_iters=100, deadline=t_lp)
            update_best(a_lp)

            rem = deadline - time.monotonic()
            if rem < 25:
                break

            # Adam with different params per round
            t_adam = time.monotonic() + 0.25 * rem
            tau_i = 10.0 + round_num * 3.0
            tau_f = 50.0 + round_num * 10.0
            lr_i = 0.025 + 0.005 * round_num
            a_ad, obj_ad = smooth_adam_v2(best_seq, max_iters=4000, deadline=t_adam,
                                          tau_init=tau_i, tau_max=tau_f, lr_init=lr_i)
            update_best(a_ad)

            rem = deadline - time.monotonic()
            if rem < 25:
                break

            # FW-away
            t_fw = time.monotonic() + 0.15 * rem
            a_fw, obj_fw = frank_wolfe_away(best_seq, max_iters=400, deadline=t_fw,
                                            tau=25.0 + round_num * 5.0)
            update_best(a_fw)

            rem = deadline - time.monotonic()
            if rem < 20:
                break

            # Final CD polish
            a_final = coord_descent_golden(best_seq, max_sweeps=10000,
                                           deadline=deadline - 5,
                                           seed=999 + round_num * 111)
            update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
