"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth log-sum-exp continuation with L-BFGS-B, LP alternating,
subgradient with momentum, and multi-n exploration.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix for LP."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def smooth_obj_and_grad(a, tau):
    """
    Compute smooth approximation of 2n*max(conv)/(sum)^2 and its gradient.
    Uses log-sum-exp for smooth max.
    Returns (obj, grad).
    """
    n = len(a)
    s = float(np.sum(a))
    conv = compute_conv(a)
    
    # Log-sum-exp for smooth max
    cmax = float(np.max(conv))
    shifted = conv / tau
    shifted -= np.max(shifted)  # for numerical stability
    log_sum = np.log(np.sum(np.exp(shifted))) + np.max(conv / tau)
    soft_max = tau * log_sum
    
    # Gradient of soft_max w.r.t. a
    weights = np.exp(conv / tau - cmax / tau)
    w_sum = float(np.sum(weights))
    weights /= w_sum
    
    # d(soft_max)/d(a[i]) = (1/tau) * sum_k weights[k] * d(conv[k])/d(a[i])
    # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (with bounds)
    # Using FFT: this is a correlation
    a_flip = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    Wf = rfft(weights, size)
    Af = rfft(a_flip, size)
    corr = irfft(Wf * Af, size)
    grad_soft = (2.0 / tau) * corr[n - 1:2 * n - 1]
    
    # Full objective: 2n * soft_max / s^2
    obj = 2.0 * n * soft_max / (s * s)
    
    # Gradient of full objective
    # d/d(a[i]) [2n * soft_max / s^2] = (2n/s^2) * grad_soft[i] - (2n * soft_max * 2 / s^3)
    grad = (2.0 * n / (s * s)) * grad_soft - (4.0 * n * soft_max / (s * s * s))
    
    return obj, grad


def smooth_optimize(a, tau_start=0.1, tau_end=0.001, n_tau_steps=20, time_budget=60.0):
    """
    Continuation method: optimize smooth approximation with decreasing tau.
    Uses L-BFGS-B.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for i in range(n_tau_steps):
        if time.monotonic() - start > time_budget:
            break
        
        tau = tau_start * (tau_end / tau_start) ** (i / max(1, n_tau_steps - 1))
        
        def func_and_grad(x):
            x = np.maximum(x, 0.0)
            s = float(np.sum(x))
            if s < 1e-10:
                return 1e10, np.zeros(len(x))
            obj, grad = smooth_obj_and_grad(x, tau)
            return obj, grad
        
        res = minimize(func_and_grad, a, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-6})
        
        a_new = np.maximum(res.x, 0.0)
        a_new = normalize(a_new)
        obj_true = compute_obj(a_new)
        
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a_new.copy()
            a = a_new

def resample_sequence(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def lp_gradient(a, p):
    """Compute gradient of ||conv(a,a)||_p w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    
    # weights[k] = conv[k]^(p-1)
    # For numerical stability, clip conv to be non-negative
    conv_clipped = np.maximum(conv, 0)
    w = conv_clipped ** (p - 1)
    
    # Scale for numerical stability
    w_max = np.max(w)
    if w_max > 0:
        w = w / w_max
    
    # Compute correlation: grad[i] = 2 * sum_k w[k] * a[k-i] * 1[0<=k-i<n]
    a_flip = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    Wf = rfft(w, size)
    Af = rfft(a_flip, size)
    corr = irfft(Wf * Af, size)
    grad = 2.0 * corr[n-1:2*n-1]
    
    # Scale back
    norm_factor = (np.sum(conv_clipped**p)) ** (1.0/p - 1.0)
    if w_max > 0:
        norm_factor /= w_max ** (p-1)
    
    grad *= norm_factor
    return grad

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Lp-norm continuation (p: 2->100) with L-BFGS-B for smooth optimization
2. Soft-max gradient descent with Nesterov momentum and backtracking
3. Peak-targeting coordinate descent
4. Aggressive multi-n exploration with resampling
5. Structured initializations (chirp, random, parent)
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_obj_and_grad(a, p):
    """
    Compute Lp-norm based objective and gradient.
    f(a) = 2n * ||conv(a,a)||_p / (sum(a))^2
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return 1e10, np.zeros(n)
    conv = compute_conv(a)
    conv_pos = np.maximum(conv, 1e-30)

    lp_power = np.sum(conv_pos ** p)
    lp = lp_power ** (1.0 / p)
    obj = 2.0 * n * lp / (s * s)

    # Gradient of lp w.r.t. a[i] = 2 * sum_k conv[k]^(p-1) * a[k-i] / lp^(p-1)
    w = conv_pos ** (p - 1)
    # Scale w for numerical stability
    w_scale = np.max(w)
    if w_scale > 1e-300:
        w = w / w_scale

    a_flip = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    Wf = rfft(w, size)
    Af = rfft(a_flip, size)
    corr = irfft(Wf * Af, size)
    corr = np.real(corr)

    # grad_lp[i] = 2 * corr[n-1+i] * w_scale / lp^(p-1)
    grad_lp = 2.0 * corr[n - 1:2 * n - 1] * w_scale / (lp ** (p - 1))

    # Full gradient of objective
    grad = (2.0 * n / (s * s)) * grad_lp - (4.0 * n * lp / (s * s * s))

    return obj, grad


def lp_continuation(a, p_values=None, time_budget=120.0):
    """
    Continuation method over p: start with small p (smooth), increase to large p (sharp).
    Uses L-BFGS-B at each p value.
    """
    if p_values is None:
        p_values = [2, 4, 8, 16, 32, 64, 128]

    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for p in p_values:
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        def func_and_grad(x):
            x = np.maximum(x, 0.0)
            s = float(np.sum(x))
            if s < 1e-10:
                return 1e10, np.zeros(len(x))
            return lp_obj_and_grad(x, p)

        max_iter = max(10, int(remaining * 0.8 / max(1, n / 700)))
        max_iter = min(max_iter, 200)

        res = minimize(func_and_grad, a, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8})

        a_new = np.maximum(res.x, 0.0)
        a_new = normalize(a_new)
        obj_true = compute_obj(a_new)

        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a_new.copy()
        a_new = normalize(a_new)

    return best_a


def softmax_gradient_step(a, max_steps=300, time_budget=60.0):
    """
    Soft-max gradient descent with Nesterov momentum and adaptive step size.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    beta = 0.85
    step = 0.03
    tau = 0.05

    size = 1
    while size < 3 * n:
        size <<= 1

    for step_i in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        s = float(np.sum(a))

        # Adaptive temperature
        tau = max(0.001, cmax / (30 + 5 * step_i))
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500, 0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= w_sum

        # Compute subgradient via FFT
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size).real
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Project to sum=0 hyperplane
        grad = grad - np.mean(grad)

        # Nesterov momentum
        v = beta * v - step * grad
        a_new = normalize(a + v)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.01, 0.08)
        else:
            a = best_a.copy()
            step *= 0.65
            v *= 0.3
            if step < 1e-9:
                break

    return best_a
