"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Improved approach with: (1) dual-informed LP direction, (2) multi-peak AdaGrad
subgradient, (3) backtracking line search, (4) differential evolution for diversity,
(5) aggressive multi-length exploration, (6) final coordinate descent polish.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns (g, dual_y, lp_value) or (None, None, None)."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                dual_y = res.unequality slack if res.unequality is not None else None
                # Actually, get the dual values properly
                y = res.ineqfun  # This might not exist
                # For HiGHS, the dual of A_ub constraints is in res
                # Let's just return the primal
                return g / s, float(-res.fun)  # -fun because we negated
    except Exception:
        pass
    return None, None


def multi_peak_subgradient(a, top_k=3, tau=0.01):
    """Compute a smoothed subgradient targeting the top-k peaks of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    
    # Select top-k peaks with soft weighting
    sorted_idx = np.argsort(conv)[::-1]
    top_indices = sorted_idx[:top_k]
    top_vals = conv[top_indices]
    weights = np.exp((top_vals - cmax) / tau)
    weights = weights / weights.sum()
    
    # Compute gradient: grad[i] = sum_j w_j * a[k_j - i]
    grad = np.zeros(n)
    for w, k in zip(weights, top_indices):
        # dc_k/da[i] = a[k-i] if 0 <= k-i < n
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        if lo <= hi:
            # a[k-i] for i in [lo, hi] means we want a at positions [k-hi, k-lo]
            a_slice = a[max(0, k - hi):k - lo + 1]
            grad[lo:hi+1] += w * a_slice
    
    return grad


def soft_max_gradient(a, tau=0.01):
    """Full soft-max over all convolution positions."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # grad[i] = sum_k weights[k] * a[k-i] (with bounds)
    # Use FFT for efficiency
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    w_ext = np.zeros(2 * n - 1)
    w_ext[:] = weights
    a_flip = a[::-1]
    Wf = rfft(w_ext, size)
    Af = rfft(a_flip, size)
    corr = irfft(Wf * Af, size)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad


def backtrack_line_search(a, direction, f0, max_iter=30, c1=1e-4):
    """Backtracking line search. direction is the update direction.
    Try a_new = normalize(a + t * direction) for decreasing t."""
    n = len(a)
    t = 1.0
    for _ in range(max_iter):
        a_try = normalize(a + t * direction)
        f_try = compute_obj(a_try)
        if f_try < f0 - c1 * t:  # Simplified Armijo (using t as proxy for grad^T*direction)
            return a_try, f_try, t
        t *= 0.5
        if t < 1e-10:
            break
    return a, f0, 0.0


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def core_optimize(a, time_budget, use_lp=True, use_gradient=True):
    """Core optimization loop combining LP and gradient methods with backtracking."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # AdaGrad state
    grad_sq = np.zeros(n)
    lr = 0.1
    lr_decay = 0.995

    lp_success = 0
    lp_fail = 0

        it += 1
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed

        if use_lp and (it % 3 != 0) and remaining > 5:

            conv = compute_conv(a)

            if g is not None:
                lp_success += 1
                direction = g - a  # Direction toward g
                a_new, f_new, t = backtrack_line_search(a, direction, best_obj)
                if f_new < best_obj:
                    best_obj = f_new
                    best_a = a_new.copy()
                    a = a_new
                else:
                    # Try smaller step
                    a_try = normalize(a + 0.1 * direction)
                    f_try = compute_obj(a_try)
                    if f_try < best_obj:
                        best_obj = f_try
                        best_a = a_try.copy()
                        a = a_try
            else:
                lp_fail += 1
        else:
            # Gradient step (soft-max or multi-peak)

                # Multi-peak subgradient (target specific peaks)
                grad = multi_peak_subgradient(a, top_k=3, tau=0.005)
            else:
                # Soft-max gradient (smoother)
                grad = soft_max_gradient(a, tau=0.01 + 0.005 * (it % 20))
            
            # Center the gradient (remove mean to preserve sum)
            grad_centered = grad - np.mean(grad)
            
            # AdaGrad update
            grad_sq += grad_centered ** 2
            a_try = normalize(a - lr * grad_centered / (np.sqrt(grad_sq) + 1e-8))

                best_obj = f_try

                    best_a = a_pert.copy()

    return best_a


def differential_evolution_step(pop, f, mutation_rate=0.8, crossover_rate=0.5, rng=None):
    """One step of differential evolution."""
    if rng is None:
        rng = np.random.default_rng()
    
    npop = len(pop)
    n = len(pop[0])
    
    if npop < 3:
        return pop, f
    
    # For each individual, create a trial
    for i in range(npop):
        # Select three distinct individuals
        candidates = [j for j in range(npop) if j != i]
        a, b, c = rng.choice(candidates, size=3, replace=False)
        
        # Mutation
        v = pop[a] + mutation_rate * (pop[b] - pop[c])
        
        # Crossover
        cross_mask = rng.random(n) < crossover_rate
        trial = np.where(cross_mask, v, pop[i])
        trial = normalize(trial)
        
        # Selection
        f_trial = compute_obj(trial)
        if f_trial < f[i]:
            pop[i] = trial
            f[i] = f_trial
    
    return pop, f


def coordinate_descent_polish(a, time_budget=15.0):
    """Pairwise mass-transfer coordinate descent."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find contributors to the peak
        contribs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)
        
        improved = False
        for _, j1, j2 in contribs[:3]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            
            # Move mass from (j1, j2) to random positions
            for attempt in range(5):
                t1 = int(rng.integers(0, n))
                t2 = int(rng.integers(0, n))
                if t1 in (j1, j2) or t2 in (j1, j2):
                    continue
                
                delta = 0.03 * max(a[j1], a[j2])
                a_try = a.copy()
                a_try[j1] = max(0, a_try[j1] - delta)
                a_try[j2] = max(0, a_try[j2] - delta)
                a_try[t1] += delta
                a_try[t2] += delta
                a_try = normalize(a_try)
                f_try = compute_obj(a_try)
                
                if f_try < best_obj:
                    best_obj = f_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved and rng.random() < 0.15:
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
    
    return best_a

"""
Improved multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses dual LP (min-peak) direction, multi-peak AdaGrad subgradient with backtracking,
multiplicative weights update, differential evolution for diversity, aggressive
multi-length exploration, and coordinate descent polishing.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        for k in range(k_start, k_end):
            A[k, j] = a[k - j]
    return A


def solve_lp_min_peak(a, time_limit=20):
    """Solve LP: min t s.t. conv(a,g) <= t*1, sum(g)=1, g>=0.
    Returns (g, t) or (None, None)."""
    n = len(a)
    m = 2 * n - 1
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.hstack([np.ones(1, n), np.zeros(1)])
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})

            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(res.x[-1])
    except Exception:

def soft_max_gradient(a, tau=0.01):
    """Compute smooth gradient using soft-max over all convolution positions."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    size = 1
    while size < 2 * n + n:
        size <<= 1
    Wf = rfft(weights, size)
    ah = np.zeros(size)
    ah[:n] = a
    Ah = rfft(ah, size)
    corr_full = irfft(Wf * np.conj(Ah), size)
    grad = 2.0 * corr_full[:n]
    return grad


def multi_peak_subgradient(a, top_k=3, tau=0.005):
    """Subgradient targeting top-k peaks of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    sorted_idx = np.argsort(conv)[::-1][:top_k]
    top_vals = conv[sorted_idx]
    weights = np.exp((top_vals - cmax) / tau)
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for w, k in zip(weights, sorted_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            a_slice = a[k - hi:k - lo + 1]
            grad[lo:hi + 1] += 2.0 * w * a_slice
    return grad


def mwu_step(a, grad, lr):
    """Multiplicative weights update: a_new = a * exp(-lr * grad)."""
    a_new = a * np.exp(-lr * grad)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return a.copy()
    return a_new / s


def backtrack_line_search(a, direction, f0, c1=1e-4, max_iter=25):
    """Backtracking line search for a + t*direction."""
    t = 1.0
    best_a = a.copy()
    best_f = f0
    for _ in range(max_iter):
        a_try = normalize(a + t * direction)
        f_try = compute_obj(a_try)
        if f_try < best_f:
            best_a = a_try
            best_f = f_try
        if f_try < f0 - c1 * t:
            return a_try, f_try, t
        t *= 0.5
        if t < 1e-10:
            break
    return best_a, best_f, t


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def core_optimize(a, time_budget, use_lp=True, use_mwu=False):
    """Core optimization loop combining LP and gradient methods."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    grad_sq = np.zeros(n)
    lr = 0.08
    it = 0
    stagnation = 0
    last_improve = best_obj

    while time.monotonic() - start < time_budget:
        it += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        # Stagnation detection
        if best_obj - last_improve < 1e-10:
            stagnation += 1
        else:
            stagnation = 0

        # Escalating perturbation on stagnation
        if stagnation > 30 and stagnation % 15 == 0:
            rng = np.random.default_rng(it * 104729 + 7)
            noise_scale = 0.02 * min(stagnation // 15, 5)
            a_pert = normalize(best_a * np.exp(rng.normal(0, noise_scale, n)))
            f_pert = compute_obj(a_pert)
            if f_pert < best_obj:
                best_obj = f_pert
                best_a = a_pert.copy()
                a = a_pert
                last_improve = best_obj

        # Choose method
        if use_lp and it % 4 != 0 and remaining > 8:
            # LP step (alternate between min-peak and max-sum)
            conv = compute_conv(a)
            M = float(np.max(conv))
            if it % 8 == 0:
                # Use relaxation for exploration
                M *= 1.03

            g = None
            if it % 2 == 1:
                g, _ = solve_lp_min_peak(a, time_limit=min(10, int(remaining * 0.3)))
            else:
                g = solve_lp_max_sum(a, M, time_limit=min(10, int(remaining * 0.3)))

            if g is not None:
                direction = g - a
                a_new, f_new, t = backtrack_line_search(a, direction, best_obj)
                if f_new < best_obj:
                    best_obj = f_new
                    best_a = a_new.copy()
                    a = a_new
                    last_improve = best_obj
                elif t > 0:
                    a = a_new
            else:
                # LP failed, fall through to gradient
                pass

        if time.monotonic() - start < time_budget - 1:
            # Gradient step (always do after LP, or as standalone)
            if it % 6 == 0:
                grad = multi_peak_subgradient(a, top_k=3, tau=0.003)
            elif it % 6 == 3:
                grad = multi_peak_subgradient(a, top_k=5, tau=0.008)
            else:
                grad = soft_max_gradient(a, tau=0.005 + 0.003 * (it % 10))

            grad_centered = grad - np.mean(grad)

            if use_mwu and it % 3 == 0:
                # MWU step
                a_try = mwu_step(a, grad_centered, lr * 0.5)
                f_try = compute_obj(a_try)
                if f_try < best_obj:
                    best_obj = f_try
                    best_a = a_try.copy()
                    a = a_try
                    last_improve = best_obj
                else:
                    lr *= 0.8
            else:
                # AdaGrad step
                grad_sq += grad_centered ** 2
                a_try = normalize(a - lr * grad_centered / (np.sqrt(grad_sq) + 1e-8))
                f_try = compute_obj(a_try)
                if f_try < best_obj:
                    best_obj = f_try
                    best_a = a_try.copy()
                    a = a_try
                    last_improve = best_obj
                else:
                    lr *= 0.7
                    if lr < 1e-8:
                        lr = 0.05
                        grad_sq *= 0.3

    return best_a


def differential_evolution_step(pop, f, mu=0.8, cr=0.6, rng=None):
    """One DE step. Returns updated population and fitness."""
    if rng is None:
        rng = np.random.default_rng()
    npop = len(pop)
    n = len(pop[0])
    if npop < 3:
        return pop, f
    for i in range(npop):
        candidates = [j for j in range(npop) if j != i]
        a_idx, b_idx, c_idx = rng.choice(candidates, size=3, replace=False)
        v = pop[a_idx] + mu * (pop[b_idx] - pop[c_idx])
        cross_mask = rng.random(n) < cr
        trial = np.where(cross_mask, v, pop[i])
        trial = normalize(trial)
        f_trial = compute_obj(trial)
        if f_trial < f[i]:
            pop[i] = trial
            f[i] = f_trial
    return pop, f


def coordinate_descent_polish(a, time_budget=15.0):
    """Pairwise mass-transfer coordinate descent targeting convolution peaks."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        contribs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)

        improved = False
        for _, j1, j2 in contribs[:4]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            for _ in range(4):
                t1 = int(rng.integers(0, n))
                t2 = int(rng.integers(0, n))
                if t1 in (j1, j2) or t2 in (j1, j2):
                    continue
                delta = 0.04 * max(a[j1], a[j2])
                a_try = a.copy()
                a_try[j1] = max(0.0, a_try[j1] - delta)
                a_try[j2] = max(0.0, a_try[j2] - delta)
                a_try[t1] += delta
                a_try[t2] += delta
                a_try = normalize(a_try)
                f_try = compute_obj(a_try)
                if f_try < best_obj:
                    best_obj = f_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            if rng.random() < 0.2:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:
                break

    return best_a


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    Phase 1: Core LP+gradient+MWU optimization at parent length
    Phase 2: Multi-length exploration with resampling
    Phase 3: Differential evolution for global diversity
    Phase 4: Final intensive polish with coordinate descent + short LP
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: n={n_parent}, score={best_score:.8f}, budget={budget_s}s")

    # =========================================================
    # Phase 1: Core optimization at parent length (45% time)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Core LP+gradient+MWU optimization (n={n_parent})")
    print(f"{'='*60}")

    phase1_budget = (deadline - start_time) * 0.45
    if best_seq is not None:
        a = normalize(best_seq.copy())
        a_opt = core_optimize(a, time_budget=phase1_budget, use_lp=True, use_mwu=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1 result: {best_score:.8f}")
        else:
            print(f"  Phase 1 result: {obj_opt:.8f} (no improvement)")

    # =========================================================
    # Phase 2: Multi-length exploration (25% time)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Multi-length exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700
    n_candidates = sorted(set([
        n_best - 200, n_best - 150, n_best - 100, n_best - 75,
        n_best - 50, n_best - 25, n_best - 10,
        n_best + 10, n_best + 25, n_best + 50, n_best + 75,
        n_best + 100, n_best + 150, n_best + 200, n_best + 300,
        n_best + 400, n_best + 500
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 1500]
    # Sort by proximity to current best
    n_candidates.sort(key=lambda x: abs(x - n_best))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if best_seq is not None:
            a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try

        alloc = min(remaining * 0.12, 35.0)
        use_lp_here = n_try <= 850
        a_try = core_optimize(a_try, time_budget=alloc, use_lp=use_lp_here, use_mwu=True)
        obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 3: Differential evolution (15% time)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Differential evolution for diversity")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    if remaining > 60 and best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed)

        pop_size = 10
        pop = [normalize(best_seq.copy())]
        f_list = [best_score]

        for i in range(1, pop_size):
            if i % 4 == 1:
                a_new = best_seq * np.exp(rng.normal(0, 0.12, n_cur))
            elif i % 4 == 2:
                freq = rng.uniform(1, 10)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.25 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
                a_new = best_seq * mod
            elif i % 4 == 3:
                a_rand = rng.dirichlet(np.ones(n_cur) * 0.5)
                mix = rng.uniform(0.2, 0.5)
                a_new = (1 - mix) * normalize(best_seq) + mix * a_rand
            else:
                # Block reversal
                a_new = best_seq.copy()
                blen = int(rng.integers(n_cur // 5, n_cur // 2))
                bstart = int(rng.integers(0, n_cur - blen))
                a_new[bstart:bstart + blen] = a_new[bstart:bstart + blen][::-1]
            pop.append(normalize(a_new))
            f_list.append(compute_obj(pop[-1]))

        de_deadline = time.monotonic() + min(remaining * 0.5, 100.0)
        de_iter = 0
        while time.monotonic() < de_deadline:
            pop, f_list = differential_evolution_step(pop, f_list, mu=0.7, cr=0.6, rng=rng)
            de_iter += 1
            fmin = min(f_list)
            if fmin < best_score:
                best_score = fmin
                best_idx = f_list.index(fmin)
                best_seq = pop[best_idx].copy()
                print(f"  DE iter {de_iter}: {best_score:.8f} **")

        # Polish the best DE result
        if best_seq is not None:
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                a_de = core_optimize(normalize(best_seq.copy()),
                                     time_budget=min(30, remaining2 * 0.4),
                                     use_lp=(len(best_seq) <= 850), use_mwu=True)
                obj_de = compute_obj(a_de)
                if obj_de < best_score:
                    best_score = obj_de
                    best_seq = a_de.copy()
                    print(f"  DE + core: {best_score:.8f}")

    # =========================================================
    # Phase 4: Final intensive polish (remaining time)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final intensive polish")
    print(f"{'='*60}")

    if best_seq is not None:
        n_final = len(best_seq)
        a_final = normalize(best_seq.copy())

        # Round A: LP-heavy
        remaining = deadline - time.monotonic()
        if remaining > 20 and n_final <= 900:
            a_pol = core_optimize(a_final, time_budget=min(35, remaining * 0.35),
                                  use_lp=True, use_mwu=False)
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish LP: {best_score:.8f}")
            else:
                a_final = a_pol
                print(f"  Polish LP: {obj_pol:.8f}")

        # Round B: Gradient + MWU
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_pol = core_optimize(a_final, time_budget=min(25, remaining * 0.35),
                                  use_lp=False, use_mwu=True)
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish Grad: {best_score:.8f}")
            else:
                a_final = a_pol
                print(f"  Polish Grad: {obj_pol:.8f}")

        # Round C: Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_pol = coordinate_descent_polish(a_final, time_budget=min(20, remaining * 0.4))
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Polish CD: {best_score:.8f}")
            else:
                print(f"  Polish CD: {obj_pol:.8f}")

        # Round D: One more LP if time
        remaining = deadline - time.monotonic()
        if remaining > 15 and n_final <= 850:
            a_pol = core_optimize(a_final, time_budget=min(20, remaining * 0.5),
                                  use_lp=True, use_mwu=True)
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Polish Final LP: {best_score:.8f}")
            else:
                print(f"  Polish Final LP: {obj_pol:.8f}")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
