def fast_conv(a):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_M(a):
    return float(fast_conv(a).max())

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    M = compute_M(a)
    return float(2 * len(a) * M / (s * s))

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.array([max(0, v[0])]) if v[0] > 0 else np.array([1.0/n])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def anderson_mix(history_x, history_Fx, m, clip_range=1.5):
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-10 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-12:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -clip_range, clip_range)
    alpha_sum = alpha.sum()
    if abs(alpha_sum) < 1e-12:
        alpha = ones / m
        alpha_sum = alpha.sum()
    alpha = alpha / alpha_sum
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)

            break
        M_cur = compute_M(a) * relax
        for it in range(iters_per_phase):
            if time.monotonic() > deadline:
                return best_a
            g = lp_best_response(a, M_cur)
            if g is None:
                return best_a
            gs = g.sum()
            if gs < 1e-10:
                return best_a
            g /= gs
            t = 0.3 / (1.0 + 0.02 * it)
            a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new /= s
            M_new = compute_M(a_new)
            if M_new < best_M * (1 - 1e-10):
                best_M = M_new
                best_a = a_new.copy()
            a = a_new
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_hump_init(n, sigma=0.3, edge_decay=2.0):
    t = np.linspace(0, 1, n, endpoint=False) + 0.5 / n
    hump = np.exp(-((t - 0.5) ** 2) / (2 * sigma * sigma))
    edge = np.minimum(t / 0.1, 1.0) * np.minimum((1 - t) / 0.1, 1.0)
    edge = np.clip(edge, 0, 1) ** edge_decay
    result = hump * edge + 0.01
    return result / result.sum()


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False) + 0.5 / n
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.005))
    inits.append(make_hump_init(n, sigma=0.35, edge_decay=1.5))
    inits.append(make_hump_init(n, sigma=0.25, edge_decay=3.0))
    inits.append(make_hump_init(n, sigma=0.45, edge_decay=1.0))
    inits.append(np.ones(n) / n)
    for alpha in [0.5, 1.0, 2.0]:
        shape = np.power(np.sin(np.pi * t), alpha)
        shape = np.maximum(shape, 0.01)
        inits.append(shape / shape.sum())
    for f in [2, 3, 5]:
        mod = np.maximum(0.6 + 0.4 * np.cos(2 * np.pi * f * t), 0.05)
        inits.append(make_hump_init(n) * (mod / mod.sum() * n) / (make_hump_init(n).sum() * n))
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_flat_spectral(a_rand, n_iters=80, beta=0.85))
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(gs_flat_spectral(p, n_iters=100, beta=0.9))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = evaluate_sequence(a.tolist())
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = make_hump_init(1024)
        best.update(parent)

    # Lengths to explore
    lengths = [768, 896, 1024, 1152, 1280, 1408, 1536]
    print(f"Lengths: {lengths}", flush=True)

    # === PHASE 1: Diverse inits + smooth optimization ===
    p1_budget = 350.0
    p1_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth Optimization ({p1_budget:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    best_per_length = {}
    n_lengths = len(lengths)
    time_per_length = p1_budget / n_lengths

    for n in lengths:
        t0 = time.monotonic()
        remaining = deadline - 300 - time.monotonic()
        if remaining < 30:
            break
        allot = min(time_per_length, remaining * 0.4)
        dl = time.monotonic() + allot
        if time.monotonic() >= dl:
            continue

        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')

        n_iters_base = max(300, int(12000 * (512.0 / n)))
        for idx, a0 in enumerate(inits):
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            frac = 1.0 / (1.0 + idx * 0.4)
            sub_dl = time.monotonic() + max(5.0, (dl - time.monotonic()) * frac)
            a = run_smooth_opt(a0, sub_dl, max_iters=n_iters_base,
                               tau_start=3.0, tau_end=300.0,
                               lr_init=0.3, lr_end=0.003)
            M = compute_M(a)
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: M={best_M:.8f} J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: LP mirror descent with Anderson (main workhorse) ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Mirror Descent + Anderson", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        est_lp_ms = 5.0 * (n / 128.0) ** 2
        print(f"  n={n}, est LP time={est_lp_ms:.0f}ms", flush=True)

        # Allocate most time here
        lp_budget = min(1000.0, max(200.0, (deadline - 120) - time.monotonic()))
        max_iters = max(80, int(lp_budget / max(0.01, est_lp_ms / 1000.0)))
        print(f"  LP budget={lp_budget:.0f}s, max_iters={max_iters}", flush=True)

        # Run 1: Anderson LP
        dl1 = time.monotonic() + lp_budget * 0.45
        a = run_lp_anderson(best.seq, dl1, max_iters=int(max_iters * 0.5),
                            anderson_window=6, t_base=0.35, t_min=0.003)
        best.update(a)

        # Run 2: Continuation LP
        if time.monotonic() < deadline - 100:
            dl2 = time.monotonic() + min(lp_budget * 0.30, deadline - 150 - time.monotonic())
            a2 = run_lp_with_continuation(a, dl2, max_iters=int(max_iters * 0.3))
            best.update(a2)

        # Run 3: More Anderson with different params
        if time.monotonic() < deadline - 80:
            dl3 = deadline - 60
            a3 = run_lp_anderson(best.seq, dl3, max_iters=int(max_iters * 0.3),
                                 anderson_window=8, t_base=0.25, t_min=0.002)
            best.update(a3)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Multi-length LP polish ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*50}", flush=True)

    if best_per_length and time.monotonic() < deadline - 60:
        sorted_lengths = sorted(best_per_length.keys(),
                                key=lambda nn: compute_J(best_per_length[nn]))
        for n_try in sorted_lengths[:3]:
            if time.monotonic() > deadline - 40:
                break
            a_cand = best_per_length[n_try].copy()
            # Quick smooth
            a_cand = run_smooth_opt(a_cand, min(time.monotonic() + 20, deadline - 35),
                                    max_iters=400, tau_start=10, tau_end=200,
                                    lr_init=0.15, lr_end=0.01)
            best.update(a_cand)
            # LP polish
            if time.monotonic() < deadline - 30:
                a_lp = run_lp_anderson(a_cand, deadline - 25,
                                       max_iters=40, anderson_window=5,
                                       t_base=0.3, t_min=0.01)
                best.update(a_lp)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Final polish ===
    p4_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Final Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 5:
        # Final Anderson with very small steps
        a_final = run_lp_anderson(best.seq, deadline - 3,
                                  max_iters=50, anderson_window=5,
                                  t_base=0.15, t_min=0.002)
        best.update(a_final)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Length perturbation ===
    if best.seq is not None and time.monotonic() < deadline - 10:
        n_best = best.seq.size
        for delta in [-64, -32, 32, 64, 128]:
            if time.monotonic() > deadline - 8:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 2560:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = run_smooth_opt(a_ext, min(time.monotonic() + 10, deadline - 5),
                                   max_iters=200, tau_start=10, tau_end=150,
                                   lr_init=0.1, lr_end=0.01)
            best.update(a_ext)

    if best.seq is None:
        best.seq = np.ones(1024)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
