import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_grad_fft(a, p):
    """Compute grad[i] = 2 * sum_j p[i+j] * a[j] using FFT cross-correlation."""
    n = len(a)
    size = 1
    while size < 2 * (2 * n - 1) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(fp * np.conj(fa), size)
    return 2.0 * cross[:n]


def compute_grad_direct(a, p):
    """Direct computation of grad[i] = 2 * sum_j p[i+j] * a[j]."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(p[i:i + n], a)
    return grad


def softmax_stable(x, beta):
    """Compute exp(beta*x)/sum(exp(beta*x)) with numerical stability."""
    bx = beta * x
    bx_max = bx.max()
    exp_bx = np.exp(bx - bx_max)
    return exp_bx / exp_bx.sum()


def run_fdmd(a_init, betas, iters_per_beta, deadline, lr_base=0.1):
    """Full-distribution mirror descent with adaptive beta schedule and line search."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for beta in betas:
        if time.monotonic() > deadline:
            break

        for it in range(iters_per_beta):
            if time.monotonic() > deadline:
                break

            b = fast_conv(a)
            M_cur = b.max()

            # Compute distribution p over k values
            p = softmax_stable(b, beta / M_cur if M_cur > 0 else 1.0)

            # Compute gradient
            grad = compute_grad_fft(a, p)

            # Mirror descent update (multiplicative weights)
            lr = lr_base / (1.0 + 0.01 * it)
            log_update = -lr * grad
            log_update -= log_update.max()
            a_new = a * np.exp(log_update)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()

            # Line search: try several step sizes
            improved = False
            for step in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
                a_trial = (1.0 - step) * a + step * a_new
                a_trial = np.maximum(a_trial, 1e-15)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-15:
                    a = a_trial.copy()
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    break

            if not improved:
                # Try smaller steps or break
                if it > 10:
                    break

    return best_a


def run_fdmd_fista(a_init, beta, max_iters, deadline, lr=0.15):
    """FISTA-style accelerated FD-MD with fixed beta."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    t = 1.0
    y = a.copy()
    a_prev = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_y = fast_conv(y)
        M_y = b_y.max()
        p = softmax_stable(b_y, beta / M_y if M_y > 0 else 1.0)
        grad = compute_grad_fft(y, p)

        # Gradient step with projection
        step = lr / (1.0 + 0.005 * it)
        a_new = y - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # Nesterov extrapolation
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-10:
            y_new = a_new.copy()
        else:
            y_new = y_new / s_y

        a_prev = a
        a = a_new
        y = y_new
        t = t_new

        if it % 3 == 0:
            M_a = fast_conv(a).max()
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()
        best_M = M_final

    return best_a


def run_cutting_plane_lp(a_init, max_iters, deadline, initial_active=30, max_active=None):
    """Cutting-plane LP: start with top-k active, grow by adding violated constraints."""
    if max_active is None:
        max_active = 4 * len(a_init) - 4

    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    active_set = set()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        # Initialize or update active set
        if not active_set:
            top_idx = np.argsort(b)[::-1][:initial_active]
            active_set = set(int(k) for k in top_idx)
        else:
            # Add all k where b[k] > best_M * 0.999 (near-active)
            near_active = np.where(b > best_M * 0.995)[0]
            for k in near_active:
                if len(active_set) < max_active:
                    active_set.add(int(k))

        # Solve LP: max sum(g) s.t. (a*g)[k] <= best_M for k in active_set, g >= 0
        active_list = sorted(active_set)
        if len(active_list) == 0:
            break

        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active_list):
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    rows.append(idx)
                    cols.append(j)
                    vals.append(a[m])

        A = csr_matrix((vals, (rows, cols)), shape=(len(active_list), n))
        if A.nnz == 0:
            break

        c = -np.ones(n)
        b_ub = np.full(len(active_list), best_M)
        bounds = [(0.0, None)] * n

        time_limit_lp = min(5.0, max(0.5, (deadline - time.monotonic()) / 3.0))
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': time_limit_lp, 'presolve': True})
            if not r.success or r.x is None:
                no_improve += 2
                if no_improve > 10:
                    break
                continue
        except Exception:
            no_improve += 3
            if no_improve > 10:
                break
            continue

        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search
        improved = False
        base_steps = [0.3, 0.2, 0.15, 0.1, 0.05, 0.02, 0.01, 0.005]
        for t in base_steps:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Try to expand active set
            if len(active_set) < max_active:
                b_cur = fast_conv(a)
                violated = np.where(b_cur > best_M + 1e-10)[0]
                for k in violated[:10]:
                    active_set.add(int(k))
                if len(violated) == 0:
                    # No violations, try adding near-max
                    near_max = np.where(b_cur > best_M * 0.99)[0]
                    for k in near_max[:5]:
                        active_set.add(int(k))

            if no_improve > 20:
                break

    return best_a


def coordinate_opt_1d(a, i, b, n):
    """Optimize coordinate i given all others. Returns optimal x >= 0."""
    x_cur = a[i]
    S_fixed = a.sum() - x_cur

    # Affected k range: k in [i, min(2n-2, i+n-1)]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # For each affected k: f_k = A_k + B_k * x + C_k * x^2
    A_vals = []
    B_vals = []
    C_vals = []

    for k in range(k_lo, k_hi + 1):
        j = k - i
        if k == 2 * i:
            # f_k contains x^2 term
            C = 1.0
            B = 0.0
            A = b[k] - x_cur * x_cur
        elif 0 <= j < n:
            C = 0.0
            B = 2.0 * a[j]
            A = b[k] - 2.0 * x_cur * a[j]
        else:
            C = 0.0
            B = 0.0
            A = b[k]
        A_vals.append(A)
        B_vals.append(B)
        C_vals.append(C)

    A_arr = np.array(A_vals)
    B_arr = np.array(B_vals)
    C_arr = np.array(C_vals)

    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    def obj_at(x):
        """Compute 2n * max_k f_k(x) / (S_fixed + x)^2"""
        if x < 0:
            x = 0.0
        denom = S_fixed + x
        if denom < 1e-10:
            return float('inf')
        vals = A_arr + B_arr * x + C_arr * x * x
        m = vals.max() if vals.size > 0 else 0.0
        if base_max > m:
            m = base_max
        return 2.0 * n * m / (denom * denom)

    # Golden section search
    lo = 0.0
    hi = max(3.0 * x_cur, np.sqrt(max(b.max(), 0.0)) * 2.0, 0.5)
    # Ensure hi gives a higher objective (minimum should be interior)
    for _ in range(15):
        if obj_at(hi) < obj_at(hi * 0.5):
            hi *= 2.0
            if hi > 100.0:
                break
        else:
            break

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    # Check boundary
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with golden section 1D optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = np.arange(n)
        rng.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False

        for i in order:
            if time.monotonic() > deadline:
                break
            x_opt = coordinate_opt_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-10:
                dx = x_opt - a[i]
                a[i] = x_opt
                # Update b efficiently
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - (x_opt - dx) * (x_opt - dx))
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * dx
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        J_cur = compute_J(a)

        if J_cur < best_J * (1 - 1e-10):
            best_J = J_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 5
        if no_improve > 15:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_full_pipeline(a_init, deadline, seed=0):
    """Run the full optimization pipeline on a single sequence."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    # Stage 1: FD-MD with various betas (40% of time)
    t1 = time.monotonic() + 0.35 * (deadline - time.monotonic())
    betas = [5, 20, 100, 500, 2000]
    for beta in betas:
        if time.monotonic() > t1:
            break
        sub_dl = min(t1, time.monotonic() + max(5, (t1 - time.monotonic()) / (len(betas) - 1)))
        a_md = run_fdmd(best_a, [beta], 200, sub_dl, lr_base=0.15)
        M_md = fast_conv(a_md).max()
        if M_md < best_M:
            best_M = M_md
            best_a = a_md.copy()

    # Stage 2: FISTA (20% of time)
    t2 = time.monotonic() + 0.25 * (deadline - time.monotonic())
    for beta in [200, 1000]:
        if time.monotonic() > t2:
            break
        a_fista = run_fdmd_fista(best_a, beta, 3000,
                                  min(t2, time.monotonic() + 20), lr=0.1)
        M_f = fast_conv(a_fista).max()
        if M_f < best_M:
            best_M = M_f
            best_a = a_fista.copy()

    # Stage 3: Cutting-plane LP (25% of time)
    t3 = time.monotonic() + 0.25 * (deadline - time.monotonic())
    max_iters_lp = max(5, int((t3 - time.monotonic()) / 3.0))
    a_lp = run_cutting_plane_lp(best_a, max_iters_lp, t3, initial_active=40)
    M_lp = fast_conv(a_lp).max()
    if M_lp < best_M:
        best_M = M_lp
        best_a = a_lp.copy()

    # Stage 4: Coordinate descent (remaining time)
    a_cd = run_coordinate_descent(best_a, 20, deadline, seed=seed)
    M_cd = fast_conv(a_cd).max()
    if M_cd < best_M:
        best_M = M_cd
        best_a = a_cd.copy()

    return best_a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: adaptive-beta FD-MD, cutting-plane LP, coordinate descent,
    multi-length exploration, and perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size

    # === PHASE 1: Intensive optimization on parent length ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Intensive optimization on n={n_parent}", flush=True)
    print(f"{'='*60}", flush=True)

    t_phase1_end = time.monotonic() + min(500, (deadline - time.monotonic()) * 0.4)

    # Run full pipeline on parent
    a_opt = run_full_pipeline(parent, t_phase1_end, seed=seed)
    tracker.update(a_opt)

    # Run with perturbed parent
    parent_pert = parent + rng.normal(0, 0.02, n_parent)
    parent_pert = np.maximum(parent_pert, 1e-4)
    parent_pert = parent_pert / parent_pert.sum()
    sub_dl = min(t_phase1_end, time.monotonic() + 100)
    a_opt2 = run_full_pipeline(parent_pert, sub_dl, seed=seed + 1)
    tracker.update(a_opt2)

    tracker.report("After Phase 1")

    # === PHASE 2: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 200:
        n_best = tracker.best.size
        candidate_lengths = set()
        for delta in [-128, -64, -32, 32, 64, 128, 256]:
            nl = n_best + delta
            if 256 <= nl <= 2048:
                candidate_lengths.add(nl)

        # Prioritize lengths close to best
        candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - n_best))[:4]

        t_per_len = (deadline - 200 - time.monotonic()) / (len(candidate_lengths) + 1)
        t_per_len = max(t_per_len, 30)

        for nl in candidate_lengths:
            if time.monotonic() > deadline - 200:
                break
            if nl == n_best:
                continue
            a_init = resample_seq(tracker.best, nl)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()
            sub_dl = min(deadline - 200, time.monotonic() + t_per_len)
            a_opt_nl = run_full_pipeline(a_init, sub_dl, seed=seed + 100 + nl)
            tracker.update(a_opt_nl)
            print(f"  n={nl}: done, best J={tracker.best_score:.8f}", flush=True)

    tracker.report("After Phase 2")

    # === PHASE 3: Intensive LP polish on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Intensive LP polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 100:
        t_lp = min(deadline - 100, time.monotonic() + 150)
        max_iters = max(10, int((t_lp - time.monotonic()) / 2.0))
        a_lp = run_cutting_plane_lp(tracker.best, max_iters, t_lp, initial_active=60)
        tracker.update(a_lp)

        # Second LP pass with different initial active set
        if time.monotonic() < deadline - 60:
            t_lp2 = min(deadline - 60, time.monotonic() + 60)
            a_lp2 = run_cutting_plane_lp(a_lp, 30, t_lp2, initial_active=20)
            tracker.update(a_lp2)

    tracker.report("After Phase 3")

    # === PHASE 4: Coordinate descent refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        max_sweeps = max(5, int((deadline - 30 - time.monotonic()) / max(1.0, tracker.best.size / 80.0)))
        a_cd = run_coordinate_descent(tracker.best, max_sweeps, deadline - 10, seed=99999)
        tracker.update(a_cd)

    tracker.report("After Phase 4")

    # === PHASE 5: Final perturbation attempts ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final perturbation attempts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 15:
        n_best = tracker.best.size
        remaining = deadline - 15 - time.monotonic()
        n_attempts = min(5, max(1, int(remaining / 30)))

        for attempt in range(n_attempts):
            if time.monotonic() > deadline - 15:
                break
            # Different perturbation scales
            scale = 0.01 * (attempt + 1)
            a_pert = tracker.best + rng.normal(0, scale, n_best)
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert = a_pert / a_pert.sum()

            # Quick optimization: FD-MD + short LP
            sub_dl = min(deadline - 15, time.monotonic() + max(10, remaining / (n_attempts - attempt)))
            a_q1 = run_fdmd(a_pert, [50, 200, 1000], 100, sub_dl, lr_base=0.1)
            tracker.update(a_q1)
            if time.monotonic() < sub_dl - 5:
                a_q2 = run_cutting_plane_lp(a_q1, 20, sub_dl, initial_active=30)
                tracker.update(a_q2)

    tracker.report("After Phase 5")

    # === Final: ensure valid return ===
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
