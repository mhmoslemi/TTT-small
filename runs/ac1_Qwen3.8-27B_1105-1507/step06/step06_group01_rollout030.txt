Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid optimization:
1. FISTA on log-sum-exp smooth approximation with adaptive temperature
2. Multiplicative weights (mirror descent) with subgradient
3. Full-constraint LP best-response for direction finding
4. Coordinate descent for final polish
Key improvements: efficient gradient via np.correlate, full LP constraints,
adaptive tau schedule, and multi-phase time allocation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time
import sys

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def lse_grad(a, tau):
    """Compute (value, gradient) of log-sum-exp(tau * conv(a,a))."""
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    val = b_max + np.log(exp_shifted.sum()) / tau
    # grad[i] = (2/tau) * sum_k p[k] * a[k-i] = (2/tau) * correlate(p, a, 'valid')[i]
    grad = (2.0 / tau) * np.correlate(p, a, 'valid')
    return val, grad

def run_fista(a_init, tau_schedule, max_iters, deadline, lr=0.5):
    """FISTA with adaptive tau schedule."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        # Adaptive tau
        if it < len(tau_schedule):
            tau = tau_schedule[it]
        else:
            tau = tau_schedule[-1]
        
        _, grad_y = lse_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        # Step size: use Lipschitz estimate
        # L ≈ 2*tau*max(a) * 2 (rough bound)
        step = lr / (1.0 + 0.1 * np.zeros(1)[0])  # fixed step for now
        # Better: backtracking
        # For simplicity, use a fixed step that works well in practice
        step = min(lr, 0.5 / (tau * y.max() + 1e-10))
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha_mom = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha_mom * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        if it % 10 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    
    M_a = fast_conv(a).max()
    if M_a < best_M:
        best_M = M_a
        best_a = a.copy()
    return best_a, best_M

Actually, the Hessian of (a*a)[k] at a is 2*M_k (where M_k = u_k u_k^T), so the Hessian of LSE(tau * conv(a,a)) is:
sum_k p[k] * 2*tau * M_k + (second-order terms from p depending on a)

The dominant term is 2*tau * sum_k p[k] * M_k = 2*tau * M_p where M_p = sum_k p[k] * M_k.
The eigenvalues of M_p are at most 2*tau * ||u_p||^2 where u_p = sum_k p[k] * u_k... this is getting complicated.

A practical bound: L <= 2*tau*2*max_k ||u_k||^2 = 2*tau*2*n (since ||u_k||^2 <= n for any k).
So L <= 4*tau*n.

For n=768, tau=100: L <= 4*100*768 = 307200. Step = 1/L ≈ 3.2e-6. That's very small!

But this is a loose bound. In practice, the effective Lipschitz constant is much smaller. Let me just use backtracking.

"""

import numpy as np

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def lse_and_grad(a, tau):
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    val = b_max + np.log(exp_s.sum()) / tau
    grad = (2.0 / tau) * np.correlate(p, a, 'valid')
    return val, grad, p, b

def fista_smooth(a_init, tau_start, tau_end, n_tau_steps, max_iters, deadline, base_lr=1.0):
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        # Adaptive tau: interpolate from tau_start to tau_end
        frac = min(1.0, it / max(1, int(0.7 * max_iters)))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y, p, b = lse_and_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        # Backtracking step size
        # Lipschitz estimate: L ≈ 2*tau * max sum |p[k]| * |a[k-i]| over i
        # Simpler: use L ≈ 4*tau * max(b) (heuristic)
        L_est = 4.0 * tau * max(b.max(), 1e-6)
        step = base_lr / (L_est + 1e-10)
        # Cap step to avoid too-large moves
        step = min(step, 1.0 / n)
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha_m = (t_mom - 1.0) / t_new
        y_new = a_new + alpha_m * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        
        if it % 5 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    
    M_f = fast_conv(a).max()
    if M_f < best_M:
        best_a = a.copy()
        best_M = M_f
    return best_a, best_M

def mirror_descent(a_init, max_iters, deadline, eta0=1.0):
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        
        # Subgradient: grad[i] = 2 * a[k_star - i] (valid indices)
        subgrad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                subgrad[i] = 2.0 * a[j]
        
        subgrad_tan = subgrad - subgrad.mean()
        
        # Adaptive step size
        eta = eta0 / (1.0 + 0.05 * it)
        
        # Multiplicative weights update
        log_w = -eta * subgrad_tan
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        a = a_new
        M = b.max()  # This is the M BEFORE the update; for tracking use new M
        # Actually compute new M
        M_new = fast_conv(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    
    return best_a, best_M

def solve_lp_best_response(a, t_limit=None):
    """Solve LP: min t s.t. (a*g)[k] <= t, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None)."""
    n = len(a)
    # Constraint matrix: row k has a[k-j] in column j for valid j
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    # Add -t column: identity for row k
    for k in range(2 * n - 1):
        rows.append(k)
        cols.append(n)  # t is variable index n
        vals.append(-1.0)
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    
    # Objective: minimize t = variable n
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        opts = {'time_limit': 10} if t_limit else {}
        r = linprog(c, A_ub=A, b_ub=np.zeros(2 * n - 1),
                    A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                    method='highs', options=opts)
        if r.success:
            g = np.maximum(r.x[:n], 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
            t_opt = r.x[n]
            return g, t_opt
    except Exception:
        pass
    return None, None

def coordinate_descent_1d(a, i, b, n):
    """Optimize a[i] given others. Returns optimal x >= 0."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    # Unaffected max
    all_ks = set(range(2 * n - 1))
    affected_ks = set(range(k_lo, k_hi + 1))
    unaffected_ks = all_ks - affected_ks
    if unaffected_ks:
        base_max = max(b[k] for k in unaffected_ks)
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if quad_pos >= 0:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    # Golden section search
    lo = 0.0
    hi = max(3.0 * x_cur, np.sqrt(max(b.max(), 0.0)) * 1.5, 0.1)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_diverse_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        if p.sum() > 1e-10:
            inits.append(p / p.sum())
        for noise_level in [0.02, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, noise_level, n), 1e-6)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(np.maximum(cos_w, 0.01) / np.maximum(cos_w, 0.01).sum())
    
    # Gaussian centered
    g = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(g / g.sum())
    
    # Two-bump
    two_bump = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2) + 0.05
    inits.append(two_bump / two_bump.sum())
    
    # Power law
    pw = (t + 0.1) ** 1.5 + (1 - t + 0.1) ** 1.5
    inits.append(pw / pw.sum())
    
    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits
