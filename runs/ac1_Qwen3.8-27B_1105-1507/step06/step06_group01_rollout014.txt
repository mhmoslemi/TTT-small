def fast_conv(a):

        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_grad_fft(a, tau):
    """Compute smooth max of conv(a,a) and its gradient using FFT. O(n log n)."""
    n = len(a)
    b = fast_conv(a)
    conv_len = len(b)
    
    # Softmax weights
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i]
    # This is 2 * cross_correlate(p, a)[i] for i in [0, n-1]
    # Using FFT: IFFT(FFT(p) * conj(FFT(a)))
    size = 1
    while size < conv_len + n - 1:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(P * np.conj(A), size)
    grad = 2.0 * corr[:n]
    
    return smooth_val, grad

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * rho_arr) > (cv - 1.0)
    if cond.any():
        rho = int(np.argmax(cond))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def fista_smooth(a_init, tau_start, tau_end, max_iters, deadline, lr=1.0):

    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()

    n = len(a)

    a_prev = a.copy()

            break
        
        frac = min(1.0, it / max(1, max_iters - 1))
        tau = tau_start * (tau_end / tau_start) ** frac

        # Project gradient to tangent of simplex (sum = 0 direction)
        grad_tan = grad_y - grad_y.mean()
        
        # Adaptive step
        step = lr / (tau + 1.0)
        # Backtracking (simplified)
        # Actually for smoothness L = O(tau * max(a) * 2) roughly
        # Let's use a fixed reasonable step
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        # Clip to maintain non-negativity approximately
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / y_new.sum()
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new

            if M_cur < best_M:

                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses FFT-accelerated FISTA (O(n log n) gradient), reflected-gradient mirror descent,
active-set LP polish, and vectorized coordinate descent. Multi-start with diverse
initializations. Key improvement over previous: 100x faster gradients via FFT.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_grad(a, tau):
    """Compute log-sum-exp smooth max of conv(a,a) and its gradient. O(n log n)."""
    n = len(a)
    b = fast_conv(a)
    conv_len = len(b)  # 2n-1
    
    # Softmax
    b_max = b.max()
    shifted = np.clip(tau * (b - b_max), -700, 700)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient via FFT cross-correlation
    # grad[i] = 2 * sum_k p[k] * a[k-i]
    size = 1
    while size < conv_len + n - 1:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(P * np.conj(A), size)
    grad = 2.0 * corr[:n]
    
    return smooth_val, grad

def proj_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * rho_arr) > (cv - 1.0)
    if cond.any():
        rho = int(np.argmax(cond))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def fista_optimize(a_init, deadline, lr_base=1.0, tau_start=50.0, tau_end=500.0, max_iters=50000):
    """FISTA with adaptive tau on smooth max approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters - 1))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_and_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        # Step size: conservative bound
        step = lr_base / (tau * np.max(a) * 2.0 + 1.0)
        
        a_new = proj_simplex(y - step * grad_tan)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / y_new.sum()
        
        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        
        # Check improvement every 20 iterations
        if it % 20 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a, best_M

def reflected_mirror_descent(a_init, deadline, step_init=0.1, n_top=5):

        top_b = b[top_idx]
        weights = np.exp(100 * (top_b - M))
        weights /= weights.sum()
        
        # Compute weighted reflected gradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * w * a[j]
        
        grad_tan = grad - grad.mean()
        
        # Exponential tilting (mirror descent step)
        step = step_init / (1.0 + 0.01 * it)
        log_w = -step * grad_tan
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        if M < best_M:
            best_M = M
            best_a = a.copy()

    if M_final < best_M:

        best_a = a.copy()
