"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Cutting-plane LP for exact best response (no top-k approximation)
2. Adaptive line search with golden-section refinement
3. Multi-length, multi-start exploration
4. Coordinate descent with golden-section search for final polish
5. Periodic perturbation to escape local minima
Key improvement: exact LP best response via constraint generation, ensuring
the mirror descent direction is truly optimal at each step.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a, b=None):
    """Linear convolution using FFT. If b is None, computes a*a."""
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def compute_M(a):
    """Compute M = max(conv(a,a)) for normalized a."""
    b = fast_conv(a)
    return float(b.max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def build_lp_matrix(a, active_indices):
    """Build sparse constraint matrix for LP. Row k: (a*g)[k] = sum_j a[k-j]*g[j]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_cutting_plane(a, M, max_rounds=50, time_limit=10.0):
    """Solve LP best response exactly using cutting plane (constraint generation).
    
    LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
    Returns normalized g or None.
    """
    n = len(a)
    t_start = time.monotonic()
    
    # Initialize with the most important constraint
    b = fast_conv(a)
    k_init = int(np.argmax(b))
    active = [k_init]
    
    prev_obj = -np.inf
    best_g = None
    
    for rnd in range(max_rounds):
        if time.monotonic() - t_start > time_limit:
            break
        
        A = build_lp_matrix(a, active)
        if A.nnz == 0:
            break
        
        c_obj = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0.0, None)] * n
        
        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs',
                        options={'time_limit': max(1, time_limit - (time.monotonic() - t_start)),
                                 'presolve': True})
        except Exception:
            break
        
        if not r.success or r.x is None:
            break
        
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        obj = -r.fun
        
        if g_sum < 1e-12:
            break
        
        # Check feasibility against all constraints
        ag = fast_conv(a, g)
        max_violation = ag.max() - M
        
        if max_violation <= 1e-7 * M:
            # Feasible - this is the exact LP optimum
            best_g = g / g_sum
            break
        
        # Add most violated constraint
        k_new = int(np.argmax(ag))
        if k_new not in active:
            active.append(k_new)
        else:
            # Numerical issue - constraint already present but violated
            # Add next most violated
            sorted_idx = np.argsort(ag)[::-1]
            added = False
            for kk in sorted_idx:
                kk = int(kk)
                if kk not in active:
                    active.append(kk)
                    added = True
                    break
            if not added:
                best_g = g / g_sum
                break
        
        # Track best feasible-so-far
        if obj > prev_obj:
            prev_obj = obj
            best_g = g / g_sum
    
    return best_g


def line_search(a, g, M_cur, n_steps=8):
    """Find best t in [0,1] for a_new = (1-t)*a + t*g.
    Returns (best_t, best_M)."""
    best_t = 0.0
    best_M = M_cur
    
    # Try a grid of t values
    t_vals = np.linspace(0, 1.0, n_steps)
    Ms = []
    for t in t_vals:
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-10:
            Ms.append(float('inf'))
            continue
        a_t = a_t / s
        Ms.append(fast_conv(a_t).max())
    Ms = np.array(Ms)
    
    # Find best on grid
    best_idx = int(np.argmin(Ms))
    best_t = t_vals[best_idx]
    best_M = Ms[best_idx]
    
    # Refine with golden section around the best
    if 0 < best_idx < len(t_vals) - 1:
        lo = t_vals[max(0, best_idx - 1)]
        hi = t_vals[min(len(t_vals) - 1, best_idx + 1)]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        
        def eval_M(t):
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                return float('inf')
            a_t = a_t / s
            return fast_conv(a_t).max()
        
        Mc = eval_M(c_pt)
        Md = eval_M(d_pt)
        
        for _ in range(20):
            if Mc < Md:
                hi = d_pt
                d_pt = c_pt
                Md = Mc
                c_pt = hi - gr * (hi - lo)
                Mc = eval_M(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                Mc = Md
                d_pt = lo + gr * (hi - lo)
                Md = eval_M(d_pt)
        
        best_t = (lo + hi) / 2.0
        best_M = eval_M(best_t)
    
    return best_t, best_M


def run_lp_descent(a_init, deadline, max_iters=500, perturb_every=30):
    """Main LP mirror descent with cutting plane."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    rng = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Periodic perturbation to escape local minima
        if it > 0 and it % perturb_every == 0 and no_improve > perturb_every // 3:
            noise = rng.normal(0, 0.02, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert = a_pert / sp
                M_pert = fast_conv(a_pert).max()
                if M_pert < best_M + 0.01:
                    a = a_pert.copy()
        
        # Solve LP best response
        M_cur = fast_conv(a).max()
        g = lp_cutting_plane(a, M_cur, max_rounds=30, time_limit=5.0)
        
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Line search
        t_opt, M_opt = line_search(a, g, M_cur, n_steps=10)
        
        if M_opt < best_M - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                best_M = M_opt
                best_a = a.copy()
                no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
    
    # Final check
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def coordinate_descent_step(a, i, n):
    """Find optimal value for a[i] given all others fixed.
    Uses golden section search on the convex objective."""
    x_cur = a[i]
    b = fast_conv(a)
    
    # Affected k range
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Precompute: for each affected k, the value as a function of x = a[i]
    # (a*a)[k] = base_k + slope_k * x + (quad? x^2 : 0)
    c_vals = np.zeros(k_hi - k_lo + 1)
    s_vals = np.zeros(k_hi - k_lo + 1)
    quad_flags = np.zeros(k_hi - k_lo + 1, dtype=bool)
    
    for idx, k in enumerate(range(k_lo, k_hi + 1)):
        if k == 2 * i:
            # (a*a)[k] contains a[i]^2
            c_vals[idx] = b[k] - x_cur * x_cur
            s_vals[idx] = 0.0
            quad_flags[idx] = True
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                s_vals[idx] = 2.0 * a[j]
            else:
                c_vals[idx] = b[k]
                s_vals[idx] = 0.0
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    def obj_at(x_test):
        vals = c_vals + s_vals * x_test
        m = base_max
        if vals.max() > m:
            m = vals.max()
        for idx in range(len(c_vals)):
            if quad_flags[idx]:
                qv = c_vals[idx] + x_test * x_test
                if qv > m:
                    m = qv
        return m
    
    # Golden section search on [0, hi]
    lo = 0.0
    hi = max(2.0 * x_cur, 0.5)
    
    # Make sure hi is large enough
    for _ in range(10):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 10.0:
            hi *= 2.0
        else:
            break
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = obj_at(c_pt)
    fd = obj_at(d_pt)
    
    for _ in range(35):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = obj_at(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = obj_at(d_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, deadline, max_sweeps=50, seed=0):
    """Coordinate descent with random ordering."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        b = fast_conv(a)
        sweep_changed = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            x_opt = coordinate_descent_step(a, i, n)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                a[i] = x_opt
                # Update b incrementally
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += x_opt * x_opt - old * old
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old)
                sweep_changed = True
        
        if sweep_changed:
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s
            M = fast_conv(a).max()
            if M < best_M * (1 - 1e-13):
                best_M = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 3
        
        if no_improve > 8:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_noisy = np.maximum(p + rng.normal(0, 0.05, n), 1e-4)
        inits.append(p_noisy / p_noisy.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # U-shaped (bathtub)
    u_shape = (2 * t - 1) ** 2 + 0.3
    inits.append(u_shape / u_shape.sum())
    
    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())
    
    # Edges emphasized
    edges = np.sin(np.pi * t) ** 0.3 + 0.1
    inits.append(edges / edges.sum())
    
    # Two peaks
    two_peak = np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2) + 0.05
    inits.append(two_peak / two_peak.sum())
    
    # Power law
    pl = (t + 0.1) ** (-0.5) + (1 - t + 0.1) ** (-0.5)
    inits.append(pl / pl.sum())
    
    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)

    print(f"{'='*60}", flush=True)
    
    # Allocate time: parent gets 40%, others split remaining
    total_search_time = deadline - time.monotonic() - 60.0  # reserve 60s
    time_per_length = []
    if lengths_to_try[0] == n_parent:
        time_per_length.append(0.4 * total_search_time)
        remaining_time = 0.6 * total_search_time
        n_other = max(1, len(lengths_to_try) - 1)
        for _ in range(n_other):
            time_per_length.append(remaining_time / n_other)
    else:
        for _ in range(len(lengths_to_try)):
            time_per_length.append(total_search_time / len(lengths_to_try))
    
    # Phase 1: Multi-length search
    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - 60:
            break
        
        t_alloc = time_per_length[li]
        sub_deadline = min(time.monotonic() + t_alloc, deadline - 60)
        
        # Minimum time per length
        if sub_deadline - time.monotonic() < 30:
            sub_deadline = time.monotonic() + 30
            if sub_deadline > deadline - 60:
                break
        
        print(f"\n  Length n={n}, budget={sub_deadline - time.monotonic():.0f}s", flush=True)
        
        # Generate initializations
        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        
        # Limit number of inits based on time
        time_per_init = (sub_deadline - time.monotonic()) / max(1, len(inits))
        max_inits = max(1, int((sub_deadline - time.monotonic()) / 40.0))
        inits = inits[:max_inits]
        
        best_for_n = None
        best_M_for_n = float('inf')
        
        for ii, a0 in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break
            if ii > 0 and (sub_deadline - time.monotonic()) < 40:
                break
            
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            
            print(f"    Init {ii}/{len(inits)}: M_init={compute_M(a0):.8f}", flush=True)
            
            # For the first init (parent), spend more time
            init_deadline = sub_deadline if ii == 0 else min(sub_deadline, time.monotonic() + t_alloc * 0.4)
            
            a_opt = run_optimization_for_length(a0, n, init_deadline)
            M_opt = fast_conv(a_opt).max()
            
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best J={J_val:.6f} (elapsed={time.monotonic()-t_alloc:.1f}s)", flush=True)
            tracker.update(best_for_n)
    
    tracker.report("After Phase 1")
    
    # Phase 2: Intensive polish on best
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 2: Intensive LP polish on best (n={tracker.best.size})", flush=True)
        print(f"{'='*60}", flush=True)
        
        t_polish = min(deadline - 20, time.monotonic() + 150)
        a_polished = run_lp_descent(tracker.best, t_polish, max_iters=200, perturb_every=20)
        tracker.update(a_polished)
        
        # Coordinate descent
        if time.monotonic() < deadline - 10:
            n_pol = tracker.best.size
            max_sw = max(10, int((deadline - 10 - time.monotonic()) / max(0.5, n_pol / 60.0)))
            a_cd = run_coordinate_descent(tracker.best, deadline - 5, max_sweeps=max_sw, seed=99)
            tracker.update(a_cd)
        
        # One more LP pass
        if time.monotonic() < deadline - 5:
            a_final = run_lp_descent(tracker.best, deadline, max_iters=30, perturb_every=15)
            tracker.update(a_final)
    
    tracker.report("After Phase 2")
    
    # Phase 3: Try a slightly different length from best
    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_cur = tracker.best.size
        for mult in [1.15, 0.85, 1.25]:
            if time.monotonic() > deadline - 20:
                break
            new_n = int(n_cur * mult)
            new_n = max(256, min(2048, new_n))
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 10, time.monotonic() + 40)
            a_opt = run_optimization_for_length(a_ext, new_n, sub_dl)
            tracker.update(a_opt)
    
    tracker.report("After Phase 3")
    
    # Final: ensure valid output
    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)
    
    # Normalize for output (scale doesn't matter for J, but keep reasonable)
    best_out = tracker.best.copy()
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_out]
