        return a

    best_M = fast_conv(a).max()

def run_subgradient_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=100.0,
                              lr_init=0.3, lr_end=0.002):
    """Fast subgradient method with Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    momentum = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = subgradient_step(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum ** 2)) / 2.0
        # Projected gradient step
        a_new = np.maximum(y - lr * grad / (n * y.sum()), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix for LP: A @ g gives (a*g)_k for k=0,...,2n-2."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    rows, cols, vals = build_A_sparse(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 5.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_extrapolate(hist_x, hist_Fx, m, max_omega=1.0):
    """Anderson acceleration: find best extrapolation of history."""
    residuals = np.array([hist_x[i] - hist_Fx[i] for i in range(m)])
    # shape (m, n)
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    # Clamp for stability
    alpha = np.clip(alpha, -max_omega, max_omega)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(hist_x[0]))
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    return np.maximum(mixed, 0)


def run_lp_adaptive(a, deadline, max_iters=500, anderson_window=6):
    """LP mirror descent with adaptive step size and Anderson acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = compute_J(a)
    no_imp = 0
    hist_x = []
    hist_Fx = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        J_curr = 2 * n * M / (a.sum() ** 2)
        
        # Get LP best response
        g = lp_best_response(a, M, n)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Try multiple step sizes (from large to small)
        improved = False
        for t in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.02, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            J_t = 2 * n * M_t / (a_t.sum() ** 2)
            if J_t < best_J - 1e-12:
                a = a_t
                best_M = M_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(((1.0 - 0.1) * a + 0.1 * g).copy())  # small step for history
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 4 and no_imp == 0:
            a_and = anderson_extrapolate(hist_x, hist_Fx, len(hist_x), max_omega=0.5)
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
                M_and = fast_conv(a_and).max()
                J_and = 2 * n * M_and / (a_and.sum() ** 2)
                if J_and < best_J * 1.01:
                    a = a_and
    
    return best_a


def run_lp_polish(a, deadline, max_iters=200):
    """Final LP polish with very small steps."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, n)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        improved = False
        for t in [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            J_t = 2 * n * M_t / (a_t.sum() ** 2)
            if J_t < best_J - 1e-14:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 10:
                break
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.001))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.001))
    
    # Various structured initializations
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t**alpha + (1-t)**alpha) / (2 * (0.5**alpha + 0.5**alpha)), 0.01))
    
    for f in [1, 2, 3, 4, 6, 8]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    
    # Gaussian
    inits.append(np.exp(-3 * (t - 0.5)**2) + 0.05)
    # Double Gaussian
    inits.append(np.exp(-6 * (t - 0.3)**2) + np.exp(-6 * (t - 0.7)**2) + 0.03)
    # Raised cosine
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    # Sine power
    inits.append(np.abs(np.sin(np.pi * t))**2 + 0.05)
    
    # Random smooth
    k = max(5, n // 10)
    kern = np.exp(-0.5 * (np.arange(-k//2, k//2+1) / (k//4))**2)
    kern /= kern.sum()
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.02, 0.01))
    
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_flat_spectral(a_rand, n_iters=80, beta=0.85))
    
    # GS from parent
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(gs_flat_spectral(p, n_iters=100, beta=0.9))
    
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):

    1. GS spectral initialization for flat autocorrelation
    2. Subgradient pre-optimization with Nesterov acceleration (fast, coarse)
    3. LP mirror descent with adaptive backtracking step size + Anderson acceleration
    4. Final LP polish with small steps
    5. Multi-start with diverse initializations and focused length exploration

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: iteratively flatten the power spectrum."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        power = 0.3 + 0.7 * beta
        mag_new = mag ** power
        mag_new = mag_new / (mag_new.max() + 1e-30) * (mag.max() + 1e-30)
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def subgradient_step(a, tau):
    """Compute subgradient of max_k (a*a)_k with softmax weighting.
    Returns (grad_a, p, b, b_max)."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad_a = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad_a[lo:hi + 1] += 2.0 * p[k] * seg
    return grad_a, p, b, b_max


def run_subgradient_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=80.0,
                              lr_init=0.3, lr_end=0.002):

    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    momentum = 0.9
    scale = 1.0 / n
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = subgradient_step(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum ** 2)) / 2.0
        a_new = np.maximum(y - lr * scale * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


class LPContext:
    """Precomputed structure for LP to avoid rebuilding matrix each iteration."""
    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.num_constraints = 2 * n - 1
        self.num_nonzeros = len(rows)

    def solve_lp(self, a, M):
        """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
        vals = a[self.rows - self.cols]
        A = csr_matrix((vals, (self.rows, self.cols)),
                       shape=(self.num_constraints, self.n))
        c = -np.ones(self.n)
        b_ub = np.full(self.num_constraints, M)
        bounds = [(0, None)] * self.n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
            if r.success and r.x is not None and r.x.sum() > 1e-15:
                return np.maximum(r.x, 0)
        except Exception:
            pass
        return None


def anderson_extrapolate(hist_x, hist_Fx, m, max_omega=0.8):
    """Anderson acceleration: find best extrapolation of history."""
    residuals = np.array([hist_x[i] - hist_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -max_omega, max_omega)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(hist_x[0]))
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    return np.maximum(mixed, 0)

        if gs < 1e-10:

        g /= gs

        improved = False

            if s < 1e-10:

        # Anderson acceleration tracking

        hist_Fx.append(np.maximum(a + 0.1 * (g - a), 0.0).copy())

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        allot = min(time_per_length, remaining * 0.35)
        if allot < 8:
            continue

        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')

        n_iters = max(400, int(6000 * (512.0 / n)))
        for idx, a0 in enumerate(inits):
            if time.monotonic() - t0 > allot:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            sub_dl = time.monotonic() + allot / (1.5 + idx * 0.5)
            a = run_subgradient_nesterov(a0, sub_dl, max_iters=n_iters,
                                          tau_start=1.0, tau_end=60.0,
                                          lr_init=0.25, lr_end=0.001)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: M={best_M:.7f} J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: LP mirror descent (main optimization) ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (adaptive)", flush=True)
    print(f"{'='*50}", flush=True)

    sorted_lengths = sorted(best_per_length.keys(), key=lambda nn: compute_J(best_per_length[nn]))

    # Focus LP on the best 2-3 lengths
    lp_lengths = sorted_lengths[:3] if len(sorted_lengths) >= 3 else sorted_lengths
    remaining_for_lp = deadline - 40 - time.monotonic()
    if remaining_for_lp > 0:
        # Allocate LP time proportionally (more to better lengths)
        weights = [1.0 / (i + 1) for i in range(len(lp_lengths))]
        total_w = sum(weights)
        weights = [w / total_w for w in weights]

        for i, n_try in enumerate(lp_lengths):
            if time.monotonic() > deadline - 30:
                break
            alloc = remaining_for_lp * weights[i] * 0.8
            if alloc < 10:
                continue
            a_start = best_per_length[n_try].copy()
            # If this is the best length, start from the global best
            if best.seq is not None and best.seq.size == n_try:
                a_start = best.seq.copy()

            lp_ctx = LPContext(n_try)
            max_iters = max(30, int(alloc / 0.15))
            print(f"  LP n={n_try}, budget={alloc:.0f}s, max_iters~{max_iters}", flush=True)

            a = run_lp_adaptive(a_start, time.monotonic() + alloc * 0.65,
                                max_iters=max_iters, anderson_window=6, lp_ctx=lp_ctx)
            best.update(a)

            # Second pass with remaining time for this length
            rem = time.monotonic() + alloc * 0.35 - time.monotonic()
            if rem > 5:
                a2 = run_lp_adaptive(a, time.monotonic() + rem,
                                     max_iters=int(max_iters * 0.4),
                                     anderson_window=6, lp_ctx=lp_ctx)
                best.update(a2)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Final LP polish ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Final LP Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 5:
        n_best = best.seq.size
        lp_ctx = LPContext(n_best)
        polish_time = min(60.0, deadline - time.monotonic() - 3)
        max_iters_p = max(20, int(polish_time / 0.15))
        a_final = run_lp_polish(best.seq, time.monotonic() + polish_time,
                                max_iters=max_iters_p, lp_ctx=lp_ctx)
        best.update(a_final)

        # One more adaptive pass
        if time.monotonic() < deadline - 3:
            a_final2 = run_lp_adaptive(a_final, deadline, max_iters=30,
                                       anderson_window=5, lp_ctx=lp_ctx)
            best.update(a_final2)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Length perturbation ===
    if best.seq is not None and time.monotonic() < deadline - 10:
        n_best = best.seq.size
        for delta in [-32, 32, -16, 16, 64, -64]:
            if time.monotonic() > deadline - 8:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2560:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = run_subgradient_nesterov(a_ext, min(time.monotonic() + 12, deadline - 5),
                                             max_iters=200, tau_start=5, tau_end=50,
                                             lr_init=0.1, lr_end=0.005)
            best.update(a_ext)
            if time.monotonic() < deadline - 5:
                lp_ctx2 = LPContext(n_try)
                a_lp = run_lp_adaptive(a_ext, min(time.monotonic() + 15, deadline - 3),
                                       max_iters=20, anderson_window=4, lp_ctx=lp_ctx2)
                best.update(a_lp)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
