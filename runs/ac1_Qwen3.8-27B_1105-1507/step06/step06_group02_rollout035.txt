def fast_conv(a):

    n = len(a)

    while size < 2 * n - 1:

    fa = np.fft.rfft(a, size)

        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad_sym(a, tau):
    """Compute gradient of smooth max of conv(a,a) for symmetric a."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    # For symmetry, only process first half of convolution lags
    # and double the contribution (except center)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # (a*a)(k) = sum_{j=lo}^{hi} a[j] * a[k-j]
        # d/d a[m] of (a*a)(k) = a[k-m] if m in [lo, hi], plus a[m] if k-m in [lo,hi]
        # For the gradient: d(a*a)(k)/da[m] = a[k-m] + (1 if m == k-m else 0) * a[m]... 
        # Actually: (a*a)(k) = sum_j a[j]*a[k-j]
        # d/d a[m] = a[k-m] (if 0 <= k-m <= n-1) + a[m] (if 0 <= m <= n-1 and m appears as k-j, i.e., j = k-m)
        # Wait, more carefully:
        # (a*a)(k) = sum_j a[j] * a[k-j]
        # If we differentiate w.r.t. a[m]:
        #   - a[m] appears as a[j] when j=m: contributes a[k-m] (if valid)
        #   - a[m] appears as a[k-j] when k-j=m, i.e., j=k-m: contributes a[k-m] = a[m] (if valid)
        # So d/d a[m] = a[k-m] (if k-m valid) + a[m] (if k-m valid) = 2*a[k-m] if k != 2m, or a[m] if k = 2m
        # Wait no. Let me be more careful.
        # (a*a)(k) = sum_{j} a[j] * a[k-j]
        # The term a[m] appears:
        #   1) As a[j] when j = m: the term is a[m] * a[k-m], derivative = a[k-m]
        #   2) As a[k-j] when k-j = m, i.e., j = k-m: the term is a[k-m] * a[m], derivative = a[k-m]
        # But case 1 and case 2 are the same term when j = m = k-m, i.e., k = 2m.
        # So d/d a[m] of (a*a)(k) = a[k-m] + a[k-m] = 2*a[k-m] if k-m is valid and k != 2m
        #                                      = a[m] if k = 2m (since a[k-m] = a[m])
        # Actually no: if k = 2m, then in the sum, the term j=m gives a[m]*a[m], 
        # and d/d a[m] of a[m]*a[m] = 2*a[m]. So it's still 2*a[m] = 2*a[k-m].
        # So the formula is: d/d a[m] of (a*a)(k) = 2*a[k-m] if 0 <= k-m <= n-1, else 0.
        # Wait, that's only if both j=m and j=k-m are in the valid range.
        # If j=m is valid (0<=m<=n-1, always true) and k-m is valid (0<=k-m<=n-1):
        #   contribution = a[k-m]
        # If j=k-m is valid (0<=k-m<=n-1) and m is valid (always):
        #   contribution = a[m]
        # But these are the same condition! k-m is valid means 0<=k-m<=n-1.
        # So if 0<=k-m<=n-1, both terms contribute, giving 2*a[k-m]... 
        # UNLESS m = k-m (i.e., k=2m), in which case it's the same term counted once.
        # Wait no. In the sum sum_j a[j]*a[k-j], each j gives one term.
        # For j=m: term = a[m]*a[k-m], d/d a[m] = a[k-m]
        # For j=k-m: term = a[k-m]*a[m], d/d a[m] = a[k-m]
        # These are two different terms in the sum (unless m = k-m).
        # If m = k-m (k=2m), then j=m=k-m, so it's the same term, and d/d a[m] of a[m]^2 = 2*a[m].
        # Either way, the total derivative is 2*a[k-m] (when k-m is valid).
        
        # So the gradient contribution from lag k to position m is:
        # 2*p[k]*a[k-m] if 0 <= k-m <= n-1, else 0.
        
        if 0 <= k - lo <= n - 1:  # This is always true by construction of lo, hi
            pass  # already handled below
        
        # More efficient: for each m in [lo, hi], grad[m] += 2 * p[k] * a[k-m]
        # But by symmetry, we can halve this work.
        for m in range(lo, hi + 1):
            grad[m] += 2.0 * pk * a[k - m]
    
    # Enforce symmetry on gradient
    grad = symmetrize(grad)
    return grad, p, b, b_max


def run_smooth_descent(a, deadline, max_iters=3000, tau_start=1.0, tau_end=500.0,
                      lr_init=0.5, lr_end=0.005, use_symmetry=True):
    """Smooth gradient descent with Nesterov acceleration and symmetry."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = symmetrize(a)
    a = np.maximum(a, 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    momentum = 0.9
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        # Adaptive tau: start small (explore), end large (exploit)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_max_grad_sym(a, tau)
        
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y = np.maximum(y - lr * grad, 0.0)
        if use_symmetry:
            y = symmetrize(y)
        y = np.maximum(y, 1e-10)
        y /= y.sum()
        
        a_new = y + ((t_k - 1.0) / t_k) * (y - a)
        if use_symmetry:
            a_new = symmetrize(a_new)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new
    
    return best_a


def build_sym_A(n, a):
    """Build the constraint matrix for the symmetric LP.
    Variables: h[0], ..., h[n//2] (first half of symmetric sequence)
    Constraints: (a*full_a)(k) <= M for k = 0, ..., n-1 (first half of convolution)
    """
    half_n = n // 2 + 1  # number of free variables
    # Full sequence from h: full[j] = h[min(j, n-1-j)]
    # For constraint k: (a*full)(k) = sum_j a[j] * full[k-j]
    #                   = sum_j a[j] * h[min(k-j, n-1-(k-j))]
    # Since a is also symmetric: a[j] = h[min(j, n-1-j)]
    
    rows = []
    cols = []
    vals = []
    
    for k in range(n):  # only first half of convolution (by symmetry)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            # full[k-j] = h[min(k-j, n-1-(k-j))]
            idx = min(k - j, n - 1 - (k - j))
            if idx < half_n:
                rows.append(k)
                cols.append(idx)
                vals.append(a[j])
    
    return rows, cols, vals, half_n


def lp_best_response_sym(a, M):
    """LP best response for symmetric sequences."""
    n = len(a)
    rows, cols, vals, half_n = build_sym_A(n, a)
    
    if len(rows) == 0:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(n, half_n))
    c = -np.ones(half_n)
    b_ub = np.full(n, M)
    bounds = [(0, None)] * half_n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            # Expand back to full symmetric sequence
            h = np.maximum(r.x, 0)
            full = np.zeros(n)
            for j in range(n):
                full[j] = h[min(j, n - 1 - j)]
            return full
    except Exception:
        pass
    return None


def run_lp_sym(a, deadline, max_iters=100, t_base=0.3, min_t=0.01):
    """LP mirror descent with symmetry."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = symmetrize(a)
    a = np.maximum(a, 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        
        g = lp_best_response_sym(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        g = symmetrize(g)
        
        t = max(min_t, t_base / (1 + 0.1 * it))
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        a_new = symmetrize(a_new)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new
    
    return best_a


def gs_symmetric(a, n_iters=200, beta=0.85):
    """Gerchberg-Saxton with symmetry constraint."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = symmetrize(a)
    a = np.maximum(a, 0)
    a /= a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Flatten the magnitude
        mag_target = np.ones_like(mag) * (mag.mean() + 1e-30)
        mag_new = (1.0 - beta) * mag + beta * mag_target
        
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = symmetrize(a_new)
        a_new = np.maximum(a_new, 0)
        sc = a_new.sum()
        if sc < 1e-15:
            break
        a_new /= sc
        
        M_c = fast_conv(a_new).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def resample_sym(src, n):
    """Resample a sequence to length n, maintaining symmetry."""
    src = np.asarray(src, dtype=np.float64)
    src = symmetrize(src)
    if src.size == n:
        return src.copy()
    # Resample the first half
    half_src = src[:src.size//2 + 1]
    half_n = n // 2 + 1
    t_old = np.linspace(0, 1, half_src.size)
    t_new = np.linspace(0, 1, half_n)
    half_new = np.maximum(np.interp(t_new, t_old, half_src), 0.0)
    # Build full symmetric sequence
    full = np.zeros(n)
    for j in range(n):
        full[j] = half_new[min(j, n - 1 - j)]
    return full


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)
    
    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def make_inits_sym(n, rng, parent):
    """Create diverse symmetric initializations."""
    inits = []
    half_n = n // 2 + 1
    t = np.linspace(0, 1, half_n)
    
    # From parent
    if parent is not None:
        p = resample_sym(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.001))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.001))
        inits.append(gs_symmetric(p, n_iters=100, beta=0.9))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Curved profiles
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        h = 0.3 + 0.7 * (1.0 - np.abs(2*t - 1)**alpha)
        full = np.zeros(n)
        for j in range(n):
            full[j] = h[min(j, n-1-j)]
        full /= full.sum()
        inits.append(full)
    
    # Cosine profiles
    for f in [1, 2, 3, 4]:
        h = 0.5 + 0.4 * np.cos(np.pi * f * t)
        h = np.maximum(h, 0.01)
        full = np.zeros(n)
        for j in range(n):
            full[j] = h[min(j, n-1-j)]
        full /= full.sum()
        inits.append(full)
    
    # Gaussian profiles
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            h = np.exp(-((t - center)**2) / (2*width**2)) + 0.05
            full = np.zeros(n)
            for j in range(n):
                full[j] = h[min(j, n-1-j)]
            full /= full.sum()
            inits.append(full)
    
    # Random smooth symmetric
    for _ in range(3):
        h = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
        # Smooth
        k = max(3, half_n // 10)
        kern = np.ones(k) / k
        h = np.convolve(h, kern, 'same')
        h = np.maximum(h, 0.01)
        full = np.zeros(n)
        for j in range(n):
            full[j] = h[min(j, n-1-j)]
        full /= full.sum()
        inits.append(full)
    
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_symmetric(a_rand, n_iters=80, beta=0.85))
    
    return inits

        pass

    p1_start = time.monotonic()

    short_n = 128

                                     lr_init=0.3, lr_end=0.02)

    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-length smooth optimization", flush=True)
    print(f"{'='*60}, flush=True)
    p2_start = time.monotonic()

    n_lengths = len(lengths)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetric sequence constraint (exploits problem symmetry for efficiency)
2. Adaptive smooth-max gradient descent with Nesterov momentum and restarts
3. Symmetric LP best-response for precise local refinement

Key improvements: symmetry exploitation, FFT-based gradient, adaptive tau scheduling,

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def symmetrize(a):
    """Enforce symmetry: a[j] = a[n-1-j]."""
    n = len(a)
    return (a + a[::-1]) / 2.0


def compute_J(a):
    """Compute the objective value."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad_sym(a, tau):
    """Compute gradient of smooth max of conv(a,a) with symmetry."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    # Shift to avoid overflow
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    # Only iterate over lags with significant weight
    threshold = 1e-10
    active = np.where(p > threshold)[0]
    for k in active:
        pk = p[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[m] += 2 * p[k] * a[k-m] for m in [lo, hi]
        # Vectorized: a[k-hi : k-lo+1] reversed gives a[k-m] for m=lo..hi
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * pk * seg

    # Enforce symmetry
    grad = symmetrize(grad)
    return grad, p, b, b_max


def run_smooth_descent(a, deadline, max_iters=5000, tau_start=1.0, tau_end=500.0,
                       lr_init=0.5, lr_end=0.005, use_restart=True):
    """Smooth gradient descent with Nesterov acceleration, symmetry, and restarts."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = symmetrize(a)
    a = np.maximum(a, 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    momentum = 0.9
    stagnant_count = 0
    no_improve_threshold = 50

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        # Adaptive tau: geometric schedule
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad_sym(a, tau)

        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
            stagnant_count = 0
        else:
            stagnant_count += 1

        # Restart mechanism: if stagnant, perturb and reduce momentum
        if use_restart and stagnant_count > no_improve_threshold:
            y = np.maximum(a + 0.05 * np.random.randn(n), 1e-6)
            y = symmetrize(y)
            y = np.maximum(y, 1e-6)
            y /= y.sum()
            momentum = min(0.95, momentum + 0.05)
            stagnant_count = 0
            no_improve_threshold = max(30, no_improve_threshold - 5)

        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y = np.maximum(y - lr * grad, 0.0)
        y = symmetrize(y)
        y = np.maximum(y, 1e-10)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y /= sy

        a_new = y + ((t_k - 1.0) / t_k) * (y - a)
        a_new = symmetrize(a_new)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new

def build_sym_A(n, a):
    """Build sparse constraint matrix for symmetric LP.
    Variables: h[0],...,h[n//2] (half of symmetric sequence)
    Constraints: (a*full_a)(k) <= M for k=0,...,n-1
    """
    half_n = n // 2 + 1
    rows = []
    cols = []
    vals = []

    for k in range(n):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            # full[k-j] = h[min(k-j, n-1-(k-j))]
            idx = k - j
            if idx > n - 1 - idx:
                idx = n - 1 - idx
            if idx < half_n:
                rows.append(k)
                cols.append(idx)
                vals.append(a[j])

    return rows, cols, vals, half_n


def lp_best_response_sym(a, M, time_limit=20):
    """LP best response for symmetric sequences."""
    n = len(a)
    rows, cols, vals, half_n = build_sym_A(n, a)

    if len(rows) == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(n, half_n))
    c = -np.ones(half_n)
    b_ub = np.full(n, M)
    bounds = [(0, None)] * half_n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            h = np.maximum(r.x, 0)
            full = np.zeros(n)
            for j in range(n):
                full[j] = h[min(j, n - 1 - j)]
            return full
    except Exception:
        pass
    return None


def run_lp_sym(a, deadline, max_iters=100, t_base=0.3, min_t=0.005):

    a = np.maximum(a.astype(np.float64), 1e-8)
    a = symmetrize(a)
    a = np.maximum(a, 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break

        g = lp_best_response_sym(a, M, time_limit=15)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        g = symmetrize(g)
        g = np.maximum(g, 0)
        gs2 = g.sum()
        if gs2 < 1e-10:
            break
        g /= gs2

        t = max(min_t, t_base / (1 + 0.08 * it))
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        a_new = symmetrize(a_new)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new

    return best_a


def gs_symmetric(a, n_iters=200, beta=0.85):
    """Gerchberg-Saxton with symmetry constraint for initialization."""
    a = np.maximum(a.astype(np.float64), 0)
    a = symmetrize(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)

        # Flatten magnitude toward mean
        mag_mean = mag.mean() + 1e-30
        mag_new = (1.0 - beta) * mag + beta * mag_mean

        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = symmetrize(a_new)
        a_new = np.maximum(a_new, 0)
        sc = a_new.sum()
        if sc < 1e-15:
            break
        a_new /= sc

        M_c = fast_conv(a_new).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_new.copy()
        a = a_new

    return best_a


def resample_sym(src, n):
    """Resample a sequence to length n, maintaining symmetry."""
    src = np.asarray(src, dtype=np.float64)
    src = symmetrize(src)
    if src.size == n:
        return src.copy()
    half_src = src[:src.size // 2 + 1]
    half_n = n // 2 + 1
    t_old = np.linspace(0, 1, half_src.size)
    t_new = np.linspace(0, 1, half_n)
    half_new = np.maximum(np.interp(t_new, t_old, half_src), 0.0)
    full = np.zeros(n)
    for j in range(n):
        full[j] = half_new[min(j, n - 1 - j)]
    return full

    for f in [1, 2, 3, 4, 6]:
        h = 0.5 + 0.45 * np.cos(np.pi * f * t)
        h = np.maximum(h, 0.01)
        full = np.zeros(n)
        for j in range(n):
            full[j] = h[min(j, n - 1 - j)]
        full /= full.sum()
        inits.append(full)

    # Gaussian mixtures
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.15, 0.2, 0.3]:
            h = np.exp(-((t - center) ** 2) / (2 * width ** 2)) + 0.03
            full = np.zeros(n)
            for j in range(n):
                full[j] = h[min(j, n - 1 - j)]
            full /= full.sum()
            inits.append(full)

    # Random smooth symmetric
    for _ in range(4):
        h = np.maximum(rng.uniform(0.2, 1.0, half_n), 0.01)
        k = max(3, half_n // 8)
        kern = np.ones(k) / k
        h = np.convolve(h, kern, 'same')
        h = np.maximum(h, 0.01)
        full = np.zeros(n)
        for j in range(n):
            full[j] = h[min(j, n - 1 - j)]
        full /= full.sum()
        inits.append(full)

    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_symmetric(a_rand, n_iters=60, beta=0.85))

    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    """Main search: symmetric smooth descent + LP polish with multi-resolution warm start."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best.update(parent)

    # Determine lengths to explore
    lengths = [512, 768, 1024, 1280, 1536, 1792, 2048]
    if parent is not None:
        pn = parent.size
        for delta in [-128, -64, 0, 64, 128, 256]:
            nl = pn + delta
            if 256 <= nl <= 2560:
                lengths.append(nl)
    lengths = sorted(set(lengths))
    print(f"Lengths to explore: {lengths}", flush=True)
    print(f"Total budget: {budget_s - 30:.0f}s", flush=True)

    # === PHASE 1: Multi-resolution warm start (20s) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-resolution warm start", flush=True)
    print(f"{'=' * 60}", flush=True)
    p1_start = time.monotonic()

    # Build up from short to long
    current = np.ones(128) / 128
    if parent is not None and parent.size <= 256:
        current = parent.copy()

    current = symmetrize(current)
    current = np.maximum(current, 0.01)
    current /= current.sum()

    # Optimize short sequence
    current = run_smooth_descent(current, min(time.monotonic() + 10, deadline - 200),
                                 max_iters=300, tau_start=2.0, tau_end=50.0,
                                 lr_init=0.4, lr_end=0.02)
    best.update(current)

    # Build up
    for target_n in [256, 512, 1024]:
        if time.monotonic() > deadline - 200:
            break
        current = resample_sym(current, target_n)
        current = np.maximum(current, 0.005)
        current /= current.sum()
        current = run_smooth_descent(current, min(time.monotonic() + 15, deadline - 150),
                                     max_iters=200, tau_start=3.0, tau_end=40.0,
                                     lr_init=0.3, lr_end=0.03)
        best.update(current)
        print(f"  Built to n={target_n}: J={best.score:.6f}", flush=True)

    print(f"Phase 1: {time.monotonic() - p1_start:.1f}s", flush=True)

    # === PHASE 2: Multi-length smooth optimization ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length smooth optimization", flush=True)
    print(f"{'=' * 60}", flush=True)
    p2_start = time.monotonic()

    best_per_length = {}

    for n in lengths:
        if time.monotonic() > deadline - 250:
            break

        # Time allocation: shorter sequences get more time (cheaper per iter)
        base_time = 50.0 * (768.0 / max(n, 256))
        base_time = max(20.0, min(80.0, base_time))
        remaining = deadline - 250 - time.monotonic()
        allot = min(base_time, remaining * 0.7)
        if allot < 12:
            continue
        dl = time.monotonic() + allot

        ref = best.seq if best.seq is not None else parent
        inits = make_inits_sym(n, rng, ref)

        best_n = None
        best_M = float('inf')
        n_iters = max(300, int(4000 * (512.0 / n)))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = symmetrize(a0)
            a0 = np.maximum(a0, 1e-6)
            sa = a0.sum()
            if sa < 1e-10:
                continue
            a0 /= sa

            # Allocate decreasing time to later inits
            time_left = dl - time.monotonic()
            frac = max(0.15, 1.0 / (1.0 + idx * 0.35))
            sub_dl = time.monotonic() + time_left * frac
            if sub_dl - time.monotonic() < 2:
                break

            a = run_smooth_descent(a0, sub_dl, max_iters=n_iters,
                                   tau_start=1.0, tau_end=400.0,
                                   lr_init=0.45, lr_end=0.003)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            elapsed = time.monotonic() - (dl - allot)
            print(f"  n={n:4d}: J={J:.6f} ({elapsed:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 2")
    print(f"Phase 2: {time.monotonic() - p2_start:.1f}s", flush=True)

    # === PHASE 3: LP refinement ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: LP refinement", flush=True)
    print(f"{'=' * 60}", flush=True)
    p3_start = time.monotonic()

    if best.seq is not None:
        n_best = best.seq.size
        # Estimate LP cost
        est_ms = 1.5 * (n_best / 512.0) ** 2.5
        print(f"  Best n={n_best}, est LP={est_ms:.0f}ms/iter", flush=True)

        lp_avail = deadline - 60 - time.monotonic()
        if lp_avail > 30:
            budget_lp = lp_avail * 0.6
            max_iters = max(15, int(budget_lp / max(0.005, est_ms / 1000.0)))
            max_iters = min(max_iters, 200)
            print(f"  LP budget={budget_lp:.0f}s, max_iters={max_iters}", flush=True)
            a_lp = run_lp_sym(best.seq, min(time.monotonic() + budget_lp, deadline - 50),
                              max_iters=max_iters, t_base=0.3, min_t=0.005)
            best.update(a_lp)

    # Try LP on other promising lengths
    if best_per_length and time.monotonic() < deadline - 50:
        sorted_ls = sorted(best_per_length.keys(), key=lambda x: compute_J(best_per_length[x]))
        for n_try in sorted_ls[:3]:
            if time.monotonic() > deadline - 40:
                break
            if n_try == best.seq.size:
                continue
            a_c = best_per_length[n_try].copy()
            # Quick smooth polish
            a_c = run_smooth_descent(a_c, min(time.monotonic() + 10, deadline - 35),
                                     max_iters=200, tau_start=10.0, tau_end=100.0,
                                     lr_init=0.15, lr_end=0.01)
            best.update(a_c)
            # LP
            if time.monotonic() < deadline - 30:
                est = 1.5 * (a_c.size / 512.0) ** 2.5
                mi = max(10, int(15.0 / max(0.005, est / 1000.0)))
                a_lp = run_lp_sym(a_c, min(time.monotonic() + 15, deadline - 25),
                                  max_iters=mi, t_base=0.25, min_t=0.01)
                best.update(a_lp)

    best.report("Phase 3")
    print(f"Phase 3: {time.monotonic() - p3_start:.1f}s", flush=True)

    # === PHASE 4: Final polish ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Final polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 10:
        # High-tau smooth descent for fine-tuning
        a_fin = run_smooth_descent(best.seq, min(time.monotonic() + 25, deadline - 5),
                                   max_iters=1000, tau_start=50.0, tau_end=1000.0,
                                   lr_init=0.08, lr_end=0.001)
        best.update(a_fin)

        # One more LP pass
        if time.monotonic() < deadline - 5:
            a_lp2 = run_lp_sym(a_fin, deadline, max_iters=30, t_base=0.1, min_t=0.003)
            best.update(a_lp2)

    # === PHASE 5: Length exploration ===
    if best.seq is not None and time.monotonic() < deadline - 15:
        n_b = best.seq.size
        for delta in [-64, -128, 64, 128, 256, -256]:
            if time.monotonic() > deadline - 8:
                break
            n_try = n_b + delta
            if n_try < 256 or n_try > 2560:
                continue
            a_ext = resample_sym(best.seq, n_try)
            a_ext = np.maximum(a_ext, 0.005)
            a_ext /= a_ext.sum()
            a_ext = run_smooth_descent(a_ext, min(time.monotonic() + 8, deadline - 5),
                                       max_iters=300, tau_start=5.0, tau_end=80.0,
                                       lr_init=0.2, lr_end=0.01)
            best.update(a_ext)

    best.report("Phase 5")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(256) / 256
        best.score = compute_J(best.seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic() - best.t0:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)
    return [float(x) for x in best.seq]
