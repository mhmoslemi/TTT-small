"""
Improved step function optimization using enhanced LP alternating with
golden-section line search, peak-flattening LP variants, multi-n exploration,
subgradient refinement, and simulated annealing restarts.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, variant=0, M_ref=None, top_k=5, alpha=0.5):
    """
    Solve various LP formulations.
    variant=0: min M s.t. conv(a,g)<=M, sum(g)=1, g>=0
    variant=1: min alpha*M + (1-alpha)*sum(conv(a,g)) s.t. conv(a,g)<=M, sum(g)=1, g>=0
    variant=2: min M s.t. conv(a,g)<=M for top-k peaks only, sum(g)=1, g>=0
    """
    n = len(a)
    a = a / np.sum(a)
    
    if variant == 2 and M_ref is not None:
        # Solve LP that only constrains peaks near the current max
        conv_a = compute_conv(a)
        M_cur = float(np.max(conv_a))
        threshold = M_cur * (1 - 0.1)  # constrain peaks within 10% of max
        active = np.where(conv_a >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv_a)])
    else:
        active = np.arange(2 * n - 1)
    
    A = build_conv_matrix(a)
    A_active = A[active, :]
    
    g = cp.Variable(n, nonneg=True)

def golden_section_line_search(a, g, lo=0.0, hi=1.0, tol=1e-4, max_iter=30):
    """
    Golden section search for best t in [lo, hi] minimizing obj((1-t)*a + t*g).
    """
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    def obj_at(t):
        a_t = (1 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-10:
            return np.inf
        a_t = a_t / s
        return compute_obj(a_t)
    
    gr = hi - lo
    c = lo + (1 - phi) * gr
    d = lo + phi * gr
    fc = obj_at(c)
    fd = obj_at(d)
    
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            gr = hi - lo
            c = lo + (1 - phi) * gr
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            gr = hi - lo
            d = lo + phi * gr
            fd = obj_at(d)
    
    t_best = (lo + hi) / 2
    return t_best, obj_at(t_best)

def peak_flatten(a, time_budget=20.0):
    """
    Directly try to reduce the top peaks by redistributing mass.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Find top peaks
        top_k = 5
        peak_positions = np.argsort(conv)[-top_k:][::-1]
        max_val = conv[peak_positions[0]]
        
        improved = False
        for k_peak in peak_positions:
            if time.monotonic() - start > time_budget:
                break
            # Find indices that contribute most to this peak
            contributors = []
            for i in range(n):
                j = k_peak - i
                if 0 <= j < n:
                    contributors.append((a[i] * a[j], i, j))
            contributors.sort(reverse=True)
            
            for contrib, i, j in contributors[:5]:
                if i >= j:
                    continue
                # Try moving mass from position i to somewhere that doesn't contribute to this peak
                for target in range(n):
                    if target == i or target == j:
                        continue
                    # Check if target contributes to this peak
                    tj = k_peak - target
                    if 0 <= tj < n and a[tj] > 0:
                        continue  # Would still contribute
                    
                    for frac in [0.01, 0.05]:
                        delta = frac * a[i]
                        if delta > a[i]:
                            continue
                        a[i] -= delta
                        a[target] += delta
                        a = np.maximum(a, 0)
                        s = float(np.sum(a))
                        if s > 1e-10:
                            a = a / s
                        obj_new = compute_obj(a)
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[target] -= delta
                            a = a / s
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            break

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
