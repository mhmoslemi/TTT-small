"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines Nesterov-accelerated subgradient, LP alternating, multiplicative updates,
and multi-n exploration with diverse initializations (including bimodal/U-shaped).

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_nesterov(a, n_iters=5000, time_budget=100.0, verbose=False):
    """
    Nesterov-accelerated subgradient method on the simplex.
    Minimizes max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov parameters
    x = a.copy()
    y = a.copy()
    x_prev = a.copy()
    lr0 = 0.02
    momentum_fn = lambda t: (t - 1) / (t + 2) if t > 1 else 0.0

    for t in range(1, n_iters + 1):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov momentum
        if t > 1:
            beta = momentum_fn(t)
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
        else:
            y = x.copy()

        # Subgradient at y
        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * y[j]
        # Center the gradient (projection onto hyperplane sum=1)
        grad = grad - np.mean(grad)

        # Step
        eta = lr0 / np.sqrt(t)
        x_new = project_simplex(y - eta * grad)

        # Check improvement
        obj_new = compute_obj(x_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = x_new.copy()

        x_prev = x
        x = x_new

        if verbose and t % 2000 == 0:
            print(f"        subgrad t={t}: obj={best_obj:.8f}")

    return best_a

    stall = 0

    for t in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv_x = compute_conv(x)
        k_star = int(np.argmax(conv_x))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * x[j]
        grad = grad - np.mean(grad)

        x_new = project_simplex(x - lr * grad)
        obj_new = compute_obj(x_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.7
                if lr < 0.001:
                    # Restart with perturbation
                    rng = np.random.default_rng(t)
                    x = best_a * np.exp(rng.normal(0, 0.05, n))
                    x = project_simplex(x)
                    best_obj = compute_obj(x)
                    best_a = x.copy()
                    lr = lr0 * 0.5
                    stall = 0
                    if lr < 0.0005:
                        break

        x = x_new

        if verbose and t % 2000 == 0:
            print(f"        subgrad_basic t={t}: obj={best_obj:.8f}, lr={lr:.5f}")

    return best_a


def multiplicative_updates(a, n_iters=5000, time_budget=50.0, verbose=False):
    """
    Multiplicative weight updates: a[i] *= exp(-eta * grad[i]), then renormalize.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1

    for t in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv_a = compute_conv(a)
        k_star = int(np.argmax(conv_a))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        grad = grad - np.mean(grad)

        # Multiplicative update with clipping
        update = np.exp(-eta * grad)
        update = np.clip(update, 0.5, 2.0)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            eta *= 0.95

        if verbose and t % 1000 == 0:
            print(f"        mult_upd t={t}: obj={best_obj:.8f}, eta={eta:.4f}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog (HiGHS) as primary solver.
    """
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Variables: g[0..n-1], M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    # Try cvxpy as fallback
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget=100.0, verbose=False):
    """
    LP alternating: iteratively solve min-peak LP and blend with current.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Line search
        best_t_obj = np.inf
        best_t_a = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_a
            stall = 0
            if verbose and it % 10 == 0:
                print(f"        LP alt it{it}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 15:
                # Perturb
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                stall = 0
                if stall > 30:
                    break

    return best_a


def coordinate_descent(a, time_budget=30.0):
    """
    Pairwise coordinate descent: swap mass between peak-contributing pairs.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from high to low within peak-contributing indices
        indices = [i for i in range(n) if 0 <= k_star - i < n]
        indices.sort(key=lambda i: -a[i])

        for i in indices[:20]:
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if j == i or j < 0 or j >= n:
                continue
            if a[i] > a[j]:
                for frac in [0.03, 0.07, 0.15]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        if not improved:
            # Try random pairs
            rng = np.random.default_rng(it)
            for _ in range(50):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                for frac in [0.02, 0.05]:
                    if a[i] > a[j]:
                        delta = frac * a[i]
                        a[i] -= delta
                        a[j] += delta
                        new_obj = compute_obj(a)
                        if new_obj < best_obj - 1e-14:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[j] -= delta
                            break
            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    from scipy.interpolate import CubicSpline
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def generate_initialization(n, mode, rng):
    """Generate diverse initializations."""
    if mode == 0:
        # Uniform
        return np.ones(n) / n
    elif mode == 1:
        # Bimodal (U-shaped)
        i = np.arange(n)
        a = 1.0 / (np.minimum(i + 1, n - i) + 0.5)
        a = a / a.sum()
        return a
    elif mode == 2:
        # Strong bimodal
        i = np.arange(n)
        center = n / 2.0
        a = np.exp(-0.3 * ((i - center) / center) ** 2)
        a = a + 0.5 * np.exp(-((i - center) / (center * 0.3)) ** 2)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
        return a
    elif mode == 3:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
        a = a / a.sum()
        return a
    elif mode == 4:
        # Random smooth
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        i = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * i / n + phase)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
        return a
    elif mode == 5:
        # Two peaks
        c1 = n // 4
        c2 = 3 * n // 4
        w = n // 8
        i = np.arange(n)
        a = np.exp(-((i - c1) ** 2) / (2 * w ** 2)) + np.exp(-((i - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
        return a
    elif mode == 6:
        # Power law at edges
        i = np.arange(n)
        d = np.minimum(i + 1, n - i)
        a = d ** (-0.5)
        a = a / a.sum()
        return a
    else:
        # Random
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()


def objective_with_n(a):
    """Full objective including n factor."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))
