Improved peak-reduction search using soft-subgradient (log-sum-exp),

Targets improvement over 1.506547 toward the theoretical lower bound of 2n/(2n-1).

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax_obj(a, tau):
    """
    Compute soft objective: 2*n*logsumexp(conv(a,a)/tau) * tau / (sum(a))^2
    This is a smooth upper bound on the true objective.
    Actually, let's use: 2*n*tau*log(sum(exp(conv/tau))) / (sum(a))^2

    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:

    conv = compute_conv(a)
    shifted = conv / tau
    shifted -= np.max(shifted)  # numerical stability
    log_sum_exp = np.max(shifted) + np.log(np.sum(np.exp(shifted)))

    soft_max = float(np.max(conv)) + tau * float(np.log(np.sum(np.exp((conv - np.max(conv)) / tau))))
    obj = 2.0 * n * soft_max / (s * s)
    return obj, conv


def soft_subgradient_step(a, tau, v, mu, step_size):

    W_ext2 = rfft(w_ext, fft_size)

    a_new = normalize_simplex(a_new)

def soft_subgradient_optimize(a, max_iters=3000, time_budget=60.0, tau_init=0.01, tau_rate=0.999):
    """
    Soft-subgradient optimization with log-sum-exp objective.
    """
    n = len(a)

    best_a = a.copy()

def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=20):
    """
    Solve: min t subject to (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(fixed_seq, M, time_limit=20):
    """
    Solve: max sum(g) subject to (fixed_seq * g)[k] <= M for all k, g >= 0.
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate(a, max_iters=50, time_budget=100.0, use_minpeak=True):

    n = len(a)

        # Line search

def generate_initial_sequences(n, rng):
    """Generate diverse initial sequences emphasizing asymmetry."""
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # 1. Uniform (baseline)
    starts.append(np.ones(n) / n)

    # 2. Asymmetric triangular (peak off-center)
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        peak = int(n * frac)
        tri = np.zeros(n)
        rise = peak
        fall = n - 1 - peak
        for j in range(n):
            if j <= peak:
                tri[j] = (j + 1) / (rise + 1) if rise > 0 else 1.0
            else:
                tri[j] = (n - j) / (fall + 1) if fall > 0 else 0.0
        tri = np.maximum(tri, 1e-6)
        starts.append(tri / np.sum(tri))

    # 3. Sine-modulated (asymmetric)
    for freq in [1, 2, 3, 5]:
        for amp in [0.5, 0.8]:
            for phase in [0, np.pi/2, np.pi]:
                a = 0.5 + 0.5 * np.sin(2 * np.pi * freq * i / n + phase)
                a = np.maximum(a, 0.01)
                starts.append(a / np.sum(a))

    # 4. Exponential ramp (strongly asymmetric)
    for rate in [0.5, 1.0, 2.0, 3.0]:
        a = np.exp(rate * i / n)
        starts.append(a / np.sum(a))
        a2 = np.exp(-rate * i / n)
        starts.append(a2 / np.sum(a2))

    # 5. Gaussian (narrow, wide, off-center)
    for sigma_frac in [0.1, 0.2, 0.3, 0.5]:
        for center_frac in [0.3, 0.5, 0.7]:
            c = n * center_frac
            a = np.exp(-((i - c) ** 2) / (2 * (n * sigma_frac) ** 2))
            a = np.maximum(a, 1e-8)
            starts.append(a / np.sum(a))

    # 6. Random structured
    for _ in range(8):
        base = np.ones(n)
        noise = rng.normal(0, 0.5, n)
        a = np.abs(base + noise)
        a = np.maximum(a, 0.01)
        starts.append(a / np.sum(a))

    # 7. "Anti-palindromic" - minimize a[i]*a[n-1-i]
    a = np.zeros(n)
    a[:n//2] = 1.0  # First half only
    a = np.maximum(a, 1e-8)  # Small epsilon to keep positive
    starts.append(a / np.sum(a))

    # 8. Cosine window (flat-top)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * i / n)
    a = np.maximum(a, 0.01)
    starts.append(a / np.sum(a))

    return starts


def projected_gradient_lse(a, max_iters=2000, time_budget=60.0, tau_init=0.005, tau_min=0.0005):
    """
    Projected gradient descent on the log-sum-exp smoothed objective.
    f(a) = 2*n*tau*log(sum(exp(conv(a,a)/tau))) / (sum(a))^2
    We minimize this with a >= 0, and handle the (sum)^2 denominator.
    
    Actually, since the objective is homogeneous, we fix sum(a)=1 and minimize:
    2*n*tau*log(sum(exp(conv(a,a)/tau)))
    
    The gradient of log(sum(exp(conv/tau))) w.r.t. a[i] is:
    (1/tau) * sum_k exp(conv[k]/tau) / sum(exp(conv/tau)) * d(conv[k])/da[i]
    = (1/tau) * sum_k w[k] * sum_j (delta[k-i-j] + delta[k-j-i]) * a[j]  ... 
    
    Actually d(conv[k])/da[i] = a[k-i] + a[i-(k-n+1)]... this is getting complex.
    Let me just use the subgradient approach with soft weights.
    """
    n = len(a)
    a = normalize_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_init
    step = 0.04
    v = np.zeros(n)
    mu = 0.75

    # Precompute FFT size
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    A_ext = rfft(np.zeros(fft_size), fft_size)  # placeholder

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        # Soft weights
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient computation via FFT
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_cur = rfft(a, fft_size)
        corr = irfft(A_cur * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]
        
        grad_centered = grad - np.mean(grad)
        gn = float(np.linalg.norm(grad_centered))
        if gn < 1e-15:
            break

        v = mu * v - step * grad_centered / gn
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        # Adaptive: decrease tau and step
        tau = max(tau * 0.998, tau_min)
        step = max(step * 0.999, 0.003)

    return best_a, best_obj


def multi_peak_subgradient(a, num_peaks=5, max_iters=2000, time_budget=40.0):
    """Subgradient focusing on top-K peaks."""
    n = len(a)
    a = normalize_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    step = 0.03
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_k = np.argsort(conv)[-num_peaks:]
        weights = np.zeros(len(conv))
        for k in top_k:
            weights[k] = 1.0 / num_peaks

        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_cur = rfft(a, fft_size)
        corr = irfft(A_cur * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]
        grad_centered = grad - np.mean(grad)
        gn = float(np.linalg.norm(grad_centered))
        if gn < 1e-15:
            break

        a_new = normalize_simplex(a - step * grad_centered / gn)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try half step
            a_try = normalize_simplex(a - 0.5 * step * grad_centered / gn)
            if compute_obj(a_try) < best_obj:
                a = a_try
                best_obj = compute_obj(a)
                best_a = a.copy()
            step *= 0.97

    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)

    return a_new

For linear convolution: `conv(a, b)[k] = sum_j a[j] * b[k-j]` (with b[i]=0 for i<0 or i>=m).

We want: `grad[i] = 2 * sum_k weights[k] * a[k-i]` where the sum is over k such that both k and k-i are valid indices (0 <= k < 2n-1, 0 <= k-i < n, i.e., i <= k < i+n).

Let's define: `grad[i] = 2 * sum_{k=i}^{min(i+n-1, 2n-2)} weights[k] * a[k-i]`
`= 2 * sum_{j=0}^{min(n-1, 2n-2-i)} weights[i+j] * a[j]`

This is the cross-correlation of `weights` and `a` at lag `i`:
`cross[i] = sum_j weights[i+j] * a[j]`

Using FFT: If we compute `ifft(FFT(a) * conj(FFT(weights)))` (with zero-padding to size >= 2n-1), we get:
`result[m] = sum_j a[j] * weights[m+j]` (for linear, no wrap-around)

So `result[i] = cross[i]` and `grad[i] = 2 * result[i]`.
