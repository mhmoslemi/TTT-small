"""
Iterated LP-based peak reduction with subgradient descent, peak-targeted
coordinate descent, and multi-start exploration for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_toeplitz_matrix(a):
    """Build matrix A where (A @ g)[k] = conv(a, g)[k]. Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_min_peak(a, time_limit=10):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0, M>=0.
    Returns (g_opt, M_opt) or (None, inf).
    """
    n = len(a)
    if n > 2000:
        return None, np.inf

    A = build_toeplitz_matrix(a)  # shape (2n-1, n)
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: min M -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality constraints: conv(a,g)[k] - M <= 0 for all k
    # A_ineq @ x <= b_ineq
    A_ineq = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ineq = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s_g = np.sum(g)
            if s_g > 1e-10:
                g = g / s_g
            M_opt = float(res.x[-1])
            return g, M_opt
    except Exception:
        pass
    return None, np.inf


def solve_lp_flexible(a, M_target, time_limit=10):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_target, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    if n > 2000:
        return None

    A = build_toeplitz_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:

    return None

def subgradient_step(a, lr, n_active=3):
    """
    One projected subgradient step for minimizing max(conv(a,a))/sum(a)^2.
    Uses the top n_active peaks for a smoothed subgradient.
    """
    n = len(a)
    conv = compute_conv(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    # Get top peaks
    flat_conv = conv.ravel()
    top_indices = np.argsort(flat_conv)[::-1][:n_active]
    top_values = flat_conv[top_indices]

    # Weight peaks by softmax for smoothness
    max_v = top_values[0]
    weights = np.exp((top_values - max_v) * 10)  # sharp weighting
    weights = weights / np.sum(weights)

    # Subgradient: sum over active peaks of weight * 2 * a[p-k]
    sg = np.zeros(n)
    for w, p in zip(weights, top_indices):
        p = int(p)
        for k in range(n):
            idx = p - k
            if 0 <= idx < n:
                sg[k] += 2.0 * w * a[idx]

    # Gradient of full objective J = 2n * max_b / S^2:
    # dJ/d(a[k]) = (2n/S^2) * sg[k] - (2n*max_b/S^3)
    max_b = float(np.max(conv))
    grad = (2.0 * n / (s * s)) * (sg - max_b / s)

    # Projected gradient step: a_new = max(0, a - lr * grad), then normalize
    a_new = np.maximum(a - lr * grad, 0)
    s_new = np.sum(a_new)
    if s_new < 1e-10:
        return a
    a_new = a_new / s_new

    return a_new

    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    lr = lr_start
    no_improve_count = 0

    while time.monotonic() - start_time < time_budget:
        a_try = subgradient_step(a, lr, n_active)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-10:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            no_improve_count = 0
            # Can increase lr slightly
            lr = min(lr * 1.05, 0.5)
        else:
            no_improve_count += 1
            lr *= 0.7
            if no_improve_count > 30:
                # Try a larger jump
                lr = lr_start * 0.5
                no_improve_count = 0
                if time.monotonic() - start_time > time_budget * 0.8:
                    break

        if lr < 1e-5:
            break

    return best_a


def peak_coordinate_descent(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent targeting the top peaks of the convolution.
    For each peak, tries to redistribute mass from contributing elements
    to non-contributing elements.
    """
    start_time = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    n_iter = 0

    while time.monotonic() - start_time < time_budget:
        n_iter += 1
        # Find top peaks
        peak_indices = np.argsort(conv)[::-1][:5]

        improved = False
        for p in peak_indices:
            if time.monotonic() - start_time > time_budget:
                break

            # Elements contributing to conv[p]
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            contributing = list(range(i_min, i_max + 1))

            # Elements NOT contributing to this peak
            non_contributing = [j for j in range(n) if j not in set(contributing)]

            if not non_contributing:
                continue

            # Try moving mass from the largest contributor to a non-contributor
            contributor_vals = [(i, a[i]) for i in contributing]
            contributor_vals.sort(key=lambda x: -x[1])

            for src in contributor_vals[:3]:
                src_idx = src[0]
                if a[src_idx] < 1e-6:
                    continue

                # Try a few fractions to move
                for frac in [0.01, 0.05, 0.1, 0.2]:
                    delta = frac * a[src_idx]

                    # Move to the non-contributing element with largest "room"
                    # (i.e., where adding mass least increases other peaks)
                    best_dst = -1
                    best_new_obj = best_obj
                    for dst_idx in non_contributing[:20]:  # limit search
                        a_try = a.copy()
                        a_try[src_idx] -= delta
                        a_try[dst_idx] += delta
                        if a_try[src_idx] < 0:
                            continue
                        s_try = np.sum(a_try)
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_new_obj:
                            best_new_obj = obj_try
                            best_dst = dst_idx

                    if best_dst >= 0 and best_new_obj < best_obj - 1e-12:
                        a = a.copy()
                        a[src_idx] -= delta
                        a[best_dst] += delta
                        a = np.maximum(a, 0)
                        a = a / np.sum(a)
                        best_obj = best_new_obj
                        best_a = a.copy()
                        conv = compute_conv(a)
                        improved = True
                        break

            if improved:
                break

        if not improved:
            # Try smaller steps
            if n_iter > 100:
                break
            # Perturb and retry
            rng = np.random.default_rng(n_iter)
            a_pert = a * (1 + rng.normal(0, 0.01, n))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / np.sum(a_pert)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_obj:
                a = a_pert
                best_a = a_pert.copy()
                best_obj = obj_pert
                conv = compute_conv(a)

    return best_a


def make_initialization(n, kind='ramp', c=5.0):
    """Create various initial sequences of length n."""
    if kind == 'ramp':
        a = np.arange(1, n + 1, dtype=np.float64)
    elif kind == 'rev_ramp':
        a = np.arange(n, 0, -1, dtype=np.float64)
    elif kind == 'soft_cliff':
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-c * (x - 0.5)))
    elif kind == 'quad':
        a = np.arange(1, n + 1, dtype=np.float64) ** 2
    elif kind == 'exp':
        a = np.exp(c * np.linspace(0, 1, n))
    elif kind == 'uniform':
        a = np.ones(n)
    elif kind == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    elif kind == 'plateau':
        frac = 0.5
        n_plat = max(1, int(frac * n))
        start = (n - n_plat) // 2
        a = np.zeros(n)
        a[start:start + n_plat] = 1.0
        # Linear ramps
        if start > 0:
            a[:start] = np.linspace(1.0, 0.0, start)[::-1]
        end = start + n_plat
        if end < n:
            a[end:] = np.linspace(1.0, 0.0, n - end)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 0.01:
                obj = compute_obj(parent)
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Heavy LP-based refinement combined with subgradient descent,
    peak-targeted coordinate descent, multi-start with diverse initializations,
    and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent / np.sum(parent)

    # === Phase 1: LP refinement on parent (main workhorse) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP refinement on parent (n={len(best_seq) if best_seq is not None else '?'})")
    print(f"{'='*60}")

    if best_seq is not None:
        n_p = len(best_seq)
        # Phase 1a: Heavy LP loop
        lp_budget = min(400.0, (deadline - time.monotonic()) * 0.35)
        print(f"  Running LP loop with budget={lp_budget:.0f}s, n={n_p}")
        a_lp = lp_refine_loop(best_seq, max_iters=200, time_budget=lp_budget,
                              verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  After LP: {best_score:.8f} ***")
        else:
            print(f"  After LP: {obj_lp:.8f} (no improvement)")

        # Phase 1b: Subgradient descent after LP
        if time.monotonic() < deadline - 100:
            sg_budget = min(100.0, (deadline - time.monotonic()) * 0.15)
            print(f"  Running subgradient with budget={sg_budget:.0f}s")
            a_sg = subgradient_descent(best_seq, time_budget=sg_budget,
                                       lr_start=0.05, n_active=3)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  After SG: {best_score:.8f} ***")

            # LP again after SG
            if time.monotonic() < deadline - 80:
                lp_budget2 = min(100.0, (deadline - time.monotonic()) * 0.2)
                a_lp2 = lp_refine_loop(best_seq, max_iters=50,
                                       time_budget=lp_budget2, verbose=False)
                obj_lp2 = compute_obj(a_lp2)
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  After LP2: {best_score:.8f} ***")

    # === Phase 2: Peak coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Peak coordinate descent")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 80:
        cd_budget = min(120.0, (deadline - time.monotonic()) * 0.15)
        print(f"  Budget={cd_budget:.0f}s")
        a_cd = peak_coordinate_descent(best_seq, time_budget=cd_budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  After CD: {best_score:.8f} ***")
        else:
            print(f"  After CD: {obj_cd:.8f}")

        # LP after CD
        if time.monotonic() < deadline - 60:
            lp_budget3 = min(80.0, (deadline - time.monotonic()) * 0.2)
            a_lp3 = lp_refine_loop(best_seq, max_iters=30,
                                   time_budget=lp_budget3, verbose=False)
            obj_lp3 = compute_obj(a_lp3)
            if obj_lp3 < best_score:
                best_score = obj_lp3
                best_seq = a_lp3.copy()
                print(f"  After LP3: {best_score:.8f} ***")

    # === Phase 3: Multi-start with diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-start exploration")
    print(f"{'='*60}")

    n_ref = len(best_seq) if best_seq is not None else 700
    n_options = [400, 500, 600, 700, 800, 900, 1000, 1100]
    # Add neighbors of current best n
    n_options.extend([n_ref - 50, n_ref, n_ref + 50, n_ref + 100])
    n_options = sorted(set([n for n in n_options if 200 <= n <= 1500]))

    init_kinds = ['ramp', 'soft_cliff', 'quad', 'exp', 'plateau', 'uniform']
    init_params = [1.0, 5.0, 10.0, 20.0, 50.0]

    start_budget_per = min(60.0, (deadline - time.monotonic()) * 0.15)

    for n_try in n_options:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        tried = 0
        for kind in init_kinds:
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # For soft_cliff and exp, try different parameters
            params_to_try = init_params if kind in ('soft_cliff', 'exp') else [1.0]

            for c_val in params_to_try[:2]:  # Limit to 2 params per kind
                a_init = make_initialization(n_try, kind=kind, c=c_val)

                # Quick LP refinement
                alloc = min(remaining * 0.15, 40.0)
                if alloc < 5:
                    continue
                a_opt = lp_refine_loop(a_init, max_iters=30,
                                       time_budget=alloc, verbose=False)
                obj = compute_obj(a_opt)

                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  n={n_try}, {kind}(c={c_val}): {best_score:.8f} *** UPDATE")
                else:
                    print(f"  n={n_try}, {kind}(c={c_val}): {obj:.8f}")

                tried += 1
                if tried >= 3:
                    break

    # === Phase 4: Intensive refinement of best ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Intensive refinement of best (n={len(best_seq) if best_seq is not None else '?'})")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        # LP with more iterations
        remaining = deadline - time.monotonic()
        lp_budget4 = min(remaining * 0.4, 150.0)
        a_final = lp_refine_loop(best_seq, max_iters=100,
                                 time_budget=lp_budget4, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  After final LP: {best_score:.8f} ***")

        # Subgradient polish
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            sg_budget = min(remaining * 0.3, 50.0)
            a_sg2 = subgradient_descent(best_seq, time_budget=sg_budget,
                                        lr_start=0.02, n_active=5)
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                print(f"  After SG2: {best_score:.8f} ***")

        # One more CD
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            cd_budget2 = min(remaining * 0.4, 40.0)
            a_cd2 = peak_coordinate_descent(best_seq, time_budget=cd_budget2)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  After CD2: {best_score:.8f} ***")

    # === Phase 5: Random restarts with perturbation ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Random restarts")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 20:
        rng = np.random.default_rng(seed)
        n_best = len(best_seq)

        for restart in range(6):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # Different perturbation strategies
            if restart % 3 == 0:
                # Gaussian multiplicative noise
                scale = 0.1 + 0.05 * restart
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            elif restart % 3 == 1:
                # Blend with a different initialization
                kind = init_kinds[(restart + 2) % len(init_kinds)]
                a_other = make_initialization(n_best, kind=kind, c=10.0)
                alpha = 0.2 + 0.1 * restart
                a_pert = (1 - alpha) * best_seq + alpha * a_other
            else:
                # Sparse random spikes
                a_pert = best_seq.copy()
                num_spikes = max(3, n_best // 30)
                for _ in range(num_spikes):
                    idx = rng.integers(0, n_best)
                    a_pert[idx] += rng.uniform(0.05, 0.3) * np.mean(best_seq)

            a_pert = np.maximum(a_pert, 0)
            s_p = np.sum(a_pert)
            if s_p < 1e-10:
                continue
            a_pert = a_pert / s_p

            alloc = min(remaining * 0.2, 30.0)
            a_opt = lp_refine_loop(a_pert, max_iters=20,
                                   time_budget=alloc, verbose=False)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart}: {obj:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure validity
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    s_final = np.sum(best_seq)
    if s_final < 0.01:
        best_seq = np.ones(100)
        best_score = 2.0

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
