import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def peak_push_step(a, step_size, top_k=5):
    """
    Direct peak-pushing step: identify top peaks and move mass away from them.
    For each peak at position p, gradient of output[p] w.r.t. a[i] is 2*a[p-i].
    To reduce the peak, decrease a[i] where a[p-i] is large.
    """
    n = len(a)
    conv = compute_conv(a)
    # Get top k peak positions
    peak_positions = np.argsort(conv)[::-1][:top_k]
    
    # Compute "peak gradient" - weighted sum of gradients of top peaks
    # weight[i] = sum over peaks p of w_p * a[p-i] (where valid)
    peak_grad = np.zeros(n)
    for p in peak_positions:
        # d(output[p])/d(a[i]) = 2*a[p-i] if 0 <= p-i < n
        for i in range(n):
            ji = p - i
            if 0 <= ji < n:
                peak_grad[i] += 2 * a[ji]
    
    # We want to decrease a where peak_grad is high and increase where it's low
    # To maintain the sum, we do: a_new = a * exp(-step_size * (peak_grad - mean(peak_grad)))
    # This is a multiplicative weight update
    pg_centered = peak_grad - np.mean(peak_grad)
    a_new = a * np.exp(-step_size * pg_centered / (np.max(np.abs(pg_centered)) + 1e-10))
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / np.sum(a_new)

    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    # d(ls)/d(a[i]) = sum_p w[p] * d(output[p])/d(a[i]) = sum_p w[p] * 2*a[p-i]

    # Full objective gradient

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj

- Let f(a) = max_k conv(a,a)[k], g(a) = (sum a)^2
- We want min f(a)/g(a)

Let me try yet another angle. The key quantity is:
`R = max_k conv(a,a)[k] / (sum a)^2`

We want to minimize 2n * R.

Note that sum_k conv(a,a)[k] = (sum a)^2, so:
`R = max_k conv(a,a)[k] / sum_k conv(a,a)[k]`

This is the ratio of the peak to the total sum of the convolution. It's like a "concentration" measure of the convolution.

For a uniform distribution over 2n-1 points, this ratio would be 1/(2n-1), giving objective 2n/(2n-1) ≈ 1.

So we want the convolution to be as "spread out" as possible.

**New idea: Entropy-based objective**

Instead of minimizing the max, we could maximize the entropy of the convolution (which encourages spreading). Then at the end, evaluate the actual max.

H(conv/S^2) = -sum_k p_k * log(p_k) where p_k = conv[k] / S^2

This is a smooth, concave objective that we can maximize with gradient methods. The gradient encourages spreading mass in the convolution, which is exactly what we want.

The gradient of entropy w.r.t. a[i]:

Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS on smoothed objective, peak-pushing subgradient, coordinate descent,
and diverse random restarts with adaptive multi-resolution approach.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_and_grad(a, beta):
    """Compute logsumexp-smoothed objective and gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj


def log_space_lbfgs(a_init, beta_schedule, iterations_per_stage=200, time_budget=100.0):
    """
    L-BFGS optimization in log-space with staged beta schedule.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x_mean = np.mean(x)
    x = x - x_mean  # center for numerical stability
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    
    for stage_beta in beta_schedule:
        if time.monotonic() - start > time_budget:
            break
        
        def func_and_grad(x_vec):
            x_c = x_vec - np.max(x_vec)
            a_vec = np.exp(x_c)
            a_vec = a_vec / np.sum(a_vec)
            obj, grad_a = smooth_obj_and_grad(a_vec, stage_beta)
            grad_x = grad_a * a_vec
            return obj, grad_x
        
        try:
            result = minimize(func_and_grad, x, method='L-BFGS-B',
                             jac=True, options={'maxiter': iterations_per_stage,
                                                'maxfun': iterations_per_stage * 2,
                                                'gtol': 1e-6})
        except Exception:
            break
        
        x = result.x
        x_c = x - np.max(x)
        a_vec = np.exp(x_c)
        a_vec = a_vec / np.sum(a_vec)
        true_obj = compute_obj(a_vec)
        
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a_vec.copy()
            print(f"    L-BFGS beta={stage_beta:.0f}: {best_obj:.8f}")
    
    return best_a


def peak_push_optimize(a_init, time_budget=60.0, n_steps=5000, top_k=3):
    """
    Direct peak-pushing optimization using subgradient method.
    At each step, identify the peak and take a step to reduce it.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    step_size = 0.01
    
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        peak_pos = np.argmax(conv)
        
        # Gradient of the peak w.r.t. a[i] = 2*a[peak_pos - i] (if valid)
        peak_grad = np.zeros(n)
        for i in range(n):
            ji = peak_pos - i
            if 0 <= ji < n:
                peak_grad[i] = 2.0 * a[ji]
        
        # Subgradient step: decrease a where peak_grad is high
        # Maintain sum by using multiplicative update
        pg = peak_grad - np.mean(peak_grad)
        # Adaptive step size
        scale = np.max(np.abs(pg)) + 1e-10
        a = a * np.exp(-step_size * pg / scale)
        a = np.maximum(a, 1e-15)
        a = a / np.sum(a)
        
        # Decay step size
        step_size *= 0.9995
        
        # Check improvement periodically
        if step % 50 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a


def topk_objective_and_grad(a, k, beta):
    """
    Objective: sum of top-k convolution values (smoothed with beta).
    This is smoother than logsumexp for large beta.
    """

    shifted = np.clip(shifted, -500, 500)

    e_k = e ** k
    e_k_sum = float(np.sum(e_k))
    w = e_k / e_k_sum
    # Approximate top-k sum
    topk_sum = cmax + k * np.log(e_k_sum) / (beta * k)  # Hmm this isn't quite right

    eff_beta = beta * k
    shifted2 = eff_beta * (conv - cmax)
    shifted2 = np.clip(shifted2, -500, 500)
    e2 = np.exp(shifted2)
    e2_sum = float(np.sum(e2))
    w2 = e2 / e2_sum
    ls_k = cmax + np.log(e2_sum) / eff_beta
    
    obj = 2.0 * n * ls_k / (sum_a * sum_a)
    
    size = 1

    W = rfft(w2, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls_k / sum_a)
    
    return obj, grad_obj


def coordinate_descent(a_init, time_budget=30.0, seed=42):
    """
    Coordinate descent focusing on elements that contribute to peaks.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)
    
    start = time.monotonic()
    iterations = 0
    
    while time.monotonic() - start < time_budget:
        iterations += 1
        conv = compute_conv(a)
        peak_pos = int(np.argmax(conv))
        
        # Elements contributing to the peak
        i_min = max(0, peak_pos - (n - 1))
        i_max = min(peak_pos, n - 1)
        
        # Try adjusting pairs of elements
        improved = False
        candidates = list(range(i_min, i_max + 1))
        rng.shuffle(candidates)
        
        for i in candidates[:20]:  # Try at most 20 random candidates
            if time.monotonic() - start > time_budget:
                break
            j = peak_pos - i
            if j < 0 or j >= n:
                continue
            if i == j:
                continue
            
            # Try moving mass from i to some other position
            # Better: try adjusting a[i] and a[j] to reduce the peak
            # The peak contribution from pair (i, j) is a[i] * a[j]
            # To reduce it, decrease one of them
            
            # Try decreasing a[i] slightly and increasing a random other position
            for delta in [0.001, 0.005, 0.01, 0.02]:
                a_try = a.copy()
                # Decrease a[i]
                a_try[i] -= delta * a[i]
                if a_try[i] < 1e-15:
                    continue
                # Increase a random position far from peak
                target = rng.integers(0, n)
                # Choose target that minimizes new peak contribution
                a_try[target] += delta * a[i]
                a_try = a_try / np.sum(a_try)
                
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
        
        if not improved and iterations > 100:
            # Try random perturbations
            a_try = a * np.exp(rng.normal(0, 0.01, n))
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / np.sum(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
            elif iterations > 500:
                break
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def generate_structured_init(n, strategy, rng):
    """Generate structured initial sequences."""
    if strategy == 0:
        # Uniform with random perturbation
        a = np.ones(n) * (1 + 0.1 * rng.standard_normal(n))
    elif strategy == 1:
        # Cosine taper
        t = np.arange(n) / (n - 1)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = a + 0.05 * rng.standard_normal(n)
    elif strategy == 2:
        # Sine modulation
        t = np.arange(n)
        freq = rng.uniform(1, 5)
        a = 1 + 0.3 * np.sin(2 * np.pi * freq * t / n)
    elif strategy == 3:
        # Exponential decay from center
        center = n / 2
        t = np.abs(np.arange(n) - center) / center
        a = np.exp(-t * rng.uniform(0.5, 2.0))
    elif strategy == 4:
        # Two-lobe
        center = n / 2
        t = np.abs(np.arange(n) - center) / center
        a = np.exp(-t * 2) + 0.5 * np.exp(-np.abs(t - 0.5) * 5)
    elif strategy == 5:
        # Periodic
        freq = int(rng.integers(2, 6))
        t = np.arange(n)
        a = 1 + 0.2 * np.cos(2 * np.pi * freq * t / n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Primary: L-BFGS in log-space with staged logsumexp smoothing.
Secondary: Projected subgradient peak-pushing, coordinate descent,
multi-n exploration, and diverse random restarts.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_and_grad(a, beta):
    """Compute logsumexp-smoothed objective and gradient w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def _lbfgs_func_grad(x_vec, beta):
    """Top-level function for L-BFGS: smoothed obj and grad in log-space."""
    x_c = x_vec - np.max(x_vec)
    a_vec = np.exp(x_c)
    a_vec = a_vec / np.sum(a_vec)
    obj, grad_a = smooth_obj_and_grad(a_vec, beta)
    grad_x = grad_a * a_vec
    # Zero out gradient components that are numerically unstable
    grad_x = np.where(np.isfinite(grad_x), grad_x, 0.0)
    return obj, grad_x


def lg_lbfgs(a_init, beta_schedule, iters_per_stage=100, time_budget=100.0):
    """L-BFGS in log-space with staged beta schedule."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x = x - np.mean(x)

    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()

    for stage_beta in beta_schedule:
        if time.monotonic() - start > time_budget:
            break

        try:
            result = minimize(_lbfgs_func_grad, x, method='L-BFGS-B',
                              jac=True, args=(stage_beta,),
                              options={'maxiter': iters_per_stage,
                                       'maxfun': iters_per_stage * 3,
                                       'gtol': 1e-7,
                                       'ftol': 1e-10})
        except Exception:
            break

        x = np.asarray(result.x, dtype=np.float64)
        if not np.all(np.isfinite(x)):
            break
        x = x - np.max(x)
        a_vec = np.exp(x)
        a_vec = a_vec / np.sum(a_vec)
        true_obj = compute_obj(a_vec)

            best_obj = true_obj
            best_a = a_vec.copy()
            print(f"    L-BFGS beta={stage_beta:.0f}: {best_obj:.8f}")

def project_simplex(v, S):
    """Project vector v onto simplex {a >= 0, sum(a) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - S)
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_push_projected(a_init, time_budget=60.0, n_steps=5000):
    """
    Projected subgradient method: directly minimize the peak of the convolution.
    Uses projection onto the simplex for feasibility.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    S = float(np.sum(a))
    a = a / S  # normalize to sum=1

    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    step_size = 0.02
    momentum = np.zeros(n)
    mu = 0.9

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        peak_pos = int(np.argmax(conv))

        # Subgradient of max w.r.t. a[i] = 2*a[peak_pos - i] if valid
        peak_grad = np.zeros(n)
        for i in range(n):
            ji = peak_pos - i
            if 0 <= ji < n:
                peak_grad[i] = 2.0 * a[ji]

        # Also account for the (sum a)^2 in denominator
        # d/d(a[i]) [max_conv / (sum a)^2] = (peak_grad[i] * (sum a)^2 - max_conv * 2 * (sum a)) / (sum a)^4
        # = peak_grad[i] / (sum a)^2 - 2 * max_conv / (sum a)^3
        # Since sum a = 1: = peak_grad[i] - 2 * max_conv
        max_conv_val = conv[peak_pos]
        grad_full = peak_grad - 2.0 * max_conv_val

        # Momentum step
        momentum = mu * momentum + (1 - mu) * grad_full

        # Projected gradient step
        a_new = a - step_size * momentum
        a_new = project_simplex(a_new, 1.0)

        # Adaptive step size
        obj_new = compute_obj(a_new)
        if obj_new > best_obj * 1.01:
            step_size *= 0.7
        else:
            step_size = min(step_size * 1.05, 0.05)

        a = a_new

        if step % 50 == 0 or step == n_steps - 1:
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()

    return best_a


def coord_descent_peak(a_init, time_budget=30.0, seed=42):
    """
    Coordinate descent: for each coordinate i, find optimal a[i] via 1D search.
    Focuses on coordinates contributing to the peak.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    S = float(np.sum(a))
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    no_imp_count = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_pos = int(np.argmax(conv))

        # Coordinates contributing to peak
        i_min = max(0, peak_pos - (n - 1))
        i_max = min(peak_pos, n - 1)
        candidates = list(range(i_min, i_max + 1))
        rng.shuffle(candidates)

        improved = False
        for i in candidates[:15]:
            if time.monotonic() - start > time_budget:
                break

            # 1D optimization over a[i]
            # Try geometric grid around current value
            base = a[i]
            for factor in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.1]:
                a_try = a.copy()
                a_try[i] = base * factor
                if a_try[i] < 1e-15:
                    continue
                # Renormalize
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break

        if not improved:
            no_imp_count += 1
            if no_imp_count > 200:
                break
        else:
            no_imp_count = 0

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

    elif kind == 2:

        a = 1 + 0.3 * np.sin(2 * np.pi * freq * t / n)

            best_seq = a_opt.copy()
            print(f"  L-BFGS high: {best_score:.8f} ***")
        else:
            print(f"  L-BFGS high: {obj:.8f}")

    # === Phase 4: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Coordinate descent")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 50:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.12, 50.0)
        a_cd = coord_descent_peak(best_seq, time_budget=alloc, seed=seed)
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  Coord descent: {best_score:.8f} ***")
        else:
            print(f"  Coord descent: {obj:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq)
    n_options = sorted(set([
        max(300, n_best - 150),
        max(300, n_best - 75),
        n_best,
        n_best + 75,
        n_best + 150,
        n_best + 300,
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500]

    for n_try in n_options:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        a_try = a_try / np.sum(a_try)

        alloc = min(remaining * 0.25, 60.0)
        a_opt = lg_lbfgs(a_try, [30, 100, 300, 800],
                         iters_per_stage=60, time_budget=alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  n={n_try}: {obj:.8f}")

    # === Phase 6: Random restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Random restarts")
    print(f"{'='*60}")

    n_best = len(best_seq)

    for restart in range(12):
        if time.monotonic() > deadline - 25:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        strat = restart % 6
        if restart % 3 == 0:
            scale = 0.02 + 0.015 * restart
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
        elif restart % 3 == 1:
            a_struct = gen_structured(n_best, strat, rng)
            alpha = 0.08 + 0.04 * (restart % 5)
            a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * a_struct
        else:
            scale = 0.04 + 0.02 * restart
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))

        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / np.sum(a_pert)

        alloc = min(remaining * 0.18, 40.0)
        a_opt = lg_lbfgs(a_pert, [20, 80, 300, 800, 1500],
                         iters_per_stage=50, time_budget=alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Restart {restart} (strat={strat}): {best_score:.8f} *** UPDATE")
        else:
            print(f"  Restart {restart} (strat={strat}): {obj:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()

        # Peak push
        if remaining > 15:
            alloc = min(remaining * 0.25, 30.0)
            a_pp = peak_push_projected(best_seq, time_budget=alloc, n_steps=20000)
            obj = compute_obj(a_pp)
            if obj < best_score:
                best_score = obj
                best_seq = a_pp.copy()
                print(f"  Final peak-push: {best_score:.8f} ***")

        # Very high beta L-BFGS
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            a = best_seq / np.sum(best_seq)
            alloc = min(remaining * 0.5, 40.0)
            a_final = lg_lbfgs(a, [800, 1500, 2500, 4000],
                               iters_per_stage=40, time_budget=alloc)
            obj = compute_obj(a_final)
            if obj < best_score:
                best_score = obj
                best_seq = a_final.copy()
                print(f"  Final L-BFGS: {best_score:.8f} ***")

        # One more coord descent
        if time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.5, 15.0)
            a_cd = coord_descent_peak(best_seq, time_budget=alloc, seed=seed + 1)
            obj = compute_obj(a_cd)
            if obj < best_score:
                best_score = obj
                best_seq = a_cd.copy()
                print(f"  Final coord: {best_score:.8f} ***")

    # Final validation
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
