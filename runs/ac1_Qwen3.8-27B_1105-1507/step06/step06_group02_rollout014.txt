import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        power = 0.4 + 0.4 * beta
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        mag_flat = mag.mean() + 1e-30
        mag_new = (1.0 - beta) * mag + beta * mag_flat
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[lo:hi + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, p, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.3, lr_end=0.002):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = normalize(a)
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new
    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, A_sparse=None):
    n = len(a)
    if A_sparse is None:
        A_sparse = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'time_limit': 10.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(history_x, history_Fx, m, max_alpha=1.5):
    if m < 2:
        return history_Fx[-1].copy()
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-10 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -max_alpha, max_alpha)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def run_lp_mirror(a, deadline, max_iters=500, anderson_window=5,
                  step_sizes=None, min_step=0.005):
    """LP mirror descent with backtracking line search and Anderson acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = normalize(a)
    n = len(a)
    if step_sizes is None:
        step_sizes = [0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.03, 0.015, 0.008, 0.004]
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    hist_x = []
    hist_Fx = []
    current_step_idx = 3
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break
        A_sparse = build_A_sparse(n, a)
        g = lp_best_response(a, M, A_sparse)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        # Backtracking line search
        best_t = step_sizes[current_step_idx] if current_step_idx < len(step_sizes) else min_step
        best_M_t = float('inf')
        for si, t in enumerate(step_sizes):
            if t < min_step:
                break
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            st = a_t.sum()
            if st < 1e-10:
                continue
            a_t = a_t / st
            M_t = fast_conv(a_t).max()
            if M_t < best_M_t:
                best_M_t = M_t
                best_t = t
                best_t_idx = si
                a_best = a_t.copy()
        # Update step index based on best step
        if best_t_idx < len(step_sizes) - 1:
            current_step_idx = best_t_idx
        else:
            current_step_idx = min(current_step_idx + 1, len(step_sizes) - 1)
        a_new = a_best
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        if len(hist_x) >= 3:
            a_anderson = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sa = a_anderson.sum()
            if sa > 1e-10:
                a_anderson = a_anderson / sa
            M_and = fast_conv(a_anderson).max()
            if M_and <= best_M_t * 1.02:
                a = a_anderson
            else:
                a = a_new
        else:
            a = a_new
    return best_a


def run_subgradient_polish(a, deadline, max_iters=200, lr_init=0.05, lr_end=0.001):
    """Exact subgradient descent with Nesterov acceleration for final polishing."""
    a = np.maximum(a.astype(np.float64), 1e-8)

        lr = lr_init * (lr_end / lr_init) ** frac
        b = fast_conv(y)
        b_max = float(b.max())
        active = np.where(np.abs(b - b_max) < b_max * 1e-6)[0]
        if len(active) == 0:

        grad = np.zeros(n)
        for k in active:

            seg = y[lo:hi + 1][::-1]
            grad[lo:hi + 1] += 2.0 * seg
        grad /= len(active)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0

        sy = y.sum()
        if sy < 1e-10:

        a = a_new

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):

            sub_dl = time.monotonic() + max(1.0, (dl - time.monotonic()) * frac_time)
            a = run_nesterov_smooth(a0, sub_dl, max_iters=n_iters_base,
                                    tau_start=1.0, tau_end=150.0,
                                    lr_init=0.3, lr_end=0.002)
            M = fast_conv(a).max()
            if M < best_M_val:
                best_M_val = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: M={best_M_val:.7f} J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Main LP Mirror Descent ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        # Estimate LP time and budget
        est_lp_ms = 2.0 * (n_best / 128.0) ** 2
        print(f"  Best n={n_best}, est LP time={est_lp_ms:.0f}ms", flush=True)
        
        lp_budget = min(900.0, max(100.0, (deadline - 150) - time.monotonic()))
        max_iters = max(30, int(lp_budget / max(0.02, est_lp_ms / 1000.0)))
        print(f"  LP budget={lp_budget:.0f}s, max_iters={max_iters}", flush=True)

        # Run 1: aggressive steps
        a = run_lp_mirror(best.seq, min(time.monotonic() + lp_budget * 0.5, deadline - 100),
                          max_iters=max_iters, anderson_window=6,
                          step_sizes=[0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01])
        best.update(a)

        # Run 2: medium steps
        if time.monotonic() < deadline - 80:
            a2 = run_lp_mirror(a, min(time.monotonic() + lp_budget * 0.35, deadline - 60),
                               max_iters=int(max_iters * 0.7), anderson_window=6,
                               step_sizes=[0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005])
            best.update(a2)

        # Run 3: fine steps
        if time.monotonic() < deadline - 40:
            a3 = run_lp_mirror(a2, deadline - 30,
                               max_iters=int(max_iters * 0.4), anderson_window=5,
                               step_sizes=[0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003])
            best.update(a3)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Try other lengths ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Multi-length LP", flush=True)
    print(f"{'='*50}", flush=True)

    if best_per_length and time.monotonic() < deadline - 50:
        sorted_lengths = sorted(best_per_length.keys(),
                                key=lambda n: compute_J(best_per_length[n]))
        tried = set()
        for n_try in sorted_lengths:
            if time.monotonic() > deadline - 40:
                break
            if n_try in tried:
                continue
            if n_try == best.seq.size:
                tried.add(n_try)
                continue
            tried.add(n_try)
            a_cand = best_per_length[n_try].copy()
            # Quick LP run
            if time.monotonic() < deadline - 30:
                a_lp = run_lp_mirror(a_cand, min(time.monotonic() + 40, deadline - 25),
                                     max_iters=50, anderson_window=4,
                                     step_sizes=[0.3, 0.2, 0.1, 0.05, 0.03, 0.01])
                best.update(a_lp)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Subgradient polish ===
    p4_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Subgradient Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 10:
        a_pol = run_subgradient_polish(best.seq, min(time.monotonic() + 60, deadline - 5),
                                       max_iters=300, lr_init=0.03, lr_end=0.0005)
        best.update(a_pol)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Final LP micro-steps ===
    p5_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Final Micro LP", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 3:
        a_final = run_lp_mirror(best.seq, deadline,
                                max_iters=100, anderson_window=5,
                                step_sizes=[0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002])
        best.update(a_final)

        # One more subgradient pass
        if time.monotonic() < deadline - 2:
            a_fin2 = run_subgradient_polish(a_final, deadline,
                                            max_iters=100, lr_init=0.01, lr_end=0.0002)
            best.update(a_fin2)

    best.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === PHASE 6: Length micro-adjustments ===
    if best.seq is not None and time.monotonic() < deadline - 5:
        n_best = best.seq.size
        for delta in [-64, -32, 32, 64, 128]:
            if time.monotonic() > deadline - 5:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 2500:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = normalize(a_ext)
            a_ext = run_nesterov_smooth(a_ext, min(time.monotonic() + 10, deadline - 4),
                                        max_iters=200, tau_start=5, tau_end=100,
                                        lr_init=0.1, lr_end=0.005)
            best.update(a_ext)
            if time.monotonic() < deadline - 4:
                a_lp = run_lp_mirror(a_ext, deadline - 3, max_iters=20,
                                     anderson_window=4,
                                     step_sizes=[0.2, 0.1, 0.05, 0.02, 0.01])
                best.update(a_lp)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
