"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based efficient gradient computation for smooth max
2. Multi-scale (multi-tau) smooth max gradient descent with adaptive step size
3. LP mirror descent with golden-section line search for precise convergence
4. Multi-start with diverse perturbations and length exploration
Key improvements: efficient gradient via correlation, multi-tau blending,
proper line search, and focused exploration around the best solution.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad_fft(a, tau):
    """Compute gradient of smooth max of conv(a,a) using FFT-based correlation.
    grad[i] = 2 * sum_k p[k] * a[k-i] * I[0<=k-i<n]
    which equals 2 * cross_correlate(p, a)[i]
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # grad[i] = 2 * sum_{m=0}^{n-1} p[i+m] * a[m] for i+m < 2n-1
    # This is cross-correlation of p and a
    # Using zero-padding FFT:
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    Fp = np.fft.rfft(p, size)
    Fa = np.fft.rfft(a, size)
    # cross-correlation: ifft(Fp * conj(Fa)) gives sum_j p[j+i]*a[j]
    corr = np.fft.irfft(Fp * np.conj(Fa), size)[:n]
    grad = 2.0 * corr
    return grad, p, b, b_max


def smooth_max_grad_direct(a, tau):
    """Direct gradient computation (slower but simpler)."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        seg = a[j_min:j_max + 1]
        # For position k, the segment of a that participates is a[j_min..j_max]
        # grad contribution: for i in [j_min, j_max], grad[i] += 2*p[k]*a[k-i]
        # a[k-i] for i in [j_min, j_max] is a[k-j_max .. k-j_min] = a[j_min..j_max] reversed
        rev_seg = seg[::-1]
        grad[j_min:j_max + 1] += 2.0 * p[k] * rev_seg
    return grad, p, b, b_max


def run_smooth_descent(a, deadline, max_iters=3000, tau_start=1.0, tau_end=500.0,
                       lr_init=0.5, lr_end=0.001, multi_tau=False):
    """Smooth max gradient descent with Nesterov momentum and multi-tau option."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    t_k = 1.0

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        if multi_tau and it > 100:
            # Multi-tau: blend gradients at tau/2, tau, 2*tau
            grad1, _, b, b_max = smooth_max_grad_fft(y, tau * 0.5)
            grad2, _, _, _ = smooth_max_grad_fft(y, tau * 1.0)
            grad3, _, _, _ = smooth_max_grad_fft(y, tau * 2.0)
            grad = (0.25 * grad1 + 0.5 * grad2 + 0.25 * grad3)
        else:
            grad, p, b, b_max = smooth_max_grad_fft(y, tau)

        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def build_A_matrix(n, a):
    """Build sparse matrix for (a*g)[k] = (A @ g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror_linesearch(a, deadline, max_iters=100):
    """LP mirror descent with golden section line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Golden section line search to find best t
        # f(t) = max(conv((1-t)*a + t*g))
        # We want to minimize f(t) over t in [0, 1]
        golden = (np.sqrt(5) - 1) / 2  # ~0.618

        def f_of_t(t):
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                return float('inf')
            a_t = a_t / s
            return fast_conv(a_t).max()

        # Golden section search in [0, 1]
        lo, hi = 0.0, 1.0
        c_pt = hi - golden * (hi - lo)
        d_pt = lo + golden * (hi - lo)
        fc = f_of_t(c_pt)
        fd = f_of_t(d_pt)

        for _ in range(20):
            if time.monotonic() > deadline:
                break
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - golden * (hi - lo)
                fc = f_of_t(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + golden * (hi - lo)
                fd = f_of_t(d_pt)

        t_best = (lo + hi) / 2
        # Also check t=0 (no update) and a few extra points
        candidates = [t_best, 0.3, 0.5, 0.7, 0.1, 0.2]
        best_t = 0.0
        best_f = f_of_t(0.0)
        for t_c in candidates:
            if t_c < 0 or t_c > 1:
                continue
            f_c = f_of_t(t_c)
            if f_c < best_f:
                best_f = f_c
                best_t = t_c

        a_new = (1.0 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break

    return best_a


def run_lp_anderson(a, deadline, max_iters=150, anderson_window=5,
                    t_base=0.3, min_t=0.005):
    """LP mirror descent with Anderson acceleration and line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    hist_x = []
    hist_Fx = []

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break

        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Determine step size with decay
        if it < 10:
            t = t_base
        else:
            t = max(min_t, t_base / (1.0 + 0.03 * it))

        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s

        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)

        if len(hist_x) >= 3:
            # Anderson mixing
            m = len(hist_x)
            residuals = np.array([hist_x[i] - hist_Fx[i] for i in range(m)])
            G = residuals @ residuals.T + 1e-12 * np.eye(m)
            ones = np.ones(m)
            try:
                G_inv_ones = np.linalg.solve(G, ones)
                denom = ones @ G_inv_ones
                if abs(denom) > 1e-15:
                    alpha = G_inv_ones / denom
                else:
                    alpha = ones / m
            except np.linalg.LinAlgError:
                alpha = ones / m
            alpha = np.clip(alpha, -0.5, 1.5)
            alpha_sum = alpha.sum()
            if abs(alpha_sum) > 1e-15:
                alpha = alpha / alpha_sum
            mixed = np.zeros(n)
            for i in range(m):
                mixed += alpha[i] * hist_Fx[i]
            a_anderson = np.maximum(mixed, 0.0)
            sa = a_anderson.sum()
            if sa > 1e-10:
                a_anderson /= sa
                M_and = fast_conv(a_anderson).max()
                M_new = fast_conv(a_new).max()
                if M_and <= M_new * 1.02:
                    a = a_anderson
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: flatten power spectrum while maintaining non-negativity."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-30)
        power = 0.5 + 0.5 * beta
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a

    def update(self, a):

        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(1024)
        best.update(parent)

    best_ref = best.seq.copy() if best.seq is not None else parent.copy()

    # === PHASE 1: Smooth gradient descent from multiple initializations ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Multi-start Smooth Descent", flush=True)
    print(f"{'='*50}", flush=True)

    lengths = [768, 896, 1024, 1152, 1280, 1536]
    phase1_budget = 350.0
    p1_end = time.monotonic() + phase1_budget

    for n in lengths:
        if time.monotonic() > p1_end:
            break
        t0 = time.monotonic()
        # Create initializations
        inits = []
        # Resample parent
        p_resamp = resample_seq(best_ref, n)
        p_resamp = np.maximum(p_resamp, 1e-4)
        p_resamp /= p_resamp.sum()
        inits.append(p_resamp)
        # Perturbed versions
        for noise_level in [0.01, 0.03, 0.08]:
            pert = np.maximum(p_resamp + rng.normal(0, noise_level, n), 1e-4)
            pert /= pert.sum()
            inits.append(pert)
        # GS init
        gs_init = gs_flat_spectral(p_resamp, n_iters=150, beta=0.88)
        inits.append(gs_init)
        # GS from random
        a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
        gs_rand = gs_flat_spectral(a_rand, n_iters=100, beta=0.85)
        inits.append(gs_rand)
        # Symmetric init
        half_n = n // 2 + 1
        half = np.maximum(p_resamp[:half_n] + rng.normal(0, 0.02, half_n), 0.01)
        full_sym = np.concatenate([half, half[-2::-1]])[:n]
        full_sym = np.maximum(full_sym, 0.01)
        full_sym /= full_sym.sum()
        inits.append(full_sym)

        n_iters = max(100, int(5000 * (768.0 / n)))
        time_per_init = (p1_end - time.monotonic()) / max(1, len(inits))

        best_n = None
        best_M = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > p1_end:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 /= a0.sum()
            sub_dl = min(time.monotonic() + max(5, time_per_init * 1.5), p1_end - 1)
            a = run_smooth_descent(a0, sub_dl, max_iters=n_iters,
                                   tau_start=1.0, tau_end=300.0,
                                   lr_init=0.4, lr_end=0.002,
                                   multi_tau=True)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            if best.seq is not None:
                best_ref = best.seq.copy()

    best.report("Phase 1")

    # === PHASE 2: LP mirror descent with line search on best ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Mirror Descent + Line Search", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 100:
        n = best.seq.size
        p2_end = min(time.monotonic() + 400, deadline - 80)
        a = run_lp_mirror_linesearch(best.seq, p2_end, max_iters=80)
        best.update(a)

    best.report("Phase 2")

    # === PHASE 3: Anderson-accelerated LP ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Anderson Accelerated LP", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 60:
        p3_end = min(time.monotonic() + 300, deadline - 40)
        a = run_lp_anderson(best.seq, p3_end, max_iters=100,
                            anderson_window=6, t_base=0.25, min_t=0.003)
        best.update(a)

    best.report("Phase 3")

    # === PHASE 4: Refinement - alternate smooth + LP ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Alternating Refinement", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 30:
        n = best.seq.size
        for round_idx in range(3):
            if time.monotonic() > deadline - 20:
                break
            # Short smooth descent
            sub_dl = min(time.monotonic() + 60, deadline - 30)
            a = run_smooth_descent(best.seq, sub_dl, max_iters=500,
                                   tau_start=10, tau_end=500,
                                   lr_init=0.1, lr_end=0.005,
                                   multi_tau=True)
            best.update(a)
            # Short LP polish
            if time.monotonic() < deadline - 20:
                sub_dl2 = min(time.monotonic() + 80, deadline - 15)
                a_lp = run_lp_mirror_linesearch(best.seq, sub_dl2, max_iters=30)
                best.update(a_lp)

    best.report("Phase 4")

    # === PHASE 5: Length exploration around best ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Length Exploration", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 15:
        n_best = best.seq.size
        for delta in [-64, -32, 32, 64, 128, -128]:
            if time.monotonic() > deadline - 10:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 3072:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-5)
            a_ext /= a_ext.sum()
            sub_dl = min(time.monotonic() + 20, deadline - 8)
            a_ext = run_smooth_descent(a_ext, sub_dl, max_iters=300,
                                       tau_start=5, tau_end=200,
                                       lr_init=0.15, lr_end=0.01,
                                       multi_tau=True)
            best.update(a_ext)

    best.report("Phase 5")

    # === PHASE 6: Final polish ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 6: Final Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 3:
        a_final = run_lp_mirror_linesearch(best.seq, deadline, max_iters=30)
        best.update(a_final)

    best.report("Phase 6")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(1024)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
