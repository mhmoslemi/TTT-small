def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    m = 2 * n - 1
    
    # LP: min M
    # s.t. A @ g - M <= 0  (m constraints)
    #      sum(g) = 1
    #      g >= 0, M >= 0
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s, float(res.x[n])
    return None, None

        # Try moving mass from high-gradient to low-gradient positions
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if j != i:
                    grad[i] = 2.0 * a[j]
                else:
                    grad[i] = 2.0 * a[i]
        
        # Sort by gradient: reduce high grad, increase low grad
        order = np.argsort(grad)
        low_end = order[:n//4]
        high_end = order[-n//4:]
        
        for i in high_end[::3]:
            for j in low_end[::3]:
                if i == j or a[i] < 1e-6:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = frac * a[i]
                    trial = best_a.copy()
                    trial[i] -= delta
                    trial[j] += delta
                    obj_try = compute_obj(trial)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = trial
                        improved = True
                        break
                    if obj_try > best_obj:
                        break
        
        if not improved:
            break
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def spectral_init(n, rng):
    """Initialize using frequency domain approach."""
    # Create a smooth low-pass |A(f)|^2
    freqs = np.arange(n)
    # Low-pass: |A(f)|^2 = exp(-f^2 / width^2)
    width = n * 0.1
    target_power = np.exp(-(freqs ** 2) / (2 * width ** 2))
    target_power[0] = 1.0  # DC is always max
    target = np.sqrt(target_power)
    
    # Random phase
    phase = rng.uniform(0, 2 * np.pi, n)
    phase[0] = 0  # DC has zero phase
    
    # Construct A(f)
    A = target * np.exp(1j * phase)
    
    # Inverse DFT
    a = np.real(np.fft.ifft(A * n))  # scale
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def smooth_init(n, mode=0):
    """Generate smooth structured initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif mode == 1:
        # Gaussian
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif mode == 2:
        # Triangular
        a = 1 - np.abs(2 * x - 1)
    elif mode == 3:
        # Dual cosine (U-shaped)
        a = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif mode == 4:
        # Flat with smooth edges
        w = 0.1
        a = np.ones(n)
        a[:n//10] = np.linspace(0.5, 1, n//10)
        a[-n//10:] = np.linspace(1, 0.5, n//10)
    else:
        a = np.ones(n)
    a = a + 0.01
    a = a / np.sum(a)
    return a

"""
Hybrid LP-alternating + subgradient chase-peak + multi-n + 2-point refinement
for minimizing 2n * max(conv(a,a)) / (sum(a))^2 with a >= 0.
Uses fast HiGHS LP solver, efficient FFT convolution, structured inits,
and careful time management to beat the 1.5027 target.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_sum = float(np.sum(a))
    if a_sum < 1e-10:
        return None, None
    a_norm = a / a_sum
    A = build_conv_matrix(a_norm)
    m = 2 * n - 1

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0.0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s, float(res.x[n])
    return None, None

    n = len(a)

    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    perturbation_scale = 0.03

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g, M_lp = solve_lp_minpeak(a)
        if g is None:
            stall += 5
            if stall > 30:
                break
            continue

        # Line search over step sizes
        best_t_obj = np.inf
        best_t_a = best_a.copy()
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

        if best_t_obj < best_obj - 1e-13:

            a = best_t_a
            stall = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"    LP-alt it{it}: obj={best_obj:.9f}, M_lp={M_lp:.6f}")
        else:
            stall += 1
            if stall > 10:
                # Perturb to escape local minimum
                rng_local = np.random.default_rng(it * 104729 + n * 31)
                scale = perturbation_scale * (1 + 0.5 * min(stall // 10, 5))
                a = best_a * np.exp(rng_local.normal(0, scale, n))
                a = np.maximum(a, 0.0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_obj = compute_obj(a)
                best_a = a.copy()
                stall = 0

    return best_a


def chase_peak(a, time_budget=20.0, max_iters=1000):
    """
    Subgradient method targeting the peak convolution bin.
    Uses the fact that grad_i conv[k*] = 2*a[k*-i] (if valid).
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.01
    min_eta = 0.0005

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient: grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0
        indices = k_star - np.arange(n)
        mask = (indices >= 0) & (indices < n)
        grad = np.zeros(n)
        grad[mask] = 2.0 * a[indices[mask]]

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        # Try a few step sizes
        improved = False
        for eta_try in [eta, eta * 0.5, eta * 0.25]:
            a_new = a - eta_try * grad_centered
            a_new = np.maximum(a_new, 0.0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                eta = min(eta * 1.02, 0.05)
                improved = True
                break

        if not improved:
            eta *= 0.8
            if eta < min_eta:
                break

    return best_a


def two_point_refine(a, time_budget=30.0, max_rounds=100):
    """
    2-point coordinate refinement: move mass from high-peak-contribution
    positions to low-peak-contribution positions.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for round_idx in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        k_star = int(np.argmax(conv))

        # Compute gradient at peak
        indices = k_star - np.arange(n)
        mask = (indices >= 0) & (indices < n)
        grad = np.zeros(n)
        grad[mask] = 2.0 * best_a[indices[mask]]

        # Positions with highest gradient (most contribute to peak)
        # Positions with lowest gradient (least contribute)
        order = np.argsort(grad)
        high_positions = order[-n // 5:]    # top 20%
        low_positions = order[:n // 5]      # bottom 20%

        improved = False
        # Try moving mass from high to low
        step = max(1, len(high_positions) // 20)
        for ii in range(0, len(high_positions), step):
            i = high_positions[ii]
            if best_a[i] < 1e-8:
                continue
            step2 = max(1, len(low_positions) // 20)
            for jj in range(0, len(low_positions), step2):
                j = low_positions[jj]
                if i == j:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08]:
                    delta = frac * best_a[i]
                    if delta > best_a[i]:
                        continue
                    trial = best_a.copy()
                    trial[i] -= delta
                    trial[j] += delta
                    obj_try = compute_obj(trial)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = trial
                        improved = True
                        break
                    elif obj_try > best_obj:
                        break

            if time.monotonic() - start > time_budget:
                break

        if not improved:
            # Try symmetric 2-point: swap partial mass between two random pairs
            rng_local = np.random.default_rng(round_idx * 7919)
            found = False
            for _ in range(50):
                i = rng_local.integers(0, n)
                j = rng_local.integers(0, n)
                if i == j or best_a[i] < 1e-8:
                    continue
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * best_a[i]
                    trial = best_a.copy()
                    trial[i] -= delta
                    trial[j] += delta
                    obj_try = compute_obj(trial)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = trial
                        found = True
                        break
                if found:
                    break
            if not found:
                break

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)


def smooth_init(n, mode=0):
    """Generate smooth structured initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif mode == 1:
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif mode == 2:
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif mode == 3:
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    elif mode == 4:
        a = np.ones(n)
        edge = max(1, n // 15)
        a[:edge] = np.linspace(0.6, 1.0, edge)
        a[-edge:] = np.linspace(1.0, 0.6, edge)
    elif mode == 5:
        # Dual peak
        a = np.exp(-8.0 * (x - 0.3) ** 2) + np.exp(-8.0 * (x - 0.7) ** 2)
    else:
        a = np.ones(n)
    a = a + 0.005
    a = a / np.sum(a)
    return a


def spectral_init(n, rng):
    """Initialize using frequency domain: create low-pass |A(f)|, random phase, IFFT, project."""
    freqs = np.arange(n)
    width = n * 0.08
    target_power = np.exp(-(freqs.astype(np.float64) ** 2) / (2.0 * width ** 2))
    target_mag = np.sqrt(target_power)

    phase = rng.uniform(0, 2.0 * np.pi, n)
    phase[0] = 0.0

    A = target_mag * np.exp(1j * phase)
    a = np.real(np.fft.ifft(A) * n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def bloch_refine(a, time_budget=30.0):
    """
    Iterative block refinement: split into blocks and optimize each block
    while fixing others, using the LP structure.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    block_size = max(10, n // 20)
    n_blocks = (n + block_size - 1) // block_size

    for sweep in range(20):
        if time.monotonic() - start > time_budget:
            break
        improved = False

        for b in range(n_blocks):
            lo = b * block_size
            hi = min(n, lo + block_size)
            if hi - lo < 2:
                continue

            # Try simple moves within this block
            block = best_a[lo:hi].copy()
            block_sum = float(np.sum(block))
            if block_sum < 1e-10:
                continue

            # Equalize within block (reduces sum of squares)
            for t in [0.1, 0.3, 0.5]:
                new_block = (1 - t) * block + t * np.full(hi - lo, block_sum / (hi - lo))
                trial = best_a.copy()
                trial[lo:hi] = new_block
                obj_try = compute_obj(trial)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = trial
                    block = trial[lo:hi].copy()
                    block_sum = float(np.sum(block))
                    improved = True
                    break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP-alternating, chase-peak subgradient,
    2-point refinement, multi-n exploration, and structured initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        if parent.size > 0 and np.isfinite(parent).all():
            s = float(np.sum(parent))
            if s > 0.01:
                best_seq = parent / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={len(best_seq)}): obj={best_score:.9f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = smooth_init(800, 0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.9f}")

    n_parent = len(best_seq)

    # === Phase 1: Intensive LP Alternating (250s) ===
    print(f"\n=== Phase 1: LP Alternating (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(250.0, remaining * 0.25)
    a = lp_alternate(best_seq, time_budget=alloc, verbose=True, max_iters=500)
    obj = compute_obj(a)
    if obj < best_score:
        best_score = obj
        best_seq = a.copy()
        print(f"  Phase 1: obj={best_score:.9f}")
    else:
        print(f"  Phase 1: obj={obj:.9f} (no improvement)")

    # === Phase 2: Chase Peak (80s) ===
    print(f"\n=== Phase 2: Chase Peak ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_cp = chase_peak(best_seq, time_budget=min(80.0, remaining * 0.1))
        obj_cp = compute_obj(a_cp)
        if obj_cp < best_score:
            best_score = obj_cp
            best_seq = a_cp.copy()
            print(f"  Chase: obj={best_score:.9f}")
        else:
            print(f"  Chase: obj={obj_cp:.9f}")

    # === Phase 3: LP Alternating with perturbations (250s) ===
    print(f"\n=== Phase 3: LP Alternating + Restarts ===")
    for attempt in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(80.0, remaining * 0.25)
        if attempt == 0:
            a3 = best_seq.copy()
        elif attempt == 1:
            a3 = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
            a3 = np.maximum(a3, 0.0)
            a3 = a3 / np.sum(a3)
        elif attempt == 2:
            # Smooth + perturb
            a3 = smooth_init(len(best_seq), attempt - 1)
            a3 = 0.5 * best_seq + 0.5 * a3
            a3 = a3 / np.sum(a3)
        else:
            a3 = best_seq * np.exp(rng.normal(0, 0.05, len(best_seq)))
            a3 = np.maximum(a3, 0.0)
            a3 = a3 / np.sum(a3)

        a_opt = lp_alternate(a3, time_budget=alloc, max_iters=200)
        obj3 = compute_obj(a_opt)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a_opt.copy()
            print(f"  Attempt {attempt}: obj={best_score:.9f} **")
        else:
            print(f"  Attempt {attempt}: obj={obj3:.9f}")

    # === Phase 4: Chase Peak again after LP (60s) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_cp2 = chase_peak(best_seq, time_budget=min(60.0, remaining * 0.15))
        obj_cp2 = compute_obj(a_cp2)
        if obj_cp2 < best_score:
            best_score = obj_cp2
            best_seq = a_cp2.copy()
            print(f"  Chase2: obj={best_score:.9f}")

    # === Phase 5: Multi-n exploration (200s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_curr = len(best_seq)
    n_options = sorted(set([
        max(300, n_curr - 100), max(300, n_curr - 50),
        n_curr,
        n_curr + 50, n_curr + 100, n_curr + 200,
        600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        if n_try == n_curr:
            a_try = best_seq.copy()
        else:
            a_try = resample(best_seq, n_try)

        alloc = min(50.0, remaining * 0.25)
        a_opt = lp_alternate(a_try, time_budget=alloc, max_iters=100)
        obj_try = compute_obj(a_opt)

        # Also try chase peak on the LP result
        remaining_inner = deadline - time.monotonic()
        if remaining_inner > 20 and obj_try < best_score + 0.01:
            a_cp = chase_peak(a_opt, time_budget=min(20.0, remaining_inner * 0.3))
            obj_cp = compute_obj(a_cp)
            if obj_cp < obj_try:
                a_opt = a_cp
                obj_try = obj_cp

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.9f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.9f}")

    # === Phase 6: 2-point refinement (100s) ===
    print(f"\n=== Phase 6: 2-Point Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_ref = two_point_refine(best_seq, time_budget=min(100.0, remaining * 0.3))
        obj_ref = compute_obj(a_ref)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  2-point: obj={best_score:.9f}")
        else:
            print(f"  2-point: obj={obj_ref:.9f}")

    # === Phase 7: Block equalization (40s) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_blk = bloch_refine(best_seq, time_budget=min(40.0, remaining * 0.3))
        obj_blk = compute_obj(a_blk)
        if obj_blk < best_score:
            best_score = obj_blk
            best_seq = a_blk.copy()
            print(f"  Block: obj={best_score:.9f}")

    # === Phase 8: Final LP polish (80s) ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(60.0, remaining * 0.5)
        a_fin = lp_alternate(best_seq, time_budget=alloc, verbose=True, max_iters=200)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final LP: obj={best_score:.9f}")

    # === Phase 9: Final chase peak (30s) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cp3 = chase_peak(best_seq, time_budget=min(30.0, remaining * 0.5))
        obj_cp3 = compute_obj(a_cp3)
        if obj_cp3 < best_score:
            best_score = obj_cp3
            best_seq = a_cp3.copy()
            print(f"  Final chase: obj={best_score:.9f}")

    # Final 2-point tick
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_ref2 = two_point_refine(best_seq, time_budget=min(20.0, remaining * 0.5))
        obj_ref2 = compute_obj(a_ref2)
        if obj_ref2 < best_score:
            best_score = obj_ref2
            best_seq = a_ref2.copy()
            print(f"  Final 2-pt: obj={best_score:.9f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.9f}, time={elapsed:.1f}s ===")
    return result
