    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

def build_conv_matrix(a):
    """A @ g = conv(a, g)"""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_min = j
        k_max = min(2*n-2, j+n-1)
        for k in range(k_min, k_max+1):
            A[k, j] = a[k-j]
    return A

def solve_lp_minpeak(a):
    """min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0"""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = np.sum(g_val)
                if s > 1e-10:
                    return g_val/s
        except:
            continue
    return None

def mirror_descent(a, time_budget=60):
    """Mirror descent for minimizing max(conv)."""
    n = len(a)
    a = np.maximum(a, 0).astype(float)
    s = np.sum(a)
    if s < 1e-10: return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 1.0
    
    for it in range(2000):
        if time.monotonic() - start > time_budget: break
        
        conv = compute_conv(a)
        k_star = np.argmax(conv)
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sg[i] = a[j]
                if j != i:
                    sg[i] += a[j]  # Count both terms
                # Actually: ∂conv[k*]/∂a[i] = a[k*-i] + a[k*-i] if i ≠ k*-i
                # = 2*a[k*-i] if i ≠ k*-i and 0 ≤ k*-i < n
                # = a[k*-i] if i = k*-i
                # Let me just use 2*a[k*-i] as an approximation
        
        # Simpler: sg[i] = 2*a[k_star - i] if 0 <= k_star - i < n else 0
        sg = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sg[i] = 2.0 * a[j]
        
        # Center the subgradient (since we're on the simplex)
        sg -= np.mean(sg)
        
        # Mirror descent update
        a_new = a * np.exp(-eta * sg)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta *= 1.01  # Slowly increase step size
        else:
            eta *= 0.95  # Decrease step size
        
        if eta < 0.01:
            eta = 1.0  # Reset
    
    return best_a
