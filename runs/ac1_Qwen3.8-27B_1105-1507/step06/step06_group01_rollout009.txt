import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix, lil_matrix

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_full_A(n, a):
    """Build the full (2n-1) x n constraint matrix for the LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response_full(a, M, n_active=None):
    """Solve LP best response using full or partial constraint set."""
    n = len(a)
    if n_active is None or n_active >= 2 * n - 1:
        A = build_full_A(n, a)
    else:
        b = fast_conv(a)
        top_idx = np.argsort(b)[::-1][:n_active]
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(top_idx):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])
        A = csr_matrix((vals, (rows, cols)), shape=(n_active, n))
    
    if A.nnz == 0:
        return None
    
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def run_lp_mirror_descent(a_init, max_iters, deadline, n_active=None, 
                          base_t=0.3, t_decay=0.95):
    """LP mirror descent with full or partial constraints."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = lp_best_response_full(a, best_M, n_active)
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        improved = False
        t = base_t
        for _ in range(12):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 0.01:
                t *= 0.5
                continue
            a_trial = a_trial / st
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
            t *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a

def smooth_j_and_grad(a, tau):
    """Compute smooth approximation of max(conv(a,a)) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: d/d a[i] of smooth_val
    # smooth_val ≈ b_max + (1/tau) * log(sum exp(tau*b[k]))
    # d smooth_val / d a[i] = sum_k p[k] * d b[k] / d a[i]
    # d b[k] / d a[i] = 2 * a[k-i] if 0 <= k-i < n, else a[i-k] if 0 <= i-k < n
    # Actually b[k] = sum_j a[j]*a[k-j], so d b[k]/d a[i] = a[k-i] + a[i-k]
    # where a[m] = 0 if m not in [0, n-1]
    # Since b is the linear convolution (not circular), d b[k]/d a[i] = a[k-i] + (a[i-k] if k-i<0... )
    # Wait: b[k] = sum_j a[j] a[k-j]
    # db[k]/da[i] = a[k-i] (from the j=i term) + sum over j where k-j=i, i.e., j=k-i:
    # Actually: b[k] = sum_j a[j] * a[k-j]
    # d b[k] / d a[i] = a[k-i] (derivative w.r.t. a[i] in the first factor, j=i)
    #                  + a[k-i] (derivative w.r.t. a[i] in the second factor, k-j=i => j=k-i)
    # Wait, the second term: d/da[i] of a[k-j] is non-zero when k-j = i, i.e., j = k-i.
    # So the term is a[j] = a[k-i] (when j = k-i is in range)
    # So d b[k] / d a[i] = a[k-i] + a[k-i] = 2*a[k-i] if 0 <= k-i < n
    # But wait, this double-counts when i = k-i, i.e., k = 2i.
    # When k = 2i: b[k] contains the term a[i]*a[i], whose derivative is 2*a[i]
    # So d b[2i] / d a[i] = 2*a[i] (correct)
    # And d b[k] / d a[i] for k != 2i: the term a[i]*a[k-i] appears once (as j=i in the sum),
    # and a[k-i]*a[i] appears once (as j=k-i in the sum). So derivative = a[k-i] + a[i]... 
    # No wait. b[k] = sum_j a[j] * a[k-j]. The term involving a[i] appears when:
    # - j = i: contribution a[i] * a[k-i]
    # - k-j = i, i.e., j = k-i: contribution a[k-i] * a[i]
    # These are the same term! So b[k] contains a[i]*a[k-i] exactly once (when j=i, which 
    # is the same as j=k-i only when i=k-i).
    # So d b[k] / d a[i] = a[k-i] if 0 <= k-i < n (from the first factor)
    #                      + a[k-i] if 0 <= k-i < n (from the second factor, when j = k-i ≠ i)
    # Wait, I'm confusing myself. Let me be very explicit.
    # 
    # b[k] = sum_{j=0}^{n-1} a[j] * a[k-j] (where a[m]=0 if m not in [0,n-1])
    # 
    # d b[k] / d a[i] = sum_j [d(a[j])/d(a[i]) * a[k-j] + a[j] * d(a[k-j])/d(a[i])]
    # = a[k-i] + sum_j a[j] * d(a[k-j])/d(a[i])
    # The second sum: d(a[k-j])/d(a[i]) = 1 if k-j = i, i.e., j = k-i.
    # So second sum = a[k-i] (if 0 <= k-i < n)
    # 
    # So d b[k] / d a[i] = a[k-i] + a[k-i] = 2*a[k-i] (if 0 <= k-i < n)
    # But when i = k-i (i.e., k = 2i), the first term is a[i] and the second term is also a[i]
    # (since k-i = i), so d b[2i]/d a[i] = 2*a[i]. This is correct because 
    # b[2i] contains the term a[i]^2, and d(a[i]^2)/d(a[i]) = 2*a[i].
    #
    # So the gradient is: grad[i] = sum_k p[k] * 2 * a[k-i] (if 0 <= k-i < n)
    
    grad = np.zeros(n)
    for i in range(n):
        # sum over k where 0 <= k-i < n, i.e., k in [i, i+n-1]
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        grad[i] = 2.0 * np.dot(p[k_lo:k_hi+1], a[0:k_hi+1-i])
    
    return smooth_val, grad

def run_lbfgs_smooth(a_init, tau_start, tau_end, deadline, max_iter=5000):
    """L-BFGS-B on the smooth approximation."""
    a0 = np.maximum(a_init.astype(np.float64), 0.0)
    s = a0.sum()
    if s < 1e-10:
        a0 = np.ones_like(a0) / len(a0)
    else:
        a0 = a0 / s
    
    best_a = a0.copy()
    best_M = fast_conv(a0).max()
    
    for tau in [tau_start, int(tau_start*2), int(tau_start*4), min(tau_end, int(tau_start*8))]:
        if time.monotonic() > deadline:
            break
        
        # Work in log-space or with a reparameterization to handle non-negativity
        # Simple approach: use the variable directly with bounds
        def obj(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-10:
                return 1e10
            x = x / s
            b = fast_conv(x)
            b_max = b.max()
            exp_b = np.exp(tau * (b - b_max))
            return b_max + np.log(exp_b.sum()) / tau
        
        def grad(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-10:
                return np.zeros_like(x)
            x = x / s
            _, g = smooth_j_and_grad(x, tau)
            return g - g.mean()  # project to tangent of simplex
        
        try:
            res = minimize(obj, a0, jac=grad, method='L-BFGS-B',
                          bounds=[(0, None)] * len(a0),
                          options={'maxiter': max_iter, 'maxfun': max_iter,
                                   'ftol': 1e-15, 'gtol': 1e-12})

                    a_opt = a_opt / s_opt
                    M_opt = fast_conv(a_opt).max()
                    if M_opt < best_M:
                        best_M = M_opt
                        best_a = a_opt.copy()
                        a0 = a_opt.copy()
        except Exception as e:
            pass
    
    return best_a

def coordinate_opt(a, i, b):
    """Optimize a[i] given all others. Returns optimal x >= 0."""
    n = len(a)
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        diff = k - i
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            quad_pos = len(c_vals) - 1
        elif 0 <= diff < n:
            c = b[k] - 2.0 * x_cur * a[diff]
            c_vals.append(c)
            slopes.append(2.0 * a[diff])
        else:
            # This shouldn't happen given k_lo and k_hi
            c_vals.append(b[k])
            slopes.append(0.0)
    
    # Unaffected max
    if len(2 * n - 1 if False else np.ones(2*n-1)) > 0:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if quad_pos >= 0:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 1.0, 0.5)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    # Use FFT-based resampling for quality
    n_old = src.size
    # Zero-pad and interpolate in frequency domain
    size = max(1, (2 * max(n_old, dest_n) - 1).__next_power__()) if False else 1
    while size < 2 * max(n_old, dest_n):
        size <<= 1
    fa = np.fft.rfft(src, size)
    # Low-pass filter
    keep = min(size // 2 + 1, n_old)
    fa[keep:] = 0
    a_resampled = np.fft.irfft(fa, size)[:dest_n]
    return np.maximum(a_resampled, 0.0)

def resample_seq_simple(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
