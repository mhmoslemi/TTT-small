def lp_best_response(a, M, active_indices):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    n = len(a)
    if len(active_indices) == 0:
        return None
    
    m = len(active_indices)
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(active_indices):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            ai = a[k - j]
            if ai > 1e-15:
                rows.append(idx)
                cols.append(j)
                vals.append(ai)
    
    if len(rows) == 0:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def run_lp_optimization(a_init, max_iters, deadline, n_active_start=30, n_active_max=150):

def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    # Gradient: d/dc_k [smooth_max] * d/dc_k [conv(a,a)][i]
    # d/dc_k [conv(a,a)]_i = 2*a[k-i] (if 0 <= k-i < n)
    # grad[i] = 2 * sum_k p[k] * a[k-i] (where a[m]=0 outside [0,n-1])
    grad = np.zeros(n)
    for i in range(n):
        # sum_k p[k] * a[k-i] where 0 <= k-i < n, i.e., i <= k <= i+n-1
        k_lo = i
        k_hi = min(i + n - 1, 2 * n - 2)
        if k_lo <= k_hi:
            grad[i] = 2.0 * np.dot(p[k_lo:k_hi+1], a[0:k_hi - k_lo + 1])
    
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    return smooth_val, grad


def run_fista(a_init, deadline, max_iters=5000):
    """FISTA on smoothed max."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    t_param = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    tau_schedule = [50, 100, 200, 500, 1000]
    tau_idx = 0
    tau = tau_schedule[0]
    
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        _, grad_y = smooth_max_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        # Step size
        step = 1.0 / (tau * 2.0 + 1.0)
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param ** 2))
        alpha = (t_param - 1.0) / t_new if t_param > 1 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t_param = t_new
        it += 1
        
        # Check progress
        if it % 10 == 0:
            M_cur = compute_M(a)
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
        
        # Update tau
        if it % 500 == 0 and tau_idx < len(tau_schedule) - 1:
            tau_idx += 1
            tau = tau_schedule[tau_idx]
    
    M_final = compute_M(a)
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def coordinate_optimize(a, i, b, n):
    """Find optimal a[i] given all others fixed. Returns optimal x >= 0."""
    x_cur = a[i]
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    s_vals = []
    quad_c = None
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            qc = b[k] - x_cur * x_cur
            quad_c = qc
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                s_vals.append(2.0 * a[j])
    
    # Unaffected max
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_lo:k_hi + 1] = False
    base_max = b[mask].max() if mask.any() else 0.0
    
    c_arr = np.array(c_vals) if c_vals else np.array([])
    s_arr = np.array(s_vals) if s_vals else np.array([])
    
    def obj(x):
        m = base_max
        if len(c_arr) > 0:
            vm = (c_arr + s_arr * x).max()
            if vm > m:
                m = vm
        if quad_c is not None:
            qv = quad_c + x * x
            if qv > m:
                m = qv
        return m
    
    golden = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(3.0 * x_cur, 0.1 / np.sqrt(n), 0.01)
    
    # Ensure hi is in a reasonable range
    for _ in range(5):
        if obj(hi) > obj(0.5 * hi) and hi < 1.0:
            hi *= 2.0
        else:
            break
    
    a_pt = hi - golden * (hi - lo)
    b_pt = lo + golden * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(30):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - golden * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + golden * (hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj(0.0) <= obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=42):
    """Coordinate descent with smart ordering."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        order = np.arange(n)
        rng.shuffle(order)
        
        sweep_changed = False
        for i in order:
            x_opt = coordinate_optimize(a, i, b, n)
            dx = x_opt - a[i]
            if abs(dx) > 1e-12:
                # Update b
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += dx * (x_opt + a[i])
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * dx
                a[i] = x_opt
                sweep_changed = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_changed:
            no_improve += 5
        if no_improve > 15:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed versions
        for sigma in [0.02, 0.05, 0.1]:
            p_pert = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(p_pert / p_pert.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various structured inits
    # Concave (parabola)
    w = 4 * t * (1 - t) + 0.1
    inits.append(w / w.sum())
    
    # Convex (U-shape)
    w = (2*t - 1)**2 + 0.1
    inits.append(w / w.sum())
    
    # Sine
    w = 0.5 + 0.4 * np.sin(2 * np.pi * t) + 0.1
    w = np.maximum(w, 0.01)
    inits.append(w / w.sum())
    
    # Cosine window
    w = np.cos(np.pi * (t - 0.5))**2 + 0.05
    inits.append(w / w.sum())
    
    # Two peaks
    w = np.exp(-20*(t-0.3)**2) + np.exp(-20*(t-0.7)**2) + 0.05
    inits.append(w / w.sum())
    
    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits

            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_multistart_lp(a0, deadline, seed):
    """Run LP optimization with multiple active set strategies."""
    best = a0.copy()
    best_M = compute_M(a0)
    n = len(a0)
    
    # Strategy 1: Standard LP
    if time.monotonic() < deadline:
        sub_dl = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.5)
        a1 = run_lp_optimization(best, 100, sub_dl, n_active_start=30, n_active_max=100)
        M1 = compute_M(a1)
        if M1 < best_M:
            best_M = M1
            best = a1
    
    # Strategy 2: LP with more active constraints
    if time.monotonic() < deadline:
        sub_dl = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.5)
        a2 = run_lp_optimization(best, 100, sub_dl, n_active_start=50, n_active_max=200)
        M2 = compute_M(a2)
        if M2 < best_M:
            best_M = M2
            best = a2
    
    return best

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)
    
    n_parent = parent.size
