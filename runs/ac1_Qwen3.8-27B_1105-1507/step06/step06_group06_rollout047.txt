import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):

            if sweep > 10:
                break
    
    return best_a, best_obj


def mirror_descent(a, max_iters=200, time_budget=30.0, eta_init=0.1):
    """
    Mirror descent (multiplicative updates) with softmax-smoothed subgradient.
    Naturally maintains non-negativity.
    """
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        
        # Smoothed subgradient with softmax
        tau = 0.01 + 0.05 * (it / max_iters)  # increasing smoothness
        shifted = (conv - c_max) / tau
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (for valid k-i)
        # This is 2 * correlate(weights, a)
        # Use FFT
        N_fft = 1
        while N_fft < len(weights) + n - 1:
            N_fft <<= 1
        fw = np.fft.fft(weights, N_fft)
        fa = np.fft.fft(a, N_fft)
        corr = np.real(np.fft.ifft(fw * np.conj(fa), N_fft))
        # corr[m] = sum_k weights[k] * a[k + m - (n-1)]... 
        # Actually: ifft(fft(w) * conj(fft(a))) gives circular correlation
        # corr[m] = sum_k w[k] * a[(k-m) mod N]
        # We want: sum_k w[k] * a[k-i] = corr step... 
        # Let me just compute directly for safety
        grad = np.zeros(n)
        for k_idx in np.where(weights > 1e-10)[0]:
            w_val = weights[k_idx]
            # grad[i] += 2 * w_val * a[k_idx - i] for 0 <= k_idx - i < n
            i_lo = max(0, k_idx - n + 1)
            i_hi = min(n, k_idx + 1)
            if i_hi > i_lo:
                j_vals = k_idx - np.arange(i_lo, i_hi)
                grad[i_lo:i_hi] += 2.0 * w_val * a[j_vals]
        
        # Multiplicative update
        eta = eta_init / (1.0 + 0.01 * it)
        a = a * np.exp(-eta * grad)
        a = project_nonneg_sum1(a)
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def chase_peak(a, n_steps=100, step_size=0.05):
    """Aggressive peak chasing with subgradient."""
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for step in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient at k_star
        grad = np.zeros(n)
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n, k_star + 1)
        if i_hi > i_lo:
            j_vals = k_star - np.arange(i_lo, i_hi)
            grad[i_lo:i_hi] = 2.0 * a[j_vals]
        
        # Center the gradient (preserve sum)
        grad = grad - np.sum(grad) * a / np.sum(a)
        
        # Try step sizes
        for ss in [step_size, step_size * 0.5, step_size * 2.0]:
            a_try = a - ss * grad
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def freq_compress(a, max_iters=100, time_budget=20.0):
    """Frequency domain compression: reduce dynamic range of power spectrum."""
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        p_max = np.max(power)
        if p_max < 1e-15:
            break
        
        # Compress: raise the floor, lower the ceiling
        p_norm = power / p_max
        p_compressed = np.maximum(p_norm, 0.3) ** 0.7
        magnitude = np.sqrt(p_compressed * p_max)
        phase = np.angle(fa)
        fa_new = magnitude * np.exp(1j * phase)
        
        a_new = np.real(np.fft.ifft(fa_new))[:n]
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj
