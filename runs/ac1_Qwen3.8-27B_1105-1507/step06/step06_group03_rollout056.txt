    term1 = conv_wr[n-1:n-1+n]  # corr[i] = sum_k w[k]*a[k-i] for i=0..n-1

    # = (w * a_rev)[n-1-i] ...

    # Actually, let me use the FFT approach correctly:

    # P[2] = w2*a0

    # w * a_flip (linear conv, length 5):
    # [0]: w0*a2
    # [1]: w0*a1 + w1*a2
    # [2]: w0*a0 + w1*a1 + w2*a2
    # [3]: w1*a0 + w2*a1
    # [4]: w2*a0

    # So P[0] = (w*a_flip)[2] = (w*a_flip)[n-1]
    # P[1] = (w*a_flip)[3] = (w*a_flip)[n]

    # Q[2] = w[0]*a[2] + w[1]*a[1] + w[2]*a[0]
    
    # w_flip = [w2, w1, w0]
    # w_flip * a (linear conv, length 5):
    # [0]: w2*a0
    # [1]: w2*a1 + w1*a0
    # [2]: w2*a2 + w1*a1 + w0*a0
    # [3]: w1*a2 + w0*a1
    # [4]: w0*a2
    #
    # Q[0] = w0*a0 = (w_flip*a)[2] = (w_flip*a)[n-1]
    # Q[1] = w0*a1 + w1*a0 = (w_flip*a)[1]... hmm that's (w_flip*a)[n-1-i] for i=1: (w_flip*a)[1]
    # Q[2] = w0*a2 + w1*a1 + w2*a0 = (w_flip*a)[0] = (w_flip*a)[n-1-2]
    # So Q[i] = (w_flip * a)[n-1-i]
    
    # Great! So:
    # P[i] = (w * a_flip)[n-1+i] for i = 0, ..., n-1
    # Q[i] = (w_flip * a)[n-1-i] for i = 0, ..., n-1
    
    a_flip = a[::-1]
    w_flip = weights[::-1]
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    W_fft = np.fft.rfft(weights, size)
    Arev_fft = np.fft.rfft(a_flip, size)
    Wrev_fft = np.fft.rfft(w_flip, size)
    A_fft = np.fft.rfft(a, size)
    
    conv_w_arev = np.fft.irfft(W_fft * Arev_fft, size)[:2*n-1]
    conv_wrev_a = np.fft.irfft(Wrev_fft * A_fft, size)[:2*n-1]
    
    P = conv_w_arev[n-1 : n-1+n]  # P[i] = conv_w_arev[n-1+i]
    Q = conv_wrev_a[n-1 : n-1+n][::-1]  # Q[i] = conv_wrev_a[n-1-i], so Q = reversed of conv_wrev_a[n-1:n-1+n]
    
    # Wait, Q[i] = conv_wrev_a[n-1-i] for i=0,...,n-1
    # Q[0] = conv_wrev_a[n-1]
    # Q[1] = conv_wrev_a[n-2]
    # Q[n-1] = conv_wrev_a[0]
    # So Q = conv_wrev_a[0:n][::-1]... no
    # Q[i] = conv_wrev_a[n-1-i]
    # Let j = n-1-i, so i = n-1-j, j = n-1, n-2, ..., 0
    # Q = [conv_wrev_a[n-1], conv_wrev_a[n-2], ..., conv_wrev_a[0]]
    # = conv_wrev_a[0:n][::-1]
    
    Q = conv_wrev_a[0:n][::-1]
    
    return P + Q


def sca_lp_bisection(a, deadline, tol=1e-6, max_iters=200, verbose=False):
    """
    SCA-LP with bisection on M.
    At each iteration:
    1. Compute c = conv(a, a), M_curr = max(c)
    2. Binary search on M: find minimum M such that LP is feasible
    3. Update a to the LP solution
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M_curr = c.max()
        
        # Bisection on M: find min M such that LP is feasible
        # LP: find b >= 0, sum(b) = 1, s.t. 2*conv(a,b)[k] - c[k] <= M for all k
        # i.e., 2*conv(a,b)[k] <= M + c[k] for all k
        
        # Lower bound: we need 2*conv(a,b)[n-1] <= M + c[n-1]
        # conv(a,b)[n-1] = a · reverse(b) >= 0, so M >= 2*0 - c[n-1] = -c[n-1] (useless)
        # Better lower bound: M >= 0 (since conv values are non-negative and we need the constraint)
        # Actually the constraint is 2*conv(a,b)[k] <= M + c[k], so M >= 2*conv(a,b)[k] - c[k]
        # The minimum M over all b with sum=1, b>=0 is what we want.
        
        # Upper bound: M_curr (current value)
        # Lower bound: 0
        
        lo = 0.0
        hi = M_curr
        
        # Do a few bisection steps
        for bis in range(8):
            mid = (lo + hi) / 2
            # Check LP feasibility for M = mid
            # Variables: b[0..n-1]
            # Constraints: 2*conv(a,b)[k] <= mid + c[k] for k=0,...,2n-2
            #              sum(b) = 1
            #              b >= 0
            #
            # conv(a,b)[k] = sum_i a[|k-i|... no
            # conv(a,b)[k] = sum_i a[i] * b[k-i] where 0<=i<n, 0<=k-i<n
            # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i]
            
            # Build constraint matrix
            rows = []
            cols = []
            vals = []
            b_ub = []
            
            for k in range(2 * n - 1):
                i_min = max(0, k - n + 1)
                i_max = min(n - 1, k)
                for i in range(i_min, i_max + 1):
                    rows.append(k)
                    cols.append(k - i)
                    vals.append(2.0 * a[i])
                b_ub.append(mid + c[k])
            
            A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
            c_obj = np.zeros(n)  # feasibility: minimize 0
            A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))), shape=(1, n)).tocsr()
            b_eq = [1.0]
            bounds = [(0, None)] * n
            
            try:
                r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method='highs',
                           options={'time_limit': 10, 'presolve': True})
                if r.success:
                    hi = mid
                else:
                    lo = mid
            except:
                    lo = mid
        
        # After bisection, solve LP at M = hi to get the optimal b
        M_opt = hi
        rows = []
        cols = []
        vals = []
        b_ub = []
        
        for k in range(2 * n - 1):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                rows.append(k)
                cols.append(k - i)
                vals.append(2.0 * a[i])
            b_ub.append(M_opt + c[k])
        
        A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
        c_obj = np.zeros(n)
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))), shape=(1, n)).tocsr()
        b_eq = [1.0]
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                       bounds=bounds, method='highs',
                       options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                b = np.maximum(r.x, 0)
                bs = b.sum()
                if bs > 1e-10:
                    b /= bs
                    # Try different step sizes
                    found = False
                    for frac in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * b
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        obj_try = 2.0 * n * fast_conv(a_try).max()
                        if obj_try < best_obj - 1e-10:
                            a = a_try
                            best_obj = obj_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except:
            no_improve += 1
        
        if verbose and it % 10 == 0:
            print(f"    SCA-bis it {it}: M={M_opt:.8f} best={best_obj:.8f}", flush=True)
        
        if no_improve > 30:
            break
    
    return best_a, best_obj / (n * 2)  # Return J = 2n*max/sum² = 2n*max (since sum=1)


def subgradient_descent(a, max_iters, deadline, lr_init=0.01, verbose=False):
    """
    Projected subgradient descent on the simplex.
    The subgradient of max_k f_k(a) is the gradient of any active f_k,
    or a convex combination. We use the LogSumExp gradient as a smooth surrogate.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    
    tau = 20.0
    tau_growth = 1.02
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()
        
        # Compute subgradient (smooth approximation)
        grad = weighted_conv_grad(a, w)
        
        # Project to simplex tangent: remove mean
        grad = grad - np.mean(grad)
        
        # Step size (Polyak-like)
        gap = c.max() - c.min()
        lr = lr_init * n / (1.0 + it * 0.01)
        
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        obj_new = 2.0 * n * fast_conv(a_new).max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        tau = min(tau * tau_growth, 500.0)
        
        if no_improve > 200:
            # Restart with perturbation
            a = best_a + np.random.default_rng(it).normal(0, 0.01, n)
            a = project_simplex(a)
            tau = 20.0
            no_improve = 0
        
        if verbose and it % 200 == 0:
            print(f"    Subgrad it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)
    
    return best_a, best_obj / (2.0 * n)  # This returns max(conv) not J... let me fix


def nelder_mead_simplex(a, max_iters, deadline, verbose=False):
    """
    Nelder-Mead on the simplex (using simplex projection after each step).
    Focus on the "active set" - coordinates that are near zero or at the max.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Use a reduced dimension approach: optimize a subset of "active" coordinates
    # Find coordinates with significant values
    threshold = a.max() * 0.1
    active = np.where(a > threshold)[0]
    if len(active) < 2:
        active = np.where(a > 0)[0]
    if len(active) < 2:
        return a, compute_J(a)
    
    # Limit to top 20 active coordinates for Nelder-Mead
    if len(active) > 20:
        top_idx = np.argsort(a[active])[::-1][:20]
        active = active[top_idx]
    
    d = len(active)
    best_obj = compute_J(a)
    best_a = a.copy()
    
    # Initial simplex in the active subspace
    rng = np.random.default_rng(42)
    vertices = [a.copy()]
    step = a[active].max() * 0.1
    for i in range(d):
        v = a.copy()
        v[active[i]] = max(0, a[active[i]] + step)
        v = project_simplex(v)
        vertices.append(v)
    
    # Nelder-Mead iteration
    alpha, gamma, rho = 1.0, 2.0, 0.5
    
    f_values = [compute_J(v) for v in vertices]
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Sort by function value
        order = np.argsort(f_values)
        vertices = [vertices[i] for i in order]
        f_values = [f_values[i] for i in order]
        
        if f_values[0] < best_obj:
            best_obj = f_values[0]
            best_a = vertices[0].copy()
        
        # Centroid of all but worst
        centroid = np.mean(vertices[:-1], axis=0)
        
        # Reflection
        x_r = centroid + alpha * (centroid - vertices[-1])
        x_r = project_simplex(x_r)
        f_r = compute_J(x_r)
        
        if f_r < f_values[0]:
            # Expansion
            x_e = centroid + gamma * (x_r - centroid)
            x_e = project_simplex(x_e)
            f_e = compute_J(x_e)
            if f_e < f_r:
                vertices[-1] = x_e
                f_values[-1] = f_e
            else:
                vertices[-1] = x_r
                f_values[-1] = f_r
        elif f_r < f_values[-2]:
            vertices[-1] = x_r
            f_values[-1] = f_r
        else:
            # Contraction
            x_c = centroid + rho * (vertices[-1] - centroid)
            x_c = project_simplex(x_c)
            f_c = compute_J(x_c)
            if f_c < f_values[-1]:
                vertices[-1] = x_c
                f_values[-1] = f_c
            else:
                # Shrink
                for i in range(1, len(vertices)):
                    vertices[i] = vertices[0] + 0.5 * (vertices[i] - vertices[0])
                    vertices[i] = project_simplex(vertices[i])
                    f_values[i] = compute_J(vertices[i])
        
        if f_values[0] - f_values[-1] < 1e-10:
            break
    
    return best_a, best_obj


def block_cd(a, block_size, deadline, verbose=False):
    """
    Block coordinate descent: optimize blocks of consecutive coordinates.
    Within each block, use the SCA-LP approach restricted to the block.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    rng = np.random.default_rng(123)
    
    while time.monotonic() < deadline:
        if no_improve > 50:
            break
        
        # Shuffle blocks
        n_blocks = n // block_size
        block_starts = list(range(0, n - block_size + 1, block_size))
        rng.shuffle(block_starts)
        
        improved = False
        for start in block_starts[:30]:  # Limit blocks per sweep
            if time.monotonic() > deadline:
                break
            
            end = start + block_size
            block = a[start:end].copy()
            outside = a.copy()
            outside[start:end] = 0
            
            # Optimize the block: minimize max(conv_block) where
            # conv_block[k] = conv(a_full, a_full)[k] with only block varying
            # = 2 * (outside * block)[k] + (block * block)[k] + conv(outside, outside)[k]
            
            # For simplicity, use coordinate descent within the block
            for j in range(block_size):
                idx = start + j
                old = a[idx]
                # Try several candidates
                for x in [0.0, old * 0.5, old * 0.8, old, old * 1.2, old * 1.5, old * 2.0]:

Let me also think about the LP formulation more carefully. The SCA-LP at point a is:

Minimize M
Subject to:
  2 * sum_i a[i] * b[k-i] - M <= c[k] for k = 0, ..., 2n-2
  where c = conv(a,a)
  sum(b) = 1
  b >= 0

The number of constraints is 2n-1 (inequality) + 1 (equality) = 2n
The number of variables is n (for b) + 1 (for M) = n+1

For n = 768: 769 variables, 1536 constraints. This should be fast with HiGHS.

Actually, instead of bisection, I can just minimize M directly as the objective. Let me use that approach since it's simpler and HiGHS should handle it well.

Let me also think about what makes the difference between 1.5065 and 1.5027. The difference is 0.0038, which in terms of max(conv) is 0.0038/(2n) * (sum(a))² ≈ 0.0038/(2*768) ≈ 2.5e-6 (with sum=1). This is a very small difference, so we need a very precise optimization.

I think the key is:
1. Get the SCA-LP to converge more tightly (more iterations, better LP solver settings)
2. Use the "active set" structure: at the optimum, several (a*a)[k] values are tied at the maximum. We can exploit this.
3. Use a final "equalization" step: identify which k values have (a*a)[k] near the max, and try to equalize them.

Optimized SCA-LP with subgradient polishing for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))².
Key innovations: direct M-minimization LP, active-set detection, multi-start SCA,

import numpy as np

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def weighted_conv_grad(a, weights):
    """Compute sum_k weights[k] * d(a*a)[k]/da using FFT."""
    n = len(a)
    a_flip = a[::-1]
    w_flip = weights[::-1]
    size = 1
    while size < 2 * n:
        size <<= 1
    W_fft = np.fft.rfft(weights, size)
    Arev_fft = np.fft.rfft(a_flip, size)
    Wrev_fft = np.fft.rfft(w_flip, size)
    A_fft = np.fft.rfft(a, size)
    conv_w_arev = np.fft.irfft(W_fft * Arev_fft, size)[:2*n-1]
    conv_wrev_a = np.fft.irfft(Wrev_fft * A_fft, size)[:2*n-1]
    P = conv_w_arev[n-1 : n-1+n]
    Q = conv_wrev_a[0:n][::-1]
    return P + Q

def sca_lp_optimize(a, deadline, max_iters=500, verbose=False, lp_timeout=10):

    Iteratively update a = b.

    a = np.maximum(a.astype(np.float64), 0)

    else:

    s = 1.0

            break

        M_curr = c.max()

        # Variables: b[0..n-1], M (index n)
        # Constraints: 2*conv(a,b)[k] - M <= c[k] for k=0,...,2n-2
        # sum(b) = 1
        # b >= 0, M free
        
        rows = []
        cols = []
        vals = []
        b_ub = []
        
        for k in range(2 * n - 1):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                rows.append(k)
                cols.append(k - i)
                vals.append(2.0 * a[i])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])
        
        A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n+1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize M

        A_eq_cols = list(range(n))
        A_eq = coo_matrix((np.ones(n), (A_eq_rows, A_eq_cols)), shape=(1, n+1)).tocsr()

                            best_obj = obj_try
                            best_a = a_try.copy()

                    if not found:

                else:

def logsumexp_gradient_descent(a, max_iters, deadline, tau_init=30.0, verbose=False):

            break
