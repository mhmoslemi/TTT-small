"""
Fast FFT-based gradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothing, Nesterov momentum, adaptive step size, and simplex projection.
Multi-start with diverse (including asymmetric) initializations and multi-n exploration.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    Compute gradient of log-sum-exp(conv/tau) with respect to a.

    Computed via FFT (cross-correlation).

    conv = compute_conv(a)
    cmax = float(np.max(conv))

    shifted = np.clip(shifted, -100.0, 100.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))

        w[np.argmax(conv)] = 1.0
    else:
        w = weights / w_sum

    # Let's define b[j] = a[n-1-j] (flipped a)
    # Then sum_j w[i+j] * a[j] = sum_j w[i+j] * b[n-1-j]
    # = (w * b)[i + n - 1] (convolution)
    # So grad[i] = 2 * (w * b)[i + n - 1]
    
    b = a[::-1]
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    
    Wf = rfft(w, size)
    Bf = rfft(b, size)
    conv_wb = irfft(Wf * Bf, size)
    
    # (w * b)[k] for k = n-1, n, ..., 2n-2
    # grad[i] = 2 * (w*b)[i + n - 1] for i = 0, ..., n-1
    grad = 2.0 * conv_wb[n - 1:2 * n - 1]
    
    return grad, float(np.log(w_sum) + cmax / tau)


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(a)
    if n == 0:
        return a
    # Simple approach: clip and renormalize (fast, approximate)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def project_simplex_exact(v):
    """Exact projection onto simplex {x >= 0, sum(x) = 1} using sorting."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_gradient_descent(a, n_steps=1000, tau_start=0.01, tau_end=0.0001, 
                          momentum=0.9, time_budget=60.0):
    """
    Soft-max gradient descent with momentum and temperature annealing.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    v = a.copy()  # momentum velocity
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    prev_grad = None
    prev_a = None
    
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature
        frac = step / max(n_steps - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute gradient
        grad, _ = compute_soft_gradient(a, tau)
        
        # Center the gradient (remove mean to preserve sum approximately)
        grad_centered = grad - np.mean(grad)
        
        # Adaptive step size (Barzilai-Borwein like)
        if prev_grad is not None and prev_a is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if sy > 1e-10:
                step_size = ss / sy
            else:
                step_size = 0.05 * tau
            step_size = np.clip(step_size, 1e-6, 0.5)
        else:
            step_size = 0.1 * tau
        
        # Nesterov momentum
        v = momentum * v + grad_centered
        a_new = a + step_size * v
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Backtracking if not improving
        if obj_new > best_obj * 1.01:
            # Reduce step
            a_new = a + 0.5 * step_size * v
            a_new = project_simplex(a_new)
            obj_new = compute_obj(a_new)
            if obj_new > best_obj * 1.01:
                a_new = a  # reject update
                obj_new = best_obj
        
        a = a_new
        prev_a = a.copy()
        prev_grad = grad.copy()
    
    return best_a


def soft_gradient_descent_v2(a, n_steps=2000, tau_start=0.005, tau_end=0.00005,
                             time_budget=60.0):
    """
    Improved soft-max gradient descent with:
    - Polyak momentum (adaptive)
    - Better step size selection
    - Periodic re-evaluation
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = 0.9
    step_size = 0.05
    no_improve_count = 0
    last_improve_obj = best_obj
    
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature
        frac = min(step / max(n_steps * 0.7, 1), 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute gradient
        grad, _ = compute_soft_gradient(a, tau)
        grad_centered = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad_centered
        v_norm = float(np.linalg.norm(v))
        if v_norm < 1e-15:
            break
        
        v_normalized = v / v_norm
        
        # Adaptive step size
        a_new = a + step_size * v_normalized
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve_count = 0
            step_size *= 1.05  # slightly increase
        else:
            no_improve_count += 1
            step_size *= 0.8  # decrease
            if no_improve_count > 50:
                # Reset momentum and try different direction
                v = np.zeros(n)
                step_size = 0.05
                no_improve_count = 0
                # Small random perturbation
                rng = np.random.default_rng(step)
                a = normalize(a + rng.normal(0, 0.01, n))
        
        a = a_new if obj_new < best_obj or no_improve_count < 10 else a
        step_size = np.clip(step_size, 1e-6, 1.0)
        
        # Decrease momentum over time
        momentum = 0.9 - 0.3 * (step / n_steps)
    
    return best_a


def soft_gradient_descent_v3(a, n_steps=3000, time_budget=60.0):
    """
    Most robust version: backtracking line search + Nesterov + annealing.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = 0.85
    base_step = 0.1
    
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        frac = step / max(n_steps - 1, 1)
        # Temperature schedule: start high, decrease
        tau = 0.01 * (1.0 - frac * 0.99) + 0.0001
        tau = max(tau, 0.00005)
        
        # Nesterov lookahead
        a_lookahead = a + momentum * v
        grad, _ = compute_soft_gradient(a_lookahead, tau)
        grad_centered = grad - np.mean(grad)
        
        # Update momentum
        v_new = momentum * v + grad_centered
        v_norm = float(np.linalg.norm(v_new))
        if v_norm < 1e-18:
            break
        
        # Backtracking line search
        step_size = base_step / (1.0 + step * 0.01)
        a_old = a.copy()
        obj_old = compute_obj(a)
        
        accepted = False
        for bt in range(10):
            a_try = a + step_size * v_new
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            
            if obj_try <= obj_old + 1e-12:
                accepted = True
                break
            step_size *= 0.5
        
        if accepted:
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()
        else:
            # Accept anyway if very close
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()
        
        v = v_new
        momentum = 0.85 - 0.3 * frac  # decrease momentum over time
    
    return best_a


def init_hann(n, asymmetric=False):
    """Hann window initialization."""
    if asymmetric:
        # Asymmetric: more weight on one side
        j = np.arange(n)
        a = np.sin(np.pi * j / n) ** 2
        a = a + 0.3 * np.exp(-((j - n * 0.3) / (n * 0.2)) ** 2)
    else:
        j = np.arange(n)
        a = np.sin(np.pi * (j + 0.5) / n) ** 2
    return normalize(a)


def init_power_law(n, beta=1.0, asymmetric=False):
    """Power-law taper initialization."""
    j = np.arange(n)
    if asymmetric:
        # Taper more on one side
        left = j / (n * 0.3)
        right = (n - 1 - j) / (n * 0.7)
        a = np.power(np.maximum(1.0 - left, 0.01), beta) * \
            np.power(np.maximum(1.0 - right, 0.01), beta / 2)
    else:
        # Symmetric power-law taper
        x = np.linspace(-1, 1, n)
        a = np.power(np.maximum(1.0 - x**2, 0.01), beta / 2)
    return normalize(a)


def init_gaussian(n, sigma_frac=0.3, center=0.5):
    """Gaussian initialization."""
    j = np.arange(n)
    a = np.exp(-((j - n * center) / (n * sigma_frac)) ** 2)
    return normalize(a)


def init_random(n, seed=0):
    """Random initialization with Dirichlet distribution."""
    rng = np.random.default_rng(seed)
    alpha = rng.uniform(0.5, 2.0)
    a = rng.dirichlet(np.ones(n) * alpha)
    return a


def init_from_parent(parent, n):
    """Resample parent to new length."""
    if len(parent) == n:
        return normalize(parent.copy())
    x_old = np.linspace(0.0, 1.0, len(parent))
    x_new = np.linspace(0.0, 1.0, n)
    a = np.interp(x_new, x_old, parent)
    return normalize(a)


def spectral_flatten(a, n_iters=20, time_budget=10.0):
    """
    Fast spectral flattening: iteratively flatten the power spectrum
    while maintaining non-negativity.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 4 * n:
        size <<= 1
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # FFT
        A = fft(a, size)
        P = np.abs(A) ** 2
        P_max = float(np.max(P))
        P_min = float(np.min(P[P > 1e-30]))
        if P_max < 1e-30:
            break
        
        # Compress power spectrum
        P_norm = P / P_max
        alpha = 0.3 + 0.4 * (it / n_iters)  # start mild, get stronger
        P_new = np.maximum(P_norm ** alpha, P_min / P_max * 0.1)
        
        # Reconstruct
        phase = np.angle(A)
        mag = np.sqrt(P_new)
        A_new = mag * np.exp(1j * phase)
        
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            a = 0.7 * a + 0.3 * a_new
            a = normalize(a)
    
    return best_a


def coordinate_polish(a, time_budget=20.0):
    """
    Fast coordinate descent: try moving mass from peak contributors to non-peak positions.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find top contributors to peak
        contribs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)
        
        improved = False
        for peak_val, j1, j2 in contribs[:3]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            
            # Try moving small mass from j1 to a random position far from peak
            for _ in range(5):
                j_target = int(rng.integers(0, n))
                if j_target in (j1, j2):
                    continue
                # Check if moving to j_target would create a new peak
                a_try = a.copy()
                delta = 0.03 * a_try[j1]
                a_try[j1] -= delta
                a_try[j_target] += delta
                a_try = np.maximum(a_try, 0)
                s = float(np.sum(a_try))
                if s < 1e-10:
                    continue
                a_try = a_try / s
                
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved:
            # Try random perturbation
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
    
    return best_a

"""
Fast FFT-based gradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothing with Nesterov momentum, adaptive step size, multiplicative
weight updates, asymmetric initializations, spectral flattening, and multi-n exploration.
Key insight: asymmetric sequences can achieve lower objective than symmetric ones.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale-invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_soft_gradient(a, tau):
    """
    Compute gradient of log-sum-exp(conv/tau) w.r.t. a using FFT.
    grad[i] = 2 * sum_j w[i+j] * a[j] where w = softmax(conv/tau)
    Computed as cross-correlation via FFT.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -200.0, 200.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w = weights / w_sum

    # grad[i] = 2 * (w * b)[i + n - 1] where b = a[::-1]
    b = a[::-1]
    size = 1
    while size < len(w) + n - 1:
        size <<= 1

    Wf = rfft(w, size)
    Bf = rfft(b, size)
    conv_wb = irfft(Wf * Bf, size)

    grad = 2.0 * conv_wb[n - 1:2 * n - 1]
    return grad


def gradient_descent_nesterov(a, n_steps=2000, time_budget=60.0, tau_start=0.01, tau_end=0.0001):
    """
    Nesterov-accelerated gradient descent with backtracking, temperature annealing.
    Works on normalized (sum=1) non-negative sequences.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = 0.8
    step_size = 0.05
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        frac = step / max(n_steps - 1, 1)
        # Temperature: logarithmic annealing
        tau = tau_start * (tau_end / tau_start) ** frac
        tau = max(tau, 1e-5)

        # Nesterov lookahead
        a_look = a + momentum * v
        a_look = np.maximum(a_look, 0)
        s_look = float(np.sum(a_look))
        if s_look < 1e-15:
            a_look = a.copy()
        else:
            a_look = a_look / s_look

        grad = compute_soft_gradient(a_look, tau)
        grad_centered = grad - np.mean(grad)

        # Momentum update
        v = momentum * v + grad_centered

        # Adaptive step size
        v_norm = float(np.linalg.norm(v))
        if v_norm < 1e-18:
            break

        # Backtracking line search
        a_old = a.copy()
        obj_old = compute_obj(a_old)
        ss = step_size
        accepted = False
        for bt in range(8):
            a_try = a + ss * v
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-15:
                ss *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try <= obj_old + 1e-13:
                accepted = True
                break
            ss *= 0.5

        if accepted:
            a = a_try
            step_size = min(ss * 1.5, 0.5)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            step_size *= 0.8

        if no_improve > 80:
            # Restart with perturbation
            rng = np.random.default_rng(step * 31 + 7)
            a = normalize(a * np.exp(rng.normal(0, 0.05, n)))
            v = np.zeros(n)
            step_size = 0.05
            no_improve = 0

        # Adapt momentum
        momentum = 0.8 - 0.3 * (step / n_steps)

    return best_a


def gradient_descent_multiplicative(a, n_steps=2000, time_budget=60.0, tau_start=0.01, tau_end=0.0001):
    """
    Multiplicative weight (mirror descent) gradient method.
    a_new[i] = a[i] * exp(-eta * grad[i]), then normalize.
    Naturally maintains non-negativity.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.1
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        frac = step / max(n_steps - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        tau = max(tau, 1e-5)

        grad = compute_soft_gradient(a, tau)
        grad_centered = grad - np.mean(grad)

        # Multiplicative update
        a_try = a * np.exp(-eta * grad_centered)
        a_try = np.maximum(a_try, 1e-300)
        s_try = float(np.sum(a_try))
        if s_try < 1e-15:
            break
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
            eta = min(eta * 1.03, 0.5)
        else:
            no_improve += 1
            eta *= 0.85
            if no_improve > 60:
                # Reset with perturbation
                rng = np.random.default_rng(step * 13 + 3)
                a = normalize(a * np.exp(rng.normal(0, 0.04, n)))
                eta = 0.1
                no_improve = 0

        # Cool eta over time
        eta *= (1.0 - 0.001 * frac)

    return best_a

def spectral_flatten(a, n_iters=30, time_budget=15.0):
    """
    Iteratively flatten power spectrum while maintaining non-negativity.
    Reduces concentration in frequency domain → flatter autocorrelation.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 4 * n:
        size <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        P_max = float(np.max(P))
        if P_max < 1e-30:
            break

        P_norm = P / P_max
        # Compress: values near 1 stay, values near 0 get raised
        alpha = 0.4 + 0.3 * (it / max(n_iters - 1, 1))
        P_new = np.maximum(P_norm ** alpha, 0.05)

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Gentle blend
            blend = 0.5 + 0.3 * (it / max(n_iters - 1, 1))
            a = (1 - blend) * a + blend * a_new
            a = normalize(a)

    return best_a


def coordinate_polish(a, time_budget=15.0):
    """
    Fast coordinate descent: move mass from peak contributors to reduce max convolution.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors
        contribs = []
        for j in range(min(n, 100)):  # limit for speed
            k2 = k_star - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        for j in range(n - min(n, 100), n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contribs[:5]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2 or a[j1] < 1e-10:
                continue

            for _ in range(4):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue

                a_try = a.copy()
                delta = 0.04 * a_try[j1]
                a_try[j1] -= delta
                a_try[j_target] += delta
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try

                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and rng.random() < 0.15:
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a

        a = np.sin(np.pi * (j + 0.5) / n) ** 2
        # Add asymmetric bump
        a += 0.5 * np.exp(-((j - n * asym_center) / (n * 0.2)) ** 2)
    else:
        a = np.sin(np.pi * (j + 0.5) / n) ** 2
    return normalize(a)


def init_power_law(n, beta=1.0, asymmetric=False):
    """Power-law edge taper."""
    j = np.arange(n)
    if asymmetric:
        left_taper = np.power(np.maximum(0.1 + 0.9 * j / n, 0.01), beta)
        right_taper = np.power(np.maximum(0.3 + 0.7 * (1 - j / n), 0.01), beta * 0.5)
        a = left_taper * right_taper
    else:
        x = (2.0 * j / (n - 1) - 1.0) if n > 1 else np.array([0.0])
        a = np.power(np.maximum(1.0 - x ** 2, 0.01), beta / 2.0)
    return normalize(a)


def init_gaussian(n, sigma_frac=0.25, center=0.4):
    """Gaussian bump at given center."""
    j = np.arange(n)
    a = np.exp(-((j - n * center) / (n * sigma_frac)) ** 2)
    return normalize(a)


def init_bimodal(n, c1=0.25, c2=0.7, sigma=0.12):
    """Two Gaussian bumps."""
    j = np.arange(n)
    a = np.exp(-((j - n * c1) / (n * sigma)) ** 2) + np.exp(-((j - n * c2) / (n * sigma)) ** 2)
    return normalize(a)


def init_ramp(n, power=1.0):
    """Monotone ramp (very asymmetric)."""
    j = np.arange(n)
    a = np.power((j + 1) / n, power)
    return normalize(a)


def init_random(n, seed=0, concentration=1.0):
    """Random Dirichlet initialization."""
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n) * concentration)
    return a


def init_from_parent(parent, n):
    """Resample parent sequence to new length."""
    if len(parent) == n:
        return normalize(parent.copy())
    x_old = np.linspace(0.0, 1.0, len(parent))
    x_new = np.linspace(0.0, 1.0, n)
    a = np.interp(x_new, x_old, parent)
    return normalize(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase strategy using fast FFT-based gradient methods.
    
    Phase 1: Optimize parent with gradient descent (Nesterov + multiplicative hybrid)
    Phase 2: Spectral flattening + gradient
    Phase 3: Diverse initializations (especially asymmetric)
    Phase 4: Multi-n exploration
    Phase 5: Intensive rounds at best n
    Phase 6: Final coordinate polish
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            best_score = obj
            best_seq = parent.copy()
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = init_hann(500, asymmetric=True)
        best_score = compute_obj(best_seq)
        print(f"No parent, using asymmetric Hann n=500: obj={best_score:.6f}")

    n_parent = len(best_seq)

    # =========================================================
    # Phase 1: Hybrid gradient descent at parent length
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Hybrid gradient at n={n_parent}")
    print(f"{'='*60}")

    a_cur = normalize(best_seq.copy())

    # 1a: Nesterov gradient
    if time.monotonic() < deadline - 300:
        a1 = gradient_descent_nesterov(a_cur, n_steps=1500, time_budget=55.0,
                                       tau_start=0.01, tau_end=0.0001)
        obj1 = compute_obj(a1)
        print(f"  Nesterov: {obj1:.8f}")
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            a_cur = a1

    # 1b: Multiplicative gradient
    if time.monotonic() < deadline - 250:
        a2 = gradient_descent_multiplicative(best_seq, n_steps=1500, time_budget=45.0,
                                             tau_start=0.008, tau_end=0.0001)
        obj2 = compute_obj(a2)
        print(f"  Multiplicative: {obj2:.8f}")
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            a_cur = a2

    # 1c: Combined (switching)
    if time.monotonic() < deadline - 200:
        a3 = gradient_descent_combined(best_seq, n_steps=2000, time_budget=50.0)
        obj3 = compute_obj(a3)
        print(f"  Combined: {obj3:.8f}")
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            a_cur = a3

    # =========================================================
    # Phase 2: Spectral flattening + gradient
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Spectral flatten + gradient")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 200:
        a_spec = spectral_flatten(best_seq, n_iters=40, time_budget=20.0)
        obj_spec = compute_obj(a_spec)
        print(f"  Spectral only: {obj_spec:.8f}")

        # Follow with gradient
        a_spec_grad = gradient_descent_combined(a_spec, n_steps=1500, time_budget=45.0)
        obj_spec_grad = compute_obj(a_spec_grad)
        print(f"  Spectral+Grad: {obj_spec_grad:.8f}")

        if obj_spec_grad < best_score:
            best_score = obj_spec_grad
            best_seq = a_spec_grad.copy()
        elif obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()

    # =========================================================
    # Phase 3: Diverse initializations
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Diverse initializations at n={n_parent}")
    print(f"{'='*60}")

    n = n_parent
    diverse_inits = {}

    # Asymmetric variants (key insight)
    diverse_inits['hann_asymp_0.3'] = init_hann(n, asymmetric=True, asym_center=0.3)
    diverse_inits['hann_asymp_0.4'] = init_hann(n, asymmetric=True, asym_center=0.4)
    diverse_inits['hann_asymp_0.6'] = init_hann(n, asymmetric=True, asym_center=0.6)
    diverse_inits['power_asymp'] = init_power_law(n, beta=1.5, asymmetric=True)
    diverse_inits['power_sym'] = init_power_law(n, beta=1.0, asymmetric=False)
    diverse_inits['gauss_left'] = init_gaussian(n, sigma_frac=0.2, center=0.3)
    diverse_inits['gauss_right'] = init_gaussian(n, sigma_frac=0.2, center=0.7)
    diverse_inits['gauss_center'] = init_gaussian(n, sigma_frac=0.3, center=0.5)
    diverse_inits['bimodal'] = init_bimodal(n, c1=0.2, c2=0.65, sigma=0.15)
    diverse_inits['ramp1'] = init_ramp(n, power=1.0)
    diverse_inits['ramp2'] = init_ramp(n, power=2.0)
    diverse_inits['ramp_inv'] = init_ramp(n, power=1.0)[::-1]  # inverted ramp
    diverse_inits['rand1'] = init_random(n, seed=7, concentration=0.5)
    diverse_inits['rand2'] = init_random(n, seed=42, concentration=2.0)

    for name, a_init in diverse_inits.items():
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        alloc = min(remaining * 0.08, 20.0)
        a_opt = gradient_descent_combined(a_init, n_steps=800, time_budget=alloc)
        obj_opt = compute_obj(a_opt)

        status = "**" if obj_opt < best_score else ""
        print(f"  {name}: {obj_opt:.8f} {status}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()

    # =========================================================
    # Phase 4: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq)
    n_options = sorted(set([
        max(150, n_best - 200),
        max(150, n_best - 100),
        max(150, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        min(2000, n_best + 400),
        min(2000, n_best + 600),
    ]))
    n_options = [n for n in n_options if 150 <= n <= 2500]

    for n_try in n_options:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Two strategies at each n: resample from best, and asymmetric init
        a_resampled = init_from_parent(best_seq, n_try)
        a_asym = init_hann(n_try, asymmetric=True, asym_center=0.35)

        alloc = min(remaining * 0.06, 15.0)

        a_opt1 = gradient_descent_combined(a_resampled, n_steps=600, time_budget=alloc)
        obj1 = compute_obj(a_opt1)

        a_opt2 = gradient_descent_combined(a_asym, n_steps=400, time_budget=alloc * 0.7)
        obj2 = compute_obj(a_opt2)

        best_at_n = min(obj1, obj2)
        best_a_at_n = a_opt1 if obj1 < obj2 else a_opt2

        status = "**" if best_at_n < best_score else ""
        print(f"  n={n_try}: {best_at_n:.8f} {status}")

        if best_at_n < best_score:
            best_score = best_at_n
            best_seq = best_a_at_n.copy()
            n_best = n_try

    # =========================================================
    # Phase 5: Intensive optimization at best n
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Intensive at n={len(best_seq)}")
    print(f"{'='*60}")

    for rnd in range(8):
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * 0.15, 35.0)

        # Perturb current best
        rng = np.random.default_rng(seed + rnd * 97)
        if rnd % 4 == 0:
            a_try = best_seq * np.exp(rng.normal(0, 0.03, len(best_seq)))
        elif rnd % 4 == 1:
            # Spectral perturbation
            sz = 1
            while sz < 2 * len(best_seq):
                sz <<= 1
            A = fft(normalize(best_seq), sz)
            A = A * np.exp(1j * rng.normal(0, 0.04, sz // 2 + 1))
            a_try = np.real(ifft(A, sz))[:len(best_seq)]
            a_try = np.maximum(a_try, 0)
        elif rnd % 4 == 2:
            # Blend with asymmetric init
            a_other = init_hann(len(best_seq), asymmetric=True, asym_center=rng.uniform(0.2, 0.5))
            a_try = 0.7 * normalize(best_seq) + 0.3 * a_other
        else:
            # Spectral flatten then use as start
            a_try = spectral_flatten(best_seq, n_iters=10, time_budget=min(5, alloc * 0.2))

        a_try = normalize(a_try)
        a_opt = gradient_descent_combined(a_try, n_steps=1200, time_budget=alloc)
        obj_opt = compute_obj(a_opt)

        status = "**" if obj_opt < best_score else ""
        print(f"  Round {rnd}: {obj_opt:.8f} {status}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()

    # =========================================================
    # Phase 6: Final polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 10:
        a_pol = coordinate_polish(best_seq, time_budget=min(15.0, deadline - time.monotonic() - 10))
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
        print(f"  CoordPolish: {best_score:.8f}")

    if time.monotonic() < deadline - 5:
        a_pol = gradient_descent_nesterov(best_seq, n_steps=500, time_budget=
            min(12.0, deadline - time.monotonic() - 5), tau_start=0.003, tau_end=0.00005)
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
        print(f"  Final Nesterov: {best_score:.8f}")

    # =========================================================
    # Final validation
    # =========================================================
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
