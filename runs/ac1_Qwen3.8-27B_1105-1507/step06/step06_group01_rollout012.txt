Optimization of non-negative step functions to minimize J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP best-response with exact line search, spectral initialization, and multi-length search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2 * len(a) * fast_conv(a).max() / (s * s))

def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * rho_arr) > (cv - 1.0)
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.max(np.where(cond)[0]))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def spectral_init(n, seed, target_flatness=0.5):
    """Generate a sequence with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    # Start with random positive sequence
    a = rng.uniform(0.5, 1.5, n)
    # Iteratively flatten the spectrum
    for it in range(50):
        b = fast_conv(a)
        M = b.max()
        # The autocorrelation should be flat, which means power spectrum should be concentrated
        # Actually, we want conv(a,a) to be flat, which means the "cross-correlation" is flat
        # For symmetric a, this is equivalent to |FFT(a)|^2 being flat
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        A = np.fft.rfft(a, size)
        P = np.abs(A)**2
        # We want P to be flat (uniform)
        P_flat = P.mean()
        # Adjust: reduce P where it's high, increase where it's low
        # But we need to keep a real and non-negative
        # Simple approach: adjust amplitude while keeping phase
        phase = np.angle(A)
        mag = np.sqrt(P)
        # Flatten magnitude
        target = np.sqrt(P_flat)
        mag_new = 1.0 + target_flatness * (target - mag) / (mag + 1e-10)
        mag_new = np.maximum(mag_new, 0.1)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 1e-6)
        a = a_new / a_new.sum()
    return a / a.sum()

def lp_direction(a, M, n_active, n=None):
    """Compute LP best-response direction.
    
    max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0
    """
    if n is None:
        n = len(a)
    b = fast_conv(a)
    
    # Get active set
    top_idx = np.argsort(b)[::-1][:min(n_active, 2*n-1)]
    active = top_idx
    
    # Build constraint matrix
    # (a*g)[k] = sum_j a[k-j] * g[j] for valid j
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(active):
        for j in range(n):
            m = k - j
            if 0 <= m < n and a[m] > 1e-15:
                rows.append(ii)
                cols.append(j)
                vals.append(a[m])
    
    if not rows:
        return None
    
    A_mat = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def golden_section_line_search(a, g, current_M, n_evals=40):
    """Find best mixing parameter t in [0, 1] to minimize max(conv((1-t)a+tg, (1-t)a+tg))."""
    lo = 0.0
    hi = 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    
    def obj(t):
        x = (1.0 - t) * a + t * g
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 0.01:
            return float('inf')
        x = x / s
        return fast_conv(x).max()
    
    f1 = obj(a1)
    f2 = obj(a2)
    
    for _ in range(n_evals):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1 = obj(a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = obj(a2)
    
    t_opt = (lo + hi) / 2.0
    # Also check t=0 and t=1
    f0 = current_M
    f1_opt = obj(t_opt)
    f1_full = obj(1.0)
    
    if f1_full < f0 and f1_full < f1_opt:
        t_opt = 1.0
        f1_opt = f1_full
    
    return t_opt, f1_opt

def run_lp_optimize(a_init, deadline, n_active_start=30, n_active_end=80, min_improve=1e-8):
    """LP-based optimization with adaptive active set and exact line search."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    n_active = n_active_start
    
    for it in range(500):
        if time.monotonic() > deadline:
            break
        
        # Get LP direction
        g = lp_direction(a, best_M, n_active, n)
        if g is None:
            no_improve += 2
            n_active = min(n_active * 2, 2 * n - 1)
            if no_improve > 10:
                break
            continue
        
        # Line search
        t_opt, M_new = golden_section_line_search(a, g, best_M, n_evals=35)
        
        if M_new < best_M - min_improve:
            a_trial = (1.0 - t_opt) * a + t_opt * g
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            a = a_trial
            best_M = M_new
            best_a = a.copy()
            no_improve = 0
            # Try to reduce active set
            if n_active > 15:
                n_active = max(15, int(n_active * 0.9))
        else:
            no_improve += 1
            n_active = min(int(n_active * 1.5), 2 * n - 1)
            if no_improve > 15:
                break
    
    return best_a, best_M

def run_coordinate_descent(a_init, deadline, max_sweeps=50, seed=0):
    """Coordinate descent with exact 1D optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = np.arange(n)
        rng.shuffle(order)
        
        b = fast_conv(a)
        improved = False
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            idx = order[i]
            
            # Compute the 1D objective: f(x) = max of affected convolution values
            # (a*a)[k] for k in [idx, min(2n-2, idx+n-1)]
            k_lo = idx
            k_hi = min(2 * n - 2, idx + n - 1)
            
            # Precompute constants and slopes
            c_vals = []
            s_vals = []
            has_quad = False
            quad_idx = -1
            
            for k in range(k_lo, k_hi + 1):
                if k == 2 * idx:
                    c_vals.append(b[k] - a[idx]**2)
                    s_vals.append(0.0)
                    has_quad = True
                    quad_idx = len(c_vals) - 1
                else:
                    j = k - idx
                    if 0 <= j < n:
                        c_vals.append(b[k] - 2.0 * a[idx] * a[j])
                        s_vals.append(2.0 * a[j])
                    else:
                        c_vals.append(b[k])
                        s_vals.append(0.0)
            
            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi+1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            
            c_arr = np.array(c_vals)
            s_arr = np.array(s_vals)
            
            def obj_1d(x):
                vals = c_arr + s_arr * x
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if has_quad:
                    qv = c_arr[quad_idx] + x * x
                    if qv > m:
                        m = qv
                return m
            
            # Golden section for optimal x
            x_lo = 0.0
            x_hi = max(2.0 * a[idx], np.sqrt(max(b.max(), 0.0)) * 1.5, 0.5)
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = x_hi - gr * (x_hi - x_lo)
            a2 = x_lo + gr * (x_hi - x_lo)
            f1 = obj_1d(a1)
            f2 = obj_1d(a2)
            
            for _ in range(35):
                if f1 < f2:
                    x_hi = a2
                    a2 = a1
                    f2 = f1
                    a1 = x_hi - gr * (x_hi - x_lo)
                    f1 = obj_1d(a1)
                else:
                    x_lo = a1
                    a1 = a2
                    f1 = f2
                    a2 = x_lo + gr * (x_hi - x_lo)
                    f2 = obj_1d(a2)
            
            x_opt = (x_lo + x_hi) / 2.0
            
            # Check boundary
            if obj_1d(0.0) <= obj_1d(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - a[idx]) > 1e-10:
                a[idx] = x_opt
                improved = True
                # Update b
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * idx:
                        b[k] = c_vals[quad_idx] + x_opt * x_opt if has_quad else b[k]
                        # Actually need to recompute
                        b[k] = sum(a[j] * a[k-j] for j in range(max(0, k-n+1), min(n, k+1)))
                    else:
                        j = k - idx
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - a[idx])
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    return best_a, best_M

def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and gradient (for FISTA)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * b - tau * b_max)
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: d/da[i] of sum_k p[k] * (a*a)[k]
    # (a*a)[k] = sum_j a[j] * a[k-j]
    # d/da[i] (a*a)[k] = a[k-i] + a[i-k] (if valid) ... wait
    # Actually for the full convolution: d/da[i] (a*a)[k] = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
    # = a[k-i] + a[k-i] ... no
    # (a*a)[k] = sum_j a[j] * a[k-j]
    # d/da[i] = sum_j [delta(j,i) * a[k-j] + a[j] * delta(k-j, i)]
    # = a[k-i] + a[k-i] ... wait
    # First term: j=i, gives a[k-i]
    # Second term: k-j=i, so j=k-i, gives a[k-i]
    # So d/da[i] (a*a)[k] = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n, else just one term
    # Actually: if 0 <= k-i < n, both terms are valid, so it's 2*a[k-i]
    # But wait, when k-i < 0 or k-i >= n, only one or zero terms are valid
    # More precisely:
    # d/da[i] sum_j a[j]*a[k-j] = a[k-i] (from first factor) + a[k-i] (from second factor, when j=k-i is valid)
    # First factor valid when 0 <= k-i < n: contributes a[k-i]
    # Second factor: j = k-i must be in [0,n-1], i.e., 0 <= k-i < n: contributes a[k-i]
    # So if 0 <= k-i < n: gradient contribution is 2*a[k-i]
    # Otherwise: 0
    
    # This is O(n^2) for the full gradient. For n=768, that's ~600K operations per eval.
    # Can optimize with FFT: grad[i] = 2 * sum_k p[k] * a[k-i]
    # This is a cross-correlation of p and a
    grad = np.zeros(n)
    # Use FFT for efficiency
    size = 1
    while size < 2 * n:
        size <<= 1
    Pf = np.fft.rfft(p, size)
    Af = np.fft.rfft(a, size)
    # cross-correlation: grad[i] = 2 * sum_k p[k] * a[k-i] (where a is zero-padded)
    # This is (p * reverse(a))[n-1-i] ... let me think
    # sum_k p[k] * a[k-i] = (p convolve reverse(a))[n-1-i]
    # Actually, let's just use the direct formula
    # grad[i] = 2 * sum_{k=max(i,0)}^{min(n-1+i, 2n-2)} p[k] * a[k-i]
    # = 2 * correlation of p and a at lag i
    # Using FFT: cross_corr = ifft(fft(p) * conj(fft(a)))
    cross = np.fft.irfft(Pf * np.conj(Af), size)[:2*n-1]
    # cross[k] = sum_j p[j] * a[j+k-n+1] ... this is getting confusing
    # Let me just do it directly for correctness
    for i in range(n):
        s = 0.0
        for k in range(i, i + n):
            if k < len(p):
                s += p[k] * a[k - i]
        grad[i] = 2.0 * s
    
    return smooth_val, grad

def run_fista(a_init, deadline, tau_start=50, tau_end=500, max_iters=2000):
    """FISTA on smooth max approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad = smooth_max_grad(y, tau)
        grad_tan = grad - grad.mean()
        
        step = 0.3 / (1.0 + 0.01 * tau)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        if it % 10 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a, best_M

def mirror_descent_step(a, b, M, step_size):
    """One mirror descent (multiplicative weights) step."""
    n = len(a)
    k_star = int(np.argmax(b))
    
    # Subgradient
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    
    grad_tan = grad - grad.mean()
    log_w = -step_size * grad_tan
    log_w -= log_w.max()
    a_new = a * np.exp(log_w)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new

def run_mirror_descent(a_init, deadline, step_init=0.3, max_iters=5000):
    """Iterated mirror descent."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        step = step_init / (1.0 + 0.005 * it)
        a = mirror_descent_step(a, b, M, step)
    
    return best_a, best_M

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Parent (resampled)
    if parent is not None and parent.size > 0:
        if parent.size != n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        else:
            p = parent.copy()
        p = np.maximum(p, 1e-6)
        inits.append(p / p.sum())
        # Perturbed parent
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            p_pert = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(p_pert / p_pert.sum())
    
    # Symmetric parent
    if parent is not None and parent.size == n:
        sym = (parent + parent[::-1]) / 2.0
        sym = np.maximum(sym, 1e-6)
        inits.append(sym / sym.sum())
    
    # Spectral init
    spec = spectral_init(n, seed=int(rng.integers(0, 10000)))
    inits.append(spec)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # U-shaped
    u_shape = ((2*t - 1)**2 + 0.1)
    inits.append(u_shape / u_shape.sum())
    
    # Cosine
    cos_w = np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.05)
    inits.append(cos_w / cos_w.sum())
    
    # Gaussian
    gauss = np.exp(-5.0 * (t - 0.5)**2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits

def resample_seq(src, dest_n):
    """Resample to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    Multi-phase optimization combining LP best-response with exact line search,

    """

    # Load parent
    parent = None
    try:

            parent = np.maximum(p, 0.0)
            print(f"Parent loaded: n={parent.size}, J={compute_J(parent):.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    global_best = parent.copy()

    def update_best(a):

        J = compute_J(a)

            print(f"  ** NEW BEST: J={J:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    print(f"Initial best: J={global_best_J:.8f} n={global_best.size}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Multi-length, multi-init LP optimization", flush=True)
    print(f"{'='*60}", flush=True)
    
    n_parent = parent.size
    lengths = [n_parent]
    for delta in [64, 128, 192, 256, -64, -128, 320, 384]:

        if 512 <= nl <= 1536 and nl not in lengths:

    lengths.sort(key=lambda x: abs(x - n_parent))
    lengths = lengths[:5]  # Try at most 5 lengths
