"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast top-k smoothed gradient descent with Nesterov acceleration (O(n*k) per step)
2. LP best-response with cutting-plane (lazy constraint) approach
3. Fast coordinate descent with incremental convolution updates
4. Multi-phase alternating with random perturbations to escape local minima
Key improvements over previous: O(n*k) gradient (vs O(n^2)), cutting-plane LP,
faster coordinate descent, better time allocation, perturbation-based exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv_cross(a, b):
    """FFT-based linear convolution of a and b."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def topk_gradient(a, b, k):
    """Compute gradient of mean of top-k elements of b=conv(a,a).
    grad[i] = (2/k) * sum_{j in top_k} a[j-i] if 0<=j-i<n, else 0.
    Returns (smooth_value, gradient).
    """
    n = len(a)
    k_eff = min(k, len(b))
    top_k = np.argsort(b)[::-1][:k_eff]
    smooth_val = b[top_k].mean()

    grad = np.zeros(n)
    for j in top_k:
        j = int(j)
        i_lo = max(0, j - n + 1)
        i_hi = min(n - 1, j)
        if i_lo <= i_hi:
            # a[j-i] for i in [i_lo, i_hi]
            # When i=i_lo: j-i = j-i_lo (largest index into a)
            # When i=i_hi: j-i = j-i_hi (smallest index into a)
            # So the slice is a[j-i_hi : j-i_lo+1] reversed
            sl = a[j - i_hi: j - i_lo + 1][::-1]
            grad[i_lo:i_hi + 1] += sl

    grad *= 2.0 / k_eff
    return smooth_val, grad


def run_topk_gradient(a_init, deadline, k_values=None, seed=0):
    """Nesterov-accelerated gradient descent on top-k smooth approximation."""
    if k_values is None:
        k_values = [10, 20, 50]

    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(seed)

    for k in k_values:
        if time.monotonic() > deadline - 1:
            break

        a_cur = best_a.copy()
        b_cur = fast_conv(a_cur)
        M_cur = b_cur.max()

        t = 1.0
        y = a_cur.copy()
        a_prev = a_cur.copy()
        lr = 0.5
        no_improve = 0

        max_iters = 20000
        for it in range(max_iters):
            if time.monotonic() > deadline - 1:
                break

            b_y = fast_conv(y)
            _, grad_y = topk_gradient(y, b_y, k)
            grad_tan = grad_y - grad_y.mean()

            step = lr / (1.0 + 0.01 * it)
            a_new = y - step * grad_tan
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = best_a.copy()
            else:
                a_new = a_new / s_new

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
            y_new = a_new + alpha * (a_new - a_prev)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-10:
                y_new = y_new / s_y

            a_prev = a_cur
            a_cur = a_new
            y = y_new
            t = t_new

            # Check improvement every 20 iterations
            if it % 20 == 0:
                M_new = fast_conv(a_cur).max()
                if M_new < best_M * (1 - 1e-10):
                    best_M = M_new
                    best_a = a_cur.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve > 200:
                        break

                M_y = fast_conv(y).max()
                if M_y < best_M * (1 - 1e-10):
                    best_M = M_y
                    best_a = y.copy()
                    no_improve = 0

                # Adaptive learning rate
                if no_improve > 100:
                    lr *= 0.7
                    if lr < 0.01:
                        break

    # Final check
    M_a = fast_conv(a_cur).max()
    if M_a < best_M:
        best_a = a_cur.copy()
        best_M = M_a

    return best_a, best_M


def build_sparse_a(n, a, active_indices):
    """Build sparse constraint matrix for LP: (a*g)[k] = sum_j a[k-j]*g[j]."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        k = int(k)
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response_cutting_plane(a, M, n_active_start=20, n_active_max=80,
                                   max_rounds=5, deadline=None):
    """LP best-response with cutting-plane (lazy constraint) approach.
    Returns normalized best response g or None.
    """
    n = len(a)
    b = fast_conv(a)
    conv_len = 2 * n - 1

    # Start with top n_active_start
    top_idx = np.argsort(b)[::-1][:min(n_active_start, conv_len)]
    active = list(top_idx.astype(int))

    g = None
    for round_num in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            return g

        if len(active) == 0:
            break

        A = build_sparse_a(n, a, active)
        if A.nnz == 0:
            break

        c = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0.0, None)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 20, 'presolve': True})
            if not r.success or r.x is None:
                break
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g_norm = g / g_sum
        except Exception:
            break

        # Check for violated constraints
        b_g = fast_conv_cross(a, g)
        violations = []
        for k in range(conv_len):
            if k not in active and b_g[k] > M * (1 + 1e-6):
                violations.append((b_g[k] - M, k))

        if not violations:
            break

        # Add most violated constraints
        violations.sort(reverse=True)
        n_add = min(10, len(violations), n_active_max - len(active))
        for _, k in violations[:n_add]:
            active.append(int(k))

        if len(active) >= n_active_max:
            break

    if g is not None and g.sum() > 1e-10:
        return g / g.sum()
    return None


def run_lp_search(a_init, deadline, n_active_start=20, n_active_max=80):
    """Run LP best-response search with line search."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(500):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        current_M = b.max()

        g = lp_best_response_cutting_plane(a, current_M,
                                           n_active_start=n_active_start,
                                           n_active_max=n_active_max,
                                           max_rounds=4,
                                           deadline=deadline - 0.5)
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-12):
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    return best_a, best_M


def coordinate_optimize(a, b, i, n):
    """Find optimal a[i] given others fixed. Returns optimal value."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Precompute c and slope for each affected k
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    # Golden section search
    lo = 0.0
    hi = max(2.0 * x_cur + 0.01, 0.5 / n)

    # Expand hi if needed
    for _ in range(5):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 10.0 / n:
            hi *= 2.0
        else:
            break

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coordinate_descent(a_init, deadline, max_sweeps=200, seed=0):
    """Run coordinate descent with incremental convolution updates."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)

        b = fast_conv(a)
        b_max = b.max()
        sweep_improved = False

        for i in order:
            i = int(i)
            x_opt = coordinate_optimize(a, b, i, n)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                a[i] = x_opt
                # Update affected b values
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = b[k] - old * old + x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] = b[k] - 2.0 * old * a[j] + 2.0 * x_opt * a[j]
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 5
        if no_improve > 30:
            break

    return best_a, best_M


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        # Resample parent to length n
        if parent.size != n:
            t_old = np.linspace(0, 1, parent.size)
            t_new = np.linspace(0, 1, n)
            p = np.interp(t_new, t_old, parent)
        else:
            p = parent.copy()
        p = np.maximum(p, 0.0)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed parent
        p1 = np.maximum(p + rng.normal(0, 0.02, n), 1e-6)
        inits.append(p1 / p1.sum())
        p2 = np.maximum(p + rng.normal(0, 0.08, n), 1e-6)
        inits.append(p2 / p2.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Parabolic (concave)
    poly = t * (1 - t) + 0.1
    inits.append(poly / poly.sum())

    # U-shaped (convex)
    poly2 = (2 * t - 1) ** 2 + 0.1
    inits.append(poly2 / poly2.sum())

    # Cosine windows
    for freq in [1, 2, 3]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())

    # Gaussian-like
    gauss = np.exp(-5.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    # Smooth random
    ksize = max(3, n // 15)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_full_pipeline(a0, n, deadline, seed=0):

    best = a0.copy()
    best_M = fast_conv(a0).max()
    elapsed_total = deadline - time.monotonic()

        if M_g < best_M:
            best = a_g
            best_M = M_g
        print(f"    After grad: M={best_M:.12f}", flush=True)

    # Phase B: LP search (30% of time)
    t_b = time.monotonic() + 0.3 * (deadline - time.monotonic())
    if time.monotonic() < t_b and time.monotonic() < deadline - 5:
        a_lp, M_lp = run_lp_search(best, min(t_b, deadline - 5),
                                   n_active_start=20, n_active_max=60)
        if M_lp < best_M:
            best = a_lp
            best_M = M_lp
        print(f"    After LP: M={best_M:.12f}", flush=True)

    # Phase C: Coordinate descent (20% of time)
    t_c = time.monotonic() + 0.2 * (deadline - time.monotonic())
    if time.monotonic() < t_c and time.monotonic() < deadline - 3:
        a_cd, M_cd = run_coordinate_descent(best, min(t_c, deadline - 3),
                                            max_sweeps=100, seed=seed + 1)
        if M_cd < best_M:
            best = a_cd
            best_M = M_cd
        print(f"    After CD: M={best_M:.12f}", flush=True)

    # Phase D: Final LP + CD alternating (10% of time)
    if time.monotonic() < deadline - 2:
        remaining = deadline - 2 - time.monotonic()
        # One more LP pass
        if remaining > 10:
            a_lp2, M_lp2 = run_lp_search(best, time.monotonic() + min(remaining * 0.5, 30),
                                         n_active_start=30, n_active_max=80)
            if M_lp2 < best_M:
                best = a_lp2
                best_M = M_lp2
        # One more CD pass
        if time.monotonic() < deadline - 2:
            a_cd2, M_cd2 = run_coordinate_descent(best, deadline - 2,
                                                  max_sweeps=50, seed=seed + 2)
            if M_cd2 < best_M:
                best = a_cd2
                best_M = M_cd2

    return best, best_M


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: fast top-k gradient -> LP cutting-plane -> coordinate descent -> final polish.
    Tries multiple initializations and lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 40.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size
    print(f"\n{'='*60}", flush=True)
    print(f"Starting optimization. Parent J={tracker.best_score:.8f}, n={n_parent}", flush=True)
    print(f"Deadline in {deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    # Determine lengths to try
    lengths_to_try = [n_parent]
    for candidate_n in [512, 768, 1024, 1280]:
        if candidate_n not in lengths_to_try and 256 <= candidate_n <= 2048:
            lengths_to_try.append(candidate_n)

    # Sort: parent first, then by distance from parent
    lengths_to_try.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    lengths_to_try = lengths_to_try[:4]  # At most 4 lengths

    n_lengths = len(lengths_to_try)
    # Time allocation: 60% to parent length, rest split
    time_per_len = {}
    time_budget = deadline - time.monotonic() - 100  # reserve 100s for final phases
    if n_lengths == 1:
        time_per_len[lengths_to_try[0]] = time_budget
    else:
        # Parent gets 50%, others split the rest
        time_per_len[lengths_to_try[0]] = 0.5 * time_budget
        remaining_time = 0.5 * time_budget
        for nl in lengths_to_try[1:]:
            time_per_len[nl] = remaining_time / (n_lengths - 1)

    # Phase 1: Multi-length optimization
    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - 100:
            break

        t_alloc = time_per_len[n]
        t_alloc = min(t_alloc, deadline - 100 - time.monotonic())
        if t_alloc < 10:
            break

        print(f"\n  Length n={n}, time={t_alloc:.0f}s", flush=True)

        # Generate inits
        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        # Limit number of inits based on available time
        time_per_init = t_alloc / max(1, len(inits))
        n_inits = min(len(inits), max(2, int(t_alloc / 20)))

        best_for_n = None
        best_M_for_n = float('inf')

        for ii in range(n_inits):
            if time.monotonic() > deadline - 100:
                break
            sub_deadline = min(time.monotonic() + time_per_init, deadline - 100)
            if sub_deadline - time.monotonic() < 5:
                break

            a0 = inits[ii]
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            a_opt, M_opt = run_full_pipeline(a0, n, sub_deadline, seed=seed + li * 100 + ii)

            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best M={best_M_for_n:.12f} J={J_val:.6f}", flush=True)
            tracker.update(best_for_n)

    tracker.report("After Phase 1 (multi-length)")

    # Phase 2: Intensive LP on best
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print(f"\n  Phase 2: Intensive LP polish", flush=True)
        t_lp = min(deadline - 20, time.monotonic() + 120)
        a_lp, M_lp = run_lp_search(tracker.best, t_lp,
                                   n_active_start=25, n_active_max=100)
        tracker.update(a_lp)

        # Second LP pass with different params
        if time.monotonic() < deadline - 30:
            t_lp2 = min(deadline - 15, time.monotonic() + 60)
            a_lp2, M_lp2 = run_lp_search(tracker.best, t_lp2,
                                         n_active_start=40, n_active_max=120)
            tracker.update(a_lp2)

    tracker.report("After Phase 2 (LP polish)")

    # Phase 3: Coordinate descent polish
    if tracker.best is not None and time.monotonic() < deadline - 20:
        print(f"\n  Phase 3: Coordinate descent polish", flush=True)
        t_cd = min(deadline - 10, time.monotonic() + 100)
        a_cd, M_cd = run_coordinate_descent(tracker.best, t_cd,
                                            max_sweeps=200, seed=99999)
        tracker.update(a_cd)

    tracker.report("After Phase 3 (CD polish)")

    # Phase 4: Perturbation + re-optimization
    if tracker.best is not None and time.monotonic() < deadline - 20:
        print(f"\n  Phase 4: Perturbation exploration", flush=True)
        n_best = tracker.best.size
        for trial in range(5):
            if time.monotonic() > deadline - 20:
                break
            noise_scale = 0.02 + 0.03 * trial
            perturbed = tracker.best + rng.normal(0, noise_scale, n_best)
            perturbed = np.maximum(perturbed, 1e-6)
            perturbed = perturbed / perturbed.sum()

            sub_dl = min(deadline - 15, time.monotonic() + 20)
            a_opt, M_opt = run_full_pipeline(perturbed, n_best, sub_dl,
                                             seed=50000 + trial)
            tracker.update(a_opt)

    tracker.report("After Phase 4 (perturbation)")

    # Phase 5: Final fine-tuning
    if tracker.best is not None and time.monotonic() < deadline - 5:
        print(f"\n  Phase 5: Final fine-tuning", flush=True)
        # Quick gradient with high k
        t_fin = min(deadline - 3, time.monotonic() + 40)
        a_fin, M_fin = run_topk_gradient(tracker.best, t_fin,
                                         k_values=[50, 100], seed=77777)
        tracker.update(a_fin)

        # Quick LP
        if time.monotonic() < deadline - 3:
            a_fin2, M_fin2 = run_lp_search(tracker.best, deadline - 3,
                                           n_active_start=30, n_active_max=80)
            tracker.update(a_fin2)

    tracker.report("After Phase 5 (final)")

    # Final safety: if we have best, return it
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
