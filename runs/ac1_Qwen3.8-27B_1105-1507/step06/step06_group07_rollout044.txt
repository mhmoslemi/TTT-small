```python
        #      = (2/|P|) * sum_{k in P} a[k-i] * 1[0<=k-i<n]
        
        # For each k in P, we add a shifted version of a (shifted by k) to g, but only where valid.
        # a[k-i] for i in [0,n-1] and 0<=k-i<n means i in [max(0,k-n+1), min(k, n-1)]
        # And a[k-i] = a[k-i], so for i in that range, g[i] += a[k-i]
        # Which is: g[i] += a[k-i], i.e., g[k-i] += a[i] for i in valid range
        # This is just a shifted copy of a.
        
        num_peaks = len(peaks)
        g = np.zeros(n)
        for k in peaks:
            # a[k-i] for i such that 0 <= i < n and 0 <= k-i < n
            # i.e., i in [max(0, k-n+1), min(k, n-1)]
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_start <= i_end:
                g[i_start:i_end+1] += a[k - i_end:k - i_start + 1][::-1]
        
        g *= 2.0 / num_peaks
        
        # Full subgradient of objective = 2*n*M/S^2 * g - 2*n*M/S^3 (since dS/da[i] = 1)
        # But we're working with S = 1 (normalized), so:
        # = 2*n * g - 2*n * max_b
        
        # Actually, let me just use the gradient of the log-sum-exp smoothed version
        # for a smoother optimization, and periodically check the true objective.
        
        # For now, use the subgradient:
        grad = g - max_b  # simplified (the 2n factor and 1/S^2 factor cancel in direction)
        
        # Update in log-space: grad_x[i] = grad[i] * a[i]
        grad_x = grad * a
        
        # Momentum update
        v = beta_m * v - lr * grad_x
        x = x + v
        
        # Periodic centering
        if it % 100 == 0:
            x -= np.mean(x)
        
        # Check true objective
        if it % 50 == 0:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            obj = compute_obj(a_check)
            if obj < best_obj:
                best_obj = obj
                best_a = a_check.copy()
        
        # Decaying learning rate
        lr *= lr_decay
    
    return best_a
```

```python
def proj_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```
