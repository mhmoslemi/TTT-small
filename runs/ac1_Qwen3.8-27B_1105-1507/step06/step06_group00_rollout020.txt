def compute_conv(a):

    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):

    if s < 0.01:

    a_norm = a / s
    conv = compute_conv(a_norm)

Actually, let me reconsider the problem from a high level. The objective is:

f(a) = 2n * max_k (a*a)[k] / (sum a)^2

By homogeneity, set sum(a) = 1:
f(a) = 2n * max_k (a*a)[k]

We want to minimize this. The LP alternating approach:
1. Given a (fixed), solve: min M s.t. conv(a, g)_k <= M, sum(g)=1, g>=0
2. Update a ← g (or blend)
3. Repeat

The LP in step 1 finds the g that "best matches" a in the sense of minimizing the peak cross-correlation.

A key insight: at the optimal solution, we have a = g (the sequence is a fixed point of the LP map). At this point, the LP is:
min M s.t. conv(a, a)_k <= M, sum(a)=1, a>=0

And M* = max_k (a*a)[k] is the optimal value.

So we're looking for a fixed point of the map T(a) = argmin{M : conv(a,T(a)) <= M, sum(T(a))=1, T(a)>=0}.

This is a non-trivial fixed point problem. The convergence of the iteration a_{t+1} = T(a_t) is not guaranteed (it's not a contraction in general).

To improve convergence:
1. **Averaging**: a_avg = (1/T) sum_{t=1}^T a_t converges to a fixed point (under mild conditions)
2. **Damping**: a_{t+1} = (1-alpha)*a_t + alpha*T(a_t) with alpha < 1
3. **Momentum**: v_{t+1} = beta*v_t + (T(a_t) - a_t); a_{t+1} = a_t + v_{t+1}

I think averaging is the most robust. Let me use that.

def lp_alternating(a, time_budget, use_average=True):
    """LP alternating with optional averaging of iterates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    avg = a.copy()
    start = time.monotonic()
    count = 0
    
    while True:
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_minpeak_lp(a)
        if g_lp is None:
            break
        
        count += 1
        
        # Update running average
        if use_average:
            avg = avg * (count - 1) / count + g_lp / count
            avg = np.maximum(avg, 0)
            sa = float(np.sum(avg))
            if sa > 1e-10:
                avg = avg / sa
        
        # Use g_lp as next iterate
        a = g_lp.copy()
        
        # Check objective
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    # Check averaged solution
    if use_average:
        avg_obj = compute_obj(avg)
        if avg_obj < best_obj:
            best_obj = avg_obj
            best_a = avg.copy()
    
    return best_a


def subgradient_refine(a, time_budget, step_size=0.01):
    """Subgradient descent on the non-convex objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = step_size
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: g[j] = 2*a[k_star - j] (with bounds)
        grad = np.zeros(n)
        for j in range(n):
            m = k_star - j
            if 0 <= m < n:
                grad[j] = 2.0 * a[m]
        
        # Center the gradient (project onto sum-preserving direction)
        grad = grad - np.mean(grad)
        
        # Take step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
            else:
                step *= 0.9
        else:
            step *= 0.5
    
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:

    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

Step function optimization: minimizes 2*n*max(conv(a,a))/(sum(a))^2.

from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_minpeak_lp(a):
    n = len(a)
    if n <= 0:
        return None, None
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None
