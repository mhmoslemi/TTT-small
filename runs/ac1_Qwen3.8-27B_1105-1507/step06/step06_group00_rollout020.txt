def compute_conv(a):
    n = len(a)

    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def soft_max_conv(a, T):
    """Compute soft-max approximation of max(conv(a,a))."""
    conv = compute_conv(a)
    # Scale to avoid overflow
    conv_scaled = conv / T
    max_c = np.max(conv_scaled)
    return T * (max_c + np.log(np.sum(np.exp(conv_scaled - max_c))))

def softmax_parameterize(z):
    """Convert z to a probability vector via softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)

def neg_smooth_obj(z, T):
    """Negative smooth objective for minimization."""
    a = softmax_parameterize(z)
    return -soft_max_conv(a, T)

"""
Combined optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max L-BFGS with adaptive temperature, FW-style gradient steps,
LP alternating, multi-n exploration, and diverse initializations.
Key improvements over parent: smooth gradient-based optimization via softmax
parameterization, adaptive soft-max temperature schedule, and more aggressive
multi-start exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_vec(z):
    """Stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def soft_max_conv_value(a, T):
    """Compute soft-max approximation of max(conv(a,a))."""
    conv = compute_conv(a)
    conv_s = conv / T
    m = np.max(conv_s)
    return T * (m + np.log(np.sum(np.exp(conv_s - m))))


def soft_max_conv_grad(z, T):

    Returns (value, grad_z).

    conv = compute_conv(a)
    conv_s = conv / T
    m = np.max(conv_s)
    p = np.exp(conv_s - m)
    p /= p.sum()

    # Let j = k-i, so k = i+j, j in [0, n-1], k in [0, 2n-2]
    # g[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j] (where i+j < 2n-1)
    # This is 2 * (p cross-correlated with a)[i] = 2 * (a * p_rev)[n-1-i]... 
    # Let me just use the direct FFT approach.

    # Define d[m] = sum_j a[j] * p[m-j] ... that's just conv(a, p) but p has length 2n-1
    # Actually g[i]/2 = sum_j a[j] * p[i+j]

    # This is the "anti-convolution" or cross-correlation.
    # g[i]/2 = (a corr q)[i] where (a corr q)[i] = sum_j a[j] q[i+j]

def softmax_lbfgs(a_init, T_schedule, max_iters=200, time_budget=120.0):
    """
    Run L-BFGS on softmax parameterization with adaptive temperature.
    a_init: initial point (normalized, non-negative).
    T_schedule: list of temperatures to use, in decreasing order.
    """
    n = len(a_init)
    start = time.monotonic()
    
    # Initialize z from a_init
    a_safe = np.maximum(a_init, 1e-15)
    a_safe = a_safe / a_safe.sum()
    z = np.log(a_safe)
    
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    
    for T in T_schedule:
        if time.monotonic() - start > time_budget:
            break
        
        def func_grad(z_curr):
            val, grad = soft_max_conv_grad(z_curr, T)
            return val, grad
        
        try:
            res = minimize(func_grad, z, jac=True, method='L-BFGS-B',
                          options={'maxiter': max_iters, 'gtol': 1e-6,
                                   'maxls': 50, 'maxfun': max_iters * 10})
            if res.x is not None:
                a_new = softmax_vec(res.x)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    z = res.x.copy()
        except Exception:
            pass
    
    return best_a, best_obj


def fw_refine(a_init, n_steps=500, time_budget=60.0):
    """
    Frank-Wolfe style refinement.
    At each step, find the subgradient direction and the FW vertex,
    then take a step towards the vertex.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init, compute_obj(a_init)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: grad[i] = 2*a[k_star - i] if 0 <= k_star-i < n
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # FW vertex: argmin <grad, v> over simplex = e_{argmin grad}
        i_min = int(np.argmin(grad))
        
        # FW step size: use 2/(it+2) (standard FW rate) or adaptive
        t_fw = 2.0 / (it + 2)
        
        # Also try a few manual step sizes
        best_t = t_fw
        best_obj_t = np.inf
        for t_try in [t_fw, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t_try) * a + t_try * np.eye(n)[i_min]
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t_try
        
        if best_obj_t < best_obj - 1e-12:
            best_obj = best_obj_t
            a_new = (1.0 - best_t) * a + best_t * np.eye(n)[i_min]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a = a_new / s_new
                best_a = a.copy()
    
    return best_a, best_obj


def chase_peak_subgradient(a_init, n_steps=1000, time_budget=60.0):
    """
    Subgradient descent on the exact (non-smooth) objective.
    Uses the subgradient at the current peak and projects onto the simplex.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init, compute_obj(a_init)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.02
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of max_conv: 2*a[k_star - i]
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Center the gradient (projection onto tangent space of simplex)
        grad_centered = grad - np.mean(grad)
        
        # Take step with backtracking
        for factor in [1.0, 0.5, 0.25, 0.1]:
            a_new = a - factor * step_size * grad_centered
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    break
                else:
                    continue
        else:
            step_size *= 0.9
        
        # Periodic annealing: occasionally try a random perturbation
        if it % 100 == 99:
            rng_local = np.random.default_rng(it)
            a_perturbed = a * np.exp(rng_local.normal(0, 0.05, n))
            a_perturbed = np.maximum(a_perturbed, 0)
            s_p = float(np.sum(a_perturbed))
            if s_p > 1e-10:
                a_perturbed = a_perturbed / s_p
                obj_p = compute_obj(a_perturbed)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_perturbed.copy()
                    a = a_perturbed
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Use scipy linprog (HiGHS) - most reliable
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_alternating(a, time_budget=60.0):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    max_iters = 300
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 30:
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                scale = 0.03
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a = a_pert / s_p
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                stall = 0
    
    return best_a, best_obj


def multi_peak_fw(a_init, n_steps=500, time_budget=60.0):
    """
    Improved FW: instead of just the single peak, consider the top-K peaks
    and use a weighted subgradient.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init, compute_obj(a_init)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    K = min(5, n)  # number of peaks to consider
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        
        # Find top K peaks
        # Use soft-max to get weights
        T_soft = 0.05 * (np.max(conv) - np.min(conv) + 1e-10)
        conv_s = conv / T_soft
        m = np.max(conv_s)
        w = np.exp(conv_s - m)
        w /= w.sum()
        
        # Weighted subgradient
        grad = np.zeros(n)
        for i in range(n):
            for k in range(len(conv)):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * w[k] * a[j]
        
        # Actually this is O(n * (2n-1)) which is O(n^2) - too slow for large n
        # Use FFT instead:
        # grad[i] = 2 * sum_k w[k] * a[k-i] * I(0<=k-i<n)
        # = 2 * (w * flip(a))[i + n - 1]
        r_a = a[::-1]
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fw = rfft(w, size)
        fa_r = rfft(r_a, size)
        conv_w_a = irfft(fw * fa_r, size)
        grad = 2.0 * conv_w_a[n - 1: n - 1 + n]
        
        # FW vertex
        i_min = int(np.argmin(grad))
        
        # Adaptive step size
        best_t = 2.0 / (it + 2)
        best_obj_t = np.inf
        for t_try in [best_t, 0.3, 0.5, 0.7]:
            a_try = (1.0 - t_try) * a + t_try * np.eye(n)[i_min]
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t_try
        
        if best_obj_t < best_obj - 1e-12:
            best_obj = best_obj_t
            a = (1.0 - best_t) * a + best_t * np.eye(n)[i_min]
            a = np.maximum(a, 0)
            s_new = float(np.sum(a))
            if s_new > 1e-10:
                a = a / s_new
                best_a = a.copy()
    
    return best_a, best_obj


def smooth_init(n, mode=0, rng=None):
    """Generate a smooth initial sequence."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    if mode == 0:
        # Uniform
        a = np.ones(n) / n
    elif mode == 1:
        # Cosine bump
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + np.cos(x)
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 2:
        # Gaussian bump
        center = n // 2
        sigma = n / 4.0
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * sigma ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine squared
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2
        a = a + 0.1  # avoid zeros
        a = a / a.sum()
    elif mode == 4:
        # Two-hump
        c1, c2 = n // 3, 2 * n // 3
        sig = n / 8.0
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * sig ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * sig ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Random smooth (low-pass filtered random)
        a_raw = rng.uniform(0, 1, n)
        # Simple moving average
        k = max(3, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a_raw, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        # Power-of-cosine with random exponent
        p = rng.uniform(1, 5)
        x = np.linspace(0, np.pi, n)
        a = (np.cos(x) ** 2) ** p + 0.05
        a = a / a.sum()
    
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def paired_swap_refine(a_init, time_budget=30.0):
    """
    Local refinement: try swapping/perturbing pairs of elements that
    contribute to the current peak.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init, compute_obj(a_init)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find the top contributors to conv[k_star]
        # contrib[i] = a[i] * a[k_star - i]
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((i, j, a[i] * a[j]))
        
        if not contribs:
            break
        
        contribs.sort(key=lambda x: -x[2])
        
        # Try to reduce the top contributor by moving mass
        i_hi, j_hi, c_hi = contribs[0]
        
        # Find a position far from the peak to move mass to
        # Positions that don't contribute to conv[k_star]
        non_contrib = [i for i in range(n) if not (0 <= k_star - i < n)]
        if not non_contrib:
            non_contrib = list(range(n))
        
        # Pick a random non-contributing position
        target = rng.choice(non_contrib)
        
        # Try moving a small fraction of mass from i_hi to target
        for frac in [0.01, 0.03, 0.05, 0.1]:
            delta = frac * a[i_hi]
            if delta < 1e-10:
                continue
            a[i_hi] -= delta
            a[target] += delta
            a = np.maximum(a, 0)
            s_new = float(np.sum(a))
            if s_new > 1e-10:
                a = a / s_new
                obj_new = compute_obj(a)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    break
                else:
                    # Revert
                    a[i_hi] += delta
                    a[target] -= delta
                    a = np.maximum(a, 0)
                    s_r = float(np.sum(a))
                    if s_r > 1e-10:
                        a = a / s_r
                    break
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):

    rng = np.random.default_rng(seed)

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = smooth_init(800, mode=1)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800, cosine): obj={best_score:.8f}")

    print(f"\n=== Phase 1: Soft-max L-BFGS ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_cur = len(best_seq)
        # Temperature schedule: start wide, narrow down
        T_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        
        # Scale T relative to typical conv magnitude

        conv_scale = float(np.max(conv_init))

        a_lbfgs, obj_lbfgs = softmax_lbfgs(

        if obj_lbfgs < best_score:

            best_seq = a_lbfgs.copy()

            if obj_mfw < best_score:
                best_score = obj_mfw

                print(f"  Multi-peak FW: obj={best_score:.8f}")

    print(f"\n=== Phase 3: Chase Peak ===")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()

        if obj_opt < best_score:
