"""
Optimization of non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent (multiplicative weight updates), frequency-domain 
Gerchberg-Saxton iterations, LP alternating, and multi-restart perturbation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1} using simplex projection."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return a
    # Use the algorithm for Euclidean projection onto the probability simplex
    if np.sum(a) <= 1e-15:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(a - theta, 0)
    s = proj.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return proj / s

            best_a = a.copy()
        else:
            # Reduce eta if no improvement
            eta *= 0.95

def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(a)
    if n == 0:
        return a

    if np.all(a < 0):

        # Line search over candidate blend factors
        ts = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        best_t_obj = np.inf
        best_t_val = 0.3
        for t in ts:
            a_try = (1.0 - t) * a + t * g
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t

        if best_t_obj < best_obj - 1e-9:
            a = (1.0 - best_t_val) * a + best_t_val * g
            a = project_simplex(a)
            best_obj = best_t_obj
            best_a = a.copy()
            stall = 0
            if verbose and (it % 10 == 0 or it < 5):
                print(f"      LP it{it}: obj={best_obj:.7f}, t={best_t_val:.3f}")
        else:
            stall += 1

    return best_a, best_obj


def mirror_descent(a, max_iters=3000, time_budget=60.0, top_k=3,
                   eta_init=0.8, eta_decay=0.9995, momentum=0.6,
                   verbose=False):
    """Mirror descent (multiplicative weight) with top-K subgradient."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    eta = eta_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Compute top-K subgradient
        if top_k <= 1:
            k_star = int(np.argmax(conv))
            g = subgradient_at(a, k_star)
        else:
            top_idx = np.argsort(conv)[::-1][:top_k]
            threshold = cmax * 0.80
            valid_idx = [int(k) for k in top_idx if conv[int(k)] >= threshold]
            if not valid_idx:
                valid_idx = [int(top_idx[0])]
            g = np.zeros(n)
            for k_star in valid_idx:
                g += subgradient_at(a, k_star)
            g /= len(valid_idx)

        # Center gradient to preserve sum
        g = g - np.mean(g)

        # Momentum
        v = momentum * v + (1.0 - momentum) * g

        # Multiplicative (mirror) update
        exponent = -eta * v
        exponent = np.clip(exponent, -40.0, 40.0)
        a_new = a * np.exp(exponent)
        s = a_new.sum()
        if s < 1e-300:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"        MD it{it}: obj={best_obj:.7f}")
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                eta *= 0.85

        eta = max(0.0005, eta_init * (eta_decay ** it))

    return best_a, best_obj


def freq_domain_gs(a, max_iters=300, time_budget=40.0, alpha_vals=None,
                   verbose=False):
    """
    Frequency-domain Gerchberg-Saxton: iteratively flatten the power spectrum
    while maintaining non-negativity in time domain.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if alpha_vals is None:
        alpha_vals = [0.6, 0.7, 0.8, 0.9]

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for alpha in alpha_vals:
        if time.monotonic() - start > time_budget:
            break
        a_cur = best_a.copy()
        for it in range(max_iters):
            if time.monotonic() - start > time_budget:
                break

            fa = np.fft.fft(a_cur, N)
            mag = np.abs(fa)
            phase = np.angle(fa)

            # Compress magnitude
            mag_max = np.max(mag)
            if mag_max < 1e-15:
                break
            mag_norm = mag / mag_max
            mag_new = mag_norm ** alpha * mag_max

            # Reconstruct
            fa_new = mag_new * np.exp(1j * phase)
            a_candidate = np.fft.ifft(fa_new).real[:n]

            # Project to non-negative
            a_candidate = np.maximum(a_candidate, 0)
            s = a_candidate.sum()
            if s < 1e-15:
                break
            a_candidate = a_candidate / s

            obj = compute_obj(a_candidate)
            if obj < best_obj:
                best_obj = obj
                best_a = a_candidate.copy()
                a_cur = a_candidate.copy()
                if verbose:
                    print(f"          GS alpha={alpha:.1f} it{it}: obj={best_obj:.7f}")

    return best_a, best_obj


def chase_peak_fast(a, n_steps=30, step_size=0.02):
    """Fast chase-peak: subgradient on the max position only."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        g = subgradient_at(a, k_star)
        g = g - np.mean(g)

        a_try = a - step_size * g
        a_try = project_simplex(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()

    return best_a, best_obj


def destroy_and_repair(a, rng, destroy_frac=0.2, n_fill_frac=0.4):
    """Destroy random positions and repair with structured fill."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    n_destroy = max(10, int(n * destroy_frac))
    destroy_idx = rng.choice(n, size=n_destroy, replace=False)
    a_d = a.copy()
    a_d[destroy_idx] = 0
    mass = a_d.sum()

    # Repair: fill at random positions with Dirichlet weights
    n_fill = max(20, int(n * n_fill_frac))
    fill_idx = rng.choice(n, size=n_fill, replace=False)
    weights = rng.dirichlet(np.ones(n_fill) * 1.5)
    a_r = np.zeros(n)
    a_r[fill_idx] = weights * mass
    a_r = project_simplex(a_r)
    return a_r


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.all(np.isfinite(parent)):
            return parent.copy()
    except Exception:
        pass
    return None


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    x = np.linspace(0, 1, n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5))) + 0.1
    inits.append(rc / rc.sum())

    # Gaussian
    x = np.linspace(-3, 3, n)
    g = np.exp(-x ** 2 / 3)
    inits.append(g / g.sum())

    # Two-peak
    p1 = int(n * 0.25)
    p2 = int(n * 0.75)
    tp = np.zeros(n)
    w = max(5, n // 20)
    tp[max(0, p1 - w):p1 + w + 1] = 1.0
    tp[max(0, p2 - w):p2 + w + 1] = 1.0
    inits.append(tp / tp.sum())

    # Random
    rnd = rng.random(n) + 0.05
    inits.append(rnd / rnd.sum())

    # Sinusoidal
    sine = np.sin(np.pi * x) ** 2 + 0.2
    inits.append(sine / sine.sum())

    if parent is not None:
        ps = resample_sequence(parent, n)
        inits.append(ps)
        pert = ps * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_simplex(pert))

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining LP alternating, mirror
    descent, frequency-domain GS, and multi-restart to minimize
    2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent = project_simplex(parent)
        parent_obj = compute_obj(parent)
        print(f"Parent (n={len(parent)}): obj={parent_obj:.7f}")
        best_seq = parent.copy()
        best_score = parent_obj
    else:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"No parent, starting uniform n={len(best_seq)}: obj={best_score:.7f}")

    n_best = len(best_seq)
    print(f"Initial best: n={n_best}, score={best_score:.8f}")

    # ===== PHASE 1: LP Alternating (40% budget) =====
    print("\n=== Phase 1: LP Alternating ===")
    p1_budget = min(500.0, (deadline - start_time) * 0.35)
    p1_start = time.monotonic()

    a_work = project_simplex(best_seq.copy())

    # Run LP alternating on the main sequence
    a_lp, obj_lp = lp_alternating(a_work, max_iters=60,
                                  time_budget=p1_budget * 0.7, verbose=True)
    if obj_lp < best_score:
        best_score = obj_lp
        best_seq = a_lp.copy()
        print(f"  LP improved: {best_score:.8f}")

    # Try on a slightly perturbed version
    remaining_p1 = p1_budget - (time.monotonic() - p1_start)
    if remaining_p1 > 30:
        a_pert = best_seq * np.exp(rng.normal(0, 0.03, n_best))
        a_pert = project_simplex(a_pert)
        a_lp2, obj_lp2 = lp_alternating(a_pert, max_iters=30,
                                        time_budget=min(remaining_p1 * 0.6, 80.0),
                                        verbose=False)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            n_best = len(best_seq)
            print(f"  LP perturbed improved: {best_score:.8f}")

    print(f"  Phase 1 best: {best_score:.8f}, n={len(best_seq)}")

    # ===== PHASE 2: Mirror Descent (25% budget) =====
    print("\n=== Phase 2: Mirror Descent ===")
    p2_budget = min(350.0, (deadline - time.monotonic()) * 0.25)
    p2_start = time.monotonic()

    a_work = project_simplex(best_seq.copy())
    n_work = len(a_work)

    # Multiple MD configurations
    md_configs = [
        {'top_k': 1, 'eta_init': 0.5, 'momentum': 0.5},
        {'top_k': 3, 'eta_init': 0.3, 'momentum': 0.6},
        {'top_k': 5, 'eta_init': 0.2, 'momentum': 0.7},
        {'top_k': 1, 'eta_init': 1.0, 'momentum': 0.3},
    ]

    per_md_budget = p2_budget / len(md_configs)
    for ci, cfg in enumerate(md_configs):
        if time.monotonic() > p2_start + p2_budget:
            break
        a_md, obj_md = mirror_descent(a_work, max_iters=2000,
                                      time_budget=per_md_budget,
                                      top_k=cfg['top_k'],
                                      eta_init=cfg['eta_init'],
                                      momentum=cfg['momentum'])
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            n_best = len(best_seq)
            a_work = a_md.copy()
            print(f"  MD config {ci} improved: {best_score:.8f}")

    # Additional MD runs with different starting points
    remaining_p2 = p2_budget - (time.monotonic() - p2_start)
    if remaining_p2 > 30:
        for attempt in range(3):
            if time.monotonic() > p2_start + p2_budget:
                break
            a_start = best_seq * np.exp(rng.normal(0, 0.1, n_best))
            a_start = project_simplex(a_start)
            a_md2, obj_md2 = mirror_descent(a_start, max_iters=1500,
                                            time_budget=min(remaining_p2 * 0.3, 40.0),
                                            top_k=3, eta_init=0.3, momentum=0.6)
            if obj_md2 < best_score:
                best_score = obj_md2
                best_seq = a_md2.copy()
                n_best = len(best_seq)
                print(f"  MD restart {attempt} improved: {best_score:.8f}")
            remaining_p2 = p2_budget - (time.monotonic() - p2_start)

    print(f"  Phase 2 best: {best_score:.8f}, n={len(best_seq)}")

    # ===== PHASE 3: Frequency Domain GS (15% budget) =====
    print("\n=== Phase 3: Frequency Domain GS ===")
    p3_budget = min(200.0, (deadline - time.monotonic()) * 0.15)
    p3_start = time.monotonic()

    a_work = project_simplex(best_seq.copy())
    a_gs, obj_gs = freq_domain_gs(a_work, max_iters=200,
                                  time_budget=p3_budget * 0.6,
                                  alpha_vals=[0.5, 0.6, 0.7, 0.8],
                                  verbose=True)
    if obj_gs < best_score:
        best_score = obj_gs
        best_seq = a_gs.copy()
        n_best = len(best_seq)
        print(f"  GS improved: {best_score:.8f}")

    # GS from perturbed start
    remaining_p3 = p3_budget - (time.monotonic() - p3_start)
    if remaining_p3 > 20:
        a_pert = best_seq * np.exp(rng.normal(0, 0.1, n_best))
        a_pert = project_simplex(a_pert)
        a_gs2, obj_gs2 = freq_domain_gs(a_pert, max_iters=150,
                                        time_budget=min(remaining_p3 * 0.5, 40.0),
                                        alpha_vals=[0.55, 0.65, 0.75])
        if obj_gs2 < best_score:
            best_score = obj_gs2
            best_seq = a_gs2.copy()
            n_best = len(best_seq)
            print(f"  GS perturbed improved: {best_score:.8f}")

    print(f"  Phase 3 best: {best_score:.8f}, n={len(best_seq)}")

    # ===== PHASE 4: LP + MD Combined Polish (20% budget) =====
    print("\n=== Phase 4: Combined Polish ===")
    p4_budget = min(250.0, (deadline - time.monotonic()) * 0.20)
    p4_start = time.monotonic()

    a_work = project_simplex(best_seq.copy())
    n_work = len(a_work)

    for round_idx in range(8):
        if time.monotonic() > p4_start + p4_budget:
            break
        remaining = p4_start + p4_budget - time.monotonic()
        if remaining < 15:
            break

        # LP step
        if n_work <= 1500:
            a_lp3, obj_lp3 = lp_alternating(a_work, max_iters=15,
                                            time_budget=min(remaining * 0.35, 30.0),
                                            verbose=False)
            if obj_lp3 < best_score:
                best_score = obj_lp3
                best_seq = a_lp3.copy()
                a_work = a_lp3.copy()
                n_work = len(a_work)
                print(f"    Polish LP r{round_idx}: {best_score:.8f}")

        # Mirror descent step
        remaining = p4_start + p4_budget - time.monotonic()
        if remaining > 10:
            a_md3, obj_md3 = mirror_descent(a_work, max_iters=800,
                                            time_budget=min(remaining * 0.4, 30.0),
                                            top_k=3, eta_init=0.2, momentum=0.6)
            if obj_md3 < best_score:
                best_score = obj_md3
                best_seq = a_md3.copy()
                a_work = a_md3.copy()
                n_work = len(a_work)
                print(f"    Polish MD r{round_idx}: {best_score:.8f}")

        # Chase peak
        remaining = p4_start + p4_budget - time.monotonic()
        if remaining > 5:
            a_cp, obj_cp = chase_peak_fast(a_work, n_steps=20, step_size=0.015)
            if obj_cp < best_score:
                best_score = obj_cp
                best_seq = a_cp.copy()
                a_work = a_cp.copy()
                n_work = len(a_work)
                print(f"    Polish CP r{round_idx}: {best_score:.8f}")

    print(f"  Phase 4 best: {best_score:.8f}, n={len(best_seq)}")

    # ===== PHASE 5: Multi-Restart (remaining time) =====
    print("\n=== Phase 5: Multi-Restart ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p5_budget = remaining * 0.7
        p5_start = time.monotonic()
        n_best = len(best_seq)

        n_restarts = min(10, int(p5_budget / 25))
        for ri in range(n_restarts):
            if time.monotonic() > p5_start + p5_budget:
                break
            per_restart = min(25.0, p5_budget / max(1, n_restarts - ri))

            # Different perturbation scales
            scale = 0.02 + 0.03 * ri
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_pert = project_simplex(a_pert)

            # Quick optimization
            a_r, obj_r = mirror_descent(a_pert, max_iters=500,
                                        time_budget=per_restart * 0.5,
                                        top_k=3, eta_init=0.3, momentum=0.6)
            if n_best <= 1200:
                a_r2, obj_r2 = lp_alternating(a_r, max_iters=5,
                                              time_budget=per_restart * 0.3,
                                              verbose=False)
                if obj_r2 < obj_r:
                    a_r, obj_r = a_r2, obj_r2

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                n_best = len(best_seq)
                print(f"  Restart {ri} (scale={scale:.3f}): {best_score:.8f}")

    # ===== Phase 6: Try Different Lengths =====
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Length Exploration (remaining {remaining:.0f}s) ===")
        n_cur = len(best_seq)
        length_tries = []
        for dl in [-100, -50, 50, 100]:
            n_try = n_cur + dl
            if 200 <= n_try <= 2000:
                length_tries.append(n_try)

        per_length = min(40.0, remaining * 0.5 / max(1, len(length_tries)))
        for n_try in length_tries:
            if time.monotonic() > deadline - 10:
                break
            a_res = resample_sequence(best_seq, n_try)
            a_opt, obj_opt = mirror_descent(a_res, max_iters=1000,
                                            time_budget=per_length * 0.6,
                                            top_k=3, eta_init=0.3, momentum=0.6)
            if n_try <= 1200:
                a_opt2, obj_opt2 = lp_alternating(a_opt, max_iters=5,
                                                  time_budget=per_length * 0.3,
                                                  verbose=False)
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")

    # ===== Final Validation =====
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"  Target: 1.5027")

    return result
