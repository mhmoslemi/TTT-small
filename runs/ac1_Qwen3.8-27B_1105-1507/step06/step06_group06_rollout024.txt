"""
Optimized search for low-peak-autocorrelation non-negative sequences.
Uses improved LP alternating, frequency-domain equalization, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix for (Toeplitz(a) @ g) = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, m)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_maxsum(a, M, time_limit=15):
    """LP: max sum(g) s.t. Toeplitz(a)@g <= M * ones, g >= 0."""
    n = len(a)
    if n > 3000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=15):
    """LP: min M s.t. Toeplitz(a)@g <= M*ones, sum(g)=1, g>=0."""
    n = len(a)
    if n > 3000:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None

def freq_equalize(a, max_iters=200, time_budget=30.0, alpha=0.8):
    """
    Frequency domain equalization: try to flatten the power spectrum |FFT(a)|^2
    by applying a soft-threshold to the magnitude, then project back.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = fft(a, N)
        mag = np.abs(fa)
        phase = np.angle(fa)
        
        # Flatten the magnitude spectrum
        mag_max = np.max(mag)
        if mag_max < 1e-15:
            break
        
        mag_norm = mag / mag_max
        # Apply power compression (alpha < 1 compresses peaks)
        mag_compressed = mag_norm ** alpha
        # Scale back
        target_mag = mag_compressed * mag_max
        # But we need to preserve the DC component (sum constraint)
        target_mag[0] = mag[0]
        if N // 2 < len(target_mag):
            target_mag[N // 2] = mag[N // 2]
        
        fa_new = target_mag * np.exp(1j * phase)
        a_new = ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Gradually reduce compression
        alpha = min(1.0, alpha * 1.01 + 0.001)

    return best_a, best_obj

def peak_reduction(a, n_steps=100, time_budget=20.0):
    """Iteratively reduce the peak by adjusting the sequence."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        c_max = conv[k_star]
        
        # Find which pairs (i, k_star - i) contribute most to the peak
        # conv[k_star] = sum_i a[i]*a[k_star-i]
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Reduce the largest contributors: decrease a[i] where contribution is large
        # and increase a[j] where j is not part of the peak
        top_contributors = np.argsort(contributions)[-max(5, n//20):]
        
        a_new = a.copy()
        # Reduce top contributors slightly
        reduction = 0.05 * c_max / max(1, a[top_contributors].sum())
        a_new[top_contributors] -= reduction * a[top_contributors]
        
        # Redistribute to positions that don't contribute to the peak
        peak_positions = set()
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                peak_positions.add(i)
                peak_positions.add(j)
        
        safe_positions = [i for i in range(n) if i not in peak_positions]
        if not safe_positions:
            safe_positions = list(range(n))
        
        mass_moved = a[top_contributors].sum() - a_new[top_contributors].sum()
        if mass_moved > 0:
            n_safe = len(safe_positions)
            a_new[safe_positions] += mass_moved / n_safe
        
        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0/new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Gaussian
    x = np.linspace(-3, 3, n)
    g = np.exp(-x**2 / 3)
    inits.append(g / g.sum())
    
    # Bimodal
    x = np.linspace(0, 1, n)
    b = np.exp(-((x-0.3)/0.08)**2) + np.exp(-((x-0.7)/0.08)**2)
    inits.append(b / b.sum())
    
    # Sine
    x = np.linspace(0, 1, n)
    s = np.sin(np.pi * x) + 0.1
    inits.append(s / s.sum())
    
    # Random
    r = rng.random(n) + 0.01
    inits.append(r / r.sum())
    
    # Lowpass filtered random
    r = rng.random(n)
    fr = fft(r, n)
    cutoff = n // 4
    fr[cutoff:] = 0
    fr[:cutoff] *= 1.0
    r_filt = ifft(fr).real
    r_filt = np.maximum(r_filt, 0)
    if r_filt.sum() > 1e-10:
        inits.append(r_filt / r_filt.sum())
    
    if parent is not None:
        resampled = resample(parent, n)
        inits.append(resampled / resampled.sum() if resampled.sum() > 0 else np.ones(n)/n)
        # Perturbed
        pert = resampled * np.exp(rng.normal(0, 0.15, n))
        pert = np.maximum(pert, 0)
        if pert.sum() > 1e-10:
            inits.append(pert / pert.sum())

    return inits

"""
Multi-strategy optimization for low-peak-autocorrelation non-negative sequences.
Combines LP alternating (improved), frequency-domain equalization, subgradient
descent with momentum, adaptive peak reduction, and multi-length exploration.
Key improvements: adaptive M in LP, momentum-based subgradient, frequency

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, m)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_maxsum(a, M, time_limit=15):
    """LP: max sum(g) s.t. Toeplitz(a)@g <= M*ones, g >= 0."""
    n = len(a)
    if n > 2500:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=15):
    """LP: min M s.t. Toeplitz(a)@g <= M*ones, sum(g)=1, g>=0."""
    n = len(a)
    if n > 2500:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def lp_alternating(a, max_iters=150, time_budget=120.0, m_scale=1.0):
    """
    LP alternating with adaptive M scaling.
    Uses m_scale < 1.0 to force g to be more spread out.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()

    start = time.monotonic()

    last_t = 0.3

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Try with slightly reduced M to force more spread
        M_eff = M * m_scale
        tl = min(rem * 0.25, 10.0)
        g = solve_lp_maxsum(a, M_eff, time_limit=tl)

        if g is None:

            if g2 is None:

                if g is None:
                    break
        else:
            pass

        # Line search with adaptive step
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        candidate_ts = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.7, 1.0]
        if 0.02 < last_t < 1.0:

            candidate_ts.append(min(1.0, last_t * 1.3))

            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)

def freq_equalize(a, max_iters=300, time_budget=40.0, alpha_init=0.7, alpha_end=0.95):
    """
    Frequency domain equalization: flatten power spectrum to reduce autocorrelation peak.
    Variable alpha for progressive compression.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        rem = time_budget - (time.monotonic() - start)
        if rem < 2:
            break

        alpha = alpha_init + (alpha_end - alpha_init) * (it / max_iters)

        fa = fft(a, N)
        mag = np.abs(fa)
        phase = np.angle(fa)

        mag_max = np.max(mag)
        if mag_max < 1e-15:
            break

        mag_norm = mag / mag_max
        mag_compressed = mag_norm ** alpha
        target_mag = mag_compressed * mag_max
        # Preserve DC and Nyquist if present
        target_mag[0] = mag[0]
        if N % 2 == 0:
            target_mag[N // 2] = mag[N // 2]

        fa_new = target_mag * np.exp(1j * phase)
        a_new = ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def subgrad_momentum(a, max_iters=500, time_budget=50.0, tau_init=0.05, lr_init=0.03, momentum=0.9):
    """
    Subgradient descent with Nesterov momentum on softmax-smoothed max convolution.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    v = np.zeros(n)

    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1

    for it in range(max_iters):
        rem = time_budget - (time.monotonic() - start)
        if rem < 2:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights
        tau = tau_init + 0.1 * (it / max_iters)
        shifted = np.clip((conv - cmax) / tau, -200.0, 0.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient via cross-correlation
        fw = fft(weights, N_fft)
        fa_fft = fft(a, N_fft)
        corr = ifft(fw * np.conj(fa_fft)).real[:n]
        grad = 2.0 * corr

        # Center (scale invariance)
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v - lr * grad_centered
        a_new = a - v
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.97
            v *= 0.5

        lr = max(lr, 0.002)

    return best_a, best_obj

def peak_directed_repair(a, n_steps=80, time_budget=20.0):
    """
    Directly target the peak convolution position and reduce it
    by adjusting elements that contribute most.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()

        c_max = conv[k_star]

        # Compute contributions to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        # Find top contributors
        n_top = max(5, n // 25)
        top_idx = np.argsort(contribs)[-n_top:]

        # Reduce top contributors, add to "safe" positions
        a_new = a.copy()
        reduction_frac = 0.03 + 0.02 * (step / n_steps)
        mass_moved = 0.0
        for i in top_idx:
            j = k_star - i
            if j == i:
                # Diagonal: reducing a[i] reduces a[i]^2
                a_new[i] *= (1 - reduction_frac * 0.5)
            else:
                # Off-diagonal: reduce the larger of a[i], a[j]
                if a[i] >= a[j]:
                    a_new[i] *= (1 - reduction_frac)
                else:
                    a_new[j] *= (1 - reduction_frac)
            mass_moved += a[i] - max(a_new[i], 0)

        # Find safe positions (not contributing to this peak)
        peak_set = set()
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                peak_set.add(i)
        
        safe = [i for i in range(n) if i not in peak_set]
        if not safe:
            safe = [i for i in range(n) if i not in set(top_idx)]
        if not safe:
            safe = list(range(n))

        if mass_moved > 1e-12:
            n_safe = len(safe)
            a_new[safe] += mass_moved / n_safe

        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian centered
    x = np.linspace(-3, 3, n)
    g = np.exp(-x**2 / 4.0)
    inits.append(g / g.sum())

    # Bimodal
    x = np.linspace(0, 1, n)
    b = np.exp(-((x - 0.25) / 0.08)**2) + np.exp(-((x - 0.75) / 0.08)**2)
    inits.append(b / b.sum())

    # Sinusoidal
    x = np.linspace(0, 1, n)
    s = np.sin(np.pi * x) + 0.05
    inits.append(s / s.sum())

    # Cosine window
    x = np.linspace(0, np.pi, n)
    c = np.cos(x / 2) ** 2
    inits.append(c / c.sum())

    # Random
    r = rng.random(n) + 0.01
    inits.append(r / r.sum())

    # Lowpass filtered random (smooth)
    r = rng.random(n)
    fr = fft(r, n)
    cutoff = max(2, n // 6)
    fr[cutoff:] = 0
    r_filt = np.maximum(ifft(fr).real, 0)
    if r_filt.sum() > 1e-10:
        inits.append(r_filt / r_filt.sum())

    # Sparse peaks
    n_peaks = max(3, n // 100)
    peak_pos = rng.choice(n, size=n_peaks, replace=False)
    sp = np.zeros(n)
    widths = rng.randint(3, max(4, n // 50), size=n_peaks)
    for p, w in zip(peak_pos, widths):
        lo = max(0, p - w)
        hi = min(n, p + w + 1)
        sp[lo:hi] += 1.0
    sp = np.maximum(sp, 0.01)
    inits.append(sp / sp.sum())

    # From parent
    if parent is not None:
        resampled = resample(parent, n)
        resampled = np.maximum(resampled, 0)
        if resampled.sum() > 1e-10:
            inits.append(resampled / resampled.sum())
            # Perturbed parent
            pert = resampled * np.exp(rng.normal(0, 0.2, n))
            pert = np.maximum(pert, 0)
            if pert.sum() > 1e-10:
                inits.append(pert / pert.sum())
            # Smoothed parent
            sm = np.convolve(resampled, np.ones(5)/5, mode='same')
            sm = np.maximum(sm, 0)
            if sm.sum() > 1e-10:
                inits.append(sm / sm.sum())

    return inits

                time_budget=min(rem * 0.25, 20.0), alpha_init=0.55, alpha_end=0.88)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_best = a_new
                n_best = len(a_best)
                print(f"  Round {rnd} (freq): obj={best_score:.8f}")

            rem2 = phase2_start + phase2_budget - time.monotonic()
            if rem2 > 10:
                a_new2, obj_new2 = subgrad_momentum(a_best, max_iters=200,
                    time_budget=min(rem2 * 0.3, 15.0), tau_init=0.02, lr_init=0.01)
                if obj_new2 < best_score:
                    best_score = obj_new2
                    best_seq = a_new2.copy()
                    a_best = a_new2
                    print(f"  Round {rnd} (subgrad): obj={best_score:.8f}")

    # === Phase 3: Condensed explosion + re-optimize (10% of time) ===
    print(f"\n=== Phase 3: Perturbation + Re-optimize ===")
    print(f"Best: {best_score:.8f}")
    phase3_budget = min(100.0, (deadline - time.monotonic()) * 0.12)
    phase3_start = time.monotonic()

    for pe in range(5):
        rem = phase3_start + phase3_budget - time.monotonic()
        if rem < 15:
            break

        # Perturb the best sequence
        frac = 0.05 + 0.03 * pe
        n_zero = max(10, int(len(best_seq) * frac))
        sorted_idx = np.argsort(best_seq)
        zero_idx = sorted_idx[:n_zero]

        a_pert = best_seq.copy()
        mass = a_pert[zero_idx].sum()
        a_pert[zero_idx] = 0.0

        # Redistribute
        fill_idx = rng.choice(len(a_pert), size=n_zero // 2 + 5, replace=False)
        weights = rng.dirichlet(np.ones(len(fill_idx)) * 3.0)
        a_pert[fill_idx] += mass * weights

        a_pert = project_nonneg_sum1(a_pert)

        # Quick optimize
        a_new, obj_new = single_track_optimize(a_pert, min(rem * 0.4, 20.0), rng)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Perturb {pe}: NEW BEST obj={best_score:.8f}")

    # === Phase 4: Fine length tuning (remaining) ===
    print(f"\n=== Phase 4: Fine Length Tuning ===")
    print(f"Best: {best_score:.8f}")
    rem_final = deadline - time.monotonic()
    if rem_final > 20:
        phase4_budget = min(100.0, rem_final * 0.6)
        phase4_start = time.monotonic()

        n_cur = len(best_seq)
        fine_ns = sorted(set([
            max(100, n_cur - 50), max(100, n_cur - 20),
            n_cur + 20, n_cur + 50, n_cur + 100, n_cur + 200
        ]))
        fine_ns = [n for n in fine_ns if 100 <= n <= 4000]

        per_fine = phase4_budget / max(1, len(fine_ns))

        for n_try in fine_ns:
            rem4 = phase4_start + phase4_budget - time.monotonic()
            if rem4 < 10:
                break
            a_rs = resample(best_seq, n_try)
            a_rs = project_nonneg_sum1(a_rs)

            if n_try <= 2000:
                a_new, obj_new = lp_alternating(a_rs, max_iters=30,
                    time_budget=min(per_fine * 0.6, 15.0), m_scale=1.0)
            else:
                a_new, obj_new = freq_equalize(a_rs, max_iters=100,
                    time_budget=min(per_fine * 0.6, 15.0))

            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: NEW BEST obj={best_score:.8f}")

    # === Phase 5: Final micro-polish ===
    print(f"\n=== Phase 5: Final Micro-Polish ===")
    rem5 = deadline - time.monotonic()
    if rem5 > 10 and best_seq is not None:
        a_fin = project_nonneg_sum1(best_seq.copy())
        n_fin = len(a_fin)

        # Quick LP if possible
        if n_fin <= 1500 and rem5 > 15:
            a_p, obj_p = lp_alternating(a_fin, max_iters=20,
                time_budget=min(rem5 * 0.3, 12.0), m_scale=0.98)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Final LP: obj={best_score:.8f}")

        # Quick peak repair
        rem5b = deadline - time.monotonic()
        if rem5b > 5:
            a_p2, obj_p2 = peak_directed_repair(best_seq, n_steps=30,
                time_budget=min(rem5b * 0.3, 8.0))
            if obj_p2 < best_score:
                best_score = obj_p2
                best_seq = a_p2.copy()
                print(f"  Final peak: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
