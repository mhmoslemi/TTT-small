"""
Optimized step function search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines smooth gradient descent with adaptive log-sum-exp temperature,
FFT-based efficient gradients, spectral shaping, LP refinement,
and multi-length exploration. Key improvements over prior work:
(1) proper gradient with factor-of-2 correction, (2) adaptive tau schedule,
(3) backtracking line search with sufficient decrease, (4) Nesterov momentum,
(5) multi-strategy spectral compression, (6) structured asymmetry exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_weights(conv, tau):
    """Compute softmax weights over convolution values."""
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= wsum
    return weights


def compute_gradient(a, weights):
    """
    Compute gradient of weighted sum of conv values w.r.t. a.
    grad[i] = 2 * sum_j weights[i+j] * a[j] for valid j.
    Uses FFT for efficiency.
    """
    n = len(a)
    w = weights  # length 2n-1
    w_rev = w[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    W = rfft(w_rev, size)
    A = rfft(a, size)
    conv_full = irfft(W * A, size)[:3 * n - 2]
    return 2.0 * conv_full[n - 1:2 * n - 1][::-1]


def smooth_gradient_descent(a, max_iters=2000, time_budget=60.0,
                            tau_init=0.001, tau_final=0.01,
                            lr_init=0.1, momentum=0.9):
    """
    Nesterov accelerated smoothed gradient descent with adaptive tau
    and backtracking line search.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2.0:
            break

        # Adaptive tau: start small (peak-focused), increase
        frac = it / max(max_iters - 1, 1)
        tau = tau_init + (tau_final - tau_init) * frac
        lr = lr_init * (1.0 - 0.5 * frac)

        # Gradient at Nesterov point
        conv_y = compute_conv(y)
        weights = compute_weights(conv_y, tau)
        grad_y = compute_gradient(y, weights)

        # Center gradient (removes the "uniform shift" component)
        grad_c = grad_y - np.mean(grad_y)

        # Nesterov momentum update
        v = momentum * v - lr * grad_c
        y = a + v

        # Project
        a_new = project_nonneg_sum1(y)

        # Backtracking line search
        obj_new = compute_obj(a_new)
        obj_cur = compute_obj(a)
        alpha = 1.0
        for _ in range(5):
            if obj_new < obj_cur - 1e-15:
                break
            alpha *= 0.5
            y_try = a + alpha * v
            a_new = project_nonneg_sum1(y_try)
            obj_new = compute_obj(a_new)
            if alpha < 0.001:
                break

        a = a_new
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

    return best_a, best_obj


def spectral_shaping(a, max_iters=100, time_budget=20.0, method="power"):
    """
    Iterative spectral shaping: compress dynamic range of |FFT(a)|^2
    while preserving non-negativity in time domain.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        ps_max = np.max(power)
        if ps_max < 1e-15:
            break

        ps_norm = power / ps_max

        if method == "power":
            alpha = 0.6 + 0.3 * (it / max_iters)
            ps_new = ps_norm ** alpha
        elif method == "log":
            ps_new = np.log1p(5.0 * ps_norm) / np.log(6.0)
        elif method == "soft_threshold":
            threshold = 0.1 + 0.5 * (it / max_iters)
            ps_new = np.maximum(ps_norm - threshold, 0) / np.maximum(1 - threshold, 0.01)
            ps_new = np.maximum(ps_new, 1e-10)
        else:
            ps_new = ps_norm

        # Reconstruct with modified magnitude, same phase
        phase = np.angle(fa)
        mag = np.sqrt(ps_new * ps_max)
        fa_new = mag * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        # Blend for stability
        a = 0.7 * a + 0.3 * a_new
        a = project_nonneg_sum1(a)

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if n > 1500:
        return None, None
    A_mat = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_mat
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternate(a, max_iters=30, time_budget=30.0):
    """LP alternating with line search."""
    n = len(a)
    if n > 1200:
        return a, compute_obj(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0
        A_mat = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 10})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s < 1e-10:
                break
            g = g / s
        except Exception:
            break

        # Line search
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = project_nonneg_sum1((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break

    return best_a, best_obj


def asymmetry_search(a, rng, time_budget=20.0):

def simulated_annealing_grad(a, rng, max_iters=300, T_init=0.1, T_min=0.001,
                             time_budget=40.0):
    """
    Simulated annealing with gradient-informed moves.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    cur_a = a.copy()
    cur_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = cur_obj
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T_init * (T_min / T_init) ** (it / max_iters)

        # Gradient-informed perturbation
        conv = compute_conv(cur_a)
        k_star = int(np.argmax(conv))
        tau = 0.002
        weights = compute_weights(conv, tau)
        grad = compute_gradient(cur_a, weights)
        grad_c = grad - np.mean(grad)

        # Mix gradient direction with random noise
        noise_scale = 0.1 * (1 + it / max_iters)
        direction = -grad_c * 0.3 + rng.normal(0, noise_scale, n)
        direction -= np.mean(direction)

        step = 0.02 * (1 + 0.5 * (1 - it / max_iters))
        a_trial = project_nonneg_sum1(cur_a + step * direction)
        trial_obj = compute_obj(a_trial)

        # Also try a window-level move
        if rng.random() < 0.3:
            wsize = max(5, n // 20)
            wstart = rng.integers(0, n - wsize)
            a_trial2 = cur_a.copy()
            a_trial2[wstart:wstart + wsize] *= np.exp(rng.normal(0, 0.5, wsize))
            a_trial2 = project_nonneg_sum1(a_trial2)
            trial_obj2 = compute_obj(a_trial2)
            if trial_obj2 < trial_obj:
                a_trial = a_trial2
                trial_obj = trial_obj2

        delta = trial_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_a = a_trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()

    return best_a, best_obj


def multi_peak_reduction(a, max_iters=200, time_budget=25.0):
    """
    Targeted multi-peak reduction: identify all significant peaks in the
    convolution and reduce them simultaneously using their combined gradient.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Find all peaks above 90% of max
        threshold = 0.9 * cmax
        peak_mask = conv >= threshold
        n_peaks = np.sum(peak_mask)

        if n_peaks == 0:
            break

        # Compute combined gradient from all significant peaks
        tau = 0.002
        weights = compute_weights(conv, tau)

        # Weight peaks more heavily
        weights[peak_mask] *= 2.0
        weights /= np.sum(weights)

        grad = compute_gradient(a, weights)
        grad_c = grad - np.mean(grad)

        # Adaptive step size
        grad_norm = np.linalg.norm(grad_c)
        if grad_norm < 1e-10:
            break
        step = 0.05 / (1.0 + it / 50.0)

        a_new = project_nonneg_sum1(a - step * grad_c / (grad_norm / n * 0.1 + 1e-10))
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization starting from parent.
    Phases: (1) gradient polish of parent, (2) spectral shaping,
    (3) LP refinement, (4) asymmetry search, (5) SA, (6) multi-length,
    (7) final intensive polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent = project_nonneg_sum1(parent)
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()
    else:
        # Fallback: use a Hann-like window
        n = 700
        x = np.linspace(0, 1, n)
        best_seq = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.01
        best_seq = project_nonneg_sum1(best_seq)
        best_score = compute_obj(best_seq)
        print(f"Generated initial (n={len(best_seq)}): obj={best_score:.8f}")

    print(f"Starting best: {best_score:.8f}, n={len(best_seq)}")

    # === Phase 1: Intensive gradient polish (30% time) ===
    print("\n=== Phase 1: Gradient Polish ===")
    p1_budget = min((deadline - start_time) * 0.25, 300.0)
    a = project_nonneg_sum1(best_seq.copy())

    # Multiple rounds with different parameters
    for round_idx in range(5):
        if time.monotonic() > start_time + p1_budget:
            break
        remaining = start_time + p1_budget - time.monotonic()
        if remaining < 5:
            break

        tau_i = 0.0005 + 0.002 * round_idx
        tau_f = 0.005 + 0.01 * round_idx
        lr = 0.08 - 0.01 * round_idx

        a_new, obj_new = smooth_gradient_descent(
            a, max_iters=500, time_budget=min(remaining * 0.6, 50.0),
            tau_init=tau_i, tau_final=tau_f,
            lr_init=lr, momentum=0.9)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  Gradient r{round_idx}: {best_score:.8f}")
        else:
            a = a_new  # still use as starting point

    # === Phase 2: Spectral shaping (15% time) ===
    print(f"\n=== Phase 2: Spectral Shaping ===")
    print(f"Best: {best_score:.8f}")
    p2_budget = min((deadline - start_time) * 0.15, 150.0)
    a = project_nonneg_sum1(best_seq.copy())

    for method in ["power", "log", "soft_threshold"]:
        if time.monotonic() > start_time + p2_budget:
            break
        remaining = start_time + p2_budget - time.monotonic()
        if remaining < 5:
            break
        a_new, obj_new = spectral_shaping(a, max_iters=100,
                                          time_budget=min(remaining * 0.5, 30.0),
                                          method=method)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  Spectral ({method}): {best_score:.8f}")

    # === Phase 3: LP refinement (15% time) ===
    print(f"\n=== Phase 3: LP Refinement ===")
    print(f"Best: {best_score:.8f}")
    p3_budget = min((deadline - start_time) * 0.15, 150.0)
    n_best = len(best_seq)

    if n_best <= 1200:
        a = project_nonneg_sum1(best_seq.copy())
        a_new, obj_new = lp_alternate(a, max_iters=30,
                                      time_budget=min(p3_budget * 0.7, 60.0))
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  LP alternate: {best_score:.8f}")

        # Min-peak LP blend
        if time.monotonic() < start_time + p3_budget - 5:
            g_lp, M_opt = solve_lp_minpeak(best_seq, time_limit=15)
            if g_lp is not None:
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_try = project_nonneg_sum1((1 - t) * best_seq + t * g_lp)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_score:
                        best_score = obj_try
                        best_seq = a_try.copy()
                        print(f"  LP blend t={t}: {best_score:.8f}")

    # === Phase 4: Asymmetry search (10% time) ===
    print(f"\n=== Phase 4: Asymmetry Search ===")
    print(f"Best: {best_score:.8f}")
    p4_budget = min((deadline - start_time) * 0.10, 100.0)
    a = project_nonneg_sum1(best_seq.copy())
    a_new, obj_new = asymmetry_search(a, rng, time_budget=p4_budget)
    if obj_new < best_score:
        best_score = obj_new
        best_seq = a_new.copy()
        print(f"  Asymmetry: {best_score:.8f}")

    # === Phase 5: Simulated Annealing (15% time) ===
    print(f"\n=== Phase 5: Simulated Annealing ===")
    print(f"Best: {best_score:.8f}")
    p5_budget = min((deadline - start_time) * 0.15, 150.0)
    a = project_nonneg_sum1(best_seq.copy())
    a_new, obj_new = simulated_annealing_grad(a, rng, max_iters=500,
                                              T_init=0.05, T_min=0.001,
                                              time_budget=p5_budget)
    if obj_new < best_score:
        best_score = obj_new
        best_seq = a_new.copy()
        print(f"  SA: {best_score:.8f}")

    # Multi-peak reduction
    if time.monotonic() < deadline - 30:
        a = project_nonneg_sum1(best_seq.copy())
        a_new, obj_new = multi_peak_reduction(a, max_iters=200, time_budget=30.0)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Multi-peak: {best_score:.8f}")

    # === Phase 6: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 6: Multi-length Exploration ===")
    print(f"Best: {best_score:.8f}, n={len(best_seq)}")
    p6_budget = min((deadline - time.monotonic()) * 0.15, 120.0)
    n_current = len(best_seq)

    n_tries = []
    for delta in [-100, -50, -20, 20, 50, 100, 200, -200]:
        n_try = n_current + delta
        if 100 <= n_try <= 2000 and n_try not in n_tries:
            n_tries.append(n_try)

    per_n_budget = max(5.0, p6_budget / max(len(n_tries), 1))

    for n_try in n_tries:
        if time.monotonic() > start_time + p6_budget - 20 or time.monotonic() > deadline - 30:
            break
        a_resampled = resample_sequence(best_seq, n_try)
        a_resampled = project_nonneg_sum1(a_resampled)

        # Quick gradient polish
        a_opt, obj_opt = smooth_gradient_descent(a_resampled, max_iters=200,
                                                 time_budget=min(per_n_budget * 0.6, 20.0),
                                                 tau_init=0.001, tau_final=0.005,
                                                 lr_init=0.05, momentum=0.9)
        # Spectral
        a_opt2, obj_opt2 = spectral_shaping(a_opt, max_iters=50,
                                            time_budget=min(per_n_budget * 0.3, 10.0),
                                            method="power")
        if obj_opt2 < obj_opt:
            a_opt, obj_opt = a_opt2, obj_opt2

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 7: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish ===")
    print(f"Best before: {best_score:.8f}")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a = project_nonneg_sum1(best_seq.copy())
        n_final = len(a)

        # Cycle through all strategies rapidly
        strategies = []
        strategies.append(("grad_fine", lambda x: smooth_gradient_descent(
            x, max_iters=300, time_budget=min(15, remaining * 0.1),
            tau_init=0.0003, tau_final=0.002, lr_init=0.03, momentum=0.92)))
        strategies.append(("spectral", lambda x: spectral_shaping(
            x, max_iters=80, time_budget=min(10, remaining * 0.08), method="power")))
        strategies.append(("multi_peak", lambda x: multi_peak_reduction(
            x, max_iters=100, time_budget=min(10, remaining * 0.08))))
        if n_final <= 1000:
            strategies.append(("lp", lambda x: lp_alternate(
                x, max_iters=20, time_budget=min(15, remaining * 0.1))))

        strat_cycle = 0
        while time.monotonic() < deadline - 5 and remaining > 5:
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            name, func = strategies[strat_cycle % len(strategies)]
            try:
                a_new, obj_new = func(a)
            except Exception:
                strat_cycle += 1
                continue
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  Final {name} cycle{strat_cycle}: {best_score:.8f}")
            else:
                # Still use the result as starting point (it might be in a better basin)
                if compute_obj(a_new) < compute_obj(a) + 0.001:
                    a = a_new
            strat_cycle += 1

            # Small perturbation every few cycles
            if strat_cycle % 4 == 0:
                a_pert = a * np.exp(rng.normal(0, 0.01, n_final))
                a_pert = project_nonneg_sum1(a_pert)
                obj_pert = compute_obj(a_pert)
                if obj_pert < compute_obj(a):
                    a = a_pert

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        val = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"evaluate_sequence confirms: {val:.8f}")

    return result
