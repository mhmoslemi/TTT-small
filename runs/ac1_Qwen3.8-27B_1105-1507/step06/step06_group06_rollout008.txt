```python
    """Successive linearization LP for minimizing max self-convolution."""
    n = len(a0)
    a = project_nonneg_sum1(a0.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv0 = compute_conv(a)
        # Build LP: min M s.t. 2*(a*a0)[k] - conv0[k] <= M, sum(a)=1, a>=0, M>=0
        # Variables: a[0..n-1], M (total n+1 variables)
        # Constraint k: sum_i 2*a0[k-i]*a[i] - M <= conv0[k]
        
        # Build A_ub: shape (2n-1, n+1)
        A_ub = np.zeros((2*n-1, n+1))
        b_ub = conv0.copy()
        
        for k in range(2*n-1):
            A_ub[k, n] = -1.0  # coefficient of M
            # Coefficients of a[i]: 2*a0[k-i] for valid i
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                A_ub[k, i] = 2.0 * a0[k - i]
        
        # Objective: minimize M -> c = [0, 0, ..., 0, 1]
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        # Equality: sum(a) = 1
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5})
            if res.status == 0:
                a_new = np.maximum(res.x[:n], 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                    # Update a for next iteration (use the LP solution regardless)
                    a = a_new
        except:
            break
    
    return best_a, best_obj
```

```python
"""
Optimize non-negative sequences to minimize 2*n*max(self_conv)/(sum)^2.
Novel approach: Bundle LP with multiple linearization cuts and constraint selection,
combined with Nesterov-accelerated gradient on log-sum-exp smoothed objective,
adaptive T scheduling, and multi-length exploration with restarts.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from scipy.sparse import csc_matrix
import time


def compute_conv(a):
    """Compute linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def smoothed_obj_grad(a, T):
    """
    Compute smoothed objective 2n*(1/T)*log(sum(exp(T*conv[k]))) and its gradient.
    Returns (smoothed_value, gradient).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    shifted = T * (conv - cmax)
    log_sum = cmax * T + np.log(np.sum(np.exp(shifted)))
    smoothed_val = 2.0 * n * log_sum / T
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= wsum
    # Gradient: 4n * sum_k weights[k] * a[k-i] for valid k-i
    size_w = len(weights)
    N_fft = 1
    while N_fft < size_w + n - 1:
        N_fft <<= 1
    fw = rfft(weights, N_fft)
    fa = rfft(a, N_fft)
    corr = irfft(fw * np.conj(fa), N_fft)
    grad = 4.0 * n * corr[:n]
    return smoothed_val, grad


def nesterov_smoothed(a_init, T_start=20.0, T_end=80.0, max_iters=3000, time_budget=60.0, lr_init=0.005):
    """Nesterov accelerated gradient on smoothed objective with adaptive T."""
    n = len(a_init)
    x = project_nonneg_sum1(a_init.astype(np.float64).copy())
    best_a = x.copy()
    best_obj = compute_obj(x)
    start = time.monotonic()
    v = np.zeros(n)
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)
        _, grad = smoothed_obj_grad(x, T)
        grad_centered = grad - np.mean(grad)
        v = 0.9 * v - lr * grad_centered
        x_new = x + v
        x_new = project_nonneg_sum1(x_new)
        obj = compute_obj(x_new)
        if obj < best_obj:
            best_obj = obj
            best_a = x_new.copy()
        x = x_new
        lr *= 0.9995
    return best_a, best_obj


def build_single_cut_lp(a_cut, conv_cut, n, top_k=None):
    """
    Build LP: min M s.t. 2*(a*a_cut)[k] - conv_cut[k] <= M for all k,
              sum(a)=1, a>=0, M>=0.
    Optionally keep only top_k most binding constraints.
    Returns (A_ub, b_ub) or None.
    """
    nconv = 2 * n - 1
    if top_k is not None and top_k < nconv:
        # Select top_k constraints with largest b_ub (most likely binding)
        idx = np.argsort(conv_cut)[::-1][:top_k]
    else:
        idx = np.arange(nconv)

    m = len(idx)
    rows = []
    cols = []
    data = []
    b_ub = np.zeros(m)

    for mi, k in enumerate(idx):
        rows.append(mi)
        cols.append(n)
        data.append(-1.0)
        b_ub[mi] = conv_cut[k]
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            rows.append(mi)
            cols.append(i)
            data.append(2.0 * a_cut[k - i])

    A_ub = csc_matrix((data, (rows, cols)), shape=(m, n + 1))
    return A_ub, b_ub

def chase_peak(a, n_steps=50, step_size=0.03):
    """Chase-the-peak subgradient method."""
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
            j2 = i + n - 1 - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]
        grad_centered = grad - np.mean(grad)
        improved = False
        for ss in [step_size, step_size * 0.5, step_size * 0.25, step_size * 2.0]:
            a_try = a - ss * grad_centered
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
        if not improved:
            break
    return best_a, best_obj


def freq_domain_smooth(a, max_iters=100, time_budget=20.0, alpha_start=0.6, alpha_end=0.85):

    best_a = a.copy()

    while N < 2 * n - 1:
        N <<= 1
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        fa = np.fft.fft(a, N)
        power_spec = np.abs(fa) ** 2
        ps_max = np.max(power_spec)
        if ps_max < 1e-15:
            break
        ps_norm = power_spec / ps_max
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        ps_compressed = np.maximum(ps_norm, 1e-12) ** alpha
        phase = np.angle(fa)
        magnitude = np.sqrt(ps_compressed * ps_max)
        fa_new = magnitude * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)
    tri = np.arange(1, n + 1, dtype=float)
    tri = np.concatenate([tri, tri[:n // 2][::-1]])[:n]
    inits.append(tri / tri.sum())
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.1
    inits.append(sine / sine.sum())
    xg = np.linspace(-3, 3, n)
    gauss = np.exp(-xg ** 2 / 4)
    inits.append(gauss / gauss.sum())
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())
    # Bimodal
    bimodal = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
    inits.append(bimodal / bimodal.sum())
    # Cosine with harmonic
    cos_seq = 1.0 + 0.5 * np.cos(2 * np.pi * x * 3)
    cos_seq = np.maximum(cos_seq, 0.01)
    inits.append(cos_seq / cos_seq.sum())
    if parent is not None:
        inits.append(resample_sequence(parent, n))
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_nonneg_sum1(pert))
    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Bundle LP + Nesterov smoothed + multi-n + perturbations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    print(f"Starting: best={best_score:.8f}, n_parent={len(best_seq)}")

    # === Phase 1: Quick multi-n screening (15% time) ===
    print("\n=== Phase 1: Multi-n Screening ===")
    phase1_budget = 0.15 * (deadline - start_time)
    phase1_end = start_time + phase1_budget

    n_parent = len(best_seq)
    n_options = [n_parent, max(100, n_parent - 50), max(100, n_parent - 100),
                 n_parent + 50, n_parent + 100, n_parent + 200]
    n_options = list(dict.fromkeys([n for n in n_options if 50 <= n <= 2000]))

    per_n_budget = max(3.0, phase1_budget / len(n_options))

    for n_try in n_options:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        inits = generate_initializations(n_try, rng, parent)
        per_init = max(0.5, (per_n_budget - (time.monotonic() - t0)) / max(1, len(inits)))

        for init in inits:
            if time.monotonic() > phase1_end:
                break
            # Quick Nesterov
            a_q, obj_q = nesterov_smoothed(init, T_start=30, T_end=60,
                                           max_iters=min(200, int(per_init * 50)),
                                           time_budget=min(per_init, 3.0), lr_init=0.003)
            # Quick chase peak
            a_q2, obj_q2 = chase_peak(a_q, n_steps=20, step_size=0.02)
            if obj_q2 < best_score:
                best_score = obj_q2
                best_seq = a_q2.copy()
                print(f"  n={n_try}: improved to {best_score:.8f}")

    print(f"  After Phase 1: best={best_score:.8f}, n={len(best_seq)}")

    # === Phase 2: Intensive Bundle LP + Nesterov on best (35% time) ===
    print("\n=== Phase 2: Bundle LP + Nesterov ===")
    phase2_budget = 0.35 * (deadline - start_time)
    phase2_end = time.monotonic() + phase2_budget
    a_work = project_nonneg_sum1(best_seq.copy())
    n_work = len(a_work)

    # Sub-phase 2a: Nesterov with increasing T
    t2a_budget = min(30.0, phase2_budget * 0.3)
    a_ns, obj_ns = nesterov_smoothed(a_work, T_start=20, T_end=100,
                                     max_iters=2000, time_budget=t2a_budget, lr_init=0.004)
    if obj_ns < best_score:
        best_score = obj_ns
        best_seq = a_ns.copy()
        a_work = a_ns.copy()
        print(f"  Nesterov: {best_score:.8f}")

    # Sub-phase 2b: Bundle LP
    t2b_budget = min(60.0, phase2_budget - (time.monotonic() - (phase2_end - phase2_budget)))
    a_bl, obj_bl = bundle_lp_solve(a_work, max_iters=40, time_budget=t2b_budget,
                                   max_cuts=6, top_k_per_cut=150)
    if obj_bl < best_score:
        best_score = obj_bl
        best_seq = a_bl.copy()
        a_work = a_bl.copy()
        print(f"  Bundle LP: {best_score:.8f}")

    # Sub-phase 2c: Nesterov polish with high T
    remain2 = phase2_end - time.monotonic()
    if remain2 > 5:
        a_ns2, obj_ns2 = nesterov_smoothed(a_work, T_start=80, T_end=150,
                                           max_iters=1000, time_budget=min(remain2, 30.0),
                                           lr_init=0.002)
        if obj_ns2 < best_score:
            best_score = obj_ns2
            best_seq = a_ns2.copy()
            print(f"  Nesterov polish: {best_score:.8f}")

    # Sub-phase 2d: More bundle LP rounds
    remain2b = phase2_end - time.monotonic()
    if remain2b > 10:
        a_bl2, obj_bl2 = bundle_lp_solve(best_seq, max_iters=30, time_budget=min(remain2b, 40.0),
                                         max_cuts=8, top_k_per_cut=200)
        if obj_bl2 < best_score:
            best_score = obj_bl2
            best_seq = a_bl2.copy()
            print(f"  Bundle LP round 2: {best_score:.8f}")

    print(f"  After Phase 2: best={best_score:.8f}, n={len(best_seq)}")

    # === Phase 3: Multi-n refinement (20% time) ===
    print("\n=== Phase 3: Multi-n Refinement ===")
    phase3_budget = 0.20 * (deadline - start_time)
    phase3_end = time.monotonic() + phase3_budget
    n_best = len(best_seq)

    n_tries = [n_best - 20, n_best - 40, n_best + 20, n_best + 40,
               n_best + 100, n_best - 100, n_best + 50, n_best - 50]
    n_tries = [n for n in n_tries if 100 <= n <= 2500]
    n_tries = list(dict.fromkeys(n_tries))

    per_n3 = max(5.0, phase3_budget / max(1, len(n_tries)))

    for n_try in n_tries:
        if time.monotonic() > phase3_end:
            break
        a_res = project_nonneg_sum1(resample_sequence(best_seq, n_try))
        # Nesterov
        a_r1, obj_r1 = nesterov_smoothed(a_res, T_start=30, T_end=80,
                                         max_iters=500, time_budget=min(per_n3 * 0.4, 15.0),
                                         lr_init=0.003)
        # Bundle LP
        a_r2, obj_r2 = bundle_lp_solve(a_r1, max_iters=15, time_budget=min(per_n3 * 0.4, 15.0),
                                       max_cuts=5, top_k_per_cut=150)
        if obj_r2 < obj_r1:
            a_r1, obj_r1 = a_r2, obj_r2
        # Chase peak
        a_r3, obj_r3 = chase_peak(a_r1, n_steps=30, step_size=0.02)
        if obj_r3 < best_score:
            best_score = obj_r3
            best_seq = a_r3.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_r3:.8f}")

    print(f"  After Phase 3: best={best_score:.8f}, n={len(best_seq)}")

    # === Phase 4: Perturbation + re-optimization (20% time) ===
    print("\n=== Phase 4: Perturbation + Re-optimization ===")
    phase4_budget = 0.20 * (deadline - start_time)
    phase4_end = time.monotonic() + phase4_budget
    n_best = len(best_seq)
    n_perturbs = min(6, int(phase4_budget / 15))

    for pi in range(n_perturbs):
        if time.monotonic() > phase4_end:
            break
        # Different perturbation strategies
        if pi % 3 == 0:
            # Multiplicative noise
            a_pert = best_seq * np.exp(rng.normal(0, 0.03 + 0.01 * pi, n_best))
        elif pi % 3 == 1:
            # Window swap
            a_pert = best_seq.copy()
            wsize = max(20, n_best // 10)
            w1 = rng.integers(0, n_best - wsize)
            w2 = rng.integers(0, n_best - wsize)
            if w2 < w1:
                w1, w2 = w2, w1
            if w1 - w2 >= wsize:
                a_pert[w1:w1 + wsize], a_pert[w2:w2 + wsize] = \
                    a_pert[w2:w2 + wsize].copy(), a_pert[w1:w1 + wsize].copy()
        else:
            # Frequency domain perturbation
            a_pert = best_seq.copy()
            Nf = 1
            while Nf < 2 * n_best - 1:
                Nf <<= 1
            fa = np.fft.fft(a_pert, Nf)
            ps = np.abs(fa) ** 2
            ps = ps * np.exp(rng.normal(0, 0.1, Nf))
            ps = np.maximum(ps, 0)
            mag = np.sqrt(ps)
            ph = np.angle(fa)
            fa_new = mag * np.exp(1j * ph)
            a_pert = np.fft.ifft(fa_new).real[:n_best]

        a_pert = project_nonneg_sum1(a_pert)

        # Optimize
        a_p1, obj_p1 = nesterov_smoothed(a_pert, T_start=40, T_end=100,
                                         max_iters=500, time_budget=10.0, lr_init=0.003)
        a_p2, obj_p2 = bundle_lp_solve(a_p1, max_iters=20, time_budget=15.0,
                                       max_cuts=6, top_k_per_cut=150)
        if obj_p2 < obj_p1:
            a_p1, obj_p1 = a_p2, obj_p2
        a_p3, obj_p3 = chase_peak(a_p1, n_steps=20, step_size=0.02)

        if obj_p3 < best_score:
            best_score = obj_p3
            best_seq = a_p3.copy()
            print(f"  Perturb {pi}: NEW BEST {best_score:.8f}")
        else:
            print(f"  Perturb {pi}: obj={obj_p3:.8f} (no improvement)")

    # === Phase 5: Final intensive polish (10% time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 10:
        n_final = len(best_seq)
        a_final = project_nonneg_sum1(best_seq.copy())

        # Multiple rounds of LP + Nesterov
        rounds = min(5, int(remaining5 / 15))
        for pr in range(rounds):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 10:
                break

            # Bundle LP
            a_pl, obj_pl = bundle_lp_solve(a_final, max_iters=20, time_budget=min(rem * 0.4, 30.0),
                                           max_cuts=8, top_k_per_cut=200)
            if obj_pl < best_score:
                best_score = obj_pl
                best_seq = a_pl.copy()
                a_final = a_pl.copy()
                print(f"  Polish LP r{pr}: {best_score:.8f}")

            # Nesterov with very high T
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                a_pn, obj_pn = nesterov_smoothed(a_final, T_start=100, T_end=200,
                                                 max_iters=500, time_budget=min(rem2 * 0.3, 15.0),
                                                 lr_init=0.001)
                if obj_pn < best_score:
                    best_score = obj_pn
                    best_seq = a_pn.copy()
                    a_final = a_pn.copy()
                    print(f"  Polish Nesterov r{pr}: {best_score:.8f}")

            # Chase peak
            rem3 = deadline - time.monotonic()
            if rem3 > 3:
                a_pc, obj_pc = chase_peak(a_final, n_steps=30, step_size=0.015)
                if obj_pc < best_score:
                    best_score = obj_pc
                    best_seq = a_pc.copy()
                    a_final = a_pc.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
```
