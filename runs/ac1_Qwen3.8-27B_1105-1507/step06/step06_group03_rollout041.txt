Multi-strategy optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k_idx = np.where(cond)[0]
    if len(k_idx) == 0:
        result = np.ones(n) / n
    else:
        lam = cssv[k_idx[-1]] / (k_idx[-1] + 1)
        result = np.maximum(v - lam, 0.0)
    return result * target_sum

    """Resample 1D sequence to new length via linear interpolation."""

    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

Actually, I just realized something important. Let me think about the problem from a different angle.

The convolution $c[k] = \sum_i a[i] a[k-i]$ can be seen as the output of a "match filter" if we think of $a$ as a signal. The max of the convolution is the "peak" of this output.

Another view: if we define the Fourier transform $A(\omega) = \sum_i a[i] e^{-i\omega i}$, then:
$c[k] = \frac{1}{2\pi} \int_{-\pi}^{\pi} |A(\omega)|^2 e^{i\omega k} d\omega$

So the max of $c[k]$ is related to the L1 norm of $|A(\omega)|^2$ (via the Riemann-Lebesgue lemma, the decay of $c[k]$ is related to the smoothness of $|A(\omega)|^2$).

To minimize the max of $c[k]$, we want $|A(\omega)|^2$ to be as smooth as possible (so its Fourier coefficients decay fast). But $|A(\omega)|^2 \geq 0$ and $\sum a[i] = A(0) = 1$, so $|A(0)|^2 = 1$.

This gives a tradeoff: we want the power spectrum to be flat (smooth) while having $|A(0)| = 1$.

This is related to the design of "low PSL" sequences. For complex sequences, the optimal solution has a flat power spectrum ( Zadoff-Chu, CAZAC sequences). For real non-negative sequences, the problem is harder.

One approach: parameterize $a[i] = |w[i]|$ where $w$ is a unit-energy sequence, and optimize the phases of $w$. But since $a \geq 0$, we're effectively fixing the phases to 0, which constrains the power spectrum.

Actually, for non-negative sequences, $A(\omega) = \sum a[i] e^{-i\omega i}$ has a specific structure. The power spectrum $|A(\omega)|^2$ must satisfy certain constraints related to the non-negativity of $a$.

This line of thinking might not lead to a practical algorithm, but it gives insight into the structure of the optimal solution.

"""
Multi-strategy optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: fast coordinate descent with golden section, SCA-LP with active set reduction,
smooth FISTA with momentum on log-softmax parameterization, and multi-length exploration.
Key improvements over prior: vectorized CD, reduced LP constraints, active-set equalization,
and adaptive time allocation.

import numpy as np

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k_idx = np.where(cond)[0]
    if len(k_idx) == 0:
        result = np.ones(n) / n
    else:
        lam = cssv[k_idx[-1]] / (k_idx[-1] + 1)
        result = np.maximum(v - lam, 0.0)
    return result * target_sum


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def sca_lp_active(a, deadline, top_k=80, max_iters=500):
    """
    SCA-LP with active set reduction. Only constrains top_k convolution lags.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        # Find top_k lags
        if 2 * n - 1 <= top_k:
            active_k = np.arange(2 * n - 1)
        else:
            active_k = np.argpartition(c, -top_k)[-top_k:]
        active_k = np.sort(active_k)

        # Build LP: minimize t
        # s.t. 2*(a*g)[k] - c[k] <= t for k in active_k
        #      sum(g) = 1, g >= 0
        # Variables: g[0..n-1], t (index n)
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in active_k:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub))
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(len(b_ub))
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        m = len(active_k)
        A = coo_matrix((vals, (rows, cols)), shape=(m, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (0, np.arange(n))), shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    # Try step fractions
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:

                        J_try = compute_J(a_try)

                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_obj


def fista_logsoftmax(a, deadline, max_iters=20000, tau0=50.0, tau1=500.0, lr0=0.1):
    """
    FISTA with momentum on log-softmax parameterization.
    Objective: J = 2n * max_k conv(a,a) (with sum(a)=1 from normalization).
    We use log-sum-exp to smooth the max.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    x = np.log(a)  # log-softmax parameterization: a = softmax(x)
    x = x - x.mean()  # center for numerical stability

    best_a = a.copy()
    best_obj = compute_J(a)

    # FISTA momentum
    y = x.copy()
    t_mom = 1.0
    no_improve = 0

    c = fast_conv(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau0 + (tau1 - tau0) * min(frac * 3.0, 1.0)
        lr = lr0 / (1.0 + 0.05 * np.sqrt(it + 1))

        # Recompute a from y (the FISTA iterate)
        y_shifted = y - y.max()
        a_y = np.exp(y_shifted)
        a_y /= a_y.sum()

        # Compute smooth gradient
        c_y = fast_conv(a_y)
        c_shifted = c_y - c_y.max()
        w = np.exp(tau * c_shifted)
        w /= w.sum()

        # Gradient of sum_k w[k] * c[k] w.r.t. a = (w * a_rev) + (w_rev * a_rev) type computation
        a_rev = a_y[::-1]
        w_rev = w[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        Wr_fft = np.fft.rfft(w_rev, size)

        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wr_fft * A_fft, size)

        grad_a = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < len(conv1):
                grad_a[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad_a[i] += conv2[idx2]

        # Project gradient to simplex tangent: subtract mean
        grad_a = grad_a - grad_a.mean()

        # FISTA update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        x_new = y - lr * grad_a
        beta = (1.0 + t_mom) / (2.0 + t_next)
        y = x_new + beta * (x_new - x)
        x = x_new
        t_mom = t_next

        # Evaluate
        a_curr = np.exp(y - y.max())
        a_curr /= a_curr.sum()
        obj = 2.0 * n * fast_conv(a_curr).max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_curr.copy()
            no_improve = 0
            c = fast_conv(a_curr)
        else:
            no_improve += 1
            if no_improve > 500:
                # Perturb and reset
                rng = np.random.default_rng(it)
                y = x + rng.normal(0, 0.1, n)
                t_mom = 1.0
                no_improve = 0

    return best_a, best_obj


def active_set_equalize(a, deadline, eps=0.05, max_iters=200):
    """
    Active set equalization: identify lags at the max, and adjust a to equalize them.
    Uses a dual-based approach: add "pressure" from active lags.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()
        # Active set: lags within eps*M of the max
        active = c > M * (1.0 - eps)

        # Subgradient: sum of gradients at active lags
        # grad_k[i] = a[k-i] + a[i-k] (with bounds)
        grad = np.zeros(n)
        k_active = np.where(active)[0]
        for k in k_active:
            # d(c[k])/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
            # But since c[k] = sum_j a[j]a[k-j], d/d(a[i]) = a[k-i] (from j=i term) + a[i-k] (from j=k-i term)
            # Wait: if i != k-i, both terms appear. If i = k-i, only one term.
            for i in range(n):
                p1 = k - i
                p2 = i - k
                val = 0.0
                if 0 <= p1 < n:
                    val += a[p1]
                if 0 <= p2 < n and p2 != p1:
                    val += a[p2]
                elif 0 <= p2 < n:
                    # p2 == p1, means i == k/2, term a[i]^2, derivative is 2*a[i]
                    # But we already counted a[i] once above (as p1 or p2)
                    val += a[p2]
                grad[i] += val

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad /= gnorm

        # Project to simplex tangent and take step
        grad = grad - grad.mean()
        step = 0.05 / (1.0 + 0.01 * it)
        a_new = a - step * grad
        a_new = project_simplex(a_new, 1.0)

        obj = compute_J(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns
    for freq in [1, 2, 3, 5, 8]:
        for amp in [0.2, 0.5, 0.8]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    # Gaussian bumps
    for center in [0.25, 0.5, 0.75]:
        for width in [0.1, 0.2, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.05))

    # Two-sided (mass at edges)
    for alpha in [1, 2, 3, 5]:
        edge = t ** alpha + (1 - t) ** alpha + 0.1
        inits.append(project_simplex(edge))

    # Power law
    for alpha in [0.5, 1, 2, 4]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    # Periodic with envelope
    for period in [8, 16, 32, 64]:
        env = 1.0 / (1.0 + 0.5 * np.sin(2 * np.pi * t))
        pat = env * (1.0 + 0.3 * np.cos(2 * np.pi * t * n / period))
        inits.append(project_simplex(pat))

    return inits


def solve_pipeline(a, deadline):
    """
    Run the full optimization pipeline on a single sequence.
    Returns (best_a, best_obj).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new, label=""):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            print(f"      [{label}] J={obj:.8f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 1: Coordinate descent (35%)
    t1 = time.monotonic() + 0.35 * remaining
    print(f"    [CD] budget={t1 - time.monotonic():.0f}s n={n}", flush=True)
    a_cd, obj_cd = cd_vectorized(a, t1, seed=42, gs_iters=18)
    try_update(a_cd, "CD")

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 2: SCA-LP (25%)
    t2 = time.monotonic() + 0.25 * remaining
    print(f"    [SCA-LP] budget={t2 - time.monotonic():.0f}s n={n}", flush=True)
    a_sca, obj_sca = sca_lp_active(best_a, t2, top_k=100, max_iters=500)
    try_update(a_sca, "SCA")
    a_curr = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 3: FISTA log-softmax (20%)
    t3 = time.monotonic() + 0.20 * remaining
    print(f"    [FISTA] budget={t3 - time.monotonic():.0f}s n={n}", flush=True)
    a_fista, obj_fista = fista_logsoftmax(a_curr, t3, max_iters=15000, tau0=30, tau1=300, lr0=0.08)
    try_update(a_fista, "FISTA")
    a_curr = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 4: Active set equalization (10%)
    t4 = time.monotonic() + 0.10 * remaining
    print(f"    [EQ] budget={t4 - time.monotonic():.0f}s n={n}", flush=True)
    a_eq, obj_eq = active_set_equalize(a_curr, t4, eps=0.03, max_iters=300)
    try_update(a_eq, "EQ")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 5: Final CD polish (10%)
    t5 = deadline - 3
    a_final, obj_final = cd_vectorized(best_a, t5, seed=999, gs_iters=22, max_sweeps=10000)
    try_update(a_final, "FinalCD")

    return best_a, best_obj

    def update_best(a, label=""):
        nonlocal best_seq, best_score

        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
