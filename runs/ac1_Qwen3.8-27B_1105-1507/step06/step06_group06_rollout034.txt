def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def project_simplex_inplace(a):
    """In-place projection (modifies a in place and returns it)."""
    a[:] = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        a[:] = 1.0 / len(a)
    else:
        a /= s
    return a

def weighted_subgradient(a, conv, tau=0.01):
    """Compute weighted subgradient of max_k h_k(a)."""
    n = len(a)
    conv_max = np.max(conv)
    # Softmax weights
    shifted = (conv - conv_max) / tau
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= wsum
    
    # grad[i] = sum_k weights[k] * 2*a[k-i] (for valid k-i)
    grad = np.zeros(n)
    for k_idx in range(len(weights)):
        if weights[k_idx] < 1e-14:
            continue
        # grad[i] += 2 * weights[k_idx] * a[k_idx - i] for 0 <= k_idx - i < n
        i_start = max(0, k_idx - n + 1)
        i_end = min(n, k_idx + 1)
        if i_end > i_start:
            grad[i_start:i_end] += 2.0 * weights[k_idx] * a[k_idx - i_start:k_idx - i_start + (i_end - i_start)][::-1]
    
    return grad

def subgradient_step(a, conv, tau, step_size):
    """One step of projected subgradient method."""
    n = len(a)
    grad = weighted_subgradient(a, conv, tau)
    # Center to maintain sum
    grad_centered = grad - np.mean(grad)
    a_new = a - step_size * grad_centered
    return project_simplex(a_new)

def solcp_step(a, r, n_active=30):
    """
    SOCP trust region step using cvxpy.
    min t s.t. h_k(a) + 2*(B_k a)^T d <= t for k in K_active
              ||d||_2 <= r, d >= -a, 1^T d = 0
    Returns the step d or None if solve fails.
    """
    if not HAS_CVXPY:
        return None
    
    n = len(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    
    # Select active lags
    sorted_idx = np.argsort(conv)[::-1]
    active_lags = sorted_idx[:min(n_active, len(conv))]
    
    # Build the linear terms: for each active lag k, the linear term in d is 2*(B_k a)^T d
    # (B_k a)[i] = a[k-i] (for valid k-i), so (B_k a)^T d = sum_i a[k-i] * d[i]
    # This is the dot product of (reversed a shifted by k) with d.
    # Let v_k[i] = a[k-i] if 0 <= k-i < n, else 0.
    # Then the constraint is: conv[k] + 2 * v_k^T d <= t
    
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.norm(d, 2) <= r,
        d >= -a,
        cp.sum(d) == 0
    ]
    
    for k in active_lags:
        # v_k[i] = a[k-i] if 0 <= k-i < n
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        v = np.zeros(n)
        v[i_start:i_end] = a[k - i_start:k - i_start + (i_end - i_start)][::-1]
        constraints.append(conv[k] + 2.0 * v @ d <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-6)
    except Exception:
        try:
            prob.solve()
        except Exception:
            return None
    
    if d.value is None:
        return None
    
    d_val = np.array(d.value).flatten()
    d_val = np.maximum(d_val, -a)  # Ensure non-negativity of a + d
    s = np.sum(d_val)
    d_val -= s / n  # Ensure sum(d) = 0 approximately
    d_val = np.maximum(d_val, -a)
    
    return d_val

def lp_alternate(a, time_limit=15):
    """LP alternating: find g that maximizes sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 2000:
        return None
    
    conv_a = compute_conv(a)
    M = float(np.max(conv_a))
    
    # Build Toeplitz matrix: (A@g)[k] = sum_j a[k-j] * g[j] = conv(a, g)[k]
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For efficiency, use the structure
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        A[j:j+n, j] = a
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def freq_flatten(a, alpha=0.5):
    """Frequency domain flattening step."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    A = np.fft.fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Flatten: new_mag = mag^alpha * mean_mag^(1-alpha)
    mean_mag = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
    new_mag = (mag ** alpha) * (mean_mag ** (1 - alpha))
    
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.fft.ifft(A_new).real[:n]
    
    return project_simplex(a_new)

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

    # Gaussian

    # Bimodal
    x = np.linspace(0, 1, n)
    bim = np.exp(-((x - 0.25)**2) / 0.03) + np.exp(-((x - 0.75)**2) / 0.03)
    inits.append(bim / bim.sum())
    
    # Random
    r = rng.random(n) + 0.01
    inits.append(r / r.sum())
    
    # One-sided (concentrated on left half)
    one_side = np.zeros(n)
    one_side[:n//2] = 1.0
    inits.append(one_side / one_side.sum())
    
    if parent is not None:
        inits.append(resample_sequence(parent, n))
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.1, n))
        inits.append(project_simplex(pert))
    
    return inits

"""
Multi-strategy optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Combines: (1) projected subgradient with momentum, (2) SOCP trust-region steps via cvxpy,
(3) LP alternating method, (4) frequency-domain spectral flattening, (5) structured
perturbations and multi-length exploration. The key novel element is the SOCP trust-region
method that linearizes the active quadratic constraints and solves a second-order cone
program for the optimal step direction within a ball constraint.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def weighted_subgradient(a, conv, tau=0.01):
    """
    Compute weighted subgradient of max_k h_k(a) using softmax weights.
    grad[i] = sum_k weights[k] * 2 * a[k-i] (for valid indices).
    """
    n = len(a)
    conv_max = np.max(conv)
    shifted = (conv - conv_max) / tau
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= wsum

    grad = np.zeros(n)
    significant = np.where(weights > 1e-12)[0]
    for k_idx in significant:
        w = weights[k_idx]
        i_min = max(0, k_idx - n + 1)
        i_max = min(n - 1, k_idx)
        # v[i] = a[k_idx - i] for i in [i_min, i_max]
        # This array is a[k_idx-i_max : k_idx-i_min+1][::-1]
        grad[i_min:i_max + 1] += 2.0 * w * a[k_idx - i_max:k_idx - i_min + 1][::-1]

    return grad


def subgradient_momentum_step(a, conv, tau, step_size, momentum_v, beta):
    """One momentum subgradient step with projection."""
    grad = weighted_subgradient(a, conv, tau)
    grad_centered = grad - np.mean(grad)
    momentum_v = beta * momentum_v + (1.0 - beta) * grad_centered
    a_new = a - step_size * momentum_v
    return project_simplex(a_new), momentum_v


def solcp_trust_region(a, r, n_active=30, timeout=8.0):
    """
    SOCP trust region step.
    min t s.t. conv[k] + 2*v_k^T d <= t for k in K_active
              ||d||_2 <= r, d >= -a, sum(d) = 0
    Returns step d or None.
    """
    if not HAS_CVXPY:
        return None

    n = len(a)
    conv = compute_conv(a)

    # Select active lags (top n_active by convolution value)
    sorted_idx = np.argsort(conv)[::-1]
    active_lags = sorted_idx[:min(n_active, 2 * n - 1)]

    d_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [
        cp.norm(d_var, 2) <= r,
        d_var >= -a,
        cp.sum(d_var) == 0
    ]

    for k in active_lags:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        v = np.zeros(n)
        # v[i] = a[k-i] for i in [i_start, i_end]
        v[i_start:i_end + 1] = a[k - i_end:k - i_start + 1][::-1]
        constraints.append(conv[k] + 2.0 * (v @ d_var) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    t0 = time.monotonic()
    try:
        prob.solve(solver=cp.SCS, max_iters=20000, eps=1e-7)
    except Exception:
        try:
            prob.solve()
        except Exception:
            return None

    if time.monotonic() - t0 > timeout:
        return None

    if d_var.value is None:
        return None

    d_val = np.array(d_var.value).flatten()
    if not np.all(np.isfinite(d_val)):
        return None

    # Enforce constraints numerically
    d_val = np.maximum(d_val, -a)
    d_val -= np.sum(d_val) / n
    d_val = np.maximum(d_val, -a)
    d_val -= np.sum(d_val) / n

    return d_val


def lp_alternate_step(a, time_limit=5):
    """
    LP alternating: find g maximizing sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    if n > 2500:
        return None

    conv_a = compute_conv(a)
    M = float(np.max(conv_a))

    # Build matrix A where A[k,j] = a[k-j] if 0<=k-j<n, else 0
    # So that (A @ g)[k] = sum_j a[k-j]*g[j] = conv(a,g)[k]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def freq_flatten(a, alpha=0.5):
    """Frequency domain flattening: reduce spectral peaks."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    A = np.fft.fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)

    # Flattening: new_mag = mag^alpha * median^（1-alpha)
    positive_mag = mag[mag > 1e-10]
    if len(positive_mag) > 0:
        ref_mag = np.median(positive_mag)
    else:
        ref_mag = 1.0
    new_mag = (mag ** alpha) * (ref_mag ** (1.0 - alpha))

    A_new = new_mag * np.exp(1j * phase)
    a_new = np.fft.ifft(A_new).real[:n]

    return project_simplex(a_new)


def freq_flatten_adaptive(a, num_iters=5):
    """Adaptive frequency flattening with varying alpha."""
    a_curr = a.copy()
    best_a = a_curr.copy()
    best_obj = compute_obj(a_curr)

    for i in range(num_iters):
        alpha = 0.3 + 0.5 * i / max(1, num_iters)
        a_trial = freq_flatten(a_curr, alpha=alpha)
        obj_trial = compute_obj(a_trial)
        if obj_trial < best_obj:
            best_obj = obj_trial
            best_a = a_trial.copy()
        a_curr = a_trial

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] / new_n if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Sine
    s = np.sin(np.pi * x) + 0.1
    inits.append(s / s.sum())

    # Gaussian
    g = np.exp(-((x - 0.5) ** 2) / 0.08)
    inits.append(g / g.sum())

    # Bimodal
    b = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
    inits.append(b / b.sum())

    # Three-peak
    t = (np.exp(-((x - 0.2) ** 2) / 0.015) +
         np.exp(-((x - 0.5) ** 2) / 0.015) +
         np.exp(-((x - 0.8) ** 2) / 0.015))
    inits.append(t / t.sum())

    # One-sided
    one = np.zeros(n)
    one[:n // 2] = np.linspace(1, 0.3, n // 2)
    inits.append(one / one.sum())

    # Random
    r = rng.random(n) + 0.01
    inits.append(r / r.sum())

    # W-shaped
    w = np.abs(np.sin(2 * np.pi * x)) ** 0.5 + 0.1
    inits.append(w / w.sum())

    if parent is not None:
        inits.append(resample_sequence(parent, n))
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.08, n))
        inits.append(project_simplex(pert))
        # Mirror perturbation
        mirror = parent[::-1].copy()
        inits.append(resample_sequence(mirror, n))

    return inits

        if strat == 0 and use_solcp and n <= 1200:
            # SOCP trust region
            d = solcp_trust_region(a, r_trust, n_active=min(40, n // 8))
            if d is not None and np.max(np.abs(d)) > 1e-10:
                a_trial = project_simplex(a + d)
                obj_trial = compute_obj(a_trial)
                if obj_trial < current_obj:
                    a = a_trial
                    current_obj = obj_trial
                    r_trust = min(0.25, r_trust * 1.15)
                    stagnation = 0
                else:
                    r_trust = max(0.01, r_trust * 0.7)
                    stagnation += 1

        elif strat == 1:
            # Subgradient with momentum
            a, momentum_v = subgradient_momentum_step(a, conv, tau, step_size, momentum_v, beta)
            obj = compute_obj(a)
            if obj < current_obj:
                current_obj = obj
                step_size = min(0.08, step_size * 1.03)
                stagnation = 0
            else:
                step_size = max(0.005, step_size * 0.93)
                stagnation += 1

        elif strat == 2 and n <= 900:
            # LP alternating
            g = lp_alternate_step(a, time_limit=max(2, int((time_budget - (time.monotonic() - start)) * 0.3)))
            if g is not None:
                best_blend_obj = current_obj
                best_blend_a = a
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_trial = project_simplex((1 - t) * a + t * g)
                    obj_trial = compute_obj(a_trial)
                    if obj_trial < best_blend_obj:
                        best_blend_obj = obj_trial
                        best_blend_a = a_trial
                if best_blend_obj < current_obj:
                    a = best_blend_a
                    current_obj = best_blend_obj
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                # Fallback: subgradient
                a, momentum_v = subgradient_momentum_step(a, conv, tau, step_size, momentum_v, beta)
                stagnation += 1

        else:
            # Frequency domain flattening
            a_trial = freq_flatten(a, alpha=0.4)
            obj_trial = compute_obj(a_trial)
            if obj_trial < current_obj:
                a = a_trial
                current_obj = obj_trial
                stagnation = 0
            else:
                # Try subgradient instead
                a, momentum_v = subgradient_momentum_step(a, conv, tau, step_size, momentum_v, beta)
                stagnation += 1

        # Track best
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a.copy()
            last_best_time = time.monotonic()

        # Perturbation on stagnation
        if stagnation > 40:
            rng_local = np.random.default_rng(iteration)
            pert_strength = 0.03 + 0.02 * (stagnation // 20)
            a = a * np.exp(rng_local.normal(0, pert_strength, n))
            a = project_simplex(a)
            momentum_v = np.zeros(n)
            step_size = 0.04
            r_trust = 0.08
            stagnation = 0

        # Adaptive tau: increase when stagnating (more diverse subgradient)
        if stagnation > 20:
            tau = min(0.05, tau * 1.1)
        else:
            tau = max(0.001, tau * 0.99)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Diverse initialization screening with quick subgradient.
    Phase 2: Intensive optimization (SOCP + subgradient + LP + freq).
    Phase 3: Multi-length exploration.
    Phase 4: Final polish with reduced step sizes.
    """
    global start_run_time
    start_run_time = time.monotonic()
    deadline = start_run_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent = project_simplex(parent.copy())
        parent_obj = compute_obj(parent)
        print(f"Parent loaded (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        n_start = 700
        best_seq = np.ones(n_start) / n_start
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"No parent. Starting with uniform n={n_start}: obj={best_score:.8f}")

    n_base = len(best_seq)
    use_solcp = HAS_CVXPY and n_base <= 1200
    print(f"Config: n_base={n_base}, use_solcp={use_solcp}, budget={budget_s}s")

    # === Phase 1: Diverse initialization screening ===
    print("\n=== Phase 1: Initialization Screening ===")
    phase1_budget = min(100.0, (deadline - start_run_time) * 0.06)
    phase1_start = time.monotonic()

    n_options = [n_base]
    for delta in [-50, -30, 30, 50, 100, -100]:
        n_try = n_base + delta
        if 100 <= n_try <= 2000:
            n_options.append(n_try)
    n_options = list(dict.fromkeys(n_options))

    for n_try in n_options:
        if time.monotonic() - phase1_start > phase1_budget:
            break
        inits = generate_diverse_inits(n_try, rng, parent)
        for ii, init in enumerate(inits):
            if time.monotonic() - phase1_start > phase1_budget:
                break
            a_opt, obj_opt = quick_subgradient_optimize(init, num_steps=40, tau=0.004, step_size=0.05)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}, init{ii}: NEW BEST {best_score:.8f}")

    print(f"Phase 1 done. Best: {best_score:.8f}, n={len(best_seq)}")

    # === Phase 2: Intensive optimization ===
    print("\n=== Phase 2: Intensive Optimization ===")
    remaining2 = deadline - time.monotonic()
    phase2_budget = remaining2 * 0.55
    phase2_start = time.monotonic()

    # Main intensive optimization on the best sequence
    a_main = best_seq.copy()
    a_main, obj_main = intensive_optimize(a_main, time_budget=phase2_budget * 0.6,
                                          use_solcp=use_solcp)
    if obj_main < best_score:
        best_score = obj_main
        best_seq = a_main.copy()
        print(f"  Main intensive: NEW BEST {best_score:.8f}")

    # Additional intensive runs from slightly perturbed versions
    remaining2_now = deadline - time.monotonic()
    if remaining2_now > 30:
        sub_budget = min(remaining2_now * 0.35, 200.0)
        for pert_idx in range(3):
            if time.monotonic() - phase2_start > phase2_budget:
                break
            pert = best_seq * np.exp(rng.normal(0, 0.04, len(best_seq)))
            pert = project_simplex(pert)
            a_p, obj_p = intensive_optimize(pert, time_budget=sub_budget / 3.0,
                                            use_solcp=use_solcp)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Perturb {pert_idx}: NEW BEST {best_score:.8f}")

    print(f"Phase 2 done. Best: {best_score:.8f}")

    # === Phase 3: Multi-length exploration ===
    print("\n=== Phase 3: Multi-length Exploration ===")
    remaining3 = deadline - time.monotonic()
    if remaining3 > 50:
        phase3_budget = min(remaining3 * 0.25, 200.0)
        phase3_start = time.monotonic()
        n_best = len(best_seq)

        n_candidates = [n_best + d for d in [20, -20, 40, -40, 60, -60, 80, -80]]
        n_candidates = [n for n in n_candidates if 100 <= n <= 2000]
        n_candidates = list(dict.fromkeys(n_candidates))

        per_n_budget = max(5.0, phase3_budget / max(1, len(n_candidates)))

        for n_try in n_candidates:
            if time.monotonic() - phase3_start > phase3_budget:
                break
            a_try = resample_sequence(best_seq, n_try)
            a_try = project_simplex(a_try)
            a_opt, obj_opt = intensive_optimize(a_try, time_budget=per_n_budget,
                                                use_solcp=(HAS_CVXPY and n_try <= 1000))
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f} (no improvement)")

    # === Phase 4: Final polish ===
    print("\n=== Phase 4: Final Polish ===")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 10 and best_seq is not None:
        a_final = best_seq.copy()
        n_final = len(a_final)
        polish_start = time.monotonic()
        polish_budget = max(5.0, remaining4 - 5.0)

        # Fine-grained subgradient with very small steps
        momentum_v = np.zeros(n_final)
        for pit in range(min(300, int(polish_budget * 5))):
            if time.monotonic() - polish_start > polish_budget:
                break
            conv = compute_conv(a_final)
            tau_p = 0.0005 + 0.002 * (pit / 300.0)
            step_p = 0.015 * (1.0 - pit / 400.0)
            a_final, momentum_v = subgradient_momentum_step(
                a_final, conv, tau_p, max(0.001, step_p), momentum_v, 0.8)
            obj = compute_obj(a_final)
            if obj < best_score:
                best_score = obj
                best_seq = a_final.copy()

        # Final LP if size allows
        if n_final <= 800 and time.monotonic() - polish_start < polish_budget - 5:
            g = lp_alternate_step(a_final, time_limit=max(3, int(polish_budget - (time.monotonic() - polish_start) - 5)))
            if g is not None:
                for t in [0.1, 0.2, 0.3, 0.5]:
                    a_t = project_simplex((1 - t) * a_final + t * g)
                    obj_t = compute_obj(a_t)
                    if obj_t < best_score:
                        best_score = obj_t
                        best_seq = a_t.copy()
                        a_final = a_t

        print(f"  Polish done. Best: {best_score:.8f}")

    # Final output
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
        print(f"\n=== FINAL: n={len(result)}, score={val:.8f} ===")
    else:
        result = [1.0] * 100
        print(f"\n=== FALLBACK: uniform n=100, score={compute_obj(np.ones(100)):.4f} ===")

    elapsed = time.monotonic() - start_run_time
    print(f"Total time: {elapsed:.1f}s")

    return result
