    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

"""
Optimized search for non-negative step function minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses enhanced LP alternating with cutting planes, FFT-based subgradient with log-sum-exp
smoothing, frequency-domain shaping, and adaptive length exploration.
Key improvements: (1) cutting-plane LP for faster solves, (2) FFT-based O(n log n) gradients,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def obj_val(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        ks = j
        ke = min(j + n, 2 * n - 1)
        A[ks:ke, j] = a[:ke - ks]
    return A


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def lp_minpeak(a, time_limit=20.0):

    n = len(a)

    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            s = g.sum()

    return None, np.inf

def subgradient_logsumexp(a, max_iters=300, time_budget=30.0, tau_init=5.0, tau_end=200.0):
    """Subgradient descent on log-sum-exp smoothed objective with Nesterov momentum."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.85
    step = 0.05
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        tau = tau_init * (tau_end / tau_init) ** (float(it) / max(1, max_iters - 1))
        conv = compute_conv(a)

        # Softmax weights (numerically stable)
        shifted = conv - np.max(conv)
        w = np.exp(tau * shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            w = np.zeros(len(conv))
            w[idx] = 1.0
        else:
            w = w / w_sum

        # Gradient via FFT: grad[i] = 4n/tau * sum_j w[i+j]*a[j]
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, N_fft)
        fw = np.fft.fft(w, N_fft)
        conv_aw = np.fft.ifft(fa_rev * fw).real
        grad = conv_aw[n - 1:2 * n - 1] * (4.0 * n / tau)

        # Center to preserve sum
        grad_c = grad - np.mean(grad)

        # Nesterov momentum
        v = mu * v - step * grad_c
        a_new = normalize(a + v)

        obj_new = obj_val(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        step = max(0.001, step * 0.997)

    return best_a, best_obj


def freq_shaping(a, max_iters=100, time_budget=20.0, alpha_start=0.5, alpha_end=0.9):
    """Iteratively flatten the power spectrum to reduce convolution peaks."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()

    while True:
        N = 1
        while N < 2 * n:
            N <<= 1
        break

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        alpha = alpha_start + (alpha_end - alpha_start) * (float(it) / max(1, max_iters - 1))
        fa = np.fft.fft(a, N)
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Compress magnitude
        mag_max = np.max(mag)
        if mag_max < 1e-15:
            break
        mag_norm = mag / mag_max
        mag_compressed = mag_norm ** alpha * mag_max

        fa_new = mag_compressed * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = normalize(a_new)

        obj_new = obj_val(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def chase_peak(a, n_steps=40, step_size=0.03):
    """Iteratively chase and reduce the maximum convolution peak."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = obj_val(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += a[j]
            j2 = i + 2 * n - 2 - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]

        grad_c = grad - np.mean(grad)
        for ss in [step_size, step_size * 0.5, step_size * 2.0]:
            a_try = normalize(a - ss * grad_c)
            obj_try = obj_val(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a, best_obj


def simulated_annealing_fast(a, rng, max_iters=400, T_init=0.3, T_min=0.005, time_budget=60.0):
    """Simulated annealing with multiple move types."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    cur_a = a.copy()
    cur_obj = obj_val(a)
    best_a = a.copy()
    best_obj = cur_obj
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T_init * (T_min / T_init) ** (float(it) / max(1, max_iters))
        mt = rng.integers(0, 3)
        a_t = cur_a.copy()

        if mt == 0:
            ws = max(10, n // 12)
            ws_start = int(rng.integers(0, n - ws))
            noise = rng.normal(0, 0.3, ws) * (np.mean(a_t[ws_start:ws_start + ws]) + 0.01)
            a_t[ws_start:ws_start + ws] += noise
            a_t = normalize(a_t)
        elif mt == 1:
            ws = max(10, n // 20)
            w1 = int(rng.integers(0, n - ws))
            w2 = int(rng.integers(0, n - ws))
            if abs(w1 - w2) >= ws:
                a_t[w1:w1 + ws], a_t[w2:w2 + ws] = a_t[w2:w2 + ws].copy(), a_t[w1:w1 + ws].copy()
                a_t = normalize(a_t)
        else:
            # Add a small perturbation weighted by current peaks
            conv = compute_conv(a_t)
            weights = conv / conv.sum()
            pert = np.zeros(n)
            for k in range(min(5, len(weights))):
                idx_k = int(np.argsort(weights)[-k - 1])
                for i in range(n):
                    j = idx_k - i
                    if 0 <= j < n:
                        pert[i] += weights[idx_k] * a_t[j]
            a_t = normalize(a_t - 0.05 * (pert - np.mean(pert)))

        obj_t = obj_val(a_t)
        delta = obj_t - cur_obj
        if delta < 0 or rng.random() < np.exp(-max(0, delta) / max(T, 1e-8)):
            cur_a = a_t
            cur_obj = obj_t
            if obj_t < best_obj:
                best_obj = obj_t
                best_a = a_t.copy()

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    x2 = np.linspace(-3, 3, n)
    gauss = np.exp(-x2 ** 2 / 3.0)
    inits.append(gauss / gauss.sum())

    rnd = rng.random(n) + 0.05
    inits.append(rnd / rnd.sum())

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Raised cosine variant
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # Two-peak
    bp = np.exp(-((x - 0.3) ** 2) / 0.03) + np.exp(-((x - 0.7) ** 2) / 0.03)
    inits.append(bp / bp.sum())

    if parent is not None:
        resampled = resample_seq(parent, n)
        inits.append(normalize(resampled))
        pert = resampled * np.exp(rng.normal(0, 0.1, n))
        inits.append(normalize(pert))

    return inits


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining LP alternating with cutting planes,
    FFT-based subgradient with log-sum-exp smoothing, frequency-domain shaping,
    simulated annealing, and adaptive length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    parent = try_load_parent()
    if parent is not None:
        p_obj = obj_val(parent)
        print(f"Parent: n={len(parent)}, obj={p_obj:.8f}")
        best_score = p_obj
        best_seq = normalize(parent.copy())
    else:
        best_seq = np.ones(700)
        best_score = obj_val(best_seq)
        print(f"No parent, starting uniform n=700, obj={best_score:.8f}")

    # === Phase 1: Quick screening (120s) ===
    print("\n=== Phase 1: Screening ===")
    p1_end = start_time + min(120.0, (deadline - start_time) * 0.08)
    n_options = [600, 650, 700, 750, 800, 900, 1000]
    n_options = [n for n in n_options if 100 <= n <= 2000]

    candidates = []
    for n_try in n_options:
        if time.monotonic() > p1_end:
            break
        inits = generate_inits(n_try, rng, best_seq if best_seq is not None else None)
        for init in inits:
            if time.monotonic() > p1_end:
                break
            a_q, obj_q = chase_peak(init, n_steps=15, step_size=0.03)
            if np.isfinite(obj_q):
                candidates.append((obj_q, a_q))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best: {candidates[0][0] if candidates else 'N/A'}")

    for obj_c, a_c in candidates[:5]:
        if obj_c < best_score:
            best_score = obj_c

        n_cur = len(a_init)
