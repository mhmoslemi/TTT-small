def toeplitz_solve(w, rhs):
    """Solve M x = rhs where M_{i,j} = w_{i+j} (i,j = 0,...,n-1)."""
    n = len(rhs)
    # Use Levinson-Durbin or just numpy's solve for moderate n
    M = np.zeros((n, n))
    for s in range(2*n - 1):
        for i in range(max(0, s-n+1), min(s+1, n)):
            M[i, s-i] = w[s]

def compute_conv(a):

        size <<= 1
    fa = rfft(a, size)

        return np.inf

    max_b = float(np.max(conv))

    s = np.sum(a)

        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a

        i_max = min(s + 1, n)
        for i in range(i_min, i_max):

            M[i, j] = w[s]

        x = np.linalg.solve(M + 1e-10 * np.eye(n), rhs)
    except np.linalg.LinAlgError:
        x, _, _, _ = np.linalg.lstsq(M + 1e-10 * np.eye(n), rhs, rcond=None)
    
    # Project onto non-negative
    x = np.maximum(x, 0)
    
    # The target direction
    return x

    S = float(np.sum(a))
    conv = compute_conv(a)

    xn = tau * conv / (S * S)
    xn_max = np.max(xn)
    w = np.exp(xn - xn_max)
    w_sum = np.sum(w)
    w = w / w_sum
    
    # Cross-correlation via FFT
    w_flip = w[::-1]  # length 2n-1
    fft_size = 1
    while fft_size < n + (2*n-1) - 1:  # = 3n - 2
        fft_size <<= 1
    
    A = rfft(a, fft_size)
    Wf = rfft(w_flip, fft_size)
    conv_result = irfft(A * Wf, fft_size)
    
    G = conv_result[n-1 : 2*n-1][::-1]  # length n
    H = float(np.dot(w, conv))
    
    # Gradient of smoothed objective
    # dF/da_i = (4n/S^2) * (G_i - H/S)
    grad = (4.0 * n / (S * S)) * (G - H / S)

"""
Dual-ascent smoothed optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key ideas: (1) Log-sum-exp smoothing with FFT-based gradients, (2) exact dual
fixed-point corrections via Toeplitz linear solves, (3) adaptive temperature
annealing, (4) multi-start diverse exploration, (5) simulated annealing for
escaping local optima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def compute_grad_smoothed(a, tau):
    n = len(a)
    S = float(np.sum(a))
    conv = compute_conv(a)
    xn = tau * conv / (S * S)
    xn_max = np.max(xn)
    w = np.exp(xn - xn_max)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
    else:
        w /= w_sum
    w_flip = w[::-1]
    fft_size = 1
    while fft_size < 3 * n - 2:
        fft_size <<= 1
    A = rfft(a, fft_size)
    Wf = rfft(w_flip, fft_size)
    conv_result = irfft(A * Wf, fft_size)
    G = conv_result[n - 1:2 * n - 1][::-1]
    H = float(np.dot(w, conv))
    grad = (4.0 * n / (S * S)) * (G - H / S)
    return grad, w, conv, H


def build_toeplitz_M(w):
    n = (len(w) + 1) // 2
    M = np.zeros((n, n))
    for s in range(len(w)):
        i_min = max(0, s - n + 1)
        i_max = min(s + 1, n)
        for i in range(i_min, i_max):
            j = s - i
            M[i, j] = w[s]
    return M


def dual_correction(a, tau, reg=1e-8):
    n = len(a)
    S = float(np.sum(a))
    conv = compute_conv(a)
    xn = tau * conv / (S * S)
    xn_max = np.max(xn)
    w = np.exp(xn - xn_max)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        return a.copy()
    w /= w_sum
    M = build_toeplitz_M(w)
    rhs = np.ones(n)
    try:
        x = np.linalg.solve(M + reg * np.eye(n), rhs)
    except np.linalg.LinAlgError:
        x, _, _, _ = np.linalg.lstsq(M + reg * np.eye(n), rhs, rcond=None)
    x = np.maximum(x, 0.0)
    s = np.sum(x)
    if s < 1e-15:
        return a.copy()
    return x / s


def smooth_optimize(a, time_budget, tau_start=2000.0, tau_end=20000.0,
                    momentum=0.7, step_init=0.15, dual_every=15,
                    perturb_every=40, seed=42):
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    step = step_init
    rng = np.random.default_rng(seed)
    tau = tau_start
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        it += 1
        frac = min(it / 500.0, 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac

        grad, w, conv, H = compute_grad_smoothed(a, tau)
        grad_centered = grad - np.mean(grad)
        gn = np.linalg.norm(grad_centered)
        if gn < 1e-14:
            break

        v = momentum * v - step * grad_centered / gn
        a_new = a + v
        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a_try = a - 0.5 * step * grad_centered / gn
            a_try = normalize_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                stall = 0
            else:
                step *= 0.9
                stall += 1

        step = min(step_init, step * 1.01)

        if dual_every > 0 and it % dual_every == 0:
            a_dual = dual_correction(a, tau)
            for t_blend in [0.3, 0.5, 0.7, 1.0]:
                a_mix = (1.0 - t_blend) * a + t_blend * a_dual
                a_mix = normalize_simplex(a_mix)
                obj_mix = compute_obj(a_mix)
                if obj_mix < best_obj - 1e-13:
                    best_obj = obj_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    break

        if stall > perturb_every and it % 3 == 0:
            scale = 0.02 + 0.03 * min(stall - perturb_every, 10) / 10.0
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = normalize_simplex(a_pert)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_obj or rng.random() < 0.3:
                a = a_pert
            stall = 0

    return best_a, best_obj

def gen_diverse_starts(n, num=10, seed=42):
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    starts.append(np.ones(n) / n)

    # Triangular
    tri = 1.0 - np.abs(i - center) / (center + 1)
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    # Gaussian
    for sig_frac in [0.15, 0.3, 0.5]:
        bell = np.exp(-((i - center) ** 2) / (2 * (n * sig_frac) ** 2))
        starts.append(bell / np.sum(bell))

    # U-shaped (low middle, high edges)
    u_shape = np.exp(((i - center) ** 2) / (2 * (n * 0.3) ** 2))
    starts.append(u_shape / np.sum(u_shape))

    # Sinusoidal modulations
    for freq in [1, 2, 3]:
        s = 0.5 + 0.5 * np.cos(2 * np.pi * freq * i / n)
        s = np.maximum(s, 0.01)
        starts.append(s / np.sum(s))

    # Random smooth
    for _ in range(3):
        noise = rng.normal(0, 1, n)
        # Smooth via cumulative
        smooth = np.convolve(noise, np.ones(n // 5 + 1) / (n // 5 + 1), mode='same')
        s = np.exp(smooth / (np.std(smooth) + 1e-10))
        starts.append(s / np.sum(s))

    return starts[:num]


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
