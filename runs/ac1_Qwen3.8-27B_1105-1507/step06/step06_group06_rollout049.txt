from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project(a):
    """Project to non-negative, sum=1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def subgradient_step(a, lr, momentum_state=None, beta=0.9):
    """One step of projected subgradient with momentum."""
    n = len(a)
    c = compute_conv(a)
    m_star = int(np.argmax(c))
    
    # Subgradient: g[j] = 2*a[m_star - j] for valid j
    g = np.zeros(n)
    # a[m_star - j] is valid when 0 <= m_star - j < n, i.e., m_star - n + 1 <= j <= m_star
    j_start = max(0, m_star - n + 1)
    j_end = min(n - 1, m_star)
    if j_end >= j_start:
        indices = np.arange(j_start, j_end + 1)
        a_indices = m_star - indices
        g[indices] = 2.0 * a[a_indices]
    
    # Center to preserve sum
    g_centered = g - np.mean(g)
    
    if momentum_state is None:
        v = -lr * g_centered
    else:
        v = beta * momentum_state - lr * g_centered
    
    a_new = a + v  # Note: we're MINIMIZING, so we go in the direction of -gradient
    # Wait, the subgradient gives us a direction of INCREASE for c[m_star].
    # We want to DECREASE c[m_star], so we step in the -g direction.
    # a_new = a - lr * g_centered (without momentum)
    # With momentum: a_new = a + v where v = beta*v_old - lr*g_centered
    
    a_new = project(a_new)
    return a_new, v

def chase_peak_subgradient(a, max_iters=10000, tau=0.5, time_budget=60.0):
    """Chase the peak using subgradient with adaptive step size."""
    n = len(a)
    a = project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    lr = 0.1
    v = np.zeros(n)
    beta = 0.7  # momentum
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        c = compute_conv(a)
        m_star = int(np.argmax(c))
        
        # Subgradient
        g = np.zeros(n)
        j_start = max(0, m_star - n + 1)
        j_end = min(n - 1, m_star)
        if j_end >= j_start:
            indices = np.arange(j_start, j_end + 1)
            a_indices = m_star - indices
            g[indices] = 2.0 * a[a_indices]
        
        g_centered = g - np.mean(g)
        
        # Momentum step
        v = beta * v - lr * g_centered
        a_new = a + v
        a_new = project(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            # Occasionally increase learning rate
            if it % 100 == 0:
                lr = min(lr * 1.05, 0.5)
        else:
            # Backtrack
            lr *= 0.8
            v = beta * v  # Don't update v fully
            # Try a smaller step
            a_try = a - lr * g_centered
            a_try = project(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
        
        # Annealing: reduce momentum over time
        beta = 0.7 + 0.2 * (it / max_iters)
    
    return best_a, best_obj
