"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nn_sum1(a):
    """Project to {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def baby_step_giant_step(a, max_iters=200, tau_init=0.01, tau_min=0.001, 
                         step_init=0.1, time_budget=30, n_smooth=50):
    """
    Baby-step giant-step method for minimizing max(conv(a,a)) subject to a>=0, sum(a)=1.
    Uses smoothed max as surrogate.
    """
    n = len(a)
    a_cur = project_nn_sum1(a.copy().astype(np.float64))
    best_a = a_cur.copy()
    best_obj = compute_obj(a_cur)
    y = a_cur.copy()  # running average
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        tau = tau_init * (tau_min / tau_init) ** (it / max_iters)
        
        # Compute gradient of smoothed max
        grad, cmax = smoothed_max_gradient(a_cur, tau)
        
        # Center gradient (to maintain sum=1 constraint)
        grad_centered = grad - np.mean(grad)
        
        # Baby step
        step = step_init * (1.0 + it / max_iters)  # gradually increase
        a_baby = a_cur - step * grad_centered
        a_baby = project_nn_sum1(a_baby)
        
        # Giant step: running average
        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * a_baby
        
        # Check objective of both a_baby and y
        obj_baby = compute_obj(a_baby)
        obj_y = compute_obj(y)
        
        if obj_baby < best_obj:
            best_obj = obj_baby
            best_a = a_baby.copy()
            a_cur = a_baby
        elif obj_y < best_obj:
            best_obj = obj_y
            best_a = y.copy()
            a_cur = y.copy()
        else:
            a_cur = a_baby  # continue from baby step even if not better
    
    return best_a, best_obj

def freq_waterfilling(a, max_iters=100, time_budget=25.0):
    """
    Frequency domain water-filling: iteratively reduce peaks in the convolution
    by modifying the magnitude of the FFT.
    """
    n = len(a)
    a = project_nn_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, N)
        power = np.abs(A) ** 2
        C = ifft(power).real[:2*n-1]
        
        c_max = np.max(C)
        c_mean = np.mean(C)
        
        if c_max <= c_mean * 1.005:
            break
        
        # Water-filling: reduce peaks in C, then verify FFT(C_new) >= 0
        # Try different levels of peak reduction
        for blend in [0.3, 0.5, 0.7]:
            C_new = C.copy()
            # Soft threshold
            threshold = c_mean + blend * (c_max - c_mean)
            C_new[C_new > threshold] = threshold
            
            # Check validity: FFT(C_new) should be >= 0
            P_new = fft(C_new, N).real
            if np.min(P_new) < -0.01 * np.max(P_new):
                continue
            
            # Reconstruct: keep phase of A, use new magnitude
            P_new = np.maximum(P_new, 0)
            phase = np.angle(A)
            mag_new = np.sqrt(P_new)
            A_new = mag_new * np.exp(1j * phase)
            a_new = ifft(A_new).real[:n]
            
            # Project
            a_new = project_nn_sum1(a_new)
            obj_new = compute_obj(a_new)
            
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                break
        
    return best_a, best_obj

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nn_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=10):
    n = len(a)
    if n > 2500:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0 or res.status == 3:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, res
    except Exception:
        pass
    return None, None

def freq_irls(a, max_iters=80, time_budget=25.0):
    """
    Iteratively Reweighted Least Squares in frequency domain.
    Reweights the power spectrum to reduce convolution peaks.
    """
    n = len(a)
    a = project_nn_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, N)
        power = np.abs(A) ** 2
        C = ifft(power).real[:2 * n - 1]

        c_max = np.max(C)
        if c_max < 1e-15:
            break

        # IRLS weights: downweight positions with high convolution values
        c_norm = C / c_max
        # Weight: lower where C is high
        alpha = 0.3 + 0.4 * (it / max_iters)
        w = 1.0 / (c_norm ** alpha + 1.0)

        # Apply weights in frequency domain
        # The relationship: C = IFFT(|A|^2), so we want to modify |A|^2
        # New target: C_new = C * w (element-wise)
        # |A_new|^2 = FFT(C_new)
        C_new = C * w
        C_new = np.maximum(C_new, 0)

        P_new = fft(C_new, N).real
        if np.min(P_new) < -0.05 * np.max(P_new):
            # Invalid, try softer reweighting
            w_soft = 1.0 / (c_norm ** (alpha * 0.5) + 1.0)
            C_new = C * w_soft
            C_new = np.maximum(C_new, 0)
            P_new = fft(C_new, N).real
            if np.min(P_new) < -0.05 * np.max(P_new):
                continue
            P_new = np.maximum(P_new, 0)

        # Reconstruct with phase preserved
        phase = np.angle(A)
        mag_new = np.sqrt(np.maximum(P_new, 0))
        A_new = mag_new * np.exp(1j * phase)
        a_new = ifft(A_new).real[:n]
        a_new = project_nn_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def smoothed_max_descent(a, max_iters=300, time_budget=30.0):
    """
    Gradient descent on the smoothed max of conv(a,a) using baby-step giant-step.
    Adaptive temperature schedule.
    """
    n = len(a)
    a = project_nn_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    y = a.copy()
    start = time.monotonic()

    N = 1
    while N < 3 * n - 2:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive temperature
        tau = 0.005 + 0.1 * (1.0 - it / max_iters) ** 2

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -500.0, 0.0)
        weights = np.exp(shifted)
        ws = np.sum(weights)
        if ws < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= ws

        # Gradient via correlation
        fw = fft(weights, N)
        fa = fft(a, N)
        corr = ifft(fw * np.conj(fa)).real[:n]
        grad = 2.0 * corr - 2.0 * np.mean(corr)

        # Baby step with adaptive step size
        step = 0.05 * (1.0 + 0.5 * it / max_iters)
        a_baby = a - step * grad
        a_baby = project_nn_sum1(a_baby)

        # Giant step (running average)
        alpha_gs = 2.0 / (it + 2.0)
        y = (1.0 - alpha_gs) * y + alpha_gs * a_baby

        # Evaluate
        obj_baby = compute_obj(a_baby)
        obj_y = compute_obj(y)

        if obj_baby < best_obj:
            best_obj = obj_baby
            best_a = a_baby.copy()
            a = a_baby
        elif obj_y < best_obj:
            best_obj = obj_y
            best_a = y.copy()
            a = y.copy()
        else:
            a = a_baby

    return best_a, best_obj


def freq_perturb(a, rng, strength=0.05):
    """
    Apply a small frequency-domain perturbation to escape local minima.
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    A = fft(a, N)
    # Add small random phase perturbation
    phase_perturb = rng.normal(0, strength, N)
    # Add small random magnitude perturbation
    mag_perturb = 1.0 + rng.normal(0, strength * 0.5, N)

    A_new = A * mag_perturb * np.exp(1j * phase_perturb)
    a_new = ifft(A_new).real[:n]
    a_new = project_nn_sum1(a_new)
    return a_new


def coordinate_descent(a, time_budget=15.0):
    """
    Coordinate descent: optimize one or two coordinates at a time.
    Uses the linearity of conv in each coordinate for efficiency.
    """
    n = len(a)
    a = project_nn_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        # Pick a random coordinate (or cycle through)
        i = it % n

        base = a[i]
        sum_others = 1.0 - base

        # Try candidate values for a[i], renormalizing the rest
        candidates = [base * 0.5, base * 0.8, base * 1.2, base * 1.5,
                      base * 0.2, 0.0, base * 2.0]
        candidates = [c for c in candidates if 0 <= c <= 1.0]

        best_local_obj = best_obj
        best_local_a = best_a

        for c in candidates:
            if abs(c - base) < 1e-10:
                continue
            a_trial = a.copy()
            # Set a[i] = c, renormalize others to maintain sum=1
            remaining = 1.0 - c
            if remaining < 0:
                continue
            a_trial[i] = c
            if sum_others > 1e-15:
                scale = remaining / sum_others
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] *= scale
            else:
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] = 0

            a_trial = project_nn_sum1(a_trial)
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_local_obj:
                best_local_obj = obj_trial
                best_local_a = a_trial.copy()

        if best_local_obj < best_obj:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a
            conv = compute_conv(a)

    return best_a, best_obj

def generate_inits(n, rng, parent=None):
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    x_g = np.linspace(-3, 3, n)
    gauss = np.exp(-x_g ** 2 / 3)
    inits.append(gauss / gauss.sum())

    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Cosine window
    cosw = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.1
    inits.append(cosw / cosw.sum())

    # Two-peak
    p1, p2 = int(n * 0.3), int(n * 0.7)
    tp = np.exp(-((x - p1 / n) ** 2) / 0.01 + (x - p2 / n) ** 2 / 0.01)
    tp = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
    inits.append(tp / tp.sum())

    if parent is not None:
        p_res = resample_sequence(parent, n)
        inits.append(project_nn_sum1(p_res))
        p_pert = p_res * np.exp(rng.normal(0, 0.08, n))
        inits.append(project_nn_sum1(p_pert))
        p_pert2 = p_res * np.exp(rng.normal(0, 0.15, n))
        inits.append(project_nn_sum1(p_pert2))

    return inits
