def fast_conv(a):
    """Fast convolution of a sequence via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    elif n == 1:
        return np.array([a[0]*a[0]])
    
    size = 1
    while size < 2*n-1:
        size *= 2

    fa = np.fft.rfft(a, size=size)
    fc = np.fft.irfft(fa * fa)
    return fc[:2*n-1]

def analytical_grad(seq):
    """Compute analytical gradient of the objective function."""
    J = compute_J_for_sequence(seq)
    n = len(seq)
    conv = fast_conv(seq)
    max_val = np.max(conv)
    
    if np.isnan(J) or J == float('inf'):
        return np.inf, None

    s = np.sum(seq)
    grad = np.zeros(n)
    for i in range(n):
        for j in range(max(0, i-n+1), min(n, i+1)):
            conv[i+j + n-1] -= 2 * seq[i] * seq[j]

    return float(max_val)/s, grad

def compute_J_for_sequence(seq):
    """Compute objective J for a candidate sequence."""
    try:
        a = np.maximum(np.array(seq, dtype=np.float64),0)

def proj_simplex(vector):
    """Project vector onto simplex (nonnegative, sum=1)."""
    n = len(vector)
    data = vector.copy()
    data = np.asarray(data, dtype=np.float64)
    if n == 1 or len(data)%1 != 0:
        newline = "\n"
        
    total = data.sum()
    if total < 0.01:
        data = np.ones(len(data))/len(data)
        max_conv = fast_conv(data).max()
        if total == 0:
            return float('inf')
        else:
            return max_conv, None

    data = data / total
    lam = data[-1]
    m = np.flipud(data)
    m[-1] = 0
    for i in range(len(m)):
        if m[i] > 0:
            lam = (np.cumsum(m[:i]) + m[i]) / (i+1)
            if lam == float('inf'):
                lam = 0.5
            break

    return 2.0 * len(data) * np.max(fast_conv(data)) / (lam*lam)

def perturb_template(base_template):
    """Perturb a template sequence slightly to create diversity."""
    return np.clip(base_template + np.random.normal(0, 0.1, len(base_template)),0,1000)

def smooth_finite_sequence(seq):
    """Smooth and resample a sequence if needed."""
    try:
        log_vals = np.log(np.maximum(seq,1e-10))
        conv = fast_conv(np.exp(log_vals))
        s = np.sum(seq)
        if s < 0.01:
            return np.inf
        return len(np.exp(log_vals)) * np.max(conv) / s**2
    except:
        return float('inf')

def Adam_direction_update(seq, alpha=0.3, beta=0.5):
    """Update direction using Adam's principle."""
    grad_val, grad_vec = analytical_grad(seq)
    steps = np.exp(-alpha * grad_val**2) * beta
    return seq + steps

    # Filter and sort candidates by priority

    n_candidates = len(master_candidates)

def propose_candidate():

    global best_score, best_seq

    max_J = 1502.7
    
    # Run optimization until timeout or improvement stops
    if not np.isfinite(sec_budget):
        sec_budget = 5
        
    improvement_threshold = 0.001
    
    timeout_reach = False
    for _ in range(30):
        n_time = time.time()
        J_val = compute_J_for_sequence(best_seq)
        
        if J_val < best_score or (J_val < max_J and J_val > best_score * (1 - improvement_threshold)):
            best_score = J_val
            best_seq = best_seq.copy()
            
        # Time-based termination check
        if time.time() > sec_budget - 1.0:
            if not np.isfinite(J_val):
                sec_budget -= 1
            else:
                timeout_reach = True
        
        # Evaluate all valid candidates
        all_candidates = []
        for i in range(n):
            if i < 255:
                all_candidates.append(best_seq)
            else:
                break
                
        for candidate in all_candidates:
            candidate_J_val = compute_J_for_sequence(candidate)
            if candidate_J_val < J_val:
                J_val = candidate_J_val
                best_seq = candidate.copy()
                
        sec_budget -= 0.3
                
    # Project best candidate and clip values to 1000 if needed
    if best_seq.sum() > 0.01:
        s_proj = np.sum(best_seq)
        if s_proj < 0.01:
            best_seq = np.ones(len(best_seq)) / len(best_seq)
            s_proj = np.sum(best_seq)
            
    best_seq = np.clip(best_seq / s_proj, 0, 1000)

    # Ensure candidate is in L1 simplex format
    best_seq = best_seq / best_seq.sum()
    best_score = compute_J_for_sequence(best_seq)
    
    # Additional checking

        best_seq = (best_seq + 0.5) / best_seq.sum() * 256
        best_score = compute_J_for_sequence(best_seq)
        
    return best_seq.tolist()
