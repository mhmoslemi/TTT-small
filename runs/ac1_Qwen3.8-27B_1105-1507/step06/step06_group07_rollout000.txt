def toeplitz_row(a, k):

    # Actually (a*a)[k] = sum_i a[i] * a[k-i]

    n = len(a)
    a = np.maximum(a, 0)
    s = np.sum(a)

    else:

L-BFGS-B based optimization with subgradient peak reduction,
LP refinement, and multi-structure initialization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses annealed smoothing, structured
initializations (plateau, cosine, Gaussian, Hann-flat), and iterative
LP + gradient cycles.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_and_grad(a, beta):
    """Compute smoothed objective (logsumexp of beta*conv) and its gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


class SmoothObj:

    def __init__(self, beta):
        self.beta = beta

    def __call__(self, x):
        return smooth_obj_and_grad(x, self.beta)[0]

    def grad(self, x):
        return smooth_obj_and_grad(x, self.beta)[1]


def l_bfgsb_anneal(a, time_budget=60.0, beta_start=10.0, beta_end=500.0,
                   max_inner_iters=150):
    """
    L-BFGS-B optimization with annealed temperature.
    Iteratively increases beta and re-optimizes.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    num_rounds = max(3, int(time_budget / 5))

    for r in range(num_rounds):
        if time.monotonic() - start > time_budget:
            break
        progress = r / max(num_rounds - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        func = SmoothObj(beta)
        bounds = [(1e-12, None)] * n

        try:
            res = minimize(
                func, a, jac=func.grad, method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': max_inner_iters, 'maxfun': max_inner_iters * 3,
                         'gtol': 1e-5}
            )
            a_new = project_simplex(res.x)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
        except Exception:
            pass

    return best_a
