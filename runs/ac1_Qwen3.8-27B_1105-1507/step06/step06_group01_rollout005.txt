    """Project onto simplex {x >= 0, sum(x) = 1}."""

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated FISTA with log-sum-exp smooth approximation
2. LP best-response with adaptive active sets

Key improvement: O(n log n) gradient computation enables 10x more iterations.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto probability simplex in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not (~cond).any():
        rho = n - 1
    else:
        rho = int(np.argmax(~cond))
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad_fft(a, tau):
    """Compute smooth max of conv(a,a) and gradient via FFT. O(n log n).
    
    Returns (smooth_value, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.minimum(shifted, 500.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    smooth_val = b_max + np.log(exp_shifted.sum()) / tau
    
    # Gradient via FFT cross-correlation
    a_rev = a[::-1]
    fft_size = 1
    while fft_size < n + (2 * n - 1):
        fft_size <<= 1
    fa = np.fft.rfft(a_rev, fft_size)
    fp = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(fa * fp, fft_size)
    grad = 2.0 * conv_result[n - 1: n - 1 + n]
    
    return smooth_val, grad


def run_fista(a_init, deadline, tau_start=50.0, tau_end=2000.0, max_iters=500000, lr=0.3):
    """FISTA with FFT-accelerated gradients and adaptive temperature."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    it = 0
    last_check = 0
    
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_grad_fft(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        # Adaptive step size
        step = lr / (1.0 + 0.01 * tau)
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        it += 1
        
        # Periodic check
        if it - last_check >= 100:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
            last_check = it
    
    # Final check
    M_a = fast_conv(a).max()
    if M_a < best_M:
        best_M = M_a
        best_a = a.copy()
    M_y = fast_conv(y).max()
    if M_y < best_M:
        best_M = M_y
        best_a = y.copy()
    
    return best_a


def lp_best_response(a, M, n_active=30, time_limit=10):
    """Solve LP best response with top n_active constraints."""
    n = len(a)
    b = fast_conv(a)
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_indices):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            m = k - j
            rows.append(idx)
            cols.append(j)
            vals.append(a[m])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_descent(a_init, deadline, n_active_start=40, max_iters=100):
    """LP mirror descent with adaptive active set."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    cur_active = n_active_start
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = lp_best_response(a, best_M, cur_active, time_limit=8)
        if g is None:
            cur_active = min(cur_active * 2, 2 * n - 1)
            no_improve += 1
            if no_improve > 5:
                break
            continue
        
        improved = False
        for t in [0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
            cur_active = max(10, int(cur_active * 0.7))
        else:
            no_improve += 1
            cur_active = min(int(cur_active * 1.5), 2 * n - 1)
            if no_improve > 10:
                break
    
    return best_a

                            j = k - i
                            if 0 <= j < n:

                    a[i] = x_try
                    sweep_improved = True
                continue
            
            # Golden section search for optimal x_i
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Precompute
            c_vals = np.zeros(k_hi - k_lo + 1)
            s_vals = np.zeros(k_hi - k_lo + 1)

            def obj_1d(x):
                vals = c_vals + s_vals * x
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if has_quad:
                    qv = c_vals[quad_idx] + x * x
                    if qv > m:
                        m = qv
                return m
            
            # Search range
            lo = 0.0
            hi = max(3.0 * x_cur, np.sqrt(best_M) + 0.5)
            
            # Golden section

            fb = obj_1d(b_pt)

def simulated_annealing(a_init, deadline, n_perturb=5, seed=42):
    """Simulated annealing: perturb random coordinates and accept/reject."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    b = fast_conv(a)
    M_cur = b.max()
    
    T_start = 0.05 * best_M
    T_end = 0.001 * best_M
    total_steps = 50000
    
    for step in range(total_steps):
        if time.monotonic() > deadline:
            break
        
        frac = step / total_steps
        T = T_start * (T_end / T_start) ** frac
        
        # Perturb n_perturb random coordinates
        indices = rng.choice(n, size=n_perturb, replace=False)
        a_trial = a.copy()
        for idx in indices:
            a_trial[idx] = max(0.0, a[idx] + rng.normal(0, 0.1 / n))
        
        s_t = a_trial.sum()
        if s_t < 0.01:
            continue
        a_trial = a_trial / s_t
        
        b_t = fast_conv(a_trial)
        M_t = b_t.max()
        
        delta = M_t - M_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_trial
            b = b_t
            M_cur = M_t
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.01, 0.03, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Concave (parabola)
    poly = (t * (1 - t)) + 0.1
    inits.append(poly / poly.sum())
    
    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())
    
    # Gaussian
    gauss = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    return inits

    print(f"{'='*60}", flush=True)
    
    t2_deadline = min(time.monotonic() + 200, deadline - 200)
    
    if tracker.best is not None:
        a_lp = run_lp_descent(tracker.best, t2_deadline, n_active_start=50)
        tracker.update(a_lp)
    
    tracker.report("After Phase 2 (LP)")
    
    # === PHASE 3: Coordinate descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Coordinate descent", flush=True)
    print(f"{'='*60}", flush=True)
    
    t3_deadline = min(time.monotonic() + 300, deadline - 100)
    
    if tracker.best is not None:
        a_cd = run_coord_descent(tracker.best, t3_deadline, seed=seed)
        tracker.update(a_cd)
    
    tracker.report("After Phase 3 (CD)")
    
    # === PHASE 4: Simulated annealing + FISTA refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: SA + refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    t4_deadline = deadline - 30
    
    if tracker.best is not None and time.monotonic() < t4_deadline:
        # SA to explore
        t_sa = min(time.monotonic() + 100, t4_deadline - 100)
        a_sa = simulated_annealing(tracker.best, t_sa, n_perturb=3, seed=seed+1)
        tracker.update(a_sa)
        
        # FISTA refinement
        if time.monotonic() < t4_deadline - 50:
            a_f = run_fista(tracker.best, t4_deadline - 50, tau_start=500, tau_end=5000, lr=0.1)
            tracker.update(a_f)
        
        # Final CD
        if time.monotonic() < t4_deadline - 10:
            a_cd2 = run_coord_descent(tracker.best, t4_deadline, seed=seed+99)
            tracker.update(a_cd2)
        
        # One more LP
        if time.monotonic() < t4_deadline - 5:
            a_lp2 = run_lp_descent(tracker.best, t4_deadline, n_active_start=30)
            tracker.update(a_lp2)
    
    tracker.report("After Phase 4 (SA + refine)")
    
    # === PHASE 5: Try nearby lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 20:
        n_cur = tracker.best.size
        for new_n in [n_cur - 16, n_cur + 16, n_cur - 32, n_cur + 32]:
            if new_n < 256 or new_n > 2048:
                continue
            if time.monotonic() > deadline - 15:
                break
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 5, time.monotonic() + 30)
            a_opt = run_fista(a_ext, sub_dl, tau_start=100, tau_end=2000, lr=0.2)
            tracker.update(a_opt)
            # Quick CD
            if time.monotonic() < deadline - 5:
                a_cd = run_coord_descent(a_opt, deadline - 3, seed=seed + new_n)
                tracker.update(a_cd)
    
    tracker.report("After Phase 5 (lengths)")
    
    # Final
    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x)
