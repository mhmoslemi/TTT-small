import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build the convolution matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            m = k - j
            rows.append(k)
            cols.append(j)
            vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def golden_line_search(a, g, M_current, n_evals=30):
    """Golden section search for best t in [0,1] minimizing max(conv((1-t)*a+t*g))."""
    a = a.astype(np.float64)
    g = g.astype(np.float64)
    
    def obj(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-10:
            return 1e10
        a_t = a_t / s
        return fast_conv(a_t).max()
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

            b_pt = a_pt

        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_conv(a)
    best_M = b.max()
    best_a = a.copy()
    no_improve = 0
    iter_time = time_per_lp + ls_evals * 0.05
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M_cur = b.max()
        
        # Solve LP
        if use_full_lp:
            g, lp_result = solve_lp_full(a, M_cur, n, time_limit=time_per_lp)
        else:
            g = solve_lp_active(a, M_cur, n, n_active=n_active, time_limit=time_per_lp)
        
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Check if g is meaningfully different from a
        diff = np.abs(g - a).sum()
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Line search
        t_opt, M_new = golden_line_search(a, g, M_cur, n_evals=ls_evals)
        
        if M_new < best_M - 1e-14:
            a_new = (1.0 - t_opt) * best_a + t_opt * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_a = a_new.copy()
                best_M = fast_conv(a_new).max()
            a = a_new
            no_improve = 0
        elif t_opt > 0.01:
            a_new = (1.0 - t_opt) * a + t_opt * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                M_check = fast_conv(a_new).max()
                a = a_new
                if M_check < best_M:
                    best_M = M_check
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Try a more aggressive step
            for t_try in [0.3, 0.5, 0.7]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                M_try = fast_conv(a_try).max()
                if M_try < best_M:
                    a = a_try
                    best_a = a.copy()
                    best_M = M_try
                    no_improve = 0
                    break
        
        if no_improve > 20:
            break
        
        if (it + 1) % 10 == 0:
            elapsed = time.monotonic()
            print(f"      it={it+1}, M={best_M:.12f}, n={n}", flush=True)
    
    return best_a, best_M


def coordinate_descent_optimize(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with golden section on each coordinate."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_conv(a)
    best_a = a.copy()
    best_M = b.max()
    no_improve = 0
    order = np.arange(n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        sweep_changes = 0
        
        b = fast_conv(a)
        for i in order:
            if time.monotonic() > deadline:
                break
            
            # Find optimal a[i] given all others
            # Affected k range: [i, min(2n-2, i+n-1)]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Precompute coefficients
            # b[k] as function of x = a[i]: b[k] = c_k + s_k * x (+ quadratic at k=2*i)
            c_arr = {}
            s_arr = {}
            quad_at = -1
            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    c_arr[k] = b[k] - a[i] * a[i]
                    s_arr[k] = 0.0
                    quad_at = k
                else:
                    j = k - i
                    if 0 <= j < n:
                        c_arr[k] = b[k] - 2.0 * a[i] * a[j]
                        s_arr[k] = 2.0 * a[j]
                    else:
                        c_arr[k] = b[k]
                        s_arr[k] = 0.0
            
            # Unaffected max
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[k_lo:k_hi + 1] = False
            unaff_max = b[mask].max() if mask.any() else 0.0
            
            def obj_cd(x):
                vals = []
                for k in range(k_lo, k_hi + 1):
                    v = c_arr[k] + s_arr[k] * x
                    if k == quad_at:
                        v += x * x
                    vals.append(v)
                return max(unaff_max, max(vals) if vals else unaff_max)
            
            # Golden section on [0, hi]
            x_lo = 0.0
            x_hi = max(2.0 * a[i] + 0.01, 0.1)
            
            # Expand hi if needed
            for _ in range(5):
                if obj_cd(x_hi) > obj_cd(x_hi * 0.5) + 1e-15 and x_hi < 1.0:
                    x_hi *= 2.0
                else:
                    break
            
            a_pt = x_hi - gr * (x_hi - x_lo)
            b_pt = x_lo + gr * (x_hi - x_lo)
            fa = obj_cd(a_pt)
            fb = obj_cd(b_pt)
            for _ in range(30):
                if fa < fb:
                    x_hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = x_hi - gr * (x_hi - x_lo)
                    fa = obj_cd(a_pt)
                else:
                    x_lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = x_lo + gr * (x_hi - x_lo)
                    fb = obj_cd(b_pt)
            
            x_opt = (x_lo + x_hi) / 2.0
            if obj_cd(0.0) <= obj_cd(x_opt):
                x_opt = 0.0
            x_opt = max(0.0, x_opt)
            
            if abs(x_opt - a[i]) > 1e-12:
                # Update b
                delta = x_opt - a[i]
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += 2 * a[i] * delta + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * delta
                a[i] = x_opt
                sweep_changes += 1
        
        # Renorm
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        b_new = fast_conv(a)
        M_new = b_new.max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if sweep_changes == 0:
            no_improve += 5
        if no_improve > 15:
            break
    
    return best_a, best_M


def perturbation_escape(a, n_perturbs, deadline, rng):
    """Try random perturbations to escape local minima."""
    a = a.astype(np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for i in range(n_perturbs):
        if time.monotonic() > deadline:
            break
        
        # Random perturbation
        scale = rng.uniform(0.01, 0.1)
        pert = rng.normal(0, scale, n)
        a_pert = a + pert
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s
        
        M_pert = fast_conv(a_pert).max()
        if M_pert < best_M:
            best_a = a_pert.copy()
            best_M = M_pert
        
        # Quick LP step from perturbed point
        g = solve_lp_active(a_pert, M_pert, n, n_active=50, time_limit=3)
        if g is not None:
            t, M_new = golden_line_search(a_pert, g, M_pert, n_evals=10)
            if M_new < best_M:
                a_new = (1 - t) * a_pert + t * g
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
                best_a = a_new.copy()
                best_M = M_new
    
    return best_a, best_M


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for noise_scale in [0.02, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, noise_scale, n), 1e-6)
            inits.append(p_n / p_n.sum())
    
    inits.append(np.ones(n) / n)
    
    # Various structured inits
    inits.append(((t * (1 - t)) + 0.05) / (((t * (1 - t)) + 0.05).sum()))
    inits.append(((2*t-1)**2 + 0.05) / (((2*t-1)**2 + 0.05).sum()))
    
    for freq in [1, 2, 3, 4]:
        w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        w = np.maximum(w, 0.01)
        inits.append(w / w.sum())
    
    gauss = np.exp(-3.0 * (t - 0.5)**2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Anti-correlation init: try to make autocorrelation flat
    a_anti = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_anti)
        M_s = bs.max()
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_anti[j]
        a_anti = a_anti - 0.15 * (corr - corr.mean())
        a_anti = np.maximum(a_anti, 0.001)
        a_anti = a_anti / a_anti.sum()
    inits.append(a_anti.copy())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)
