    # = ifft(FFT(a)*conj(FFT(p)))[0] for m = 0

    # With corr[0] = R[0] (since -0 = 0 = N-0 mod N)

    # conv(a, q)[k] = sum_n a[n] * q[k-n]

    #

    # And conv(a, q) can be computed via FFT:
    # conv(a, q) = ifft(FFT(a) * FFT(q)) (with appropriate zero-padding)
    # 
    # So: R = ifft(FFT(a) * FFT(p[::-1]))[2n-2-m] for m = 0, ..., 2n-2
    # = ifft(FFT(a, N) * FFT(p[::-1], N))[0:2n-1][::-1]
    
    # Let me verify with the small example:
    # a = [a0, a1] (n=2), p = [p0, p1, p2] (length 3 = 2n-1)
    # q = p[::-1] = [p2, p1, p0]
    # conv(a, q) has length 2+3-1 = 4:
    # conv[0] = a0*p2
    # conv[1] = a0*p1 + a1*p2
    # conv[2] = a0*p0 + a1*p1
    # conv[3] = a1*p0
    # 
    # R[m] = conv[2n-2-m] = conv[2-m]:
    # R[0] = conv[2] = a0*p0 + a1*p1 ✓
    # R[1] = conv[1] = a0*p1 + a1*p2 ✓
    # R[2] = conv[0] = a0*p2 ✓
    
    # So: R = ifft(FFT(a, N) * FFT(p[::-1], N))[:2n-1][::-1]
    # where N >= n + (2n-1) - 1 = 3n-2
    
    # And the gradient is:
    # grad[i] = 2*R[i] - p[2*i]*a[i] for i = 0, ..., n-1
    # (since 2i <= 2n-2 for all i in [0, n-1])
    
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p[::-1], size)
    R = np.fft.irfft(A * P, size)[:2 * n - 1][::-1]
    
    # grad[i] = 2*R[i] - p[2*i]*a[i]
    grad = 2.0 * R[:n] - p[2 * np.arange(n)] * a
    return grad

import numpy as np
from scipy.optimize import linprog

import time


def fast_autocorr(a):
    """Compute full autocorrelation a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective value J(a)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_grad_fft(a, p):
    """
    Compute gradient of sum_k p[k]*b[k] w.r.t. a, where b = a*a.
    Uses FFT for O(n log n) computation.
    p: weights on autocorrelation lags (length 2n-1), should sum to 1.
    """
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p[::-1], size)
    R = np.fft.irfft(A * P, size)[:2 * n - 1][::-1]
    # grad[i] = 2*R[i] - p[2*i]*a[i]
    grad = 2.0 * R[:n].copy()
    idx = 2 * np.arange(n)
    grad -= p[idx] * a
    return grad


def smooth_max_weights(b, tau):
    """Compute softmax weights for smooth max approximation."""
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    s = exp_b.sum()
    if s < 1e-300:
        return np.full_like(b, 1.0 / len(b))
    return exp_b / s


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: flatten power spectrum while maintaining non-negativity."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        # Power-law compression
        power = 0.4 + 0.4 * beta
        mag_alt = (mag ** power)
        mag_alt = mag_alt / (mag_alt.max() + 1e-30) * mag.max()
        # Linear compression
        mag_lin = (1.0 - beta * 0.5) * mag + beta * 0.5 * (mag.mean() + 1e-30)
        mag_final = 0.5 * mag_lin + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_autocorr(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def run_adam_curriculum(a, deadline, max_iters=3000, tau_start=0.5, tau_end=500.0,
                        lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with curriculum on tau (soft-to-hard max)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    stall_count = 0
    no_improve_threshold = 200
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        
        frac = it / max_iters
        # Curriculum: increasing tau (soft to hard)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        
        b = fast_autocorr(a)
        b_max = b.max()
        
        if b_max < best_M * (1 - 1e-9):
            best_M = b_max
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > no_improve_threshold and frac > 0.3:
            # Try to escape: add small perturbation
            if stall_count > no_improve_threshold * 2:
                break
            if stall_count % 100 == 0:
                a = a + 0.01 * np.random.default_rng(it * 7919).normal(0, 1, n)
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                m = np.zeros(n)
                v = np.zeros(n)
                stall_count = 0
                continue
        
        p = smooth_max_weights(b, tau)
        grad = compute_grad_fft(a, p)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        # Learning rate schedule: decay slowly
        lr_t = lr * (0.5 + 0.5 * (1 - frac))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-10)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    
    return best_a

def build_A_matrix(n, a):
    """Build sparse constraint matrix for LP: (a*g)(k) <= M for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    """Solve LP: maximize sum(g) s.t. (a*g)(k) <= M, g >= 0."""
    n = len(a)
    rows, cols, vals = build_A_matrix(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def perturbation_restart(a, rng, n_perturb=5, pert_size=0.05):
    """Apply random perturbations and keep the best."""
    n = len(a)
    best = a.copy()
    best_M = fast_autocorr(a).max()
    for _ in range(n_perturb):
        a_pert = a + pert_size * rng.normal(0, 1, n)
        a_pert = np.maximum(a_pert, 1e-8)
        a_pert = a_pert / a_pert.sum()
        M = fast_autocorr(a_pert).max()
        if M < best_M:
            best_M = M
            best = a_pert.copy()
    return best


def resample_seq(src, n):
    """Resample a sequence to a new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent):
    """Generate diverse initializations for optimization."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + 0.03 * rng.normal(0, 1, n), 0.01))
        inits.append(np.maximum(p + 0.08 * rng.normal(0, 1, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Power-law shapes
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        shape = t**alpha + (1 - t)**alpha
        inits.append(np.maximum(shape, 0.01))
    
    # Cosine-based
    for f in [1, 2, 3, 4, 6]:
        inits.append(np.maximum(0.5 + 0.45 * np.cos(2 * np.pi * f * t), 0.01))
    
    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        inits.append(np.exp(-8 * (t - center)**2) + 0.05)
    
    # Double bump
    inits.append(np.exp(-10 * (t - 0.3)**2) + np.exp(-10 * (t - 0.7)**2) + 0.03)
    inits.append(np.exp(-6 * (t - 0.25)**2) + np.exp(-6 * (t - 0.75)**2) + 0.05)
    
    # W-shaped
    inits.append(np.maximum(1.0 - 1.2 * (4 * t - 1)**2 * (4 * t - 3)**2, 0.02))
    
    # Sine-squared
    inits.append(np.sin(np.pi * t)**2 + 0.05)
    
    # Random smooth
    k = max(5, n // 15)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.02))
    
    # High-frequency random
    freqs = rng.integers(1, 5, size=min(5, n // 10))
    phases = rng.uniform(0, 2 * np.pi, size=min(5, n // 10))
    amps = rng.uniform(0.1, 0.4, size=min(5, n // 10))
    sig = 0.6
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * f * t + ph)
    inits.append(np.maximum(sig, 0.02))
    
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.02)
    inits.append(gs_flat_spectral(a_rand, n_iters=80, beta=0.85))
    
    # GS from parent
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(gs_flat_spectral(p, n_iters=100, beta=0.9))
    
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.02)
    if n % 2 == 0:
        full = np.concatenate([half, half[-2::-1]])[:n]
    else:
        full = np.concatenate([half, half[-1::-1]])[:n]
    inits.append(full)
    
    # Anti-symmetric correction (make it more "flat" in frequency)
    if parent is not None:
        p = resample_seq(parent, n)
        # Try complement: 1 - normalized parent
        p_norm = p / p.max()
        inits.append(np.maximum(1.0 - 0.7 * p_norm, 0.02))
    
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0
    best = BestTracker()
    
    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent loaded: n={parent.size}, J={best.score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass
    
    if parent is None:
        parent = np.ones(256)
        best.update(parent)
    
    # === PHASE 1: Multi-start Adam optimization ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-start Adam Optimization", flush=True)
    print(f"{'='*60}", flush=True)
    
    phase1_budget = 500.0
    p1_deadline = time.monotonic() + phase1_budget
    
    # Focus on lengths around the parent and some alternatives
    base_n = parent.size
    candidate_lengths = [640, 768, 896, 1024, 1152, 1280, 1408]
    # Prioritize lengths close to parent
    candidate_lengths.sort(key=lambda n: abs(n - base_n))
    
    best_per_length = {}
    time_per_length = phase1_budget / len(candidate_lengths)
    
    for n in candidate_lengths:
        t0 = time.monotonic()
        remaining = p1_deadline - time.monotonic()
        if remaining < 30:
            break
        allot = min(time_per_length, remaining * 0.6)
        if allot < 15:
            continue
        
        inits = make_initializations(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')
        
        # Try top initializations with time allocation
        n_inits = len(inits)
        time_per_init = allot / min(n_inits, 6)
        
        for idx, a0 in enumerate(inits[:8]):  # Limit to 8 inits per length
            if time.monotonic() > time.monotonic() + time_per_init:
                if idx > 3:
                    break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 = a0 / a0.sum()
            
            sub_dl = min(time.monotonic() + time_per_init * 1.5, p1_deadline - 10)
            a = run_adam_multi_tau(a0, sub_dl, max_iters=3000)
            M = fast_autocorr(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()
        
        if best_n is not None:
            J = compute_J(best_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M:.8f} J={J:.6f} ({elapsed:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()
    
    best.report("After Phase 1")
    print(f"Phase 1 complete: {time.monotonic() - (deadline - phase1_budget - 20):.0f}s used", flush=True)
    
    # === PHASE 2: Refined Adam on best candidates ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Refined Adam on Top Candidates", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_deadline = deadline - 400.0  # Reserve 400s for LP
    if time.monotonic() < p2_deadline - 30:
        # Sort by current best
        sorted_lengths = sorted(best_per_length.keys(),
                                key=lambda n: compute_J(best_per_length[n]))
        
        for n_try in sorted_lengths[:3]:
            if time.monotonic() > p2_deadline - 20:
                break
            a_cand = best_per_length[n_try].copy()
            remaining = p2_deadline - time.monotonic() - 15
            if remaining < 30:
                break
            allot = remaining / max(1, 3 - sorted_lengths.index(n_try))
            
            # Run longer Adam with harder tau
            a_refined = run_adam_curriculum(a_cand, time.monotonic() + allot * 0.7,
                                            max_iters=5000, tau_start=50.0,
                                            tau_end=500.0, lr=0.05)
            best.update(a_refined)
            
            # Try perturbation + re-optimize
            if time.monotonic() < time.monotonic() + allot * 0.25:
                a_pert = perturbation_restart(a_refined, rng, n_perturb=8, pert_size=0.03)
                a_pert = run_adam_curriculum(a_pert, time.monotonic() + allot * 0.2,
                                             max_iters=2000, tau_start=100.0,
                                             tau_end=800.0, lr=0.03)
                best.update(a_pert)
    
    best.report("After Phase 2")
    
    # === PHASE 3: LP Refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best.seq is not None:
        n = best.seq.size
        # Estimate LP cost
        est_lp_time = 0.5 * (n / 256.0)**2  # seconds per LP iteration (rough)
        max_lp_iters = max(20, min(80, int(200.0 / max(0.1, est_lp_time))))
        
        # LP on the best sequence
        lp_dl = min(deadline - 60, time.monotonic() + 250)
        a_lp = run_lp_refine(best.seq, lp_dl, max_iters=max_lp_iters)
        best.update(a_lp)
        
        # Second LP pass with smaller steps
        if time.monotonic() < deadline - 50:
            a_lp2 = run_lp_refine(a_lp, min(deadline - 40, time.monotonic() + 150),
                                  max_iters=max(10, max_lp_iters // 2))
            best.update(a_lp2)
    
    best.report("After Phase 3")
    
    # === PHASE 4: Final polish and exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best.seq is not None and time.monotonic() < deadline - 10:
        # Short Adam polish with very hard tau
        a_polish = run_adam_curriculum(best.seq, min(time.monotonic() + 60, deadline - 20),
                                       max_iters=3000, tau_start=200.0,
                                       tau_end=2000.0, lr=0.01)
        best.update(a_polish)
        
        # LP on polished
        if time.monotonic() < deadline - 15:
            a_final_lp = run_lp_refine(a_polish, deadline - 5, max_iters=30)
            best.update(a_final_lp)
        
        # Try nearby lengths
        n_best = best.seq.size
        for delta in [32, -32, 64, -64]:
            if time.monotonic() > deadline - 15:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 2048:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()
            a_ext = run_adam_curriculum(a_ext, min(time.monotonic() + 20, deadline - 10),
                                        max_iters=1000, tau_start=50.0,
                                        tau_end=300.0, lr=0.05)
            best.update(a_ext)
    
    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best.seq]
