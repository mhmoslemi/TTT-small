def fast_conv(a):

def build_lp_matrix(n, a):
    """Build the sparse constraint matrix for the LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def build_lp_matrix_fast(n, a):
    """Build sparse LP matrix using vectorized approach."""
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        rows_list.append(np.full(len(js), k))
        cols_list.append(js)
        vals_list.append(a[k - js])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return rows, cols, vals


def solve_lp_best_response(a, M, n):
    """Solve LP to find best response g."""
    rows, cols, vals = build_lp_matrix_fast(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-8,
                            'primal_feasibility_tolerance': 1e-8})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception as e:
        print(f"  LP error: {e}", flush=True)
    return None


def anderson_mix(x_hist, Fx_hist, m):
    """Anderson mixing: find optimal convex combination of history."""
    if m < 2:
        return Fx_hist[-1].copy()
    
    residuals = np.array([x_hist[i] - Fx_hist[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-10 * np.eye(m)
    
    try:
        ones = np.ones(m)
        alpha = np.linalg.solve(G, ones)
        alpha = alpha / alpha.sum()
        # Clamp for stability
        alpha = np.clip(alpha, -0.5, 1.5)
        alpha = alpha / alpha.sum()
    except np.linalg.LinAlgError:
        alpha = np.ones(m) / m
    
    mixed = np.zeros(len(Fx_hist[0]))
    for i in range(m):
        mixed += alpha[i] * Fx_hist[i]
    return np.maximum(mixed, 0)


def lp_mirror_descent(a0, deadline, max_iters=500, anderson_win=6):

            break

        M = b.max()
        
        if M < best_M * (1 - 1e-10):
            if best_M / M > 1.001:
                print(f"    LP iter {it}: M={M:.8f} (best {best_M:.8f})", flush=True)
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break
            # Decrease step size
            t = max(0.005, t * 0.95)
        
        g = solve_lp_best_response(a, M, n)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        a_new = np.maximum((1 - t) * a + t * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s

        x_hist.append(a.copy())
        Fx_hist.append(a_new.copy())

def coordinate_descent_polish(a0, deadline, max_iters=2000):
    """Coordinate descent on the most impactful variables."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Find the index of the max
        k_max = b.argmax()
        
        # The variables that affect b[k_max] are a[lo:hi+1] where
        # lo = max(0, k_max - n + 1), hi = min(n-1, k_max)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        
        # Try to reduce b[k_max] by adjusting a[lo:hi+1]
        # dc_k/da[j] = 2*a[k-j], so we want to decrease a[j] proportional to a[k-j]
        # But we also need to keep b[j] below the new max for other j
        
        best_local_M = best_M
        best_local_a = a.copy()
        
        for j in range(lo, hi + 1):
            # Try decreasing a[j]
            for delta_frac in [0.01, 0.02, 0.05, 0.1]:
                a_try = a.copy()
                decrease = delta_frac * a[j]
                if decrease < 1e-12:
                    continue
                a_try[j] -= decrease
                if a_try[j] < 0:
                    continue
                # Redistribute to maintain sum
                a_try /= a_try.sum()
                b_try = fast_conv(a_try)
                M_try = b_try.max()
                if M_try < best_local_M:
                    best_local_M = M_try
                    best_local_a = a_try.copy()
        
        if best_local_M < best_M * (1 - 1e-12):
            best_M = best_local_M
            best_a = best_local_a.copy()
            a = best_local_a
        else:
            # Try increasing some variable
            # (sometimes increasing a non-critical variable helps)
            improved = False
            for j in range(n):
                if a[j] / a.max() > 0.5:
                    continue  # Skip already-large variables
                a_try = a.copy()
                a_try[j] += 0.01 * a.max()
                a_try /= a_try.sum()
                b_try = fast_conv(a_try)
                if b_try.max() < best_M * (1 - 1e-12):
                    best_M = b_try.max()
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if not improved:
                break
    
    return best_a


def resample(a, n_new):
    """Resample sequence to new length."""
    a = np.asarray(a, dtype=np.float64)
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    t_old = np.linspace(0, 1, n_old)
    t_new = np.linspace(0, 1, n_new)
    return np.maximum(np.interp(t_new, t_old, a), 0)

def fast_conv(a):
    """FFT-based full convolution."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def gs_flat(a0, n_iters=200, beta=0.85):
    """Gerchberg-Saxton spectral flattening for good initializations."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()

    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_mean = mag.mean()
        # Adaptive compression: stronger when dynamic range is large
        dr = (mag.max() - mag.min()) / (mag_mean + 1e-30)
        eff_beta = min(0.95, beta * min(1.0, 2.0 / (1.0 + dr * 0.1)))
        mag_new = (1 - eff_beta) * mag + eff_beta * mag_mean

def chirp_sequence(n):
    """Chirp (FM) sequence - has good autocorrelation properties."""
    t = np.arange(n) / n

def smooth_grad(a, tau, top_k=80):
    """Compute gradient of smooth max (log-sum-exp) of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    
    if top_k >= len(b):
        top_idx = np.arange(len(b))
        b_top = b
    else:
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        b_top = b[top_idx]
    
    b_top_max = b_top.max()
    exp_b = np.exp(tau * (b_top - b_top_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        wi = 2.0 * p[i]
        if wi < 1e-14:
            continue
        k = top_idx[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        kj = k - np.arange(lo, hi + 1)
        grad[lo:hi + 1] += wi * a[kj]
    
    return grad, b

def build_lp_data(n, a):
    """Build sparse matrix data for LP constraint: conv(a, g) <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        js = np.arange(j_min, j_max + 1)
        rows.append(np.full(len(js), k))
        cols.append(js)
        vals.append(a[k - js])
    return np.concatenate(rows), np.concatenate(cols), np.concatenate(vals)

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                            'primal_feasibility_tolerance': 1e-9,
                            'time_limit': max(1.0, (deadline_global - time.monotonic()) * 0.05)})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(x_hist, Fx_hist):
    """Anderson mixing with extrapolation."""
    m = len(x_hist)
    if m < 2:
        return Fx_hist[-1].copy()
    
    residuals = np.array([x_hist[i] - Fx_hist[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-12 * np.eye(m)
    
    try:
        ones = np.ones(m)
        alpha = np.linalg.solve(G, ones)
        s = alpha.sum()
        if abs(s) < 1e-15:
            alpha = ones / m
        else:
            alpha = alpha / s
        # Allow mild extrapolation but clamp for stability
        alpha = np.clip(alpha, -0.3, 1.3)
        alpha = alpha / alpha.sum()
    except np.linalg.LinAlgError:
        alpha = np.ones(m) / m
    
    mixed = np.zeros(len(Fx_hist[0]))
    for i in range(m):
        mixed += alpha[i] * Fx_hist[i]
    return np.maximum(mixed, 0)


def lp_mirror(a0, deadline, max_iters=500, anderson_win=6):
    """LP mirror descent with Anderson acceleration."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    x_hist = []
    Fx_hist = []
    no_imp = 0
    t = 0.25
    consec_no_lp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                t = max(0.01, t * 0.9)
                # Try a small perturbation to escape
                if no_imp > 80:
                    a_pert = a + 0.01 * np.random.default_rng(it).normal(0, 1, n)
                    a_pert = np.maximum(a_pert, 0)
                    a_pert /= a_pert.sum()
                    M_pert = fast_conv(a_pert).max()
                    if M_pert < M:
                        a = a_pert
                        no_imp = 0
                        t = 0.2
                    elif no_imp > 150:
                        break
        
        g = solve_lp(a, M, n)
        if g is None:
            consec_no_lp += 1
            if consec_no_lp > 5:
                break
            continue
        consec_no_lp = 0
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        a_new = np.maximum((1 - t) * a + t * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        # Anderson acceleration
        x_hist.append(a.copy())
        Fx_hist.append(a_new.copy())
        if len(x_hist) > anderson_win:
            x_hist.pop(0)
            Fx_hist.pop(0)
        
        if len(x_hist) >= 3:
            a_and = anderson_mix(x_hist, Fx_hist)
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
            M_and = fast_conv(a_and).max()
            M_new = fast_conv(a_new).max()
            if M_and <= M_new * 1.01:
                a = a_and
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def active_set_polish(a0, deadline, max_iters=1000):
    """Polish by focusing on the active set (convolution values near max)."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Identify active set: convolution values within 1% of max
        threshold = b_max * 0.99
        active_mask = b >= threshold
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            break
        
        # Compute weighted gradient: equalize active values
        # We want to reduce the max, so push down the highest active values
        b_active = b[active_idx]
        weights = (b_active - b_active.min()) / (b_active.max() - b_active.min() + 1e-30)
        weights = weights / weights.sum()
        
        # Gradient of weighted sum of active convolution values
        grad = np.zeros(n)
        for i, k in enumerate(active_idx):
            w = weights[i]
            if w < 1e-8:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            kj = k - np.arange(lo, hi + 1)
            grad[lo:hi + 1] += 2.0 * w * a[kj]
        
        # Projected gradient step (stay in simplex: a >= 0, sum = 1)
        lr = 0.05 * (1.0 - 0.5 * it / max_iters)
        a_new = np.maximum(a - lr * grad, 0)
        a_new -= (a_new - a).sum() * a  # Project to zero-sum perturbation
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-10:
            break
        a_new /= sa
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M * (1 - 1e-12):
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for lr_try in [0.01, 0.005, 0.001]:
                a_try = np.maximum(a - lr_try * grad, 0)
                a_try -= (a_try - a).sum() * a
                a_try = np.maximum(a_try, 0)
                if a_try.sum() < 1e-10:
                    continue
                a_try /= a_try.sum()
                M_try = fast_conv(a_try).max()
                if M_try < best_M * (1 - 1e-12):
                    best_M = M_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                break
    
    return best_a


def resample(a, n_new):
    """Resample sequence to new length using linear interpolation."""
    a = np.asarray(a, dtype=np.float64)
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    t_old = np.linspace(0, 1, n_old)
    t_new = np.linspace(0, 1, n_new)
    return np.maximum(np.interp(t_new, t_old, a), 0)

    best_per_len = {}

        inits.append(('chirp_gs', a_ch_gs))
        
        # Random smooth
        k = max(5, n // 25)
        sigma = k * 0.4
        x = np.arange(-2*k, 2*k+1)
        kern = np.exp(-0.5 * (x / sigma) ** 2)
        kern /= kern.sum()
        raw = rng.uniform(0.2, 1.0, n + 4*k)

        # Symmetric random
        half_n = n // 2 + 1
        half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.05)
        full = np.concatenate([half, half[-2::-1]])[:n]
