So `df/da[i] = S^{1/p - 1} * sum_k conv[k]^{p-1} * 2*a[k-i]` (for valid k-i)
`= 2 * S^{1/p - 1} * sum_{j=0}^{n-1} conv[i+j]^{p-1} * a[j]`

The key sum is `sum_j conv[i+j]^{p-1} * a[j]` for j = 0, ..., n-1.
This is a cross-correlation of `w = conv^{p-1}` (length 2n-1) with `a` (length n).

Specifically, `g[i] = sum_{j=0}^{n-1} w[i+j] * a[j]` for i = 0, ..., n-1.

We can compute this with FFT:
- Pad w to length >= 3n-2 (since i+j can go up to (n-1)+(n-1) = 2n-2, and w has length 2n-1, so we need index up to 2n-2)
- Actually, g[i] = sum_j w[i+j]*a[j]. Let's flip a: `a_flip[j] = a[n-1-j]`.
- Then g[i] = sum_j w[i+j]*a[n-1-j] = (w * a_flip)[i + n - 1] where * is convolution.

So: `g = convolve(w, a_flip)[n-1 : 2n-1]` (taking the right part).

We can compute this with FFT in O(n log n).

```python
def fast_pnorm_descent(a, p_start=2, p_end=20, n_iter=500, time_budget=60):
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(np.convolve(a, a))
    
    start = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive p
        p = p_start + (p_end - p_start) * min(1.0, it / (n_iter * 0.5))
        
        # Compute convolution
        conv = full_conv(a)  # length 2n-1
        S = np.sum(conv**p)
        if S < 1e-30:
            break
        
        # Compute gradient
        w = conv ** (p-1)  # length 2n-1
        a_flip = a[::-1]
        # g[i] = sum_j w[i+j] * a[j] for i=0..n-1
        # = (w * a_flip)[i + n - 1]
        conv_w_a = full_conv(w, a_flip)  # length (2n-1) + n - 1 = 3n-2
        g = conv_w_a[n-1 : 2*n-1]  # length n
        
        grad = 2.0 * (S ** (1.0/p - 1.0)) * g
        
        # Project gradient: subtract mean to stay on sum=const hyperplane
        grad_centered = grad - np.mean(grad)
        
        # Adaptive step size
        step = 0.1 / (np.max(np.abs(grad_centered)) + 1e-10)
        
        # Try a few step sizes
        best_step_obj = np.inf
        best_step_a = best_a
        for mult in [0.1, 0.3, 0.5, 1.0, 1.5, 2.0]:
            a_try = a - mult * step * grad_centered
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * np.max(full_conv(a_try))
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()
        
        if best_step_obj < best_obj - 1e-10:
            best_obj = best_step_obj
            best_a = best_step_a
            a = best_a
        else:
            # Stuck, try to perturb
            if it % 50 == 49:
                a = best_a * (1 + 0.05 * np.random.randn(n))
                a = np.maximum(a, 0)
                a = a / a.sum()
    
    return best_a
```

"""
Log-sum-exp gradient descent + LP refinement for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Key innovations:
1. Recursive "flat autocorrelation" initialization
2. FFT-accelerated log-sum-exp gradient with adaptive temperature
3. No symmetry enforcement (exploits asymmetry for improvement)
4. Multi-n exploration with resampling
5. LP-based final polish
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def full_conv(a, b=None):
    """Full linear convolution a*b via FFT."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = full_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def recursive_init(n):
    """
    Build sequence by enforcing conv[k] = M for k=0..n-1 recursively.
    Returns normalized sequence or None if invalid.
    """
    a = np.zeros(n)
    a[0] = 1.0  # Scale later
    M = 1.0  # = a[0]^2
    
    for k in range(1, n):
        # a[k] = (M - sum_{i=1}^{k-1} a[i]*a[k-i]) / (2*a[0])
        s = 0.0
        for i in range(1, k):
            s += a[i] * a[k-i]
        a[k] = (M - s) / (2.0 * a[0])
        if a[k] < -1e-10:
            # Negative: try with a[0] adjusted
            return None
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return None
    return a / s

def lse_gradient_descent(a, time_budget=60.0, tau_start=5.0, tau_end=50.0, verbose=False):
    """
    Log-sum-exp gradient descent to minimize max(conv(a,a)).
    a is assumed normalized (sum=1).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(full_conv(a)))
    start = time.monotonic()
    
    lr = 0.1
    momentum = 0.9
    v = np.zeros(n)
    n_iter = 0
    
    while time.monotonic() - start < time_budget:
        n_iter += 1
        tau = tau_start + (tau_end - tau_start) * min(1.0, n_iter / 500.0)
        
        # Compute convolution
        conv = full_conv(a)  # length 2n-1
        max_conv = float(np.max(conv))
        
        # Log-sum-exp gradient
        # w[k] = exp(tau * conv[k] / max_conv) / sum (shift for numerical stability)
        shifted = tau * (conv - max_conv)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j] for i=0..n-1
        # This is cross-correlation: (w * a_flip)[i + n - 1]
        a_flip = a[::-1]
        # conv_w_a = full_conv(w, a_flip) has length (2n-1) + n - 1 = 3n-2
        # We want indices n-1 to 2n-2 (length n)
        conv_w_a = full_conv(w, a_flip)
        g = conv_w_a[n-1 : 2*n-1]  # length n
        grad = 2.0 * g
        
        # Center the gradient (to maintain sum=const)
        grad_centered = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad_centered
        
        # Adaptive learning rate
        grad_norm = np.max(np.abs(v)) + 1e-10
        step = lr / grad_norm
        
        # Try a few step sizes
        best_local_obj = np.inf
        best_local_a = best_a.copy()
        for mult in [0.3, 0.5, 0.7, 1.0, 1.5]:
            a_try = a - mult * step * v
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(full_conv(a_try)))
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a
            if verbose and n_iter % 100 == 0:
                print(f"    LSE it{n_iter}: obj={best_obj:.8f}, tau={tau:.1f}")
        else:
            # Stuck: reduce learning rate and perturb
            lr *= 0.9
            if lr < 0.001:
                # Reset with perturbation
                a = best_a * np.exp(np.random.randn(n) * 0.05)
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
                lr = 0.05
        
        # Decay momentum slightly
        momentum = max(0.5, momentum * 0.999)
    
    return best_a

def solve_lp_minpeak_cvxpy(a, time_limit=5.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a = a / np.sum(a)
    
    # Build convolution matrix A (2n-1 x n)
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2*n-1, n))
    for i in range(n):
        # A[k, i] = a[k-i] for k=i, i+1, ..., i+n-1
        A[i:i+n, i] = a
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        # min M s.t. A@g - M <= 0, sum(g) = 1, g >= 0
        # Variables: [g_0, ..., g_{n-1}, M]
        c = np.zeros(n+1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
        b_ub = np.zeros(2*n-1)
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = [1.0]
        bounds = [(0, None)]*n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
