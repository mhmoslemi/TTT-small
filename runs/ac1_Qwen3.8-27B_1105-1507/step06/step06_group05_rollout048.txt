```python
    # Numerical cleanup

def solve_lp(a, M, time_limit=20):

                      options={'presolve': True, 'time_limit': time_limit})

            s = float(np.sum(g))
```

```python
def softmax_gradient(a, tau, conv=None):
    if conv is None:
        conv = compute_conv(a)
    n = len(a)
    cmax = float(np.max(conv))
    shifted = (conv - cmax) / tau
    weights = np.exp(np.clip(shifted, -500, 500))
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # grad[i] = 2 * sum_j a[j] * w[i+j]
    # = 2 * (a_rev * w)[i + n - 1] where a_rev = reverse(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    a_rev = a[::-1]
    Af = rfft(a_rev, size)
    Wf = rfft(weights, size)
    conv_result = irfft(Af * Wf, size)
    grad = 2.0 * conv_result[n-1:2*n-1]
    return grad, weights
```
