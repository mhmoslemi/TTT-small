Uses LP alternating, frequency-domain spectral shaping, and adaptive
subgradient methods with multi-n exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa*fa, size)
    return conv[:2*n-1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0*n*max_b/(s*s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a))/len(a)
    return a/s


def build_toeplitz_fast(a):
    n = len(a)
    m = 2*n-1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    return A


def solve_lp(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz_fast(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)]*n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g/s
    except Exception:
        pass
    return None

def subgradient_optimize(a, max_steps=300, time_budget=30.0):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2*n-1:
        size <<= 1
    
    step_size = 0.05
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Softmax weighting over peaks
        tau = cmax / (10 + step*0.5)
        tau = max(tau, 1e-8)
        shifted = np.clip((conv - cmax)/tau, -50, 50)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: grad[i] = 2 * sum_j weights[i+j] * a[j]
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf*Af, size)
        grad = 2.0*corr[n-1:2*n-1]
        
        # Center the gradient
        grad -= np.mean(grad)
        
        a_new = normalize(a - step_size*grad)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step_size = min(step_size*1.1, 0.1)
        else:
            step_size *= 0.7
            if step_size < 1e-8:
                break
    
    return best_a

        P = np.abs(A)**2

        # the gradient of ac[k] w.r.t. P[m] is e^{2πimk/N} (real part for our purposes)
        # So we want to decrease P[m] where the weighted sum of e^{2πimk/N} is positive
        
        # Compute the weighted modulation

        # (using real part since P is real)

        # The real part of W gives the gradient direction
        grad_real = np.real(W)

        step = 0.05 * (1 - 0.5*it/max_iters)

        P_new = np.maximum(P_new, 0)
        
        # Reconstruct
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new, N))[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-15:

        a_new = a_new / s

            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

            P_new = P * (1 - step * grad_real / max(np.max(np.abs(grad_real)), 1e-10))
            P_new = np.maximum(P_new, 0)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with adaptive line search, frequency-domain peak reduction,
soft-subgradient descent, multi-n exploration, and local coordinate descent.
Focuses on efficient LP solves and broader n-range exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Clip negatives and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build (2n-1)x n Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_max_sum(a, M, time_limit=20):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_optimize(a, max_iters=80, time_budget=100.0, verbose=False):
    """LP alternating with adaptive blend search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break
        conv = compute_conv(a)
        M = float(np.max(conv))
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 4:
            break

        g = solve_lp_max_sum(a, M, time_limit=max(3, int(remaining * 0.65)))
        if g is None:
            break

        # Adaptive blend search: try key fractions
        best_t = 0.0
        best_o = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = normalize((1 - t) * a + t * g)
            o = compute_obj(a_try)
            if o < best_o:
                best_o = o
                best_t = t

        if best_t == 0.0:
            # No improvement with blend; try small perturbation
            stall_count += 1
            if stall_count > 5:
                rng = np.random.default_rng((it * 104729 + 31) % (2**31))
                a = normalize(best_a * np.exp(rng.normal(0, 0.04 + 0.02 * stall_count, n)))
                stall_count = 0
            continue

        stall_count = 0
        a_new = normalize((1 - best_t) * a + best_t * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and (it % 10 == 0):
                print(f"    LP it={it}: obj={best_obj:.8f}")
        else:
            # Try a smaller step
            t_small = best_t * 0.3
            a_try = normalize((1 - t_small) * a + t_small * g)
            o_try = compute_obj(a_try)
            if o_try < best_obj:
                best_obj = o_try
                best_a = a_try.copy()
                a = a_try

    return best_a, best_obj


def freq_peak_reduction(a, max_iters=100, time_budget=30.0):
    """Frequency-domain peak reduction using softmax-weighted spectral tilt."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, N)
        P = np.abs(A) ** 2
        ac = irfft(P, N)[:2 * n - 1]
        ac_max = float(np.max(ac))
        if ac_max < 1e-15:
            break

        # Softmax weights on peaks
        tau = max(ac_max / 25.0, 1e-10)
        exponents = (ac - ac_max) / tau
        weights = np.exp(np.clip(exponents, -50, 0))
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(len(ac))
            weights[int(np.argmax(ac))] = 1.0
        else:
            weights /= w_sum

        # Gradient of weighted ac w.r.t. P: IFFT(weights) gives the spectral tilt
        W = fft(weights, N)
        grad_real = np.real(W) / max(float(np.max(np.abs(W))), 1e-10)

        # Adaptively reduce power where gradient says to
        step = 0.08 * (1.0 - 0.5 * it / max_iters)
        P_new = P * (1.0 - step * grad_real)
        P_new = np.maximum(P_new, 1e-15)

        # Reconstruct preserving phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, N))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend to maintain progress
            a = normalize(0.6 * a + 0.4 * a_new)

    return best_a, best_obj


def subgradient_optimize(a, max_steps=300, time_budget=25.0):
    """Soft-subgradient descent targeting multiple autocorrelation peaks."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    step_size = 0.06
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Adaptive temperature
        tau = max(cmax / (15 + step * 0.3), 1e-10)
        shifted = np.clip((conv - cmax) / tau, -50, 0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient via correlation
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Center gradient to preserve sum
        grad -= np.mean(grad)

        a_new = normalize(a - step_size * grad)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step_size = min(step_size * 1.05, 0.12)
        else:
            step_size *= 0.7
            if step_size < 1e-9:
                break

    return best_a, best_obj


def coordinate_descent(a, time_budget=20.0):
    """Targeted coordinate descent on top peak contributors."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        pairs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                pairs.append((a[j] * a[k2], j, k2))
        pairs.sort(key=lambda x: x[0], reverse=True)

        improved = False
        for val, j1, j2 in pairs[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            # Try moving mass from peak contributor to a low-impact position
            for _ in range(2):
                j_tgt = int(rng.integers(0, n))
                if j_tgt in (j1, j2):
                    continue
                a_try = a.copy()
                delta = 0.04 * a_try[j1]
                if a_try[j1] < delta:
                    continue
                a_try[j1] -= delta
                a_try[j_tgt] += delta
                a_try = normalize(a_try)
                o = compute_obj(a_try)
                if o < best_obj - 1e-13:
                    best_obj = o
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            # Random kick
            if rng.random() < 0.15:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:
                break

    return best_a, best_obj


def resample_seq(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)
