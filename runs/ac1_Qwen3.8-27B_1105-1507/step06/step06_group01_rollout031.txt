import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def smooth_max_grad(a, tau, n_peaks=5):
    """Compute smooth max of conv(a,a) using log-sum-exp and gradient."""
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (valid k-i)
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        # sum_{k} p[k] * a[k-i] where 0 <= k-i < n
        lo = max(0, i)
        hi = min(2*n-1, i+n)
        if lo < hi:
            grad[i] = 2.0 * np.dot(p[lo:hi], a[0:hi-lo])
    return smooth_val, grad, b

def run_fista(a_init, deadline, tau_start=50, tau_end=500, max_iters=10000):
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr = 0.8
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y, _ = smooth_max_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        # Step size: use Lipschitz-like bound
        step = lr / (1.0 + 0.01 * tau)
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        if it % 10 == 0:
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
    
    M = fast_conv(a).max()
    if M < best_M:
        best_M = M
        best_a = a.copy()
    
    return best_a

def multi_peak_subgradient(a, n_peaks=10, eps=0.05):
    """Compute a combined subgradient from the top n_peaks of the convolution."""
    b = fast_conv(a)
    M = b.max()
    n = len(a)
    
    # Find peaks within eps*M of the max
    threshold = M * (1 - eps)
    peak_indices = np.where(b >= threshold)[0]
    
    if len(peak_indices) == 0:
        peak_indices = np.array([np.argmax(b)])
    
    # Cap at n_peaks for efficiency
    if len(peak_indices) > n_peaks:
        # Keep the top n_peaks
        peak_vals = b[peak_indices]
        top_idx = np.argsort(peak_vals)[::-1][:n_peaks]
        peak_indices = peak_indices[top_idx]
        peak_weights = peak_vals[top_idx] / peak_vals[top_idx].sum()
    else:
        peak_weights = np.ones(len(peak_indices)) / len(peak_indices)
    
    # Combined subgradient
    grad = np.zeros(n)
    for w, k in zip(peak_weights, peak_indices):
        for i in range(n):
            j = int(k) - i
            if 0 <= j < n:
                grad[i] += 2.0 * w * a[j]
    
    return grad, b, M

def run_multi_peak_descent(a_init, deadline, max_iters=5000, n_peaks=10, eps=0.05):
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    step_base = 0.3
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, b, M = multi_peak_subgradient(a, n_peaks, eps)
        grad_tan = grad - grad.mean()
        
        step = step_base / (1.0 + 0.05 * it)
        a_new = a - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Evaluate
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def lp_best_response(a, n_active=30, M_target=None):
    """LP best response: find g that minimizes max(a*g) given a."""
    n = len(a)
    b = fast_conv(a)
    M = b.max() if M_target is None else M_target
    
    top_indices = np.argsort(b)[::-1][:min(n_active, 2*n-1)]
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_indices):
        for j in range(n):
            m = int(k) - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def run_lp_descent(a_init, deadline, max_iters=100, n_active=30):
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    cur_na = n_active
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = lp_best_response(a, cur_na)
        if g is None:
            cur_na = min(cur_na * 2, 2*len(a)-1)
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1-t)*a + t*g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 0.01:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
            if cur_na > 10:
                cur_na = max(10, int(cur_na * 0.9))
        else:
            no_improve += 1
            cur_na = min(int(cur_na * 1.5), 2*len(a)-1)
            if no_improve > 20:
                break
    
    return best_a

def coord_optimize(a, i, b, n):
    """Optimize single coordinate a[i] by golden section search."""
    x_cur = a[i]
    
    k_lo = i
    k_hi = min(2*n-2, i+n-1)
    
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi+1):
        if k == 2*i:
            c = b[k] - x_cur*x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals)-1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0*x_cur*a[j]
                c_vals.append(c)
                slopes.append(2.0*a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    if k_lo > 0 or k_hi < 2*n-2:
        mask = np.ones(2*n-1, dtype=bool)
        mask[k_lo:k_hi+1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals) if c_vals else np.array([0.0])
    s_arr = np.array(slopes) if slopes else np.array([0.0])
    
    def obj(x):
        vals = c_arr + s_arr * x
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x*x
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0*x_cur, 0.5, np.sqrt(max(b.max(), 0.0)) + 0.5)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr*(hi - lo)
    b_pt = lo + gr*(hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr*(hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr*(hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj(0.0) <= obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

def run_coord_descent(a_init, deadline, max_sweeps=100, seed=0):
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = np.arange(n)
        rng.shuffle(order)
        b = fast_conv(a)
        improved = False
        
        for i in order:
            x_opt = coord_optimize(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2*n-2, i+n-1)
                for k in range(k_lo, k_hi+1):
                    if k == 2*i:
                        b[k] += x_opt*x_opt - old*old
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0*a[j]*(x_opt - old)
                improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not improved:
            no_improve += 5
        if no_improve > 15:
            break
    
    return best_a

def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = parent / parent.sum()
        inits.append(p.copy())
        p_noisy = np.maximum(p + rng.normal(0, 0.05, n), 1e-6)
        inits.append(p_noisy / p_noisy.sum())
        p_noisy2 = np.maximum(p + rng.normal(0, 0.15, n), 1e-6)
        inits.append(p_noisy2 / p_noisy2.sum())
    
    inits.append(np.ones(n) / n)
    
    # U-shaped
    u = t**2 + (1-t)**2 + 0.1
    inits.append(u / u.sum())
    
    # W-shaped
    w = (4*t - 1)**2 * (4*t - 3)**2 + 0.1
    inits.append(w / w.sum())
    
    # Cosine
    for f in [2, 3, 5]:
        c = 0.5 + 0.5*np.cos(2*np.pi*f*t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Gaussian
    g = np.exp(-3.0*(t-0.5)**2) + 0.05
    inits.append(g / g.sum())
    
    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    sr = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sr = np.maximum(sr, 0.01)
    inits.append(sr / sr.sum())
    
    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA on log-sum-exp smooth approximation with adaptive tau
2. Multi-peak subgradient descent (combines top-m peaks)
3. LP best-response with adaptive active sets
4. 1D golden-section coordinate optimization
5. Structured perturbation for escaping local minima
6. Multi-start with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto probability simplex {x>=0, sum(x)=1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) via log-sum-exp and full gradient."""
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau

    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        lo = max(0, i)
        hi = min(2 * n - 1, i + n)
        if lo < hi:
            grad[i] = 2.0 * np.dot(p[lo:hi], a[0:hi - lo])
    return smooth_val, grad, b


def run_fista(a_init, deadline, tau_start=50, tau_end=500, max_iters=10000, lr=0.8):
    """FISTA on smooth max with adaptive tau."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    t = 1.0
    y = a.copy()
    a_prev = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac

        _, grad_y, _ = smooth_max_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()

        step = lr / (1.0 + 0.01 * tau)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t = t_new

        if it % 10 == 0:
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()

    M = fast_conv(a).max()
    if M < best_M:
        best_M = M
        best_a = a.copy()

    return best_a


def multi_peak_subgradient(a, n_peaks=10, eps=0.05):
    """Combined subgradient from top n_peaks of convolution."""
    b = fast_conv(a)
    M = b.max()
    n = len(a)

    threshold = M * (1.0 - eps)
    peak_indices = np.where(b >= threshold)[0]

    if len(peak_indices) == 0:
        peak_indices = np.array([np.argmax(b)])

    if len(peak_indices) > n_peaks:
        peak_vals = b[peak_indices]
        top_idx = np.argsort(peak_vals)[::-1][:n_peaks]
        peak_indices = peak_indices[top_idx]
        peak_weights = peak_vals[top_idx] / peak_vals[top_idx].sum()
    else:
        peak_weights = np.ones(len(peak_indices)) / len(peak_indices)

    grad = np.zeros(n)
    for w, k in zip(peak_weights, peak_indices):
        ki = int(k)
        for i in range(n):
            j = ki - i
            if 0 <= j < n:
                grad[i] += 2.0 * w * a[j]

    return grad, b, M


def run_multi_peak_descent(a_init, deadline, max_iters=5000, n_peaks=10, eps=0.05, step_base=0.3):
    """Multi-peak subgradient descent with decaying step."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, M = multi_peak_subgradient(a, n_peaks, eps)
        grad_tan = grad - grad.mean()

        step = step_base / (1.0 + 0.03 * it)
        a_new = a - step * grad_tan
        a_new = proj_simplex(a_new)

        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()

        a = a_new

    return best_a


def lp_best_response(a, n_active=30):
    """LP best response: find g minimizing max(a*g) given a."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()

    top_indices = np.argsort(b)[::-1][:min(n_active, 2 * n - 1)]

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_indices):
        ki = int(k)
        for j in range(n):
            m = ki - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(top_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_descent(a_init, deadline, max_iters=100, n_active=30):
    """LP-based mirror descent with adaptive active set."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    cur_na = n_active

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = lp_best_response(a, cur_na)
        if g is None:
            cur_na = min(cur_na * 2, 2 * n - 1)
            no_improve += 1
            if no_improve > 10:
                break
            continue

        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 0.01:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
            if cur_na > 10:
                cur_na = max(10, int(cur_na * 0.9))
        else:
            no_improve += 1
            cur_na = min(int(cur_na * 1.5), 2 * n - 1)
            if no_improve > 20:
                break

    return best_a


def coord_optimize(a, i, b, n):
    """Optimize single coordinate a[i] via golden section search on max constraint."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals) if c_vals else np.array([0.0])
    s_arr = np.array(slopes) if slopes else np.array([0.0])

    def obj_at(x):
        vals = c_arr + s_arr * x
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x * x
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(2.0 * x_cur, 0.5, np.sqrt(max(b.max(), 0.0)) + 0.5)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coord_descent(a_init, deadline, max_sweeps=100, seed=0):
    """Coordinate descent with random ordering."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = np.arange(n)
        rng.shuffle(order)
        b = fast_conv(a)
        improved = False

        for i in order:
            x_opt = coord_optimize(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += x_opt * x_opt - old * old
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old)
                improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not improved:
            no_improve += 5
        if no_improve > 15:
            break

    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = parent / parent.sum()
        inits.append(p.copy())
        p1 = np.maximum(p + rng.normal(0, 0.04, n), 1e-6)
        inits.append(p1 / p1.sum())
        p2 = np.maximum(p + rng.normal(0, 0.1, n), 1e-6)
        inits.append(p2 / p2.sum())
        p3 = np.maximum(p + rng.normal(0, 0.2, n), 1e-6)
        inits.append(p3 / p3.sum())

    inits.append(np.ones(n) / n)

    u = t**2 + (1.0 - t)**2 + 0.1
    inits.append(u / u.sum())

    w = (4.0 * t - 1.0)**2 * (4.0 * t - 3.0)**2 + 0.1
    inits.append(w / w.sum())

    for f in [2, 3, 5]:
        c = 0.5 + 0.5 * np.cos(2.0 * np.pi * f * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    g = np.exp(-3.0 * (t - 0.5)**2) + 0.05
    inits.append(g / g.sum())

    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    sr = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sr = np.maximum(sr, 0.01)
    inits.append(sr / sr.sum())

    # Spectral init: flatten |FFT|^2
    a_spec = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_spec)
        M_s = bs.max()
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.1 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())

    return inits


def run_combined_optimize(a_init, deadline, n, seed):
    """Run a full optimization pipeline on a single initialization."""
    a = a_init.copy()
    best_M = fast_conv(a).max()

    # Stage 1: FISTA
    t1 = min(deadline, time.monotonic() + max(5, (deadline - time.monotonic()) * 0.4))
    if time.monotonic() < t1:
        a_f = run_fista(a, t1, tau_start=30, tau_end=400, lr=0.6)
        M_f = fast_conv(a_f).max()
        if M_f < best_M:
            best_M = M_f
            a = a_f.copy()

    # Stage 2: Multi-peak
    t2 = min(deadline, time.monotonic() + max(5, (deadline - time.monotonic()) * 0.3))
    if time.monotonic() < t2:
        a_mp = run_multi_peak_descent(a, t2, n_peaks=10, eps=0.05, step_base=0.25)
        M_mp = fast_conv(a_mp).max()
        if M_mp < best_M:
            best_M = M_mp
            a = a_mp.copy()

    # Stage 3: LP
    t3 = min(deadline, time.monotonic() + max(5, (deadline - time.monotonic()) * 0.2))
    if time.monotonic() < t3:
        a_lp = run_lp_descent(a, t3, max_iters=30, n_active=25)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            a = a_lp.copy()

    # Stage 4: Coordinate descent
    if time.monotonic() < deadline:
        a_cd = run_coord_descent(a, deadline, max_sweeps=20, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            a = a_cd.copy()

    return a
