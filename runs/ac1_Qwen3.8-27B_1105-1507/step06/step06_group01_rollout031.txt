import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def smooth_max_grad_fft(a, tau):
    """Compute log-sum-exp of conv(a,a) and its gradient using FFT.
    
    Returns (smooth_value, gradient) where gradient has same length as a.
    O(n log n) complexity.
    """
    n = len(a)
    b = fast_conv(a)
    
    # Log-sum-exp
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = b_max + np.log(exp_shifted.sum()) / tau
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient: grad_i = 2 * sum_k p[k] * a[k-i] (with a out of range = 0)
    # This is the cross-correlation of p (padded to length 2n-1) with a
    # corr[i] = sum_k p[k] * a[k-i]
    # Using FFT: pad p to length >= 2n, compute FFT, multiply by conj(FFT(a_padded))
    
    # Actually, let's think about this more carefully.
    # conv(a,a)[k] = sum_j a[j] * a[k-j] for k = 0, ..., 2n-2
    # grad_i = sum_k p[k] * d(conv(a,a)[k])/da_i
    # d(conv(a,a)[k])/da_i = a[k-i] + a[i-k] (where a[m]=0 if m<0 or m>=n)
    # Since conv(a,a) is symmetric (conv(a,a)[k] = conv(a,a)[2n-2-k]),
    # and p is symmetric too (since it's a function of conv(a,a)),
    # we have: grad_i = 2 * sum_k p[k] * a[k-i] (with a[m]=0 outside [0,n-1])
    
    # sum_k p[k] * a[k-i]: let m = k-i, so k = m+i
    # = sum_m p[m+i] * a[m] (where m+i is in [0, 2n-2] and m is in [0, n-1])
    # This is the cross-correlation of p (length 2n-1) with a (length n)
    
    # Using FFT:
    size = 1
    while size < 2 * n:
        size <<= 1
    p_padded = np.zeros(size)
    p_padded[:len(p)] = p
    a_padded = np.zeros(size)
    a_padded[:n] = a
    
    P = np.fft.fft(p_padded)
    A = np.fft.fft(a_padded)
    # Cross-correlation: ifft(conj(P) * A) gives sum_k p[k] * a[k+i]... 
    # Actually, let me be more careful.
    # Circular cross-correlation: (p ⋆ a)[i] = sum_k p[k] * conj(a[k-i])
    # For real signals: (p ⋆ a)[i] = sum_k p[k] * a[k-i]
    # In FFT: ifft(conj(FFT(p)) * FFT(a))
    
    corr = np.fft.ifft(np.conj(P) * A).real[:n]
    grad = 2.0 * corr
    
    return log_sum, grad

def run_fista_fft(a_init, tau_schedule, max_iters, deadline, lr_init=1.0):
    """FISTA with FFT-accelerated gradient and adaptive tau."""
    a = np.maximum(a_init.astype(np.float64), 0.0)

    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    # Lipschitz constant estimate (rough)
    L = 4.0 * n  # crude upper bound
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        # Interpolate tau
        tau = tau_schedule[0] * (tau_schedule[1] / tau_schedule[0]) ** frac
        
        _, grad_y = smooth_max_grad_fft(y, tau)
        
        # Project gradient to tangent of simplex
        grad_tan = grad_y - grad_y.mean()
        
        # Step size (Nesterov: 1/L)
        step = 1.0 / (tau * L * (1.0 + 0.1 * frac))
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        if it % 10 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a

def lp_best_response(a, n_active=50):
    """Solve LP best response using top n_active convolution coefficients."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    
    # Get active set
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    active = top_indices.tolist()
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def palindromic_fold(a):
    """Make a sequence palindromic by averaging with its reverse."""
    n = len(a)
    return (a + a[::-1]) / 2.0

def generate_inits(n, rng, parent=None):

    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

        if p.sum() > 0:

            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-4)
            inits.append(p_n / p_n.sum())
        # Palindromic version
        p_pal = palindromic_fold(p)
        if p_pal.sum() > 0:
            inits.append(p_pal / p_pal.sum())
    
    inits.append(np.ones(n) / n)
    
    # Various shape families
    for alpha in [0.5, 1, 2, 3, 5]:
        poly = t ** alpha + (1-t) ** alpha + 0.01
        inits.append(poly / poly.sum())
    
    # Gaussian
    for sigma in [0.1, 0.2, 0.3]:
        gauss = np.exp(-(t - 0.5) ** 2 / (2 * sigma ** 2)) + 0.01
        inits.append(gauss / gauss.sum())
    
    # Cosine windows
    for freq in [1, 2, 3]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.001)
        inits.append(cos_w / cos_w.sum())
    
    # "U-shaped" (high at edges)
    for alpha in [0.5, 1, 2]:
        u_shape = t ** (-alpha) if alpha < 1 else ((t + (1-t)) ** alpha) 
        u_shape = np.maximum(t ** alpha + (1-t) ** alpha, 0.01)
        inits.append(u_shape / u_shape.sum())

    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')

def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad_fft(a, tau):
    """Compute log-sum-exp of conv(a,a) and its gradient using FFT.
    
    Returns (smooth_value, gradient). O(n log n) complexity.
    """
    n = len(a)
    b = fast_conv(a)
    
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = b_max + np.log(exp_shifted.sum()) / tau
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient via FFT cross-correlation
    size = 1
    while size < 2 * n:
        size <<= 1
    p_padded = np.zeros(size)
    p_padded[:len(p)] = p
    a_padded = np.zeros(size)
    a_padded[:n] = a
    
    P = np.fft.fft(p_padded)
    A = np.fft.fft(a_padded)
    corr = np.fft.ifft(np.conj(P) * A).real[:n]
    grad = 2.0 * corr
    
    return log_sum, grad

        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_iteration(a_init, max_iters, deadline, n_active_start=40, n_active_end=100):
    """LP mirror descent with adaptive active set size."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        n_active = int(n_active_start + (n_active_end - n_active_start) * frac)
        
        g = lp_best_response(a, n_active)
        if g is None:
            no_improve += 2
            if no_improve > 12:
                break
            continue
        
        improved = False
        for t in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                break
    
    return best_a


def coordinate_descent_1d(a, i, b, n):
    """Find optimal value for a[i] given all others via golden section."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 1.0, 0.5)
    for _ in range(8):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 50.0:
            hi *= 2.0
        else:
            break
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, deadline, seed=0):
    """Run coordinate descent sweeps with shuffled order."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    while time.monotonic() < deadline:
        order = np.arange(n)
        rng.shuffle(order)
        
        b = fast_conv(a)
        sweep_improved = False
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                delta = x_opt - a[i]
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += 2 * x_opt * delta + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * delta
                a[i] = x_opt
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 3
        if no_improve > 12:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def palindromic_fold(a):
    """Make a sequence palindromic."""
    return (a + a[::-1]) / 2.0


def parametric_init(n, family, param, rng=None):
    """Generate a parametric sequence from a given family."""
    t = np.linspace(0, 1, n)
    if family == 'cosine':
        a = 0.5 + param * 0.5 * np.cos(2 * np.pi * t)
    elif family == 'parabolic':
        a = 1.0 + param * (2 * t - 1) ** 2
    elif family == 'gaussian':
        a = np.exp(-param * (2 * t - 1) ** 2)
    elif family == 'power_triangle':
        a = (1.0 - np.abs(2 * t - 1)) ** param + 0.01
    elif family == 'u_shape':
        a = t ** param + (1 - t) ** param
    elif family == 'bi_gaussian':
        c = 0.3
        a = np.exp(-((t - c) ** 2) / 0.02) + np.exp(-((t - (1-c)) ** 2) / 0.02) + 0.01 * param
    elif family == 'flat_rounded':
        edge = 0.15 * param
        a = np.ones(n)
        for i in range(n):
            dist = min(t[i], 1 - t[i])
            if dist < edge:
                a[i] = 1.0 - 0.5 * (1 - np.cos(np.pi * dist / edge))
        a = a + 0.001
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.001)
    if rng is not None:
        a = a + rng.normal(0, 0.005, n)
        a = np.maximum(a, 0.001)
    return a / a.sum()


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.01, 0.02, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-4)
            inits.append(p_n / p_n.sum())
        p_pal = palindromic_fold(p)
        if p_pal.sum() > 0:
            inits.append(p_pal / p_pal.sum())
    
    inits.append(np.ones(n) / n)
    
    # Parametric families
    inits.append(parametric_init(n, 'cosine', 0.3, rng))
    inits.append(parametric_init(n, 'cosine', 0.6, rng))
    inits.append(parametric_init(n, 'parabolic', 0.2, rng))
    inits.append(parametric_init(n, 'parabolic', 0.5, rng))
    inits.append(parametric_init(n, 'parabolic', 1.0, rng))
    inits.append(parametric_init(n, 'gaussian', 2.0, rng))
    inits.append(parametric_init(n, 'gaussian', 5.0, rng))
    inits.append(parametric_init(n, 'gaussian', 10.0, rng))
    inits.append(parametric_init(n, 'power_triangle', 0.5, rng))
    inits.append(parametric_init(n, 'power_triangle', 2.0, rng))
    inits.append(parametric_init(n, 'power_triangle', 5.0, rng))
    inits.append(parametric_init(n, 'u_shape', 0.5, rng))
    inits.append(parametric_init(n, 'u_shape', 0.2, rng))
    inits.append(parametric_init(n, 'bi_gaussian', 1.0, rng))
    inits.append(parametric_init(n, 'flat_rounded', 1.0, rng))
    inits.append(parametric_init(n, 'flat_rounded', 2.0, rng))
    
    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.001)
    inits.append(smooth / smooth.sum())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_combined_optimize(a0, deadline, seed=0):
    """Run combined optimization: FISTA -> LP -> CD."""
    best = np.maximum(a0.astype(np.float64), 0.0)
    s = best.sum()
    if s < 1e-10:
        best = np.ones_like(best) / len(best)
    else:
        best = best / s
    n = len(best)
    best_M = fast_conv(best).max()
    
    t_start = time.monotonic()
    total_time = max(1, deadline - t_start)
    
    # Stage 1: FFT FISTA (50%)
    t1 = t_start + 0.5 * total_time
    if time.monotonic() < t1:
        # Multi-tau FISTA
        time_per_tau = (t1 - t_start) / 3.0
        for tau_pair in [(30, 150), (80, 400), (200, 1000)]:
            if time.monotonic() > t1:
                break
            sub_dl = min(t1, time.monotonic() + time_per_tau)
            a_f = run_fista_fft(best, tau_pair[0], tau_pair[1], 50000, sub_dl)
            M_f = fast_conv(a_f).max()
            if M_f < best_M:
                best_M = M_f
                best = a_f.copy()
    
    # Stage 2: LP best response (30%)
    t2 = t_start + 0.8 * total_time
    if time.monotonic() < t2:
        a_lp = run_lp_iteration(best, 200, t2, n_active_start=40, n_active_end=100)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
    
    # Stage 3: Coordinate descent (20%)
    if time.monotonic() < deadline:
        a_cd = run_coordinate_descent(best, deadline, seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
    
    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining FFT-accelerated FISTA, LP best-response,
    parametric initialization, and coordinate descent across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)
    
    # Determine lengths to try
    n_parent = parent.size
    lengths_to_try = [n_parent]
    for delta in [-64, -32, 32, 64, 128, 256, -128]:
        nl = n_parent + delta
        if 256 <= nl <= 2048:
            lengths_to_try.append(nl)
    for nl in [384, 512, 640, 768, 896, 1024, 1280]:
        if 256 <= nl <= 2048:
            lengths_to_try.append(nl)
    
    lengths_to_try = sorted(set(lengths_to_try))
    lengths_to_try.sort(key=lambda x: abs(x - n_parent))
    lengths_to_try = lengths_to_try[:8]
    
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-length search: {lengths_to_try}", flush=True)
    print(f"{'='*60}", flush=True)
    
    reserve = 350
    time_per_length = max(50, (deadline - time.monotonic() - reserve) / len(lengths_to_try))
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - reserve:
            break
        
        remaining = deadline - reserve - time.monotonic()
        if remaining < 40:
            break
        
        time_alloc = min(time_per_length, remaining)
        print(f"\n  n={n}, alloc={time_alloc:.0f}s", flush=True)
        
        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        
        # Select top init by quick J evaluation to save time
        scored_inits = []
        for a0 in inits:
            J0 = compute_J(a0)
            scored_inits.append((J0, a0))
        scored_inits.sort(key=lambda x: x[0])
        
        # Use top 6 inits
        top_inits = [a for _, a in scored_inits[:6]]
        time_per_init = time_alloc / max(1, len(top_inits))
        
        best_for_n = None
        best_M_n = float('inf')
        
        for ai, a0 in enumerate(top_inits):
            if time.monotonic() > deadline - reserve:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            
            sub_dl = min(time.monotonic() + time_per_init, deadline - reserve)
            if time.monotonic() >= sub_dl:
                M = compute_J(a0)
                if M < best_M_n:
                    best_M_n = M
                    best_for_n = a0.copy()
                continue
            
            a_opt = run_combined_optimize(a0, sub_dl, seed=seed + n * 10 + ai)
            M_opt = fast_conv(a_opt).max()
            
            if M_opt < best_M_n:
                best_M_n = M_opt
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best J={J_val:.6f} (elapsed={time.monotonic()-time.monotonic()+0:.0f}s)", flush=True)
            tracker.update(best_for_n)
    
    tracker.report("After Phase 1")
    
    # Phase 2: Intensive LP refinement
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Intensive LP refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t_dl = min(deadline - 80, time.monotonic() + 120)
        a_lp = run_lp_iteration(tracker.best, 300, t_dl, n_active_start=50, n_active_end=120)
        tracker.update(a_lp)
        
        # Palindromic refinement
        if time.monotonic() < deadline - 60:
            a_pal = palindromic_fold(a_lp)
            a_pal = np.maximum(a_pal, 0)
            s = a_pal.sum()
            if s > 0:
                a_pal = a_pal / s
                a_pal_opt = run_combined_optimize(a_pal, min(deadline - 40, time.monotonic() + 50), seed=555)
                tracker.update(a_pal_opt)
    
    tracker.report("After Phase 2")
    
    # Phase 3: High-tau FISTA for fine-tuning
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: High-tau FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 50:
        for tau_pair in [(500, 2000), (1000, 4000)]:
            if time.monotonic() > deadline - 40:
                break
            a_f = run_fista_fft(tracker.best, tau_pair[0], tau_pair[1],
                                50000, min(deadline - 30, time.monotonic() + 40))
            tracker.update(a_f)
    
    tracker.report("After Phase 3")
    
    # Phase 4: Final coordinate descent + LP
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Final polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 15:
        a_cd = run_coordinate_descent(tracker.best, deadline - 15, seed=77777)
        tracker.update(a_cd)
        
        if time.monotonic() < deadline - 10:
            a_final = run_lp_iteration(a_cd, 40, deadline, n_active_start=30, n_active_end=70)
            tracker.update(a_final)
    
    tracker.report("After Phase 4")
    
    # Phase 5: Try a different length from best
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 20:
        n_cur = tracker.best.size
        for mult in [1.3, 0.7]:
            if time.monotonic() > deadline - 15:
                break
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 2048)
            new_n = (new_n // 16) * 16
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 10, time.monotonic() + 40)
            a_opt = run_combined_optimize(a_ext, sub_dl, seed=seed + 999)
            tracker.update(a_opt)
    
    tracker.report("After Phase 5")
    
    # Final safety
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
