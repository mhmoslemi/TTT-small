import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax_obj_and_grad(x, beta):
    """
    Soft-max objective and gradient.
    x is the raw parameter, a = softmax(x) ensures positivity and sum=1.
    """
    # Convert to a via softmax
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)
    
    n = len(a)
    conv = compute_conv(a)
    
    # Soft-max
    beta_conv = beta * conv
    beta_conv_max = np.max(beta_conv)
    exp_bc = np.exp(beta_conv - beta_conv_max)
    p = exp_bc / np.sum(exp_bc)
    
    # Objective: (2n/beta) * (log(sum exp(beta*conv)) ) = (2n/beta) * (beta_max + log(sum exp(bc - beta_max)))
    obj = (2.0 * n / beta) * (beta_conv_max + np.log(np.sum(exp_bc)))
    
    # Gradient with respect to a:
    # d(obj)/d(a[i]) = 4n * sum_j p[i+j] * a[j]  (cross-correlation)
    # But we need d(obj)/d(x) = sum_i (d(obj)/d(a[i])) * (d(a[i])/d(x))
    # d(a[i])/d(x[k]) = a[i]*(delta_ik - a[k])
    # So d(obj)/d(x[k]) = sum_i grad_a[i] * a[i] * (delta_ik - a[k])
    # = grad_a[k] * a[k] - a[k] * sum_i grad_a[i] * a[i]
    
    # Compute grad_a
    a_flipped = a[::-1]
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    p_fl = np.fft.fft(p, N)
    a_fl_fl = np.fft.fft(a_flipped, N)
    cross = np.fft.ifft(p_fl * a_fl_fl).real[:n]  # cross[i] = sum_j p[i+j]*a[j]
    
    grad_a = 4.0 * n * cross
    
    # Chain rule to get grad_x
    grad_x = grad_a * a - a * np.dot(grad_a, a)
    
    return obj, grad_x

        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, np.maximum(a, 0))
    return a_new


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_active(a, M, active_indices, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_indices, g >= 0."""
    n = len(a)
    if n > 2500:
        return None
    A = build_toeplitz(a)
    A_active = A[active_indices, :]
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_active, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_full(a, M, time_limit=10):
    """Solve LP with all constraints."""
    n = len(a)
    if n > 2500:
        return None
    idx = list(range(2 * n - 1))
    return solve_lp_active(a, M, idx, time_limit)


def get_active_set(conv, epsilon=0.0):
    """Get indices where conv[k] >= max - epsilon."""
    max_val = np.max(conv)
    threshold = max_val - epsilon
    return np.where(conv >= threshold)[0]


def lp_alternate_active(a, max_iters=100, time_budget=60.0, eps_init=0.01,
                        eps_min=0.001, expand_factor=3):
    """
    LP alternating with adaptive active set.
    Starts with a small active set and expands it as needed.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Get active set with adaptive epsilon
        eps = max(eps_min, eps_init * (0.95 ** it))
        active = get_active_set(conv, epsilon=eps * M)

        # Expand active set to include neighbors
        if len(active) < 2 * n - 1:
            expanded = set(active.tolist())
            for k in active:
                for dk in range(-expand_factor, expand_factor + 1):
                    if 0 <= k + dk < 2 * n - 1:
                        expanded.add(k + dk)
            active = np.array(sorted(expanded))

        g = solve_lp_active(a, M, active, time_limit=8)
        if g is None:
            g = solve_lp_full(a, M, time_limit=10)
        if g is None:
            break

        # Line search
        best_t_obj = np.inf
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

    return best_a, best_obj

def softmax_lbfgs(a, beta_start=30.0, beta_end=300.0, max_iters=200, time_budget=60.0):
    """
    Soft-max L-BFGS optimization.
    Uses smooth approximation of max to escape local minima.
    """
    n = len(a)
    a_cur = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a_cur.copy()
    best_obj = compute_obj(a_cur)
    start = time.monotonic()

    def objective_and_grad(x):
        a_local = project_nonneg_sum1(x)
        conv = compute_conv(a_local)
        bc = beta_cur * conv
        bc_max = np.max(bc)
        exp_bc = np.exp(bc - bc_max)
        sum_exp = np.sum(exp_bc)
        obj = (2.0 * n / beta_cur) * (bc_max + np.log(sum_exp))

        # Gradient
        p = exp_bc / sum_exp
        a_flipped = a_local[::-1]
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        cross = np.fft.ifft(np.fft.fft(p, N) * np.fft.fft(a_flipped, N)).real[:n]
        grad = 4.0 * n * cross

        return obj, grad

    # Schedule beta
    for beta_cur in [beta_start, beta_start * 2, beta_start * 4,
                     beta_start * 8, min(beta_end, beta_start * 16),
                     min(beta_end, beta_start * 32), beta_end]:
        if time.monotonic() - start > time_budget:
            break

        # Set closure variable
        beta_cur = beta_cur

        def obj_grad(x, bc=beta_cur):
            a_local = project_nonneg_sum1(x)
            conv = compute_conv(a_local)
            bconv = bc * conv
            bconv_max = np.max(bconv)
            exp_bc = np.exp(bconv - bconv_max)
            sum_exp = np.sum(exp_bc)
            obj = (2.0 * n / bc) * (bconv_max + np.log(sum_exp))

            p = exp_bc / sum_exp
            a_flipped = a_local[::-1]
            N = 1
            while N < 2 * n - 1:
                N <<= 1
            cross = np.fft.ifft(np.fft.fft(p, N) * np.fft.fft(a_flipped, N)).real[:n]
            grad = 4.0 * n * cross
            return obj, grad

        try:
            result = minimize(obj_grad, a_cur, method='L-BFGS-B',
                              bounds=[(0, None)] * n,
                              options={'maxiter': max_iters // 7, 'ftol': 1e-10,
                                       'gtol': 1e-8})
            a_new = project_nonneg_sum1(result.x)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a_cur = a_new.copy()
        except Exception:
            pass

    return best_a, best_obj


def freq_domain_flatten(a, max_iters=150, time_budget=25.0):
    """
    Frequency-domain approach: iteratively flatten the power spectrum
    to reduce auto-correlation peaks.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2

        # Target: flatten power spectrum
        # Use logarithmic compression
        p_max = np.max(power)
        if p_max < 1e-20:
            break

        # Iteratively reduce peaks in power spectrum
        alpha = 0.85 + 0.1 * (it / max_iters)
        p_norm = power / p_max
        p_compressed = p_norm ** alpha
        p_compressed = np.maximum(p_compressed, 1e-10)

        # Reconstruct with compressed magnitude, keep phase
        magnitude = np.sqrt(p_compressed * p_max)
        phase = np.angle(fa)
        fa_new = magnitude * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def chase_peak(a, n_steps=50, step_size=0.03):
    """Chase-the-peak: reduce the maximum convolution peak."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += a[j]
            j2 = i + n - 1 - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]

        grad_centered = grad - np.mean(grad)

        for ss in [step_size, step_size * 0.5, step_size * 1.5]:
            a_try = a - ss * grad_centered
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a, best_obj


def symmetric_optimize(a, max_iters=50, time_budget=30.0):
    """
    Optimize under the symmetry constraint a[i] = a[n-1-i].
    This reduces the problem size and may find better local minima.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))

    # Enforce symmetry
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = (a[i] + a[n - 1 - i]) / 2
        a_sym[n - 1 - i] = a_sym[i]
    a_sym = project_nonneg_sum1(a_sym)

    best_a = a_sym.copy()
    best_obj = compute_obj(a_sym)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a_sym)
        M = float(np.max(conv))

        # Solve LP with symmetry constraint
        # Variables: a[0], ..., a[half-1] (n//2 or (n+1)//2 variables)
        # a[n-1-i] = a[i]
        # Constraint: (a*g)[k] <= M where g is also symmetric

        # Build effective constraint matrix for symmetric variables
        # conv[k] = sum_i a[i]*a[k-i], with a[n-1-j] = a[j]
        # This is complex, so use the full LP and project to symmetry

        g = solve_lp_full(a_sym, M, time_limit=8)
        if g is None:
            break

        # Project g to symmetry
        g_sym = np.zeros(n)
        for i in range(half):
            g_sym[i] = (g[i] + g[n - 1 - i]) / 2
            g_sym[n - 1 - i] = g_sym[i]
        g_sym = project_nonneg_sum1(g_sym)

        best_t_obj = np.inf
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a_sym + t * g_sym
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            a_sym = best_a
        else:
            if it > 5:
                break

    return best_a, best_obj


def subgradient_descent(a, max_iters=300, time_budget=30.0, tau=0.3):
    """
    Subgradient method with softmax weighting over peaks.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    momentum = 0.9
    step = 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Soft-max weights
        shifted = (conv - cmax) / tau
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient: grad[i] = sum_k weights[k] * 2*a[k-i] (for valid k-i)
        # Use FFT for efficiency
        a_flipped = a[::-1]
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        cross = np.fft.ifft(np.fft.fft(weights, N) * np.fft.fft(a_flipped, N)).real[:n]
        grad = 4.0 * n * cross  # factor of 2 from chain rule, 2n from objective

        # Actually for the exact max (not soft-max), the gradient is:
        # grad[i] = 2*a[k*-i] (for valid) where k* is argmax
        # For soft-max: grad[i] = 2 * sum_k p_k * a[k-i]
        # Let me use the soft-max gradient
        # cross[i] = sum_j weights[i+j] * a[j]
        # grad[i] = 2 * cross[i] * (2n factor for objective scaling)
        # Wait, the objective is 2*n*max(conv), so:
        # d_obj/d_a[i] = 2*n * d_max/d_a[i] = 2*n * sum_k p_k * 2*a[k-i]
        # = 4*n * cross[i]
        # That's what I have. Good.

        grad_centered = grad - np.mean(grad)
        v = momentum * v - step * grad_centered
        a_new = a - v
        a_new = project_nonneg_sum1(a_new)

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        step *= 0.999

    return best_a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []

    inits.append(np.ones(n) / n)

    # Raised cosine
    x = np.linspace(0, 1, n)
    rc = np.sin(np.pi * x) ** 2
    inits.append(rc / rc.sum())

    # Gaussian
    x = np.linspace(-3, 3, n)
    g = np.exp(-x ** 2 / 3)
    inits.append(g / g.sum())

    # Bimodal
    x = np.linspace(0, 1, n)
    bi = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
    inits.append(bi / bi.sum())

    # Raised cosine with different width
    x = np.linspace(0, 1, n)
    rc2 = np.cos(np.pi * x) ** 2 + 0.01
    inits.append(rc2 / rc2.sum())

    # Flat-top window (approximate)
    a = np.ones(n)
    fade = max(5, n // 20)
    for i in range(fade):
        a[i] = 0.5 * (1 - np.cos(np.pi * i / fade))
        a[n - 1 - i] = a[i]
    inits.append(a / a.sum())

    # Random smooth
    rnd = rng.random(n) + 0.01
    # Smooth it
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    rnd_smooth = np.convolve(rnd, kernel, mode='same')
    inits.append(rnd_smooth / rnd_smooth.sum())

    if parent is not None:
        inits.append(resample_sequence(parent, n))
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.15, n))
        inits.append(project_nonneg_sum1(pert))
        pert2 = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.3, n))
        inits.append(project_nonneg_sum1(pert2))

    return inits


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def multi_strategy_optimize(a_init, time_budget, rng, allow_symmetric=True):
    """
    Run multiple strategies in cycle on a given initial sequence.
    """
    n = len(a_init)
    a = project_nonneg_sum1(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    cycle = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        strategy = cycle % 6
        alloc = min(remaining * 0.4, 40.0)

        if strategy == 0:
            # LP alternating with active set
            a_new, obj_new = lp_alternate_active(a, max_iters=30, time_budget=alloc,
                                                 eps_init=0.05, eps_min=0.005)
        elif strategy == 1:
            # Soft-max L-BFGS
            a_new, obj_new = softmax_lbfgs(a, beta_start=30, beta_end=200,
                                           max_iters=100, time_budget=alloc)
        elif strategy == 2:
            # Frequency domain flattening
            a_new, obj_new = freq_domain_flatten(a, max_iters=80, time_budget=alloc)
        elif strategy == 3:
            # Chase peak
            a_new, obj_new = chase_peak(a, n_steps=40, step_size=0.03)
        elif strategy == 4 and allow_symmetric:
            # Symmetric optimization
            a_new, obj_new = symmetric_optimize(a, max_iters=30, time_budget=alloc)
        else:
            # Subgradient
            a_new = subgradient_descent(a, max_iters=100, time_budget=alloc, tau=0.3)
            obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

        cycle += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None and parent.size > 0:
        parent_obj = compute_obj(parent)
        print(f"Parent loaded (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score and parent_obj < np.inf:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700)
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    print(f"Initial best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 1: Quick screening with diverse initializations (60s) ===
    print("\n=== Phase 1: Initialization Screening ===")
    phase1_budget = min(60.0, (deadline - start_time) * 0.04)
    phase1_start = time.monotonic()

    n_parent = len(best_seq)
    n_options = [n_parent, max(100, n_parent - 50), n_parent + 50,
                 n_parent + 100, min(2000, n_parent + 200),
                 max(100, n_parent - 100)]
    n_options = list(dict.fromkeys([n for n in n_options if 50 <= n <= 3000]))

    candidates = []
    per_n_budget = max(5.0, phase1_budget / len(n_options))

    for n_try in n_options:
        if time.monotonic() - phase1_start > phase1_budget:
            break
        inits = generate_initializations(n_try, rng, parent)
        per_init_time = max(1.0, per_n_budget / len(inits))

        for init in inits:
            if time.monotonic() - phase1_start > phase1_budget:
                break
            a_q, obj_q = chase_peak(init, n_steps=15, step_size=0.03)
            if obj_q < np.inf:
                candidates.append((obj_q, a_q, n_try))

    candidates.sort(key=lambda x: x[0])
    print(f"  Screened {len(candidates)} candidates, best: {candidates[0][0] if candidates else 'N/A'}")

    for obj_c, a_c, n_c in candidates[:3]:
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  New best from screening: n={n_c}, obj={best_score:.8f}")

    # === Phase 2: Intensive multi-strategy optimization (700s) ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    print(f"Best before: {best_score:.8f}")
    phase2_budget = min(700.0, (deadline - start_time) * 0.50)
    phase2_start = time.monotonic()

    # Focus on top candidates
    top_candidates = [(best_score, best_seq.copy(), len(best_seq))]
    for obj_c, a_c, n_c in candidates[:2]:
        if obj_c < best_score + 0.1:
            top_candidates.append((obj_c, a_c.copy(), n_c))

    for ci, (obj_init, a_init, n_init) in enumerate(top_candidates):
        if time.monotonic() - phase2_start > phase2_budget:
            break
        alloc = (phase2_budget - (time.monotonic() - phase2_start)) / max(1,
                 len(top_candidates) - ci)
        alloc = min(alloc, 350.0)
        if alloc < 20:
            break

        print(f"  Candidate {ci} (n={n_init}): obj={obj_init:.8f}, budget={alloc:.0f}s")

        a_opt, obj_opt = multi_strategy_optimize(a_init, alloc, rng, allow_symmetric=True)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"    ** NEW BEST: n={n_init}, obj={best_score:.8f} **")

    # === Phase 3: Resilience - try to escape local minimum (200s) ===
    print(f"\n=== Phase 3: Escape Local Minima ===")
    print(f"Best before: {best_score:.8f}")
    phase3_budget = min(200.0, (deadline - time.monotonic()) * 0.20)
    phase3_start = time.monotonic()

    # Multiple perturbations + re-optimization
    n_best = len(best_seq)
    for attempt in range(min(8, int(phase3_budget / 20))):
        if time.monotonic() - phase3_start > phase3_budget:
            break

        # Random perturbation
        noise_scale = 0.1 + 0.1 * attempt
        a_pert = best_seq * np.exp(rng.normal(0, noise_scale, n_best))
        a_pert = project_nonneg_sum1(a_pert)

        # Quick optimization
        a_opt, obj_opt = multi_strategy_optimize(a_pert, 15.0, rng, allow_symmetric=False)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Attempt {attempt}: NEW BEST {best_score:.8f}")

    # === Phase 4: Multi-length exploration (150s) ===
    print(f"\n=== Phase 4: Multi-Length Exploration ===")
    print(f"Best before: {best_score:.8f}")
    phase4_budget = min(150.0, (deadline - time.monotonic()) * 0.15)
    phase4_start = time.monotonic()

    n_current = len(best_seq)
    n_tries = sorted(set([n_current - 30, n_current - 50, n_current - 80,
                          n_current + 30, n_current + 50, n_current + 80,
                          n_current + 150, n_current + 200]))
    n_tries = [n for n in n_tries if 100 <= n <= 3000]
    n_tries = list(dict.fromkeys(n_tries))

    per_n_budget = max(5.0, phase4_budget / max(1, len(n_tries)))

    for n_try in n_tries:
        if time.monotonic() - phase4_start > phase4_budget:
            break
        a_res = resample_sequence(best_seq, n_try)
        a_res = project_nonneg_sum1(a_res)

        a_opt, obj_opt = multi_strategy_optimize(a_res, per_n_budget, rng,
                                                 allow_symmetric=True)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 10 and best_seq is not None:
        # Focus all remaining time on the best sequence
        a_final, obj_final = multi_strategy_optimize(
            best_seq, remaining5 - 5.0, rng, allow_symmetric=True)

        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Polish improved to: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
