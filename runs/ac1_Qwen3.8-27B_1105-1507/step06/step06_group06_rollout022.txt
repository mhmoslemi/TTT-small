import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog

def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

    # Dirichlet kernel: sum_{k=0}^{2n-2} W_N^{mk}
    k_max = 2 * n - 2
    # Use closed form: sum_{k=0}^{M} r^k = (1 - r^{M+1})/(1-r) where r = W_N^m
    # For m=0: sum = M+1 = 2n-1
    # For m!=0: sum = (1 - W_N^{m*(2n-1)}) / (1 - W_N^m)
    target = np.ones(N, dtype=np.complex128)
    mask = m != 0
    m_vals = m[mask]
    W = np.exp(-2j * np.pi * m_vals / N)
    W_pow = W ** (2 * n - 1)
    denom = 1.0 - W
    # Handle numerical issues where denom is very small
    small = np.abs(denom) < 1e-12
    for idx in range(len(m_vals)):
        if small[idx]:
            target[m_vals[idx]] = 2 * n - 1
        else:
            target[m_vals[idx]] = (1.0 - W_pow[idx]) / denom[idx]
    target /= (2 * n - 1)
    return target


def gs_target_iteration(a, n_lambda=0.5, max_iters=200, time_budget=30.0,
                        lambda_schedule=None):
    """
    GS iteration with target spectrum: drive F(a)^2 toward the flat-convolution target.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1
    N = max(N, 4 * n)  # extra zero-padding for stability

    target = build_target_spectrum(n, N)
    # target should have |target[0]| = 1 and be real-positive at DC
    # Normalize target so that target[0] = 1 (since sum(a)=1 means F[0]=1)
    target = target.copy()
    target[0] = 1.0 + 0j

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adapt lambda
        if lambda_schedule == 'anneal':
            lam = n_lambda * (1.0 - it / max(float(max_iters), 1)) + 0.05
        elif lambda_schedule == 'constant':
            lam = n_lambda
        else:
            lam = n_lambda

        F = fft(a, N)
        F_sq = F * F

        # Blend toward target
        F_sq_new = (1.0 - lam) * F_sq + lam * target

        # Complex square root
        F_new = np.sqrt(F_sq_new + 0j)

        # IFFT and project
        a_new = ifft(F_new).real[:n]
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def gs_magnitude_iteration(a, max_iters=200, time_budget=30.0, alpha_init=0.6, alpha_final=0.2):
    """
    Alternative GS: flatten the magnitude of F(a)^2 toward the target magnitude.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1
    N = max(N, 4 * n)

    target = build_target_spectrum(n, N)
    target_mag = np.abs(target)
    target_mag[0] = 1.0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        alpha = alpha_init + (alpha_final - alpha_init) * (it / max(float(max_iters), 1))

        F = fft(a, N)
        F_sq = F * F
        F_sq_mag = np.abs(F_sq)
        F_sq_phase = np.angle(F_sq)

        # Blend magnitude toward target
        # Use geometric interpolation (log domain)
        F_sq_mag = np.maximum(F_sq_mag, 1e-15)
        target_mag_safe = np.maximum(target_mag, 1e-15)
        log_mag = (1.0 - alpha) * np.log(F_sq_mag) + alpha * np.log(target_mag_safe)
        F_sq_mag_new = np.exp(log_mag)

        # Reconstruct F_sq with new magnitude, keep phase
        F_sq_new = F_sq_mag_new * np.exp(1j * F_sq_phase)
        F_sq_new[0] = 1.0 + 0j  # enforce DC

        F_new = np.sqrt(F_sq_new + 0j)
        a_new = ifft(F_new).real[:n]
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def compute_gradient_fft(a, tau):
    """
    Compute gradient of phi_tau(conv(a,a)) where phi_tau is smooth max (log-sum-exp).
    Returns gradient vector of length n.
    grad_i = 2 * sum_j a[j] * w[i+j] where w = softmax(conv/tau)
    Computed via FFT-based convolution.
    """
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    c_max = float(np.max(conv))

    # Softmax weights (numerically stable)
    shifted = (conv - c_max) / tau
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum

    # grad_i = 2 * sum_j a[j] * weights[i+j]
    # = 2 * (a * w_rev)[2n-2-i] where w_rev[k] = weights[2n-2-k]
    w_rev = weights[::-1]

    # Compute linear convolution via FFT
    conv_size = n + (2 * n - 1) - 1
    size = 1
    while size < conv_size:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    conv_result = irfft(fa * fw, size)

    # grad[i] = 2 * conv_result[2n-2-i] for i = 0, ..., n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]

    return grad


def projected_gradient_nesterov(a, max_iters=500, time_budget=40.0,
                                 tau_init=0.05, tau_final=0.01,
                                 step_init=0.1, momentum=0.9):
    """
    Projected gradient descent with Nesterov momentum on the simplex.
    Minimizes phi_tau(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    step = step_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        tau = tau_init + (tau_final - tau_init) * (it / max(float(max_iters), 1))
        grad = compute_gradient_fft(a, tau)

        # Center gradient (project onto tangent space of simplex)
        grad_centered = grad - np.mean(grad)

        # Nesterov step
        v = momentum * v - step * grad_centered
        a_new = a + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            step = min(step * 1.02, 0.3)  # slight increase on success
        else:
            step *= 0.9  # decrease on failure
            a = project_simplex(a + 0.5 * v)  # half step

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 1500:
        return None
    try:
        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate(a, max_iters=30, time_budget=30.0):
    """LP alternating with line search."""
    n = len(a)
    if n > 1500:
        return a, compute_obj(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_max_sum(a, M, time_limit=10)
        if g is None:
            break

        # Line search
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = project_simplex((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break

    return best_a, best_obj


def chase_peak(a, n_steps=40, step_size=0.04, time_budget=15.0):
    """Iteratively reduce the maximum convolution peak."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for _ in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient for the max peak
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += a[j]
            j2 = 2 * n - 2 - k_star + i
            if 0 <= j2 < n:
                grad[i] += a[j2]

        grad_centered = grad - np.mean(grad)

        # Try a few step sizes
        improved = False
        for ss in [step_size, step_size * 0.5, step_size * 2.0, step_size * 0.25]:
            a_try = project_simplex(a - ss * grad_centered)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_obj


def simulated_annealing(a, rng, max_iters=300, T_init=0.3, T_min=0.005, time_budget=60.0):
    """Simulated annealing with diverse moves."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    cur_a = a.copy()
    cur_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = cur_obj
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T_init * (T_min / T_init) ** (it / max(float(max_iters), 1))
        T = max(T, 1e-6)

        move_type = rng.integers(0, 4)
        a_trial = cur_a.copy()

        if move_type == 0:
            # Window noise
            wsize = max(5, n // 20)
            wstart = int(rng.integers(0, n - wsize))
            noise = rng.normal(0, 0.3, wsize)
            a_trial[wstart:wstart + wsize] += noise
        elif move_type == 1:
            # Window swap
            wsize = max(5, n // 25)
            w1 = int(rng.integers(0, n - wsize))
            w2 = int(rng.integers(0, n - wsize))
            if abs(w1 - w2) >= wsize:
                a_trial[w1:w1 + wsize], a_trial[w2:w2 + wsize] = \
                    a_trial[w2:w2 + wsize].copy(), a_trial[w1:w1 + wsize].copy()
        elif move_type == 2:
            # Gaussian perturbation
            a_trial = a_trial * np.exp(rng.normal(0, 0.1, n))
        else:
            # Sparse update: modify a few random positions
            n_mod = max(5, n // 30)
            idx = rng.choice(n, size=n_mod, replace=False)
            a_trial[idx] += rng.normal(0, 0.1, n_mod)

        a_trial = project_simplex(a_trial)
        trial_obj = compute_obj(a_trial)

        delta = trial_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_a = a_trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular
    x = np.linspace(0, 1, n)
    tri = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
    tri = tri + 0.1
    inits.append(tri / tri.sum())

    # Sinusoidal
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    # Gaussian center
    gauss = np.exp(-((x - 0.5) ** 2) / 0.05)
    inits.append(gauss / gauss.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Bimodal
    bimodal = np.exp(-((x - 0.3) ** 2) / 0.01) + np.exp(-((x - 0.7) ** 2) / 0.01) + 0.05
    inits.append(bimodal / bimodal.sum())

    # Arcsine-like
    arcs = 1.0 / np.sqrt(np.maximum(x * (1 - x), 1e-6))
    inits.append(arcs / arcs.sum())

    # If parent available
    if parent is not None:
        parent_resampled = resample_sequence(parent, n)
        inits.append(project_simplex(parent_resampled))
        pert = parent_resampled * np.exp(rng.normal(0, 0.15, n))
