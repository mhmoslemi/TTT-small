    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 + np.arange(1, n+1) * u - cssv > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

The dual function is: `inf_{a>=0, M} L = inf_M [M(1 - sum λ_k)] + inf_{a>=0} [sum_k λ_k (a*a)_k - μ sum(a) - sum_i ν_i a_i]`

For the inf over M to be finite, we need `sum λ_k = 1`.

For the inf over `a >= 0`: the objective is `sum_k λ_k (a*a)_k - (μ + ν) • a` where `ν` is a vector.

`sum_k λ_k (a*a)_k = sum_k λ_k sum_i a_i a_{k-i} = sum_i sum_k λ_k a_i a_{k-i} = sum_i a_i * (sum_k λ_k a_{k-i})`

Let `w_i = sum_k λ_k a_{k-i}` (this is `(λ * a)_i` with appropriate indexing).

Then the inf over `a >= 0` of `sum_i a_i * w_i - sum_i (μ_i + ν_i) a_i` where `μ_i + ν_i = μ + ν_i` (scalar μ plus vector ν_i).

For the inf to be finite, we need `w_i >= μ + ν_i` for all i, i.e., `w_i - μ >= 0` for all i (since we can set `ν_i = w_i - μ` if `w_i >= μ`).

So the dual is: max `-μ` subject to `sum_k λ_k = 1`, `λ_k >= 0`, `(λ*a)_i >= μ` for all i.

But this still has `a` in it, so it's not a clean dual. The problem is non-convex.

Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent (multiplicative weights), stochastic subgradient,
LP-based refinement, and multi-size exploration.

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(1 + np.arange(1, n+1) * u - cssv > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def subgradient_convolve(a, k):
    """Subgradient of (a*a)_k w.r.t. a. Returns vector of length n."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            g[i] += 2.0 * a[j]
    return g

def subgradient_max_conv(a):
    """Subgradient of max_k (a*a)_k w.r.t. a."""
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    g = subgradient_convolve(a, k_star)
    return g, k_star, conv

def mirror_descent(a, n_iter=5000, lr0=0.1, decay=0.999, time_budget=60.0):
    """Mirror descent (multiplicative weights) with subgradient."""
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr0
    
    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        
        g, k_star, conv = subgradient_max_conv(a)
        
        # Center the gradient to maintain sum constraint
        g_centered = g - np.mean(g)
        
        # Multiplicative update
        a_new = a * np.exp(-lr * g_centered)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / np.sum(a_new)
        
        obj = 2.0 * len(a) * float(np.max(compute_conv(a_new)))
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        lr *= decay
    
    return best_a

def stochastic_mirror_descent(a, n_iter=50000, lr0=0.05, decay=0.9999, time_budget=60.0, batch_size=10):
    """Stochastic mirror descent: sample random k values instead of finding max."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr0
    rng = np.random.default_rng(42)
    
    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Sample random convolution indices
        ks = rng.integers(0, 2*n-1, size=batch_size)
        # Also always include the current max
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        ks = np.append(ks, k_max)
        
        # Compute subgradient for the max of sampled values
        vals = np.array([float(conv[k]) for k in ks])
        k_star = ks[int(np.argmax(vals))]
        g = subgradient_convolve(a, k_star)
        
        g_centered = g - np.mean(g)
        
        a_new = a * np.exp(-lr * g_centered)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / np.sum(a_new)
        
        if it % 100 == 0:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        
        a = a_new
        lr *= decay
    
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    return best_a

from scipy.fft import rfft, irfft
import time

def compute_conv(a):

    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def obj_value(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a / s)))

# --- Optimization methods ---

def subgrad_step(a, k_star, lr):
    """One subgradient step (mirror descent style)."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = 2.0 * a[j]
    g -= np.mean(g)
    a_new = a * np.exp(-lr * g)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new

def mirror_descent_polish(a, time_budget=60.0, lr0=0.08, decay=0.9995, verbose=False):
    """Mirror descent with periodic restarts."""
    n = len(a)
    a = a.copy().astype(np.float64) / a.sum()
    best_a = a.copy()
    best_obj = obj_value(a)
    start = time.monotonic()
    
    lr = lr0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        a = subgrad_step(a, k_star, lr)
        
        if int((time.monotonic() - start) * 10) % 100 == 0:  # Check every ~100ms
            obj = obj_value(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        
        if stall > 200:
            # Restart: perturb and increase lr
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            a /= a.sum()
            lr = lr0

def lp_minpeak(a, use_scipy=True):
    """Solve LP: min M s.t. conv(a, g)_k <= M, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a = a / a.sum()
    
    # Build constraint matrix: A @ g = conv(a, g)
    # conv(a,g)_k = sum_i a_i * g_{k-i}
    # So A[k, j] = a_{k-j} if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for valid k
        ks = np.arange(j, j + 2*n-1)
        valid = (ks < 2*n-1)
        for k_idx in range(len(ks)):
            k = ks[k_idx]
            if k < 2*n-1:
                idx = k - j
                if 0 <= idx < n:
                    A[k_idx, j] = a[idx]
    
    # Simpler: A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k+1, n)):
            A[k, j] = a[k-j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = g_val.sum()
                if s > 1e-10:
                    return g_val / s, float(M.value)

            continue

    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    bounds = [(0, None)] * (n+1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = g_val.sum()
        if s > 1e-10:

    return None, None
