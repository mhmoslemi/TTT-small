import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def subgradient_descent(a, max_iters, deadline, lr_init=0.5, verbose=False):
    """
    Projected subgradient descent on the simplex.
    Uses the exact subgradient of J(a) = 2n*max(conv)/S^2.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = conv.max()
        S = a.sum()

        # Find active set (all k where conv[k] is close to max)
        threshold = M * (1.0 - 1e-8)
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute subgradient
        # dJ/da_i = (4n/S^2) * sum_{k in A} a[k-i]*I(0<=k-i<n) * w_k - (4n*M/S^3) * a_i
        # Weight active set uniformly
        w = 1.0 / len(active)
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += w * 2.0 * a[ki]

        grad = (4.0 * n / (S * S)) * grad - (4.0 * n * M / (S * S * S)) * a

        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.05 * it ** 0.5)

        # Projected gradient step on simplex
        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Reduce learning rate
                lr_init *= 0.5
                no_improve = 0

        a = a_new

        if verbose and it % 50 == 0:
            print(f"    SGD it {it}: obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def multi_cut_lp(a, max_iters, deadline, num_cuts_max=50, verbose=False):
    """
    Multi-cut SCA-LP: accumulate linearization cuts from multiple points
    and solve an increasingly constrained LP.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    # Store cuts: each cut is (c_vec, b_val) where c_vec @ g <= b_val
    cuts_c = []
    cuts_b = []

    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv_a = fast_conv(a)
        M = conv_a.max()
        S = a.sum()

        # Add a new cut at current point:
        # Linearize (g*g)[k] <= M at point a:
        # 2*(a*g)[k] - (a*a)[k] <= M
        # Sum over active k (weighted):
        threshold = M * (1.0 - 1e-7)
        active = np.where(conv_a >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv_a)])
        w = 1.0 / len(active)

        c_vec = np.zeros(n)
        b_val = 0.0
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                c_vec[j] += w * 2.0 * a[k - j]
            b_val += w * (M + conv_a[k])  # 2*(a*g)[k] <= M + (a*a)[k] = 2M (at active points)

        # Actually the correct linearization: (g*g)[k] ≈ 2*(a*g)[k] - (a*a)[k]
        # So constraint: 2*(a*g)[k] - (a*a)[k] <= M
        # i.e., 2*(a*g)[k] <= M + (a*a)[k]
        # For active k: (a*a)[k] = M, so 2*(a*g)[k] <= 2M
        # i.e., (a*g)[k] <= M

        # Let's use the simpler form: for each active k, (a*g)[k] <= M
        c_vec = np.zeros(n)
        b_val = 0.0
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                c_vec[j] += w * a[k - j]
            b_val += w * M

        cuts_c.append(c_vec.copy())
        cuts_b.append(b_val)

        # Limit number of cuts
        if len(cuts_c) > num_cuts_max:
            cuts_c = cuts_c[-num_cuts_max:]
            cuts_b = cuts_b[-num_cuts_max:]

        # Solve LP: minimize t
        # s.t. cuts_c[m] @ g <= cuts_b[m] for all m
        #      sum(g) = 1
        #      g >= 0
        #      (a*g)[k] <= M + t for all k  (the new linearization bound)
        #
        # Actually, let's use: minimize max_k (a*g)[k] subject to cuts and simplex
        # Variables: g[0..n-1], t
        # Minimize t
        # s.t. (a*g)[k] - t <= 0 for all k
        #      cuts_c[m] @ g <= cuts_b[m] for all m
        #      sum(g) = 1, g >= 0

        total_rows = (2 * n - 1) + len(cuts_c)
        rows = []
        cols = []
        vals = []
        b_ub = []

        # Constraints: (a*g)[k] - t <= 0
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(0.0)

        # Cut constraints: cuts_c[m] @ g <= cuts_b[m]
        for m in range(len(cuts_c)):
            offset = 2 * n - 1
            for j in range(n):
                if abs(cuts_c[m][j]) > 1e-15:
                    rows.append(offset + m)
                    cols.append(j)
                    vals.append(cuts_c[m][j])
            b_ub.append(cuts_b[m])

        A = coo_matrix((vals, (rows, cols)), shape=(total_rows, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_eq_rows = [0] * n
        A_eq_cols = list(range(n))
        A_eq = coo_matrix((np.ones(n), (A_eq_rows, A_eq_cols)), shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        bounds = [(0, None)] * n + [(None, None)]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    J_g = compute_J(g)
                    if J_g < best_obj - 1e-12:
                        best_obj = J_g
                        best_a = g.copy()
                        no_improve = 0
                    else:
                        # Try fractional steps
                        found = False
                        for frac in [0.5, 0.3, 0.2, 0.1]:
                            a_try = (1 - frac) * a + frac * g
                            a_try = np.maximum(a_try, 0)
                            a_try /= a_try.sum()
                            J_try = compute_J(a_try)
                            if J_try < best_obj - 1e-12:
                                a = a_try
                                best_obj = J_try
                                best_a = a_try.copy()
                                no_improve = 0
                                found = True
                                break
                        if not found:
                            no_improve += 1
                    a = best_a.copy()
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 15:
            break

        if verbose and it % 5 == 0:
            print(f"    MCLP it {it}: obj={best_obj:.8f} cuts={len(cuts_c)}", flush=True)

    return best_a, best_obj


def generate_inits_from_parent(parent, n, rng, num_inits=8):
    """Generate diverse initializations based on parent."""
    inits = []
    p = resample_seq(parent, n)
    ps = p.sum()
    if ps > 1e-10:
        p = p / ps
    else:
        p = np.ones(n) / n

    inits.append(p.copy())

    t = np.linspace(0, 1, n, endpoint=False)
    # Noise perturbations
    for noise_scale in [0.02, 0.05, 0.1, 0.2]:
        inits.append(project_simplex(p + rng.normal(0, noise_scale, n)))

    # Multiplicative perturbations
    for scale in [0.5, 0.8, 1.2, 1.5]:
        inits.append(project_simplex(p * scale))

    # Shifted versions
    for shift in [0.02, -0.02, 0.05, -0.05]:
        shifted = np.roll(p, int(shift * n))
        inits.append(project_simplex(shifted))

    # Mirror
    inits.append(project_simplex(p[::-1]))

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses golden-section CD, subgradient descent, multi-cut LP, and restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Intensive golden-section CD at parent length ===
    phase1_deadline = t_start + 0.40 * (budget_s - 25.0)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Golden-section CD at n={n_parent} (budget {phase1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Start from parent (normalized)
    a_curr = parent.copy().astype(np.float64)
    a_curr = np.maximum(a_curr, 0)
    s = a_curr.sum()
    if s > 1e-10:
        a_curr = a_curr / s

    # Run golden-section CD with subgradient ordering
    a_opt = golden_section_cd(a_curr, max_sweeps=5000, deadline=phase1_deadline,
                              subgradient_order=True, seed=42)
    update_best(a_opt)

    # Also try without subgradient ordering (random)
    remaining_p1 = phase1_deadline - time.monotonic()
    if remaining_p1 > 30:
        a_opt2 = golden_section_cd(best_seq.astype(np.float64), max_sweeps=3000,
                                   deadline=time.monotonic() + remaining_p1 * 0.6,
                                   subgradient_order=False, seed=123)
        update_best(a_opt2)

    remaining_p1 = phase1_deadline - time.monotonic()
    if remaining_p1 > 20:
        a_opt3 = golden_section_cd(best_seq.astype(np.float64), max_sweeps=2000,
                                   deadline=time.monotonic() + remaining_p1,
                                   subgradient_order=True, seed=777)
        update_best(a_opt3)

    # === PHASE 2: Multi-cut LP polishing ===
    phase2_deadline = t_start + 0.65 * (budget_s - 25.0)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-cut LP polishing (budget {phase2_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if time.monotonic() < phase2_deadline - 20:
        a_lp, obj_lp = multi_cut_lp(best_seq.astype(np.float64), max_iters=100,
                                    deadline=phase2_deadline, num_cuts_max=30)
        update_best(a_lp)

        # CD after LP
        if time.monotonic() < phase2_deadline - 10:
            a_cd = golden_section_cd(best_seq.astype(np.float64), max_sweeps=2000,
                                     deadline=phase2_deadline - 5,
                                     subgradient_order=True, seed=456)
            update_best(a_cd)

    # === PHASE 3: Subgradient + perturbation restarts ===
    phase3_deadline = t_start + 0.85 * (budget_s - 25.0)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Subgradient + restarts (budget {phase3_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Subgradient descent
    if time.monotonic() < phase3_deadline - 30:
        sg_deadline = time.monotonic() + 0.3 * (phase3_deadline - time.monotonic())
        a_sg, obj_sg = subgradient_descent(best_seq.astype(np.float64),
                                           max_iters=3000, deadline=sg_deadline,
                                           lr_init=0.3)
        update_best(a_sg)
        # CD after subgradient
        if time.monotonic() < phase3_deadline - 20:
            a_cd2 = golden_section_cd(best_seq.astype(np.float64), max_sweeps=2000,
                                      deadline=phase3_deadline - 10,
                                      subgradient_order=True, seed=321)
            update_best(a_cd2)

    # Perturbation restarts
    remaining_p3 = phase3_deadline - time.monotonic()
    if remaining_p3 > 60:
        print(f"  Perturbation restarts (budget {remaining_p3:.0f}s)...", flush=True)
        inits = generate_inits_from_parent(best_seq, n_parent, rng, num_inits=10)
        per_init_time = remaining_p3 / len(inits)

        for idx in range(len(inits)):
            if time.monotonic() > phase3_deadline - 10:
                break
            alloc = min(per_init_time, phase3_deadline - time.monotonic() - 5)
            if alloc < 15:
                break

            a0 = inits[idx]
            a0 = np.maximum(a0, 0)
            if a0.sum() < 1e-10:
                continue
            a0 = a0 / a0.sum()

            a_opt_r = golden_section_cd(a0, max_sweeps=1000, deadline=time.monotonic() + alloc,
                                        subgradient_order=True, seed=1000 + idx)
            update_best(a_opt_r)
            print(f"    Restart {idx}: J={compute_J(a_opt_r):.8f}", flush=True)

    # === PHASE 4: Multi-length exploration ===
    phase4_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Multi-length + final polish (budget {phase4_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    remaining = phase4_deadline - time.monotonic()
    if remaining > 60:
        # Try nearby lengths
        try_lengths = [n_parent - 2, n_parent - 1, n_parent + 1, n_parent + 2,
                       n_parent - 4, n_parent + 4]
        try_lengths = [n for n in try_lengths if 100 <= n <= 3000]
        per_len_time = remaining / (len(try_lengths) + 2)

        for new_n in try_lengths[:4]:
            if time.monotonic() > phase4_deadline - 20:
                break
            alloc = min(per_len_time, phase4_deadline - time.monotonic() - 10)
            if alloc < 15:
                break

            a_ext = resample_seq(best_seq, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()

            a_opt_n = golden_section_cd(a_ext, max_sweeps=1000, deadline=time.monotonic() + alloc,
                                        subgradient_order=True, seed=2000 + new_n)
            update_best(a_opt_n)
            print(f"    n={new_n}: J={compute_J(a_opt_n):.8f}", flush=True)

    # Final CD polish on best
    if time.monotonic() < deadline - 10:
        print(f"  Final CD polish...", flush=True)
        a_final = golden_section_cd(best_seq.astype(np.float64), max_sweeps=5000,
                                    deadline=deadline - 5,
                                    subgradient_order=True, seed=9999)
        update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
