from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)

"""
Peak-targeted optimization with LP refinement for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Combines gradient descent on active peaks, LP-based direction finding,
and coordinate descent for local refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def cross_conv(a, d):
    """Compute cross-convolution a*d using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fd = rfft(d, size)
    conv = irfft(fa * fd, size)
    return conv[:2 * n - 1]


def peak_gradient_single(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a. Only non-zero at participating indices."""
    n = len(a)
    grad = np.zeros(n)
    i_min = max(0, k - n + 1)
    i_max = min(k, n - 1)
    if i_min > i_max:
        return grad
    indices = np.arange(i_min, i_max + 1)
    partners = k - indices
    # For each pair (i, k-i), the gradient at position i gets a[k-i]
    # and at position k-i gets a[i]
    grad[indices] += a[partners]
    grad[partners] += a[indices]
    # Fix diagonal: where i == k-i, we added a[i] twice, which equals 2*a[i], correct
    return grad


def active_peak_gradient(a, top_k=5, weight_alpha=1.0):
    """
    Compute weighted gradient from top peaks.
    weight_alpha controls concentration: higher = more focused on max peak.
    """
    n = len(a)
    conv = compute_conv(a)
    # Get top peaks
    peak_indices = np.argsort(conv)[::-1][:top_k]
    peak_values = conv[peak_indices]
    
    # Weights
    if weight_alpha < 1e10:
        weights = peak_values ** weight_alpha
    else:
        weights = np.zeros(len(peak_indices))
        weights[0] = 1.0
    weights = weights / np.sum(weights)
    
    # Combined gradient
    grad = np.zeros(n)
    for w, k in zip(weights, peak_indices):
        grad += w * peak_gradient_single(a, k)
    
    return grad, conv, peak_indices


def lp_direction(a, active_lags, time_limit=10):
    """
    Find direction d that reduces the max over active lags.
    LP: min t s.t. cross_conv(a,d)[k] <= t for k in active_lags,
        sum(d) = 0, -1 <= d[i] <= 1
    Returns d (normalized) and optimal t, or None if infeasible/no improvement.
    """
    n = len(a)
    if n > 1000:
        return None, 0.0
    
    # Build constraint matrix for cross_conv(a,d)[k] <= t
    # cross_conv(a,d)[k] = sum_i a[i]*d[k-i] (valid indices)
    # This is linear in d: sum_i (coefficient) * d[i]
    # For lag k: d[i] appears with coefficient a[k-i] (when k-i is valid)
    #             and a[i] (when i is valid, as d[k-i] = d[j] where j = k-i)
    # Actually: cross_conv(a,d)[k] = sum_i a[i]*d[k-i]
    #           = sum_j d[j]*a[k-j]  (substituting j = k-i)
    # So the coefficient of d[j] is a[k-j] (when k-j is in [0,n-1])
    
    m = len(active_lags)
    # A_ub has m rows (one per active lag) + we'll add the t variable
    # Variables: d[0],...,d[n-1], t (total n+1 variables)
    # Constraint: cross_conv(a,d)[k] - t <= 0
    # i.e., sum_j a[k-j]*d[j] - t <= 0
    
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for row, k in enumerate(active_lags):
        for j in range(n):
            partner = k - j
            if 0 <= partner < n:
                A_ub[row, j] = a[partner]
        A_ub[row, n] = -1.0  # coefficient of t
    
    # Objective: minimize t => c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Equality: sum(d) = 0 => A_eq = [1, 1, ..., 1, 0], b_eq = [0]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -1 <= d[i] <= 1, t unbounded below but bounded above by 0 (for feasibility)
    # Actually t can be anything, but we're minimizing it
    bounds = [(-1.0, 1.0)] * n + [(-np.inf, 0.0)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            d = res.x[:n]
            t_val = res.x[n]
            if t_val < -1e-10:  # Only useful if t is negative
                # Normalize d
                d_norm = np.linalg.norm(d)
                if d_norm > 1e-10:
                    d = d / d_norm
                return d, t_val
    except Exception:
        pass
    return None, 0.0


def peak_grad_descent(a, time_budget=100.0, top_k=5, weight_alpha=1.0,
                      lr_start=0.01, lr_end=0.001, momentum=0.9):
    """
    Peak-targeted gradient descent with momentum.
    Works in normalized space (sum(a)=1, a>=0).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)  # momentum
    
    start = time.monotonic()
    it = 0
    max_iters = 100000
    
    while it < max_iters and time.monotonic() - start < time_budget:
        # Adaptive learning rate (geometric decay)
        progress = min(it / max_iters, 1.0)
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Compute gradient
        grad, conv, peak_indices = active_peak_gradient(a, top_k, weight_alpha)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        a_new = a - lr * v
        
        # Project: non-negativity + normalization
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a  # reject
        else:
            a_new = a_new / s_new
        
        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        
        it += 1
        
        if it % 1000 == 0:
            print(f"    PGD iter {it}: obj={best_obj:.8f}, lr={lr:.6f}")
    
    return best_a


def lp_refine(a, time_budget=30.0):

    n = len(a)

        return a

    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)

        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))

            for alpha in [0.01, 0.03, 0.05, 0.1, 0.2]:
                a_try = a - alpha * d
                a_try = np.maximum(a_try, 0.0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    best_alpha = -alpha
        
        a = best_a.copy()
    
    return best_a


def multiplicative_update(a, time_budget=60.0, top_k=5, weight_alpha=1.0,
                          lr_start=0.05, lr_end=0.005):
    """
    Multiplicative (mirror) descent: a[i] *= exp(-lr * g[i])
    Automatically maintains non-negativity.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    it = 0
    max_iters = 50000
    
    while it < max_iters and time.monotonic() - start < time_budget:
        progress = min(it / max_iters, 1.0)
        lr = lr_start * (lr_end / lr_start) ** progress
        
        grad, conv, _ = active_peak_gradient(a, top_k, weight_alpha)
        
        # Multiplicative update in log-space
        # a_new[i] = a[i] * exp(-lr * grad[i])
        # For numerical stability, center the gradient
        grad_centered = grad - np.mean(grad)
        log_update = -lr * grad_centered
        log_update = np.clip(log_update, -5, 5)  # prevent overflow
        a_new = a * np.exp(log_update)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        
        it += 1
        if it % 5000 == 0:
            print(f"    MR iter {it}: obj={best_obj:.8f}")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def coordinate_descent_peaks(a, time_budget=20.0, seed=0):
    """
    Coordinate descent: for each peak, try adjusting pairs of coordinates
    that contribute to it.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy().astype(np.float64)
    s = np.sum(best_a)
    if s < 1e-10:
        return best_a
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(seed)
    
    conv = compute_conv(best_a)
    top_peaks = np.argsort(conv)[::-1][:5]
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        for p in top_peaks:
            # Find indices contributing to conv[p]
            i_min = max(0, p - n + 1)
            i_max = min(p, n - 1)
            # Try a few random pairs from the contributing indices
            indices = np.arange(i_min, i_max + 1)
            if len(indices) < 2:
                continue
            for trial in range(10):
                if time.monotonic() - start > time_budget:
                    break
                # Pick two random indices from contributors
                i1, i2 = rng.choice(indices, size=2, replace=False)
                j1, j2 = p - i1, p - i2
                j1 = int(j1)
                j2 = int(j2)
                if j1 < 0 or j1 >= n or j2 < 0 or j2 >= n:
                    continue
                # Try transferring mass
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    delta = frac * best_a[i1]
                    a_try = best_a.copy()
                    a_try[i1] -= delta
                    a_try[j2] += delta  # or some other transfer
                    if a_try[i1] < 0:
                        continue
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
        
        if not improved and it > 30:
            # Recompute peaks
            conv = compute_conv(best_a)
            top_peaks = np.argsort(conv)[::-1][:5]
            if it > 100:
                break
    
    return best_a


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf
