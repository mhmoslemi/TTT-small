```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Successive Convex Approximation (SCA) with LP subproblems (primary optimizer)
2. FFT-accelerated FISTA on log-sum-exp smooth approximation (fast pre-optimizer)
3. Coordinate descent with golden-section 1D optimization (polisher)
4. Multi-length, multi-start search with diverse spectral initializations
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * rho_arr) > (cv - 1.0)
    if not cond.any():
        theta = max(0.0, (cv[-1] - 1.0) / n)
    else:
        rho = int(np.argmax(cond))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
        if theta < 0:
            theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad_fft(a, tau):
    """Compute smooth max value and gradient of conv(a,a) using FFT.
    
    Returns (smooth_value, gradient) where gradient has same length as a.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # Use FFT cross-correlation
    N = 1
    while N < 2 * n:
        N <<= 1
    N = max(N, 2 * n - 1 + n - 1)  # Need N >= 3n-2 for full correlation
    N = 1
    while N < (2*n - 1) + n - 1:
        N <<= 1
    
    p_ext = np.zeros(N)
    p_ext[:len(p)] = p
    a_ext = np.zeros(N)
    a_ext[:n] = a
    
    # Cross-correlation: ifft(fft(p) * conj(fft(a))) gives r[i] = sum_j p[j+i]*a[j]
    P = np.fft.fft(p_ext)
    A = np.fft.fft(a_ext)
    r = np.fft.ifft(P * np.conj(A)).real[:n]
    
    grad = 2.0 * r
    return smooth_val, grad

    """FISTA with FFT-accelerated smooth max gradient."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_grad_fft(y, tau)
```

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Successive Convex Approximation (SCA) with LP subproblems (primary method)
2. FFT-accelerated FISTA on log-sum-exp smooth approximation
3. LP best-response direction finding (complement to SCA)
4. Coordinate descent with golden-section 1D optimization
5. Multi-length, multi-start search with diverse initializations
Key new idea: SCA linearizes the quadratic autocorrelation constraints at the
current point, yielding a linear program that can be solved exactly with HiGHS.
This provides a globally optimal step within the local linear approximation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * rho_arr) > (cv - 1.0)
    if not cond.any():
        theta = max(0.0, (cv[-1] - 1.0) / n)
    else:
        rho = int(np.argmax(cond))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
        if theta < 0:
            theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad_fft(a, tau):
    """Compute smooth max value and gradient of conv(a,a) using FFT cross-correlation.
    
    Returns (smooth_value, gradient) where gradient has same length as a.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b -
