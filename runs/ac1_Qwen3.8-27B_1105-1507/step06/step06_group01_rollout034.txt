    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where (A@g)[k] = (a*g)[k] for all k."""
    n = len(a)
    # Each column j has nonzeros at rows j..j+n-1 with values a[0]..a[n-1]
    col_idx = np.repeat(np.arange(n), n)
    row_idx = np.concatenate([np.arange(j, j + n) for j in range(n)])
    vals = np.tile(a, n)
    return csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))

                    method='highs', options={'time_limit': time_limit})

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using full-constraint LP best response
with exact 1D golden-section line search, adaptive coordinate descent, subgradient
fallback, and multi-start strategy across sequence lengths.
Key improvements over previous: (1) ALL (2n-1) LP constraints instead of top-k active set,
(2) precomputed sparse matrix structure for speed, (3) adaptive line search,
(4) subgradient Frank-Wolfe fallback when LP stalls, (5) more aggressive multi-start.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_structure(n):
    """Precompute (row_idx, col_idx) for the (2n-1) x n convolution matrix."""
    col_idx = np.repeat(np.arange(n), n)
    segments = [np.arange(j, j + n) for j in range(n)]
    row_idx = np.concatenate(segments)
    return row_idx, col_idx


def make_conv_matrix(a, row_idx, col_idx):
    """Create sparse convolution matrix with new values but precomputed structure."""
    n = len(a)
    vals = np.tile(a, n)
    return csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))


def lp_full(a, row_idx, col_idx, M, time_limit=2.0):
    """Full-constraint LP best response. Returns normalized g or None."""
    n = len(a)
    A = make_conv_matrix(a, row_idx, col_idx)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                    method='highs', options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def _mix_peak(a, g, t):
    """Compute max of conv of mixed sequence (1-t)*a + t*g."""
    mix = (1.0 - t) * a + t * g
    return fast_conv(mix).max()


def line_search(a, g, n_evals=55):
    """Golden section search for best t in [0,1]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    a1 = lo + (1.0 - gr) * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = _mix_peak(a, g, a1)
    f2 = _mix_peak(a, g, a2)
    for _ in range(n_evals):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = lo + (1.0 - gr) * (hi - lo)
            f1 = _mix_peak(a, g, a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = _mix_peak(a, g, a2)
    return (lo + hi) / 2.0


def _golden_section_1d(obj, lo, hi, n_iter=35):
    """General golden section minimization on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = lo + (1.0 - gr) * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = obj(a1)
    f2 = obj(a2)
    for _ in range(n_iter):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = lo + (1.0 - gr) * (hi - lo)
            f1 = obj(a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = obj(a2)
    best_x = a1 if f1 <= f2 else a2
    # Check boundary
    f0 = obj(0.0)
    if f0 < obj(best_x):
        return 0.0
    return best_x


def cd_optimize_coord(a, i, b):
    """Optimize coordinate i given convolution b. Returns optimal new value."""
    n = len(a)
    x = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    s_vals = []
    has_quad = False
    quad_idx = -1

    for k in range(k_lo, k_hi + 1):
        other = k - i
        if other == i:
            c_vals.append(b[k] - x * x)
            s_vals.append(0.0)
            has_quad = True
            quad_idx = len(c_vals) - 1
        elif 0 <= other < n:
            c_vals.append(b[k] - 2.0 * x * a[other])
            s_vals.append(2.0 * a[other])
        else:
            c_vals.append(b[k])
            s_vals.append(0.0)

    affected = np.zeros(2 * n - 1, dtype=bool)
    affected[k_lo:k_hi + 1] = True
    base_max = b[~affected].max() if not affected.all() else 0.0

    c_arr = np.array(c_vals, dtype=np.float64)
    s_arr = np.array(s_vals, dtype=np.float64)

    def obj(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        vm = vals.max()
        if vm > m:
            m = vm
        if has_quad:
            qv = c_arr[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m

    hi = max(2.0 * x, 0.3)
    for _ in range(5):
        if obj(hi) > obj(hi * 0.5) and hi < 50.0:
            hi *= 2.0
        else:
            break

    return max(0.0, _golden_section_1d(obj, 0.0, hi, 30))


def run_cd_sweeps(a, n_sweeps, deadline, seed=0):
    """Run coordinate descent with in-place convolution updates."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        rng = np.random.default_rng(seed + sweep)
        order = rng.permutation(n)
        b = fast_conv(a)
        improved = False

        for i in order:
            x_new = cd_optimize_coord(a, i, b)
            if abs(x_new - a[i]) > 1e-13:
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    other = k - i
                    if other == i:
                        b[k] = b[k] - a[i] * a[i] + x_new * x_new
                    elif 0 <= other < n:
                        b[k] += 2.0 * a[other] * (x_new - a[i])
                a[i] = x_new
                improved = True

        s = a.sum()
        if s > 1e-10:
            a = a / s
            b = fast_conv(a)
            M = b.max()
            if M < best_M * (1.0 - 1e-11):
                best_M = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        if not improved:
            no_improve += 5
        if no_improve > 15:
            break

    return best_a


def subgradient_step(a, M, b, step_size):
    """Frank-Wolfe style subgradient step using argmax of autocorrelation."""
    n = len(a)
    k_star = int(np.argmax(b))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    # Project to tangent of simplex
    grad_tan = grad - grad.mean()
    a_new = a - step_size * grad_tan
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def run_lp_optimization(a_init, row_idx, col_idx, deadline, max_iters=5000,
                        cd_every=15, seed=0):
    """Main LP-based optimization with periodic CD and subgradient fallback."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        if M < best_M:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic coordinate descent
        if it > 0 and it % cd_every == 0:
            cd_deadline = min(deadline, time.monotonic() + 20.0)
            a = run_cd_sweeps(a, 3, cd_deadline, seed=seed + it)
            b = fast_conv(a)
            M = b.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
                no_improve = 0
            continue

        # Subgradient fallback when LP stalls
        if no_improve > 10 and no_improve % 5 == 0:
            step = max(0.005, 0.1 / (1.0 + 0.01 * it))
            a_sg = subgradient_step(a, M, b, step)
            M_sg = fast_conv(a_sg).max()
            if M_sg < best_M:
                a = a_sg
                best_M = M_sg
                best_a = a.copy()
                no_improve = 0
                continue
            # Try multiplicative weights step
            grad = np.zeros(n)
            k_star = int(np.argmax(b))
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] = 2.0 * a[j]
            grad_tan = grad - grad.mean()
            mw_step = max(0.01, 0.3 / (1.0 + 0.05 * it))
            log_w = -mw_step * grad_tan
            log_w -= log_w.max()
            a_mw = a * np.exp(log_w)

            M_mw = fast_conv(a_mw).max()
            if M_mw < best_M:
                a = a_mw
                best_M = M_mw
                best_a = a.copy()
                no_improve = 0
                continue

        # Perturbation to escape local minima

            sigma = min(sigma, 0.15)

            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / a_pert.sum()
            M_pert = fast_conv(a_pert).max()
            if M_pert < best_M:
                a = a_pert
                no_improve = 0
                continue
            no_improve = 25  # Reset to avoid repeated large perturbations

        # Solve full LP
        g = lp_full(a, row_idx, col_idx, M, time_limit=2.5)
        if g is not None:
            if np.max(np.abs(a - g)) > 1e-9:
                t_star = line_search(a, g, n_evals=50)
                if t_star > 1e-4:
                    a_new = (1.0 - t_star) * a + t_star * g
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                        M_new = fast_conv(a_new).max()
                        if M_new < M - 1e-14:
                            a = a_new
                            if M_new < best_M:
                                best_M = M_new
                                best_a = a_new.copy()
                                no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            # LP failed, use subgradient
            step = max(0.01, 0.2 / (1.0 + 0.02 * it))
            a = subgradient_step(a, M, b, step)
            no_improve += 1

        if it % 100 == 0 and it > 0:
            elapsed = time.monotonic() - t_start
            print(f"    iter {it}: best_M={best_M:.12f} t={elapsed:.0f}s", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
