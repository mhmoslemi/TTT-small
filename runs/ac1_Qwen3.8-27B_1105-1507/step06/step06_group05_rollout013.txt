```python
"""
Multi-scale multiplicative gradient descent with LP polishing for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FFT-based cross-correlation for fast gradients, geometric annealing,
multi-start exploration, LP direction finding, and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def fast_gradient(a, T):
    """
    Compute the smoothed gradient of 2n*max(conv) using log-sum-exp.
    Returns grad array of length n where grad[i] = 2*sum_k p[k]*a[k-i].
    Uses FFT cross-correlation for O(n log n) computation.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    # Soft peak distribution
    scaled = (conv - cmax) / T
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / np.sum(exp_scaled)

    # Compute grad[i] = 2 * sum_k p[k] * a[k-i] via FFT cross-correlation
    # (p * a_rev)[n-1-i] = sum_k p[k]*a[k-i]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    a_rev = a[::-1]
    Pf = rfft(p, size)
    Af = rfft(a_rev, size)
    conv_pa = irfft(Pf * Af, size)
    # grad[i] = 2 * conv_pa[n-1-i]
    grad = 2.0 * conv_pa[n - 1:2 * n - 1][::-1]
    return grad, conv, cmax


def multiplicative_descent(a, T_start=1.0, T_end=0.005, n_steps=800, eta=0.15,
                           time_budget=60.0, verbose=False):
    """
    Multiplicative (mirror) gradient descent on the simplex with geometric annealing.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_start
    if n_steps > 1:
        T_ratio = (T_end / T_start) ** (1.0 / n_steps)
    else:
        T_ratio = 1.0

    no_improve = 0
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        grad, conv, cmax = fast_gradient(a, T)

        # Multiplicative update
        log_scale = -eta * grad
        log_scale = np.clip(log_scale, -50.0, 50.0)
        scaling = np.exp(log_scale)
        a_new = a * scaling
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                eta *= 0.8
                no_improve = 0
            if eta < 1e-10:
                break

        T *= T_ratio

    return best_a


def lp_polish(a, max_iters=30, time_budget=60.0):
    """LP-based polishing with golden section line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        time_per_lp = max(3.0, min(15.0, remaining * 0.5 / max(1, max_iters - it)))
        g = solve_lp(a, M, time_limit=int(time_per_lp))
        if g is None:
            break

        # Golden section for step size t in [0, 1]
        def obj_at_t(t):
            a_try = normalize((1.0 - t) * a + t * g)
            return compute_obj(a_try)

        lo, hi = 0.0, 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc, fd = obj_at_t(c), obj_at_t(d)
        for _ in range(12):
            if hi - lo < 0.01:
                break
            if fc < fd:
                hi, d, fd = d, c, fc
                c = hi - gr * (hi - lo)
                fc = obj_at_t(c)
            else:
                lo, c, fc = c, d, fc
                d = lo + gr * (hi - lo)
                fd = obj_at_t(d)

        t_opt = (lo + hi) / 2.0
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Perturb to escape
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.05, n)))

    return best_a


def coordinate_descent(a, time_budget=20.0):
    """Pairwise mass-transfer coordinate descent targeting the peak."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        contribs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(key=lambda x: -x[0])

        improved = False
        for _, j1, j2 in contribs[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            for _ in range(2):
                j_tgt = int(rng.integers(0, n))
                if j_tgt in (j1, j2):
                    continue
                a_try = a.copy()
                delta = 0.03 * a_try[j1]
                a_try[j1] -= delta
                a_try[j_tgt] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            if rng.random() < 0.15:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:
                break

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def multi_start_gradient(a_base, n_starts=6, time_budget=120.0, verbose=False):
    """Run multiplicative descent from multiple diverse starting points."""
    n = len(a_base)
    best_a = a_base.copy()
    best_obj = compute_obj(a_base)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    per_start = time_budget / n_starts

    for s in range(n_starts):
        if time.monotonic() - start > time_budget:
            break

        # Generate diverse starting point
        if s == 0:
            a_try = a_base.copy()
        elif s == 1:
            # Gaussian perturbation
            a_try = a_base * np.exp(rng.normal(0, 0.1, n))
        elif s == 2:
            # Sinusoidal modulation
            freq = rng.uniform(1.0, 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            a_try = a_base * np.maximum(mod, 0.1)
        elif s == 3:
            # Blend with uniform
            a_try = 0.7 * normalize(a_base) + 0.3 * np.ones(n) / n
        elif s == 4:
            # Block reversal
            a_try = a_base.copy()
            blen = int(rng.integers(n // 3, n // 2))
            bstart = int(rng.integers(0, n - blen))
            a_try[bstart:bstart + blen] = a_try[bstart:bstart + blen][::-1]
        else:
            # Cosine window blend
            cos_w = 0.5 * (1 + np.cos(np.pi * np.arange(n) / n))
            a_try = a_base * (0.5 + 0.5 * cos_w)

        a_try = normalize(a_try)
        remaining = time_budget - (time.monotonic() - start)
        alloc = min(per_start, remaining * 0.9)
        if alloc < 5:
            break

        # Run multiplicative descent
        a_res = multiplicative_descent(a_try, T_start=0.5, T_end=0.01,
                                       n_steps=400, eta=0.12, time_budget=alloc)
        obj_res = compute_obj(a_res)
        if obj_res < best_obj:
            best_obj = obj_res
            best_a = a_res.copy()
            if verbose:
                print(f"    Start {s}: {best_obj:.8f} **")
        elif verbose:
            print(f"    Start {s}: {obj_res:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-scale gradient descent + LP polish + length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            best_score = compute_obj(parent)
            best_seq = parent.copy()
            print(f"Parent loaded: n={parent.size}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(700)
        best_score = compute_obj(best_seq)
        print(f"No parent, starting uniform: n=700, score={best_score:.8f}")

    n_cur = len(best_seq)

    # ============================================================
    # Phase 1: Multi-start gradient descent (45% time)
    # ============================================================
    phase1_budget = (deadline - start_time) * 0.45
    print(f"\nPhase 1: Multi-start gradient descent (budget={phase1_budget:.0f}s)")

    a_work = normalize(best_seq.copy())
    a_new = multi_start_gradient(a_work, n_starts=6, time_budget=phase1_budget, verbose=True)
    obj_new = compute_obj(a_new)
    if obj_new < best_score:
        best_score = obj_new
        best_seq = a_new.copy()
        print(f"  Phase 1 result: {best_score:.8f}")
    else:
        print(f"  Phase 1: no improvement ({obj_new:.8f} vs {best_score:.8f})")

    # ============================================================
    # Phase 2: Annealed gradient with restarts (25% time)
    # ============================================================
    phase2_budget = (deadline - start_time) * 0.25
    print(f"\nPhase 2: Annealed gradient with restarts (budget={phase2_budget:.0f}s)")

    a_work = normalize(best_seq.copy())
    n_cool = 5
    per_cool = phase2_budget / n_cool
    rng2 = np.random.default_rng(seed + 1)

    for cool in range(n_cool):
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        alloc = min(per_cool, remaining * 0.85)
        if alloc < 10:
            break

        # Different temperature schedules
        T_s = 1.0 / (cool + 1)
        T_e = T_s * 0.01
        eta_s = 0.2 / (cool + 1)

        a_res = multiplicative_descent(a_work, T_start=T_s, T_end=T_e,
                                       n_steps=600, eta=eta_s, time_budget=alloc * 0.8)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            a_work = normalize(a_res.copy())
            print(f"  Cool {cool}: {best_score:.8f} **")
        else:
            # Perturb for next restart
            a_work = normalize(a_res * np.exp(rng2.normal(0, 0.06 / (cool + 1), len(a_res))))
            print(f"  Cool {cool}: {obj_res:.8f} (best={best_score:.8f})")

    # ============================================================
    # Phase 3: LP polishing (15% time)
    # ============================================================
    phase3_budget = (deadline - start_time) * 0.15
    print(f"\nPhase 3: LP polishing (budget={phase3_budget:.0f}s)")

    if n_cur <= 900:
        a_work = normalize(best_seq.copy())
        a_res = lp_polish(a_work, max_iters=20, time_budget=phase3_budget * 0.7)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  LP: {best_score:.8f} **")
        else:
            print(f"  LP: {obj_res:.8f} (no improvement)")

        # Second LP pass from perturbed start
        if time.monotonic() < deadline - 50:
            rng3 = np.random.default_rng(seed + 2)
            a_pert = normalize(best_seq * np.exp(rng3.normal(0, 0.04, n_cur)))
            remaining = min(phase3_budget * 0.25, deadline - time.monotonic() - 40)
            if remaining > 10:
                a_res2 = lp_polish(a_pert, max_iters=10, time_budget=remaining)
                obj_res2 = compute_obj(a_res2)
                if obj_res2 < best_score:
                    best_score = obj_res2
                    best_seq = a_res2.copy()
                    print(f"  LP perturbed: {best_score:.8f} **")

    # ============================================================
    # Phase 4: Length exploration (10% time)
    # ============================================================
    phase4_budget = (deadline - start_time) * 0.10
    print(f"\nPhase 4: Length exploration (budget={phase4_budget:.0f}s)")

    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100), max(200, n_best - 50),
        n_best,
        min(1200, n_best + 50), min(1200, n_best + 100), min(1200, n_best + 200)
    ]))

    for n_try in n_options:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(phase4_budget / len(n_options), remaining * 0.8)

        if n_try == n_best:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample_seq(best_seq, n_try))

        # Quick gradient optimization at this length
        a_try = multiplicative_descent(a_try, T_start=0.3, T_end=0.01,
                                       n_steps=200, eta=0.1, time_budget=alloc)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # ============================================================
    # Phase 5: Final coordinate descent polish (5% time)
    # ============================================================
    phase5_budget = (deadline - start_time) * 0.05
    print(f"\nPhase 5: Final coordinate descent (budget={phase5_budget:.0f}s)")

    a_final = normalize(best_seq.copy())
    a_res = coordinate_descent(a_final, time_budget=phase5_budget)
    obj_res = compute_obj(a_res)
    if obj_res < best_score:
        best_score = obj_res
        best_seq = a_res.copy()
        print(f"  Coord: {best_score:.8f} **")
    else:
        print(f"  Coord: {obj_res:.8f}")

    # One more gradient pass if time allows
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 10
        if remaining > 15:
            a_final = multiplicative_descent(normalize(best_seq), T_start=0.1,
                                             T_end=0.003, n_steps=300,
                                             eta=0.08, time_budget=remaining)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final gradient: {best_score:.8f} **")

    # ============================================================
    # Final validation
    # ============================================================
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
```
