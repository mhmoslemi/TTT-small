        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]

"""
Improved multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: parabolic line search, adaptive M in LP, peak-targeted variants,
better time allocation, and diverse restarts with simulated annealing acceptance.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, obj_vec=None, time_limit=10):
    """Solve LP: max obj_vec^T g s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    if obj_vec is None:
        obj_vec = np.ones(n)
    c = -obj_vec.astype(np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def spectral_equalization(a, max_iters=40, time_budget=30.0):
    """POCS-style spectral equalization with gradual compression."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Gradual compression: start gentle, increase over time
        alpha = 0.85 + 0.1 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.7 * a + 0.3 * a_new)

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def soft_max_gradient_step(a, tau_init=0.001, step_size=0.04, max_steps=100, time_budget=30.0):
    """Soft-max (log-sum-exp) weighted gradient descent."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 3 * n - 1:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        tau = tau_init + 0.3 * cmax / (n * (step + 1))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_ext = np.zeros(2 * n - 1)
        w_ext[:] = weights
        a_flip = a[::-1]
        Wf = rfft(w_ext, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        grad_centered = grad - np.mean(grad)

        a_new = normalize(a - step_size * grad_centered)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.65
            if step_size < 1e-8:
                break
        tau_init *= 0.98

    return best_a, best_obj

            else:
                a = a_new
                print(f"  Round {round_idx} (m={m_start}): {obj_new:.8f}")

    # ============================================================
    # Phase 2: Peak-targeted LP (8% of time)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Peak-targeted LP")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 80:
        a = normalize(best_seq.copy())
        alloc = min(45.0, (deadline - time.monotonic()) * 0.15)
        a_new, obj_new = lp_optimize_peak_targeted(a, max_time=alloc)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Peak-targeted: {best_score:.8f} **")
        else:
            print(f"  Peak-targeted: {obj_new:.8f}")

        # Follow up with core LP
        if time.monotonic() < deadline - 60:
            a_follow, obj_follow = lp_optimize_core(a_new, max_time=20.0, m_start=1.002, m_end=1.001)
            if obj_follow < best_score:
                best_score = obj_follow
                best_seq = a_follow.copy()
                print(f"  Follow-up LP: {best_score:.8f} **")

    # ============================================================
    # Phase 3: Spectral + gradient (7% of time)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Spectral + gradient")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 60:
        a = normalize(best_seq.copy())
        a_spec, obj_spec = spectral_equalization(a, max_iters=30, time_budget=20.0)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: {best_score:.8f} **")
        else:
            print(f"  Spectral: {obj_spec:.8f}")

        a_soft, obj_soft = soft_max_gradient_step(a, tau_init=0.001, step_size=0.04,
                                                  max_steps=100, time_budget=20.0)
        if obj_soft < best_score:
            best_score = obj_soft
            best_seq = a_soft.copy()
            print(f"  SoftMax: {best_score:.8f} **")
        else:
            print(f"  SoftMax: {obj_soft:.8f}")

    # ============================================================
    # Phase 4: Multi-n exploration (15% of time)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700
    n_options = sorted(set([
        max(200, n_best - 200),
        max(200, n_best - 100),
        max(200, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 300,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1200]

    phase4_budget = min(120.0, (deadline - time.monotonic()) * 0.25)
    per_n_budget = phase4_budget / max(len(n_options), 1)

    for n_try in n_options:
        if time.monotonic() > deadline - 40:
            break
        if per_n_budget < 10:
            break

        if best_seq is not None:
            a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try

        # Quick LP optimization at this n
        a_try, obj_try = lp_optimize_core(a_try, max_time=per_n_budget * 0.6,
                                          m_start=1.02, m_end=1.002, perturb_every=20)

        # Quick gradient step
        if per_n_budget > 10:
            a_try, obj_try2 = soft_max_gradient_step(a_try, tau_init=0.001, step_size=0.03,
                                                     max_steps=40, time_budget=per_n_budget * 0.3)
            if obj_try2 < obj_try:
                obj_try = obj_try2

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # ============================================================
    # Phase 5: Diverse restarts (10% of time)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Diverse restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed)
        phase5_budget = min(80.0, (deadline - time.monotonic()) * 0.2)
        n_restarts = min(8, max(3, int(phase5_budget / 10)))
        per_restart = phase5_budget / n_restarts

        for restart in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            if per_restart < 8:
                break

            # Diverse perturbation
            strategy = restart % 4
            if strategy == 0:
                a_try = best_seq * np.exp(rng.normal(0, 0.06, n_cur))
            elif strategy == 1:
                freq = rng.uniform(0.5, 4.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
                a_try = best_seq * mod
            elif strategy == 2:
                a_rand = rng.dirichlet(np.ones(n_cur) * 0.3)
                mix = rng.uniform(0.15, 0.4)
                a_try = (1 - mix) * normalize(best_seq) + mix * a_rand
            else:
                a_try = best_seq.copy()
                block_len = int(rng.integers(n_cur // 3, n_cur // 2))
                start_idx = int(rng.integers(0, max(1, n_cur - block_len)))
                a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]

            a_try = normalize(a_try)
            a_try, obj_try = lp_optimize_core(a_try, max_time=per_restart * 0.7,
                                              m_start=1.01, m_end=1.002, perturb_every=15)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart} (strat={strategy}): {best_score:.8f} **")
            else:
                print(f"  Restart {restart} (strat={strategy}): {obj_try:.8f}")

    # ============================================================
    # Phase 6: Final intensive polish (10% of time)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 10:
        n_final = len(best_seq)
        a_final = normalize(best_seq.copy())
        alloc = min(60.0, deadline - time.monotonic())

        # Intensive LP with very tight M
        a_pol, obj_pol = lp_optimize_core(a_final, max_time=alloc * 0.5,
                                          m_start=1.002, m_end=1.0005, perturb_every=60)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            a_final = a_pol
            print(f"  Polish LP: {best_score:.8f} **")

        # Peak targeted
        if time.monotonic() < deadline - 5:
            a_pol2, obj_pol2 = lp_optimize_peak_targeted(a_final, max_time=alloc * 0.3)
            if obj_pol2 < best_score:
                best_score = obj_pol2
                best_seq = a_pol2.copy()
                print(f"  Polish peak: {best_score:.8f} **")

        # Final gradient step
        if time.monotonic() < deadline - 3:
            a_pol3, obj_pol3 = soft_max_gradient_step(a_final, tau_init=0.0005, step_size=0.015,
                                                      max_steps=50, time_budget=min(10, alloc * 0.2))
            if obj_pol3 < best_score:
                best_score = obj_pol3
                best_seq = a_pol3.copy()
                print(f"  Polish grad: {best_score:.8f} **")

    # ============================================================
    # Final validation and return
    # ============================================================
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate with the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
