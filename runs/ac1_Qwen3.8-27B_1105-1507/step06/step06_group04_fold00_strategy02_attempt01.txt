            if isinstance(x, bool) or not isinstance(x, (int, float)):
                return np.inf
            if np.isnan(x) or np.isinf(x):
                return np.inf
        # Convert to floats and defend against negatives
        seq = [float(x) for x in seq]
        seq = [max(0.0, x) for x in seq]
        
        # Compute sum and protect against too small
        s = sum(seq)
        if s < 0.01:
            return np.inf
            
        # Convolution (full) calculation is expensive for large sequences
        n = len(seq)
        if n == 0:
            return np.inf
        conv_size = 2*n - 1
        # There's no FFT-wrapper in the provided code, so we do direct convolution
        conv = np.convolve(seq, seq, mode='full')
        max_conv = np.max(conv)
        obj = float(2.0*n*max_conv/(s**2))
        return obj
        
    except Exception as e:
        # Any exception means invalid
        return np.inf

def pulse_position_modulation(base_sequence, height_factor, width_factor):
    """Create a sequence by PPM modulation with a single peak effect.

    This function models the effect of having two groups of pulses that interfere
    with each other's autocorrelation. It is a more sophisticated initial shape.

    Args:
        base_sequence (numpy array): Base sequence (doesn't have to be normalized).
        height_factor (float): Height of the base for normalization.
        width_factor (float): Width factor for the pulse.

    Returns:
        np.array: The transformed sequence.
    """
    # Create original base sequence and check shape
    n = len(base_sequence)
    # Simple: just add a PPM component
    # This is alternative implementation parameterized by new factors
    shifted_base = np.zeros(n)
    shifted_base[0] = height_factor * 0.2
    shifted_base[n//2] = height_factor * 0.1
    shifted_base[2*n//3] = height_factor * 0.3
    
    # Modify the base_sequence if available
    transformed = base_sequence + shifted_base
    return transformed

def probabilistic_decay(sequence, decay_factor=0.95, time_remain=60.0):
    """Implement a probabilistic coordinate decay to explore the simplex faster.

    Instead of uniform updates, we use a time-based decay on ergodic coordinates.

    Args:
        sequence (numpy.array): Initial sequence.
        decay_factor (float): Probability decay for updating coordinates.
        time_remain (float): Time remaining for the search.

    Returns:
        np.array: New sequence after decay update.
    """
    # This is a simplified version but with probabilistic updates
    # We don't have a clear way to cut corners with constraints
    n_seq = len(sequence)
    decay_prob = 1.0 / (n_seq**(time_remain/1000.0))  # Time-based decay
    decay_mask = np.zeros(n_seq, dtype=bool)
    
    # Apply decay on some coordinates
    for i in range(n_seq):
        decay_value = 1.0 - decay_factor * (np.random.rand() < decay_prob)
        sequence[i] = sequence[i] * decay_value
        # But we must maintain simplex: clip negative and re-normalize if needed
        if sequence[i] < 0.0:
            sequence[i] = 0.0
            
    s_sum = np.sum(sequence) + 1e-10  # Avoid division by infinity
    if s_sum > 1e-10:
        sequence = pulse_position_modulation(sequence, 0.5, 0.4)  # Reinjection of PPM
    
    return sequence
