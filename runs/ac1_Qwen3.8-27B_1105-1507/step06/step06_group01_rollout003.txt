    # This is the same as (p * flip(a)) convolved... let me just use FFT
    # Actually the simplest: the linear cross-correlation of p with a
    # (corr)[i] = sum_j p[i+j] * a[j] for j=0..n-1 (p out of range = 0)
    # This equals (flip(p) * a)[n-1 + i] where * is linear convolution
    # Hmm, let me just compute it with the direct formula using matrix-like ops
    # For n up to ~2000, O(n^2) is 4M ops which is fine
    conv_len = 2 * n - 1
    grad = np.zeros(n)
    for i in range(n):
        p_slice = p[i:i + n]
        grad[i] = 2.0 * np.dot(p_slice, a)
    return logsumexp, grad


def run_fista_smooth(a_init, tau_start, tau_end, max_iters, deadline, lr=0.3):
    """FISTA on smooth max approximation with adaptive tau."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()

    it = 0
    last_eval = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac

        _, grad_y = smooth_max_and_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()

        step = lr / (1.0 + 0.02 * tau * frac)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        it += 1

        # Evaluate periodically
        if it % 10 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()

    M_a = fast_conv(a).max()
    if M_a < best_M:
        best_M = M_a
        best_a = a.copy()

    return best_a


def build_lp_matrix(a, active_indices):
    """Build sparse LP constraint matrix for active indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    if len(rows) == 0:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def solve_lp_best_response(a, M, active_indices, time_limit=8):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a, active_indices)
    if A is None:
        return None
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'mip_rel_gap': 0})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-15:
                return g / g_sum
    except Exception:
        pass
    return None


def get_violated_constraints(a, g, M, active_set, top_k=10):
    """Find constraint indices most violated by (a*g) exceeding M."""
    ag = cross_conv(a, g)
    violations = ag - M
    # Only consider indices NOT in active_set
    active_set_arr = np.array(active_set)
    mask = np.ones(len(violations), dtype=bool)
    mask[active_set_arr] = False
    violations_masked = violations.copy()
    violations_masked[~mask] = -np.inf
    # Get top-k most violated
    n_take = min(top_k, int(mask.sum()))
    if n_take <= 0:
        return []
    top_idx = np.argsort(violations_masked)[::-1][:n_take]
    return [int(i) for i in top_idx if violations_masked[i] > 0]


def run_lp_mirror_descent(a_init, max_iters, deadline, n_active_init=50,
                          n_active_max=300, time_per_lp=8):
    """LP mirror descent with cutting-plane active set growth."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    # Initialize active set
    b = fast_conv(a)
    active = list(np.argsort(b)[::-1][:min(n_active_init, 2 * n - 1)])

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Solve LP with current active set
        g = solve_lp_best_response(a, best_M, active, time_limit=time_per_lp)
        if g is None:
            no_improve += 1
            if no_improve > 5:
                break
            # Try with more active constraints
            if len(active) < n_active_max:
                b = fast_conv(a)
                active = list(np.argsort(b)[::-1][:min(len(active) * 2, n_active_max)])
            continue

        # Find violated constraints and add to active set
        violated = get_violated_constraints(a, g, best_M, active, top_k=5)
        if violated:
            for v_idx in violated:
                if v_idx not in active:
                    active.append(v_idx)
            active = sorted(active)[:n_active_max]

        # Line search for best step
        improved = False
        b = fast_conv(a)
        for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                break

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def coord_opt_1d(a, i, b, n):
    """Find optimal value for a[i] given all others fixed. Golden section search."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Precompute c and slope for each affected k
    c_vals = np.zeros(k_hi - k_lo + 1)
    slopes = np.zeros(k_hi - k_lo + 1)
    has_quad = False
    quad_idx = -1

    for idx, k in enumerate(range(k_lo, k_hi + 1)):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            slopes[idx] = 0.0
            has_quad = True
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                slopes[idx] = 2.0 * a[j]
            else:
                c_vals[idx] = b[k]
                slopes[idx] = 0.0

    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    def obj_at(x_test):
        vals = c_vals + slopes * x_test
        m = base_max
        vm = vals.max()
        if vm > m:
            m = vm
        if has_quad:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.5)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with exact 1D minimization per coordinate."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)

        b = fast_conv(a)
        sweep_improved = False

        for i in order:
            x_opt = coord_opt_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-10:
                x_old = a[i]
                a[i] = x_opt
                # Update b in-place
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += x_opt * x_opt - x_old * x_old
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - x_old)
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 2
        if no_improve > 15:
            break

    return best_a


def fwiess_wolfe_step(a, n, max_iters=200, deadline=None):
    """Frank-Wolfe style: find best vertex direction using subgradient at peak."""
    a = np.maximum(a.astype(np.float64), 0.0) / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_star = int(np.argmax(b))

        # Subgradient at peak: grad[i] = 2*a[k_star - i] if valid
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # Frank-Wolfe: best vertex is argmin grad[i]
        # Since we're on simplex, the FW step goes toward e_{i_min}
        i_min = int(np.argmin(grad))

        # Step size: use 2/(it+2) rule or line search
        t = 2.0 / (it + 2.0)
        a_new = (1.0 - t) * a + t * np.zeros(n)
        a_new[i_min] = t
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.02, n), 1e-4)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.08, n), 1e-4)
        inits.append(p_n2 / p_n2.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())

    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    # Beta-like (concentrated in middle)
    beta_w = np.sin(np.pi * t) ** 2 + 0.1
    inits.append(beta_w / beta_w.sum())

    # Spectrum-flattening init
    a_spec = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_spec)
        k_s = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_s - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.05 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())

    # Smooth random
    ksize = max(3, n // 15)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # Triangular
    tri = 1.0 - np.abs(2 * t - 1)
    tri = np.maximum(tri, 0.05)
    inits.append(tri / tri.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_combined_pipeline(a0, n, time_budget, seed):
    """Run the full pipeline: FISTA -> LP -> CD on a single initialization."""
    t_start = time.monotonic()
    deadline = t_start + time_budget
    best = a0.copy()
    best_M = fast_conv(a0).max()

    # Phase 1: FISTA (30% of time)
    t1 = t_start + 0.30 * time_budget
    if time.monotonic() < t1:
        for tau in [50, 150, 300]:
            if time.monotonic() > t1:
                break
            a_f = run_fista_smooth(best, tau, tau * 3, 2000, t1, lr=0.3)
            M_f = fast_conv(a_f).max()
            if M_f < best_M:
                best_M = M_f
                best = a_f.copy()

    # Phase 2: LP mirror descent (45% of time)
    t2 = t_start + 0.75 * time_budget
    if time.monotonic() < t2:
        a_lp = run_lp_mirror_descent(best, 200, t2, n_active_init=40,
                                     n_active_max=250, time_per_lp=6)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()

    # Phase 3: Coordinate descent (25% of time)
    if time.monotonic() < deadline:
        max_sw = max(5, int((deadline - time.monotonic()) / max(0.3, n / 30.0)))
        a_cd = run_coordinate_descent(best, max_sw, deadline, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()

    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: FISTA + LP cutting plane + coordinate descent
    across multiple initializations and sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    deadline = t_global_start + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError, UnboundLocalError):
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size

    # Determine lengths to try
    lengths_to_try = []
    if n_parent >= 128:
        lengths_to_try.append(n_parent)
    # Try nearby lengths
    for cand in [512, 640, 768, 896, 1024]:
        if cand not in lengths_to_try and 256 <= cand <= 1536:
            lengths_to_try.append(cand)
    # Sort: parent first, then by closeness
    lengths_to_try.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    lengths_to_try = lengths_to_try[:5]

    print(f"\n{'='*60}", flush=True)
    print(f"Lengths to try: {lengths_to_try}", flush=True)
    print(f"{'='*60}", flush=True)

    # Reserve time for final polish
    reserve_time = 200.0
    available = deadline - t_global_start - reserve_time

    n_lengths = len(lengths_to_try)
    time_per_length = available / n_lengths
    time_per_length = max(time_per_length, 60.0)

    # === Main search: for each length, try multiple inits ===
    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - reserve_time:
            break
        t0 = time.monotonic()
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 30:
            break

        time_alloc = min(time_per_length, remaining / max(1, n_lengths - li))
        time_alloc = max(time_alloc, 30.0)
        print(f"\n  --- Length n={n}, time={time_alloc:.0f}s ---", flush=True)

        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_M_for_n = float('inf')

        # Time per init
        n_inits = len(inits)
        time_per_init = time_alloc / max(1, n_inits)
        time_per_init = max(time_per_init, 15.0)

        for ii, a0 in enumerate(inits):
            if time.monotonic() > deadline - reserve_time:
                break
            sub_deadline = min(t0 + time_alloc, deadline - reserve_time)
            if time.monotonic() >= sub_deadline - 5:
                break

            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            a_opt = run_combined_pipeline(a0, n, time_per_init, seed=seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()

            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
                J_val = compute_J(best_for_n)
                print(f"    init {ii}/{n_inits}: M={best_M_for_n:.10f} J={J_val:.6f} "
                      f"(t={time.monotonic()-t0:.0f}s)", flush=True)

        if best_for_n is not None:
            tracker.update(best_for_n)

    tracker.report("After main search")

    # === Final polish phase ===
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL POLISH PHASE", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        n_best = tracker.best.size
        polish_deadline = deadline - 10

        # Extended LP polish
        t_lp = min(polish_deadline, time.monotonic() + 120)
        a_lp = run_lp_mirror_descent(tracker.best, 100, t_lp,
                                     n_active_init=60, n_active_max=400, time_per_lp=8)
        tracker.update(a_lp)

        # High-tau FISTA for fine-tuning
        if time.monotonic() < polish_deadline - 30:
            t_fista = min(polish_deadline - 20, time.monotonic() + 60)
            for tau in [500, 1000]:
                if time.monotonic() > t_fista:
                    break
                a_f = run_fista_smooth(tracker.best, tau, tau * 1.5, 5000,
                                       min(t_fista, time.monotonic() + 30), lr=0.1)
                tracker.update(a_f)

        # Coordinate descent final pass
        if time.monotonic() < polish_deadline - 5:
            max_sw = max(10, int((polish_deadline - 5 - time.monotonic()) / max(0.3, n_best / 40.0)))
            a_cd = run_coordinate_descent(tracker.best, max_sw, polish_deadline - 2, seed=99999)
            tracker.update(a_cd)

        # One more LP pass after CD
        if time.monotonic() < deadline - 2:
            a_final_lp = run_lp_mirror_descent(a_cd, 20, deadline - 1,
                                               n_active_init=30, n_active_max=100, time_per_lp=3)
            tracker.update(a_final_lp)

    tracker.report("After final polish")

    # === Try a couple of different lengths from best ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        n_cur = tracker.best.size
        for new_n in [int(n_cur * 1.2), int(n_cur * 0.8)]:
            new_n = max(256, min(1536, new_n))
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128 or time.monotonic() > deadline - 10:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6) / a_ext.sum()
            a_opt = run_combined_pipeline(a_ext, new_n,
                                          min(40, deadline - time.monotonic() - 5),
                                          seed=seed + 777)
            tracker.update(a_opt)

    # === Fallback ===
    if tracker.best is None:
        tracker.best = np.ones(768) / 768.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-t_global_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
