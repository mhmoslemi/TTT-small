import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_and_grad(a, tau):
    """Compute smooth max (log-sum-exp) of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(p[i:i + n], a)
    return smooth_val, grad


def get_adaptive_active_set(b, M, eps_rel=0.05, min_count=10, max_count=200):
    """Get adaptive active set: all indices within eps_rel of max, bounded by min/max."""
    threshold = M * (1.0 - eps_rel)
    active = np.where(b >= threshold)[0].tolist()
    if len(active) < min_count:
        # Extend to top min_count
        top_idx = np.argsort(b)[::-1][:min_count]
        active = list(set(active) | set(top_idx.tolist()))
    if len(active) > max_count:
        # Keep only top max_count
        active = [int(x) for x in np.argsort(b)[::-1][:max_count]]
    return sorted(active)


def build_sparse_A(n, a, active_indices):
    """Build sparse constraint matrix for LP: (a*g)[k] for k in active_indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response_adaptive(a, M, eps_rel=0.05, max_active=100, lp_time_limit=10):
    """Solve LP best response with adaptive active set.
    
    Returns (g_normalized, dual_values) or (None, None).
    """
    n = len(a)
    b = fast_conv(a)
    
    active = get_adaptive_active_set(b, M, eps_rel=eps_rel, min_count=10, max_count=max_active)
    
    A = build_sparse_A(n, a, active)
    if A.nnz == 0:
        return None, None
    
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': lp_time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                dual = r.ineqlin.marginals if r.ineqlin is not None else None
                return g / g_sum, dual
    except Exception:
        pass
    return None, None

def run_fista_smooth(a_init, tau_start, tau_end, max_iters, deadline, lr=0.3):
    """FISTA on smooth max approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)

        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_and_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        step = lr / (1.0 + 0.03 * tau * it / max_iters)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        
        if it % 10 == 0 or it == max_iters - 1:

            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()

                best_M = M_y
                best_a = y.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:

def subgradient_polyak(a_init, max_iters, deadline, lower_bound=0.0):
    """Subgradient method with Polyak step size for diversification."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Track running min for Polyak step
    running_min = best_M
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Subgradient: grad of max(at k_star) = 2*a[k_star - i] for valid i
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        grad_tan = grad - grad.mean()
        g_norm = np.linalg.norm(grad_tan)
        if g_norm < 1e-15:
            break
        
        # Polyak step size
        gap = M - running_min
        if gap < 1e-15:
            gamma = 0.01 / (1.0 + 0.01 * it)
        else:
            gamma = gap / (g_norm * g_norm)
            gamma = min(gamma, 1.0)
        
        a_new = a - gamma * grad_tan
        a_new = proj_simplex(a_new)
        
        M_new = fast_conv(a_new).max()
        if M_new < running_min:
            running_min = M_new
        
        a = a_new
        
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    
    return best_a

            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for noise_level in [0.02, 0.05, 0.1, 0.2]:
            p_n = np.maximum(p + rng.normal(0, noise_level, n), 1e-4)
            inits.append(p_n / p_n.sum())
    
    inits.append(np.ones(n) / n)
    
    # Parabolic (concave)
    poly = (t * (1 - t)) + 0.05
    inits.append(poly / poly.sum())
    
    # U-shaped (convex)
    poly2 = ((2 * t - 1) ** 2) + 0.05
    inits.append(poly2 / poly2.sum())
    
    # Cosine windows
    for freq in [1, 2, 3, 5]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.005)
        inits.append(cos_w / cos_w.sum())
    
    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.03
    inits.append(gauss / gauss.sum())
    
    # Raised cosine (Hann window)
    hann = 0.5 * (1 + np.cos(2 * np.pi * t))
    hann = np.maximum(hann, 0.001)
    inits.append(hann / hann.sum())
    
    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.005)
    inits.append(smooth / smooth.sum())
    
    # Spectral flattening init
    a_spec = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_spec)
        M_s = bs.max()
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.08 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    
    # "Anti-peak" init: try to reduce the peak
    a_anti = np.ones(n) / n
    for _ in range(20):
        bs = fast_conv(a_anti)
        M_s = bs.max()
        # Reduce elements that contribute most to peak
        k_star = int(np.argmax(bs))
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a_anti[j]
        a_anti = a_anti - 0.05 * contrib
        a_anti = np.maximum(a_anti, 0.01)
        a_anti = a_anti / a_anti.sum()
    inits.append(a_anti.copy())
    
    return inits


def enforce_symmetry(a):
    """Make a symmetric: a[i] = (a[i] + a[n-1-i])/2, then project to simplex."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-10:
        a_sym = a_sym / s
    return a_sym


def run_optimization_pipeline(a0, n, deadline, seed=0):
    """Run full optimization pipeline on a single initialization."""
    best = a0.copy()
    best_M = fast_conv(a0).max()
    t_start = time.monotonic()
    total_time = max(1.0, deadline - t_start)
    
    # Stage 1: FISTA (30% of time)
    t1 = t_start + 0.30 * total_time
    if time.monotonic() < t1:
        a_f = run_fista_smooth(best, 50, 200, 3000, t1, lr=0.25)
        M_f = fast_conv(a_f).max()
        if M_f < best_M:
            best_M = M_f
            best = a_f.copy()
    
    # Stage 2: Adaptive LP (50% of time)
    t2 = t_start + 0.80 * total_time
    if time.monotonic() < t2:
        est_iter_time = 0.3
        max_iters = max(10, int((t2 - time.monotonic()) / est_iter_time))
        a_lp = run_lp_adaptive(best, max_iters, t2, eps_rel=0.05, max_active=60)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
    
    # Stage 3: Subgradient diversification (10%)
    t3 = t_start + 0.90 * total_time
    if time.monotonic() < t3:
        a_sg = subgradient_polyak(best, 2000, t3, lower_bound=best_M * 0.99)
        M_sg = fast_conv(a_sg).max()
        if M_sg < best_M:
            best_M = M_sg
            best = a_sg.copy()
    
    # Stage 4: Final LP polish (5%)
    if time.monotonic() < deadline - 1:
        a_final = run_lp_adaptive(best, 50, deadline - 1, eps_rel=0.02, max_active=100)
        M_fin = fast_conv(a_final).max()
        if M_fin < best_M:
            best_M = M_fin
            best = a_final.copy()
    
    return best


def run_symmetric_pipeline(a0, n, deadline, seed=0):
    """Pipeline that enforces symmetry at each step."""
    a0 = enforce_symmetry(a0)
    best = a0.copy()
    best_M = fast_conv(a0).max()
    t_start = time.monotonic()
    total_time = max(1.0, deadline - t_start)
    
    # FISTA with symmetry projection
    t1 = t_start + 0.4 * total_time
    if time.monotonic() < t1:
        a = best.copy()
        t_mom = 1.0
        y = a.copy()
        a_prev = a.copy()
        for it in range(2000):
            if time.monotonic() > t1:
                break
            tau = 100
            _, grad_y = smooth_max_and_grad(y, tau)
            grad_y = enforce_symmetry(grad_y)
            grad_tan = grad_y - grad_y.mean()
            step = 0.2 / (1.0 + 0.02 * it)
            a_new = y - step * grad_tan
            a_new = enforce_symmetry(a_new)
            a_new = proj_simplex(a_new)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            alpha = (t_mom - 1.0) / t_mom if t_mom > 1.0 else 0.0
            y_new = a_new + alpha * (a_new - a_prev)
            y_new = enforce_symmetry(y_new)
            y_new = proj_simplex(y_new)
            a_prev = a
            a = a_new
            y = y_new
            t_mom = t_new
            if it % 10 == 0:
                M_cur = fast_conv(a).max()
                if M_cur < best_M:
                    best_M = M_cur
                    best = a.copy()
    
    # LP with symmetry
    t2 = deadline - 1
    if time.monotonic() < t2:
        # Build LP on half the variables (symmetric)
        half_n = (n + 1) // 2
        for it in range(200):
            if time.monotonic() > t2:
                break
            g = lp_best_response_adaptive(best, best_M, eps_rel=0.05, max_active=50)[0]
            if g is None:
                break
            g_sym = enforce_symmetry(g)
            for t in [0.3, 0.15, 0.07, 0.03, 0.01]:
                a_trial = (1.0 - t) * best + t * g_sym
                a_trial = enforce_symmetry(a_trial)
                a_trial = proj_simplex(a_trial)
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M:
                    best = a_trial
                    best_M = M_trial
                    break
    
    return best


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)
