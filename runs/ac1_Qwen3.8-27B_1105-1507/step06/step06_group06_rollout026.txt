import numpy as np
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = next_fast_len(2*n - 1)
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (s*s)

def project_nonneg_sum1(a):
    """Project to non-negative, sum=1."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a):
    """Build the matrix A where (A @ g)[k] = (conv(a, g))[k]."""
    n = len(a)
    # (conv(a,g))[k] = sum_{i=0}^{k} a[i]*g[k-i] for k < n
    #                 = sum_{i=k-n+1}^{n-1} a[i]*g[k-i] for k >= n
    # This is a Toeplitz-like matrix of size (2n-1) x n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # g[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # with coefficient a[k-j]
        end = min(j + n, 2*n-1)
        A[j:end, j] = a[:end-j]
    return A

def solve_lp_minpeak(a, time_limit=5):
    """
    Solve: min M s.t. (a*g)[k] <= M for k=0,...,2n-2, sum(g)=1, g>=0
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    
    # Variables: g[0],...,g[n-1], M
    # Objective: minimize M (so c = [0,...,0,1])
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # Inequality: A@g - M <= 0 for each k
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)]*n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception as e:
        print(f"LP error: {e}")
    return None, None

def gerchberg_saxton(a, n_iters=100, target='flat', time_budget=10):
    """
    Gerchberg-Saxton spectral shaping.
    Target: flat power spectrum.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = next_fast_len(2*n-1)
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = np.fft.fft(a, N)
        magnitude = np.abs(fa)
        phase = np.angle(fa)
        
        if target == 'flat':
            # Target: uniform magnitude
            target_mag = np.full(N, np.mean(magnitude))
        else:
            # Compress dynamic range
            max_mag = np.max(magnitude)
            min_mag = np.min(magnitude[magnitude > 0]) if np.any(magnitude > 0) else 1e-10
            # Geometric mean as target
            target_mag = np.sqrt(magnitude * np.full(N, min_mag))
        
        # Blend current and target
        alpha = 0.3 + 0.5 * (it / n_iters)
        new_mag = alpha * target_mag + (1-alpha) * magnitude
        
        fa_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a, best_obj

def spectral_compress(a, power=0.7, n_iters=50, time_budget=10):
    """Compress the power spectrum to reduce peaks."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    N = next_fast_len(2*n-1)
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = np.fft.fft(a, N)
        ps = np.abs(fa)**2
        ps_max = np.max(ps)
        if ps_max < 1e-30:
            break
        
        ps_norm = ps / ps_max
        ps_comp = ps_norm ** power
        # Reconstruct with compressed magnitude
        new_mag = np.sqrt(ps_comp) * np.sqrt(ps_max)
        phase = np.angle(fa)
        fa_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def lp_alternate_aggressive(a, max_iters=1000, time_budget=300.0):
    """
    Aggressive LP alternating with large steps and periodic spectral shaping.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Solve LP
        g, M_opt = solve_lp_minpeak(a, time_limit=3)
        if g is None:
            # Fallback: subgradient step
            conv = compute_conv(a)
            k_star = int(np.argmax(conv))
            grad = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] = a[j]
            grad_centered = grad - np.mean(grad)
            a = project_nonneg_sum1(a - 0.05 * grad_centered)
            stall_count += 1
            continue
        
        # Line search
        best_t_obj = np.inf
        best_t = 0.5
        best_a_try = a.copy()
        
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1-t)*a + t*g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_try = a_try.copy()
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_try.copy()
            a = best_a_try.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Periodic diversification
        if stall_count > 30:
            stall_count = 0
            # Try spectral shaping
            a_spec, obj_spec = spectral_compress(a, power=0.7, n_iters=20, time_budget=5)
            if obj_spec < best_obj:
                best_obj = obj_spec
                best_a = a_spec.copy()
            # Try perturbation
            a_pert = a * np.exp(np.random.default_rng(it).normal(0, 0.05, n))
            a_pert = project_nonneg_sum1(a_pert)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
            a = best_a.copy()
    
    return best_a, best_obj

1. The objective is scale-invariant, so I can normalize `sum(a)=1` and minimize `2*n*max(conv)`
2. The LP alternating method (minimizing peak of `a*g` over `g`) is the most principled approach
3. The gap to target is small (~0.004), so I need aggressive LP optimization with good diversification
4. I should try the iterative LP with large steps and periodic spectral shaping (Gerchberg-Saxton) to escape local optima

My approach:
- Heavy focus on LP alternating with aggressive line search (t close to 1.0)
- Periodic spectral compression (power spectrum flattening) for diversification
- Multi-length exploration (600-900 range)
- Warm restarts and perturbation when stuck

"""
Aggressive LP alternating with spectral compression for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses min-peak LP as the core driver,
Gerchberg-Saxton spectral shaping for diversification, and multi-length
exploration. Focuses on n=700 (parent) with periodic perturbations.

import numpy as np
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def project_nonneg_sum1(a):
    """Project to non-negative with sum=1."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_minpeak(a, time_limit=5.0):
    """
    Solve: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_max_sum(a, M, time_limit=5.0):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    A = build_conv_matrix(a)

    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def spectral_flatten(a, n_iters=80, time_budget=15.0, blend_rate=0.4):
    """
    Gerchberg-Saxton: iteratively flatten the power spectrum.
    Keeps phase, replaces magnitude with geometric mean of current and uniform.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    N = next_fast_len(2 * n - 1)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Target: uniform magnitude (flat spectrum)
        target_mag = np.full(N, float(np.mean(mag)))

        # Geometric blend
        alpha = blend_rate * (0.5 + 0.5 * it / max(1, n_iters))
        new_mag = np.sqrt((1 - alpha) * mag**2 + alpha * target_mag**2)

        fa_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def spectral_compress(a, power=0.6, n_iters=60, time_budget=12.0):
    """Compress power spectrum dynamic range."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    N = next_fast_len(2 * n - 1)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        ps = np.abs(fa) ** 2
        ps_max = float(np.max(ps))
        if ps_max < 1e-30:
            break

        # Gradually reduce compression
        p = power + (1.0 - power) * (it / max(1, n_iters))
        ps_norm = ps / ps_max
        ps_comp = ps_norm ** p

        new_mag = np.sqrt(ps_comp) * np.sqrt(ps_max)
        phase = np.angle(fa)
        fa_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def chase_peak_subgrad(a, n_steps=50, step_size=0.05, time_budget=10.0):
    """Subgradient descent chasing the peak."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for _ in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        k_star = int(np.argmax(conv))

        # Subgradient of max(conv) w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        grad_centered = grad - np.mean(grad)

        for ss in [step_size, step_size * 0.5, step_size * 2.0]:
            a_try = a - ss * grad_centered
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def lp_alternate_aggressive(a, max_iters=5000, time_budget=400.0,
                            t_values=None, diversify_every=50):
    """
    Aggressive LP alternating with periodic diversification.
    Solves min-peak LP, does line search, diversifies when stuck.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)

    if t_values is None:
        t_values = [0.5, 0.7, 0.85, 1.0]

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Solve min-peak LP
        g, M_opt = solve_lp_minpeak(a, time_limit=3.0)

        if g is None:
            # Fallback: subgradient step
            stall += 1
            a_sub, obj_sub = chase_peak_subgrad(a, n_steps=5, step_size=0.03,
                                                time_budget=2.0)
            if obj_sub < best_obj:
                best_obj = obj_sub
                best_a = a_sub.copy()
            a = a_sub.copy()
            continue

        # Line search over t values
        best_t_obj = np.inf
        best_t = 0.5
        best_a_try = a.copy()

        for t in t_values:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_try = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_try.copy()
            a = best_a_try.copy()
            stall = 0
        else:
            stall += 1

        # Diversification when stuck
        if stall >= diversify_every:
            stall = 0
            # Try spectral methods
            a_spec, obj_spec = spectral_flatten(a, n_iters=30, time_budget=8.0)
            if obj_spec < best_obj:
                best_obj = obj_spec
                best_a = a_spec.copy()

            a_comp, obj_comp = spectral_compress(a, power=0.5, n_iters=30,
                                                 time_budget=5.0)
            if obj_comp < best_obj:
                best_obj = obj_comp
                best_a = a_comp.copy()

            # Perturbation
            a_pert = best_a * np.exp(rng.normal(0, 0.03, n))
            a_pert = project_nonneg_sum1(a_pert)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()

            a = best_a.copy()

    return best_a, best_obj


def lp_alternate_maxsum(a, max_iters=2000, time_budget=200.0):
    """
    LP alternating using max-sum formulation.
    Given M = max(conv(a)), find g maximizing sum(g) s.t. (a*g)[k] <= M.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Try slightly reduced M for more aggressive peak reduction
        for M_factor in [1.0, 0.99, 0.98]:
            M_try = M * M_factor
            g = solve_lp_max_sum(a, M_try, time_limit=3.0)
            if g is not None:
                break
        else:
            g = None

        if g is None:
            stall += 1
            if stall > 20:
                break
            continue

        # Line search
        best_t_obj = np.inf
        best_a_try = a.copy()
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_try = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_try.copy()
            a = best_a_try.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break

    return best_a, best_obj

def generate_initialization(n, rng, parent=None, method='uniform'):
    """Generate initial sequence of length n."""
    if parent is not None:
        a = resample_sequence(parent, n)
        if method == 'perturb':
            a = a * np.exp(rng.normal(0, 0.1, n))
        return project_nonneg_sum1(a)

    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'sinusoidal':
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 + 0.05
        return a / a.sum()
    elif method == 'gaussian':
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2 / 3)
        return a / a.sum()
    elif method == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / 0.01) + np.exp(-((x - 0.7) ** 2) / 0.01) + 0.01
        return a / a.sum()
    elif method == 'random':
        a = rng.random(n) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
