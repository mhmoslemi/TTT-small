"""
Optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact coordinate descent, LP alternating, soft-max gradient descent,
and multi-length exploration. Exploits symmetry and FFT-based convolutions.
"""
import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def coordinate_descent(a, max_passes=50, time_limit=None, start_time=None):
    """
    Exact coordinate descent: for each coordinate i, find optimal value
    by 1D minimization of max(conv) subject to x >= 0.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    if time_limit is None:
        time_limit = 60.0
    if start_time is None:
        start_time = time.monotonic()
    
    for p in range(max_passes):
        if time.monotonic() - start_time > time_limit:
            break
        
        conv = compute_conv(a)
        
        for i in range(n):
            if time.monotonic() - start_time > time_limit:
                break
            
            ai = a[i]
            if ai < 1e-15:
                continue
            
            # Compute base convolution (removing a[i]'s contribution)
            # conv[k] gets contribution from a[i]*a[k-i] and a[k-i]*a[i]
            # i.e., 2*a[i]*a[k-i] for k-i valid (or a[i]^2 if k-i=i)
            k_vals = np.arange(i, i + n)  # k such that k-i in [0, n-1]
            k_valid = (k_vals >= 0) & (k_vals < 2 * n - 1)
            k_vals = k_vals[k_valid]
            a_vals = a[k_vals - i]  # a[k-i] for these k
            
            # Base conv: remove contribution
            base = conv.copy()
            base[k_vals] -= 2.0 * ai * a_vals
            # Fix the double-counting at k=2i
            if 2 * i < 2 * n - 1:
                base[2 * i] += ai * ai  # we subtracted 2*ai^2 but should subtract ai^2
            
            # Now find x >= 0 minimizing max_k (base[k] + 2*x*a[k-i]) with quad at k=2i
            # h(x) = max(max_k_linear(base[k] + 2*x*b[k]), base[2i] + x^2)
            # where b[k] = a[k-i] for valid k
            
            # Bisection on subgradient
            lo = 0.0
            hi = max(ai * 3.0, np.max(a) + 1e-6)
            
            for _ in range(25):
                mid = (lo + hi) / 2.0
                # Compute h'(mid) (subgradient)
                # Linear part: max_k (base[k] + 2*mid*b[k])
                linear_vals = base[k_vals] + 2.0 * mid * a_vals
                lin_max = np.max(linear_vals)
                lin_argmax = np.argmax(linear_vals)
                
                # Quadratic part
                if 2 * i < 2 * n - 1:
                    quad_val = base[2 * i] + mid * mid
                else:
                    quad_val = -np.inf
                
                total_max = max(lin_max, quad_val)
                
                # Subgradient: direction of increase
                if total_max == lin_max and quad_val < lin_max:
                    grad = 2.0 * a_vals[lin_argmax]
                elif total_max == quad_val and lin_max < quad_val:
                    grad = 2.0 * mid
                else:
                    # Both are max: subgradient is in [2*mid, 2*a_vals[lin_argmax]]
                    grad = 2.0 * mid  # if quad >= lin, use quad gradient
                    if lin_max > quad_val:
                        grad = 2.0 * a_vals[lin_argmax]
                    else:
                        # Use the one that gives positive gradient (to push down)
                        if 2.0 * a_vals[lin_argmax] > 2.0 * mid:
                            grad = 2.0 * a_vals[lin_argmax]
                        else:
                            grad = 2.0 * mid
                
                if grad > 0:
                    hi = mid
                else:
                    lo = mid
            
            new_ai = (lo + hi) / 2.0
            new_ai = max(0.0, new_ai)
            
            if abs(new_ai - ai) > 1e-10:
                a[i] = new_ai
                a = project_simplex(a)
                conv = compute_conv(a)
                obj = 2.0 * n * np.max(conv)  # sum is 1 after projection
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
    
    return best_a, best_obj


def lp_alternate(a, max_iters=30, time_limit=None, start_time=None):
    """LP alternating: fix a, solve LP for g, blend."""
    n = len(a)
    if n > 2000:
        return a.copy(), compute_obj(a)
    
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    if time_limit is None:
        time_limit = 60.0
    if start_time is None:
        start_time = time.monotonic()
    
    # Build Toeplitz matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_limit:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv)) * 1.001  # Slight relaxation for numerical issues
        
        # LP: max sum(g) s.t. A*g <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 10})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s < 1e-10:
                break
            g = g / s
        except Exception:
            break
        
        # Line search
        best_t_obj = best_obj
        best_t_val = 0.5
        best_a_local = best_a.copy()
        
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = project_simplex((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local
            a = best_a_local
            # Update Toeplitz
            for j in range(n):
                k_start = j
                k_end = min(j + n, 2 * n - 1)
                A[k_start:k_end, j] = a[:k_end - k_start]
        else:
            break
    
    return best_a, best_obj


def softmax_gradient_descent(a, max_iters=200, time_limit=None, start_time=None,
                              tau_init=50.0, lr_init=0.1, momentum=0.8):
    """Gradient descent on soft-max of convolution with Nesterov momentum."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    if time_limit is None:
        time_limit = 60.0
    if start_time is None:
        start_time = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    step = lr_init
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_limit:
            break
        
        tau = tau_init * (1.0 + 0.1 * it)
        
        # Compute conv of y
        fy = rfft(y, N)
        conv_y = irfft(fy * fy, N)[:2 * n - 1]
        
        cmax = float(np.max(conv_y))
        shifted = np.clip((conv_y - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv_y))
            weights[int(np.argmax(conv_y))] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (correlation)
        # Use FFT for cross-correlation
        fw = rfft(weights, N)
        fa = rfft(y, N)
        # corr[m] = sum_k weights[k] * y[k-m+n-1] ... 
        # Let's compute sum_k weights[k] * y[k-i] for i = 0,...,n-1
        # This is the correlation of weights and y
        # = ifft(fft(weights) * conj(fft(y))) gives circular correlation
        # We need the linear part
        
        corr = irfft(fw * np.conj(fa), N)
        # corr[m] = sum_k weights[k+m] * y[k] (circular)
        # What we want: sum_k weights[k] * y[k-i]
        # = sum_k weights[k] * y[k-i] 
        # Let j = k-i: sum_j weights[j+i] * y[j]
        # = corr_circular at offset i (with proper indexing)
        
        # Actually let me just compute it directly for clarity
        # grad[i] = 2 * sum_{k=i}^{i+n-1} weights[k] * y[k-i]
        # = 2 * (linear correlation of weights and y at lag i)
        
        grad = np.zeros(n)
        for i in range(n):
            k_start = i
            k_end = i + n
            if k_end <= len(weights):
                grad[i] = 2.0 * np.dot(weights[k_start:k_end], y[:n])
            else:
                valid_end = min(k_end, len(weights))
                grad[i] = 2.0 * np.dot(weights[k_start:valid_end], y[:valid_end - k_start])
        
        # Center gradient (tangent to simplex)
        grad_centered = grad - np.mean(grad)
        
        # Nesterov update
        v = momentum * v - step * grad_centered
        a_new = project_simplex(y + v)
        y = a_new  # Simplified (not full Nesterov)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        step *= 0.995
    
    return best_a, best_obj


def freq_domain_flatten(a, max_iters=100, time_limit=None, start_time=None, alpha=0.8):
    """Flatten the self-convolution in frequency domain."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    if time_limit is None:
        time_limit = 30.0
    if start_time is None:
        start_time = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_limit:
            break
        
        fa = fft(a, N)
        power = np.abs(fa) ** 2
        
        # Compute self-convolution
        conv = irfft(fa * fa, N)[:2 * n - 1]
        cmax = np.max(conv)
        if cmax < 1e-15:
            break
        
        # Compress peaks in convolution
        conv_mod = conv.copy()
        # Soft clip: anything above threshold gets compressed
        threshold = cmax * 0.7
        above = conv_mod > threshold
        if np.any(above):
            conv_mod[above] = threshold + (conv_mod[above] - threshold) * alpha
        
        conv_mod = np.maximum(conv_mod, 0)
        
        # Take square root in frequency domain
        cf = fft(conv_mod, N)
        fa_new = np.sqrt(cf * 0.5 + 0j)  # * 0.5 because ifft(fft(x)^2) = circular conv
        a_new = ifft(fa_new).real[:n]
        
        a_new = project_simplex(a_new)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a, best_obj


def symmetrize(a):
    """Make sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return project_simplex(a_sym)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Symmetric triangular
    x = np.linspace(0, 1, n)
    tri = 1.0 - np.abs(2.0 * x - 1.0)
    inits.append(tri / tri.sum())
    
    # Sinusoidal
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())
    
    # Gaussian
    xg = np.linspace(-3, 3, n)
    gauss = np.exp(-xg ** 2 / 3.0)
    inits.append(gauss / gauss.sum())
    
    # Flat top with smooth edges
    edge = int(n * 0.15)
    flat = np.ones(n)
    if edge > 2:
        for i in range(edge):
            t = (i + 1) / (edge + 1)
            flat[i] = 0.5 * (1 - np.cos(np.pi * t))
            flat[n - 1 - i] = 0.5 * (1 - np.cos(np.pi * t))
    inits.append(flat / flat.sum())
    
    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    inits.append(rc / rc.sum())
    
    if parent is not None:
        # Resampled parent
        inits.append(resample_sequence(parent, n))
        # Perturbed parent
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_simplex(pert))
        # Symmetrized parent
        inits.append(symmetrize(resample_sequence(parent, n)))
    
    return inits

                    print(f"  Cycle {cycle}: PERTURB UPDATE obj={best_score:.8f}")
    
    print(f"After Phase 2: best={best_score:.8f}, n={len(best_seq)}")
    
    # === Phase 3: Multi-length exploration (20% time) ===
    phase3_end = start_time + 0.80 * (budget_s - 30)
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    
    n_current = len(best_seq)
    # Try various lengths
    n_candidates = []
    for delta in [-100, -50, 50, 100, 200, 300, 500]:
        n_try = n_current + delta
        if 100 <= n_try <= 5000:
            n_candidates.append(n_try)
    # Also try some absolute values
    for n_abs in [300, 500, 800, 1200, 1500, 2000]:
        if n_abs not in n_candidates and 100 <= n_abs <= 5000:
            n_candidates.append(n_abs)
    n_candidates = list(dict.fromkeys(n_candidates))
    rng.shuffle(n_candidates)
    
    per_n_budget = max(5.0, (phase3_end - time.monotonic()) / max(1, len(n_candidates)))
    
    for n_try in n_candidates:
        if time.monotonic() > phase3_end:
            break
        if n_try == n_current:
            continue
        
        a_resampled = resample_sequence(best_seq, n_try)
        a_resampled = project_simplex(a_resampled)
        
        # Optimize
        a_opt, obj_opt = coordinate_descent(a_resampled, max_passes=10,
                                             time_limit=min(per_n_budget * 0.4, 15.0),
                                             start_time=time.monotonic())
        if n_try <= 1500:
            a_opt2, obj_opt2 = lp_alternate(a_opt, max_iters=8,
                                             time_limit=min(per_n_budget * 0.3, 15.0),
                                             start_time=time.monotonic())
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2
        a_opt3, obj_opt3 = coordinate_descent(a_opt, max_passes=10,
                                               time_limit=min(per_n_budget * 0.2, 10.0),
                                               start_time=time.monotonic())
        if obj_opt3 < obj_opt:
            a_opt, obj_opt = a_opt3, obj_opt3
        
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    print(f"After Phase 3: best={best_score:.8f}, n={len(best_seq)}")
    
    # === Phase 4: Final polish (remaining time) ===
    print(f"\n=== Phase 4: Final Polish ===")
    
    a_final = project_simplex(best_seq.copy())
    n_final = len(a_final)
    
    while time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        # Coordinate descent (most reliable)
        t_cd = min(remaining * 0.4, 30.0)
        a_cd, obj_cd = coordinate_descent(a_final, max_passes=20,
                                           time_limit=t_cd,
                                           start_time=time.monotonic())
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a_final = a_cd.copy()
            print(f"  Polish CD: {best_score:.8f}")
        
        if time.monotonic() > deadline - 5:
            break
        
        # LP if feasible
        if n_final <= 2000:
            remaining = deadline - time.monotonic()
            if remaining > 15:
                t_lp = min(remaining * 0.3, 25.0)
                a_lp, obj_lp = lp_alternate(a_final, max_iters=10,
                                             time_limit=t_lp,
                                             start_time=time.monotonic())
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_final = a_lp.copy()
                    print(f"  Polish LP: {best_score:.8f}")
        
        if time.monotonic() > deadline - 5:
            break
        
        # Try symmetrization
        a_s = symmetrize(a_final)
        obj_s = compute_obj(a_s)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_s.copy()
            a_final = a_s.copy()
            print(f"  Polish SYM: {best_score:.8f}")
            # Further optimize the symmetric version
            a_cd2, obj_cd2 = coordinate_descent(a_final, max_passes=15,
                                                 time_limit=15.0,
                                                 start_time=time.monotonic())
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                a_final = a_cd2.copy()
                print(f"  Polish SYM+CD: {best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
