import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_feasible(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz_reduced(a, active_indices):
    """Build Toeplitz matrix only for active constraints."""
    n = len(a)
    K = len(active_indices)
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(active_indices):
        # (a*g)[k] = sum_i a[i] * g[k-i] for valid i, k-i
        # For g[j], it contributes to (a*g)[k] when k-j is a valid index of a
        # i.e., a[k-j] * g[j] contributes when 0 <= k-j < n
        # So column j of the row for constraint k is a[k-j] if 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[row, j] = a[k - j]
    return A

def solve_lp_reduced(a, M, active_indices, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_indices, g >= 0."""
    n = len(a)

    c = -np.ones(n)

    bounds = [(0.0, None)] * n

            s = float(np.sum(g))

    except Exception:

"""
Optimized LP-alternating with reduced constraints, Anderson acceleration,
and multi-strategy polish to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: (1) reduced LP with adaptive constraint selection,
(2) over-relaxation with backtracking, (3) dual LP as alternative direction,
(4) adaptive K (number of active constraints), (5) periodic random restarts,
(6) efficient FFT-based convolution throughout.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_feasible(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a, indices):
    """Build constraint matrix rows for given convolution indices."""
    n = len(a)
    K = len(indices)
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        cols = np.arange(j_start, j_end + 1)
        A[row, cols] = a[k - cols]
    return A


def solve_lp_max_sum(a, M, active_indices, time_limit=8):
    """LP: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a, active_indices)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=10):
    """LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 1500:
        return None, None
    A_full = build_toeplitz(a, np.arange(2 * n - 1))
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_full
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None

def subgradient_descent(a, max_iters=300, time_budget=60.0, momentum=0.8):
    """Subgradient with Nesterov momentum on smoothed max."""
    n = len(a)
    a = project_feasible(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    step = 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights
        tau = 0.01 + 0.1 * (it / max_iters)
        shifted = np.clip(conv / tau - cmax / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= wsum

        # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j] (where valid)
        grad = np.zeros(n)
        # Use FFT for efficiency
        # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
        # This is 2 * (w * a)[n-1-j] ... let me compute via convolution
        # Actually: sum_k w[k] * a[k-j] = (w_rev * a)[n-1-j] where w_rev = w[::-1]
        # Or more directly: it's the cross-correlation
        # Let's use FFT: ifft(fft(w) * conj(fft(a))) gives cross-correlation
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        fw = np.fft.fft(weights, N)
        fa = np.fft.fft(a, N)
        corr = np.fft.ifft(fw * np.conj(fa)).real
        # corr[m] = sum_k w[k] * conj(a[k-m]) = sum_k w[k] * a[k-m] (real)
        # We want: grad[j] = 2 * sum_k w[k] * a[k-j] = 2 * corr[j]
        # But corr has period N, and we need j in [0, n-1]
        grad = 2.0 * corr[:n]

        # Nesterov momentum
        v = momentum * v + grad
        a_new = a + step * v  # We want to MINIMIZE max, so go AGAINST gradient
        # Wait: we want to minimize max(conv). The gradient of max(conv) w.r.t. a
        # points in the direction of increase. So we should go a - step * grad.
        # But with momentum: v = momentum*v + grad, then a = a - step*v
        # Let me redo:
        v = momentum * v + grad
        a_new = a - step * v
        a_new = project_feasible(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Don't update a, just reduce step
            step *= 0.95

    return best_a, best_obj


def chase_peak_fast(a, n_steps=100, step_size=0.02):
    """Chase the maximum peak with subgradient and line search."""
    n = len(a)
    a = project_feasible(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient at k_star
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        cols = np.arange(j_start, j_end + 1)
        grad[cols] = 2.0 * a[k_star - cols]

        # Center
        grad_centered = grad - np.mean(grad)

        # Try decreasing step sizes
        improved = False
        for ss in [step_size, step_size * 0.5, step_size * 0.25, step_size * 2.0]:
            a_try = a - ss * grad_centered
            a_try = project_feasible(a_try)

            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_obj


def freq_domain_flat(a, max_iters=100, time_budget=30.0):
    """
    Frequency domain: iteratively flatten the power spectrum |FFT(a)|^2
    to reduce the peak of the time-domain convolution.
    """
    n = len(a)
    a = project_feasible(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2

        # Target: flatten power spectrum (reduces peak of ifft)
        # Use iterative amplitude modification
        # We want ifft(fa*fa) to be flat, i.e., fa*fa should be close to constant
        # So we want |fa|^2 to be close to constant
        # This means |fa| should be close to constant
        target_mag = np.mean(power)
        if target_mag < 1e-30:
            break

        # Blend current magnitude toward target
        mag = np.sqrt(power)
        alpha = 0.1 + 0.2 * (it / max_iters)
        mag_new = (1 - alpha) * mag + alpha * np.sqrt(target_mag)

        # Preserve phase
        phase = np.angle(fa)
        fa_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_feasible(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] / new_n if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_feasible(a_new)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []

    # Sinusoidal bump

    # Super-Gaussian
    for p in [2, 4, 8]:
        sg = np.exp(-((2 * x - 1) ** 2) ** p) + 0.001
        inits.append(sg / sg.sum())

    # Polynomial bump
    poly = (1 - (2 * x - 1) ** 2) ** 3 + 0.001
    inits.append(poly / poly.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Two-bump (asymmetric)
    two = np.exp(-((x - 0.35) ** 2) / 0.01) + 0.6 * np.exp(-((x - 0.65) ** 2) / 0.01)

        # Perturbed parent
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.1, n))

    print(f"\n=== Phase 2: LP Alternating (main engine) ===")
    print(f"Best before: {best_score:.8f}")
    phase2_budget = (deadline - start_time) * 0.55
    p2_start = time.monotonic()

    a_main = project_feasible(best_seq.copy())
    n_main = len(a_main)

    # Split phase 2 into sub-rounds with different strategies
    num_rounds = min(8, int(phase2_budget / 60))
    round_budget = phase2_budget / max(1, num_rounds)

    for rnd in range(num_rounds):
        if time.monotonic() - p2_start > phase2_budget:
            break
        rem = p2_start + phase2_budget - time.monotonic()
        if rem < 10:
            break

        alloc = min(round_budget, rem)
        strategy = rnd % 4

        if strategy == 0:
            # Main LP alternating with reduced constraints
            a_new, obj_new = lp_alternate_fast(
                a_main, max_iters=200, time_budget=alloc * 0.9,
                K_init=max(100, n_main // 3), K_min=50, K_max=min(2*n_main-1, n_main))
        elif strategy == 1:
            # LP alternating with minpeak LP direction
            a_new, obj_new = a_main.copy(), compute_obj(a_main)
            sub_start = time.monotonic()
            stall = 0
            while time.monotonic() - sub_start < alloc * 0.9 and stall < 30:
                g, M_opt = solve_lp_minpeak(a_main, time_limit=min(10, alloc * 0.1))
                if g is None:
                    break
                # Line search
                best_t_obj = np.inf
                best_t = 0
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - t) * a_main + t * g
                    a_try = project_feasible(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_t_obj:
                        best_t_obj = obj_try
                        best_t = t
                if best_t_obj < obj_new - 1e-12:
                    obj_new = best_t_obj
                    a_new = (1 - best_t) * a_main + best_t * g
                    a_new = project_feasible(a_new)
                    a_main = a_new.copy()
                    stall = 0
                else:
                    stall += 1
        elif strategy == 2:
            # Subgradient with momentum
            a_new, obj_new = subgradient_descent(
                a_main, max_iters=200, time_budget=alloc * 0.8)
        else:
            # Chase peak + freq domain
            a_tmp, obj_tmp = chase_peak_fast(a_main, n_steps=50, step_size=0.02)
            a_new, obj_new = freq_domain_flat(a_tmp, max_iters=80, time_budget=alloc * 0.5)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_main = a_new.copy()
            print(f"  Round {rnd} (strat {strategy}): NEW BEST {best_score:.8f}")
        else:
            a_main = a_new.copy()  # still use the improved local solution
            print(f"  Round {rnd} (strat {strategy}): obj={obj_new:.8f} (best={best_score:.8f})")

    # === Phase 3: Multi-length exploration (20% of time) ===
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    print(f"Best before: {best_score:.8f}")
    phase3_budget = (deadline - start_time) * 0.20
    p3_start = time.monotonic()

    n_current = len(best_seq)
    n_tries = [n_current - 20, n_current - 50, n_current - 100,
               n_current + 20, n_current + 50, n_current + 100,
               n_current + 200, max(200, n_current - 200)]
    n_tries = list(dict.fromkeys([n for n in n_tries if 100 <= n <= 2500]))

    per_n_budget = max(5.0, phase3_budget / len(n_tries))

    for n_try in n_tries:
        if time.monotonic() - p3_start > phase3_budget:
            break
        a_res = resample_sequence(best_seq, n_try)
        a_opt, obj_opt = chase_peak_fast(a_res, n_steps=30, step_size=0.02)

        if n_try <= 1000:
            a_opt2, obj_opt2 = lp_alternate_fast(
                a_opt, max_iters=50, time_budget=min(per_n_budget * 0.5, 30.0),
                K_init=max(50, n_try // 4), K_min=30)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 4: Intensive final polish (remaining time) ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"Best before: {best_score:.8f}")
    remaining = deadline - time.monotonic()
    if remaining > 15 and best_seq is not None:
        a_final = project_feasible(best_seq.copy())
        n_final = len(a_final)

        # Multiple polishing rounds
        num_polish = min(10, int(remaining / 10))
        for pr in range(num_polish):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            # LP alternating (most effective)
            if n_final <= 1500:
                a_p, obj_p = lp_alternate_fast(
                    a_final, max_iters=min(100, int(rem / 2)),
                    time_budget=rem * 0.4,
                    K_init=max(80, n_final // 4), K_min=40)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Polish LP {pr}: {best_score:.8f}")

            # Chase peak
            rem = deadline - time.monotonic()
            if rem > 5:
                a_p2, obj_p2 = chase_peak_fast(a_final, n_steps=min(50, int(rem)), step_size=0.01)
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_final = a_p2.copy()
                    print(f"  Polish CP {pr}: {best_score:.8f}")

            # Small perturbation + re-optimize
            rem = deadline - time.monotonic()
            if rem > 10:
                a_pert = a_final * np.exp(rng.normal(0, 0.01, n_final))
                a_pert = project_feasible(a_pert)
                a_pert, obj_pert = chase_peak_fast(a_pert, n_steps=30, step_size=0.01)
                if n_final <= 1000:
                    a_pert2, obj_pert2 = lp_alternate_fast(
                        a_pert, max_iters=30, time_budget=min(rem * 0.3, 15.0),
                        K_init=max(50, n_final // 5), K_min=30)
                    if obj_pert2 < obj_pert:
                        a_pert, obj_pert = a_pert2, obj_pert2
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_final = a_pert.copy()
                    print(f"  Polish pert {pr}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
