```python
import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def conv_with_fixed(f, g):
    """Compute conv(f, g) where f is fixed, using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    ff = rfft(f, size)
    fg = rfft(g, size)
    return irfft(ff * fg, size)[:2 * n - 1]

def lp_minpeak(a, use_fft=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build A matrix or use FFT-based approach
    # For CVXPY, we need explicit A
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                s = float(np.sum(g_val))
                if s > 1e-10:

    return None, None
```

```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed gradient descent with simplex projection as the main engine,
combined with LP alternating for polishing and multi-n exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def conv_xy(x, y):
    """Compute conv(x, y) via FFT."""
    n = len(x)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fx = rfft(x, size)
    fy = rfft(y, size)
    return irfft(fx * fy, size)[:2 * n - 1]

def simplex_project(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    d = np.arange(1, n + 1, dtype=np.float64)
    cond = u * d > cv
    m = int(np.sum(cond))
    if m == 0:
        return np.ones(n) / n
    theta = (float(np.sum(v[:m])) - 1.0) / m
    # Wait, v is not sorted. I need to use the sorted values to find theta,
    # but apply it to the original v.
    # theta = (sum of top m values of v - 1) / m
    # But I sorted u = sort(v)[::-1], so the top m values are u[0:m]
    theta = (float(np.sum(u[:m])) - 1.0) / m
    return np.maximum(v - theta, 0.0)

def compute_grad_smoothed(a, p):
    """
    Compute gradient of smoothed objective sum_k p_k * conv_k(a).
    p is array of length 2n-1 (softmax weights).
    a is array of length n.
    Returns gradient of length n.
    """

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        pk = p[k]
        # Term 1: a[k-j] for valid j

        hi1 = min(n, k + 1)
        if lo1 < hi1:

            grad[lo1:hi1] += pk * a[idx]

            # j-k ranges from lo2-k to hi2-k-1
            idx = np.arange(lo2 - k, hi2 - k)
            grad[lo2:hi2] += pk * a[idx]
    
    return grad

    """
    Smoothed gradient descent for minimizing max(conv_k(a)) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    tau = tau0
    it = 0
    
    # Compute initial conv
    conv = compute_conv(a)
    
    while time.monotonic() - start < time_budget:
        it += 1
        
        # Compute softmax weights
        conv_max = float(np.max(conv))
        # Shift for numerical stability
        exp_vals = np.exp((conv - conv_max) / tau)
        p = exp_vals / np.sum(exp_vals)
        
        # Compute gradient
        grad = compute_grad_smoothed(a, p)
        
        # Center gradient (to maintain sum constraint approximately)
        grad_centered = grad - np.mean(grad)
        
        # Step size
        eta = eta0 / np.sqrt(it + 1)
        
        # Take step
        y = a - eta * grad_centered
        
        # Project onto simplex
        a_new = simplex_project(y)
        
        # Compute objective
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = compute_conv(a)
        
        # Update temperature
        tau *= tau_decay
        if tau < 1e-8:
            tau = 1e-8
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      SGD it{it}: obj={best_obj:.8f}, tau={tau:.2e}, elapsed={elapsed:.1f}s")
    
    return best_a

def lp_alternating(a, time_budget=100.0, verbose=False):
    """
    LP alternating optimization.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        # Solve LP: min M s.t. conv(a, g)[k] <= M, sum(g)=1, g>=0
        a_norm = a / np.sum(a)
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a_norm[:r]
        
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        g_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    break
            except Exception:
                continue
        
        if g_val is None:
            break
        
        gs = float(np.sum(g_val))
        if gs < 1e-10:
            break
        g_val = g_val / gs
        
        # Line search
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()
        
        for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose and it % 10 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t_val:.2f}")
        else:
            # Try perturbation
            rng = np.random.default_rng(hash((it, n, it * 7)) % (2**32))
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
        
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def random_init(n, rng):
    """Generate random non-negative sequence."""
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Smooth
        a = np.cumsum(rng.uniform(0, 1, n)) + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        center = n // 2
        width = max(1, n // 5)
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a += 0.5 * np.exp(-((x - center * 0.6) ** 2) / (2 * (width // 2) ** 2))
        a = a / np.sum(a)
    else:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.3
        a = a / np.sum(a)
    return a
```

Wait, I need to double-check my `simplex_project` function. The standard algorithm is:

Given v, find x = argmin ||x - v||^2 subject to sum(x) = 1, x >= 0.

The solution is x_j = max(v_j - theta, 0) where theta is chosen so sum(x) = 1.

To find theta:
- Sort v in decreasing order: v_(1) >= v_(2) >= ... >= v_(n)
- For m = 1, ..., n: define theta_m = (sum_{j=1}^m v_(j) - 1) / m
- The correct m is the largest one such that v_(m) > theta_m
- Then theta = theta_m

Let me fix my implementation:

```python
def simplex_project(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)

    u = np.sort(v)[::-1]

    return result
```

v_(m) > (sum_{j=1}^m v_(j) - 1) / m

Which is: v_(m) * m > sum_{j=1}^m v_(j) - 1
i.e., v_(m) * m + 1 > sum_{j=1}^m v_(j)
i.e., v_(m) * m > cumsum(u)[m-1] - 1

In 0-indexed: u[m-1] * m > cumsum(u)[m-1] - 1

Which is: u[i] * (i+1) > cumsum(u)[i] - 1 for i = 0, ..., n-1 (where i = m-1)

So `cond[i] = u[i] * (i+1) > cv[i]` where `cv[i] = cumsum(u)[i] - 1`.

Let me also think about whether I should use **DAGM** (Distributed Average Greedy Method) or **mirror descent** instead. Mirror descent with the negative entropy DRO (relative to the uniform distribution) gives multiplicative updates:
a_new[j] ∝ a[j] * exp(-eta * grad[j])

This naturally preserves positivity and might converge faster. The "projection" in mirror descent with entropy DRO is just normalization (since the Bregman projection onto {x: sum(x)=1} with negative entropy DRO is x_new = x * exp(-eta*grad) / sum(x * exp(-eta*grad))).

```python
def mirror_descent(a, time_budget=100.0, tau0=0.01, tau_decay=0.999, eta0=1.0):

    it = 0

        it += 1
        
        conv = compute_conv(a)
        conv_max = float(np.max(conv))
        exp_vals = np.exp((conv - conv_max) / tau)
        p = exp_vals / np.sum(exp_vals)
        
        grad = compute_grad_smoothed(a, p)
        
        eta = eta0 / np.sqrt(it + 1)
        
        # Multiplicative update (mirror descent with entropy)
        a_new = a * np.exp(-eta * (grad - np.mean(grad)))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        
        tau *= tau_decay
        if tau < 1e-8:
            tau = 1e-8
    
    return best_a
```

```python
def active_set_qp(a, K=10, time_budget=30.0):
    """Solve QP with active set constraints."""
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    
    # Find top K bins
    top_k = np.argsort(conv)[::-1][:K]
    t_current = float(conv[top_k[0]])
    
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1, t_var >= 0]
    
    for k in top_k:
        # conv_k = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
        # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
```

```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent (multiplicative updates), smoothed gradient descent with
simplex projection, LP alternating, active-set QP, and multi-n exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def conv_xy(x, y):
    """Compute linear conv(x, y) via FFT."""
    n = len(x)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fx = rfft(x, size)
    fy = rfft(y, size)
    return irfft(fx * fy, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def simplex_project(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    d = np.arange(1, n + 1, dtype=np.float64)
    cond = u * d > cv
    m = int(np.sum(cond))
    if m == 0:
        return np.ones(n) / n
    theta = (float(np.sum(u[:m])) - 1.0) / m
    result = np.maximum(v - theta, 0.0)
    s = float(np.sum(result))
    if s > 0:
        result = result / s
    return result


def compute_grad_smoothed(a, p):
    """
    Compute gradient of sum_k p[k] * conv_k(a).
    p: array of length 2n-1 (weights).
    a: array of length n.
    Returns gradient of length n.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        pk = p[k]
        # Term 1: a[k-j] for valid j (0 <= k-j <= n-1)
        lo1 = max(0, k - n + 1)
        hi1 = min(n, k + 1)
        if lo1 < hi1:
            idx = np.arange(k - lo1, k - hi1, -1)
            grad[lo1:hi1] += pk * a[idx]
        # Term 2: a[j-k] for valid j (0 <= j-k <= n-1)
        lo2 = max(0, k)
        hi2 = min(n, k + n)
        if lo2 < hi2:
            idx = np.arange(lo2 - k, hi2 - k)
            grad[lo2:hi2] += pk * a[idx]
    return grad


def softmax_weights(conv, tau):
    """Compute softmax weights from convolution values."""
    conv_max = float(np.max(conv))
    exp_vals = np.exp((conv - conv_max) / tau)
    s = float(np.sum(exp_vals))
    if s < 1e-300:
        return np.full(len(conv), 1.0 / len(conv))
    return exp_vals / s


def mirror_descent(a, time_budget=100.0, tau0=0.01, tau_decay=0.999, eta0=1.0, verbose=False):
    """
    Mirror descent with negative entropy (multiplicative updates).
    Minimizes max(conv_k(a)) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau0
    it = 0
    patience = 0
    max_patience = 500

    while time.monotonic() - start < time_budget:
        it += 1

        conv = compute_conv(a)
        p = softmax_weights(conv, tau)
        grad = compute_grad_smoothed(a, p)
        grad_centered = grad - np.mean(grad)

        eta = eta0 / np.sqrt(it + 1)
        a_new = a * np.exp(-eta * grad_centered)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            a = a_new
            patience += 1

        tau *= tau_decay
        if tau < 1e-8:
            tau = 1e-8

        if patience > max_patience:
            # Reset with perturbation
            rng = np.random.default_rng(hash((it, n)) % (2**32))
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-15)
            a = a / np.sum(a)
            patience = 0
            tau = max(tau, tau0 * 0.1)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"      MD it{it}: obj={best_obj:.8f}, tau={tau:.2e}, eta={eta:.4f}, elapsed={elapsed:.1f}s")

    return best_a

def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        a_norm = a / np.sum(a)
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a_norm[:r]

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        g_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    break
            except Exception:
                continue

        if g_val is None:
            break

        gs = float(np.sum(g_val))
        if gs < 1e-10:
            break
        g_val = g_val / gs

        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t_val:.2f}")
        else:
            rng = np.random.default_rng(hash((it, n, 7)) % (2**32))
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s

    return best_a


def active_set_qp(a, K=8, time_budget=30.0):
    """Solve QP with top-K active constraints."""
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    top_k = np.argsort(conv)[::-1][:K]

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1, t_var >= 0.0]

    for k in top_k:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            terms = [a_var[i] * a_var[k - i] for i in range(lo, hi + 1)]
            constraints.append(cp.sum(terms) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.maximum(a_var.value, 0)
                s = float(np.sum(a_val))
                if s > 1e-10:
                    return a_val / s
        except Exception:
            continue

    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        a = np.cumsum(rng.uniform(0, 1, n)) + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        center = n // 2
        width = max(1, n // 5)
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - center) ** 2) / (2.0 * width ** 2))
        a += 0.5 * np.exp(-((x - center * 0.6) ** 2) / (2.0 * max(1, width // 2) ** 2))
        a = a / np.sum(a)
    else:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.3
        a = a / np.sum(a)
    return a


def chase_peak_refine(a, time_budget=20.0):
    """Chase-the-peak subgradient method targeting the maximum convolution bin."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.02

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= k_star - j < n:
                val += a[k_star - j]
            if 0 <= j - k_star < n:
                val += a[j - k_star]
            grad[j] = val

        grad_centered = grad - np.mean(grad)
        a_new = a - step_size * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                step_size = min(step_size * 1.05, 0.1)
            else:
                step_size *= 0.7
                a_new2 = a - step_size * grad_centered
                a_new2 = np.maximum(a_new2, 0)
                s2 = float(np.sum(a_new2))
                if s2 > 1e-10:
                    a_new2 = a_new2 / s2
                    obj2 = compute_obj(a_new2)
                    if obj2 < best_obj:
                        best_obj = obj2
                        best_a = a_new2.copy()
                        a = a_new2
        else:
            step_size *= 0.5

    return best_a

    if remaining > 30:
        alloc = min(remaining * 0.15, 100.0)
        a_mom = momentum_sgd(best_seq, time_budget=alloc, tau0=1.0 / n_parent, tau_decay=0.999, eta0=0.5, momentum=0.9)
        obj_mom = compute_obj(a_mom)
        if obj_mom < best_score:
            best_score = obj_mom
            best_seq = a_mom.copy()
            print(f"  MomSGD: obj={best_score:.8f} **")
        else:
            print(f"  MomSGD: obj={obj_mom:.8f}")

    # Phase 4: LP alternating
    print(f"\n=== Phase 4: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 150.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # Phase 5: Active set QP
    print(f"\n=== Phase 5: Active Set QP ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for K in [5, 10, 15]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_qp = active_set_qp(best_seq, K=K, time_budget=min(remaining * 0.25, 30.0))
            obj_qp = compute_obj(a_qp)
            if obj_qp < best_score:
                best_score = obj_qp
                best_seq = a_qp.copy()
                print(f"  QP K={K}: obj={best_score:.8f} **")
            else:
                print(f"  QP K={K}: obj={obj_qp:.8f}")

    # Phase 6: Chase peak
    print(f"\n=== Phase 6: Chase Peak ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.1, 30.0)
        a_chase = chase_peak_refine(best_seq, time_budget=alloc)
        obj_chase = compute_obj(a_chase)
        if obj_chase < best_score:
            best_score = obj_chase
            best_seq = a_chase.copy()
            print(f"  Chase: obj={best_score:.8f} **")
        else:
            print(f"  Chase: obj={obj_chase:.8f}")

    # Phase 7: Multi-n exploration
    print(f"\n=== Phase 7: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([n_best - 100, n_best - 50, n_best + 50, n_best + 100, 600, 700, 900, 1000, 1100]))
    n_options = [n for n in n_options if 300 <= n <= 1200 and n != n_best]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.15, 60.0)
        a_opt = mirror_descent(a_try, time_budget=alloc * 0.6, tau0=1.0 / n_try, tau_decay=0.9995, eta0=1.0)

        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_opt = lp_alternating(a_opt, time_budget=min(remaining2 * 0.5, 30.0))

        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 8: Random restarts
    print(f"\n=== Phase 8: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    n_restarts = min(5, int(remaining // 50))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if r % 3 == 0:
            a_r = random_init(n_best, rng)
        elif r % 3 == 1:
            a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            n_r = max(300, min(1200, n_best + rng.integers(-200, 200)))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, n_r))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        alloc = min(remaining * 0.25, 80.0)
        a_opt = mirror_descent(a_r, time_budget=alloc * 0.5, tau0=1.0 / len(a_r), tau_decay=0.999, eta0=1.0)
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_opt = lp_alternating(a_opt, time_budget=min(remaining2 * 0.6, 30.0))

        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # Phase 9: Final intensive polish
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    if remaining > 30:
        # Mirror descent with fine params
        alloc = min(remaining * 0.3, 60.0)
        a_p1 = mirror_descent(best_seq, time_budget=alloc, tau0=0.5 / n_final, tau_decay=0.9999, eta0=0.5)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Polish MD: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.3, 40.0)
            a_p2 = lp_alternating(best_seq, time_budget=alloc)
            obj_p2 = compute_obj(a_p2)
            if obj_p2 < best_score:
                best_score = obj_p2
                best_seq = a_p2.copy()
                print(f"  Polish LP: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.3, 30.0)
            a_p3 = chase_peak_refine(best_seq, time_budget=alloc)
            obj_p3 = compute_obj(a_p3)
            if obj_p3 < best_score:
                best_score = obj_p3
                best_seq = a_p3.copy()
                print(f"  Polish Chase: obj={best_score:.8f}")

        # One more mirror descent round
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.4, 50.0)
            a_p4 = mirror_descent(best_seq, time_budget=alloc, tau0=0.3 / n_final, tau_decay=0.99995, eta0=0.3)
            obj_p4 = compute_obj(a_p4)
            if obj_p4 < best_score:
                best_score = obj_p4
                best_seq = a_p4.copy()
                print(f"  Final MD: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
