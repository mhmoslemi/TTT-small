```python
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smooth_max_grad(conv, beta):
    """Compute softmax weights and gradient of smooth max of conv w.r.t. a."""
    n = len(conv)
    # Numerical stability
    conv_shifted = conv - conv.max()
    w = np.exp(beta * conv_shifted)
    w /= w.sum()
    return w
```

```python
def solve_minpeak_lp(a):
    """Solve min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Variable: x = [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A@g - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_val = g_val / s
            M_val = float(res.x[n])
            return g_val, M_val
    return None, None
```

```python
    n = len(a)

        return a

        conv_max = conv.max()
        w = np.exp(beta * (conv - conv_max))
        w /= w.sum()
        
        # Gradient: grad[i] = sum_k w[k] * a[k-i] * 1(0<=k-i<n)
        grad = np.zeros(n)
        for i in range(n):
            k_start = i  # k >= i (so k-i >= 0)
            k_end = min(2*n-2, i+n-1)  # k-i <= n-1
            if k_start <= k_end:
                grad[i] = np.dot(w[k_start:k_end+1], a[0:k_end-k_start+1])
        
        # Center the gradient (to maintain sum=1 approximately)
        grad -= np.mean(grad)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
            a = a_new
        else:
            a = best_a.copy()
    
    return best_a
```

```python
"""
Optimized search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast LP alternating (scipy HiGHS), smooth gradient descent with
softmax max, multi-n exploration, and simulated annealing with
structured moves. Designed to beat 1.5027 target.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A@g = conv(a,g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[:]
    return A


def solve_minpeak_lp(a):
    """Solve min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    return None, None


def smooth_grad(a, w):
    """Compute grad[i] = sum_j w[i+j]*a[j] for valid j."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j_max = min(n, 2 * n - 1 - i)
        if j_max > 0:
            grad[i] = np.dot(w[i:i + j_max], a[:j_max])
    return grad


def smooth_gradient_descent(a, time_budget=60.0, beta_start=5.0, beta_end=200.0):
    """Smooth gradient descent with annealed beta and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    momentum = 0.85
    lr = 0.02
    total_iters = 0

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        # Anneal beta
        progress = min(1.0, it / 5000.0)
        beta = beta_start + (beta_end - beta_start) * progress

        conv = compute_conv(a)
        conv_max = conv.max()
        w = np.exp(beta * (conv - conv_max))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        grad = smooth_grad(a, w)
        grad -= np.mean(grad)

        v = momentum * v + grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            total_iters += 1
            if total_iters % 100 == 0:
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
            a = a_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            lr *= 0.5

    return best_a

    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    count = 0

    max_iters = 100000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:

        g_lp, M_lp = solve_minpeak_lp(a)
        if g_lp is None:

        # Try multiple blend factors

        best_t_val = 0.5
        best_a_local = best_a.copy()

def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def simulated_annealing(a, time_budget=60.0, n_blocks=10):
    """Simulated annealing with block moves."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    temp = 0.01
    rng = np.random.default_rng(12345)
    block_size = max(1, n // n_blocks)

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Pick a random block
        b_start = rng.integers(0, max(1, n - block_size))
        b_end = min(n, b_start + block_size)

        a_trial = a.copy()
        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Add noise to block
            a_trial[b_start:b_end] += rng.normal(0, temp, b_end - b_start)
        elif move_type == 1:
            # Scale block
            a_trial[b_start:b_end] *= rng.uniform(0.5, 2.0)
        elif move_type == 2:
            # Swap with another block
            b2_start = rng.integers(0, max(1, n - block_size))
            b2_end = min(n, b2_start + block_size)
            tmp = a_trial[b_start:b_end].copy()
            a_trial[b_start:b_end] = a_trial[b2_start:b2_end]
            a_trial[b2_start:b2_end] = tmp
        else:
            # Flatten block toward mean
            blk_mean = np.mean(a_trial[b_start:b_end])
            a_trial[b_start:b_end] = (1 - 0.3) * a_trial[b_start:b_end] + 0.3 * blk_mean

        a_trial = np.maximum(a_trial, 0)
        s_trial = float(np.sum(a_trial))
        if s_trial < 1e-10:
            temp *= 0.99
            continue
        a_trial = a_trial / s_trial

        obj_trial = compute_obj(a_trial)
        delta = obj_trial - best_obj
        if delta < 0 or rng.random() < np.exp(-delta / (temp + 1e-10)):
            a = a_trial
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a = a_trial.copy()
        # Cool down
        temp *= 0.9995

    return best_a


def coordinate_ascent(a, time_budget=30.0):
    """Pairwise coordinate swap to reduce peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from positions that contribute to peak
        step = max(1, n // 100)
        for i in range(0, n, step):
            j = k_star - i
            if 0 <= j < n and i != j and a[i] > 1e-10:
                for frac in [0.01, 0.05, 0.1]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break
        if not improved:
            break

    return best_a


def gen_structured_init(n, rng, mode=0):
    """Generate structured initial sequences."""
    x = np.arange(n) / max(1, n - 1)
    if mode == 0:
        # U-shaped
        a = 1.0 + 2.0 * (x - 0.5) ** 2
    elif mode == 1:
        # Cosine bump
        a = np.cos(np.pi * (x - 0.5)) ** 2 + 0.3
    elif mode == 2:
        # Bimodal
        a = np.exp(-10 * (x - 0.2) ** 2) + np.exp(-10 * (x - 0.8) ** 2) + 0.2
    elif mode == 3:
        # Random smooth
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * x + phase)
    elif mode == 4:
        # Inverted parabola with noise
        a = 1.0 - (x - 0.5) ** 2 + 0.1 * rng.normal(0, 1, n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating + smooth gradient descent + multi-n exploration.
    Targets score below 1.5027.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = gen_structured_init(n_init, rng, 0)
        best_score = compute_obj(best_seq)
        print(f"Init (n={n_init}): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"\n=== Phase 1: LP Alternating (n={n_parent}) ===")
    phase1_budget = min(400.0, (deadline - start_time) * 0.35)
    a_work = best_seq.copy()

    # Run LP alternating in chunks
    chunk_time = 60.0
    chunks = 0
    while time.monotonic() < start_time + 40.0 + phase1_budget:
        alloc = min(chunk_time, deadline - time.monotonic())
        if alloc < 5:
            break
        a_new = lp_alternating(a_work, time_budget=alloc, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-10:
            best_score = obj_new
            best_seq = a_new.copy()
            a_work = a_new.copy()
            print(f"  LP chunk {chunks}: obj={best_score:.8f}")
        else:
            print(f"  LP chunk {chunks}: obj={obj_new:.8f} (no improvement)")
        chunks += 1

    # Phase 2: Smooth gradient descent
    print(f"\n=== Phase 2: Smooth Gradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(200.0, remaining * 0.2)
        a_sgd = smooth_gradient_descent(best_seq, time_budget=alloc)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score - 1e-10:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  SGD: obj={best_score:.8f}")
        else:
            print(f"  SGD: obj={obj_sgd:.8f}")

        # Another run with different beta schedule
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(100.0, remaining * 0.15)
            a_sgd2 = smooth_gradient_descent(best_seq, time_budget=alloc,
                                             beta_start=50.0, beta_end=500.0)
            obj_sgd2 = compute_obj(a_sgd2)
            if obj_sgd2 < best_score - 1e-10:
                best_score = obj_sgd2
                best_seq = a_sgd2.copy()
                print(f"  SGD2: obj={best_score:.8f}")

    # Phase 3: Multi-n exploration
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_options = sorted(set([
        max(200, n_parent - 200), max(200, n_parent - 100),
        n_parent,
        n_parent + 100, n_parent + 200, n_parent + 400,
        500, 700, 900, 1100, 1500, 2000
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        # Add slight perturbation
        a_try = a_try * np.exp(rng.normal(0, 0.02, n_try))
        a_try = np.maximum(a_try, 0)
        a_try = a_try / np.sum(a_try)

        alloc = min(40.0, remaining * 0.15)
        a_opt = lp_alternating(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score - 1e-10:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 4: Intensive LP polish on best
    print(f"\n=== Phase 4: Intensive LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_work = best_seq.copy()
        alloc = min(300.0, remaining * 0.4)
        a_new = lp_alternating(a_work, time_budget=alloc, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-10:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Polish: obj={best_score:.8f}")
        else:
            print(f"  Polish: obj={obj_new:.8f}")

    # Phase 5: Simulated annealing
    print(f"\n=== Phase 5: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(80.0, remaining * 0.15)
        a_sa = simulated_annealing(best_seq, time_budget=alloc)
        obj_sa = compute_obj(a_sa)
        if obj_sa < best_score - 1e-10:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA: obj={best_score:.8f}")
        else:
            print(f"  SA: obj={obj_sa:.8f}")

    # Phase 6: Coordinate refinement
    print(f"\n=== Phase 6: Coordinate Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(50.0, remaining * 0.1)
        a_cd = coordinate_ascent(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-10:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f}")

    # Phase 7: Final LP squeeze
    print(f"\n=== Phase 7: Final Squeeze ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(60.0, remaining * 0.5)
        a_final = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-10:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final: obj={best_score:.8f}")

    # Validate
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
```
