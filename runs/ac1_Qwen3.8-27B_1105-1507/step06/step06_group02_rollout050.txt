**Additional idea: Use the LP to directly certify optimality**

The LP dual gives us:
minimize M * sum(y)
subject to A^T y >= 1
y >= 0

At the optimum, if we can find y* such that A^T y* = 1 (all constraints tight) and y* >= 0, then the dual objective is M * sum(y*). If this equals the primal objective (sum(g*)), we have strong duality and can certify the solution.

More importantly, the dual variables y* tell us which constraints are "important" (active). If y*_k > 0, then constraint k is active, meaning (a*a)_k = M at the optimum.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where (a*g)(k) = (A g)_k for k=0,...,2n-2."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, M, n, A_cached=None):
    """Solve the LP: max 1^T g s.t. A g <= M, g >= 0"""
    if A_cached is None:
        A = build_A_sparse(n, a)
    else:
        A = A_cached
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(history_x, history_Fx, m, clip_range=(-0.5, 1.5)):
    """Anderson mixing: find alpha minimizing ||sum alpha_i * r_i||^2 s.t. sum alpha = 1."""
    if m < 2:
        return history_Fx[-1].copy()
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T + 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, clip_range[0], clip_range[1])
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(history_Fx[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def lp_mirror_descent(a_init, deadline, max_iters=5000, anderson_window=10,
                      t_init=0.4, t_min=0.005):
    """LP mirror descent with Anderson acceleration and adaptive step size."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Build A once (it depends on a, so we need to update it)
    # Actually A depends on a, so we rebuild each time
    # But for speed, we can rebuild less frequently
    
    hist_x = []
    hist_Fx = []
    no_imp = 0
    t = t_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Build A matrix
        A = build_A_sparse(n, a)
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                # Try a perturbation
                pert = a + 0.01 * np.random.randn(n)
                pert = np.maximum(pert, 1e-10)
                pert = pert / pert.sum()
                M_pert = fast_conv(pert).max()
                if M_pert < best_M:
                    a = pert
                    best_a = a.copy()
                    best_M = M_pert
                    no_imp = 0
                else:
                    if no_imp > 100:
                        break
        
        # Solve LP
        g = lp_solve(a, M, n, A)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Step
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        M_new = fast_conv(a_new).max()
        
        # Adaptive step: if no improvement, reduce t
        if M_new > M * (1 + 1e-10):
            t = max(t_min, t * 0.7)
        elif M_new < M * (1 - 1e-10):
            t = min(0.6, t * 1.1)
        
        # Anderson mixing
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and = a_and / sa
                M_and = fast_conv(a_and).max()
                if M_and <= M_new * 1.01:
                    a = a_and
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def gs_flat_spectrum(a_init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton for flat spectrum."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-10)
        # Power law compression
        p = 0.3 + 0.4 * beta
        mag_alt = (mag ** p) / ((mag ** p).max() + 1e-10) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 1e-10)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    if parent is not None:

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with aggressive Anderson acceleration (window=15)
2. Adaptive step size with backtracking/forward acceptance

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, M, n, A_mat):
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(hist_x, hist_Fx, m, low=-1.0, high=2.0):
    if m < 2:
        return hist_Fx[-1].copy()
    residuals = np.array([hist_x[i] - hist_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, low, high)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(hist_Fx[0]))
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    return np.maximum(mixed, 0)


def lp_mirror_descent(a_init, deadline, max_iters=10000, anderson_window=15,
                      t_init=0.5, t_min=0.003, perturb_every=40):
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    hist_x = []
    hist_Fx = []
    no_imp = 0
    t = t_init
    rng_local = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = build_A_sparse(n, a)
        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp == perturb_every:
                pert = a + 0.005 * rng_local.randn(n)
                pert = np.maximum(pert, 1e-10)
                pert = pert / pert.sum()
                M_p = fast_conv(pert).max()
                if M_p < best_M * 0.999:
                    a = pert
                    no_imp = 0
            elif no_imp > perturb_every * 3:
                break

        g = lp_solve(a, M, n, A)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_conv(a_new).max()

        if M_new > M * (1 + 0.5e-6):
            t = max(t_min, t * 0.6)
        elif M_new < M * (1 - 0.5e-6):
            t = min(0.7, t * 1.15)

        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)

        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and = a_and / sa
                M_and = fast_conv(a_and).max()
                if M_and <= min(M_new, M) * 1.005:
                    a = a_and
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new

    return best_a


def gs_flat_spectrum(a_init, n_iters=200, beta=0.87):
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-10)
        p = 0.3 + 0.5 * beta
        mag_alt = (mag ** p) / ((mag ** p).max() + 1e-10) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 1e-10)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_max_descent(a_init, deadline, max_iters=3000, tau=50.0, lr=0.15):
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta_mom = 0.9

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = float(b.max())
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        grad = np.zeros(n)
        active = np.where(p > 1e-12)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * p[k] * seg

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_mom**2)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta_mom * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new

        M_cur = fast_conv(a).max()
        if M_cur < best_M * (1 - 1e-12):
            best_M = M_cur
            best_a = a.copy()

    return best_a
