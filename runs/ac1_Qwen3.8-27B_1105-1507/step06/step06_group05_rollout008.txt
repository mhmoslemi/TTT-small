"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. New LP formulation that directly minimizes the convolution peak
2. Multiplicative weights (mirror descent) optimization on the simplex
3. Parameterized initializations (U-shape, Gaussian, cosine families)
4. Adaptive strategy switching with progress monitoring
5. Multi-scale construction (optimize small n, upscale, optimize large n)

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz_conv(a):
    """Build matrix A where (A @ g)[k] = conv(a,g)[k] for k=0..2n-2."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_lo = j
        k_hi = min(j + n, m)
        A[k_lo:k_hi, j] = a[:k_hi - k_lo]
    return A


def lp_min_peak(a, time_limit=20):
    """
    LP: minimize t s.t. conv(a,g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g_optimal, t_optimal) or (None, None).
    """
    n = len(a)
    A = build_toeplitz_conv(a)
    m = 2 * n - 1
    # Variables: g_0..g_{n-1}, t
    # Minimize t: c = [0,0,...,0,1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    # Constraints: conv(a,g)_k - t <= 0 for all k
    # A_ub @ [g,t] <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # Bounds: g_j >= 0, t free
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            t_val = res.x[n]
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
            return g, t_val
    except Exception:
        pass
    return None, None


def lp_max_sum(a, M, time_limit=20):
    """
    LP: maximize sum(g) s.t. conv(a,g)_k <= M for all k, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    A = build_toeplitz_conv(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate_new(b, max_iters=40, time_budget=80.0):
    """
    LP alternating with new 'minimize peak' formulation.
    b is normalized (sum=1).
    """
    n = len(b)
    b = normalize(b.copy().astype(np.float64))
    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # New LP: minimize peak of conv(b, g) s.t. sum(g)=1, g>=0
        tl = max(3, min(int(remaining * 0.5), 25))
        g, t_val = lp_min_peak(b, time_limit=tl)

        if g is None:
            # Fallback to old formulation
            conv = compute_conv(b)
            M = float(np.max(conv))
            g = lp_max_sum(b, M, time_limit=tl)
            if g is None:
                break

        # Line search on alpha in [0, 1]: b_new = (1-alpha)*b + alpha*g
        best_alpha = 0.0
        best_trial = best_obj
        for alpha in [0.3, 0.5, 0.7, 0.85, 1.0, 0.15, 0.4, 0.6, 0.9]:
            b_try = normalize((1.0 - alpha) * b + alpha * g)
            obj_try = compute_obj(b_try)
            if obj_try < best_trial:
                best_trial = obj_try
                best_alpha = alpha

        b_new = normalize((1.0 - best_alpha) * b + best_alpha * g)
        obj_new = compute_obj(b_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_b = b_new.copy()
            b = b_new
        else:
            # Perturb to escape local minimum
            rng = np.random.default_rng((it * 6271 + 42) % (2**31))
            b = normalize(best_b * np.exp(rng.normal(0, 0.03 + 0.01 * (it % 4), n)))

    return best_b


def multiplicative_weights(b, max_steps=300, time_budget=50.0, tau_init=0.1):
    """
    Mirror descent with multiplicative weights on the simplex.
    b_new_j = b_j * exp(-eta * v_j) / Z
    where v_j is the subgradient of max_k (b*b)_k.
    Uses soft-max weighting for smoother gradient.
    """
    n = len(b)
    b = normalize(b.copy().astype(np.float64))
    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()
    eta = 0.05
    tau = tau_init

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(b)
        cmax = float(np.max(conv))

        # Soft-max weights over peaks
        tau = tau_init * np.exp(-step / 50.0) + 0.01 * cmax / n
        shifted = np.clip((conv - cmax) / max(tau, 1e-10), -300.0, 300.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute subgradient: v_j = sum_k weights[k] * b[k-j] (where valid)
        # v_j = (weights * b_flipped) correlation, extracted appropriately
        # conv_b_w[j] = sum_k weights[k] * b[k-j]... 
        # Actually: v_j = sum_k w_k * b_{k-j} for valid indices
        # This is the correlation of weights with b
        # Let's compute it directly: v[j] = sum_k w[k] * b[k-j] where 0<=k-j<n
        # Equivalently: v = ifft(fft(weights_extended) * conj(fft(b_extended)))... 
        # Simpler: use the fact that v[j] contributes to conv at positions k where b[k-j] is used
        # v[j] = sum_{k: 0<=k-j<n and 0<=k<2n-1} w[k] * b[k-j]
        # Let l = k-j, so k = j+l, and we need 0<=j+l<2n-1 and 0<=l<n
        # v[j] = sum_{l=0}^{n-1} w[j+l] * b[l] (where j+l < 2n-1)
        # This is the cross-correlation!
        size = 1
        while size < 3 * n - 1:
            size <<= 1
        w_ext = np.zeros(2 * n - 1)
        w_ext[:] = weights
        Wf = rfft(w_ext, size)
        Bf = rfft(b, size)
        corr = irfft(Wf * Bf, size)
        # v[j] = sum_l w_ext[j+l] * b[l] = corr at appropriate index
        # Actually corr[m] = sum_l w_ext[l] * b[m-l]... let me just do it directly for clarity
        # v[j] = sum_{l=0}^{n-1} min(w[j+l], valid) * b[l]
        # Use the correlation result: irfft(fft(w) * conj(fft(b))) gives sum w[k] * b[k-j] shifted
        # Let me just compute v directly:
        v = np.zeros(n)
        for j in range(n):
            # v[j] = sum_{l=0}^{n-1} w[j+l] * b[l] where j+l < 2n-1
            l_max = min(n - 1, 2 * n - 2 - j)
            if l_max >= 0:
                v[j] = np.dot(w_ext[j:j + l_max + 1], b[:l_max + 1])

        # Multiplicative update
        log_weights = np.log(np.maximum(b, 1e-300)) - eta * v
        log_weights -= np.max(log_weights)  # numerical stability
        new_b = np.exp(log_weights)
        s = float(np.sum(new_b))
        if s < 1e-300:
            break
        new_b = new_b / s

        obj_new = compute_obj(new_b)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_b = new_b.copy()
            b = new_b
            eta *= 1.02  # slight increase
        else:
            eta *= 0.7
            if eta < 1e-6:
                break
            # Small perturbation
            rng = np.random.default_rng(step * 31 + 7)
            b = normalize(best_b * np.exp(rng.normal(0, 0.02, n)))

    return best_b


def gradient_descent_soft(b, max_steps=300, time_budget=50.0, lr_init=0.1):
    """
    Gradient descent on the simplex using soft-max peak weighting.
    Uses centering to maintain sum constraint, then clips and renormalizes.
    """
    n = len(b)
    b = normalize(b.copy().astype(np.float64))
    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()
    lr = lr_init
    tau = 0.05

    size = 1
    while size < 3 * n - 1:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(b)
        cmax = float(np.max(conv))

        # Adaptive temperature
        tau = 0.05 * cmax * (1.0 + 50.0 / (step + 1))
        shifted = np.clip((conv - cmax) / max(tau, 1e-12), -300.0, 300.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum > 1e-300:
            weights /= w_sum
        else:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0

        # Gradient: v[j] = sum_l w[j+l] * b[l]
        v = np.zeros(n)
        for j in range(min(n, 200)):  # Optimization: only compute for active region if possible
            l_max = min(n - 1, 2 * n - 2 - j)
            if l_max >= 0:
                v[j] = np.dot(weights[j:j + l_max + 1], b[:l_max + 1])
        # For efficiency, compute all using FFT
        w_ext = np.zeros(2 * n - 1)
        w_ext[:] = weights
        Wf = rfft(w_ext, size)
        Bf = rfft(b, size)
        corr = irfft(Wf * Bf, size)
        # v[j] should be sum_l w[j+l]*b[l]; corr gives sum_k w[k]*b[k-j+j]... 
        # Let me verify: irfft(fft(w) * fft(b)) = circular conv of w and b (padded)
        # lin_conv[m] = sum_k w[k] * b[m-k]
        # We want v[j] = sum_l w[j+l] * b[l] = sum_k w[k] * b[k-j] (with k=j+l)
        # = (w * b_flipped convolved) ... this is getting confusing. Let me just do direct.
        v = np.zeros(n)
        for j in range(n):
            l_max = min(n - 1, 2 * n - 2 - j)
            if l_max >= 0:
                v[j] = float(np.dot(w_ext[j:j + l_max + 1], b[:l_max + 1]))

        # Center the gradient (remove mean to preserve sum approximately)
        v_centered = v - np.mean(v)

        # Update
        b_new = b - lr * v_centered
        b_new = np.maximum(b_new, 0.0)
        s = float(np.sum(b_new))
        if s < 1e-10:
            break
        b_new = b_new / s

        obj_new = compute_obj(b_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_b = b_new.copy()
            b = b_new
        else:
            lr *= 0.7
            if lr < 1e-8:
                break

    return best_b


def spectral_shaping(b, max_iters=80, time_budget=40.0):
    """
    Frequency-domain power spectrum flattening.
    """
    n = len(b)
    b = normalize(b.copy().astype(np.float64))
    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        B = fft(b, size)
        P = np.abs(B) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        alpha = 0.3 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(B)
        B_new = np.sqrt(P_new) * np.exp(1j * phase)
        b_new = np.real(ifft(B_new, size))[:n]
        b_new = np.maximum(b_new, 0.0)
        s = float(np.sum(b_new))
        if s < 1e-10:
            break
        b_new = b_new / s

        obj_new = compute_obj(b_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_b = b_new.copy()
            b = b_new
        else:
            b = normalize(0.6 * b + 0.4 * b_new)

    return best_b


def coordinate_descent_polish(b, time_budget=25.0):
    """
    Targeted mass redistribution: move mass away from peak contributors.
    """
    n = len(b)
    b = normalize(b.copy().astype(np.float64))
    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(b)
        k_star = int(np.argmax(conv))

        # Find top contributors to peak
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((b[j] * b[k2], j, k2))
        contrib.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            for frac in [0.02, 0.05, 0.1, 0.15]:
                delta = frac * b[j1]
                b_try = b.copy()
                b_try[j1] = max(0, b_try[j1] - delta)
                # Distribute to random positions far from both
                targets = rng.choice(n, size=2, replace=False)
                for tgt in targets:
                    if tgt != j1 and tgt != j2:
                        b_try[tgt] += delta / 2
                b_try = normalize(b_try)
                obj_try = compute_obj(b_try)

                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_b = b_try.copy()
                    b = b_try
                    improved = True
                    break

        if not improved:
            # Random perturbation
            noise = rng.normal(0, 0.02, n)
            b = normalize(b * np.exp(noise))

    return best_b


def parameterized_init(n, family='u_p2'):
    """Generate parameterized initial sequences."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    if family == 'uniform':
        a = np.ones(n)
    elif family == 'u_p1':
        xn = j / center - 1.0
        a = 1.0 + 0.5 * xn ** 2
    elif family == 'u_p2':
        xn = j / center - 1.0
        a = 1.0 + 1.0 * xn ** 2
    elif family == 'u_p3':
        xn = j / center - 1.0
        a = 0.5 + 2.0 * xn ** 2
    elif family == 'gauss_inv':
        # Inverse Gaussian (peaked at edges)
        a = 1.0 / (0.1 + (xn ** 2) * 0.5)
        a = a / max(a)
        a = 1.0 + a  # Add baseline
    elif family == 'cosine':
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * j / n)
    elif family == 'cosine2':
        a = 1.0 + 0.5 * np.cos(4.0 * np.pi * j / n)
    elif family == 'v_shape':
        a = np.abs(xn) + 0.1
    elif family == 'bathtub':
        a = 0.3 + 0.7 * (xn ** 2)
        a = a ** 1.5
    elif family == 'edge_conc':
        decay = 0.05
        a = np.exp(-decay * np.abs(j - n // 2))
        a = a / max(a)
        a = 0.2 + 0.8 * a
    else:
        a = np.ones(n)

    return normalize(a)


def resample_sequence(a, new_n):
    """Resample via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def multi_scale_optimize(n_target, budget_s=120.0):
    """
    Multi-scale: start small, optimize, upscale, optimize.
    """
    scales = []
    s = max(50, n_target // 16)
    while s < n_target:
        scales.append(s)
        s = min(s * 2, n_target)
    scales.append(n_target)

    a = parameterized_init(scales[0], 'u_p2')
    elapsed = 0.0
    start = time.monotonic()

    for i, n_i in enumerate(scales):
        if i > 0:
            a = resample_sequence(a, n_i)
        remaining = budget_s - elapsed
        if remaining < 5:
            break
        alloc = remaining * 0.6

        # Optimize at this scale
        if n_i <= 500:
            a = lp_alternate_new(a, max_iters=10, time_budget=alloc * 0.5)
        a = multiplicative_weights(a, max_steps=50, time_budget=alloc * 0.3)
        a = gradient_descent_soft(a, max_steps=50, time_budget=alloc * 0.2)

        elapsed = time.monotonic() - start
        obj = compute_obj(a)
        print(f"  Multi-scale n={n_i}: obj={obj:.6f}, elapsed={elapsed:.1f}s")

    return a

    a = normalize(best_seq.copy())

    if time.monotonic() < deadline - 250:
        a_mw = multiplicative_weights(a, max_steps=200, time_budget=45.0, tau_init=0.1)
        obj_mw = compute_obj(a_mw)
        if obj_mw < best_score:
            best_score = obj_mw
            best_seq = a_mw.copy()
            a = a_mw
            print(f"  MultWeights: {best_score:.8f} **")
        else:
            print(f"  MultWeights: {obj_mw:.8f}")

    if time.monotonic() < deadline - 200:
        a_gd = gradient_descent_soft(a, max_steps=200, time_budget=40.0)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            a = a_gd
            print(f"  GradientDesc: {best_score:.8f} **")
        else:
            print(f"  GradientDesc: {obj_gd:.8f}")

    # Follow-up LP
    if time.monotonic() < deadline - 180:
        a_new = lp_alternate_new(a, max_iters=20, time_budget=50.0)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  Post-Phase3 LP: {best_score:.8f} **")
        else:
            print(f"  Post-Phase3 LP: {obj_new:.8f}")

    # =========================================================
    # Phase 4: Multi-scale construction
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-scale construction")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 200:
        a_ms = multi_scale_optimize(n_parent, budget_s=80.0)
        obj_ms = compute_obj(a_ms)
        if obj_ms < best_score:
            best_score = obj_ms
            best_seq = a_ms.copy()
            print(f"  MultiScale: {best_score:.8f} **")
        else:
            print(f"  MultiScale: {obj_ms:.8f}")

    # =========================================================
    # Phase 5: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq)
    n_options = []
    for delta in [-200, -150, -100, -75, -50, -25, 0, 25, 50, 75, 100, 150, 200, 300, 400]:
        n_try = n_best + delta
        if 200 <= n_try <= 2000:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if n_try == n_best:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample_sequence(best_seq, n_try))

        alloc = min(remaining * 0.1, 30.0)
        if n_try <= 1000:
            a_try = lp_alternate_new(a_try, max_iters=8, time_budget=alloc * 0.5)
        a_try = multiplicative_weights(a_try, max_steps=40, time_budget=alloc * 0.3)
        a_try = gradient_descent_soft(a_try, max_steps=40, time_budget=alloc * 0.2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 6: Random restarts with diverse strategies
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Random restarts")
    print(f"{'='*60}")

    n_cur = len(best_seq)
    rng = np.random.default_rng(seed)

    for restart in range(10):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.1, 30.0)

        # Diverse perturbation strategies
        r = restart % 5
        if r == 0:
            # Gaussian multiplier
            a_try = best_seq * np.exp(rng.normal(0, 0.06, n_cur))
        elif r == 1:
            # Sinusoidal modulation
            freq = rng.uniform(1.0, 8.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.2 * np.sin(2.0 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
            a_try = best_seq * mod
        elif r == 2:
            # Blend with Dirichlet
            a_rand = rng.dirichlet(np.ones(n_cur) * 0.3)
            mix = rng.uniform(0.1, 0.4)
            a_try = (1 - mix) * normalize(best_seq) + mix * a_rand
        elif r == 3:
            # Block reversal
            a_try = best_seq.copy()
            block_len = int(rng.integers(n_cur // 5, n_cur // 3))
            start_idx = int(rng.integers(0, n_cur - block_len))
            a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]
            a_try = normalize(a_try)
        else:
            # Shift + smooth
            shift = int(rng.integers(-n_cur // 10, n_cur // 10 + 1))
            a_try = np.roll(best_seq, shift)
            # Smooth with small kernel
            kernel = np.ones(5) / 5.0
            a_try = np.convolve(a_try, kernel, mode='same')
            a_try = normalize(a_try)

        a_try = normalize(a_try)

        if n_cur <= 1000 and alloc > 10:
            a_try = lp_alternate_new(a_try, max_iters=8, time_budget=alloc * 0.4)
        a_try = multiplicative_weights(a_try, max_steps=40, time_budget=alloc * 0.3)
        a_try = gradient_descent_soft(a_try, max_steps=30, time_budget=alloc * 0.3)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = len(best_seq)
            print(f"  Restart {restart}: {best_score:.8f} **")
        else:
            print(f"  Restart {restart}: {obj_try:.8f}")

    # =========================================================
    # Phase 7: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")

    n_final = len(best_seq)
    a_final = normalize(best_seq.copy())

    for polish_round in range(5):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * 0.22, 30.0)

        # LP
        if n_final <= 1000 and alloc > 15:
            a_pol = lp_alternate_new(a_final, max_iters=12, time_budget=alloc * 0.4)
        else:
            a_pol = a_final.copy()

        # Multiplicative weights
        a_pol = multiplicative_weights(a_pol, max_steps=60, time_budget=alloc * 0.25, tau_init=0.05)

        # Gradient descent
        a_pol = gradient_descent_soft(a_pol, max_steps=60, time_budget=alloc * 0.2, lr_init=0.05)

        # Coordinate descent
        if alloc > 8:
            a_pol = coordinate_descent_polish(a_pol, time_budget=min(8, alloc * 0.15))

        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            a_final = a_pol
            print(f"  Polish {polish_round}: {best_score:.8f} **")
        else:
            a_final = a_pol
            print(f"  Polish {polish_round}: {obj_pol:.8f}")

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
