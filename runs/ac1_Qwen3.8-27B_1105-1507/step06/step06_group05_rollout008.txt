    Uses log-sum-exp for smooth max, with backtracking line search.

def kkt_fixed_point(a, max_iter=200, time_budget=40.0):
    """
    KKT-inspired fixed-point iteration.
    At optimum: sum_k mu_k * 2*a[k-i] = lambda for all i in support(a).
    We iterate: compute weights on peaks, compute 'pressure', update a.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Find peak set with soft threshold
        threshold = cmax * 0.999
        peak_mask = conv >= threshold
        n_peaks = int(np.sum(peak_mask))
        if n_peaks == 0:
            peak_mask[int(np.argmax(conv))] = True
            n_peaks = 1

        mu = peak_mask.astype(np.float64) / n_peaks

        # Compute pressure: h[i] = sum_k mu[k] * 2*a[k-i]
        h = 2.0 * np.correlate(mu, a, mode='valid')

        # Lambda: mean pressure on support
        support = a > 1e-10
        if not np.any(support):
            break
        lam = float(np.mean(h[support]))

        # Update: increase where pressure < lambda, decrease where > lambda
        diff = lam - h
        diff[~support] = np.minimum(diff[~support], 0)  # Don't increase outside support

        # Adaptive step size
        eta = max(0.01, 0.5 / (it + 1))
        a_try = a + eta * diff
        a_try = np.maximum(a_try, 0.0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            break
        a_try = a_try / s_try

        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_a = a_try.copy()
            a = a_try
        else:
            # Blend for stability
            a_try = 0.7 * a + 0.3 * a_try
            a_try = normalize(a_try)
            a = a_try

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternation(a, max_iters=40, time_budget=90.0):
    """LP alternating with adaptive blending."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        lp_time = max(3, min(15, int(remaining * 0.5)))
        g = solve_lp(a, M, time_limit=lp_time)
        if g is None:
            break

        # Try several blend factors and pick best
        best_t = 0.0
        best_obj_t = best_obj
        for t in [0.3, 0.5, 0.7, 0.85, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t

        a_new = normalize((1.0 - best_t) * a + best_t * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Small perturbation to escape stagnation
            rng = np.random.default_rng((it * 6151 + 7) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def spectral_init(n, rng):
    """Generate initial sequence with flat-ish power spectrum."""
    # Random phases, flat magnitude
    target_power = 1.0 / n
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.sqrt(target_power) * np.exp(1j * phases)
    # Enforce Hermitian symmetry for real signal
    if n % 2 == 0:
        A[:n//2+1] = np.sqrt(target_power) * np.exp(1j * rng.uniform(0, 2*np.pi, n//2+1))
        A[n//2+1:] = np.conj(A[n//2:0:-1])
    else:
        A[:n//2+1] = np.sqrt(target_power) * np.exp(1j * rng.uniform(0, 2*np.pi, n//2+1))
        A[n//2+1:] = np.conj(A[n//2:0:-1])
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining smooth gradient,
    LP alternation, KKT fixed-point, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # =========================================================
    # Phase 1: Smooth gradient descent from parent
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Smooth gradient descent (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a1 = normalize(best_seq.copy())
        a1 = smooth_gradient_descent(a1, tau_start=100.0, tau_end=3000.0,
                                     max_iter=400, time_budget=120.0)
        obj1 = compute_obj(a1)
        print(f"  Smooth gradient: {obj1:.8f}")
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()

    # =========================================================
    # Phase 2: LP alternation (primary optimizer)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP alternation")
    print(f"{'='*60}")

    if best_seq is not None:
        a2 = normalize(best_seq.copy())
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.25, 250.0)

        for cycle in range(3):
            if time.monotonic() > deadline - 300:
                break
            a2 = lp_alternation(a2, max_iters=30, time_budget=alloc / 3.0)
            obj2 = compute_obj(a2)
            print(f"  LP cycle {cycle}: {obj2:.8f}")
            if obj2 < best_score:
                best_score = obj2
                best_seq = a2.copy()

    # =========================================================
    # Phase 3: KKT fixed-point + smooth gradient
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: KKT fixed-point + gradient")
    print(f"{'='*60}")

    if best_seq is not None:
        a3 = normalize(best_seq.copy())
        a3 = kkt_fixed_point(a3, max_iter=200, time_budget=40.0)
        obj3 = compute_obj(a3)
        print(f"  KKT: {obj3:.8f}")
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()

        a3 = smooth_gradient_descent(a3, tau_start=500.0, tau_end=5000.0,
                                     max_iter=300, time_budget=50.0)
        obj3b = compute_obj(a3)
        print(f"  Post-KKT gradient: {obj3b:.8f}")
        if obj3b < best_score:
            best_score = obj3b
            best_seq = a3.copy()

    # =========================================================
    # Phase 4: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700
    n_options = sorted(set([
        n_best - 200, n_best - 150, n_best - 100, n_best - 75,
        n_best - 50, n_best - 25, n_best,
        n_best + 25, n_best + 50, n_best + 75, n_best + 100,
        n_best + 150, n_best + 200, n_best + 300
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1300]

    rng_explore = np.random.default_rng(seed + 100)

    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if best_seq is not None:
            if n_try == n_best:
                a_try = normalize(best_seq.copy())
            else:
                a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = spectral_init(n_try, rng_explore)

        # Optimize: gradient then LP
        alloc = min(remaining * 0.15, 40.0)
        a_try = smooth_gradient_descent(a_try, tau_start=200.0, tau_end=5000.0,
                                        max_iter=200, time_budget=alloc * 0.5)
        if n_try <= 900 and alloc > 10:
            a_try = lp_alternation(a_try, max_iters=10, time_budget=alloc * 0.4)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 5: Random restarts with diverse strategies
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Random restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed + 200)

        for restart in range(10):
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.12, 40.0)

            if restart % 5 == 0:
                # Gaussian perturbation
                a_try = best_seq * np.exp(rng.normal(0, 0.1, n_cur))
            elif restart % 5 == 1:
                # Sinusoidal modulation
                freq = rng.uniform(0.3, 4.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
                a_try = best_seq * mod
            elif restart % 5 == 2:
                # Blend with Dirichlet
                a_rand = rng.dirichlet(np.ones(n_cur) * 0.3)
                mix = rng.uniform(0.3, 0.6)
                a_try = (1 - mix) * normalize(best_seq) + mix * a_rand
            elif restart % 5 == 3:
                # Block reversal
                a_try = best_seq.copy()
                block_len = int(rng.integers(n_cur // 3, n_cur // 2 + 1))
                start_idx = int(rng.integers(0, n_cur - block_len))
                a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]
            else:
                # Spectral re-initialization blended with best
                a_spec = spectral_init(n_cur, rng)
                mix = rng.uniform(0.2, 0.4)
                a_try = (1 - mix) * normalize(best_seq) + mix * a_spec

            a_try = normalize(a_try)

            # Optimize
            a_try = smooth_gradient_descent(a_try, tau_start=300.0, tau_end=8000.0,
                                            max_iter=150, time_budget=alloc * 0.5)
            if n_cur <= 850 and alloc > 10:
                a_try = lp_alternation(a_try, max_iters=8, time_budget=alloc * 0.4)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart}: {best_score:.8f} **")
            else:
                print(f"  Restart {restart}: {obj_try:.8f}")

    # =========================================================
    # Phase 6: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        n_final = len(best_seq)
        a_final = normalize(best_seq.copy())

        for polish_round in range(5):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            alloc = min(remaining * 0.2, 40.0)

            # Gradient with very high tau (nearly exact max)
            a_pol = smooth_gradient_descent(a_final, tau_start=2000.0, tau_end=20000.0,
                                            max_iter=200, time_budget=alloc * 0.4)

            # LP for final push
            if n_final <= 900 and alloc > 15:
                a_pol = lp_alternation(a_pol, max_iters=10, time_budget=alloc * 0.4)

            # KKT
            if alloc > 8:
                a_pol = kkt_fixed_point(a_pol, max_iter=100, time_budget=alloc * 0.2)

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish_round}: {best_score:.8f} **")
            else:
                a_final = a_pol
                print(f"  Polish {polish_round}: {obj_pol:.8f}")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
