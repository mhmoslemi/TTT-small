def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = (1.0 + u*rho) > cv
    if not cond.any():
        theta = 0.0
    else:

    w = np.maximum(v - theta, 0.0)

For coordinate descent, the key optimization is: given all coordinates except a[i], find the optimal a[i] to minimize max(conv(a,a)).

This is a 1D convex optimization problem (the max of linear functions of a[i], plus possibly a quadratic term at k=2i).

Let me implement this carefully:

For coordinate i, the affected conv values are:
- b[k] for k in [i, i+n-1] (since changing a[i] affects b[k] = sum_j a[j]*a[k-j] when j=i or k-j=i)

For each affected k:

- b[k] = b_base[k] + a[i]^2 (if k = 2i)

The objective is max(base_max, max over affected k of (b_base[k] + 2*a[i]*a[k-i]), b_base[2i] + a[i]^2)

This is a convex function of a[i] (max of linear and one quadratic function). We can find its minimum using ternary search or by analyzing the breakpoints.

def lp_best_response(a, M, n_active=30, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top n_active k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    
    top_indices = np.argsort(b)[::-1][:min(n_active, 2*n-1)]
    active = top_indices.tolist()
    
    # Build constraint matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None
