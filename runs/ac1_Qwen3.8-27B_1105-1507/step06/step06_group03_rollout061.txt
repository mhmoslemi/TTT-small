"""
Enhanced optimizer using exact coordinate descent with bisection,
adaptive perturbation, and multi-length exploration.
Improves upon previous approach by using exact 1D subproblem solving
instead of fixed candidate sets.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def exact_coord_descent(a, max_sweeps, deadline, seed=42, verbose=False):
    """
    Coordinate descent with exact 1D subproblem solving via bisection.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n) if sweep % 3 == 0 else range(n)
        improved = False

        for i in order:
            old = a[i]
            if old < 1e-14:
                # Try small values
                candidates = [0.0, 1e-5, 1e-4, 1e-3, 0.01]
                best_x = old
                best_j = best_obj
                for x in candidates:
                    delta = x - old
                    lo = max(0, i)
                    hi = min(2 * n - 2, i + n - 1)
                    a_slice = a[lo - i:hi - i + 1].copy()
                    conv[lo:hi + 1] += 2.0 * delta * a_slice
                    if 2 * i <= 2 * n - 2:
                        conv[2 * i] += delta * delta
                    new_s = s + delta
                    if new_s > 1e-10:

                        if new_obj < best_j:

                            best_x = x
                    conv[lo:hi + 1] -= 2.0 * delta * a_slice
                    if 2 * i <= 2 * n - 2:

                if best_x != old:

                    lo = max(0, i)
                    hi = min(2 * n - 2, i + n - 1)
                    a_slice = a[lo - i:hi - i + 1].copy()
                    conv[lo:hi + 1] += 2.0 * delta * a_slice
                    if 2 * i <= 2 * n - 2:
                        conv[2 * i] += delta * delta
                    a[i] = best_x
                    s += delta
                    improved = True
                continue

            # For non-zero coordinates, use bisection on [0, old]
            # and also try extending beyond old
            lo_bound = 0.0
            hi_bound = old * 2.0  # allow some extension
            
            # Define the objective as a function of x
            # We'll use a ternary search or bisection on the derivative
            
            def obj_at_x(x):
                delta = x - old
                lo = max(0, i)
                hi = min(2 * n - 2, i + n - 1)
                a_slice = a[lo - i:hi - i + 1].copy()
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] += delta * delta
                new_s = s + delta
                if new_s < 1e-10:
                    result = float('inf')
                else:
                    result = 2.0 * n * conv.max() / (new_s * new_s)
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] -= delta * delta
                return result

            # Golden section search on [0, hi_bound]
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
            a_g = lo_bound
            b_g = hi_bound
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = obj_at_x(c_g)
            fd = obj_at_x(d_g)
            
            for _ in range(30):  # 30 iterations of golden section
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = obj_at_x(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = obj_at_x(d_g)
            
            x_opt = (a_g + b_g) / 2
            # Also check boundaries
            candidates = [0.0, x_opt, old, hi_bound]
            best_x = old
            best_j = best_obj
            for x in candidates:
                x = max(0.0, min(x, hi_bound))
                j = obj_at_x(x)
                if j < best_j:

                    best_x = x

                hi = min(2 * n - 2, i + n - 1)
                a_slice = a[lo - i:hi - i + 1].copy()

                a[i] = best_x

            if current_obj < best_obj - 1e-13:

                best_a = a.copy()
                no_improve = 0
            else:

        else:

            print(f"    CD sweep {sweep}: obj={best_obj:.8f}", flush=True)

    return best_a, best_obj

"""
Enhanced optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses exact coordinate descent with adaptive candidate generation,
Fourier-domain perturbation for escaping local minima, and multi-length exploration.
Key improvements: (1) golden-section 1D subproblem solving, (2) spectral jolts,
(3) pair-exchange moves, (4) adaptive relaxation of the step candidate set.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent_exact(a, max_sweeps, deadline, seed=42, verbose=False, pair_mode=False):
    """
    Coordinate descent with golden-section 1D subproblem solving.
    Optionally uses pair-exchange moves for better local exploration.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 100

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved = False

        for i in order:
            old = a[i]

            if old < 1e-13:
                # Try activating this coordinate
                candidates = [0.0, 1e-5, 1e-4, 2e-4, 5e-4, 1e-3]
                best_x = 0.0
                best_j = 2.0 * n * conv.max() / (s * s)
                lo_k = max(0, i)
                hi_k = min(2 * n - 2, i + n - 1)
                for x in candidates:
                    delta = x
                    a_slice = a[lo_k - i:hi_k - i + 1].copy()
                    conv[lo_k:hi_k + 1] += 2.0 * delta * a_slice
                    if 2 * i <= 2 * n - 2:
                        conv[2 * i] += delta * delta
                    new_s = s + delta
                    if new_s > 1e-10:
                        new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                        if new_obj < best_j:
                            best_j = new_obj
                            best_x = x
                    conv[lo_k:hi_k + 1] -= 2.0 * delta * a_slice
                    if 2 * i <= 2 * n - 2:
                        conv[2 * i] -= delta * delta
                if best_x > 0:
                    delta = best_x
                    a_slice = a[lo_k - i:hi_k - i + 1].copy()
                    conv[lo_k:hi_k + 1] += 2.0 * delta * a_slice
                    if 2 * i <= 2 * n - 2:
                        conv[2 * i] += delta * delta
                    a[i] = best_x
                    s += delta
                    improved = True
                continue

            lo_k = max(0, i)
            hi_k = min(2 * n - 2, i + n - 1)
            a_slice = a[lo_k - i:hi_k - i + 1].copy()
            has_self = (2 * i <= 2 * n - 2)

            def eval_x(x):
                delta = x - old
                conv[lo_k:hi_k + 1] += 2.0 * delta * a_slice
                if has_self:
                    conv[2 * i] += delta * delta
                new_s = s + delta
                if new_s < 1e-10:
                    result = float('inf')
                else:
                    result = 2.0 * n * conv.max() / (new_s * new_s)
                conv[lo_k:hi_k + 1] -= 2.0 * delta * a_slice
                if has_self:
                    conv[2 * i] -= delta * delta
                return result

            # Coarse candidates
            ratios = [0.0, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 1.0, 1.15, 1.4]
            candidates = [r * old for r in ratios]
            best_x = old
            best_j = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x_val = max(0.0, x)
                if abs(x_val - old) < 1e-14:
                    continue
                j = eval_x(x_val)
                if j < best_j:
                    best_j = j
                    best_x = x_val

            # Refine with golden section around best candidate
            if best_x != old:
                # Determine bracket
                if best_x < old:
                    lo_b = 0.0
                    hi_b = old
                else:
                    lo_b = old
                    hi_b = old * ratios[-1]

                # Golden section in [lo_b, hi_b]
                a_g = lo_b
                b_g = hi_b
                c_g = b_g - gr * (b_g - a_g)
                d_g = a_g + gr * (b_g - a_g)
                fc = eval_x(c_g)
                fd = eval_x(d_g)
                for _ in range(20):
                    if fc < fd:
                        b_g = d_g
                        d_g = c_g
                        fd = fc
                        c_g = b_g - gr * (b_g - a_g)
                        fc = eval_x(c_g)
                    else:
                        a_g = c_g
                        c_g = d_g
                        fc = fd
                        d_g = a_g + gr * (b_g - a_g)
                        fd = eval_x(d_g)
                x_gs = (a_g + b_g) / 2.0
                j_gs = eval_x(x_gs)
                if j_gs < best_j:
                    best_j = j_gs
                    best_x = x_gs

            if abs(best_x - old) > 1e-12:
                delta = best_x - old
                conv[lo_k:hi_k + 1] += 2.0 * delta * a_slice
                if has_self:
                    conv[2 * i] += delta * delta
                a[i] = best_x
                s += delta
                improved = True

        if improved:
            current_obj = 2.0 * n * conv.max() / (s * s)
            if current_obj < best_obj - 1e-13:
                best_obj = current_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 50 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f} no_improve={no_improve}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a, compute_J(a)


def fourier_jolt(a, rng, amplitude=0.02):
    """
    Add a Fourier-domain perturbation: modify the DFT of a by adding
    small random phase shifts to non-DC bins, then invert and project.
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    # Add random phase perturbation to non-DC bins
    phases = rng.uniform(-amplitude, amplitude, A.shape[0])
    phases[0] = 0.0
    A_pert = A * np.exp(1j * phases)
    a_pert = np.fft.irfft(A_pert, size)[:n]
    a_pert = np.maximum(a_pert, 0)
    if a_pert.sum() < 1e-10:
        return a.copy()
    a_pert /= a_pert.sum()
    return a_pert


def spectral_shaping(a, rng, alpha=0.5, amplitude=0.05):
    """
    Spectral shaping: modify |DFT(a)|^alpha to flatten the spectrum,
    then invert and project.
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    mag = np.abs(A)
    phase = np.angle(A)
    # Flatten: reduce peaks in the magnitude spectrum
    mag_target = np.full_like(mag, np.mean(mag))
    mag_new = (1 - amplitude) * mag + amplitude * mag_target
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, size)[:n]
    a_new = np.maximum(a_new, 0)
    if a_new.sum() < 1e-10:
        return a.copy()
    a_new /= a_new.sum()
    return a_new


def lp_mirror_descent(a, max_iters, deadline, verbose=False):
    """LP mirror descent: solve LP for descent direction at each step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

        A_mat = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 25:
            break

    return best_a, best_obj


def smooth_gradient_descent(a, max_iters, deadline, tau=50.0, lr=0.01, seed=0):
    """
    Smooth gradient descent using log-sum-exp approximation of max.
    Uses FFT-based gradient computation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    rng = np.random.default_rng(seed)
    no_improve = 0

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal tau
        frac = min(1.0, it / max(1, max_iters * 0.5))
        tau_curr = tau + 100.0 * frac

        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau_curr * c_shifted)
        w = exp_c / exp_c.sum()

        # Gradient: d/d a[i] of sum_k w[k] * conv(a,a)[k]
        # = sum_k w[k] * (a[k-i] + a[i-k]) with bounds
        # Use FFT: grad = ifft(fft(w) * fft(a_flip)) appropriately
        a_flip = a[::-1]
        size_g = 1
        while size_g < 2 * n:
            size_g <<= 1
        W_fft = np.fft.rfft(w, size_g)
        A_fft = np.fft.rfft(a_flip, size_g)
        conv_g = np.fft.irfft(W_fft * A_fft, size_g)

        grad = np.zeros(n)
        for i in range(n):
            idx = n - 1 + i
            if idx < size_g:
                grad[i] = conv_g[idx]

        # Project gradient to simplex tangent
        grad = grad - np.mean(grad)

        # Adam update
        t_count += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        lr_curr = lr / (1.0 + 0.05 * t_count ** 0.5)
        a_new = a - lr_curr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                # Perturb and reset
                a = best_a + rng.normal(0, 0.01, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

    return best_a, best_obj


def solve_single(a, deadline, verbose=False, use_lp=False, use_adam=False):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return a
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 1: Coordinate descent (40% of time)
    t1 = time.monotonic() + 0.40 * remaining
    a = try_update(coord_descent_exact(a, max_sweeps=50000, deadline=t1, seed=42)[0])

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 2: Fourier jolt + CD (20% of time)
    t2 = time.monotonic() + 0.20 * remaining
    rng = np.random.default_rng(123)
    a_jolt = fourier_jolt(best_a, rng, amplitude=0.03)
    a_jolt = try_update(coord_descent_exact(a_jolt, max_sweeps=20000, deadline=t2, seed=77)[0])

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 3: LP mirror descent (20% of time)
    if use_lp:
        t3 = time.monotonic() + 0.20 * remaining
        a_lp, obj_lp = lp_mirror_descent(best_a, max_iters=100, deadline=t3)
        try_update(a_lp)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 4: Adam (15% of time)
    if use_adam:
        t4 = time.monotonic() + 0.15 * remaining
        a_adam, obj_adam = smooth_gradient_descent(best_a, max_iters=3000, deadline=t4,
                                                   tau=30.0, lr=0.02, seed=55)
        try_update(a_adam)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 5: Final CD polish (5% of time)
    t5 = deadline - 2
    a = try_update(coord_descent_exact(best_a, max_sweeps=20000, deadline=t5, seed=999)[0])

    return best_a, best_obj

        update_best(a_opt2)
        print(f"  Jolt {j}: amp={amp:.3f} J={obj_opt2:.8f}", flush=True)

    # === PHASE 3: Multi-length exploration (300s) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Multi-length exploration (300s budget)", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase3_end = t_start + 1100.0
    candidate_ns = []
    # Generate candidate lengths around parent
    for delta in [-128, -64, -32, -16, 16, 32, 64, 128, 256]:
        cn = n_parent + delta
        if 256 <= cn <= 2048:
            candidate_ns.append(cn)
    # Also add some "round" numbers
    for rn in [512, 640, 768, 896, 1024, 1152, 1280]:
        if rn != n_parent and 256 <= rn <= 2048:
            candidate_ns.append(rn)

    candidate_ns = sorted(set(candidate_ns), key=lambda x: abs(x - n_parent))

    for new_n in candidate_ns[:6]:
        remaining_check = phase3_end - time.monotonic()
        if remaining_check < 40:
            break
        alloc = min(50.0, remaining_check - 10)
        if alloc < 30:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        a_opt3, obj_opt3 = solve_single(a_ext, time.monotonic() + alloc, use_lp=False, use_adam=False)
        update_best(a_opt3)
        print(f"  n={new_n}: J={obj_opt3:.8f}", flush=True)

    # === PHASE 4: Final intensive polish on best (remaining time) ===
    print(f"\n{'=' * 60}", flush=True)
    remaining = deadline - time.monotonic()
    print(f"PHASE 4: Final polish (budget {remaining:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and remaining > 20:
        # Multiple CD passes with different seeds
        n_passes = min(4, int(remaining // 30) + 1)
        per_pass = remaining / (n_passes + 1)

        for p in range(n_passes):
            t_pass_end = time.monotonic() + per_pass
            if time.monotonic() >= deadline - 5:
                break
            a_p = coord_descent_exact(best_seq, max_sweeps=50000, deadline=t_pass_end,
                                      seed=1000 + 7 * p)[0]
            update_best(a_p)

        # One LP pass if time allows
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_lp, obj_lp = lp_mirror_descent(best_seq, max_iters=50, deadline=deadline - 5)
            update_best(a_lp)

        # Final CD on the LP result
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_fin = coord_descent_exact(best_seq, max_sweeps=20000, deadline=deadline - 3,
                                        seed=77777)[0]
            update_best(a_fin)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
