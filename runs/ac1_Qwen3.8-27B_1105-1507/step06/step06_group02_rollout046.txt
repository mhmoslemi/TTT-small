def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (a*g)(k) = (A @ g)(k)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M):
    """Solve max sum(g) s.t. a*g <= M, g >= 0. Returns (g, dual_vars) or (None, None)."""

    cols = []

            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            dual = np.zeros(r.unevaluated if r.unevaluated is None else len(r.unevaluated))

            # Extract dual variables if available

                duals = -r.mult_A_ub  # dual variables for A_ub constraints
            return np.maximum(r.x, 0), duals
    except Exception:

    return None, None

        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = y[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * exp_b[k] * seg
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_mom ** 2)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta_mom * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def gs_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton for flat spectral initialization."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        power = 0.5 + 0.3 * beta
        mag_pow = mag ** power
        mag_new = mag_pow / (mag_pow.max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0.0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lp_fixed_point(a, deadline, max_iters=200, use_backtrack=True,
                   t_init=0.5, t_min=0.005, perturb=False, rng=None):
    """LP-based fixed point with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    prev_g = None

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break

        # Solve LP
        g = solve_lp_simple(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Determine step size
        if use_backtrack:
            best_t = t_min
            best_M_t = M
            for t in [t_init * (0.5 ** i) for i in range(8)]:
                t = max(t, t_min)
                a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-10:
                    continue
                a_t /= s_t
                M_t = fast_conv(a_t).max()
                if M_t < best_M_t:
                    best_M_t = M_t
                    best_t = t
            a = np.maximum((1.0 - best_t) * a + best_t * g, 0.0)
            s = a.sum()
            if s < 1e-10:
                a = best_a.copy()
            else:
                a /= s
        else:
            t = max(t_min, t_init / (1 + 0.1 * it))
            a = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a.sum()
            if s < 1e-10:
                a = best_a.copy()
            else:
                a /= s

        # Optional perturbation
        if perturb and rng is not None and it % 20 == 19:
            noise = rng.normal(0, 0.001, n)
            a_pert = np.maximum(a + noise, 1e-8)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert /= sp
                M_pert = fast_conv(a_pert).max()
                if M_pert < M * 1.01:
                    a = a_pert

        # Decay step size
        t_init = max(t_min, t_init * 0.995)

    return best_a, best_M


def lp_two_direction(a, deadline, max_iters=100, t_init=0.3):
    """Two-LP direction: use M and M*(1-delta) to get a better direction."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break

        # Solve LP with current M
        g1 = solve_lp_simple(a, M)
        if g1 is None:
            break
        g1 = g1 / g1.sum()

        # Solve LP with slightly reduced M for better direction
        delta = 0.001 + 0.002 * (1.0 / (1.0 + 0.01 * it))
        g2 = solve_lp_simple(a, M * (1 - delta))
        if g2 is not None and g2.sum() > 1e-10:
            g2 = g2 / g2.sum()
            direction = g2 - g1
            # Normalize direction
            dir_norm = np.linalg.norm(direction)
            if dir_norm > 1e-10:
                direction = direction / dir_norm
                # Step in -direction (to reduce M)
                for t in [t_init, t_init * 0.5, t_init * 0.25, t_init * 0.1]:
                    a_t = np.maximum(a - t * 0.1 * direction, 0.0)
                    s_t = a_t.sum()
                    if s_t < 1e-10:
                        continue
                    a_t /= s_t
                    M_t = fast_conv(a_t).max()
                    if M_t < M * (1 - 1e-14):
                        a = a_t
                        break
                else:
                    # Fallback to standard LP step
                    a = np.maximum((1.0 - 0.3) * a + 0.3 * g1, 0.0)
                    a = a / a.sum()
            else:
                a = np.maximum((1.0 - 0.3) * a + 0.3 * g1, 0.0)
                a = a / a.sum()
        else:
            # Standard LP step
            a = np.maximum((1.0 - 0.3) * a + 0.3 * g1, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s

    return best_a, best_M


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)
            return True
        return False

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.005))
    inits.append(np.ones(n, dtype=np.float64))
    for alpha in [0.5, 1.0, 2.0]:
        inits.append(np.maximum(0.3 + 0.7 * (t**alpha + (1 - t)**alpha) / 2, 0.01))
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.05)
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_spectral(a_rand, n_iters=80, beta=0.85))
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(gs_spectral(p, n_iters=100, beta=0.9))
    return inits


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 25.0
    best = BestTracker()

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(512)
        best.update(parent)

    n_parent = parent.size
    print(f"Starting with n={n_parent}", flush=True)

    # === PHASE 1: Quick smooth optimization with diverse starts (300s) ===
    p1_end = time.monotonic() + 300.0
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth Optimization", flush=True)
    print(f"{'='*50}", flush=True)

    # Focus on lengths near parent and some longer ones
    lengths_to_try = sorted(set([
        n_parent,
        max(256, n_parent - 64),
        n_parent + 64,
        n_parent + 128,
        n_parent + 256,
        n_parent + 512,
    ]))
    # Filter to reasonable range
    lengths_to_try = [n for n in lengths_to_try if 256 <= n <= 2500]

    best_per_length = {}
    for n in lengths_to_try:
        if time.monotonic() > p1_end - 30:
            break
        t0 = time.monotonic()
        sub_budget = min(45.0, p1_end - time.monotonic() - 10)
        if sub_budget < 10:
            continue

        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > p1_end - 15:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            s0 = a0.sum()
            if s0 < 1e-10:
                continue
            a0 /= s0
            # Shorter time for later inits
            frac = 0.8 / (1.0 + 0.5 * idx)
            sub_dl = time.monotonic() + sub_budget * frac
            max_it = max(200, int(2000 * (512.0 / n)))
            a = nesterov_smooth(a0, min(sub_dl, p1_end - 15),
                                max_iters=max_it, tau_start=1.0, tau_end=100.0,
                                lr_init=0.3, lr_end=0.003)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")

    # === PHASE 2: LP Fixed-Point with backtracking (main optimization) ===
    p2_end = time.monotonic() + 650.0
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Fixed-Point Optimization", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        print(f"  Optimizing n={n_best} sequence", flush=True)

        # Multiple rounds of LP optimization with decreasing perturbation
        a_current = best.seq.copy()
        for round_num in range(5):
            if time.monotonic() > p2_end - 30:
                break
            # Slight perturbation in later rounds to escape local minima
            if round_num > 0:
                noise_scale = 0.003 * (round_num - 1) + 0.001
                a_pert = np.maximum(a_current + rng.normal(0, noise_scale, n_best), 1e-8)
                a_pert /= a_pert.sum()
                # Only use perturbed if it doesn't worsen too much
                M_pert = fast_conv(a_pert).max()
                M_curr = fast_conv(a_current).max()
                if M_pert < M_curr * 1.005:
                    a_current = a_pert
                else:
                    a_current = best.seq.copy()

            sub_end = min(time.monotonic() + 120.0, p2_end - 20)
            max_iters = max(50, int(120.0 / max(0.05, 2.5 * (n_best / 512.0) ** 2)))
            a_lp, M_lp = lp_fixed_point(a_current, sub_end,
                                       max_iters=max_iters,
                                       use_backtrack=True,
                                       t_init=0.4, t_min=0.003,
                                       perturb=False, rng=None)
            best.update(a_lp)
            a_current = a_lp if a_lp is not None else a_current
            print(f"  Round {round_num+1}: M={M_lp:.10f} J={compute_J(a_current):.6f}", flush=True)

    best.report("Phase 2")

    # === PHASE 3: Two-LP direction optimization ===
    p3_end = time.monotonic() + 200.0
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Two-LP Direction", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < p3_end - 30:
        a_two, M_two = lp_two_direction(best.seq, min(time.monotonic() + 180, p3_end - 10),
                                        max_iters=100, t_init=0.2)
        best.update(a_two)
        print(f"  Two-LP: M={M_two:.10f}", flush=True)

    best.report("Phase 3")

    # === PHASE 4: Alternative lengths ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Alternative Lengths", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        alt_lengths = []
        for delta in [-32, -16, 16, 32, -64, 64]:
            n_try = n_best + delta
            if 256 <= n_try <= 2500:
                alt_lengths.append(n_try)

        for n_try in alt_lengths[:4]:
            if time.monotonic() > deadline - 30:
                break
            a_res = resample_seq(best.seq, n_try)
            a_res = np.maximum(a_res, 1e-6)
            a_res /= a_res.sum()
            # Quick smooth + LP
            a_smooth = nesterov_smooth(a_res, min(time.monotonic() + 20, deadline - 25),
                                       max_iters=300, tau_start=5, tau_end=80,
                                       lr_init=0.15, lr_end=0.01)
            best.update(a_smooth)
            if time.monotonic() < deadline - 20:
                a_lp, _ = lp_fixed_point(a_smooth, min(time.monotonic() + 40, deadline - 15),
                                         max_iters=60, use_backtrack=True,
                                         t_init=0.3, t_min=0.005)
                best.update(a_lp)

    best.report("Phase 4")

    # === PHASE 5: Final intensive LP polish ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Final LP Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 5:
        # Multiple short LP passes with tiny steps
        a_final = best.seq.copy()
        for _ in range(3):
            if time.monotonic() > deadline - 3:
                break
            sub_dl = min(time.monotonic() + (deadline - time.monotonic()) * 0.4, deadline - 2)
            a_final, M_fin = lp_fixed_point(a_final, sub_dl,
                                           max_iters=50, use_backtrack=True,
                                           t_init=0.1, t_min=0.001)
            best.update(a_final)

    best.report("Phase 5")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(512)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
