def evaluate_sequence(seq):

        return np.inf
    if len(seq) == 0:
        return np.inf
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)) or np.isnan(x) or np.isinf(x):
            return np.inf
    # Convert potentials to floats with bounds
    seq_converted = [float(x) for x in seq]
    seq_clipped = [max(0, min(1000, x)) for x in seq_converted]
    n = len(seq_clipped)
    conv_sequence = compute_conv(seq_clipped)
    max_conv = np.max(conv_sequence)
    total_sum = np.sum(seq_clipped)
    if total_sum < 0.01:
        return np.inf
    return 2 * n * max_conv / (total_sum ** 2)
