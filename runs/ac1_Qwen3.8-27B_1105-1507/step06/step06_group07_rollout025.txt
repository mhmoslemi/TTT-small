Uses Adam in log-space with aggressive temperature annealing,

from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def soft_max_grad(a, beta):
    """
    Returns (obj_val, grad) where obj_val = 2n*logsumexp(beta*conv(a,a))/S^2
    and grad = d(obj_val)/d(a).
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = np.clip(beta * (conv - cmax), -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    # Gradient of ls w.r.t. a:
    size = 1
    while size < 2 * len(conv):
        size <<= 1

    A = rfft(a, size)
