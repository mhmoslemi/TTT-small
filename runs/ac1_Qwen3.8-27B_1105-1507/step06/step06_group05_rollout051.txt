def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1

    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_qp_proximal(a, M, lambda_reg=0.01, time_limit=60):
    """Solve QP: max sum(g) - lambda*||g-a||^2 s.t. conv(a,g) <= M, g >= 0.
    Returns normalized g or None."""
    if not HAS_CVXPY:
        return None
    n = len(a)
    g = cp.Variable(n)
    A = build_toeplitz(a)
    constraints = [A @ g <= M, g >= 0]
    objective = cp.Maximize(cp.sum(g) - lambda_reg * cp.sum_squares(g - a))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, time_limit=time_limit)
        if g.value is not None:
            g_val = np.maximum(g.value, 0.0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def backtracking_line_search(a, direction, obj_current, max_alpha=1.0, c=0.01, rho=0.5):
    """Backtracking line search for minimize f(a + t*direction)."""
    # Compute directional derivative (approximate)
    # f'(a; d) = subgradient dot d
    # We'll just use a simple backtracking on the function value
    t = max_alpha
    best_t = 0.0
    best_obj = obj_current
    for _ in range(30):
        a_try = a + t * direction
        a_try = normalize(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
        t *= rho
        if t < 1e-6:
            break
    return best_t, best_obj


def exact_subgradient(a, conv, tolerance=1e-6):
    """Compute the exact subgradient using all active (near-max) convolution positions."""
    n = len(a)
    cmax = float(np.max(conv))
    # Find all positions where conv is within tolerance of max
    active = np.where(conv >= cmax - tolerance * cmax)[0]
    
    # Subgradient at position k: grad[i] = 2 * a[k-i] if 0 <= k-i < n
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(active)
    return grad


def soft_max_subgradient(a, conv, tau=0.001):
    """Compute soft-max (log-sum-exp) subgradient for smoother optimization."""
    n = len(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # Compute gradient via FFT
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    
    w_ext = np.zeros(2 * n - 1)
    w_ext[:] = weights
    a_flip = a[::-1]
    
    Wf = rfft(w_ext, size)
    Af = rfft(a_flip, size)
    corr = irfft(Wf * Af, size)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    
    return grad


def lp_alternate_improved(a, max_iters=50, time_budget=100.0, use_qp=False):

def peak_shaving_gradient(a, max_steps=200, time_budget=60.0, method='softmax'):
    """Gradient descent using peak-targeting subgradient with backtracking."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.1
    tau = 0.001
    
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        
        if method == 'exact':
            grad = exact_subgradient(a, conv)
        else:
            grad = soft_max_subgradient(a, conv, tau)
        
        # Center the gradient (projection onto sum=const)
        grad_centered = grad - np.mean(grad)
        
        # Try the step
        a_try = normalize(a - step_size * grad_centered)
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            step_size = min(step_size * 1.05, 0.5)  # Slight increase
        else:
            step_size *= 0.5
            if step_size < 1e-10:
                # Perturb to escape
                rng = np.random.default_rng(step * 31 + 7)
                a = normalize(best_a * np.exp(rng.normal(0, 0.02, n)))
                step_size = 0.1
        
        # Cool temperature for soft-max
        tau *= 0.98
    
    return best_a


def spectral_flatten(a, max_iters=50, time_budget=40.0):
    """POCS-style spectral equalization."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        
        # Progressive compression
        alpha = 0.3 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** (1.0 / alpha)) * p_max  # Note: 1/alpha > 1 compresses
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.7 * a + 0.3 * a_new)
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

Improved multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=60):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_qp_proximal(a, M, lambda_reg=0.005, time_limit=60):
    """Solve QP: max sum(g) - lambda*||g-a||^2 s.t. conv(a,g) <= M, g >= 0."""
    if not HAS_CVXPY:
        return None
    n = len(a)
    if n > 900:
        return None
    g = cp.Variable(n)
    A = build_toeplitz(a)
    constraints = [A @ g <= M, g >= 0]
    objective = cp.Maximize(cp.sum(g) - lambda_reg * cp.sum_squares(g - a))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, time_limit=time_limit)
        if g.value is not None:
            g_val = np.maximum(g.value, 0.0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, time_limit=time_limit)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            pass
    return None


def exact_subgradient_fft(a, conv, tolerance=0.0):
    """Compute exact subgradient using FFT-based cross-correlation."""
    n = len(a)
    cmax = float(np.max(conv))
    if tolerance == 0.0:
        active_mask = conv >= cmax - 1e-10 * cmax
    else:
        active_mask = conv >= cmax - tolerance * cmax
    w = active_mask.astype(np.float64)
    w_sum = float(np.sum(w))
    if w_sum <= 0:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
        return 2.0 * _single_subgrad_fft(a, idx, n)
    w /= w_sum

    size = 1
    while size < 3 * n:
        size <<= 1

    W = fft(w, size)
    B = fft(a, size)
    cross_corr = ifft(W * np.conj(B), size).real[:n]
    grad = 2.0 * cross_corr
    return grad


def _single_subgrad_fft(a, k, n):
    """Subgradient for a single active position k."""
    grad = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            grad[i] = a[j]
    return grad


def soft_max_subgradient(a, conv, tau=0.001):
    """Compute soft-max subgradient via FFT."""
    n = len(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    size = 1
    while size < 3 * n:
        size <<= 1

    W = fft(weights, size)
    B = fft(a, size)
    cross_corr = ifft(W * np.conj(B), size).real[:n]
    grad = 2.0 * cross_corr
    return grad


def backtracking_step(a, direction, obj_current, c=0.01, rho=0.7, max_iter=25):
    """Backtracking line search. Returns (best_t, best_a, best_obj)."""
    t = 1.0
    best_t = 0.0
    best_a = a.copy()
    best_obj = obj_current
    for _ in range(max_iter):
        a_try = normalize(a + t * direction)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try
            best_t = t
        t *= rho
        if t < 1e-7:
            break
    return best_t, best_a, best_obj


def lp_alternate_improved(a, max_iters=50, time_budget=80.0, use_qp=False):
    """LP alternating with backtracking line search and momentum."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum_vec = np.zeros(n)
    momentum_alpha = 0.3

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 4:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = None
        if use_qp and HAS_CVXPY and n <= 850:
            g = solve_qp_proximal(a, M, lambda_reg=0.005,
                                  time_limit=max(4, int(remaining * 0.35)))
        if g is None:
            g = solve_lp_max_sum(a, M, time_limit=max(4, int(remaining * 0.55)))
        if g is None:
            break

        direction = g - a
        # Add momentum
        direction = direction + momentum_alpha * momentum_vec
        momentum_vec = direction - momentum_alpha * momentum_vec

        t, a_new, obj_new = backtracking_step(a, direction, best_obj)

        if t > 0 and obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try without momentum
            t2, a_new2, obj_new2 = backtracking_step(a, g - a, best_obj)
            if t2 > 0 and obj_new2 < best_obj - 1e-14:
                best_obj = obj_new2
                best_a = a_new2.copy()
                a = a_new2
            else:
                # Perturb to escape

                sigma = 0.04 * (0.93 ** it)
                a = normalize(best_a * np.exp(rng.normal(0, max(sigma, 0.005), n)))
                momentum_vec = np.zeros(n)

    return best_a


def gradient_descent_momentum(a, max_steps=300, time_budget=60.0,
                              method='exact', tau_init=0.002,
                              lr_init=0.15, momentum=0.7):
    """Gradient descent with momentum and adaptive learning rate."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    tau = tau_init

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)

        if method == 'exact':
            grad = exact_subgradient_fft(a, conv, tolerance=0.05)
        else:
            grad = soft_max_subgradient(a, conv, tau)

        # Center gradient (project onto sum=const hyperplane)
        grad_centered = grad - np.mean(grad)

        # Momentum update
        v = momentum * v + grad_centered
        a_try = normalize(a - lr * v)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            lr = min(lr * 1.02, 0.8)
        else:
            lr *= 0.5
            v *= momentum * 0.5
            if lr < 1e-10:
                rng = np.random.default_rng(step * 2654435761 + 97)
                a = normalize(best_a * np.exp(rng.normal(0, 0.015, n)))
                v = np.zeros(n)
                lr = lr_init * 0.5

def spectral_flatten(a, max_iters=50, time_budget=40.0):
    """POCS-style spectral equalization with progressive compression."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        alpha = 0.4 + 0.35 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** (1.0 / alpha)) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.6 * a + 0.4 * a_new)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

        a_new = lp_alternate_improved(a, max_iters=30, time_budget=alloc, use_qp=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
        print(f"  Post-gradient LP: {obj_new:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700
    n_options = sorted(set([
        max(150, n_best - 200), max(150, n_best - 100),
        n_best - 50, n_best, n_best + 50,
        n_best + 100, n_best + 200, n_best + 400
    ]))
    n_options = [n for n in n_options if 150 <= n <= 1800]

    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if best_seq is not None:
            if n_try == n_best:
                a_try = normalize(best_seq.copy())
            else:
                a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try

        alloc = min(remaining * 0.12, 35.0)
        if n_try <= 900 and alloc > 10:
            a_try = lp_alternate_improved(a_try, max_iters=10, time_budget=alloc * 0.45)
        a_try = gradient_descent_momentum(a_try, max_steps=80, time_budget=alloc * 0.45,
                                          method='softmax', tau_init=0.002,
                                          lr_init=0.08, momentum=0.7)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 5: Diverse random restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Random restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed)

        for restart in range(12):
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            alloc = min(remaining * 0.09, 28.0)

            strategy = restart % 6
            if strategy == 0:
                a_try = best_seq * np.exp(rng.normal(0, 0.05, n_cur))
            elif strategy == 1:
                freq = rng.uniform(0.3, 6.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.12 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
                a_try = best_seq * mod
            elif strategy == 2:
                a_rand = rng.dirichlet(np.ones(n_cur) * 0.2)
                mix = rng.uniform(0.1, 0.25)
                a_try = (1 - mix) * normalize(best_seq) + mix * a_rand
            elif strategy == 3:
                # Block shuffle
                a_try = best_seq.copy()
                n_blocks = int(rng.integers(4, 10))
                block_size = n_cur // n_blocks
                indices = rng.permutation(n_blocks)
                a_new = np.zeros(n_cur)
                for old_i, new_i in enumerate(indices):
                    s = old_i * block_size
                    e = s + block_size if old_i < n_blocks - 1 else n_cur
                    ns = new_i * block_size
                    ne = ns + (e - s)
                    if ne > n_cur:
                        ne = n_cur
                    a_new[ns:ne] = a_try[s:e]
                a_try = a_new
            elif strategy == 4:
                # Edge concentration
                a_try = best_seq * (1.0 + 0.3 * np.abs(2.0 * np.arange(n_cur) / n_cur - 1.0))
            else:
                # Smooth noise
                a_try = best_seq * (1.0 + 0.08 * rng.standard_normal(n_cur))

            a_try = normalize(a_try)

            if n_cur <= 900 and alloc > 8:
                a_try = lp_alternate_improved(a_try, max_iters=8, time_budget=alloc * 0.4)
            a_try = gradient_descent_momentum(a_try, max_steps=60, time_budget=alloc * 0.45,
                                              method='exact', tau_init=0.001,
                                              lr_init=0.06, momentum=0.7)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart} (strat={strategy}): {best_score:.8f} **")
            else:
                print(f"  Restart {restart} (strat={strategy}): {obj_try:.8f}")

    # === Phase 6: Final intensive polishing ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polishing")
    print(f"{'='*60}")

    if best_seq is not None:
        n_final = len(best_seq)
        a_final = normalize(best_seq.copy())

        for polish_round in range(6):
            if time.monotonic() > deadline - 3:
                break
            remaining = deadline - time.monotonic()
            if remaining < 4:
                break
            alloc = min(remaining * 0.28, 28.0)

            if n_final <= 900 and alloc > 8:
                a_pol = lp_alternate_improved(a_final, max_iters=20, time_budget=alloc * 0.35,
                                              use_qp=True)
            else:
                a_pol = a_final.copy()

            a_pol = gradient_descent_momentum(a_pol, max_steps=100, time_budget=alloc * 0.35,
                                              method='exact', tau_init=0.0005,
                                              lr_init=0.05, momentum=0.8)
            a_pol = gradient_descent_momentum(a_pol, max_steps=80, time_budget=alloc * 0.3,
                                              method='softmax', tau_init=0.0005,
                                              lr_init=0.04, momentum=0.8)

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish_round}: {best_score:.8f} **")
            else:
                a_final = a_pol
                print(f"  Polish {polish_round}: {obj_pol:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
