Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous:

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direct_peak(a, time_limit=30):
    """
    Solve LP: min t s.t. conv(a,g)[k] <= t for all k, sum(g) = 1, g >= 0.
    Returns g (normalized to sum 1) or None.
    """
    n = len(a)
    m = 2 * n - 1
    A = build_toeplitz(a)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality: A @ g - t <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_max_sum(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:

                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:

    except Exception:

                best_a = g.copy()
                a = g
            else:

    return best_a


def soft_max_gradient(a, max_steps=300, time_budget=50.0):
    """
    Improved soft-max gradient descent with adaptive temperature and step size.
    Uses multiple peaks simultaneously.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 3 * n - 1:
        size <<= 1

    # Adaptive parameters
    tau = 0.0005  # initial temperature
    step_size = 0.05
    min_step = 1e-8
    stagnation = 0

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Adaptive temperature: wider early, narrower later
        tau = max(0.0001, 0.0005 * (1.0 - 0.5 * step / max_steps) + 0.5 * cmax / (n * (step + 1)))
        
        # Compute soft-max weights
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient via FFT
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Project onto sum=const (remove mean)
        grad_centered = grad - np.mean(grad)

        # Adaptive step size with backtracking
        a_new = normalize(a - step_size * grad_centered)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stagnation = 0
            step_size = min(step_size * 1.05, 0.15)  # slightly increase
        else:
            step_size *= 0.7
            stagnation += 1
            if step_size < min_step or stagnation > 20:
                break

        # Cool temperature
        tau *= 0.98

    return best_a


def peak_chasing(a, max_iters=200, time_budget=40.0):
    """
    Multiplicative update that directly targets the top K peaks.
    For each peak at position k, reduce a[j] proportionally to a[k-j].
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    K = 5  # number of top peaks to target
    eta = 0.02  # learning rate for multiplicative update

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Find top K peaks (above threshold)
        threshold = cmax * 0.95
        peak_positions = np.where(conv >= threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = [int(np.argmax(conv))]
        peak_positions = peak_positions[:K]

        # Compute the "pressure" on each a[j] from all targeted peaks
        pressure = np.zeros(n)
        for k in peak_positions:
            for j in range(n):
                k2 = int(k) - j
                if 0 <= k2 < n:
                    pressure[j] += a[k2]

        # Multiplicative update: reduce a[j] where pressure is high
        # a_new[j] = a[j] * exp(-eta * pressure[j] / max_pressure)
        max_p = float(np.max(pressure))
        if max_p < 1e-15:
            break
        a_new = a * np.exp(-eta * pressure / max_p)
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.1, 0.1)
        else:
            eta *= 0.8
            if eta < 0.001:
                break

    return best_a


def spectral_equalization_v2(a, max_iters=80, time_budget=40.0):
    """
    Improved POCS: alternate between spectral flattening and time-domain projection.
    Uses soft clipping instead of hard projection.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Soft spectral compression
        # Use tanh-based soft clipping
        alpha = 0.3 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        # Soft clip: values above 1 get compressed
        P_soft = np.where(P_norm > 1.0, 1.0 + np.tanh(P_norm - 1.0) * 0.5, P_norm)
        P_new = P_soft * p_max

        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        # Blended update
        blend = 0.3 + 0.4 * (it / float(max_iters))
        a_candidate = normalize(blend * a + (1 - blend) * a_new)
        obj_new = compute_obj(a_candidate)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_candidate.copy()
            a = a_candidate
        else:
            # Try full replacement
            obj_full = compute_obj(a_new)
            if obj_full < best_obj:
                best_obj = obj_full
                best_a = a_new.copy()
                a = a_new

    return best_a


def coordinate_descent_v3(a, time_budget=30.0):
    """
    Improved coordinate descent: target the top peaks and move mass
    from high-pressure positions to low-pressure positions.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Find top 3 peaks
        top_peaks = np.argsort(conv)[-3:][::-1]

        improved = False
        for k_star in top_peaks:
            if time.monotonic() - start > time_budget:
                break

            k_star = int(k_star)
            # Compute pressure on each position
            pressure = np.zeros(n)
            for j in range(n):
                k2 = k_star - j
                if 0 <= k2 < n:
                    pressure[j] = a[k2]
                else:
                    pressure[j] = 0.0

            # Find high-pressure and low-pressure positions
            high_idx = np.where(pressure > np.median(pressure) * 2)[0]
            low_idx = np.where(pressure < np.median(pressure) * 0.5)[0]

            if len(high_idx) == 0 or len(low_idx) == 0:
                continue

            # Move mass from high to low pressure
            for _ in range(5):
                if len(high_idx) == 0:
                    break
                j_src = high_idx[int(rng.integers(0, len(high)))] if len(high_idx) > 1 else high_idx[0]
                j_dst = low_idx[int(rng.integers(0, len(low_idx)))]
                
                a_try = a.copy()
                delta = 0.03 * a_try[j_src]
                if delta < 1e-10:
                    continue
                a_try[j_src] -= delta
                a_try[j_dst] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)

                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True

        if not improved and rng.random() < 0.15:
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)
