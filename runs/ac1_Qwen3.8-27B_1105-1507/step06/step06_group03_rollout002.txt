def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                phi = 0.6180339887498949
                a_g = x_lo + (1 - phi) * (x_hi - x_lo)
                b_g = x_lo + phi * (x_hi - x_lo)
                
                # Precompute the slice of conv that depends on a[i]
                lo = max(0, i)
                hi = min(2 * n - 2, i + n - 1)
                a_slice = a[lo - i:hi - i + 1].copy()
                kq = 2 * i
                valid_kq = kq <= 2 * n - 2
                
                def eval_at_x(x):
                    delta = x - old
                    conv[lo:hi + 1] += 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] += delta * delta
                    new_s = s - old + x
                    if new_s < 1e-10:
                        result = float('inf')
                    else:
                        result = 2.0 * n * conv.max() / (new_s * new_s)
                    conv[lo:hi + 1] -= 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] -= delta * delta
                    return result
                
                f_a = eval_at_x(a_g)
                f_b = eval_at_x(b_g)
                
                for _ in range(10):  # 10 iterations of golden section
                    if f_a < f_b:
                        x_hi = b_g
                        b_g = a_g
                        f_b = f_a
                        a_g = x_lo + (1 - phi) * (x_hi - x_lo)
                        f_a = eval_at_x(a_g)
                    else:
                        x_lo = a_g
                        a_g = b_g
                        f_a = f_b
                        b_g = x_lo + phi * (x_hi - x_lo)
                        f_b = eval_at_x(b_g)
                
                best_x = (x_lo + x_hi) / 2
                best_f = min(f_a, f_b)
                
                # Compare with current
                if best_f < current_obj - 1e-14:
                    delta = best_x - old
                    conv[lo:hi + 1] += 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] += delta * delta
                    a[i] = best_x
                    s = s - old + best_x
                    improved = True
            else:
                # Grid search (faster per coordinate)
                if old < 1e-10:
                    candidates = np.array([0.0, 0.5/n, 1.0/n, 2.0/n, 3.0/n])
                else:
                    ratios = np.array([0.0, 0.25, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.6])
                    candidates = ratios * old
                
                conv_max = conv.max()
                best_val = old
                best_local = 2.0 * n * conv_max / (s * s)
                
                lo = max(0, i)
                hi = min(2 * n - 2, i + n - 1)
                a_slice = a[lo - i:hi - i + 1].copy()
                kq = 2 * i
                valid_kq = kq <= 2 * n - 2
                
                for x in candidates:
                    x = float(max(0.0, x))
                    delta = x - old
                    if abs(delta) < 1e-15:
                        continue
                    conv[lo:hi + 1] += 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] += delta * delta
                    
                    new_s = s - old + x
                    if new_s < 1e-10:
                        new_obj = float('inf')
                    else:
                        new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                    
                    if new_obj < best_local:
                        best_local = new_obj
                        best_val = x
                    
                    conv[lo:hi + 1] -= 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] -= delta * delta
                
                if best_val != old and abs(best_val - old) > 1e-13:
                    delta = best_val - old
                    conv[lo:hi + 1] += 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] += delta * delta
                    a[i] = best_val
                    s = s - old + best_val
                    improved = True

        # Update best
        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 200 == 0:
            print(f"    CD v2 sweep {sweep}: obj={best_obj:.8f}", flush=True)

    # Final cleanup
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def sca_lp_step(a, max_iters, deadline, top_k=20, verbose=False):
    """
    SCA-LP with only top-k active constraints for speed.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()
        
        # Only use top-k constraints for speed
        # Find the top-k lags
        k_indices = np.array(range(2 * n - 1))
        c_sorted = np.sort(c)[::-1]
        top_k_actual = min(top_k, 2 * n - 1)
        # Get indices of top k values
        threshold = c_sorted[min(top_k_actual, len(c_sorted)) - 1]
        active_k = np.where(c >= threshold * 0.99)[0]
        
        if len(active_k) == 0:
            active_k = np.array([np.argmax(c)])
        
        # Build LP with only active constraints
        rows = []
        cols = []
        vals = []
        b_ub = []
        
        for k in active_k:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub))
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(len(b_ub))
            cols.append(n)  # t variable
            vals.append(-1.0)
            b_ub.append(c[k])
        
        A = coo_matrix((vals, (rows, cols)), shape=(len(active_k), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        bounds = [(0, None)] * n + [(None, None)]
        
        A_eq = csr_matrix((np.ones(n), (0, range(n))), shape=(1, n + 1))
        b_eq = [1.0]
        
        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    J_g = compute_J(g)
                    if J_g < best_obj:
                        a = g.copy()
                        best_obj = J_g
                        best_a = g.copy()
                        no_improve = 0
                    else:
                        found = False
                        for frac in [0.5, 0.3, 0.2, 0.1]:
                            a_try = (1 - frac) * a + frac * g
                            a_try = np.maximum(a_try, 0)
                            if a_try.sum() > 1e-10:
                                a_try /= a_try.sum()
                            J_try = compute_J(a_try)
                            if J_try < best_obj - 1e-12:
                                a = a_try
                                best_obj = J_try
                                best_a = a_try.copy()
                                no_improve = 0
                                found = True
                                break
                        if not found:
                            no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 15:
            break

    return best_a, best_obj


def smooth_adam_v2(a, max_iters, deadline, tau_init=20.0, tau_max=150.0,
                   lr_init=0.04, verbose=False):
    """
    Improved Adam with log-sum-exp smoothing.
    Uses FFT for gradient computation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_count = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal tau (faster schedule)
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_init + (tau_max - tau_init) * (frac ** 0.5)

        # Compute smooth gradient
        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()

        # Gradient via FFT
        a_rev = a[::-1]
        w_rev = w[::-1]
        size = 1
        while size < 3 * n - 1:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        Wrev_fft = np.fft.rfft(w_rev, size)

        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wrev_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad[i] += conv2[idx2]

        # Project gradient to simplex tangent space
        grad = grad - np.mean(grad)

        # Adam update
        t_count += 1
        lr = lr_init / (1.0 + 0.05 * t_count ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                # Restart with perturbation
                perturb_scale = 0.05 + 0.1 * min(1.0, no_improve / 200.0)
                a = best_a + rng.normal(0, perturb_scale, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

        if verbose and it % 200 == 0:
            print(f"    Adam v2 it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def generate_inits_v2(n, rng, parent=None):
    """Generate diverse initial sequences with focus on structured patterns."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        
        # Small perturbations
        for scale in [0.01, 0.02, 0.05, 0.1, 0.2]:
            inits.append(project_simplex(p + rng.normal(0, scale, n)))
        
        # Frequency-domain perturbations
        for freq_scale in [0.05, 0.1, 0.3]:
            A = np.fft.rfft(p)
            A_mag = np.abs(A)
            A_phase = np.angle(A)
            # Multiply by smooth random function
            smooth_rand = np.exp(-0.5 * (np.linspace(0, 1, len(A)) * 5) ** 2) * rng.uniform(0.5, 1.5, len(A))
            A_new = A_mag * smooth_rand * np.exp(1j * A_phase)
            p_new = np.fft.irfft(A_new, n)
            inits.append(project_simplex(p_new))
        
        # Structured bumps
        for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
            for width in [0.05, 0.1, 0.2]:
                bump = np.exp(-((t - center) / width) ** 2)
                inits.append(project_simplex(p * (1 + 0.3 * (bump - bump.mean()))))

    # Base patterns
    inits.append(np.ones(n) / n)
    
    # Raised cosine window
    rc = 0.5 * (1 + np.cos(2 * np.pi * t))
    inits.append(project_simplex(rc + 0.1))
    
    # Gaussian
    for width in [0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-((t - 0.5) / width) ** 2)
        inits.append(project_simplex(g + 0.05))
    
    # Sine power patterns
    for power in [2, 3, 4, 6]:
        s = np.sin(np.pi * t) ** power
        inits.append(project_simplex(s + 0.05))
    
    # B-spline-like (triangular convolutions)
    tri = np.maximum(1 - 2 * np.abs(t - 0.5), 0)
    inits.append(project_simplex(tri + 0.1))
    
    # Smooth random
    ksize = max(5, n // 10)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    return inits
