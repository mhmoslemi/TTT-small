import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def obj_from_conv(conv, n, s):
    """Compute objective from pre-computed convolution and sum."""
    if s < 1e-10:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)


def coord_descent_exact(a, max_sweeps, deadline, random_order=False, seed=0,
                        n_candidates=12, refine=True):
    """
    Coordinate descent with exact 1D minimization per coordinate.
    For each coordinate, evaluates objective at multiple candidate points
    and optionally refines with golden section search.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = obj_from_conv(conv, n, s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 500

    # Precompute unaffected max for each coordinate
    # For coordinate i, affected taps are in [i, i+n-1]
    # Unaffected max = max of conv outside [i, i+n-1]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved = False

        for i in order:
            old_val = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            kq = 2 * i  # quadratic term index

            # Get affected slice of a (for the cross terms)
            # conv[k] += 2*delta*a[k-i] for k in [i, i+n-1], k != 2i
            # conv[2i] += 2*delta*a[i] + delta^2
            a_idx = np.arange(lo - i, hi - i + 1)
            a_idx = a_idx[(a_idx >= 0) & (a_idx < n)]
            if len(a_idx) == 0:
                continue
            a_slice = a[a_idx]

            # Unaffected max
            if lo > 0:
                unaff_left = conv[:lo].max() if lo > 0 else 0.0
            else:
                unaff_left = 0.0
            if hi < 2 * n - 2:
                unaff_right = conv[hi + 1:].max() if hi + 1 < 2 * n - 1 else 0.0
            else:
                unaff_right = 0.0
            M_unaff = max(unaff_left, unaff_right) if n > 1 else 0.0
            if n == 1:
                M_unaff = 0.0

            # Current conv in affected region
            conv_aff = conv[lo:hi + 1].copy()

            def eval_x(x):
                """Evaluate objective for a[i] = x"""
                x = max(0.0, float(x))
                delta = x - old_val
                if abs(delta) < 1e-15:
                    new_max = max(M_unaff, conv_aff.max())
                else:
                    new_conv = conv_aff + 2.0 * delta * a_slice
                    if 0 <= kq - lo <= (hi - lo):
                        new_conv[kq - lo] += delta * delta
                    new_max = max(M_unaff, new_conv.max())
                new_s = s - old_val + x
                if new_s < 1e-10:
                    return float('inf')
                return 2.0 * n * new_max / (new_s * new_s)

            # Generate candidates
            if old_val < 1e-12:
                candidates = [0.0, 1e-4, 1e-3, 5e-3, 1e-2, 5e-2, 0.1]
            else:
                # Spread candidates around current value
                candidates = np.concatenate([
                    [0.0],
                    np.linspace(0, 3.0 * old_val, n_candidates)
                ])
                # Add some far-out points
                candidates = np.unique(np.concatenate([candidates, [5.0 * old_val, 10.0 * old_val]]))

            # Evaluate all candidates
            best_x = old_val
            best_local = eval_x(old_val)
            for x in candidates:
                if abs(x - old_val) < 1e-14:
                    continue
                v = eval_x(x)
                if v < best_local:
                    best_local = v
                    best_x = x

            # Golden section refinement around best
            if refine and abs(best_x - old_val) > 1e-13:
                # Find bracket
                if best_x == 0.0:
                    hi_g = max(old_val, 0.1 * best_x if best_x > 0 else old_val)
                    lo_g = 0.0
                elif best_x == candidates.max():
                    lo_g = best_x * 0.5
                    hi_g = best_x * 2.0
                else:
                    # Find neighbors in candidates
                    idx = np.where(candidates == best_x)[0]
                    if idx[0] > 0:
                        lo_g = candidates[idx[0] - 1]
                    else:
                        lo_g = max(0.0, best_x * 0.5)
                    if idx[0] < len(candidates) - 1:
                        hi_g = candidates[idx[0] + 1]
                    else:
                        hi_g = best_x * 2.0

                # Golden section
                gr = (np.sqrt(5) - 1) / 2
                a_g, b_g = lo_g, hi_g
                c_g = b_g - gr * (b_g - a_g)
                d_g = a_g + gr * (b_g - a_g)
                fc = eval_x(c_g)
                fd = eval_x(d_g)
                for _ in range(8):
                    if fc < fd:
                        b_g = d_g
                        d_g = c_g
                        fd = fc
                        c_g = b_g - gr * (b_g - a_g)
                        fc = eval_x(c_g)
                    else:
                        a_g = c_g
                        c_g = d_g
                        fc = fd
                        d_g = a_g + gr * (b_g - a_g)
                        fd = eval_x(d_g)
                mid = (a_g + b_g) / 2
                v_mid = eval_x(mid)
                if v_mid < best_local:
                    best_local = v_mid
                    best_x = mid

            # Apply best
            if abs(best_x - old_val) > 1e-13:
                delta = best_x - old_val
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if 0 <= kq < 2 * n - 1:
                    conv[kq] += delta * delta
                a[i] = best_x
                s = s - old_val + best_x
                improved = True

        current_obj = obj_from_conv(conv, n, s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def sca_lp_adaptive(a, max_iters, deadline, epsilon_frac=0.05, verbose=False):
    """
    SCA-LP with adaptive active set.
    Only includes constraints that are near-active (within epsilon_frac of max).
    This makes the LP much smaller and faster.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()
        threshold = M - epsilon_frac * M

        # Active set: taps near the maximum
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Build LP: minimize t
        # s.t. 2*(a*g)[k] - c[k] <= t for k in active
        #      sum(g) = 1, g >= 0
        rows = []
        cols = []
        vals = []
        b_ub = []

        for idx, k in enumerate(active):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(idx)
            cols.append(n)  # t variable
            vals.append(-1.0)
            b_ub.append(c[k])

        A_ub = coo_matrix((vals, (rows, cols)), shape=(len(active), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * (n + 1)
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs

                    # Line search
                    found = False
                    for frac in [0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        # Adaptively tighten epsilon
        if it % 5 == 0:
            epsilon_frac = max(0.01, epsilon_frac * 0.9)

        if verbose and it % 10 == 0:
            print(f"    SCA it {it}: obj={best_obj:.8f} active={len(active)}", flush=True)

        if no_improve > 30:
            break

    return best_a, best_obj


def lp_mirror_desc(a, max_iters, deadline, verbose=False):
    """
    LP mirror descent: solve LP to find best response direction.
    LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Use only near-active constraints for speed
        threshold = M - 0.02 * M
        active = np.where(c >= threshold)[0]

        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])

        A_ub = coo_matrix((vals, (rows, cols)), shape=(len(active), n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [0.5, 0.3, 0.2, 0.15, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 10 == 0:
            print(f"    LPMD it {it}: obj={best_obj:.8f}", flush=True)

        if no_improve > 25:
            break

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))
        # Perturbed reflections
        inits.append(project_simplex(p[::-1]))
        inits.append(project_simplex(0.7 * p + 0.3 * p[::-1]))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns
    for freq in [1, 2, 3, 5, 7, 11]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    # Two-bump patterns (asymmetric)
    for c1, w1, c2, w2 in [(0.2, 0.1, 0.8, 0.1), (0.1, 0.05, 0.9, 0.05),
                            (0.3, 0.15, 0.7, 0.15), (0.15, 0.08, 0.85, 0.08)]:
        base = np.exp(-((t - c1) / w1) ** 2) + np.exp(-((t - c2) / w2) ** 2) + 0.02
        inits.append(project_simplex(base))

    # Power-law (asymmetric)
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(project_simplex((t + 0.001) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.001) ** alpha))

    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) / (ksize * 0.5)) ** 2)
    kernel /= kernel.sum()
    for _ in range(8):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    # Wiggly patterns
    for freq in [3, 5, 8]:
        for amp in [0.3, 0.5]:
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t) *
                                         np.cos(2 * np.pi * freq * 0.3 * t)))

    # Edge-concentrated
    for edge_w in [0.05, 0.1, 0.2]:
        inits.append(project_simplex(np.exp(-((t - edge_w) / edge_w) ** 2) +
                                     np.exp(-((t - 1 + edge_w) / edge_w) ** 2) + 0.01))

    return inits


def solve_single(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 1: Coordinate descent (40% of time)
    t1 = time.monotonic() + 0.40 * remaining
    if verbose:
        print(f"    CD (budget {t1 - time.monotonic():.0f}s)", flush=True)
    a = coord_descent_exact(a, max_sweeps=50000, deadline=t1,
                            random_order=True, seed=42, n_candidates=10, refine=True)
    try_update(a)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 2: SCA-LP (25% of time)
    t2 = time.monotonic() + 0.25 * remaining
    if verbose:
        print(f"    SCA-LP (budget {t2 - time.monotonic():.0f}s)", flush=True)
    a_sca, obj_sca = sca_lp_adaptive(a, max_iters=200, deadline=t2)
    if obj_sca < best_obj:
        best_a = a_sca.copy()
        best_obj = obj_sca
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 3: LP mirror descent (15% of time)
    t3 = time.monotonic() + 0.15 * remaining
    if verbose:
        print(f"    LP-MD (budget {t3 - time.monotonic():.0f}s)", flush=True)
    a_lp, obj_lp = lp_mirror_desc(a, max_iters=100, deadline=t3)
    if obj_lp < best_obj:
        best_a = a_lp.copy()
        best_obj = obj_lp
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 4: Final CD polish (20% of time)
    t4 = deadline - 3
    a = coord_descent_exact(a, max_sweeps=50000, deadline=t4,
                            random_order=True, seed=777, n_candidates=8, refine=True)
    try_update(a)
    a = best_a.copy()

    # One more SCA on the polished result if time remains
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_sca2, obj_sca2 = sca_lp_adaptive(a, max_iters=50, deadline=deadline - 2)
        if obj_sca2 < best_obj:
            best_a = a_sca2.copy()
            best_obj = obj_sca2

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses exact coordinate descent, adaptive SCA-LP, LP mirror descent,
    and multi-n multi-init exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Deep optimization at parent length ===
    phase1_deadline = t_start + min(500.0, 0.35 * budget_s)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Deep optimization at n={n_parent}", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Try a few inits at parent length
    inits_p1 = generate_inits(n_parent, rng, parent)
    n_tries = min(5, len(inits_p1))
    per_try_time = (phase1_deadline - time.monotonic()) / max(1, n_tries) - 10

    for idx in range(n_tries):
        rem = phase1_deadline - time.monotonic()
        if rem < 30:
            break
        alloc = min(per_try_time, rem - 10)
        if alloc < 20:
            break

        a0 = inits_p1[idx]
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt, obj_opt = solve_single(a0, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  P1 Init {idx}: J={obj_opt:.8f} n={n_parent}", flush=True)

    # === PHASE 2: Multi-length exploration ===
    phase2_deadline = t_start + min(1000.0, 0.70 * budget_s)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Carefully choose n values to try
    base_ns = [384, 512, 640, 768, 896, 1024, 1152, 1280]
    if n_parent not in base_ns:
        base_ns.append(n_parent)
    # Sort by priority: near parent first, then round numbers
    candidate_ns = sorted([n for n in base_ns if n != n_parent and 256 <= n <= 2048],
                          key=lambda x: abs(x - n_parent))

    for new_n in candidate_ns[:4]:
        rem = phase2_deadline - time.monotonic()
        if rem < 80:
            break
        alloc = min(150.0, rem - 30)
        if alloc < 50:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = solve_single(a_ext, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive refinement on global best ===
    phase3_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive refinement (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            # Multiple alternating passes
            a_curr = best_seq.copy()
            passes = [
                ("CD", 0.25),
                ("SCA", 0.20),
                ("CD", 0.20),
                ("LPMD", 0.15),
                ("CD", 0.20),
            ]

            for p_idx, (method, frac) in enumerate(passes):
                rem = phase3_deadline - time.monotonic()
                if rem < 20:
                    break
                alloc = min(frac * (phase3_deadline - t_start) * 0.3, rem - 10)
                if alloc < 15:
                    break

                if method == "CD":
                    print(f"  Pass {p_idx}: CD ({alloc:.0f}s)", flush=True)
                    a_curr = coord_descent_exact(a_curr, max_sweeps=100000,
                                                 deadline=time.monotonic() + alloc,
                                                 random_order=True, seed=100 + p_idx,
                                                 n_candidates=10, refine=True)
                    update_best(a_curr)

                elif method == "SCA":
                    print(f"  Pass {p_idx}: SCA-LP ({alloc:.0f}s)", flush=True)
                    a_opt, obj_opt = sca_lp_adaptive(a_curr, max_iters=200,
                                                     deadline=time.monotonic() + alloc)
                    update_best(a_opt)
                    a_curr = best_seq.copy() if best_seq is not None else a_curr

                elif method == "LPMD":
                    print(f"  Pass {p_idx}: LP-MD ({alloc:.0f}s)", flush=True)
                    a_opt, obj_opt = lp_mirror_desc(a_curr, max_iters=100,
                                                    deadline=time.monotonic() + alloc)
                    update_best(a_opt)
                    a_curr = best_seq.copy() if best_seq is not None else a_curr

            # Final micro-polish: very long CD
            rem = phase3_deadline - time.monotonic()
            if rem > 30:
                print(f"  Final CD polish ({rem - 5:.0f}s)", flush=True)
                a_curr = best_seq.copy()
                a_curr = coord_descent_exact(a_curr, max_sweeps=200000,
                                             deadline=phase3_deadline - 3,
                                             random_order=True, seed=999,
                                             n_candidates=12, refine=True)
                update_best(a_curr)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
