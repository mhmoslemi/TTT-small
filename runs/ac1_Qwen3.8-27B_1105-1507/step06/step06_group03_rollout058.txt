def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k_arr = np.where(cond)[0]
    if len(k_arr) == 0:
        return np.ones(n) / n
    lam = cssv[k_arr[-1]] / (k_arr[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)

    if src.size == 0:

    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                    conv[k_lo:k_hi + 1] += 2.0 * delta * coeffs
                if k_quad >= 0 and k_quad <= 2 * n - 2:
                    conv[k_quad] += delta * delta
                a[i] = x_opt
                s = s - old + x_opt
                best_obj = f_opt
                best_a = a.copy()
                improved = True

        if improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps > 40:
                break

        if verbose and sweep % 200 == 0:
            print(f"    CD-GS sweep {sweep}: obj={best_obj:.8f}", flush=True)

    a = np.maximum(best_a, 0)
    if a.sum() > 1e-10:
        a = a / a.sum()
    return a


def cd_simple(a, deadline, max_sweeps=5000, seed=42):
    """Simple coordinate descent with a few discrete candidates (fast, for quick passes)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    conv = fast_conv(a).copy()
    rng = np.random.default_rng(seed)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            old = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            k_quad = 2 * i if 2 * i < 2 * n - 1 else -1
            a_idx_lo = 0
            a_idx_hi = min(2 * n - 2 - i, n - 1)
            coeffs = a[a_idx_lo:a_idx_hi + 1] if a_idx_hi >= a_idx_lo else np.array([])

            candidates = np.unique(np.concatenate([[0.0], [0.2 * old, 0.5 * old, 0.7 * old,
                                                          1.0 * old, 1.3 * old, 1.7 * old]]))
            best_local = 2.0 * n * conv.max() / (s * s)
            best_val = old

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                if len(coeffs) > 0:
                    conv[k_lo:k_hi + 1] += 2.0 * delta * coeffs
                if k_quad >= 0 and k_quad <= 2 * n - 2:
                    conv[k_quad] += delta * delta
                s_new = s - old + x
                if s_new > 1e-10:
                    obj = 2.0 * n * conv.max() / (s_new * s_new)
                    if obj < best_local:
                        best_local = obj
                        best_val = x
                if len(coeffs) > 0:
                    conv[k_lo:k_hi + 1] -= 2.0 * delta * coeffs
                if k_quad >= 0 and k_quad <= 2 * n - 2:
                    conv[k_quad] -= delta * delta

            if abs(best_val - old) > 1e-12:
                delta = best_val - old
                if len(coeffs) > 0:
                    conv[k_lo:k_hi + 1] += 2.0 * delta * coeffs
                if k_quad >= 0 and k_quad <= 2 * n - 2:
                    conv[k_quad] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        cur_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 50:
            break

    a = np.maximum(best_a, 0)
    if a.sum() > 1e-10:
        a = a / a.sum()
    return a


def lp_mirror_step(a, deadline, max_iters=500, time_per_lp=5.0, verbose=False):
    """LP mirror descent: solve LP for best response direction, then blend."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        c = fast_conv(a)
        M = c.max()

        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

        if len(rows) == 0:
            break

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        t_lp_start = time.monotonic()
        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': time_per_lp, 'presolve': True})
            if not r.success or r.x is None:
                no_improve += 1
                if no_improve > 20:
                    break
                continue

            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs < 1e-10:
                no_improve += 1
                continue
            g = g / gs

            found = False
            for frac in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05]:
                a_try = (1.0 - frac) * a + frac * g
                a_try = np.maximum(a_try, 0)
                try_sum = a_try.sum()
                if try_sum < 1e-10:
                    continue
                a_try = a_try / try_sum
                J_try = compute_J(a_try)
                if J_try < best_obj - 1e-12:
                    a = a_try
                    best_obj = J_try
                    best_a = a_try.copy()
                    no_improve = 0
                    found = True
                    break
            if not found:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

        if verbose and it % 10 == 0:
            print(f"    LP it {it}: obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def sca_lp_step(a, deadline, max_iters=200, time_per_lp=10.0, verbose=False):
    """SCA-LP: Successive Convex Approximation with LP subproblems."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        c = fast_conv(a)
        M = c.max()

        rows = []
        cols = []
        vals = []
        b_ub = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0, None)] * n + [(None, None)]

        A_eq_rows = [0] * n
        A_eq_cols = list(range(n))
        A_eq = coo_matrix((np.ones(n), (A_eq_rows, A_eq_cols)), shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': time_per_lp, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                if g.sum() > 1e-10:
                    g = g / g.sum()
                    found = False
                    for frac in [0.5, 0.3, 0.2, 0.1, 0.05]:
                        a_try = (1.0 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try_sum = a_try.sum()
                        if a_try_sum < 1e-10:
                            continue
                        a_try = a_try / a_try_sum
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 25:
            break

        if verbose and it % 5 == 0:
            print(f"    SCA it {it}: obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences for optimization."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    inits.append(np.ones(n) / n)

    # Fractal-like patterns (related to low PAPR)
    for freq in [1, 2, 3, 4, 6, 8]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))

    # Smooth random
    ksize = max(5, n // 12)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.02))

    # Asymmetric patterns
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1.0 - t) + 0.01) ** alpha))

    # Two-peak patterns
    for center in [0.25, 0.35, 0.5]:
        for width in [0.08, 0.15, 0.25]:
            bump = np.exp(-((t - center) / width) ** 2)
            inits.append(project_simplex(0.3 + bump))

    # Sinc-like (prolate spheroidal approximation)
    for bw in [0.3, 0.5, 0.7]:
        inits.append(project_simplex(np.sinc(bw * (t - 0.5)) ** 2 + 0.05))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: golden section CD + LP mirror + multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_seq = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            try_update(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        try_update(parent)

    # === PHASE 1: Quick optimization at parent length ===
    n_parent = len(parent)
    phase1_time = 150.0
    phase1_deadline = t_start + phase1_time
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Quick CD at n={n_parent} (budget {phase1_time}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    a0 = parent.copy()
    if a0.sum() > 1e-10:
        a0 = a0 / a0.sum()

    # Quick simple CD first
    t_quick = t_start + 40.0
    a_quick = cd_simple(a0, t_quick, max_sweeps=5000, seed=1)
    try_update(a_quick)

    # Then golden section CD
    t_gs1 = phase1_deadline - 5.0
    a_gs1 = coord_descent_gs(a_quick, t_gs1, max_sweeps=100000, golden_evals=20, seed=2)
    try_update(a_gs1)

    # === PHASE 2: Intensive optimization on best ===
    phase2_deadline = t_start + 900.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Intensive CD-GS + LP (budget {phase2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        a_curr = best_seq.copy()
        a_curr = np.maximum(a_curr, 0)
        a_curr = a_curr / a_curr.sum()

        # Alternating CD and LP
        n_cd_blocks = 5
        n_lp_blocks = 3
        block_time = (phase2_deadline - time.monotonic()) / (n_cd_blocks + n_lp_blocks)
        block_time = max(block_time, 20.0)

        for block in range(n_cd_blocks + n_lp_blocks):
            if time.monotonic() > phase2_deadline - 30:
                break
            block_deadline = min(time.monotonic() + block_time, phase2_deadline - 10)
            if block_deadline <= time.monotonic() + 5:
                break

            if block % (n_cd_blocks + n_lp_blocks) < n_cd_blocks or block % 2 == 0:
                # CD block
                print(f"  CD block {block} (budget {block_deadline - time.monotonic():.0f}s)...", flush=True)
                a_curr = coord_descent_gs(a_curr, block_deadline, max_sweeps=100000,
                                          golden_evals=22, seed=100 + block)
                try_update(a_curr)
            else:
                # LP block
                print(f"  LP block {block} (budget {block_deadline - time.monotonic():.0f}s)...", flush=True)
                a_lp, obj_lp = lp_mirror_step(a_curr, block_deadline, max_iters=200,
                                              time_per_lp=8.0)
                try_update(a_lp)
                if obj_lp < best_score:
                    a_curr = a_lp.copy()

        # SCA-LP refinement
        remaining = phase2_deadline - time.monotonic()
        if remaining > 30 and best_seq is not None:
            print(f"  SCA-LP refinement (budget {remaining - 10:.0f}s)...", flush=True)
            a_sca, obj_sca = sca_lp_step(best_seq, phase2_deadline - 10, max_iters=100,
                                         time_per_lp=15.0)
            try_update(a_sca)

    # === PHASE 3: Multi-n exploration ===
    phase3_deadline = t_start + 1250.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Multi-n exploration (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        base_n = best_seq.size
        candidate_ns = []
        for n_try in [256, 320, 384, 512, 640, 768, 896, 1024, 1152, 1280, 1536]:
            if n_try != base_n and n_try <= 2048 and n_try >= 128:
                candidate_ns.append(n_try)
        candidate_ns.sort(key=lambda n: abs(n - base_n))

        for new_n in candidate_ns[:5]:
            remaining = phase3_deadline - time.monotonic()
            if remaining < 40:
                break
            alloc = min(80.0, remaining - 15)
            if alloc < 30:
                break

            print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
            a_ext = resample_seq(best_seq, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()

            a_opt = coord_descent_gs(a_ext, time.monotonic() + alloc,
                                     max_sweeps=100000, golden_evals=20, seed=500 + new_n)
            try_update(a_opt)
            print(f"  n={new_n}: J={compute_J(a_opt):.8f}", flush=True)

    # === PHASE 4: Final intensive polish ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Final polish (budget {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        a_curr = best_seq.copy()
        a_curr = np.maximum(a_curr, 0)
        if a_curr.sum() > 1e-10:
            a_curr = a_curr / a_curr.sum()

        # Long golden section CD
        if remaining > 60:
            t_cd = time.monotonic() + 0.5 * remaining
            print(f"  Final CD-GS (budget {t_cd - time.monotonic():.0f}s)...", flush=True)
            a_final = coord_descent_gs(a_curr, t_cd, max_sweeps=200000,
                                       golden_evals=25, seed=9999)
            try_update(a_final)
            a_curr = best_seq.copy()
            a_curr = a_curr / a_curr.sum()

        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t_lp = time.monotonic() + 0.35 * remaining
            print(f"  Final LP (budget {t_lp - time.monotonic():.0f}s)...", flush=True)
            a_lp, obj_lp = lp_mirror_step(a_curr, t_lp, max_iters=300, time_per_lp=10.0)
            try_update(a_lp)

        # One more quick CD on the LP result
        remaining = deadline - time.monotonic()
        if remaining > 20 and best_seq is not None:
            a_curr = best_seq.copy()
            a_curr = a_curr / a_curr.sum()
            t_last = deadline - 5.0
            print(f"  Last CD (budget {t_last - time.monotonic():.0f}s)...", flush=True)
            a_last = coord_descent_gs(a_curr, t_last, max_sweeps=200000,
                                      golden_evals=22, seed=77777)
            try_update(a_last)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
