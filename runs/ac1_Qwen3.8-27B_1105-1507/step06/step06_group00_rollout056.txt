def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

    n = len(a)

        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def compute_smooth_obj(a, beta):

    conv_max = np.max(conv)

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Efficient algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_gradient_descent(a, time_budget=60.0, beta_start=50.0, beta_end=500.0):

    """

        return a
    a = a / s

    min_lr = 1e-6
    
    max_iters = int(time_budget * 10)  # rough estimate

        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive beta: start small, increase over time
        progress = min(1.0, it / max(1, max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient
        grad = compute_smooth_grad(a, beta)
        
        # Project gradient onto tangent space of simplex (remove mean component)

def solve_lp_minpeak_fast(a, max_solvers=3):
    """Solve min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a = a / np.sum(a)
    
    # Build conv matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers[:max_solvers]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def multi_scale_search(a, time_budget=60.0):
    """
    Multi-scale: solve at low resolution, then refine at full resolution.
    """
    n = len(a)
    start = time.monotonic()
    
    # Downsample to n/4
    n_small = max(50, n // 4)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, n_small)
    a_small = np.interp(x_new, x_old, a)
    a_small = a_small / np.sum(a_small)
    
    # Optimize at small scale
    budget_small = time_budget * 0.3
    a_small_opt = smooth_gradient_descent(a_small, time_budget=budget_small)
    
    # Upsample back
    a_up = np.interp(x_old, x_new, a_small_opt)
    a_up = np.maximum(a_up, 0)
    a_up = a_up / np.sum(a_up)
    
    # Blend with original
    a_blend = 0.5 * a + 0.5 * a_up
    a_blend = a_blend / np.sum(a_blend)
    
    # Refine at full scale
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_opt = smooth_gradient_descent(a_blend, time_budget=remaining)
    else:
        a_opt = a_blend
    
    return a_opt


def structured_init(n, dtype):
    """Generate structured initializations."""
    x = np.linspace(0, 1, n)
    if dtype == 0:
        # Uniform
        a = np.ones(n)
    elif dtype == 1:
        # Cosine window
        a = np.cos(np.pi * (x - 0.5)) ** 2 + 0.1
    elif dtype == 2:
        # Triangular
        a = 1.0 - 2.0 * np.abs(x - 0.5)
        a = a ** 2 + 0.1
    elif dtype == 3:
        # Sinc-like
        k = x * 10 - 5
        a = np.sinc(k / 5) ** 2 + 0.05
    elif dtype == 4:
        # Two bumps
        a = np.exp(-((x - 0.3) ** 2) / 0.05) + np.exp(-((x - 0.7) ** 2) / 0.05) + 0.1
    elif dtype == 5:
        # Flat top with smooth edges
        a = np.ones(n)
        edge = max(2, n // 10)
        for i in range(edge):
            a[i] *= (i + 1) / (edge + 1)
            a[n-1-i] *= (i + 1) / (edge + 1)
    elif dtype == 6:
        # Based on Ramanujan sum structure
        a = np.ones(n)
        for k in range(1, min(n//2, 20)):
            a += 0.5 * np.cos(2 * np.pi * k * x) * np.exp(-k * 0.2)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smooth gradient descent
(log-sum-exp approximation), LP alternating polish, stochastic local search,
multi-scale refinement, and diverse structured initializations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def compute_smooth_and_grad(a, beta):
    """
    Compute smooth objective value and gradient using log-sum-exp.
    Assumes sum(a) = 1.
    Returns (smooth_obj, gradient).
    """
    n = len(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    shifted = conv - conv_max
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / np.sum(exp_vals)
    
    # Smooth objective: (2*n/beta) * log(sum(exp(beta*conv))) 
    # = (2*n/beta) * (beta*conv_max + log(sum(exp(beta*(conv-conv_max)))))
    # = 2*n*conv_max + (2*n/beta)*log(sum(w))
    # = 2*n*conv_max + (2*n/beta)*log(1)  ... no wait
    # log(sum(exp(beta*conv))) = beta*conv_max + log(sum(exp(beta*(conv-conv_max))))
    # = beta*conv_max + log(sum(w) * sum(exp(...)))... 
    # Actually: sum(exp(beta*conv)) = exp(beta*conv_max) * sum(exp(beta*(conv-conv_max)))
    # log(sum(exp(beta*conv))) = beta*conv_max + log(sum(exp(beta*(conv-conv_max))))
    # = beta*conv_max + log(sum(exp_vals))
    # = beta*conv_max + log(n_eff) where n_eff = sum(exp_vals)
    # But w = exp_vals / sum(exp_vals), so sum(exp_vals) = 1/mean(w) = sum(exp_vals)
    # Let's just compute: log_sum = conv_max + log(np.sum(exp_vals)) / beta... 
    # No: smooth_obj = (2*n/beta) * (beta*conv_max + log(np.sum(exp_vals)))
    # = 2*n*conv_max + (2*n/beta)*log(np.sum(exp_vals))
    
    log_sum_exp = beta * conv_max + np.log(np.sum(exp_vals))
    smooth_obj = (2.0 * n / beta) * log_sum_exp
    
    # Gradient: grad[i] = 4*n * sum_j w[i+j] * a[j]
    # Cross-correlation of w (len 2n-1) and a (len n)
    N = 1
    while N < 3 * n:
        N <<= 1
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)
    grad = 4.0 * n * corr[:n]
    
    return smooth_obj, grad


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = np.arange(1, n + 1) * u > (cssv - 1)
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_gradient_descent(a, time_budget=60.0, beta_start=20.0, beta_end=500.0):
    """
    Smooth gradient descent with log-sum-exp approximation, projected onto simplex.
    Adaptive learning rate with backtracking.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.05
    min_lr = 1e-8
    stall = 0
    max_stall = 30

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / (time_budget * 0.9))
        beta = beta_start * (beta_end / beta_start) ** progress

        smooth_val, grad = compute_smooth_and_grad(a, beta)
        
        # Project gradient onto tangent space of simplex
        grad_tangent = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_tangent)
        if grad_norm < 1e-12:
            break

        # Backtracking line search
        found = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr_try = lr * factor
            a_new = project_simplex(a - lr_try * grad_tangent)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                lr = min(lr * 1.1, 0.2)
                stall = 0
                found = True
                break

        if not found:
            lr *= 0.5
            stall += 1
            if lr < min_lr:
                break
            if stall > max_stall:
                # Perturb to escape
                rng_local = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                a = best_a * np.exp(rng_local.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = 0.05
                stall = 0

    return best_a


def solve_lp_minpeak(a, max_solvers=3):
    """Solve min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a = a / np.sum(a)

    # Build convolution matrix A (2n-1 x n) where A @ g = conv(a, g)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers[:max_solvers]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def stochastic_local_search(a, time_budget=30.0, seed=12345):
    """Stochastic local search with simulated annealing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    temperature = 0.0005
    cool_rate = 0.998

    while time.monotonic() - start < time_budget:
        pert_type = rng.integers(0, 5)
        a_trial = a.copy()

        if pert_type == 0:
            # Move mass between two random positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-6:
                delta = a[i] * rng.uniform(0.05, 0.3)
                a_trial[i] -= delta
                a_trial[j] += delta
        elif pert_type == 1:
            # Small multiplicative perturbation
            a_trial = a * np.exp(rng.normal(0, 0.03, n))
        elif pert_type == 2:
            # Symmetric swap
            i = rng.integers(0, n // 2)
            j = n - 1 - i
            if i != j:
                diff = a[i] - a[j]
                delta = diff * rng.uniform(0.1, 0.5)
                a_trial[i] -= delta
                a_trial[j] += delta
        elif pert_type == 3:
            # Block perturbation
            block_size = int(rng.integers(3, max(4, n // 15)))
            i = int(rng.integers(0, n - block_size))
            block = a_trial[i:i + block_size]
            block = block * np.exp(rng.normal(0, 0.15, block_size))
            a_trial[i:i + block_size] = block
        else:
            # Targeted: move mass away from peak contributors
            conv = compute_conv(a)
            k_star = int(np.argmax(conv))
            i = int(rng.integers(0, n))
            j = k_star - i
            if 0 <= j < n and j != i and a[i] > a[j] * 2:
                delta = (a[i] - a[j]) * rng.uniform(0.1, 0.4)
                a_trial[i] -= delta
                a_trial[j] += delta
            else:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j and a[i] > 1e-6:
                    delta = a[i] * 0.1
                    a_trial[i] -= delta
                    a_trial[j] += delta

        a_trial = np.maximum(a_trial, 0)
        s_trial = float(np.sum(a_trial))
        if s_trial < 1e-10:
            continue
        a_trial = a_trial / s_trial

        obj_trial = compute_obj(a_trial)
        delta_obj = obj_trial - current_obj

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(temperature, 1e-10)):
            a = a_trial
            current_obj = obj_trial
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()

        temperature *= cool_rate

    return best_a


def multi_scale_refine(a, time_budget=60.0):
    """Multi-scale: optimize at low resolution, upsample, refine at full resolution."""
    n = len(a)
    start = time.monotonic()

    factors = [4, 3, 2]
    current = a.copy()

    for factor in factors:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        n_small = max(30, n // factor)
        x_old = np.linspace(0, 1, len(current))
        x_new = np.linspace(0, 1, n_small)
        a_small = np.interp(x_new, x_old, current)
        a_small = np.maximum(a_small, 0)
        a_small = a_small / np.sum(a_small)

        budget_small = min(remaining * 0.4, 20.0)
        a_small_opt = smooth_gradient_descent(a_small, time_budget=budget_small,
                                              beta_start=10.0, beta_end=100.0)

        # Upsample
        x_full = np.linspace(0, 1, len(current))
        a_up = np.interp(x_full, x_new, a_small_opt)
        a_up = np.maximum(a_up, 0)
        a_up = a_up / np.sum(a_up)

        # Blend
        current = 0.5 * current + 0.5 * a_up
        current = current / np.sum(current)

    # Final refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        current = smooth_gradient_descent(current, time_budget=remaining,
                                          beta_start=50.0, beta_end=500.0)

    return current


def structured_init(n, dtype, rng=None):
    """Generate structured initializations of length n."""
    x = np.linspace(0, 1, n)
    if dtype == 0:
        a = np.ones(n)
    elif dtype == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2 + 0.1
    elif dtype == 2:
        a = (1.0 - 2.0 * np.abs(x - 0.5)) ** 2 + 0.1
    elif dtype == 3:
        k = x * 8 - 4
        a = np.sinc(k / 4) ** 2 + 0.05
    elif dtype == 4:
        a = np.exp(-((x - 0.3) ** 2) / 0.03) + np.exp(-((x - 0.7) ** 2) / 0.03) + 0.1
    elif dtype == 5:
        a = np.ones(n)
        edge = max(2, n // 8)
        for i in range(edge):
            a[i] *= (i + 1) / (edge + 1)
            a[n - 1 - i] *= (i + 1) / (edge + 1)
    elif dtype == 6:
        a = np.ones(n)
        for k in range(1, min(n // 3, 30)):
            a += 0.3 * np.cos(2 * np.pi * k * x) * np.exp(-k * 0.15)
        a = np.maximum(a, 0.01)
    elif dtype == 7:
        # Step function with few levels
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.integers(1, 5, n).astype(float)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a


def pairwise_local_search(a, time_budget=30.0):
    """
    Targeted local search: find peak of convolution, then try to reduce it
    by moving mass from contributing positions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99999)

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from positions that contribute to conv[k_star]
        for _ in range(10):
            i = int(rng.integers(0, n))
            j = k_star - i
            if 0 <= j < n and j != i:
                if a[i] > a[j] + 1e-8:
                    for frac in [0.05, 0.1, 0.2, 0.3]:
                        delta = frac * (a[i] - a[j])
                        a[i] -= delta
                        a[j] += delta
                        a[j2] = a[j]  # no-op, just for clarity
                        s_check = float(np.sum(a))
                        if s_check > 1e-10:
                            a_norm = a / s_check
                            obj_check = compute_obj(a_norm)
                            if obj_check < best_obj - 1e-12:
                                best_obj = obj_check
                                best_a = a.copy()
                                improved = True
                                break
                            else:
                                a[i] += delta
                                a[j] -= delta
                                break
                else:
                    # Try in other direction
                    for frac in [0.05, 0.1, 0.2]:
                        delta = frac * (a[j] - a[i])
                        a[j] -= delta
                        a[i] += delta
                        s_check = float(np.sum(a))
                        if s_check > 1e-10:
                            a_norm = a / s_check
                            obj_check = compute_obj(a_norm)
                            if obj_check < best_obj - 1e-12:
                                best_obj = obj_check
                                best_a = a.copy()
                                improved = True
                                break
                            else:
                                a[j] += delta
                                a[i] -= delta
                                break

        if not improved and it > 100:
            # Perturb and continue
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            best_obj = compute_obj(a)

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining smooth gradient descent, LP alternating,
    chase-peak, stochastic local search, multi-scale, and diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Smooth gradient descent ===
    print(f"\n=== Phase 1: Smooth Gradient Descent ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.2, 180.0)
    a1 = smooth_gradient_descent(best_seq.copy(), time_budget=alloc,
                                  beta_start=15.0, beta_end=400.0)
    obj1 = compute_obj(a1)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        n_best = len(best_seq)
        print(f"  Grad descent: obj={best_score:.8f} **")
    else:
        print(f"  Grad descent: obj={obj1:.8f}")

    # === Phase 2: LP Alternating ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: LP Alternating ===")
        alloc = min(remaining * 0.15, 120.0)
        a2 = lp_alternating(best_seq.copy(), time_budget=alloc, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            n_best = len(best_seq)
            print(f"  LP alt: obj={best_score:.8f} **")
        else:
            print(f"  LP alt: obj={obj2:.8f}")

    # === Phase 3: Chase peak ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 3: Chase Peak ===")
        alloc = min(remaining * 0.08, 60.0)
        a3 = chase_peak(best_seq.copy(), time_budget=alloc)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            n_best = len(best_seq)
            print(f"  Chase peak: obj={best_score:.8f} **")
        else:
            print(f"  Chase peak: obj={obj3:.8f}")

    # === Phase 4: Smooth gradient round 2 (higher beta) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: Smooth Gradient Round 2 ===")
        alloc = min(remaining * 0.15, 120.0)
        a4 = smooth_gradient_descent(best_seq.copy(), time_budget=alloc,
                                      beta_start=50.0, beta_end=1500.0)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            n_best = len(best_seq)
            print(f"  Grad descent 2: obj={best_score:.8f} **")
        else:
            print(f"  Grad descent 2: obj={obj4:.8f}")

    # === Phase 5: Pairwise local search ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Pairwise Local Search ===")
        alloc = min(remaining * 0.08, 50.0)
        a5 = pairwise_local_search(best_seq.copy(), time_budget=alloc)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            n_best = len(best_seq)
            print(f"  Pairwise: obj={best_score:.8f} **")
        else:
            print(f"  Pairwise: obj={obj5:.8f}")

    # === Phase 6: Stochastic local search ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Stochastic Local Search ===")
        alloc = min(remaining * 0.1, 60.0)
        a6 = stochastic_local_search(best_seq.copy(), time_budget=alloc, seed=777)

    remaining = deadline - time.monotonic()

        print(f"\n=== Phase 7: Multi-Scale ===")
        alloc = min(remaining * 0.12, 80.0)
        a7 = multi_scale_refine(best_seq.copy(), time_budget=alloc)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            n_best = len(best_seq)
            print(f"  Multi-scale: obj={best_score:.8f} **")
        else:
            print(f"  Multi-scale: obj={obj7:.8f}")

    # === Phase 8: Diverse initializations ===
    print(f"\n=== Phase 8: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    for dtype in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_init = structured_init(n_best, dtype, rng)
        alloc = min(remaining * 0.08, 30.0)
        a_d = smooth_gradient_descent(a_init, time_budget=alloc,
                                       beta_start=15.0, beta_end=200.0)
        obj_d = compute_obj(a_d)
        if obj_d < best_score:
            best_score = obj_d
            best_seq = a_d.copy()
            n_best = len(best_seq)
            print(f"  Init type {dtype}: obj={best_score:.8f} **")
        else:
            print(f"  Init type {dtype}: obj={obj_d:.8f}")

    # === Phase 9: Multi-n exploration ===
    print(f"\n=== Phase 9: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        max(300, n_best - 100), max(300, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200,
        600, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)

        alloc = min(remaining * 0.12, 35.0)
        a_n = smooth_gradient_descent(a_try, time_budget=alloc,
                                       beta_start=20.0, beta_end=300.0)
        obj_n = compute_obj(a_n)
        if obj_n < best_score:
            best_score = obj_n
            best_seq = a_n.copy()
            n_best = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_n:.8f}")

    # === Phase 10: Final LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 10: Final LP Polish ===")
        alloc = min(remaining * 0.25, 60.0)
        a_final = lp_alternating(best_seq.copy(), time_budget=alloc)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_f:.8f}")

    # === Phase 11: Final gradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_f2 = smooth_gradient_descent(best_seq.copy(), time_budget=min(remaining * 0.3, 30.0),
                                        beta_start=100.0, beta_end=3000.0)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final grad: obj={best_score:.8f} **")

    # === Phase 12: Final local search ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_f3 = stochastic_local_search(best_seq.copy(), time_budget=min(remaining * 0.5, 20.0),
                                         seed=4242)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final stochastic: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
