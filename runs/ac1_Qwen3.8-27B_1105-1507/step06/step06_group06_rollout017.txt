"""
Spectral shaping genetic algorithm for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Core idea: iteratively flatten the power spectrum of the sequence while maintaining
non-negativity, using a population-based search with condensed explosion mutation
and frequency-domain crossover.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def spectral_shape(a, alpha=0.7, blend=0.3, N=None):
    """Apply spectral shaping: compress the power spectrum."""
    n = len(a)
    if N is None:
        N = 1
        while N < 2 * n:
            N <<= 1
    
    F = fft(a, N)
    mag = np.abs(F)
    phase = np.angle(F)
    
    # Power compression
    mag_max = np.max(mag)
    if mag_max < 1e-15:
        return a.copy()
    
    mag_norm = mag / mag_max
    mag_new = mag_norm ** alpha
    mag_new = (1 - blend) * mag + blend * mag_new * mag_max
    
    F_new = mag_new * np.exp(1j * phase)
    a_new = ifft(F_new).real[:n]
    return project_nonneg_sum1(a_new)

def spectral_flatten(a, t=0.1, N=None):
    """Blend toward flat spectrum."""
    n = len(a)
    if N is None:
        N = 1
        while N < 2 * n:
            N <<= 1
    
    F = fft(a, N)
    mag = np.abs(F)
    phase = np.angle(F)
    
    # Target: flat magnitude
    # Use DC component as target (preserves energy)
    target_mag = np.zeros(N)
    target_mag[0] = mag[0]  # preserve DC
    # For non-DC, target is the mean of non-DC magnitudes
    non_dc = mag[1:]
    target_non_dc = np.mean(non_dc) if len(non_dc) > 0 else 0
    target_mag[1:] = target_non_dc
    
    mag_new = (1 - t) * mag + t * target_mag
    F_new = mag_new * np.exp(1j * phase)
    a_new = ifft(F_new).real[:n]
    return project_nonneg_sum1(a_new)

def condensed_explosion(a, rng, frac=0.1):
    """Remove low-magnitude elements and redistribute their mass."""
    n = len(a)
    a = a.copy()
    n_remove = max(1, int(n * frac))
    
    # Remove the lowest elements
    sorted_idx = np.argsort(a)
    remove_idx = sorted_idx[:n_remove]
    mass = a[remove_idx].sum()
    a[remove_idx] = 0
    
    # Redistribute to random positions (or to neighbors of removed)
    n_add = n_remove
    add_idx = rng.choice(n, size=n_add, replace=False)
    for idx in add_idx:
        a[idx] += mass / n_add
    
    return project_nonneg_sum1(a)

def freq_crossover(a1, a2, rng, N=None):
    """Crossover in frequency domain."""
    n = len(a1)
    if N is None:
        N = 1
        while N < 2 * n:
            N <<= 1
    
    F1 = fft(a1, N)
    F2 = fft(a2, N)
    
    # Random crossover mask
    t = rng.random()
    F_new = t * F1 + (1 - t) * F2
    
    a_new = ifft(F_new).real[:n]
    return project_nonneg_sum1(a_new)

def chase_peak_ff(a, n_steps=50, step_size=0.05):
    """Chase the peak using FFT for speed."""
    n = len(a)

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if n > 1500:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_nonneg_sum1(a_new)

def genetic_algorithm_population(population, objs, rng, n_gen=20, time_budget=60.0):
    """
    Population-based GA with spectral operations.
    """
    start = time.monotonic()
    best_idx = np.argmin(objs)
    best_a = population[best_idx].copy()
    best_obj = objs[best_idx]
    pop_size = len(population)
    n = len(population[0])
    
    N = 1
    while N < 2 * n:
        N <<= 1
    
    gen = 0
    while time.monotonic() - start < time_budget and gen < n_gen:
        gen += 1
        new_pop = []
        new_objs = []
        
        # Keep elites (top 30%)
        n_elite = max(1, pop_size // 3)
        sorted_idx = np.argsort(objs)
        for i in range(n_elite):
            new_pop.append(population[sorted_idx[i]].copy())
            new_objs.append(objs[sorted_idx[i]])
        
        # Generate offspring
        while len(new_pop) < pop_size:
            if time.monotonic() - start > time_budget:
                break
            
            # Selection: tournament of 3
            idxs = rng.choice(pop_size, size=3, replace=False)
            parent_idx = idxs[np.argmin(objs[idxs])]
            
            # Choose operation
            op = rng.integers(0, 5)
            
            if op == 0 and len(new_pop) >= 1:
                # Frequency crossover with another parent
                other_idx = rng.choice(pop_size)
                child = freq_crossover(population[parent_idx], population[other_idx], rng, N)
            elif op == 1:
                # Spectral shaping (power compression)
                alpha = rng.uniform(0.5, 0.9)
                blend = rng.uniform(0.1, 0.5)
                child = spectral_shape(population[parent_idx], alpha=alpha, blend=blend, N=N)
            elif op == 2:
                # Spectral flattening
                t = rng.uniform(0.05, 0.3)
                child = spectral_flatten(population[parent_idx], t=t, N=N)
            elif op == 3:
                # Condensed explosion
                frac = rng.uniform(0.05, 0.2)
                child = condensed_explosion(population[parent_idx], rng, frac=frac)
            else:
                # Gaussian perturbation
                noise = rng.normal(0, 0.05, n)
                child = project_nonneg_sum1(population[parent_idx] * np.exp(noise))
            
            # Quick local optimization
            child, child_obj = chase_peak_ff(child, n_steps=10, step_size=0.03)
            
            new_pop.append(child)
            new_objs.append(child_obj)
        
        population = new_pop[:pop_size]
        objs = np.array(new_objs[:pop_size])
        
        new_best = np.argmin(objs)
        if objs[new_best] < best_obj:
            best_obj = objs[new_best]
            best_a = population[new_best].copy()

def main_optimize(a, rng, time_budget=100.0):

    n = len(a)

    start = time.monotonic()
    
    strategies = []
    
    # Strategy: spectral shaping with varying parameters
    for alpha in [0.5, 0.6, 0.7, 0.8]:
        for blend in [0.1, 0.2, 0.3, 0.5]:
            strategies.append(('shape', alpha, blend))
    
    # Strategy: spectral flattening
    for t in [0.05, 0.1, 0.2, 0.3]:
        strategies.append(('flatten', t, 0))
    
    # Strategy: chase peak
    strategies.append(('chase', 0, 0))
    
    # Strategy: condensed explosion
    strategies.append(('explosion', 0, 0))
    
    for strat in strategies:
        if time.monotonic() - start > time_budget:
            break
        
        if strat[0] == 'shape':
            _, alpha, blend = strat
            a_new = spectral_shape(best_a, alpha=alpha, blend=blend)
        elif strat[0] == 'flatten':
            _, t, _ = strat
            a_new = spectral_flatten(best_a, t=t)
        elif strat[0] == 'chase':
            a_new, _ = chase_peak_ff(best_a, n_steps=20, step_size=0.03)
        elif strat[0] == 'explosion':
            a_new = condensed_explosion(best_a, rng, frac=0.1)
            a_new, _ = chase_peak_ff(a_new, n_steps=10, step_size=0.02)
        else:
            continue
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
    
    return best_a, best_obj

"""
Spectral shaping genetic algorithm for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines population-based GA with frequency-domain crossover, condensed explosion
mutation, spectral flattening, LP polishing, and adaptive multi-length search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def spectral_shape(a, alpha=0.7, blend=0.3, N=None):
    """Apply spectral power compression to flatten the convolution."""
    n = len(a)
    if N is None:
        N = 1
        while N < 2 * n:
            N <<= 1
    F = fft(a, N)
    mag = np.abs(F)
    phase = np.angle(F)
    mag_max = np.max(mag)
    if mag_max < 1e-15:
        return a.copy()
    mag_norm = mag / mag_max
    mag_compressed = mag_norm ** alpha
    mag_new = (1.0 - blend) * mag + blend * mag_compressed * mag_max
    F_new = mag_new * np.exp(1j * phase)
    a_new = ifft(F_new).real[:n]
    return project_nonneg_sum1(a_new)


def spectral_flatten(a, t=0.1, N=None):
    """Blend spectrum toward flat magnitude (energy-preserving)."""
    n = len(a)
    if N is None:
        N = 1
        while N < 2 * n:
            N <<= 1
    F = fft(a, N)
    mag = np.abs(F)
    phase = np.angle(F)
    target_mag = np.zeros(N)
    target_mag[0] = mag[0]
    non_dc = mag[1:]
    target_non_dc = np.mean(non_dc) if len(non_dc) > 0 else 0.0
    target_mag[1:] = target_non_dc
    mag_new = (1.0 - t) * mag + t * target_mag
    F_new = mag_new * np.exp(1j * phase)
    a_new = ifft(F_new).real[:n]
    return project_nonneg_sum1(a_new)


def spectral_bandpass(a, lo_frac=0.1, hi_frac=0.4, boost=1.3, N=None):
    """Boost mid-frequency components to reshape the time-domain profile."""
    n = len(a)
    if N is None:
        N = 1
        while N < 2 * n:
            N <<= 1
    F = fft(a, N)
    mag = np.abs(F)
    phase = np.angle(F)
    lo = max(1, int(N * lo_frac))
    hi = min(N // 2, int(N * hi_frac))
    boost_vec = np.ones(N)
    boost_vec[lo:hi] = boost
    boost_vec[N - hi + 1:N - lo + 1] = boost
    mag_new = mag * boost_vec
    F_new = mag_new * np.exp(1j * phase)
    a_new = ifft(F_new).real[:n]
    return project_nonneg_sum1(a_new)


def condensed_explosion(a, rng, frac=0.1):
    """Remove lowest-magnitude elements and redistribute their mass randomly."""
    n = len(a)
    a = a.copy()
    n_remove = max(1, int(n * frac))
    sorted_idx = np.argsort(a)
    remove_idx = sorted_idx[:n_remove]
    mass = a[remove_idx].sum()
    a[remove_idx] = 0.0
    n_add = n_remove
    add_idx = rng.choice(n, size=n_add, replace=False)
    for idx in add_idx:
        a[idx] += mass / n_add
    return project_nonneg_sum1(a)


def freq_crossover(a1, a2, rng, N=None):
    """Crossover in frequency domain with random blend."""
    n = len(a1)
    if N is None:
        N = 1
        while N < 2 * n:
            N <<= 1
    F1 = fft(a1, N)
    F2 = fft(a2, N)
    t = rng.random()
    F_new = t * F1 + (1.0 - t) * F2
    a_new = ifft(F_new).real[:n]
    return project_nonneg_sum1(a_new)


def chase_peak_ff(a, n_steps=50, step_size=0.04, top_k=1, time_budget=10.0):
    """
    Chase-the-peak with optional top-k softening.
    Gradient of conv[k] wrt a[i] = 2*a[k-i] if 0<=k-i<n, else 0.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        conv = irfft(fa * fa, N)[:2 * n - 1]

        if top_k == 1:
            k_star = int(np.argmax(conv))
            grad = np.zeros(n)
            j = k_star - np.arange(n)
            valid = (j >= 0) & (j < n)
            grad[valid] = 2.0 * a[j[valid]]
        else:
            # Soft top-k: weight by convexity of conv
            cmax = np.max(conv)
            cmin = np.min(conv)
            if cmax - cmin > 1e-15:
                weights = (conv - cmin) / (cmax - cmin)
                weights = weights ** 3.0  # sharpen toward peaks
            else:
                weights = np.full(len(conv), 1.0 / len(conv))
            weights /= np.sum(weights)
            grad = np.zeros(n)
            for k_idx in range(len(conv)):
                if weights[k_idx] < 1e-10:
                    continue
                j = k_idx - np.arange(n)
                valid = (j >= 0) & (j < n)
                grad[valid] += weights[k_idx] * 2.0 * a[j[valid]]

        grad_centered = grad - np.mean(grad)
        a_try = a - step_size * grad_centered
        a_try = project_nonneg_sum1(a_try)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if n > 1200:
        return None, None
    try:
        A = build_toeplitz(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0.0, None)] * n + [(0.0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 1200:
        return None
    try:
        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_nonneg_sum1(a_new)


def try_load_parent():

def genetic_algorithm_population(population, objs, rng, n_gen=80, time_budget=120.0):
    """
    Population-based GA with spectral operations and condensed explosion.
    """
    start = time.monotonic()
    best_idx = int(np.argmin(objs))
    best_a = population[best_idx].copy()
    best_obj = float(objs[best_idx])
    pop_size = len(population)
    n = len(population[0])

    N = 1
    while N < 2 * n:
        N <<= 1

    gen = 0
    while time.monotonic() - start < time_budget and gen < n_gen:
        gen += 1
        new_pop = []
        new_objs = []

        # Elitism: keep top 25%
        n_elite = max(2, pop_size // 4)
        sorted_idx = np.argsort(objs)
        for i in range(n_elite):
            new_pop.append(population[sorted_idx[i]].copy())
            new_objs.append(float(objs[sorted_idx[i]]))

        # Generate offspring
        while len(new_pop) < pop_size:
            if time.monotonic() - start > time_budget:
                break

            # Tournament selection
            idxs = rng.choice(pop_size, size=min(3, pop_size), replace=False)
            parent_idx = int(idxs[np.argmin(objs[idxs])])

            # Choose operation
            op = int(rng.integers(0, 6))

            if op == 0:
                # Frequency crossover
                other_idx = int(rng.integers(pop_size))
                child = freq_crossover(population[parent_idx], population[other_idx], rng, N)
            elif op == 1:
                # Spectral power compression
                alpha = float(rng.uniform(0.4, 0.85))
                blend = float(rng.uniform(0.1, 0.5))
                child = spectral_shape(population[parent_idx], alpha=alpha, blend=blend, N=N)
            elif op == 2:
                # Spectral flattening
                t = float(rng.uniform(0.05, 0.35))
                child = spectral_flatten(population[parent_idx], t=t, N=N)
            elif op == 3:
                # Condensed explosion
                frac = float(rng.uniform(0.05, 0.2))
                child = condensed_explosion(population[parent_idx], rng, frac=frac)
            elif op == 4:
                # Bandpass boost
                lo = float(rng.uniform(0.05, 0.2))
                hi = float(rng.uniform(0.3, 0.55))
                boost = float(rng.uniform(1.1, 1.6))
                child = spectral_bandpass(population[parent_idx], lo, hi, boost, N)
            else:
                # Gaussian multiplicative perturbation
                noise = rng.normal(0, 0.08, n)
                child = project_nonneg_sum1(population[parent_idx] * np.exp(noise))

            # Quick local optimization
            child, child_obj = chase_peak_ff(child, n_steps=8, step_size=0.03, time_budget=3.0)
            new_pop.append(child)
            new_objs.append(child_obj)

        population = new_pop[:pop_size]
        objs = np.array(new_objs[:pop_size])

        new_best = int(np.argmin(objs))
        if objs[new_best] < best_obj:
            best_obj = float(objs[new_best])
            best_a = population[new_best].copy()

    return best_a, best_obj


def lp_alternate_step(a, time_limit=12):
    """One LP alternating step: find optimal g, then blend."""
    conv = compute_conv(a)
    M = float(np.max(conv))
    g = solve_lp_max_sum(a, M, time_limit=time_limit)
    if g is None:
        return None, None
    best_t_obj = np.inf
    best_t_a = a.copy()
    for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_blend = project_nonneg_sum1((1.0 - t_val) * a + t_val * g)
        obj_blend = compute_obj(a_blend)
        if obj_blend < best_t_obj:
            best_t_obj = obj_blend
            best_t_a = a_blend.copy()
    return best_t_a, best_t_obj


def spiky_gradient_descent(a, rng, max_iters=200, time_budget=30.0):
    """
    Subgradient-style method with adaptive top-k peaking.
    Uses softmax-weighted peaks for smoother gradient.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    step_size = 0.08
    momentum = np.zeros(n)
    mom_factor = 0.8

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        conv = irfft(fa * fa, N)[:2 * n - 1]
        cmax = float(np.max(conv))

        if cmax < 1e-15:
            break

        # Softmax weights over convolution peaks
        tau = 0.01 + 0.02 * (it / max_iters)
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute weighted gradient
        grad = np.zeros(n)
        for k_idx in range(len(conv)):
            w = weights[k_idx]
            if w < 1e-12:
                continue
            j = k_idx - np.arange(n)
            valid = (j >= 0) & (j < n)
            grad[valid] += w * 2.0 * a[j[valid]]

        grad_centered = grad - np.mean(grad)
        momentum = mom_factor * momentum - step_size * grad_centered
        a_new = project_nonneg_sum1(a - momentum)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

        step_size *= 0.997

    return best_a, best_obj

            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
