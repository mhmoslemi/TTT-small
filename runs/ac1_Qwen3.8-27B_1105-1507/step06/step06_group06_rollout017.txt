"""
Optimized self-convolution peak minimization using:
1. FFT-based exact subgradient with Nesterov momentum
2. Frequency-domain flat-magnitude (Gerchberg-Saxton) refinement
3. Cyclic coordinate descent on individual elements
4. Multi-scale annealing with adaptive soft-max temperature
5. Multi-length exploration around the parent length

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute full linear convolution a*a."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    N = 1
    while N < size:
        N <<= 1
    fa = rfft(a, N)
    conv = irfft(fa * fa, N)
    return conv[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def normalize_pos(a):
    """Project to non-negative, sum=1."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def grad_weighted_conv(a, w):
    """
    Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad[m] = 2 * sum_k w[k] * a[k-m] (for valid k-m in [0,n)).
    Computed via FFT cross-correlation.
    """
    n = len(a)
    L = len(w)
    N = 1
    while N < L + n - 1:
        N <<= 1
    fw = rfft(w, N)
    fa = rfft(a, N)
    grad = 2.0 * irfft(fw * np.conj(fa), N)[:n]
    return grad


def subgradient_step(a, tau, lr, momentum_buf, momentum=0.9):
    """
    One step of subgradient descent with momentum.
    a: current sequence (sum=1, non-negative)
    tau: soft-max temperature
    lr: learning rate
    momentum_buf: previous velocity buffer
    Returns: (new_a, new_momentum_buf, grad)
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    if cmax < 1e-15:
        return a, momentum_buf, np.zeros(n)

    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -50.0, 50.0)
    w = np.exp(shifted)
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws

    grad = grad_weighted_conv(a, w)
    # Center the gradient (removes the constant component that doesn't affect direction on simplex)
    grad = grad - np.mean(grad)

    # Nesterov momentum
    velocity = momentum * momentum_buf - lr * grad
    a_new = a + velocity
    a_new = normalize_pos(a_new)

    return a_new, velocity, grad


def freq_flat_magnitude(a, n_iters=200, target='flat', time_budget=20.0):
    """
    Gerchberg-Saxton: iteratively flatten the magnitude of FFT(a)
    while maintaining a >= 0 in time domain.
    """
    n = len(a)
    a = normalize_pos(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        phase = np.angle(fa)
        mag = np.abs(fa)

        if target == 'flat':
            # Target: constant magnitude (DC dominates for non-negative a)
            # Use geometric mean with constant to gradually flatten
            mag_target = mag * 0.5 + np.full_like(mag, np.mean(mag)) * 0.5
        elif target == 'soft_flat':
            # Power compression
            mag_norm = mag / (np.max(mag) + 1e-300)
            mag_target = (mag_norm ** 0.5) * np.max(mag)
        else:
            mag_target = mag.copy()

        # Blend current and target magnitude
        blend = 0.3 + 0.5 * (it / max(n_iters, 1))
        mag_new = (1 - blend) * mag + blend * mag_target

        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, N)[:n]
        a_new = normalize_pos(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def freq_shaped_target(a, n_iters=150, time_budget=15.0):
    """
    Frequency domain: shape the magnitude to minimize peak autocorrelation.
    Uses iterative amplitude adjustment based on where the peak is in time domain.
    """
    n = len(a)
    a = normalize_pos(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        conv = irfft(fa * fa, N)[:2 * n - 1]
        cmax = np.max(conv)
        k_peak = int(np.argmax(conv))

        phase = np.angle(fa)
        mag = np.abs(fa)

        # Reduce magnitude at frequencies that contribute to the peak
        # The peak at position k_peak in autocorrelation corresponds to
        # frequency components exp(2*pi*i*f*k_peak/N)
        # Reduce overall magnitude at high frequencies (they create more structure)
        freq_idx = np.arange(len(mag))
        # Apply a low-pass to reduce high-frequency content
        lowpass = np.exp(-(freq_idx / (N * 0.3)) ** 2)
        # Blend: keep some high frequency for sharpness
        alpha = 0.3 + 0.4 * (it / max(n_iters, 1))
        mag_new = mag * ((1 - alpha) + alpha * lowpass)

        # Also apply soft flattening
        mag_mean = np.mean(mag_new)
        mag_new = mag_new * 0.7 + mag_mean * 0.3

        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, N)[:n]
        a_new = normalize_pos(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def cyclic_descent(a, n_sweeps=20, time_budget=10.0):
    """
    Cyclic coordinate descent: optimize each element one at a time.
    For element i, the objective as a function of a[i] is:
    f(a[i]) = 2n * max_k (conv_without_i[k] + 2*a[i]*a[k-i] + ...) / (sum)^2
    Simplified: for the peak constraint, find the max allowed a[i].
    """
    n = len(a)
    a = normalize_pos(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break
        for i in range(n):
            if time.monotonic() - start > time_budget:
                    break
            # Try adjusting a[i]
            orig_val = a[i]
            if orig_val < 1e-15:
                continue

            # Compute the "partial" convolution without a[i]'s contribution
            # conv[k] = sum_j a[j]*a[k-j]
            # Terms involving a[i]: a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i]
            # (when k-i is valid, i.e., 0 <= k-i < n)
            # Also the term a[i]*a[i] when k = 2i... wait no.
            # conv[k] = sum_j a[j]*a[k-j], terms with j=i: a[i]*a[k-i]
            # terms with k-j=i, i.e., j=k-i: a[k-i]*a[i]
            # These are the same term! So contribution of a[i] to conv[k] is a[i]*a[k-i] (one term)
            # unless i = k-i, i.e., k = 2i, in which case it's a[i]^2 (one term, not two)

            # Actually let me just try a few values and pick the best
            rest_sum = np.sum(a) - orig_val
            best_local_val = orig_val
            best_local_obj = best_obj

            # Try reducing a[i] (to reduce peak)
            for frac in [0.5, 0.7, 0.3, 0.9, 0.1]:
                a[i] = orig_val * frac
                s = rest_sum + a[i]
                if s < 1e-15:
                    a[i] = orig_val
                    continue
                a_norm = a / s
                conv = compute_conv(a_norm)
                obj_try = 2.0 * n * np.max(conv)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_local_val = a[i]

            # Try increasing slightly
            for frac in [1.2, 1.5, 2.0]:
                a[i] = orig_val * frac
                s = rest_sum + a[i]
                a_norm = a / s
                conv = compute_conv(a_norm)
                obj_try = 2.0 * n * np.max(conv)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_local_val = a[i]

            a[i] = best_local_val

            # Update best
            a_norm = normalize_pos(a)
            obj_cur = compute_obj(a_norm)
            if obj_cur < best_obj:
                best_obj = obj_cur
                best_a = a_norm.copy()
                a = a_norm.copy()

    return best_a, best_obj


def peak_chasing(a, n_steps=50, time_budget=20.0):
    """
    Focus on reducing the maximum peak specifically.
    At each step, identify the peak location and reduce contributions to it.
    """
    n = len(a)
    a = normalize_pos(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_peak = int(np.argmax(conv))
        cmax = float(conv[k_peak])

        # Gradient at the peak: grad[i] = 2*a[k_peak - i] (for valid)
        grad = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        grad = grad - np.mean(grad)

        # Try different step sizes
        best_step_obj = best_obj
        best_step_a = a.copy()
        for lr in [0.01, 0.02, 0.05, 0.1, 0.005]:
            a_try = a - lr * grad
            a_try = normalize_pos(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()

        if best_step_obj < best_obj:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = best_step_a.copy()
        else:
            # No improvement, try reducing the peak element in a directly
            # The peak at k_peak: conv[k_peak] = sum_i a[i]*a[k_peak-i]
            # Reduce a[i] where a[i]*a[k_peak-i] is largest
            products = np.zeros(n)
            for i in range(n):
                j = k_peak - i
                if 0 <= j < n:
                    products[i] = a[i] * a[j]

            top_idx = int(np.argmax(products))
            for frac in [0.5, 0.7, 0.3]:
                a_try = a.copy()
                a_try[top_idx] *= frac
                a_try = normalize_pos(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

    return best_a, best_obj


def soft_threshold_step(a, n_iters=200, time_budget=20.0):
    """
    Iterate: compute conv, find peaks, reduce elements contributing to peaks.
    Use a soft threshold that adapts to the current state.
    """
    n = len(a)
    a = normalize_pos(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = 0.01  # initial soft-max temperature
    lr = 0.05

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        if cmax < 1e-15:
            break

        # Soft-max weights
        tau = 0.01 + 0.1 * (it / n_iters)
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -100.0, 0.0)
        w = np.exp(shifted)
        ws = np.sum(w)
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= ws

        # Gradient
        grad = grad_weighted_conv(a, w)
        grad = grad - np.mean(grad)

        # Adaptive learning rate
        a_new = a - lr * grad
        a_new = normalize_pos(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(lr * 1.05, 0.5)
        else:
            a = a_new  # still update (for exploration)
            lr *= 0.9

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize_pos(a_new)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Sine window
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    # Gaussian
    x = np.linspace(-3, 3, n)
    gauss = np.exp(-x ** 2 / 3.0)
    inits.append(gauss / gauss.sum())

    # Two Gaussian bumps
    x = np.linspace(0, 1, n)
    bimodal = np.exp(-((x - 0.3) ** 2) / 0.03) + np.exp(-((x - 0.7) ** 2) / 0.03)
    inits.append(bimodal / bimodal.sum())

    # Random smooth
    rough = rng.random(n)
    # Smooth with a simple moving average
    kernel_size = max(3, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rough, kernel, mode='same')
    inits.append(smooth / smooth.sum() if smooth.sum() > 0 else np.ones(n) / n)

    # Cosine window (Hann)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.01
    inits.append(hann / hann.sum())

    # Flat-top-ish
    flat = np.ones(n)
    flat[:n // 4] *= 0.8
    flat[-n // 4:] *= 0.8
    inits.append(flat / flat.sum())

    # Random with structure (blocks)
    n_blocks = rng.integers(3, 10)
    block_vals = rng.dirichlet(np.ones(n_blocks) * 2.0)
    block_a = np.zeros(n)
    for b in range(n_blocks):
        start = int(b * n / n_blocks)
        end = int((b + 1) * n / n_blocks)
        block_a[start:end] = block_vals[b]
    inits.append(block_a / block_a.sum() if block_a.sum() > 0 else np.ones(n) / n)

    if parent is not None:
        # Resampled parent
        inits.append(resample_sequence(parent, n))
        # Perturbed parent
        base = resample_sequence(parent, n)
        pert = base * np.exp(rng.normal(0, 0.1, n))
        inits.append(normalize_pos(pert))
        # Strongly perturbed parent
        pert2 = base * np.exp(rng.normal(0, 0.3, n))
        inits.append(normalize_pos(pert2))

    return inits


def simulate_annealing(a, rng, max_iters=300, T_init=0.1, T_min=0.001, time_budget=30.0):
    """Simulated annealing with various move types."""
    n = len(a)
    a = normalize_pos(a.copy())
    cur = a.copy()
    cur_obj = compute_obj(a)
    best = a.copy()
    best_obj = cur_obj
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T_init * (T_min / T_init) ** (it / max_iters)
        T = max(T, 1e-7)

        move = rng.integers(0, 4)
        trial = cur.copy()

        if move == 0:
            # Window noise
            wsize = max(5, n // 20)
            wstart = int(rng.integers(0, n - wsize))
            scale = cur[wstart:wstart + wsize].mean() + 0.01
            trial[wstart:wstart + wsize] += rng.normal(0, 0.2 * scale, wsize)
        elif move == 1:
            # Element swap (shift mass)
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2:
                amt = min(cur[i1], 0.05) * cur[i1]
                trial[i1] -= amt
                trial[i2] += amt
        elif move == 2:
            # Window smooth
            wsize = max(5, n // 15)
            wstart = int(rng.integers(0, n - wsize - 1))
            # Blend with neighbor
            w2start = wstart + wsize
            trial[wstart:wstart + wsize] = 0.5 * cur[wstart:wstart + wsize] + 0.5 * cur[w2start:w2start + wsize]
        else:
            # Reshape: apply a random smooth perturbation
            pert = np.cumsum(rng.normal(0, 0.1, n))
            pert = pert - np.mean(pert)
            trial = cur * np.exp(pert * 0.1)

        trial = normalize_pos(trial)
        trial_obj = compute_obj(trial)

        delta = trial_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best = trial.copy()

    return best, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.all(np.isfinite(parent)):
            return parent.copy()
    except (NameError, Exception):
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-strategy optimization to minimize
    2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent loaded: n={len(parent)}, obj={parent_obj:.8f}")
        best_seq = parent.copy()
        best_score = parent_obj
    else:
        parent = np.ones(500) / 500
        best_seq = parent.copy()
        best_score = compute_obj(parent)
        print(f"No parent found. Starting with uniform n={len(parent)}, obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # === Phase 1: Intensive optimization on parent ===
    print("\n=== Phase 1: Intensive Optimization on Parent ===")
    phase1_budget = min(n_parent * 1.5, (deadline - start_time) * 0.40)
    phase1_start = time.monotonic()
    a = normalize_pos(best_seq.copy())
    n = len(a)
    local_best = a.copy()
    local_obj = best_score

    # Strategy cycle
    momentum_buf = np.zeros(n)
    strategies = ['subgrad', 'freq_flat', 'peak_chase', 'soft_thresh', 'cyclic', 'freq_shaped']

    cycle_count = 0
    while time.monotonic() - phase1_start < phase1_budget:
        remaining = phase1_budget - (time.monotonic() - phase1_start)
        if remaining < 5:
            break

        strat = strategies[cycle_count % len(strategies)]
        alloc = min(remaining * 0.3, 30.0)

        if strat == 'subgrad':
            # Subgradient with momentum
            a_cur = local_best.copy()
            v = momentum_buf.copy()
            tau = 0.05
            lr = 0.02
            for it in range(min(200, int(alloc * 5))):
                if time.monotonic() - phase1_start > phase1_budget:
                    break
                a_cur, v, _ = subgradient_step(a_cur, tau, lr, v, momentum=0.85)
                obj_cur = compute_obj(a_cur)
                if obj_cur < local_obj:
                    local_obj = obj_cur
                    local_best = a_cur.copy()
                    if local_obj < best_score:
                        best_score = local_obj
                        best_seq = local_best.copy()
                        print(f"  [subgrad it{it}] NEW BEST: {best_score:.8f}")
                lr *= 0.999
                tau = 0.01 + 0.1 * (it / 200)
            momentum_buf = v

        elif strat == 'freq_flat':
            a_new, obj_new = freq_flat_magnitude(local_best, n_iters=150,
                                                  target='flat', time_budget=alloc)
            if obj_new < local_obj:
                local_obj = obj_new
                local_best = a_new.copy()
                if local_obj < best_score:
                    best_score = local_obj
                    best_seq = local_best.copy()
                    print(f"  [freq_flat] NEW BEST: {best_score:.8f}")

        elif strat == 'freq_shaped':
            a_new, obj_new = freq_shaped_target(local_best, n_iters=100, time_budget=alloc)
            if obj_new < local_obj:
                local_obj = obj_new
                local_best = a_new.copy()
                if local_obj < best_score:
                    best_score = local_obj
                    best_seq = local_best.copy()
                    print(f"  [freq_shaped] NEW BEST: {best_score:.8f}")

        elif strat == 'peak_chase':
            a_new, obj_new = peak_chasing(local_best, n_steps=40, time_budget=alloc)
            if obj_new < local_obj:
                local_obj = obj_new
                local_best = a_new.copy()
                if local_obj < best_score:
                    best_score = local_obj
                    best_seq = local_best.copy()
                    print(f"  [peak_chase] NEW BEST: {best_score:.8f}")

        elif strat == 'soft_thresh':
            a_new, obj_new = soft_threshold_step(local_best, n_iters=200, time_budget=alloc)
            if obj_new < local_obj:
                local_obj = obj_new
                local_best = a_new.copy()
                if local_obj < best_score:
                    best_score = local_obj
                    best_seq = local_best.copy()
                    print(f"  [soft_thresh] NEW BEST: {best_score:.8f}")

        elif strat == 'cyclic' and n <= 800:
            a_new, obj_new = cyclic_descent(local_best, n_sweeps=10, time_budget=alloc)
            if obj_new < local_obj:
                local_obj = obj_new
                local_best = a_new.copy()
                if local_obj < best_score:
                    best_score = local_obj
                    best_seq = local_best.copy()
                    print(f"  [cyclic] NEW BEST: {best_score:.8f}")

        cycle_count += 1

    print(f"Phase 1 done. Best: {best_score:.8f}")

    # === Phase 2: Simulated Annealing ===
    print("\n=== Phase 2: Simulated Annealing ===")
    phase2_budget = min(120.0, (deadline - time.monotonic()) * 0.15)
    if phase2_budget > 10:
        a_sa, obj_sa = simulate_annealing(best_seq, rng, max_iters=500,
                                          T_init=0.05, T_min=0.001,
                                          time_budget=phase2_budget)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA improved: {best_score:.8f}")
        else:
            print(f"  SA best: {obj_sa:.8f} (no improvement)")

    # === Phase 3: Multi-length exploration ===
    print("\n=== Phase 3: Multi-length Exploration ===")
    remaining3 = deadline - time.monotonic()
    if remaining3 > 60:
        phase3_budget = min(remaining3 * 0.25, 300.0)
        phase3_start = time.monotonic()

        n_current = len(best_seq)
        # Try various lengths
        n_tries = []
        for delta in [-100, -50, -30, -10, 10, 30, 50, 100, 200]:
            n_try = n_current + delta
            if 100 <= n_try <= 2000:
                n_tries.append(n_try)
        # Also try some specific lengths
        for n_spec in [300, 400, 500, 600, 700, 800, 1000, 1200]:
            if n_spec not in n_tries:
                n_tries.append(n_spec)
        n_tries = list(dict.fromkeys(n_tries))  # deduplicate, keep order
        # Sort by proximity to current best length
        n_tries.sort(key=lambda x: abs(x - n_current))

        per_n_budget = max(5.0, phase3_budget / len(n_tries))

        for n_try in n_tries:
            if time.monotonic() - phase3_start > phase3_budget:
                break
            if time.monotonic() > deadline:
                break

            a_res = resample_sequence(best_seq, n_try)
            a_res, obj_res = peak_chasing(a_res, n_steps=30, time_budget=5.0)
            a_res, obj_res2 = soft_threshold_step(a_res, n_iters=100, time_budget=5.0)
            if obj_res2 < obj_res:
                obj_res = obj_res2

            # Frequency domain refinement
            a_res, obj_res3 = freq_flat_magnitude(a_res, n_iters=80,
                                                   target='flat', time_budget=5.0)
            if obj_res3 < obj_res:
                obj_res = obj_res3

            status = ""
            if obj_res < best_score:
                best_score = obj_res
                best_seq = a_res.copy()
                status = " ** NEW BEST **"
                print(f"  n={n_try}: {best_score:.8f}{status}")
            else:
                print(f"  n={n_try}: {obj_res:.8f} (no improvement)")

    # === Phase 4: Final intensive polish ===
    print("\n=== Phase 4: Final Polish ===")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 20 and best_seq is not None:
        a_final = normalize_pos(best_seq.copy())
        n_final = len(a_final)
        local_best_final = a_final.copy()
        local_obj_final = best_score
        momentum_buf = np.zeros(n_final)

        # Rapid strategy cycling
        while time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic()
            if remaining < 3:
                break

            # Subgradient
            a_cur = local_best_final.copy()
            v = momentum_buf.copy()
            for it in range(min(100, int(remaining * 3))):
                if time.monotonic() > deadline - 5:
                    break
                tau = 0.02 + 0.08 * (it / 100)
                lr = 0.01 * (1.0 - it / 200)
                a_cur, v, _ = subgradient_step(a_cur, tau, max(lr, 0.001), v, momentum=0.9)
                obj_cur = compute_obj(a_cur)
                if obj_cur < local_obj_final:
                    local_obj_final = obj_cur
                    local_best_final = a_cur.copy()
                    if local_obj_final < best_score:
                        best_score = local_obj_final
                        best_seq = local_best_final.copy()
                        print(f"  [final subgrad] {best_score:.8f}")
            momentum_buf = v

            # Quick freq flat
            remaining = deadline - time.monotonic()
            if remaining > 5:
                a_f, obj_f = freq_flat_magnitude(local_best_final, n_iters=50,
                                                  target='soft_flat', time_budget=min(remaining * 0.3, 10.0))
                if obj_f < local_obj_final:
                    local_obj_final = obj_f
                    local_best_final = a_f.copy()
                    if local_obj_final < best_score:
                        best_score = local_obj_final
                        best_seq = local_best_final.copy()
                        print(f"  [final freq] {best_score:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    # Convert to list of floats, clip to [0, 1000]
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with the actual evaluator
    val = evaluate_sequence(result)
    print(f"\nEvaluated score: {val:.8f}")
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        print("Warning: evaluation failed, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
