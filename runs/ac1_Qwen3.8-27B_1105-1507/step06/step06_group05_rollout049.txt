def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Solve: find lambda such that sum(max(v - lambda, 0)) = target_sum
    v_sorted = np.sort(v)[::-1]
    cv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cv[-1] - target_sum))[0][-1]
    lambda_ = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - lambda_, 0.0)

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def spectral_flat(a, max_iters=80, time_budget=50.0):
    """Improved spectral equalization with proper DC handling."""
    n = len(a)
    x = normalize(a.copy().astype(np.float64))
    best_x = x.copy()
    best_obj = compute_obj(x)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(x, size)
        P = np.abs(A) ** 2
        P_max = float(np.max(P[1:])) if size > 1 else 0
        if P_max < 1e-30:
            break
        
        # Target: make P[1:] as flat as possible (keep DC fixed)
        # DC bin: A[0] = sum(x) = 1 (real, positive)
        # We want P[k] for k>=1 to be close to some target
        target_power = P[0] ** 2 / (size - 1)  # average non-DC power
        
        # Blend current spectrum toward target
        alpha = 0.3 + 0.4 * (it / max_iters)
        P_new = P.copy()
        for k in range(1, size // 2 + 1):
            if k < size - k:  # avoid double-counting for real signals
                P_new[k] = (1 - alpha) * P[k] + alpha * target_power
                if k + (size - k) < size:
                    P_new[size - k] = P_new[k]
        
        # Reconstruct
        mag_new = np.sqrt(np.maximum(P_new, 0))
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        
        # Force Hermitian symmetry
        A_new[0] = 1.0  # DC
        for k in range(1, size // 2 + 1):
            A_new[size - k] = np.conj(A_new[k])
        
        x_new = np.real(ifft(A_new, size))[:n]
        x_new = np.maximum(x_new, 0.0)
        s = float(np.sum(x_new))
        if s < 1e-10:
            break
        x_new = x_new / s
        
        obj = compute_obj(x_new)
        if obj < best_obj:
            best_obj = obj
            best_x = x_new.copy()
            x = x_new
        else:
            x = 0.7 * x + 0.3 * x_new
            x = normalize(x)
    
    return best_x


def coordinate_descent(a, time_budget=30.0):
    """Coordinate descent targeting convolution peaks."""
    n = len(a)
    x = normalize(a.copy().astype(np.float64))
    best_x = x.copy()
    best_obj = compute_obj(x)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(x)
        k_star = int(np.argmax(conv))
        
        # Find top contributors
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((x[j] * x[k2], j, k2))
        contrib.sort(reverse=True)
        
        improved = False
        for _, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            
            # Move mass from peak contributor to a low-impact position
            for _ in range(3):
                j_target = int(rng.integers(0, n))
                if j_target in (j1, j2):
                    continue
                
                for frac in [0.1, 0.2, 0.05]:
                    delta = frac * x[j1]
                    x_try = x.copy()
                    x_try[j1] -= delta
                    x_try[j_target] += delta
                    x_try = normalize(x_try)
                    obj_try = compute_obj(x_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_x = x_try.copy()
                        x = x_try
                        improved = True
                        break

            x = normalize(x * np.exp(rng.normal(0, 0.02, n)))
    
    return best_x


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key new approaches: multiplicative weights (mirror descent) with soft/hard subgradients,
improved LP alternating with multi-direction diversity, spectral equalization with
proper DC handling, and careful multi-n exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum} via bisection on lambda."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cv[-1] - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    lambda_ = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - lambda_, 0.0)


def compute_subgradient_peak(x, k):
    """Compute subgradient of (x*x)[k] at x. Returns array of length n."""
    n = len(x)
    i_start = max(0, k - n + 1)
    i_end = min(k, n - 1)
    g = np.zeros(n)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        j = k - idx
        g[idx] = 2.0 * x[j]
    return g


def multiplicative_weights_soft(a, max_iters=800, time_budget=80.0, tau_init=0.005):
    """Mirror descent with KL divergence using soft subgradient (softmax over peaks)."""
    n = len(a)
    x = normalize(a.copy().astype(np.float64))
    best_x = x.copy()
    best_obj = compute_obj(x)
    start = time.monotonic()
    step = 0.4
    tau = tau_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        max_val = float(np.max(conv))

        # Softmax weights over convolution values
        shifted = (conv - max_val) / tau
        weights = np.exp(np.clip(shifted, -500.0, 500.0))
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx_max = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx_max] = 1.0
        else:
            weights /= w_sum

        # Compute soft subgradient: g[i] = sum_k w[k] * 2*x[k-i] (valid indices)
        g = np.zeros(n)
        for k in range(len(weights)):
            wk = weights[k]
            if wk < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                j = k - idx
                g[idx] += wk * 2.0 * x[j]

        # Center gradient for sum constraint
        g_centered = g - np.mean(g)

        # Multiplicative weights update
        log_update = -step * g_centered
        log_update = np.clip(log_update, -50.0, 50.0)
        x_new = x * np.exp(log_update)
        x_new = np.maximum(x_new, 0.0)
        s = float(np.sum(x_new))
        if s < 1e-300:
            break
        x_new = x_new / s

        obj = compute_obj(x_new)
        if obj < best_obj:
            best_obj = obj
            best_x = x_new.copy()
            x = x_new
            step = min(step * 1.003, 1.2)
        else:
            step *= 0.65
            if step < 1e-7:
                break

        # Cool temperature
        tau = max(tau * 0.995, 0.0005)

    return best_x

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    if n > 1500:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate_multi(a, max_iters=25, time_budget=80.0):
    """LP alternating with multiple M values and backtracking line search."""
    n = len(a)
    x = normalize(a.copy().astype(np.float64))
    best_x = x.copy()
    best_obj = compute_obj(x)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(x)
        M = float(np.max(conv))

        # Try different M values for diverse directions
        M_fracs = [1.0, 0.92, 0.85]
        directions = []
        for frac in M_fracs:
            if time.monotonic() - start > time_budget - 3:
                break
            g = solve_lp(x, M * frac, time_limit=max(3, int(remaining * 0.25)))
            if g is not None:
                directions.append(g)

        if not directions:
            break

        improved = False
        for g in directions:
            for alpha in [1.0, 0.6, 0.4, 0.25, 0.15, 0.1]:
                x_try = normalize((1.0 - alpha) * x + alpha * g)
                obj_try = compute_obj(x_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_x = x_try.copy()
                    x = x_try
                    improved = True
                    break

        if not improved:
            rng = np.random.default_rng(it * 104729 + 1)
            x = normalize(best_x * np.exp(rng.normal(0, 0.04, n)))

    return best_x


def spectral_flat(a, max_iters=80, time_budget=50.0):
    """Spectral equalization: flatten power spectrum while preserving DC."""
    n = len(a)
    x = normalize(a.copy().astype(np.float64))
    best_x = x.copy()
    best_obj = compute_obj(x)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(x, size)
        P = np.abs(A) ** 2

        # Target power for non-DC bins
        dc_power = P[0]
        non_dc_mean = float(np.mean(P[1:]))
        target = (0.5 * non_dc_mean + 0.5 * dc_power / (size - 1))

        # Blend toward target
        alpha = 0.2 + 0.3 * (it / max(float(max_iters), 1))
        P_new = P.copy()
        P_new[1:] = (1.0 - alpha) * P[1:] + alpha * target

        # Reconstruct
        mag_new = np.sqrt(np.maximum(P_new, 0.0))
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)

        # Enforce Hermitian symmetry
        A_new[0] = float(A_new[0].real)
        half = size // 2
        for k in range(1, half + 1):
            A_new[size - k] = np.conj(A_new[k])

        x_new = np.real(ifft(A_new, size))[:n]
        x_new = np.maximum(x_new, 0.0)
        s = float(np.sum(x_new))
        if s < 1e-10:
            break
        x_new = x_new / s

        obj = compute_obj(x_new)
        if obj < best_obj:
            best_obj = obj
            best_x = x_new.copy()
            x = x_new
        else:
            x = normalize(0.7 * x + 0.3 * x_new)

    return best_x


def peak_targeting(a, time_budget=40.0):
    """Directly target the top peaks by moving mass from contributing positions."""
    n = len(a)
    x = normalize(a.copy().astype(np.float64))
    best_x = x.copy()
    best_obj = compute_obj(x)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(x)
        sorted_peaks = np.argsort(conv)[::-1][:3]

        improved = False
        for k_star in sorted_peaks:
            if time.monotonic() - start > time_budget:
                break
            k_star = int(k_star)

            # Find top contributors to this peak
            contrib = []
            i_start = max(0, k_star - n + 1)
            i_end = min(k_star, n - 1)
            for i in range(i_start, i_end + 1):
                j = k_star - i
                contrib.append((x[i] * x[j], i, j))
            contrib.sort(reverse=True)

            for _, i1, i2 in contrib[:6]:
                if i1 == i2:
                    continue
                # Move mass from this pair to random distant positions
                for _ in range(4):
                    t1 = int(rng.integers(0, n))
                    t2 = int(rng.integers(0, n))
                    if t1 in (i1, i2) or t2 in (i1, i2):
                        continue
                    for frac in [0.08, 0.15, 0.25]:
                        delta1 = frac * x[i1]
                        delta2 = frac * x[i2]
                        x_try = x.copy()
                        x_try[i1] -= delta1
                        x_try[i2] -= delta2
                        x_try[t1] += delta1
                        x_try[t2] += delta2
                        x_try = normalize(x_try)
                        obj_try = compute_obj(x_try)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_x = x_try.copy()
                            x = x_try
                            improved = True
                            break
                    if improved:
                        break

        if not improved and rng.random() < 0.08:
            x = normalize(x * np.exp(rng.normal(0, 0.03, n)))

    return best_x


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)
