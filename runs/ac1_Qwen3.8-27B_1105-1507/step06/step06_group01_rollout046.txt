"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

Uses a multi-phase approach:
1. Adaptive mirror descent with log-sum-exp smoothing (main workhorse)
2. LP best-response with adaptive active sets
3. Coordinate descent with golden section search
4. Multi-start diversification with restarts

Key features:
- FFT-based fast convolution
- Adaptive temperature schedule for smoothing
- Polyak step size heuristic
- Multiple initialization strategies
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """Compute (smooth_max_value, gradient) using log-sum-exp.
    
    smooth_max ≈ (1/tau) * log(sum_k exp(tau * b_k))
    grad[i] = 2 * sum_k w_k * a[k-i] where w_k = exp(tau*b_k)/sum(exp(tau*b))
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    w = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} w[j+i] * a[j]
    grad = np.zeros(n)
    for i in range(n):
        k_min = i
        k_max = i + n - 1
        grad[i] = 2.0 * np.dot(w[k_min:k_max+1], a)
    
    return smooth_val, grad


def mirror_descent_adaptive(a_init, deadline, tau_start=50.0, tau_end=2000.0, 
                            max_iters=50000, verbose=False):
    """Adaptive mirror descent with log-sum-exp smoothing.
    
    Uses multiplicative weights with adaptive step size and temperature.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Initial step size
    step = 0.5
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Adaptive temperature
        frac = min(1.0, it / max(1, max_iters // 2))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute smooth max and gradient
        _, grad = smooth_max_grad(a, tau)
        grad_tan = grad - grad.mean()
        
        # Adaptive step size (Polyak-like)
        # Scale step inversely with gradient magnitude and tau
        grad_norm = np.sqrt(np.sum(grad_tan ** 2))
        if grad_norm > 1e-15:
            step = 0.3 / (1.0 + 0.1 * tau * 0.01 + 0.5 * grad_norm)
        else:
            step = 0.5
        
        # Mirror descent update
        log_weights = -step * grad_tan
        log_weights -= log_weights.max()
        a_new = a * np.exp(log_weights)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        # Track best
        M_cur = fast_conv(a).max()
        if M_cur < best_M - 1e-12:
            best_M = M_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart with perturbation if stuck
        if no_improve > 200 and it % 500 == 0:
            a = best_a + np.random.randn(n) * 0.02
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            no_improve = 0
    
    return best_a


def lp_best_response(a, n_active=30, time_limit=10):
    """Solve LP best response using top n_active convolution coefficients."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    active = top_indices.tolist()
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_polish(a_init, deadline, n_active_start=20, n_active_max=100):
    """LP mirror descent polish."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    n_active = n_active_start
    
    while time.monotonic() < deadline and no_improve < 30:
        g = lp_best_response(a, n_active, time_limit=8)
        if g is None:
            n_active = min(n_active + 10, n_active_max)
            no_improve += 2
            continue
        
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
            n_active = max(10, int(n_active * 0.9))
        else:
            no_improve += 1
            n_active = min(n_active + 10, n_active_max)
    
    return best_a


def coordinate_descent_1d(a, i, b, n):
    """Optimize a single coordinate using golden section search."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = np.zeros(k_hi - k_lo + 1)
    s_vals = np.zeros(k_hi - k_lo + 1)
    has_quad = False
    quad_idx = -1
    
    for idx, k in enumerate(range(k_lo, k_hi + 1)):
        j = k - i
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            s_vals[idx] = 0.0
            has_quad = True
            quad_idx = idx
        elif 0 <= j < n:
            c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
            s_vals[idx] = 2.0 * a[j]
        else:
            c_vals[idx] = b[k]
            s_vals[idx] = 0.0
    
    # Unaffected max
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_lo:k_hi + 1] = False
    base_max = b[mask].max() if mask.any() else 0.0
    
    def obj_at(x):
        vals = c_vals + s_vals * x
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_vals[quad_idx] + x * x
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(3.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.3)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, deadline, seed=0):
    """Coordinate descent with random ordering."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    while time.monotonic() < deadline and no_improve < 30:
        rng.shuffle(order)
        b = fast_conv(a)
        improved = False
        
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                delta = x_opt - a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += 2 * a[i] * delta + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2 * a[j] * delta
                improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.01, 0.05, 0.1]:
            pn = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(pn / pn.sum())
    
    inits.append(np.ones(n) / n)
    
    # Skewed distributions
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        skewed = t ** alpha + 0.01
        inits.append(skewed / skewed.sum())
        skewed2 = (1 - t) ** alpha + 0.01
        inits.append(skewed2 / skewed2.sum())
    
    # Cosine windows
    for freq in [1, 2, 3]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    # Random smooth
    ksize = max(3, n // 10)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits

    best = np.maximum(a0.copy(), 1e-15)

    t0 = time.monotonic()
