Spectral-Subgradient-LP hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key innovations: (1) geometric/power-law initializations that provably beat uniform,
(2) adaptive spectral peak-shaving in frequency domain, (3) momentum-based subgradient

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

# Global storage
_best_seq = None
_best_score = np.inf


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-15:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):

def solve_lp_max_sum(a, M, time_limit=15):

def solve_lp_minpeak(a, time_limit=15):

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))

def spectral_peak_shave(a, max_iters=200, time_budget=30.0, alpha_start=0.5, alpha_end=0.95, shave_frac=0.3):
    """
    Spectral peak shaving: iteratively reduce peaks in the power spectrum.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        
        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        ps_max = np.max(power)
        if ps_max < 1e-30:
            break
        
        # Peak shaving: reduce peaks above threshold
        threshold = np.median(power[power > 0]) * (1 + shave_frac)
        if threshold < ps_max:
            power_clipped = power.copy()
            mask = power > threshold
            power_clipped[mask] = threshold + (power[mask] - threshold) * 0.1
        
        # Power compression
        ps_norm = power_clipped / np.max(power_clipped)
        ps_compressed = ps_norm ** alpha
        
        # Reconstruct
        magnitude = np.sqrt(ps_compressed * ps_max)
        phase = np.angle(fa)
        fa_new = magnitude * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

    while size < 2 * n - 1:
        size <<= 1

    conv = irfft(fa * fa, size)
