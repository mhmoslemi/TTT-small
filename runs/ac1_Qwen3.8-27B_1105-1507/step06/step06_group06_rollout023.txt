import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_step(a, eps=0.01, time_limit=10):
    """
    One LP step: find direction d that reduces the max convolution peak.
    Linearize conv_k at current a, solve LP to minimize worst-case increase.
    """
    n = len(a)
    conv = compute_conv_fft(a)
    cmax = float(np.max(conv))
    
    # Active set: positions where conv is close to max
    active_mask = conv >= (1 - eps) * cmax
    active_indices = np.where(active_mask)[0]
    
    if len(active_indices) == 0:
        active_indices = np.array([int(np.argmax(conv))])
    
    # Build gradient matrix: G[k_idx, i] = d(conv_{active_indices[k_idx]})/da[i]
    # grad_{k}[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    m = len(active_indices)
    G = np.zeros((m, n))
    for idx, k in enumerate(active_indices):
        # For position i, grad = 2*a[k-i] if 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            i_range = np.arange(lo, hi)
            a_indices = k - i_range
            G[idx, i_range] = 2.0 * a[a_indices]
    
    # LP: minimize t
    # subject to: G @ d <= t - conv[active_indices]  =>  G @ d - t <= -conv[active_indices]
    #             sum(d) = 0
    #             a + d >= 0  =>  -d <= a
    #             d is unbounded above (but constrained by -d <= a)
    
    # Variables: [d_0, ..., d_{n-1}, t]
    # c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ [d, t] <= b_ub
    # 1) G @ d - t <= -conv[active_indices]  =>  [G, -1] @ [d, t] <= -conv[active_indices]
    A_ub1 = np.hstack([G, -np.ones((m, 1))])
    b_ub1 = -conv[active_indices]
    
    # 2) -d <= a  =>  [-I, 0] @ [d, t] <= a
    A_ub2 = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b_ub2 = a
    
    A_ub = np.vstack([A_ub1, A_ub2])
    b_ub = np.concatenate([b_ub1, b_ub])
    
    # Equality: sum(d) = 0  =>  [1, ..., 1, 0] @ [d, t] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: t is unbounded (but will be constrained), d is unconstrained (but -d <= a handles lower bound)
    bounds = [(None, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            d = res.x[:n]
            t_opt = res.x[n]
            return d, t_opt, conv[active_indices]
    except Exception:
        pass
    return None, None, None

def lp_iteration(a, eps=0.01, max_steps_per_call=5, time_budget=30.0):
    """Multiple LP iterations with line search."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for step in range(max_steps_per_call):
        if time.monotonic() - start > time_budget:
            break
        
        d, t_opt, active_conv = lp_step(a, eps=eps, time_limit=8)
        if d is None:
            break
        
        # Line search: try different step sizes
        best_step_obj = best_obj
        best_step_a = a.copy()
        
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = a + alpha * d
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()
        
        if best_step_obj < best_obj - 1e-14:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = best_step_a
        else:
            break
    
    return best_a, best_obj

    """
    Find LP direction to reduce max convolution peak.
    
    If include_all=True, use all 2n-1 convolution positions.
    If include_all=False, use only active positions (within eps of max).
    """
    n = len(a)
    conv = compute_conv_fft(a)
    cmax = float(np.max(conv))
    
    if include_all:
        indices = np.arange(2 * n - 1)
    else:
        if eps > 0:
            active = conv >= (1 - eps) * cmax
            indices = np.where(active)[0]
        else:
            indices = np.array([int(np.argmax(conv))])
    
    m = len(indices)
    
    # Build gradient matrix G: G[idx, i] = 2*a[k-i] if 0 <= k-i < n
    G = np.zeros((m, n))
    for idx, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            i_range = np.arange(lo, hi)
            a_indices = k - i_range
            G[idx, i_range] = 2.0 * a[a_indices]
    
    # LP: minimize t
    # s.t. G @ d <= t + (cmax - conv[indices])  [allow up to (cmax - conv[k]) increase for inactive, 0 for active]
    #     Actually: conv[k] + G[k,:] @ d <= t  for all k in indices
    #     => G[k,:] @ d - t <= -conv[k]
    #     => [G, -1] @ [d, t] <= -conv[indices]
    #     -d <= a  (non-negativity of a+d)
    #     sum(d) = 0
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub_rows = []
    b_ub_rows = []
    
    # Convolution constraints
    A_ub_rows.append(np.hstack([G, -np.ones((m, 1))]))
    b_ub_rows.append(-conv[indices])
    
    # Non-negativity: -d <= a
    A_ub_rows.append(np.hstack([-np.eye(n), np.zeros((n, 1))]))
    b_ub_rows.append(a.copy())
    
    A_ub = np.vstack(A_ub_rows)
    b_ub = np.concatenate(b_ub_rows)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(None, None)] * (n + 1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            d = res.x[:n]
            t_val = res.x[n]
            # The optimal t tells us the best achievable max (linearized)
            improvement = cmax - t_val
            return d, t_val, improvement
    except Exception as e:
        pass
    return None, None, None


def lp_optimize(a, max_iters=30, time_budget=60.0, use_all=True, eps=0.0):
    """Run multiple LP iterations with line search."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        d, t_val, improvement = lp_direction(a, include_all=use_all, eps=eps, time_limit=8)
        if d is None:
            break
        
        if improvement < 1e-12:
            # Try with just the top peak
            d, t_val, improvement = lp_direction(a, include_all=False, eps=0.0, time_limit=8)
            if d is None or improvement < 1e-12:
                break
        
        # Line search
        best_alpha = 0
        best_alpha_obj = best_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = a + alpha * d
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj:
                best_alpha_obj = obj_try
                best_alpha = alpha
        
        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-14:
            best_obj = best_alpha_obj
            best_a = a.copy()
            a = a + best_alpha * d
            a = project_nonneg_sum1(a)
            stall_count = 0
            if it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, alpha={best_alpha:.2f}, improvement={improvement:.2e}")
        else:
            stall_count += 1
            if stall_count > 5:
                break
    
    return best_a, best_obj


def freq_smooth(a, max_iters=50, time_budget=15.0):
    """Frequency domain smoothing to reduce peaks."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = np.fft.fft(a, N)
        power_spec = np.abs(fa) ** 2
        
        # Compress dynamic range of power spectrum
        ps_max = np.max(power_spec)
        if ps_max < 1e-15:
            break
        ps_norm = power_spec / ps_max
        alpha = 0.8 + 0.15 * (it / max_iters)
        ps_compressed = np.maximum(ps_norm, 1e-10) ** alpha
        
        phase = np.angle(fa)
        magnitude = np.sqrt(ps_compressed * ps_max)
        fa_new = magnitude * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
    
    return best_a, best_obj


def chase_peak(a, n_steps=50, step_size=0.05):
    """Chase the peak using subgradient method."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for _ in range(n_steps):
        conv = compute_conv_fft(a)
        k_star = int(np.argmax(conv))
        
        # Gradient of conv_{k_star}
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        if hi > lo:
            i_range = np.arange(lo, hi)
            a_indices = k_star - i_range
            grad[i_range] = 2.0 * a[a_indices]
        
        grad_centered = grad - np.mean(grad)
        
        for ss in [step_size, step_size * 0.5, step_size * 2]:
            a_try = a - ss * grad_centered
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break
    
    return best_a, best_obj


def softmax_subgradient(a, max_iters=100, time_budget=20.0):
    """Subgradient with softmax weighting over peaks."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv_fft(a)
        cmax = float(np.max(conv))
        tau = 0.001 * cmax + 1e-8
        shifted = (conv - cmax) / tau
        weights = np.exp(np.clip(shifted, -30, 30))
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
        
        # Weighted gradient
        grad = np.zeros(n)
        active_ks = np.where(weights > 1e-6)[0]
        for k in active_ks:
            w = weights[k]
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if hi > lo:
                i_range = np.arange(lo, hi)
                a_indices = k - i_range
                grad[i_range] += w * 2.0 * a[a_indices]
        
        grad_centered = grad - np.mean(grad)
        
        step = 0.05 / (1 + 0.01 * it)
        a_try = a - step * grad_centered
        a_try = project_nonneg_sum1(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

"""
LP-based active-set optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: (1) Full-constraint LP direction finding that simultaneously prevents
peak migration across all 2n-1 convolution positions, (2) adaptive line search with
quadratic prediction, (3) multi-tier active set refinement, (4) frequency-domain
polishing combined with LP re-optimization, (5) perturbation-based restarts to
escape local minima.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lp_direction(a, mode="all", eps=0.0, time_limit=10):
    """
    Find LP direction to reduce max convolution peak.
    mode="all": use all 2n-1 convolution positions (prevents peak migration)
    mode="active": use only positions within eps fraction of max
    mode="topk": use top-k positions
    """
    n = len(a)
    conv = compute_conv_fft(a)
    cmax = float(np.max(conv))
    total = 2 * n - 1

    if mode == "all":
        indices = np.arange(total)
    elif mode == "active":
        if eps > 0:
            active = conv >= (1.0 - eps) * cmax
            indices = np.where(active)[0]
        else:
            indices = np.array([int(np.argmax(conv))])
    elif mode == "topk":
        k = min(50, total)
        indices = np.argsort(conv)[-k:]
    else:
        indices = np.arange(total)

    m = len(indices)

    # Build gradient matrix G: G[idx, i] = 2*a[k-i] if 0 <= k-i < n
    G = np.zeros((m, n))
    for idx, k in enumerate(indices):
        lo = max(0, int(k) - n + 1)
        hi = min(n, int(k) + 1)
        if hi > lo:
            i_range = np.arange(lo, hi)
            a_indices = int(k) - i_range
            G[idx, i_range] = 2.0 * a[a_indices]

    # LP: minimize t
    # s.t. G @ d - t <= -conv[indices]
    #      -d <= a
    #      sum(d) = 0
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.hstack([G, -np.ones((m, 1))])
    b_ub_top = -conv[indices]

    A_ub_nonneg = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b_ub_nonneg = a.copy()

    A_ub_full = np.vstack([A_ub, A_ub_nonneg])
    b_ub_full = np.concatenate([b_ub_top, b_ub_nonneg])

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    bounds = [(None, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            d = res.x[:n]
            t_val = float(res.x[n])
            improvement = cmax - t_val
            return d, t_val, improvement
    except Exception:
        pass
    return None, None, None


def lp_optimize(a, max_iters=50, time_budget=60.0, mode="all", eps=0.0,
                line_search_points=None):
    """Run multiple LP iterations with line search."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    if line_search_points is None:
        line_search_points = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        d, t_val, improvement = lp_direction(a, mode=mode, eps=eps, time_limit=8)
        if d is None:
            break

        if improvement < 1e-13:
            # Try with just active peaks
            d2, t2, imp2 = lp_direction(a, mode="active", eps=0.0, time_limit=5)
            if d2 is not None and imp2 > improvement:
                d, t_val, improvement = d2, t2, imp2
            else:
                break

        # Line search
        best_alpha = 0.0
        best_alpha_obj = best_obj
        best_alpha_a = None

        for alpha in line_search_points:
            a_try = a + alpha * d
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj:
                best_alpha_obj = obj_try
                best_alpha = alpha
                best_alpha_a = a_try.copy()

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-14:
            best_obj = best_alpha_obj
            best_a = best_alpha_a
            a = best_alpha_a
            stall_count = 0
            if it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.10f}, alpha={best_alpha:.3f}, lin_improvement={improvement:.2e}")
        else:
            stall_count += 1
            if stall_count > 8:
                break

    return best_a, best_obj


def freq_smooth(a, max_iters=80, time_budget=20.0):
    """Frequency domain smoothing to reduce peaks."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power_spec = np.abs(fa) ** 2

        ps_max = np.max(power_spec)
        if ps_max < 1e-15:
            break
        ps_norm = power_spec / ps_max
        alpha = 0.85 + 0.12 * (it / max_iters)
        ps_compressed = np.maximum(ps_norm, 1e-12) ** alpha

        phase = np.angle(fa)
        magnitude = np.sqrt(ps_compressed * ps_max)
        fa_new = magnitude * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def freq_smooth_aggressive(a, max_iters=60, time_budget=20.0):
    """More aggressive frequency domain smoothing with peak targeting."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute convolution in frequency domain
        fa = np.fft.fft(a, N)
        conv = np.fft.ifft(fa * fa).real[:2 * n - 1]
        cmax = np.max(conv)

        # Soft-compress the convolution peaks
        tau = 0.05 * cmax
        shifted = np.clip((conv - cmax) / tau, -200, 200)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            break
        weights /= w_sum

        # Modify power spectrum to reduce weighted peaks
        # Reduce power at frequencies that contribute to peaks
        ps = np.abs(fa) ** 2
        ps_max = np.max(ps)
        if ps_max < 1e-15:
            break

        # Compress: blend toward uniform
        blend = 0.1 + 0.2 * (it / max_iters)
        ps_uniform = np.full_like(ps, ps.sum() / N)
        ps_new = (1 - blend) * ps + blend * ps_uniform
        ps_new = np.maximum(ps_new, 1e-15)

        phase = np.angle(fa)
        magnitude = np.sqrt(ps_new)
        fa_new = magnitude * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def chase_peak(a, n_steps=50, step_size=0.05):
    """Chase the peak using subgradient method."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv_fft(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        if hi > lo:
            i_range = np.arange(lo, hi)
            a_indices = k_star - i_range
            grad[i_range] = 2.0 * a[a_indices]

        grad_centered = grad - np.mean(grad)

        for ss in [step_size, step_size * 0.5, step_size * 1.5, step_size * 0.25]:
            a_try = a - ss * grad_centered
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a, best_obj


def softmax_subgradient(a, max_iters=80, time_budget=20.0):
    """Subgradient with softmax weighting over peaks."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv_fft(a)
        cmax = float(np.max(conv))
        tau = max(0.005 * cmax, 1e-8)
        shifted = (conv - cmax) / tau
        weights = np.exp(np.clip(shifted, -30, 30))
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum

        grad = np.zeros(n)
        active_ks = np.where(weights > 1e-8)[0]
        for k in active_ks:
            w = weights[k]
            lo = max(0, int(k) - n + 1)
            hi = min(n, int(k) + 1)
            if hi > lo:
                i_range = np.arange(lo, hi)
                a_indices = int(k) - i_range
                grad[i_range] += w * 2.0 * a[a_indices]

        grad_centered = grad - np.mean(grad)
        step = 0.04 / (1.0 + 0.005 * it)

        a_try = a - step * grad_centered
        a_try = project_nonneg_sum1(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def perturbed_restart(a, rng, sigma=0.02):

            if remain < 12:
                break

            # LP full
            t_lp = min(remain * 0.35, 35.0)
            if remain > 25:
                a_p, obj_p = lp_optimize(a_final, max_iters=15, time_budget=t_lp, mode="all")
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Polish LP {cycle}: {best_score:.10f}")

            # Freq smooth
            remain = deadline - time.monotonic()
            if remain > 15:
                a_p2, obj_p2 = freq_smooth(a_final, max_iters=30, time_budget=min(remain * 0.25, 20.0))
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_final = a_p2.copy()
                    print(f"  Polish FD {cycle}: {best_score:.10f}")

            # Aggressive freq
            remain = deadline - time.monotonic()
            if remain > 15:
                a_p3, obj_p3 = freq_smooth_aggressive(a_final, max_iters=25, time_budget=min(remain * 0.2, 15.0))
                if obj_p3 < best_score:
                    best_score = obj_p3
                    best_seq = a_p3.copy()
                    a_final = a_p3.copy()
                    print(f"  Polish FD-agg {cycle}: {best_score:.10f}")

            # Softmax subgradient
            remain = deadline - time.monotonic()
            if remain > 10:
                a_p4, obj_p4 = softmax_subgradient(a_final, max_iters=40, time_budget=min(remain * 0.2, 12.0))
                if obj_p4 < best_score:
                    best_score = obj_p4
                    best_seq = a_p4.copy()
                    a_final = a_p4.copy()
                    print(f"  Polish SG {cycle}: {best_score:.10f}")

            # Chase peak
            remain = deadline - time.monotonic()
            if remain > 8:
                a_p5, obj_p5 = chase_peak(a_final, n_steps=25, step_size=0.03)
                if obj_p5 < best_score:
                    best_score = obj_p5
                    best_seq = a_p5.copy()
                    a_final = a_p5.copy()

            cycle += 1

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
