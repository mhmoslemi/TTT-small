```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. L-BFGS-B on smooth log-sum-exp approximation with softmax reparametrization
2. Nesterov accelerated projected gradient (FISTA) on the simplex
3. LP-based Frank-Wolfe best response for global-style moves
4. Coordinate descent for fine-grained polishing
Multi-start with diverse initializations and adaptive tau schedule.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def lse_and_grad_a(a, tau):
    """Compute log-sum-exp of conv(a,a) and its gradient w.r.t. a.
    
    Returns (lse_value, grad_a) where grad_a = d/d a [LSE(tau * conv(a,a))]
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    lse_val = b_max + np.log(exp_b.sum()) / tau
    p = exp_b / exp_b.sum()
    
    # Gradient w.r.t. a: h_i = 2 * sum_j p[i+j] * a[j]
    # Using FFT cross-correlation
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(P * np.conj(A), N)
    grad_a = 2.0 * corr[:n]
    
    return lse_val, grad_a


def lbfgs_objective(z, tau, n):
    """Objective and gradient for L-BFGS-B on softmax reparametrization."""
    a = softmax(z)
    lse_val, grad_a = lse_and_grad_a(a, tau)
    f_val = 2.0 * n * lse_val
    dot_a_g = np.dot(a, grad_a)
    grad_z = 2.0 * n * a * (grad_a - dot_a_g)
    return f_val, grad_z


def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def run_fista(a_init, tau_start, tau_end, max_iters, deadline):
    """FISTA on smooth max approximation."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    L = 1.0  # Lipschitz estimate
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        lse_y, grad_y = lse_and_grad_a(y, tau)
        f_y = 2.0 * n * lse_y
        
        # Project gradient to tangent of simplex
        grad_tan = grad_y - np.mean(grad_y)
        
        # Backtracking line search for step size
        step = 1.0 / L
        
        a_new = y - step * grad_tan
        # Check if descent (using the smooth objective)
        a_new_raw = y - step * grad_tan
        # Project to simplex
        a_new = proj_simplex(a_new_raw)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        # Track best (every 3 iterations)
        if it % 3 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def build_lp_matrix(n, a, active_indices):
    """Build sparse LP matrix for active constraints.
    
    Primal: max sum(g) s.t. (a*g)[k] <= 1 for k in active, g >= 0
    """
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response(a, M, n_active, deadline_lp):
    """Solve LP best response."""
    n = len(a)
    b = fast_conv(a)
    top_indices = np.argsort(b)[::-1][:min(n_active, 2*n-1)]
    
    A = build_lp_matrix(n, a, top_indices.tolist())
    if A is None or A.nnz == 0:
        return None
    
    c = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': min(10, max(1, (deadline_lp - time.monotonic())))})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_fw(a_init, max_iters, deadline, n_active=30):
    """LP-based Frank-Wolfe / best response iteration."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = lp_best_response(a, best_M, n_active, deadline)
        if g is None:
            no_improve += 2
            if no_improve > 10:
                break
            continue
        
        # Exact line search along a -> g
        best_t = 0.0
        best_f = best_M
        for t in [0.3, 0.5, 0.2, 0.7, 0.1, 0.4, 0.6, 0.15, 0.25]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_f:
                best_f = M_trial
                best_t = t
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_M = best_f
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break
    
    return best_a


def run_coord_descent(a_init, max_sweeps, deadline):
    """Coordinate descent with exact 1D optimization."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        improved = False
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            x_cur = a[i]
            
            # Compute affected convolution values
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Base convolution values (without a[i] contribution)
            c_vals = np.zeros(k_hi - k_lo + 1)
            slopes = np.zeros(k_hi - k_lo + 1)
            has_quad = False
            quad_idx = -1
            
            for k in range(k_lo, k_hi + 1):
                idx = k - k_lo
                if k == 2 * i:
                    c_vals[idx] = b[k] - x_cur * x_cur
                    slopes[idx] = 0.0
                    has_quad = True
                    quad_idx = idx
                else:
                    j = k - i
                    if 0 <= j < n:
                        c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                        slopes[idx] = 2.0 * a[j]
                    else:
                        c_vals[idx] = b[k]
                        slopes[idx] = 0.0
            
            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            
            def obj_at(x):
                vals = c_vals + slopes * x
                m = base_max
                if vals.size > 0:
                    m = max(m, vals.max())
                if has_quad:
                    m = max(m, c_vals[quad_idx] + x * x)
                return m
            
            # Golden section search
            lo = 0.0
            hi = max(2.0 * x_cur + 0.1, 0.5)
            
            # Ensure hi is large enough
            for _ in range(5):
                if obj_at(hi) > obj_at(hi * 0.5) and hi < 5.0:
                    hi *= 2.0
                else:
                    break
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = obj_at(c_pt)
            fd = obj_at(d_pt)
            
            for _ in range(30):
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = obj_at(c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = obj_at(d_pt)
            
            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0
            
            x_opt = max(0.0, x_opt)
            
            if abs(x_opt - a[i]) > 1e-12:
                # Update b
                for k in range(k_lo, k_hi + 1):
                    idx = k - k_lo
                    if k == 2 * i:
                        b[k] = c_vals[idx] + x_opt * x_opt
                    else:
                        b[k] = c_vals[idx] + slopes[idx] * x_opt
                a[i] = x_opt
                improved = True
        
        a = a / a.sum()
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 10:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.02, n), 1e-4)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.05, n), 1e-4)
        inits.append(p_n2 / p_n2.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())
    
    # Parabolic
    poly = (t * (1 - t)) + 0.1
    inits.append(poly / poly.sum())
    
    # Gaussian
    gauss = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Pseudo-random (smoothed)
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits
