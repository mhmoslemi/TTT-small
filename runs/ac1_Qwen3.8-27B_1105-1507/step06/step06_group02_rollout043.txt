def lp_close_direction(a, eps_pen=1e-6, active_eps=1e-5):
    n=len(a)
    c=fast_conv(a); M=float(c.max())
    active = np.where(c >= M - active_eps*M)[0]
    rows,cols,vals,offs = get_sparsity(n)
    adata = a[vals]
    active_count = len(active)
    nabs = 2*n
    total_rows = active_count + nabs
    nvar = 1 + 2*n
    q_start=1; u_start=1+n
    r_rows=[]; r_cols=[]; r_vals=[]
    b_ub = np.zeros(total_rows)
    # active constraints
    for pos,k in enumerate(active):
       r = pos
       # -t
       r_rows.append(r); r_cols.append(0); r_vals.append(-1.0)
       start=offs[int(k)]; end=offs[int(k)+1]
       for idx in range(start,end):
           r_rows.append(r); r_cols.append(q_start + int(cols[idx])); r_vals.append(float(adata[idx]))
       # rhs 0
    # abs constraints
    row_idx = active_count
    for i in range(n):
       # q_i - u_i <= p_i
       r_rows.append(row_idx); r_cols.append(q_start+i); r_vals.append(1.0)
       r_rows.append(row_idx); r_cols.append(u_start+i); r_vals.append(-1.0)
       b_ub[row_idx] = float(a[i])
       row_idx +=1
       # -q_i - u_i <= -p_i
       r_rows.append(row_idx); r_cols.append(q_start+i); r_vals.append(-1.0)
       r_rows.append(row_idx); r_cols.append(u_start+i); r_vals.append(-1.0)
       b_ub[row_idx] = -float(a[i])
       row_idx +=1
    A = csr_matrix((r_vals, (r_rows, r_cols)), shape=(total_rows, nvar))
    cobj = np.zeros(nvar)
    cobj[0]=1.0
    cobj[u_start:u_start+n] = eps_pen
    eq_data=np.ones(n); eq_rows=np.zeros(n); eq_cols=np.arange(q_start,u_start)
    Aeq=csr_matrix((eq_data,(eq_rows,eq_cols)), shape=(1,nvar))
    bounds=[(None,None)] + [(0,None)]*n + [(0,None)]*n
    try:
       r=linprog(cobj, A_ub=A, b_ub=b_ub, A_eq=Aeq, b_eq=np.array([1.0]), bounds=bounds, method='highs', options={'presolve':True})
    except Exception:
       return None
    if r.success:
       q=r.x[q_start:u_start]
       q=np.maximum(q,0); s=q.sum()
       if s>1e-12: q/=s
       return q
    return None
```

"""
Search for a non-negative sequence a minimizing
    J(a) = 2*n*max_k conv(a,a)_k / (sum a)^2
using a hybrid of:
  1) spectral polynomial-square projection (P^2 target flattening),
  2) annealed log-sum-exp projected gradient / Nesterov,
  3) active-set LP direction search,
  4) LP best-response mirror descent,
  5) exact coordinate descent polish.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

_SPARSITY_CACHE = {}


def _next_pow2(x):
    s = 1
    while s < x:
        s <<= 1
    return s


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(a.size, dtype=np.float64)
        return a / float(a.size)
    return a / s


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-12:
        return float("inf")
    b = fast_conv(a)
    if b.size == 0:
        return float("inf")
    return float(2.0 * a.size * float(b.max()) / (s * s))


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == n:
        return src.copy()
    if src.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    t_old = np.linspace(0.0, 1.0, src.size, dtype=np.float64)
    t_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(t_new, t_old, src, dtype=np.float64)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return out / s


def uniform_target_root(n):
    m = 2 * n - 1
    d = np.ones(m, dtype=np.float64) / float(m)
    size = _next_pow2(m)
    D = np.fft.fft(d, size)
    P = np.sqrt(D)
    q = np.fft.ifft(P).real[:n]
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s <= 1e-300:
        q = np.ones(n, dtype=np.float64)
        return q / float(n)
    return q / s


def random_smooth_init(n, rng):
    a = rng.uniform(0.2, 1.0, size=n)
    k = max(3, n // 25)
    kernel = np.ones(k, dtype=np.float64) / float(k)
    a = np.convolve(a, kernel, mode="same")
    a = np.maximum(a, 0.0)
    a += 0.05
    return normalize_nonneg(a)


def spectral_square_optimize(a0, deadline, max_iters=20000, mode="floor",
                             start=0.95, end=0.3, step=1.0, smooth_k=0):
    """
    Alternating projections:
      time domain: truncate to n, nonnegative, normalize
      frequency domain: P = sqrt(FFT(target)), so P^2 matches target
    """
    a = normalize_nonneg(a0)
    n = a.size
    if n == 0:
        return a

    best = a.copy()
    best_m = float(fast_conv(a).max())

    m_len = 2 * n - 1
    size = _next_pow2(m_len)
    U = np.ones(m_len, dtype=np.float64) / float(m_len)

    kern = None
    if smooth_k > 0:
        w = 2 * int(smooth_k) + 1
        kern = np.ones(w, dtype=np.float64) / float(w)

    it = 0
    while it < max_iters:
        if time.monotonic() >= deadline:
            break
        it += 1
        c = fast_conv(a)
        M = float(c.max())
        if M < best_m * (1.0 - 1e-13):
            best_m = M
            best = a.copy()

        frac = min(1.0, float(it) / 2000.0)
        f = float(start) + float(end - start) * frac

        if mode == "uniform":
            d = U.copy()
        elif mode == "mix":
            cn = c.copy()
            cs = float(cn.sum())
            if cs > 0:
                cn /= cs
            d = (1.0 - f) * cn + f * U
        else:  # floor
            d = np.maximum(c, f * M)

        ds = float(d.sum())
        if ds <= 1e-300:
            break
        d = d / ds

        if kern is not None:
            d = np.convolve(d, kern, mode="same")
            d = np.maximum(d, 0.0)
            ds = float(d.sum())
            if ds > 0.0:
                d = d / ds

        D = np.fft.fft(d, size)
        P = np.sqrt(D)
        q = np.fft.ifft(P).real[:n]
        q = np.maximum(q, 0.0)
        qs = float(q.sum())
        if qs <= 1e-300:
            continue
        q = q / qs

        if step < 1.0:
            q = (1.0 - step) * a + step * q
            qs = float(q.sum())
            if qs <= 1e-300:
                q = a.copy()
            else:
                q = q / qs

        Mq = float(fast_conv(q).max())
        if Mq < best_m * (1.0 - 1e-13):
            best_m = Mq
            best = q.copy()
        a = q

    return best


def smooth_grad_fft(a, tau):
    n = a.size
    b = fast_conv(a)
    bmax = float(b.max())
    x = tau * (b - bmax)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 1e-300 or not np.isfinite(s):
        w = np.zeros_like(b)
        idx = int(np.argmax(b))
        if 0 <= idx < b.size:
            w[idx] = 1.0
        s = 1.0
    else:
        w = ex / s
    lse = float(np.log(s))
    val = bmax + (1.0 / tau) * lse

    size = _next_pow2(3 * n - 1)
    Fp = np.fft.rfft(a[::-1], size)
    Fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(Fp * Fw, size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    grad[~np.isfinite(grad)] = 0.0
    return val, grad, b, bmax


def run_nesterov_smooth(a0, deadline, max_iters=5000,
                        tau_start=1.0, tau_end=5000.0,
                        lr_init=0.2, lr_end=0.003, beta=0.9):
    a = normalize_nonneg(a0)
    best = a.copy()
    cur_m = float(fast_conv(a).max())
    best_m = cur_m
    y = a.copy()

    t_const = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
    momentum = (t_const - 1.0) / (t_const + 1.0)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = (it + 1) / float(max_iters)
        if tau_end <= tau_start:
            tau = tau_start
        else:
            tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_init * (lr_end / lr_init) ** frac

        try:
            _v, grad, _b, _bm = smooth_grad_fft(y, tau)
        except Exception:
            break

        a_prev = a
        update = grad + beta * (y - a_prev)
        un = float(np.linalg.norm(update))
        if not np.isfinite(un):
            update = grad
            un = float(np.linalg.norm(update))
        if un > 50.0:
            update = update * (50.0 / un)

        a_try = y - lr * update
        a_try = np.maximum(a_try, 0.0)
        s_try = float(a_try.sum())
        if s_try > 1e-300:
            a_try = a_try / s_try
            m_try = float(fast_conv(a_try).max())
        else:
            a_try = a_prev.copy()
            m_try = cur_m

        chosen = a_prev
        chosen_m = cur_m
        if m_try <= cur_m * 1.001 + 1e-15:
            chosen = a_try
            chosen_m = m_try
        else:
            a_half = y - 0.5 * lr * update
            a_half = np.maximum(a_half, 0.0)
            sh = float(a_half.sum())
            if sh > 1e-300:
                a_half = a_half / sh
                m_half = float(fast_conv(a_half).max())
                if m_half <= cur_m * 1.001 + 1e-15 and m_half < m_try:
                    chosen = a_half
                    chosen_m = m_half

        if chosen_m < best_m:
            best_m = chosen_m
            best = chosen.copy()
        a = chosen
        cur_m = chosen_m

        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = float(y.sum())
        if sy > 1e-300:
            y = y / sy
        else:
            y = a.copy()

    return best


def run_adam_smooth(a0, deadline, max_iters=5000, lr=0.003,
                    tau_start=10.0, tau_end=100000.0,
                    beta1=0.9, beta2=0.999):
    a = normalize_nonneg(a0)
    best = a.copy()

    m = np.zeros_like(a)
    v = np.zeros_like(a)

        if tau_end <= tau_start:

        else:

        try:
            _val, grad, _b, _bm = smooth_grad_fft(a, tau)
        except Exception:
            break

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        mhat = m / max(bc1, 1e-12)
        vhat = v / max(bc2, 1e-12)
        step = lr * mhat / (np.sqrt(vhat) + 1e-8)

        chosen = a

                chosen_m = mc
                break

        if chosen_m < best_m:

            best = chosen.copy()
        a = chosen
        cur_m = chosen_m

def get_sparsity(n):
    if n in _SPARSITY_CACHE:
        return _SPARSITY_CACHE[n]
    rows = []
    cols = []
    vals = []
    offsets = [0]
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(k - j)
        offsets.append(len(rows))
    res = (
        np.asarray(rows, dtype=np.int32),
        np.asarray(cols, dtype=np.int32),
        np.asarray(vals, dtype=np.int32),
        np.asarray(offsets, dtype=np.int32),
    )
    _SPARSITY_CACHE[n] = res
    return res


def lp_best_response(a, M):
    n = a.size
    if n == 0 or M <= 0.0:
        return None
    rows, cols, vals, _off = get_sparsity(n)
    A = csr_matrix((a[vals], (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, float(M), dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True}
        )
    except Exception:
        return None
    if r.success and r.x is not None:
        q = np.asarray(r.x, dtype=np.float64)
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s > 1e-12:
            return q / s
    return None


def line_search_mix(p, q, cur_m):
    best_p = p.copy()
    best_m = cur_m
    alphas = (1.5, 1.2, 1.0, 0.7, 0.5, 0.3, 0.15, 0.07, 0.03, 0.01)
    for alpha in alphas:
        r = p + alpha * (q - p)
        r = np.maximum(r, 0.0)
        s = float(r.sum())
        if s <= 1e-300:
            continue
        r = r / s
        m = float(fast_conv(r).max())
        if m < best_m - 1e-15:
            best_m = m
            best_p = r.copy()
    return best_p, best_m

def active_lp_direction(a, active_eps=1e-3, eps_pen=1e-6, bound=None):
    n = a.size
    if n == 0:
        return None
    c = fast_conv(a)
    M = float(c.max())
    if M <= 0.0:
        return None

    if bound is None:
        mean = float(a.sum()) / float(n)
        mx = float(a.max())
        bound = max(0.05, 5.0 * mean, 2.0 * mx)

    eps = max(1e-8, active_eps * M)
    active = np.where(c >= M - eps)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(c))], dtype=np.int32)

    rows, cols, vals, offsets = get_sparsity(n)
    adata = 2.0 * a[vals]

    active_count = int(active.size)
    total_rows = active_count + 2 * n
    nvar = 1 + 3 * n
    d_start = 1
    u_start = 1 + n
    v_start = 1 + 2 * n

    rr = []
    cc = []
    vv = []
    b_ub = np.zeros(total_rows, dtype=np.float64)

    for pos, k in enumerate(active):
        r = pos
        rr.append(r)
        cc.append(0)
        vv.append(-1.0)
        start = int(offsets[int(k)])
        end = int(offsets[int(k) + 1])
        for idx in range(start, end):
            rr.append(r)
            cc.append(int(d_start + cols[idx]))
            vv.append(float(adata[idx]))
        b_ub[r] = -float(c[int(k)])

    for i in range(n):
        r1 = active_count + i
        rr.append(r1)
        cc.append(d_start + i)
        vv.append(1.0)
        rr.append(r1)
        cc.append(u_start + i)
        vv.append(-1.0)

        r2 = active_count + n + i
        rr.append(r2)
        cc.append(d_start + i)
        vv.append(-1.0)
        rr.append(r2)
        cc.append(v_start + i)
        vv.append(-1.0)

    A = csr_matrix((vv, (rr, cc)), shape=(total_rows, nvar))
    cobj = np.zeros(nvar, dtype=np.float64)
    cobj[0] = 1.0
    cobj[u_start:u_start + n] = eps_pen
    cobj[v_start:v_start + n] = eps_pen

    eq_cols = np.arange(d_start, d_start + n, dtype=np.int32)
    Aeq = csr_matrix((np.ones(n, dtype=np.float64), (np.zeros(n, dtype=np.int32), eq_cols)),

    b_eq = np.array([0.0], dtype=np.float64)

    try:

            cobj,
            A_ub=A,
            b_ub=b_ub,
            A_eq=Aeq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True}
        )
    except Exception:
        return None

    if r.success and r.x is not None:
        d = np.asarray(r.x[d_start:d_start + n], dtype=np.float64)
        d[~np.isfinite(d)] = 0.0
        if float(np.max(np.abs(d))) > 0.0:
            return d
    return None


def line_search_direction(p, d, cur_m):
    best_p = p.copy()
    best_m = cur_m
    dmax = float(np.max(np.abs(d)))
    if dmax <= 1e-14:
        return best_p, best_m

    pmean = float(p.sum()) / float(p.size)
    base = pmean / dmax
    alphas = (
        0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 1.0, 3.0,
        base * 0.1, base * 0.3, base * 1.0, base * 3.0
    )
    for alpha in alphas:
        if alpha <= 0.0 or not np.isfinite(alpha):
            continue
        r = p + alpha * d
        r = np.maximum(r, 0.0)
        s = float(r.sum())
        if s <= 1e-300:
            continue
        r = r / s
        m = float(fast_conv(r).max())
        if m < best_m - 1e-15:
            best_m = m
            best_p = r.copy()
    return best_p, best_m

def coordinate_polish(a0, deadline, rng=None, max_cycles=30):

    n = a.size

        return a

    stagnation = 0

    while cycle < max_cycles:
        if time.monotonic() >= deadline:
            break
        if rng is not None:
            rng.shuffle(order)

        a = best.copy()
        c = fast_conv(a)
        B = float(c.max())
        a_sum = 1.0

        improved = False

                break
            i = int(order[ii])
            x = float(a[i])
            S0 = a_sum - x
            mean = a_sum / float(n)

            if i > 0:
                left_max = float(c[:i].max())
            else:
                left_max = -1e300
            out_start = i + n
            if out_start < c.size:
                right_max = float(c[out_start:].max())
            else:
                right_max = -1e300
            Umax = max(0.0, left_max, right_max)

            if x > 0.0:

                y1 = x * np.asarray(ratios, dtype=np.float64)
                extra = mean * np.asarray((0.5, 1.0, 2.0), dtype=np.float64)
                y_cands = np.concatenate((y1, extra))
            else:
                y_cands = mean * np.asarray((0.1, 0.3, 1.0, 3.0, 10.0), dtype=np.float64)

            y_cands = y_cands[np.isfinite(y_cands)]
            y_cands = y_cands[y_cands >= 0.0]
            ymax = max(1.0, 5.0 * mean, 20.0 * x if x > 0.0 else 0.0)
            y_cands = y_cands[y_cands <= ymax]
            if y_cands.size == 0:
                y_cands = np.asarray([x], dtype=np.float64)
            y_cands = np.unique(y_cands)

            c_aff = c[i:i + n]
            deltas = y_cands - x
            new_aff = c_aff[None, :] + 2.0 * deltas[:, None] * a[None, :]
            new_aff[:, i] += deltas * deltas
            Bcands = np.maximum(Umax, new_aff.max(axis=1))
            Scands = S0 + y_cands
            Scands = np.maximum(Scands, 1e-12)
            objs = (2.0 * n * Bcands) / (Scands * Scands)

            cur_local = (2.0 * n * B) / (a_sum * a_sum)
            idx = int(np.argmin(objs))
            best_y = float(y_cands[idx])
            best_local = float(objs[idx])

            if abs(best_y - x) > 1e-13 and best_local < cur_local - 1e-13:
                old_aff = c[i:i + n].copy()
                old_c2 = c[2 * i]
                delta = best_y - x

                c[i:i + n] = old_aff + 2.0 * delta * a
                c[2 * i] = old_c2 + delta * delta
                a[i] = best_y
                a_sum += delta

                aff_max = float(c[i:i + n].max())
                B = max(Umax, aff_max)
                cur_obj = best_local
                improved = True

        a_norm = normalize_nonneg(a)
        score = compute_J(a_norm)
        if np.isfinite(score) and score < best_obj - 1e-14:
            best_obj = score
            best = a_norm.copy()
            stagnation = 0
        else:
            if improved:
                stagnation += 1
                if stagnation > 3:
                    break
            else:
                break

        cycle += 1

    return best


class BestTracker:
    def __init__(self):
        self.best = None
        self.score = float("inf")
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s <= 1e-12:
            a = normalize_nonneg(a)
        else:
            a = a / s
        score = compute_J(a)
        if not np.isfinite(score):
            return
        if score < self.score - 1e-12:
            self.best = a.copy()
            self.score = score
            el = time.monotonic() - self.t0
            print(f"  ** improved J={score:.9f} n={a.size} t={el:.1f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(60.0, float(budget_s) - 25.0)
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = normalize_nonneg(p)
    except Exception:
        parent = None

    if parent is None:
        parent = uniform_target_root(1024)
        print("No valid parent found; using uniform-target root.", flush=True)

    best.update(parent)

    n_parent = parent.size
    lengths_set = set()
    for factor in (0.875, 1.0, 1.125, 1.25, 1.5):
        nn = int(round(n_parent * factor))
        if 128 <= nn <= 3000:
            lengths_set.add(nn)
    lengths = sorted(lengths_set)
    if len(lengths) == 0:
        lengths = [n_parent]
    print(f"Trying lengths: {lengths}", flush=True)

    per_len = {}
    for nn in lengths:
        a0 = resample_seq(parent, nn)
        s0 = compute_J(a0)
        per_len[nn] = (s0, a0.copy())
        best.update(a0)

    # Phase A: spectral square-root evolution, cheap and globally informative.
    spec_total = min(180.0, max(60.0, (deadline - t0) * 0.18))
    spec_deadline = t0 + spec_total
    total_runs = max(1, len(lengths) * 3 * 2)
    alloc = max(2.0, spec_total / float(total_runs))

    for nn in lengths:
        if time.monotonic() > spec_deadline:
            break
        init1 = resample_seq(best.best if best.best is not None else parent, nn)
        init2 = uniform_target_root(nn)
        init3 = random_smooth_init(nn, rng)

        for a0 in (init1, init2, init3):
            sub = min(time.monotonic() + alloc, spec_deadline)
            if sub <= time.monotonic():
                continue
            a1 = spectral_square_optimize(
                a0, sub,
                max_iters=100000,
                mode="floor",
                start=0.95, end=0.3,
                step=1.0, smooth_k=0
            )
            a2 = a1
            if time.monotonic() + 1.0 < spec_deadline:
                sub2 = min(time.monotonic() + alloc * 0.5, spec_deadline)
                a2 = spectral_square_optimize(
                    a1, sub2,
                    max_iters=100000,
                    mode="mix",
                    start=0.1, end=0.7,
                    step=0.8, smooth_k=0
                )

            for cand in (a1, a2):
                if cand is None:
                    continue
                sc = compute_J(cand)
                if np.isfinite(sc):
                    best.update(cand)
                    old = per_len.get(nn)
                    if old is None or sc < old[0]:
                        per_len[nn] = (sc, cand.copy())

    # Extra global spectral push on the current best sequence.
    if time.monotonic() < deadline and best.best is not None:
        extra = min(120.0, max(30.0, (deadline - time.monotonic()) * 0.12))
        sub = time.monotonic() + extra
        auni = spectral_square_optimize(
            best.best, sub,
            max_iters=200000,
            mode="uniform",
            step=0.8, smooth_k=1
        )
        sc = compute_J(auni)
        if np.isfinite(sc):
            best.update(auni)
            nn = auni.size
            old = per_len.get(nn)
            if old is None or sc < old[0]:
                per_len[nn] = (sc, auni.copy())

    # Phase B: annealed smooth optimization on the strongest lengths.
    rem = deadline - time.monotonic()
    smooth_total = min(350.0, max(60.0, rem * 0.30))
    if smooth_total > 30.0:
        items = [(float(s), int(nn), a) for nn, (s, a) in per_len.items()]
        items.sort()
        top = items[:3]
        share = smooth_total / max(1, len(top))
        for s, nn, a in top:
            if time.monotonic() > deadline - 60.0:
                break
            sub = min(time.monotonic() + max(10.0, share), deadline - 60.0)
            aout = smooth_sequence(a, sub)
            sc = compute_J(aout)
            if np.isfinite(sc):
                best.update(aout)
                old = per_len.get(nn)
                if old is None or sc < old[0]:
                    per_len[nn] = (sc, aout.copy())

    # Phase C: LP best-response mirror descent.
    rem = deadline - time.monotonic()
    lp_total = min(650.0, max(80.0, rem * 0.55))
    if lp_total > 30.0:
        items = [(float(s), int(nn), a) for nn, (s, a) in per_len.items()]
        items.sort()
        top = items[:2]
        share = lp_total / max(1, len(top))
        for s, nn, a in top:
            if time.monotonic() > deadline - 80.0:
                break
            sub = min(time.monotonic() + max(20.0, share), deadline - 80.0)
            if nn <= 1200:
                iters = int(max(80, share / 0.25))
            else:
                iters = int(max(30, share / 0.8))
            aout = run_lp_mirror(a, sub, max_iters=iters)
            sc = compute_J(aout)
            if np.isfinite(sc):
                best.update(aout)
                old = per_len.get(nn)
                if old is None or sc < old[0]:
                    per_len[nn] = (sc, aout.copy())

    # Phase D: active LP direction search on the current best.
    rem = deadline - time.monotonic()
    act_total = min(200.0, max(40.0, rem * 0.25))
    if act_total > 30.0 and best.best is not None:
        sub = time.monotonic() + act_total
        aout = run_active_lp(best.best, sub, max_iters=300, active_eps=1e-3)
        sc = compute_J(aout)
        if np.isfinite(sc):
            best.update(aout)
            nn = aout.size
            old = per_len.get(nn)
            if old is None or sc < old[0]:
                per_len[nn] = (sc, aout.copy())

    # Phase E: exact coordinate polish.
    rem = deadline - time.monotonic()
    coord_total = min(180.0, max(30.0, rem * 0.80))
    if coord_total > 20.0 and best.best is not None:
        sub = time.monotonic() + coord_total
        aout = coordinate_polish(best.best, sub, rng=rng, max_cycles=30)
        sc = compute_J(aout)
        if np.isfinite(sc):
            best.update(aout)

    if best.best is None:
        best.best = parent
        best.score = compute_J(parent)

    out = normalize_nonneg(best.best)
    print(f"FINAL J={compute_J(out):.9f} n={out.size} t={time.monotonic()-t0:.1f}s", flush=True)
    return [float(x) for x in out]
