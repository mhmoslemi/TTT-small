def evaluate_sequence(seq: np.ndarray) -> float:
    """Evaluates a sequence of coefficients applying clipping and checks sum constraint.
    Implementation details: Clips negatives to 0 and values >1000 to 1000.
    Normalizes to sum at least 0.01 (raises np.inf if not).
    Computes full linear convolution (size 2n-1) and extracts the maximum.
    Returns objective = 2n * max(conv) / (sum(seq))**2.
    - Parameters:seq (ndarray): Input sequence of coefficients (must be nonnegative first!).
    - Returns:float: The objective value or np.inf if invalid.
    """
    seq = seq.astype(float)

    seq = np.clip(seq, 0, 1000)
