"""
Advanced multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses corrected spectral equalization (targeting rectangular window spectrum),
Nesterov-accelerated soft-subgradient descent, LP alternating refinement,
structured random restarts, and coordinate descent polishing.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft, rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp(a, M, time_limit=60):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section_search(f, lo, hi, tol=0.01, max_iter=20):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def spectral_equalization_v2(a, max_iters=80, time_budget=60.0):
    """
    POCS-style spectral equalization with CORRECT target spectrum.
    Target: power spectrum proportional to |DFT(rectangular window of length 2n-1)|^2.
    This corresponds to a flat linear autocorrelation.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    # Target power spectrum: |DFT of rect window|
    rect = np.zeros(N)
    rect[:2 * n - 1] = 1.0
    target_P = np.abs(fft(rect, N)) ** 2
    target_P = target_P / (np.sum(target_P) + 1e-30)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, N)
        P = np.abs(A) ** 2
        P = P / (np.sum(P) + 1e-30)
        phase = np.angle(A)

        # Geometric interpolation toward target (better than arithmetic)
        alpha = 0.15 + 0.25 * (it / float(max_iters))
        P_new = P ** (1 - alpha) * (target_P + 1e-15) ** alpha
        P_new = P_new / (np.sum(P_new) + 1e-30)

        mag_new = np.sqrt(P_new)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, N))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.6 * a + 0.4 * a_new
            a = normalize(a)

    return best_a

def accelerated_subgradient(a, max_steps=500, time_budget=120.0,
                            init_lr=0.08, momentum=0.9):
    """
    Nesterov-accelerated projected subgradient with soft-max weighting.
    Uses backtracking line search and adaptive temperature.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)  # Momentum variable
    lr = init_lr
    tau = 0.01  # Softmax temperature

    size = 1
    while size < 3 * n:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov look-ahead
        y = a + momentum * v

        # Compute convolution at y
        conv = compute_conv(y)
        cmax = float(np.max(conv))

        # Adaptive temperature: start broad, narrow down
        tau = 0.005 + 0.05 / (1.0 + step * 0.05)
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient: g[i] = 2 * sum_k weights[k] * (T_k * a)[i]
        # (T_k * a)[i] = a[k-i] + a[i+k] (with valid index checks)
        # Efficiently via correlation
        w_ext = weights
        a_flip = a[::-1]
        Wf = rfft(w_ext, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Project gradient onto sum=1 hyperplane
        grad_centered = grad - np.mean(grad)

        # Update with momentum
        v = momentum * v - lr * grad_centered
        a_candidate = normalize(a + v)

        # Backtracking line search
        obj_candidate = compute_obj(a_candidate)
        if obj_candidate < best_obj:
            best_obj = obj_candidate
            best_a = a_candidate.copy()
            a = a_candidate
            lr = min(lr * 1.05, 0.15)
        else:
            # Try smaller step
            for _ in range(3):
                lr *= 0.5
                v = momentum * v - lr * grad_centered
                a_candidate = normalize(a + v)
                obj_candidate = compute_obj(a_candidate)
                if obj_candidate < compute_obj(a):
                    a = a_candidate
                    break
            else:
                # Revert momentum
                v = 0.5 * v

    return best_a


def soft_max_gradient(a, tau_init=0.01, step_size=0.05, max_steps=300, time_budget=60.0):
    """Standard soft-max gradient descent (no momentum) for reliability."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 3 * n:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        tau = tau_init * (0.95 ** step) + 0.001
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        grad_centered = grad - np.mean(grad)

        a_new = normalize(a - step_size * grad_centered)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step_size = min(step_size * 1.02, 0.1)
        else:
            step_size *= 0.6
            if step_size < 1e-8:
                break

    return best_a


def lp_alternate(a, max_iters=30, time_budget=120.0):
    """LP alternating with golden section line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=max(5, int(remaining * 0.5)))
        if g is None:
            break

        def obj_at_t(t):
            a_try = normalize((1.0 - t) * a + t * g)
            return compute_obj(a_try)

        t_opt = golden_section_search(obj_at_t, 0.0, 1.0, tol=0.02, max_iter=12)
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            rng = np.random.default_rng((it * 6151 + 777) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def coordinate_descent(a, time_budget=60.0):
    """Targeted coordinate descent: move mass away from peak contributors."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        pairs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                pairs.append((a[j] * a[k2], j, k2))
        pairs.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in pairs[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            # Try moving mass from the larger contributor
            src = j1 if a[j1] >= a[j2] else j2
            for _ in range(4):
                dst = int(rng.integers(0, n))
                if dst == src:
                    continue
                # Prefer moving to positions that don't contribute to the peak
                if dst + (k_star - src) == k_star and 0 <= k_star - src < n:
                    continue

                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[src]
                    if delta <= 0:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[dst] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved and rng.random() < 0.05:
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.1, 0.9, n)
    x_new = np.linspace(0.1, 0.9, new_n)
    return np.interp(x_new, x_old, a)

        else:
            print(f"  Skipping LP (n={n_cur} > 900)")

    # =========================================================
    # Phase 4: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best - 50, n_best,
            n_best + 50, n_best + 100, n_best + 200
        ]))
        n_options = [n for n in n_options if 200 <= n <= 1100]

        for n_try in n_options:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            if n_try == len(best_seq):
                a_try = normalize(best_seq.copy())
            else:
                a_try = normalize(resample_sequence(best_seq, n_try))

            alloc = min(remaining * 0.15, 40.0)

            # Quick optimization at this n
            a_try = soft_max_gradient(a_try, tau_init=0.008, step_size=0.04,
                                      max_steps=80, time_budget=alloc * 0.5)
            if n_try <= 800 and alloc > 20:
                a_try = lp_alternate(a_try, max_iters=8, time_budget=alloc * 0.4)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 5: Final polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        n_final = len(best_seq)
        a_final = normalize(best_seq.copy())

        for rnd in range(3):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # Coordinate descent
            alloc_cd = min(remaining * 0.3, 40.0)
            a_cd = coordinate_descent(a_final, time_budget=alloc_cd)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                a_final = a_cd
                print(f"  CD round {rnd}: {best_score:.8f} **")

            # Short LP if feasible
            if n_final <= 800 and remaining > 30:
                a_lp = lp_alternate(a_final, max_iters=10,
                                    time_budget=min(remaining * 0.25, 40.0))
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_final = a_lp
                    print(f"  LP round {rnd}: {best_score:.8f} **")

            # Short accelerated subgradient
            if remaining > 20:
                a_sub = accelerated_subgradient(a_final, max_steps=100,
                                                time_budget=min(remaining * 0.2, 25.0),
                                                init_lr=0.02, momentum=0.8)
                obj_sub = compute_obj(a_sub)
                if obj_sub < best_score:
                    best_score = obj_sub
                    best_seq = a_sub.copy()
                    a_final = a_sub
                    print(f"  Subgrad round {rnd}: {best_score:.8f} **")

            if not (obj_cd < best_score or (n_final <= 800 and remaining > 30)):
                break

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
