            best_obj = true_obj

            print(f"  L-BFGS stage p=[{p_s:.0f},{p_e:.0f}]: {true_obj:.8f} ***")
        else:
            print(f"  L-BFGS stage p=[{p_s:.0f},{p_e:.0f}]: {true_obj:.8f}")
    
    return best_a, best_obj


def adam_optimize(a_init, time_budget, beta_start=10.0, beta_end=500.0,
                  lr_start=0.05, lr_decay=0.99998, max_iters=50000):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a_init)
    s = np.sum(a_init)
    if s < 1e-10:
        return a_init
    a_init = a_init / s
    a_init = np.maximum(a_init, 1e-15)
    
    x = np.log(a_init)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_c = x - np.max(x)
        a = np.exp(x_c)
        a = a / np.sum(a)
        
        if beta >= 50:
            _, grad_a, _, _ = logsumexp_grad(a, beta=beta)
        else:
            _, grad_a, _, _ = soft_pnorm_grad(a, max(beta, 2.0))
        
        grad_x = grad_a * a
        
        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 500 == 0:
            x -= np.mean(x)
        
        if it % 200 == 0 or it == max_iters - 1:
            x_cc = x - np.max(x)
            a_check = np.exp(x_cc)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()
    
    return best_a


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_refine(a, time_budget=40.0):
    """LP-based refinement."""
    n = len(a)
    if n > 600:
        return a
    
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    
    conv = compute_conv(a_norm)
    M = float(np.max(conv))
    
    A = build_toeplitz(a_norm)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': min(time_budget * 0.5, 20)})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            sg = float(np.sum(g))
            if sg > 1e-10:
                g = g / sg
                best_a = a.copy()
                best_obj = compute_obj(a)
                
                for t_blend in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - t_blend) * a_norm + t_blend * g
                    a_try = np.maximum(a_try, 0)
                    st = np.sum(a_try)
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a
    except Exception:
        pass
    
    return a


def directed_peak_reduction(a, time_budget=30.0, peak_threshold=0.95):
    """
    Directed peak reduction: iteratively reduce the hottest peaks by adjusting
    the sequence elements that contribute most to them.
    """
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for iteration in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(best_a)
        max_b = float(np.max(conv))
        threshold = peak_threshold * max_b
        hot_lags = np.where(conv >= threshold)[0]
        
        if len(hot_lags) == 0:
            break
        
        # For each hot lag, find the elements contributing most
        moved = False
        for k in hot_lags[:5]:  # Focus on top 5 peaks
            # c[k] = sum_j a[j]*a[k-j]
            # Compute contribution of each pair
            i_vals = np.arange(max(0, k - n + 1), min(k, n - 1) + 1)
            j_vals = k - i_vals
            contribs = best_a[i_vals] * best_a[j_vals]
            
            if len(contribs) == 0:
                continue
            
            # Find the pair with maximum contribution
            max_idx = np.argmax(contribs)
            i = i_vals[max_idx]
            j = j_vals[max_idx]
            
            # Try to reduce one of i or j and compensate elsewhere
            # Choose the element to reduce: the one with more "freedom"
            # (i.e., contributing less to other hot peaks)
            
            # Try reducing a[i] by a small amount and adding to a random position
            for delta_frac in [0.002, 0.005, 0.01, 0.02]:
                delta = delta_frac * best_a[i]
                if delta < 1e-10:
                    continue
                
                # Find a position to add the mass that doesn't hurt hot peaks
                # Try several candidate positions
                for _ in range(10):
                    target = rng.integers(0, n)
                    if target == i:
                        continue
                    
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[target] += delta
                    
                    if a_try[i] < 0 or a_try[target] < 0:
                        continue
                    
                    st = np.sum(a_try)
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        moved = True
                        break
                
                if moved:
                    break
        
        if not moved and iteration > 20:
            # Try broader moves
            for _ in range(50):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.01 * best_a[i]
                a_try = best_a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                st = np.sum(a_try)
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    moved = True
                    break
        
        if not moved:
            # Relax threshold
            peak_threshold = max(0.8, peak_threshold - 0.02)
    
    return best_a


def generate_structured_init(n, strategy='mixed'):
    """Generate diverse structured initializations."""
    if strategy == 'uniform':
        return np.ones(n) / n
    elif strategy == 'triangle':
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)
        return a / np.sum(a)
    elif strategy == 'gaussian':
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
        return a / np.sum(a)
    elif strategy == 'exponential_ramp':
        a = np.geomspace(0.1, 1.0, n)
        return a / np.sum(a)
    elif strategy == 'two_bump':
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
        return a / np.sum(a)
    elif strategy == 'cosine_modulated':
        x = np.arange(n)
        a = np.abs(np.cos(np.pi * x / n)) + 0.3
        return a / np.sum(a)
    elif strategy == 'random_smooth':
        rng = np.random.default_rng(42)
        # Low-frequency random signal
        freqs = rng.uniform(0, 3, n)
        a = np.abs(np.sin(2 * np.pi * freqs * np.arange(n) / n)) + 0.1
        return a / np.sum(a)
    elif strategy == 'spectrally_flat':
        """Try to create a sequence with flat magnitude spectrum."""
        rng = np.random.default_rng(77)
        # Random phase chirp
        x = np.arange(n)
        phase = 2 * np.pi * (x ** 2) / (4 * n) + rng.uniform(0, 2*np.pi, n) * 0.1
        a = np.abs(np.cos(phase)) + 0.5
        return a / np.sum(a)
    elif strategy == 'mixed':
        strategies = ['triangle', 'two_bump', 'cosine_modulated', 'gaussian']
        idx = hash('init') % len(strategies)
        return generate_structured_init(n, strategies[idx])
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            seq = [float(x) for x in parent]
            val = evaluate_sequence(seq)
            if val < np.inf:
                print(f"Parent (n={parent.size}): obj={val:.8f}")
                return parent.copy(), val
    except Exception:
        pass
    return None, np.inf
