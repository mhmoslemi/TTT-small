from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    mb = float(np.max(conv))
    return 2.0 * n * mb / (s * s)

def cross_corr(w, a):
    """
    Compute cross-correlation c[i] = sum_k w[k]*a[k-i] for i=0..len(a)-1.
    w has length 2*len(a)-1, a has length len(a).
    """
    n = len(a)
    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    c = irfft(W * np.conj(A), size)
    return c[:n]

def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def soft_gradient(a, beta):
    """
    Compute the gradient of the soft-max smoothed objective.
    Returns (smoothed_obj, gradient).
    The smoothed function is (1/beta)*log(sum_k exp(beta*(a*a)[k])).
    Its gradient is 2 * cross_corr(softmax_weights, a).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    p = e / np.sum(e)
    smooth_val = cmax + np.log(np.sum(e)) / beta
    grad = 2.0 * cross_corr(p, a)
    return smooth_val, grad, p

                if 0 <= j_src < n:
                    mask[j_src] = False

                # Prefer positions with low values that are far from peak
                cand_positions = np.where(mask)[0]
                if len(cand_positions) == 0:
                    continue

                # Score: prefer low a[i] positions
                scores = best_a[cand_positions]
                # Add small random perturbation for diversity
                scores += rng.uniform(0, 1e-4, len(scores))
                dst = cand_positions[np.argmin(scores)]

                for frac in [0.002, 0.005, 0.01, 0.02]:
                    delta = frac * best_a[i_src]
                    if delta > 0.9 * best_a[i_src]:
                        continue
                    a_try = best_a.copy()
                    a_try[i_src] -= delta
                    a_try[dst] += delta
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break

        if not improved and iteration > 50:
            break

    return best_a
