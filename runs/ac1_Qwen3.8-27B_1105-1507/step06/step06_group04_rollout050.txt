"""
Improved peak-reduction search for step function coefficient sequences.
Uses probabilistic interpretation (minimize peak of self-convolution of a
probability distribution), alternating LP with multiple formulations,
adaptive subgradient with log-sum-exp smoothing, and systematic exploration
of sequence lengths and structures.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize_simplex(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a

def build_toeplitz(a, n_other=None):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    if n_other is None:
        n_other = n
    A = np.zeros((n + n_other - 1, n_other), dtype=np.float64)
    for j in range(n_other):
        lo = j
        hi = min(j + n, n + n_other - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A

def solve_minpeak_lp(fixed_seq, time_limit=20):
    n = len(fixed_seq)
    A = build_toeplitz(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None

def solve_maxsum_lp(fixed_seq, M, time_limit=20):
    n = len(fixed_seq)
    A = build_toeplitz(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:

    except Exception:

def soft_max_subgradient(a, tau, max_iters=2000, time_budget=60.0):
    """
    Subgradient method using log-sum-exp smoothing of the peak.
    Objective: 2*n*soft_max(conv(a,a), tau) / (sum(a))^2
    where soft_max(x, tau) = tau * log(sum(exp(x[k]/tau)))
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    fft_size = 1
    while fft_size < 2 * n -  1:
        fft_size <<= 1
    
    v = np.zeros(n)
    mu = 0.9
    step = 0.05
    tau = 0.01  # smoothing parameter
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Soft max weights
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Gradient of soft_max w.r.t. a_i:
        # soft_max = tau * log(sum_k exp(conv[k]/tau))
        # d(soft_max)/da_i = sum_k weights[k] * d(conv[k])/da_i
        # d(conv[k])/da_i = 2*a_{k-i} (for k != 2i) or a_i (for k = 2i)
        #                    (when 0 <= k-i < n)
        
        # Using FFT: grad[i] = 2 * sum_k weights[k] * a[k-i] (approximately)
        # More precisely: grad[i] = sum_k weights[k] * (d conv[k]/da_i)
        # For non-negative a and the structure of conv:
        # d conv[k]/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} (if k != 2i, 0<=k-i<n)
        #                = a_i (if k = 2i)
        #                = 0 (otherwise)
        
        # Using the correlation trick:
        # sum_k weights[k] * a[k-i] = cross_correlation of weights and a at lag i
        # = ifft(fft(a) * conj(fft(weights_padded)))[i]
        
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        # corr[i] = sum_j a[j] * w_ext[i+j] = sum_k weights[k] * a[k-i] (approximately)
        # Actually: corr[m] = sum_j a[j] * w[m+j]
        # We want sum_k weights[k] * a[k-i] = sum_j a[j] * weights[i+j] (sub k = i+j)
        # = corr[i]
        
        grad = 2.0 * corr[:n]
        
        # The full objective is F = 2*n*soft_max / (sum(a))^2
        # dF/da_i = 2*n * (d(soft_max)/da_i * (sum)^2 - soft_max * 2*sum * 1) / (sum)^4
        #         = 2*n / (sum)^2 * (d(soft_max)/da_i - soft_max / sum)
        # Since sum = 1 (we normalize), this simplifies to:
        #         = 2*n * (grad[i] - soft_max)
        
        soft_max_val = cmax + tau * np.log(w_sum)  # approximation
        # Actually soft_max = tau * log(sum(exp(conv/tau))) = cmax + tau * log(sum(exp((conv-cmax)/tau)))
        soft_max_val = cmax + tau * np.log(np.sum(np.exp(np.clip((conv - cmax) / tau, -500, 500))))
        
        grad_obj = 2.0 * n * (grad - soft_max_val)  # since sum = 1
        grad_centered = grad_obj - np.mean(grad_obj)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break
        
        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive step and temperature
        step *= 0.9995
        tau *= 0.999
        if tau < 0.001:
            tau = 0.001
    
    return best_a, best_obj

def coordinate_descent(a, max_iters=500, time_budget=30.0):
    """
    Exact coordinate descent: for each coordinate i, find the optimal value
    of a_i (with others fixed) that minimizes the objective.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        
        for i in range(n):
            # Optimize a_i with others fixed
            # conv(k) as function of a_i:
            # conv(k) = base[k] + 2*a_{k-i}*a_i (for k where 0<=k-i<n, k!=2i)
            #        = base[k] + a_i^2 (for k=2i)
            # where base[k] = sum_{j != i} a_j * a_{k-j} (adjusting for the i terms)
            
            # Actually, let's compute the contribution of a_i to each conv[k]:
            # conv[k] = sum_j a_j * a_{k-j}
            # The terms involving a_i are: a_i * a_{k-i} (j=i) and a_{k-i} * a_i (k-j=i)
            # If k != 2i: both terms are a_i * a_{k-i}, so total = 2*a_i*a_{k-i}
            # If k = 2i: both terms are a_i*a_i, so total = a_i^2
            
            # Let's compute base (conv without a_i contribution) and the linear/quadratic coefficients
            a_old_i = a[i]
            a[i] = 0
            conv_base = compute_conv(a)  # conv with a_i = 0
            a[i] = a_old_i
            
            # Now conv[k] = conv_base[k] + a_i * (coefficient)
            # For k where 0 <= k-i < n and k != 2i: conv[k] = conv_base[k] + 2*a_i*a_{k-i}
            # For k = 2i: conv[k] = conv_base[k] + a_i^2
            # (Note: conv_base already has a_i=0, so we need to add back the full contribution)
            
            # Actually conv_base was computed with a[i]=0, so:
            # conv[k] = conv_base[k] + 2*a_i*a_{k-i} (for valid k-i, k != 2i)
            #         = conv_base[k] + a_i^2 (for k = 2i)

            # sum = S_old - a_old_i + a_i where S_old = sum(a)
            # Let S = sum of all a except a_i (which is fixed)
            S = np.sum(a) - a_old_i

            # 2*n * max_k (conv_base[k] + c[k]*a_i + q[k]*a_i^2) / (S + a_i)^2

            # at several candidate points.

            # The maximum of these is a piecewise quadratic function.
            # The objective is this max divided by (S+a_i)^2.
            
            # Simple approach: evaluate at a few candidate points
            # Good candidates: a_old_i, 0, and values where the "active set" changes

                         a_old_i * 0.8, a_old_i * 1.2,
                         np.mean(a), np.max(a) * 0.5]
            candidates = [max(0, c) for c in candidates]

            top_k = np.argsort(conv_base)[-3:]
            for k in top_k:
                if 0 <= k - i < n and k != 2 * i:
                    # conv[k] = conv_base[k] + 2*a_i*a_{k-i}
                    # To make conv[k] = (S + a_i)^2 / (2n) * target_val
                    # Skip this for now
                    pass
            
            best_val = a_old_i
            best_f = np.inf
            for av in candidates:
                a[i] = av
                total_sum = S + av
                if total_sum < 0.01:
                    continue
                conv_full = conv_base.copy()
                for k in range(2 * n - 1):
                    ki = k - i
                    if 0 <= ki < n:
                        if k == 2 * i:
                            conv_full[k] += av * av
                        else:
                            conv_full[k] += 2 * av * a[ki]
                max_b = np.max(conv_full)
                f = 2.0 * n * max_b / (total_sum * total_sum)
                if f < best_f:
                    best_f = f
                    best_val = av
            
            if best_val != a_old_i:
                a[i] = best_val
                improved = True
            
            if improved:
                a = normalize_simplex(a)
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_obj

    for _ in range(3):

Improved peak-reduction search for step function coefficient sequences.
Uses alternating LP (min-peak and max-sum formulations) with adaptive blending,
log-sum-exp smoothed subgradient, coordinate-level targeted perturbations guided

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=20):
    """Solve: min t s.t. (fixed_seq*g)[k] <= t, sum(g)=1, g>=0."""
    n = len(fixed_seq)
    A = build_toeplitz(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(fixed_seq, M, time_limit=20):
    """Solve: max sum(g) s.t. (fixed_seq*g)[k] <= M, g >= 0."""
    n = len(fixed_seq)
    A = build_toeplitz(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)

    except Exception:

    stall = 0
    last_t = 0.3
    use_reg = False
    reg_val = 0.1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        remaining = time_budget - (time.monotonic() - start)
        tl = min(20, max(3, remaining * 0.3))

        # Choose LP formulation based on iteration
        if stall > 6 and it % 3 == 2:
            # Try regularized LP to encourage spreading
            cap = 4.0 / n * (1.0 + 0.5 * (stall % 5))
            g, t_opt = solve_minpeak_reg_lp(a, cap, time_limit=tl)
        elif it % 3 == 0:
            g, t_opt = solve_minpeak_lp(a, time_limit=tl)
        else:
            conv = compute_conv(a)
            M = float(np.max(conv))
            g = solve_maxsum_lp(a, M, time_limit=tl)

        if g is None:
            stall += 1
            if stall > 15:
                break
            continue

        # Binary search on blending parameter
        lo, hi = 0.01, 1.0
        best_t = last_t
        best_t_obj = best_obj
        best_a_local = best_a.copy()

        # First evaluate at last_t
        a_try = (1.0 - last_t) * a + last_t * g
        a_try = normalize_simplex(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_t_obj:
            best_t_obj = obj_try
            best_t = last_t
            best_a_local = a_try.copy()

        # Binary search for better t
        for _ in range(5):
            mid1 = lo + (hi - lo) * 0.33
            mid2 = lo + (hi - lo) * 0.67
            a1 = normalize_simplex((1.0 - mid1) * a + mid1 * g)
            a2 = normalize_simplex((1.0 - mid2) * a + mid2 * g)
            o1 = compute_obj(a1)
            o2 = compute_obj(a2)
            if o1 < o2:
                hi = mid2
                if o1 < best_t_obj:
                    best_t_obj = o1
                    best_t = mid1
                    best_a_local = a1.copy()
            else:
                lo = mid1
                if o2 < best_t_obj:
                    best_t_obj = o2
                    best_t = mid2
                    best_a_local = a2.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local
            a = best_a_local
            last_t = best_t
            stall = 0
        else:
            stall += 1
            if stall % 4 == 0:
                # Targeted perturbation: push mass away from peak-contributing coords
                conv = compute_conv(a)
                peak_idx = np.argmax(conv)
                # Perturb coordinates that contribute most to the peak
                rng = np.random.default_rng(it * 104729 + 7)
                noise = rng.normal(0, 0.02, n)
                # Extra push on coordinates near the peak
                for k in range(n):
                    dist = abs(k - peak_idx % n)
                    if dist < n // 4:
                        noise[k] -= 0.01
                a = best_a * np.exp(noise)
                a = normalize_simplex(a)
    
    return best_a, best_obj


def subgradient_smooth(a, max_iters=3000, time_budget=60.0, tau_init=0.01, tau_min=0.001):
    """
    FFT-accelerated subgradient with log-sum-exp smoothing and Nesterov momentum.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    v = np.zeros(n)
    mu = 0.88
    step = 0.06
    tau = tau_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
            w_sum = 1.0
        else:
            weights /= w_sum

        # Gradient via FFT correlation
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]

        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Adaptive step: decrease slowly, increase if stalled
        step *= 0.9997
        tau = max(tau_min, tau * 0.9995)

    return best_a, best_obj


def multi_peak_subgradient(a, num_peaks=5, max_iters=2000, time_budget=40.0):
    """
    Subgradient using only the top-K peaks (sharper gradient signal).
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    v = np.zeros(n)
    mu = 0.85
    step = 0.04

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Use top K peaks with softmax weighting
        top_k = np.argsort(conv)[-num_peaks:]
        top_vals = conv[top_k]
        tau_k = 0.005
        w_top = np.exp((top_vals - top_vals[-1]) / tau_k)
        w_top /= np.sum(w_top)

        weights = np.zeros(len(conv))
        for idx, k in enumerate(top_k):
            weights[k] = w_top[idx]

        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]

        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        v = mu * v - step * grad_centered / grad_norm
        a_new = normalize_simplex(a - v)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        step *= 0.9998

    return best_a, best_obj


def full_polish(a, time_budget=100.0):
    """
    Full polishing pipeline: LP alternation + subgradient + multi-peak.
    """
    n = len(a)
    start = time.monotonic()
    best_a = normalize_simplex(a.copy())
    best_obj = compute_obj(best_a)

    # Cycle through methods
    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        alloc = min(remaining * 0.3, 40.0)
        elapsed_frac = (time.monotonic() - start) / time_budget

        # In early phase, use LP more; later, use subgradient
        if n <= 900 and elapsed_frac < 0.6:
            a_new, obj_new = lp_alternate(best_a, max_iters=30, time_budget=alloc)
        else:
            if remaining > 30:
                a_new, obj_new = subgradient_smooth(best_a, max_iters=int(alloc * 20),
                                                    time_budget=alloc * 0.6,
                                                    tau_init=0.005, tau_min=0.0005)
                a_new2, obj_new2 = multi_peak_subgradient(a_new, num_peaks=5,
                                                          max_iters=int(alloc * 5),
                                                          time_budget=alloc * 0.4)
                if obj_new2 < obj_new:
                    a_new, obj_new = a_new2, obj_new2

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            # Perturb and try again
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            scale = 0.01 + 0.01 * (int(elapsed_frac * 10) % 5)
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = normalize_simplex(a_pert)
            a_try, obj_try = subgradient_smooth(a_pert, max_iters=200,
                                                time_budget=min(10, remaining * 0.2),
                                                tau_init=0.01)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a, best_obj


def generate_starts(n, num_starts=8, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    starts.append(np.ones(n) / n)

    tri = 1.0 - np.abs(i - center) / (center + 1)
    starts.append(tri / np.sum(tri))

    for sigma_frac in [0.1, 0.15, 0.25, 0.4]:
        bell = np.exp(-((i - center) ** 2) / (2 * (n * sigma_frac) ** 2))
        starts.append(bell / np.sum(bell))

    # U-shaped
    u = ((i - center) / center) ** 2 + 0.05
    starts.append(u / np.sum(u))

    # Sinusoidal
    for freq in [2, 4]:
        s = 1.0 + 0.4 * np.cos(2 * np.pi * freq * i / n + rng.random() * 2 * np.pi)
        s = np.maximum(s, 0.05)
        starts.append(s / np.sum(s))

    # Random smooth
    r = np.abs(rng.normal(1, 0.3, n))
    # Smooth it
    kernel = np.ones(5) / 5
    r = np.convolve(r, kernel, mode='same')
    starts.append(r / np.sum(r))

    return starts[:num_starts]


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start exploration, intensive LP+subgradient polishing,
    perturbation-based local escape, and length variation.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: score={best_score:.8f}, n={n_parent}")

    # Phase 1: Length and structure exploration
    print("\n=== Phase 1: Exploration ===")
    n_values = sorted(set([
        n_parent, n_parent - 100, n_parent - 50, n_parent + 50,
        n_parent + 100, n_parent + 200, n_parent - 200,
        max(300, n_parent - 300), min(1500, n_parent + 300)
    ]))
    n_values = [n for n in n_values if 250 <= n <= 1500]

    candidates = []

    for n in n_values:
        if time.monotonic() > deadline - 400:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n == n_parent and best_seq is not None:
            starts = [normalize_simplex(best_seq.copy())]
            starts.extend(generate_starts(n, num_starts=5, seed=seed + n))
        else:
            starts = generate_starts(n, num_starts=5, seed=seed + n * 31)

        alloc_per = min(remaining / len(starts), 25.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 350:
                break
            t_alloc = min(alloc_per, 20.0)
            if t_alloc < 5:
                break

            a_opt, obj_opt = subgradient_smooth(s, max_iters=int(t_alloc * 15),
                                                time_budget=t_alloc * 0.5,
                                                tau_init=0.01)

            if n <= 900 and t_alloc > 10:
                a_lp, obj_lp = lp_alternate(a_opt, max_iters=20, time_budget=t_alloc * 0.45)
                if obj_lp < obj_opt:
                    a_opt, obj_opt = a_lp, obj_lp

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {obj_opt:.8f} ** NEW BEST **")

            candidates.append((obj_opt, a_opt.copy()))

    candidates.sort(key=lambda x: x[0])
    print(f"Top 5: {[f'{c[0]:.6f}' for c in candidates[:5]]}")
    print(f"Best so far: {best_score:.8f}")

    # Phase 2: Intensive optimization of top candidates
    print(f"\n=== Phase 2: Intensive (best: {best_score:.8f}) ===")
    top_k = min(3, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_cand = candidates[ci][1].copy()
        n_cand = len(a_cand)
        alloc = min(remaining * 0.35, 200.0)

        print(f"  Cand {ci} (n={n_cand}, obj={candidates[ci][0]:.8f}): {alloc:.0f}s budget")

        a_pol, obj_pol = full_polish(a_cand, time_budget=alloc)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {best_score:.8f} (n={len(a_pol)}) **")
        else:
            print(f"  No improvement ({obj_pol:.8f})")

    # Phase 3: Targeted perturbations on the best
    print(f"\n=== Phase 3: Perturbations (best: {best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 77777)

        for pidx in range(10):
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # Different perturbation strategies
            if pidx % 3 == 0:
                # Gaussian multiplicative
                scale = 0.005 + 0.005 * (pidx // 3)
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            elif pidx % 3 == 1:
                # Sparse perturbation (change few coordinates)
                a_pert = best_seq.copy()
                num_changes = max(5, n_best // 20)
                idxs = rng.choice(n_best, size=num_changes, replace=False)
                a_pert[idxs] *= (1.0 + rng.normal(0, 0.1, num_changes))
                a_pert = np.maximum(a_pert, 0)
            else:
                # Shift mass from peaks to valleys
                conv = compute_conv(best_seq)
                peak_region = np.where(conv > np.max(conv) * 0.9)[0]
                a_pert = best_seq.copy()
                # Reduce contributions to peak
                for k in peak_region[:10]:
                    j = int(k) % n_best
                    a_pert[j] *= 0.95
                    # Add to a random other coordinate
                    other = rng.integers(0, n_best)
                    a_pert[other] += a_pert[j] * 0.05
                a_pert = np.maximum(a_pert, 0)

            a_pert = normalize_simplex(a_pert)

            t_alloc = min(remaining * 0.12, 30.0)
            if t_alloc < 5:
                break

            a_opt, obj_opt = subgradient_smooth(a_pert, max_iters=int(t_alloc * 12),
                                                time_budget=t_alloc * 0.5,
                                                tau_init=0.005)
            if n_best <= 900:
                a_lp, obj_lp = lp_alternate(a_opt, max_iters=15, time_budget=t_alloc * 0.45)
                if obj_lp < obj_opt:
                    a_opt, obj_opt = a_lp, obj_lp

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Perturb {pidx}: ** {best_score:.8f} **")

    # Phase 4: Resample best to different lengths and re-optimize
    print(f"\n=== Phase 4: Length refinement (best: {best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = sorted(set([n_best - 50, n_best - 30, n_best + 30,
                                   n_best + 50, n_best - 100, n_best + 100]))
        test_lengths = [x for x in test_lengths if 200 <= x <= 1500]

        for n_test in test_lengths:
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_res = normalize_simplex(resample_sequence(best_seq, n_test))
            alloc = min(remaining * 0.2, 30.0)
            a_pol, obj_pol = full_polish(a_res, time_budget=alloc)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** {best_score:.8f} **")

    # Phase 5: Final intensive LP on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        if n_best <= 1000 and remaining > 25:
            print(f"\n=== Phase 5: Final LP (n={n_best}, {remaining:.0f}s) ===")
            a_lp, obj_lp = lp_alternate(best_seq, max_iters=60, time_budget=remaining * 0.7)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
