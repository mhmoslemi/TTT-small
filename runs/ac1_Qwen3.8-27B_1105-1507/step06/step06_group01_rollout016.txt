def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def smooth_max_and_grad_fft(a, tau):
    """Compute smooth max of conv(a,a) and its gradient via FFT. Returns (value, grad)."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    smooth_val = b_max + np.log(exp_s.sum()) / tau
    # grad[i] = 2 * sum_j p[i+j] * a[j] = 2 * cross_correlation(p, a)[i]
    fa = np.fft.fft(a, size)
    fp = np.fft.fft(p, size)
    cross = np.fft.ifft(fp * np.conj(fa)).real[:n]
    grad = 2.0 * cross
    return smooth_val, grad


def compute_M(a):
    """Compute max(conv(a,a)) for normalized (sum=1) non-negative a. Returns 2*n*max."""
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max())


def compute_M_raw(a):
    """Compute max(conv(a,a)) without the 2n factor, for normalized a."""
    b = fast_conv(a)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if cond.any():
        rho = int(np.argmax(~cond))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def run_fista_fft(a_init, tau_start, tau_end, max_iters, deadline, lr=0.3):
    """FISTA with FFT-based gradient on log-sum-exp smooth approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M_raw(a)
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        _, grad_y = smooth_max_and_grad_fft(y, tau)
        grad_tan = grad_y - grad_y.mean()
        step = lr / (1.0 + 0.02 * np.sqrt(it + 1))
        a_new = y - step * grad_tan
        a_new = project_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = project_simplex(y_new)
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        it += 1
        if it % 10 == 0 or it == max_iters - 1:
            M_a = compute_M_raw(a)
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()
            M_y = compute_M_raw(y)
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    M_final = compute_M_raw(a)
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    return best_a


def build_lp_matrix(a, active_indices, n):
    """Build sparse LP constraint matrix for given active convolution indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response_cutting(a, M_target, deadline, max_constraints=500):
    """LP best-response with cutting-plane: iteratively add violated constraints.
    Returns best response g (normalized) or None."""
    n = len(a)
    b = fast_conv(a)
    # Initial active set: top indices
    top_k = 30
    active = np.argsort(b)[::-1][:top_k].tolist()
    g = None
    for round_idx in range(50):
        if time.monotonic() > deadline:
            break
        if len(active) > max_constraints:
            break
        A = build_lp_matrix(a, active, n)
        if A.nnz == 0:
            return None
        c = -np.ones(n)
        b_ub = np.full(len(active), M_target)
        bounds = [(0.0, None)] * n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if not r.success or r.x is None:
                return None
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                return None
            g = g / g_sum
        except Exception:
            return None
        # Check for violated constraints
        bg = fast_conv(a) * 0.0  # placeholder
        # Actually compute conv(a, g)
        size = next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, size)
        fg = np.fft.rfft(g, size)
        ag = np.fft.irfft(fa * fg, size)[:2 * n - 1]
        violated = np.where(ag > M_target * (1 + 1e-8))[0]
        if len(violated) == 0:
            break
        # Add worst violators

            active.extend(violated.tolist())
        else:
            worst = violated[np.argsort(ag[violated])[::-1][:10]]
            active.extend(worst.tolist())
        active = list(set(active))
        active.sort()
    if g is not None:

    return None

def coordinate_descent_1d_fast(a, i, n, b):
    """Optimize a[i] given all others. Uses golden section on the max constraint.
    Returns optimal x >= 0 for a[i]."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    if 2 * n - 1 > (k_hi - k_lo + 1):
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(3.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.1)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with golden section on each coordinate."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:

        a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M_raw(a)
    no_improve = 0
    order = np.arange(n)
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False
        for i in order:
            x_opt = coordinate_descent_1d_fast(a, i, n, b)
            dx = x_opt - a[i]
            if abs(dx) > 1e-12:
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - (x_opt - dx) * (x_opt - dx))
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * dx
                sweep_improved = True
        s = a.sum()
        if s < 1e-10:

        a = a / s
        M = compute_M_raw(a)
        if M < best_M * (1 - 1e-10):

            best_a = a.copy()
            no_improve = 0
        else:

def pairwise_cd(a, k_active, n, b):
    """Pairwise coordinate descent: for active peak k, optimize pairs (i, k-i)."""
    improved = False
    for i in range(n):
        j = k_active - i
        if not (0 <= j < n) or j <= i:
            continue
        if i == j:
            continue
        # Optimize a[i] and a[j] jointly
        # Contribution to b[k_active]: 2*a[i]*a[j]
        # Also affects other bins
        # Simple approach: for each, do 1D opt considering the other is fixed
        x_i = coordinate_descent_1d_fast(a, i, n, b)
        if abs(x_i - a[i]) > 1e-12:
            dx = x_i - a[i]
            a[i] = x_i
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    b[k] += (x_i * x_i - (x_i - dx) * (x_i - dx))
                else:
                    jj = k - i
                    if 0 <= jj < n:
                        b[k] += 2.0 * a[jj] * dx
            improved = True
    return a, b, improved


def simulated_annealing(a_init, deadline, n_perturb=3, T_start=0.001, T_end=0.00001):
    """Simulated annealing: randomly perturb coordinates, accept with Metropolis criterion."""
    rng = np.random.default_rng(int(time.time()) % 100000)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M_raw(a)
    it = 0
    max_iters = 50000
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max_iters)
        T = T_start * (T_end / T_start) ** frac
        a_trial = a.copy()
        # Perturb 1 to n_perturb coordinates
        npert = rng.integers(1, n_perturb + 1)
        idx = rng.integers(0, n, size=npert)
        for idx_i in idx:
            scale = rng.uniform(0.3, 2.0)
            a_trial[idx_i] = a[idx_i] * scale
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-10:
            it += 1
            continue
        a_trial = a_trial / st
        M_trial = compute_M_raw(a_trial)
        delta = M_trial - best_M
        if delta < 0:
            a = a_trial
            if M_trial < best_M:
                best_M = M_trial
                best_a = a_trial.copy()
        else:
            # Accept with probability exp(-delta/T)
            if T > 1e-15 and rng.uniform() < np.exp(-delta / T):
                a = a_trial
        it += 1
    return best_a


def spectral_init(n, max_iters=100, deadline=None):
    """Initialize by iteratively reducing peak of |FFT|^2 (time domain)."""
    a = np.ones(n) / n
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        # Reduce a where it contributes to the peak
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        grad_tan = grad - grad.mean()
        a = a - 0.1 * grad_tan
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    return a


def water_filling_init(n, max_iters=50):
    """Water-filling inspired initialization: equalize autocorrelation values."""
    a = np.ones(n) / n
    for it in range(max_iters):
        b = fast_conv(a)
        b_target = b.mean()
        # Where b is above target, reduce contributing elements
        excess = np.maximum(b - b_target * 1.1, 0)
        # Find which a values contribute most to excess
        for k in np.where(excess > 0)[0][:10]:
            j_lo = max(0, k - (n - 1))
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                a[j] *= 0.995
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None and parent.size > 0:
        p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        p = np.maximum(p, 0.0)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-5)
            inits.append(p_n / p_n.sum())
    # Uniform
    inits.append(np.ones(n) / n)
    # Parabolic
    poly = t * (1 - t) + 0.05
    inits.append(poly / poly.sum())
    # U-shaped
    poly2 = (2 * t - 1) ** 2 + 0.05
    inits.append(poly2 / poly2.sum())
    # Cosine windows
    for freq in [1, 2, 3, 5]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.03
    inits.append(gauss / gauss.sum())
    # Spectral init
    inits.append(spectral_init(n, max_iters=50))
    # Water-filling
    inits.append(water_filling_init(n, max_iters=30))
    # Smooth random
    ksize = max(3, n // 15)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_full_pipeline(a0, n, deadline, seed=0):
    """Run the full optimization pipeline on a single initial sequence."""
    best = a0.copy()
    best_M = compute_M_raw(a0)
    t_start = time.monotonic()
    remaining = deadline - t_start
    if remaining < 5:
        return best
    # Stage 1: FISTA (40% of time)
    t1 = t_start + 0.4 * remaining
    a_f = best
    for tau in [50, 100, 300]:
        if time.monotonic() > t1:
            break
        sub_dl = min(t1, time.monotonic() + max(5, (t1 - time.monotonic()) / 2))
        a_f = run_fista_fft(a_f, tau, tau * 3, 5000, sub_dl, lr=0.3)
    M_f = compute_M_raw(a_f)
    if M_f < best_M:
        best_M = M_f
        best = a_f.copy()
    # Stage 2: LP mirror descent (30% of time)
    t2 = t_start + 0.7 * remaining
    if time.monotonic() < t2:
        a_lp = run_lp_mirror_descent(best, t2, n_active_start=20, max_iters=100)
        M_lp = compute_M_raw(a_lp)
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
    # Stage 3: Coordinate descent (20% of time)
    t3 = t_start + 0.9 * remaining
    if time.monotonic() < t3:
        n_sweeps = max(5, int((t3 - time.monotonic()) / max(0.5, n * 0.01)))
        a_cd = run_coordinate_descent(best, n_sweeps, t3, seed=seed)
        M_cd = compute_M_raw(a_cd)
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
    # Stage 4: Final LP (10% of time)
    if time.monotonic() < deadline:
        a_final = run_lp_mirror_descent(best, deadline, n_active_start=30, max_iters=30)
        M_fin = compute_M_raw(a_final)
        if M_fin < best_M:
            best_M = M_fin
            best = a_final.copy()
    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: FISTA (FFT gradient) -> LP cutting-plane -> simulated annealing
    -> coordinate descent -> final polish. Multi-start with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size
    # Focus on parent length and a few nearby
    lengths = [n_parent]
    for delta in [-64, 64, -128, 128, 256, -256]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl not in lengths:
            lengths.append(nl)
    lengths = lengths[:5]

    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-start FISTA on lengths {lengths}", flush=True)
    print(f"{'='*60}", flush=True)

    time_per_length = (deadline - time.monotonic() - 400) / len(lengths)
    time_per_length = max(time_per_length, 30)

    best_overall = None
    best_M_overall = float('inf')

    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - 400:
            break
        remaining = deadline - 400 - time.monotonic()
        if remaining < 15:
            break
        time_alloc = min(time_per_length, remaining / (len(lengths) - li))
        print(f"\n  n={n}, time={time_alloc:.0f}s", flush=True)

        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_M_n = float('inf')
        time_per_init = time_alloc / max(1, len(inits))

        for ii, a0 in enumerate(inits):
            if time.monotonic() > deadline - 400:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            sub_dl = min(time.monotonic() + time_per_init * 1.5, deadline - 400)
            if time.monotonic() >= sub_dl:
                break
            a_opt = run_full_pipeline(a0, n, sub_dl, seed=seed + li * 100 + ii)
            M_opt = compute_M_raw(a_opt)
            if M_opt < best_M_n:
                best_M_n = M_opt
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_M(best_for_n)
            print(f"  n={n}: best J={J_val:.6f} (elapsed={time.monotonic()-tracker.t_start:.0f}s)", flush=True)
            tracker.update(best_for_n)
            if J_val < best_M_overall:
                best_M_overall = J_val
                best_overall = best_for_n.copy()

    tracker.report("After Phase 1")

    # Phase 2: Simulated annealing on best
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Simulated annealing", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 100:
        t_sa_deadline = min(deadline - 100, time.monotonic() + 200)
        a_sa = simulated_annealing(tracker.best, t_sa_deadline, n_perturb=3,
                                   T_start=0.0005, T_end=0.00001)
        tracker.update(a_sa)
        # Also try with different perturbation sizes
        if time.monotonic() < deadline - 80:
            a_sa2 = simulated_annealing(a_sa, min(deadline - 80, time.monotonic() + 100),
                                        n_perturb=5, T_start=0.0003, T_end=0.000005)
            tracker.update(a_sa2)

    tracker.report("After Phase 2")

    # Phase 3: Intensive LP + coordinate descent
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: Intensive LP + CD polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 50:
        # LP polish
        t_lp_dl = min(deadline - 50, time.monotonic() + 150)
        a_lp = run_lp_mirror_descent(tracker.best, t_lp_dl, n_active_start=30, max_iters=100)
        tracker.update(a_lp)
        # Coordinate descent
        if time.monotonic() < deadline - 30:
            n_sweeps = max(10, int((deadline - 30 - time.monotonic()) / max(1.0, tracker.best.size * 0.015)))
            a_cd = run_coordinate_descent(tracker.best, n_sweeps, deadline - 30, seed=99999)
            tracker.update(a_cd)

    tracker.report("After Phase 3")

    # Phase 4: Final FISTA with high tau
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: High-tau FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 20:
        for tau in [200, 500, 1000]:
            if time.monotonic() > deadline - 15:
                break
            a_f = run_fista_fft(tracker.best, tau, tau * 2, 3000,
                                min(deadline - 15, time.monotonic() + 30), lr=0.15)
            tracker.update(a_f)

    tracker.report("After Phase 4")

    # Phase 5: Try a different length
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        n_cur = tracker.best.size
        for mult in [1.25, 0.8, 1.5]:
            if time.monotonic() > deadline - 8:
                break
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 2048)
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 5, time.monotonic() + 40)
            a_opt = run_full_pipeline(a_ext, new_n, sub_dl, seed=seed + 777)
            tracker.update(a_opt)

    tracker.report("After Phase 5")

    # Ensure valid output
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_M(tracker.best)

    # Final validation
    final_score = float(evaluate_sequence([float(x) for x in tracker.best]))
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
