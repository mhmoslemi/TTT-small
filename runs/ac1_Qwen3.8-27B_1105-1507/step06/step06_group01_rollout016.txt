import numpy as np

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def spectral_shape(a, alpha=0.5, iterations=50):
    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(iterations):
        # Compute power spectrum
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        A = np.fft.rfft(a, size)
        P = np.abs(A)**2
        
        # Flatten: compress dynamic range
        P_flat = np.power(P, alpha)  # alpha < 1 compresses
        
        # Reconstruct with modified magnitude, same phase
        phase = A / (np.abs(A) + 1e-30)
        A_new = phase * np.sqrt(P_flat + 1e-30)
        
        # Inverse transform
        a_new = np.fft.irfft(A_new, size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        else:
            # Try blending
            for t in [0.1, 0.3, 0.5]:
                a_blend = (1-t)*best_a + t*a_new
                a_blend = np.maximum(a_blend, 0)
                a_blend = a_blend / a_blend.sum()
                M_blend = fast_conv(a_blend).max()
                if M_blend < best_M:
                    best_M = M_blend
                    best_a = a_blend.copy()
                    break
            if M_new >= best_M:
                # Reduce aggressiveness
                alpha = alpha * 0.9
                if alpha < 0.8:
                    break
            a = best_a.copy()
    
    return best_a

def lp_best_response(a, M_target, n_active=20, time_limit=5):
    """Solve LP for best response g that keeps conv(a,g) <= M_target."""
    n = len(a)
    b = fast_conv(a)
    
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    active = top_indices.tolist()
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    if A.nnz == 0:
        return None
    
    c = -np.ones(n)
    b_ub = np.full(len(active), M_target)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except:
        pass
    return None

def run_lp_iter(a_init, max_iters, deadline, n_active_start=30):

    a = np.maximum(a_init.astype(np.float64), 0.0)

    best_M = fast_conv(a).max()
    no_improve = 0
    n_active = n_active_start
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = lp_best_response(a, best_M, n_active, time_limit=3)
        if g is None:
            n_active = min(n_active * 2, 2 * n - 1)
            no_improve += 1
            if no_improve > 8:
                break
            continue
        
        improved = False
        for t in [0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-12):
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()

            n_active = max(10, int(n_active * 0.9))

            n_active = min(int(n_active * 1.5), 2 * n - 1)

                break
    
    return best_a, best_M

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def spectral_shape(a, alpha, n_iters=20):

def fourier_mask_perturb(a, rng, n_modes=10, strength=0.3):

def lp_best_response(a, M_target, n_active, time_limit=4):
    """Solve LP for best response g subject to conv(a,g) <= M_target at active indices."""
    n = len(a)
    b = fast_conv(a)
    
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    active = top_indices.tolist()
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M_target)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:

    except Exception:
        pass
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                corr[i] += a_sf[k - i]
        corr /= len(top_k)
        a_sf = a_sf - 0.12 * (corr - corr.mean())
        a_sf = np.maximum(a_sf, 1e-7)
        a_sf = a_sf / a_sf.sum()
    inits.append(a_sf.copy())
    
    # Spectral flattened
    a_fl = np.ones(n) / n
    for alpha in [0.6, 0.4, 0.25]:
        a_fl = spectral_shape(a_fl, alpha, n_iters=8)
    inits.append(a_fl.copy())
    
    # Cosine window variants
    for freq in [1, 2, 3, 4]:
        cw = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t + rng.normal(0, 1))
        cw = np.maximum(cw, 0.01)
        inits.append(cw / cw.sum())
    
    # Gaussian
    gc = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gc / gc.sum())
    
    # U-shaped
    uc = ((2 * t - 1) ** 2) + 0.1
    inits.append(uc / uc.sum())
    
    # Periodic structures
    for period in [2, 3, 5]:
        a_p = np.ones(n)
        for i in range(n):
            if i % period == 0:
                a_p[i] = 2.0
            elif i % period == 1:
                a_p[i] = 0.5
        a_p += rng.normal(0, 0.05, n)
        a_p = np.maximum(a_p, 0.01)
        inits.append(a_p / a_p.sum())
    
    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best_score = score
            self.best = a.copy()
            elapsed = time.monotonic() - self.t_start
            print("  ** NEW BEST: J={:.8f} n={} t={:.0f}s".format(score, a.size, elapsed), flush=True)
    
    def report(self, label):
        if self.best is not None:
            print("  {}: J={:.8f} n={}".format(label, self.best_score, self.best.size), flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: spectral shaping, Fourier-mask exploration,
    LP best-response, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = Tracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print("Parent: n={}, J={:.8f}".format(parent.size, tracker.best_score), flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        tracker.update(parent)
        print("No parent, using uniform n=512, J={:.8f}".format(tracker.best_score), flush=True)
    
    # === PHASE 1: Spectral shaping from parent ===
    print("\n" + "=" * 50, flush=True)
    print("PHASE 1: Spectral shaping", flush=True)
    print("=" * 50, flush=True)
    
    n_base = parent.size
    if n_base < 256 or n_base > 2048:
        n_base = 768
    
    t_p1_end = min(deadline - 500, time.monotonic() + 90)
    
    for alpha in [0.9, 0.8, 0.7, 0.55, 0.4, 0.3]:
        if time.monotonic() > t_p1_end:
            break
        a_sh = spectral_shape(parent, alpha, n_iters=25)
        tracker.update(a_sh)
    
    # Spectral from current best
    if tracker.best is not None and time.monotonic() < t_p1_end:
        a_sh2 = spectral_shape(tracker.best, 0.85, n_iters=30)
        tracker.update(a_sh2)
    
    tracker.report("End Phase 1")
    
    # === PHASE 2: Multi-start LP optimization ===
    print("\n" + "=" * 50, flush=True)
    print("PHASE 2: Multi-start LP optimization", flush=True)
    print("=" * 50, flush=True)
    
    ref = tracker.best if tracker.best is not None else parent
    inits = gen_inits(n_base, rng, ref)
    
    t_p2_end = deadline - 250
    time_per_init = max(8, (t_p2_end - time.monotonic()) / len(inits))
    
    for ii, a0 in enumerate(inits):
        if time.monotonic() > t_p2_end:
            break
        sub_dl = min(t_p2_end, time.monotonic() + time_per_init)
        a0 = np.maximum(a0, 1e-7)
        a0 = a0 / a0.sum()
        a_opt, M_opt = run_lp_iter(a0, max_iters=60, deadline=sub_dl, n_active_start=25)
        tracker.update(a_opt)
    
    tracker.report("End Phase 2")
    
    # === PHASE 3: Intensive LP polish ===
    print("\n" + "=" * 50, flush=True)
    print("PHASE 3: Intensive LP polish", flush=True)
    print("=" * 50, flush=True)
    
    if tracker.best is not None:
        t_p3 = deadline - 120
        a_lp, M_lp = run_lp_iter(tracker.best, max_iters=80, deadline=t_p3, n_active_start=50)
        tracker.update(a_lp)
        
        if time.monotonic() < t_p3:
            a_lp2, _ = run_lp_iter(a_lp, max_iters=40, deadline=min(t_p3, time.monotonic() + 50), n_active_start=12)
            tracker.update(a_lp2)
    
    tracker.report("End Phase 3")
    
    # === PHASE 4: Fourier-mask exploration + LP ===
    print("\n" + "=" * 50, flush=True)
    print("PHASE 4: Fourier-mask exploration", flush=True)
    print("=" * 50, flush=True)
    
    if tracker.best is not None:
        t_p4 = deadline - 80
        cur_best = tracker.best.copy()
        best_M = fast_conv(cur_best).max()
        
        for attempt in range(30):
            if time.monotonic() > t_p4:
                break
            
            # Random Fourier perturbation
            a_pert = fourier_mask_perturb(cur_best, rng, n_modes=rng.integers(3, 15),
                                          strength=0.2 + 0.3 * rng.random())
            if a_pert is None:
                continue
            
            # LP polish the perturbation
            sub_dl = min(t_p4, time.monotonic() + 8)
            a_opt, M_opt = run_lp_iter(a_pert, max_iters=20, deadline=sub_dl, n_active_start=20)
            
            if M_opt < best_M:
                best_M = M_opt
                cur_best = a_opt.copy()
                tracker.update(a_opt)

            if time.monotonic() < t_p4 - 5:

                a_opt2, M_opt2 = run_lp_iter(a_sp, max_iters=10, deadline=min(t_p4, time.monotonic() + 5),

        if time.monotonic() < deadline - 5:
            a_final, _ = run_lp_iter(a_cd, max_iters=25, deadline=deadline - 2, n_active_start=30)

        for delta in [-128, -64, 64, 128, -256, 256]:
            nn = n_cur + delta
            if 256 <= nn <= 2048:
                candidates_n.append(nn)
        
        for nn in candidates_n[:4]:
            if time.monotonic() > deadline - 10:
                break
            a_res = resample_seq(tracker.best, nn)
            a_res = np.maximum(a_res, 1e-6)
            a_res = a_res / a_res.sum()
            sub_dl = min(deadline - 5, time.monotonic() + 12)
            a_opt, _ = run_lp_iter(a_res, max_iters=15, deadline=sub_dl, n_active_start=25)
            tracker.update(a_opt)
    
    tracker.report("End Phase 6")
    
    # === FINAL ===
    print("\n" + "=" * 50, flush=True)
    print("FINAL: J={:.8f} n={}".format(tracker.best_score, tracker.best.size if tracker.best is not None else 0), flush=True)
    print("Total time: {:.0f}s".format(time.monotonic() - tracker.t_start), flush=True)
    print("=" * 50, flush=True)
    
    if tracker.best is None:
        return [1.0 / 512] * 512
    return [float(x) for x in tracker.best]
