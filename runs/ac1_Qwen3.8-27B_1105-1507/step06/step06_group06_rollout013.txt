"""
Optimized low-peak self-convolution sequence design using FFT-accelerated
subgradient methods with adaptive log-sum-exp smoothing, simplex projection,
Nesterov momentum, and LP refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    conv = irfft(fa * fa, N)
    return conv[:2*n-1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al. 2011
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u > (cssv - 1) / rho
    rho_val = np.max(rho[cond]) if np.any(cond) else 1
    theta = (cssv[rho_val-1] - 1) / rho_val
    return np.maximum(v - theta, 0)

def compute_gradient_fft(a, weights, N_fft):
    """
    Compute gradient of sum_k weights[k] * conv(a,a)[k] with respect to a.
    grad[i] = 2 * sum_j weights[i+j] * a[j] for j=0..n-1
    Using FFT: grad = 2 * cross_corr(weights, a)[0:n]
    """
    n = len(a)
    # cross_corr(weights, a)[i] = sum_j weights[i+j] * a[j]
    # = IFFT(FFT(weights) * conj(FFT(a)))
    W = fft(weights, N_fft)
    A = fft(a, N_fft)
    cross = ifft(W * np.conj(A)).real
    grad = 2.0 * cross[:n]
    return grad

def logsumexp_optimize(a, max_iters=5000, time_budget=60.0, t_init=0.1, t_end=0.01, momentum=0.9):
    """
    Optimize using log-sum-exp smoothed max with projected gradient + momentum.
    a is normalized to sum=1.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    N_fft = 1
    while N_fft < 3*n - 1:
        N_fft <<= 1
    
    v = np.zeros(n)  # momentum velocity
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive temperature
        progress = it / max_iters
        t = t_init * (t_end / t_init) ** progress
        
        # Compute convolution
        conv = compute_conv_fft(a)
        cmax = float(np.max(conv))
        
        # Compute softmax weights
        shifted = (conv - cmax) / t
        shifted = np.clip(shifted, -50, 300)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            w = np.zeros(len(conv))
            w[idx] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient via FFT
        grad = compute_gradient_fft(a, w, N_fft)
        
        # Center the gradient (since sum(a)=1 constraint means uniform shifts don't change feasibility)
        # Actually, for the constraint sum(a)=1, the Lagrange multiplier handles this.
        # The projected gradient step already handles it via the projection.
        
        # Momentum update

        a_new = a + v  # Wait, this should be a - step * grad
        
        # Actually, let me use a proper step size
        # The Lipschitz constant of the gradient of the smoothed function is bounded by
        # L = 2 * max |conv values| / t (roughly)
        # Let's use a heuristic step size
        step = t / (2.0 * cmax + t)
        a_new = a - step * grad
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def nesterov_logsumexp(a, max_iters=10000, time_budget=60.0, t_init=0.05, t_end=0.005):
    """
    Nesterov accelerated gradient for log-sum-exp smoothed max convolution.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    y = a.copy()  # Nesterov look-ahead
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    N_fft = 1
    while N_fft < 3*n - 1:
        N_fft <<= 1
    
    start = time.monotonic()
    mu = 1.0  # momentum parameter, updated each iteration
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max_iters
        t = t_init * (t_end / t_init) ** progress
        
        # Compute convolution at y
        conv_y = compute_conv_fft(y)
        cmax = float(np.max(conv_y))
        
        # Softmax weights
        shifted = (conv_y - cmax) / t
        shifted = np.clip(shifted, -50, 300)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv_y))
            w = np.zeros(len(conv_y))
            w[idx] = 1.0
        else:
            w /= w_sum
        
        # Gradient at y
        grad = compute_gradient_fft(y, w, N_fft)
        
        # Step size (Lipschitz constant estimate)
        # The Hessian of the smoothed function is bounded by 4*cmax/t (roughly)
        L = 4.0 * cmax / t
        step = 1.0 / L
        
        # Nesterov update
        mu_new = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        a_new = project_simplex(y - step * grad)
        y = a_new + ((mu - 1.0) / mu_new) * (a_new - a)
        mu = mu_new
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def chase_peak_fftm(a, n_steps=200, time_budget=30.0):
    """
    Chase the peak using FFT-accelerated gradient.
    At each step, find the peak lag and take a gradient step to reduce it.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    N_fft = 1
    while N_fft < 3*n - 1:
        N_fft <<= 1
    
    start = time.monotonic()
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv_fft(a)
        k_star = int(np.argmax(conv))
        
        # Gradient for single peak: grad[i] = 2*a[k_star - i] if 0 <= k_star-i < n
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)
        if j_end > j_start:
            indices = np.arange(j_start, j_end)
            a_indices = k_star - indices
            grad[indices] = 2.0 * a[a_indices]
        
        # Try multiple step sizes
        cmax = conv[k_star]
        best_step = 0
        best_obj_try = best_obj
        
        for step in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_try = a - step * grad / cmax
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_step = step
                best_a = a_try.copy()
                a = a_try.copy()
        
        if best_step == 0:
            break
    
    return best_a, best_obj

def lp_refine(a, time_budget=20.0):
    """LP refinement: minimize peak of conv(a, g) subject to sum(g)=1, g>=0."""
    n = len(a)
    if n > 2000:
        return a, compute_obj(a)
    
    a = project_simplex(a.copy())
    conv = compute_conv_fft(a)
    M = float(np.max(conv))
    
    # Build the constraint matrix: (a * g)[k] <= M for all k
    # (a * g)[k] = sum_j a[j] * g[k-j] for valid j
    # This is a Toeplitz-like matrix
    
    # For efficiency, only include constraints that are "active" (close to M)
    threshold = M * 0.8
    active = np.where(conv >= threshold)[0]
    
    if len(active) == 0:
        return a, compute_obj(a)
    
    # Build constraint matrix for active constraints
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # But g has length n, and conv(a,g) has length 2n-1
    # (a*g)[k] = sum_{j=0}^{n-1} a[j] * g[k-j] for valid k-j
    
    # Actually, let me think about this differently.
    # We want to minimize M s.t. (a*g)[k] <= M for all k, sum(g) = 1, g >= 0
    # Variables: g[0], ..., g[n-1], M
    # Constraints: (a*g)[k] - M <= 0 for k = 0, ..., 2n-2
    #             sum(g) = 1
    #             g[i] >= 0, M >= 0
    
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for j in range(n):
            if 0 <= k-j < n:
                A_ub[k, j] = a[j]
        A_ub[k, n] = -1.0
    
    b_ub = np.zeros(2*n-1)
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize M
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_budget*0.7})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                # Blend with current a
                best_a = a.copy()
                best_obj = compute_obj(a)
                for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
                    a_try = project_simplex((1-t)*a + t*g)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a, best_obj
    except Exception:
        pass
    
    return a, compute_obj(a)

def water_filling_iter(a, max_iters=1000, time_budget=30.0):
    """
    Water-filling fixed point iteration:
    1. Compute conv(a,a) and find the "active set" of peaks
    2. The optimal a should satisfy: for all i with a[i] > 0, grad[i] = constant
    3. Update: a[i] = max(0, lambda - grad[i]) where lambda ensures sum(a)=1
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    N_fft = 1
    while N_fft < 3*n - 1:
        N_fft <<= 1
    
    start = time.monotonic()
    damping = 0.5
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv_fft(a)
        cmax = float(np.max(conv))
        
        # Use a moderate temperature for the weights
        t = cmax * 0.1 + 1e-10
        shifted = (conv - cmax) / t
        shifted = np.clip(shifted, -50, 300)
        w = np.exp(shifted)
        w /= np.sum(w)
        
        # Compute gradient
        grad = compute_gradient_fft(a, w, N_fft)
        
        # Water-filling: a_new[i] = max(0, lambda - grad[i])
        # Find lambda such that sum(a_new) = 1
        # This is equivalent to projecting (lambda - grad) onto the simplex
        # But we need to find the right lambda...
        # Actually, the projection onto {x>=0, sum(x)=1} of any vector v
        # gives us the right form. Let's use a "dual" approach:
        # a_new = project_simplex(-grad + some_shift)
        # But the shift is determined by the constraint.
        # The simplex projection of v gives max(0, v - theta) where theta is chosen
        # so that sum = 1. This is exactly what we want with lambda = shift + theta.
        
        # So: a_new = project_simplex(-grad)
        # But this might give too aggressive a step. Use damping.
        a_wf = project_simplex(-grad)
        a_new = (1 - damping) * a + damping * a_wf
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        # Adaptive damping
        if it % 50 == 0:
            damping = max(0.1, damping * 0.95)
        
        a = a_new
    
    return best_a, best_obj

def multi_temperature_optimize(a, max_iters=5000, time_budget=60.0):
    """
    Multi-temperature log-sum-exp optimization.
    Cycles through different temperatures to explore different regions.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    N_fft = 1
    while N_fft < 3*n - 1:
        N_fft <<= 1
    
    start = time.monotonic()
    temperatures = [0.02, 0.05, 0.1, 0.2, 0.5]
    
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Cycle through temperatures with emphasis on smaller ones as we progress
        progress = it / max_iters
        t_idx = min(int(progress * len(temperatures) * 1.5), len(temperatures) - 1)
        t = temperatures[max(0, t_idx - 1)]  # bias toward smaller t
        
        conv = compute_conv_fft(a)
        cmax = float(np.max(conv))
        
        shifted = (conv - cmax) / t
        shifted = np.clip(shifted, -50, 300)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            w = np.zeros(len(conv))
            w[idx] = 1.0
        else:
            w /= w_sum
        
        grad = compute_gradient_fft(a, w, N_fft)
        
        # Step size based on Lipschitz constant estimate
        L = 4.0 * cmax / t
        step = 1.0 / L
        
        # Momentum
        v = 0.9 * v - step * grad
        a_new = project_simplex(a + v)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a
