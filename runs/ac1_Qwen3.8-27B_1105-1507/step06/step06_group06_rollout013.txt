"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: (1) Correlation-targeting Newton steps that solve the linearized
Toeplitz system to flatten the autocorrelation, (2) refined LP alternating with
adaptive line search, (3) spectral equalization blended with Newton steps,
(4) adaptive perturbation schedule for escaping local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix T where (T@g)[k] = conv(a, g)[k]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        T[k_start:k_end, j] = a[:k_end - k_start]
    return T


def newton_targeting_step(a, max_iters=50, time_budget=20.0, reg_init=0.01):
    """
    Correlation-targeting Newton step: solve the linearized system
    2 * T(a) @ delta = c_target - c(a) to flatten the autocorrelation.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    reg = reg_init

    T = build_toeplitz(a)
    TT = T.T @ T  # n x n Gram matrix

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        c_target = np.full(2 * n - 1, 1.0 / (2 * n - 1))

        # Focus on peaks: only correct lags above target
        residual = c_target - conv
        # Weight the residual: prioritize large positive peaks
        weights = np.where(conv > c_max * 0.7, 1.0, 0.3)

        # Solve regularized least squares: (4*TT + reg*I) delta = 2*T^T * (weights*residual)
        rhs = 2.0 * T.T @ (weights * residual)
        A = 4.0 * TT + reg * np.eye(n)

        try:
            delta = np.linalg.solve(A, rhs)
        except np.linalg.LinAlgError:
            delta = np.linalg.lstsq(A, rhs, rcond=None)[0]

        # Adaptive step: try multiple step sizes
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_try = a + alpha * delta
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-10:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                # Update T and TT for next iteration
                T = build_toeplitz(a)
                TT = T.T @ T
                break

        if not improved:
            reg *= 2.0
            if reg > 100.0:
                reg = reg_init
                # Try a random perturbation

                a = a_pert

                TT = T.T @ T
            if it > 30 and not improved:

        # Gradually reduce regularization (take larger steps as we converge)
        reg = max(reg_init * 0.5, reg * 0.95)

    return best_a, best_obj


def lp_step(a, M=None, time_limit=15):
    """
    Solve LP: max sum(g) s.t. conv(a, g) <= M, g >= 0.
    Returns normalized g or None if infeasible.
    """
    n = len(a)
    if M is None:
        conv = compute_conv(a)
        M = float(np.max(conv))

    if n > 2000:
        return None

    T = build_toeplitz(a)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=T, b_ub=np.full(2 * n - 1, M),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def spectral_equalize(a, max_iters=100, time_budget=20.0, alpha_init=0.3):
    """
    Spectral equalization: flatten |FFT(a)|^2 to reduce autocorrelation peaks.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        F = fft(a, N)
        power = np.abs(F) ** 2
        power_max = np.max(power)
        if power_max < 1e-15:
            break

        alpha = alpha_init * (1.0 - 0.5 * it / max_iters)

        # Blend toward flat spectrum (but keep DC fixed)
        power_flat = np.full(N, power_max / N)
        power_new = (1.0 - alpha) * power + alpha * power_flat
        power_new[0] = power[0]  # Keep DC

        # Reconstruct with same phase
        phase = np.angle(F)
        mag_new = np.sqrt(power_new)
        F_new = mag_new * np.exp(1j * phase)

        a_new = ifft(F_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def peak_chase_fast(a, n_steps=30, step_size=0.05):
    """Fast peak-chasing: iteratively reduce the max convolution peak."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient for the peak lag
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += a[j]
            j2 = i - (k_star - (n - 1))
            if 0 <= j2 < n and j2 != j:
                grad[i] += a[j2]

        # Avoid constant gradient
        grad -= np.mean(grad)

        # Try a few step sizes
        for ss in [step_size, step_size * 0.5, step_size * 2.0]:
            a_try = a - ss * grad
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-10:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a, best_obj


def diverse_init(n, rng, parent=None):

    inits.append(exp_decay / exp_decay.sum())

    # Reverse exponential
    exp_decay_r = np.exp(-np.linspace(4, 0, n))
    inits.append(exp_decay_r / exp_decay_r.sum())

    # Sinusoidal (one full period)

    inits.append(sine / sine.sum())

    u_shape = np.cos(np.pi * (x - 0.5)) ** 2 + 0.05
    inits.append(u_shape / u_shape.sum())

    # Two asymmetric peaks
    x2 = np.linspace(0, 1, n)
    two_pk = np.exp(-((x2 - 0.25) ** 2) / 0.02) + 0.7 * np.exp(-((x2 - 0.7) ** 2) / 0.015)
    inits.append(two_pk / two_pk.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=float)
    inits.append(ramp / ramp.sum())

    # Anti-symmetric-like (high left, low right)
    anti = np.linspace(n, 1, n) + 1.0
    inits.append(anti / anti.sum())

    # Bathtub (V-shape)
    vat = np.abs(np.linspace(-1, 1, n)) + 0.1
    inits.append(vat / vat.sum())

    if parent is not None:
        par = resample_sequence(parent, n)
        inits.append(par / par.sum())
        pert = par * np.exp(rng.normal(0, 0.1, n))
        inits.append(project_nonneg_sum1(pert))

    return inits

            if time.monotonic() - p1_start > phase1_budget:
                break
            a_quick, obj_quick = peak_chase_fast(init, n_steps=15, step_size=0.04)
            if obj_quick < np.inf:
                candidates.append((obj_quick, a_quick))
                if obj_quick < best_score:
                    best_score = obj_quick
                    best_seq = a_quick.copy()

    candidates.sort(key=lambda x: x[0])
    print(f"  Candidates: {len(candidates)}, best={candidates[0][0] if candidates else 'N/A'}")
    if candidates and candidates[0][0] < best_score:
        best_score = candidates[0][0]
        best_seq = candidates[0][1].copy()

    # === Phase 2: Main Optimization (65% of time) ===
    phase2_budget = min(950.0, (deadline - time.monotonic()) * 0.70)
    print(f"\n=== Phase 2: Main Optimization (budget={phase2_budget:.0f}s) ===")
    p2_start = time.monotonic()

    a_main = project_nonneg_sum1(best_seq.copy())
    n_main = len(a_main)
    local_best = a_main.copy()
    local_obj = best_score

    # Main loop: cycle through strategies with adaptive allocation
    strategy_cycle = 0
    no_improve_count = 0

    while time.monotonic() < p2_start + phase2_budget:
        remaining = p2_start + phase2_budget - time.monotonic()
        if remaining < 10:
            break

        # Choose strategy based on cycle and remaining time
        strat = strategy_cycle % 4
        strategy_cycle += 1

        if strat == 0:
            # Newton targeting (primary method)
            a_new, obj_new = newton_targeting_step(
                a_main, max_iters=30,
                time_budget=min(remaining * 0.3, 30.0), reg_init=0.005)
        elif strat == 1:
            # LP alternating
            if n_main <= 1500:
                a_new, obj_new = lp_alternate_refined(
                    a_main, max_iters=15,
                    time_budget=min(remaining * 0.3, 35.0))
            else:
                a_new, obj_new = newton_targeting_step(
                    a_main, max_iters=20,
                    time_budget=min(remaining * 0.3, 25.0), reg_init=0.01)
        elif strat == 2:
            # Spectral equalization
            a_new, obj_new = spectral_equalize(
                a_main, max_iters=50,
                time_budget=min(remaining * 0.2, 20.0),
                alpha_init=0.2 + 0.1 * (strategy_cycle % 3))
        else:
            # Hybrid Newton + LP
            a_new, obj_new = hybrid_newton_lp(
                a_main, max_iters=20,
                time_budget=min(remaining * 0.3, 40.0))

        if obj_new < local_obj - 1e-9:
            local_obj = obj_new
            local_best = a_new.copy()
            a_main = a_new.copy()
            no_improve_count = 0
            if local_obj < best_score:
                best_score = local_obj
                best_seq = local_best.copy()
                print(f"  ** UPDATE strat={strat} cycle={strategy_cycle}: obj={best_score:.8f} **")
        else:
            no_improve_count += 1
            # Perturb when stalled
            if no_improve_count >= 4:
                pert_strength = 0.02 + 0.01 * no_improve_count
                a_pert = a_main * (1.0 + pert_strength * rng.standard_normal(n_main))
                a_pert = project_nonneg_sum1(a_pert)
                a_pert, obj_pert = newton_targeting_step(
                    a_pert, max_iters=15, time_budget=15.0, reg_init=0.01)
                if obj_pert < local_obj:
                    local_obj = obj_pert
                    local_best = a_pert.copy()
                    a_main = a_pert.copy()
                    if local_obj < best_score:
                        best_score = local_obj
                        best_seq = local_best.copy()
                        print(f"  ** PERTURB UPDATE: obj={best_score:.8f} **")
                no_improve_count = 0

    print(f"  Phase 2 done: local_obj={local_obj:.8f}")

    # === Phase 3: Multi-length exploration (15% of time) ===
    remaining3 = deadline - time.monotonic()
    if remaining3 > 30:
        phase3_budget = min(250.0, remaining3 * 0.35)
        print(f"\n=== Phase 3: Multi-length (budget={phase3_budget:.0f}s) ===")
        p3_start = time.monotonic()

        n_current = len(best_seq)
        n_tries = []
        for offset in [-100, -50, -25, 25, 50, 100, 150, 200]:
            n_try = n_current + offset
            if 100 <= n_try <= 2500:
                n_tries.append(n_try)
        n_tries = list(dict.fromkeys(n_tries))

        per_n_budget = max(8.0, phase3_budget / max(1, len(n_tries)))

        for n_try in n_tries:
            if time.monotonic() - p3_start > phase3_budget:
                break

            a_res = resample_sequence(best_seq, n_try)
            a_res = project_nonneg_sum1(a_res)

            # Optimize at this length
            a_opt, obj_opt = newton_targeting_step(
                a_res, max_iters=20, time_budget=min(per_n_budget * 0.4, 20.0))
            if n_try <= 1200:
                a_opt2, obj_opt2 = lp_alternate_refined(
                    a_opt, max_iters=8, time_budget=min(per_n_budget * 0.3, 15.0))
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2
            a_opt3, obj_opt3 = peak_chase_fast(a_opt, n_steps=20, step_size=0.02)
            if obj_opt3 < obj_opt:
                a_opt, obj_opt = a_opt3, obj_opt3

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 4: Final Polish (remaining time) ===
    remaining4 = deadline - time.monotonic()
    if remaining4 > 15 and best_seq is not None:
        print(f"\n=== Phase 4: Final Polish (budget={remaining4:.0f}s) ===")
        a_final = project_nonneg_sum1(best_seq.copy())
        n_final = len(a_final)

        rounds = min(8, int(remaining4 / 12))
        for pr in range(rounds):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            # Newton polish with very small regularization
            a_p, obj_p = newton_targeting_step(
                a_final, max_iters=20, time_budget=min(rem * 0.3, 15.0), reg_init=0.001)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  Polish N r{pr}: {best_score:.8f}")

            remain = deadline - time.monotonic()
            if remain > 5 and len(a_final) <= 1200:
                a_p2, obj_p2 = lp_alternate_refined(
                    a_final, max_iters=8, time_budget=min(remain * 0.3, 12.0))
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_final = a_p2.copy()
                    print(f"  Polish LP r{pr}: {best_score:.8f}")

            # Tiny perturbation + re-optimize
            remain = deadline - time.monotonic()
            if remain > 8:
                a_pt = a_final * (1.0 + 0.005 * rng.standard_normal(n_final))
                a_pt = project_nonneg_sum1(a_pt)
                a_pt, obj_pt = newton_targeting_step(
                    a_pt, max_iters=15, time_budget=min(remain * 0.3, 10.0), reg_init=0.002)
                if obj_pt < best_score:
                    best_score = obj_pt
                    best_seq = a_pt.copy()
                    a_final = a_pt.copy()
                    print(f"  Polish PT r{pr}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
