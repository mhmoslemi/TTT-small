"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: (1) LP-dual-weighted subgradient method, (2) adaptive LP alternation
with warm-starting and full-replacement mode, (3) spectral domain flattening with
phase preservation, (4) multi-resolution length exploration, (5) active-set
coordinate descent exploiting the clean gradient structure d(a*a)[k]/da[j]=2*a[k-j].
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, x0=None, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g_normalized, dual_vars) or (None, None)."""
    n = len(a)
    if n > 5000:
        return None, None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g_norm = g / s
                # Get dual variables
                dual = res.ineqlin.marginals if hasattr(res, 'ineqlin') and res.ineqlin is not None else None
                return g_norm, dual
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 5000:
        return None, None
    A = build_toeplitz(a)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def lp_alternate(a, max_iters=200, time_budget=100.0, full_replace=False,
                 t_init=0.5, t_decay=0.98):
    """LP alternation with adaptive blending and optional full replacement."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    t = t_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, dual = solve_lp_max_sum(a, M, time_limit=min(5.0, time_budget * 0.05))
        if g is None:
            break

        # Try blending
        if full_replace:
            candidates = [g]
        else:
            candidates = [(1.0 - t) * a + t * g,
                          (1.0 - t * 0.5) * a + t * 0.5 * g,
                          (1.0 - min(1.0, t * 1.5)) * a + min(1.0, t * 1.5) * g]

        best_local_obj = np.inf
        best_local_a = best_a

        for cand in candidates:
            cand = project_nonneg_sum1(cand)
            obj = compute_obj(cand)
            if obj < best_local_obj:
                best_local_obj = obj
                best_local_a = cand.copy()

        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_local_a
            stall = 0
            t = max(0.1, t * 1.02)  # increase step slowly
        else:
            stall += 1
            t = max(0.05, t * t_decay)
            if stall > 15:
                break

    return best_a, best_obj


def dual_weighted_subgradient(a, max_iters=200, time_budget=60.0):
    """
    Novel method: Use LP dual variables as weights for subgradient descent.
    The LP dual gives information about which convolution values are most
    "binding", providing a much better descent direction than just the argmax.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    v = np.zeros(n)
    momentum = 0.8

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Solve LP to get dual variables
        g, dual_vars = solve_lp_max_sum(a, M, time_limit=min(3.0, time_budget * 0.03))

        if dual_vars is not None and np.max(dual_vars) > 0:
            # Use dual variables as weights (normalized)
            w = np.maximum(dual_vars, 0)
            w_sum = np.sum(w)
            if w_sum > 1e-15:
                w = w / w_sum
            else:
                w = np.zeros(len(conv))
                w[np.argmax(conv)] = 1.0
        else:
            # Fallback: use softmax weights on top peaks
            top_frac = 0.05
            n_top = max(1, int(len(conv) * top_frac))
            top_idx = np.argsort(conv)[-n_top:]
            w = np.zeros(len(conv))
            w[top_idx] = 1.0 / n_top

        # Compute weighted gradient: grad[j] = sum_k w[k] * 2 * a[k-j] (for valid k-j)
        grad = np.zeros(n)
        for j in range(n):
            # grad[j] = 2 * sum_k w[k] * a[k-j] where 0 <= k-j < n
            # = 2 * (w convolved with a)[n-1-j] ... let me compute via correlation
            # Actually: sum_k w[k] * a[k-j] for 0 <= k-j < n
            # Let m = k-j, so k = m+j, m in [0, n-1]
            # grad[j] = 2 * sum_{m=0}^{n-1} w[m+j] * a[m]
            # This is the cross-correlation of w and a at lag j
            if j < len(w):
                overlap = min(n, len(w) - j)
                if overlap > 0:
                    grad[j] = 2.0 * np.dot(w[j:j + overlap], a[:overlap])

        # Center gradient
        grad_centered = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - step * grad_centered
        a_new = a - v
        a_new = project_nonneg_sum1(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

        step *= 0.995
        momentum = min(0.95, momentum + 0.001)

    return best_a, best_obj


def spectral_flatten(a, max_iters=100, time_budget=30.0):
    """
    Frequency domain approach: iteratively flatten the power spectrum.
    Use proper phase handling to maintain real-valued output.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = fft(a, N)
        power = np.abs(fa) ** 2
        p_max = np.max(power)
        if p_max < 1e-15:
            break

        # Apply compression: reduce dynamic range
        # Use a soft power law: p_new = p^alpha where alpha < 1 compresses
        progress = it / max_iters
        alpha = 0.6 + 0.3 * progress  # start aggressive, ease off
        p_norm = power / p_max
        p_compressed = np.maximum(p_norm, 1e-12) ** alpha
        p_compressed = p_compressed * p_max

        # Reconstruct with original phase
        phase = np.angle(fa)
        mag = np.sqrt(p_compressed)
        fa_new = mag * np.exp(1j * phase)

        a_new = ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def coordinate_descent_peaks(a, max_iters=300, time_budget=60.0):
    """
    Coordinate descent: at each step, find the peak k*, then for each j,
    compute the optimal delta that reduces (a*a)[k*] most while respecting
    non-negativity and sum constraint.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient of (a*a)[k_star] wrt a[j] = 2*a[k_star-j] for valid k_star-j
        grad = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                grad[j] = 2.0 * a[kj]

        # Try to reduce: decrease a[j] where grad[j] is large, increase where small
        # But must maintain sum=1 and a>=0
        # Simple approach: move in -grad direction, project
        for step_frac in [0.01, 0.02, 0.05, 0.1]:
            a_try = a - step_frac * (grad - np.mean(grad))
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break
        else:
            # If no improvement, try a random coordinate perturbation
            j1 = rng.integers(0, n)
            j2 = rng.integers(0, n)
            if j1 != j2:
                delta = 0.01 * a.mean() * rng.random()
                a_try = a.copy()
                a_try[j1] -= delta
                a_try[j2] += delta
                a_try = project_nonneg_sum1(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()

    return best_a, best_obj


def multi_peak_subgradient(a, max_iters=200, time_budget=50.0):
    """Subgradient using multiple top peaks with adaptive weights."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    momentum = 0.85
    step = 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Use top peaks with exponential weighting
        tau = 0.01 * cmax + 0.001
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -200, 0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum

        # Compute weighted gradient
        grad = np.zeros(n)
        for j in range(n):
            # grad[j] = 2 * sum_k weights[k] * a[k-j] (for valid k-j)
            if j < len(weights):
                overlap = min(n, len(weights) - j)
                if overlap > 0:
                    grad[j] = 2.0 * np.dot(weights[j:j + overlap], a[:overlap])

        grad_centered = grad - np.mean(grad)

        v = momentum * v - step * grad_centered
        a_new = a - v
        a_new = project_nonneg_sum1(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

        step *= 0.997

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular
    x = np.linspace(0, 1, n)
    tri = 1 - np.abs(2 * x - 1)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    # Gaussian center
    x = np.linspace(-3, 3, n)
    gauss = np.exp(-x ** 2 / 3)
    inits.append(gauss / gauss.sum())

    # Cosine window
    x = np.linspace(0, 1, n)
    cos_win = 0.54 + 0.46 * np.cos(np.pi * (2 * x - 1))
    cos_win = np.maximum(cos_win, 0.01)
    inits.append(cos_win / cos_win.sum())

    # Sinusoidal
    x = np.linspace(0, 2 * np.pi, n)
    sine = (1 + np.sin(x)) / 2
    sine = np.maximum(sine, 0.01)
    inits.append(sine / sine.sum())

    # Bimodal
    x = np.linspace(0, 1, n)
    bi = np.exp(-((x - 0.3) ** 2) / 0.01) + np.exp(-((x - 0.7) ** 2) / 0.01)
    inits.append(bi / bi.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Power law (heavy at one end)
    x = np.linspace(0.01, 1, n)
    pl = x ** 2
    inits.append(pl / pl.sum())

    # Inverse power law
    pl2 = (1 / x) ** 0.5
    inits.append(pl2 / pl2.sum())

    if parent is not None:
        inits.append(resample_sequence(parent, n))
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_nonneg_sum1(pert))

    return inits

    mode_idx = 0
    lap_count = 0

    while time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        if remaining < 5:
            break

        mode = modes[mode_idx % len(modes)]
        mode_idx += 1

        if mode == 'blend':
            a_new, obj_new = lp_alternate(a_work, max_iters=50, time_budget=min(remaining * 0.3, 60.0),
                                          full_replace=False, t_init=0.5)
        elif mode == 'full':
            a_new, obj_new = lp_alternate(a_work, max_iters=30, time_budget=min(remaining * 0.3, 40.0),
                                          full_replace=True)
        elif mode == 'dual_sg':
            a_new, obj_new = dual_weighted_subgradient(a_work, max_iters=50, time_budget=min(remaining * 0.3, 40.0))
        elif mode == 'multi_sg':
            a_new, obj_new = multi_peak_subgradient(a_work, max_iters=50, time_budget=min(remaining * 0.3, 40.0))
        else:
            a_new, obj_new = lp_alternate(a_work, max_iters=50, time_budget=min(remaining * 0.3, 60.0),
                                          full_replace=False, t_init=0.3)

        if obj_new < local_obj:
            improvement = local_obj - obj_new
            local_obj = obj_new
            local_best = a_new.copy()
            a_work = a_new.copy()
            if local_obj < best_score:
                best_score = local_obj
                best_seq = local_best.copy()
                print(f"  L{lap_count}: NEW BEST obj={best_score:.8f} (improvement={improvement:.2e})")
            lap_count += 1
        else:
            # Try a small perturbation to escape local minimum
            a_pert = a_work * np.exp(rng.normal(0, 0.01, len(a_work)))
            a_pert = project_nonneg_sum1(a_pert)
            a_work = a_pert

        if lap_count % 10 == 0 and lap_count > 0:
            print(f"  Lap {lap_count}: obj={local_obj:.8f}, time_used={time.monotonic()-start_time:.0f}s")

    print(f"  Phase 2 done: obj={local_obj:.8f}")

    # === Phase 3: Try different lengths (20% of budget) ===
    print(f"\n=== Phase 3: Multi-Length Exploration ===")
    print(f"Best so far: {best_score:.8f} (n={len(best_seq)})")
    phase3_budget = min(300.0, (deadline - time.monotonic()) * 0.20)
    phase3_end = time.monotonic() + phase3_budget

    n_current = len(best_seq)
    n_tries = [n_current - 100, n_current + 100, n_current + 200, n_current + 300,
               max(200, n_current - 200)]
    n_tries = list(dict.fromkeys([n for n in n_tries if 200 <= n <= 3000]))

    for n_try in n_tries:
        if time.monotonic() > phase3_end:
            break
        a_res = resample_sequence(best_seq, n_try)
        a_res = project_nonneg_sum1(a_res)

        # Optimize at this length
        a_opt, obj_opt = lp_alternate(a_res, max_iters=30,
                                      time_budget=min(remaining if (remaining := phase3_end - time.monotonic()) > 0 else 10, 30.0),
                                      full_replace=False)
        if obj_opt > local_obj:
            # Try different mode
            a_opt2, obj_opt2 = multi_peak_subgradient(a_res, max_iters=30,
                                                      time_budget=min(phase3_end - time.monotonic(), 20.0))
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f} (no improvement)")

    # === Phase 4: Spectral + coordinate descent (15% of budget) ===
    print(f"\n=== Phase 4: Spectral + Coordinate Descent ===")
    print(f"Best so far: {best_score:.8f}")
    phase4_budget = min(200.0, (deadline - time.monotonic()) * 0.15)
    phase4_end = time.monotonic() + phase4_budget

    a_work2 = project_nonneg_sum1(best_seq.copy())

    # Spectral flattening
    a_sp, obj_sp = spectral_flatten(a_work2, max_iters=80, time_budget=min(phase4_budget * 0.4, 40.0))
    if obj_sp < best_score:
        best_score = obj_sp
        best_seq = a_sp.copy()
        print(f"  Spectral: NEW BEST {best_score:.8f}")
    a_work2 = project_nonneg_sum1(best_seq.copy())

    # Coordinate descent
    a_cd, obj_cd = coordinate_descent_peaks(a_work2, max_iters=200, time_budget=min(phase4_budget * 0.3, 30.0))
    if obj_cd < best_score:
        best_score = obj_cd
        best_seq = a_cd.copy()
        print(f"  CoordDesc: NEW BEST {best_score:.8f}")
    a_work2 = project_nonneg_sum1(best_seq.copy())

    # Dual-weighted subgradient
    a_dw, obj_dw = dual_weighted_subgradient(a_work2, max_iters=100, time_budget=min(phase4_budget * 0.3, 40.0))
    if obj_dw < best_score:
        best_score = obj_dw
        best_seq = a_dw.copy()
        print(f"  DualSG: NEW BEST {best_score:.8f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()

    if remaining5 > 10 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())

        # Cycle through all methods with decreasing time
        polish_cycle = 0
        while time.monotonic() < deadline - 5:
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            polish_cycle += 1
            strat = (polish_cycle) % 5

            if strat == 0:
                # LP blend
                a_p, obj_p = lp_alternate(a_final, max_iters=30, time_budget=min(rem * 0.4, 40.0),
                                          full_replace=False, t_init=0.4)
            elif strat == 1:
                # LP full replace
                a_p, obj_p = lp_alternate(a_final, max_iters=20, time_budget=min(rem * 0.3, 30.0),
                                          full_replace=True)
            elif strat == 2:
                # Multi-peak subgradient
                a_p, obj_p = multi_peak_subgradient(a_final, max_iters=50, time_budget=min(rem * 0.3, 30.0))
            elif strat == 3:
                # Spectral
                a_p, obj_p = spectral_flatten(a_final, max_iters=50, time_budget=min(rem * 0.3, 25.0))
            else:
                # Coordinate descent
                a_p, obj_p = coordinate_descent_peaks(a_final, max_iters=100, time_budget=min(rem * 0.3, 30.0))

            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  Polish {polish_cycle} (strat {strat}): NEW BEST {best_score:.8f}")
            else:
                # Small perturbation for diversity
                a_final = a_final * np.exp(rng.normal(0, 0.005, len(a_final)))
                a_final = project_nonneg_sum1(a_final)

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
