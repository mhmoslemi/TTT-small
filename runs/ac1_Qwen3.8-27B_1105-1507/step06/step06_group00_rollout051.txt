Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of: (1) adaptive subgradient "chase-the-peak" with momentum,
(2) pair-splitting heuristic for peak reduction, (3) LP with constraint generation,
(4) multi-start with diverse initializations, (5) frequency-domain projection.

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def build_conv_matrix(a, k_indices=None):
    """Build matrix A (len(k_indices) x n) such that A @ g = conv(a, g)[k_indices]."""
    n = len(a)
    if k_indices is None:
        k_indices = range(2 * n - 1)
    k_indices = list(k_indices)
    m = len(k_indices)
    A = np.zeros((m, n))
    for idx, k in enumerate(k_indices):
        for j in range(n):
            l = k - j
            if 0 <= l < n:
                A[idx, j] = a[l]
    return A

def solve_lp_cvg(a, active_set, M_target=None):
    """
    Solve LP with constraint generation.
    min M s.t. conv(a,g)[k] <= M for k in active_set, sum(g)=1, g>=0.
    If M_target given, solve max sum(g) s.t. conv(a,g)[k] <= M_target.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    
    g = np.ones(n) / n  # initial guess for warm start wouldn't work with linprog
    
    A = build_conv_matrix(a, active_set)
    m = len(active_set)
    
    # Variables: g[0..n-1], M
    # Minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: A @ g - M <= 0  =>  A @ g + (-1)*M <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    
    return None, None

def pair_split(a, n_iters=100, time_budget=20.0, verbose=False):
    """
    Pair-splitting heuristic: for the peak bin, make contributing pairs more unequal.
    """

                # Move mass to a position that doesn't contribute to this bin.
                # Find a position p that doesn't contribute to bin k_star.
                # p contributes if k_star - p is in [0, n-1].
                # So p doesn't contribute if k_star - p < 0 or k_star - p >= n.
                # i.e., p > k_star or p < k_star - (n-1).

                        candidates.append(p)

                    a[p] += delta

                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[p] -= delta
                continue

            # To minimize product given fixed sum s = a[i]+a[j], make them unequal.
            # Current product: a[i]*a[j]
            # If we move delta from larger to smaller: 
            #   new product = (a[large]-d)*(a[small]+d)

            #   This increases if a[large] > a[small]!
            # Wait, that's the opposite of what we want!
            # 
            # Actually: to DECREASE the product a[i]*a[j] with fixed sum:
            # Make them MORE unequal. So move mass FROM the smaller TO the larger.
            # (a[small]-d)*(a[large]+d) = a[small]*a[large] + d*(a[small]-a[large]) - d^2
            # = a[small]*a[large] - d*(a[large]-a[small]) - d^2

            if a[i] > a[j]:
                large, small = i, j
            else:
                large, small = j, i
            
            # Move from small to large (making them more unequal)
            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = frac * a[small]
                if delta >= a[small]:
                    delta = a[small] * 0.9
                a[small] -= delta
                a[large] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[small] += delta
                    a[large] -= delta

def freq_projection(a, n_iters=500, time_budget=30.0, verbose=False):
    """
    Frequency-domain projection method (inspired by error reduction in crystallography).
    Works with circular convolution approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    # Pad to power of 2
    N = 1
    while N < 2 * n:
        N *= 2
    
    a_padded = np.zeros(N)
    a_padded[:n] = a
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # FFT
        A = fft(a_padded)
        
        # Power spectrum
        P = np.abs(A) ** 2
        
        # Autocorrelation (circular)
        r = np.real(ifft(P))
        
        # Find peak and reduce power at contributing frequencies
        k_star = int(np.argmax(r))
        
        # The peak r[k_star] = (1/N) sum_k P[k] * exp(2*pi*i*k_star*k/N)
        # To reduce r[k_star], reduce P[k] where cos(2*pi*k_star*k/N) is large positive
        
        # Reduce power at the peak-contributing frequencies
        phase = np.cos(2 * np.pi * k_star * np.arange(N) / N)
        # Reduce P where phase is positive (contributing to peak)
        alpha = 0.01
        P_new = P * (1 - alpha * np.maximum(phase, 0))
        P_new[0] = P[0]  # Keep DC (sum constraint)
        P_new = np.maximum(P_new, 0)
        
        # Recovery step: project onto non-negativity
        A_new_mag = np.sqrt(P_new)
        A_new_phase = np.angle(A)
        A_new = A_new_mag * np.exp(1j * A_new_phase)
        a_new = np.real(ifft(A_new))
        a_new = np.maximum(a_new[:n], 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a_padded = np.zeros(N)
                a_padded[:n] = a_new
            else:
                # Decrease alpha
                alpha *= 0.8
                # Reset
                a_padded = np.zeros(N)
                a_padded[:n] = best_a
        
        # Slowly increase alpha
        if it % 100 == 0 and alpha < 0.05:
            alpha *= 1.1
    
    return best_a

def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 6)
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    elif mode == 2:
        # Triangle
        center = n / 2
        a = 1 - 2 * np.abs(x - center) / n
        a = np.maximum(a, 0.01)
    elif mode == 3:
        # Double bump
        c1 = n * 0.3
        c2 = n * 0.7
        a = np.exp(-((x - c1) ** 2) / (n * 0.05)) + np.exp(-((x - c2) ** 2) / (n * 0.05))
    elif mode == 4:
        # Random smooth
        a = rng.uniform(0.5, 1.5, n)
        # Smooth
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
    else:
        # Uniform random
        a = rng.uniform(0, 1, n)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) momentum-based chase-the-peak gradient, (2) LP alternating with
constraint generation via HiGHS, (3) pair-unequalization heuristic,
(4) frequency-domain projection, (5) multi-start with diverse initializations,
(6) simulated annealing for escaping local optima, (7) multi-n exploration.

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix_rows(a, k_indices):
    """Build rows of convolution matrix for specific k values. A @ g = conv(a,g)[k_indices]."""
    n = len(a)
    m = len(k_indices)
    A = np.zeros((m, n))
    for idx, k in enumerate(k_indices):
        for j in range(n):
            l = k - j
            if 0 <= l < n:
                A[idx, j] = a[l]
    return A


def solve_lp_minpeak(a, k_indices=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in k_indices, sum(g)=1, g>=0.
    Uses HiGHS. Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    if k_indices is None:
        k_indices = list(range(2 * n - 1))
    else:
        k_indices = list(k_indices)
    m = len(k_indices)
    if m == 0:
        return np.ones(n) / n, 0.0

    A = build_conv_matrix_rows(a, k_indices)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def chase_peak(a, n_iters=5000, time_budget=30.0, momentum=0.85, verbose=False):
    """
    Chase-the-peak with Nesterov momentum and adaptive step size.
    Works on normalized sequence (sum=1).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    step_size = 0.005
    stalled = 0
    best_step = step_size
    rng = np.random.default_rng(12345)
    arange_n = np.arange(n)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0
        shifted = k_star - arange_n
        valid = (shifted >= 0) & (shifted < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * a[shifted[valid]]

        # Center gradient (project onto sum=0)
        grad -= np.mean(grad)

        # Nesterov momentum
        v = momentum * v - step_size * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)

            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stalled = 0
                if step_size < 0.02:
                    step_size = min(step_size * 1.02, 0.02)
            else:
                a = best_a.copy()
                stalled += 1
                step_size *= 0.92
                if stalled > 30:
                    # Perturbation to escape
                    scale = 0.005 * min(1 + stalled / 50, 3.0)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    v = np.zeros(n)
                    step_size = best_step
                    stalled = 0

        if verbose and it % 500 == 0:
            print(f"      chase it{it}: obj={best_obj:.8f}, step={step_size:.5f}")

    return best_a


def lp_alternate(a, n_iters=200, time_budget=60.0, verbose=False):
    """
    LP alternating with constraint generation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Start with all constraints, can reduce later
    all_k = list(range(2 * n - 1))
    prev_t = 0.5

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP
        g, M = solve_lp_minpeak(a, all_k)
        if g is None:
            break

        # Line search with adaptive t
        candidate_ts = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if 0 < prev_t < 1:
            candidate_ts = sorted(set(candidate_ts + [max(0.05, prev_t * 0.5), min(1.0, prev_t * 1.5)]))

        best_t = 0
        best_t_obj = best_obj
        best_t_a = best_a.copy()

        for t in candidate_ts:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_t_a = a_try.copy()

        if best_t > 0 and best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a.copy()
            prev_t = best_t
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, M={M:.6f}")
        else:
            # Try perturbation
            if it > 10:
                rng = np.random.default_rng(it * 7 + n)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)

    return best_a


def pair_unequalize(a, n_iters=100, time_budget=20.0, verbose=False):
    """
    For the peak bin, make contributing pairs more unequal to reduce the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    arange_n = np.arange(n)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        cur_peak = conv[k_star]

        improved = False
        # Process pairs for the peak bin
        for i in range(n):
            j = k_star - i
            if j < i or j >= n or j < 0:
                continue

            if i == j:
                # Middle term a[i]^2: try to reduce by moving mass elsewhere
                # Find position p not contributing to bin k_star
                # p contributes if 0 <= k_star - p < n
                # p doesn't contribute if k_star - p < 0 or k_star - p >= n
                # i.e., p > k_star or p < k_star - n + 1
                candidates = []
                for p in range(0, n, max(1, n // 200)):
                    if p != i and (k_star - p < 0 or k_star - p >= n):
                        candidates.append(p)
                if candidates:
                    p = min(candidates, key=lambda x: a[x])
                    for frac in [0.03, 0.08, 0.15]:
                        delta = frac * a[i]
                        a[i] -= delta
                        a[p] += delta
                        new_obj = 2.0 * n * float(np.max(compute_conv(a)))
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[p] -= delta
                continue

            # Pair (i, j) with i < j, i + j = k_star
            # Product a[i]*a[j]: to reduce, make more unequal
            # Move from smaller to larger
            if a[i] <= a[j]:
                small_i, large_i = i, j
            else:
                small_i, large_i = j, i

            for frac in [0.05, 0.1, 0.2, 0.35]:
                delta = frac * a[small_i]
                if delta > a[small_i]:
                    delta = a[small_i] * 0.8
                a[small_i] -= delta
                a[large_i] += delta
                new_obj = 2.0 * n * float(np.max(compute_conv(a)))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[small_i] += delta
                    a[large_i] -= delta

        if not improved:
            # Try all peak bins (not just the argmax)
            top_k_indices = np.argsort(conv)[-5:]
            found = False
            for k_try in top_k_indices:
                if k_try == k_star:
                    continue
                for i in range(0, n, 3):
                    j = k_try - i
                    if j < 0 or j >= n or j <= i:
                        continue
                    if a[i] <= a[j]:
                        small_i, large_i = i, j
                    else:
                        small_i, large_i = j, i
                    delta = 0.1 * a[small_i]
                    a[small_i] -= delta
                    a[large_i] += delta
                    new_obj = 2.0 * n * float(np.max(compute_conv(a)))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                    else:
                        a[small_i] += delta
                        a[large_i] -= delta
                if found:
                    break

            if not improved:
                break

    return best_a


def freq_domain_proj(a, n_iters=300, time_budget=30.0, verbose=False):
    """
    Frequency-domain projection: iteratively modify spectrum to flatten autocorrelation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 4 * n:
        N *= 2

    a_pad = np.zeros(N)
    a_pad[:n] = a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.02
    arange_N = np.arange(N)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a_pad)
        P = np.abs(A) ** 2
        r = np.real(ifft(P))

        # Look at the region corresponding to linear conv (center of circular)
        # The linear conv peak is at index n-1 in the 2n-1 array
        # In circular conv of size N, this corresponds to index (N//2 + n - 1) % N
        # Actually, let's just use the full circular autocorrelation
        k_star = int(np.argmax(r))

        # Reduce power where it contributes to the peak
        phase = np.cos(2 * np.pi * k_star * arange_N / N)
        P_new = P.copy()
        P_new[1:] *= (1 - alpha * np.maximum(phase[1:], 0))
        P_new[0] = P[0]  # Keep DC
        P_new = np.maximum(P_new, 0)

        # Recovery
        mag = np.sqrt(P_new)
        phase_A = np.angle(A)
        A_new = mag * np.exp(1j * phase_A)
        a_new = np.real(ifft(A_new))
        a_new = np.maximum(a_new[:n], 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a_pad = np.zeros(N)
                a_pad[:n] = a_new
                if alpha < 0.08:
                    alpha *= 1.01
            else:
                a_pad = np.zeros(N)
                a_pad[:n] = best_a
                alpha *= 0.95

        if verbose and it % 100 == 0:
            print(f"      freq it{it}: obj={best_obj:.8f}, alpha={alpha:.4f}")

    return best_a


def simulated_annealing(a, n_iters=2000, time_budget=30.0, T0=0.01, verbose=False):
    """
    Simulated annealing: randomly perturb pairs of elements.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(999)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T0 * (0.999 ** it)
        if T < 1e-8:
            T = 1e-8

        # Random pair swap
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Move small fraction of mass from i to j (or vice versa)
        frac = rng.uniform(0.01, 0.1)
        if a[i] > a[j]:
            delta = frac * a[i]
            a[i] -= delta
            a[j] += delta
        else:
            delta = frac * a[j]
            a[j] -= delta
            a[i] += delta

        new_obj = compute_obj(a)

        if new_obj < cur_obj or rng.random() < np.exp(-(new_obj - cur_obj) / T):
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            # Revert
            if a[i] > a[j]:
                a[i] += delta
                a[j] -= delta
            else:
                a[j] += delta
                a[i] -= delta

        if verbose and it % 500 == 0:
            print(f"      SA it{it}: best={best_obj:.8f}, T={T:.6e}")

    return best_a


def multi_peak_chase(a, n_iters=3000, time_budget=30.0, verbose=False):

def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    obj2 = compute_obj(a2)
    if obj2 < best_score:
        best_score = obj2
        best_seq = a2.copy()
        print(f"  LP: {best_score:.8f}")

    # Phase 3: Multi-peak chase
    print(f"\n=== Phase 3: Multi-Peak Chase ===")
    remaining = deadline - start_time
    a3 = multi_peak_chase(best_seq, n_iters=5000, time_budget=min(remaining * 0.1, 50.0), verbose=True)
    obj3 = compute_obj(a3)
    if obj3 < best_score:
        best_score = obj3
        best_seq = a3.copy()
        print(f"  Multi: {best_score:.8f}")

    # Phase 4: Pair unequalization
    print(f"\n=== Phase 4: Pair Unequalize ===")
    remaining = deadline - start_time
    a4 = pair_unequalize(best_seq, n_iters=200, time_budget=min(remaining * 0.08, 30.0))
    obj4 = compute_obj(a4)
    if obj4 < best_score:
        best_score = obj4
        best_seq = a4.copy()
        print(f"  Pair: {best_score:.8f}")

    # Phase 5: Frequency projection
    print(f"\n=== Phase 5: Frequency Projection ===")
    remaining = deadline - start_time
    a5 = freq_domain_proj(best_seq, n_iters=500, time_budget=min(remaining * 0.08, 30.0), verbose=True)
    obj5 = compute_obj(a5)
    if obj5 < best_score:
        best_score = obj5
        best_seq = a5.copy()
        print(f"  Freq: {best_score:.8f}")

    # Phase 6: Simulated annealing
    print(f"\n=== Phase 6: Simulated Annealing ===")
    remaining = deadline - start_time
    a6 = simulated_annealing(best_seq, n_iters=5000, time_budget=min(remaining * 0.08, 30.0), verbose=True)
    obj6 = compute_obj(a6)
    if obj6 < best_score:
        best_score = obj6
        best_seq = a6.copy()
        print(f"  SA: {best_score:.8f}")

    # Phase 7: Multi-n exploration
    print(f"\n=== Phase 7: Multi-n Exploration ===")
    remaining = deadline - start_time
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best,
        n_best + 50, n_best + 100, n_best + 150, n_best + 200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - start_time
        if remaining < 25:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
            a_try = a_try / np.sum(a_try)

        # Quick chase + LP
        alloc = min(remaining * 0.12, 25.0)
        a_opt = chase_peak(a_try, n_iters=3000, time_budget=alloc * 0.6)
        obj_try = compute_obj(a_opt)
        if remaining > 15:
            alloc2 = min(remaining * 0.08, 15.0)
            a_opt2 = lp_alternate(a_opt, n_iters=30, time_budget=alloc2)
            obj_try2 = compute_obj(a_opt2)
            if obj_try2 < obj_try:
                a_opt = a_opt2
                obj_try = obj_try2

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # Phase 8: Random restarts
    print(f"\n=== Phase 8: Random Restarts ===")
    remaining = deadline - start_time
    n_best = len(best_seq)

    for r in range(12):
        remaining = deadline - start_time
        if remaining < 12:
            break

        # Diverse initialization
        x = np.arange(n_best)
        mode = r % 6
        if mode == 0:
            a_r = 0.5 + 0.5 * np.cos(2 * np.pi * x / n_best)
        elif mode == 1:
            a_r = np.exp(-((x - n_best / 2) ** 2) / (2 * (n_best / 6) ** 2)) + 0.1
        elif mode == 2:
            a_r = 1.0 + 0.3 * np.sin(4 * np.pi * x / n_best)
        elif mode == 3:
            a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
        elif mode == 4:
            a_r = best_seq * np.exp(rng.normal(0, 0.15, n_best))
            # Flip a random section
            lo = rng.integers(0, n_best // 2)
            hi = rng.integers(n_best // 2, n_best)
            a_r[lo:hi] = best_seq[lo:hi] * 0.5
        else:
            a_r = 1.0 + 0.2 * rng.normal(0, 1, n_best)

        a_r = np.maximum(a_r, 0)
        a_r = a_r / np.sum(a_r)

        alloc = min(remaining * 0.08, 15.0)
        a_opt = chase_peak(a_r, n_iters=2000, time_budget=alloc)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): {best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): {obj_r:.8f}")

    # Phase 9: Final intensive polish
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - start_time

    if remaining > 15:
        # Chase peak (long)
        alloc = min(remaining * 0.25, 60.0)
        a_fp = chase_peak(best_seq, n_iters=20000, time_budget=alloc, verbose=True)
        obj_fp = compute_obj(a_fp)
        if obj_fp < best_score:
            best_score = obj_fp
            best_seq = a_fp.copy()
            print(f"  Final chase: {best_score:.8f}")

        # Multi-peak
        remaining = deadline - start_time
        if remaining > 15:
            alloc = min(remaining * 0.2, 40.0)
            a_mp = multi_peak_chase(best_seq, n_iters=10000, time_budget=alloc, verbose=True)
            obj_mp = compute_obj(a_mp)
            if obj_mp < best_score:
                best_score = obj_mp
                best_seq = a_mp.copy()
                print(f"  Final multi: {best_score:.8f}")

        # LP
        remaining = deadline - start_time
        if remaining > 15:
            alloc = min(remaining * 0.3, 60.0)
            a_lpf = lp_alternate(best_seq, n_iters=50, time_budget=alloc, verbose=True)
            obj_lpf = compute_obj(a_lpf)
            if obj_lpf < best_score:
                best_score = obj_lpf
                best_seq = a_lpf.copy()
                print(f"  Final LP: {best_score:.8f}")

        # Pair unequalize
        remaining = deadline - start_time
        if remaining > 10:
            a_pf = pair_unequalize(best_seq, n_iters=100, time_budget=min(remaining * 0.2, 20.0))
            obj_pf = compute_obj(a_pf)
            if obj_pf < best_score:
                best_score = obj_pf
                best_seq = a_pf.copy()
                print(f"  Final pair: {best_score:.8f}")

        # One more chase
        remaining = deadline - start_time
        if remaining > 5:
            a_last = chase_peak(best_seq, n_iters=5000, time_budget=min(remaining * 0.4, 15.0))
            obj_last = compute_obj(a_last)
            if obj_last < best_score:
                best_score = obj_last
                best_seq = a_last.copy()
                print(f"  Last chase: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
