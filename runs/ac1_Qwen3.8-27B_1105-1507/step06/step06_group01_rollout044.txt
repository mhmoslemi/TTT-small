2. Adaptive growing-active-set LP best-response

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv
    if not cond.any():
        return np.ones(n) / n
    rho_idx = int(np.max(np.where(cond)[0]))
    theta = (cv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def fw_step(a, grad, lam):
    """Frank-Wolfe step: a_new = (1-lam)*a + lam*e_j where j = argmin grad."""
    j = int(np.argmin(grad))
    e_j = np.zeros(len(a))
    e_j[j] = 1.0
    return (1.0 - lam) * a + lam * e_j

def run_frank_wolfe(a_init, max_iters, deadline, p=8.0, adaptive=True):
    """Frank-Wolfe method with p-norm smoothing of max(conv(a,a))."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_pos = np.maximum(b, 1e-30)
        pn = (np.sum(b_pos ** p)) ** (1.0 / p)

        w = b_pos ** (p - 1)
        grad = np.zeros(n)
        for i in range(n):
            end = min(i + n, len(b))
            if end > i:
                grad[i] = 2.0 * (pn ** (1 - p)) * np.dot(w[i:end], a[:end - i])
        
        # FW step
        if adaptive:
            lam = 2.0 / (it + 2)
        else:
            lam = 0.1
        
        a_new = fw_step(a, grad, lam)
        
        # Check improvement
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Projected gradient with p-norm smoothing (L1-regularized smooth max)
2. Growing active-set LP best-response with dual-guided constraint addition
3. Exact 1D coordinate descent with golden section
4. Multi-init, multi-length exploration
Key innovations: growing active set (add most-violated constraint each iter),
dual solution for constraint selection, interleaved LP+CD, chirp init.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv
    if not cond.any():
        return np.ones(n) / n
    rho_idx = int(np.max(np.where(cond)[0]))
    theta = (cv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def pnorm_conv(a, p):
    """Compute p-norm of conv(a,a) and gradient."""
    b = fast_conv(a)
    b_pos = np.maximum(b, 1e-30)
    pn = (np.sum(b_pos ** p)) ** (1.0 / p)
    w = b_pos ** (p - 1)
    corr = np.correlate(w, a, mode='valid')
    grad = 2.0 * (pn ** (1 - p)) * corr
    return pn, grad, b

def proj_grad_step(a_init, max_iters, deadline, p=16.0, lr=None):
    """Projected gradient descent with p-norm smoothing."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    if lr is None:
        lr = 2.0 / (p * n)  # heuristic
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    prev_pn = float('inf')
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        pn, grad, b = pnorm_conv(a, p)
        
        # Backtracking
        t = lr
        for _ in range(20):
            a_trial = proj_simplex(a - t * grad)
            b_trial = fast_conv(a_trial)
            pn_trial = (np.sum(np.maximum(b_trial, 1e-30) ** p)) ** (1.0 / p)
            if pn_trial <= pn - 0.5 * t * np.dot(grad, a - a_trial):
                break
            t *= 0.5
        
        a = a - t * grad
        a = proj_simplex(a)
        
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
        
        if abs(pn - prev_pn) < 1e-12:
            break
        prev_pn = pn
    
    return best_a

def build_lp_matrix(a, active_indices, n):
    """Build sparse LP constraint matrix.
    Row k (for convolution index active_indices[k]):
    A[k, j] = a[active_indices[k] - j] if 0 <= active_indices[k] - j < n
    """
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A

def solve_lp_best_response(a, M, active_indices):
    """Solve LP: max sum(g) s.t. A*g <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_lp_matrix(a, active_indices, n)
    if A.nnz == 0:
        return None
    
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g
    except Exception:
        pass
    return None

def run_growing_lp(a_init, max_iters, deadline, k_start=10, k_max=100):
    """LP mirror descent with growing active set.
    
    Each iteration:
    1. Solve LP with current active set
    2. Get best response g
    3. Blend toward g
    4. Check for violated constraints; add most violated to active set
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Initial active set: top k_start
    b = fast_conv(a)
    active = np.argsort(b)[::-1][:k_start].tolist()
    
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Solve LP
        g = solve_lp_best_response(a, best_M, active)
        if g is None:
            # Expand active set
            b = fast_conv(a)
            new_active = np.argsort(b)[::-1][:min(len(active) * 2, k_max, 2 * n - 1)].tolist()
            active = new_active
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        g_s = g.sum()
        if g_s < 1e-10:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        g_norm = g / g_s
        
        # Line search for blend parameter
        improved = False
        base_t = 0.5
        for t in [base_t, base_t*0.5, base_t*0.25, base_t*0.125, 0.02, 0.01, 0.005, 0.001]:
            a_trial = (1.0 - t) * a + t * g_norm
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-14):
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
            # After improvement, check for new violated constraints
            b = fast_conv(a)
            # Find most violated constraint not in active set
            current_max = M_trial  # Actually this is the new best_M
            # We want to add constraints where b > best_M
            # But b is the convolution of the NEW a, not the old one
            # For the next LP, we want constraints that are close to being tight
            # Update active set: top k indices of current b
            k_cur = min(len(active) + 2, k_max, 2 * n - 1)
            active = np.argsort(b)[::-1][:k_cur].tolist()
        else:
            no_improve += 1
            # Expand active set
            b = fast_conv(a)
            k_cur = min(len(active) * 2, k_max, 2 * n - 1)
            active = np.argsort(b)[::-1][:k_cur].tolist()
            if no_improve > 15:
                break
    
    return best_a

def coordinate_descent_1d(a, i, b, n):
    """Find optimal a[i] given all others fixed. Returns optimal value."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Collect affected values: b[k] for k in [k_lo, k_hi]
    # For each k, b[k] = C_k + S_k * x where:
    #   if k == 2*i: b[k] = C_k + x^2 (quadratic)
    #   else: b[k] = C_k + 2*a[k-i]*x (linear)
    # where C_k = b[k] - (current contribution of a[i])
    
    c_vals = []
    slopes = []
    has_quad = False
    quad_idx = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_idx = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    # Unaffected max
    k_max_total = 2 * n - 2
    if k_lo > 0 or k_hi < k_max_total:
        mask = np.ones(k_max_total + 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals) if c_vals else np.array([0.0])
    s_arr = np.array(slopes) if slopes else np.array([0.0])
    
    def obj_at(x):
        vals = c_arr + s_arr * x
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_idx] + x * x
            if qv > m:
                m = qv
        return m
    
    # Golden section search on [0, hi]
    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) * 1.5, 0.1)
    
    # Ensure hi is sufficient
    for _ in range(5):
        if obj_at(hi) > obj_at(0.0) and hi < 10.0:
            hi *= 1.5
        else:
            break
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = lo + (1.0 - gr) * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = lo + (1.0 - gr) * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with random order."""
    rng = np.random.default_rng(seed)

    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = np.arange(n)
        rng.shuffle(order)
        b = fast_conv(a)
        sweep_changed = False
        
        for i in order:
            x_opt = coordinate_descent_1d(a, int(i), b, n)
            if abs(x_opt - a[i]) > 1e-12:
                # Update b
                k_lo = int(i)
                k_hi = min(2 * n - 2, int(i) + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * int(i):
                        b[k] += x_opt * x_opt - a[i] * a[i]
                    else:
                        j = k - int(i)
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - a[i])
                a[int(i)] = x_opt
                sweep_changed = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-14):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_changed:
            no_improve += 5
        if no_improve > 20:
            break
    
    return best_a
