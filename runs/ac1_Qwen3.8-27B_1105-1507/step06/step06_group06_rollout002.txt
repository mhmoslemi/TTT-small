    """Compute gradient of log-sum-exp smoothed max of convolution."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    conv_shifted = conv - conv.max()
    weights = np.exp(beta * conv_shifted)
    weights /= weights.sum()
    # Gradient: grad[i] = sum_k weights[k] * 2 * a[k-i] (for valid k-i)

            if 0 <= j < n:

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(a, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = a.copy()
    n = len(a)
    # Bisection on threshold
    lo = a.min() - 1
    hi = a.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.maximum(a - mid, 0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.maximum(a - (lo + hi) / 2, 0)
    s = projected.sum()
    if s < 1e-15:
        return np.ones(n) * target_sum / n
    return projected * (target_sum / s)


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. T_a * g <= M, g >= 0. Returns normalized g."""
    n = len(a)
    if n > 2500:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def backtracking_line_search(a, direction, initial_obj, max_iter=50):
    """Find step size t for a + t*(direction - a) that reduces objective."""
    n = len(a)
    best_t = 0.0
    best_obj = initial_obj
    t = 1.0
    for _ in range(max_iter):
        a_try = a + t * (direction - a)
        a_try = project_simplex(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
        t *= 0.7
        if t < 1e-6:
            break
    return best_t, best_obj, a + best_t * (direction - a) if best_t > 0 else a

def spectral_flatten(a, max_iters=100, time_budget=30.0, compress_power=0.7):
    """
    Spectral domain power spectrum compression.
    Compress |FFT(a)|^2 to be flatter, keep phase, project back.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = fft(a, N)
        power = np.abs(fa) ** 2
        p_max = power.max()
        if p_max < 1e-15:
            break
        
        p_norm = power / p_max
        # Compress: p_new = p^alpha (alpha < 1 compresses peaks)
        alpha = compress_power + 0.1 * (it / max_iters)
        p_compressed = np.maximum(p_norm, 1e-10) ** alpha
        
        # Reconstruct
        phase = np.angle(fa)
        mag = np.sqrt(p_compressed * p_max)
        fa_new = mag * np.exp(1j * phase)
        a_new = ifft(fa_new).real[:n]
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
    
    return best_a, best_obj


def smoothed_gradient_descent(a, max_iters=300, time_budget=60.0, beta_start=3.0, beta_end=50.0):
    """
    Gradient descent on log-sum-exp smoothed max of convolution.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 3 * n:
        N <<= 1

    step_size = 0.05
    
    for it in range(max_iters):

        # This is 2 * correlate(weights, a)

        # corr[m] = sum_k w[k] * a[(k-m) mod N]

        # Actually, let me use a different approach
        # conv_w[i] = sum_k w[k] * a[k-i] (for 0 <= k-i < n)

        # That's fine for a few hundred iterations

        # Vectorized: for each k, grad[i] += 2*w[k]*a[k-i] for valid i

            N_full <<= 1

        a_rev_fft = fft(a_rev, N_full)

def chase_peak_refined(a, n_steps=100, time_budget=30.0):

def resample(a, new_n):
    n = len(a)
