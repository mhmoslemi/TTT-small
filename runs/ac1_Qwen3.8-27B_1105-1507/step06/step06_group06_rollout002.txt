        # Target: |A[k]|^2 = 1/(2n-1) for all k (approximately)

        # So sum_k |A[k]|^2 >= N, meaning average power is >= 1/N
        # DC power is |A[0]|^2 = 1
        # Remaining power for N-1 bins: sum_{k!=0} |A[k]|^2 = N*sum(a^2) - 1
        # For flat: |A[k]|^2 = (N*sum(a^2) - 1)/(N-1) for k != 0
        
        dc = power[0]
        remaining = power.sum() - dc
        target_power = remaining / (N - 1)
        
        # Blend toward target
        alpha = 0.1  # blending factor
        new_power = power.copy()
        new_power[1:] = (1-alpha) * power[1:] + alpha * target_power
        new_power[0] = dc  # keep DC
        
        # Reconstruct
        new_mag = np.sqrt(new_power)
        phase = np.angle(A)
        new_A = new_mag * np.exp(1j * phase)
        new_a = np.fft.ifft(new_A).real[:n]
        
        # Project to non-negative
        new_a = np.maximum(new_a, 0)
        s_new = new_a.sum()
        if s_new < 1e-10:

        new_a = new_a / s_new

            best_a = new_a.copy()

"""
Spectral water-filling + LP alternating + multi-length search for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Key innovations: (1) aggressive iterative spectral
flattening with adaptive water-filling, (2) LP alternating with adaptive M,
(3) multi-start spectral approaches, (4) careful length exploration around parent.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        val = a[0] * n if n > 0 else 1.0
        return np.full(new_n, val / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def spectral_water_filling(a, max_iters=2000, time_budget=60.0, alpha_init=0.3, alpha_min=0.02):
    """
    Aggressive spectral water-filling: iteratively flatten the power spectrum
    to minimize the peak of the autocorrelation. Uses adaptive alpha and
    water-filling to compress dynamic range.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters, 1)
        alpha = alpha_init * (alpha_min / alpha_init) ** progress

        A = np.fft.fft(a, N)
        power = np.abs(A) ** 2

        # Water-filling approach: compress toward the mean of non-DC bins
        dc_power = power[0]
        non_dc_power = power.copy()
        non_dc_power[0] = 0
        total_non_dc = non_dc_power.sum()

        if total_non_dc < 1e-30:
            break

        target_power = total_non_dc / (N - 1)

        # Water-filling: cap the top, fill the bottom
        new_power = power.copy()
        # Compress high bins
        high_mask = non_dc_power > target_power * 1.5
        new_power[high_mask] = non_dc_power[high_mask] * (1 - alpha) + target_power * 1.5 * alpha
        # Slightly boost low bins
        low_mask = non_dc_power < target_power * 0.5
        new_power[low_mask] = non_dc_power[low_mask] * (1 - alpha * 0.5) + target_power * 0.5 * alpha * 0.5
        # Middle: gentle compression
        mid_mask = ~(high_mask | low_mask | (np.arange(N) == 0))
        new_power[mid_mask] = non_dc_power[mid_mask] * (1 - alpha * 0.3) + target_power * alpha * 0.3
        new_power[0] = dc_power

        # Reconstruct
        new_mag = np.sqrt(np.maximum(new_power, 0))
        phase = np.angle(A)
        new_A = new_mag * np.exp(1j * phase)
        new_a = np.fft.ifft(new_A).real[:n]

        # Project
        new_a = project_nonneg_sum1(new_a)
        obj = compute_obj(new_a)

        if obj < best_obj:
            best_obj = obj
            best_a = new_a.copy()

        a = new_a

    return best_a, best_obj


def spectral_log_compression(a, max_iters=2000, time_budget=60.0, gamma_init=2.0, gamma_min=1.1):
    """
    Spectral compression via log/power scaling of the magnitude spectrum.
    More aggressive than simple water-filling for initial flattening.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters, 1)
        gamma = gamma_init * (gamma_min / gamma_init) ** progress

        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Power compression: mag_new = mag^beta where beta < 1 compresses peaks
        mag_max = np.max(mag)
        if mag_max < 1e-30:
            break
        mag_norm = mag / mag_max
        # Keep DC fixed
        dc_idx = 0
        mag_new = mag_norm ** (1.0 / gamma)
        mag_new[dc_idx] = 1.0  # keep DC at max
        mag_new = mag_new * mag_max

        new_A = mag_new * np.exp(1j * phase)
        new_a = np.fft.ifft(new_A).real[:n]
        new_a = project_nonneg_sum1(new_a)

        obj = compute_obj(new_a)
        if obj < best_obj:
            best_obj = obj
            best_a = new_a.copy()
        a = new_a

    return best_a, best_obj


def lp_alternate(a, max_iters=30, time_budget=40.0):
    """LP alternating: maximize sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 2000:
        return a, compute_obj(a)

    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Build constraint matrix: (a*g)[k] = sum_j a[max(0,k-n+1):min(n,k+1)] * g[...]
        # More efficiently: A_ub[k, j] = a[j] if 0 <= k-j < n, else 0
        A_ub = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_max >= j_min:
                A_ub[k, j_min:j_max + 1] = a[j_min:j_max + 1]

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': min(15, time_budget - (time.monotonic() - start))})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                s = float(np.sum(g))
                if s > 1e-10:
                    g = g / s
                    # Line search
                    best_t_obj = np.inf
                    best_t_a = best_a.copy()
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_try = (1 - t) * a + t * g
                        a_try = project_nonneg_sum1(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_t_obj:
                            best_t_obj = obj_try
                            best_t_a = a_try.copy()
                    if best_t_obj < best_obj - 1e-12:
                        best_obj = best_t_obj
                        best_a = best_t_a.copy()
                        a = best_t_a
        except Exception:
            break

    return best_a, best_obj

def subgradient_peaks(a, max_iters=500, time_budget=30.0, n_peaks=5):
    """Subgradient descent using top-K peaks with softmax weighting."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    step = 0.05
    momentum = 0.8

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Top-K peaks with softmax weighting
        tau = 0.01 * cmax + 0.001
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -50, 0)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum

        # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j] for valid k-j
        grad = np.zeros(n)
        # Use FFT for efficiency
        N_fft = 1
        while N_fft < 2 * n - 1:
            N_fft <<= 1
        fw = rfft(weights, N_fft)
        fa = rfft(a, N_fft)
        # cross-correlation: sum_k w[k] * a[k-j]
        # = ifft(conj(fft(w)) * fft(a)) ... 
        # Actually: correlation(w, a)[j] = sum_k w[k] * a[k-j]
        # In FFT: ifft(conj(W) * A) gives circular correlation
        corr = irfft(np.conj(fw) * fa, N_fft)
        # corr[j] = sum_k w[k] * a[(k-j) mod N_fft]
        # For linear: we want sum_{k: 0<=k-j<n} w[k] * a[k-j]
        # corr[j] for j in [0, n-1] should give us what we want (if N_fft is large enough)
        grad = 2.0 * corr[:n]

        grad_centered = grad - np.mean(grad)
        v = momentum * v - step * grad_centered
        a_new = a - v
        a_new = project_nonneg_sum1(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        step *= 0.997

    return best_a, best_obj


def chase_peak(a, n_steps=80, step_size=0.02):
    """Chase the maximum peak in the convolution."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        if j_max >= j_min:
            grad[j_min:j_max + 1] = 2.0 * a[j_min:j_max + 1]

        grad_centered = grad - np.mean(grad)

        for ss in [step_size, step_size * 0.5, step_size * 2.0]:
            a_try = a - ss * grad_centered
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a, best_obj

    phase3_budget = min(400.0, (deadline - start_time) * 0.30)
    phase3_start = time.monotonic()

    n_cycles = min(15, int(phase3_budget / 25))
    for ci in range(n_cycles):
        if time.monotonic() > phase3_start + phase3_budget:
            break
        cycle_budget = min(30.0, (phase3_start + phase3_budget) - time.monotonic())
        if cycle_budget < 10:
            break

        a_new, obj_new = combined_iteration(best_seq, time_budget=cycle_budget)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Cycle {ci}: NEW BEST {best_score:.8f}")
        else:
            print(f"  Cycle {ci}: obj={obj_new:.8f} (no improvement)")

        # Occasionally try a perturbation
        if ci % 3 == 2:
            a_pert = best_seq * np.exp(rng.normal(0, 0.03, len(best_seq)))
            a_pert = project_nonneg_sum1(a_pert)
            a_pert, obj_pert = chase_peak(a_pert, n_steps=30, step_size=0.015)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb cycle {ci}: NEW BEST {best_score:.8f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    print(f"Best so far: {best_score:.8f}")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 40:
        phase4_budget = min(250.0, remaining4 * 0.45)
        phase4_start = time.monotonic()

        n_current = len(best_seq)
        n_tries = [n_current - 50, n_current - 100, n_current + 50, n_current + 100,
                   n_current + 200, n_current - 200, n_current + 300]
        n_tries = [n for n in n_tries if 100 <= n <= 2500]
        n_tries = list(dict.fromkeys(n_tries))

        per_n_budget = max(10.0, phase4_budget / len(n_tries))

        for ni, n_try in enumerate(n_tries):
            if time.monotonic() > phase4_start + phase4_budget:
                break
            b_left = phase4_budget - (time.monotonic() - phase4_start)
            if b_left < 8:
                break

            a_resampled = resample_sequence(best_seq, n_try)

            # Quick spectral + chase
            a_opt, obj_opt = spectral_water_filling(a_resampled, max_iters=500,
                                                     time_budget=min(per_n_budget * 0.4, 20.0))
            a_opt, obj_opt = chase_peak(a_opt, n_steps=40, step_size=0.02)

            if n_try <= 1200:
                remain_n = b_left - (time.monotonic() - phase4_start) - (time.monotonic() - phase4_start)
                a_opt2, obj_opt2 = lp_alternate(a_opt, max_iters=8, time_budget=min(per_n_budget * 0.3, 15.0))
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 15 and best_seq is not None:
        n_final = len(best_seq)
        polish_rounds = min(8, int(remaining5 / 12))

        for pr in range(polish_rounds):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 10:
                break

            # Short spectral flush
            a_p, obj_p = spectral_water_filling(best_seq, max_iters=300,
                                                 time_budget=min(rem * 0.2, 10.0),
                                                 alpha_init=0.08, alpha_min=0.01)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Polish spectral r{pr}: {best_score:.8f}")

            # Chase peak
            rem = deadline - time.monotonic()
            if rem > 5:
                a_p2, obj_p2 = chase_peak(best_seq, n_steps=50, step_size=0.01)
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    print(f"  Polish chase r{pr}: {best_score:.8f}")

            # LP if feasible
            rem = deadline - time.monotonic()
            if rem > 10 and n_final <= 1200:
                a_p3, obj_p3 = lp_alternate(best_seq, max_iters=8, time_budget=min(rem * 0.3, 15.0))
                if obj_p3 < best_score:
                    best_score = obj_p3
                    best_seq = a_p3.copy()
                    print(f"  Polish LP r{pr}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
