import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        hi = min(j + n, 2 * n - 1)
        A[j:hi, j] = a[:hi - j]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=25):
    """
    Solve: min t subject to (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None

    ag = fft_convolve(a, g)

    # Coarse search
    for t in np.linspace(0, 1, coarse_evals):
        val = (1 - t) ** 2 * aa + 2 * t * (1 - t) * ag + t ** 2 * gg
        m = float(np.max(val))
        if m < best_max:
            best_max = m
            best_t = t

    # Fine search around best
    if best_t > 0.02 and best_t < 0.98:
        lo = max(0.0, best_t - 0.06)
        hi = min(1.0, best_t + 0.06)
        for t in np.linspace(lo, hi, fine_evals):
            val = (1 - t) ** 2 * aa + 2 * t * (1 - t) * ag + t ** 2 * gg
            m = float(np.max(val))
            if m < best_max:
                best_max = m
                best_t = t

    return best_t


def adaptive_subgradient(a, max_iters=2000, time_budget=60.0, tau_init=None, step_init=0.06):
    """
    FFT-accelerated subgradient with adaptive temperature.
    Temperature adapts based on flatness of convolution.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.82
    step = step_init

    # Adaptive initial temperature based on convolution spread
    conv_init = compute_conv(a)
    cmax = float(np.max(conv_init))
    cmean = float(np.mean(conv_init))
    if cmax > 1e-15:
        flatness = cmean / cmax  # 1.0 means perfectly flat
        if tau_init is None:
            tau_init = 0.003 + 0.02 * (1.0 - flatness)
    else:
        tau_init = 0.01

    tau = tau_init
    tau_min = 0.0008

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 0.5:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        cmean = float(np.mean(conv))

        # Adaptive temperature
        if cmax > 1e-15:
            flatness = cmean / cmax
            tau_target = 0.002 + 0.015 * (1.0 - flatness)
            tau = 0.95 * tau + 0.05 * max(tau_target, tau_min)
        tau = max(tau, tau_min)

        shifted = np.clip((conv - cmax) / tau, -400.0, 400.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient via FFT: grad[i] = 2 * sum_j a[j] * w[i+j]
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        # Nesterov momentum
        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Adaptive step: decrease slowly, increase if making progress
        step *= 0.9992
        if obj < best_obj:
            step = min(step * 1.01, 0.15)
        step = max(step, 0.015)

    return best_a, best_obj


def subgradient_exact(a, max_iters=500, time_budget=30.0):
    """
    Subgradient using exact subdifferential (uniform over argmax set).
    More aggressive peak targeting.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.04

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 0.5:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        # Exact subgradient: uniform over argmax
        tol = cmax * 1e-6
        active = np.where(conv >= cmax - tol)[0]
        weights = np.zeros(len(conv))
        weights[active] = 1.0 / len(active)

        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        a_new = a - step * grad_centered / grad_norm
        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try half step
            a_half = a - 0.5 * step * grad_centered / grad_norm
            a_half = normalize_simplex(a_half)
            obj_half = compute_obj(a_half)
            if obj_half < best_obj:
                best_obj = obj_half
                best_a = a_half.copy()
                a = a_half
            step *= 0.92

    return best_a, best_obj


def lp_with_quadratic_search(a, max_iters=30, time_budget=120.0, use_minpeak=True):
    """
    LP alternating with quadratic line search for step size.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        remaining = time_budget - (time.monotonic() - start)
        lp_time = min(25.0, max(5.0, remaining - 10.0))

        if use_minpeak:
            g, t_opt = solve_minpeak_lp(a, time_limit=lp_time)
        else:
            conv = compute_conv(a)
            M = float(np.max(conv))
            g = solve_maxsum_lp(a, M, time_limit=lp_time)
            if g is not None:
                g = normalize_simplex(g)

        if g is None:
            stall += 1
            if stall > 8:
                break
            continue

        g = normalize_simplex(g)

        # Quadratic line search
        t_opt = quadratic_line_search(a, g)

        if t_opt < 0.01:
            # LP direction doesn't help; try small step
            t_opt = 0.05

        a_try = (1.0 - t_opt) * a + t_opt * g
        a_try = normalize_simplex(a_try)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
        else:
            # Try blend of t_opt and smaller values
            improved = False
            for t_frac in [0.5, 0.3, 0.2, 0.7]:
                t_try = t_opt * t_frac
                a_try2 = (1.0 - t_try) * a + t_try * g
                a_try2 = normalize_simplex(a_try2)
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-12:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = a_try2
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall > 6:
                    # Perturb and retry
                    rng = np.random.default_rng(it * 7919 + 3)
                    a = best_a * np.exp(rng.normal(0, 0.03, n))
                    a = normalize_simplex(a)
                    stall = 0
                    break

    return best_a, best_obj


def peak_peeling(a, num_rounds=200, time_budget=30.0):
    """
    Targeted peak reduction: identify the peak, find the most impactful
    coordinate to reduce, and do so carefully.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    for rnd in range(num_rounds):
        if time.monotonic() - start > time_budget - 1.0:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        peak_k = int(np.argmax(conv))

        # Find which a[i] most contributes to the peak
        # (a*a)[peak_k] = sum_i a[i]*a[peak_k - i]
        # Contribution of a[i]: a[i] * a[peak_k - i]
        # Reducing a[i] by delta reduces peak by delta * a[peak_k - i] (to first order)
        # But also affects other conv[k] values.
        
        # Compute sensitivity: for each i, how much does reducing a[i] reduce the peak
        # vs how much does it increase other peaks
        if peak_k < 2 * n - 1:
            contrib = np.zeros(n)
            for i in range(n):
                j = peak_k - i
                if 0 <= j < n:
                    contrib[i] = a[i] * a[j]
        
        total_contrib = np.sum(contrib)
        if total_contrib < 1e-15:
            break

        # Choose the coordinate with highest contribution to reduce
        # But also consider: reducing a[i] affects all conv[k] where i is involved
        # Use a simple heuristic: reduce the coordinate with max contribution
        i_star = int(np.argmax(contrib))
        
        # How much to reduce: try a small reduction
        delta = a[i_star] * 0.02
        a[i_star] -= delta
        a = normalize_simplex(a)
        obj = compute_obj(a)

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
        else:
            # Revert and try a different coordinate
            a = best_a.copy()
            # Try reducing the second-highest contributor
            contrib[i_star] = 0
            i_second = int(np.argmax(contrib))
            if contrib[i_second] < 1e-15:
                break
            a[i_second] -= a[i_second] * 0.015
            a = normalize_simplex(a)
            obj = compute_obj(a)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
            else:
                a = best_a.copy()
                break

    return best_a, best_obj


def generate_diverse_starts(n, num_starts=10, seed=42):
    """Generate diverse initial sequences for multi-start."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # 1. Uniform
    starts.append(np.ones(n) / n)

    # 2. Triangular
    tri = 1.0 - np.abs(i - center) / (center + 1)
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    # 3. Gaussian (narrow)
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 8.0) ** 2))
    starts.append(bell / np.sum(bell))

    # 4. Gaussian (wide)
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 3.5) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # 5. Uniform + noise
    noisy = np.abs(np.ones(n) + rng.normal(0, 0.4, n))
    starts.append(noisy / np.sum(noisy))

    # 6. Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * i / n))
    cos_w = np.maximum(cos_w, 0.01)
    starts.append(cos_w / np.sum(cos_w))

    # 7-10: Parametric with sinusoidal modulation
    for k in range(4):
        sigma = n * (0.15 + 0.4 * rng.random())
        freq = int(1 + rng.integers(0, 6))
        amp = 0.3 + 0.5 * rng.random()
        phase = 2 * np.pi * rng.random()
        bell_p = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
        s = bell_p * mod
        s = np.maximum(s, 0)
        ss = np.sum(s)
        if ss < 1e-10:
            s = np.ones(n) / n
        else:
            s = s / ss
        starts.append(s)

    # Deduplicate
    unique = [starts[0]]
    for s in starts[1:]:
        if all(np.linalg.norm(s - u) > 0.03 for u in unique):
            unique.append(s)

    return unique


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def polish_sequence(a, time_budget=120.0):
    """
    Full polishing pipeline: alternate between LP, subgradient, and peak peeling.
    """
    n = len(a)
    start = time.monotonic()
    best_a = normalize_simplex(a.copy())
    best_obj = compute_obj(best_a)
    stall_global = 0

    methods = ['lp_minpeak', 'sg', 'lp_maxsum', 'sg_exact', 'peel', 'sg', 'lp_minpeak']
    method_idx = 0

    while time.monotonic() - start < time_budget - 3:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        method = methods[method_idx % len(methods)]
        method_idx += 1

        alloc = min(remaining * 0.2, 40.0)
        if alloc < 3:
            break

        if method == 'lp_minpeak' and n <= 1000:
            a_new, obj_new = lp_with_quadratic_search(best_a, max_iters=8, time_budget=alloc, use_minpeak=True)
        elif method == 'lp_maxsum' and n <= 1000:
            a_new, obj_new = lp_with_quadratic_search(best_a, max_iters=8, time_budget=alloc, use_minpeak=False)
        elif method == 'sg':
            a_new, obj_new = adaptive_subgradient(best_a, max_iters=int(alloc * 15), time_budget=alloc)
        elif method == 'sg_exact':
            a_new, obj_new = subgradient_exact(best_a, max_iters=int(alloc * 8), time_budget=alloc)
        else:  # peel
            a_new, obj_new = peak_peeling(best_a, num_rounds=int(alloc * 8), time_budget=alloc)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_global = 0
        else:
            stall_global += 1
            if stall_global >= 5:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                scale = 0.015 + 0.015 * (stall_global % 4)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = normalize_simplex(a_pert)
                a_pert, obj_pert = adaptive_subgradient(a_pert, max_iters=150, time_budget=4.0)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                stall_global = 0

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multi-start diverse initialization,
    LP with quadratic line search, adaptive subgradient, peak peeling,
    and multi-length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    
    Key innovations over previous approach:
    1. Quadratic line search for LP step sizes (second-order model)
    2. Adaptive subgradient temperature based on convolution flatness
    3. Exact subgradient (uniform over argmax) as complementary method
    4. Peak peeling heuristic for targeted local improvements
    5. Wider length exploration (400-2000)
    6. More aggressive LP usage with better time allocation
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Phase 0: Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # Phase 1: Multi-start diverse exploration across lengths
    print("\n=== Phase 1: Multi-start exploration ===")

    # Determine which lengths to try
    n_values = set()
    n_values.add(n_parent)
    for dn in [-200, -100, -50, 50, 100, 200, 300, 400, 500]:
        nv = n_parent + dn
        if 400 <= nv <= 2000:
            n_values.add(nv)
    # Also try some absolute values
    for nv in [400, 500, 600, 700, 800, 900, 1000, 1200, 1500]:
        n_values.add(nv)
    n_values = sorted(set(n_values))

    # Limit to reasonable number of lengths
    if len(n_values) > 8:
        # Pick a diverse subset
        n_values = sorted(n_values)[::max(1, len(n_values) // 8)][:8]

    candidates = []  # list of (obj, seq)

    for n in n_values:
        if time.monotonic() > deadline - 250:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Generate diverse starts
        if n == n_parent and best_seq is not None:
            starts = [normalize_simplex(best_seq.copy())]
            starts.extend(generate_diverse_starts(n, num_starts=6, seed=seed + n))
        else:
            starts = generate_diverse_starts(n, num_starts=6, seed=seed + n * 7)
            # Also try resampled parent if available
            if best_seq is not None and n != n_parent:
                starts.append(normalize_simplex(resample_sequence(best_seq, n)))

        alloc_per_start = min(remaining / (len(starts) + 1), 25.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 200:
                break
            if alloc_per_start < 2:
                break

            # Quick optimization
            t_alloc = min(alloc_per_start, 20.0)
            a_opt, obj_opt = adaptive_subgradient(s, max_iters=int(t_alloc * 8), time_budget=t_alloc * 0.5)

            # If we have time, do LP
            remaining2 = deadline - time.monotonic()
            if remaining2 > 80 and n <= 900 and t_alloc > 5:
                a_lp, obj_lp = lp_with_quadratic_search(a_opt, max_iters=4, time_budget=min(remaining2 * 0.15, 30.0), use_minpeak=True)
                if obj_lp < obj_opt:
                    a_opt, obj_opt = a_lp, obj_lp

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {obj_opt:.8f} ** NEW BEST **")

            candidates.append((obj_opt, a_opt.copy(), n))

    # Sort candidates by objective
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates: {[f'{c[0]:.6f}(n={c[2]})' for c in candidates[:6]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 2: Intensive optimization of top candidates
    print("\n=== Phase 2: Intensive optimization ===")

    # Deduplicate by length (keep best per length)
    best_per_len = {}
    for obj, seq, n in candidates:
        if n not in best_per_len or obj < best_per_len[n][0]:
            best_per_len[n] = (obj, seq)

    top_candidates = sorted(best_per_len.items(), key=lambda x: x[1][0])[:4]

    for idx, (n_cand, (obj_cand, a_cand)) in enumerate(top_candidates):
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand = a_cand.copy()
        alloc = min(remaining * 0.3, 150.0)

        print(f"  Candidate {idx} (n={n_cand}, obj={obj_cand:.8f}): polishing with {alloc:.0f}s budget")

        a_pol, obj_pol = polish_sequence(a_cand, time_budget=alloc)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {obj_pol:.8f} **")
        else:
            print(f"  No improvement (got {obj_pol:.8f})")

    # Phase 3: Length fine-tuning around best
    print(f"\n=== Phase 3: Length fine-tuning (best: {best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = []
        for dn in [-50, -30, -20, -10, 10, 20, 30, 50, 75, 100]:
            nv = n_best + dn
            if 300 <= nv <= 2500:
                test_lengths.append(nv)
        test_lengths = sorted(set(test_lengths))

        for n_test in test_lengths[:6]:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_resampled = normalize_simplex(resample_sequence(best_seq, n_test))
            alloc = min(remaining * 0.15, 30.0)
            if alloc < 5:
                break

            a_opt, obj_opt = adaptive_subgradient(a_resampled, max_iters=int(alloc * 10), time_budget=alloc * 0.4)

            remaining2 = deadline - time.monotonic()
            if remaining2 > 50 and n_test <= 900:
                a_lp, obj_lp = lp_with_quadratic_search(a_opt, max_iters=3, time_budget=min(remaining2 * 0.15, 20.0))
                if obj_lp < obj_opt:
                    a_opt, obj_opt = a_lp, obj_lp

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_test}: ** UPDATE to {best_score:.8f} **")
            else:
                print(f"  n={n_test}: {obj_opt:.8f}")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_final, obj_final = polish_sequence(best_seq, time_budget=remaining * 0.6)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final polish: {best_score:.8f}")
            else:
                print(f"  Final polish: {obj_final:.8f} (no improvement)")

        # Batch perturbations
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_final = len(best_seq)
            rng = np.random.default_rng(seed + 77777)
            for perturb_idx in range(15):
                if time.monotonic() > deadline - 10:
                    break
                scale = 0.008 + 0.008 * (perturb_idx % 5)
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
                a_pert = normalize_simplex(a_pert)

                t_alloc = min(remaining * 0.06, 8.0)
                if t_alloc < 2:
                    break

                a_pert, obj_pert = adaptive_subgradient(a_pert, max_iters=int(t_alloc * 12), time_budget=t_alloc)

                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb {perturb_idx} (scale={scale:.4f}): ** {best_score:.8f} **")

        # Final LP on the absolute best
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        if n_best <= 900 and remaining > 30:
            print(f"  Final LP (n={n_best}, {remaining:.0f}s remaining)")
            a_lp, obj_lp = lp_with_quadratic_search(best_seq, max_iters=15, time_budget=min(remaining * 0.5, 60.0), use_minpeak=True)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP: {best_score:.8f} ** UPDATE **")

            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_lp2, obj_lp2 = lp_with_quadratic_search(best_seq, max_iters=10, time_budget=min(remaining * 0.5, 40.0), use_minpeak=False)
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  MaxSum LP: {best_score:.8f} ** UPDATE **")

        # One more subgradient pass
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sg, obj_sg = adaptive_subgradient(best_seq, max_iters=int(remaining * 12), time_budget=remaining * 0.7)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Final SG: {best_score:.8f} ** UPDATE **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
