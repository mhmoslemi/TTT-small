LP alternating with golden-section search, Nesterov-accelerated subgradient,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_search(f, lo, hi, max_iter=20):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = f(c), f(d)
    for _ in range(max_iter):
        if hi - lo < 0.005:
            break
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0

    best_obj = compute_obj(a)

    return best_a


def subgradient_nesterov(a, max_steps=200, time_budget=50.0):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Soft-max weighting
        tau = 0.5 * cmax / (n * (1 + step * 0.01))
        shifted = np.clip((conv - cmax) / max(tau, 1e-10), -500, 500)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Gradient via FFT
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        grad -= np.mean(grad)
        
        # Nesterov momentum
        lr = 1.0 / (1.0 + 0.1 * step)
        v = 0.9 * v - lr * grad
        a_new = normalize(a + v)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            v *= 0.5
            a = best_a.copy()
    
    return best_a


def ramp_init(n, direction=1):
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64)
    elif direction == 2:
        a = np.arange(n, 0, -1, dtype=np.float64)
    elif direction == 3:
        mid = (n - 1) / 2.0
        a = np.abs(np.arange(n) - mid) + 1
    elif direction == 4:
        a = np.exp(2.0 * np.arange(n) / n)
    elif direction == 5:
        a = np.exp(-2.0 * np.abs(np.arange(n) - n/2) / n)
    else:
        a = np.ones(n)
    return a / a.sum()


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])

    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)
