"""
Optimized step function minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Golden-section coordinate descent with incremental conv updates,
(2) Active-set SCA-LP for escaping local minima,
(3) Subgradient method with Nesterov acceleration.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def cd_golden(a, max_sweeps, deadline, seed=42, gs_iters=20):
    """Coordinate descent with golden section search."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 100
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Adaptive golden section iterations
        if sweep < 50:
            git = max(10, gs_iters - 5)
        else:
            git = gs_iters

        if sweep % 5 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        for i in order:
            old = a[i]
            aff_lo = i
            aff_hi = min(2 * n - 2, i + n - 1)
            aff_len = aff_hi - aff_lo + 1

            a_aff = a[:aff_len]
            conv_aff = conv[aff_lo:aff_hi + 1]

            # Outer max
            if aff_lo > 0 and aff_hi < 2 * n - 2:
                outer = max(conv[:aff_lo].max(), conv[aff_hi + 1:].max())
            elif aff_lo > 0:
                outer = conv[:aff_lo].max()
            elif aff_hi < 2 * n - 2:
                outer = conv[aff_hi + 1:].max()
            else:
                outer = -1e30

            two_i = 2 * i
            ti_off = two_i - aff_lo if (aff_lo <= two_i <= aff_hi) else -1

            # Search range for x = a[i]
            x_lo = 0.0
            x_hi = min(s, 3.0 * old + 2.0 * s / n)

            # Golden section
            c = x_hi - gr * (x_hi - x_lo)
            d = x_lo + gr * (x_hi - x_lo)

            def eval_f(x, _old=old, _s=s, _conv_aff=conv_aff, _a_aff=a_aff, _outer=outer, _ti=ti_off):
                delta = x - _old
                ns = _s + delta
                if ns < 1e-10:
                    return 1e30
                new_aff = _conv_aff + 2.0 * delta * _a_aff
                if _ti >= 0:
                    # Need to add delta^2 at position _ti
                    # But we can't modify _conv_aff, so compute max directly
                    vals = _conv_aff + 2.0 * delta * _a_aff
                    vals[_ti] += delta * delta
                    m = vals.max()
                else:
                    m = (_conv_aff + 2.0 * delta * _a_aff).max()
                if _outer > m:
                    m = _outer
                return 2.0 * n * m / (ns * ns)

            fc = eval_f(c)
            fd = eval_f(d)

            for _ in range(git):
                if fc < fd:
                    x_hi = d
                    d = c
                    fd = fc
                    c = x_hi - gr * (x_hi - x_lo)
                    fc = eval_f(c)
                else:
                    x_lo = c
                    c = d
                    fc = fd
                    d = x_lo + gr * (x_hi - x_lo)
                    fd = eval_f(d)

            x_opt = (x_lo + x_hi) * 0.5
            f_opt = eval_f(x_opt)

            # Check boundary x=0
            f0 = eval_f(0.0)
            if f0 < f_opt:
                x_opt = 0.0

            f_cur = 2.0 * n * max(outer, conv.max()) / (s * s)
            if f_opt < f_cur - 1e-14:
                delta = x_opt - old
                conv[aff_lo:aff_hi + 1] += 2.0 * delta * a_aff
                if ti_off >= 0:
                    conv[two_i] += delta * delta
                a[i] = x_opt
                s += delta

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    return best_a, best_obj


def sca_lp(a, max_iters, deadline, active_frac=0.9, max_constraints=200):
    """SCA-LP with active set reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = conv.max()

        # Select active constraints
        threshold = M * active_frac
        active_idx = np.where(conv >= threshold)[0]
        if len(active_idx) > max_constraints:
            # Take the top max_constraints
            top_idx = np.argsort(conv)[::-1][:max_constraints]
            active_idx = top_idx

        # Build LP: minimize t
        # s.t. 2 * (a * g)[k] - t <= conv[k]  for k in active_idx
        #      sum(g) = 1, g >= 0
        # where (a*g)[k] = sum_j a[k-j] * g[j]
        
        # A_ub[k, j] = 2 * a[k-j] (for k in active_idx, j in 0..n-1)
        # A_ub[k, n] = -1 (for t variable)
        # b_ub[k] = conv[k]
        
        n_con = len(active_idx)
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active_idx):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(idx)
            cols.append(n)
            vals.append(-1.0)

        A_ub = coo_matrix((vals, (rows, cols)), shape=(n_con, n + 1)).tocsr()
        b_ub = conv[active_idx].copy()
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [1.0]
        
        bounds = [(0, None)] * n + [(None, None)]

        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 15:
            break

    return best_a, best_obj


def gs_spectral(a, n_iters=50, deadline=time.time() + 1000):
    """Gerchberg-Saxton inspired spectral step."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0)
    if a.sum() < 1e-10:
        return a
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Compute FFT
        sz = 1
        while sz < 2 * n - 1:
            sz <<= 1
        A = np.fft.rfft(a, sz)
        
        # Target magnitude: try to flatten
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Blend current magnitude with flat target
        target_mag = np.ones_like(mag)
        target_mag[0] = mag[0]  # Keep DC (sum)
        alpha = 0.3  # Blending factor
        
        new_mag = (1 - alpha) * mag + alpha * target_mag * (mag.sum() / max(target_mag.sum(), 1e-10))
        new_mag = np.maximum(new_mag, 0)
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_candidate = np.fft.irfft(A_new, sz)[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_candidate, 0)
        if a_new.sum() < 1e-10:
            continue
        a_new = a_new / a_new.sum()
        
        obj = compute_J(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a
