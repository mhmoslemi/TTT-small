Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def gradient_descent_armijo(a, tau_init=0.005, max_steps=500, time_budget=60.0, 
                              step_init=0.1, verbose=False):
    """
    Gradient descent with Armijo backtracking line search.
    Uses soft-max smoothing of the max function.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    tau = tau_init
    step = step_init
    
    for step_i in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient
        grad, conv = compute_soft_gradient(a, tau)
        
        # Project gradient: remove mean component (preserve sum=1)
        grad_centered = grad - np.mean(grad)
        
        # Armijo backtracking
        obj_current = compute_obj(a)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-12:
            break
        
        alpha = step
        improved = False
        for backtrack in range(20):
            a_try = a - alpha * grad_centered
            a_try = np.maximum(a_try, 0.0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            
            if obj_try < obj_current - 1e-4 * alpha * grad_norm * grad_norm:
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            a = a_try
            obj_new = compute_obj(a)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
            step = min(step * 1.2, 0.5)
        else:
            step *= 0.5
            if step < 1e-10:
                break
        
        # Anneal temperature
        tau *= 0.995
    
    return best_a


def gradient_descent_log_param(a, tau_init=0.003, max_steps=300, time_budget=50.0):
    """
    Gradient descent in log-parameterization.
    a[j] = exp(x[j]) / sum(exp(x))
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    # Initialize x from a
    a_safe = np.maximum(a, 1e-10)
    x = np.log(a_safe)
    x -= np.max(x)  # numerical stability
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    tau = tau_init
    step = 0.1
    
    for step_i in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute current a from x
        exp_x = np.exp(x - np.max(x))
        a = exp_x / np.sum(exp_x)
        
        # Compute gradient with respect to a
        grad_a, conv = compute_soft_gradient(a, tau)
        
        # Transform to x-space: dx[m] = grad_a[m] * a[m] - (sum grad_a[j]*a[j]) * a[m]
        # Actually: d/dx[m] f(a(x)) = sum_j (df/da[j]) * (da[j]/dx[m])
        # da[j]/dx[m] = a[j] * (delta(j,m) - a[m])
        # So grad_x[m] = sum_j grad_a[j] * a[j] * (delta(j,m) - a[m])
        # = a[m] * (grad_a[m] - sum_j grad_a[j] * a[j])
        
        dot = float(np.sum(grad_a * a))
        grad_x = a * (grad_a - dot)
        
        # Gradient step in x-space
        x_new = x - step * grad_x
        
        # Check objective
        exp_xn = np.exp(x_new - np.max(x_new))
        a_new = exp_xn / np.sum(exp_xn)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            x = x_new
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.7
            if step < 1e-10:
                break
        
        tau *= 0.997
    
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate(a, max_iters=30, time_budget=50.0):
    """LP alternating with simple line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        remaining = time_budget - (time.monotonic() - start)
        g = solve_lp_max_sum(a, M, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break
        
        # Try multiple blend factors
        best_t = 0.0
        best_obj_t = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = normalize((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t
        
        a_new = normalize((1.0 - best_t) * a + best_t * g)
        if best_obj_t < best_obj - 1e-12:
            best_obj = best_obj_t
            best_a = a_new.copy()
            a = a_new
        else:
            # Small perturbation to escape
            rng = np.random.default_rng(it * 31 + 7)
            a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))
    
    return best_a


def spectral_equalization(a, max_iters=40, time_budget=30.0):
    """POCS-style spectral equalization."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        
        # Progressive compression
        alpha = 0.3 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = P_norm ** alpha * p_max
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.6 * a + 0.4 * a_new)
    
    return best_a


def peak_shaving(a, max_iters=200, time_budget=30.0):
    """
    Direct peak-shaving: identify elements contributing most to the peak
    and redistribute their mass to positions that reduce the peak.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find top contributors to peak at k_star
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)
        
        if not contrib:
            break
        
        improved = False
        # Try moving mass from top contributors to random positions
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            
            for _ in range(5):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue
                
                a_try = a.copy()
                # Try different amounts of mass transfer
                for frac in [0.03, 0.05, 0.08, 0.12]:
                    delta = frac * a_try[j1]
                    a_try_temp = a_try.copy()
                    a_try_temp[j1] -= delta
                    a_try_temp[j_target] += delta
                    a_try_temp = np.maximum(a_try_temp, 0.0)
                    s = float(np.sum(a_try_temp))
                    if s < 1e-10:
                        continue
                    a_try_temp = a_try_temp / s
                    obj_try = compute_obj(a_try_temp)
                    
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try_temp.copy()
                        a = a_try_temp
                        improved = True
                        break
        
        if not improved and rng.random() < 0.15:
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

    inits.append(g / np.sum(g))
    
    # Triangular
    t = np.minimum(x, n - 1 - x) + 0.5
    inits.append(t / np.sum(t))
    
    # Cosine bump
    c = np.cos(np.pi * x / n) ** 2 + 0.01
    inits.append(c / np.sum(c))
    
    # Sine-modulated uniform
    freq = rng.uniform(1, 3)
    s = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x / n)
    inits.append(s / np.sum(s))
    
    # Two-hump
    h = np.exp(-0.5 * ((x - n*0.25) / (n*0.1)) ** 2) + np.exp(-0.5 * ((x - n*0.75) / (n*0.1)) ** 2) + 0.01
    inits.append(h / np.sum(h))

    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Power law
    p = (x + 1) ** 0.5
    inits.append(p / np.sum(p))
    
    return inits

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous:
1. Corrected and vectorized FFT-based gradient computation for soft-max smoothed objective
2. Nesterov momentum gradient descent with Armijo backtracking (much faster per-iteration)
3. Log-parameterization variant for better non-negativity handling
4. Peak-shaving local search targeting specific convolution peaks
5. Diverse structured initializations (Gaussian, triangular, multi-hump, etc.)
6. Aggressive multi-n exploration with resampling
7. LP used only for final polishing at smaller n values
8. Spectral equalization with adaptive compression
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_soft_gradient(a, tau):
    """
    Compute gradient of soft-max smoothed objective f(a) = 2n*tau*log(sum exp((a*a)[k]/tau)).
    Returns (gradient, convolution).
    grad[m] = 4n * sum_j w[j+m] * a[j]
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    # Soft-max weights
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum

    # Compute grad[m] = 4n * sum_j w[j+m] * a[j] via FFT
    # Using convolution: let a_pad = [a, 0,...,0] (length 2n-1), w_rev = reverse(weights)
    # Then sum_j a[j]*w[j+m] = (a_pad * w_rev)[2n-2-m]
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    w_rev = weights[::-1]

    size = 1
    while size < 4 * n - 3:
        size <<= 1
    A = rfft(a_pad, size)
    W = rfft(w_rev, size)
    conv_result = irfft(A * W, size)
    grad_raw = conv_result[n - 2:2 * n - 1][::-1]
    grad = 4.0 * n * grad_raw

    return grad, conv


def gradient_descent_momentum(a, tau_init=0.003, max_steps=500, time_budget=60.0,
                              step_init=0.1, momentum=0.9, verbose=False):
    """
    Nesterov momentum gradient descent with Armijo backtracking.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    tau = tau_init
    step = step_init
    v = np.zeros(n)  # momentum velocity

    for step_i in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov lookahead
        a_look = a - momentum * v
        a_look = np.maximum(a_look, 0.0)
        s_look = float(np.sum(a_look))
        if s_look > 1e-10:
            a_look = a_look / s_look
        grad, conv = compute_soft_gradient(a_look, tau)

        # Center gradient (preserve sum)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-12:
            break

        # Update velocity
        v = momentum * v - step * grad_centered

        # Trial step with Armijo backtracking
        obj_current = compute_obj(a)
        alpha = 1.0
        improved = False
        for backtrack in range(15):
            a_try = a + alpha * v
            a_try = np.maximum(a_try, 0.0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)

            if obj_try < obj_current - 1e-4 * alpha * np.dot(v, grad_centered):
                improved = True
                break
            alpha *= 0.5

        if improved:
            a = a_try
            obj_new = compute_obj(a)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
            # Increase step slightly if we used full alpha
            if alpha > 0.9:
                step = min(step * 1.05, 0.3)
        else:
            v = np.zeros(n)
            step *= 0.6
            if step < 1e-12:
                break

        # Anneal temperature
        tau *= 0.997
        # Decay momentum slightly
        momentum = max(0.5, momentum * 0.999)

    return best_a


def gradient_descent_log_param(a, tau_init=0.003, max_steps=300, time_budget=50.0):
    """
    Gradient descent in log-parameterization: a[j] = exp(x[j]) / sum(exp(x)).
    Naturally handles non-negativity.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    a_safe = np.maximum(a, 1e-12)
    x = np.log(a_safe)
    x -= np.max(x)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    tau = tau_init
    step = 0.5

    for step_i in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        # Compute a from x
        x_shifted = x - np.max(x)
        exp_x = np.exp(x_shifted)
        a = exp_x / np.sum(exp_x)

        # Gradient with respect to a
        grad_a, conv = compute_soft_gradient(a, tau)

        # Transform to x-space: grad_x[m] = a[m] * (grad_a[m] - sum(grad_a * a))
        dot = float(np.sum(grad_a * a))
        grad_x = a * (grad_a - dot)

        # Armijo backtracking in x-space
        obj_current = compute_obj(a)
        alpha = step
        improved = False
        for backtrack in range(15):
            x_try = x - alpha * grad_x
            x_try_shifted = x_try - np.max(x_try)
            exp_try = np.exp(x_try_shifted)
            a_try = exp_try / np.sum(exp_try)
            obj_try = compute_obj(a_try)

            if obj_try < obj_current - 1e-4 * alpha * float(np.sum(grad_x * grad_x)):
                improved = True
                break
            alpha *= 0.5

        if improved:
            x = x_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
            if alpha > 0.9 * step:
                step = min(step * 1.1, 2.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break

        tau *= 0.997

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def spectral_equalization(a, max_iters=40, time_budget=30.0):
    """POCS-style spectral equalization with adaptive compression."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        alpha = 0.3 + 0.5 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = P_norm ** alpha * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.6 * a + 0.4 * a_new)

    return best_a


def peak_shaving(a, max_iters=300, time_budget=30.0):
    """
    Direct peak-shaving: identify elements contributing most to the convolution peak

        contrib.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contrib[:10]:
            if time.monotonic() - start > time_budget:

                    a_try = a.copy()
                    delta = frac * a_try[j1]
                    a_try[j1] -= delta
                    a_try[j_target] += delta
                    a_try = np.maximum(a_try, 0.0)
                    s = float(np.sum(a_try))
                    if s < 1e-10:

                    a_try = a_try / s

                    if obj_try < best_obj - 1e-12:

                        best_a = a_try.copy()

                        improved = True
                        break

        if not improved:
            if rng.random() < 0.2:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:
                # Try moving mass from peak to low-convolution region

                    obj_try = compute_obj(a_try)

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([np.sum(a)])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def diverse_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian centered
    sigma = n / 4.0
    g = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    inits.append(g / np.sum(g))

    # Triangular
    t = np.minimum(x, n - 1 - x) + 0.5
    inits.append(t / np.sum(t))

    # Cosine bump
    c = np.cos(np.pi * x / n) ** 2 + 0.01
    inits.append(c / np.sum(c))

    # Sine-modulated
    freq = rng.uniform(1.0, 3.0)
    s = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x / n)
    inits.append(s / np.sum(s))

    # Two-hump
    h = (np.exp(-0.5 * ((x - n * 0.25) / (n * 0.12)) ** 2) +
         np.exp(-0.5 * ((x - n * 0.75) / (n * 0.12)) ** 2) + 0.01)
    inits.append(h / np.sum(h))

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Power law symmetric
    d = np.abs(x - n / 2) + 1
    p = d ** (-0.3)
    inits.append(p / np.sum(p))

    return inits

    # Gradient after spectral

        a_new = gradient_descent_momentum(best_seq, tau_init=0.0005,
                                           max_steps=300, time_budget=35.0)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Post-spectral gradient: {best_score:.8f} **")

    # =========================================================
    # Phase 4: LP optimization
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: LP optimization")
    print(f"{'='*60}")

    n_best = len(best_seq)

    if n_best <= 900:
        for lp_cycle in range(3):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.15, 45.0)

            a_lp = lp_alternate(best_seq, max_iters=20, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP cycle {lp_cycle}: {best_score:.8f} **")
            else:
                print(f"  LP cycle {lp_cycle}: {obj_lp:.8f}")

    # =========================================================
    # Phase 5: Diverse random restarts
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Random restarts with diverse strategies")
    print(f"{'='*60}")

    n_cur = len(best_seq)

    for restart in range(12):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.08, 25.0)

        strategy = restart % 6
        if strategy == 0:
            a_try = best_seq * np.exp(rng.normal(0, 0.05, n_cur))
        elif strategy == 1:
            freq = rng.uniform(0.5, 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.15 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
            a_try = best_seq * mod
        elif strategy == 2:
            x = np.arange(n_cur)
            center = rng.uniform(0.2, 0.8) * n_cur
            width = rng.uniform(0.08, 0.2) * n_cur
            g = np.exp(-0.5 * ((x - center) / width) ** 2) + 0.01
            mix = rng.uniform(0.2, 0.4)
            a_try = (1 - mix) * normalize(best_seq) + mix * g / np.sum(g)
        elif strategy == 3:
            a_try = best_seq.copy()
            block_len = int(rng.integers(n_cur // 5, max(2, n_cur // 3)))
            start_idx = int(rng.integers(0, max(1, n_cur - block_len)))
            a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]
        elif strategy == 4:
            shift = int(rng.integers(1, max(2, n_cur // 5)))
            a_try = np.roll(best_seq, shift)
        else:
            # Two-scale perturbation
            coarse = rng.normal(1.0, 0.03, n_cur)
            fine = np.exp(2j * np.pi * rng.uniform(2, 8) * np.arange(n_cur) / n_cur)
            a_try = best_seq * coarse * (1.0 + 0.05 * np.real(fine))

        a_try = normalize(a_try)
        a_try = gradient_descent_momentum(a_try, tau_init=0.003,
                                           max_steps=200, time_budget=alloc)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  Restart {restart} (strat={strategy}): {best_score:.8f} **")
        else:
            print(f"  Restart {restart} (strat={strategy}): {obj_try:.8f}")

    # =========================================================
    # Phase 6: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    for polish_round in range(6):
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break
        alloc = min(remaining * 0.20, 35.0)

        # Gradient with very low temperature
        a_pol = gradient_descent_momentum(best_seq, tau_init=0.0003,
                                           max_steps=200, time_budget=alloc * 0.4)

        # Peak shaving
        if alloc > 12:
            a_pol = peak_shaving(a_pol, max_iters=200, time_budget=alloc * 0.3)

        # LP if small enough
        if n_best <= 700 and alloc > 20:
            a_pol = lp_alternate(a_pol, max_iters=8, time_budget=alloc * 0.25)

        # Final gradient
        if alloc > 8:
            a_pol = gradient_descent_momentum(a_pol, tau_init=0.0002,
                                               max_steps=100, time_budget=min(10, alloc * 0.15))

        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Polish {polish_round}: {best_score:.8f} **")
        else:
            print(f"  Polish {polish_round}: {obj_pol:.8f}")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
