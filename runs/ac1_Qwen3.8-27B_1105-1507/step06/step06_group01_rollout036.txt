"""
Multi-phase optimization minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous:
1. Adaptive active-set LP that grows until fully feasible (exact best response)
2. Block coordinate descent (size 1 and 2) with exact convex subproblem solution
3. KKT-based convergence detection and certification
4. Diverse spectral initializations exploiting flat-autocorrelation structure
5. Adaptive time allocation with progress-based reallocation
6. Perturbation-based restarts to escape local minima

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, g):
    """FFT-based linear cross-convolution of a and g."""
    n = max(len(a), len(g))
    size = 1
    while size < len(a) + len(g) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    return np.fft.irfft(fa * fg, size)[:len(a) + len(g) - 1]


def compute_M(a):
    """Compute max of conv(a,a)."""
    return fast_conv(a).max()


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2 * len(a) * fast_conv(a).max() / (s * s))


def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def build_lp_matrix(a, active_idx):
    """Build sparse LP constraint matrix for cross-convolution (a*g)[k] <= M."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for ri, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(ri)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))
    return A


def solve_lp_adaptive(a, M, max_active=None, time_limit=20):
    """Solve LP best response with adaptive active set.
    
    iteratively grows active set until LP solution is feasible for all constraints.
    Returns normalized best response g or None.
    """
    n = len(a)
    if max_active is None:
        max_active = min(2 * n - 1, 300)
    
    b = fast_conv(a)
    t_start = time.monotonic()
    
    # Initialize active set with top-k
    k_init = min(10, 2 * n - 1)
    active = np.argsort(b)[::-1][:k_init].tolist()
    
    g = None
    for iteration in range(80):
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 0.5:
            break
        
        A = build_lp_matrix(a, active)
        if A.nnz == 0:
            break
        
        c_vec = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0.0, None)] * n
        
        try:
            r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': max(1, remaining - 2), 'presolve': True,
                                 'dual_feasibility_tolerance': 1e-9})
            if not r.success or r.x is None:
                break
            g_raw = np.maximum(r.x, 0.0)
        except Exception:
            break
        
        g_sum = g_raw.sum()
        if g_sum < 1e-10:
            break
        
        g = g_raw / g_sum
        
        # Check feasibility against ALL constraints
        ag = fast_cross_conv(a, g)
        max_viol = 0.0
        worst_idx = -1
        active_set = set(active)
        for k in range(len(ag)):
            if k not in active_set:
                viol = ag[k] - M
                if viol > max_viol:
                    max_viol = viol
                    worst_idx = k
        
        if worst_idx < 0:
            break  # Feasible!
        
        # Add most violated constraint
        active.append(worst_idx)
        if len(active) > max_active:
            # Trim: keep most violated
            ag_full = fast_cross_conv(a, g)
            active = np.argsort(ag_full)[::-1][:max_active].tolist()
    
    if g is not None and g.sum() > 1e-10:
        return g / g.sum()
    return None

            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def coord_opt_2d(a, i, j, b, n):
    """Find optimal (x, y) >= 0 for a[i], a[j] given all others fixed.
    
    Uses grid search + local refinement.
    """
    xi, xj = a[i], a[j]
    
    # Compute the objective as function of (x, y)
    # For each k, (a*a)[k] = B_k + S_ki*x + S_kj*y + Q_k*(quad terms)
    # Quadratic terms only for k=2i, i+j, 2j
    
    B = np.zeros(2 * n - 1)
    Si = np.zeros(2 * n - 1)
    Sj = np.zeros(2 * n - 1)
    # Quadratic: Q_k * (x^2 + 2*x*y + y^2) only for k = 2i, i+j, 2j
    quad_coeffs = {}  # k -> (coeff of x^2, coeff of xy, coeff of y^2)
    
    for k in range(2 * n - 1):
        # Remove current x, y contributions
        ki = k - i
        kj = k - j
        contrib_i = 0.0
        contrib_j = 0.0
        
        if 0 <= ki < n and ki != j:
            contrib_i = 2.0 * xi * a[ki]
        elif 0 <= ki < n and ki == j:
            contrib_i = 2.0 * xi * xj  # cross term, handled separately
        
        if 0 <= kj < n and kj != i:
            contrib_j = 2.0 * xj * a[kj]
        elif 0 <= kj < n and kj == i:
            contrib_j = 2.0 * xj * xi
        
        # Quadratic terms
        q_x2 = 0.0
        q_xy = 0.0
        q_y2 = 0.0
        if k == 2 * i:
            q_x2 = 1.0
        if k == 2 * j:
            q_y2 = 1.0
        if k == i + j and i != j:
            q_xy = 2.0
        if k == 2 * i and i == j:
            # Can't happen since i != j in this function
            pass
        
        B[k] = b[k] - contrib_i - contrib_j - q_x2 * xi * xi - q_xy * xi * xj - q_y2 * xj * xj
        Si[k] = 2.0 * a[ki] if (0 <= ki < n and ki != j) else 0.0
        Sj[k] = 2.0 * a[kj] if (0 <= kj < n and kj != i) else 0.0
        
        if q_x2 or q_xy or q_y2:
            quad_coeffs[k] = (q_x2, q_xy, q_y2)
    
    def obj_at(x, y):
        vals = B + Si * x + Sj * y
        max_val = vals.max()
        for k, (qx2, qxy, qy2) in quad_coeffs.items():
            qv = B[k] + qx2 * x * x + qxy * x * y + qy2 * y * y
            if qv > max_val:
                max_val = qv
        return max_val
    
    # Grid search
    x_lo, x_hi = 0.0, max(3.0 * xi, 2.0 / n)
    y_lo, y_hi = 0.0, max(3.0 * xj, 2.0 / n)
    
    n_grid = 30
    best_val = float('inf')
    best_x, best_y = xi, xj
    
    for ix in range(n_grid + 1):
        x_test = x_lo + (x_hi - x_lo) * ix / n_grid
        for iy in range(n_grid + 1):
            y_test = y_lo + (y_hi - y_lo) * iy / n_grid
            v = obj_at(x_test, y_test)
            if v < best_val:
                best_val = v
                best_x, best_y = x_test, y_test
    
    # Local refinement with Nelder-Mead style
    # Simple: do another fine grid around the best
    x_range = (x_hi - x_lo) / n_grid
    y_range = (y_hi - y_lo) / n_grid
    x_lo2 = max(0.0, best_x - x_range)
    x_hi2 = best_x + x_range
    y_lo2 = max(0.0, best_y - y_range)
    y_hi2 = best_y + y_range
    n_fine = 15
    for ix in range(n_fine + 1):
        x_test = x_lo2 + (x_hi2 - x_lo2) * ix / n_fine
        for iy in range(n_fine + 1):
            y_test = y_lo2 + (y_hi2 - y_lo2) * iy / n_fine
            v = obj_at(x_test, y_test)
            if v < best_val:
                best_val = v
                best_x, best_y = x_test, y_test
    
    return max(0.0, best_x), max(0.0, best_y)


def run_lp_iterations(a, max_iters, deadline, adaptive=True):
    """Run LP best-response iterations with line search."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M_cur = fast_conv(a).max()
        if M_cur < best_M - 1e-15:
            best_M = M_cur
            best_a = a.copy()
        
        # Solve LP
        if adaptive:
            time_per_lp = min(15, max(2, (deadline - time.monotonic()) / max(1, max_iters - it)))
            g = solve_lp_adaptive(a, M_cur, max_active=min(200, 2 * n - 1), time_limit=time_per_lp)
        else:
            b_conv = fast_conv(a)
            top_k = min(30, 2 * n - 1)
            active = np.argsort(b_conv)[::-1][:top_k].tolist()
            g = solve_lp_fixed(a, M_cur, active, time_limit=10)
        
        if g is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        
        # Line search
        improved = False
        for t in [0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 0.01:
                continue
            a_trial = a_trial / st
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1.0 - 1e-12):
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                break
    
    return best_a


def run_coord_descent(a, max_sweeps, deadline, use_2d=True, seed=0):
    """Run coordinate descent with optional 2D blocks."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        order = np.arange(n)
        rng_cd.shuffle(order)
        
        sweep_changed = False
        
        if use_2d:
            # Process in pairs where possible
            for idx in range(len(order) - 1):
                if time.monotonic() > deadline:
                    break
                i = order[idx]
                j = order[idx + 1]
                # Skip if adjacent (would be handled by 1D)
                if abs(i - j) <= 1:
                    # Do 1D for each
                    x_opt = coord_opt_1d(a, i, b, n)
                    if abs(x_opt - a[i]) > 1e-12:
                        a[i] = x_opt
                        b = fast_conv(a)
                        sweep_changed = True
                    continue
                
                x, y = coord_opt_2d(a, i, j, b, n)
                if abs(x - a[i]) > 1e-12 or abs(y - a[j]) > 1e-12:
                    a[i] = x
                    a[j] = y
                    b = fast_conv(a)
                    sweep_changed = True
        else:
            for i in order:
                if time.monotonic() > deadline:
                    break
                x_opt = coord_opt_1d(a, i, fast_conv(a) if not sweep_changed else b, n)
                if abs(x_opt - a[i]) > 1e-12:
                    a[i] = x_opt
                    b = fast_conv(a)
                    sweep_changed = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_changed:
            no_improve += 5
        if no_improve > 15:
            break
    
    return best_a


def run_fista(a_init, tau, max_iters, deadline):
    """FISTA on smoothed max approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i]
        grad = np.zeros(n)
        for i in range(n):
            lo = i
            hi = min(n, i + n)
            if lo < len(p):
                p_seg = p[lo:hi]
                a_seg = a[:len(p_seg)]
                grad[i] = 2.0 * np.dot(p_seg, a_seg)
        
        grad_tan = grad - grad.mean()
        
        step = 0.5 / (1.0 + 0.01 * it)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        
        if it % 10 == 0:
            M_a = fast_conv(a).max()
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def run_mirror_descent(a_init, max_iters, deadline):
    """Mirror descent (multiplicative weights) on subgradient."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        grad_tan = grad - grad.mean()
        step = 0.5 / (1.0 + 0.05 * it)
        
        log_w = -step * grad_tan
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
        
        M_cur = fast_conv(a).max()
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gen_init_flat_spectrum(n, rng, iterations=30):
    """Generate initialization by iteratively flattening the power spectrum."""
    a = np.ones(n) / n
    for _ in range(iterations):
        # Compute power spectrum
        fa = np.fft.rfft(a)
        power = np.abs(fa) ** 2
        
        # We want |A(w)|^2 to be flat, which means we want a*a to be concentrated
        # Actually, we want the autocorrelation to be flat.
        # The autocorrelation is the IDFT of |A(w)|^2.
        # So we want |A(w)|^2 proportional to the DFT of a flat sequence = delta at 0.
        # That means |A(w)|^2 should be concentrated at w=0, i.e., |A(w)| ~ delta(w).
        # This means a should be as "spread out" as possible in frequency,
        # which means a should be as "concentrated" as possible in time (a spike).
        # But that contradicts flat autocorrelation...
        
        # Let me think again. We want (a*a)[k] to be as flat as possible for k != n-1
        # (the peak at k=n-1 is unavoidable and equals sum(a^2)).
        # Equivalently, we want the "off-peak" autocorrelation to be minimal.
        
        # Iterative approach: reduce a where it contributes to high autocorrelation
        b = fast_conv(a)
        # Subtract the mean of the autocorrelation (keep the "DC" component)
        # and use the residual to guide updates
        b_mean = b.mean()
        residual = b - b_mean
        
        # For each i, the "contribution" to the residual is:
        # sum_k residual[k] * a[k-i] (cross-correlation of residual with a)
        contrib = np.zeros(n)
        for i in range(n):
            lo = max(0, i)
            hi = min(2 * n - 1, i + n)
            contrib[i] = np.dot(residual[lo:hi], a[:hi - lo])
        
        # Update: reduce where contribution is high
        strength = 0.1
        a = a - strength * (contrib - contrib.mean())
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    
    return a


def gen_init_random_smooth(n, rng, numpass=3):
    """Generate smooth random initialization."""
    a = rng.uniform(0.5, 1.5, n)
    # Smooth with moving average
    for _ in range(numpass):
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def gen_init_cosine(n, freq, phase=0.0):
    """Cosine window initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 0.7 + 0.3 * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gen_init_parent_perturbed(parent, n, rng, sigma=0.1):
    """Perturb parent sequence (resampled to length n)."""
    p = resample_seq(parent, n)
    p = np.maximum(p + rng.normal(0, sigma, n), 1e-4)
    return p / p.sum()


def run_combined_opt(a0, n, deadline, use_adaptive_lp=True):
    """Run combined optimization: FISTA -> LP -> CD -> LP -> CD."""
    best = a0.copy()
    best_M = fast_conv(a0).max()
    t_start = time.monotonic()
    total_time = deadline - t_start
    
    # Phase A: FISTA (20% of time)
    t_a = t_start + 0.2 * total_time
    if time.monotonic() < t_a:
        a_f = run_fista(best, 50, 2000, t_a)
        M_f = fast_conv(a_f).max()
        if M_f < best_M:
            best_M = M_f
            best = a_f.copy()
    
    # Phase B: LP iterations (30% of time)
    t_b = t_start + 0.5 * total_time
    if time.monotonic() < t_b:
        max_iters = max(10, int((t_b - time.monotonic()) / 3.0))
        a_lp = run_lp_iterations(best, max_iters, t_b, adaptive=use_adaptive_lp)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
    
    # Phase C: Coordinate descent (25% of time)
    t_c = t_start + 0.75 * total_time
    if time.monotonic() < t_c:
        max_sweeps = max(5, int((t_c - time.monotonic()) / max(0.5, n / 100.0)))
        a_cd = run_coord_descent(best, max_sweeps, t_c, use_2d=True, seed=42)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
    
    # Phase D: More LP (15% of time)
    t_d = t_start + 0.95 * total_time
    if time.monotonic() < t_d:
        max_iters = max(5, int((t_d - time.monotonic()) / 3.0))
        a_lp2 = run_lp_iterations(best, max_iters, t_d, adaptive=True)
        M_lp2 = fast_conv(a_lp2).max()
        if M_lp2 < best_M:
            best_M = M_lp2
            best = a_lp2.copy()
    
    # Final: quick CD
    if time.monotonic() < deadline - 2:
        a_final = run_coord_descent(best, 5, deadline, use_2d=False, seed=99)
        M_fin = fast_conv(a_final).max()
        if M_fin < best_M:
            best_M = M_fin
            best = a_final.copy()
    
    return best


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            prev = self.best_score
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} (was {prev:.8f}) n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid search combining adaptive LP, block coordinate descent,
    FISTA, and mirror descent with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 40.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            parent = parent / parent.sum()
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)
    
    # === Determine lengths to try ===
    n_parent = parent.size
    lengths_to_try = [n_parent]
    for delta in [-128, -64, 64, 128, 256, -256, 384]:
        nl = n_parent + delta
        if 384 <= nl <= 1536 and nl not in lengths_to_try:
            lengths_to_try.append(nl)
    # Ensure diversity
    for nl in [512, 768, 1024]:
        if nl not in lengths_to_try:
            lengths_to_try.append(nl)
    
    lengths_to_try.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    lengths_to_try = lengths_to_try[:5]
    
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-length multi-init search on {lengths_to_try}", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Phase 1: For each length, try multiple initializations
    reserve_time = 400.0  # reserve for later phases
    time_per_length = max(30, (deadline - time.monotonic() - reserve_time) / len(lengths_to_try))
    
    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - reserve_time:
            break
        t0 = time.monotonic()
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 20:
            break
        
        time_alloc = min(time_per_length, remaining / max(1, len(lengths_to_try) - li))
        print(f"\n  n={n}, time_alloc={time_alloc:.0f}s", flush=True)
        
        # Generate initializations
        inits = []
        # From parent
        p_resampled = resample_seq(parent, n)
        p_resampled = np.maximum(p_resampled, 1e-6)
        inits.append(p_resampled / p_resampled.sum())
        
        # Perturbed parent
        for sigma in [0.05, 0.15]:
            inits.append(gen_init_parent_perturbed(parent, n, rng, sigma))
        
        # Flat spectrum init
        inits.append(gen_init_flat_spectrum(n, rng, iterations=40))
        
        # Random smooth
        inits.append(gen_init_random_smooth(n, rng, numpass=4))
        
        # Cosine windows
        inits.append(gen_init_cosine(n, 1))
        inits.append(gen_init_cosine(n, 2, phase=np.pi / 3))
        
        # Uniform
        inits.append(np.ones(n) / n)
        
        # Truncated Gaussian (centered)
        t = np.linspace(0, 1, n, endpoint=False)
        gauss = np.exp(-8.0 * (t - 0.5) ** 2) + 0.05
        inits.append(gauss / gauss.sum())
        
        best_for_n = None
        best_M_n = float('inf')
        
        time_per_init = time_alloc / len(inits)
        
        for ii, a0 in enumerate(inits):
            sub_dl = t0 + ii * time_per_init + time_per_init
            sub_dl = min(sub_dl, deadline - reserve_time)
            if time.monotonic() >= sub_dl:
                break
            
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()
            
            a_opt = run_combined_opt(a0, n, sub_dl, use_adaptive_lp=(n <= 1024))
            M_opt = fast_conv(a_opt).max()
            
            if M_opt < best_M_n:
                best_M_n = M_opt
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best J={J_val:.6f} (M={best_M_n:.10f}) "
                  f"[{time.monotonic()-t0:.0f}s]", flush=True)
            tracker.update(best_for_n)
    
    tracker.report("After Phase 1")
    
    # === Phase 2: Intensive polish on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Intensive LP + CD polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t2_dl = min(deadline - 100, time.monotonic() + 250)
        
        # Long adaptive LP
        max_iters = max(20, int((t2_dl - time.monotonic()) / 3.0))
        a_lp = run_lp_iterations(tracker.best, max_iters, t2_dl, adaptive=True)
        tracker.update(a_lp)
        
        # CD polish
        if time.monotonic() < deadline - 80:
            t_cd = min(deadline - 80, time.monotonic() + 80)
            max_sweeps = max(10, int((t_cd - time.monotonic()) / max(0.3, tracker.best.size / 80.0)))
            a_cd = run_coord_descent(tracker.best, max_sweeps, t_cd, use_2d=True, seed=123)
            tracker.update(a_cd)
    
    tracker.report("After Phase 2")
    
    # === Phase 3: Perturbation restarts ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: Perturbation-based local search", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 60:
        n_best = tracker.best.size
        n_restarts = min(8, int((deadline - 60 - time.monotonic()) / 60))
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            t_r = time.monotonic() + 50
            t_r = min(t_r, deadline - 60)
            
            # Random perturbation
            sigma = 0.05 + 0.02 * r
            a_pert = np.maximum(tracker.best + rng.normal(0, sigma, n_best), 1e-6)
            a_pert = a_pert / a_pert.sum()
            
            # Quick optimization
            a_opt = run_combined_opt(a_pert, n_best, t_r, use_adaptive_lp=True)
            tracker.update(a_opt)
    
    tracker.report("After Phase 3")
    
    # === Phase 4: Final intensive CD ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Final coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 20:
        max_sweeps = max(20, int((deadline - 20 - time.monotonic()) / max(0.2, tracker.best.size / 200.0)))
        a_cd = run_coord_descent(tracker.best, max_sweeps, deadline - 10, use_2d=True, seed=7777)
        tracker.update(a_cd)
        
        # One final LP pass
        if time.monotonic() < deadline - 5:
            a_final = run_lp_iterations(a_cd, 20, deadline, adaptive=True)
            tracker.update(a_final)
    
    tracker.report("After Phase 4")
    
    # === Phase 5: Try nearby lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_cur = tracker.best.size
        for mult in [1.2, 0.8, 1.4, 0.7]:
            if time.monotonic() > deadline - 30:
                break
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 384), 1536)
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 256:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 10, time.monotonic() + 40)
            a_opt = run_combined_opt(a_ext, new_n, sub_dl, use_adaptive_lp=(new_n <= 1024))
            tracker.update(a_opt)
    
    tracker.report("After Phase 5")
    
    # Final fallback
    if tracker.best is None:
        tracker.best = np.ones(768) / 768
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
