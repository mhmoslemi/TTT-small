"""
Hybrid LP + Adam optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses peak-shaving LP, original LP, log-space Adam with annealing,
multi-n exploration, and block coordinate refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # a[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # with coefficient a[k-j]
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_peak_shaving(a, time_limit=30):
    """
    LP: min t s.t. (a*g)[k] <= t for all k, sum(g) = 1, g >= 0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(a)
    if n > 2000:
        return None, None
    m = 2 * n - 1
    A_conv = build_conv_matrix(a)
    # Variables: [g_0, ..., g_{n-1}, t]
    # Constraints: A_conv @ g - t <= 0  =>  [A_conv, -1] @ [g,t] <= 0
    A_ub = np.hstack([A_conv, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    # Equality: sum(g) = 1
    A_eq = np.hstack([np.ones(n), [[0.0]]])
    b_eq = [1.0]
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            t_opt = res.x[-1]
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
            return g, t_opt
    except Exception:
        pass
    return None, None

    n = len(a)
    if n > 2000:
        return None
    m = 2 * n - 1
    A_conv = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_conv, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_tight_bounds(a, time_limit=30):
    """
    LP: max sum(g) s.t. (a*g)[k] <= conv(a,a)[k] for all k, g >= 0.
    This is tighter than the original LP since it uses actual values as bounds.
    Returns normalized g or None.
    """
    n = len(a)
    if n > 2000:
        return None
    m = 2 * n - 1
    A_conv = build_conv_matrix(a)
    conv_a = compute_conv(a)
    c = -np.ones(n)
    b_ub = conv_a.copy()
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_conv, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def smooth_obj_grad(a, beta):
    """
    Compute smoothed objective and gradient.
    Uses logsumexp of beta*conv as smooth approximation of max.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf, np.zeros(n)

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    # Gradient of soft_max w.r.t. a[i]
    # d(smax)/d(a[i]) = 2 * sum_j w[j] * a[j] where j+i ranges over valid indices
    # This is 2 * (w cross-correlate a)[i]
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def adam_log_opt(a, max_iters=5000, time_budget=60.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start

    best_a = a.copy()
    best_true_obj = compute_obj(a)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)

        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a

        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 200 == 0:
            x -= np.mean(x)

        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

    return best_a


def lp_refine_iterative(a, time_budget=60.0, n_iters=20, blend_start=0.3, blend_end=1.0):
    """
    Iteratively apply the peak-shaving LP and blend.
    Tries both peak-shaving LP and original LP.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    s = np.sum(best_a)
    if s > 1e-10:
        best_a = best_a / s

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        t_step = blend_start + (blend_end - blend_start) * (it / max(n_iters - 1, 1))
        t_step = min(t_step, 1.0)

        # Try peak-shaving LP
        g, t_opt = lp_peak_shaving(best_a, time_limit=min(remaining * 0.3, 15))
        if g is not None:
            a_try = (1 - t_step) * best_a + t_step * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()

        # Try original LP
        conv = compute_conv(best_a)
        M = float(np.max(conv))
        g2 = lp_original(best_a, M, time_limit=min(remaining * 0.3, 15))
        if g2 is not None:
            a_try2 = (1 - t_step) * best_a + t_step * g2
            a_try2 = np.maximum(a_try2, 0)
            s_try2 = np.sum(a_try2)
            if s_try2 > 1e-10:
                a_try2 = a_try2 / s_try2
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj:
                    best_obj = obj_try2
                    best_a = a_try2.copy()

        # Try tight-bounds LP
        g3 = lp_tight_bounds(best_a, time_limit=min(remaining * 0.2, 10))
        if g3 is not None:
            a_try3 = (1 - t_step) * best_a + t_step * g3
            a_try3 = np.maximum(a_try3, 0)
            s_try3 = np.sum(a_try3)
            if s_try3 > 1e-10:
                a_try3 = a_try3 / s_try3
                obj_try3 = compute_obj(a_try3)
                if obj_try3 < best_obj:
                    best_obj = obj_try3
                    best_a = a_try3.copy()

    return best_a


def block_refine(a, time_budget=30.0, block_size=100):
    """
    Block coordinate descent: optimize blocks of the sequence using LP.
    """
    n = len(a)
    n_blocks = max(1, n // block_size)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for sweep in range(10):
        if time.monotonic() - start > time_budget:
            break
        # Pick the worst-performing block (contributes most to peak)
        conv = compute_conv(best_a)
        peak_idx = int(np.argmax(conv))
        # Elements contributing to this peak
        i_min = max(0, peak_idx - (n - 1))
        i_max = min(peak_idx, n - 1)
        # Find which block contains the center of contributing elements
        center_i = (i_min + i_max) // 2
        block_idx = min(center_i // block_size, n_blocks - 1)

        b_start = block_idx * block_size
        b_end = min(b_start + block_size, n)

        # Optimization: fix all elements except the block, optimize the block
        # Using a local LP or gradient step
        # Simple approach: perturb the block slightly and keep if improved
        a_try = best_a.copy()
        # Try scaling the block
        block_mean = np.mean(a_try[b_start:b_end])
        for scale in [0.8, 0.9, 1.1, 1.2]:
            a_try = best_a.copy()
            a_try[b_start:b_end] *= scale
            s_try = np.sum(a_try)
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining LP refinement, Adam optimization, multi-n exploration,
    and block coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        s = np.sum(best_seq)
        if s > 1e-10:
            best_seq = best_seq / s

    if best_seq is None:
        # Fallback initialization
        n_init = 700
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)

    n_current = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n={n_current}, budget={budget_s}s")

    # === Phase 1: LP-based refinement on parent length ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP refinement (n={n_current})")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 120.0)
        a_lp = lp_refine_iterative(best_seq, time_budget=alloc, n_iters=30,
                                    blend_start=0.2, blend_end=0.8)
        obj_lp = compute_obj(a_lp)
        print(f"  LP iterative: {obj_lp:.8f}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  *** New best: {best_score:.8f}")

    # === Phase 2: Adam optimization on current best ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Adam log-space optimization (n={len(best_seq)})")
    print(f"{'='*60}")

    rounds = [
        (5.0, 50.0, 0.05, 0.99998),
        (10.0, 100.0, 0.04, 0.99998),
        (20.0, 200.0, 0.03, 0.99999),
        (40.0, 400.0, 0.02, 0.99999),
        (80.0, 600.0, 0.015, 0.99999),
        (150.0, 800.0, 0.01, 0.99999),
        (300.0, 1200.0, 0.005, 0.99999),
    ]

    for r_idx, (b_start, b_end, lr0, lr_dec) in enumerate(rounds):
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * 0.1, 60.0)
        a_new = adam_log_opt(best_seq, max_iters=20000, time_budget=alloc,
                              beta_start=b_start, beta_end=b_end,
                              lr_start=lr0, lr_decay=lr_dec)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Round {r_idx} (beta {b_start:.0f}-{b_end:.0f}): {best_score:.8f} ***")
        else:
            print(f"  Round {r_idx} (beta {b_start:.0f}-{b_end:.0f}): {obj:.8f}")

    # === Phase 3: Second LP round after Adam ===
    print(f"\n{'='*60}")
    print(f"Phase 3: LP re-refinement (n={len(best_seq)})")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.12, 60.0)
        a_lp2 = lp_refine_iterative(best_seq, time_budget=alloc, n_iters=20,
                                     blend_start=0.1, blend_end=0.6)
        obj_lp2 = compute_obj(a_lp2)
        print(f"  LP re-refine: {obj_lp2:.8f}")
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  *** New best: {best_score:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq)
    n_options = sorted(set([
        max(300, n_best - 100),
        n_best,
        n_best + 100,
        n_best + 200,
        n_best + 400,
        min(2000, n_best + 600),
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2500]

    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        s = np.sum(a_try)
        if s > 1e-10:
            a_try = a_try / s

        # Adam optimization
        alloc_adam = min(remaining * 0.12, 50.0)
        a_opt = adam_log_opt(a_try, max_iters=15000, time_budget=alloc_adam,
                              beta_start=10.0, beta_end=300.0,
                              lr_start=0.03, lr_decay=0.99998)

        # LP refinement
        if time.monotonic() < deadline - 30:
            alloc_lp = min((deadline - time.monotonic()) * 0.08, 25.0)
            a_opt = lp_refine_iterative(a_opt, time_budget=alloc_lp, n_iters=10,
                                         blend_start=0.2, blend_end=0.7)

        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  n={n_try}: {obj:.8f}")

    # === Phase 5: Alternative initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Alternative initializations")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 60:
        n_target = len(best_seq)
        rng = np.random.default_rng(seed)

        inits = []
        # Edge-heavy (V-shape)
        x = np.arange(n_target)
        v_shape = 1.0 + 0.5 * np.abs(2.0 * x / (n_target - 1) - 1.0)
        inits.append(('V-shape', v_shape / np.sum(v_shape)))
        # U-shape (more extreme)
        u_shape = 1.0 + 1.0 * (2.0 * x / (n_target - 1) - 1.0) ** 2
        inits.append(('U-shape', u_shape / np.sum(u_shape)))
        # Smooth bump (Gaussian-like in middle)
        center = (n_target - 1) / 2.0
        gauss = np.exp(-(x - center) ** 2 / (2 * (n_target / 4) ** 2))
        inits.append(('Gauss', gauss / np.sum(gauss)))
        # Blend with parent
        parent_norm = best_seq / np.sum(best_seq)
        blend = 0.5 * parent_norm + 0.5 * (v_shape / np.sum(v_shape))
        inits.append(('Blend', blend))
        # Random smooth
        rand_smooth = np.abs(np.cumsum(rng.normal(0, 1, n_target))) + 0.1
        inits.append(('Random', rand_smooth / np.sum(rand_smooth)))

        for name, a_init in inits:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            alloc = min(remaining * 0.12, 40.0)
            a_opt = adam_log_opt(a_init, max_iters=10000, time_budget=alloc,
                                  beta_start=5.0, beta_end=200.0,
                                  lr_start=0.04, lr_decay=0.99998)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  {name}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  {name}: {obj:.8f}")

    # === Phase 6: Intensive final optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final intensive optimization (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None:
        # High-beta Adam for precise peak targeting
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.25, 60.0)
            a_final = adam_log_opt(best_seq, max_iters=20000, time_budget=alloc,
                                    beta_start=200.0, beta_end=2000.0,
                                    lr_start=0.005, lr_decay=0.99999)
            obj_final = compute_obj(a_final)
            print(f"  High-beta Adam: {obj_final:.8f}")
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.8f}")

        # LP final pass
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.3, 30.0)
            a_final2 = lp_refine_iterative(best_seq, time_budget=alloc, n_iters=10,
                                            blend_start=0.05, blend_end=0.3)
            obj_final2 = compute_obj(a_final2)
            print(f"  Final LP: {obj_final2:.8f}")
            if obj_final2 < best_score:
                best_score = obj_final2
                best_seq = a_final2.copy()
                print(f"  *** New best: {best_score:.8f}")

        # One more Adam after LP
        if time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.5, 15.0)
            a_final3 = adam_log_opt(best_seq, max_iters=10000, time_budget=alloc,
                                     beta_start=500.0, beta_end=3000.0,
                                     lr_start=0.003, lr_decay=0.99999)
            obj_final3 = compute_obj(a_final3)
            print(f"  Final Adam: {obj_final3:.8f}")
            if obj_final3 < best_score:
                best_score = obj_final3
                best_seq = a_final3.copy()
                print(f"  *** New best: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
