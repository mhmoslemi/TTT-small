def spectral_shape(a, time_budget=30.0):
    """
    Shape the sequence in the frequency domain to flatten the power spectrum,
    which reduces the peak of the autocorrelation.
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Work in log-magnitude spectrum
    # a = IDFT(A), where |A| is the magnitude spectrum
    # We want to make |A|^2 as flat as possible (water-filling)
    # But we need a >= 0, so this is constrained.
    
    # Simple approach: iteratively project
    # 1. Compute FFT of a
    # 2. Modify magnitude to be flatter
    # 3. Take IFFT, project to non-negative
    # 4. Repeat
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        A = rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Water-filling: reduce high-magnitude bins, increase low-magnitude bins
        mag_mean = np.mean(mag)
        # Compress the magnitude
        alpha = 0.95  # compression factor
        mag_new = mag_mean + alpha * (mag - mag_mean)
        mag_new = np.maximum(mag_new, 1e-10)
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = irfft(A_new, size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            # Try different alpha
            for alpha_try in [0.8, 0.7, 0.5]:
                mag_new2 = mag_mean + alpha_try * (mag - mag_mean)
                mag_new2 = np.maximum(mag_new2, 1e-10)
                A_new2 = mag_new2 * np.exp(1j * phase)
                a_new2 = irfft(A_new2, size)[:n]
                a_new2 = np.maximum(a_new2, 0)
                s_new2 = np.sum(a_new2)
                if s_new2 < 1e-10:
                    continue
                a_new2 = a_new2 / s_new2
                obj_new2 = compute_obj(a_new2)
                if obj_new2 < best_obj:
                    best_obj = obj_new2
                    best_a = a_new2.copy()
                    a = a_new2.copy()
                    break
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def coordinate_refine(a, time_budget=20.0, seed=0):
    n = len(a)
    start = time.monotonic()

        for p in peaks:

                    a_try[j] += delta
                    # Also try the reverse

                        a_try[i] += delta

                        break
        if not improved and it > 50:
            break

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines subgradient methods exploiting the exact structure of the max,
Adam log-space optimization, adaptive LP refinement, peak-driven coordinate
descent, spectral shaping, and multi-scale exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """Smoothed logsumexp objective and gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def subgradient_step(a, lr, momentum_vec=None, momentum=0.9):
    """
    One subgradient step for the exact (non-smoothed) objective.
    The subgradient of max_k C[k] w.r.t. a[i] is 2*a[k*-i] where k* = argmax C.
    Projected onto the sum=0 constraint and positivity.
    """
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    # Find all positions achieving the max (for subgradient averaging)
    peak_positions = np.where(conv >= max_b - 1e-15)[0]
    
    # Subgradient: average over all peak positions
    grad = np.zeros(n)
    for k in peak_positions:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(peak_positions)
    
    # Project onto sum(a)=1 constraint: feasible directions have sum(d)=0
    grad_proj = grad - np.mean(grad)
    
    if momentum_vec is not None:
        v = momentum * momentum_vec - lr * grad_proj
    else:
        v = -lr * grad_proj
    
    return v, grad_proj


def subgradient_method(a, time_budget=60.0, max_iters=5000):
    """
    Subgradient method with adaptive learning rate and momentum.
    Directly minimizes the non-smooth objective.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1 / np.sqrt(n)
    momentum_vec = np.zeros(n)
    momentum = 0.85
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        v, grad_proj = subgradient_step(a, lr, momentum_vec, momentum)
        momentum_vec = v
        
        # Try multiple step sizes (line search)
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:

                break
        else:

            # Occasionally increase lr slightly

def multiplicative_update(a, time_budget=40.0, max_iters=3000):
    """
    Multiplicative weight update specialized for this problem.
    Updates: a[i] *= exp(-lr * (a[k*-i] - mean))
    Naturally maintains positivity.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a

def adam_log_opt(a, max_iters=5000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995):
    """Adam in log-space with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    x = np.log(a)
    x -= np.mean(x)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)
        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a
        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        if it % 200 == 0:
            x -= np.mean(x)
        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def peak_driven_search(a, time_budget=40.0, seed=0):
    """
    Directly target convolution peaks by identifying contributing elements
    and redistributing mass to less impactful positions.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy().astype(np.float64)
    s = np.sum(best_a)
    if s < 1e-10:
        return best_a
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(seed)
    
    for it in range(300):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(best_a)
        max_b = float(np.max(conv))
        
        # Focus on top peaks
        top_k = np.argsort(conv)[::-1][:3]
        
        improved = False
        for k_star in top_k:
            if time.monotonic() - start > time_budget:
                break
            
            # Find elements contributing to C[k_star]
            i_min = max(0, k_star - (n - 1))
            i_max = min(k_star, n - 1)
            if i_min > i_max:
                continue
            
            # Compute contribution of each element
            contributions = []
            for i in range(i_min, i_max + 1):
                j = k_star - i
                if 0 <= j < n:
                    contrib = best_a[i] * best_a[j]
                    # Leverage: how much peak decreases per unit decrease
                    leverage = best_a[j] if i != j else 2.0 * best_a[i]
                    contributions.append((contrib, leverage, i, j))
            
            if not contributions:
                continue
            
            # Sort by contribution (largest first)
            contributions.sort(reverse=True)
            
            for contrib, leverage, i, j in contributions[:3]:
                if time.monotonic() - start > time_budget:
                    break
                if leverage < 1e-10:
                    continue
                
                # Find best place to add mass: minimize impact on all peaks
                # Approximation: position where the "total pressure" is lowest
                total_pressure = np.zeros(n)
                for k_p in top_k:
                    for ii in range(n):
                        jj = k_p - ii
                        if 0 <= jj < n:
                            total_pressure[ii] += best_a[jj]
                
                # Find position with lowest pressure (excluding i)
                pressure_no_i = total_pressure.copy()
                pressure_no_i[i] = np.inf
                j_add = int(np.argmin(pressure_no_i))
                
                if j_add == i:
                    continue
                
                # Try moving mass
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    delta = frac * best_a[i]
                    if delta <= 0:
                        continue
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j_add] += delta
                    if a_try[i] < 0:
                        continue
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break

            a_pert = np.maximum(a_pert, 1e-12)

            obj_pert = compute_obj(a_pert)

def spectral_shape(a, time_budget=30.0):

    size = 1
    while size < 2 * n:
        size <<= 1

        if time.monotonic() - start > time_budget:
            break
        
        A = rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        best_alpha = 0.95
        best_a_iter = a
        best_obj_iter = best_obj
        
        for alpha in [0.95, 0.9, 0.8, 0.7, 0.5, 0.3]:

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

    best_seq = None
    best_score = np.inf
    
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            best_seq = parent.copy()
            best_score = obj
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(700)
        best_score = compute_obj(best_seq)
    
    print(f"Starting: best={best_score:.8f}, n={len(best_seq)}, budget={budget_s}s")
    
    a = best_seq.copy()
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    
    # === Phase 1: Subgradient method (exploits exact structure) ===
    print(f"\n{'='*50}")
    print(f"Phase 1: Subgradient method")
    print(f"{'='*50}")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.15, 70.0)
    a_sub = subgradient_method(a, time_budget=alloc, max_iters=10000)
    obj = compute_obj(a_sub)
    if obj < best_score:
        best_score = obj
        best_seq = a_sub.copy()
        a = a_sub.copy()
        print(f"  Subgradient: {best_score:.8f} ***")
    else:
        print(f"  Subgradient: {obj:.8f} (no imp)")
    
    # === Phase 2: Multiplicative update ===
    print(f"\n{'='*50}")
    print(f"Phase 2: Multiplicative update")
    print(f"{'='*50}")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.1, 40.0)
    a_mu = multiplicative_update(a, time_budget=alloc, max_iters=5000)
    obj = compute_obj(a_mu)
    if obj < best_score:
        best_score = obj
        best_seq = a_mu.copy()
        a = a_mu.copy()
        print(f"  Multiplicative: {best_score:.8f} ***")
    else:
        print(f"  Multiplicative: {obj:.8f} (no imp)")
    
    # === Phase 3: Adam log-space (coarse to fine) ===
    print(f"\n{'='*50}")
    print(f"Phase 3: Adam log-space optimization")
    print(f"{'='*50}")
    rounds = [
        (5.0, 50.0, 0.05, 0.99998),
        (20.0, 150.0, 0.03, 0.99998),
        (50.0, 300.0, 0.02, 0.99999),
        (100.0, 600.0, 0.01, 0.99999),
        (200.0, 1000.0, 0.005, 0.99999),
    ]
    for r_idx, (b_start, b_end, lr0, lr_dec) in enumerate(rounds):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * 0.1, 50.0)
        a_new = adam_log_opt(a, max_iters=20000, time_budget=alloc,
                             beta_start=b_start, beta_end=b_end,
                             lr_start=lr0, lr_decay=lr_dec)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Adam r{r_idx} (beta {b_start:.0f}-{b_end:.0f}): {best_score:.8f} ***")
        else:
            print(f"  Adam r{r_idx} (beta {b_start:.0f}-{b_end:.0f}): {obj:.8f} (no imp)")
    
    # === Phase 4: Subgradient again (fine-tuning after Adam) ===
    print(f"\n{'='*50}")
    print(f"Phase 4: Fine subgradient")
    print(f"{'='*50}")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.1, 40.0)
    a_sub2 = subgradient_method(a, time_budget=alloc, max_iters=10000)
    obj = compute_obj(a_sub2)
    if obj < best_score:
        best_score = obj
        best_seq = a_sub2.copy()
        a = a_sub2.copy()
        print(f"  Fine subgradient: {best_score:.8f} ***")
    else:
        print(f"  Fine subgradient: {obj:.8f} (no imp)")
    
    # === Phase 5: Peak-driven search ===
    print(f"\n{'='*50}")
    print(f"Phase 5: Peak-driven search")
    print(f"{'='*50}")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.1, 30.0)
    a_pds = peak_driven_search(a, time_budget=alloc, seed=seed)
    obj = compute_obj(a_pds)
    if obj < best_score:
        best_score = obj
        best_seq = a_pds.copy()
        a = a_pds.copy()
        print(f"  Peak-driven: {best_score:.8f} ***")
    else:
        print(f"  Peak-driven: {obj:.8f} (no imp)")
    
    # === Phase 6: LP refinement ===
    print(f"\n{'='*50}")
    print(f"Phase 6: LP refinement")
    print(f"{'='*50}")
    if len(best_seq) <= 900:
        for r_idx in range(4):
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.12, 25.0)
            a_lp = lp_refine_adaptive(a, time_budget=alloc)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                a = a_lp.copy()
                print(f"  LP {r_idx}: {best_score:.8f} ***")
            else:
                print(f"  LP {r_idx}: {obj:.8f} (no imp)")
    
    # === Phase 7: Spectral shaping ===
    print(f"\n{'='*50}")
    print(f"Phase 7: Spectral shaping")
    print(f"{'='*50}")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.08, 25.0)
    a_spec = spectral_shape(a, time_budget=alloc)
    obj = compute_obj(a_spec)
    if obj < best_score:
        best_score = obj
        best_seq = a_spec.copy()
        a = a_spec.copy()
        print(f"  Spectral: {best_score:.8f} ***")
    else:
        print(f"  Spectral: {obj:.8f} (no imp)")
    
    # === Phase 8: Multi-n exploration ===
    print(f"\n{'='*50}")
    print(f"Phase 8: Multi-n exploration")
    print(f"{'='*50}")
    n_best = len(best_seq)
    n_options = sorted(set([n_best - 50, n_best + 50, n_best + 100, n_best + 200]))
    n_options = [n for n in n_options if 200 <= n <= 1500]
    
    for n_try in n_options:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_try = resample_sequence(best_seq, n_try)
        s = np.sum(a_try)
        if s > 1e-10:
            a_try = a_try / s
        # Quick subgradient + Adam
        alloc_sub = min(remaining * 0.05, 10.0)
        a_sub_n = subgradient_method(a_try, time_budget=alloc_sub, max_iters=3000)
        alloc_adam = min(remaining * 0.05, 10.0)
        a_opt = adam_log_opt(a_sub_n, max_iters=5000, time_budget=alloc_adam,
                             beta_start=20.0, beta_end=200.0,
                             lr_start=0.03, lr_decay=0.99998)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  n={n_try}: {obj:.8f}")
    
    # === Phase 9: Final intensive polish ===
    print(f"\n{'='*50}")
    print(f"Phase 9: Final polish")
    print(f"{'='*50}")
    
    # Subgradient final
    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 15.0)
        a_fin = subgradient_method(a, time_budget=alloc, max_iters=5000)
        obj = compute_obj(a_fin)
        if obj < best_score:
            best_score = obj
            best_seq = a_fin.copy()
            a = a_fin.copy()
            print(f"  Final subgrad: {best_score:.8f} ***")
    
    # Pair swap
    if time.monotonic() < deadline - 12:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 12.0)
        a_ps = pair_swap_search(a, time_budget=alloc, seed=seed+1)
        obj = compute_obj(a_ps)
        if obj < best_score:
            best_score = obj
            best_seq = a_ps.copy()
            a = a_ps.copy()
            print(f"  Pair swap: {best_score:.8f} ***")
    
    # One more multiplicative
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 10.0)
        a_mu2 = multiplicative_update(a, time_budget=alloc, max_iters=3000)
        obj = compute_obj(a_mu2)
        if obj < best_score:
            best_score = obj
            best_seq = a_mu2.copy()
            a = a_mu2.copy()
            print(f"  Final multiplicative: {best_score:.8f} ***")
    
    # Final Adam with high beta
    if time.monotonic() < deadline - 8:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 10.0)
        a_fin2 = adam_log_opt(a, max_iters=5000, time_budget=alloc,
                              beta_start=300.0, beta_end=1500.0,
                              lr_start=0.003, lr_decay=0.99999)
        obj = compute_obj(a_fin2)
        if obj < best_score:
            best_score = obj
            best_seq = a_fin2.copy()
            print(f"  Final Adam high-beta: {best_score:.8f} ***")
    
    # === Final validation ===
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")
    
    return result
