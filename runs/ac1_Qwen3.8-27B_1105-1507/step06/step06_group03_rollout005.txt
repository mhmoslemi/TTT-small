Advanced optimizer for J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) fast SCA-LP with adaptive constraint selection,
(2) enhanced coordinate descent with candidate generation,
(3) smoothed gradient descent with dual-weighted steps,

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fft_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def solve_sca_lp(a, deadline, verbose=False):

    n = len(a)

    best_a = a.copy()

    no_improve = 0
    K = 40

    for it in range(50000):

            break

        c = fft_conv(a)
        M = c.max()
        K_use = min(K, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:K_use]

        rows = []
        cols = []
        vals = []
        b_ub = []

        for idx, k in enumerate(top_k):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(idx)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(K_use, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (list(range(n)), list(range(n)))), shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

def coord_descent_fast(a, deadline, seed=42, verbose=False):
    """Coordinate descent with smart candidate generation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fft_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(100000):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            kq = 2 * i
            has_kq = (kq <= 2 * n - 2)

            if lo > 0 or hi < 2 * n - 2:
                outer = np.ones(2 * n - 1, dtype=bool)
                outer[lo:hi + 1] = False
                outer_max = conv[outer].max() if outer.any() else 0.0
            else:
                outer_max = 0.0

            a_slice = a[lo - i:hi - i + 1]

            if old < 1e-8:
                candidates = np.array([0.0, 1e-4, 5e-4, 1e-3, 3e-3, 1e-2, 3e-2])
            else:
                candidates = np.array([0.0, old * 0.1, old * 0.3, old * 0.5, old * 0.7,
                                       old, old * 1.3, old * 1.7, old * 2.5])

            best_val = old
            best_f = 2.0 * n * max(outer_max, conv[lo:hi+1].max()) / (s * s)

            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                aff = conv[lo:hi + 1].copy() + 2.0 * delta * a_slice
                if has_kq:
                    aff[kq - lo] += delta * delta
                inner_max = aff.max()
                total_max = max(outer_max, inner_max)
                new_s = s - old + x
                if new_s < 1e-10:
                    f = float('inf')
                else:
                    f = 2.0 * n * total_max / (new_s * new_s)
                if f < best_f:
                    best_f = f
                    best_val = x

            if best_val != old and abs(best_val - old) > 1e-14:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if has_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        if verbose and sweep % 50 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def smooth_gradient_descent(a, deadline, verbose=False):
    """Smoothed gradient descent with log-sum-exp and adaptive tau."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0
    no_improve = 0
    tau = 50.0
    lr = 0.05

    for it in range(50000):
        if time.monotonic() > deadline:
            break

        c = fft_conv(a)
        c_shift = c - c.max()
        wc = np.exp(tau * c_shift)
        w = wc / wc.sum()

        # Gradient via FFT
        a_rev = a[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        conv1 = np.fft.irfft(W_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx = n - 1 + i
            if 0 <= idx < len(conv1):
                grad[i] = conv1[idx]

        grad = grad - np.mean(grad)

        t_count += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            # Animate tau slower when improving
            tau *= 0.995
        else:
            no_improve += 1
            tau *= 1.02
            if no_improve > 80:
                # Restart
                a = best_a + np.random.default_rng(it).normal(0, 0.015, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0
                lr = 0.03
                tau = 30.0

        a = a_new
        lr *= 0.9995

        if verbose and it % 200 == 0:
            print(f"    Grad it {it}: tau={tau:.1f} lr={lr:.5f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def peak_shaving(a, deadline, verbose=False):
    """
    Targeted peak shaving: identify max conv positions and
    selectively reduce coordinates that contribute to those peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(20000):
        if time.monotonic() > deadline:
            break

        c = fft_conv(a)
        M = c.max()
        k_star = np.argmax(c)

        # Find coordinates that contribute most to conv at k_star
        # conv[k_star] = sum_j a_j * a_{k_star - j}
        # Gradient w.r.t. a_i: a_{k_star - i} (if valid) + a_{k_star - i} (if k_star-i != i)
        # More precisely: d(conv[k_star])/da_i = a[(k_star - i) % ... ]
        # In the convolution, a_i contributes to conv[k] for k in [i, i+n-1]
        # The contribution to conv[k_star] from a_i is:
        #   a_{k_star - i} if 0 <= k_star - i < n and k_star - i != i
        #   a_i (i.e., a_{k_star-i} = a_i) if k_star - i == i, i.e., k_star = 2i
        # So the "influence" of coordinate i on the peak is a[max(0, min(n-1, k_star-i))] approximately

        # Simply: reduce coordinates with high a[k_star - i]
        influence = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                influence[i] = a[j]
            # Also the other direction
            j2 = k_star - i + n  # wrap-around doesn't apply for linear conv
            # Actually for linear convolution, a_i contributes to conv[k] only if 0 <= k-i < n
            # So influence is just a[k_star - i] if 0 <= k_star - i < n, else 0

        # Reduce top contributors, increase bottom contributors
        order_red = np.argsort(influence)[::-1]
        order_inc = np.argsort(influence)

        nr = min(n // 4, 100)
        ni = min(n // 4, 100)

        total_red = 0.0
        for idx in range(nr):
            i = order_red[idx]
            reduce_amt = a[i] * 0.05
            a[i] -= reduce_amt
            total_red += reduce_amt

        # Distribute to low-influence coordinates
        for idx in range(ni):
            i = order_inc[idx]
            a[i] += total_red / ni

        a = np.maximum(a, 0)
        if a.sum() > 1e-10:
            a /= a.sum()

        obj = compute_J(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

    return best_a, best_obj

            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        # Quick SCA-LP
        a_opt, obj_opt = solve_sca_lp(a_ext, time.monotonic() + alloc * 0.6)
        update_best(a_opt)

        # Quick CD
        if time.monotonic() < phase3_end - 20:
            a_cd2 = coord_descent_fast(a_opt, time.monotonic() + alloc * 0.3, seed=99)
            update_best(a_cd2)

        print(f"  n={new_n}: best J so far={best_score:.8f}", flush=True)

    # === PHASE 4: Intensive local optimization on best ===
    print(f"\n=== PHASE 4: Intensive optimization on best (n={best_seq.size if best_seq is not None else '?'}) ===", flush=True)
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            # SCA-LP with more iterations
            t_sca = time.monotonic() + 0.35 * min(remaining, 400)
            print(f"  SCA-LP intensive ({t_sca - time.monotonic():.0f}s)...", flush=True)
            a_best_local = best_seq.copy()
            a_best_local = np.maximum(a_best_local, 1e-12)
            a_best_local = a_best_local / a_best_local.sum()
            a_opt, obj_opt = solve_sca_lp(a_best_local, t_sca)
            update_best(a_opt)

            remaining = deadline - time.monotonic()
            if remaining > 30:
                # CD
                t_cd = time.monotonic() + 0.25 * min(remaining, 200)
                print(f"  CD intensive ({t_cd - time.monotonic():.0f}s)...", flush=True)
                a_cd3 = coord_descent_fast(best_seq, t_cd, seed=777)
                update_best(a_cd3)

            remaining = deadline - time.monotonic()
            if remaining > 30:
                # Gradient descent
                t_grad = time.monotonic() + 0.2 * min(remaining, 150)
                print(f"  Grad descent ({t_grad - time.monotonic():.0f}s)...", flush=True)
                a_grad, obj_grad = smooth_gradient_descent(best_seq, t_grad)
                update_best(a_grad)

            remaining = deadline - time.monotonic()
            if remaining > 20:
                # Peak shaving
                t_ps = time.monotonic() + 0.1 * min(remaining, 100)
                print(f"  Peak shaving ({t_ps - time.monotonic():.0f}s)...", flush=True)
                a_ps, obj_ps = peak_shaving(best_seq, t_ps)
                update_best(a_ps)

            # Final SCA on the best
            remaining = deadline - time.monotonic()
            if remaining > 20:
                t_final = deadline - 5
                print(f"  Final SCA-LP ({t_final - time.monotonic():.0f}s)...", flush=True)
                a_fl = best_seq.copy()
                a_fl = np.maximum(a_fl, 1e-12)
                a_fl = a_fl / a_fl.sum()
                a_final, obj_final = solve_sca_lp(a_fl, t_final)
                update_best(a_final)

                # One last CD
                if time.monotonic() < deadline - 10:
                    a_last = coord_descent_fast(best_seq, deadline - 5, seed=1)
                    update_best(a_last)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
