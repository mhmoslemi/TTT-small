def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1, dtype=np.float64)

        # Part 1: sum_k w[k] * a[k-i] for i=0,...,n-1
        # = correlation of w and a = (w_rev * a)[n-1 + i]
        corr = np.fft.irfft(np.fft.rfft(w_rev, size2) * np.fft.rfft(a, size2), size2)
        part1 = corr[n-1 : n-1+n]
        
        # Part 2: sum_k w[k] * a[i-k] for i=0,...,n-1
        # = (w * a_rev)[3n-3-i] where a_rev = a[::-1]
        corr2 = np.fft.irfft(np.fft.rfft(w, size2) * np.fft.rfft(a_rev, size2), size2)
        part2 = corr2[3*n-3 : 3*n-3+n-1:-1]  # reversed
        
        # Actually let me be more careful:
        # Part 2: sum_k w[k] * a[i-k]
        # Let j = i-k, so k = i-j: sum_j w[i-j] * a[j] = (w_rev * a)[n-1 + i]... 
        # No wait: sum_j w[i-j]*a[j] is the cross-correlation (w ⋆ a)[i] = (w_rev * a)[n-1+i]
        # But w_rev[j] = w[n-1-j], so (w_rev * a)[n-1+i] = sum_j w_rev[j]*a[n-1+i-j] = sum_j w[n-1-j]*a[n-1+i-j]
        # Let m = n-1-j: sum_m w[m]*a[i+m-1+n-1-n+1]... this is getting confusing.
        
        # Let me just use a simpler but correct approach:
        # The gradient of (a*a)[k] w.r.t. a[i] is:
        #   a[k-i] (if 0 <= k-i < n) + a[i-k] (if 0 <= i-k < n)
        # Note: these are the same term when k-i = i-k, i.e., k = 2i.
        
        # For the weighted sum: grad[i] = sum_k w[k] * (a[max(0,k-i)] * (k-i<n) + a[max(0,i-k)] * (i-k<n))
        
        # I'll compute this using two correlations:
        # C1[i] = sum_k w[k] * a[k-i] (where a is zero-padded)
        # C2[i] = sum_k w[k] * a[i-k] (where a is zero-padded)
        
        # C1[i] = sum_{k=i}^{i+n-1} w[k] * a[k-i] (since a[k-i] is nonzero only for 0<=k-i<n)
        #       = sum_{j=0}^{n-1} w[i+j] * a[j]
        # This is a "sliding dot product"
        
        # C2[i] = sum_{k=0}^{i} w[k] * a[i-k] (since a[i-k] is nonzero only for 0<=i-k<n, i.e., k>=i-n+1)
        #       = sum_{j=0}^{n-1} w[i-j] * a[j] (where w is zero-padded)
        # This is also a "sliding dot product"
        
        # Both can be computed via FFT:
        # C1[i] = sum_j w[i+j]*a[j] = (w * a_pad)[i] where w is padded... 
        
        # OK let me just use a direct approach with FFT:
        # Pad a to length 2n-1 with zeros at the end
        # C1[i] = sum_j w[i+j]*a[j] for j=0..n-1, i=0..n-1
        # Let W_full = w padded to length 2n-1
        # C1 = ifft(fft(W_full) * fft(a_padded)) -- but this gives circular correlation
        
        # I'll use the standard trick:
        # C1[i] = (w_rev_full * a_pad)[(2n-2) - i] where appropriate padding
        
        # You know what, let me just do it the simple O(n^2) way for the gradient
        # since n ~ 768 and we only need it once per iteration.
        # Actually for n=768, O(n^2) = ~600K operations per iteration, which is fine.
        
        # Simple O(n * n_active) computation:
        # Only the active lags contribute significantly to the gradient
        active_lags = np.where(w > 1e-6)[0]
        grad = np.zeros(n)
        for k in active_lags:
            wk = w[k]
            if wk < 1e-8:
                continue
            for i in range(n):
                ki = k - i
                ik = i - k
                g = 0.0
                if 0 <= ki < n:
                    g += a[ki]
                if 0 <= ik < n:
                    g += a[ik]
                grad[i] += wk * g
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        # Nesterov update
        beta = 2.0 / (it + 5.0)
        v = beta * v + grad
        y_new = y - 0.01 * v
        a_new = project_simplex(y_new)
        
        obj_new = compute_J(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        y = a_new
    
    return best_a, best_obj

def truncate_and_optimize(a, deadline, min_length=128):
    """
    Try trimming near-zero values from the ends of the sequence.
    Each trim reduces n, which directly reduces J.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0)
    s = a.sum()
    if s < 1e-10:
        return a, compute_J(a)
    
    best_a = a.copy()
    best_obj = compute_J(a)
    
    # Try trimming from the end
    for trim in range(1, n // 4):
        if time.monotonic() > deadline:
            break
        a_trimmed = a[:-trim]
        s_trim = a_trimmed.sum()
        if s_trim < 0.01:
            continue
        obj = compute_J(a_trimmed)
        if obj < best_obj - 1e-8:
            best_a = a_trimmed.copy()
            best_obj = obj
        elif trim > 50:
            # If we've trimmed a lot without improvement, stop
            if trim > 20:
                break
    
    # Try trimming from the beginning
    a = best_a.copy()
    n = len(a)
    for trim in range(1, n // 4):
        if time.monotonic() > deadline:
            break
        a_trimmed = a[trim:]
        s_trim = a_trimmed.sum()
        if s_trim < 0.01:
            continue
        obj = compute_J(a_trimmed)
        if obj < best_obj - 1e-8:
            best_a = a_trimmed.copy()
            best_obj = obj
        elif trim > 50:
            if trim > 20:
                break
    
    # Also try trimming from both ends
    n = len(a)
    for t1 in range(0, min(20, n//4)):
        for t2 in range(0, min(20, n//4)):
            if time.monotonic() > deadline:
                break
            if t1 == 0 and t2 == 0:
                continue
            a_trimmed = a[t1:n-t2] if t2 > 0 else a[t1:]
            if len(a_trimmed) < min_length:
                continue
            s_trim = a_trimmed.sum()
            if s_trim < 0.01:
                continue
            obj = compute_J(a_trimmed)
            if obj < best_obj - 1e-8:
                best_a = a_trimmed.copy()
                best_obj = obj
    
    return best_a, best_obj

def smooth_adam_v2(a, deadline, max_iters=3000, tau=50.0, lr=0.03):
    """Improved smooth Adam with better gradient computation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()
        
        # Compute gradient using FFT
        a_rev = a[::-1]
        w_rev = w[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        
        # grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
        # Part 1: sum_k w[k] * a[k-i]
        # = (w * a_padded)[...] let me use the correlation formula
        # C1[i] = sum_k w[k] * a[k-i] = (w_conv_a)[n-1+i] where w_conv_a = ifft(fft(w_padded)*fft(a_padded))
        # Actually: (w * a)[m] = sum_j w[j] * a[m-j]
        # C1[i] = sum_k w[k]*a[k-i] = sum_j w[j+i]*a[j] (letting j=k-i)
        # This is the cross-correlation (w ⋆ a)[i]
        # = (flip(w) * a)[n-1+i] in linear convolution
        
        W_fft = np.fft.rfft(w, size)
        A_fft = np.fft.rfft(a, size)
        
        # (w * a)[m] = sum_j w[j]*a[m-j]
        conv_wa = np.fft.irfft(W_fft * A_fft, size)
        
        # C1[i] = sum_j w[j+i]*a[j] = (w * a)[n-1+i]... 
        # (w*a)[m] = sum_j w[j]*a[m-j]. For m = n-1+i:
        # sum_j w[j]*a[n-1+i-j]. Let k = n-1+i-j, so j = n-1+i-k:
        # sum_k w[n-1+i-k]*a[k]
        # We want: sum_k w[k]*a[k-i]. These are different.
        
        # Let me use: C1[i] = sum_k w[k]*a[k-i]
        # = (a * w_rev)[n-1+i] where w_rev[k] = w[n-1-k]
        # (a * w_rev)[m] = sum_j a[j]*w_rev[m-j] = sum_j a[j]*w[n-1-m+j]
        # For m = n-1+i: sum_j a[j]*w[n-1-(n-1-i)+j] = sum_j a[j]*w[i+j]
        # That's sum_j a[j]*w[i+j], which is what we want (C1[i] = sum_j a[j]*w[i+j])!
        
        Wr = w[::-1]
        Wr_fft = np.fft.rfft(Wr, size)
        conv_awa = np.fft.irfft(A_fft * Wr_fft, size)
        C1 = conv_awa[n-1 : n-1+n]
        
        # C2[i] = sum_k w[k]*a[i-k]
        # = (w * a_rev)[3n-3-i]... let me think
        # C2[i] = sum_k w[k]*a[i-k] (a[i-k] = 0 if i-k < 0 or i-k >= n)
        # Let j = i-k: sum_j w[i-j]*a[j] = (w_rev * a)[n-1+i]... 
        # (w_rev * a)[m] = sum_j w_rev[j]*a[m-j] = sum_j w[n-1-j]*a[m-j]
        # Let k = n-1-j: sum_k w[k]*a[m-n+1+k]
        # For m = n-1+i: sum_k w[k]*a[i+k]
        # That's C1[i+1]ish... not quite.
        
        # Alternative: C2[i] = sum_k w[k]*a[i-k]
        # = (a * w)[2i]... no
        # Let me just use: C2[i] = sum_j w[i-j]*a[j] (where w is zero outside [0,n-1])
        # This is the convolution (w * a) evaluated at... 
        # (w*a)[m] = sum_j w[j]*a[m-j]
        # C2[i] = sum_j w[i-j]*a[j] = sum_k w[k]*a[i-k] (letting k=i-j) 
        # = (w * a)[i]... wait: (w*a)[i] = sum_j w[j]*a[i-j]. Let k=i-j: = sum_k w[i-k]*a[k]. 
        # But we want sum_k w[k]*a[i-k], which is (w*a)[i] with w and a swapped = (a*w)[i] = (w*a)[i] (convolution is commutative).
        # So C2[i] = (w*a)[i]? No: sum_k w[k]*a[i-k] = (w*a)[i]. Yes!
        # And (w*a)[i] = conv_wa[i].
        
        # Wait, let me double-check:
        # (w*a)[i] = sum_j w[j]*a[i-j]
        # Let k = j: sum_k w[k]*a[i-k]. Yes! That's exactly C2[i].
        
        C2 = conv_wa[:n]  # (w*a)[0], (w*a)[1], ..., (w*a)[n-1]
        
        # But wait, C2[i] should be sum_k w[k]*a[i-k] where a[i-k] is 0 if i-k < 0 or i-k >= n.
        # (w*a)[i] = sum_j w[j]*a[i-j] where a is zero-padded. 
        # For i < n: a[i-j] is nonzero for 0 <= i-j < n, i.e., i-n+1 <= j <= i.
        # Since w[j] is nonzero for 0 <= j < n, the effective range is max(0, i-n+1) <= j <= min(i, n-1).
        # This is correct for the linear convolution.
        
        # Similarly for C1: C1[i] = sum_j a[j]*w[i+j] where w[i+j] is 0 if i+j >= n.
        # This requires i+j < n, i.e., j < n-i. So effectively j = 0 to min(n-1, n-1-i).
        # (a * w_rev)[n-1+i] = sum_j a[j]*w_rev[n-1+i-j] = sum_j a[j]*w[n-1-(n-1+i-j)] = sum_j a[j]*w[j-i]
        # Hmm that gives w[j-i], not w[i+j]. Let me redo.
        
        # w_rev[k] = w[n-1-k]
        # (a * w_rev)[m] = sum_j a[j]*w_rev[m-j] = sum_j a[j]*w[n-1-m+j]
        # For m = n-1-i: sum_j a[j]*w[n-1-(n-1-i)+j] = sum_j a[j]*w[i+j]
        # Yes! So C1[i] = (a * w_rev)[n-1-i]
        
        conv_awr = np.fft.irfft(A_fft * Wr_fft, size)  # same as conv_awa
        # C1[i] = conv_awr[n-1-i]
        C1_corrected = conv_awr[n-1 : n-1+n-1:-1]  # reversed
        
        # So: grad[i] = C1[i] + C2[i] = conv_awr[n-1-i] + conv_wa[i]
        grad = conv_awr[n-1 : n-1+n-1:-1] + conv_wa[:n]
        
        # Project gradient to simplex tangent space
        grad = grad - np.mean(grad)
        
        # Adam update
        t_count += 1
        lr_t = lr / (1.0 + 0.05 * np.sqrt(t_count))
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t_count)
        v_hat = v / (1 - beta2**t_count)
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Restart
                a = best_a + np.random.default_rng(it).normal(0, 0.01, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0
        
        a = a_new
    
    return best_a, best_obj
