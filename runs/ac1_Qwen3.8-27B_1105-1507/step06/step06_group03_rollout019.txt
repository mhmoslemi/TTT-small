"""
Optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. SCA-LP with efficient sparse LP solving
2. Fast coordinate descent with incremental updates
3. Subgradient mirror descent with exact max
4. Multi-length exploration and diverse initializations
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import cvxpy as cp


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def sca_lp(a, max_iters, deadline, verbose=False):
    """
    SCA-LP: minimize t s.t. 2*(a*x)[k] - t <= (a*a)[k], sum(x)=1, x>=0.
    Uses scipy linprog with HiGHS.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    
    # Precompute sparsity pattern
    rows_list = []
    cols_list = []
    vals_template = []  # template for a-dependent values
    
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_template.append(j)  # store j to look up a[j] later
        rows_list.append(k)
        cols_list.append(n)  # t variable
        vals_template.append(-1)  # coefficient for t
    
    rows_arr = np.array(rows_list)
    cols_arr = np.array(cols_list)
    
    # Equality constraint: sum(x) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = c.max()
        
        # Build right-hand side and coefficients
        vals = np.empty(len(rows_arr))
        # For each non-t entry, value is 2*a[j]
        # For t entries, value is -1
        a_deps = vals_template  # j indices or -1 for t
        
        # Vectorized: for j != -1, val = 2*a[j]; for j == -1, val = -1
        is_t = (a_deps == -1)
        vals[is_t] = -1.0
        vals[~is_t] = 2.0 * a[a_deps[~is_t]]
        
        A_ub = coo_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n + 1)).tocsr()
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t
        b_ub = c  # (a*a)[k]
        bounds = [(0, None)] * (n + 1)
        
        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 5, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    
                    # Line search with bisection
                    best_frac = 0
                    best_J = best_obj
                    lo, hi = 0.0, 1.0
                    for _ in range(15):
                        mid = (lo + hi) / 2
                        a_try = (1 - mid) * a + mid * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            hi = mid
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_J:
                            best_J = J_try
                            best_frac = mid
                            lo = mid
                        else:
                            hi = mid
                    
                    if best_frac > 1e-6:
                        a = (1 - best_frac) * a + best_frac * g
                        a = np.maximum(a, 0)
                        a /= a.sum()
                        best_obj = best_J
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1
        
        if verbose and it % 10 == 0:
            print(f"    SCA it {it}: obj={best_obj:.10f}", flush=True)
        
        if no_improve > 30:
            break
    
    return best_a, best_obj


def coord_descent_fast(a, deadline, max_sweeps=50000, random_order=True, seed=42):
    """
    Fast coordinate descent with incremental convolution updates.
    Tries multiple candidate values per coordinate.
    """
    n = len(a)

"""
Optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. SCA-LP with efficient sparse LP solving and bisection line search
2. Fast coordinate descent with incremental convolution updates
3. Subgradient mirror descent with exact active-set max
4. Multi-length exploration and diverse initializations
5. Periodic restarts to escape local minima
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def sca_lp(a, max_iters, deadline, verbose=False):
    """
    SCA-LP: minimize t s.t. 2*(a*x)[k] - t <= (a*a)[k], sum(x)=1, x>=0.
    Uses scipy linprog with HiGHS and bisection line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    # Precompute sparsity pattern (rows, cols, and which a-index each entry depends on)
    rows_list = []
    cols_list = []
    a_idx_list = []  # index into a, or -1 for t variable

    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            a_idx_list.append(j)
        rows_list.append(k)
        cols_list.append(n)  # t variable at index n
        a_idx_list.append(-1)

    rows_arr = np.array(rows_list, dtype=np.int32)
    cols_arr = np.array(cols_list, dtype=np.int32)
    a_idx_arr = np.array(a_idx_list, dtype=np.int32)

    # Equality constraint
    A_eq_data = np.ones(n + 1)
    A_eq_data[n] = 0.0
    A_eq = coo_matrix((A_eq_data, (np.zeros(n + 1, dtype=np.int32), np.arange(n + 1))),
                      shape=(1, n + 1)).tocsr()
    b_eq = np.array([1.0])

                        best_J = J_full

                            a_try = (1.0 - mid) * a + mid * g
                            a_try = np.maximum(a_try, 0)
                            if a_try.sum() < 1e-10:
                                hi_frac = mid
                                continue
                            a_try /= a_try.sum()
                            J_try = compute_J(a_try)
                            if J_try < best_J - 1e-14:
                                best_J = J_try
                                best_frac = mid
                                lo_frac = mid
                            else:
                                hi_frac = mid

                    if best_frac > 1e-8:
                        a = (1.0 - best_frac) * a + best_frac * g
                        a = np.maximum(a, 0)
                        a /= a.sum()

        except Exception:
            no_improve += 1

        if it > 0 and it % 200 == 0:

            a_pert = best_a + rng.normal(0, 0.02, n)
            a_pert = project_simplex(a_pert)
            J_pert = compute_J(a_pert)
            if J_pert < best_obj:
                a = a_pert
                best_a = a_pert.copy()
                best_obj = J_pert
            else:
                a = best_a.copy()

                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 1000:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def smooth_adam(a, max_iters, deadline, tau_init=30.0, tau_max=300.0,
                lr_init=0.05, seed=42):
    """
    Adam optimization with log-sum-exp smoothing of max(conv(a,a)).
    Works on the simplex using a rebalancing trick.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_count = 0
    no_improve = 0
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_init + (tau_max - tau_init) * frac

        conv = fast_conv(a)
        c_shifted = conv - conv.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()

        # Compute gradient using FFT
        # grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i in [0,n-1])
        # This is a cross-correlation of w with a
        a_rev = a[::-1]
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        W_fft = np.fft.rfft(w, size)
        A_fft = np.fft.rfft(a_rev, size)
        corr = np.fft.irfft(W_fft * A_fft, size)

        # corr[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-(m-k)]
        # We want: sum_k w[k] * a[k-i]
        # Let j = k-i, then k = j+i, sum_j w[j+i] * a[j]
        # This is the cross-correlation at lag i
        # corr[i] = sum_k w[k] * a_rev[i-k] ... let me just compute directly for simplicity
        # Actually let me use a different approach:
        # grad[i] = 2 * (w * a)[i + n - 1] where (w*a) is full convolution
        # (w * a)[k] = sum_j w[j] * a[k-j]
        # We want sum_k w[k] * a[k-i] = (a * w)[i] ... hmm

        # Let's just do it with FFT properly
        # g[i] = sum_{k=0}^{2n-2} w[k] * (a[k-i] if 0<=k-i<n else 0)
        # = (a_reversed * w)[n-1+i] approximately
        # where a_reversed[m] = a[n-1-m]
        # (a_rev * w)[m] = sum_j a_rev[j] * w[m-j] = sum_j a[n-1-j] * w[m-j]
        # Let i = n-1-j, j = n-1-i:
        # = sum_i a[i] * w[m-(n-1-i)] = sum_i a[i] * w[m-n+1+i]
        # For m = n-1+i0: = sum_i a[i] * w[i0+i]
        # That's what we want for grad[i0] / 2!

        w_fft = np.fft.rfft(w, size)
        arev_fft = np.fft.rfft(a_rev, size)
        conv_warev = np.fft.irfft(w_fft * arev_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx = n - 1 + i
            if 0 <= idx < len(conv_warev):
                grad[i] = 2.0 * conv_warev[idx]

        # Project to simplex tangent (make sum zero)
        grad = grad - np.mean(grad)

        # Adam update
        t_count += 1
        lr = lr_init / (1.0 + 0.05 * np.sqrt(t_count))
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                # Restart with perturbation
                a = best_a + rng.normal(0, 0.03, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

        if verbose and it % 100 == 0:
            print(f"    Adam it {it}: tau={tau:.1f} obj={best_obj:.10f}", flush=True)

    return best_a, best_obj


def subgradient_method(a, max_iters, deadline, verbose=False):
    """
    Subgradient method using exact active set of max(conv).
    Uses projected subgradient with step size decay.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    no_improve = 0
    step = 0.5
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        max_val = conv.max()

        # Use all active coordinates (within tolerance of max)
        active = np.where(conv >= max_val * 0.999)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Subgradient: g[i] = 2n * (1/|active|) * sum_{k in active} (a[k-i]+a[i-k])/2
        # Simplified: for each active k, the gradient of (a*a)[k] w.r.t. a[i] is:
        # a[k-i] if 0<=k-i<n, plus a[i-k] if 0<=i-k<n (but these are the same term counted twice
        # only when k-i==i-k, i.e., k=2i... actually no)
        # (a*a)[k] = sum_j a[j]*a[k-j]
        # d/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n
        # Wait: d/d(a[i]) sum_j a[j]*a[k-j]
        # = sum_j [d(a[j])/d(a[i])*a[k-j] + a[j]*d(a[k-j])/d(a[i])]
        # = a[k-i] + a[k-i]  (first term: j=i; second term: k-j=i => j=k-i)
        # = 2*a[k-i] if 0 <= k-i < n

        g = np.zeros(n)
        for k in active:
            j = k
            # For each i, d(conv[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n
            lo_i = max(0, k - n + 1)
            hi_i = min(n - 1, k)
            for i in range(lo_i, hi_i + 1):
                g[i] += 2.0 * a[k - i]
        g /= len(active)
        g *= 2.0 * n  # scale for J = 2n*max(conv)

        # Projected subgradient step
        a_new = a - step * g
        a_new = project_simplex(a_new)

        obj_new = 2.0 * n * fast_conv(a_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            a = best_a.copy()

        # Step size schedule
        step = max(0.001, 0.5 / np.sqrt(it + 1))

        # Restart
        if no_improve > 100:
            a = best_a + rng.normal(0, 0.02, n)
            a = project_simplex(a)
            step = 0.3
            no_improve = 0

        if no_improve > 500:
            break

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine windows (various)
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        w = np.cos(np.pi * t ** alpha) ** 2 + 0.1
        inits.append(project_simplex(w))

    # Sinusoidal perturbations of uniform
    for freq in [1, 2, 3, 4, 5]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            w = 1.0 + amp * np.cos(2 * np.pi * freq * t)
            inits.append(project_simplex(np.maximum(w, 0.01)))

    # Gaussian bumps
    for center in [0.25, 0.5, 0.75]:
        for width in [0.1, 0.15, 0.2, 0.3]:
            w = np.exp(-((t - center) / width) ** 2) + 0.05
            inits.append(project_simplex(w))

    # Power laws
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        w = (t + 0.01) ** alpha + 0.01
        inits.append(project_simplex(w))
        w2 = ((1 - t) + 0.01) ** alpha + 0.01
        inits.append(project_simplex(w2))

    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.03))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses SCA-LP, coordinate descent, smooth Adam, subgradient, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)
    print(f"Starting optimization. Parent n={n_parent}, J={best_score:.8f}", flush=True)

    # === PHASE 1: SCA-LP on parent (most important, 35% of time) ===
    p1_deadline = t_start + 0.35 * budget_s
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: SCA-LP on n={n_parent} (deadline {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    a_curr = parent.copy()
    a_curr = np.maximum(a_curr, 1e-10)
    a_curr /= a_curr.sum()

    a_sca, obj_sca = sca_lp(a_curr, max_iters=5000, deadline=p1_deadline, verbose=True)
    update_best(a_sca)
    print(f"  SCA-LP done: J={obj_sca:.10f}", flush=True)

    # === PHASE 2: Coordinate descent polish (15% of time) ===
    p2_deadline = t_start + 0.50 * budget_s
    remaining = p2_deadline - time.monotonic()
    if remaining > 20:
        print(f"\n{'=' * 60}", flush=True)
        print(f"PHASE 2: Coord Descent (budget {remaining:.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        a_cd = coord_descent(best_seq.copy(), deadline=p2_deadline, seed=42)
        update_best(a_cd)
        print(f"  CD done: J={compute_J(a_cd):.10f}", flush=True)

    # === PHASE 3: Smooth Adam (15% of time) ===
    p3_deadline = t_start + 0.65 * budget_s
    remaining = p3_deadline - time.monotonic()
    if remaining > 20:
        print(f"\n{'=' * 60}", flush=True)
        print(f"PHASE 3: Smooth Adam (budget {remaining:.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        a_adam, obj_adam = smooth_adam(best_seq.copy(), max_iters=20000,
                                       deadline=p3_deadline, tau_init=20.0,
                                       tau_max=200.0, lr_init=0.03)
        update_best(a_adam)
        print(f"  Adam done: J={obj_adam:.10f}", flush=True)

    # === PHASE 4: More SCA from best (15% of time) ===
    p4_deadline = t_start + 0.80 * budget_s
    remaining = p4_deadline - time.monotonic()
    if remaining > 20:
        print(f"\n{'=' * 60}", flush=True)
        print(f"PHASE 4: SCA-LP round 2 (budget {remaining:.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        a_sca2, obj_sca2 = sca_lp(best_seq.copy(), max_iters=5000, deadline=p4_deadline, verbose=True)
        update_best(a_sca2)
        print(f"  SCA-LP 2 done: J={obj_sca2:.10f}", flush=True)

        # Quick CD polish
        if time.monotonic() < deadline - 30:
            a_cd2 = coord_descent(best_seq.copy(), deadline=min(p4_deadline, time.monotonic() + 60), seed=77)
            update_best(a_cd2)

    # === PHASE 5: Try different lengths (10% of time) ===
    p5_deadline = t_start + 0.90 * budget_s
    remaining = p5_deadline - time.monotonic()
    if remaining > 30:
        print(f"\n{'=' * 60}", flush=True)
        print(f"PHASE 5: Multi-length exploration (budget {remaining:.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        # Try lengths near and far from parent
        candidate_ns = sorted(set([
            max(64, n_parent // 2),
            n_parent,
            n_parent + n_parent // 4,
            n_parent * 2,
            min(n_parent * 2, 2048),
        ]))
        candidate_ns = [n for n in candidate_ns if n != n_parent and 64 <= n <= 4096]

        for new_n in candidate_ns[:3]:
            if time.monotonic() > p5_deadline - 30:
                break
            alloc = min(60.0, p5_deadline - time.monotonic() - 10)
            if alloc < 30:
                break

            print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
            a_ext = resample_seq(best_seq, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext /= a_ext.sum()

            # Quick SCA
            a_opt, obj_opt = sca_lp(a_ext, max_iters=2000, deadline=time.monotonic() + alloc * 0.7)
            update_best(a_opt)

            # Quick CD
            if time.monotonic() < time.monotonic() + alloc * 0.3:
                a_cd3 = coord_descent(a_opt, deadline=time.monotonic() + alloc * 0.3, seed=new_n)
                update_best(a_cd3)

            print(f"  n={new_n}: best J so far={best_score:.10f}", flush=True)

    # === PHASE 6: Final intensive polish (10% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n{'=' * 60}", flush=True)
        print(f"PHASE 6: Final polish (budget {remaining:.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        n_best = len(best_seq)

        # Round 1: SCA
        t1 = time.monotonic() + 0.3 * remaining
        a_f1, obj_f1 = sca_lp(best_seq.copy(), max_iters=3000, deadline=t1, verbose=True)
        update_best(a_f1)

        # Round 2: CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t2 = time.monotonic() + 0.3 * remaining
            a_f2 = coord_descent(best_seq.copy(), deadline=t2, seed=999)
            update_best(a_f2)

        # Round 3: SCA again
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_f3, obj_f3 = sca_lp(best_seq.copy(), max_iters=2000, deadline=deadline - 5, verbose=True)
            update_best(a_f3)

        # Final CD
        if time.monotonic() < deadline - 5:
            a_f4 = coord_descent(best_seq.copy(), deadline=deadline - 2, seed=1234)
            update_best(a_f4)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
