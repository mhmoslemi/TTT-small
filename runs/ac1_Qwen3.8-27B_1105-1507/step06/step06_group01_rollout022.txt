```python
def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s
```

```python
def smooth_max_grad_fft(a, tau):
    """Compute log-sum-exp smooth max of conv(a,a) and its gradient using FFT.
    
    Returns (smooth_value, gradient) where gradient has same length as a.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    # Numerically stable log-sum-exp
    shifted = tau * (b - b_max)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    log_sum_exp = max_shifted + np.log(exp_shifted.sum())
    smooth_val = b_max + log_sum_exp / tau
    
    # Weights
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient: grad_i = 2 * sum_j p[i+j] * a[j] for j=0..n-1
```

```python
conv_pa = np.fft.irfft(np.fft.rfft(p, size) * np.fft.rfft(a_rev, size), size)
grad = 2.0 * conv_pa[n-1 : 2*n-1]
```

This is O(n log n)! Much faster than the O(n^2) direct computation in the previous program.

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated FISTA on log-sum-exp smooth approximation (O(n log n) per iteration)
2. Exact subgradient mirror descent (multiplicative weights)
3. LP best-response with adaptive active set
4. Fast coordinate descent with golden section search on 1D subproblems
5. Spectral flattening initialization and intermediate refinement
6. Multi-start, multi-length search with diverse initializations
Key innovation: FFT-based gradient computation makes FISTA ~100x faster than O(n^2) direct,
allowing many more optimization iterations within the time budget.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        neg = ~cond
        if not neg.any():
            rho = n - 1
        else:
            rho = int(np.argmax(neg))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_and_grad_fft(a, tau):
    """Compute log-sum-exp smooth max of conv(a,a) and its gradient using FFT.
    
    Returns (smooth_value, gradient_array) where gradient has same length as a.
    All operations are O(n log n).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    # Numerically stable log-sum-exp
    shifted = tau * (b - b_max)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    log_sum = max_s + np.log(exp_s.sum())
    smooth_val = b_max + log_sum / tau
    
    # Weights p_k
    p = exp_s / exp_s.sum()
    
    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # = 2 * conv(p, a_rev)[n-1 + i] for i = 0..n-1
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1  # = 3n - 2
    size = 1
    while size < conv_len + 1:
        size <<= 1
    p_fft = np.fft.rfft(p, size)
    ar_fft = np.fft.rfft(a_rev, size)
    conv_result = np.fft.irfft(p_fft * ar_fft, size)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    
    return smooth_val, grad


def exact_subgradient(a):
    """Compute the exact subgradient of max(conv(a,a)) at a.
    
    The subgradient at the argmax index k* is: g[i] = 2*a[k*-i] for valid i, else 0.
    Returns (M, k_star, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    k_star = int(np.argmax(b))
    
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    
    return M, k_star, grad


def run_fista_fft(a_init, tau_start, tau_end, max_iters, deadline, lr=0.5):
    """FISTA with FFT-based gradient on smooth log-sum-exp approximation.
    
    Operates on the simplex (sum=1, a>=0) for scale invariance.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters - 1))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_and_grad_fft(y, tau)
        
        # Project gradient to tangent space of simplex
        grad_tan = grad_y - grad_y.mean()
        
        # Adaptive step size
        step = lr / (1.0 + 0.01 * tau)
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        it += 1
        
        # Track best (check periodically)
        if it % 10 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    
    # Final check
    M_a = fast_conv(a).max()
    if M_a < best_M:
        best_M = M_a
        best_a = a.copy()
    
    return best_a


def run_mirror_descent_exact(a_init, max_iters, deadline, step_init=0.5):
    """Mirror descent with exact subgradient (multiplicative weights).
    
    Uses the actual subgradient at argmax for potentially faster convergence
    than smoothed versions when the max is well-defined.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Use a mixture: track multiple "hot" indices for better subgradient
    n_hot = 5
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Get top n_hot indices for a "smoothed" subgradient
        top_idx = np.argsort(b)[::-1][:n_hot]
        # Weights proportional to how close to max
        b_top = b[top_idx]
        w = np.exp(50.0 * (b_top - b_top[0]))
        w = w / w.sum()
        
        # Combined subgradient
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            ki = int(ki)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += wi * 2.0 * a[j]
        
        grad_tan = grad - grad.mean()
        
        # Exponential mirror descent step
        step = step_init / (1.0 + 0.005 * it)
        log_w = -step * grad_tan
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        M_cur = fast_conv(a).max()
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()
    
    return best_a


def build_active_lp_matrix(n, a, active_indices):
    """Build sparse constraint matrix for LP: (a*g)[k] <= M for k in active_indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        k = int(k)
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response(a, M_target, n_active=20, time_limit=10):
    """Solve LP best response: max sum(g) s.t. (a*g)[k] <= M_target for active k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    
    top_indices = np.argsort(b)[::-1][:min(n_active, 2 * n - 1)]
    active = top_indices.tolist()
    
    A = build_active_lp_matrix(n, a, active)
    if A.nnz == 0:
        return None
    
    c = -np.ones(n)
    b_ub = np.full(len(active), M_target)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_descent(a_init, max_iters, deadline, n_active_init=20):
    """LP-based mirror descent with adaptive active set."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    cur_active = n_active_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = lp_best_response(a, best_M, cur_active, time_limit=10)
        if g is None:
            cur_active = min(cur_active * 2, 2 * n - 1)
            no_improve += 1
            if no_improve > 8:
                break
            continue
        
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
            if cur_active > 8:
                cur_active = max(8, int(cur_active * 0.85))
        else:
            no_improve += 1
            cur_active = min(int(cur_active * 1.5), 2 * n - 1)
            if no_improve > 20:
                break
    
    return best_a


def coordinate_opt_1d(a, i, b, n):
    """Find optimal a[i] given all others fixed, using golden section search.
    
    The objective max_k b_k(x) is convex in x >= 0.
    """
    x_cur = a[i]
    
    # Determine affected range
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Precompute base values and slopes for affected k
    # b_k(x) = base_k + slope_k * x for k != 2i
    # b_{2i}(x) = base_quad + x^2
    c_const = []
    slopes = []
    quad_active = False
    quad_base = 0.0
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            quad_active = True
            quad_base = b[k] - x_cur * x_cur
            c_const.append(0.0)
            slopes.append(0.0)
        else:
            j = k - i
            if 0 <= j < n:
                c_const.append(b[k] - 2.0 * x_cur * a[j])
                slopes.append(2.0 * a[j])
            else:
                c_const.append(b[k])
                slopes.append(0.0)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max_unaffected = b[mask].max() if mask.any() else 0.0
    else:
        base_max_unaffected = 0.0
    
    c_arr = np.array(c_const)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max_unaffected
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if quad_active:
            qv = quad_base + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    # Golden section search on [0, x_upper]
    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 0.5, 1.0)
    
    # Ensure hi is large enough
    for _ in range(5):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 200.0:
            hi *= 2.0
        else:
            break
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    
    # Check boundary
    if obj_at(0.0) <= obj_at(x_opt) + 1e-12:
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with random ordering and incremental b updates."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        
        b = fast_conv(a)
        sweep_improved = False
        
        for i in order:
            x_opt = coordinate_opt_1d(a, i, b, n)
            dx = x_opt - a[i]
            if abs(dx) > 1e-12:
                # Update b incrementally
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = b[k] - a[i] * a[i] + x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * dx
                a[i] = x_opt
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        # Recompute b after normalization (scale factor changed)
        scale = a.sum()  # should be 1 now
        b = fast_conv(a)
        
        M = b.max()
        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 2
        if no_improve > 15:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n)
    if src.size == 1:
        return np.full(dest_n, float(src[0]))
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def spectral_init(n, rng, iterations=50):
    """Generate initial sequence by iteratively trying to flatten the power spectrum.
    
    The idea: minimize variance of |FFT(a)|^2 (excluding DC) to make autocorrelation
    closer to a delta function.
    """
    # Start with random non-negative
    a = np.abs(rng.standard_normal(n)) + 0.1
    a = a / a.sum()
    
    for _ in range(iterations):
        # Compute power spectrum
        a_fft = np.fft.rfft(a)
        power = np.abs(a_fft) ** 2
        
        # Target: flat spectrum (excluding DC)
        power_ac = power[1:]
        target = power_ac.mean()
        
        # Compute gradient-like update: reduce a where it contributes to spectral peaks
        # Simple approach: a_new = a * exp(-alpha * (power_deviation projected back))
        # Alternative: just reduce a where the autocorrelation peak is
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Subgradient direction
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        grad_tan = grad - grad.mean()
        alpha = 0.3 / (1.0 + 0.01 * _)
        a = a * np.exp(-alpha * grad_tan)
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
    
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n1 = np.maximum(p + rng.normal(0, 0.05, n), 1e-4)
        inits.append(p_n1 / p_n1.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.15, n), 1e-4)
        inits.append(p_n2 / p_n2.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Concave (parabola)
    poly = (t * (1 - t)) + 0.1
    inits.append(poly / poly.sum())
    
    # Convex (U-shaped)
    poly2 = ((2 * t - 1) ** 2) + 0.1
    inits.append(poly2 / poly2.sum())
    
    # Cosine windows
    for freq in [1, 2, 3, 5]:
        cos_w = 0.5 + 0.4 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Spectral init
    inits.append(spectral_init(n, rng, iterations=30))
    
    # Another spectral init with different start
    a_sp2 = rng.uniform(0.5, 1.5, n)
    a_sp2 = a_sp2 / a_sp2.sum()
    for _ in range(30):
        b = fast_conv(a_sp2)
        k_star = int(np.argmax(b))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a_sp2[j]
        grad_tan = grad - grad.mean()
        a_sp2 = a_sp2 * np.exp(-0.2 * grad_tan)
        a_sp2 = np.maximum(a_sp2, 1e-8)
        a_sp2 = a_sp2 / a_sp2.sum()
    inits.append(a_sp2)
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print("  ** NEW BEST: J={:.8f} n={} t={:.0f}s".format(score, a.size, elapsed), flush=True)

    def report(self, label=""):
        if self.best is not None:
            print("  {}: J={:.8f} n={}".format(label, self.best_score, self.best.size), flush=True)


def run_combined_phase(a0, deadline_phase, seed):
    """Run full optimization pipeline on a single starting point."""
    best = np.maximum(a0.astype(np.float64), 0.0)
    s = best.sum()
    if s < 1e-10:
        best = np.ones_like(best) / len(best)
    else:
        best = best / s
    best_M = fast_conv(best).max()
    
    t_start = time.monotonic()
    remaining = deadline_phase - t_start
    if remaining < 5:
        return best
    
    # Stage 1: FFT FISTA (50% of time)
    t1 = t_start + 0.5 * remaining
    if time.monotonic() < t1:
        for tau_pair in [(30, 300), (100, 1000), (300, 3000)]:
            if time.monotonic() > t1:
                break
            sub_dl = min(t1, time.monotonic() + max(10, (t1 - time.monotonic()) * 0.4))
            a_f = run_fista_fft(best, tau_pair[0], tau_pair[1], 100000, sub_dl, lr=0.8)
            M_f = fast_conv(a_f).max()
            if M_f < best_M:
                best_M = M_f
                best = a_f.copy()
    
    # Stage 2: Mirror descent (20% of time)
    t2 = t_start + 0.7 * remaining
    if time.monotonic() < t2:
        a_md = run_mirror_descent_exact(best, 50000, t2, step_init=0.4)
        M_md = fast_conv(a_md).max()
        if M_md < best_M:
            best_M = M_md
            best = a_md.copy()
    
    # Stage 3: LP descent (20% of time)
    t3 = t_start + 0.9 * remaining
    if time.monotonic() < t3:
        a_lp = run_lp_descent(best, 500, t3, n_active_init=25)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
    
    # Stage 4: Coordinate descent (10% of time)
    if time.monotonic() < deadline_phase:
        a_cd = run_coordinate_descent(best, 500, deadline_phase, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
    
    return best

            tracker.update(parent)

    print("=" * 60, flush=True)

    lengths = [n_parent]

    reserve_later = 250

    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - reserve_later:

        t0 = time.monotonic()
        remaining_pm1 = deadline - reserve_later - time.monotonic()

        time_alloc = min(time_per_length, remaining_pm1 / max(1, len(lengths) - li))

            if time.monotonic() > deadline - reserve_later:
                break
            sub_dl = min(t0 + time_alloc, deadline - reserve_later)

            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()
            
            a_opt = run_combined_phase(a0, sub_dl, seed=seed + li * 1000 + ii)
            M_opt = fast_conv(a_opt).max()

                best_M_n = M_opt
                best_for_n = a_opt.copy()

            tracker.update(best_for_n)
    
    tracker.report("After Phase 1")
```
