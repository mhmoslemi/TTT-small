- We have n bins, each with value a_i, sum = 1
- The convolution has 2n-1 bins
- We want max(conv) to be as small as possible
- The minimum possible max(conv) is at least 1/(2n-1) (by averaging)
- So J >= 2n/(2n-1) ≈ 1

So we need to be very precise in our optimization. The LP approach is probably the right one, but we need to be more careful about convergence.

Let me focus on making the LP mirror descent more effective:

1. **Better active set**: Instead of taking the top-k by value, take all k where b[k] > M - delta for a small delta. This captures all "near-max" constraints.

2. **Larger LP**: Try including more constraints (up to all 2n-1). For n=768, that's 1535 constraints and 768 variables. This is a moderate LP that HiGHS should handle in under a second.

3. **Better step size**: Use a proper line search (golden section or parabolic interpolation) instead of a fixed set of trial values.

4. **Averaged iterate**: In mirror descent / multiplicative weights, the theoretical guarantee is for the AVERAGED iterate, not the current one. So we should also track and evaluate the average.

5. **Multiple LP restarts**: Occasionally restart the LP with a different active set to escape local minima.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using an enhanced LP-based
mirror descent with:
- Full-constraint LP (all 2n-1 constraints, not just top-k)
- Golden section line search for optimal step size
- Averaged iterate tracking
- Dual-guided active set refinement
- Coordinate descent polish with exact 1D optimization
- Multi-start with diverse initializations
- Palindromic and structured initialization variants

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def lp_best_response_full(a, M, max_time=5.0):
    """Solve LP best response with ALL constraints.
    
    max sum(g) s.t. (a*g)[k] <= M for all k=0..2n-2, g >= 0
    
    Returns (g_normalized, dual_values) or (None, None).
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # Build constraint matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    # This is a banded/Toeplitz-like matrix
    rows = []
    cols = []
    vals = []
    for k in range(conv_len):
        for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
    
    c = -np.ones(n)
    b_ub = np.full(conv_len, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': max_time, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs
                # Get dual values (on A_ub constraints)
                duals = r.ineqlin.marginals if hasattr(r, 'ineqlin') and r.ineqlin is not None else None
                return g, duals
    except Exception as e:
        pass
    return None, None

def lp_best_response_active(a, M, active_indices, max_time=5.0):
    """Solve LP best response with only active constraints."""
    n = len(a)
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': max_time, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs
                duals = None
                try:
                    if r.ineqlin is not None:
                        duals = r.ineqlin.marginals
                except:
                    pass
                return g, duals
    except Exception:
        pass
    return None, None

def line_search_1d(a, g, best_M, n_steps=30):
    """Golden section search for optimal t in [0,1] minimizing J((1-t)*a + t*g)."""
    def obj(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 0.01:
            return float('inf')
        a_t = a_t / s
        return fast_conv(a_t).max()
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(n_steps):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    t_opt = (lo + hi) / 2.0
    obj_opt = min(fa, fb)
    # Also check t=0 and t=1
    obj_0 = obj(0.0)
    obj_1 = obj(1.0)
    if obj_0 < obj_opt:
        t_opt = 0.0
        obj_opt = obj_0
    if obj_1 < obj_opt:
        t_opt = 1.0
        obj_opt = obj_1
    return t_opt, obj_opt

def run_lp_mirror_descent_v2(a_init, max_iters, deadline, use_full_lp=True):
    """Improved LP mirror descent with full constraints and line search."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Also track averaged iterate
    avg_a = a.copy()
    avg_count = 1.0
    
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            break
        
        # Solve LP
        if use_full_lp:
            g, duals = lp_best_response_full(a, M, max_time=3.0)
        else:
            # Use top-50 active constraints
            top_k = np.argsort(b)[::-1][:min(50, 2*n-1)]
            g, duals = lp_best_response_active(a, M, top_k.tolist(), max_time=3.0)
        
        if g is None:
            break
        
        # Line search
        t_opt, M_opt = line_search_1d(a, g, best_M, n_steps=25)
        
        if t_opt > 0.001 and M_opt < M * (1 - 1e-10):
            a = (1.0 - t_opt) * a + t_opt * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            
            # Update average
            avg_a = avg_a * (avg_count / (avg_count + 1)) + a / (avg_count + 1)
            avg_count += 1
        
        # Check averaged iterate
        M_avg = fast_conv(avg_a).max()
        if M_avg < best_M:
            best_M = M_avg
            best_a = avg_a.copy()
    
    # Final check
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a

def run_mirror_descent_averaged(a_init, max_iters, deadline, step_init=0.3):
    """Mirror descent with averaged iterate tracking."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    avg_a = a.copy()
    avg_count = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        # Use all indices near the max
        threshold = M * 0.95
        near_max = np.where(b >= threshold)[0]
        
        if len(near_max) == 0:
            near_max = np.array([np.argmax(b)])
        
        # Average subgradient over near-max indices
        grad = np.zeros(n)
        for k in near_max:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= len(near_max)
        grad_tan = grad - grad.mean()
        
        # Step with decay
        step = step_init / np.sqrt(it + 1)
        log_w = -step * grad_tan
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
        
        # Update average (uniform averaging)
        avg_a = avg_a * (avg_count / (avg_count + 1)) + a / (avg_count + 1)
        avg_count += 1
        
        # Check
        if it % 10 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_avg = fast_conv(avg_a).max()
            if M_avg < best_M:
                best_M = M_avg
                best_a = avg_a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    M_avg = fast_conv(avg_a).max()
    if M_avg < best_M:
        best_M = M_avg
        best_a = avg_a.copy()
    
    return best_a

def coordinate_descent_1d(a, i, b, n):
    """Find optimal a[i] given all others fixed. Returns optimal value."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 1.0, 0.5)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)

def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Run coordinate descent with exact 1D optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False
        
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                old_val = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] = b[k] - 2.0 * old_val * a[j] + 2.0 * x_opt * a[j]
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 2
        if no_improve > 30:
            break
    
    return best_a

def resample_seq(src, dest_n):
    """Resample sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits_v2(n, rng, parent=None):
    """Generate diverse initializations with emphasis on U-shaped and structured patterns."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.01, 0.05, 0.1]:
            pn = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(pn / pn.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # U-shaped (various strengths)
    for beta in [2, 4, 8, 16]:
        u = (2 * np.abs(t - 0.5)) ** beta + 0.01
        inits.append(u / u.sum())
    
    # V-shaped (linear)
    v = np.abs(t - 0.5) * 2 + 0.01
    inits.append(v / v.sum())
    
    # Power-law at edges
    for alpha in [0.5, 1.0, 2.0]:
        edge = np.minimum(t, 1 - t)
        pl = edge ** (-alpha) * 0.01 + 0.01
        inits.append(pl / pl.sum())
    
    # Cosine (inverted - low in middle)
    for freq in [1, 2]:
        cos_w = 0.5 - 0.3 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    # Two peaks
    peak1 = np.exp(-50 * (t - 0.1) ** 2)
    peak2 = np.exp(-50 * (t - 0.9) ** 2)
    inits.append((peak1 + peak2 + 0.01) / (peak1 + peak2 + 0.01).sum())
    
    # Three peaks
    p3_1 = np.exp(-50 * (t - 0.05) ** 2)
    p3_2 = np.exp(-50 * (t - 0.5) ** 2)
    p3_3 = np.exp(-50 * (t - 0.95) ** 2)
    inits.append((p3_1 + p3_2 + p3_3 + 0.01) / (p3_1 + p3_2 + p3_3 + 0.01).sum())
    
    # Fourier-based: a_i = 1 + sum c_k cos(2*pi*k*i/n)
    # Choose coefficients to make |A(f)|^2 flat
    for n_harm in [1, 2, 3]:
        a_fourier = np.ones(n)
        for k in range(1, n_harm + 1):
            a_fourier += 0.3 * np.cos(2 * np.pi * k * t)
        a_fourier = np.maximum(a_fourier, 0.01)
        inits.append(a_fourier / a_fourier.sum())
    
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)
