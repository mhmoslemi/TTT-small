"""

restarts. Key innovations over prior approach: (1) continuous temperature

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    conv = irfft(fa * fa, N)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def gradient_weighted(a, weights):
    n = len(a)
    N = 1
    while N < 2 * (2 * n - 1) - 1:
        N <<= 1
    fw = rfft(weights, N)
    fa = rfft(a, N)
    corr = irfft(fw * np.conj(fa), N)
    return 2.0 * corr[:n]


def smoothed_gd(a, max_iters=1000, time_budget=60.0, tau_init=5.0, tau_final=500.0,
                lr_init=0.1, momentum=0.9, stagnation_limit=200):
    """Smoothed gradient descent with Nesterov momentum and adaptive temperature."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    x_mom = a.copy()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        tau = tau_init * (tau_final / tau_init) ** progress
        lr = lr_init * (0.95 ** it)

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = (conv - cmax) * tau
        shifted = np.clip(shifted, -500.0, 500.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= wsum

        grad = gradient_weighted(a, weights)
        grad_centered = grad - np.mean(grad)

        x_mom = a + momentum * (a - x_mom)
        a_new = x_mom - lr * grad_centered
        a_new = project_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > stagnation_limit:
                break

        a = a_new

    return best_a, best_obj

def multi_peak_chase(a, max_iters=200, time_budget=30.0, top_k=5, lr_init=0.05):
    """Chase top-k peaks simultaneously with adaptive learning rate."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights over top-k peaks
        tau_top = 5.0 + 20.0 * (it / max_iters)
        shifted = np.clip((conv - cmax) * tau_top, -300.0, 300.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= wsum

        grad = gradient_weighted(a, weights)
        grad_centered = grad - np.mean(grad)

        # Backtracking line search
        found = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
            a_new = a - trial_lr * grad_centered
            a_new = project_simplex(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                if obj_new < best_obj + 1e-13:
                    lr = trial_lr
                found = True
                stall = 0
                break

        if not found:
            lr *= 0.7
            stall += 1
            if stall > 30:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def generate_initializations(n, rng, parent=None):
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian centered
    x = np.linspace(-3, 3, n)
    g = np.exp(-x ** 2 / 4)
    inits.append(g / g.sum())

    # Raised cosine
    x = np.linspace(0, 1, n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # Optimal-ish: inverse of triangular
    tri = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(float)
    inv_tri = 1.0 / tri
    inits.append(inv_tri / inv_tri.sum())

    # Sine with dead zone
    x = np.linspace(0, 1, n)
    sine = np.abs(np.sin(np.pi * x * 2)) ** 0.5 + 0.3
    inits.append(sine / sine.sum())

    # Random smooth
    t = rng.random(n)
    from scipy.ndimage import gaussian_filter1d
    smooth_t = gaussian_filter1d(t, sigma=max(5, n // 20))
    inits.append(project_simplex(smooth_t))

    # Two-peak Gaussian
    x = np.linspace(0, 1, n)
    two_g = np.exp(-((x - 0.25) ** 2) / 0.03) + np.exp(-((x - 0.75) ** 2) / 0.03)
    inits.append(two_g / two_g.sum())

    if parent is not None:
        resampled = resample_sequence(parent, n)
        inits.append(resampled)
        # Perturbed parent
        noise = np.exp(rng.normal(0, 0.05, n))
        inits.append(project_simplex(resampled * noise))

    return inits

                tau_final=method_params['tau_final']
            )
        elif method_name == 'lp_alt':
            if n_main <= 2000:
                a_new, obj_new = lp_alternate_v2(a_main, max_iters=method_params['max_iters'],
                                                 time_budget=alloc_time)
            else:
                a_new, obj_new = multi_peak_chase(a_main, max_iters=100, time_budget=alloc_time)
        else:  # peak_chase
            a_new, obj_new = multi_peak_chase(
                a_main, max_iters=method_params['max_iters'],
                time_budget=alloc_time, top_k=method_params['top_k']
            )

        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            a_main = a_new.copy()
            n_main = len(a_main)
            print(f"  Cycle {cycle} ({method_name}): NEW BEST {best_score:.8f}")
        elif obj_new < compute_obj(a_main) - 1e-12:
            a_main = a_new.copy()

    print(f"  Phase 2 final: {best_score:.8f}")

    # === Phase 3: Restart and intensify (20% of time) ===
    phase3_budget = min(300.0, (deadline - start_time) * 0.20)
    phase3_start = time.monotonic()
    print(f"\n=== Phase 3: Restart & Intensify (budget={phase3_budget:.0f}s) ===")

    n_restarts = min(8, int(phase3_budget / 25))
    for r in range(n_restarts):
        if time.monotonic() - start_time > phase3_start + phase3_budget:
            break
        remaining = phase3_start + phase3_budget - time.monotonic()
        if remaining < 20:
            break

        # Create perturbed version
        noise_scale = 0.02 + 0.03 * r
        a_pert = best_seq * np.exp(rng.normal(0, noise_scale, len(best_seq)))
        a_pert = project_simplex(a_pert)

        # Optimize the perturbed version
        alloc = min(25.0, remaining * 0.4)
        a_new, obj_new = smoothed_gd(a_pert, max_iters=200, time_budget=alloc * 0.5,
                                     tau_init=5.0, tau_final=200.0)
        if n_main <= 1500:
            a_new2, obj_new2 = lp_alternate_v2(a_new, max_iters=15, time_budget=alloc * 0.3)
            if obj_new2 < obj_new:
                a_new, obj_new = a_new2, obj_new2
        a_new3, obj_new3 = multi_peak_chase(a_new, max_iters=50, time_budget=alloc * 0.2, top_k=5)
        if obj_new3 < obj_new:
            a_new, obj_new = a_new3, obj_new3

        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            n_main = len(a_new)
            print(f"  Restart {r}: NEW BEST {best_score:.8f}")
        else:
            print(f"  Restart {r}: obj={obj_new:.8f} (no improvement)")

    # === Phase 4: Multi-length exploration (12% of time) ===
    phase4_budget = min(180.0, (deadline - start_time) * 0.12)
    phase4_start = time.monotonic()
    print(f"\n=== Phase 4: Multi-Length (budget={phase4_budget:.0f}s) ===")

    n_best = len(best_seq)
    n_tries = [n_best - 30, n_best - 60, n_best + 30, n_best + 60,
               n_best + 100, n_best - 100]
    n_tries = sorted(set(n for n in n_tries if 100 <= n <= 2000))

    per_n = max(5.0, phase4_budget / len(n_tries))

    for n_try in n_tries:
        if time.monotonic() - start_time > phase4_start + phase4_budget:
            break
        remaining = phase4_start + phase4_budget - time.monotonic()
        if remaining < 5:
            break

        a_res = resample_sequence(best_seq, n_try)
        a_res = project_simplex(a_res)

        # Optimize at this length
        a_opt, obj_opt = smoothed_gd(a_res, max_iters=150, time_budget=per_n * 0.4,
                                     tau_init=5.0, tau_final=150.0)
        if n_try <= 1200:
            a_opt2, obj_opt2 = lp_alternate_v2(a_opt, max_iters=10, time_budget=per_n * 0.3)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2
        a_opt3, obj_opt3 = multi_peak_chase(a_opt, max_iters=40, time_budget=per_n * 0.2, top_k=5)
        if obj_opt3 < obj_opt:
            a_opt, obj_opt = a_opt3, obj_opt3

        if obj_opt < best_score - 1e-12:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    remaining5 = deadline - time.monotonic()
    print(f"\n=== Phase 5: Final Polish (budget={remaining5:.0f}s) ===")

    if remaining5 > 10 and best_seq is not None:
        a_final = project_simplex(best_seq.copy())
        n_final = len(a_final)

        # Aggressive LP with many iterations
        if n_final <= 1500:
            a_p, obj_p = lp_alternate_v2(a_final, max_iters=50, time_budget=min(remaining5 * 0.3, 100.0))
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  Final LP: {best_score:.8f}")

        # Very high temperature smoothed GD (fine-tuning)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_p2, obj_p2 = smoothed_gd(a_final, max_iters=500, time_budget=min(remaining * 0.3, 60.0),
                                       tau_init=500.0, tau_final=2000.0, lr_init=0.02)
            if obj_p2 < best_score:
                best_score = obj_p2
                best_seq = a_p2.copy()
                a_final = a_p2.copy()
                print(f"  Final smoothGD: {best_score:.8f}")

        # Peak chase with very fine steps
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_p3, obj_p3 = multi_peak_chase(a_final, max_iters=200, time_budget=min(remaining * 0.3, 30.0),
                                            top_k=10, lr_init=0.01)
            if obj_p3 < best_score:
                best_score = obj_p3
                best_seq = a_p3.copy()
                print(f"  Final peak chase: {best_score:.8f}")

        # One more LP pass
        remaining = deadline - time.monotonic()
        if remaining > 10 and n_final <= 1500:
            a_p4, obj_p4 = lp_alternate_v2(a_final, max_iters=30, time_budget=min(remaining * 0.5, 40.0))
            if obj_p4 < best_score:
                best_score = obj_p4
                best_seq = a_p4.copy()
                print(f"  Final LP2: {best_score:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200) / 200

    # Ensure all values are valid
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Normalize to ensure sum > 0.01 (it will be 1.0 from project_simplex)
    if np.sum(best_seq) < 0.01:
        best_seq = np.ones(100)

    # Convert to list of floats
    result = [float(x) for x in best_seq]

    # Final evaluation check
    val = evaluate_sequence(result)
    if np.isinf(val):
        # Fallback
        result = [1.0] * 100
        val = evaluate_sequence(result)

    print(f"\n=== FINAL: n={len(result)}, score={val:.8f}, time={time.monotonic()-start_time:.1f}s ===")
    return result
