import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

multiplicative (mirror descent) updates, LP alternating, and frequency-domain polishing.
Key innovations: (1) correct FFT cross-correlation for O(n log n) gradients,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_gradient_fft(w, a, N_fft):
    """
    Compute grad[i] = sum_k w[k] * a[k-i] for i=0..n-1 (valid k-i in [0,n)).
    Uses FFT cross-correlation. w has length 2n-1, a has length n.
    N_fft must be >= (2n-1) + n - 1 = 3n-2.
    """
    fw = fft(w, N_fft)
    fa = fft(a, N_fft)
    corr = ifft(fw * np.conj(fa)).real[:len(a)]
    return corr


def subgradient_reduce(a, max_time=60.0, verb=False):
    """
    Optimized subgradient method with:
    - FFT-based gradient computation
    - Adaptive softmax temperature
    - Multiplicative (mirror descent) updates
    - Backtracking step size
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1

    step = 0.05
    it = 0
    stall = 0
    use_mult = True

    while time.monotonic() - start < max_time:
        it += 1
        elapsed_frac = min(it / 300.0, 1.0)
        tau = 0.4 * (0.05 / 0.4) ** elapsed_frac + 0.05

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        s = float(np.sum(a))

        # Softmax weights over convolution
        shifted = np.clip((conv - cmax) / tau, -50.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Gradient via FFT cross-correlation
        grad = 2.0 * compute_gradient_fft(w, a, N_fft)

        # Center gradient
        gmean = grad.mean()
        grad_c = grad - gmean

        # Multiplicative (mirror descent) update
        if use_mult:
            a_new = a * np.exp(-step * grad_c)
            a_new = project_simplex(a_new)
        else:
            a_new = project_simplex(a + step * grad_c)
            # Actually we want to REDUCE the objective, so we go AGAINST gradient
            a_new = project_simplex(a - step * grad_c)

        obj_new = compute_obj(a_new)
        obj_cur = compute_obj(a)

        if obj_new < obj_cur:
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                stall = 0
                step = min(step * 1.05, 0.3)
            else:
                stall += 1
        else:
            step *= 0.8
            stall += 1

        # Switch between multiplicative and additive

            use_mult = not use_mult
            stall = 0
            step = 0.05

        if verb and it % 100 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, step={step:.4f}, tau={tau:.4f}")

    return best_a, best_obj


def soft_max_polish(a, max_time=40.0):
    """
    Iteratively reduce peaks in the convolution by directly modifying a.
    Uses a "peaking" penalty approach.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    gamma = 0.1
    decay = 0.999

    while time.monotonic() - start < max_time:
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        cmean = float(np.mean(conv))

        # Create a "target" convolution that's flatter
        # Reduce peaks by pulling them toward the mean
        peak_mask = conv > cmean * 1.1
        if not np.any(peak_mask):
            break

        # Compute the amount of excess at peaks
        excess = np.maximum(conv - cmean * 1.05, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            break

        # Distribute excess reduction as gradient
        # For each peak position k, the gradient contribution is a[k-i]
        # We want to reduce a at positions that contribute to peaks
        penalty = np.zeros(n)
        for k_idx in range(len(conv)):
            if excess[k_idx] < 1e-12:
                continue
            for i in range(n):
                j = k_idx - i
                if 0 <= j < n:
                    penalty[i] += excess[k_idx] * a[j]

        # Normalize penalty
        pmax = penalty.max()
        if pmax < 1e-15:
            break
        penalty /= pmax

        # Update: reduce where penalty is high
        alpha = gamma * decay
        a_new = a * (1 - alpha * penalty) + alpha * 0.1 * (1 / n)
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            gamma *= 0.9

    return best_a, best_obj


def freq_domain_compress(a, max_time=30.0):
    """
    Frequency domain approach: compress the power spectrum to reduce
    convolution peaks, then extract new sequence via complex square root.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    blend = 0.3
    for it in range(200):
        if time.monotonic() - start > max_time:
            break

        # Compute convolution in frequency domain
        fa = fft(a, N)
        cf = fa * fa
        c = ifft(cf).real[:2 * n - 1]
        c = np.maximum(c, 0)

        cmax = c.max()
        if cmax < 1e-15:
            break

        # Compress peaks in convolution
        c_median = np.median(c[c > 0]) if np.any(c > 0) else cmax * 0.3
        # Soft thresholding: pull peaks toward median
        c_mod = c.copy()
        peak_mask = c_mod > c_median * 1.3
        if np.any(peak_mask):
            c_mod[peak_mask] = c_mod[peak_mask] * (1 - blend) + c_median * 1.1 * blend

        # Ensure non-negative
        c_mod = np.maximum(c_mod, 0)

        # Extract new a via complex square root
        cf_new = fft(c_mod, N)
        # Take square root in frequency domain
        fa_new = np.sqrt(cf_new + 0j)
        a_new = ifft(fa_new).real[:n]

        # Project
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        # Gradually reduce blend
        blend *= 0.995

    return best_a, best_obj


def lp_alternate(a, max_time=30.0):
    """LP alternating: fix a, solve LP for g maximizing sum(g) s.t. conv(a,g) <= M."""
    n = len(a)
    if n > 1500:
        return a, compute_obj(a)

    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Build Toeplitz matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]

    while time.monotonic() - start < max_time:
        conv = compute_conv(a)
        M = float(np.max(conv))

        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(5, max_time * 0.3)})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                sg = float(np.sum(g))
                if sg > 1e-10:
                    g = g / sg
                    # Line search
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_try = project_simplex((1 - t) * a + t * g)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                    if compute_obj(a) >= best_obj - 1e-10 and time.monotonic() - start > max_time * 0.3:
                        # Check if we're stuck
                        if abs(compute_obj(a) - best_obj) < 1e-10:
                            break
        except Exception:
            break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    rc = 1 - 0.5 * np.cos(2 * np.pi * x)
    inits.append(rc / rc.sum())

    # Hann
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-6)
    inits.append(hann / hann.sum())

    # Gaussian (wide)
    gx = np.linspace(-2.5, 2.5, n)
    gauss = np.exp(-gx ** 2 / 2)
    inits.append(gauss / gauss.sum())

    # Asymmetric ramp with taper
    ramp = np.arange(1, n + 1, dtype=float)
    taper = np.ones(n)
    taper[:n // 10] = np.linspace(0.3, 1, n // 10)
    taper[-n // 10:] = np.linspace(1, 0.3, n // 10)
    asym = ramp * taper
    inits.append(asym / asym.sum())

    # Asymmetric: heavy on one side
    skew = (1 + 2 * x) ** 2
    inits.append(skew / skew.sum())

    # Two-hump asymmetric
    hump1 = np.exp(-((x - 0.35) ** 2) / 0.03)
    hump2 = np.exp(-((x - 0.7) ** 2) / 0.05) * 0.7
    bimodal = hump1 + hump2 + 0.05
    inits.append(bimodal / bimodal.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    if parent is not None:
        parent_resampled = resample_sequence(parent, n)
        inits.append(project_simplex(parent_resampled))
        pert = parent_resampled * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_simplex(pert))

    return inits

    phase1_time = min(120.0, (deadline - time.monotonic()) * 0.12)
    p1_start = time.monotonic()

    n_options = [n_parent, n_parent - 50, n_parent + 50, n_parent - 100,
                 n_parent + 100, 500, 800, 1000]
    n_options = list(dict.fromkeys([n for n in n_options if 100 <= n <= 2000]))
    per_n_time = max(5.0, phase1_time / len(n_options))

    candidates = []
    for n_try in n_options:
        if time.monotonic() - p1_start > phase1_time:
            break
        inits = generate_inits(n_try, rng, parent)
        for init in inits:
            if time.monotonic() - p1_start > phase1_time:
                break
            # Quick subgradient
            a_quick, obj_quick = subgradient_reduce(init, max_time=min(per_n_time * 0.3, 8.0))
            if obj_quick < np.inf:
                candidates.append((obj_quick, a_quick.copy(), n_try))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best: {candidates[0][0] if candidates else 'N/A'}")

    for obj_c, a_c, n_c in candidates[:8]:
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  Updated: n={n_c}, obj={best_score:.8f}")

    # === Phase 2: Intensive optimization (50% time) ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    print(f"  Best entering: {best_score:.8f}")
    phase2_time = min((deadline - time.monotonic()) * 0.50, 700.0)
    p2_start = time.monotonic()

    # Focus on top 3 candidates
    top_list = [(best_score, best_seq.copy(), len(best_seq))]
    for obj_c, a_c, n_c in candidates[:5]:
        if obj_c < best_score + 0.01 and (obj_c, n_c) not in [(x[0], x[2]) for x in top_list]:
            top_list.append((obj_c, a_c.copy(), n_c))

    for ci, (obj_init, a_init, n_init) in enumerate(top_list[:3]):
        if time.monotonic() - p2_start > phase2_time:
            break
        remaining = phase2_time - (time.monotonic() - p2_start)
        if remaining < 15:
            break
        alloc = remaining / max(1, 3 - ci)

        a = project_simplex(a_init.copy())
        local_best = a.copy()
        local_obj = compute_obj(a)
        print(f"  Cand {ci} (n={n_init}): obj={local_obj:.8f}, alloc={alloc:.0f}s")

        # Cycle through optimization strategies
        strat_cycle = 0
        while time.monotonic() - p2_start < p2_start + phase2_time:
            rem = p2_start + phase2_time - time.monotonic()
            if rem < 10:
                break

            strat = strat_cycle % 4
            t_budget = min(rem * 0.25, 30.0)

            if strat == 0:
                a_new, obj_new = subgradient_reduce(a, max_time=t_budget)
            elif strat == 1 and n_init <= 1200:
                a_new, obj_new = lp_alternate(a, max_time=t_budget)
            elif strat == 2:
                a_new, obj_new = freq_domain_compress(a, max_time=t_budget)
            else:
                a_new, obj_new = soft_max_polish(a, max_time=t_budget)

            if obj_new < local_obj:
                local_obj = obj_new
                local_best = a_new.copy()
                a = a_new
                if local_obj < best_score:
                    best_score = local_obj
                    best_seq = local_best.copy()
                    print(f"    ** UPD strat={strat}: obj={best_score:.8f} **")
            strat_cycle += 1

        print(f"  Cand {ci} done: obj={local_obj:.8f}")

    # === Phase 3: Frequency domain + fine subgradient (20% time) ===
    print(f"\n=== Phase 3: Freq + Fine Polish ===")
    print(f"  Best: {best_score:.8f}")
    phase3_time = min((deadline - time.monotonic()) * 0.20, 250.0)
    p3_start = time.monotonic()

    a = project_simplex(best_seq.copy())
    while time.monotonic() - p3_start < p3_start + phase3_time:
        rem = p3_start + phase3_time - time.monotonic()
        if rem < 8:
            break

        # Frequency domain
        a_f, obj_f = freq_domain_compress(a, max_time=min(rem * 0.3, 20.0))
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            a = a_f
            print(f"  FD: {best_score:.8f}")

        # Fine subgradient with small step
        a_s, obj_s = subgradient_reduce(a, max_time=min(rem * 0.3, 15.0))
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_s.copy()
            a = a_s
            print(f"  SG: {best_score:.8f}")

    # === Phase 4: Multi-length fine exploration (10% time) ===
    print(f"\n=== Phase 4: Multi-length Fine ===")
    print(f"  Best: {best_score:.8f}")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 20:
        phase4_time = min(remaining4 * 0.35, 150.0)
        p4_start = time.monotonic()
        n_cur = len(best_seq)
        n_tries = [n_cur - 20, n_cur + 20, n_cur - 40, n_cur + 40,
                   n_cur + 60, n_cur - 60]
        n_tries = list(dict.fromkeys([n for n in n_tries if 100 <= n <= 2000]))
        per_n = max(5.0, phase4_time / len(n_tries))

        for n_try in n_tries:
            if time.monotonic() - p4_start > p4_start + phase4_time:
                break
            a_res = project_simplex(resample_sequence(best_seq, n_try))
            a_opt, obj_opt = subgradient_reduce(a_res, max_time=min(per_n * 0.5, 12.0))
            if n_try <= 1000:
                a_opt2, obj_opt2 = lp_alternate(a_opt, max_time=min(per_n * 0.3, 8.0))
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Best before: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 10 and best_seq is not None:
        a_final = project_simplex(best_seq.copy())
        n_final = len(a_final)

        while time.monotonic() < deadline - 5:
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            # Rotation of strategies
            a1, o1 = subgradient_reduce(a_final, max_time=min(rem * 0.2, 15.0))
            if o1 < best_score:
                best_score = o1
                best_seq = a1.copy()
                a_final = a1
            rem = deadline - time.monotonic()
            if rem < 5:
                break

            if n_final <= 1000:
                a2, o2 = lp_alternate(a_final, max_time=min(rem * 0.2, 10.0))
                if o2 < best_score:
                    best_score = o2
                    best_seq = a2.copy()
                    a_final = a2
            rem = deadline - time.monotonic()
            if rem < 5:
                break

            a3, o3 = freq_domain_compress(a_final, max_time=min(rem * 0.2, 10.0))
            if o3 < best_score:
                best_score = o3
                best_seq = a3.copy()
                a_final = a3
            rem = deadline - time.monotonic()
            if rem < 3:
                break

            a4, o4 = soft_max_polish(a_final, max_time=min(rem * 0.15, 8.0))
            if o4 < best_score:
                best_score = o4
                best_seq = a4.copy()
                a_final = a4

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - t0
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
