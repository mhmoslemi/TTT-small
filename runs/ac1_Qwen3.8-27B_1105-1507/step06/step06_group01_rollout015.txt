"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a dual-guided optimization:
1. LP cutting-plane method to find exact best responses (with optimality certificate)
2. Nesterov-accelerated projected gradient on surrogate quadratic objectives
   constructed from LP dual variables
3. Smart 1D line search along improving directions
4. Multi-length exploration with perturbation-based local minimum escape
5. Final coordinate descent polish
Key innovation over previous: exact LP via cutting planes gives optimality
certificates, and dual variables build surrogate QPs for faster convergence.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def linear_conv(a, b):
    """Linear convolution of a and b via FFT."""
    na, nb = len(a), len(b)
    if na == 0 or nb == 0:
        return np.array([])
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * rho_arr + 1.0) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(cond))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def solve_lp_cutting_plane(a, M, max_rounds=80, time_limit=120, init_active=None):
    """Solve LP max sum(g) s.t. (a*g)[k] <= M, g >= 0 using cutting planes.
    
    Returns (g, lp_value, dual_vars, active_set) or (None, ...) on failure.
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    if init_active is None:
        # Initialize with positions where (a*a) is near max
        b = fast_conv(a)
        thresh = M * 0.95
        active = np.where(b >= thresh)[0].tolist()
        if len(active) < 5:
            active = np.argsort(b)[::-1][:20].tolist()
    else:
        active = list(init_active)
    
    t_start = time.monotonic()
    g = None
    lp_value = 0.0
    dual = None
    
    for rnd in range(max_rounds):
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit:
            break
        
        # Build constraint matrix
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active):
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])
        
        if not rows:
            break
        
        A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
        
        c_obj = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0.0, None)] * n
        
        remaining_time = max(1.0, time_limit - (time.monotonic() - t_start))
        
        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs',
                        options={'time_limit': remaining_time, 'presolve': True,
                                 'mip_gap': 0, 'dual_feasibility_tolerance': 1e-9})
            if not r.success or r.x is None:
                break
        except Exception:
            break
        
        g = np.maximum(r.x, 0.0)
        lp_value = g.sum()
        
        # Get dual variables if available
        if hasattr(r, 'ineqlin_slack') and r.ineqlin_slack is not None:
            dual = np.zeros(len(active))
            # HiGHS dual variables for inequality constraints
            if hasattr(r, 'ineqlin_dual') and r.ineqlin_dual is not None:
                dual = np.abs(r.ineqlin_dual)
        
        # Check for violations of non-active constraints
        ag = linear_conv(a, g)
        if len(ag) < conv_len:
            ag = np.pad(ag, (0, conv_len - len(ag)))
        
        violations = ag[:conv_len] - M
        max_viol = violations.max()
        
        if max_viol < 1e-7 * max(M, 1e-10):
            # No significant violation - we have the true optimum
            break
        
        # Add most violated constraint
        worst_k = int(np.argmax(violations))
        if worst_k not in active:
            active.append(worst_k)
            # Also add symmetric partner
            sym_k = conv_len - 1 - worst_k
            if sym_k != worst_k and sym_k not in active:
                active.append(sym_k)
        
        if len(active) >= conv_len:
            break
    
    return g, lp_value, dual, active


def surrogate_qp_grad(a, lam):
    """Compute gradient of surrogate quadratic sum_k lam[k]*(a*a)[k].
    
    grad[i] = 2 * sum_k lam[k] * a[k-i] (for 0 <= k-i < n)
    """
    n = len(a)
    grad = np.zeros(n)
    # (Q*a)[i] = sum_j lam[i+j] * a[j]
    # This is a reversed correlation
    # grad[i] = 2 * (Q*a)[i]
    # (Q*a)[i] = sum_{j=0}^{n-1} lam[i+j] * a[j]
    # Let's compute this efficiently
    # For each i, we need sum of lam[i+j]*a[j] for j=0..n-1
    # This is like: for each pair (i,j), add lam[i+j]*a[j] to grad[i]
    # Alternative: use the fact that this is a "Hankel matrix times vector"
    
    # Direct O(n^2) computation - fine for n <= 2048
    # (Q*a)[i] = sum_j lam[i+j]*a[j]
    # Let m = i+j, so j = m-i, m ranges from i to i+n-1
    # (Q*a)[i] = sum_{m=i}^{i+n-1} lam[m]*a[m-i]
    
    # This is a cross-correlation of lam and a!
    # np.correlate(lam, a, mode='full')[i+n-1] = sum_j lam[i+j]*a[j]
    # Actually: np.correlate(lam, a, 'full')[k] = sum_j lam[j+k]*a[j] for appropriate k
    
    # Let me just do it directly with numpy for clarity
    # lam has length 2n-1 (indices 0 to 2n-2)
    # (Q*a)[i] = sum_{j=0}^{n-1} lam[i+j] * a[j]
    
    # Using the full correlation:
    # full = np.correlate(lam, a, mode='full')
    # full[k] = sum_j lam[j] * a[k-j] ... this isn't quite right
    
    # Let me just use the direct approach
    for i in range(n):
        # sum_j lam[i+j] * a[j]
        idx = np.arange(i, i + n)
        idx = idx[idx < len(lam)]
        valid_j = idx - i
        grad[i] = 2.0 * np.dot(lam[idx], a[valid_j])
    
    return grad


def nesterov_qp(a_init, lam, max_iters, deadline, lr=None):
    """Minimize sum_k lam[k]*(a*a)[k] subject to a>=0, sum(a)=1 using Nesterov."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    if lr is None:
        lr = 0.1
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_val = np.inf
    
    lam = np.asarray(lam, dtype=np.float64)
    lam_len = len(lam)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        grad_y = np.zeros(n)
        for i in range(n):
            idx = np.arange(i, i + n)
            idx = idx[idx < lam_len]
            valid_j = idx - i
            grad_y[i] = 2.0 * np.dot(lam[idx], y[valid_j])
        
        # Project gradient to tangent of simplex
        grad_tan = grad_y - grad_y.mean()
        
        step = lr / (1.0 + 0.01 * it)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        # Track best
        if it % 10 == 0:
            val = np.dot(fast_conv(a), lam[:len(fast_conv(a))])
            if val < best_val:
                best_val = val
                best_a = a.copy()
    
    return best_a


def line_search(a, g, M, n_crit=20):
    """1D line search: minimize max(conv(a_t, a_t)) where a_t = (1-t)*a + t*g.
    
    Uses critical coordinates for efficiency.
    """
    n = len(a)
    
    # Find critical coordinates (where convolution is near max)
    b = fast_conv(a)
    conv_len = 2 * n - 1
    critical = np.argsort(b)[::-1][:n_crit]
    
    def obj_at_t(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        st = a_t.sum()
        if st < 1e-10:
            return np.inf
        a_t = a_t / st
        b_t = fast_conv(a_t)
        return b_t.max()
    
    # Check endpoints
    f0 = obj_at_t(0.0)
    f1 = obj_at_t(1.0)
    
    if f1 >= f0:
        return 0.0
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at_t(a_pt)
    fb = obj_at_t(b_pt)
    
    for _ in range(30):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at_t(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at_t(b_pt)
    
    return (lo + hi) / 2.0

def coordinate_descend(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent for final polish."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        
        b = fast_conv(a)
        sweep_changed = False
        
        for i in order:
            # Save old value
            old_i = a[i]
            
            # Compute affected convolution values
            # (a*a)[k] changes for k in [i, i+n-1]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Base values (without the a[i] contribution)
            c_vals = []
            slopes = []
            has_quad = False
            quad_idx = -1
            
            for k in range(k_lo, k_hi + 1):
                j = k - i
                if 0 <= j < n:
                    if k == 2 * i:
                        c = b[k] - old_i * old_i
                        c_vals.append(c)
                        slopes.append(0.0)
                        has_quad = True
                        quad_idx = len(c_vals) - 1
                    else:
                        c = b[k] - 2.0 * old_i * a[j]
                        c_vals.append(c)
                        slopes.append(2.0 * a[j])
                else:
                    c_vals.append(b[k])
                    slopes.append(0.0)
            
            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            
            c_arr = np.array(c_vals)
            s_arr = np.array(slopes)
            
            def obj_at(x):
                vals = c_arr + s_arr * x
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if has_quad:
                    qv = c_arr[quad_idx] + x * x
                    if qv > m:
                        m = qv
                return m
            
            # Golden section on [0, hi]
            hi = max(2.0 * old_i + 0.1, np.sqrt(max(b.max(), 0.0)) + 0.5)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_at(a_pt)
            fb = obj_at(b_pt)
            
            for _ in range(35):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_at(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)
            
            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - old_i) > 1e-10:
                a[i] = max(0.0, x_opt)
                # Update b
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if k == 2 * i:
                            b[k] += (x_opt * x_opt - old_i * old_i)
                        else:
                            b[k] += 2.0 * a[j] * (x_opt - old_i)
                sweep_changed = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_changed:
            no_improve += 2
        if no_improve > 15:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search with LP cutting-plane optimization, dual-guided QP,
    and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512) / 512.0
        tracker.update(parent)
        print(f"No parent, using uniform n=512", flush=True)
    
    n_parent = parent.size
    
    # === PHASE 1: Intensive LP mirror descent on parent length ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP cutting-plane optimization (n={n_parent})", flush=True)
    print(f"{'='*60}", flush=True)
    
    phase1_deadline = time.monotonic() + min(500, 0.4 * (deadline - time.monotonic()))
    
    a_p1 = parent / parent.sum()
    a_p1 = np.maximum(a_p1, 0.0)
    a_p1 = a_p1 / a_p1.sum()
    
    # Run LP mirror descent with cutting planes
    a_opt = lp_mirror_descent_v2(a_p1, max_rounds=30, deadline=phase1_deadline, tol=1e-5)
    tracker.update(a_opt)
    
    # Try from a perturbed version
    if time.monotonic() < phase1_deadline:
        a_pert = np.maximum(a_opt + rng.normal(0, 0.03, n_parent), 1e-6)
        a_pert = a_pert / a_pert.sum()
        sub_dl = min(phase1_deadline, time.monotonic() + 120)
        a_opt2 = lp_mirror_descent_v2(a_pert, max_rounds=15, deadline=sub_dl, tol=1e-5)
        tracker.update(a_opt2)
    
    tracker.report("After Phase 1")
    
    # === PHASE 2: Dual-guided QP refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Dual-guided QP refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 200:
        n_best = tracker.best.size
        a_cur = tracker.best / max(tracker.best.sum(), 1e-10)
        
        phase2_deadline = min(deadline - 200, time.monotonic() + 200)
        
        # Get LP dual to construct surrogate QP
        M_cur = fast_conv(a_cur).max()
        g, lp_val, dual, active = solve_lp_cutting_plane(
            a_cur, M_cur, max_rounds=40, time_limit=60
        )
        
        if dual is not None and len(dual) > 0:
            # Expand dual to full length
            lam_full = np.zeros(2 * n_best - 1)
            # Map active indices to full
            for idx, k in enumerate(active):
                lam_full[k] = dual[idx]
            
            if lam_full.sum() > 1e-10:
                # Run Nesterov on the surrogate QP
                a_qp = nesterov_qp(a_cur, lam_full, max_iters=5000,
                                   deadline=min(phase2_deadline, time.monotonic() + 80),
                                   lr=0.05)
                tracker.update(a_qp)
        
        # Second QP pass with updated dual
        if time.monotonic() < phase2_deadline and tracker.best is not None:
            a_cur2 = tracker.best / max(tracker.best.sum(), 1e-10)
            M_cur2 = fast_conv(a_cur2).max()
            g2, lp_val2, dual2, active2 = solve_lp_cutting_plane(
                a_cur2, M_cur2, max_rounds=40, time_limit=60
            )
            if dual2 is not None and len(dual2) > 0:
                lam_full2 = np.zeros(2 * n_best - 1)
                for idx, k in enumerate(active2):
                    lam_full2[k] = dual2[idx]
                if lam_full2.sum() > 1e-10:
                    a_qp2 = nesterov_qp(a_cur2, lam_full2, max_iters=5000,
                                        deadline=min(phase2_deadline, time.monotonic() + 60),
                                        lr=0.03)
                    tracker.update(a_qp2)
    
    tracker.report("After Phase 2")
    
    # === PHASE 3: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if time.monotonic() < deadline - 150:
        # Try lengths near parent
        candidate_lengths = []
        for delta in [-64, 64, -128, 128, 256]:
            nl = n_parent + delta
            if 256 <= nl <= 2048:
                candidate_lengths.append(nl)
        
        phase3_deadline = min(deadline - 150, time.monotonic() + 300)
        time_per_len = max(30, (phase3_deadline - time.monotonic()) / max(1, len(candidate_lengths)))
        
        for nl in candidate_lengths[:4]:
            if time.monotonic() > phase3_deadline:
                break
            sub_dl = min(phase3_deadline, time.monotonic() + time_per_len)
            
            a_init = resample_seq(tracker.best, nl)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()
            
            a_opt = lp_mirror_descent_v2(a_init, max_rounds=10, deadline=sub_dl, tol=1e-4)
            tracker.update(a_opt)
    
    tracker.report("After Phase 3")
    
    # === PHASE 4: Final coordinate descent polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_best = tracker.best.size
        # Estimate time per sweep
        n = n_best
        est_sweep_time = n * 0.005  # rough estimate
        max_sweeps = max(5, int((deadline - 30 - time.monotonic()) / max(0.1, est_sweep_time)))
        max_sweeps = min(max_sweeps, 200)
        
        a_cd = coordinate_descend(tracker.best, max_sweeps, deadline - 5, seed=12345)
        tracker.update(a_cd)
    
    tracker.report("After Phase 4")
    
    # === PHASE 5: One more LP pass on the best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 10:
        a_final = lp_mirror_descent_v2(tracker.best, max_rounds=10,
                                       deadline=deadline - 5, tol=1e-6)
        tracker.update(a_final)
    
    tracker.report("After Phase 5")
    
    # Ensure valid output
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
    
    # Scale to ensure sum >= 0.01 (it should be since J is scale invariant)
    final = tracker.best.copy()
    if final.sum() < 0.01:
        final = final * (0.1 / max(final.sum(), 1e-15))
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in final]
