"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with analytical 1D optimization via bisection on the ratio objective
2. Nesterov-accelerated gradient descent on adaptive log-sum-exp smooth approximation
3. Cutting-plane LP best-response with warm starts
4. Diverse spectral and anti-peak initializations
5. Multi-start, multi-length search with careful time management
Key improvements: analytical CD (O(n) per coord), adaptive FISTA with momentum restart,
cutting-plane LP (grow active set adaptively), and anti-peak spectral initialization.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    if not (~cond).any():
        rho = n - 1
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad(a, tau):
    """Smooth max of conv(a,a) via log-sum-exp and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    scaled = tau * (b - b_max)
    scaled -= scaled.max()
    exp_b = np.exp(scaled)
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: d/da[i] of sum_k p[k] * (a*a)[k]
    # = 2 * sum_k p[k] * a[k-i]
    grad = np.zeros(n)
    for i in range(n):
        lo = max(0, i)
        hi = min(2 * n - 1, i + n)
        if lo < hi:
            grad[i] = 2.0 * np.dot(p[lo:hi], a[i - lo:i - lo + (hi - lo)])
    return smooth_val, grad


def run_fista_adaptive(a_init, max_iters, deadline, tau_start=10.0, tau_end=500.0):
    """FISTA with adaptive tau, momentum restart, and backtracking step."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    x_prev = a.copy()
    step = 0.3
    stale_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters * 0.8))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_grad(y, tau)
        grad_tan = grad_y - np.mean(grad_y)
        
        # Backtracking: try current step, reduce if no progress
        x_new = y - step * grad_tan
        x_new = proj_simplex(x_new)
        
        # Momentum restart: if oscillating, reset momentum
        diff = x_new - x_prev
        if it > 2:
            dot = np.dot(diff, x_prev - y)
            if dot > 0:
                t = 1.0
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new
        y_new = x_new + alpha * (x_new - x_prev)
        y_new = proj_simplex(y_new)
        
        x_prev = a
        a = x_new
        y = y_new
        t = t_new
        stale_count += 1
        
        # Periodic evaluation
        if it % 10 == 0 or it == max_iters - 1:
            M_a = fast_conv(a).max()
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()
                stale_count = 0
                # Increase step slightly on improvement
                step = min(step * 1.05, 1.0)
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
                stale_count = 0
            else:
                # Reduce step if stuck
                if stale_count > 30:
                    step *= 0.9
                    stale_count = 0
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()
        best_M = M_final
    
    return best_a, best_M


def coord_1d_optim(a, i, b):
    """Fast 1D coordinate optimization for a[i] minimizing J (scale-invariant).
    
    We minimize g(x) = M(x) / (S+x)^2 where:
    - S = sum of a[j] for j != i
    - M(x) = max of conv values when a[i] = x
    
    Returns optimal x >= 0.
    """
    n = len(a)
    x_old = a[i]
    S = a.sum() - x_old  # sum of others
    
    if S < 1e-15:
        return 0.0
    
    # Determine affected k range
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Build the pieces
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        U = b[mask].max() if mask.any() else 0.0
    else:
        U = 0.0
    
    # Affected pieces (linear and quadratic)
    c_lin = []
    s_lin = []
    c_quad = 0.0
    has_quad = False
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c_quad = b[k] - x_old * x_old
            has_quad = True
        else:
            j = k - i
            if 0 <= j < n:
                c_lin.append(b[k] - 2.0 * x_old * a[j])
                s_lin.append(2.0 * a[j])
    
    c_lin = np.array(c_lin) if c_lin else np.array([0.0])
    s_lin = np.array(s_lin) if s_lin else np.array([0.0])
    
    def M_of_x(x):
        """Max conv value when a[i] = x."""
        m = U
        if c_lin.size > 0:
            lm = (c_lin + s_lin * x).max()
            if lm > m:
                m = lm
        if has_quad:
            qv = c_quad + x * x
            if qv > m:
                m = qv
        return m
    
    def g_of_x(x):
        return M_of_x(x) / ((S + x) * (S + x))
    
    # Find minimum of g(x) for x >= 0
    # Candidates: x=0, analytical solutions, and golden section
    g_0 = g_of_x(0.0)
    
    # Analytical candidates
    candidates = [0.0, x_old]
    
    # Quadratic piece active: x = c_quad / S
    if has_quad and c_quad > 0:
        x_q = c_quad / S
        if x_q > 0:
            candidates.append(x_q)
    
    # Linear pieces active: x = S - 2*c_j/s_j
    for j in range(min(len(c_lin), 50)):
        if s_lin[j] > 1e-15:
            x_cand = S - 2.0 * c_lin[j] / s_lin[j]
            if 0 < x_cand < 10.0 * S:
                candidates.append(x_cand)
    
    # Also add some heuristic candidates
    candidates.append(0.5 * x_old)
    candidates.append(2.0 * x_old)
    candidates.append(S / n)  # uniform-like
    
    # Evaluate all candidates
    best_x = 0.0
    best_g = g_0
    for xc in candidates:
        xc = max(0.0, xc)
        gx = g_of_x(xc)
        if gx < best_g:
            best_g = gx
            best_x = xc
    
    # Golden section refinement around best candidate
    lo = max(0.0, best_x * 0.5 - 0.5 * S / n)
    hi = best_x * 1.5 + 0.5 * S / n
    hi = min(hi, 5.0 * S)  # cap
    
    if hi > lo:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_pt = hi - gr * (hi - lo)
        b_pt = lo + gr * (hi - lo)
        fa = g_of_x(a_pt)
        fb = g_of_x(b_pt)
        for _ in range(30):
            if fa < fb:
                hi = b_pt
                b_pt = a_pt
                fb = fa
                a_pt = hi - gr * (hi - lo)
                fa = g_of_x(a_pt)
            else:
                lo = a_pt
                a_pt = b_pt
                fa = fb
                b_pt = lo + gr * (hi - lo)
                fb = g_of_x(b_pt)
        x_gs = (lo + hi) / 2.0
        if g_of_x(x_gs) < best_g:
            best_x = x_gs
            best_g = g_of_x(x_gs)
    
    return max(0.0, best_x)


def run_coord_descent_fast(a_init, max_sweeps, deadline, seed=0):
    """Fast coordinate descent using analytical 1D optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        
        b = fast_conv(a)
        sweep_changed = False
        
        for i in order:
            x_opt = coord_1d_optim(a, i, b)
            if abs(x_opt - a[i]) > 1e-12 * max(1.0, a[i]):
                # Update b incrementally
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                old_x = a[i]
                a[i] = x_opt
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += x_opt * x_opt - old_x * old_x
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old_x)
                sweep_changed = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        J = compute_J(a)
        if J < best_J * (1 - 1e-10):
            best_J = J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_changed:
            no_improve += 5
        if no_improve > 15:
            break
    
    return best_a, best_J


def cutting_plane_lp(a, M_target, n_active_start=10, n_active_max=80, max_rounds=20, time_limit=30.0):
    """Cutting-plane LP: start with small active set, grow by adding most violated constraint."""
    n = len(a)
    b = fast_conv(a)
    
    # Initial active set: top n_active_start
    top_idx = np.argsort(b)[::-1][:min(n_active_start, 2 * n - 1)]
    active_set = set(top_idx.tolist())
    
    c_vec = -np.ones(n)
    best_g = None
    best_M = M_target
    
    t_start = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - t_start > time_limit:
            break
        
        active_sorted = sorted(active_set)
        
        # Build constraint matrix
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active_sorted):
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    rows.append(idx)
                    cols.append(j)
                    vals.append(a[m])
        
        if not rows:
            break
        
        A = csr_matrix((vals, (rows, cols)), shape=(len(active_sorted), n))
        b_ub = np.full(len(active_sorted), M_target)
        bounds = [(0.0, None)] * n
        
        try:
            r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'time_limit': 5, 'presolve': True})
            if not r.success or r.x is None:
                break
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g = g / g_sum
            
            # Check violation on ALL constraints
            b_g = fast_conv(g)
            violations = b_g - M_target
            max_viol = violations.max()
            
            if max_viol < 1e-10:
                # Feasible! Return g
                return g
            else:
                # Add the most violated constraint
                worst_k = int(np.argmax(b_g))
                active_set.add(worst_k)
                
                # Also add a few more near the worst
                top_viol = np.argsort(violations)[::-1]
                for vk in top_viol[:3]:
                    active_set.add(int(vk))
                
                # Cap active set size
                if len(active_set) > n_active_max:
                    # Keep only the most violated
                    viol_arr = np.array([violations[k] for k in active_set])
                    keep = np.argsort(viol_arr)[::-1][:n_active_max]
                    active_set = set([active_set[k] for k in keep])
        except Exception:
            break
    
    return None


def run_lp_polish(a_init, deadline, n_active_start=10, n_active_max=60):
    """LP-based polish with cutting planes and line search."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t_start = time.monotonic()
    no_improve = 0
    
    while time.monotonic() < deadline and no_improve < 20:
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        time_limit = min(20.0, remaining * 0.3)
        
        g = cutting_plane_lp(a, best_M, n_active_start, n_active_max,
                            max_rounds=15, time_limit=time_limit)
        if g is None:
            no_improve += 3
            # Try with slightly relaxed M
            g = cutting_plane_lp(a, best_M * 1.001, n_active_start, n_active_max,
                                max_rounds=10, time_limit=min(10.0, time_limit))
            if g is None:
                no_improve += 3
                continue
        
        # Line search
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial = a_trial / st
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-12):
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
    
    return best_a, best_M


def anti_peak_init(n, rng, iters=500):
    """Initialize by iteratively reducing peak autocorrelation (spectral-ish approach)."""
    a = np.ones(n) / n
    for _ in range(iters):
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        # Subgradient: reduce a[i] where a[k_star - i] is large
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        # Also consider second and third peaks
        top_k = np.argsort(b)[::-1][:3]
        for k2 in top_k[1:]:
            for i in range(n):
                j = k2 - i
                if 0 <= j < n:
                    grad[i] += 0.5 * 2.0 * a[j]
        
        grad_tan = grad - grad.mean()
        step = 0.05
        a = a - step * grad_tan
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / a_sum
    
    return a


def random_phase_init(n, rng):
    """Initialize using random phase correlations to get flat-ish autocorrelation."""
    # Generate random complex phases
    phases = rng.uniform(0, 2 * np.pi, n)
    z = np.exp(1j * phases)
    # The "amplitude" that would give flat autocorrelation if we could use complex
    # Instead, use the real part magnitudes as a starting point
    a = np.abs(z)  # all ones, not useful
    # Better: use a random vector and take absolute values after FFT
    x = rng.standard_normal(n) + 1j * rng.standard_normal(n)
    X = np.fft.fft(x)
    # Flatten the spectrum
    X_flat = X / (np.abs(X) + 1e-10) * np.abs(X).mean()
    a = np.abs(np.fft.ifft(X_flat).real)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        from scipy.interpolate import interp1d
        if parent.size != n:
            t_old = np.linspace(0, 1, parent.size)
            t_new = np.linspace(0, 1, n)
            p = np.interp(t_new, t_old, parent)
        else:
            p = parent.copy()
        p = np.maximum(p, 0.0)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_noisy = np.maximum(p + rng.normal(0, 0.02, n), 1e-4)
        inits.append(p_noisy / p_noisy.sum())
        p_noisy2 = np.maximum(p + rng.normal(0, 0.08, n), 1e-4)
        inits.append(p_noisy2 / p_noisy2.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Anti-peak (iterative peak reduction)
    inits.append(anti_peak_init(n, rng, iters=200))
    
    # Random phase
    inits.append(random_phase_init(n, rng))
    
    # Cosine
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())
    
    # Double cosine
    cos_w2 = 0.5 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t)
    cos_w2 = np.maximum(cos_w2, 0.01)
    inits.append(cos_w2 / cos_w2.sum())
    
    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # U-shaped
    u_shape = ((2 * t - 1) ** 2) + 0.1
    inits.append(u_shape / u_shape.sum())
    
    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def optimize_single(a0, n, deadline, seed):
    """Run full optimization pipeline on a single initialization."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t_start = time.monotonic()
    total_time = deadline - t_start
    
    # Stage 1: FISTA (30% time)
    t1 = t_start + 0.30 * total_time
    if time.monotonic() < t1:
        a_f, M_f = run_fista_adaptive(a, 5000, t1, tau_start=5.0, tau_end=300.0)
        if M_f < best_M:
            best_M = M_f
            best_a = a_f.copy()
        # Second FISTA pass with higher tau
        if time.monotonic() < t1:
            a_f2, M_f2 = run_fista_adaptive(best_a, 3000, t1, tau_start=100.0, tau_end=800.0)
            if M_f2 < best_M:
                best_M = M_f2
                best_a = a_f2.copy()
        
        print(f"    FISTA: M={best_M:.10f} t={time.monotonic()-t_start:.0f}s", flush=True)
    
    # Stage 2: Coordinate descent (30% time)
    t2 = t_start + 0.60 * total_time
    if time.monotonic() < t2:
        a_cd, J_cd = run_coord_descent_fast(best_a, 50, t2, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best_a = a_cd.copy()
        print(f"    CD: M={best_M:.10f} t={time.monotonic()-t_start:.0f}s", flush=True)
    
    # Stage 3: LP polish (25% time)
    t3 = t_start + 0.85 * total_time
    if time.monotonic() < t3:
        a_lp, M_lp = run_lp_polish(best_a, t3, n_active_start=10, n_active_max=50)
        if M_lp < best_M:
            best_M = M_lp
            best_a = a_lp.copy()
        print(f"    LP: M={best_M:.10f} t={time.monotonic()-t_start:.0f}s", flush=True)
    
    # Stage 4: Final CD + FISTA refinement (15% time)
    if time.monotonic() < deadline:
        a_cd2, _ = run_coord_descent_fast(best_a, 30, min(deadline - 5, time.monotonic() + 0.4 * (deadline - time.monotonic())), seed=seed + 1)
        M_cd2 = fast_conv(a_cd2).max()
        if M_cd2 < best_M:
            best_M = M_cd2
            best_a = a_cd2.copy()
        
        if time.monotonic() < deadline - 5:
            a_f3, M_f3 = run_fista_adaptive(best_a, 2000, deadline - 2, tau_start=200.0, tau_end=1000.0)
            if M_f3 < best_M:
                best_M = M_f3
                best_a = a_f3.copy()
    
    return best_a, best_M


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: diverse inits -> FISTA -> CD -> LP -> refinement.
    Tries multiple lengths and initializations with careful time management.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768", flush=True)
    
    n_parent = parent.size
    
    # Determine lengths to try
    lengths = [n_parent]
    for delta in [-64, 64, -128, 128, 256, -256]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl not in lengths:
            lengths.append(nl)
    
    # Sort: parent first, then by closeness
    lengths.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    lengths = lengths[:5]  # At most 5 lengths
    
    print(f"\n{'='*60}", flush=True)
    print(f"Search plan: lengths={lengths}, budget={budget_s}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Phase 1: Multi-length, multi-init search
    # Reserve time for final polish
    polish_time = 200
    search_time = max(100, (deadline - time.monotonic()) - polish_time)
    
    # Allocate time per length (more time for parent length)
    n_lengths = len(lengths)
    # Weight: parent gets 40%, others split remaining 60%
    weights = np.zeros(n_lengths)
    for i, nl in enumerate(lengths):
        if nl == n_parent:
            weights[i] = 0.4
        else:
            weights[i] = 0.6 / (n_lengths - 1) if n_lengths > 1 else 0.6
    weights = weights / weights.sum()
    
    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - polish_time - 30:
            break
        time_alloc = search_time * weights[li]
        t0 = time.monotonic()
        print(f"\n  --- Length n={n}, time={time_alloc:.0f}s ---", flush=True)
        
        # Generate inits
        cur_parent = tracker.best if tracker.best is not None else parent
        if cur_parent.size != n:
            cur_parent_n = resample_seq(cur_parent, n)
        else:
            cur_parent_n = cur_parent.copy()
        
        inits = generate_inits(n, rng, cur_parent_n)
        
        # Time per init
        time_per_init = time_alloc / max(1, len(inits))
        time_per_init = max(time_per_init, 15.0)  # minimum 15s per init
        
        best_for_n = None
        best_M_for_n = float('inf')
        
        for ii, a0 in enumerate(inits):
            if time.monotonic() > deadline - polish_time - 30:
                break
            sub_deadline = t0 + time_per_init * (ii + 1)
            sub_deadline = min(sub_deadline, deadline - polish_time - 30)
            if time.monotonic() >= sub_deadline:
                break
            
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            
            a_opt, M_opt = optimize_single(a0, n, sub_deadline, seed=seed + li * 100 + ii)
            
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best M={best_M_for_n:.10f} J={J_val:.6f} "
                  f"(elapsed={time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_for_n)
    
    tracker.report("After Phase 1 (multi-length)")
    
    # Phase 2: Intensive polish on the best
    if tracker.best is not None:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 2: Intensive polish", flush=True)
        print(f"{'='*60}", flush=True)
        
        best_n = tracker.best.size
        t_polish_start = time.monotonic()
        t_polish_deadline = deadline - 10
        
        # Multiple rounds of CD + LP
        a_cur = tracker.best.copy()
        for rnd in range(5):
            if time.monotonic() > t_polish_deadline - 20:
                break
            # CD
            cd_time = min(30.0, (t_polish_deadline - time.monotonic()) / 3)
            a_cd, _ = run_coord_descent_fast(a_cur, 30, time.monotonic() + cd_time, seed=seed + 500 + rnd)
            M_cd = fast_conv(a_cd).max()
            if M_cd < fast_conv(a_cur).max():
                a_cur = a_cd
                tracker.update(a_cur)
            
            # LP
            lp_time = min(30.0, (t_polish_deadline - time.monotonic()) / 3)
            if time.monotonic() < t_polish_deadline - 10:
                a_lp, M_lp = run_lp_polish(a_cur, time.monotonic() + lp_time,
                                           n_active_start=15, n_active_max=60)
                if M_lp < fast_conv(a_cur).max():
                    a_cur = a_lp
                    tracker.update(a_cur)
            
            # FISTA refinement
            if time.monotonic() < t_polish_deadline - 10:
                fista_time = min(20.0, t_polish_deadline - time.monotonic() - 10)
                a_f, M_f = run_fista_adaptive(a_cur, 3000, time.monotonic() + fista_time,
                                             tau_start=100.0, tau_end=500.0)
                if M_f < fast_conv(a_cur).max():
                    a_cur = a_f
                    tracker.update(a_cur)
        
        print(f"  Polish elapsed: {time.monotonic()-t_polish_start:.1f}s", flush=True)
    
    tracker.report("After Phase 2 (polish)")
    
    # Phase 3: Try a few more diverse approaches on the best
    if tracker.best is not None and time.monotonic() < deadline - 20:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 3: Diverse refinement", flush=True)
        print(f"{'='*60}", flush=True)
        
        best_n = tracker.best.size
        rng2 = np.random.default_rng(seed + 9999)
        
        # Try perturbations of the best
        for pert_size in [0.03, 0.01, 0.005]:
            if time.monotonic() > deadline - 20:
                break
            a_pert = np.maximum(tracker.best + rng2.normal(0, pert_size, best_n), 1e-6)
            a_pert = a_pert / a_pert.sum()
            # Quick optimize
            sub_dl = min(deadline - 15, time.monotonic() + 40)
            a_opt, M_opt = optimize_single(a_pert, best_n, sub_dl, seed=seed + 7777)
            tracker.update(a_opt)
        
        # Try length change
        for mult in [1.25, 0.75]:
            new_n = int(best_n * mult)
            new_n = max(256, min(2048, new_n))
            if new_n == best_n:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 10, time.monotonic() + 50)
            a_opt, M_opt = optimize_single(a_ext, new_n, sub_dl, seed=seed + 8888)
            tracker.update(a_opt)
    
    tracker.report("After Phase 3 (diverse)")
    
    # Final safety: ensure we have a valid result
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
    
    # Final evaluation
    final_score = float(evaluate_sequence(tracker.best.tolist()))
    if np.isfinite(final_score):
        tracker.best_score = final_score
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
