Let me also think about a different approach: the "spectral flatness" method.

The idea: we want the DFT of a (zero-padded to length N) to have a flat magnitude spectrum. If |A[f]|^2 is flat, then the circular autocorrelation is flat (all zeros except at lag 0), which means the peak is minimized.

The iterative reweighted approach:

3. Set |A_new| = 1/sqrt(N) (flat)
4. A_new = |A_new| * exp(i*angle(A))
5. a_new = IDFT(A_new)
6. Project a_new to be non-negative (clip)
7. Repeat

This is like the Gerchberg-Saxton algorithm. But the projection to non-negative will distort the spectrum, so it won't converge to a perfectly flat spectrum. However, it should drive the peak down.

An alternative: instead of projecting to non-negative, use a "soft" projection: a_new = sigmoid(a_raw / temperature) scaled to sum to 1. As temperature decreases, this approaches the hard projection.

def fast_conv(a):

        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def compute_gradient_conv(a, v, n):
    """
    Compute grad_a where grad_a[i] = 2 * sum_m v[i+m] * a[m].
    v has length 2n-1, a has length n.
    Returns array of length n.
    """
    # We want h[i] = sum_{m=0}^{n-1} v[i+m] * a[m] for i = 0, ..., n-1
    # This is the cross-correlation of v and a, which can be computed via:
    # h = (v * a_rev)[n-1 : 2n-1] where a_rev = a[::-1]
    # using linear convolution of v (len 2n-1) and a_rev (len n)
    # Result has length (2n-1) + n - 1 = 3n-2
    # We want elements [n-1, n, ..., 2n-2] (n elements)

    size = 1

    V = np.fft.rfft(v, size)
    A = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(V * A, size)
    h = conv[n-1 : 2*n-1]  # n elements
    return 2.0 * h

    sweep = 0
    while time.monotonic() < deadline and sweep < n_sweeps_max:
        sweep += 1
        order = rng.permutation(n) if sweep % 3 != 0 else range(n)
        improved = False
        
        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            has_kq = kq <= 2 * n - 2
            
            candidates = np.unique(np.concatenate([[0.0], multipliers * old]))
            if old < 1e-10:
                candidates = np.unique(np.concatenate([candidates, [1e-4, 1e-3, 1e-2, 0.01/n]]))
            
            best_val = old
            best_local = 2.0 * n * conv.max()
            
            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if has_kq:
                    conv[kq] += delta * delta
                
                new_obj = 2.0 * n * conv.max()
                
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if has_kq:
                    conv[kq] -= delta * delta
            
            if best_val != old:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if has_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                improved = True
        
        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
    
    best_a = np.maximum(best_a, 0)
    if best_a.sum() > 1e-10:
        best_a /= best_a.sum()
    return best_a


def spectral_flatten(a, deadline, n_iters=1000):
    """Spectral flattening via Gerchberg-Saxton-like iterations."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute spectrum
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude (with DC preserved)
        # Use a "soft" target: blend between current and flat
        alpha = 0.3 + 0.5 * min(1.0, it / 100.0)
        target_mag = np.ones_like(mag)
        target_mag[0] = mag[0]  # Preserve DC (sum)
        new_mag = (1 - alpha) * mag + alpha * target_mag
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            a_new = a
        else:
            a_new = a_new / a_new.sum()
        
        obj = 2.0 * n * fast_conv(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def lp_polish(a, deadline):
    """LP-based mirror descent polish (from previous approach, streamlined)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    no_improve = 0
    
    while time.monotonic() < deadline and no_improve < 50:
        c = fast_conv(a)
        M = c.max()
        
        # Build LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        
        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 30, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                if g.sum() > 1e-10:
                    g /= g.sum()
                    found = False
                    for frac in [0.3, 0.2, 0.1, 0.05]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = 2.0 * n * fast_conv(a_try).max()
                        if J_try < best_obj - 1e-10:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1
    
    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        if p.sum() > 1e-10:
            p = p / p.sum()
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))
    
    inits.append(np.ones(n) / n)
    
    # Cosine patterns
    for freq in [1, 2, 3, 4, 6]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
    
    # Edge-concentrated
    for width in [0.2, 0.3, 0.5]:
        inits.append(project_simplex(np.exp(-((t - 0) / width)**2) + 0.1))
        inits.append(project_simplex(np.exp(-((t - 1) / width)**2) + 0.1))
    
    # Two-bump
    for sep in [0.3, 0.4, 0.5]:
        inits.append(project_simplex(np.exp(-((t-sep/2)/0.1)**2) + np.exp(-((t-1+sep/2)/0.1)**2) + 0.05))
    
    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))
    
    return inits


def optimize_sequence(a0, deadline):
    """Full optimization pipeline for a single initial sequence."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_J(a)
    
    def update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
