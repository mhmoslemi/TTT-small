**Algorithm: Pairwise Minimax Optimization**

1. Normalize a, sum = 1.
2. For each pair (i,j) (or a random subset):
   a. Compute the current conv.
   b. For the pair, find the optimal split (a_i', a_j') with a_i' + a_j' = a_i + a_j that minimizes the peak of the modified conv.
   c. Apply the split.
3. Repeat until convergence.

The 1D minimax for a pair:
- conv_k(new) = conv_k(old) + δ * (a[k-i] - a[k-j]) where δ = a_i' - a_i
- We want min_δ max_k [conv_k + δ * (a[k-i] - a[k-j])]
- subject to max(0, a_i+δ) + max(0, a_j-δ) = c (or a_i' >= 0, a_j' >= 0)

This is a linear minimax problem in δ! The optimal δ is where the maximum is minimized, which occurs at the intersection of the two "active" lines.

This can be solved in O(n) time per pair (since there are 2n-1 convolution entries, and we need to find the δ that minimizes the max of 2n-1 linear functions of δ).

This is a 1D linear programming problem, solvable in O(n) time.

```python
def pairwise_minimax_step(a, n_pairs=50, rng=None):
    """Optimize pairs of elements to minimize the convolution peak."""
    n = len(a)
    conv = compute_conv(a)
    current_peak = np.max(conv)
    
    for _ in range(n_pairs):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        c = a[i] + a[j]
        # conv_k change = δ * (a[k-i] - a[k-j]) where a[k-out] = 0
        diff = np.zeros(2*n-1)
        for k in range(2*n-1):
            ki = k - i
            kj = k - j
            val_i = a[ki] if 0 <= ki < n else 0
            val_j = a[kj] if 0 <= kj < n else 0
            diff[k] = val_i - val_j
        
        # Find δ that minimizes max_k (conv[k] + δ * diff[k])
        # This is: min_δ max_k (conv[k] + δ * diff[k])
        # subject to 0 <= a[i] + δ <= c (i.e., -a[i] <= δ <= a[j])
        
        # Linear minimax: find δ* that minimizes the max
        # The optimal δ* is where the two largest lines cross
        # Lines: f_k(δ) = conv[k] + δ * diff[k]
        # We need min_δ max_k f_k(δ)
        
        # For efficiency, only consider k where diff[k] != 0
        active = np.nonzero(np.abs(diff) > 1e-15)[0]
        if len(active) == 0:
            continue
        
        # The minimum of the max of linear functions occurs at a vertex
        # of the upper envelope. We can find it by computing the upper hull.
        # Simpler: use the fact that the optimal δ is where two lines cross
        # and the max is minimized.
        
        # Use binary search on the target level T:
        # Can we make all conv[k] + δ*diff[k] <= T for some δ in [-a[i], a[j]]?
        # For each k: if diff[k] > 0, δ <= (T - conv[k]) / diff[k]
        #              if diff[k] < 0, δ >= (T - conv[k]) / diff[k]
        #              if diff[k] = 0, need conv[k] <= T
        # So we need: max_k [(T - conv[k]) / diff[k]] for diff[k] < 0
        #            <= min_k [(T - conv[k]) / diff[k]] for diff[k] > 0
        # And also -a[i] <= δ <= a[j]
        
        # Binary search on T:
        lo = current_peak  # can't do better than current peak (approximately)
        hi = np.max(conv) + np.max(np.abs(diff)) * c  # upper bound
        
        # Actually, let's just do a few iterations of the subgradient method on δ
        # Or use the fact that this is a 1D LP
        
        # Better: compute the optimal δ directly
        # The value function v(δ) = max_k (conv[k] + δ*diff[k]) is convex in δ
        # We want to minimize v(δ) over δ ∈ [-a[i], a[j]]
        # v'(δ) = diff[k*(δ)] where k*(δ) = argmax_k (conv[k] + δ*diff[k])
        # So v is piecewise linear, and the minimum is at a kink point or boundary
        
        # Find all kink points: where two lines cross
        # Line k and line l cross at: δ = (conv[l] - conv[k]) / (diff[k] - diff[l])
        
        # This is O(n^2) in the number of active lines, which might be too slow
        # for large n. Let's use a simpler approach.
        
        # Simple approach: ternary search on δ (since v is convex)
        lo_d = -a[i]
        hi_d = a[j]
        
        def v(delta):
            new_conv = conv + delta * diff
            return np.max(new_conv)
        
        # Ternary search
        for _ in range(30):
            m1 = lo_d + (hi_d - lo_d) / 3
            m2 = hi_d - (hi_d - lo_d) / 3
            if v(m1) < v(m2):
                hi_d = m2
            else:
                lo_d = m1
        
        delta_opt = (lo_d + hi_d) / 2
        new_peak = v(delta_opt)
        
        if new_peak < current_peak - 1e-15:
            a[i] += delta_opt
            a[j] -= delta_opt
            a = np.maximum(a, 0)
            # Renormalize (shouldn't be needed since we preserved the sum)
            current_peak = new_peak
    
    return a
```

```python
def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    r = rho_idx[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(conv, beta):
    """Compute softmax weights for smooth max approximation."""
    m = float(np.max(conv))
    shifted = beta * (conv - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= s
    return w


def smooth_gradient(a, p):
    """Gradient of (1/beta)*log(sum(exp(beta*conv_k))) w.r.t. a.
    p is the softmax weight vector of length 2n-1.
    """
    n = len(a)
    a_flip = a[::-1]
    size = 1
    while size < len(p) + n:
        size <<= 1
    Pf = rfft(p, size)
    Af = rfft(a_flip, size)
    corr = irfft(Pf * Af, size)

def nesterov_optimize(a, beta=50.0, max_steps=500, time_budget=200.0,
                      eta_init=0.1, restart_every=100):
    """Nesterov-accelerated gradient descent on log-sum-exp smooth max."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    a_prev = a.copy()
    stalled = 0

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        p = softmax_weights(conv, beta)
        g = smooth_gradient(a, p)

        # Nesterov momentum
        if step > 0:
            alpha = (step - 1) / (step + 2.0)
            if step % restart_every == 0:
                alpha = 0.0
            y = a + alpha * (a - a_prev)
        else:
            y = a.copy()

        # Project y to simplex (for validity)
        y = project_simplex(y)

        # Gradient at y
        conv_y = compute_conv(y)
        p_y = softmax_weights(conv_y, beta)
        g_y = smooth_gradient(y, p_y)

        # Center gradient (project onto sum=1 hyperplane)
        g_y = g_y - np.mean(g_y)

        # Backtracking line search
        a_new = None
        for backtrack in range(20):
            a_trial = project_simplex(y - eta * g_y)
            obj_trial = compute_obj(a_trial)
            if obj_trial < compute_obj(y) + 1e-4:  # simplified Armijo
                a_new = a_trial
                break
            eta *= 0.5

        if a_new is None:
            eta = max(eta, 1e-6)
            a_new = project_simplex(y - eta * g_y)

        a_prev = a.copy()
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stalled = 0
        else:
            stalled += 1

        if stalled >= 20:
            # Restart momentum, increase beta slightly
            a_prev = a.copy()
            beta *= 1.1
            eta = eta_init
            stalled = 0

    return best_a, best_obj


def pairwise_minimax_optimize(a, n_sweeps=200, pairs_per_sweep=100,
                               time_budget=100.0):

    n = len(a)

    best_a = a.copy()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated gradient descent (log-sum-exp), pairwise minimax,
LP alternating, and spectral equalization with multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return np.maximum(conv[:2 * n - 1], 0.0)


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    r = idx[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(conv, beta):
    m = float(np.max(conv))
    shifted = np.clip(beta * (conv - m), -500.0, 500.0)
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= s
    return w

    Af = rfft(a_flip, size)
    corr = irfft(Pf * Af, size)
    return 2.0 * corr[n - 1:2 * n - 1]
