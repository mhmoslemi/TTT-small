def projected_subgradient(a_init, deadline, tau_start=5, tau_end=500, max_iters=50000, seed=0):
    """Projected subgradient with log-sum-exp smoothing and adaptive step size."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Barzilai-Borwein state
    x_prev = a.copy()
    g_prev = None
    step = 1.0 / np.sqrt(n)
    
    no_improve = 0
    it = 0
    
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        # Geometric tau schedule
        tau = tau_start * (tau_end / tau_start) ** frac
        
        smooth_val, grad = smooth_max_grad(a, tau)
        # Project gradient onto tangent space of simplex
        grad_tan = grad - grad.mean()
        
        # Barzilai-Borwein step size
        if g_prev is not None:
            s_vec = a - x_prev
            y_vec = grad_tan - g_prev
            s_norm = np.dot(s_vec, s_vec)
            y_norm = np.dot(y_vec, y_vec)
            sy = np.dot(s_vec, y_vec)
            if s_norm > 1e-15 and y_norm > 1e-15:
                if sy > 0:
                    step_bb = s_norm / sy  # BB1
                else:
                    step_bb = 1.0 / y_norm  # BB2 (for non-convex)
                step = 0.5 * (step + np.clip(step_bb, 1e-8, 10.0 / tau))
        
        a_new = a - step * grad_tan
        a_new = project_simplex(a_new)
        
        x_prev = a.copy()
        g_prev = grad_tan.copy()
        a = a_new
        
        it += 1
        
        # Periodic evaluation
        if it % 10 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M - 1e-12:
                best_M = M_cur
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Restart if stuck
        if no_improve > 200:
            # Restart from best with perturbation
            a = best_a + rng.normal(0, 0.01 * best_a.mean(), n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
            step = 1.0 / np.sqrt(n)
            g_prev = None
    
    return best_a, best_M

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Projected subgradient with adaptive log-sum-exp smoothing and BB step sizes
2. LP best-response with greedy active set expansion

6. Frequency-domain informed initialization

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_hat = int(np.max(rho[cond]))
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad(a, tau):
    """Log-sum-exp of conv(a,a) and gradient w.r.t. a. Returns (value, grad)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    e = np.exp(tau * (b - b_max))
    e_sum = e.sum()
    p = e / e_sum
    val = b_max + np.log(e_sum) / tau
    # grad[i] = 2 * sum_j p[i+j] * a[j]
    grad = np.zeros(n)
    # Vectorized: for each i, dot p[i:i+n] with a (trimmed appropriately)
    # Use a sliding dot product via FFT for large n, direct for small
    if n <= 3000:
        # Direct: grad[i] = 2 * sum_{j=0}^{min(n-1, 2n-2-i)} p[i+j] * a[j]
        # Build a matrix of views for efficiency
        for i in range(n):
            jmax = min(n, 2 * n - 1 - i)
            if jmax > 0:
                grad[i] = 2.0 * np.dot(p[i:i + jmax], a[:jmax])
    else:
        # FFT approach: grad[i] = 2 * (p_reversed_convolve_a)
        # grad[i] = 2 * sum_j p[i+j] * a[j]
        # = 2 * ifft(fft(p) * fft(a_flipped))[i + n - 1] ... 
        # Use direct for simplicity (n won't exceed 3000 in practice)
        for i in range(n):
            jmax = min(n, 2 * n - 1 - i)
            if jmax > 0:
                grad[i] = 2.0 * np.dot(p[i:i + jmax], a[:jmax])
    return val, grad


def run_subgradient(a_init, deadline, tau_start=5.0, tau_end=500.0, max_iters=100000, seed=0):
    """Projected subgradient with BB step size and geometric tau."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    x_prev = a.copy()
    g_prev = None
    step = 1.0 / (tau_start * np.sqrt(n))
    no_improve = 0
    it = 0
    last_eval = 0
    
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad = smooth_max_grad(a, tau)
        grad_tan = grad - grad.mean()
        
        # BB step size
        if g_prev is not None:
            s_vec = a - x_prev
            y_vec = grad_tan - g_prev
            s_sq = np.dot(s_vec, s_vec)
            y_sq = np.dot(y_vec, y_vec)
            sy = np.dot(s_vec, y_vec)
            if s_sq > 1e-14 and y_sq > 1e-14:
                if sy > 1e-10:
                    step_bb = s_sq / sy
                else:
                    step_bb = 1.0 / y_sq
                step_bb = np.clip(step_bb, 1e-10, 50.0 / tau)
                step = 0.5 * (step + step_bb)
        
        a_new = a - step * grad_tan
        a_new = project_simplex(a_new)
        
        x_prev = a.copy()
        g_prev = grad_tan.copy()
        a = a_new
        it += 1
        
        if it >= last_eval:
            M_cur = fast_conv(a).max()
            if M_cur < best_M - 1e-13:
                best_M = M_cur
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            # Adaptive evaluation frequency
            if no_improve < 50:
                last_eval = it + 5
            elif no_improve < 200:
                last_eval = it + 10
            else:
                last_eval = it + 20
        
        # Restart from best with perturbation
        if no_improve > 500:
            a = best_a + rng.normal(0, 0.02 / np.sqrt(n), n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
            step = 1.0 / (tau * np.sqrt(n))
            g_prev = None
            last_eval = it
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a, best_M


def lp_best_response(a, M, n_active=30, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top n_active k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    top_idx = np.argsort(b)[::-1][:min(n_active, len(b))]
    active = top_idx.tolist()
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def run_lp_iterations(a_init, deadline, n_active_start=20, max_rounds=100):

            continue

        for t in [0.3, 0.5, 0.7, 0.2, 0.1, 0.8, 0.9]:

    return best_a, best_M


def coordinate_descent_1d(a, i, b, n):
    """Optimize a[i] via golden section on max convolution."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c_vals.append(b[k] - x_cur * x_cur)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c_vals.append(b[k] - 2.0 * x_cur * a[j])
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(3.0 * x_cur, 0.5, np.sqrt(max(b.max(), 0.0)) + 0.5)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt) + 1e-12:
        x_opt = 0.0
    
    return max(0.0, x_opt)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a).copy()
        sweep_changed = False
        
        for i in order:

        M = fast_conv(a).max()
        
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_changed:
            no_improve += 5
        if no_improve > 30:
            break
    
    return best_a, best_M


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.02, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-5)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t + 0.5 / n)))
    hann = np.maximum(hann, 1e-4)
    inits.append(hann / hann.sum())
    
    # Raised cosine (wider)
    rc = 0.5 * (1.0 + np.cos(np.pi * (2 * t - 1)))
    rc = np.maximum(rc, 1e-4)
    inits.append(rc / rc.sum())
    
    # Sinc-like (main lobe only, clipped)
    center = (n - 1) / 2.0
    sinc = np.sinc((np.arange(n) - center) / (n * 0.4))
    sinc = np.maximum(sinc, 0.0)
    # Add small floor to avoid zeros
    sinc = sinc + 0.01 * sinc.max()
    inits.append(sinc / sinc.sum())
    
    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.02
    inits.append(gauss / gauss.sum())
    
    # Cosine (single period)
    cos1 = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    cos1 = np.maximum(cos1, 0.01)
    inits.append(cos1 / cos1.sum())
    
    # Cosine squared
    cos2 = (0.5 + 0.5 * np.cos(2.0 * np.pi * t)) ** 2
    cos2 = np.maximum(cos2, 0.005)
    inits.append(cos2 / cos2.sum())
    
    # "U-shaped" (convex)
    u_shape = 0.1 + (2.0 * t - 1.0) ** 2
    inits.append(u_shape / u_shape.sum())
    
    # Two-lobe
    two_lobe = np.exp(-6.0 * (t - 0.25) ** 2) + np.exp(-6.0 * (t - 0.75) ** 2) + 0.02
    inits.append(two_lobe / two_lobe.sum())
    
    # Smooth random (low frequency)
    ksize = max(3, n // 16)
    kernel = np.hanning(ksize)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0.3, 1.0, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Spectral flatness init: iterate to flatten |FFT|^2
    a_spec = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_spec)
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.15 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.005)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    
    return inits


def run_optimization_pipeline(a0, n, deadline, seed=0):
    """Full optimization pipeline: subgradient -> LP -> coordinate descent."""
    best = a0.copy()
    best_M = fast_conv(a0).max()
    t_start = time.monotonic()
    phase_duration = max(5.0, deadline - t_start)
    
    # Stage 1: Subgradient (50% time)
    t1 = t_start + 0.5 * phase_duration
    if time.monotonic() < t1:
        a_sg, M_sg = run_subgradient(best, t1, tau_start=5, tau_end=500,
                                     max_iters=50000, seed=seed)
        if M_sg < best_M:
            best_M = M_sg
            best = a_sg.copy()
    
    # Stage 2: LP iterations (25% time)
    t2 = t_start + 0.75 * phase_duration
    if time.monotonic() < t2:
        a_lp, M_lp = run_lp_iterations(best, t2, n_active_start=20, max_rounds=200)
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
    
    # Stage 3: Coordinate descent (20% time)
    t3 = t_start + 0.95 * phase_duration
    if time.monotonic() < t3:
        a_cd, M_cd = run_coordinate_descent(best, t3, seed=seed + 1, max_sweeps=100)
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
    
    # Stage 4: Final LP polish (5% time)
    if time.monotonic() < deadline:
        a_final, M_final = run_lp_iterations(best, deadline, n_active_start=15, max_rounds=50)
        if M_final < best_M:
            best_M = M_final
            best = a_final.copy()
    
    return best, best_M


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self.history = []

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0:
            return
        s = a.sum()
        if s < 0.01:
            return
        score = compute_J(a)
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            self.history.append((score, a.size, elapsed))
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: multi-start subgradient -> LP -> coordinate descent,
    across multiple sequence lengths with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent, "(parent)")
                print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(768) * 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(768) / 768.0))
        parent = np.maximum(parent, 1e-4)
        parent = parent / parent.sum()
        tracker.update(parent, "(default)")
    
    n_parent = parent.size if parent.size > 0 else 768
    print(f"Starting search. Parent n={n_parent}, J={tracker.best_score:.8f}", flush=True)
    
    # === Phase 1: Multi-length quick exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Choose lengths to try
    lengths = []
    for nl in [n_parent]:
        if 256 <= nl <= 2048:
            lengths.append(nl)
    for nl in [n_parent - 128, n_parent - 64, n_parent + 64, n_parent + 128]:
        if 256 <= nl <= 2048 and nl not in lengths:
            lengths.append(nl)
    # Ensure diversity
    for nl in [512, 640, 768, 896, 1024]:
        if nl not in lengths and len(lengths) < 6:
            lengths.append(nl)
    
    lengths = lengths[:5]
    lengths.sort(key=lambda x: abs(x - n_parent))
    
    print(f"Lengths to try: {lengths}", flush=True)
    
    t1_end = time.monotonic() + min(400, (deadline - time.monotonic()) * 0.3)
    time_per_length = (t1_end - time.monotonic()) / len(lengths)
    
    for li, n in enumerate(lengths):
        if time.monotonic() > t1_end:
            break
        t0 = time.monotonic()
        t_alloc = min(time_per_length, t1_end - time.monotonic())
        if t_alloc < 15:
            break
        t_alloc = max(t_alloc, 15)
        
        print(f"\n  n={n}, alloc={t_alloc:.0f}s", flush=True)
        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        
        best_for_n = None
        best_M_for_n = float('inf')
        time_per_init = t_alloc / len(inits)
        
        for ii, a0 in enumerate(inits):
            sub_dl = t0 + (ii + 1) * time_per_init
            if time.monotonic() > sub_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            
            # Quick subgradient (short)
            a_opt, M_opt = run_subgradient(a0, sub_dl, tau_start=10, tau_end=200,
                                           max_iters=20000, seed=seed + li * 100 + ii)
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best M={best_M_for_n:.10f} J={J_val:.6f} "
                  f"(elapsed={time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_for_n, f"(P1 n={n})")
    
    tracker.report("After Phase 1")
    
    # === Phase 2: Deep optimization on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Deep optimization", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 100:
        n_best = tracker.best.size
        t2_end = min(deadline - 100, time.monotonic() + 600)
        
        # Run full pipeline on best
        a_opt, M_opt = run_optimization_pipeline(tracker.best, n_best, t2_end, seed=seed + 1000)

        # Additional subgradient with different tau range
        if time.monotonic() < deadline - 100:

                                         tau_start=50, tau_end=2000,
                                         max_iters=50000, seed=seed + 2000)
            tracker.update(a_sg, "(P2 high-tau)")
        
        # LP with larger active set
        if time.monotonic() < deadline - 100:
            a_lp, M_lp = run_lp_iterations(tracker.best, min(deadline - 100, time.monotonic() + 150),
                                           n_active_start=50, max_rounds=100)
            tracker.update(a_lp, "(P2 LP-large)")
    
    tracker.report("After Phase 2")
    
    # === Phase 3: Coordinate descent + fine LP ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Fine-tuning", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 30:
        # Coordinate descent
        a_cd, M_cd = run_coordinate_descent(tracker.best, min(deadline - 30, time.monotonic() + 150),
                                            seed=9999, max_sweeps=200)
        tracker.update(a_cd, "(P3 CD)")

        if time.monotonic() < deadline - 30:
            a_lp, M_lp = run_lp_iterations(tracker.best, min(deadline - 30, time.monotonic() + 60),
                                           n_active_start=10, max_rounds=50)

        # Another CD pass
        if time.monotonic() < deadline - 20:

                                                  seed=8888, max_sweeps=100)
            tracker.update(a_cd2, "(P3 CD2)")
    
    tracker.report("After Phase 3")
    
    # === Phase 4: Try different lengths from best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_cur = tracker.best.size
        for mult in [1.3, 0.7, 1.15, 0.85]:
            if time.monotonic() > deadline - 30:
                break
            new_n = int(n_cur * mult)
            new_n = max(256, min(2048, new_n))

            a_ext = resample_seq(tracker.best, new_n)

                                           max_iters=20000, seed=seed + 5000)

            if new_n2 != n_cur:

                                     tau_start=500, tau_end=2000,
                                     max_iters=20000, seed=77777)
        tracker.update(a_gt, "(P5 high-tau)")
        
        # Final coordinate descent
        if time.monotonic() < deadline - 5:
            a_fc, M_fc = run_coordinate_descent(tracker.best, deadline - 5,
                                                seed=12345, max_sweeps=50)

    # Final validation
    if tracker.best is None:

          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
