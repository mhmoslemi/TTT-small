def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale-invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def soft_max_gradient_fast(a, tau):
    """
    Compute gradient of soft-max of conv(a,a) using log-sum-exp.
    Returns (gradient, conv, cmax).
    Uses efficient FFT-based correlation.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500.0, 500.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # grad[m] = 2 * sum_j a[j] * weights[j+m] for m=0..n-1
    # This is 2 * np.correlate(weights, a, mode='valid')
    grad = 2.0 * np.correlate(weights, a, mode='valid')

    return grad, conv, cmax


def fista_soft_max(a_init, max_steps=500, tau_init=0.005, tau_decay=0.995,
                   time_budget=120.0, step_mode='nesterov'):
    """
    FISTA (Nesterov-accelerated) descent on the soft-max objective.
    Minimize soft_max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_init
    # Nesterov momentum
    beta = 0.9

    # Initialize: compute obj at current point
    conv0 = compute_conv(a)
    cmax0 = float(np.max(conv0))

    # FISTA variables
    y = a.copy()  # auxiliary variable
    t_k = 1.0     # momentum parameter

    # Compute initial gradient for step size estimation
    grad0, _, _ = soft_max_gradient_fast(y, tau)
    grad_norm0 = float(np.linalg.norm(grad0))
    if grad_norm0 < 1e-15:
        grad_norm0 = 1.0
    L_init = n * cmax0 / max(grad_norm0 * grad_norm0, 1e-10) * 2.0
    L = max(L_init, 1.0)

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        # Compute gradient at y
        grad_y, _, _ = soft_max_gradient_fast(y, tau)

        # Gradient step with backtracking
        L = L_init * 0.5
        x = y.copy()
        for _bt in range(20):
            x_trial = y - grad_y / L
            x_trial = np.maximum(x_trial, 0.0)
            s = float(np.sum(x_trial))
            if s < 1e-15:
                break
            x_trial = x_trial / s

            grad_x, conv_x, _ = soft_max_gradient_fast(x_trial, tau)
            # Check Lipschitz: ||grad(x_trial) - grad(y)|| <= L * ||x_trial - y||
            diff_grad = np.linalg.norm(grad_x - grad_y)
            diff_x = np.linalg.norm(x_trial - y)
            if diff_x < 1e-15:
                break
            if diff_grad <= L * diff_x + 1e-8:
                x = x_trial
                break
            L *= 1.5

        # Nesterov update for y
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y = x + beta * (x - a)
        y = np.maximum(y, 0.0)
        s_y = float(np.sum(y))
        if s_y > 1e-15:
            y = y / s_y
        a = x
        t_k = t_new

        # Evaluate
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Decay temperature
        tau = max(tau * tau_decay, 1e-6)

    return best_a, best_obj


def water_filling(a_init, max_iters=100, time_budget=60.0, n_peaks=5):
    """
    Water-filling approach: iteratively equalize the top N convolution peaks.
    For each peak, compute the direction that reduces it, combine directions,
    and take a projected step.
    """
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Find top peaks
        top_indices = np.argsort(conv)[::-1][:n_peaks]

        # Compute combined gradient: equal weight on top peaks
        # grad[m] = 2 * sum_{k in top} a[k-m] * I(0<=k-m<n)
        grad = np.zeros(n)
        for k in top_indices:
            k = int(k)
            # grad[m] += 2 * a[k-m] for m where 0 <= k-m < n
            m_min = max(0, k - (n - 1))
            m_max = min(n - 1, k)
            for m in range(m_min, m_max + 1):
                grad[m] += 2.0 * a[k - m]

        # Center the gradient (project onto sum-preserving subspace)
        grad_centered = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = cmax / (gnorm * gnorm) * 0.5
        step = min(step, 0.5)

        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Halve step and retry
            a_new = a - step * 0.3 * grad_centered
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Perturb to escape
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    # A[k, j] = a[j] if k >= j and k-j < n, else 0
    # Actually conv(a,g)[k] = sum_j a[j] * g[k-j]
    # So A[k, j] = a[j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, M, time_limit=20):
    """
    Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns normalized g or None.
    """
    from scipy.optimize import linprog
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate(a_init, max_iters=30, time_budget=60.0):
    """LP alternating with golden section line search."""
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        remaining = time_budget - (time.monotonic() - start)
        g = solve_lp_direction(a, M, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break

        # Golden section search for blend parameter t
        def obj_at_t(t):
            blend = normalize((1.0 - t) * a + t * g)
            return compute_obj(blend)

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = obj_at_t(c_pt)
        fd = obj_at_t(d_pt)
        for _ in range(12):
            if hi - lo < 0.02:
                break
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = obj_at_t(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = obj_at_t(d_pt)
        t_opt = (lo + hi) / 2.0

        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Perturb
            rng = np.random.default_rng((it * 104729 + 42) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def cosine_window(n, power=1.0):
    """Generate a cosine-windowed initialization."""
    j = np.arange(n)
    w = np.cos(np.pi * j / (2.0 * n)) ** (2.0 * power)
    # Ensure non-negative and add small floor
    w = np.maximum(w, 0.01)
    return normalize(w)


def chirp_init(n):
    """Generate a chirp-like initialization."""
    j = np.arange(n)
    # Modulated power
    w = 1.0 + 0.3 * np.sin(2.0 * np.pi * 1.5 * j / n)
    w = np.maximum(w, 0.05)
    return normalize(w)


def random_dirichlet(n, alpha=0.5, rng=None):
    """Generate a random Dirichlet-distributed initialization."""
    if rng is None:
        rng = np.random.default_rng()
    return rng.dirichlet(np.ones(n) * alpha)


def multi_peak_equalize(a_init, max_iters=80, time_budget=60.0, n_peaks=3):
    """
    More aggressive peak equalization: target the top n_peaks simultaneously
    using a weighted gradient that emphasizes the most prominent peaks.
    """
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    tau = 0.01

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Focus on top peaks with exponential weighting
        top_k = min(n_peaks, len(conv))
        top_indices = np.argsort(conv)[::-1][:top_k]
        top_vals = conv[top_indices]
        top_range = top_vals[0] - top_vals[-1] + 1e-10
        weights = np.exp(-top_range / (tau * cmax + 1e-10))
        weights /= np.sum(weights)

        # Compute combined gradient
        grad = np.zeros(n)
        for idx, w in enumerate(zip(top_indices, weights)):
            k = int(w[0])
            wt = w[1]
            m_min = max(0, k - (n - 1))
            m_max = min(n - 1, k)
            for m in range(m_min, m_max + 1):
                grad[m] += 2.0 * wt * a[k - m]

        grad_centered = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-15:
            break

        step = 0.4 * cmax / (gnorm * gnorm + 1e-10)
        step = min(step, 0.3)

        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a_new = a - step * 0.3 * grad_centered
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                tau *= 0.9

    return best_a, best_obj


def coordinate_polish(a_init, time_budget=20.0):
    """Pairwise mass-transfer coordinate descent."""
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            for _ in range(4):
                j_target = int(rng.integers(0, n))
                if j_target in (j1, j2):
                    continue
                a_try = a.copy()
                delta = 0.04 * max(a_try[j1], a_try[j2])
                src = j1 if a_try[j1] >= a_try[j2] else j2
                a_try[src] -= delta
                a_try[j_target] += delta
                if a_try[src] < 0:
                    continue
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            if rng.random() < 0.15:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:
                # Try a different peak
                second_peak = int(np.argsort(conv)[::-1][1])
                a_try = a.copy()
                delta = 0.02 * a_try.mean()
                src = int(np.argmax(a_try))
                dst = int(np.argmin(a_try))
                a_try[src] -= delta
                a_try[dst] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase strategy:
    Phase 1: Diverse initializations + quick FISTA screening
    Phase 2: Intensive FISTA on best candidate
    Phase 3: LP-based refinement
    Phase 4: Water-filling peak equalization
    Phase 5: Multi-n exploration
    Phase 6: Final polish
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={parent_obj:.8f}")
            best_seq = parent.copy()
            best_score = parent_obj
    except Exception:
        pass

    if best_seq is None:
        best_seq = normalize(np.ones(700))
        best_score = compute_obj(best_seq)
        print(f"No parent. Starting with uniform n=700: obj={best_score:.8f}")

    n_parent = len(best_seq)
    rng_global = np.random.default_rng(seed)

    # =========================================================
    # Phase 1: Diverse initializations + quick FISTA
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Diverse initializations")
    print(f"{'='*60}")

    inits = []
    # Parent-based
    inits.append(("parent", normalize(best_seq.copy())))
    # Cosine window at parent n
    inits.append((f"cos_{n_parent}", cosine_window(n_parent, power=1.0)))
    inits.append((f"cos2_{n_parent}", cosine_window(n_parent, power=2.0)))
    # Chirp
    inits.append((f"chirp_{n_parent}", chirp_init(n_parent)))
    # Various n with cosine
    for n_try in [n_parent - 100, n_parent + 100, n_parent + 200, 500, 900, 1100]:
        if n_try >= 200:
            inits.append((f"cos_{n_try}", cosine_window(n_try, power=1.0)))
            inits.append((f"cos2_{n_try}", cosine_window(n_try, power=2.0)))

    # Quick FISTA screening (short budget per init)
    for name, a_init in inits:
        if time.monotonic() > deadline - 600:
            break
        a_screen, obj_screen = fista_soft_max(a_init, max_steps=60,
                                              tau_init=0.004, tau_decay=0.99,
                                              time_budget=15.0)
        if obj_screen < best_score:
            best_score = obj_screen
            best_seq = a_screen.copy()
            print(f"  {name}: {obj_screen:.8f} **")
        else:
            print(f"  {name}: {obj_screen:.8f}")

    # =========================================================
    # Phase 2: Intensive FISTA on best candidate
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Intensive FISTA (n={len(best_seq)})")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    alloc2 = min(remaining * 0.30, 300.0)

    a_best = normalize(best_seq.copy())
    for fista_round in range(3):
        if time.monotonic() > deadline - 300:
            break
        sub_alloc = alloc2 / (3 - fista_round)
        tau_start = 0.005 * (0.7 ** fista_round)
        a_fista, obj_fista = fista_soft_max(a_best, max_steps=600,
                                            tau_init=tau_start, tau_decay=0.997,
                                            time_budget=sub_alloc)
        if obj_fista < best_score:
            best_score = obj_fista
            best_seq = a_fista.copy()
            a_best = a_fista
            print(f"  FISTA round {fista_round}: {best_score:.8f} **")
        else:
            # Blend and try again from different angle
            a_best = normalize(0.7 * a_best + 0.3 * a_fista)
            print(f"  FISTA round {fista_round}: {obj_fista:.8f} (no improvement, blending)")

    # =========================================================
    # Phase 3: LP-based refinement
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: LP refinement (n={len(best_seq)})")
    print(f"{'='*60}")

    n_cur = len(best_seq)
    if n_cur <= 900:
        remaining = deadline - time.monotonic()
        alloc3 = min(remaining * 0.20, 180.0)
        a_lp, obj_lp = lp_alternate(normalize(best_seq.copy()),
                                    max_iters=25, time_budget=alloc3)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: {best_score:.8f} **")
        else:
            print(f"  LP: {obj_lp:.8f} (no improvement)")
    else:
        print(f"  Skipping LP (n={n_cur} too large)")

    # =========================================================
    # Phase 4: Water-filling peak equalization
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Water-filling (n={len(best_seq)})")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    alloc4 = min(remaining * 0.20, 150.0)

    # Water filling with different n_peaks
    for n_peaks in [3, 5, 8]:
        if time.monotonic() > deadline - 150:
            break
        sub_alloc = alloc4 / 3.0
        a_wf, obj_wf = water_filling(normalize(best_seq.copy()),
                                     max_iters=80, time_budget=sub_alloc,
                                     n_peaks=n_peaks)
        if obj_wf < best_score:
            best_score = obj_wf
            best_seq = a_wf.copy()
            print(f"  Water-fill (k={n_peaks}): {best_score:.8f} **")
        else:
            print(f"  Water-fill (k={n_peaks}): {obj_wf:.8f}")

    # Multi-peak equalize
    if time.monotonic() < deadline - 100:
        a_mp, obj_mp = multi_peak_equalize(normalize(best_seq.copy()),
                                           max_iters=80,
                                           time_budget=40.0, n_peaks=4)
        if obj_mp < best_score:
            best_score = obj_mp
            best_seq = a_mp.copy()
            print(f"  Multi-peak: {best_score:.8f} **")
        else:
            print(f"  Multi-peak: {obj_mp:.8f}")

    # Follow up with FISTA
    if time.monotonic() < deadline - 80:
        a_post, obj_post = fista_soft_max(normalize(best_seq.copy()),
                                          max_steps=200, tau_init=0.002,
                                          tau_decay=0.995, time_budget=50.0)
        if obj_post < best_score:
            best_score = obj_post
            best_seq = a_post.copy()
            print(f"  Post WF FISTA: {best_score:.8f} **")

    # =========================================================
    # Phase 5: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq)
    n_options = []
    for delta in [-200, -150, -100, -50, 0, 50, 100, 150, 200, 300, 400, 500]:
        n_try = n_best + delta
        if 300 <= n_try <= 1600:
            n_options.append(n_try)
    # Also add some absolute values
    for n_abs in [400, 550, 700, 850, 1000, 1200, 1400]:
        if n_abs not in n_options and 300 <= n_abs <= 1600:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining_sub = deadline - time.monotonic()
        if remaining_sub < 15:
            break

        if n_try == n_best:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample_sequence(best_seq, n_try))

        # Quick optimization at this n
        alloc_n = min(remaining_sub * 0.15, 30.0)
        if n_try <= 900:
            a_try, _ = fista_soft_max(a_try, max_steps=80, tau_init=0.003,
                                      tau_decay=0.995, time_budget=alloc_n * 0.6)
        a_try, obj_try = water_filling(a_try, max_iters=40,
                                       time_budget=alloc_n * 0.4, n_peaks=4)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 6: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish (n={len(best_seq)})")
    print(f"{'='*60}")

    n_final = len(best_seq)
    a_final = normalize(best_seq.copy())

    for polish_round in range(5):
        if time.monotonic() > deadline - 10:
            break
        remaining_sub = deadline - time.monotonic()
        if remaining_sub < 10:
            break
        alloc_pol = min(remaining_sub * 0.30, 40.0)

        # FISTA
        a_pol, obj_pol = fista_soft_max(a_final, max_steps=200,
                                        tau_init=0.001 * (0.8 ** polish_round),
                                        tau_decay=0.997, time_budget=alloc_pol * 0.5)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            a_final = a_pol

        # Water filling
        if time.monotonic() < deadline - 15:
            a_pol2, obj_pol2 = multi_peak_equalize(a_final, max_iters=50,
                                                    time_budget=alloc_pol * 0.3,
                                                    n_peaks=3)
            if obj_pol2 < best_score:
                best_score = obj_pol2
                best_seq = a_pol2.copy()
                a_final = a_pol2

        # Coordinate descent
        if time.monotonic() < deadline - 12:
            a_pol3, obj_pol3 = coordinate_polish(a_final, time_budget=min(8, alloc_pol * 0.2))
            if obj_pol3 < best_score:
                best_score = obj_pol3
                best_seq = a_pol3.copy()
                a_final = a_pol3

        print(f"  Polish {polish_round}: {best_score:.8f}")

    # =========================================================
    # Final FISTA with very low temperature for fine-tuning
    # =========================================================
    if time.monotonic() < deadline - 15:
        a_final2, obj_final2 = fista_soft_max(normalize(best_seq.copy()),
                                              max_steps=300, tau_init=0.0005,
                                              tau_decay=0.998,
                                              time_budget=min(40, deadline - time.monotonic() - 10))
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final FISTA: {best_score:.8f} **")

    # =========================================================
    # Validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate with the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
        print(f"\n{'='*60}")
        print(f"FINAL: n={len(result)}, score={best_score:.8f}, "
              f"evaluate={val:.8f}, time={time.monotonic()-start_time:.1f}s")
        print(f"{'='*60}")
    else:
        print(f"\nVALIDATION FAILED. Using fallback.")
        result = [1.0] * 10

    return result
