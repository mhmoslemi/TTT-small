```python
Optimized step function sequence generator using LP alternating method
with aggressive polishing via coordinate descent and multi-n exploration.
Focuses on tight convergence of the LP fixed-point iteration and careful
pairwise mass transfer to minimize the convolution peak.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=60):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, 0.0

    LP alternating fixed-point iteration.
```

```python
"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP fixed-point iteration with line search, proximal gradient with Nesterov
momentum, aggressive coordinate polish targeting convolution peaks, and
systematic multi-n exploration. Focuses on tight convergence and careful
pairwise mass transfer to close the gap to the target bound.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=60):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section(f, lo, hi, max_iter=20, tol=0.01):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def lp_fixed_point(a, max_iters=150, time_budget=300.0, tol=1e-11):
    """LP alternating with adaptive line search for the blend parameter."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 10:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        lp_time = max(3, min(30, int(remaining * 0.4)))
        g = solve_lp(a, M, time_limit=lp_time)
        if g is None:
            break

        diff = np.max(np.abs(a - g))
        if diff < tol:
            return best_a

        # Line search on blend parameter t in [0, 1.2]
        def obj_at_t(t):
            a_try = normalize((1.0 - t) * a + t * g)
            return compute_obj(a_try)

        t_opt = golden_section(obj_at_t, 0.0, 1.2, max_iter=12, tol=0.02)
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

        if it % 25 == 0:
            print(f"    LP iter {it}: obj={best_obj:.8f}, diff={diff:.2e}, t={t_opt:.3f}")

    return best_a


def nesterov_gradient(a, time_budget=90.0, tau_start=0.01, tau_end=0.0005):
    """Nesterov-accelerated proximal gradient on softmax-smoothed peak."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    y = a.copy()
    momentum = 0.9
    step_size = 0.03
    n_steps = 0
    total_steps = max(100, int(time_budget / 0.05))

    while time.monotonic() - start < time_budget:
        progress = min(1.0, n_steps / float(total_steps))
        tau = tau_start * (tau_end / tau_start) ** progress

        conv = compute_conv(y)
        cmax = float(np.max(conv))

        shifted = np.clip((conv - cmax) / tau, -500, 500)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum > 1e-300:
            weights /= w_sum
        else:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0

        # Gradient via cross-correlation
        w_ext = np.zeros(2 * n - 1)
        w_ext[:] = weights
        a_flip = y[::-1]
        Wf = rfft(w_ext, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        grad -= np.mean(grad)

        # Proximal gradient step
        a_try = normalize(y - step_size * grad)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

        # Nesterov update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        momentum = min(momentum + 0.001, 0.99)
        y = a_try + ((momentum - 1.0) / beta_new) * (a_try - y)
        y = np.maximum(y, 0.0)

        a = a_try

        if obj_try > best_obj + 1e-10:
            step_size *= 0.8
            if step_size < 1e-9:
                break

        n_steps += 1

    return best_a


def coordinate_polish(a, time_budget=120.0):
    """
    Systematic pairwise coordinate descent targeting the convolution peak.
    For pairs contributing to the peak, try moving mass to positions that
    minimize the new peak.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    iter_count = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Build list of (i, j) pairs contributing to peak, sorted by contribution
        pairs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                pairs.append((j, k2, a[j] * a[k2]))
        pairs.sort(key=lambda x: -x[2])

        improved = False
        n_top = min(10, len(pairs))

        for pi in range(n_top):
            if time.monotonic() - start > time_budget - 5:
                break

            idx1, idx2, prod = pairs[pi]
            if idx1 == idx2:
                continue

            # Find target positions that don't contribute much to near-peak values
            targets = []
            for t in range(n):
                if t == idx1 or t == idx2:
                    continue
                # Check contribution to peak
                t2 = k_star - t
                contrib_peak = 0.0
                if 0 <= t2 < n:
                    contrib_peak = a[t] * a[t2]
                # Check contribution to near-peak (within 3 positions)
                contrib_near = 0.0
                for dk in range(-3, 4):
                    if dk == 0:
                        continue
                    k_near = k_star + dk
                    if 0 <= k_near < 2 * n - 1:
                        t2_near = k_near - t
                        if 0 <= t2_near < n:
                            contrib_near = max(contrib_near, conv[k_near] * a[t2_near] * a[t] / max(a[t], 1e-15))
                if contrib_peak < 0.3 * prod:
                    targets.append((t, contrib_peak))

            if not targets:
                continue

            # Try best targets
            targets.sort(key=lambda x: x[1])
            for target, _ in targets[:3]:
                if time.monotonic() - start > time_budget - 5:
                    break

                # Ternary search for optimal delta
                lo = 0.0
                hi = 0.4 * a[idx1]
                if hi < 1e-12:
                    continue

                best_local_obj = best_obj
                best_local_delta = 0.0

                for _ in range(10):
                    mid1 = lo + (hi - lo) / 3.0
                    mid2 = hi - (hi - lo) / 3.0

                    a_try1 = a.copy()
                    a_try1[idx1] -= mid1
                    a_try1[target] += mid1
                    a_try1 = normalize(a_try1)
                    obj1 = compute_obj(a_try1)

                    a_try2 = a.copy()
                    a_try2[idx1] -= mid2
                    a_try2[target] += mid2
                    a_try2 = normalize(a_try2)
                    obj2 = compute_obj(a_try2)

                    if obj1 < obj2:
                        hi = mid2
                    else:
                        lo = mid1

                    if obj1 < best_local_obj:
                        best_local_obj = obj1
                        best_local_delta = mid1
                    if obj2 < best_local_obj:
                        best_local_obj = obj2
                        best_local_delta = mid2

                if best_local_delta > 1e-8 and best_local_obj < best_obj - 1e-13:
                    a[idx1] -= best_local_delta
                    a[target] += best_local_delta
                    a = normalize(a)
                    best_obj = best_local_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Occasional perturbation to escape local minima
            if rng.random() < 0.2:
                strength = 0.01 + 0.02 * (iter_count % 3)
                a = normalize(a * np.exp(rng.normal(0, strength, n)))

        iter_count += 1
        if iter_count % 30 == 0:
            print(f"    Coord iter {iter_count}: obj={best_obj:.8f}")

    return best_a


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def spectral_init(n, rng):
    """Create a sequence with good spectral properties for initialization."""
    # Random phase in frequency domain with specific magnitude shaping
    freqs = np.arange(n)
    # Emphasize low and high frequencies, suppress mid (reduces concentration)
    mag = 1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(0.5, 3.0) * freqs / n)
    phase = rng.uniform(0, 2 * np.pi, n)
    spectrum = mag * np.exp(1j * phase)
    a = np.real(np.fft.ifft(spectrum))[:n]
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a

            else:
                print(f"  Restart {restart}: {obj_try:.8f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
```
