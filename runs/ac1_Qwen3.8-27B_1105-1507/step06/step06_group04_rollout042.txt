"""
Peak-equalizing optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative subgradient, LP alternating, coordinate descent, 
and simulated annealing with adaptive length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=25):
    """Solve: min t s.t. (fixed_seq * g)[k] <= t, sum(g)=1, g>=0."""
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(fixed_seq, M, time_limit=25):
    """Solve: max sum(g) s.t. (fixed_seq * g)[k] <= M, g >= 0."""
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def multiplicative_subgradient(a, max_iters=2000, time_budget=60.0):
    """
    Multiplicative subgradient: a_i <- a_i / (1 + lambda * g_i)
    where g_i is the weighted contribution to convolution peaks.
    Multiplicative updates naturally maintain non-negativity.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    lambda_ = 0.1
    tau = 0.01  # temperature for softmax over conv values
    tau_min = 0.001
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Softmax weights over convolution values
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient: g_i = sum_k w_k * a_{k-i}
        # Using FFT: corr[m] = sum_j a[j] * w[m+j]
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]

        # Multiplicative update: a_i <- a_i / (1 + lambda * grad_i / max(grad))
        grad_max = np.max(grad)
        if grad_max < 1e-15:
            break
        grad_normalized = grad / grad_max
        
        a_new = a / (1.0 + lambda_ * grad_normalized)
        a_new = normalize_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Decay lambda slightly
            lambda_ *= 0.999
        else:
            a = a  # keep old
            stall += 1
            if stall > 50:
                # Reset with smaller lambda and higher tau
                lambda_ = max(0.01, lambda_ * 0.5)
                tau = min(0.05, tau * 1.5)
                # Perturb
                rng = np.random.default_rng(it * 7919 + 3)
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = normalize_simplex(a)
                stall = 0

        # Slowly decrease temperature
        tau = max(tau_min, tau * 0.9997)

    return best_a, best_obj


def coordinate_descent_polish(a, time_budget=30.0, num_passes=5):
    """
    Coordinate descent: for each element, try small perturbations
    and keep the best. Cycles through all elements num_passes times.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute convolution for initial state
    for p in range(num_passes):
        if time.monotonic() - start > time_budget:
            break
        # Random order for diversity
        rng = np.random.default_rng(p * 31 + 7)
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            # Try a few perturbations
            delta_options = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005]
            best_delta = 0.0
            for d in delta_options:
                a_try = a.copy()
                a_try[i] = max(0, a_try[i] + d * np.mean(a))
                s = np.sum(a_try)
                if s < 1e-15:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_delta = d
                    best_a = a_try.copy()
            if best_delta != 0.0:
                a = best_a.copy()
                # Reset delta options to smaller values
                delta_options = [0.003, -0.003, 0.001, -0.001]

    return best_a, best_obj


def simulated_annealing(a, time_budget=60.0, T_init=0.05, T_min=0.001):
    """
    Simulated annealing: random perturbations with cooling.
    Accept worse solutions with probability exp(-delta/T).
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = T_init
    rng = np.random.default_rng(12345)

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        if T < T_min:
            break

        # Random perturbation
        scale = 0.01 + 0.03 * T / T_init
        a_new = a * np.exp(rng.normal(0, scale, n))
        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)

        delta = obj_new - best_obj
        if delta < 0:
            a = a_new
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
        else:
            # Accept with probability exp(-delta/T)
            if rng.random() < np.exp(-delta / max(T, 1e-10)):
                a = a_new

        T *= 0.9995

    return best_a, best_obj


def lp_alternate_refined(a, max_iters=50, time_budget=120.0):
    """
    Refined LP alternating with more aggressive line search.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        # Alternate between min-peak and max-sum LP
        if it % 2 == 0:
            g, t_opt = solve_minpeak_lp(a, time_limit=min(20, remaining - 10))
        else:
            conv = compute_conv(a)
            M = float(np.max(conv))
            g = solve_maxsum_lp(a, M, time_limit=min(20, remaining - 10))
            if g is None:
                g, t_opt = solve_minpeak_lp(a, time_limit=min(20, remaining - 10))
                if g is None:
                    stall += 1
                    if stall > 10:
                        break
                    continue

        if g is None:
            stall += 1
            if stall > 10:
                break
            continue

        # Extensive line search
        # Use golden section
        def blend_obj(t):
            a_try = (1.0 - t) * a + t * g
            a_try = normalize_simplex(a_try)
            return compute_obj(a_try)

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = blend_obj(c)
        fd = blend_obj(d)

        for _ in range(12):
            if hi - lo < 0.01:
                break
            if fc < fd:

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def generate_starts(n, num_starts=10, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    starts.append(np.ones(n) / n)

    # Gaussian (narrow)
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 5.0) ** 2))
    starts.append(bell / np.sum(bell))

    # Gaussian (wide)
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 3.0) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # U-shaped
    u = (np.abs(i - center) / center + 0.1)
    starts.append(u / np.sum(u))

    # V-shaped (inverted U)
    v = 1.0 - (np.abs(i - center) / center)
    v = np.maximum(v, 0.01)
    starts.append(v / np.sum(v))

    # Random parametric
    for _ in range(5):
        sigma = n * (0.15 + 0.3 * rng.random())
        freq = int(1 + rng.integers(0, 6))
        amp = 0.2 + 0.4 * rng.random()
        phase = 2 * np.pi * rng.random()
        bell_p = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
        a = np.maximum(bell_p * mod, 0)
        s = np.sum(a)
        starts.append(a / s if s > 1e-15 else np.ones(n) / n)

    return starts

"""
Peak-equalizing optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative subgradient (log-space gradient descent), LP alternating,
coordinate descent polishing, simulated annealing, and adaptive length exploration.
Key innovation: multiplicative updates in log-space for faster convergence
and automatic non-negativity preservation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=25):
    """Solve: min t s.t. (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0."""
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(fixed_seq, M, time_limit=25):
    """Solve: max sum(g) s.t. (fixed_seq * g)[k] <= M, g >= 0."""
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def compute_subgradient_fft(a, weights, fft_size):
    """Compute gradient g[i] = 2 * sum_j a[j] * weights[i+j] via FFT."""
    w_ext = np.zeros(fft_size)
    w_ext[:len(weights)] = weights
    W_ext = rfft(w_ext, fft_size)
    A_ext = rfft(a, fft_size)
    corr = irfft(A_ext * np.conj(W_ext), fft_size)
    return 2.0 * corr[:len(a)]


def multiplicative_subgradient(a, max_iters=3000, time_budget=60.0):
    """
    Multiplicative subgradient: a_i <- a_i / (1 + lambda * g_i / g_max).
    Equivalent to gradient descent in log-space, naturally maintains non-negativity.
    Adaptive temperature and step size.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    lambda_ = 0.15
    tau = 0.015
    tau_min = 0.002
    stall = 0
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient via FFT
        grad = compute_subgradient_fft(a, weights, fft_size)
        grad_max = np.max(grad)
        if grad_max < 1e-15:
            break
        grad_norm = grad / grad_max

        # Multiplicative update
        a_new = a / (1.0 + lambda_ * grad_norm)
        a_new = normalize_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lambda_ = max(0.02, lambda_ * 0.9995)
        else:
            stall += 1
            if stall > 30:
                stall_count += 1
                # Reset with perturbation
                rng = np.random.default_rng(it * 7919 + 3 + stall_count)
                perturb_scale = 0.01 + 0.01 * min(stall_count, 5)
                a = best_a * np.exp(rng.normal(0, perturb_scale, n))
                a = normalize_simplex(a)
                lambda_ = max(0.01, lambda_ * 0.7)
                tau = min(0.05, tau * 1.5)
                stall = 0

        # Decay temperature
        tau = max(tau_min, tau * 0.9998)

    return best_a, best_obj


def additive_subgradient_nesterov(a, max_iters=3000, time_budget=60.0):
    """
    Additive subgradient with Nesterov momentum and adaptive step.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    v = np.zeros(n)
    mu = 0.8
    step = 0.08
    tau = 0.02
    tau_min = 0.002

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        grad = compute_subgradient_fft(a, weights, fft_size)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        # Nesterov momentum
        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            a = a  # revert
            step *= 0.95
            if it % 200 == 199:
                rng = np.random.default_rng(it * 31 + 7)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = normalize_simplex(a)

        step = max(0.01, step * 0.9997)
        tau = max(tau_min, tau * 0.9996)

    return best_a, best_obj


def coordinate_descent_polish(a, time_budget=30.0, num_passes=10):
    """
    Coordinate descent: for each element, try perturbations and keep best.
    Uses adaptive delta sizes that shrink over passes.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    base_delta = 0.02

    for p in range(num_passes):
        if time.monotonic() - start > time_budget:
            break
        # Shrink delta as we polish
        delta_scale = base_delta * (0.7 ** p)
        rng = np.random.default_rng(p * 31 + 7)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break
            mean_a = np.mean(a)
            deltas = [delta_scale, -delta_scale, 0.5 * delta_scale, -0.5 * delta_scale,
                      2.0 * delta_scale, -2.0 * delta_scale]
            improved = False
            for d in deltas:
                a_try = a.copy()
                a_try[i] = max(0.0, a_try[i] + d * mean_a)
                s = np.sum(a_try)
                if s < 1e-15:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            # If no improvement, try smaller deltas
            if not improved and p < num_passes - 2:
                small_deltas = [0.1 * delta_scale, -0.1 * delta_scale]
                for d in small_deltas:
                    a_try = a.copy()
                    a_try[i] = max(0.0, a_try[i] + d * mean_a)
                    s = np.sum(a_try)
                    if s < 1e-15:
                        continue
                    a_try = a_try / s
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        break

    return best_a, best_obj


def simulated_annealing(a, time_budget=60.0, T_init=0.03, T_min=0.001):
    """
    Simulated annealing for escaping local minima.
    Multiplicative perturbation with geometric cooling.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = T_init
    rng = np.random.default_rng(54321)

    it = 0
    while time.monotonic() - start < time_budget and T > T_min:
        scale = 0.005 + 0.04 * (T / T_init)
        a_new = a * np.exp(rng.normal(0, scale, n))
        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)

        delta = obj_new - best_obj
        if delta < 0:
            a = a_new
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
        else:
            if scale > 0 and rng.random() < np.exp(-delta / max(T, 1e-10)):
                a = a_new

        T *= 0.999
        it += 1
        if it > 20000:
            break

    return best_a, best_obj


def lp_alternate_refined(a, max_iters=40, time_budget=120.0):
    """
    LP alternating with golden-section line search.
    Alternates between min-peak LP and max-sum LP.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        # Choose LP type
        if it % 2 == 0:
            g, t_opt = solve_minpeak_lp(a, time_limit=min(20, max(5, remaining - 10)))
        else:
            conv = compute_conv(a)
            M = float(np.max(conv))
            g = solve_maxsum_lp(a, M, time_limit=min(20, max(5, remaining - 10)))
            if g is None:
                g, _ = solve_minpeak_lp(a, time_limit=min(15, max(5, remaining - 10)))

        if g is None:
            stall += 1
            if stall > 8:
                break
            continue

        # Golden-section line search
        def blend_eval(t):
            a_try = (1.0 - t) * a + t * g
            a_try = normalize_simplex(a_try)
            return compute_obj(a_try)

        lo, hi = 0.0, 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = blend_eval(c)
        fd = blend_eval(d)

        for _ in range(10):
            if hi - lo < 0.02:
                break
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = blend_eval(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = blend_eval(d)

        t_best = (lo + hi) / 2.0
        a_try = (1.0 - t_best) * a + t_best * g
        a_try = normalize_simplex(a_try)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
        else:
            stall += 1
            if stall > 6:
                rng = np.random.default_rng(it * 104729 + 17)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = normalize_simplex(a)
                stall = 0

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def generate_starts(n, num_starts=8, seed=42):
    """Generate diverse initial sequences for multi-start."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    starts.append(np.ones(n) / n)

    # Gaussian narrow
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 5.0) ** 2))
    starts.append(bell / np.sum(bell))

    # Gaussian wide
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 3.0) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # U-shaped
    u = np.abs(i - center) / center + 0.1
    starts.append(u / np.sum(u))

    # Cosine bump
    cos_bump = np.maximum(np.cos(np.pi * (i - center) / center), 0) ** 2
    s = np.sum(cos_bump)
    starts.append(cos_bump / s if s > 1e-15 else np.ones(n) / n)

    # Random parametric
    for _ in range(max(0, num_starts - 5)):
        sigma = n * (0.15 + 0.3 * rng.random())
        freq = int(1 + rng.integers(0, 6))
        amp = 0.2 + 0.4 * rng.random()
        phase = 2 * np.pi * rng.random()
        bell_p = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
        a = np.maximum(bell_p * mod, 0)
        s = np.sum(a)
        starts.append(a / s if s > 1e-15 else np.ones(n) / n)

    return starts


def polish_pipeline(a, time_budget=100.0):
    """
    Full polishing pipeline: multiplicative SG -> LP -> additive SG -> CD -> SA.
    Each method gets a fraction of the time budget.
    """
    n = len(a)
    start = time.monotonic()
    best_a = normalize_simplex(a.copy())
    best_obj = compute_obj(best_a)

    # Phase A: Multiplicative subgradient (fast initial convergence)
    remaining = time_budget - (time.monotonic() - start)
    alloc = min(remaining * 0.25, 30.0)
    if alloc > 5:
        a_new, obj_new = multiplicative_subgradient(best_a, max_iters=int(alloc * 20), time_budget=alloc)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

    # Phase B: LP alternating (if n is small enough)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20 and n <= 750:
        alloc = min(remaining * 0.30, 45.0)
        a_new, obj_new = lp_alternate_refined(best_a, max_iters=30, time_budget=alloc)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

    # Phase C: Additive subgradient with Nesterov (different direction)
    remaining = time_budget - (time.monotonic() - start)
    alloc = min(remaining * 0.20, 25.0)
    if alloc > 5:
        a_new, obj_new = additive_subgradient_nesterov(best_a, max_iters=int(alloc * 15), time_budget=alloc)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

    # Phase D: Another multiplicative pass (finer)
    remaining = time_budget - (time.monotonic() - start)
    alloc = min(remaining * 0.15, 20.0)
    if alloc > 5:
        a_new, obj_new = multiplicative_subgradient(best_a, max_iters=int(alloc * 25), time_budget=alloc)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

    # Phase E: Coordinate descent (fine polishing)
    remaining = time_budget - (time.monotonic() - start)
    alloc = min(remaining * 0.15, 25.0)
    if alloc > 5:
        a_new, obj_new = coordinate_descent_polish(best_a, time_budget=alloc, num_passes=5)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

    # Phase F: Simulated annealing (escape local minima)
    remaining = time_budget - (time.monotonic() - start)
    alloc = min(remaining * 0.10, 15.0)
    if alloc > 5:
        a_new, obj_new = simulated_annealing(best_a, time_budget=alloc, T_init=0.02, T_min=0.003)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

    # Final CD polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_new, obj_new = coordinate_descent_polish(best_a, time_budget=remaining * 0.5, num_passes=15)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multiplicative subgradient, LP alternating,
    coordinate descent, simulated annealing, and length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_norm = normalize_simplex(parent.copy())
            parent_obj = compute_obj(parent_norm)
            print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
            best_score = parent_obj
            best_seq = parent_norm.copy()
    except Exception:
        pass

    if best_seq is None:
        n_init = 700
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Initialized uniform n=700: obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n={n_parent}")

    # === Phase 1: Intensive optimization of parent ===
    print("\n=== Phase 1: Intensive optimization of parent ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.25, 250.0)
    a_pol, obj_pol = polish_pipeline(best_seq, time_budget=alloc)
    if obj_pol < best_score:
        best_score = obj_pol
        best_seq = a_pol.copy()
        print(f"  Phase 1 done: {best_score:.8f}")
    else:
        print(f"  Phase 1 done: {obj_pol:.8f} (no improvement)")

    # === Phase 2: Length exploration ===
    print(f"\n=== Phase 2: Length exploration (best: {best_score:.8f}) ===")
    n_best = len(best_seq)
    test_lengths = sorted(set([
        n_best - 200, n_best - 100, n_best - 50,
        n_best + 50, n_best + 100, n_best + 200,
        500, 600, 800, 900, 1000, 1100, 1200
    ]))
    test_lengths = [x for x in test_lengths if 300 <= x <= 1400]

    for n_test in test_lengths:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        if n_test == n_best:
            continue

        a_resampled = resample_sequence(best_seq, n_test)
        a_resampled = normalize_simplex(a_resampled)
        obj_init = compute_obj(a_resampled)

        alloc = min(remaining * 0.12, 60.0)
        if alloc < 10:
            break
        a_pol2, obj_pol2 = polish_pipeline(a_resampled, time_budget=alloc)

        if obj_pol2 < best_score:
            best_score = obj_pol2
            best_seq = a_pol2.copy()
            print(f"  n={n_test}: ** UPDATE {best_score:.8f} **")
        else:
            print(f"  n={n_test}: {obj_pol2:.8f} (init {obj_init:.6f})")

    # === Phase 3: Multi-start diverse exploration ===
    print(f"\n=== Phase 3: Multi-start (best: {best_score:.8f}) ===")
    n_best = len(best_seq)
    explore_lengths = sorted(set([n_best, max(400, n_best - 150), min(1100, n_best + 150)]))

    for n_exp in explore_lengths:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        starts = generate_starts(n_exp, num_starts=6, seed=seed + n_exp)
        # Include current best as a start if length matches
        if len(best_seq) == n_exp:
            starts.insert(0, best_seq.copy())

        for si, s in enumerate(starts[:4]):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            alloc = min(remaining * 0.12, 45.0)
            if alloc < 10:
                break

            a_pol3, obj_pol3 = polish_pipeline(s, time_budget=alloc)
            if obj_pol3 < best_score:
                best_score = obj_pol3
                best_seq = a_pol3.copy()
                print(f"  n={n_exp}, start {si}: ** {best_score:.8f} **")
            else:
                print(f"  n={n_exp}, start {si}: {obj_pol3:.8f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = remaining * 0.4
        a_final, obj_final = polish_pipeline(best_seq, time_budget=alloc)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  After final polish: {best_score:.8f}")
        else:
            print(f"  Final polish: {obj_final:.8f}")

        # Extended coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_cd, obj_cd = coordinate_descent_polish(best_seq, time_budget=remaining * 0.4, num_passes=30)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  After extended CD: {best_score:.8f}")

        # Another LP pass
        remaining = deadline - time.monotonic()
        if remaining > 25 and len(best_seq) <= 750:
            a_lp, obj_lp = lp_alternate_refined(best_seq, max_iters=25, time_budget=remaining * 0.4)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  After final LP: {best_score:.8f}")

    # === Phase 5: Perturbation search ===
    remaining = deadline - time.monotonic()
    if remaining > 40 and best_seq is not None:
        print(f"\n=== Phase 5: Perturbation search (best: {best_score:.8f}) ===")
        n_final = len(best_seq)
        rng = np.random.default_rng(seed + 77777)

        for pidx in range(20):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            scale = 0.003 + 0.002 * pidx
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
            a_pert = normalize_simplex(a_pert)

            alloc = min(remaining * 0.12, 15.0)
            if alloc < 5:
                break

            a_pol5, obj_pol5 = polish_pipeline(a_pert, time_budget=alloc)
            if obj_pol5 < best_score:
                best_score = obj_pol5
                best_seq = a_pol5.copy()
                print(f"  Perturb {pidx} (scale={scale:.4f}): ** {best_score:.8f} **")
            elif pidx < 5:
                print(f"  Perturb {pidx}: {obj_pol5:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
