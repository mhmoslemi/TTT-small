import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax_param(x):
    """Convert parameter x to positive sequence via softmax (numerically stable)."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)

def subgradient_at_peak(a, k_star):
    """Compute subgradient of (a*a)[k_star] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        k2 = k_star - j
        if 0 <= k2 < n:
            grad[j] = a[k2]
    return grad

def projected_subgradient_nesterov(a_init, max_iter=200, alpha_init=0.01, time_budget=50.0):
    """
    Nesterov accelerated projected subgradient method.
    Optimizes: minimize max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    Uses Euclidean projection onto the simplex.
    """
    n = len(a_init)
    a = normalize(a_init.copy())
    a_prev = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = alpha_init
    t = 1.0
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)
        y = y / np.sum(y)
        
        # Compute subgradient at y
        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))
        grad = subgradient_at_peak(y, k_star)
        
        # Normalize gradient (it's in the unit ball of the dual)
        # Scale alpha appropriately
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        
        # Gradient step
        a_new = y - alpha * grad
        a_prev = a
        a = project_simplex(a_new)
        t = t_new
        
        # Evaluate
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive alpha: decrease slowly
        alpha = alpha_init / (1.0 + 0.01 * it)
    
    return best_a

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def lp_alternate_improved(a, max_iters=30, time_budget=80.0):
    """
    Improved LP alternating: solve LP max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Then blend with golden-section line search.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    build_toeplitz_mat = None  # cache
    n_prev = -1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Build Toeplitz (cache if n hasn't changed)
        if n_prev != n:
            build_toeplitz_mat = build_toeplitz(a)
            n_prev = n
        A = build_toeplitz_mat
        
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(3, int(remaining * 0.5))})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g
        except Exception:
            break
        
        # Line search for blend parameter
        best_t = 0.0
        best_obj_t = best_obj
        for t_try in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = normalize((1.0 - t_try) * a + t_try * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t_try
        
        a_new = normalize((1.0 - best_t) * a + best_t * g)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try perturbation
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.05, n)))
    
    return best_a

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def soft_max_gradient_v2(a, tau_init=0.002, step_size=0.03, max_steps=300, time_budget=50.0):
    """
    Improved soft-max gradient with adaptive temperature and
    multiple peak targeting.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        cmean = float(np.mean(conv))
        
        # Adaptive temperature
        tau = max(tau_init, (cmax - cmean) / (10.0 + step * 0.5))
        
        # Soft-max weights: emphasize peaks more than the previous version
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -300.0, 0.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Gradient via correlation: grad[i] = sum_j weights[i+j] * a[j] (approximately)
        # Using FFT for efficiency
        a_flip = a[::-1]
        nf = 1
        while nf < 3 * n:
            nf <<= 1
        Wf = rfft(np.concatenate([weights, np.zeros(nf - len(weights))]), nf)
        Af = rfft(np.concatenate([a_flip, np.zeros(nf - len(a_flip))]), nf)

        grad = 2.0 * corr[n - 1:2 * n - 1]
        
        # Center gradient (remove component along all-ones to maintain sum)
        grad_centered = grad - np.mean(grad)

        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

def window_initialization(n, window_type='tukey'):
    """Create a sequence from a window function."""
    x = np.arange(n) / (n - 1)
    if window_type == 'tukey':
        from scipy.signal import tukey
        a = tukey(n, 0.5)
    elif window_type == 'hamming':
        a = np.hamming(n)
    elif window_type == 'flattop':
        # Flat-top window (good for flat spectral response)
        a = (0.21557895 - 0.41663158 * np.cos(2*np.pi*x) + 0.277263158 * np.cos(4*np.pi*x)
             - 0.083578947 * np.cos(6*np.pi*x) + 0.006947368 * np.cos(8*np.pi*x))
    elif window_type == 'gaussian':
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - 0.5) * n / sigma) ** 2)
    else:
        a = np.ones(n)
    return normalize(a)

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def diverse_restart(a_best, n, rng, restart_type):
    """Generate a diverse perturbation of the best sequence."""
    if restart_type == 0:
        return normalize(a_best * np.exp(rng.normal(0, 0.1, n)))
    elif restart_type == 1:
        # Frequency modulation
        freq = rng.uniform(0.5, 8.0)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        return normalize(a_best * mod)
    elif restart_type == 2:
        # Blend with new window
        window_types = ['tukey', 'hamming', 'flattop', 'gaussian']
        wt = window_types[int(rng.integers(0, 4))]
        w = window_initialization(n, wt)
        mix = rng.uniform(0.3, 0.7)
        return normalize((1 - mix) * normalize(a_best) + mix * w)
    elif restart_type == 3:
        # Block permutation
        a_try = a_best.copy()
        num_blocks = int(rng.integers(3, 8))
        block_size = n // num_blocks
        perm = rng.permutation(num_blocks)
        new_a = np.zeros(n)
        for i in range(num_blocks):
            start = i * block_size
            end = start + block_size if i < num_blocks - 1 else n
            p_start = perm[i] * block_size
            p_end = p_start + (end - start)
            new_a[p_start:p_end] = a_try[start:end]
        return normalize(new_a)
    elif restart_type == 4:
        # Mirror + noise
        half = n // 2
        a_try = a_best.copy()
        a_try[half:] = a_try[:half][::-1] if n % 2 == 0 else a_try[:half+1][::-1]
        return normalize(a_try * np.exp(rng.normal(0, 0.05, n)))
    else:
        # Random Dirichlet blend
        a_rand = rng.dirichlet(np.ones(n) * 0.3)
        mix = rng.uniform(0.4, 0.8)
        return normalize((1 - mix) * normalize(a_best) + mix * a_rand)

"""
Optimization of 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov accelerated projected subgradient on the simplex
2. LP alternating with cached Toeplitz and adaptive line search
3. Soft-max gradient with adaptive temperature
4. Simulated annealing with targeted peak-deflation moves
5. Multi-resolution exploration and diverse restarts
6. Population-based local search for escaping basins of attraction

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_at_peak(a, k_star):
    """Compute gradient of (a*a)[k_star] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        k2 = k_star - j
        if 0 <= k2 < n:
            grad[j] = a[k2]
    return grad


def nesterov_subgradient(a_init, max_iter=500, alpha_init=0.02, time_budget=50.0):
    """
    Nesterov accelerated projected subgradient method.
    Minimizes max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    a = normalize(a_init.copy())
    a_prev = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t = 1.0

    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = a + momentum * (a - a_prev)
        y = project_simplex(y)

        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))
        grad = subgradient_at_peak(y, k_star)

        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm > 1e-15:
            grad_centered = grad_centered / grad_norm

        alpha = alpha_init / (1.0 + 0.005 * it)
        a_candidate = y - alpha * grad_centered
        a_prev = a
        a = project_simplex(a_candidate)
        t = t_new

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def soft_max_gradient(a, tau_init=0.002, step_size=0.04, max_steps=300, time_budget=50.0):
    """
    Soft-max (log-sum-exp) weighted gradient descent.
    Targets multiple peaks simultaneously for smoother convergence.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 3 * n:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        cmean = float(np.mean(conv))

        tau = max(tau_init * 0.5, (cmax - cmean) / (5.0 + step * 0.3))

        shifted = np.clip((conv - cmax) / tau, -200.0, 0.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        a_flip = a[::-1]
        Wp = np.zeros(size)
        Wp[:len(weights)] = weights
        Ap = np.zeros(size)
        Ap[:len(a_flip)] = a_flip
        Wf = rfft(Wp)
        Af = rfft(Ap)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        grad_centered = grad - np.mean(grad)

        a_new = project_simplex(a - step_size * grad_centered)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.75
            if step_size < 1e-9:
                rng = np.random.default_rng(step * 31 + 7)
                a = normalize(best_a * np.exp(rng.normal(0, 0.04, n)))
                step_size = 0.03
                best_a = a.copy()

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_alternate(a, max_iters=25, time_budget=60.0):
    """
    LP alternating: solve max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Blend with line search.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    A = build_toeplitz(a)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(3, int(remaining * 0.4))})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g
        except Exception:
            break

        # Multi-point line search
        best_t = 0.0
        best_obj_t = best_obj
        for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = normalize((1.0 - t_try) * a + t_try * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t_try

        a_new = normalize((1.0 - best_t) * a + best_t * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            rng = np.random.default_rng((it * 104729 + 12345) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.04, n)))

    return best_a


def annealed_peak_deflation(a, n_iter=200, temp_init=0.01, cool=0.97, time_budget=40.0):
    """
    Simulated annealing with peak-targeted moves.
    At each step, identify the peak lag and try moves that reduce it.
    Accept worse moves with probability exp(-delta/temp).
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    temp = temp_init
    rng = np.random.default_rng(777)

    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        # Pick a random contributor and try to reduce it
        if contrib:
            # Weighted random selection from top contributors
            top_k = min(10, len(contrib))
            idx = int(rng.integers(0, top_k))
            _, j1, j2 = contrib[idx]

            if j1 != j2:
                # Try moving mass from the larger of j1,j2 to a random distant position
                src = j1 if a[j1] > a[j2] else j2
                dst = int(rng.integers(0, n))
                if dst != src:
                    delta = rng.uniform(0.02, 0.08) * a[src]
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[dst] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    delta_obj = obj_try - best_obj

                    if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(temp, 1e-10)):
                        a = a_try
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a.copy()

        # Occasionally do a broader perturbation
        if it % 20 == 0:
            a = normalize(best_a * np.exp(rng.normal(0, 0.03 * temp * 100, n)))

        temp *= cool

    return best_a


def window_init(n, wtype):
    """Create initialization from a window function."""
    x = np.arange(n) / max(n - 1, 1)
    if wtype == 0:  # Tukey
        from scipy.signal import tukey
        a = tukey(n, 0.5)
    elif wtype == 1:  # Hamming
        a = np.hamming(n)
    elif wtype == 2:  # Flat-top
        a = (0.21557895 - 0.41663158 * np.cos(2*np.pi*x) + 0.27726316 * np.cos(4*np.pi*x)
             - 0.08357895 * np.cos(6*np.pi*x) + 0.00694737 * np.cos(8*np.pi*x))
    elif wtype == 3:  # Gaussian
        sigma = n / 5.0
        a = np.exp(-0.5 * ((x - 0.5) * n / sigma) ** 2)
    elif wtype == 4:  # Raised cosine variant
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    return normalize(a)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def targeted_repair(a, time_budget=30.0):
    """
    Targeted repair: iteratively identify the worst lag and perform
    a local optimization to reduce that specific peak.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(555)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Compute which pairs contribute to this peak
        # (a*a)[k_star] = sum_j a[j]*a[k_star-j]
        # Try pairwise adjustments among top contributors
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n and j != k2:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        improved = False
        for _, j1, j2 in contrib[:6]:
            if time.monotonic() - start > time_budget:
                break

            # Try several delta values: shift mass from j1 to j2 or vice versa
            for frac in [0.05, 0.1, 0.2, 0.3]:
                for direction in [1, -1]:
                    a_try = a.copy()
                    delta = frac * min(a[j1], a[j2]) * direction
                    a_try[j1] -= delta
                    a_try[j2] += delta
                    if np.any(a_try < -1e-10):
                        continue
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
        if not improved:
            # Perturb to escape
            a = normalize(best_a * np.exp(rng.normal(0, 0.02, n)))

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main multi-phase search combining Nesterov subgradient, LP alternating,
    soft-max gradient, simulated annealing, targeted repair, and
    multi-resolution exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent = normalize(parent)
            best_score = compute_obj(parent)
            best_seq = parent.copy()
            print(f"Parent loaded (n={parent.size}): obj={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 700
        best_seq = window_init(n_init, 0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.6f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # ============ Phase 1: Core local optimization ============
    print(f"\n{'='*60}")
    print(f"Phase 1: Core local optimization (n={n_best})")
    print(f"{'='*60}")

    for cycle in range(4):
        if time.monotonic() > deadline - 500:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.12, 60.0)

        # Nesterov
        a_new = nesterov_subgradient(best_seq, max_iter=400, alpha_init=0.02, time_budget=alloc * 0.35)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  N1_{cycle}: {best_score:.8f} **")

        # Soft-max gradient
        a_new = soft_max_gradient(best_seq, tau_init=0.001, step_size=0.04,
                                  max_steps=300, time_budget=alloc * 0.35)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  SM_{cycle}: {best_score:.8f} **")

        # Targeted repair
        a_new = targeted_repair(best_seq, time_budget=alloc * 0.3)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  TR_{cycle}: {best_score:.8f} **")

    # ============ Phase 2: LP alternating ============
    print(f"\n{'='*60}")
    print(f"Phase 2: LP alternating (n={n_best})")
    print(f"{'='*60}")

    for cycle in range(3):
        if time.monotonic() > deadline - 350:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.15, 70.0)

        a_new = lp_alternate(best_seq, max_iters=15, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  LP_{cycle}: {best_score:.8f} **")
        else:
            print(f"  LP_{cycle}: {obj_new:.8f}")

    # ============ Phase 3: Simulated annealing ============
    print(f"\n{'='*60}")
    print(f"Phase 3: Simulated annealing")
    print(f"{'='*60}")

    for cycle in range(3):
        if time.monotonic() > deadline - 250:
            break
        alloc = 35.0
        a_new = annealed_peak_deflation(best_seq, n_iter=300, temp_init=0.02,
                                        cool=0.96, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  SA_{cycle}: {best_score:.8f} **")
        else:
            print(f"  SA_{cycle}: {obj_new:.8f}")

    # ============ Phase 4: Multi-n exploration ============
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_cur = len(best_seq)
    n_options = sorted(set([
        max(200, n_cur - 100), max(200, n_cur - 50), max(200, n_cur - 25),
        n_cur,
        n_cur + 25, n_cur + 50, n_cur + 100, n_cur + 150
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1100]

    for n_try in n_options:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if n_try == n_cur:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample_sequence(best_seq, n_try))

        alloc = min(remaining * 0.12, 25.0)

        a_try = nesterov_subgradient(a_try, max_iter=200, alpha_init=0.02, time_budget=alloc * 0.4)
        a_try = soft_max_gradient(a_try, tau_init=0.001, step_size=0.03,
                                  max_steps=150, time_budget=alloc * 0.4)
        a_try = targeted_repair(a_try, time_budget=alloc * 0.2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")
            n_cur = len(best_seq)

    # ============ Phase 5: Diverse restarts ============
    print(f"\n{'='*60}")
    print(f"Phase 5: Diverse restarts")
    print(f"{'='*60}")

    rng = np.random.default_rng(seed)
    n_cur = len(best_seq)

    for restart in range(8):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.1, 28.0)

        # Different perturbation strategies
        if restart % 3 == 0:
            # Frequency modulation
            freq = rng.uniform(1.0, 6.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.15 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
            a_try = normalize(best_seq * mod)
        elif restart % 3 == 1:
            # Blend with window
            wt = int(rng.integers(0, 5))
            w = window_init(n_cur, wt)
            mix = rng.uniform(0.3, 0.6)
            a_try = normalize((1 - mix) * normalize(best_seq) + mix * w)
        else:
            # Gaussian noise
            a_try = normalize(best_seq * np.exp(rng.normal(0, 0.1, n_cur)))

        a_try = nesterov_subgradient(a_try, max_iter=150, alpha_init=0.02, time_budget=alloc * 0.35)
        a_try = soft_max_gradient(a_try, tau_init=0.001, step_size=0.03,
                                  max_steps=100, time_budget=alloc * 0.35)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  R{restart}: {best_score:.8f} **")
        else:
            print(f"  R{restart}: {obj_try:.8f}")

    # ============ Phase 6: Final polish ============
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    n_final = len(best_seq)

    for polish in range(5):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * 0.2, 35.0)

        a_cur = normalize(best_seq.copy())

        # LP for the best (if n is manageable)
        if n_final <= 800 and alloc > 20:
            a_cur = lp_alternate(a_cur, max_iters=10, time_budget=alloc * 0.35)

        # Nesterov
        a_cur = nesterov_subgradient(a_cur, max_iter=200, alpha_init=0.01, time_budget=alloc * 0.25)

        # Soft-max
        a_cur = soft_max_gradient(a_cur, tau_init=0.0005, step_size=0.02,
                                  max_steps=100, time_budget=alloc * 0.2)

        # Targeted repair
        a_cur = targeted_repair(a_cur, time_budget=alloc * 0.2)

        obj_cur = compute_obj(a_cur)
        if obj_cur < best_score:
            best_score = obj_cur
            best_seq = a_cur.copy()
            print(f"  P{polish}: {best_score:.8f} **")
        else:
            print(f"  P{polish}: {obj_cur:.8f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
