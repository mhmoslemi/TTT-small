Actually, let me think about this problem from a different angle. The optimal solution to:
minimize max_k (a*a)_k subject to sum(a) = 1, a >= 0

is characterized by the KKT conditions. At the optimum:
- Let S = {k : (a*a)_k = M} be the active set (lags where the max is achieved).
- The gradient of the objective with respect to a_j (for a_j > 0) must be in the span of the active constraint gradients and the sum constraint gradient.

The gradient of (a*a)_k with respect to a_j is 2*a_{k-j} (as derived earlier).

So the KKT condition is: for all j with a_j > 0:
sum_{k in S} lambda_k * 2 * a_{k-j} = nu (constant, from the sum constraint)

This means: sum_{k in S} lambda_k * a_{k-j} = constant for all j with a_j > 0.

This is a system of linear equations in the unknowns lambda_k and the constant. If we can solve this system, we can identify the active set S.

In practice, this suggests that at the optimum, the "shadow price" of each active lag, weighted by the dual variables, gives a constant "potential" at each occupied site j.

This is similar to the equilibrium condition in a multi-agent system: each agent (site j) is at a "potential" that's the sum of interactions from active lags.

**Algorithm idea: Use the KKT conditions to guide the optimization.**

1. Start with an initial sequence.
2. Identify the active set S (lags where conv_k is close to the max).
3. Solve the KKT linear system to find the optimal lambdas.
4. Use the residual (violation of KKT conditions) as a gradient direction.

def kkt_gradient(a, n_active=10):
    """
    Compute KKT-based gradient.
    1. Find the top n_active lags.
    2. Solve for lambda_k such that sum_k lambda_k * a_{k-j} is as constant as possible.
    3. The residual is the gradient direction.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Get top lags
    top_k = np.argsort(conv)[-n_active:]
    
    # Solve: find lambda_k to minimize var(sum_k lambda_k * a_{k-j}) over j
    # This is: find lambda to make B * lambda as constant as possible
    # where B[j, k_idx] = a_{top_k[k_idx] - j} (if valid)
    
    B = np.zeros((n, n_active))
    for idx, k in enumerate(top_k):
        for j in range(n):
            k2 = k - j
            if 0 <= k2 < n:
                B[j, idx] = a[k2]
    
    # We want B * lambda to be as close to a constant as possible.
    # Equivalently, minimize ||B * lambda - c * 1||^2 over lambda, c.
    # This is a least squares problem.
    # Augment: [B, 1] * [lambda; -c] = 0 (in the least squares sense)
    # Or: minimize ||B * lambda - mean(B * lambda) * 1||^2
    # = minimize ||(I - P) * B * lambda||^2 where P = (1/n) * 1 * 1^T
    # = minimize ||(I - P) * B * lambda||^2
    
    P = np.ones((n, n)) / n
    B_centered = (np.eye(n) - P) @ B
    
    # Least squares: minimize ||B_centered * lambda||^2
    # Solution: lambda = 0 (trivial). We need a constraint on lambda.
    # Let's constrain sum(lambda) = 1.
    # Then it's a constrained least squares: minimize ||B_centered * lambda||^2 s.t. sum(lambda) = 1, lambda >= 0
    
    # Use simple approach: project uniform lambda
    # Or use the SVD to find the minimum-norm solution
    U, s, Vt = np.linalg.svd(B_centered)
    # Minimum variance direction: Vt[0] gives the direction of maximum variance
    # We want the direction of minimum variance: Vt[-1]
    # But we need sum(lambda) = 1 and lambda >= 0
    
    # Simpler: just use the top lag
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    for j in range(n):
        k2 = k_star - j
        if 0 <= k2 < n:
            grad[j] = 2.0 * a[k2]
    
    return grad

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear autocorrelation a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    """Build Toeplitz matrix for conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=20):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status in [0, 3]:  # 0=optimal, 3=time limit with feasible solution
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, res
    except Exception:
        pass
    return None, None

def line_search(a, g, f_max_iter=20):
    """
    Find best t in [0,1] to minimize obj((1-t)*a + t*g).
    Returns (best_t, best_a, best_obj).

    g = normalize(g)
    best_t = 0.0

def l2_gradient_polish(a, max_steps=100, time_budget=30.0):
    """
    Polish using gradient of ||conv(a,a)||_2^2 (smooth surrogate).
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    step_size = 0.05
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        
        # Gradient of sum(conv_k^2) w.r.t. a:
        # d/da_j [sum_k conv_k^2] = 2 * sum_k conv_k * d(conv_k)/da_j
        # = 2 * sum_k conv_k * 2*a_{k-j} (where a_x=0 outside [0,n-1])
        # = 4 * sum_k conv_k * a_{k-j}
        # = 4 * (cross-correlation of conv and a)[j]
        
        Cft = rfft(conv, N)
        Aft = rfft(a, N)
        # cross_corr[j] = sum_k conv[k] * a[k-j] = ifft(fft(conv) * conj(fft(a)))[j]
        corr = irfft(Cft * np.conj(Aft), N)[:n]
        grad = 4.0 * corr
        
        # Project: remove mean (to preserve sum)
        grad = grad - np.mean(grad)
        
        a_new = normalize(a - step_size * grad)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step_size = min(step_size * 1.1, 0.2)
        else:
            step_size *= 0.5
            if step_size < 1e-10:
                break
    
    return best_a
