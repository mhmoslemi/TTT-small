"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Enhanced LP alternating with multi-point line search
2. Subgradient method with Nesterov-type running average
3. Careful n-sweeping around the parent length
4. Diverse random restarts for escaping local minima
Parent: n=700, score=1.506548. Target: < 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_optimize(a, time_budget, max_iters=80):
    """LP alternating with multi-point line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        lp_time = min(remaining * 0.4, 15.0)
        g = solve_lp(a, M, time_limit=lp_time)
        if g is None:
            break

        # Multi-point line search
        best_t = 0.0
        best_obj_t = best_obj
        t_candidates = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8]
        for t in t_candidates:
            a_try = normalize((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t

        if best_t > 0:
            a = normalize((1.0 - best_t) * a + best_t * g)
        
        obj_new = compute_obj(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

    return best_a


def subgradient_optimize(a, time_budget, max_steps=5000):
    """Subgradient method with running average."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Running average for Nesterov smoothing
    avg_a = a.copy()
    avg_obj = best_obj

    step_scale = 5.0

    for step in range(max_steps):
        if time.monotonic() - start > time_budget - 1:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        # Use soft-max weighting instead of hard active set for smoother gradient
        tau = cmax / (10 + step * 0.01)
        shifted = np.clip((conv - cmax) / tau, -100.0, 100.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum

        # Compute weighted subgradient
        grad = np.zeros(n)
        for k in range(len(weights)):
            if weights[k] < 1e-15:
                continue
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo > i_hi:
                continue
            k_minus_i = k - np.arange(i_lo, i_hi + 1)
            grad[i_lo:i_hi + 1] += weights[k] * 2.0 * a[k_minus_i]
            if k % 2 == 0 and i_lo <= k // 2 <= i_hi:
                grad[k // 2] -= weights[k] * a[k // 2]

        grad -= np.mean(grad)

        # Adaptive step size
        step_size = step_scale / np.sqrt(step + 1)
        a_new = normalize(a - step_size * grad)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.5 * a + 0.5 * a_new)

        # Update running average every 10 steps
        if (step + 1) % 10 == 0:
            t = (step + 1) // 10
            avg_a = avg_a * (t - 1) / t + a / t
            avg_obj = compute_obj(avg_a)
            if avg_obj < best_obj:
                best_obj = avg_obj
                best_a = avg_a.copy()

    return best_a


def spectral_smooth(a, time_budget, max_iters=100):
    """Spectral smoothing: flatten power spectrum while maintaining non-negativity."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 1:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Compress spectrum: P_new = P^alpha * p_max^(1-alpha)
        alpha = 0.7 + 0.2 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = P_norm ** alpha * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.6 * a + 0.4 * a_new)

    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

**Alternative: Use the fact that the problem is equivalent to a "minimax" problem.**

We want min_a max_k (a*a)[k] / (sum a)^2 * 2n.

This is a zero-sum game! Player 1 chooses a (from the simplex), Player 2 chooses k (from {0,...,2n-2}). Player 1's payoff is -(a*a)[k]/(sum a)^2 * 2n.

By von Neumann's minimax theorem, the value of this game equals:
min_{a in simplex} max_k (a*a)[k]/(sum a)^2 * 2n = max_{q in simplex_{2n-1}} min_{a in simplex} sum_k q[k] * (a*a)[k] / (sum a)^2 * 2n

The dual is: max over probability distributions q on {0,...,2n-2} of [min over a in simplex of sum_k q[k] * (a*a)[k]] * 2n.

The inner minimization is: min_{a: sum=1, a>=0} sum_k q[k] * (a*a)[k] = min_a a^T (sum_k q[k] Q_k) a = min_a a^T Q a

where Q = sum_k q[k] Q_k.

If Q is positive definite, the minimum over the simplex is achieved at an interior point (if the unconstrained minimum is on the simplex) or at a boundary point.

This dual formulation suggests an algorithm:
1. Start with some q (e.g., uniform)
2. Solve the inner QP: min a^T Q a s.t. sum a = 1, a >= 0
3. Update q: put more weight on the k's where (a*a)[k] is largest
4. Repeat

This is essentially the Lagrangian dual / alternating method. Let me think about whether this is different from the LP approach.

The LP approach: max sum(g) s.t. (a*g)[k] <= M, g >= 0. This is different - it's a primal approach that finds a direction g to move in.

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses enhanced LP alternating with dual-guided subgradient, spectral smoothing,
n-sweeping, and diverse restarts. Builds on parent n=700 (score 1.506548).
Target: < 1.5027.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns (g, dual_vars)."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            dual = None
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                dual = np.maximum(res.ineqlin, 0.0)
            if s > 1e-10:
                return g / s, dual
    except Exception:
        pass
    return None, None


def compute_subgradient(a, weights=None):
    """Compute (weighted) subgradient of max_k (a*a)[k]."""
    n = len(a)
    conv = compute_conv(a)
    if weights is None:
        cmax = float(np.max(conv))
        active = np.where(conv >= cmax - 1e-10)[0]
        weights = np.zeros(len(conv))
        weights[active] = 1.0 / len(active)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-15:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo > i_hi:
            continue
        k_minus_i = k - np.arange(i_lo, i_hi + 1)
        grad[i_lo:i_hi + 1] += w * 2.0 * a[k_minus_i]
        if k % 2 == 0 and i_lo <= k // 2 <= i_hi:
            grad[k // 2] -= w * a[k // 2]
    grad -= np.mean(grad)
    return grad


def lp_optimize(a, time_budget, max_iters=80):
    """LP alternating with multi-point line search and dual-guided subgradient."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        lp_time = min(remaining * 0.4, 12.0)
        g, dual = solve_lp(a, M, time_limit=lp_time)
        if g is None:
            break

        # Multi-point line search
        best_t = 0.0
        best_obj_t = best_obj
        t_candidates = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.5, 0.6, 0.7, 0.8]
        for t in t_candidates:
            a_try = normalize((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t

        if best_t > 0:
            a = normalize((1.0 - best_t) * a + best_t * g)

        # If no improvement from LP direction, try subgradient step
        if best_t == 0:
            if dual is not None and float(np.sum(dual)) > 1e-10:
                # Use dual variables as weights for subgradient
                d = dual / float(np.sum(dual))
                grad = compute_subgradient(a, weights=d)
            else:
                grad = compute_subgradient(a)
            step_size = 5.0 / np.sqrt(it + 1)
            a_new = normalize(a - step_size * grad)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                a = a_new

        obj_new = compute_obj(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

    return best_a


def subgradient_optimize(a, time_budget, max_steps=5000):
    """Subgradient method with adaptive step and running average."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    avg_a = a.copy()
    step_scale = 8.0

    for step in range(max_steps):
        if time.monotonic() - start > time_budget - 1:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Soft-max weighting over convolution peaks
        tau = max(cmax / 15.0, 1e-10)
        shifted = np.clip((conv - cmax) / tau, -80.0, 80.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum

        grad = compute_subgradient(a, weights=weights)

        step_size = step_scale / np.sqrt(step + 1)
        a_new = normalize(a - step_size * grad)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Small blend to maintain diversity
            a = normalize(0.7 * a + 0.3 * a_new)

        # Running average check every 20 steps
        if (step + 1) % 20 == 0:
            t = (step + 1) // 20
            avg_a = avg_a * (t - 1) / t + a / t
            avg_obj = compute_obj(avg_a)
            if avg_obj < best_obj:
                best_obj = avg_obj
                best_a = avg_a.copy()

    return best_a


def spectral_smooth(a, time_budget, max_iters=100):
    """Spectral smoothing: iteratively flatten power spectrum."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 1:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Progressive compression
        alpha = 0.6 + 0.35 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = P_norm ** alpha * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.65 * a + 0.35 * a_new)

    return best_a


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

            rng = np.random.default_rng(cyc * 7919 + 42)
            a = normalize(a * np.exp(rng.normal(0, 0.025 + 0.01 * cyc, n_parent)))

    # Phase 2: Dual QP + Subgradient + Spectral
    print(f"\n--- Phase 2: Dual QP + Subgradient + Spectral ---")

    if time.monotonic() < deadline - 200:
        a_dual = dual_qp_optimize(best_seq, time_budget=50.0, max_iters=40)
        obj_dual = compute_obj(a_dual)
        if obj_dual < best_score:
            best_score = obj_dual
            best_seq = a_dual.copy()
            print(f"  Dual QP: {best_score:.8f} ***")
        else:
            print(f"  Dual QP: {obj_dual:.8f}")

    if time.monotonic() < deadline - 180:
        a_sub = subgradient_optimize(best_seq, time_budget=50.0, max_steps=4000)
        obj_sub = compute_obj(a_sub)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  Subgradient: {best_score:.8f} ***")
        else:
            print(f"  Subgradient: {obj_sub:.8f}")

    if time.monotonic() < deadline - 150:
        a_spec = spectral_smooth(best_seq, time_budget=35.0, max_iters=80)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: {best_score:.8f} ***")
        else:
            print(f"  Spectral: {obj_spec:.8f}")
            # Blend with current best for next LP
            a = normalize(0.7 * best_seq + 0.3 * a_spec)

    # LP follow-up after Phase 2
    if time.monotonic() < deadline - 120:
        a = lp_optimize(best_seq, time_budget=70.0, max_iters=40)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Post-Phase2 LP: {best_score:.8f} ***")
        else:
            print(f"  Post-Phase2 LP: {obj:.8f}")

    # Phase 3: n-sweeping
    print(f"\n--- Phase 3: n-sweeping ---")
    n_cur = len(best_seq)
    n_options = sorted(set([
        n_cur - 100, n_cur - 75, n_cur - 50, n_cur - 25,
        n_cur + 25, n_cur + 50, n_cur + 75, n_cur + 100,
        n_cur + 150, n_cur + 200, n_cur + 300
    ]))
    n_options = [x for x in n_options if 300 <= x <= 1500]

    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        a_try = normalize(resample(best_seq, n_try))
        alloc = min(remaining * 0.1, 20.0)

        if n_try <= 900:
            a_try = lp_optimize(a_try, time_budget=alloc * 0.6, max_iters=12)
        a_try = subgradient_optimize(a_try, time_budget=alloc * 0.35, max_steps=500)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: {best_score:.8f} ***")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # Phase 4: Random restarts
    print(f"\n--- Phase 4: Random restarts ---")
    n_cur = len(best_seq)
    rng = np.random.default_rng(seed)

    for restart in range(6):
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.12, 30.0)

        if restart % 3 == 0:
            a_try = best_seq * np.exp(rng.normal(0, 0.05 + 0.01 * restart, n_cur))
        elif restart % 3 == 1:
            freq = rng.uniform(1.0, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.08 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
            a_try = best_seq * mod
        else:
            a_rand = rng.dirichlet(np.ones(n_cur) * 0.4)
            mix = 0.15 + 0.05 * restart
            a_try = (1 - mix) * normalize(best_seq) + mix * a_rand

        a_try = normalize(a_try)
        if n_cur <= 900:
            a_try = lp_optimize(a_try, time_budget=alloc * 0.5, max_iters=12)
        a_try = subgradient_optimize(a_try, time_budget=alloc * 0.35, max_steps=400)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  Restart {restart}: {best_score:.8f} ***")
        else:
            print(f"  Restart {restart}: {obj_try:.8f}")

    # Phase 5: Final intensive polish
    print(f"\n--- Phase 5: Final polish ---")
    if time.monotonic() < deadline - 5:
        a_final = normalize(best_seq.copy())

        for polish in range(3):
            if time.monotonic() > deadline - 3:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            alloc = min(remaining * 0.4, 25.0)

            a_final = lp_optimize(a_final, time_budget=alloc, max_iters=25)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Polish {polish+1}: {best_score:.8f} ***")
            else:
                print(f"  Polish {polish+1}: {obj_final:.8f}")

    # Validate and return
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
