from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def coordinate_update(a, i):
    """Find optimal value for a[i] using golden section search."""
    n = len(a)
    s_others = float(np.sum(a)) - a[i]
    if s_others < 1e-15:
        return a.copy()
    
    # Upper bound: we don't want a[i] to be too large
    # If a[i] = s_others, then sum = 2*s_others
    # A reasonable upper bound is 2 * mean
    upper = max(2.0 * np.mean(a), a[i] * 3.0, 0.1 / n)
    lower = 0.0
    
    # Define objective as function of t = a[i]
    def obj_at_t(t):
        a_try = a.copy()
        a_try[i] = t
        return compute_obj(a_try)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = upper - gr * (upper - lower)
    d = lower + gr * (upper - lower)
    fc = obj_at_t(c)
    fd = obj_at_t(d)
    
    for _ in range(20):
        if upper - lower < 1e-8:
            break
        if fc < fd:
            upper = d
            d = c
            fd = fc
            c = upper - gr * (upper - lower)
            fc = obj_at_t(c)
        else:
            lower = c
            c = d
            fc = fd
            d = lower + gr * (upper - lower)
            fd = obj_at_t(d)
    
    best_t = (lower + upper) / 2.0
    a_new = a.copy()
    a_new[i] = best_t
    return a_new

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent,
simulated annealing, spectral shaping, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize_a(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def coordinate_descent(a, max_passes=100, time_budget=60.0, rng=None):
    """Exact coordinate descent with golden section search per coordinate."""
    n = len(a)
    a = normalize_a(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute FFT size
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    coord_order = np.arange(n)
    
    for pass_num in range(max_passes):
        if time.monotonic() - start > time_budget:
            break
        
        # Shuffle coordinate order for diversity
        if rng is not None:
            rng.shuffle(coord_order)
        
        improved = False
        for i in coord_order:
            # Current value
            a_i = a[i]
            # Search range
            # Lower bound: 0
            # Upper bound: something reasonable
            # The objective is scale-invariant, so the "scale" is determined by the sum
            # A reasonable upper bound for a[i] is max(a) * 3 or 2/n (twice the uniform)
            upper = max(3.0 * a_i if a_i > 0 else 0.0, 4.0 / n, 0.1 / n)
            lower = 0.0
            
            # Golden section search
            c = upper - gr * (upper - lower)
            d = lower + gr * (upper - lower)
            
            a[i] = c
            fc = compute_obj(a)
            a[i] = d
            fd = compute_obj(a)
            
            for _ in range(15):
                if upper - lower < 1e-10:
                    break
                if fc < fd:
                    upper = d
                    d = c
                    fd = fc
                    c = upper - gr * (upper - lower)
                    a[i] = c
                    fc = compute_obj(a)
                else:
                    lower = c
                    c = d
                    fc = fd
                    d = lower + gr * (upper - lower)
                    a[i] = d
                    fd = compute_obj(a)
            
            # Set to best
            best_t = (lower + upper) / 2.0
            a[i] = best_t
            
            # Check improvement
            obj_i = compute_obj(a)
            if obj_i < best_obj - 1e-12:
                best_obj = obj_i
                best_a = a.copy()
                improved = True
        
        if not improved and pass_num > 5:
            # Try perturbation to escape local min
            if rng is not None:
                sigma = 0.02 / (1 + 0.1 * pass_num)
                a = normalize_a(a * np.exp(rng.normal(0, sigma, n)))
                if compute_obj(a) > best_obj:
                    a = best_a.copy()
    
    return best_a

def soft_max_gradient(a, time_budget=30.0, tau_init=0.001, step_size=0.05, max_steps=100):
    """Soft-max (log-sum-exp) weighted gradient descent."""
    n = len(a)
    a = normalize_a(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Adaptive temperature
        tau = max(tau_init, cmax / (n * (step + 1) * 2))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (circular correlation trick)
        # Use FFT: grad = 2 * IFFT(FFT(weights) * FFT(a_reversed))
        a_flip = a[::-1]
        Wf = rfft(weights, fft_size)
        Af = rfft(a_flip, fft_size)
        corr = irfft(Wf * Af, fft_size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        
        # Center the gradient (projection onto sum=const)
        grad_centered = grad - np.mean(grad)
        
        # Update
        a_new = a - step_size * grad_centered
        a_new = normalize_a(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.7
            if step_size < 1e-8:
                break
            a = a  # keep current
    
    return best_a

def simulated_annealing(a, time_budget=60.0, rng=None):
    """Simulated annealing with various perturbation strategies."""
    n = len(a)
    a = normalize_a(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if rng is None:
        rng = np.random.default_rng()
    
    T = 0.01  # Initial temperature
    T_min = 0.0001
    cooling = 0.995
    
    while time.monotonic() - start < time_budget:
        if T < T_min:
            break
        
        # Choose perturbation type
        r = rng.random()
        if r < 0.3:
            # Single coordinate change
            i = int(rng.integers(0, n))
            a_try = a.copy()
            a_try[i] = max(0, a_try[i] * np.exp(rng.normal(0, 0.3)))
            a_try = normalize_a(a_try)
        elif r < 0.5:
            # Pair swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a_try = a.copy()
            delta = 0.1 * (a_try[i] - a_try[j])
            a_try[i] -= delta
            a_try[j] += delta
            a_try = normalize_a(a_try)
        elif r < 0.7:
            # Block reversal
            block_len = int(rng.integers(max(2, n//10), max(3, n//3)))
            start_idx = int(rng.integers(0, n - block_len))
            a_try = a.copy()
            a_try[start_idx:start_idx+block_len] = a_try[start_idx:start_idx+block_len][::-1]
            a_try = normalize_a(a_try)
        elif r < 0.85:
            # Gaussian perturbation
            sigma = 0.05 + 0.1 * T / 0.01
            a_try = normalize_a(a * np.exp(rng.normal(0, sigma, n)))
        else:
            # Move mass from peak to random
            conv = compute_conv(a)
            k_peak = int(np.argmax(conv))
            # Find contributor
            contributors = []
            for j in range(n):
                k2 = k_peak - j
                if 0 <= k2 < n:
                    contributors.append(j)
            if contributors:
                i = int(rng.choice(contributors))
                j = int(rng.integers(0, n))
                if i != j:
                    a_try = a.copy()
                    delta = 0.05 * a_try[i]
                    a_try[i] -= delta
                    a_try[j] += delta
                    a_try = normalize_a(a_try)
                else:
                    T *= cooling
                    continue
            else:
                T *= cooling
                continue
        
        obj_try = compute_obj(a_try)
        obj_cur = compute_obj(a)
        
        # Metropolis acceptance
        if obj_try < obj_cur or rng.random() < np.exp(-(obj_try - obj_cur) / T):
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()
        
        T *= cooling
    
    return best_a

def spectral_equalization(a, max_iters=40, time_budget=30.0):
    """POCS-style spectral equalization."""
    n = len(a)
    a = normalize_a(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        
        alpha = 0.3 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.6 * a + 0.4 * a_new
            a = normalize_a(a)
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def generate_initializations(n, rng):
    """Generate diverse initializations for a given n."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # U-shape
    k = np.arange(n)
    u = (2.0 * k / (n - 1) - 1.0) ** 2
    inits.append(u / np.sum(u))
    # Inverted U
    iu = 1.0 - u
    inits.append(iu / np.sum(iu))
    # W-shape
    w = (np.cos(4 * np.pi * k / (n - 1)) + 1.0) / 2.0
    inits.append(w / np.sum(w))
    # Exponential ramp
    for alpha in [1.0, 2.0]:
        e = np.exp(alpha * k / (n - 1))
        inits.append(e / np.sum(e))
    # Random
    inits.append(rng.dirichlet(np.ones(n)) )
    # Two peaks
    peak1 = np.exp(-((k - n*0.25)**2) / (2 * (n*0.1)**2))
    peak2 = np.exp(-((k - n*0.75)**2) / (2 * (n*0.1)**2))
    inits.append((peak1 + peak2) / (np.sum(peak1) + np.sum(peak2)))
    return inits

from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]
