    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):

    s = float(np.sum(a))

    return a / s

    m = 2 * n - 1

        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def mirror_descent(a, max_iters=500, time_budget=300.0, tau_init=0.05, eta_init=0.1):
    """Mirror descent with soft-max subgradient and multiplicative weight updates."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_init
    eta = eta_init
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, conv, M = soft_max_subgradient(a, tau)
        # Center the gradient (project onto sum=const hyperplane)
        grad_centered = grad - np.mean(grad)

        # Multiplicative weight update
        a_new = a * np.exp(-eta * grad_centered)
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve_count = 0
        else:
            no_improve_count += 1
            eta *= 0.7
            if no_improve_count > 30:
                # Restart with perturbation
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                a = normalize(best_a * np.exp(rng.normal(0, 0.05, n)))
                best_a = a.copy()
                no_improve_count = 0

        # Cool temperature
        tau *= 0.995
        # Recover step size periodically
        if it % 20 == 0 and eta < eta_init * 0.1:
            eta = eta_init * 0.5

    return best_a


def lp_alternate(a, max_iters=30, time_budget=300.0):
    """LP alternating with golden-section line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_at_t(t):
        a_try = normalize((1.0 - t) * a + t * g)
        return compute_obj(a_try)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=max(5, int(remaining * 0.5)))
        if g is None:
            break

        # Golden section search for t
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = obj_at_t(c)
        fd = obj_at_t(d)
        for _ in range(12):
            if hi - lo < 0.01:
                break
            if fc < fd:
                hi = d; d = c; fd = fc
                c = hi - gr * (hi - lo); fc = obj_at_t(c)
            else:
                lo = c; c = d; fc = fd
                d = lo + gr * (hi - lo); fd = obj_at_t(d)
        t_opt = (lo + hi) / 2.0

        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            rng = np.random.default_rng((it * 104729 + 31) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))

    return best_a

        # Since we're optimizing F(x) = 2n * L(x/s) / s^2 where s = sum(x)
        # dF/dx[m] = 2n * [dL/dxn[m] / s * s^2 - L * 2*s] / s^4
        # = 2n * [dL/dxn[m] / s - 2*L/s^2] ... wait let me be more careful

        # F(x) = 2n * L(xn) / s^2 where xn = x/s
        # dF/dx[m] = 2n * [dL/dxn[m] * dxn[m]/dx[m] * 1/s^2 + L * d(1/s^2)/dx[m]]
        # dxn[m]/dx[m] = 1/s - x[m]/s^2 = (s-x[m])/s^2
        # d(1/s^2)/dx[m] = -2/s^3
        # dF/dx[m] = 2n * [dL/dxn[m] * (s-x[m])/s^4 - 2*L/s^3]

        # Compute dL/dxn
        dLdxn = np.zeros(n)
        for m in range(n):
            k_lo = m
            k_hi = min(m + n - 1, 2 * n - 2)
            if k_lo <= k_hi:
                dLdxn[m] = (2.0 / tau) * float(np.dot(w[k_lo:k_hi+1], xn[0:k_hi-k_lo+1][::-1]))

        # Full gradient
        s2 = s * s
        s4 = s2 * s2
        s3 = s2 * s
        grad = 2.0 * n * (dLdxn * (s - x) / s4 - 2.0 * log_sum / s3)

        val = 2.0 * n * log_sum / s2
        return val, grad

    # Anneal tau
    n_phases = 4
    phase_time = time_budget / n_phases
    current = a.copy()

    for phase in range(n_phases):
        if time.monotonic() - start > time_budget:
            break
        frac = float(phase) / max(1, n_phases - 1)
        tau = tau_start * (tau_end / tau_start) ** frac

        phase_deadline = time.monotonic() + phase_time - 2

        def obj_wrap(x):
            if time.monotonic() > phase_deadline:
                raise TimeoutError()
            val, _ = smooth_obj_and_grad(x, tau)
            return val

        def obj_grad_wrap(x):
            if time.monotonic() > phase_deadline:
                raise TimeoutError()
            return smooth_obj_and_grad(x, tau)

        try:
            res = fmin_l_bfgs_b(obj_wrap, current, fprime=None,
                                  approx_grad=True,
                                  pgtol=1e-6, maxiter=50, maxfun=200,
                                  factr=1e4,
                                  bounds=[(0.0, None)] * n)
            if res[0] < best_obj:
                obj_check = compute_obj(res[0])
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = normalize(res[0]).copy()
                    current = best_a.copy()
        except (TimeoutError, Exception) as e:
            # Try with smaller maxiter
            try:
                res = fmin_l_bfgs_b(obj_wrap, current, fprime=None,
                                      approx_grad=True,
                                      pgtol=1e-4, maxiter=20, maxfun=80,
                                      factr=1e4,
                                      bounds=[(0.0, None)] * n)
                if res[0] < best_obj:
                    obj_check = compute_obj(res[0])
                    if obj_check < best_obj:
                        best_obj = obj_check
                        best_a = normalize(res[0]).copy()
                        current = best_a.copy()
            except Exception:
                pass

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def coordinate_polish(a, time_budget=30.0):
    """Coordinate descent: try moving mass from peak contributors to random positions."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contrib[:4]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            for _ in range(4):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue
                a_try = a.copy()
                delta = 0.03 * a_try[j1]
                a_try[j1] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and rng.random() < 0.15:
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a


def load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase strategy:
    Phase 1: Mirror descent (fast iterations, good for initial progress)
    Phase 2: LP alternating (strong local optimization)
    Phase 3: L-BFGS on smoothed objective (second-order fine-tuning)
    Phase 4: Multi-n exploration
    Phase 5: Final polishing (coordinate descent + short LP)
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_best = len(best_seq) if best_seq is not None else 600
    print(f"Initial: n={n_best}, score={best_score:.8f}")

    # =========================================================
    # Phase 1: Mirror descent (main workhorse) - ~300s
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Mirror descent (n={n_best})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())

        # Multiple mirror descent restarts with different parameters
        for md_round in range(3):
            if time.monotonic() > deadline - 600:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.15, 100.0)

            if md_round == 0:
                tau_init = 0.05
            elif md_round == 1:
                tau_init = 0.02
            else:
                tau_init = 0.01

            a_md = mirror_descent(a, max_iters=400, time_budget=alloc,
                                  tau_init=tau_init, eta_init=0.08)
            obj_md = compute_obj(a_md)
            if obj_md < best_score:
                best_score = obj_md
                best_seq = a_md.copy()
                a = a_md
                print(f"  Mirror descent round {md_round}: {best_score:.8f}")
            else:
                print(f"  Mirror descent round {md_round}: {obj_md:.8f} (no improvement)")
                a = a_md  # still use it as starting point for next phase

    # =========================================================
    # Phase 2: LP alternating - ~400s
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP alternating (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None and len(best_seq) <= 1000:
        a = normalize(best_seq.copy())

        for lp_round in range(4):
            if time.monotonic() > deadline - 350:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.2, 100.0)

            a_lp = lp_alternate(a, max_iters=25, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a = a_lp
                print(f"  LP round {lp_round}: {best_score:.8f}")
            else:
                print(f"  LP round {lp_round}: {obj_lp:.8f} (no improvement)")
                # Perturb and continue
                rng = np.random.default_rng((lp_round * 997 + 7) % (2**31))
                a = normalize(a * np.exp(rng.normal(0, 0.03, len(a))))

    # =========================================================
    # Phase 3: L-BFGS on smoothed objective - ~250s
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: L-BFGS smoothed (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None and len(best_seq) <= 800:
        if time.monotonic() < deadline - 250:
            a_lbs = l_bfgs_smoothed(best_seq, time_budget=min(120, (deadline - time.monotonic()) * 0.3),
                                     tau_start=0.03, tau_end=0.002)
            obj_lbs = compute_obj(a_lbs)
            if obj_lbs < best_score:
                best_score = obj_lbs
                best_seq = a_lbs.copy()
                print(f"  L-BFGS smoothed: {best_score:.8f}")
            else:
                print(f"  L-BFGS smoothed: {obj_lbs:.8f} (no improvement)")

    # =========================================================
    # Phase 4: Multi-n exploration - ~250s
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_cur = len(best_seq) if best_seq is not None else 600
    n_candidates = sorted(set([
        n_cur - 200, n_cur - 150, n_cur - 100, n_cur - 75,
        n_cur - 50, n_cur - 25,
        n_cur,
        n_cur + 25, n_cur + 50, n_cur + 75, n_cur + 100,
        n_cur + 150, n_cur + 200, n_cur + 300
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 1100]

    # Sort by distance from current n (try nearby first)
    n_candidates.sort(key=lambda x: abs(x - n_cur))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if n_try == n_cur:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample_sequence(best_seq, n_try))

        # Quick optimization at this n
        alloc = min(remaining * 0.1, 40.0)
        if n_try <= 800:
            a_try = mirror_descent(a_try, max_iters=100, time_budget=alloc * 0.5,
                                   tau_init=0.03, eta_init=0.05)
        if n_try <= 900:
            a_try = lp_alternate(a_try, max_iters=8, time_budget=alloc * 0.3)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 5: Final polishing - ~100s
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        n_final = len(a_final)

        for polish_round in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.25, 35.0)

            # Coordinate descent
            a_pol = coordinate_polish(a_final, time_budget=min(15, alloc * 0.4))
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish_round} (coord): {best_score:.8f}")

            # Short LP if time allows
            if n_final <= 800 and alloc > 20:
                a_pol2 = lp_alternate(a_final, max_iters=8, time_budget=min(15, alloc * 0.3))
                obj_pol2 = compute_obj(a_pol2)
                if obj_pol2 < best_score:
                    best_score = obj_pol2
                    best_seq = a_pol2.copy()
                    a_final = a_pol2
                    print(f"  Polish {polish_round} (LP): {best_score:.8f}")
                else:
                    a_final = a_pol2

            # Short mirror descent
            if alloc > 15:
                a_pol3 = mirror_descent(a_final, max_iters=50, time_budget=min(10, alloc * 0.2),
                                        tau_init=0.005, eta_init=0.03)
                obj_pol3 = compute_obj(a_pol3)
                if obj_pol3 < best_score:
                    best_score = obj_pol3
                    best_seq = a_pol3.copy()
                    a_final = a_pol3
                    print(f"  Polish {polish_round} (MD): {best_score:.8f}")
                else:
                    a_final = a_pol3

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(400)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 200

    # Validate with the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
