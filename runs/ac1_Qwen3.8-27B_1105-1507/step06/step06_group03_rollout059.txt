"""
Optimized step function search using spectral flattening (Gerchberg-Saxton),
ramp-based initialization, and coordinate descent. Targets J < 1.5027.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the set {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def normalize_to_sum1(a):
    """Normalize to sum=1, keeping non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def gerchberg_saxton(a, max_iters, deadline, target_flat=True):
    """
    Gerchberg-Saxton style spectral flattening.
    Tries to make the power spectrum flat (minimizing autocorrelation peaks).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = normalize_to_sum1(a)
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Forward FFT
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: flat magnitude (except DC)
        target_mag = np.full(size // 2 + 1, 1.0 / np.sqrt(n))
        target_mag[0] = mag[0]  # Preserve DC = sum(a) = 1

        # Blend: partial step toward target
        A_new = phase * np.where(mag > 1e-15,
                                  mag * 0.5 + target_mag * 0.5,
                                  target_mag)
        A_new[0] = mag[0]

        # Inverse FFT
        a_new = np.fft.irfft(A_new, size)[:n]

        # Project to non-negative, sum=1
        a_new = np.maximum(a_new, 0)
        a_new = normalize_to_sum1(a_new)

        obj = compute_J(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_obj


def gs_with_cyclic_shift(a, max_iters, deadline):
    """
    GS with cyclic shift trick: shift the sequence before FFT to exploit
    circular vs linear correlation differences. Tries multiple shifts.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = normalize_to_sum1(a)
    best_a = a.copy()
    best_obj = compute_J(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for shift in range(min(5, n // 8)):
        if time.monotonic() > deadline:
            break

        working = np.roll(a, shift)
        no_improve = 0

        for it in range(max_iters // max(1, min(5, n // 8))):
            if time.monotonic() > deadline:
                break

            A = np.fft.rfft(working, size)
            mag = np.abs(A)
            phase = np.angle(A)

            target_mag = np.full(size // 2 + 1, 1.0 / np.sqrt(n))
            target_mag[0] = mag[0]

            A_new = phase * (0.7 * mag + 0.3 * target_mag)
            A_new[0] = mag[0]

            working_new = np.fft.irfft(A_new, size)[:n]
            working_new = np.maximum(working_new, 0)
            working_new = normalize_to_sum1(working_new)

            working = working_new
            obj_unrolled = compute_J(np.roll(working_new, -shift))
            if obj_unrolled < best_obj - 1e-10:
                best_obj = obj_unrolled
                best_a = np.roll(working_new, -shift).copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 30:
                break

    return best_a, best_obj


def coord_descent_fast(a, max_sweeps, deadline, random_order=True, seed=42):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = normalize_to_sum1(a)
    s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved = False

        for i in order:
            old = a[i]
            if old < 1e-12:
                candidates = [0.0, 1e-4, 1e-3, 5e-3]
            else:
                candidates = np.array([0.0, 0.25 * old, 0.5 * old, 0.75 * old,
                                       1.0 * old, 1.25 * old, 1.5 * old, 2.0 * old])

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()

            best_local = best_obj
            best_val = old

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                kq = 2 * i
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta

                new_s = s + delta
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                kq = 2 * i
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s + delta
                improved = True

        a = normalize_to_sum1(a)
        s = 1.0
        # Recompute conv after normalization fix
        if abs(s - 1.0) > 1e-12:
            conv = fast_conv(a)

        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

    return best_a, best_obj

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()

            # Try a range of multipliers
            best_local = best_obj
            best_val = old
            for mult in [0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 1.7, 2.0, 3.0]:
                x = old * mult
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                kq = 2 * i
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta

                new_obj = 2.0 * n * conv.max()  # s stays 1 since we'll renormalize

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                kq = 2 * i
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = best_val
                improved = True

        a = normalize_to_sum1(a)
        if improved:
            conv = fast_conv(a)
        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            break

    return best_a, best_obj


def lp_step(a, max_iters, deadline):
    """LP-based step: solve LP to find direction that reduces max autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = normalize_to_sum1(a)
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # LP: minimize t s.t. conv(a, g)[k] <= M * (1 + t) for all k, sum(g)=1, g>=0
        # Simplified: minimize t s.t. conv(a,g)[k] - t <= M for all k
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(M)

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_eq = coo_matrix((np.ones(n), (list(range(n)), list(range(n)))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]
        bounds = [(0, None)] * n + [(None, None)]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                g = normalize_to_sum1(g)
                found = False
                for frac in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
                    a_try = (1 - frac) * a + frac * g
                    a_try = normalize_to_sum1(a_try)
                    J_try = compute_J(a_try)
                    if J_try < best_obj - 1e-12:
                        a = a_try
                        best_obj = J_try
                        best_a = a_try.copy()
                        no_improve = 0
                        found = True
                        break
                if not found:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 15:
            break

    return best_a, best_obj


def make_ramp(n, alpha=1.0, shift=0.0):
    """Create a power-law ramp: a[i] = (i+1+shift)^alpha, normalized."""
    t = np.arange(n, dtype=np.float64) + 1.0 + shift
    a = np.power(np.maximum(t, 1e-6), alpha)
    return normalize_to_sum1(a)


def make_modulated_ramp(n, alpha=1.0, freq=0.0, amp=0.0):
    """Ramp with sinusoidal modulation."""
    t = np.arange(n, dtype=np.float64)
    base = (t + 1.0) ** alpha
    if freq > 0:
        mod = 1.0 + amp * np.sin(2 * np.pi * freq * t / n)
        base = base * np.maximum(mod, 0.1)
    return normalize_to_sum1(base)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def solve_pipeline(a, deadline, verbose=False):
    """Full optimization pipeline: GS -> CD -> LP -> CD."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = normalize_to_sum1(a)
    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = normalize_to_sum1(a_new)
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            if verbose:
                print(f"      Updated: J={obj:.8f}", flush=True)
        return best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 1: Gerchberg-Saxton (40% of time)
    t1 = time.monotonic() + 0.40 * remaining
    if verbose:
        print(f"    GS (budget {t1-time.monotonic():.0f}s)", flush=True)
    a_gs, obj_gs = gerchberg_saxton(a, max_iters=2000, deadline=t1)
    if obj_gs < best_obj:
        best_a = a_gs.copy()
        best_obj = obj_gs
        a = a_gs.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 2: Coordinate descent (35% of time)
    t2 = time.monotonic() + 0.35 * remaining
    if verbose:
        print(f"    CD (budget {t2-time.monotonic():.0f}s)", flush=True)
    a_cd, obj_cd = coord_descent_fast(a, max_sweeps=50000, deadline=t2,
                                       random_order=True, seed=42)
    if obj_cd < best_obj:
        best_a = a_cd.copy()
        best_obj = obj_cd
        a = a_cd.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 3: LP (20% of time)
    t3 = time.monotonic() + 0.20 * remaining
    if verbose:
        print(f"    LP (budget {t3-time.monotonic():.0f}s)", flush=True)
    a_lp, obj_lp = lp_step(a, max_iters=50, deadline=t3)
    if obj_lp < best_obj:
        best_a = a_lp.copy()
        best_obj = obj_lp
        a = a_lp.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 4: Final CD (5% of time)
    a_final, obj_final = coord_descent_fast(a, max_sweeps=20000,
                                             deadline=deadline - 1,
                                             random_order=True, seed=99)
    if obj_final < best_obj:
        best_a = a_final.copy()
        best_obj = obj_final

    return best_a, best_obj
