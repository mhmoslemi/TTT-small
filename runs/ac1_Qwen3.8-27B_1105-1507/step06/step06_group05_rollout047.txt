def momentum_subgradient(a, max_steps=200, time_budget=40.0, momentum=0.9, lr_init=0.05):
    """
    Subgradient descent with Heavy Ball momentum.
    Directly targets the peak of the convolution.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    velocity = np.zeros(n)
    lr = lr_init
    stall_count = 0

    size = 1
    while size < 3 * n:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        # Find all indices achieving the max
        peak_mask = (conv >= cmax - 1e-12)
        
        # Compute subgradient: grad[i] = 2 * sum_k peak_mask[k] * a[k-i] (if 0<=k-i<n)
        # Using FFT: grad = 2 * ifft(FFT(peak_mask) * conj(FFT(a)))[:n]
        A_f = rfft(a, size)
        P_f = rfft(peak_mask.astype(np.float64), size)
        corr = irfft(P_f * np.conj(A_f), size)
        grad = 2.0 * corr[:n]
        
        # Center the gradient
        grad_centered = grad - np.mean(grad)
        
        # Momentum update
        velocity = momentum * velocity - lr * grad_centered
        a_new = a + velocity
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            # Slowly decrease learning rate
            lr *= 0.999
        else:
            stall_count += 1
            if stall_count > 30:
                # Reset momentum and try smaller step
                velocity = np.zeros(n)
                lr *= 0.5
                stall_count = 0
                if lr < 1e-6:
                    break

    return best_a


def spectral_shaping_v2(a, max_iters=80, time_budget=50.0):
    """
    Improved spectral shaping: iterative peak reduction in frequency domain.
    Directly minimizes the peak of IFFT(|A|^2) by adjusting |A|.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Compute the time-domain peak from the spectrum
        # IFFT(|A|^2) gives the circular autocorrelation (which equals linear conv if size >= 2n-1)
        ac = np.real(ifft(P, size))[:2*n-1]
        ac_max = float(np.max(ac))
        
        # Strategy: reduce the spectral peaks that contribute most to the time-domain peak
        # The time-domain peak at lag k is: ac[k] = (1/size) * sum_m P[m] * exp(2*pi*i*m*k/size)
        # To reduce ac[k_max], we want to reduce P[m] where exp(2*pi*i*m*k_max/size) is "aligned"
        
        # Simpler approach: use a power spectrum compression that's adapted to the current peak
        # Higher compression where the spectrum is already concentrated
        p_norm = P / p_max
        
        # Adaptive compression exponent
        # Start aggressive, ease off
        alpha = 0.3 + 0.4 * (it / float(max_iters))
        
        # Also add a "flatten" component: push toward uniform
        uniform_target = np.sum(P) / size
        P_target = alpha * p_norm**0.5 * p_max + (1-alpha) * uniform_target
        
        # Blend
        P_new = 0.7 * P + 0.3 * P_target
        
        # Reconstruct
        mag = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = mag * np.exp(1j * phase)
        
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try gentler update
            P_gentle = 0.9 * P + 0.1 * P_target
            mag_g = np.sqrt(P_gentle)
            A_g = mag_g * np.exp(1j * phase)
            a_g = np.real(ifft(A_g, size))[:n]
            a_g = np.maximum(a_g, 0.0)
            s_g = float(np.sum(a_g))
            if s_g > 1e-10:
                a_g = a_g / s_g
                obj_g = compute_obj(a_g)
                if obj_g < best_obj:
                    best_obj = obj_g
                    best_a = a_g.copy()
                    a = a_g
                else:
                    # No improvement, try even gentler or break
                    if it > 30:
                        # Try blending with best
                        a = normalize(0.5 * best_a + 0.5 * a_new)

    return best_a


def chirp_init(n, chirp_rate=1.0):
    """Create a chirp-like initial sequence."""
    # A chirp in frequency domain: A[k] = exp(i * pi * rate * k^2 / n)
    # This gives a "spread out" autocorrelation
    k = np.arange(n)
    phase = np.pi * chirp_rate * k ** 2 / n
    A = np.exp(1j * phase)
    a = np.real(ifft(A, n))
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def raised_cosine_init(n):
    """Raised cosine window as initial sequence."""
    k = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * k / n))
    a = np.maximum(a, 1e-10)
    return a / float(np.sum(a))


def random_fourier_init(n, seed=42):
    """Random Fourier features as initial sequence."""
    rng = np.random.default_rng(seed)
    A = rng.normal(0, 1, n) + 1j * rng.normal(0, 1, n)
    # Ensure DC component is positive (for non-negativity)
    A[0] = np.abs(A[0]) + 1.0
    a = np.real(ifft(A, n))
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def coordinate_descent(a, time_budget=20.0):

    n = len(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = float(conv[k_star])

        # Find top contributors to the peak
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        improved = False
        for peak_val_c, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            # Try moving a fraction of mass from j1 to a random position
            for _ in range(5):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue

                for frac in [0.03, 0.05, 0.08, 0.12]:
                    delta = frac * a[j1]
                    if a[j1] - delta < 0:
                        continue
                    a_try = a.copy()
                    a_try[j1] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)

                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved:
            # Random perturbation
            if rng.random() < 0.3:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            # If we've been stuck, break
            if time.monotonic() - start > time_budget * 0.8:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf
