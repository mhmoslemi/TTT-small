def compute_conv(a):

    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:

                      options={'presolve': True, 'time_limit': time_limit})

            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:

    except Exception:
        pass
    return None

        s = float(np.sum(a_new))

        a = a_new / s

"""
Accelerated mirror descent (exponential weights) with annealed log-sum-exp smoothing,
LP corrections, spectral warm-start, and multi-n exploration for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: uses exponential weights update (a * exp(-eta*grad)) instead of additive
gradient descent, which naturally maintains the simplex constraint and provides
different optimization trajectories to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_gradient_and_conv(a, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*conv))) w.r.t. a,
    along with the convolution itself.
    grad[i] = 2 * sum_j a[j] * w[i+j] where w = softmax(beta * conv)
    """
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    exp_shifted = np.exp(np.clip(shifted, -500, 500))
    w_sum = float(np.sum(exp_shifted))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w = exp_shifted / w_sum
    # Cross-correlation: corr[k] = sum_j a[j] * w[j+k]
    # Using FFT: corr = irfft(conj(FFT(a)) * FFT(w))
    size = 1
    while size < 3 * n:
        size <<= 1
    A_f = rfft(a, size)
    W_f = rfft(w, size)
    corr = irfft(np.conj(A_f) * W_f, size)
    grad = 2.0 * corr[:n]
    return grad, conv, cmax


def exp_weights_step(a, grad, eta):
    """Exponential weights (mirror descent) step: a_new = a * exp(-eta*grad) / Z."""
    log_step = -eta * grad
    log_step = np.clip(log_step, -80.0, 80.0)
    a_new = a * np.exp(log_step)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=25):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def spectral_equalize(a, n_iters=20, alpha=0.6):
    """POCS-style spectral flattening: alternate frequency-domain compression with time-domain projection."""
    n = len(a)
    a = normalize(a.copy())
    size = 1
    while size < 2 * n:
        size <<= 1
    for it in range(n_iters):
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a = a_new / s
    return a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def mirror_descent_optimize(a, eta_init, mom, beta_start, beta_end,
                            max_steps, time_budget, eval_every=5):
    """
    Core accelerated mirror descent with exponential weights.
    Returns (best_a, best_obj, steps_done).
    """
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    eta = eta_init
    G = np.zeros(n)  # Accumulated gradient (FTRL-style)
    start = time.monotonic()
    steps = 0

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        progress = step / float(max(1, max_steps))
        beta = beta_start + (beta_end - beta_start) * progress

        grad, conv, cmax = compute_gradient_and_conv(a, beta)

        # Momentum update on gradient
        G = mom * G + (1.0 - mom) * grad

        # Exponential weights step
        a_new = exp_weights_step(a, G, eta)

        # Evaluate objective periodically
        if (step + 1) % eval_every == 0 or step == max_steps - 1:
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            # Adaptive step size
            old_obj = 2.0 * n * cmax  # approx old objective (sum normalized to 1)
            if obj_new <= old_obj + 1e-8:
                eta = min(eta * 1.005, 50.0 / n)
            else:
                eta *= 0.6
                if eta < 0.01 / n:
                    eta = 1.0 / n
                    G = grad.copy()

        a = a_new
        steps += 1

    return best_a, best_obj, steps


def lp_alternate(a, max_rounds=5, time_budget=120.0):
    """LP alternating method for global correction."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for r in range(max_rounds):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_max_sum(a, M, time_limit=max(5, int(remaining * 0.7)))
        if g is None:
            break

        # Try multiple blend factors
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = normalize((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a, best_obj


def coordinate_polish(a, time_budget=30.0, seed=123):
    """Local coordinate descent: move mass away from peak contributors."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    stagnant = 0

    while time.monotonic() - start < time_budget and stagnant < 50:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        pairs = []
        for j in range(n):
            j2 = k_star - j
            if 0 <= j2 < n:
                pairs.append((a[j] * a[j2], j, j2))
        pairs.sort(reverse=True)

        improved = False
        for _, j1, j2 in pairs[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            if a[j1] < 1e-8 and a[j2] < 1e-8:
                continue

            # Try moving mass to a random position
            for attempt in range(4):
                target = int(rng.integers(0, n))
                if target == j1 or target == j2:
                    continue
                frac = rng.uniform(0.02, 0.1)
                a_try = a.copy()
                src = j1 if a[j1] >= a[j2] else j2
                delta = frac * a_try[src]
                a_try[src] -= delta
                a_try[target] += delta
                if a_try[src] < 0:
                    continue
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    stagnant = 0
                    break

        if not improved:
            stagnant += 1
            # Random perturbation to escape
            if rng.random() < 0.3:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function using accelerated mirror descent (exponential weights)
    with annealed log-sum-exp, LP corrections, spectral warm-start, multi-n exploration,
    and coordinate descent polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # =========================================================
    # Phase 1: Spectral warm-start
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Spectral warm-start")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        a = spectral_equalize(a, n_iters=20, alpha=0.7)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  After spectral: {best_score:.8f}")
        else:
            # Blend: keep best, use spectral as starting point
            a = 0.5 * normalize(best_seq) + 0.5 * normalize(a)
            print(f"  Spectral: {obj:.8f} (no improvement, blending)")

    # =========================================================
    # Phase 2: Main mirror descent (exponential weights)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Accelerated mirror descent (exponential weights)")
    print(f"{'='*60}")

    if best_seq is not None:
        n = len(best_seq)
        a = normalize(best_seq.copy())

        # First pass: moderate beta (smooth)
        eta0 = 1.0 / n
        a_new, obj_new, steps = mirror_descent_optimize(
            a, eta_init=eta0, mom=0.7,
            beta_start=2.0, beta_end=20.0,
            max_steps=20000, time_budget=250.0, eval_every=3
        )
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Pass 1 (beta 2→20): {best_score:.8f} ({steps} steps)")
        else:
            print(f"  Pass 1 (beta 2→20): {obj_new:.8f} (no improvement)")
            a = normalize(0.7 * normalize(best_seq) + 0.3 * normalize(a_new))

        # Second pass: higher beta (sharper max approximation)
        if time.monotonic() < deadline - 300:
            a_start = normalize(best_seq.copy())
            eta1 = 0.5 / n
            a_new2, obj_new2, steps2 = mirror_descent_optimize(
                a_start, eta_init=eta1, mom=0.8,
                beta_start=20.0, beta_end=100.0,
                max_steps=30000, time_budget=300.0, eval_every=3
            )
            if obj_new2 < best_score:
                best_score = obj_new2
                best_seq = a_new2.copy()
                print(f"  Pass 2 (beta 20→100): {best_score:.8f} ({steps2} steps)")
            else:
                print(f"  Pass 2 (beta 20→100): {obj_new2:.8f} (no improvement)")

    # =========================================================
    # Phase 3: LP corrections (global optimization)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: LP corrections")
    print(f"{'='*60}")

    if best_seq is not None:
        n = len(best_seq)
        if n <= 900 and time.monotonic() < deadline - 200:
            a = normalize(best_seq.copy())
            a_lp, obj_lp = lp_alternate(a, max_rounds=6, time_budget=180.0)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  After LP: {best_score:.8f}")
            else:
                print(f"  LP: {obj_lp:.8f} (no improvement)")

    # =========================================================
    # Phase 4: Second round of mirror descent (fine tuning)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Fine-tuning mirror descent")
    print(f"{'='*60}")

    if best_seq is not None:
        n = len(best_seq)
        a = normalize(best_seq.copy())

        # Very high beta, small steps
        eta_fine = 0.1 / n
        a_fine, obj_fine, steps_fine = mirror_descent_optimize(
            a, eta_init=eta_fine, mom=0.85,
            beta_start=50.0, beta_end=200.0,
            max_steps=30000, time_budget=250.0, eval_every=5
        )
        if obj_fine < best_score:
            best_score = obj_fine
            best_seq = a_fine.copy()
            print(f"  Fine-tune: {best_score:.8f} ({steps_fine} steps)")
        else:
            print(f"  Fine-tune: {obj_fine:.8f} (no improvement)")

        # Another LP pass if time allows
        if n <= 900 and time.monotonic() < deadline - 100:
            a = normalize(best_seq.copy())
            a_lp2, obj_lp2 = lp_alternate(a, max_rounds=4, time_budget=80.0)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  LP pass 2: {best_score:.8f}")

    # =========================================================
    # Phase 5: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        n_options = sorted(set([
            n_cur - 100, n_cur - 75, n_cur - 50, n_cur - 25,
            n_cur, n_cur + 25, n_cur + 50, n_cur + 75, n_cur + 100,
            n_cur + 150
        ]))
        n_options = [n for n in n_options if 200 <= n <= 1200]

        for n_try in n_options:
            if time.monotonic() > deadline - 60:
                break
            if n_try == n_cur:
                continue

            a_try = normalize(resample_seq(best_seq, n_try))

            # Quick mirror descent at this n
            eta_n = 1.0 / n_try
            a_opt, obj_opt, _ = mirror_descent_optimize(
                a_try, eta_init=eta_n, mom=0.7,
                beta_start=5.0, beta_end=50.0,
                max_steps=5000, time_budget=40.0, eval_every=5
            )

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_opt:.8f}")

    # =========================================================
    # Phase 6: Final polish (coordinate descent + short mirror descent)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        n_fin = len(best_seq)
        a_fin = normalize(best_seq.copy())

        # Coordinate descent
        if time.monotonic() < deadline - 30:
            a_cd, obj_cd = coordinate_polish(a_fin, time_budget=30.0, seed=999)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                a_fin = a_cd
                print(f"  Coordinate: {best_score:.8f}")

        # Very short mirror descent with highest beta
        if time.monotonic() < deadline - 10:
            a_ms, obj_ms, _ = mirror_descent_optimize(
                a_fin, eta_init=0.05 / n_fin, mom=0.9,
                beta_start=100.0, beta_end=300.0,
                max_steps=10000, time_budget=30.0, eval_every=5
            )
            if obj_ms < best_score:
                best_score = obj_ms
                best_seq = a_ms.copy()
                print(f"  Final mirror: {best_score:.8f}")

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Ensure sum constraint
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate with the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
