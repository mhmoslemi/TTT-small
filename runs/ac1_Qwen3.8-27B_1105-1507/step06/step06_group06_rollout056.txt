"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Adam-based smoothed gradient descent on the max-convolution objective,
(2) FFT-based efficient gradient computation, (3) frequency-domain spectrum
flattening, (4) multi-start with asymmetric initializations, (5) LP-based
peak reduction, (6) adaptive temperature annealing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute full objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    a_norm = a / sum_a
    return compute_obj_normalized(a_norm)


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(a)
    if n == 0:
        return a
    # Use sort-based method
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
   # Find the largest rho such that u[rho-1] > (cssv[rho-1] - 1) / rho
    condition = u * rho_candidates > (cssv - 1)
    if not np.any(condition):
        return np.ones(n) / n
    rho = np.max(rho_candidates[condition])
    theta = (cssv[rho - 1] - 1.0) / rho
    w = np.maximum(a - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def softmax_weights(conv, tau):
    """Compute softmax weights over convolution values."""
    shifted = (conv - np.max(conv)) / tau
    # Clip to prevent overflow
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        idx = np.argmax(conv)
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights = weights / w_sum
    return weights


def compute_gradient_fft(a, weights):
    """
    Compute gradient of sum_k weights[k] * conv[k] with respect to a.
    grad[i] = 2 * sum_j a[j] * weights[i+j]
    Using FFT-based cross-correlation.
    """
    n = len(a)
    a_flip = a[::-1]
    N = 1
    while N < n + len(weights) - 1:
        N <<= 1
    af = rfft(a_flip, N)
    wf = rfft(weights, N)
    conv_result = irfft(af * wf, N)[:n + len(weights) - 1]
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad


def adam_smoothed_descent(a, max_iters=2000, time_budget=60.0, tau_init=0.05, 
                           tau_end=0.5, lr_init=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimization on the smoothed max-convolution objective.
    f(a) = 2n * tau * log(sum exp(conv[k]/tau)) (with sum(a)=1 constraint)
    Uses adaptive temperature (annealing from tau_init to tau_end).
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    conv = compute_conv(a)
    tau = tau_init
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive temperature
        progress = it / max(max_iters - 1, 1)
        tau = tau_init + (tau_end - tau_init) * progress
        # Also adapt based on the spread of conv values
        conv_range = np.max(conv) - np.min(conv)
        if conv_range > 1e-15:
            tau = max(tau, conv_range * 0.1)  # tau should be at least 10% of range
            tau = min(tau, conv_range * 2.0)  # but not more than 2x range
        
        # Compute weights
        weights = softmax_weights(conv, tau)
        
        # Compute gradient
        grad = compute_gradient_fft(a, weights)
        
        # Center gradient to preserve sum(a) = 1
        grad_centered = grad - np.mean(grad)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Learning rate with decay
        lr = lr_init / (1.0 + it * lr_init * 0.01)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Update
        a = a - step
        
        # Project
        a = project_simplex(a)
        
        # Compute new conv and objective
        conv = compute_conv(a)
        obj = compute_obj_normalized(a)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def peak_shaving(a, n_steps=100, time_budget=30.0):
    """
    Iteratively identify the peak convolution value and make targeted
    moves to reduce it.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    step_size = 0.05
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        
        # Find top peaks
        cmax = np.max(conv)
        peak_lag = 0.05  # Consider peaks within 5% of max
        
        peak_mask = conv >= cmax * (1 - peak_lag)
        if peak_mask.sum() == 0:
            peak_mask = conv >= cmax - 1e-10
        
        # Compute weighted gradient (uniform weights on peaks)
        w = peak_mask.astype(float)
        w = w / w.sum()
        
        grad = compute_gradient_fft(a, w)
        grad_centered = grad - np.mean(grad)
        
        # Try multiple step sizes
        best_step_obj = best_obj
        best_step_a = a.copy()
        
        for ss in [step_size, step_size * 0.5, step_size * 2, step_size * 0.25]:
            a_try = a - ss * grad_centered / (np.std(grad_centered) + 1e-10)
            a_try = project_simplex(a_try)
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()
        
        if best_step_obj < best_obj:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = best_step_a.copy()
        else:
            # Reduce step size
            step_size *= 0.7
    
    return best_a, best_obj


def spectrum_flatten(a, max_iters=200, time_budget=30.0, alpha=0.3):
    """
    Frequency-domain approach: iteratively flatten the power spectrum.
    The idea: if the power spectrum |A(f)|^2 is flat, the autocorrelation
    is a delta (peak at 0 only). We want to reduce the peak of the
    linear autocorrelation, so we iteratively adjust the spectrum.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, N)
        power = np.abs(A) ** 2
        
        # Target: flatten the power spectrum
        p_mean = np.mean(power)
        p_max = np.max(power)
        
        if p_max < 1e-15:
            break
        
        # Compress the power spectrum
        power_norm = power / p_max
        power_compressed = power_norm ** (1.0 + alpha)  # >1 compresses peaks
        power_compressed = power_compressed / p_mean * p_mean  # preserve total power
        
        # Reconstruct: keep phase, modify magnitude
        phase = np.angle(A)
        magnitude = np.sqrt(power_compressed)
        A_new = magnitude * np.exp(1j * phase)
        
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj


def lp_peak_reduction(a, M, time_budget=20.0):
    """
    Solve LP: maximize sum(g) subject to (a*g)[k] <= M for all k, g >= 0.
    Then blend g with a.
    """
    n = len(a)
    if n > 1500:
        return a, compute_obj_normalized(a)
    
    # Build constraint matrix: (a*g)[k] = sum_j g[j] * a[k-j] (with valid indices)
    # For each k = 0, ..., 2n-2: sum_j g[j] * a[k-j] <= M
    # This is a Toeplitz-like system
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, build the matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    
    c = -np.ones(n)  # Minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': min(time_budget, 15)})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
                # Blend
                best_a = a.copy()
                best_obj = compute_obj_normalized(a)
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - t) * a + t * g
                    a_try = project_simplex(a_try)
                    obj_try = compute_obj_normalized(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a, best_obj
    except Exception:
        pass
    
    return a, compute_obj_normalized(a)


def asymmetric_init(n, alpha, rng):
    """Create an asymmetric (exponential ramp) initialization."""
    x = np.arange(n, dtype=float)
    a = np.exp(alpha * x / max(n - 1, 1))
    # Add some noise
    a = a * np.exp(rng.normal(0, 0.1, n))
    a = project_simplex(a)
    return a


def multi_peak_gradient_descent(a, max_iters=500, time_budget=60.0, 
                                  tau_init=0.1, n_peaks=5):
    """
    Gradient descent focusing on the top n_peaks of the convolution.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    step_size = 0.03
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = np.max(conv)
        
        # Select top n_peaks
        top_k = np.argsort(conv)[::-1][:n_peaks]
        # Use softmax weights on the top peaks
        top_vals = conv[top_k]
        tau = tau_init * cmax
        w_top = np.exp((top_vals - top_vals[0]) / max(tau, 1e-10))
        w_top = w_top / w_top.sum()
        
        # Build full weight vector
        weights = np.zeros(len(conv))
        for i, k in enumerate(top_k):
            weights[k] = w_top[i]
        
        grad = compute_gradient_fft(a, weights)
        grad_centered = grad - np.mean(grad)
        
        # Adaptive step
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-10:
            step = 0
        else:
            step = step_size
        
        a_try = a - step * grad_centered / grad_norm
        a_try = project_simplex(a_try)
       obj_try = compute_obj_normalized(a_try)
        
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
        else:
            step_size *= 0.9
    
    return best_a, best_obj
