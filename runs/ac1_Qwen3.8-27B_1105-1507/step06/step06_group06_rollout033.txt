import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg(a, target_sum=None):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    if target_sum is not None:
        return a * (target_sum / s)
    return a / s

"""
Optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) FFT-based subgradient with Nesterov momentum, (2) LP alternating, 
(3) spectral shaping, (4) multi-scale exploration, (5) adaptive restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import sys

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg(a, target_sum=1.0):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * (target_sum / s)

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0/new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def build_toeplitz_conv(a):
    """Build matrix T such that (T @ g) = conv(a, g) (length 2n-1)."""
    n = len(a)
    T = np.zeros((2*n-1, n))
    for j in range(n):
        end = min(j + n, 2*n - 1)
        T[j:end, j] = a[:end-j]
    return T

def solve_lp_direction(a, M, time_limit=10):
    """Solve: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0. Returns normalized g."""
    n = len(a)
    if n > 2500:
        return None
    T = build_toeplitz_conv(a)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=T, b_ub=np.full(2*n-1, M), 
                      bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def solve_lp_minpeak(a, time_limit=10):
    """Solve: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    if n > 2500:
        return None, None
    T = build_toeplitz_conv(a)
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n+1)
    c[n] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)]*n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = g.sum()
            if s > 1e-10:
                return g/s, float(res.x[n])
    except Exception:
        pass
    return None, None

def fft_subgradient(a, target_sum=1.0, max_iters=3000, time_budget=60.0, 
                    tau_frac=0.15, mom_init=0.9, mom_decay=0.002, step_init=0.5):
    """
    FFT-based subgradient with Nesterov momentum.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=target_sum.
    """
    n = len(a)
    a = project_nonneg(a.copy(), target_sum)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    step = step_init
    mom = mom_init
    
    # FFT size for gradient computation
    N_fft = 1
    while N_fft < 3 * n - 2:
        N_fft <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        if cmax < 1e-15:
            break
        
        # Soft-max weights
        tau = max(cmax * tau_frac, 1e-10)
        shifted = (conv - cmax) / tau
        weights = np.exp(np.clip(shifted, -40, 40))
        wsum = weights.sum()
        if wsum < 1e-300:
            idx = np.argmax(conv)
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= wsum
        
        # Compute gradient via FFT
        # grad[i] = sum_k w[k] * (a[k-i] + a[i]) * 1_{0<=k-i<n}
        # Part 1: cross-correlation of w and a
        a_rev = a[::-1]
        corr = irfft(rfft(a_rev, N_fft) * rfft(weights, N_fft), N_fft)
        grad1 = corr[n-1 : 2*n-1]
        
        # Part 2: a[i] * sum_{k=i}^{i+n-1} w[k]
        w_cumsum = np.concatenate([[0.0], np.cumsum(weights)])
        i_range = np.arange(n)
        k_end = np.minimum(i_range + n, 2*n - 1)
        part2_coeff = w_cumsum[k_end] - w_cumsum[i_range]
        
        grad = grad1 + a * part2_coeff
        
        # Nesterov momentum step
        v = mom * v - step * grad
        a_new = a - v
        a_new = project_nonneg(a_new, target_sum)
        a = a_new
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive parameters
        step *= (1 - step_init * 0.001)  # slow decay
        mom = max(0.5, mom - mom_decay)
        
        # Periodic momentum reset
        if it > 0 and it % 200 == 0:
            v = v * 0.5  # dampen velocity
    
    return best_a, best_obj

def spectral_shaping(a, target_sum=1.0, max_iters=200, time_budget=30.0):
    """
    Spectral shaping: modify the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a)
    a = project_nonneg(a.copy(), target_sum)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2*n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        F = np.fft.fft(a, N)
        power = np.abs(F)**2
        total_power = power.sum()
        
        # Compute current autocorrelation
        acorr = np.fft.ifft(power).real[:2*n-1]
        ac_max = np.max(acorr)
        
        # Identify peaks
        target = np.median(acorr[acorr > 0]) * 1.2
        peak_mask = acorr > target
        
        # In frequency domain, reducing a time-domain peak at position k
        # corresponds to reducing the contribution of e^{2πikf/N} to the power spectrum
        # We approximate this by reducing power at frequencies that correlate with the peak
        
        # Simpler approach: apply a "flattening" operation to the power spectrum
        # that encourages a flatter autocorrelation
        
        # Method: iteratively reduce the "peakedness" of the power spectrum
        p_max = np.max(power)
        if p_max < 1e-15:
            break
        
        # Apply power-law compression to the power spectrum
        alpha = 0.85 + 0.1 * (it / max_iters)  # gradually approach 1 (no compression)
        p_norm = power / p_max
        p_compressed = np.maximum(p_norm, 1e-10) ** alpha
        p_compressed *= p_max  # restore scale
        
        # Reconstruct: keep phase, use new magnitude
        phase = np.angle(F)
        magnitude = np.sqrt(p_compressed)
        F_new = magnitude * np.exp(1j * phase)
        
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = project_nonneg(a_new, target_sum)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a, best_obj

def lp_alternating(a, time_budget=30.0, max_iters=30):
    """LP alternating method."""
    n = len(a)
    a = project_nonneg(a.copy(), 1.0)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp_direction(a, M, time_limit=max(2, min(10, remaining * 0.3)))
        if g is None:
            break
        
        # Line search
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = project_nonneg((1-t)*a + t*g, 1.0)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
        else:
            break
    
    return best_a, best_obj

def simulated_annealing(a, rng, time_budget=60.0, T_init=0.3, T_min=0.01):
    """Simulated annealing with diverse move types."""
    n = len(a)
    target_sum = 1.0
    a = project_nonneg(a.copy(), target_sum)
    cur_a = a.copy()
    cur_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = cur_obj
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        T = T_init * (T_min/T_init) ** (it / 5000)
        T = max(T, T_min)
        
        move_type = rng.integers(0, 4)
        a_trial = cur_a.copy()
        
        if move_type == 0:
            # Local window noise
            wsize = max(5, n // 20)
            wstart = int(rng.integers(0, n - wsize))
            scale = cur_a[wstart:wstart+wsize].mean() + 0.001
            a_trial[wstart:wstart+wsize] += rng.normal(0, 0.2 * scale, wsize)
        elif move_type == 1:
            # Window swap
            wsize = max(10, n // 15)
            w1 = int(rng.integers(0, n - wsize))
            w2 = int(rng.integers(0, n - wsize))
            if abs(w1 - w2) >= wsize:
                a_trial[w1:w1+wsize], a_trial[w2:w2+wsize] = \
                    a_trial[w2:w2+wsize].copy(), a_trial[w1:w1+wsize].copy()
        elif move_type == 2:
            # Random spike
            n_spikes = max(5, n // 50)
            spike_idx = rng.choice(n, size=n_spikes, replace=False)
            mass = rng.random() * 0.1
            a_trial[spike_idx] += mass / n_spikes
        else:
            # Global multiplicative noise
            a_trial = a_trial * np.exp(rng.normal(0, 0.05, n))
        
        a_trial = project_nonneg(a_trial, target_sum)
        trial_obj = compute_obj(a_trial)
        
        delta = trial_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-max(delta, -50) / T):
            cur_a = a_trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()
    
    return best_a, best_obj

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    x = np.linspace(0, np.pi, n)
    c = np.cos(x/2)**2 + 0.01
    inits.append(c / c.sum())
    
    # Sine (peak in middle)
    x = np.linspace(0, 1, n)
    s = np.sin(np.pi * x) + 0.05
    inits.append(s / s.sum())
    
    # Gaussian
    x = np.linspace(-2.5, 2.5, n)
    g = np.exp(-x**2 / 2)
    inits.append(g / g.sum())
    
    # Two bumps
    x = np.linspace(0, 1, n)
    b = np.exp(-((x-0.3)/0.1)**2) + np.exp(-((x-0.7)/0.1)**2) + 0.01
    inits.append(b / b.sum())
    
    # Random
    r = rng.random(n) + 0.01
    inits.append(r / r.sum())
    
    # Alternating (high-low pattern)
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits.append(alt / alt.sum())
    
    # Raised cosine
    x = np.linspace(0, 1, n)
    rc = 0.5 * (1 + np.cos(2*np.pi*x)) + 0.05
    inits.append(rc / rc.sum())
    
    if parent is not None:
        p_resampled = resample_sequence(parent, n)
        inits.append(p_resampled / p_resampled.sum())
        # Perturbed parent
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_nonneg(pert, 1.0))
    
    return inits

```python
"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous: (1) Efficient FFT-based subgradient with proper
gradient computation via cross-correlation, (2) Heavy LP alternating with
adaptive line search, (3) Spectral power-shaping, (4) Peak-chasing with
adaptive step, (5) Multi-scale exploration with resampling, (6) Adaptive
temperature cycling in subgradient.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg(a, target_sum=1.0):
    """Project onto {a >= 0, sum(a) = target_sum}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * (target_sum / s)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def build_toeplitz_conv(a):
    """Build matrix T where (T @ g)[k] = (conv(a,g))[k]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        T[j:end, j] = a[:end - j]
    return T


def solve_lp_direction(a, M, time_limit=10):
    """Solve: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0. Return normalized g or None."""
    n = len(a)
    if n > 2500:
        return None
    T = build_toeplitz_conv(a)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=T, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(g.sum())
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def compute_fft_gradient(a, weights, N_fft):
    """
    Compute gradient of sum_k weights[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = sum_k w[k]*(a[k-i]+a[i])*1_{0<=k-i<n}
    Using FFT cross-correlation for the first part.
    """
    n = len(a)
    # Part 1: sum_{k=i}^{i+n-1} w[k]*a[k-i] = conv(a_rev, w)[n-1 : 2n-1]
    a_rev = a[::-1]
    corr = irfft(rfft(a_rev, N_fft) * rfft(weights, N_fft), N_fft)
    grad1 = corr[n - 1:2 * n - 1]

    # Part 2: a[i] * sum_{k=i}^{min(i+n-1, 2n-2)} w[k]
    w_cumsum = np.concatenate([[0.0], np.cumsum(weights)])
    i_range = np.arange(n)
    k_end = np.minimum(i_range + n, 2 * n - 1)
    part2_coeff = w_cumsum[k_end] - w_cumsum[i_range]

    return grad1 + a * part2_coeff
