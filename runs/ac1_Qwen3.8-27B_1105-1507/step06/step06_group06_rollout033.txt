"""
Optimized step function search using LP alternating method with spectral shaping,
multi-peak subgradient descent, and adaptive multi-length exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build matrix C where (C @ g)[k] = conv(a, g)[k] for k=0..2n-2."""
    n = len(a)
    m = 2 * n - 1
    # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = np.arange(m)[:, None]
    cols = np.arange(n)[None, :]
    idx = rows - cols
    valid = (idx >= 0) & (idx < n)
    idx_clipped = np.clip(idx, 0, n - 1)
    C = np.where(valid, a[idx_clipped], 0.0)
    return C


def solve_lp_minpeak(a, time_limit=10):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None) on failure.
    """
    n = len(a)
    if n > 3000:
        return None, None
    
    C = build_conv_matrix(a)
    m = 2 * n - 1
    
    # Variables: g[0..n-1], M (total n+1 variables)
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: C @ g - M <= 0 for each k
    # A_ub @ [g, M] <= b_ub
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g[i] >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(a, M, time_limit=10):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g>=0.
    Returns g (sum=1 normalized) or None.
    """
    n = len(a)
    if n > 3000:
        return None
    
    C = build_conv_matrix(a)
    m = 2 * n - 1
    
    # Objective: max sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Inequality: C @ g <= M * ones
    A_ub = C
    b_ub = np.full(m, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def spectral_shape(a, alpha=0.7, n_iters=5):
    """
    Frequency-domain peak compression of the self-convolution.
    Repeatedly compress peaks in conv(a,a) and extract new a via complex sqrt.
    """
    n = len(a)
    a = project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for _ in range(n_iters):
        # Compute self-convolution
        fa = np.fft.fft(a, N)
        c = np.fft.ifft(fa * fa).real[:2 * n - 1]
        
        c_max = np.max(c)
        if c_max < 1e-15:
            break
        
        # Compress: raise the floor, lower the peaks
        c_mod = c.copy()
        c_pos = c[c > 0]
        if len(c_pos) > 0:
            floor = np.median(c_pos) * 0.5
            c_mod = np.maximum(c_mod, floor)
            # Power compression
            c_normalized = c_mod / c_mod.max()
            c_mod = c_normalized ** alpha * c_max
        
        # Extract new a via complex square root
        cf = np.fft.fft(c_mod, N)
        # Ensure cf is "close to a perfect square" by taking principal sqrt
        fa_new = np.sqrt(cf + 0j)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            # Try with current a (didn't improve)
            break
    
    return best_a, best_obj


def multi_peak_subgradient(a, n_top=5, n_steps=50, step_init=0.05):
    """
    Subgradient descent using weighted combination of top peaks.
    """
    n = len(a)
    a = project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    step = step_init
    
    for _ in range(n_steps):
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        
        # Find top peaks
        top_idx = np.argsort(conv)[-n_top:]
        top_val = conv[top_idx]
        
        # Weights: exponential based on height
        w = np.exp((top_val - c_max) / 0.01)
        w /= np.sum(w)
        
        # Gradient: sum of weighted a[k*-i] for each peak
        grad = np.zeros(n)
        for w_k, k in zip(w, top_idx):
            k = int(k)
            # grad[i] += w_k * 2 * a[k-i] for valid k-i
            i_max = min(n, k + 1)
            i_min = max(0, k - n + 1)
            if i_max > i_min:
                idx = np.arange(i_min, i_max)
                a_idx = k - idx
                grad[idx] += 2.0 * w_k * a[a_idx]
        
        # Center to preserve sum
        grad_c = grad - np.mean(grad)
        
        # Try step
        a_try = project(a - step * grad_c)
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
        else:
            step *= 0.7
        
        step = max(step, 0.001)
    
    return best_a, best_obj


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project(a_new)

"""
Optimized step function search using intensive LP alternating method with
spectral shaping, multi-peak subgradient descent, and adaptive multi-length
exploration. Focuses on maximizing LP solver throughput and using diverse
initializations to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build matrix C where (C @ g)[k] = conv(a, g)[k] for k=0..2n-2."""
    n = len(a)
    m = 2 * n - 1
    rows = np.arange(m)[:, None]
    cols = np.arange(n)[None, :]
    idx = rows - cols
    valid = (idx >= 0) & (idx < n)
    idx_clipped = np.clip(idx, 0, n - 1)
    C = np.where(valid, a[idx_clipped], 0.0)
    return C


def solve_lp_minpeak(a, time_limit=8):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None) on failure.
    """
    n = len(a)
    if n > 2500:
        return None, None

    C = build_conv_matrix(a)
    m = 2 * n - 1

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(a, M, time_limit=8):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g>=0.
    Returns g (sum=1 normalized) or None.
    """
    n = len(a)
    if n > 2500:
        return None

    C = build_conv_matrix(a)
    m = 2 * n - 1

    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def spectral_shape(a, alpha=0.7, n_iters=5):
    """
    Frequency-domain peak compression of the self-convolution.
    """
    n = len(a)
    a = project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for _ in range(n_iters):
        fa = np.fft.fft(a, N)
        c = np.fft.ifft(fa * fa).real[:2 * n - 1]

        c_max = np.max(c)
        if c_max < 1e-15:
            break

        c_mod = c.copy()
        c_pos = c[c > 0]
        if len(c_pos) > 0:
            floor = np.median(c_pos) * 0.5
            c_mod = np.maximum(c_mod, floor)
            c_normalized = c_mod / c_mod.max()
            c_mod = c_normalized ** alpha * c_max

        cf = np.fft.fft(c_mod, N)
        fa_new = np.sqrt(cf + 0j)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            break

    return best_a, best_obj


def multi_peak_subgradient(a, n_top=5, n_steps=50, step_init=0.05):
    """Subgradient descent using weighted combination of top peaks."""
    n = len(a)
    a = project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    step = step_init

    for _ in range(n_steps):
        conv = compute_conv(a)
        c_max = float(np.max(conv))

        top_idx = np.argsort(conv)[-n_top:]
        top_val = conv[top_idx]

        w = np.exp((top_val - c_max) / 0.01)
        w /= np.sum(w)

        grad = np.zeros(n)
        for w_k, k in zip(w, top_idx):
            k = int(k)
            i_max = min(n, k + 1)
            i_min = max(0, k - n + 1)
            if i_max > i_min:
                idx = np.arange(i_min, i_max)
                a_idx = k - idx
                grad[idx] += 2.0 * w_k * a[a_idx]

        grad_c = grad - np.mean(grad)

        a_try = project(a - step * grad_c)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
        else:
            step *= 0.7

        step = max(step, 0.001)

    return best_a, best_obj


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project(a_new)

    seen = {n_parent}
    for obj_c, a_c, n_c in candidates:
        if n_c not in seen and len(top_cands) < 4:
            top_cands.append((obj_c, a_c.copy(), n_c))
            seen.add(n_c)

    for ci, (obj_init, a_init, n_init) in enumerate(top_cands):
        rem = phase2_deadline - time.monotonic()
        if rem < 20:
            break
        alloc = rem / (len(top_cands) - ci)
        alloc = max(20, min(alloc, 400))

        print(f"  Cand {ci}: n={n_init}, obj={obj_init:.8f}, budget={alloc:.0f}s")
        a_work = a_init.copy()
        local_best = obj_init
        round_idx = 0

        while time.monotonic() < phase2_deadline:
            rem_inner = phase2_deadline - time.monotonic()
            if rem_inner < 10:
                break

            # LP burst
            burst_budget = min(25, rem_inner * 0.5)
            a_new, obj_new = lp_alternate_optimize(a_work, max_iters=40,
                                                   time_budget=burst_budget,
                                                   verbose=(ci == 0))
            if obj_new < local_best - 1e-10:
                a_work = a_new
                local_best = obj_new
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_work.copy()
                    print(f"    ** UPDATE r{round_idx}: obj={best_score:.8f} **")
            round_idx += 1

            # Spectral shaping every 3 rounds
            if round_idx % 3 == 0 and n_init <= 2000:
                rem_inner = phase2_deadline - time.monotonic()
                if rem_inner < 5:
                    break
                a_sh, obj_sh = spectral_shape(a_work, alpha=0.65, n_iters=3)
                if obj_sh < local_best:
                    a_work = a_sh
                    local_best = obj_sh
                    if obj_sh < best_score:
                        best_score = obj_sh
                        best_seq = a_work.copy()
                        print(f"    ** SPECTRAL r{round_idx}: obj={best_score:.8f} **")

            # Multi-peak subgradient every 5 rounds
            if round_idx % 5 == 0:
                rem_inner = phase2_deadline - time.monotonic()
                if rem_inner < 5:
                    break
                a_sg, obj_sg = multi_peak_subgradient(a_work, n_top=5,
                                                      n_steps=min(30, int(rem_inner)),
                                                      step_init=0.03)
                if obj_sg < local_best:
                    a_work = a_sg
                    local_best = obj_sg
                    if obj_sg < best_score:
                        best_score = obj_sg
                        best_seq = a_work.copy()
                        print(f"    ** SUBGRAD r{round_idx}: obj={best_score:.8f} **")

    # === Phase 3: Multi-length exploration (20% time) ===
    phase3_time = 0.20 * float(budget_s)
    phase3_deadline = start_time + phase1_time + phase2_time + phase3_time
    print(f"\n=== Phase 3: Multi-length (budget={phase3_time:.0f}s) ===")
    print(f"Best: {best_score:.8f}, n={len(best_seq)}")

    n_best = len(best_seq)
    n_fine = [n_best + d for d in [-80, -60, -40, -20, 20, 40, 60, 80, 120, -120]]
    n_fine = [n for n in n_fine if 50 <= n <= 2500]
    n_fine = list(dict.fromkeys(n_fine))

    per_n = max(8.0, phase3_time / max(1, len(n_fine)))

    for n_try in n_fine:
        if time.monotonic() > phase3_deadline:
            break
        a_res = resample_seq(best_seq, n_try)
        a_opt, obj_opt = lp_alternate_optimize(a_res, max_iters=25,
                                               time_budget=per_n * 0.6)
        if n_try <= 2000:
            a_sp, obj_sp = spectral_shape(a_opt, alpha=0.7, n_iters=3)
            if obj_sp < obj_opt:
                a_opt, obj_opt = a_sp, obj_sp
        a_opt2, obj_opt2 = lp_alternate_optimize(a_opt, max_iters=15,
                                                 time_budget=per_n * 0.3)
        if obj_opt2 < obj_opt:
            a_opt, obj_opt = a_opt2, obj_opt2

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: {obj_opt:.8f}")

    # === Phase 4: Final polish (remaining) ===
    remaining4 = deadline - time.monotonic()
    if remaining4 > 10:
        print(f"\n=== Phase 4: Final Polish ({remaining4:.0f}s) ===")
        print(f"Best: {best_score:.8f}")

        a_final = best_seq.copy()
        n_final = len(a_final)

        for pr in range(int(remaining4 / 8)):
            if time.monotonic() > deadline - 3:
                break
            rem = deadline - time.monotonic()
            if rem < 5:
                break

            a_p, obj_p = lp_alternate_optimize(a_final, max_iters=min(30, int(rem)),
                                               time_budget=min(rem * 0.4, 25))
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  Polish LP r{pr}: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem < 5:
                continue

            a_sg, obj_sg = multi_peak_subgradient(a_final, n_top=3,
                                                  n_steps=min(20, int(rem)),
                                                  step_init=0.02)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                a_final = a_sg.copy()
                print(f"  Polish SG r{pr}: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem < 8:
                continue

            if n_final <= 2000:
                a_sp, obj_sp = spectral_shape(a_final, alpha=0.6, n_iters=3)
                if obj_sp < best_score:
                    best_score = obj_sp
                    best_seq = a_sp.copy()
                    a_final = a_sp.copy()
                    print(f"  Polish SP r{pr}: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem > 8:
                a_pert = a_final * np.exp(rng.normal(0, 0.03, n_final))
                a_pert = project(a_pert)
                a_pert, obj_pert = lp_alternate_optimize(a_pert, max_iters=10,
                                                         time_budget=min(rem * 0.3, 8))
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_final = a_pert.copy()
                    print(f"  Polish PERT r{pr}: {best_score:.8f}")

    # Final output
    if best_seq is None:
        best_seq = np.ones(200)

    result_arr = np.clip(best_seq, 0, 1000)
    result = [float(x) for x in result_arr]

    val = evaluate_sequence(result)
    print(f"\nEvaluate: {val:.8f} (tracked best: {best_score:.8f})")

    if val >= np.inf:
        result = [1.0] * 100
        val = evaluate_sequence(result)
        print(f"Fallback: {val:.8f}")
    elif val > best_score + 0.01:
        # Mismatch - use evaluated value
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"=== FINAL: n={len(result)}, score={val:.8f}, time={elapsed:.1f}s ===")

    return result
