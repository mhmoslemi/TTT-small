import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (S * S)

def project_nonneg_sum(a, target_sum=1.0):
    """Project a onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones_like(a)
        s = len(a)
    return a * (target_sum / s)

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

Optimized subgradient method with log-sum-exp smoothing, Nesterov momentum,
and adaptive step size for minimizing 2n*max(conv(p,p)) over probability vectors p.
Includes length exploration and periodic perturbation for escaping local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (S * S)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def compute_smoothed_grad(p, c, tau):
    """
    Compute gradient of the log-sum-exp smoothed max.
    grad[i] = (2/tau) * sum_k softmax_k * p[k-i]
    """
    n = len(p)
    # Softmax weights
    c_shifted = (c - c.max()) / tau
    weights = np.exp(c_shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        # Fallback: uniform over argmax
        weights = np.zeros_like(c)
        max_idx = np.argmax(c)
        weights[max_idx] = 1.0
    else:
        weights /= w_sum
    
    # grad[i] = 2 * sum_k weights[k] * p[k-i] (for valid k-i)
    # This is the cross-correlation of p and weights
    # Use FFT for efficiency
    N_fft = 1
    while N_fft < n + len(weights) - 1:
        N_fft <<= 1
    
    # cross_corr[m] = sum_j p[j] * weights[j+m]
    # FFT(cross_corr) = conj(FFT(p)) * FFT(weights) ... 
    # Actually: (x ⋆ y)[m] = sum_n x[n] * y[n+m]
    # In frequency: X(ω) * conj(Y(ω)) gives ifft = (x ⋆ y)
    # Wait, let me be careful.
    # Cross-correlation: (x ⋆ y)[m] = sum_n x[n+m] * conj(y[n])
    # Or equivalently: (x ⋆ y)[m] = sum_n x[n] * conj(y[n-m])
    # 
    # I want: sum_j p[j] * w[j+i] for each i.
    # Let me define: r[i] = sum_j p[j] * w[j+i]
    # This is the "offset dot product" or "cross-correlation" of p and w.
    # 
    # Using the property: sum_j p[j] * w[j+i] = (p_rev ⋆ w_rev)[...] 
    # 
    # Simpler approach: 
    # r[i] = sum_j p[j] * w[j+i]
    # Let u = p reversed, v = w reversed.
    # Then r[i] = conv(u, v)[n-1-i + len(w)-1] ... this is getting confusing.
    # 
    # Let me just use the direct FFT formula:
    # r = ifft(fft(w, N) * conj(fft(p, N))) -- this gives circular cross-correlation
    # For linear, we need N >= n + len(w) - 1
    
    fp = fft(p, N_fft)
    fw = fft(weights, N_fft)
    # cross-correlation: ifft(conj(fp) * fw) gives sum_j p[j] * w[j+i]? 
    # Let me verify: 
    # ifft(conj(fft(p)) * fft(w))[m] = sum_n p[n] * w[n+m] (circular)
    # Yes! That's exactly what we want.
    corr = ifft(np.conj(fp) * fw).real
    # corr[i] = sum_j p[j] * w[j+i] (for the linear cross-correlation, valid for i in [0, len(w)-1])
    
    grad = 2.0 * corr[:n]
    return grad, weights


def refined_subgradient(p, n_iters=5000, time_budget=100.0, tau_init=None, tau_min=None, verbose=False):
    """
    Refined subgradient optimization with log-sum-exp smoothing and momentum.
    Minimizes max(conv(p,p)) subject to p >= 0, sum(p) = 1.
    """
    n = len(p)
    p = project_nonneg_sum1(p.copy())
    best_p = p.copy()
    best_obj = compute_obj(p)
    
    c = compute_conv(p)
    c_max = c.max()
    
    if tau_init is None:
        tau_init = c_max / 10.0
    if tau_min is None:
        tau_min = c_max / 100.0
    
    tau = tau_init
    alpha = 0.5
    v = np.zeros(n)  # momentum
    beta = 0.7  # momentum coefficient (will increase)
    
    start = time.monotonic()
    no_improve = 0
    max_no_improve = 200
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient
        grad, weights = compute_smoothed_grad(p, c, tau)
        
        # Center the gradient (to preserve sum approximately)
        grad_centered = grad - grad.mean()
        
        # Momentum update
        beta = min(0.95, 0.7 + 0.25 * it / max(1, n_iters))
        v = beta * v - alpha * grad_centered
        
        # Step
        p_new = p + v
        p_new = project_nonneg_sum1(p_new)
        
        # Compute new objective
        c_new = compute_conv(p_new)
        c_max_new = float(c_new.max())
        obj_new = 2.0 * n * c_max_new  # since sum(p)=1
        
        # Weighted peak (for monitoring)
        weighted_peak = float(np.sum(weights * c_new))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_p = p_new.copy()
            no_improve = 0
            # Slightly increase alpha if improving consistently
            if no_improve == 0 and it % 50 == 0:
                alpha = min(alpha * 1.02, 1.0)
        else:
            no_improve += 1
            if no_improve == 50:
                alpha *= 0.8
            elif no_improve == 100:
                alpha *= 0.7
            elif no_improve > max_no_improve:
                # Restart momentum
                v *= 0.1
                alpha = max(0.01, alpha * 0.5)
        
        # Update p and c
        p = p_new
        c = c_new
        
        # Adapt temperature
        tau = max(tau_min, tau * 0.9995)
        
        if verbose and it % 100 == 0:
            print(f"    it={it}, obj={obj_new:.8f}, best={best_obj:.8f}, tau={tau:.6f}, alpha={alpha:.4f}")
    
    return best_p, best_obj


def chase_peak_refined(p, n_steps=100, step_size=0.05):
    """
    Chase the single peak with refined step sizes.
    """
    n = len(p)
    p = project_nonneg_sum1(p.copy())
    best_p = p.copy()
    best_obj = compute_obj(p)
    
    for _ in range(n_steps):
        c = compute_conv(p)
        k_star = int(np.argmax(c))
        
        # Subgradient at the peak
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * p[j]
        
        grad_centered = grad - grad.mean()
        
        # Try multiple step sizes
        best_step_obj = np.inf
        best_step_p = p
        for ss in [step_size, step_size * 0.5, step_size * 0.25, step_size * 1.5, step_size * 2.0]:
            p_try = p - ss * grad_centered
            p_try = project_nonneg_sum1(p_try)
            obj_try = compute_obj(p_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_p = p_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_p = p_try.copy()
        
        if best_step_obj >= best_obj + 1e-12:
            # No improvement, try smaller steps
            for ss in [step_size * 0.1, step_size * 0.05]:
                p_try = p - ss * grad_centered
                p_try = project_nonneg_sum1(p_try)
                obj_try = compute_obj(p_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_p = p_try.copy()
                    break
            else:
                break
        
        p = project_nonneg_sum1(p + (best_step_p - p))
        # Actually just use the best step
        p = best_step_p.copy()
        if p is best_p:
            p = best_p.copy()
    
    return best_p, best_obj


def multi_peak_subgradient(p, n_steps=200, time_budget=50.0, n_peaks=5, tau=0.01):
    """
    Subgradient using top-k peaks with softmax weighting.
    """
    n = len(p)
    p = project_nonneg_sum1(p.copy())
    best_p = p.copy()
    best_obj = compute_obj(p)
    start = time.monotonic()
    alpha = 0.3
    v = np.zeros(n)
    beta = 0.8
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        c = compute_conv(p)
        c_max = c.max()
        
        # Top-k peaks with softmax
        if n_peaks == 1:
            idx = int(np.argmax(c))
            weights = np.zeros(len(c))
            weights[idx] = 1.0
        else:
            shifted = (c - c_max) / tau
            weights = np.exp(shifted)
            weights /= weights.sum()
        
        # Gradient
        grad = np.zeros(n)
        for k_idx in range(len(c)):
            if weights[k_idx] < 1e-10:
                continue
            for i in range(max(0, k_idx - n + 1), min(n, k_idx + 1)):
                j = k_idx - i
                if 0 <= j < n:
                    grad[i] += 2.0 * weights[k_idx] * p[j]
        
        grad_centered = grad - grad.mean()
        
        # Momentum step
        v = beta * v - alpha * grad_centered
        p_new = project_nonneg_sum1(p + v)
        
        obj_new = compute_obj(p_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_p = p_new.copy()
            p = p_new
        
        alpha *= 0.999
        beta = min(0.95, beta + 0.0002)
    
    return best_p, best_obj


def perturbation_restart(p, rng, strength=0.05):
    """Apply a random perturbation to escape local minima."""
    n = len(p)
    # Multiplicative perturbation
    pert = p * np.exp(rng.normal(0, strength, n))
    return project_nonneg_sum1(pert)


def window_perturbation(p, rng, wsize=None, strength=0.3):
    """Perturb a random window."""
    n = len(p)
    if wsize is None:
        wsize = max(5, n // 20)
    start = rng.integers(0, n - wsize)
    p_new = p.copy()
    noise = rng.normal(0, strength * p[start:start+wsize].max(), wsize)
    p_new[start:start+wsize] += noise
    return project_nonneg_sum1(p_new)

def compute_conv(a):

    n = len(a)

        return np.array([0.0])
    size = 1

def compute_obj(a):
