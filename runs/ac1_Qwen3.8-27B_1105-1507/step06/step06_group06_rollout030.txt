"""
Minimize 2*n*max(autoconv(a))/(sum(a))^2 for non-negative sequence a.
Strategy: (1) adaptive p-norm gradient descent with FFT-based gradients,
(2) iterative spectral flattening, (3) peak-targeting moves,
(4) multi-length and multi-init exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def grad_pnorm(a, p):
    """Gradient of sum_k (a*a)[k]^p w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    # weights[k] = conv[k]^(p-1), clipped
    conv_pos = np.maximum(conv, 0)
    max_cv = conv_pos.max()
    if max_cv < 1e-15:
        return np.zeros(n), conv
    # Use log to avoid overflow for large p
    # w[k] = exp((p-1) * log(conv[k]))
    log_w = (p - 1) * np.log(np.maximum(conv_pos, 1e-300) / max_cv)
    log_w = np.clip(log_w, -700, 700)
    w = np.exp(log_w)
    scale = max_cv ** (p - 1)
    
    # grad[i] = 2 * scale * sum_{j=0}^{n-1} a[j] * w[i+j]
    # = 2 * scale * (a * w_rev)[2n-2-i]
    w_rev = w[::-1]
    N = 1
    while N < n + 2 * n - 1:
        N <<= 1
    fa = np.fft.fft(a, N)
    fw = np.fft.fft(w_rev, N)
    c = np.fft.ifft(fa * fw).real
    # c[k] = sum_j a[j] * w_rev[k-j]
    # We want sum_j a[j] * w[i+j] = c[2n-2-i]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * scale * c[idx]
    return grad, conv


def pnorm_optimize(a, p_start=10, p_end=500, max_iters=2000, lr=0.05, momentum=0.9, time_budget=30):
    """Optimize sum_k conv[k]^p with projected gradient + momentum."""
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)
    start = time.monotonic()
    it = 0
    
    while it < max_iters and (time.monotonic() - start) < time_budget:
        # Adaptive p
        progress = it / max_iters
        p = p_start + (p_end - p_start) * progress ** 0.5
        
        grad, conv = grad_pnorm(a, p)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Momentum step
        v = momentum * v - lr * grad
        a_new = project(a + v)
        
        # Check improvement
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        it += 1
    
    return best_a, best_obj


def spectral_flatten(a, n_iters=100, time_budget=20):
    """Iteratively flatten the power spectrum."""
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(n_iters):
        if (time.monotonic() - start) > time_budget:
            break
        
        fa = np.fft.fft(a, N)
        ps = np.abs(fa)**2
        
        # Flatten: compress dynamic range
        ps_max = ps.max()
        ps_min = np.median(ps[ps > 0]) if np.any(ps > 0) else ps_max * 0.1
        if ps_max < 1e-15:
            break
        
        # Adaptive compression: blend between current and flat
        alpha = 0.1 + 0.1 * (it / n_iters)
        ps_target = np.full_like(ps, np.median(ps[ps > 0]) if np.any(ps > 0) else ps_max / 2)
        ps_new = (1 - alpha) * ps + alpha * ps_target
        
        # Also apply power compression
        ps_norm = ps_new / ps_new.max()
        power = 0.85 + 0.1 * (it / n_iters)
        ps_new = ps_norm ** power * ps_new.max()
        
        # Reconstruct with original phase
        phase = np.angle(fa)
        fa_new = np.sqrt(ps_new + 0j) * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj


def peak_targeting(a, n_steps=100, time_budget=20):
    """Identify peaks in auto-convolution and apply targeted reduction moves."""
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for step in range(n_steps):
        if (time.monotonic() - start) > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = conv.max()
        
        # Find top peaks
        top_k = min(5, len(conv))
        peak_indices = np.argsort(conv)[-top_k:][::-1]
        
        # For each peak, compute the gradient contribution and reduce
        grad = np.zeros(n)
        for k in peak_indices:
            if conv[k] < cmax * 0.5:
                continue
            weight = (conv[k] / cmax) ** 4  # emphasize top peaks
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += weight * a[j]
        
        # Also add the p=2 gradient (quadratic term) for stability
        # d/d a[i] of sum_k conv[k]^2 = 4 * sum_k conv[k]*a[k-i]
        # This is 4 * (conv * a_rev)[...]
        conv_pos = np.maximum(conv, 0)
        a_rev = a[::-1]
        N = 1
        while N < 2 * n - 1 + n - 1:
            N <<= 1
        fc = np.fft.fft(conv_pos, N)
        fr = np.fft.fft(a_rev, N)
        c = np.fft.ifft(fc * fr).real
        # (conv * a_rev)[k] = sum_j conv[j] a_rev[k-j] = sum_j conv[j] a[n-1-k+j]
        # We want sum_k conv[k] a[k-i] = (conv * a_rev)[n-1-i + ...]
        # Let me just use the direct formula:
        # d/d a[i] sum_k conv[k]^2 = 4 * sum_{k: 0<=k-i<n} conv[k] * a[k-i]
        # = 4 * sum_{j=0}^{n-1} conv[i+j] * a[j]
        # Same as before with w = conv^1 = conv
        w = conv_pos
        w_rev = w[::-1]
        N2 = 1
        while N2 < n + 2*n - 1:
            N2 <<= 1
        fa2 = np.fft.fft(a, N2)
        fw2 = np.fft.fft(w_rev, N2)
        c2 = np.fft.ifft(fa2 * fw2).real
        idx = 2*n - 2 - np.arange(n)
        quad_grad = 4.0 * c2[idx]
        
        total_grad = grad + 0.1 * quad_grad
        total_grad -= np.mean(total_grad)  # center
        
        gn = np.linalg.norm(total_grad)
        if gn < 1e-15:
            break
        total_grad /= gn
        
        # Try a few step sizes
        improved = False
        for lr in [0.02, 0.01, 0.005, 0.05]:
            a_try = project(a - lr * total_grad)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_obj


def self_conv_square_root(a, n_iters=100, time_budget=20):
    """
    Iterative method: modify the auto-convolution to be flatter, then extract a via sqrt(FFT).
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1

        if (time.monotonic() - start) > time_budget:
            break
        
        # Current auto-conv
        fa = np.fft.fft(a, N)
        conv = np.fft.ifft(fa * fa).real[:2*n-1]

        cmax = conv.max()
        if cmax < 1e-15:
            break
        
        # Target: flatten the convolution
        # Blend toward uniform
        target = np.full(2*n-1, conv.sum() / (2*n-1))

        blend = 0.05 + 0.1 * (it / n_iters)
        conv_new = (1 - blend) * conv + blend * target
        
        # Also apply soft thresholding to reduce peaks

        conv_new = np.maximum(conv_new, 0)
        
        # Extract a via complex square root of FFT

        fa_new = np.sqrt(cf + 0j)
        a_new = np.fft.ifft(fa_new).real[:n]
        
        a_new = project(a_new)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0/new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def generate_inits(n, rng, parent=None):
    inits = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sine
    s = np.sin(np.pi * x) + 0.1
    inits.append(s / s.sum())
    
    # Gaussian
    g = np.exp(-2 * (x - 0.5)**2 / 0.1)
    inits.append(g / g.sum())
    
    # Flat top with tapered edges
    t = np.ones(n)
    edge = max(5, n // 20)
    t[:edge] = np.linspace(0, 1, edge)
    t[-edge:] = np.linspace(1, 0, edge)
    inits.append(t / t.sum())
    
    # Two peaks
    p1 = np.exp(-((x - 0.3)**2) / 0.01)
    p2 = np.exp(-((x - 0.7)**2) / 0.01)
    inits.append((p1 + p2) / (p1 + p2).sum())
    
    # Random
    r = rng.random(n) + 0.01
    inits.append(r / r.sum())
    
    # Power law
    pl = x ** 2 + 0.1
    inits.append(pl / pl.sum())
    
    # Cosine window
    cw = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(cw / cw.sum())
    
    if parent is not None:
        pr = resample(parent, n)
        inits.append(project(pr))
        # Perturbed
        pert = pr * np.exp(rng.normal(0, 0.1, n))
        inits.append(project(pert))
    
    return inits

    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project(a):
    """Project to non-negative simplex (sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def grad_pnorm(a, p, conv=None):
    """
    Gradient of sum_k (a*a)[k]^p w.r.t. a.
    Returns (gradient, conv).
    grad[i] = 2 * sum_{j=0}^{n-1} (conv[i+j])^{p-1} * a[j]  (for i+j < 2n-1)
    Computed via FFT in O(n log n).
    """
    n = len(a)
    if conv is None:
        conv = compute_conv(a)
    len_c = len(conv)  # 2n-1

    conv_pos = np.maximum(conv, 0.0)
    max_cv = conv_pos.max()
    if max_cv < 1e-15:
        return np.zeros(n), conv

    # w[k] = conv[k]^(p-1), computed in log domain for stability
    if p <= 1:
        w = np.ones(len_c)
        scale = 1.0
    else:
        # Normalize to [0,1] first
        w_norm = conv_pos / max_cv
        log_w = (p - 1) * np.log(np.maximum(w_norm, 1e-300))
        log_w = np.clip(log_w, -700.0, 700.0)
        w = np.exp(log_w)
        scale = max_cv ** (p - 1)

    # grad[i] = 2*scale * sum_{j=0}^{n-1} a[j] * w[i+j]
    # = 2*scale * (a * w_rev)[2n-2-i]  where w_rev = w[::-1]
    w_rev = w[::-1]
    N = 1
    while N < n + len_c - 1:
        N <<= 1
    fa = np.fft.fft(a, N)
    fw = np.fft.fft(w_rev, N)
    c = np.fft.ifft(fa * fw).real
    # c[k] = sum_j a[j] * w_rev[k-j] = sum_j a[j] * w[len_c-1-(k-j)]
    # We want sum_j a[j] * w[i+j]: set len_c-1-(k-j) = i+j => k = len_c-1-i
    idx = len_c - 1 - np.arange(n)
    grad = 2.0 * scale * c[idx]
    return grad, conv


def pnorm_descent(a, p_start=4.0, p_end=2000.0, max_iters=3000,
                  lr_init=0.1, momentum=0.92, time_budget=30.0,
                  cooldown=50):
    """
    Projected gradient descent on sum_k conv[k]^p with adaptive p and momentum.
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)
    start = time.monotonic()
    it = 0
    lr = lr_init

    while it < max_iters and (time.monotonic() - start) < time_budget:
        progress = min(1.0, it / max_iters)
        # Adaptive p: start smooth, get sharper
        p = p_start + (p_end - p_start) * progress ** 0.4
        # Learning rate decay
        lr = lr_init * (0.999 ** it) * (1.0 + 0.5 * (1.0 - progress))

        grad, conv = grad_pnorm(a, p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Nesterov momentum
        v = momentum * v - lr * (grad / gnorm)
        a_new = project(a + v)

        # Evaluate (every iteration for small n, less often for large)
        if n <= 1500 or it % 3 == 0:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        else:
            # Quick proxy: just track the p-norm value
            pass

        a = a_new
        it += 1

        # Cooldown: periodically re-evaluate
        if it % cooldown == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    # Final check
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def spectral_flatten(a, n_iters=100, blend_rate=0.03, time_budget=15.0):
    """
    Iterative spectral flattening: modify |FFT(a)|^2 to be more uniform,
    then extract new a via phase-preserving reconstruction.
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(n_iters):
        if (time.monotonic() - start) > time_budget:
            break

        fa = np.fft.fft(a, N)
        ps = np.abs(fa) ** 2 + 1e-30

        # Compress dynamic range
        ps_max = ps.max()
        if ps_max < 1e-15:
            break
        ps_norm = ps / ps_max

        # Target: geometric mean (log-uniform)
        log_ps = np.log(ps_norm + 1e-30)
        log_target = np.mean(log_ps)
        # Blend toward uniform in log domain
        alpha = blend_rate * (1.0 + 0.5 * it / n_iters)
        log_ps_new = (1.0 - alpha) * log_ps + alpha * log_target
        ps_new = np.exp(np.clip(log_ps_new, -30, 30))

        # Also apply power compression for extra flattening
        ps_new = ps_new / ps_new.max()
        power = 0.8 + 0.15 * (it / n_iters)
        ps_new = ps_new ** power

        # Reconstruct with original phase
        phase = np.angle(fa)
        fa_new = np.sqrt(ps_new / ps_new.mean() * ps_max) * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def sqrt_conv_extract(a, n_iters=80, blend=0.08, time_budget=15.0):
    """
    Modify the auto-convolution to be flatter, then extract a via sqrt(FFT(conv)).
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(n_iters):
        if (time.monotonic() - start) > time_budget:
            break

        fa = np.fft.fft(a, N)
        conv = np.fft.ifft(fa * fa).real[:2 * n - 1]
        conv = np.maximum(conv, 0.0)
        cmax = conv.max()
        if cmax < 1e-15:
            break

        # Flatten: blend toward mean, clip peaks
        cmean = conv.mean()
        conv_new = (1.0 - blend) * conv + blend * cmean

        # Soft clip peaks
        threshold = np.percentile(conv_new, 85)
        mask = conv_new > threshold
        conv_new[mask] = threshold + 0.4 * (conv_new[mask] - threshold)
        conv_new = np.maximum(conv_new, 0.0)

        # Extract a via complex square root
        cf = np.fft.fft(conv_new, N)
        # Ensure valid PSD by taking real part of log
        fa_new = np.sqrt(cf + 0j)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def peak_multiplicative(a, n_iters=200, time_budget=20.0):
    """
    Multiplicative weight updates: reduce a[i] proportional to its contribution
    to the peaks of the auto-convolution.
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_iters):
        if (time.monotonic() - start) > time_budget:
            break

        conv = compute_conv(a)
        cmax = conv.max()
        if cmax < 1e-15:
            break
        cmean = conv.mean()

        # Target: values above this are "peaks"
        target = cmean + 0.5 * (cmax - cmean)

        # Compute contribution of each a[i] to peaks
        # a[i] contributes a[i]*a[j] to conv[i+j] for j=0..n-1
        # contribution[i] = sum_j a[j] * max(0, conv[i+j] - target)
        # This is the same cross-correlation as in grad_pnorm but with a different weight
        w = np.maximum(conv - target, 0.0)
        if w.max() < 1e-15:
            break

        w_rev = w[::-1]
        N = 1
        while N < n + len(conv) - 1:
            N <<= 1
        fa = np.fft.fft(a, N)
        fw = np.fft.fft(w_rev, N)
        c = np.fft.ifft(fa * fw).real
        idx = len(conv) - 1 - np.arange(n)
        contribution = c[idx]

        # Multiplicative update
        rate = 0.05 * (1.0 - 0.5 * it / n_iters)
        contrib_norm = contribution / (contribution.max() + 1e-15)
        a_new = a * np.exp(-rate * contrib_norm)
        a_new = project(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Sine
    s = np.sin(np.pi * x) + 0.05
    inits.append(s / s.sum())

    # Gaussian center
    g = np.exp(-4.0 * (x - 0.5) ** 2 / 0.05)
    inits.append(g / g.sum())

    # Flat top with tapered edges
    t = np.ones(n)
    edge = max(5, n // 15)
    t[:edge] = 0.5 * (1 + np.sin(np.pi * np.arange(edge) / edge - np.pi / 2))
    t[-edge:] = 0.5 * (1 + np.sin(np.pi * np.arange(edge) / edge - np.pi / 2))[::-1]
    inits.append(t / t.sum())

    # Two peaks
    p1 = np.exp(-((x - 0.3) ** 2) / 0.005)
    p2 = np.exp(-((x - 0.7) ** 2) / 0.005)
    inits.append((p1 + p2) / (p1 + p2).sum())

    # Random smooth
    r = rng.random(n) + 0.01
    # Smooth with moving average
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    r = np.convolve(r, kernel, mode='same')
    inits.append(r / r.sum())

    # Cosine window
    cw = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    inits.append(cw / cw.sum())

    # Majority flat with small perturbation
    base = np.ones(n) / n
    pert = rng.normal(0, 0.1, n)
    base_pert = np.maximum(base + pert * base.mean(), 0)
    inits.append(base_pert / base_pert.sum())

    if parent is not None:
        pr = resample(parent, n)
        inits.append(project(pr))
        # Scaled perturbation
        for sigma in [0.05, 0.1, 0.2]:
            pert = pr * np.exp(rng.normal(0, sigma, n))
            inits.append(project(pert))

    return inits


def local_optimize(a, time_budget=30.0, n=None):
    """
    Run a cycle of local optimization strategies.
    Returns best sequence and objective found.
    """
    if n is None:
        n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while (time.monotonic() - start) < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        # Alternate strategies
        chunk = min(remaining * 0.3, 15.0)

        # 1. p-norm descent
        a1, obj1 = pnorm_descent(best_a, p_start=8.0, p_end=500.0,
                                 max_iters=min(1000, int(chunk * 50)),
                                 lr_init=0.08, momentum=0.93,
                                 time_budget=chunk)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()

        if (time.monotonic() - start) >= time_budget:
            break

        # 2. Spectral flattening
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5.0:
            a2, obj2 = spectral_flatten(best_a, n_iters=50,
                                        blend_rate=0.04, timebudget_placeholder=None) if False else spectral_flatten(best_a, n_iters=50, blend_rate=0.04, time_budget=min(remaining * 0.3, 10.0))
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a2.copy()

        if (time.monotonic() - start) >= time_budget:
            break

        # 3. Peak multiplicative
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5.0:
            a3, obj3 = peak_multiplicative(best_a, n_iters=100,
                                           time_budget=min(remaining * 0.3, 10.0))
            if obj3 < best_obj:
                best_obj = obj3
                best_a = a3.copy()

        if (time.monotonic() - start) >= time_budget:
            break

        # 4. Sqrt-conv extract
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5.0:
            a4, obj4 = sqrt_conv_extract(best_a, n_iters=50,
                                         blend=0.06, time_budget=min(remaining * 0.3, 10.0))
            if obj4 < best_obj:
                best_obj = obj4
                best_a = a4.copy()

    return best_a, best_obj


def fine_pnorm_optimize(a, time_budget=30.0):
    """

    """
    n = len(a)
    return pnorm_descent(a, p_start=100.0, p_end=5000.0,
                         max_iters=int(time_budget * 100),

        rem3 = phase3_budget - (time.monotonic() - phase3_start)
        if rem3 > 5.0:
            a_pk, obj_pk = peak_multiplicative(a_polish, n_iters=200,
                                               time_budget=min(rem3 * 0.4, 20.0))
            if obj_pk < best_score:
                best_score = obj_pk
                best_seq = a_pk.copy()
                a_polish = a_pk.copy()
                print(f"  Peak mult: {best_score:.8f}")

        # Another round of very fine p-norm
        rem3 = phase3_budget - (time.monotonic() - phase3_start)
        if rem3 > 5.0:
            a_fine2, obj_fine2 = pnorm_descent(a_polish, p_start=500.0, p_end=10000.0,
                                               max_iters=int(rem3 * 50),
                                               lr_init=0.02, momentum=0.96,
                                               time_budget=rem3 * 0.8, cooldown=10)
            if obj_fine2 < best_score:
                best_score = obj_fine2
                best_seq = a_fine2.copy()
                print(f"  Ultra-fine p-norm: {best_score:.8f}")

    # === Phase 4: Try small perturbations and re-optimize (remaining time) ===
    print(f"\n=== Phase 4: Perturbation Search (best={best_score:.8f}) ===")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 10.0:
        n_best = len(best_seq)
        n_perturb = int(remaining4 / 15)
        n_perturb = max(2, min(10, n_perturb))
        per_perturb = min(12.0, remaining4 / n_perturb)

        for pi in range(n_perturb):
            if time.monotonic() > deadline - 10:
                break

            sigma = 0.03 + 0.02 * pi
            a_pert = best_seq * np.exp(rng.normal(0, sigma, n_best))
            a_pert = project(a_pert)

            # Quick local opt
            a_opt, obj_opt = local_optimize(a_pert, time_budget=per_perturb)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Perturb {pi} (sigma={sigma:.3f}): NEW BEST {best_score:.8f}")
            else:
                print(f"  Perturb {pi}: obj={obj_opt:.8f} (no improvement)")

    # === Final validation ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val):
            if val < best_score:
                best_score = val
            print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f} ===")
        else:
            print(f"\nEvaluation returned non-finite, using fallback")
            result = [1.0] * 100
            best_score = 2.0
    except Exception as e:
        print(f"Evaluation error: {e}")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s")
    return result
