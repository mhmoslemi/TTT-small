Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP alternating, Nesterov-accelerated projected subgradient,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=30):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.all(v <= 0):
        return np.ones(n) / n
    # Find threshold using binary search
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    v = np.zeros(n)

        # d c[k]/d a[j] = a[k-j] + a[k-j] * [k != 2j]

        # grad[j] = sum_k w[k] * (a[k-j] if k-j in [0,n) else 0) * multiplicity
        # For efficiency, use FFT
        size = 1
        while size < 3 * n - 1:
            size <<= 1
        a_flip = a_cur[::-1]
        wf = rfft(w, size)
        af = rfft(a_flip, size)
        corr = irfft(wf * af, size)
        grad = 2.0 * corr[n-1:2*n-1]
        # Subtract the diagonal correction (for k=2j terms, the factor is 1 not 2)
        # This is a minor correction, skip for now
        
        # Nesterov step
        v = beta * v + grad
        a_trial = a_cur - alpha * v
        a_trial = project_simplex(a_trial)
        
        obj_try = compute_obj(a_trial)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_trial.copy()
            a_cur = a_trial
            # Decay alpha slowly
            alpha *= 0.999
        else:
            alpha *= 0.7
            if alpha < 1e-8:
                break
    
    return best_a

def lp_alternate(a, max_iters=50, time_budget=80.0):
    """LP alternating with line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp(a, M, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break
        
        # Line search for blend parameter
        best_t = 0.0
        best_obj_try = best_obj
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = normalize((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_t = t
        
        a_new = normalize((1.0 - best_t) * a + best_t * g)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Perturb
            rng = np.random.default_rng((it * 104729 + 42) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def population_search(best_a, population_size=12, generations=15, time_budget=200):
    """Population-based search with blend crossover and diverse mutations."""
    start = time.monotonic()
    n = len(best_a)
    rng = np.random.default_rng(12345)
    
    # Initialize population
    population = [normalize(best_a.copy())]
    for i in range(population_size - 1):
        if i % 3 == 0:
            a_try = best_a * np.exp(rng.normal(0, 0.1, n))
        elif i % 3 == 1:
            freq = rng.uniform(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            a_try = best_a * mod
        else:
            a_rand = rng.dirichlet(np.ones(n) * 0.3)
            a_try = 0.7 * normalize(best_a) + 0.3 * a_rand
        population.append(normalize(a_try))
    
    # Evaluate initial population
    objs = [compute_obj(pop) for pop in population]
    best_idx = int(np.argmin(objs))
    best_obj = objs[best_idx]
    best_found = population[best_idx].copy()
    
    for gen in range(generations):
        if time.monotonic() - start > time_budget - 30:
            break
        
        # Sort by objective
        order = np.argsort(objs)
        population = [population[i] for i in order]
        objs = [objs[i] for i in order]
        
        # Keep elite (top half)
        elite_count = population_size // 2
        
        # Generate offspring
        new_pops = []
        for i in range(population_size - elite_count):
            if time.monotonic() - start > time_budget - 10:
                break
            
            # Select parents (tournament)
            p1_idx = int(rng.integers(0, elite_count))
            p2_idx = int(rng.integers(0, elite_count))
            
            # Blend crossover
            mix = rng.uniform(0.3, 0.7)
            child = mix * population[p1_idx] + (1 - mix) * population[p2_idx]
            
            # Mutation
            mut_type = int(rng.integers(0, 4))
            if mut_type == 0:
                child = child * np.exp(rng.normal(0, 0.05, n))
            elif mut_type == 1:
                # Block swap
                block_len = int(rng.integers(n // 10, n // 4))
                start_idx = int(rng.integers(0, n - block_len))
                child[start_idx:start_idx+block_len], child[start_idx+block_len:start_idx+2*block_len] = \
                    (child[start_idx+block_len:start_idx+2*block_len].copy(), 
                     child[start_idx:start_idx+block_len].copy()) if start_idx + 2*block_len <= n else (child[start_idx:start_idx+block_len], child[start_idx:start_idx+block_len])
            elif mut_type == 2:
                # Sinusoidal
                freq = rng.uniform(0.5, 5)
                child = child * (1.0 + 0.1 * np.sin(2 * np.pi * freq * np.arange(n) / n))
            else:
                # Sparse perturbation
                mask = rng.random(n) < 0.1
                child[mask] *= np.exp(rng.normal(0, 0.15, mask.sum()))
            
            child = normalize(child)
            new_pops.append(child)
        
        # Evaluate new population
        population = population[:elite_count] + new_pops
        for i in range(elite_count, len(population)):
            objs.append(compute_obj(population[i]))
        
        # Update best
        if objs[0] < best_obj:
            best_obj = objs[0]
            best_found = population[0].copy()
    
    return best_found, best_obj

def coordinate_descent(a, time_budget=20.0):
    """Pairwise mass-transfer coordinate descent."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)
        
        if not contrib:
            break
        
        improved = False
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
        
            for _ in range(5):
                j_target = int(rng.integers(0, n))
                if j_target in (j1, j2):
                    continue
        
                a_try = a.copy()
                for frac in [0.03, 0.05, 0.1]:
                    delta = frac * max(a_try[j1], a_try[j2])
                    src = j1 if a_try[j1] > a_try[j2] else j2
                    a_try[src] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
        
        if not improved:
            if rng.random() < 0.05:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:
                break
    
    return best_a

def spectral_equalization(a, max_iters=50, time_budget=30.0):
    """POCS-style spectral equalization."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        
        alpha = 0.4 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.5 * a + 0.5 * a_new)
    
    return best_a

"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix for conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def golden_section_search(f, lo, hi, tol=0.02, max_iter=20):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def lp_alternate(a, max_iters=50, time_budget=80.0):
    """LP alternating with golden section line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(2024)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break

        def obj_at_t(t):
            a_try = normalize((1.0 - t) * a + t * g)
            return compute_obj(a_try)

        t_opt = golden_section_search(obj_at_t, 0.0, 1.0, tol=0.02, max_iter=15)

        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Targeted perturbation: focus on peak contributors
            conv_new = compute_conv(a)
            k_peak = int(np.argmax(conv_new))
            pert = rng.normal(0, 0.03, n)
            # Extra perturbation near peak contributors
            for j in range(n):
                if 0 <= k_peak - j < n:
                    pert[j] += rng.normal(0, 0.02)
                    pert[k_peak - j] += rng.normal(0, 0.02)
            a = normalize(best_a * np.exp(pert))

    return best_a


def nesterov_subgradient(a, p_init=15, p_final=50, alpha_init=0.04,
                         beta=0.9, max_steps=300, time_budget=60):
    """Nesterov-accelerated projected subgradient with adaptive p-smoothing."""
    n = len(a)
    a_cur = normalize(a.copy().astype(np.float64))
    best_a = a_cur.copy()
    best_obj = compute_obj(a_cur)
    start = time.monotonic()
    v = np.zeros(n)
    alpha = alpha_init

    size = 1
    while size < 3 * n - 1:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a_cur)
        cmax = float(np.max(conv))

        # Adaptive p: start broad, narrow down
        progress = step / float(max(1, max_steps - 1))
        p = p_init + (p_final - p_init) * progress
        tau = cmax / p + 1e-15

        shifted = np.clip((conv - cmax) / tau, -200.0, 200.0)
        w = np.exp(shifted)
        w_sum = float(np.sum(w))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            w = np.zeros(len(conv))
            w[idx] = 1.0
        else:
            w /= w_sum

        # Gradient via FFT cross-correlation
        wf = rfft(w, size)
        af = rfft(a_cur, size)
        corr = irfft(wf * np.conj(af), size)
        grad = 2.0 * corr[:n]

        # Nesterov momentum step
        v = beta * v + grad
        a_trial = project_simplex(a_cur - alpha * v)

        obj_try = compute_obj(a_trial)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_trial.copy()
            a_cur = a_trial
            alpha = max(alpha * 0.998, 1e-6)
        else:
            alpha *= 0.65
            v *= 0.5
            if alpha < 1e-8:
                break

    return best_a


def spectral_equalization(a, max_iters=50, time_budget=30.0):
    """POCS-style spectral equalization."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        alpha = 0.3 + 0.5 * (it / float(max(iters_done := max_iters, 1)))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.6 * a + 0.4 * a_new)

    return best_a


def coordinate_descent(a, time_budget=20.0):
    """Pairwise mass-transfer coordinate descent targeting the peak."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        if not contrib:
            break

        improved = False
        for peak_val, j1, j2 in contrib[:10]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            for frac in [0.03, 0.05, 0.08, 0.12]:
                j_target = int(rng.integers(0, n))
                if j_target in (j1, j2):
                    continue
                a_try = a.copy()
                src = j1 if a_try[j1] >= a_try[j2] else j2
                delta = frac * a_try[src]
                a_try[src] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            if rng.random() < 0.08:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:
                break

    return best_a


def population_search(best_a, population_size=14, generations=15, time_budget=200):
    """Population-based search with blend crossover and diverse mutations."""
    start = time.monotonic()
    n = len(best_a)
    rng = np.random.default_rng(54321)

    # Initialize diverse population
    population = [normalize(best_a.copy())]

                    0.5 * np.exp(-((x - (1 - center))**2) / (2 * spread**2))
        population.append(normalize(a_try))

    objs = [compute_obj(pop) for pop in population]
    best_obj = min(objs)
    best_found = population[int(np.argmin(objs))].copy()

    for gen in range(generations):
        if time.monotonic() - start > time_budget - 20:
            break

        order = np.argsort(objs)
        population = [population[i] for i in order]
        objs = [objs[i] for i in order]

        elite_count = max(2, population_size // 3)
        new_pops = []

        for i in range(population_size - elite_count):
            if time.monotonic() - start > time_budget - 5:
                break

            # Tournament selection
            p1_idx = int(rng.integers(0, elite_count))
            p2_idx = int(rng.integers(0, elite_count))

            # Blend crossover
            mix = rng.uniform(0.3, 0.7)
            child = mix * population[p1_idx] + (1.0 - mix) * population[p2_idx]

            # Diverse mutation
            mut_type = int(rng.integers(0, 5))
            if mut_type == 0:
                child = child * np.exp(rng.normal(0, 0.04, n))
            elif mut_type == 1:

                s_idx = int(rng.integers(0, n - blen))

            elif mut_type == 2:

            else:
                # Shift-like perturbation
                shift = int(rng.integers(-3, 4))

        population = population[:elite_count] + new_pops

            objs.append(compute_obj(population[i]))

        curr_best = min(objs)

            best_obj = curr_best

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

    if time.monotonic() < deadline - 400:
        a_spec = spectral_equalization(a_cur, max_iters=50, time_budget=35)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            a_cur = a_spec
            print(f"  Spectral: {best_score:.8f} **")
        else:
            a_cur = normalize(0.7 * a_cur + 0.3 * a_spec)
            print(f"  Spectral: {obj_spec:.8f}")

    # Follow-up LP
    if time.monotonic() < deadline - 350:
        a_new = lp_alternate(a_cur, max_iters=30, time_budget=55)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new
            print(f"  Post-Phase2 LP: {best_score:.8f} **")
        else:
            print(f"  Post-Phase2 LP: {obj_new:.8f}")

    # =========================================================
    # Phase 3: Population-based search
    # =========================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Population search")
    print(f"{'=' * 60}")

    if time.monotonic() < deadline - 300:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.25, 220.0)
        pop_best, pop_obj = population_search(normalize(best_seq),
                                               population_size=16,
                                               generations=14,
                                               time_budget=alloc)
        if pop_obj < best_score:
            best_score = pop_obj
            best_seq = pop_best.copy()
            print(f"  Population: {best_score:.8f} **")
        else:
            print(f"  Population: {pop_obj:.8f}")

        # Polish best population member with LP
        if time.monotonic() < deadline - 50:
            a_pol = lp_alternate(pop_best, max_iters=15, time_budget=40)
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Pop+LP: {best_score:.8f} **")
            else:
                print(f"  Pop+LP: {obj_pol:.8f}")

    # =========================================================
    # Phase 4: Multi-n exploration
    # =========================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'=' * 60}")

    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100),
        max(200, n_best - 50), max(200, n_best - 25),
        n_best,
        n_best + 25, n_best + 50, n_best + 100, n_best + 200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1300]

    for n_try in n_options:
        if time.monotonic() > deadline - 70:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if n_try == n_best:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample_sequence(best_seq, n_try))

        alloc = min(remaining * 0.12, 35.0)
        if alloc < 10:
            break

        if n_try <= 950:
            a_try = lp_alternate(a_try, max_iters=10, time_budget=alloc * 0.45)
        a_try = nesterov_subgradient(a_try, p_init=15, p_final=40,
                                     alpha_init=0.025, beta=0.88,
                                     max_steps=60, time_budget=alloc * 0.35)
        if alloc > 8:
            a_try = coordinate_descent(a_try, time_budget=min(8, alloc * 0.2))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 5: Final intensive polish
    # =========================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Final polish")
    print(f"{'=' * 60}")

    n_final = len(best_seq)
    a_final = normalize(best_seq.copy())

    for rnd in range(6):
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * 0.18, 35.0)
        if alloc < 8:
            break

        # LP if size allows
        if n_final <= 950 and alloc > 15:
            a_pol = lp_alternate(a_final, max_iters=15, time_budget=alloc * 0.4)
        else:
            a_pol = a_final.copy()

        # Nesterov with tighter parameters
        a_pol = nesterov_subgradient(a_pol, p_init=25, p_final=80,
                                     alpha_init=0.012, beta=0.92,
                                     max_steps=80, time_budget=alloc * 0.35)

        # Coordinate descent
        if alloc > 10:
            a_pol = coordinate_descent(a_pol, time_budget=min(10, alloc * 0.2))

        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            a_final = a_pol
            print(f"  Polish {rnd}: {best_score:.8f} **")
        else:
            a_final = a_pol
            print(f"  Polish {rnd}: {obj_pol:.8f}")

    # Final: one more LP pass if time allows
    if time.monotonic() < deadline - 5 and n_final <= 950:
        a_final2 = lp_alternate(a_final, max_iters=20, time_budget=deadline - time.monotonic() - 3)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final LP: {best_score:.8f} **")

    # =========================================================
    # Validation and return
    # =========================================================
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"Target: 1.5027")
    print(f"{'=' * 60}")

    return result
