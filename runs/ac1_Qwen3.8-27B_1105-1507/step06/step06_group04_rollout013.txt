"""
Enhanced sequence optimization using FISTA with temperature annealing,
L-BFGS continuation method, block coordinate descent, and LP polishing.
Focuses on high-quality local optimization with smart restarts to beat 1.506547.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Compute linear autocorrelation a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_simplex(a, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), target / len(a))
    return a * (target / s)


def fft_cross_corr(a, w, fft_size):
    """Compute cross-correlation: corr[i] = sum_j a[j]*w[i+j] for i=0..n-1."""
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    return corr


def fista_anneal(a_init, tau_list, iters_per_tau=2000, time_budget=60.0):
    """FISTA with temperature annealing. Products list of (tau, n_iters)."""
    n = len(a_init)
    a = project_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        # FISTA for this temperature
        y = a.copy()
        t_k = 1.0
        L = 4.0 * n / tau  # Lipschitz constant estimate
        step = 1.0 / L

        for it in range(iters_per_tau):
            if time.monotonic() - start > time_budget:
                break

            # Compute gradient at y
            conv_y = compute_conv(y)
            c = conv_y / tau
            c_max = np.max(c)
            c_shifted = c - c_max
            w = np.exp(c_shifted)
            ws = np.sum(w)
            if ws < 1e-300:
                w = np.zeros_like(conv_y)
                w[int(np.argmax(conv_y))] = 1.0
            else:
                w /= ws

            # grad_i = (4n/tau) * sum_j w[i+j] * y[j]
            corr = fft_cross_corr(y, w, fft_size)
            grad = (4.0 * n / tau) * corr[:n]

            # Project gradient onto tangent space of simplex
            grad_centered = grad - np.mean(grad)
            grad_norm = np.linalg.norm(grad_centered)
            if grad_norm < 1e-12:
                break

            # FISTA step
            a_new = y - step * grad_centered
            a_new = project_simplex(a_new)

            # Evaluate
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

            # Momentum update
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = a_new + ((t_k - 1.0) / t_next) * (a_new - a)
            y = project_simplex(y)
            t_k = t_next
            a = a_new

    return best_a, best_obj


def lbfgs_refine(a_init, tau, time_budget=60.0, maxiter=2000):
    """L-BFGS-B refinement using softmax parameterization on smoothed objective."""
    n = len(a_init)
    a = project_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    # Parameterize: a_i = exp(theta_i) / sum(exp(theta_j))
    theta = np.log(a + 1e-12)

    def obj_grad(theta):
        a_s = np.exp(theta - np.max(theta))
        a_s = a_s / np.sum(a_s)
        conv = compute_conv(a_s)
        c = conv / tau
        c_max = np.max(c)
        c_shifted = c - c_max
        w = np.exp(c_shifted)
        ws = np.sum(w)
        if ws < 1e-300:
            obj = 2.0 * n * np.max(conv)
            # Numerical gradient
            g = np.zeros(n)
            for i in range(min(n, 50)):
                eps = 1e-6
                tp = theta.copy()
                tp[i] += eps
                ap = np.exp(tp - np.max(tp))
                ap /= np.sum(ap)
                obj_p = 2.0 * n * np.max(compute_conv(ap))
                g[i] = (obj_p - obj) / eps
            return obj, g
        w /= ws
        obj = 2.0 * n * (c_max + np.log(ws)) * tau

        # Gradient w.r.t. a
        corr = fft_cross_corr(a_s, w, fft_size)
        grad_a = (4.0 * n / tau) * corr[:n] * tau  # = 4n * corr[:n]
        # Actually: d/dx [logsumexp(x)] = softmax(x), so d/d a [2n*logsumexp(conv/tau)]
        # = (2n/tau) * sum_k w_k * 2*a_{k-i} = (4n/tau) * corr[i]
        # But obj = 2n*(c_max + log(ws))*tau = 2n*(logsumexp(conv/tau))
        # So grad_a_i = (2n/tau) * sum_k w_k * 2*a_{k-i} = (4n/tau) * corr[i]
        grad_a = (4.0 * n / tau) * corr[:n]

        # Chain rule for softmax
        dot_ga = np.dot(grad_a, a_s)
        grad_theta = a_s * (grad_a - dot_ga)

        return obj, grad_theta

    try:
        result = minimize(obj_grad, theta, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'maxfun': maxiter * 3,
                                   'ftol': 1e-12, 'gtol': 1e-8})
        theta_opt = result.x
        a_opt = np.exp(theta_opt - np.max(theta_opt))
        a_opt = a_opt / np.sum(a_opt)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt
    except Exception:
        pass

    return best_a, best_obj


def continuation_lbfgs(a_init, tau_list, time_per_tau=30.0, time_budget=300.0):
    """Continuation method: L-BFGS at decreasing temperatures."""
    n = len(a_init)
    best_a = project_simplex(a_init.copy())
    best_obj = compute_obj(best_a)
    start = time.monotonic()

def build_convolve_matrix(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=20):
    """Solve: min t s.t. (fixed_seq * g)[k] <= t, sum(g)=1, g>=0."""
    n = len(fixed_seq)
    A = build_convolve_matrix(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None

def block_cd(a_init, block_size=30, time_budget=60.0, max_sweeps=10):
    """Block coordinate descent with exact block optimization."""
    n = len(a_init)
    a = project_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    blocks = []
    for i in range(0, n, block_size):
        blocks.append((i, min(i + block_size, n)))

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for start_idx, end_idx in blocks:
            if time.monotonic() - start > time_budget:
                break
            b = end_idx - start_idx
            if b < 2:
                continue

            a_fixed = a.copy()
            block_sum = np.sum(a_fixed[start_idx:end_idx])

            def block_obj(block_vals):
                a_try = a_fixed.copy()
                a_try[start_idx:end_idx] = block_vals
                a_try = project_simplex(a_try)
                conv = compute_conv(a_try)
                return np.max(conv)

            # Optimize the block
            x0 = a_fixed[start_idx:end_idx].copy()
            try:
                res = minimize(block_obj, x0, method='Nelder-Mead',
                               options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-12})
                if res.fun < block_obj(x0) - 1e-14:
                    a[start_idx:end_idx] = np.maximum(res.x, 0)
                    a = project_simplex(a)
                    obj_new = compute_obj(a)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass

        if not improved:
            break

    return best_a, best_obj


def generate_starts(n, seed=42, num_starts=10):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    starts.append(np.ones(n) / n)

    # Triangular
    tri = np.maximum(1.0 - np.abs(i - center) / (center + 1), 0.01)
    starts.append(tri / np.sum(tri))

    # Gaussian (various widths)
    for width_factor in [0.15, 0.25, 0.35, 0.5]:
        bell = np.exp(-((i - center) ** 2) / (2 * (n * width_factor) ** 2))
        starts.append(bell / np.sum(bell))

    # Cosine窗
    cos_win = np.cos(np.pi * (i - center) / (center + 1)) ** 2
    cos_win = np.maximum(cos_win, 0.001)
    starts.append(cos_win / np.sum(cos_win))

    # Random
    for _ in range(3):
        r = np.abs(rng.normal(1.0, 0.5, n))
        starts.append(r / np.sum(r))

    # Structured: sum of Gaussians
    mu1, mu2 = 0.3 * n, 0.7 * n
    sig = n * 0.1
    dual = (np.exp(-(i - mu1) ** 2 / (2 * sig ** 2)) +
            np.exp(-(i - mu2) ** 2 / (2 * sig ** 2)))
    starts.append(dual / np.sum(dual))

    # Flat top with tapered edges
    flat = np.ones(n)
    taper = max(5, n // 20)
    for j in range(taper):
        flat[j] = j / taper
        flat[n - 1 - j] = j / taper
    starts.append(flat / np.sum(flat))

    return starts


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_pipeline(a_init, time_budget=120.0):
    """Full polishing pipeline: FISTA anneal -> L-BFGS continuation -> LP -> Block CD."""
    n = len(a_init)
    a = project_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: FISTA with temperature annealing
    remaining = time_budget - (time.monotonic() - start)
    tau_list_1 = [0.05, 0.03, 0.02, 0.01, 0.005]
    if remaining > 20:
        a_new, obj_new = fista_anneal(best_a, tau_list_1,
                                      iters_per_tau=1500,
                                      time_budget=min(remaining * 0.4, 50.0))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new
            print(f"    FISTA: {best_obj:.8f}")

    # Phase 2: L-BFGS continuation
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        tau_list_2 = [0.02, 0.01, 0.005, 0.002]
        a_new, obj_new = continuation_lbfgs(best_a, tau_list_2,
                                            time_per_tau=min(remaining * 0.1, 20.0),
                                            time_budget=min(remaining * 0.5, 80.0))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new
            print(f"    L-BFGS: {best_obj:.8f}")

    # Phase 3: LP polishing
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15 and n <= 900:
        a_new, obj_new = lp_alternate(best_a, max_iters=30,
                                      time_budget=min(remaining * 0.3, 40.0))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new
            print(f"    LP: {best_obj:.8f}")

    # Phase 4: Block CD
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_new, obj_new = block_cd(best_a, block_size=20,
                                  time_budget=min(remaining * 0.3, 30.0))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new
            print(f"    BlockCD: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(budget_s=1500):
    """
    Main search: multi-phase optimization combining FISTA annealing,
    L-BFGS continuation, LP alternating, and block coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting best: {best_score:.8f} (n={n_parent})")

    # Phase 1: Quick exploration of different lengths and starts
    print("\n=== Phase 1: Exploration ===")
    candidates = []  # (obj, seq)

    n_values = sorted(set([n_parent - 100, n_parent - 50, n_parent,
                           n_parent + 50, n_parent + 100,
                           n_parent + 200]))
    n_values = [x for x in n_values if 350 <= x <= 1300]

    for n in n_values:
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        starts = generate_starts(n, seed=42 + n, num_starts=8)

        # If this is the parent length and we have a parent, include it
        if n == n_parent and best_seq is not None:
            starts.insert(0, project_simplex(best_seq.copy()))

        alloc_per = min(remaining / (len(starts) * 1.5), 15.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 250:
                break
            if alloc_per < 5:
                break

            # Quick FISTA
            tau_quick = [0.03, 0.01, 0.005]
            a_opt, obj_opt = fista_anneal(s, tau_quick,
                                          iters_per_tau=800,
                                          time_budget=alloc_per * 0.6)

            # Quick L-BFGS
            remaining2 = deadline - time.monotonic()
            if remaining2 > 100 and alloc_per > 8:
                a_opt2, obj_opt2 = lbfgs_refine(a_opt, 0.005,
                                                 time_budget=min(alloc_per * 0.4, 15.0))
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {obj_opt:.8f} ** NEW BEST **")

            candidates.append((obj_opt, a_opt.copy()))

    candidates.sort(key=lambda x: x[0])
    print(f"\nTop 5: {[f'{c[0]:.6f}' for c in candidates[:5]]}")
    print(f"Best so far: {best_score:.8f}")

    # Phase 2: Deep optimization of top candidates
    print("\n=== Phase 2: Deep optimization ===")
    top_k = min(5, len(candidates))
    seen_seqs = []

    for ci in range(top_k):
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand = candidates[ci][1].copy()
        obj_cand = candidates[ci][0]

        # Skip if too similar to an already-optimized candidate
        if any(np.linalg.norm(a_cand - s) < 0.01 for s in seen_seqs):
            continue
        seen_seqs.append(a_cand.copy())

        alloc = min(remaining * 0.35, 180.0)
        print(f"  Cand {ci} (n={len(a_cand)}, obj={obj_cand:.8f}): {alloc:.0f}s budget")

        a_pol, obj_pol = polish_pipeline(a_cand, time_budget=alloc)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"    ** UPDATE: {best_score:.8f} **")

    # Phase 3: Length exploration on best
    print(f"\n=== Phase 3: Length exploration (best: {best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        test_ns = [n_best - 50, n_best + 50, n_best - 100, n_best + 100,
                   n_best + 150, n_best - 150]
        test_ns = sorted(set(x for x in test_ns if 350 <= x <= 1400))

        for n_t in test_ns:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_res = resample_seq(best_seq, n_t)
            a_res = project_simplex(a_res)

            # Quick polish
            alloc = min(remaining * 0.25, 50.0)
            a_pol, obj_pol = polish_pipeline(a_res, time_budget=alloc)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_t}: ** {best_score:.8f} **")
            else:
                print(f"  n={n_t}: {obj_pol:.8f}")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_final, obj_final = polish_pipeline(best_seq, time_budget=min(remaining * 0.6, 120.0))
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: {best_score:.8f}")

        # Try perturbations
        remaining = deadline - time.monotonic()
        if remaining > 20 and best_seq is not None:
            n_f = len(best_seq)
            rng = np.random.default_rng(12345)
            for p in range(5):
                if time.monotonic() > deadline - 10:
                    break
                scale = 0.01 + 0.005 * p
                a_pert = project_simplex(best_seq * np.exp(rng.normal(0, scale, n_f)))
                t_alloc = min(remaining * 0.15, 20.0)
                if t_alloc < 5:
                    break
                a_p, obj_p = fista_anneal(a_pert, [0.01, 0.005, 0.002],
                                          iters_per_tau=500, time_budget=t_alloc)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Perturb {p}: {best_score:.8f}")

    # Phase 5: Final LP on absolute best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_b = len(best_seq)
        if n_b <= 850 and remaining > 20:
            a_lp, obj_lp = lp_alternate(best_seq, max_iters=20,
                                         time_budget=min(remaining * 0.7, 40.0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
