"""
Peak-reduction search for step function convolution optimization.
Combines smoothed gradient descent (log-sum-exp with adaptive temperature),
LP-based block coordinate descent, Nesterov momentum, and multi-start
diverse search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(a, time_limit=30):
    """
    Solve: min t subject to (a * g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(a)
    A = build_toeplitz_convolve_mat(a)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def smoothed_gradient_descent(a, max_iters=5000, time_budget=60.0, 
                               tau_init=0.005, tau_final=0.0005,
                               momentum=0.9, lr_init=0.05):
    """
    Gradient descent on the smoothed objective log(sum exp(c[k]/tau))
    with adaptive temperature and Nesterov momentum.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    v = np.zeros(n)
    lr = lr_init
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive temperature (exponential decay)
        frac = min(it / max(max_iters, 1), 1.0)
        tau = tau_init * (tau_final / tau_init) ** frac
        
        # Compute convolution
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Softmax weights
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_shifted = np.exp(shifted)
        Z = np.sum(exp_shifted)
        if Z < 1e-300:
            idx = int(np.argmax(conv))
            p = np.zeros(len(conv))
            p[idx] = 1.0
        else:
            p = exp_shifted / Z
        
        # Gradient: grad[i] = 2 * sum_j p[i+j] * a[j]
        # Using FFT: corr = ifft(fft(a) * conj(fft(p_padded)))
        # corr[m] = sum_j a[j] * p[m+j] (for linear, with zero-padding)
        p_padded = np.zeros(fft_size)
        p_padded[:len(p)] = p
        A_fft = rfft(a, fft_size)
        P_fft = rfft(p_padded, fft_size)
        corr = irfft(A_fft * np.conj(P_fft), fft_size)
        grad = 2.0 * corr[:n]
        
        # Center gradient (project onto sum=const)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break
        
        # Nesterov momentum
        v = momentum * v - lr * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive learning rate
        if obj_new > best_obj:
            lr *= 0.95
        else:
            lr = min(lr * 1.01, 0.1)
        
        # Decay learning rate slowly
        lr *= 0.9999

    return best_a, best_obj


def lp_alternate(a, max_iters=50, time_budget=120.0):
    """
    LP-based block coordinate descent: alternate min-peak LP solves
    with line search for blending.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        lp_time = min(remaining * 0.3, 25.0)
        g, t_opt = solve_minpeak_lp(a, time_limit=lp_time)
        if g is None:
            stall += 1
            if stall > 8:
                break
            continue
        
        # Line search for blending parameter
        # Try a range of t values
        cands = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]
        if last_t not in cands:
            cands.append(last_t)
            cands.append(min(1.0, last_t * 1.5))
            cands.append(max(0.05, last_t * 0.5))
        cands = sorted(set(cands))
        
        best_t = 0.3
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        
        for t in cands:
            a_try = (1.0 - t) * a + t * g
            a_try = normalize_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            last_t = best_t
            stall = 0
        else:
            stall += 1
            if stall > 6:
                # Perturb and try again
                rng = np.random.default_rng(it * 12345 + 67890)
                scale = 0.03 + 0.02 * (stall % 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = normalize_simplex(a)

                                                    time_budget=min(5.0, remaining*0.1),
                                                    tau_init=0.003, tau_final=0.001)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                stall = 0

    return best_a, best_obj
