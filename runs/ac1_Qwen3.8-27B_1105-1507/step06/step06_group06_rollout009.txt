Multi-resolution L-BFGS optimization with log-sum-exp smoothing

from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = compute_conv(a)
    return 2.0 * n * np.max(c) / (s * s)

def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def smoothed_obj_grad(a, tau):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return 1e10, np.zeros(n)
    c = compute_conv(a)
    c_max = np.max(c)
    # Log-sum-exp for numerical stability
    shifted = (c - c_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    log_sum_exp = c_max + tau * np.log(np.sum(np.exp(shifted)))
    obj = 2.0 * n * log_sum_exp / (s * s)
    
    # Probabilities
    p = np.exp(shifted - np.log(np.sum(np.exp(shifted))))
    # This is exp((c - c_max)/tau) / sum(exp((c-c_max)/tau))
    
    # Cross-correlation r_i = sum_k p[k] * a[k-i] * [0<=k-i<n]
    N = 1
    while N < 3*n - 2:
        N <<= 1
    fp = np.fft.fft(p, N)
    fa = np.fft.fft(a, N)
    r = np.fft.ifft(fp * np.conj(fa)).real[:n]
    
    # Gradient: grad_i = (4*n*tau/s^2) * (r_i - L/s)
    L = log_sum_exp
    grad = (4.0 * n * tau / (s * s)) * (r - L / s)
    
    return obj, grad

def lbfgs_optimize(a_init, tau, max_iter=500, time_limit=30.0):
    a = a_init.copy()
    s = np.sum(a)
    n = len(a)
    start = time.monotonic()
    
    def func_and_grad(x):
        if time.monotonic() - start > time_limit:
            raise Exception("timeout")
        x_pos = np.maximum(x, 0)
        return smoothed_obj_grad(x_pos, tau)
    
    try:
        res = minimize(func_and_grad, a, method='L-BFGS-B',
                      bounds=[(0, None)] * n,
                      options={'maxiter': max_iter, 'maxfun': max_iter * 3, 'gtol': 1e-10})
        a_new = np.maximum(res.x, 0)
    except Exception:
        a_new = a
    
    a_new = project_nonneg_sum1(a_new)
    return a_new

def anneal_optimize(a_init, time_budget=60.0, verbose=False):
    """Multi-round L-BFGS with decreasing tau."""
    a = project_nonneg_sum1(a_init.copy())
    n = len(a)
    s = np.sum(a)
    c = compute_conv(a)
    c_max = np.max(c)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Tau schedule: start from a fraction of c_max, decrease
    taus = [c_max * 0.1, c_max * 0.03, c_max * 0.01, c_max * 0.003, c_max * 0.001, c_max * 0.0003, c_max * 0.0001]
    taus = [t for t in taus if t > 1e-10]

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:

        # For n=700, roughly 0.01-0.05 seconds per iteration
        est_time_per_iter = max(0.01, n * 10 / (10**7))

        a = project_nonneg_sum1(a)
        obj = compute_obj(a)
        
        if verbose:

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

    b = np.exp(-((x - 0.25) ** 2) / 0.01) + np.exp(-((x - 0.75) ** 2) / 0.01)
    b = np.maximum(b, 0.01)
    inits.append(b / b.sum())
    # Smoothed random
    rnd = rng.random(n)
    # Smooth via cumulative sum differences
    from scipy.ndimage import uniform_filter1d
    rnd_s = uniform_filter1d(rnd, size=max(3, n // 20), mode='wrap')
    rnd_s = np.maximum(rnd_s, 0.01)
    inits.append(rnd_s / rnd_s.sum())
    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=float)
    inits.append(ramp / ramp.sum())
    # Triangle
    tri = np.minimum(x, 1.0 - x) if x is not None else None
    x = np.linspace(0, 1, n)
    tri = np.minimum(x, 1.0 - x) + 0.05
    inits.append(tri / tri.sum())

    if parent is not None:
        inits.append(resample_sequence(parent, n))
        # Perturbed parent
        p = resample_sequence(parent, n)
        pert = p * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_nonneg_sum1(pert))

    return inits


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    parent = try_load_parent()
    if parent is not None:
        p_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={p_obj:.8f}")
        if p_obj < best_score:
            best_score = p_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"No parent, using uniform n=700: obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Multi-length, multi-init screening ===
    print("\n=== Phase 1: Multi-length Screening ===")
    phase1_budget = min(120.0, (deadline - start_time) * 0.10)
    n_list = sorted(set([
        max(200, n_parent - 100),
        max(200, n_parent - 50),
        n_parent,
        n_parent + 50,
        n_parent + 100,
        n_parent + 200,
        min(1500, n_parent + 300),
        min(2000, n_parent + 500),
    ]))
    n_list = [n for n in n_list if 100 <= n <= 2500]

    candidates = []
    per_n_time = max(3.0, phase1_budget / len(n_list) / 3.0)

    for n_try in n_list:
        if time.monotonic() > start_time + phase1_budget:
            break
        inits = generate_inits(n_try, rng, parent)
        for ii, init in enumerate(inits):
            if time.monotonic() > start_time + phase1_budget:
                break
            t_budget = min(per_n_time, (start_time + phase1_budget) - time.monotonic())
            if t_budget < 2:
                break
            a_opt, obj_opt = anneal_optimize(init, time_budget=t_budget, verbose=False, tag=f"n={n_try},i={ii}")
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}, init {ii}: NEW BEST {best_score:.8f}")
            candidates.append((obj_opt, a_opt.copy(), n_try))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done. Best: {candidates[0][0] if candidates else 'N/A'}")

    # === Phase 2: Intensive optimization on top candidates ===
    print(f"\n=== Phase 2: Intensive Optimization (best={best_score:.8f}) ===")
    phase2_budget = min(700.0, (deadline - start_time) * 0.50)
    phase2_start = time.monotonic()

    top_k = min(4, len(candidates))
    seen_ns = set()
    top_candidates = []
    for obj_c, a_c, n_c in candidates:
        if n_c not in seen_ns:
            top_candidates.append((obj_c, a_c, n_c))
            seen_ns.add(n_c)
        if len(top_candidates) >= top_k:
            break

    # Always include the current best
    if (best_score, len(best_seq)) not in [(o, n) for o, _, n in top_candidates]:
        top_candidates.insert(0, (best_score, best_seq.copy(), len(best_seq)))

    for ci, (obj_init, a_init, n_init) in enumerate(top_candidates):
        if time.monotonic() > phase2_start + phase2_budget:
            break
        remaining = phase2_start + phase2_budget - time.monotonic()
        if remaining < 20:
            break
        alloc = min(150.0, remaining / max(1, len(top_candidates) - ci))
        print(f"  Candidate {ci} (n={n_init}): obj={obj_init:.8f}, alloc={alloc:.0f}s")

        # Multiple rounds of annealed optimization with perturbations
        a_cur = project_nonneg_sum1(a_init.copy())
        obj_cur = compute_obj(a_cur)
        stall_count = 0

        for round_i in range(min(8, int(alloc / 15))):
            if time.monotonic() > phase2_start + phase2_budget:
                break
            rem = phase2_start + phase2_budget - time.monotonic()
            if rem < 10:
                break
            t_b = min(rem * 0.8 / max(1, 8 - round_i), 60.0)

            a_new, obj_new = anneal_optimize(a_cur, time_budget=t_b,
                                             verbose=(round_i == 0), tag=f"c{ci}r{round_i}")
            if obj_new < obj_cur - 1e-10:
                a_cur = a_new.copy()
                obj_cur = obj_new
                stall_count = 0
                if obj_cur < best_score:
                    best_score = obj_cur
                    best_seq = a_cur.copy()
                    print(f"    ** c{ci}r{round_i}: NEW BEST {best_score:.8f} **")
            else:
                stall_count += 1

            # Perturb if stalled
            if stall_count >= 2:
                a_pert = a_cur * np.exp(rng.normal(0, 0.03 + 0.01 * round_i, len(a_cur)))
                a_cur = project_nonneg_sum1(a_pert)
                # Quick re-optimize
                a_cur, _ = anneal_optimize(a_cur, time_budget=min(15.0, rem * 0.4),
                                           verbose=False, tag="pert")
                obj_cur = compute_obj(a_cur)
                stall_count = 0

        print(f"  Candidate {ci} final: obj={obj_cur:.8f}")

    # === Phase 3: Fine-tuning on the best ===
    print(f"\n=== Phase 3: Fine-tuning (best={best_score:.8f}) ===")
    phase3_start = time.monotonic()
    phase3_budget = min(400.0, deadline - phase3_start)
    n_best = len(best_seq)

    # Very fine tau annealing
    a_fine = project_nonneg_sum1(best_seq.copy())
    obj_fine = compute_obj(a_fine)

    c_fine = compute_conv(a_fine)
    c_max_fine = float(np.max(c_fine))
    s_fine = float(np.sum(a_fine))

    fine_taus = [c_max_fine * 0.0001, c_max_fine * 3e-05, c_max_fine * 1e-05,
                 c_max_fine * 3e-06, c_max_fine * 1e-06, c_max_fine * 3e-07]
    fine_taus = [t for t in fine_taus if t > 1e-14]

    for idx, tau in enumerate(fine_taus):
        if time.monotonic() > phase3_start + phase3_budget:
            break
        rem = phase3_start + phase3_budget - time.monotonic()
        if rem < 5:
            break
        t_b = min(rem * 0.6, 40.0)
        max_iter = min(1000, max(100, int(t_b / 0.03)))
        a_new = lbfgs_round(a_fine, tau, max_iter=max_iter, time_limit=t_b)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_fine = a_new.copy()
            print(f"  Fine tau={tau:.2e}: NEW BEST {best_score:.8f}")
        else:
            a_fine = a_new.copy()

    # === Phase 4: Length perturbation ===
    print(f"\n=== Phase 4: Length Perturbation (best={best_score:.8f}) ===")
    phase4_start = time.monotonic()
    phase4_budget = min(200.0, deadline - phase4_start)

    n_tries = sorted(set([n_best - 20, n_best - 40, n_best + 20,
                          n_best + 40, n_best + 80, n_best - 80]))
    n_tries = [n for n in n_tries if 100 <= n <= 2500]
    per_n = max(5.0, phase4_budget / max(1, len(n_tries)))

    for n_try in n_tries:
        if time.monotonic() > phase4_start + phase4_budget:
            break
        a_res = resample_sequence(best_seq, n_try)
        a_res = project_nonneg_sum1(a_res)
        t_b = min(per_n, (phase4_start + phase4_budget) - time.monotonic())
        if t_b < 3:
            break
        a_opt, obj_opt = anneal_optimize(a_res, time_budget=t_b, verbose=False, tag=f"n={n_try}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")

    # === Phase 5: Final perturbation restarts ===
    print(f"\n=== Phase 5: Final Restarts (best={best_score:.8f}) ===")
    phase5_start = time.monotonic()
    phase5_budget = max(10.0, deadline - phase5_start)
    n_final = len(best_seq)

    for ri in range(min(20, int(phase5_budget / 10))):
        if time.monotonic() > phase5_start + phase5_budget:
            break
        rem = phase5_start + phase5_budget - time.monotonic()
        if rem < 8:
            break

        # Different perturbation strategies
        strat = ri % 4
        if strat == 0:
            a_pert = best_seq * np.exp(rng.normal(0, 0.01, n_final))
        elif strat == 1:
            a_pert = best_seq.copy()
            idx_sel = rng.choice(n_final, size=max(10, n_final // 10), replace=False)
            a_pert[idx_sel] *= rng.uniform(0.5, 1.5, len(idx_sel))
        elif strat == 2:
            # Window swap
            ws = max(10, n_final // 20)
            w1 = rng.integers(0, n_final - ws)
            w2 = rng.integers(0, n_final - ws)
            if w2 < w1:
                w1, w2 = w2, w1
            if w1 - w2 >= ws:
                a_pert = best_seq.copy()
                a_pert[w1:w1 + ws], a_pert[w2:w2 + ws] = \
                    a_pert[w2:w2 + ws].copy(), a_pert[w1:w1 + ws].copy()
            else:
                a_pert = best_seq * np.exp(rng.normal(0, 0.005, n_final))
        else:
            a_pert = best_seq * np.exp(rng.normal(0, 0.02, n_final))

        a_pert = project_nonneg_sum1(a_pert)
        t_b = min(15.0, rem * 0.7)
        a_opt, obj_opt = anneal_optimize(a_pert, time_budget=t_b,
                                         verbose=False, tag=f"r{ri}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {ri} (strat {strat}): NEW BEST {best_score:.8f}")

    # Final validation
    print(f"\n=== Final: n={len(best_seq)}, obj={best_score:.8f} ===")
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
        print(f"  Validated: {best_score:.8f}")
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = evaluate_sequence(result)
        print(f"  Fallback: {best_score:.8f}")

    elapsed = time.monotonic() - start_time
    print(f"  Total time: {elapsed:.1f}s")
    return result
