def compute_conv(array: list[float]) -> list[float]:
    """Compute full linear convolution using FFT."""
    n = len(array)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = rfft(array, size)
    conv = np.real(irfft(fa * fa, size))
    return conv[:2*n-1]
