def smooth_max_value(c, tau):
    """Compute the smoothed max: (1/tau)*log(sum(exp(tau*c)))."""
    c_max = c.max()
    return c_max + np.log(np.exp(tau * (c - c_max)).sum()) / tau


def smoothed_objective(a, tau):
    """Compute the smoothed objective: 2*n*smooth_max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    c = fast_conv(a)
    sm = smooth_max_value(c, tau)
    return 2.0 * n * sm / (s * s)
