Multi-phase optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Key innovations: circular shift exploration, cutting-plane LP, 
simulated annealing, and structured initialization.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k_idx = np.where(cond)[0]
    if len(k_idx) == 0:
        return np.ones(n) / n
    lam = cssv[k_idx[-1]] / (k_idx[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def try_shifts(a, n_shifts=None):
    """Try circular shifts and return the best one."""
    n = len(a)
    if n_shifts is None:
        n_shifts = n
    best_a = a.copy()
    best_J = compute_J(a)
    # Only try a subset of shifts if n is large
    shifts = np.linspace(0, n - 1, min(n, n_shifts), dtype=int)
    for s in shifts:
        if s == 0:
            continue
        a_shifted = np.roll(a, s)
        J = compute_J(a_shifted)
        if J < best_J:
            best_J = J
            best_a = a_shifted.copy()
    return best_a, best_J

import numpy as np
import time
from scipy.optimize import linprog

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k_idx = np.where(cond)[0]
    if len(k_idx) == 0:
        return np.ones(n) / n
    lam = cssv[k_idx[-1]] / (k_idx[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def try_all_shifts(a):

        J_s = compute_J(a_s)

def coord_descent_fast(a, max_sweeps, deadline, random_order=False, seed=0):
    """Fast coordinate descent with incremental convolution update."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if random_order and sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved = False
        for i in order:
            old = a[i]
            if old < 1e-12:
                candidates = [0.0, 1e-4, 1e-3]
            else:
                ratios = [0.0, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]
                candidates = list(set([old * r for r in ratios]))

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i

            best_val = old
            best_local = best_obj

            for x in candidates:
                x = max(0.0, x)
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-14:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        if s > 1e-10:
            current_obj = 2.0 * n * conv.max() / (s * s)
        else:
            current_obj = float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def lp_best_response(a, max_cp_iters=50, initial_peaks=30, deadline=None):
    """
    LP best response with cutting plane.
    Solves: max sum(g) s.t. (a*g)[k] <= M for k in S, g >= 0.
    Returns normalized g.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    c = fast_conv(a)
    M = c.max()
    if M < 1e-12:
        return a.copy()

    # Initial set of peak positions
    threshold = 0.90 * M
    S = set(np.where(c >= threshold)[0].tolist())
    if len(S) == 0:
        S = {int(np.argmax(c))}

    for cp_iter in range(max_cp_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        S_list = sorted(S)
        K = len(S_list)

        # Build constraint matrix: for each k in S_list, (a*g)[k] <= M
        # (a*g)[k] = sum_j a[k-j] * g[j] for valid j
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(S_list):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])

        if len(vals) == 0:
            return a.copy()

        A = coo_matrix((vals, (rows, cols)), shape=(K, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(K, M)

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                        method='highs', options={'time_limit': 15})
            if not r.success or r.x is None:
                return a.copy()
            g = np.maximum(r.x, 0)
        except Exception:
            return a.copy()

        if g.sum() < 1e-10:
            return a.copy()

        # Check all constraints for violations
        ag = np.convolve(a, g, mode='full')
        violations = np.where(ag > M + 1e-10)[0]
        if len(violations) > 0:
            # Add the most violated constraint
            worst = violations[np.argmax(ag[violations] - M)]
            S.add(int(worst))
            if len(S) > 2 * initial_peaks:
                # Cap the number of constraints
                S = set(np.argsort(c)[::-1][:initial_peaks])
                S.add(int(worst))
        else:
            break

    return g


def lp_sca_step(a, max_iters, deadline):
    """SCA-LP: iterate LP best response with line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = lp_best_response(a, max_cp_iters=30, initial_peaks=40,
                             deadline=min(time.monotonic() + 20, deadline))
        if g is None or g.sum() < 1e-10:
            no_improve += 5
            continue
        g = g / g.sum()

        found = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            a_try = (1.0 - frac) * a + frac * g
            a_try = np.maximum(a_try, 0)
            ts = a_try.sum()
            if ts < 1e-10:
                continue
            a_try /= ts
            J_try = compute_J(a_try)
            if J_try < best_obj - 1e-14:
                a = a_try
                best_obj = J_try
                best_a = a_try.copy()
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1

        if no_improve > 15:
            break

    return best_a, best_obj


def simulated_annealing(a, deadline, n_iter=5000, T_init=0.01, T_min=0.0001):
    """Simulated annealing with coordinate and block moves."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)

    best_a = a.copy()
    best_obj = compute_J(a)

    no_improve = 0

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        T = T_init * (T_min / T_init) ** (it / max(1, n_iter))

        # Choose move type
        move_type = rng.integers(3)
        a_new = a.copy()

            # Single coordinate perturbation

            factor = rng.uniform(0.3, 2.0)

            if total < 1e-12:
                continue
            frac = rng.uniform(0.0, 1.0)
            a_new[i] = total * frac
            a_new[j] = total * (1.0 - frac)
        else:
            # Block shift
            blen = int(rng.integers(2, max(3, n // 20) + 1))
            start = rng.integers(0, n - blen)
            shift = int(rng.integers(-n // 4, n // 4 + 1))
            block = a_new[start:start + blen].copy()
            a_new[start:start + blen] = 0
            new_start = start + shift
            if 0 <= new_start and new_start + blen <= n:
                a_new[new_start:new_start + blen] = np.maximum(
                    a_new[new_start:new_start + blen] + block, 0)

        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            continue
        a_new /= a_new_sum

        new_obj = compute_J(a_new)
        delta = new_obj - cur_obj

        if delta < 0:
            accept = True
        else:
            accept = np.log(rng.uniform()) > delta / max(T, 1e-15)

        if accept:
            a = a_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 2000:
            # Restart with perturbation of best
            a = best_a + rng.normal(0, 0.05, n)
            a = project_simplex(a)
            cur_obj = compute_J(a)
            no_improve = 0

    return best_a, best_obj


def generate_structured_inits(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for noise in [0.02, 0.05, 0.1, 0.2]:
            inits.append(project_simplex(p + rng.normal(0, noise, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Hanning window
    hann = 0.5 * (1.0 + np.cos(2 * np.pi * t))
    inits.append(project_simplex(hann + 0.01))

    # Blackman window
    black = (0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t))
    inits.append(project_simplex(black + 0.01))

    # Sine patterns
    for freq in [1, 2, 3, 5, 8]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t) + 0.01))
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t) + 0.01))

    # Power-law tapers
    for alpha in [0.3, 0.5, 1.0, 2.0, 3.0]:
        inits.append(project_simplex((t + 0.01) ** alpha + 0.01))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha + 0.01))
        # Symmetric taper (high in middle, low at edges)
        sym = np.minimum(t, 1 - t) + 0.01
        inits.append(project_simplex(sym ** alpha + 0.01))

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.02))

    # Sparse patterns (random subset)
    for density in [0.3, 0.5, 0.7]:
        mask = (rng.random(n) < density).astype(float)
        if mask.sum() > 0:
            inits.append(project_simplex(mask + 0.01))

    # Smooth random
    ksize = max(5, n // 12)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    return inits


def optimize_single(a, deadline, verbose=False):
    """Full pipeline: CD -> LP -> CD -> shift -> SA -> final CD."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            return
        a_new = a_new / s
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 1: Quick CD
    t1 = time.monotonic() + 0.20 * remaining
    a_cd = coord_descent_fast(a, max_sweeps=5000, deadline=t1, random_order=True, seed=42)
    try_update(a_cd)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 2: LP-based optimization
    t2 = time.monotonic() + 0.35 * remaining
    a_lp, obj_lp = lp_sca_step(a, max_iters=50, deadline=t2)
    if obj_lp < best_obj:
        best_a = a_lp.copy()
        best_obj = obj_lp
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Try shifts
    a_shifted, obj_shifted = try_all_shifts(a)
    try_update(a_shifted)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 3: Simulated annealing
    t3 = time.monotonic() + 0.20 * remaining
    a_sa, obj_sa = simulated_annealing(a, deadline=t3, n_iter=3000,
                                       T_init=0.005, T_min=0.0001)
    if obj_sa < best_obj:
        best_a = a_sa.copy()
        best_obj = obj_sa
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 4: Final CD polish
    a_final = coord_descent_fast(a, max_sweeps=10000, deadline=deadline - 5,
                                 random_order=True, seed=99)
    try_update(a_final)

    # Final shift
    a_shifted, obj_shifted = try_all_shifts(best_a)
    try_update(a_shifted)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses circular shift exploration, cutting-plane LP, structured initialization,
    simulated annealing, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    # === CRITICAL: Try all shifts of parent immediately ===
    if time.monotonic() < deadline:
        parent_shifted, J_shifted = try_all_shifts(parent)
        update_best(parent_shifted)
        if J_shifted < best_score:
            parent = parent_shifted.copy()
            print(f"  Shift improved parent to J={J_shifted:.8f}", flush=True)

    n_parent = len(parent)

    # === PHASE 1: Multi-init optimization at parent length ===
    phase1_deadline = t_start + 450.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-init at n={n_parent} (budget {phase1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_structured_inits(n_parent, rng, parent)
    # Deduplicate and limit
    seen = set()
    unique_inits = []
    for init in inits:
        key = tuple(np.round(init, 4).tolist())
        if key not in seen:
            seen.add(key)
            unique_inits.append(init)
    inits = unique_inits[:15]

    n_init = len(inits)
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, n_init) - 2
    per_init_time = max(10, min(per_init_time, 60))

    for idx in range(n_init):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < per_init_time + 5:
            break
        alloc = min(per_init_time, remaining_check - 5)
        if alloc < 10:
            break

        a0 = inits[idx]
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt, obj_opt = optimize_single(a0, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  Init {idx}/{n_init-1}: J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Multi-length exploration ===
    phase2_deadline = t_start + 900.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length (budget {phase2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidate_ns = sorted(set([
        n_parent - 128, n_parent + 128,
        n_parent - 256, n_parent + 256,
        n_parent // 2, n_parent * 2,
        384, 512, 640, 768, 896, 1024, 1280, 1536
    ]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048 and n != n_parent]
    # Prioritize near parent length
    candidate_ns.sort(key=lambda n: abs(n - n_parent))
    candidate_ns = candidate_ns[:6]

    for new_n in candidate_ns:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 60:
            break
        alloc = min(80.0, remaining_check - 20)
        if alloc < 40:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        base = best_seq if best_seq is not None else parent
        a_ext = resample_seq(base, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = optimize_single(a_ext, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive optimization on best ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive opt (budget {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        n_best = len(best_seq)

        # Pass 1: Long LP
        t_p1 = time.monotonic() + 0.25 * remaining
        print(f"  LP pass ({t_p1 - time.monotonic():.0f}s)...", flush=True)
        a_lp, obj_lp = lp_sca_step(best_seq, max_iters=80, deadline=t_p1)
        update_best(a_lp)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Pass 2: Simulated annealing
            t_p2 = time.monotonic() + 0.25 * remaining
            print(f"  SA pass ({t_p2 - time.monotonic():.0f}s)...", flush=True)
            a_sa, obj_sa = simulated_annealing(best_seq, deadline=t_p2,
                                               n_iter=8000, T_init=0.003, T_min=0.00005)
            update_best(a_sa)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Pass 3: CD + shift
            t_p3 = time.monotonic() + 0.20 * remaining
            print(f"  CD pass ({t_p3 - time.monotonic():.0f}s)...", flush=True)
            a_cd = coord_descent_fast(best_seq, max_sweeps=20000, deadline=t_p3,
                                      random_order=True, seed=777)
            update_best(a_cd)
            a_shifted, obj_sh = try_all_shifts(best_seq)
            update_best(a_shifted)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Pass 4: Another LP round on the improved sequence
            t_p4 = time.monotonic() + 0.25 * remaining
            print(f"  Final LP ({t_p4 - time.monotonic():.0f}s)...", flush=True)
            a_final_lp, obj_final = lp_sca_step(best_seq, max_iters=60, deadline=t_p4)
            update_best(a_final_lp)

        # Final: try shifts and one more CD
        if time.monotonic() < deadline - 10:
            a_shifted, obj_sh = try_all_shifts(best_seq)
            update_best(a_shifted)
            if time.monotonic() < deadline - 5:
                a_final_cd = coord_descent_fast(best_seq, max_sweeps=5000,
                                                deadline=deadline - 3,
                                                random_order=True, seed=4242)
                update_best(a_final_cd)
                a_shifted2, obj_sh2 = try_all_shifts(best_seq)
                update_best(a_shifted2)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
