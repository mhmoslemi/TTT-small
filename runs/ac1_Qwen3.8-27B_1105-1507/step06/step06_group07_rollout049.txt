"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses peak-shaving gradient descent, LP-based direction finding,
Adam in log-space with adaptive temperature, and NLP refinement.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def conv_gradient(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j1 = k - i
        j2 = i - k
        if 0 <= j1 < n:
            grad[i] += a[j1]
        if 0 <= j2 < n:
            grad[i] += a[j2]
    # Fix double counting when j1 == j2 (i.e., k = 2i)
    # Actually, the formula grad[i] = a[k-i] + a[i-k] already handles this correctly
    # because if k-i == i-k, then k=2i, and we add a[i] twice, which is correct
    # since d/d(a[i]) [a[i]*a[i]] = 2*a[i]
    return grad


def conv_gradient_fast(a, active_k):
    """Compute sum of gradients for active peak positions using FFT."""
    n = len(a)
    # The gradient of conv(a,a)[k] w.r.t. a is the cross-correlation of a with itself shifted by k
    # Sum over active k: sum_k grad_k = sum_k (a shifted and summed)
    # This is equivalent to: for each i, grad_sum[i] = sum_k (a[k-i] + a[i-k]) for valid k
    # = sum_k a[k-i] + sum_k a[i-k]
    # = (number of valid k for each i) terms
    # Simplified: grad_sum[i] = sum_{k in active} a[k-i] + sum_{k in active} a[i-k]
    #             = sum_{j: k-i=j, k in active} a[j] + sum_{j: i-k=j, k in active} a[j]
    #             = sum_{j: j+i in active} a[j] + sum_{j: i-j in active} a[j]
    
    # More efficiently: create indicator vector w[k] = 1 if k in active, else 0
    # Then grad_sum = 2 * cross_correlation(w, a) (with some edge effects)
    
    w = np.zeros(2 * n - 1)
    for k in active_k:
        w[k] = 1.0
    
    # cross_correlation(w, a)[i] = sum_j w[j+i] * a[j]
    size = 1
    while size < 2 * len(w):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    grad_sum = 2.0 * cc[:n]
    return grad_sum


def smooth_obj_grad(a, beta):
    """Compute smoothed objective (logsumexp of beta*conv) and gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj


def adam_log_opt(a, max_iters=20000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995):
    """Adam in log-space with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_centered = x - np.max(x)
        a_cur = np.exp(x_centered)
        a_cur = a_cur / np.sum(a_cur)
        
        obj_sm, grad_a = smooth_obj_grad(a_cur, beta)
        grad_x = grad_a * a_cur
        
        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 200 == 0:
            x -= np.mean(x)
        
        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

                    a = a_try.copy()

    return best_a


def peak_shave(a, time_budget=30.0, K=5):
    """
    Peak shaving: directly reduce the top peaks by adjusting a.
    For each top peak, compute the gradient and take a step that reduces it.
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    lr = 0.1
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_val = np.max(conv)
        
        # Find top K peaks
        top_k = np.argsort(conv)[::-1][:K]
        
        # Compute combined gradient (push to reduce all top peaks)
        grad_total = np.zeros(n)
        for k in top_k:
            # Gradient of conv[k] w.r.t. a
            g_k = np.zeros(n)
            for i in range(n):
                j1 = k - i
                j2 = i - k
                if 0 <= j1 < n:
                    g_k[i] += a[j1]
                if 0 <= j2 < n:
                    g_k[i] += a[j2]
            # Weight by how close to max
            weight = 1.0 + (conv[k] / max_val)
            grad_total += weight * g_k
        
        # We want to DECREASE the peaks, so move in -grad direction
        # But we need to maintain non-negativity and sum
        # Use a "push-pull" approach: reduce a where grad is large, increase where grad is small
        grad_mean = np.mean(grad_total)
        direction = -grad_total + grad_mean  # zero-mean direction
        
        # Scale direction
        step = lr * direction / (np.max(np.abs(direction)) + 1e-10)
        
        a_try = a + step
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            lr *= 0.5
            continue
        a_try = a_try / s_try
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-10:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            # Decay learning rate slowly
            lr *= 0.999
        else:
            lr *= 0.8
        
        if lr < 1e-6:
            break
    
    return best_a
