"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. KL mirror descent (multiplicative updates) with annealed soft-max
2. Adam optimizer with log-sum-exp gradient
3. Subgradient method with adaptive step size
4. LP mirror descent with Anderson acceleration for final polish
5. Multi-length exploration with frequency-domain initialization
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Compute linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def evaluate_sequence_safe(a):
    """Safely evaluate a sequence."""
    try:
        return evaluate_sequence([float(x) for x in a])
    except Exception:
        return float('inf')


def soft_max_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*b_k)) where b = conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    # Threshold to skip negligible contributions
    threshold = max(1e-14, p.max() * 1e-10)
    active = np.where(p > threshold)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def mirror_desc_kl(a, deadline, max_iters=5000, eta_start=1.0, eta_end=0.01,
                   tau_start=5.0, tau_end=500.0):
    """KL mirror descent (multiplicative updates) with annealed soft-max."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    best_M = fast_conv(a).max()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break
        frac = it / max_iters
        # Anneal temperature (higher tau = sharper approximation of max)
        tau = tau_start * (tau_end / tau_start) ** frac
        # Anneal step size
        eta = eta_start * (eta_end / eta_start) ** frac
        
        grad, p, b, b_max = soft_max_grad(a, tau)
        
        if b_max < best_M:
            best_M = b_max
            best_J = compute_J(a)
            best_a = a.copy()
        
        # Multiplicative update: a_new = a * exp(-eta * grad) / Z
        log_update = -eta * grad
        log_update = np.clip(log_update, -100, 100)
        a_new = a * np.exp(log_update)
        a_new = project_simplex(a_new)
        
        a = a_new
    
    return best_a


def adam_smooth(a, deadline, max_iters=10000, tau_start=5.0, tau_end=500.0,
                lr=0.05):
    """Adam optimizer with log-sum-exp smooth max."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** frac
        
        grad, p, b, b_max = soft_max_grad(a, tau)
        
        if b_max < best_M:
            best_M = b_max
            best_a = a.copy()
        
        # Adam update
        t = it
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr_t = lr / (1.0 + 0.001 * it)
        a_candidate = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_candidate)
    
    return best_a


def subgradient_descent(a, deadline, max_iters=5000):
    """Subgradient method: use exact subgradient at argmax."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break
        b = fast_conv(a)
        k_star = int(b.argmax())
        b_max = b[k_star]
        
        if b_max < best_M:
            best_M = b_max
            best_a = a.copy()
        
        # Subgradient: sg_i = 2 * a[k_star - i] for valid i
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        sg = np.zeros(n)
        sg[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
        
        # Adaptive step size (decreasing)
        eta = 1.0 / (1.0 + 0.01 * it)
        
        a_candidate = a - eta * sg
        a = project_simplex(a_candidate)
    
    return best_a


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: flatten the power spectrum."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress dynamic range
        mag_norm = mag / (mag.max() + 1e-30)
        power = 0.5 + 0.5 * beta
        mag_new = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def resample_seq(src, n):
    """Resample a sequence to length n."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def build_A_values(n, a):
    """Build sparse matrix for LP: (a*g)(k) = sum_j a[k-j]*g[j]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    rows, cols, vals = build_A_values(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_solve(history_x, history_Fx, m):
    """Anderson mixing: find optimal convex combination."""
    residuals = []
    for i in range(m):
        residuals.append(history_x[i] - history_Fx[i])
    residuals = np.array(residuals)
    G = residuals @ residuals.T
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -1.0, 1.0)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def run_lp_anderson(a, deadline, max_iters=100, anderson_window=5,
                    t_base=0.3, min_t=0.01):
    """LP mirror descent with Anderson acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    hist_x = []
    hist_Fx = []
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        if it < 5:
            t = t_base
        else:
            t = max(min_t, t_base / (1.0 + 0.03 * it))
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        if len(hist_x) >= 3:
            a_anderson = anderson_solve(hist_x, hist_Fx, len(hist_x))
            sa = a_anderson.sum()
            if sa > 1e-10:
                a_anderson = a_anderson / sa
            M_and = fast_conv(a_anderson).max()
            M_new = fast_conv(a_new).max()
            if M_and <= M_new * 1.02:
                a = a_anderson
            else:
                a = a_new
        else:
            a = a_new
    return best_a


def run_lp_plain(a, deadline, max_iters=80):
    """Plain LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        improved = False
        for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break
    return best_a
