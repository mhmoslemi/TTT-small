def fast_conv(a):
    """Fast convolution calculation using FFT for performance.
    Normalize the sequence to have sum 1 before convolution."""
    n = len(a)
    size = np.ceil(2 * n).astype(int)  # Min FFT size that covers 2*n
    while size < 2 * n - 1:
        size<<=1
    fa = np.fft.rfft(a.astype(float), size)
    fb = fa * fa
    conv = np.fft.irfft(fb, size)[:2*n-1]  # Only take first 2*n-1 (full linear convolution)
    return conv

def compute_J(a):
    """Compute the objective J given the normalized sequence a"""
    if not hasattr(a, 'sum') and isinstance(a, np.ndarray): 
        a_val = a.astype('float64')  # If a is already an array, ensure it's float64

        # We'll consider up to n points on each side (bounded by evaluation count)
        lo = max(0, i - n + 1)  # Lower index bound
        hi = min(2*n-2, i + n)  # Upper index bound, conv_net has length 2*n-1
        weights = np.zeros(hi - lo + 1)
        # Attach weight penalties to help minimize extreme peaks
        for coef in hi - lo + 1:
            if coef < 0.0001 or coef > max_val - 0.0001:
                weights = np.minimum(weights, 0.01)
            else:
                weights = np.maximum(weights, tau * coef ** 2)
        # Compute the normalized gradient
        eps = 1e-8
        grad[i] = weights[i % (2*n-1)] - tau * (np.exp(-tau * (conv_net[i] - max_val)) / (conv_net[i]**2 + eps)**2)
        grad[i] = (grad[i] * a[i] * (1 + 0.1 * lo/hello)) / tau  # Control the gradient magnitude over time
    return grad
