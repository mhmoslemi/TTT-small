    while size < 2 * n - 1:

    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    """Build sparse matrix A such that (A @ x)_k = sum_j fixed[k-j] * x[j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(fixed[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    return None

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, fixed, constraint_indices=None):
    rows = []
    cols = []
    vals = []
    if constraint_indices is None:
        constraint_indices = range(2 * n - 1)
    for idx, k in enumerate(constraint_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(fixed[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))
    return A


def solve_best_response(fixed, M, n, constraint_indices=None):
    """Solve: max sum(x) s.t. (fixed*x)_k <= M for k in constraint_indices, x >= 0."""
    if constraint_indices is None:
        constraint_indices = list(range(2 * n - 1))
    if len(constraint_indices) == 0:
        return np.ones(n)
    A = build_A_sparse(n, fixed, constraint_indices)
    c = -np.ones(n)
    b_ub = np.full(len(constraint_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 5})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def select_constraints(b, M, frac=0.7):
    """Select constraints where b_k > frac * M (the 'active' ones)."""
    threshold = frac * M
    idx = np.where(b > threshold)[0]
    if len(idx) < 10:
        idx = np.where(b > 0.5 * M)[0]
    if len(idx) < 5:
        idx = np.argsort(b)[-int(0.3 * len(b)):]
    return list(idx)


def smooth_grad(a, tau):
    """Compute smooth approximation of gradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, b, b_max


def run_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=300.0,
               lr_init=0.4, lr_end=0.003):
    """Nesterov accelerated smooth max gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def anderson_mix(hist_x, hist_Fx, m):
    """Anderson mixing: find alpha minimizing ||sum alpha_i * (x_i - F(x_i))||^2 s.t. sum alpha=1."""
    n = len(hist_x[0])
    residuals = np.array([hist_x[i] - hist_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-10 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
    except np.linalg.LinAlgError:
        return np.maximum(np.mean(hist_Fx, axis=0), 0)
    denom = ones @ G_inv_ones
    if abs(denom) < 1e-15:
        alpha = ones / m
    else:
        alpha = G_inv_ones / denom
    alpha = np.clip(alpha, -0.5, 2.0)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(n)
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    return np.maximum(mixed, 0)


def parabolic_line_search(a, g, n, current_M):
    """Parabolic interpolation line search on (1-t)*a + t*g."""
    a_n = a / a.sum()
    g_n = g / g.sum()
    # Evaluate at t=0, 0.3, 0.6, 1.0
    ts = [0.0, 0.3, 0.6, 1.0]
    Ms = []
    for t in ts:
        cand = np.maximum((1 - t) * a_n + t * g_n, 0)
        s = cand.sum()
        if s < 1e-10:
            Ms.append(float('inf'))
        else:
            Ms.append(fast_conv(cand / s).max())
    # Find best from these
    best_i = int(np.argmin(Ms))
    best_t = ts[best_i]
    best_val = Ms[best_i]
    # Refine with bisection around the best
    if best_i == 0:
        lo, hi = 0.0, ts[1]
    elif best_i == 1:
        lo, hi = ts[0], ts[2]
    elif best_i == 2:
        lo, hi = ts[1], ts[3]
    else:
        lo, hi = ts[2], 1.0
    for _ in range(10):
        mid = (lo + hi) / 2
        cand = np.maximum((1 - mid) * a_n + mid * g_n, 0)
        s = cand.sum()
        if s < 1e-10:
            hi = mid
            continue
        val = fast_conv(cand / s).max()
        if val < best_val:
            best_val = val
            best_t = mid
        if val <= Ms[best_i] * (1 + 0.01):
            if mid > best_t:
                lo = mid
            else:
                hi = mid
        else:
            if mid > best_t:
                hi = mid
            else:
                lo = mid
    return best_t, best_val

def gs_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: flatten the power spectrum."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        # Compress: raise to power < 1 to flatten
        power = 0.3 + 0.4 * beta
        mag_new = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_cand = np.maximum(a_new, 0)
        sc = a_cand.sum()
        if sc < 1e-15:
            break
        a_cand /= sc
        M_c = fast_conv(a_cand).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_cand.copy()
        a = a_cand
    return best_a


def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.001))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.001))
    # Monotone decreasing
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(1.0 - alpha * t, 0.01))
    # Monotone increasing
    for alpha in [0.5, 1.0, 2.0]:
        inits.append(np.maximum(alpha * t + 0.1, 0.01))
    # Symmetric (bell)
    for sigma in [0.1, 0.2, 0.3, 0.4]:
        inits.append(np.exp(-((t - 0.5) ** 2) / (2 * sigma ** 2)) + 0.01)
    # Uniform
    inits.append(np.ones(n) * 0.5 + 0.5)
    # Cosine
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    # Two bumps
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.01)
    # Smooth random
    k = max(5, n // 16)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.02, 0.01))
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_spectral(a_rand, n_iters=80, beta=0.85))
    # GS from parent
    if parent is not None:
        p = resample(parent, n)
        inits.append(gs_spectral(p, n_iters=100, beta=0.9))
    return inits

        inits = make_inits(n, rng, p)
        best_n_a = None
        best_n_M = float('inf')
        n_iters = max(100, int(5000 * (512.0 / n)))
        for idx, a0 in enumerate(inits):
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 /= a0.sum()
            sub_dl = min(dl, time.monotonic() + (dl - time.monotonic()) * (1.0 / (1.0 + idx * 0.4)))
            a = run_smooth(a0, sub_dl, max_iters=n_iters,
                           tau_start=1.0, tau_end=200.0,
                           lr_init=0.35, lr_end=0.003)
            M = fast_conv(a).max()
            if M < best_n_M:
                best_n_M = M
                best_n_a = a.copy()
        if best_n_a is not None:
            J = compute_J(best_n_a)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_n_a, f"phase1 n={n}")
            best_per_n[n] = best_n_a.copy()

    print(f"Phase 1 best: J={tracker.score:.6f}", flush=True)

    # Phase 2: LP refinement on top candidates
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if best_per_n:
        sorted_ns = sorted(best_per_n.keys(), key=lambda x: compute_J(best_per_n[x]))
        # Focus on top 3 lengths
        for n_try in sorted_ns[:3]:
            if time.monotonic() > deadline - 60:
                break
            a_init = best_per_n[n_try]
            n = a_init.size
            print(f"  LP refining n={n}...", flush=True)
            lp_dl = min(time.monotonic() + 200, deadline - 60)
            a_lp = run_lp_refine(a_init, lp_dl, max_iters=300, anderson_window=5)
            tracker.update(a_lp, f"lp n={n}")

    print(f"Phase 2 best: J={tracker.score:.6f}", flush=True)

    # Phase 3: Two-sided LP on best
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Two-Sided LP", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.seq is not None and time.monotonic() < deadline - 60:
        a_best = tracker.seq.copy()
        n = a_best.size
        print(f"  Two-sided LP on n={n}...", flush=True)
        ts_dl = min(time.monotonic() + 300, deadline - 30)
        a_ts = run_two_sided_lp(a_best, ts_dl, max_iters=200)
        tracker.update(a_ts, "two-sided")

        # Additional LP refinement after two-sided
        if time.monotonic() < deadline - 30:
            a_lp2 = run_lp_refine(a_ts, min(time.monotonic() + 150, deadline - 10),
                                  max_iters=200, anderson_window=6)
            tracker.update(a_lp2, "lp after two-sided")

    print(f"Phase 3 best: J={tracker.score:.6f}", flush=True)

    # Phase 4: Length exploration near best
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.seq is not None and time.monotonic() < deadline - 40:
        n_best = tracker.seq.size
        for delta in [-128, -64, 64, 128, 256, 512]:
            if time.monotonic() > deadline - 20:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 2560:
                continue
            a_ext = resample(tracker.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            # Quick smooth
            a_ext = run_smooth(a_ext, min(time.monotonic() + 30, deadline - 15),
                               max_iters=500, tau_start=3, tau_end=100,
                               lr_init=0.2, lr_end=0.01)
            tracker.update(a_ext, f"len n={n_try}")
            # Quick LP
            if time.monotonic() < deadline - 15:
                a_lp = run_lp_refine(a_ext, min(time.monotonic() + 40, deadline - 5),
                                     max_iters=100, anderson_window=4)
                tracker.update(a_lp, f"lp n={n_try}")

    print(f"Phase 4 best: J={tracker.score:.6f}", flush=True)

    # Phase 5: Final polish
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.seq is not None and time.monotonic() < deadline - 5:
        a_final = run_lp_refine(tracker.seq, deadline - 2, max_iters=300, anderson_window=6)
        tracker.update(a_final, "final polish")

        if time.monotonic() < deadline - 3:
            a_final2 = run_two_sided_lp(a_final, deadline - 1, max_iters=100)
            tracker.update(a_final2, "final two-sided")

    # Fallback
    if tracker.seq is None:
        tracker.seq = np.ones(256)
        tracker.score = compute_J(tracker.seq)

    total_time = time.monotonic() - tracker.t0
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.score:.8f} n={tracker.seq.size} time={total_time:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in tracker.seq]
