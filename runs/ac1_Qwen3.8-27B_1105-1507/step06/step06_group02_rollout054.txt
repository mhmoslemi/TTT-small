import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a, b=None):
    """Compute linear convolution a*b (or a*a if b is None) using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute the objective J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where (A @ g)_k = (a*g)(k) for k=0..2n-2."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        rows.extend([k] * (j_max - j_min + 1))
        cols.extend(js.tolist())
        vals.extend((a[js]).tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def build_A_subset(n, a, active_k):
    """Build sparse constraint matrix for only the active constraints."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        rows.extend([idx] * (j_max - j_min + 1))
        cols.extend(js.tolist())
        vals.extend((a[js]).tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
    return A


def lp_solve(A, b_ub, n):
    """Solve max sum(g) s.t. A@g <= b_ub, g >= 0."""
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 10})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_best_response_full(a, M):
    """Full LP best response: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    b_ub = np.full(2 * n - 1, M)
    return lp_solve(A, b_ub, n)


def lp_best_response_cg(a, M, max_rounds=20, init_count=100):
    """LP best response with constraint generation (faster for large n)."""
    n = len(a)
    b_aa = fast_conv(a)
    sorted_idx = np.argsort(-b_aa)

    # Start with the most active constraints
    active_k = sorted_idx[:min(init_count, 2 * n - 1)]
    g = None

    for _ in range(max_rounds):
        A_sub = build_A_subset(n, a, active_k)
        b_sub = np.full(len(active_k), M)
        g = lp_solve(A_sub, b_sub, n)
        if g is None or g.sum() < 1e-15:
            return g
        # Check all constraints
        ag = fast_conv(a, g)
        violations = np.where(ag > M * (1 + 1e-10) + 1e-12)[0]
        if len(violations) == 0:
            return g
        # Add violated constraints (up to 50 new ones per round)
        new_k = violations[:50]
        active_k = np.unique(np.concatenate([active_k, new_k]))
        # Cap total constraints
        if len(active_k) > 2 * n - 1:
            active_k = sorted(np.argsort(-ag)[:2 * n - 1])

    # Final: solve with full constraint set if we keep finding violations
    A_full = build_A_sparse(n, a)
    b_full = np.full(2 * n - 1, M)
    return lp_solve(A_full, b_full, n)


def run_lp_mirror(a, deadline, max_iters=5000, use_cg=True, cg_init=100):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    t_prev = 0.3

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break

        # Compute best response
        if use_cg:
            g = lp_best_response_cg(a, M, max_rounds=10, init_count=cg_init)
        else:
            g = lp_best_response_full(a, M)

        if g is None or g.sum() < 1e-12:
            break
        g = g / g.sum()

        # Adaptive step size with line search
        dist = np.linalg.norm(g - a) / max(1, np.sqrt(n))
        if it < 3:
            t_candidates = [0.4, 0.25, 0.15, 0.1]
        elif dist > 0.5:
            t_candidates = [0.5, 0.35, 0.25, 0.15]
        elif dist > 0.2:
            t_candidates = [0.3, 0.2, 0.12, 0.07]
        else:
            t_candidates = [0.15, 0.1, 0.05, 0.02]

        best_t = t_candidates[-1]
        best_M_t = float('inf')
        best_a_t = None
        for t in t_candidates:
            a_t = np.maximum((1 - t) * a + t * g, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = fast_conv(a_t).max()
            if M_t < best_M_t:
                best_M_t = M_t
                best_t = t
                best_a_t = a_t

        if best_a_t is not None:
            a = best_a_t
            t_prev = best_t
        else:
            break

    return best_a


def run_lp_mirror_nesterov(a, deadline, max_iters=3000, use_cg=True, cg_init=100):
    """LP mirror descent with Nesterov momentum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    y = a.copy()  # momentum variable
    t_k = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(y)
        M = float(b.max())

        if use_cg:
            g = lp_best_response_cg(y, M, max_rounds=8, init_count=cg_init)
        else:
            g = lp_best_response_full(y, M)

        if g is None or g.sum() < 1e-12:
            break
        g = g / g.sum()

        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        beta = (t_k - 1) / t_new
        a_new = np.maximum((1 - 0.3) * y + 0.3 * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = y.copy()
        else:
            a_new = a_new / s
        y_new = np.maximum(a_new + beta * (a_new - a), 0)
        sy = y_new.sum()
        if sy < 1e-10:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new
        t_k = t_new

        M_new = fast_conv(a).max()
        if M_new < best_M * (1 - 1e-12):
            best_M = M_new
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 60:
                break

    return best_a


def subgradient_polish(a, deadline, max_iters=200, lr_init=0.05):
    """Projected subgradient method using exact active set of max autocorrelation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break

        # Find active set (lags where b[k] is close to M)
        tol = max(1e-10, M * 0.01)
        active = np.where(b >= M - tol)[0]

        # Subgradient: grad[j] = 2 * sum_{k in active} w_k * a[k-j]
        # Use uniform weights over active set
        w = 1.0 / len(active)
        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            js = np.arange(j_min, j_max + 1)
            grad[js] += w * a[k - js]
        grad *= 2.0

        # Adaptive learning rate (Polyak-like)
        lr = lr_init / (1 + 0.02 * it)
        a_new = np.maximum(a - lr * grad, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        # Also try a small step in the direction of reducing the max
        a = a_new

    return best_a


def gs_flat_spectral(init, n_iters=200, beta=0.8):
    """Gerchberg-Saxton: flatten power spectrum while maintaining non-negativity."""
    a = np.maximum(np.asarray(init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        # Compress dynamic range
        power = 0.3 + 0.4 * beta
        mag_new = (mag_norm ** power) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate

    return best_a


def smooth_max_grad(a, tau):
    """Compute gradient of smooth max of autocorrelation."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        grad[js] += 2.0 * p[k] * a[k - js]
    return grad, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=0.5, tau_end=100.0,
                        lr_init=0.3, lr_end=0.005):
    """Nesterov-accelerated gradient descent on smooth max autocorrelation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    t_k = 1.0

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_max_grad(y, tau)

        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        a_new = np.maximum(y - lr * grad, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        beta = (t_k - 1) / t_new
        y = np.maximum(a_new + beta * (a_new - a), 0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y = y / sy

        a = a_new
        t_k = t_new

    return best_a


def resample_seq(src, n):
    """Resample a sequence to a new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_asymmetric_inits(n, rng):
    """Create initializations with asymmetric (ramp-like) structure."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Linear ramp
    inits.append(np.maximum(0.3 + 0.7 * t, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * (1 - t), 0.01))

    # Quadratic ramp
    for alpha in [1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.2 + 0.8 * t**alpha, 0.01))
        inits.append(np.maximum(0.2 + 0.8 * (1 - t)**alpha, 0.01))

    # Exponential ramp
    for rate in [2, 4, 6]:
        inits.append(np.maximum(0.1 + 0.9 * np.exp(-rate * t), 0.01))
        inits.append(np.maximum(0.1 + 0.9 * np.exp(-rate * (1 - t)), 0.01))

    # Bump + tail
    inits.append(np.maximum(np.exp(-3 * (t - 0.3)**2) + 0.3 * t, 0.01))
    inits.append(np.maximum(np.exp(-3 * (t - 0.7)**2) + 0.3 * (1 - t), 0.01))

    # Sine-based asymmetric
    inits.append(np.maximum(0.4 + 0.4 * np.sin(np.pi * t), 0.01))
    inits.append(np.maximum(0.4 + 0.4 * np.cos(np.pi * t), 0.01))

    # Random smooth asymmetric
    k = max(3, n // 15)
    kern = np.ones(k) / k
    raw = rng.uniform(0.2, 1.0, n) * (1 + t)
    inits.append(np.maximum(np.convolve(raw, kern, 'same') + 0.05, 0.01))

    return inits

        J = compute_J(a_lp)

class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):

            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            s = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** NEW BEST: J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)

        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)
        print("No parent, using uniform init", flush=True)

    # === PHASE 1: Multi-scale optimization to get a strong starting point ===
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-scale optimization", flush=True)
    print(f"{'='*60}", flush=True)

    target_n = parent.size if parent.size >= 512 else 1024
    # Allocate ~30% of time to multi-scale
    ms_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.3
    a_multiscale = multi_scale_optimize(target_n, ms_deadline, seed=seed)
    if a_multiscale is not None:
        best.update(a_multiscale)
    best.report("Phase 1 (multi-scale)")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Diverse smooth optimization at target length ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Diverse smooth optimization", flush=True)
    print(f"{'='*60}", flush=True)

    start_a = best.seq.copy() if best.seq is not None else a_multiscale
    n = start_a.size

    # Generate diverse initializations
    inits = make_asymmetric_inits(n, rng)
    # Add parent-resampled variants
    inits.append(resample_seq(start_a, n))
    inits.append(np.maximum(resample_seq(start_a, n) + rng.normal(0, 0.02, n), 0.01))
    inits.append(np.maximum(resample_seq(start_a, n) * (1 + 0.15 * rng.normal(0, 1, n)), 0.01))
    # GS from a couple initializations
    inits.append(gs_flat_spectral(inits[0], n_iters=150, beta=0.85))
    inits.append(gs_flat_spectral(start_a, n_iters=150, beta=0.9))

    p2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.2
    best_n = start_a.copy()
    best_M_n = fast_conv(start_a).max()

    for idx, a0 in enumerate(inits):
        if time.monotonic() > p2_deadline:
            break
        a0 = np.maximum(a0.astype(np.float64), 1e-8)
        a0 = a0 / a0.sum()
        # Allocate decreasing time to later inits
        frac = 1.0 / (1.0 + idx * 0.5)
        sub_dl = time.monotonic() + min(20 * frac, p2_deadline - time.monotonic())
        a_opt = run_nesterov_smooth(a0, sub_dl, max_iters=2000,
                                    tau_start=0.5, tau_end=80,
                                    lr_init=0.25, lr_end=0.005)
        M_opt = fast_conv(a_opt).max()
        if M_opt < best_M_n:
            best_M_n = M_opt
            best_n = a_opt.copy()

    best.update(best_n)
    best.report("Phase 2 (smooth diverse)")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Main LP mirror descent (the workhorse) ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP mirror descent (main optimization)", flush=True)
    print(f"{'='*60}", flush=True)

    a_main = best_n.copy() if best_M_n < fast_conv(start_a).max() else start_a.copy()
    # Ensure we start from the best available
    M_check = fast_conv(a_main).max()
    M_start = fast_conv(start_a).max()
    if M_start < M_check:
        a_main = start_a.copy()

    use_cg = n > 128
    cg_init = max(100, min(300, int(0.15 * (2 * n - 1))))

    # Run LP mirror descent with adaptive step sizes
    p3_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.5
    a_lp = run_lp_mirror(a_main, p3_deadline, max_iters=10000,
                         use_cg=use_cg, cg_init=cg_init)
    best.update(a_lp)

    # Nesterov LP variant for a second pass
    if time.monotonic() < deadline - 60:
        a_lp2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.4
        a_lp2 = run_lp_mirror_nesterov(a_lp, a_lp2_deadline, max_iters=5000,
                                       use_cg=use_cg, cg_init=cg_init)
        best.update(a_lp2)

    best.report("Phase 3 (LP mirror)")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Subgradient polishing ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Subgradient polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 10:
        a_sg = subgradient_polish(best.seq, deadline - 10, max_iters=100, lr_init=0.03)
        best.update(a_sg)

        # Another LP pass after subgradient
        if time.monotonic() < deadline - 15:
            a_lp3 = run_lp_mirror(a_sg, deadline - 5, max_iters=50,
                                  use_cg=use_cg, cg_init=cg_init)
            best.update(a_lp3)

    best.report("Phase 4 (subgradient)")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Length exploration ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 40:
        n_best = best.seq.size
        # Try a few nearby lengths
        for delta in [32, -32, 64, -64, 128, -128]:
            if time.monotonic() > deadline - 20:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2048:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()
            # Quick smooth
            a_ext = run_nesterov_smooth(a_ext, min(time.monotonic() + 8, deadline - 15),
                                        max_iters=300, tau_start=3, tau_end=50,
                                        lr_init=0.15, lr_end=0.01)
            # Quick LP
            if time.monotonic() < deadline - 10:
                a_ext_lp = run_lp_mirror(a_ext, min(time.monotonic() + 15, deadline - 5),
                                         max_iters=100, use_cg=n_try > 128,
                                         cg_init=max(50, n_try // 4))
                best.update(a_ext_lp)
            best.update(a_ext)

    best.report("Phase 5 (lengths)")
    print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === PHASE 6: Final long LP run if time remains ===
    if best.seq is not None and time.monotonic() < deadline - 10:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 6: Final LP extension", flush=True)
        print(f"{'='*60}", flush=True)
        n_final = best.seq.size
        use_cg_f = n_final > 128
        cg_init_f = max(100, min(300, int(0.15 * (2 * n_final - 1))))
        a_final = run_lp_mirror(best.seq, deadline, max_iters=5000,
                                use_cg=use_cg_f, cg_init=cg_init_f)
        best.update(a_final)
        best.report("Phase 6 (final LP)")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} "
          f"time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best.seq]
