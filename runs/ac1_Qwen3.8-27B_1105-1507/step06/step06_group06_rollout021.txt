"""
Convolution peak minimization using log-sum-exp smoothed subgradient descent with
Nesterov momentum, spectral domain flattening, and LP polishing.
The objective 2*n*max(conv(a,a))/(sum(a))^2 is scale-invariant, so we work on
the simplex sum(a)=1, a>=0, minimizing 2*n*max(conv(a,a)).
Key novel elements: adaptive log-sum-exp smoothing for rich gradient signal,
spectral peak-shaving that directly modifies |FFT(a)|^2, and a two-phase
coarse-to-fine strategy for the smoothing parameter tau.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def hankel_grad(a, w):
    """Compute grad[i] = 2 * sum_j a[j]*w[i+j] using vectorized Hankel matrix."""
    n = len(a)
    i = np.arange(n)[:, None]
    j = np.arange(n)[None, :]
    idx = i + j
    M = w[idx]
    return 2.0 * (M @ a)


def lse_subgradient(a, time_budget=30.0, tau_start=0.05, tau_end=0.001,
                    max_iters=500, momentum=0.9, verbose=False):
    """
    Log-sum-exp smoothed subgradient descent with Nesterov momentum.
    tau is annealed from tau_start to tau_end.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    step = 0.03

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Anneal tau
        frac = it / max(max_iters - 1, 1)
        tau = tau_start + (tau_end - tau_start) * frac

        # Evaluate at Nesterov point
        y = project_simplex(y)
        conv = compute_conv(y)
        cmax = float(np.max(conv))

        # Softmax weights
        shifted = np.clip((conv - cmax) / tau, -60.0, 60.0)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        # Gradient
        grad = hankel_grad(y, w)
        grad = grad - np.mean(grad)

        # Nesterov update
        v = momentum * v - step * grad
        a_new = y + v
        a_new = project_simplex(a_new)

        # Evaluate
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        # Update Nesterov point
        y = a_new + momentum * (a_new - a)
        a = a_new

        # Decay step size
        step *= 0.997

        if verbose and it % 100 == 0:
            print(f"    LSE it={it}: obj={best_obj:.6f}, tau={tau:.5f}")

    return best_a, best_obj


def spectral_flatten(a, time_budget=20.0, n_iters=100, verbose=False):
    """
    Modify the magnitude of FFT(a) to flatten the time-domain peak of conv(a,a).
    Since conv(a,a) has spectrum |FFT(a)|^2, we directly shape the spectrum.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    fa = np.fft.fft(a, N)
    magnitude = np.abs(fa)
    phase = np.angle(fa)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # Current spectrum
        spectrum = magnitude ** 2

        # Inverse FFT to get current convolution (circular, but N >= 2n-1 so ok)
        conv_circ = np.fft.ifft(spectrum).real
        conv_lin = conv_circ[:2 * n - 1]
        cmax = float(np.max(conv_lin))

        # Identify peak regions in time domain
        # Want to reduce the peaks: find where conv is highest
        threshold = cmax * 0.8
        peak_mask = conv_circ > threshold

        # Reduce the spectrum at frequencies that contribute to peaks
        # Use the "peak pressure" approach: for each peak position, reduce
        # the spectral components that contribute
        # Simpler: apply a soft threshold to the spectrum
        alpha = 0.85 + 0.1 * (it / max(n_iters - 1, 1))  # gradually reduce compression
        spec_norm = spectrum / (np.max(spectrum) + 1e-30)
        spec_compressed = spec_norm ** alpha
        magnitude_new = np.sqrt(spec_compressed * np.max(spectrum))

        # Reconstruct
        fa_new = magnitude_new * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            # Update spectrum
            fa = np.fft.fft(a_new, N)
            magnitude = np.abs(fa)
            phase = np.angle(fa)

    return best_a, best_obj


def spectral_chase(a, time_budget=20.0, n_iters=150, verbose=False):
    """
    More direct spectral approach: iteratively reduce the contribution of
    spectral components that cause time-domain peaks.
    Uses a water-filling analogy in the frequency domain.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        conv = np.fft.ifft(fa * fa).real[:2 * n - 1]
        cmax = float(np.max(conv))

        if cmax < 1e-15:
            break

        # Find top peaks
        top_k = max(5, len(conv) // 20)
        peak_indices = np.argsort(conv)[-top_k:]
        peak_height = conv[peak_indices].mean()

        # Create "pressure" map: high where peaks are
        pressure = np.zeros(N)
        for k_idx in peak_indices:
            if 0 <= k_idx < N:
                pressure[k_idx] = 1.0

        # Compute which frequency bins contribute to peaks
        # freq_pressure[m] = sum_k pressure[k] * |FFT contribution|
        # Simplified: reduce magnitude at frequencies where the "beat" creates peaks
        # Use a different approach: compute the gradient in frequency domain
        # d(conv[k])/d|a_hat[m]| relates to the phase coherence

        # Practical approach: reduce the dynamic range of the spectrum
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Apply adaptive compression: compress high-magnitude bins more
        mag_max = np.max(mag)
        if mag_max < 1e-30:
            break

        # Target: make the spectrum closer to "flat" (like white noise)
        # But we need to keep the DC component (for the sum constraint)
        target = np.ones(N) * (mag ** 2).sum() ** 0.5 / np.sqrt(N)
        target[0] = mag[0]  # preserve DC

        alpha = 0.05 + 0.15 * (it / max(n_iters - 1, 1))
        mag_new = mag * (1 - alpha) + target * alpha
        mag_new = np.maximum(mag_new, 0)

        fa_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def lp_polish(a, time_budget=20.0):
    """
    LP-based polishing: solve min t s.t. conv(a*g)[k] <= t, sum(g)=1, g>=0
    and blend the solution with current a.
    Only works for smaller n.
    """
    n = len(a)
    if n > 600:
        return a, compute_obj(a)

    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(20):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Build Toeplitz: (Toep(a) @ g)[k] = conv(a, g)[k]
        # conv(a,g)[k] = sum_j a[j] g[k-j]
        # As a matrix: A[k,j] = a[k-j] if 0<=k-j<n, else 0
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A[k, j] = a[idx]

        # LP: min M_new s.t. A @ g <= M_new * ones, sum(g) = 1, g >= 0
        # Variables: g (n) + M_new (1) = n+1 variables
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize M_new

        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0  # A@g - M <= 0
        b_ub = np.zeros(2 * n - 1)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0.0, None)] * n + [(0.0, None)]

        try:
            t_limit = max(1.0, time_budget * 0.3)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': t_limit})
            if res.status == 0:
                g = np.maximum(res.x[:n], 0)
                s = g.sum()
                if s > 1e-10:
                    g = g / s
                    # Blend
                    for t in [0.3, 0.5, 0.7, 1.0]:
                        a_try = (1 - t) * a + t * g
                        a_try = project_simplex(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            break
        except Exception:
            break

    return best_a, best_obj


def pressure_descent(a, time_budget=30.0, n_steps=200, verbose=False):
    """
    Pressure-based descent: identify which elements of a contribute most to
    convolution peaks and selectively reduce them while boosting valley contributors.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Find top peaks
        n_peaks = max(3, len(conv) // 30)
        peak_indices = np.argsort(conv)[-n_peaks:]

        # Compute "pressure" on each element
        # pressure[i] = sum of conv[k] for peaks k where a[i] contributes
        pressure = np.zeros(n)
        for k in peak_indices:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += conv[k] * a[j]

        # Normalize pressure
        p_max = np.max(pressure)
        if p_max > 1e-30:
            pressure /= p_max

        # Direction: reduce high-pressure elements, boost low-pressure
        direction = 1.0 - 2.0 * pressure  # -1 to +1
        direction -= np.mean(direction)

        # Adaptive step size
        step = 0.02 * (1.0 - 0.5 * it / n_steps)

        a_new = a + step * direction
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def noise_injection(a, rng, strength=0.1):
    """Add multiplicative noise and project."""
    n = len(a)
    noise = np.exp(rng.normal(0, strength, n))
    a_noisy = a * noise
    return project_simplex(a_noisy)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def generate_structured_init(n, rng, parent=None):
    """Generate a diverse set of structured initializations."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Sine-based (low autocorrelation tendency)
    x = np.linspace(0, n, n, endpoint=False)
    for freq in [2, 3, 5, 7]:
        s = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x / n)
        inits.append(s / s.sum())

    # Cosine taper
    x = np.linspace(0, np.pi, n)
    c = 0.5 + 0.5 * np.cos(x)
    inits.append(c / c.sum())

    # Raised cosine
    x = np.linspace(0, 2 * np.pi, n)
    rc = 0.5 + 0.5 * np.cos(x)
    inits.append(rc / rc.sum())

    # Two-scale: base + fine noise
    base = np.ones(n) / n
    fine = rng.normal(0, 1, n)
    fine -= fine.mean()
    fine /= (np.std(fine) + 1e-10)
    for amp in [0.1, 0.2, 0.3]:
        s = 1.0 + amp * fine
        s = np.maximum(s, 0.1)
        inits.append(s / s.sum())

    # "Random paper" (white noise)
    r = rng.random(n) + 0.5
    inits.append(r / r.sum())

    # Banded: concentrated in center
    center = n // 2
    half = n // 4
    banded = np.zeros(n)
    banded[center - half: center + half] = 1.0
    inits.append(banded / banded.sum())

    if parent is not None:
        # Resampled parent
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        p = resample_seq(parent, n)
        p = noise_injection(p, rng, 0.05)
        inits.append(p)

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-phase optimization combining log-sum-exp
    subgradient descent, spectral flattening, pressure descent, and LP polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(500) / 500.0
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}")

    # === Phase 1: Coarse screening with structured inits (10% time) ===
    print("\n=== Phase 1: Coarse Screening ===")
    phase1_budget = max(30.0, (deadline - start_time) * 0.10)
    phase1_start = time.monotonic()

    # Try several lengths
    n_options = [n_parent]
    for dl in [-50, -25, 25, 50, 75, 100]:
        n_try = n_parent + dl
        if 100 <= n_try <= 3000:
            n_options.append(n_try)
    n_options = list(dict.fromkeys(n_options))

    candidates = []
    per_n_budget = max(5.0, phase1_budget / len(n_options))

    for n_try in n_options:
        if time.monotonic() > phase1_start + phase1_budget:
            break
        inits = generate_structured_init(n_try, rng, parent)
        per_init_time = max(1.0, per_n_budget / (len(inits) * 2))

        for init in inits:
            if time.monotonic() > phase1_start + phase1_budget:
                break
            # Quick LSE subgradient
            a_quick, obj_quick = lse_subgradient(init, time_budget=per_init_time,
                                                 tau_start=0.1, tau_end=0.02,
                                                 max_iters=50)
            candidates.append((obj_quick, a_quick, n_try))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.6f}")

    for obj_c, a_c, n_c in candidates[:3]:
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  NEW BEST from screening: n={n_c}, obj={best_score:.8f}")

    # === Phase 2: Intensive optimization (50% time) ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    phase2_budget = max(100.0, (deadline - start_time) * 0.50)
    phase2_start = time.monotonic()

    a_main = project_simplex(best_seq.copy())
    n_main = len(a_main)
    local_best = a_main.copy()
    local_obj = compute_obj(a_main)

    # Cycle through strategies
    strategies = ['lse', 'spectral', 'pressure', 'spectral_chase']
    n_cycles = max(1, int(phase2_budget / 25))

    for cyc in range(n_cycles):
        if time.monotonic() > phase2_start + phase2_budget:
            break
        remaining = phase2_start + phase2_budget - time.monotonic()
        if remaining < 10:
            break

        strat = strategies[cyc % len(strategies)]
        budget_here = min(remaining * 0.4, 40.0)

        if strat == 'lse':
            a_new, obj_new = lse_subgradient(local_best, time_budget=budget_here,
                                             tau_start=0.05, tau_end=0.001,
                                             max_iters=200, momentum=0.9)
        elif strat == 'spectral':
            a_new, obj_new = spectral_flatten(local_best, time_budget=budget_here,
                                              n_iters=80)
        elif strat == 'pressure':
            a_new, obj_new = pressure_descent(local_best, time_budget=budget_here,
                                              n_steps=150)
        else:
            a_new, obj_new = spectral_chase(local_best, time_budget=budget_here,
                                            n_iters=100)

        if obj_new < local_obj:
            local_obj = obj_new
            local_best = a_new.copy()
            if local_obj < best_score:
                best_score = local_obj
                best_seq = local_best.copy()
                print(f"  Cycle {cyc} [{strat}]: NEW BEST {best_score:.8f}")
            else:
                print(f"  Cycle {cyc} [{strat}]: local={local_obj:.8f}")
        else:
            print(f"  Cycle {cyc} [{strat}]: no improvement (local={local_obj:.8f})")

    # === Phase 3: LP polishing + fine LSE (20% time) ===
    print(f"\n=== Phase 3: LP + Fine Polish ===")
    phase3_budget = max(30.0, (deadline - time.monotonic()) * 0.20)
    phase3_start = time.monotonic()

    a_polish = project_simplex(local_best.copy())
    n_polish = len(a_polish)

    print(f"  Before polish: obj={compute_obj(a_polish):.8f}")

    # LP if small enough
    if n_polish <= 600:
        a_lp, obj_lp = lp_polish(a_polish, time_budget=min(30.0, phase3_budget * 0.5))
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_polish = a_lp.copy()
            print(f"  LP improved: {best_score:.8f}")

    # Fine LSE with very small tau
    remaining3 = phase3_start + phase3_budget - time.monotonic()
    if remaining3 > 10:
        a_fine, obj_fine = lse_subgradient(a_polish, time_budget=min(40.0, remaining3),
                                           tau_start=0.02, tau_end=0.002,
                                           max_iters=300, momentum=0.95)
        if obj_fine < best_score:
            best_score = obj_fine
            best_seq = a_fine.copy()
            print(f"  Fine LSE improved: {best_score:.8f}")

    # === Phase 4: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    phase4_budget = max(20.0, (deadline - time.monotonic()) * 0.10)
    phase4_start = time.monotonic()

    n_current = len(best_seq)
    n_tries = list(dict.fromkeys([
        n_current - 30, n_current - 60, n_current + 30, n_current + 60,
        n_current + 100, n_current - 100,
        max(100, n_current // 2), min(3000, n_current * 2)
    ]))
    n_tries = [n for n in n_tries if 100 <= n <= 3000]

    per_n4 = max(5.0, phase4_budget / max(len(n_tries), 1))

    for n_try in n_tries:
        if time.monotonic() > phase4_start + phase4_budget:
            break
        a_res = resample_seq(best_seq, n_try)
        a_res = project_simplex(a_res)

        # Quick optimization
        a_opt, obj_opt = lse_subgradient(a_res, time_budget=min(per_n4 * 0.7, 20.0),
                                         tau_start=0.05, tau_end=0.005,
                                         max_iters=100)
        if per_n4 > 15:
            a_opt2, obj_opt2 = spectral_flatten(a_opt, time_budget=min(per_n4 * 0.3, 10.0),
                                                n_iters=50)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: {obj_opt:.8f} (no improvement)")

    # === Phase 5: Final ultra-fine polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining5 = deadline - time.monotonic()
    print(f"  Best before final: {best_score:.8f}, time left: {remaining5:.1f}s")

    a_final = project_simplex(best_seq.copy())

    if remaining5 > 10:
        # Very fine LSE
        a_p1, obj_p1 = lse_subgradient(a_final, time_budget=min(remaining5 * 0.4, 60.0),
                                       tau_start=0.01, tau_end=0.001,
                                       max_iters=500, momentum=0.95)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            a_final = a_p1.copy()
            print(f"  Final LSE: {best_score:.8f}")

        remaining5b = deadline - time.monotonic()
        if remaining5b > 10:
            # Pressure descent very fine
            a_p2, obj_p2 = pressure_descent(a_final, time_budget=min(remaining5b * 0.3, 30.0),
                                            n_steps=300)
            if obj_p2 < best_score:
                best_score = obj_p2
                best_seq = a_p2.copy()
                a_final = a_p2.copy()
                print(f"  Final pressure: {best_score:.8f}")

        remaining5c = deadline - time.monotonic()
        if remaining5c > 5:
            # One more spectral pass
            a_p3, obj_p3 = spectral_flatten(a_final, time_budget=min(remaining5c * 0.5, 20.0),
                                            n_iters=100)
            if obj_p3 < best_score:
                best_score = obj_p3
                best_seq = a_p3.copy()
                print(f"  Final spectral: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
