"""
LP-alternating with adaptive active-set constraints, exact line search,
and multi-peak subgradient polishing to minimize 2*n*max(conv(a,a))/(sum(a))^2
for non-negative sequences. Improvements over previous: tighter active-set LP,
adaptive constraint tightening schedule, exact line search, multi-peak weighted
subgradient, and asymmetric initializations exploiting off-center peak structure.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_direction(a, M, active_idx, time_limit=15):
    n = len(a)
    T = build_toeplitz(a)
    Tsub = T[active_idx, :]
    c_vec = -np.ones(n)
    b_ub = np.full(len(active_idx), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=Tsub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def get_active_set(conv, frac=0.15, min_count=20):
    n_conv = len(conv)
    threshold = np.percentile(conv, 100.0 * (1.0 - frac))
    active = np.where(conv >= threshold)[0]
    if len(active) < min_count:
        threshold = np.percentile(conv, 100.0 * (1.0 - max(frac, min_count / n_conv)))
        active = np.where(conv >= threshold)[0]
    return active


def exact_line_search(a, g, max_trials=15):
    best_obj = compute_obj(a)
    best_t = 0.0
    best_a = a.copy()
    t_vals = np.concatenate([
        np.linspace(0.01, 0.3, 10),
        np.linspace(0.3, 1.0, 5)
    ])
    for t in t_vals:
        a_try = (1.0 - t) * a + t * g
        a_try = project_nonneg_sum1(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
            best_a = a_try.copy()
    return best_a, best_obj, best_t


def multi_peak_subgradient(a, n_steps=100, tau=0.01, step_size=0.05):
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        if cmax < 1e-15:
            break
        shifted = (conv - cmax) / tau
        weights = np.exp(np.clip(shifted, -500, 0))
        wsum = weights.sum()
        if wsum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= wsum

        grad = np.zeros(n)
        for k_idx in range(len(weights)):
            if weights[k_idx] < 1e-10:
                continue
            j_start = max(0, k_idx - n + 1)
            j_end = min(n, k_idx + 1)
            if j_end > j_start:
                indices = np.arange(j_start, j_end)
                a_indices = k_idx - indices
                grad[indices] += 2.0 * weights[k_idx] * a[a_indices]

        grad_centered = grad - np.mean(grad)
        a_try = a - step_size * grad_centered
        a_try = project_nonneg_sum1(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        else:
            step_size *= 0.7
            if step_size < 0.001:
                step_size = 0.05
                break

    return best_a, best_obj


def freq_domain_flatten(a, max_iters=100, time_budget=20.0):
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        pmax = np.max(power)
        if pmax < 1e-15:
            break

        ps_norm = power / pmax
        alpha = 0.6 + 0.3 * (it / max_iters)
        ps_comp = np.maximum(ps_norm, 1e-12) ** alpha

        phase = np.angle(fa)
        mag = np.sqrt(ps_comp * pmax)
        fa_new = mag * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def generate_inits(n, rng, parent=None):
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    for p in [0.5, 1.0, 2.0, 4.0]:
        s = np.sin(np.pi * x) ** p + 0.01
        inits.append(s / s.sum())

    # Asymmetric: concentrated on left
    left = np.exp(-np.linspace(0, 3, n))
    inits.append(left / left.sum())
    # Asymmetric: concentrated on right
    right = np.exp(-np.linspace(3, 0, n))
    inits.append(right / right.sum())
    # Two peaks (asymmetric positions)
    peak1 = int(n * 0.2)
    peak2 = int(n * 0.75)
    two = np.zeros(n)
    for i in range(n):
        two[i] = np.exp(-(i - peak1) ** 2 / (n * 0.01)) + np.exp(-(i - peak2) ** 2 / (n * 0.02))
    inits.append(two / two.sum())
    # Rand
    r = rng.random(n) + 0.05
    inits.append(r / r.sum())

    # Log-uniform spaced
    log_sp = 1.0 / np.exp(np.linspace(0, 2, n))
    inits.append(log_sp / log_sp.sum())

    if parent is not None:
        pr = resample_sequence(parent, n)
        inits.append(pr)
        pert = pr * np.exp(rng.normal(0, 0.1, n))
        inits.append(project_nonneg_sum1(pert))
        # Flip
        inits.append(pr[::-1] / pr[::-1].sum())

    return inits
