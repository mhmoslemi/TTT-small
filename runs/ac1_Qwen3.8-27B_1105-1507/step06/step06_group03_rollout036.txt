def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

- Compute ifft(conj(fft(a, L)) * fft(w, L))

- If the current peaks are at lags k1, k2, k3, ..., and we change a[i], the peaks change.
- The optimal a[i] would be where the top peaks are as equal as possible (minimax principle).
- This is hard to compute exactly, but we can approximate: if the top peak is at k*, try values of a[i] that would reduce conv[k*] to match the second peak.

For conv[k*] = sum_j a[j]*a[k*-j], the derivative with respect to a[i] is 2*a[k*-i] (if valid). So to reduce conv[k*] by δ, we need to change a[i] by δ/(2*a[k*-i]).

"""
Multi-phase optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) smoothed gradient descent with adaptive tau, (2) multi-peak SCA-LP,
(3) coordinate descent with peak-aware candidates, (4) multi-length exploration,
(5) bisection-based feasibility SCA for lower bound estimation.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2. Assumes a >= 0."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_grad_descent(a, max_iters, deadline, tau_init=20.0, tau_max=400.0,
                        lr_init=0.03, beta1=0.85, no_improve_limit=300):
    """Smoothed gradient descent with momentum on the probability simplex."""
    n = len(a)
    a = normalize(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    t_count = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_init + (tau_max - tau_init) * frac * tau

        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(np.minimum(tau * c_shifted, 50.0))
        w = exp_c / exp_c.sum()

        # Compute grad[i] = 2 * (a *star w)[i] = 2 * sum_j a[j] * w[j+i]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a, size)
        W_fft = np.fft.rfft(w, size)
        cross = np.fft.irfft(np.conj(A_fft) * W_fft, size)
        grad = 2.0 * cross[:n]

        grad = grad - np.mean(grad)
        t_count += 1
        m = beta1 * m + (1.0 - beta1) * grad
        m_hat = m / (1.0 - beta1 ** t_count)

        lr = lr_init / (1.0 + 0.3 * np.sqrt(t_count))
        a_new = a - lr * m_hat
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > no_improve_limit:
                rng = np.random.default_rng(it * 13 + 7)
                a = best_a + rng.standard_normal(n) * 0.02
                a = project_simplex(a)
                m = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

    return best_a, best_obj


def sca_lp_multi_peak(a, n_peaks, max_iters, deadline, lp_time_limit=10):
    """
    SCA-LP with multiple active peaks.
    Linearizes conv(g,g) around current a for the top n_peaks lags.
    """
    n = len(a)
    a = normalize(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        top_k = np.argsort(c)[::-1][:n_peaks]

        rows = []
        cols = []
        vals = []
        b_ub = []

        for kk in top_k:
            row_idx = len(b_ub)
            j_min = max(0, kk - n + 1)
            j_max = min(n - 1, kk)
            for j in range(j_min, j_max + 1):
                rows.append(row_idx)
                cols.append(j)
                vals.append(2.0 * a[kk - j])
            rows.append(row_idx)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[kk])

        if len(rows) == 0:
            no_improve += 1
            continue

        A_ub = coo_matrix((vals, (rows, cols)), shape=(len(b_ub), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0, None)] * n + [(-1e6, 1e6)]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=np.array(b_ub), A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': lp_time_limit, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
                        a_try = (1.0 - frac) * a + frac * g
                        a_try = normalize(a_try)
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 40:
            break

    return best_a, best_obj


def coord_descent(a, max_sweeps, deadline, random_order=False, seed=42):
    """Coordinate descent with fixed candidate ratios."""
    n = len(a)
    a = normalize(a.astype(np.float64))
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 150

    ratios = np.array([0.0, 0.1, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 1.75, 2.0, 2.5, 3.0])

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)

        for i in order:
            old = a[i]
            if old < 1e-12:
                candidates = [0.0, 0.5/n, 1.0/n, 2.0/n, 3.0/n]
            else:
                candidates = np.array([0.0]) * 0 + ratios * old

            a_slice_start = max(0, i)
            a_slice_end = min(2 * n - 2, i + n - 1)
            a_slice = a[a_slice_start - i : a_slice_end - i + 1].copy()
            kq = 2 * i
            valid_kq = 0 <= kq < 2 * n - 1

            best_val = old
            best_local = 2.0 * n * conv.max()

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                conv[a_slice_start:a_slice_end + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                f = 2.0 * n * conv.max()
                conv[a_slice_start:a_slice_end + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

                if f < best_local:
                    best_local = f
                    best_val = x

            if abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[a_slice_start:a_slice_end + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val

        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-11:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = normalize(p)
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.03, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.08, n)))

    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3, 4, 5, 7]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.05, 0.1, 0.15, 0.25, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.02))

    for alpha in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    ksize = max(5, n // 12)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.02))

    # Triangle wave
    inits.append(project_simplex(1.0 + 0.5 * np.abs(np.sin(np.pi * t * 4))))

    # Two bumps
    inits.append(project_simplex(
        np.exp(-((t - 0.3) / 0.12) ** 2) + np.exp(-((t - 0.7) / 0.12) ** 2) + 0.05))

    return inits


def optimize_pipeline(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence:
    1. CD (quick)
    2. Smooth gradient descent
    3. SCA-LP
    4. CD (detailed)
    5. SCA-LP again
    6. Final CD
    """
    n = len(a)
    a = normalize(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = normalize(np.maximum(a_new, 0))
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 1: Quick CD (15% of time)
    t1 = time.monotonic() + 0.15 * remaining
    if verbose:
        print(f"    CD1 (budget {t1 - time.monotonic():.0f}s)", flush=True)
    a = coord_descent(a, max_sweeps=5000, deadline=t1, random_order=True, seed=1)
    a = try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 2: Smooth gradient descent (25% of time)
    t2 = time.monotonic() + 0.25 * remaining
    if verbose:
        print(f"    GradDesc (budget {t2 - time.monotonic():.0f}s)", flush=True)
    a_g, obj_g = smooth_grad_descent(a, max_iters=5000, deadline=t2,
                                     tau_init=20.0, tau_max=300.0, lr_init=0.03)
    if obj_g < best_obj:
        best_obj = obj_g
        best_a = a_g.copy()
        if verbose:
            print(f"      Grad improved: {obj_g:.8f}", flush=True)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 3: SCA-LP (25% of time)
    t3 = time.monotonic() + 0.25 * remaining
    if verbose:
        print(f"    SCA-LP (budget {t3 - time.monotonic():.0f}s)", flush=True)
    a_lp, obj_lp = sca_lp_multi_peak(a, n_peaks=30, max_iters=200, deadline=t3)
    if obj_lp < best_obj:
        best_obj = obj_lp
        best_a = a_lp.copy()
        if verbose:
            print(f"      LP improved: {obj_lp:.8f}", flush=True)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 4: Detailed CD (20% of time)
    t4 = time.monotonic() + 0.20 * remaining
    if verbose:
        print(f"    CD2 (budget {t4 - time.monotonic():.0f}s)", flush=True)
    a = coord_descent(a, max_sweeps=15000, deadline=t4, random_order=True, seed=77)
    a = try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 5: Final SCA-LP (10% of time)
    t5 = time.monotonic() + 0.10 * remaining
    if verbose:
        print(f"    SCA-LP2 (budget {t5 - time.monotonic():.0f}s)", flush=True)
    a_lp2, obj_lp2 = sca_lp_multi_peak(best_a, n_peaks=50, max_iters=100, deadline=t5,
                                       lp_time_limit=8)
    if obj_lp2 < best_obj:
        best_obj = obj_lp2
        best_a = a_lp2.copy()
        if verbose:
            print(f"      LP2 improved: {obj_lp2:.8f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 3:
        a_final = coord_descent(best_a, max_sweeps=5000, deadline=deadline - 1,
                                random_order=True, seed=99)
        a_final = try_update(a_final)

    return best_a, best_obj

                # Pass 2: SCA-LP with many peaks
                t_p2 = time.monotonic() + 0.30 * remaining
                print(f"  SCA-LP polish (budget {t_p2 - time.monotonic():.0f}s)...", flush=True)
                a_lp, obj_lp = sca_lp_multi_peak(best_seq, n_peaks=80, max_iters=300,
                                                 deadline=t_p2, lp_time_limit=12)
                update_best(a_lp)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 15:
                # Pass 3: Gradient descent with different params
                t_p3 = time.monotonic() + 0.25 * remaining
                print(f"  GradDesc polish (budget {t_p3 - time.monotonic():.0f}s)...", flush=True)
                a_g, obj_g = smooth_grad_descent(best_seq, max_iters=8000, deadline=t_p3,
                                                 tau_init=50.0, tau_max=500.0, lr_init=0.01)
                update_best(a_g)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 10:
                # Pass 4: Final CD
                t_p4 = deadline - 5
                print(f"  Final CD (budget {t_p4 - time.monotonic():.0f}s)...", flush=True)
                a_final = coord_descent(best_seq, max_sweeps=20000, deadline=t_p4,
                                        random_order=True, seed=54321)
                update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
