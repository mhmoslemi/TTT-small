```python
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj


def l_bfgs_b_optimize(a, beta, maxiter=2000, time_budget=60.0):
    """Optimize using L-BFGS-B on the smoothed objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    a = a / s
    
    start = time.monotonic()
    
    def fun_and_grad(x):
        if time.monotonic() - start > time_budget:
            raise KeyboardInterrupt
        # x is in log-space
        x_shifted = x - np.max(x)
        a_local = np.exp(x_shifted)
        a_local = a_local / np.sum(a_local)
        obj, grad_a = smooth_obj_and_grad(a_local, beta)
        # Chain rule: grad_x = grad_a * a
        grad_x = grad_a * a_local
        return obj, grad_x
    
    # Start in log-space
    x0 = np.log(np.maximum(a, 1e-15))
    x0 -= np.mean(x0)
    
    try:
        result = fmin_l_bfgs_b(fun_and_grad, x0, maxiter=maxiter, 
                               pgtol=1e-10, IPrint=0)
    except KeyboardInterrupt:
        # Use last known best
        return a
    
    x_opt = result[0]
    x_shifted = x_opt - np.max(x_opt)
    a_opt = np.exp(x_shifted)
    a_opt = a_opt / np.sum(a_opt)
    
    return a_opt
```

In log-space with the constraint that the sum is 1, we can use the reparameterization: `a[i] = exp(x[i] - logsumexp(x))`. This automatically maintains the sum=1 constraint.

```python
        a_grad_a = np.dot(a_local, grad_a)
        grad_x = a_local * (grad_a - a_grad_a)

        x_opt = x  # return initial

    a_opt = np.exp(x_shifted)
    a_opt = a_opt / np.sum(a_opt)
    
    return a_opt
```

```python
"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2:
1. Diverse initialization with L-BFGS-B on log-sum-exp smoothed objective
2. LP-based peak reduction
3. Peak-targeted coordinate descent
4. Multi-n exploration
5. Final high-temperature polish
"""

import numpy as np
from scipy.optimize import fmin_l_bfgs_b, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """Compute log-sum-exp smoothed objective and gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    # Gradient via FFT cross-correlation
    size = 1
    while size < 4 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def lbfbsb_smooth(a_init, beta_schedule, maxiter_per_stage=3000, total_time=600.0):
    """
    L-BFGS-B optimization with progressive beta sharpening.
    beta_schedule: list of (beta_start, beta_end) pairs for each stage.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for stage, (b_start, b_end) in enumerate(beta_schedule):
        time_left = total_time - (time.monotonic() - start)
        if time_left < 5:
            break
        
        # Sub-stages within each stage: increase beta gradually
        n_sub = 3
        for sub in range(n_sub):
            time_left = total_time - (time.monotonic() - start)
            if time_left < 10:
                    break
            
            beta = b_start * (b_end / b_start) ** (sub / max(n_sub - 1, 1))
            
            # Log-space with softmax
            x = np.log(a)
            x -= np.mean(x)
            
            stage_start = time.monotonic()
            stage_budget = min(time_left * 0.3, 120.0)
            
            def fun_grad(x):
                if time.monotonic() - stage_start > stage_budget:
                    raise TimeoutError()
                x_s = x - np.max(x)
                e = np.exp(x_s)
                a_loc = e / np.sum(e)
                obj, grad_a = smooth_obj_grad(a_loc, beta)
                a_dot_g = np.dot(a_loc, grad_a)
                grad_x = a_loc * (grad_a - a_dot_g)
                return obj, grad_x
            
            try:
                res = fmin_l_bfgs_b(fun_grad, x, maxiter=maxiter_per_stage,
                                    pgtol=1e-14, IPrint=0)
                x_opt = res[0]
            except (TimeoutError, Exception):
                x_opt = x
            
            x_s = x_opt - np.max(x_opt)
            a_new = np.exp(x_s)
            a_new = a_new / np.sum(a_new)
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
            
            print(f"    Stage {stage} sub {sub} (beta={beta:.1f}): obj={obj_new:.8f}")
    
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_max = min(n, m - j)
        A[j:j+k_max, j] = a[:k_max]
    return A


def lp_solve_direction(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, time_budget=60.0):
    """LP-based refinement with line search."""
    n = len(a)
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    
    conv = compute_conv(a_norm)
    M = float(np.max(conv))
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Multiple LP solves with decreasing M
    for m_frac in [1.0, 0.95, 0.9, 0.85, 0.8]:
        if time.monotonic() - start > time_budget * 0.7:
            break
        M_cur = M * m_frac
        alloc = min(time_budget * 0.2, 30.0)
        g = lp_solve_direction(a_norm, M_cur, time_limit=alloc)
        if g is None:
            continue
        
        # Line search
        for t in np.linspace(0.1, 1.0, 10):
            a_try = (1 - t) * a_norm + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                # Update reference
                conv = compute_conv(a_norm)
                M = float(np.max(conv))
    
    return best_a


def peak_shave(a, time_budget=60.0, n_peaks=5):
    """Peak-targeted coordinate descent: directly reduce top peaks."""
    n = len(a)
    start = time.monotonic()
    best_a = a.astype(np.float64).copy()
    best_obj = compute_obj(best_a)
    
    for iteration in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(best_a)
        peaks = np.argsort(conv)[::-1][:n_peaks]
        
        improved = False
        for p in peaks:
            if time.monotonic() - start > time_budget:
                break
            
            # Find elements contributing most to this peak
            # conv[p] = sum_i a[i]*a[p-i] (valid indices)
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            
            contributions = []
            for i in range(i_min, i_max + 1):
                j = p - i
                if 0 <= j < n:
                    contributions.append((i, j, best_a[i] * best_a[j]))
            
            contributions.sort(key=lambda x: -x[2])
            
            # Try reducing top contributors
            for idx in range(min(3, len(contributions))):
                i, j, _ = contributions[idx]
                # Try: reduce a[i], increase a[j'] where j' doesn't contribute to peak
                # Find a j' that has low contribution to peak p
                best_j_prime = -1
                best_reduction = 0
                
                for candidate in range(n):
                    if candidate == i or candidate == j:
                        continue
                    # Check if candidate contributes to peak p
                    k1 = p - candidate
                    k2 = candidate - p
                    contrib = 0
                    if 0 <= k1 < n:
                        contrib += best_a[candidate] * best_a[k1]
                    if 0 <= k2 < n:
                        contrib += best_a[candidate] * best_a[k2]
                    # We want low contrib
                    if contrib < best_reduction or best_j_prime == -1:
                        # Actually we want to find the one with lowest contribution
                        pass
                
                # Simpler: just try moving a small amount of mass
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                    delta = frac * best_a[i]
                    if delta > 0.5 * best_a[i]:
                        delta = 0.5 * best_a[i]
                    
                    # Try moving from i to a random position
                    for j_try in [0, n-1, n//4, 3*n//4]:
                        if j_try == i:
                            continue
                        a_try = best_a.copy()
                        a_try[i] -= delta
                        a_try[j_try] += delta
                        if a_try[i] < -1e-12:
                            continue
                        a_try = np.maximum(a_try, 0)
                        s_try = np.sum(a_try)
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved and iteration > 30:
            break
    
    return best_a


def generate_structured(n, kind='hann', seed=0):
    """Generate structured initial sequences."""
    x = np.arange(n) / max(n - 1, 1)
    if kind == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == 'triangle':
        a = 1 - 2 * np.abs(x - 0.5)
    elif kind == 'gauss_center':
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
    elif kind == 'gauss_left':
        a = np.exp(-((x - 0.2) ** 2) / (2 * 0.15 ** 2))
    elif kind == 'exp_decay':
        a = np.exp(-3 * x)
    elif kind == 'exp_decay_rev':
        a = np.exp(-3 * (1 - x))
    elif kind == 'power':
        a = (1 - x) ** 2
    elif kind == 'power_rev':
        a = x ** 2
    elif kind == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    return a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
```

```python
"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2:
1. L-BFGS-B with progressive log-sum-exp smoothing from diverse starts
2. LP-based direction search for peak reduction
3. Peak-targeted gradient coordinate descent
4. Multi-n exploration
5. Final high-temperature polish
Key improvements over Adam: second-order convergence, adaptive scheduling,
peak-directed corrections.

import numpy as np
from scipy.optimize import fmin_l_bfgs_b, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """
    Compute log-sum-exp smoothed objective and gradient w.r.t. a.
    Uses soft-max weighting of convolution peaks.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    # Gradient via FFT: grad_ls[j] = 2 * sum_k w[k] * a[|k-j|] type expression
    # More precisely: d(logsumexp)/d(a[j]) = 2 * (w * a_cross)[j]
    size = 1
    while size < 4 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def lbfbsb_single(a, beta, time_budget=30.0):
    """
    Single L-BFGS-B optimization at fixed beta using softmax parameterization.
    Returns optimized a (sum=1, all positive).
    """
    n = len(a)
    x = np.log(np.maximum(a, 1e-15))
    x -= np.mean(x)

    # Benchmark one evaluation to estimate timing
    t0 = time.monotonic()
    x_s = x - np.max(x)
    e_test = np.exp(x_s)
    a_test = e_test / np.sum(e_test)
    _, _ = smooth_obj_grad(a_test, beta)
    t_eval = time.monotonic() - t0
    if t_eval < 5e-5:
        t_eval = 5e-4

    max_evals = max(200, int(time_budget / t_eval * 0.75))
    max_evals = min(max_evals, 80000)

    def fun_grad(x):
        x_s = x - np.max(x)
        e = np.exp(x_s)
        a_loc = e / np.sum(e)
        obj, grad_a = smooth_obj_grad(a_loc, beta)
        a_dot_g = float(np.dot(a_loc, grad_a))
        grad_x = a_loc * (grad_a - a_dot_g)
        return obj, grad_x

    try:
        res = fmin_l_bfgs_b(fun_grad, x, maxfun=max_evals, pgtol=1e-14, IPrint=0)

        x_opt = x

    x_s = x_opt - np.max(x_opt)
    a_opt = np.exp(x_s)
    s = np.sum(a_opt)

    return a_opt

def build_toeplitz(a):
    """Build matrix A where (A @ g) = convolve(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_max = min(n, m - j)
        if k_max > 0:
            A[j:j + k_max, j] = a[:k_max]
    return A


def lp_solve(a, M, time_limit=15.0):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, time_budget=60.0):

def peak_gradient_descent(a, time_budget=40.0, n_peaks=3, n_steps=500):
    """
    Coordinate descent targeting top convolution peaks.
    For each peak, compute gradient of that peak value and move mass
    from high-gradient to low-gradient positions.
    """
    n = len(a)
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for p in peak_indices:
            if time.monotonic() - start > time_budget:
                break

            # Gradient of conv[p] w.r.t. a[j] = a[p-j] + a[j-p] (valid only)
            grad_p = np.zeros(n)
            for j in range(n):
                k1 = p - j
                k2 = j - p
                val = 0.0
                if 0 <= k1 < n:
                    val += a[k1]
                if 0 <= k2 < n:
                    val += a[k2]
                grad_p[j] = val

            # We want to REDUCE conv[p]. Move mass from high grad to low grad.
            # Sort by gradient (descending)
            order = np.argsort(grad_p)[::-1]
            # Top candidates to decrease
            top_decrease = [order[i] for i in range(min(5, n)) if grad_p[order[i]] > 0.01 * np.max(grad_p)]
            # Bottom candidates to increase
            bottom_increase = [order[i] for i in range(min(5, n), n) if grad_p[order[i]] < 0.01 * np.max(grad_p)]

            if not top_decrease or not bottom_increase:
                continue

            for i in top_decrease[:2]:
                for j in bottom_increase[:2]:
                    for frac in [0.002, 0.005, 0.01, 0.02]:
                        delta = frac * a[i]
                        if delta >= 0.9 * a[i]:
                            continue
                        a_try = a.copy()
                        a_try[i] -= delta
                        a_try[j] += delta
                        a_try = np.maximum(a_try, 0)
                        s_try = np.sum(a_try)
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved and step > 50:
            break

    return best_a


def generate_structured(n, kind):
    """Generate a structured initial sequence of length n."""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if kind == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif kind == 'triangle':
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    elif kind == 'gauss_center':
        a = np.exp(-((x - 0.5) ** 2) / (2.0 * 0.12 ** 2))
    elif kind == 'gauss_left':
        a = np.exp(-((x - 0.25) ** 2) / (2.0 * 0.12 ** 2))
    elif kind == 'gauss_right':
        a = np.exp(-((x - 0.75) ** 2) / (2.0 * 0.12 ** 2))
    elif kind == 'exp_decay':
        a = np.exp(-4.0 * x)
    elif kind == 'exp_decay_fast':
        a = np.exp(-8.0 * x)
    elif kind == 'exp_growth':
        a = np.exp(-4.0 * (1.0 - x))
    elif kind == 'power2':
        a = (1.0 - x) ** 2
    elif kind == 'power3':
        a = (1.0 - x) ** 3
    elif kind == 'power_growth':
        a = x ** 2

    return a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

    # --- Load parent ---
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: best={best_score:.10f}, n_parent={n_parent}")

    # === Phase 1: Intensive L-BFGS-B on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: L-BFGS-B optimization on parent (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.astype(np.float64).copy()
        a = np.maximum(a, 1e-15)
        a = a / np.sum(a)

        time_left = deadline - time.monotonic()
        schedule_early = [(5, 30), (30, 80), (80, 200)]
        a = lbfbsb_progressive(a, schedule_early, time_budget=min(time_left * 0.4, 350))
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Early stages: {best_score:.10f} ***")

        time_left = deadline - time.monotonic()
        if time_left > 60:
            schedule_late = [(200, 500), (500, 1200), (1200, 3000)]
            a = lbfbsb_progressive(a, schedule_late, time_budget=min(time_left * 0.5, 300))
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Late stages: {best_score:.10f} ***")

    # === Phase 2: Diverse structured starts ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse starting points (n={n_parent})")
    print(f"{'='*60}")

    rng = np.random.default_rng(seed)
    structures = ['hann', 'gauss_left', 'exp_decay', 'sin_one_sided',
                  'power2', 'gauss_right', 'exp_decay_fast', 'triangle']

    for struct_name in structures:
        time_left = deadline - time.monotonic()
        if time_left < 25:
            break

        a_struct = generate_structured(n_parent, struct_name)
        # Small perturbation to avoid exact symmetry
        a_struct = a_struct * np.exp(rng.normal(0, 0.03, n_parent))
        a_struct = np.maximum(a_struct, 1e-15)
        a_struct = a_struct / np.sum(a_struct)

        alloc = min(time_left * 0.12, 50.0)
        schedule = [(10, 100), (100, 500), (500, 2000)]
        a_opt = lbfbsb_progressive(a_struct, schedule, time_budget=alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  {struct_name}: {best_score:.10f} *** UPDATE")
        else:
            print(f"  {struct_name}: {obj:.10f}")

    # === Phase 3: LP refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 3: LP refinement (n={len(best_seq) if best_seq is not None else '?'})")
    print(f"{'='*60}")

    if best_seq is not None and len(best_seq) <= 900:
        for round_idx in range(5):
            time_left = deadline - time.monotonic()
            if time_left < 25:
                break

            alloc = min(time_left * 0.2, 50.0)
            a_lp = lp_refine(best_seq, time_budget=alloc)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  LP {round_idx}: {best_score:.10f} ***")
            else:
                print(f"  LP {round_idx}: {obj:.10f}")

            # Follow-up L-BFGS-B after LP
            time_left = deadline - time.monotonic()
            if time_left > 30:
                alloc2 = min(time_left * 0.15, 30.0)
                a_post = lbfbsb_progressive(a_lp, [(200, 1000), (1000, 3000)],
                                           time_budget=alloc2)
                obj_post = compute_obj(a_post)
                if obj_post < best_score:
                    best_score = obj_post
                    best_seq = a_post.copy()
                    print(f"  LP+LBFGS {round_idx}: {best_score:.10f} ***")

    # === Phase 4: Peak-targeted coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Peak-targeted coordinate descent")
    print(f"{'='*60}")

    if best_seq is not None:
        time_left = deadline - time.monotonic()
        if time_left > 30:
            alloc = min(time_left * 0.1, 50.0)
            a_shaved = peak_gradient_descent(best_seq, time_budget=alloc,
                                            n_peaks=3, n_steps=500)
            obj = compute_obj(a_shaved)
            if obj < best_score:
                best_score = obj
                best_seq = a_shaved.copy()
                print(f"  Peak descent: {best_score:.10f} ***")
            else:
                print(f"  Peak descent: {obj:.10f}")

            # L-BFGS-B polish after peak descent
            time_left = deadline - time.monotonic()
            if time_left > 20:
                alloc2 = min(time_left * 0.15, 25.0)
                a_fine = lbfbsb_progressive(best_seq, [(500, 3000)],
                                           time_budget=alloc2)
                obj_fine = compute_obj(a_fine)
                if obj_fine < best_score:
                    best_score = obj_fine
                    best_seq = a_fine.copy()
                    print(f"  Post-peak LBFGS: {best_score:.10f} ***")

    # === Phase 5: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        # Try a range of n values
        n_candidates = sorted(set([
            n_best - 100, n_best - 50, n_best,
            n_best + 50, n_best + 100, n_best + 150,
            n_best + 250, n_best + 400, n_best + 600
        ]))
        n_candidates = [n for n in n_candidates if 150 <= n <= 2500]

        for n_try in n_candidates:
            time_left = deadline - time.monotonic()
            if time_left < 20:
                break

            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample_seq(best_seq, n_try)

            alloc = min(time_left * 0.12, 40.0)
            schedule = [(10, 100), (100, 500), (500, 2000)]
            a_opt = lbfbsb_progressive(a_try, schedule, time_budget=alloc)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.10f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.10f}")

    # === Phase 6: Random perturbation restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Perturbation restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        for restart in range(6):
            time_left = deadline - time.monotonic()
            if time_left < 20:
                break

            if restart % 2 == 0:
                # Multiplicative noise
                scale = 0.02 + 0.01 * restart
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_cur))
            else:
                # Blend with structured
                alpha = 0.1 + 0.05 * restart
                struct = generate_structured(n_cur, 'gauss_left' if restart % 4 == 1 else 'exp_decay')
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * struct

            a_pert = np.maximum(a_pert, 1e-15)
            a_pert = a_pert / np.sum(a_pert)

            alloc = min(time_left * 0.1, 30.0)
            a_opt = lbfbsb_progressive(a_pert, [(50, 500), (500, 2000)],
                                       time_budget=alloc)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.10f} *** UPDATE")
            else:
                print(f"  Restart {restart}: {obj:.10f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish (n={len(best_seq) if best_seq is not None else '?'})")
    print(f"{'='*60}")

    if best_seq is not None:
        # Very high beta L-BFGS-B
        time_left = deadline - time.monotonic()
        if time_left > 15:
            alloc = min(time_left * 0.3, 40.0)
            a_final = lbfbsb_progressive(best_seq, [(1000, 5000), (5000, 10000)],
                                        time_budget=alloc)
            obj = compute_obj(a_final)
            if obj < best_score:
                best_score = obj
                best_seq = a_final.copy()
                print(f"  Final LBFGS: {best_score:.10f} ***")
            else:
                print(f"  Final LBFGS: {obj:.10f}")

        # Final LP
        if len(best_seq) <= 900:
            time_left = deadline - time.monotonic()
            if time_left > 15:
                alloc = min(time_left * 0.3, 25.0)
                a_lp = lp_refine(best_seq, time_budget=alloc)
                obj = compute_obj(a_lp)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_lp.copy()
                    print(f"  Final LP: {best_score:.10f} ***")

        # One more peak descent
        time_left = deadline - time.monotonic()
        if time_left > 10:
            alloc = min(time_left * 0.3, 20.0)
            a_ps = peak_gradient_descent(best_seq, time_budget=alloc,
                                        n_peaks=5, n_steps=200)
            obj = compute_obj(a_ps)
            if obj < best_score:
                best_score = obj
                best_seq = a_ps.copy()
                print(f"  Final peak desc: {best_score:.10f} ***")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
```
