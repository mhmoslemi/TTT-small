From my earlier analysis, if a is symmetric, objective >= 2. But if a is asymmetric, the center lag can be below sum^2/n. The question is: can all lags be below sum^2/n simultaneously?

For a[i] = r^i (r < 1, very small r, say r = 0.5, n = 700):
- a is heavily weighted toward index 0.
- conv[0] = a[0]^2 = 1 (dominant)
- conv[n-1] = sum_i a[i]*a[n-1-i] = sum_i r^i * r^{n-1-i} = r^{n-1} * n (very small for large n)
- The max is at conv[0] = a[0]^2.

So for extreme asymmetry, the peak is at the edge (conv[0] or conv[2n-2]).

The optimal is somewhere in between: enough asymmetry to reduce the center peak, but not so much that the edge peaks dominate.

For a[i] = r^i, the peak transitions from center to edge as r decreases. The optimal r (for a given n) balances these.

This gives a 1-parameter family of solutions. The previous solver probably found something in this neighborhood but with more complex structure.

"""
L-BFGS optimization of power-mean smoothed convolution peak objective,
combined with geometric initialization, multi-n exploration, and 
coordinate descent polishing. Targets improvement over 1.506548.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def power_mean_obj_grad(a, p):
    """
    Compute power-mean smoothed objective and gradient.
    f(a) = 2n * (sum_k conv[k]^p)^(1/p) / (sum(a))^2
    """
    n = len(a)
    conv = compute_conv(a)
    conv = np.maximum(conv, 0)  # ensure non-negative
    S = float(np.sum(a))
    
    conv_p = conv ** p
    T = float(np.sum(conv_p))
    if T <= 0:
        return 0.0, np.zeros(n)
    S_p = T ** (1.0 / p)
    
    obj = 2.0 * n * S_p / (S * S)
    
    # Gradient
    # dS_p/da_i = 2 * S_p^(1-p) * sum_k conv[k]^(p-1) * a[k-i]
    h = conv ** (p - 1)  # length 2n-1
    
    # cross-correlation: cc[i] = sum_k h[k] * a[k-i] = (h ⋆ a)[i]
    # = ifft(FFT(h) * conj(FFT(a)))[:n]
    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    H = rfft(h, size)
    A = rfft(a, size)
    cc = irfft(H * np.conj(A), size)[:n]
    
    # dS_p/da_i / S_p = 2 * S_p^(-p) * cc[i] = 2 * cc[i] / T
    grad_Sp_over_Sp = 2.0 * cc / T
    
    grad_f = (2.0 * n / (S * S)) * (grad_Sp_over_Sp - 1.0)
    
    return obj, grad_f


def lbfgs_optimize(a, p_start=100.0, p_end=10000.0, n_restarts=3, 
                   time_budget=60.0, max_iter=2000):
    """
    Optimize using L-BFGS with decreasing p (increasing sharpness).
    """
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S  # normalize
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Schedule of p values (increasing sharpness)
    p_values = []
    for frac in np.linspace(0, 1, n_restarts):
        p_val = p_start * (p_end / p_start) ** frac
        p_values.append(p_val)
    
    current_a = a.copy()
    
    for p in p_values:
        if time.monotonic() - start > time_budget:
            break
        
        def func_grad(x):
            x = np.maximum(x, 0)
            obj, grad = power_mean_obj_grad(x, p)
            return obj, grad
        
        try:
            res = minimize(func_grad, current_a, method='L-BFGS-B',
                          jac=True, bounds=[(0, None)] * n,
                          options={'maxiter': max_iter, 'ftol': 1e-12, 
                                   'gtol': 1e-8, 'maxls': 50})
            if res.success or res.fun < compute_obj(current_a):
                new_a = np.maximum(res.x, 0)
                S_new = np.sum(new_a)
                if S_new > 1e-10:
                    new_a = new_a / S_new
                    new_obj = compute_obj(new_a)
                    if new_obj < best_obj:
                        best_obj = new_obj
                        best_a = new_a.copy()
                    current_a = new_a.copy()
        except Exception as e:
            pass
    
    return best_a

"""
L-BFGS power-mean smoothed optimization with geometric initialization,
LP natural gradient, multi-n exploration, and coordinate descent polish.
Uses continuation in smoothing parameter p for sharp convergence to the true max.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def power_mean_obj_grad(a, p):
    """
    Compute power-mean smoothed objective and gradient.
    f(a) = 2n * (sum_k conv[k]^p)^(1/p) / (sum(a))^2
    """
    n = len(a)
    conv = compute_conv(a)
    conv = np.maximum(conv, 1e-30)
    S = float(np.sum(a))
    if S < 1e-10:
        return 1e10, np.zeros(n)

    conv_p = conv ** p
    T = float(np.sum(conv_p))
    if T <= 0:
        return 0.0, np.zeros(n)
    S_p = T ** (1.0 / p)

    obj = 2.0 * n * S_p / (S * S)

    h = conv ** (p - 1)

    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    H = rfft(h, size)
    A = rfft(a, size)
    cc = irfft(H * np.conj(A), size)[:n]

    grad_Sp_over_Sp = 2.0 * cc / T
    grad_f = (2.0 * n / (S * S)) * (grad_Sp_over_Sp - 1.0)

    return obj, grad_f


def logsumexp_obj_grad(a, beta):
    """Compute logsumexp smoothed objective and gradient."""
    n = len(a)
    conv = compute_conv(a)
    conv = np.maximum(conv, 1e-30)
    S = float(np.sum(a))
    if S < 1e-10:
        return 1e10, np.zeros(n)

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (S * S)

    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)[:n]

    grad_ls = 2.0 * cc
    grad_f = (2.0 * n / (S * S)) * (grad_ls - ls / S)

    return obj, grad_f


def lbfgs_optimize(a, p_schedule=None, time_budget=60.0, max_iter=3000):
    """L-BFGS with continuation in p."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if p_schedule is None:
        p_schedule = [20, 50, 100, 200, 500, 1000, 3000, 8000, 20000]

    for p in p_schedule:
        if time.monotonic() - start > time_budget:
            break

        def func_grad(x):
            x = np.maximum(x, 1e-30)
            obj, grad = power_mean_obj_grad(x, p)
            grad = np.maximum(grad, -1e6)
            grad = np.minimum(grad, 1e6)
            return obj, grad

        try:
            res = minimize(func_grad, a, method='L-BFGS-B',
                          jac=True, bounds=[(1e-30, None)] * n,
                          options={'maxiter': max_iter, 'ftol': 1e-14,
                                   'gtol': 1e-10, 'maxls': 100})
            new_a = np.maximum(res.x, 0)
            S_new = np.sum(new_a)
            if S_new > 1e-10:
                new_a = new_a / S_new
                new_obj = compute_obj(new_a)
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_a = new_a.copy()
                a = new_a.copy()
        except Exception:
            pass

    return best_a


def lbfgs_logspace_optimize(a, beta_schedule=None, time_budget=60.0, max_iter=3000):
    """L-BFGS with logsumexp smoothing and continuation in beta."""

        except Exception:
            pass

def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_direction(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_natural_gradient_step(a, time_budget=30.0):
    """LP-based direction finding with line search."""
    n = len(a)
    start = time.monotonic()
    S = np.sum(a)
    if S < 1e-10:
        return a
    a_norm = a / S

    conv = compute_conv(a_norm)
    M = float(np.max(conv))

    g = lp_direction(a_norm, M, time_limit=min(time_budget * 0.4, 15))
    if g is None:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)

    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        if time.monotonic() - start > time_budget:
            break
        a_try = (1 - t) * a_norm + t * g
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def geometric_sequence(n, r):
    """Create normalized geometric sequence a[i] = r^i."""
    a = r ** np.arange(n, dtype=np.float64)
    s = np.sum(a)
    return a / s


def cosine_sequence(n):
    """Create cosine-tapered sequence."""
    a = np.cos(np.pi * np.arange(n) / (2.0 * (n - 1))) ** 2
    a = np.maximum(a, 0)
    s = np.sum(a)
    return a / s


def exponential_sequence(n, alpha):
    """Create exponential decay sequence a[i] = exp(-alpha*i)."""
    a = np.exp(-alpha * np.arange(n, dtype=np.float64))
    s = np.sum(a)
    return a / s


def coordinate_descent(a, time_budget=30.0, seed=42):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    for outer in range(50):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        top_peaks = np.argsort(conv)[::-1][:3]

        improved = False
        for p in top_peaks:
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            for i in range(i_min, i_max + 1):
                if time.monotonic() - start > time_budget:
                    break
                j = p - i
                if j < 0 or j >= n or j == i:
                    continue
                # Try transferring mass from i to reduce peak
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    delta = frac * best_a[i]
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
        if not improved:
            # Try random perturbations
            idx1 = rng.integers(0, n)
            idx2 = rng.integers(0, n)
            if idx1 != idx2:
                for frac in [0.005, 0.01, 0.02]:
                    delta = frac * best_a[idx1]
                    a_try = best_a.copy()
                    a_try[idx1] -= delta
                    a_try[idx2] += delta
                    if a_try[idx1] < 0:
                        continue
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
            if not improved and outer > 10:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

            if time.monotonic() < deadline - 60:
                a_post = lbfgs_optimize(a_lp, p_schedule=[200, 1000, 5000],
                                        time_budget=min(20.0, (deadline - time.monotonic()) * 0.05),
                                        max_iter=2000)
                obj_post = compute_obj(a_post)
                if obj_post < best_score:
                    best_score = obj_post
                    best_seq = a_post.copy()
                    print(f"  LP+L-BFGS {i}: {best_score:.8f} ***")

    # === Phase 3: Geometric and structured initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Structured initializations")
    print(f"{'='*60}")

    n_options_geo = [n_parent, n_parent + 50, n_parent + 100, n_parent + 200]
    n_options_geo = [n for n in n_options_geo if 200 <= n <= 2000]

    for n_try in n_options_geo:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Try several geometric ratios
        for r in [0.999, 0.9995, 0.998, 0.995, 0.99, 0.98]:
            if time.monotonic() > deadline - 80:
                break
            a_geo = geometric_sequence(n_try, r)
            alloc = min(remaining * 0.08, 30.0)
            a_opt = lbfgs_optimize(a_geo, p_schedule=[50, 200, 1000, 5000],
                                   time_budget=alloc, max_iter=3000)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Geo n={n_try}, r={r}: {best_score:.8f} ***")
            else:
                print(f"  Geo n={n_try}, r={r}: {obj:.8f}")

    # Try exponential sequences
    for n_try in [n_parent, n_parent + 100]:
        if time.monotonic() > deadline - 80:
            break
        for alpha in [0.005, 0.01, 0.02, 0.05]:
            if time.monotonic() > deadline - 60:
                break
            a_exp = exponential_sequence(n_try, alpha)
            alloc = min(15.0, (deadline - time.monotonic()) * 0.05)
            a_opt = lbfgs_optimize(a_exp, p_schedule=[100, 500, 3000],
                                   time_budget=alloc, max_iter=2000)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Exp n={n_try}, a={alpha}: {best_score:.8f} ***")
            else:
                print(f"  Exp n={n_try}, a={alpha}: {obj:.8f}")

    # Try cosine
    if time.monotonic() < deadline - 60:
        a_cos = cosine_sequence(n_parent)
        alloc = min(20.0, (deadline - time.monotonic()) * 0.05)
        a_opt = lbfgs_optimize(a_cos, p_schedule=[50, 300, 2000],
                               time_budget=alloc, max_iter=2000)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Cosine: {best_score:.8f} ***")
        else:
            print(f"  Cosine: {obj:.8f}")

    # === Phase 4: Logsumexp L-BFGS on best so far ===
    print(f"\n{'='*60}")
    print(f"Phase 4: L-BFGS logsumexp on best (n={len(best_seq)})")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        a_ls = best_seq / np.sum(best_seq)
        alloc = min(remaining * 0.15, 150.0)
        a_opt = lbfgs_logspace_optimize(a_ls,
                                        beta_schedule=[20, 60, 150, 400, 1000, 3000, 8000],
                                        time_budget=alloc, max_iter=5000)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Logsumexp L-BFGS: {best_score:.8f} ***")
        else:
            print(f"  Logsumexp L-BFGS: {obj:.8f}")

    # === Phase 5: Multi-n exploration (resample + optimize) ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best,
        n_best + 50, n_best + 100, n_best + 200, n_best + 400
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2500]

    for n_try in n_options:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        S_t = np.sum(a_try)
        if S_t > 1e-10:
            a_try = a_try / S_t

        alloc = min(remaining * 0.12, 40.0)
        a_opt = lbfgs_optimize(a_try, p_schedule=[50, 300, 1500, 8000],
                               time_budget=alloc, max_iter=3000)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  n={n_try}: {obj:.8f}")

    # === Phase 6: Random restarts with perturbation ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Random restarts")
    print(f"{'='*60}")

    rng = np.random.default_rng(seed)
    n_best = len(best_seq)

    for restart in range(6):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if restart % 3 == 0:
            scale = 0.02 + 0.01 * restart
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
        elif restart % 3 == 1:
            a_pert = best_seq.copy()
            for _ in range(max(1, n_best // 100)):
                idx = rng.integers(0, n_best)
                a_pert[idx] += rng.uniform(0.01, 0.05) * np.mean(best_seq)
        else:
            # Blend with geometric
            alpha = rng.uniform(0.1, 0.3)
            r = rng.uniform(0.99, 0.999)
            geo = geometric_sequence(n_best, r)
            a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * geo

        a_pert = np.maximum(a_pert, 1e-15)
        S_p = np.sum(a_pert)
        if S_p < 1e-10:
            continue
        a_pert = a_pert / S_p

        alloc = min(remaining * 0.1, 25.0)
        a_opt = lbfgs_optimize(a_pert, p_schedule=[50, 300, 2000, 10000],
                               time_budget=alloc, max_iter=3000)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  Restart {restart}: {obj:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish (n={len(best_seq)})")
    print(f"{'='*60}")

    a = best_seq / np.sum(best_seq)

    # High-p L-BFGS
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.25, 100.0)
        a_high = lbfgs_optimize(a, p_schedule=[500, 2000, 8000, 30000, 100000],
                                time_budget=alloc, max_iter=5000)
        obj = compute_obj(a_high)
        if obj < best_score:
            best_score = obj
            best_seq = a_high.copy()
            a = a_high.copy()
            print(f"  High-p L-BFGS: {best_score:.8f} ***")
        else:
            print(f"  High-p L-BFGS: {obj:.8f}")

    # Logsumexp high-beta
    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 60.0)
        a_ls = lbfgs_logspace_optimize(a,
                                       beta_schedule=[500, 2000, 8000, 20000],
                                       time_budget=alloc, max_iter=4000)
        obj = compute_obj(a_ls)
        if obj < best_score:
            best_score = obj
            best_seq = a_ls.copy()
            print(f"  High-beta LS: {best_score:.8f} ***")
        else:
            print(f"  High-beta LS: {obj:.8f}")

    # Coordinate descent
    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 40.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, seed=seed)
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  Coord descent: {best_score:.8f} ***")
        else:
            print(f"  Coord descent: {obj:.8f}")

    # Final micro-polish
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 30.0)
        a_final = lbfgs_optimize(best_seq / np.sum(best_seq),
                                 p_schedule=[5000, 20000, 100000],
                                 time_budget=alloc, max_iter=3000)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final micro: {best_score:.8f} ***")
        else:
            print(f"  Final micro: {obj:.8f}")

    # Final validation
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
