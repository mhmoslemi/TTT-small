Actually, let me simplify. Instead of a full golden section, let me do a "zigzag + binary search" approach:
1. Evaluate at x=0, x=old, x=2*old (or x=0.05 if old is small)
2. Find the best of these three
3. If the best is interior, do binary search refinement (5-10 iterations)

This is simpler and more robust.

Actually, even simpler: let me just evaluate at 10-15 points in [0, 2*old+epsilon] and pick the best. For n=768, this means 10-15 evaluations per coordinate, each O(n) for the max computation. def coord_descent_simple(a, deadline, n_evals=12, seed=42, verbose=False):
    """Coordinate descent with uniform grid evaluation per coordinate."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n); s = 1.0
    else:
        a = a / s; s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    sweep = 0
    while time.monotonic() < deadline:
        sweep += 1
        if sweep % 5 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break
            old = a[i]
            # Candidate x values in [0, 2*old + 0.05]
            hi = 2.0 * old + 0.05
            candidates = np.linspace(0, hi, n_evals)
            
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            kq = 2 * i
            kq_valid = kq < 2 * n - 1
            
            # Pre-compute the "unchanged" max
            unchanged_max = 0.0
            if k_lo > 0:
                unchanged_max = max(unchanged_max, conv[:k_lo].max())
            if k_hi < 2 * n - 2:
                unchanged_max = max(unchanged_max, conv[k_hi+1:].max())
            
            # a[k-i] for k in [k_lo, k_hi]
            a_slice = a[k_lo - i:k_hi - i + 1]  # a[0] to a[k_hi-i]
            
            best_x = old
            best_f = 2.0 * n * conv.max() / (s * s)
            
            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                new_s = s - old + x
                if new_s < 1e-10:
                    continue
                # Compute new max in affected range
                changed = conv[k_lo:k_hi+1] + 2.0 * delta * a_slice
                if kq_valid and kq >= k_lo and kq <= k_hi:
                    changed = changed.copy()
                    changed[kq - k_lo] += delta * delta
                new_max = max(unchanged_max, changed.max())
                f = 2.0 * n * new_max / (new_s * new_s)
                if f < best_f:
                    best_f = f
                    best_x = x
            
            if best_f < best_obj - 1e-14 and abs(best_x - old) > 1e-12:
                delta = best_x - old
                conv[k_lo:k_hi+1] += 2.0 * delta * a_slice
                if kq_valid:
                    conv[kq] += delta * delta
                a[i] = best_x
                s += delta
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            break
        if verbose and sweep % 50 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}", flush=True)

    return best_a, best_obj
