"""
Primal-dual multiplicative weight + refined coordinate descent + LP escape
for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Key innovation: exploits the KKT structure (cross-correlation of softmax weights
and a should be constant on support) via Sinkhorn-like multiplicative updates.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = mu - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def compute_cross_corr(w, a):
    """Compute s[j] = sum_k w[k] * a[k-j] (valid indices only)."""
    n = len(a)
    nw = len(w)
    w_rev = w[::-1]
    size = 1
    while size < n + nw - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    Wr = np.fft.rfft(w_rev, size)
    full_conv = np.fft.irfft(A * Wr, size)
    s = full_conv[2*n-2 : n-2 : -1]
    return s


def primal_dual_iterate(a, tau_schedule, alpha_schedule, max_iters, deadline, verbose=False):
    """
    Sinkhorn-like multiplicative weight updates.
    At each step: compute softmax weights on conv, compute cross-correlation,
    apply multiplicative update a <- a * (M/s)^alpha, normalize.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_schedule(frac)
        alpha = alpha_schedule(frac)

        conv = fast_conv(a)
        M = conv.max()

        # Compute softmax weights (focus on peak)
        exp_vals = np.exp(tau * (conv - M))
        w = exp_vals / exp_vals.sum()

        # Cross-correlation: s[j] = sum_k w[k] * a[k-j]
        s = compute_cross_corr(w, a)

        # Multiplicative update: a * (M / s)^alpha
        # Clip to avoid extreme values
        ratio = M / (s + 1e-15)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_new = a * (ratio ** alpha)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if verbose and it % 50 == 0:
            print(f"    PD it {it}: tau={tau:.1f} alpha={alpha:.3f} obj={best_obj:.8f} no_imp={no_improve}", flush=True)

        if no_improve > 200:
            break

    return best_a, best_obj


def refined_coord_descent(a, max_sweeps, deadline, verbose=False):

    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    no_improve = 0
    max_no_improve = 400
    rng = np.random.default_rng(12345)

    def eval_x(x, old, i, conv_local, s_local):
        delta = x - old
        if abs(delta) < 1e-15:
            return 2.0 * n * conv_local.max() / (s_local * s_local)
        lo = max(0, i)
        hi = min(2 * n - 2, i + n - 1)
        a_slice = a[lo - i:hi - i + 1].copy()
        convlocal = conv_local.copy()
        convlocal[lo:hi+1] += 2.0 * delta * a_slice
        kq = 2 * i
        if kq <= 2 * n - 2:
            convlocal[kq] += delta * delta
        ns = s_local - old + x
        if ns < 1e-10:
            return float('inf')
        return 2.0 * n * convlocal.max() / (ns * ns)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n) if sweep % 3 == 0 else range(n)
        improved = False

        for i in order:
            old = a[i]
            if old < 1e-14:
                candidates = [0.0, 1e-5, 1e-4, 1e-3]
            else:
                # Golden section search on [0, 2.5*old]
                lo_b = 0.0
                hi_b = 2.5 * old
                # Evaluate at golden ratio points
                gr = (np.sqrt(5) - 1) / 2
                x1 = lo_b + (1 - gr) * (hi_b - lo_b)
                x2 = lo_b + gr * (hi_b - lo_b)
                f1 = eval_x(x1, old, i, conv, s)
                f2 = eval_x(x2, old, i, conv, s)
                # Also check original and boundary
                f0 = 2.0 * n * conv.max() / (s * s)
                f_lo = eval_x(0.0, old, i, conv, s)
                # Pick 2 best regions and refine
                candidates = [0.0, old, x1, x2, 0.5*old, 1.5*old]
                # Evaluate all
                best_val = old
                best_f = f0
                for x in candidates:
                    f = eval_x(x, old, i, conv, s)
                    if f < best_f:
                        best_f = f
                        best_val = x
                # If interior point won, do a few more evaluations nearby
                if 0.01 < best_val/old < 0.99 and best_f < f0 - 1e-10:
                    for frac in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
                        x = best_val * frac
                        f = eval_x(x, old, i, conv, s)
                        if f < best_f:
                            best_f = f
                            best_val = x

            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                lo = max(0, i)
                hi = min(2 * n - 2, i + n - 1)
                a_slice = a[lo - i:hi - i + 1]
                conv[lo:hi+1] += 2.0 * delta * a_slice
                kq = 2 * i
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-12:

            no_improve = 0
        else:

    a = np.maximum(a, 0)

def lp_escape(a, deadline, max_iters=50):
    """LP-based direction finding to escape local minima."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve_local = 0
                            found = True
                            break
                    if not found:
                        return best_a, best_obj
                else:
                    return best_a, best_obj
            else:
                return best_a, best_obj
        except Exception:
            return best_a, best_obj

    return best_a, best_obj


def gradient_ascent_on_neg(a, deadline, max_iters=2000, lr_init=0.02):
    """
    Projected gradient descent with log-sum-exp smoothing.
    Uses the subgradient structure: grad = weighted cross-correlation - mean.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    tc = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = 20.0 + 80.0 * frac

        conv = fast_conv(a)
        M = conv.max()
        exp_vals = np.exp(tau * (conv - M))
        w = exp_vals / exp_vals.sum()

        # Gradient: s[j] = cross_corr(w, a), then grad = s - mean(s)
        s = compute_cross_corr(w, a)
        grad = s - np.mean(s)

        tc += 1
        lr = lr_init / (1.0 + 0.05 * tc ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** tc)
        v_hat = v / (1 - beta2 ** tc)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                rng = np.random.default_rng(it * 31 + 7)
                a = best_a + rng.normal(0, 0.03, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                tc = 0
                no_improve = 0

        a = a_new

    return best_a, best_obj


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))
    inits.append(np.ones(n) / n)
    for freq in [1, 2, 3, 5, 8]:
        inits.append(project_simplex(1.0 + 0.5 * np.cos(2 * np.pi * freq * t)))
        inits.append(project_simplex(1.0 + 0.3 * np.sin(2 * np.pi * freq * t)))
    for center in [0.3, 0.5, 0.7]:
        inits.append(project_simplex(np.exp(-((t - center) / 0.15) ** 2) + 0.05))
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))
    return inits


def solve_pipeline(a, deadline, verbose=False):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new, tag=""):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return a_new
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj - 1e-14:
            best_a = a_new.copy()
            best_obj = obj
            if verbose:
                print(f"      [{tag}] J={obj:.10f}", flush=True)
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 1: Primal-dual multiplicative updates (30% time)
    t1 = time.monotonic() + 0.30 * remaining
    if verbose:
        print(f"    PD stage (budget {t1-now_time():.0f}s)", flush=True)
    tau_sched = lambda f: 30.0 + 70.0 * f
    alpha_sched = lambda f: 0.8 * (1.0 - 0.5 * f)
    a_pd, obj_pd = primal_dual_iterate(a, tau_sched, alpha_sched, 5000, t1, verbose=verbose)
    a = try_update(a_pd, "PD")

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 2: Coordinate descent (30% time)
    t2 = time.monotonic() + 0.30 * remaining
    if verbose:
        print(f"    CD stage (budget {t2-now_time():.0f}s)", flush=True)
    a_cd = refined_coord_descent(best_a, 5000, t2, verbose=verbose)
    a = try_update(a_cd, "CD")

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 3: LP escape (20% time)
    t3 = time.monotonic() + 0.20 * remaining
    if verbose:
        print(f"    LP stage (budget {t3-now_time():.0f}s)", flush=True)
    a_lp, obj_lp = lp_escape(best_a, t3, max_iters=80)
    a = try_update(a_lp, "LP")

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 4: Gradient with smoothing (15% time)
    t4 = time.monotonic() + 0.15 * remaining
    if verbose:
        print(f"    Grad stage (budget {t4-now_time():.0f}s)", flush=True)
    a_g, obj_g = gradient_ascent_on_neg(best_a, t4, max_iters=3000, lr_init=0.03)
    a = try_update(a_g, "Grad")

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 5: Final CD polish (5% time)
    t5 = deadline - 2
    a_final = refined_coord_descent(best_a, 3000, t5)
    a = try_update(a_final, "FinalCD")

    return best_a, best_obj


def now_time():
    return time.monotonic()


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search using primal-dual multiplicative updates,
    refined coordinate descent, LP escape, and gradient methods.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  >> NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)
    parent_norm = parent / parent.sum()

    # === PHASE 1: Intensive optimization at parent length ===
    phase1_end = t_start + 400.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Optimize at n={n_parent}", flush=True)
    print(f"{'='*60}", flush=True)

    inits = generate_inits(n_parent, rng, parent_norm)
    n_init = min(len(inits), 6)
    per_time = (phase1_end - time.monotonic()) / max(1, n_init) - 3

    for idx in range(n_init):
        rem = phase1_end - time.monotonic()
        if rem < 20:
            break
        alloc = min(per_time, rem - 3)
        if alloc < 10:
            break

        a0 = inits[idx].copy()
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt, obj_opt = solve_pipeline(a0, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Try nearby lengths ===
    phase2_end = t_start + 900.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    # Prioritize lengths: parent, then multiples and nearby values
    candidate_ns = []
    for mult in [0.75, 1.25, 1.5, 2.0]:
        candidate_ns.append(int(n_parent * mult))
    for offset in [-64, -32, 32, 64, 128, -128]:
        candidate_ns.append(n_parent + offset)
    candidate_ns.extend([512, 640, 768, 896, 1024, 1280, 1536])
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2560 and n != n_parent]
    candidate_ns = sorted(set(candidate_ns), key=lambda x: abs(x - n_parent))

    for new_n in candidate_ns[:6]:
        rem = phase2_end - time.monotonic()
        if rem < 50:
            break
        alloc = min(80.0, rem - 20)
        if alloc < 30:
            break

        print(f"  n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent_norm, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = solve_pipeline(a_ext, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Focused optimization on best ===
    phase3_end = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Focused optimization (budget {phase3_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        rem = phase3_end - time.monotonic()
        if rem > 30:
            n_best = len(best_seq)
            a_curr = best_seq.copy()

            # 3a: Multi-pass primal-dual with different schedules
            t_3a = time.monotonic() + 0.25 * rem
            for pi in range(3):
                tau_s = lambda f, pi=pi: (20.0 + pi * 20.0) + 60.0 * f
                alpha_s = lambda f, pi=pi: max(0.2, 0.7 - pi * 0.15) * (1.0 - 0.3 * f)
                a_pd2, obj_pd2 = primal_dual_iterate(a_curr, tau_s, alpha_s, 3000, t_3a, verbose=False)
                update_best(a_pd2)
                a_curr = best_seq.copy()

            rem = phase3_end - time.monotonic()
            if rem > 30:
                # 3b: Long coordinate descent
                t_3b = time.monotonic() + 0.30 * rem
                a_cd2 = refined_coord_descent(best_seq, 10000, t_3b, verbose=False)
                update_best(a_cd2)

            rem = phase3_end - time.monotonic()
            if rem > 30:
                # 3c: LP escape + re-optimize
                t_3c = time.monotonic() + 0.20 * rem
                a_lp2, obj_lp2 = lp_escape(best_seq, t_3c, max_iters=120)
                update_best(a_lp2)

            rem = phase3_end - time.monotonic()
            if rem > 30:
                # 3d: Gradient method
                t_3d = time.monotonic() + 0.15 * rem
                a_g2, obj_g2 = gradient_ascent_on_neg(best_seq, t_3d, max_iters=5000, lr_init=0.015)
                update_best(a_g2)

            rem = phase3_end - time.monotonic()
            if rem > 20:
                # 3e: Final CD
                t_3e = phase3_end - 3
                a_f = refined_coord_descent(best_seq, 8000, t_3e, verbose=False)
                update_best(a_f)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
