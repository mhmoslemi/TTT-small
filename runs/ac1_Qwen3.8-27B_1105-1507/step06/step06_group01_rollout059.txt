from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def build_full_lp_matrix(a):
    """Build the full (2n-1) x n matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            m = k - j
            if 0 <= m < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response_full(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g or None."""
    n = len(a)
    A = build_full_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def fast_conv(a):

    fa = np.fft.rfft(a, size)

def compute_J(a):

    a = np.maximum(a, 0.0)
    s = a.sum()

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        if not (~cond).any():
            rho = n - 1
        else:
            rho = int(np.argmax(~cond))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

            rows.append(k)
            cols.append(j)

    k = 2 * n - 1
    vals = a[rows - cols]  # A[k,j] = a[k-j]
    A = csr_matrix((vals, (rows, cols)), shape=(k, n))
    return A


def lp_best_response(a, M, rows, cols):
    """Full LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a, rows, cols)

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:

            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs, gs
    except Exception:
        pass
    return None, 0.0

Let me also think about the connection to signal processing. We want a "good" autocorrelation sequence - one with low peaks. This is the "low PAPR" problem (peak-to-average power ratio) in communications, which is well-studied.

Algorithms for low PAPR:
- **GM (Gibbs-Monotone) algorithm**: Successive over-relaxation on the phase (for complex sequences).
- **FL (Frank-Lens) algorithm**: Gradient descent on the phase.
- **Selected Mapping (SLM)**: Try multiple phase rotations, pick the best.
- **Tx Active Phase Control (Tx-APC)**: Optimize the phase of individual subcarriers.

For our real non-negative problem, the analog would be:
- Gradient descent on a (with projection to non-negative simplex)
- Coordinate descent (optimize one a_i at a time)
- Random restarts

The GM algorithm is particularly interesting. It works by:
1. Compute the autocorrelation b = a*a.
2. Find the peak b[k*].
3. Update a to reduce b[k*]: a_i ← a_i - μ * b[k*] * a[k*-i] (for the relevant i).
4. This monotonically decreases the peak.

This is essentially the "subgradient" approach but with a specific step size that guarantees monotonic decrease.

Let me implement a GM-like algorithm:

```python
def gm_iteration(a, mu=0.5):
    """One GM iteration: reduce the peak autocorrelation."""
    n = len(a)
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    b_max = b[k_star]
    
    # Subgradient direction: reduce a where it contributes to b[k_star]
    # (a*a)[k_star] = sum_i a_i * a[k_star-i]
    # To reduce this, we want to decrease a_i where a[k_star-i] is large
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    
    # Project gradient to tangent of simplex
    grad_tan = grad - grad.mean()
    
    # Update
    a_new = a - mu * grad_tan
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()
    
    return a_new
```

import numpy as np

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        if not (~cond).any():
            rho = n - 1
        else:
            rho = int(np.argmax(~cond))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def get_lp_sparsity(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def build_lp(a, rows, cols):
    n = len(a)
    vals = a[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, rows, cols, time_limit=4):
    n = len(a)
    A = build_lp(a, rows, cols)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def smooth_max_val_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    corr = np.correlate(p, a, 'full')
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return smooth_val, grad


def run_fista(a0, tau_start, tau_end, max_iters, deadline, lr=0.4):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters - 1))
        tau = tau_start * (tau_end / tau_start) ** frac
        _, grad_y = smooth_max_val_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        step = lr / (1.0 + 0.03 * tau * it / max_iters)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        a_prev = a
        a = a_new
        y = y_new
        t = t_new

            if M_cur < best_M:

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gen_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.05, n), 1e-4)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.15, n), 1e-4)
        inits.append(p_n2 / p_n2.sum())
    inits.append(np.ones(n) / n)
    poly = t * (1 - t) + 0.05
    inits.append(poly / poly.sum())
    for freq in [1, 2, 3]:
        cw = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cw = np.maximum(cw, 0.01)
        inits.append(cw / cw.sum())
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    return inits

    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - reserve:
            break
        remaining = deadline - reserve - time.monotonic()
        if remaining < 30:
            break
        time_alloc = min(time_per, remaining / max(1, n_len - li))
        print(f"\n  n={n}, alloc={time_alloc:.0f}s", flush=True)
        inits = gen_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_M_n = float('inf')
        n_inits = min(len(inits), max(3, int(time_alloc / 40)))
        time_per_init = time_alloc / max(1, n_inits)
        for ii in range(n_inits):
            if time.monotonic() > deadline - reserve:
                break
            sub_dl = min(time.monotonic() + time_per_init, deadline - reserve)
            if time.monotonic() >= sub_dl:
                break
            a0 = np.maximum(inits[ii], 1e-6)
            a0 = a0 / a0.sum()
            a_opt = optimize_sequence(a0, n, sub_dl, seed=seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()
            if M_opt < best_M_n:
                best_M_n = M_opt
                best_for_n = a_opt.copy()
        if best_for_n is not None:
            J_v = compute_J(best_for_n)
            print(f"  n={n}: M={best_M_n:.10f} J={J_v:.6f}", flush=True)
            tracker.update(best_for_n)

    tracker.report("After Phase 1")

    if tracker.best is not None and time.monotonic() < deadline - 60:
        n_b = tracker.best.size
        rows, cols = get_lp_sparsity(n_b)
        print(f"\n  Phase 2: Intensive LP polish (n={n_b})", flush=True)
        t_lp = min(deadline - 30, time.monotonic() + 180)
        a_lp = lp_iterate(tracker.best, rows, cols, 300, t_lp)
        tracker.update(a_lp)
        if time.monotonic() < deadline - 30:
            a_lp2 = lp_iterate(a_lp, rows, cols, 100, min(deadline - 20, time.monotonic() + 80))
            tracker.update(a_lp2)

    tracker.report("After Phase 2")

    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_b = tracker.best.size
        print(f"\n  Phase 3: FISTA + GM refinement", flush=True)
        t3 = min(deadline - 20, time.monotonic() + 80)
        for tau in [300, 800, 1500]:
            if time.monotonic() > t3:
                break
            a_f = run_fista(tracker.best, tau, tau * 1.5, 3000,
                            min(t3, time.monotonic() + 25), lr=0.2)
            tracker.update(a_f)
        if time.monotonic() < t3:
            a_g = gm_iterate(tracker.best, 3000, t3, mu_init=0.15)
            tracker.update(a_g)

    tracker.report("After Phase 3")

    if tracker.best is not None and time.monotonic() < deadline - 20:
        print(f"\n  Phase 4: Coordinate descent polish", flush=True)
        n_b = tracker.best.size
        max_sw = max(10, int((deadline - 15 - time.monotonic()) / max(0.3, n_b / 80.0)))
        a_cd = coordinate_descent(tracker.best, max_sw, deadline - 10, seed=12345)
        tracker.update(a_cd)
        if time.monotonic() < deadline - 5:
            rows_b, cols_b = get_lp_sparsity(n_b)
            a_final = lp_iterate(a_cd, rows_b, cols_b, 50, deadline)
            tracker.update(a_final)

    tracker.report("After Phase 4")

    if tracker.best is not None and time.monotonic() < deadline - 20:
        n_cur = tracker.best.size
        for mult in [1.2, 0.8]:
            if time.monotonic() > deadline - 15:
                break
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 1536)
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 5, time.monotonic() + 40)
            rows_e, cols_e = get_lp_sparsity(new_n)
            a_opt = optimize_sequence(a_ext, new_n, sub_dl, seed=seed + 555)
            tracker.update(a_opt)

    tracker.report("After Phase 5")

    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
