import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a

def compute_grad_multipeak(a, weights, fft_size):
    """
    Compute gradient of weighted sum of convolution peaks.
    grad[p] = 2 * sum_k weights[k] * a[k-p] (for valid k-p)
    Computed via FFT cross-correlation.
    """
    n = len(a)
    w_ext = np.zeros(fft_size)
    w_ext[:len(weights)] = weights
    W = rfft(w_ext, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    # corr[m] = sum_j a[j]*w_ext[m+j]
    # grad[p] = 2 * sum_k w[k]*a[k-p] = 2 * corr[p] (for p = 0, ..., n-1)

def multipeak_subgradient(a, time_budget=60.0, tau_init=0.01, tau_min=0.0005,

    Uses FFT for fast gradient computation.

    a = normalize_simplex(a.copy())

    while fft_size < 2 * n - 1:
        fft_size <<= 1

    tau = tau_init
    step = step_init
    stall = 0
    best_stall_obj = best_obj

        if time.monotonic() - start > time_budget:
            break

        cmax = float(np.max(conv))

            weights = np.zeros(len(conv))
            weights[idx] = 1.0

            weights /= w_sum

        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break
        
        # Momentum update
        v = momentum * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:

                a = normalize_simplex(a)
                v = np.zeros(n)
                step = step_init
                stall = 0
                best_stall_obj = best_obj
                continue

        # Decay

        tau *= 0.9998
        if tau < tau_min:

def peak_matching_lp(a, M, time_budget=20.0):

"""
Asymmetric multi-peak subgradient optimizer with adaptive temperature scheduling,
FFT-accelerated gradient computation, LP-based refinement, and diversity-preserving
perturbations. Exploits the key insight that optimal sequences must be asymmetric
to achieve objective < 2, by using off-center/skewed initializations and
multi-peak targeting to reduce the maximum autocorrelation value.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

start_time_global = 0.0


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def compute_grad_multipeak(a, weights, fft_size):
    """
    Compute gradient of weighted sum of convolution values.
    grad[p] = 2 * sum_k weights[k] * a[k-p] (for valid k-p in [0,n-1]).
    Uses FFT cross-correlation for efficiency.
    """
    n = len(a)
    w_ext = np.zeros(fft_size)
    w_ext[:len(weights)] = weights
    W = rfft(w_ext, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    return 2.0 * corr[:n]


def multipeak_subgradient(a, time_budget=60.0, tau_init=0.01, tau_min=0.0003,
                          step_init=0.05, momentum=0.85, max_iters=10000):
    """
    Multi-peak subgradient method with adaptive temperature and momentum.
    Temperature controls focus: high tau = flat weights (explore), low tau = peak-focused.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    v = np.zeros(n)
    tau = tau_init
    step = step_init
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        grad = compute_grad_multipeak(a, weights, fft_size)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        v = momentum * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                rng = np.random.default_rng((it * 7919 + n) % (2**31))
                scale = 0.015 + 0.025 * min(stall / 80.0, 1.0)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = normalize_simplex(a)
                v = np.zeros(n)
                step = max(step_init * 0.5, 0.01)
                stall = 0
                continue

        a = a_new
        step *= 0.9997
        tau *= 0.9999
        if tau < tau_min:
            tau = tau_min
        if step < 0.003:
            step = 0.003

    return best_a, best_obj


def targeted_peak_reduction(a, time_budget=30.0, num_target_peaks=5,
                            step_size=0.02, max_iters=2000):
    """
    Directly target the top-K convolution peaks and reduce them via
    gradient steps. More aggressive than soft temperature-based focusing.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    step = step_size

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Find top-K peaks (including near-duplicates within a window)
        sorted_idx = np.argsort(conv)[::-1]
        target_peaks = []
        last_pos = -100
        for idx in sorted_idx:
            if idx - last_pos > 3:
                target_peaks.append(idx)
                last_pos = idx
            if len(target_peaks) >= num_target_peaks:
                break

        weights = np.zeros(len(conv))
        for p in target_peaks:
            # Tapered weight: highest at the actual peak
            rel = conv[p] / cmax
            weights[p] = rel

        if weights.sum() < 1e-10:
            break
        weights /= weights.sum()

        grad = compute_grad_multipeak(a, weights, fft_size)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        a_new = a - step * grad_centered / grad_norm
        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.95
            if step < 0.001:
                break

    return best_a, best_obj


def build_convolve_matrix(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k] for all k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def maxsum_lp(a, M, time_limit=15.0):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns g normalized to sum=1, or None.
    """
    n = len(a)
    A = build_convolve_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def minpeak_lp(a, time_limit=15.0):
    """
    Solve: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g >= 0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(a)
    A = build_convolve_matrix(a)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None

        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        starts = generate_starts(nv, num_starts=8, seed=seed + nv)
        if best_seq is not None and nv == n_base:
            starts.insert(0, best_seq.copy())

        alloc_per = min(remaining / len(starts), 18.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 200:
                break
            t_alloc = min(alloc_per, 15.0)
            if t_alloc < 5:
                break

            a_opt, obj_opt = multipeak_subgradient(s, time_budget=t_alloc,
                                                   tau_init=0.01, tau_min=0.0005,
                                                   step_init=0.05, momentum=0.85,
                                                   max_iters=int(t_alloc * 25))

            # Quick targeted refinement
            if time.monotonic() < deadline - 200:
                a_opt2, obj_opt2 = targeted_peak_reduction(a_opt, time_budget=min(5.0, t_alloc * 0.3),
                                                           num_target_peaks=4, step_size=0.012)
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={nv}, start {si}: {obj_opt:.8f} ** NEW BEST **")

            candidates.append((obj_opt, a_opt.copy(), nv))

    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates: {[f'{c[0]:.6f}' for c in candidates[:6]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 2: Intensive polishing of top candidates
    print("\n=== Phase 2: Intensive polishing ===")

    top_k = min(4, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand = candidates[ci][1].copy()
        obj_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        alloc = min(remaining * 0.3, 100.0)

        print(f"  Candidate {ci} (n={n_cand}, obj={obj_cand:.8f}): polishing with {alloc:.0f}s")

        a_pol, obj_pol = polish_sequence(a_cand, time_budget=alloc)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {obj_pol:.8f} **")
        else:
            print(f"  No improvement (got {obj_pol:.8f})")

    # Phase 3: Length exploration by resampling
    print(f"\n=== Phase 3: Length exploration (best: {best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = sorted(set([x for x in [n_best - 50, n_best + 50,
                                               n_best - 100, n_best + 100,
                                               n_best - 150, n_best + 150]
                                   if 400 <= x <= 1200]))

        for n_test in test_lengths:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_res = normalize_simplex(resample_sequence(best_seq, n_test))
            alloc = min(remaining * 0.15, 30.0)
            a_pol, obj_pol = polish_sequence(a_res, time_budget=alloc)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** UPDATE to {obj_pol:.8f} **")
            else:
                print(f"  n={n_test}: {obj_pol:.8f}")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_final, obj_final = polish_sequence(best_seq, time_budget=remaining * 0.5)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final polish: {obj_final:.8f}")

        # Perturbation-based diversity
        remaining = deadline - time.monotonic()
        if remaining > 30 and best_seq is not None:
            n_fin = len(best_seq)
            rng = np.random.default_rng(seed + 77777)
            for pi in range(8):
                if time.monotonic() > deadline - 15:
                    break
                scale = 0.01 + 0.008 * pi
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_fin))
                a_pert = normalize_simplex(a_pert)

                t_alloc = min(remaining * 0.1, 12.0)
                if t_alloc < 4:
                    break

                a_pert, obj_pert = multipeak_subgradient(a_pert, time_budget=t_alloc,
                                                         tau_init=0.008, tau_min=0.0005,
                                                         step_init=0.04,
                                                         max_iters=int(t_alloc * 20))
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb {pi}: ** {obj_pert:.8f} **")

    # Phase 5: Final LP on best
    if best_seq is not None and time.monotonic() < deadline - 25:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        if n_best <= 800 and remaining > 25:
            print(f"\n=== Phase 5: Final LP (n={n_best}, {remaining:.0f}s remaining) ===")
            a_lp, obj_lp = lp_alternating(best_seq, time_budget=min(remaining * 0.4, 40.0),
                                          method='maxsum')
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  MaxSum LP: {obj_lp:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_lp2, obj_lp2 = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 25.0),
                                                method='minpeak')
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  MinPeak LP: {obj_lp2:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj
