import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize_simplex(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a

def build_toeplitz_mat(a):
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j+n, 2*n-1)
        A[lo:hi, j] = a[:hi-lo]
    return A

def solve_minpeak_lp(fixed_seq, time_limit=10):
    n = len(fixed_seq)
    A = build_toeplitz_mat(fixed_seq)
    c = np.zeros(n+1)
    c[n] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n+1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except:
        pass
    return None, None

"""
Step function optimization using adaptive LP alternating with dual-guided
blending, P-norm annealing for smooth convergence, multi-length exploration,
and aggressive exploitation of the parent solution.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz_mat(a):
    """Build A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=10):
    """Solve: min t s.t. (fixed_seq * g)[k] <= t, sum(g)=1, g>=0."""
    n = len(fixed_seq)
    A = build_toeplitz_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None

    """Solve: max sum(g) s.t. (fixed_seq * g)[k] <= M, g >= 0."""
    n = len(fixed_seq)
    A = build_toeplitz_mat(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',

                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:

                scale = 0.01 + 0.015 * min(stall, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = normalize_simplex(a)
                stall = 0

    return best_a, best_obj


def maxsum_lp_alternate(a, max_iters=40, time_budget=40.0):
    """Max-sum LP alternating as complementary approach."""
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining_budget = time_budget - (time.monotonic() - start)
        if remaining_budget < 3:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_maxsum_lp(a, M, time_limit=min(12, max(3, int(remaining_budget - 2))))
        if g is None:
            stall += 2
            if stall > 6:
                break
            continue

        best_t_val = best_obj
        best_a_local = best_a.copy()

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = normalize_simplex((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_val:
                best_t_val = obj_try
                best_a_local = a_try

        if best_t_val < best_obj - 1e-14:
            best_obj = best_t_val
            best_a = best_a_local
            a = best_a_local
            stall = 0
        else:
            stall += 1
            if stall > 5:
                rng = np.random.default_rng(it * 999983 + 11)
                scale = 0.02
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = normalize_simplex(a)
                stall = 0

    return best_a, best_obj


def p_anneal_refine(a, time_budget=30.0):
    """
    P-norm annealing: optimize smooth L_p approximation of max, then sharpen.
    Uses FFT-accelerated gradient with Nesterov momentum.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    v = np.zeros(n)
    mu = 0.85
    step = 0.06

    p_values = [4, 8, 16, 32, 64, 128, 256]
    per_phase = time_budget / len(p_values)

    for p in p_values:
        phase_start = time.monotonic()
        max_iters = int(per_phase * 40)

        for it in range(max_iters):
            if time.monotonic() - phase_start > per_phase:
                break
            if time.monotonic() - start > time_budget - 2:
                break

            conv = compute_conv(a)
            cmax = np.max(conv)
            if cmax < 1e-30:
                break

            ratios = conv / cmax
            weights = np.power(ratios, p)
            wsum = np.sum(weights)
            if wsum < 1e-300:
                idx = int(np.argmax(conv))
                weights = np.zeros(len(conv))
                weights[idx] = 1.0
            else:
                weights /= wsum

            # Gradient via FFT cross-correlation
            w_ext = np.zeros(fft_size)
            w_ext[:len(weights)] = weights
            W = rfft(w_ext, fft_size)
            A = rfft(a, fft_size)
            corr = irfft(A * np.conj(W), fft_size)
            grad = 2.0 * corr[:n]

            grad = grad - np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break

            v = mu * v - step * grad / gn
            a_new = normalize_simplex(a - v)
            a = a_new
            obj = compute_obj(a)

            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

            step *= 0.9992

        # Reset momentum between phases
        v = np.zeros(n)
        step = 0.05

    return best_a, best_obj


def subgradient_refine(a, max_iters=500, time_budget=20.0):
    """Direct subgradient on the max with adaptive temperature."""
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    v = np.zeros(n)
    mu = 0.8
    step = 0.04
    tau = 0.01

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= wsum

        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W = rfft(w_ext, fft_size)
        A = rfft(a, fft_size)
        corr = irfft(A * np.conj(W), fft_size)
        grad = 2.0 * corr[:n]

        grad = grad - np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        v = mu * v - step * grad / gn
        a_new = normalize_simplex(a - v)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        step *= 0.999
        tau *= 0.9997

    return best_a, best_obj


def generate_diverse_starts(n, parent, seed=42, num=12):
    """Generate diverse initial sequences for multi-start."""
    rng = np.random.default_rng(seed)
    starts = []

    # Parent resampled
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n)
    starts.append(normalize_simplex(np.interp(x_new, x_old, parent)))

    # Uniform
    starts.append(np.ones(n) / n)

    # Asymmetric ramp
    ramp = np.linspace(0.1, 1.0, n)
    starts.append(ramp / np.sum(ramp))

    # Reverse ramp
    ramp2 = np.linspace(1.0, 0.1, n)
    starts.append(ramp2 / np.sum(ramp2))

    # Bell (centered)
    i = np.arange(n)
    center = (n - 1) / 2.0
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 5.0) ** 2))
    starts.append(bell / np.sum(bell))

    # Asymmetric bell (shifted center)
    center2 = n * 0.3
    bell2 = np.exp(-((i - center2) ** 2) / (2 * (n / 4.0) ** 2))
    starts.append(bell2 / np.sum(bell2))

    # Sinusoidal modulation
    for k in range(5):
        freq = 1 + k
        amp = 0.3 + 0.3 * rng.random()
        phase = 2 * np.pi * rng.random()
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
        a = np.maximum(bell * mod, 0)
        s = np.sum(a)
        if s > 1e-10:
            starts.append(a / s)
        else:
            starts.append(bell / np.sum(bell))

    # Random
    rand = np.abs(rng.normal(1.0, 0.5, n))
    starts.append(rand / np.sum(rand))

    return starts


def resample_to_n(a, new_n):
    """Resample sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-length exploration with adaptive LP alternating,
    P-norm annealing, and subgradient refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent: n={parent.size}, obj={obj:.10f}")
            best_seq = parent.copy()
            best_score = obj
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(500) / 500
        best_score = compute_obj(best_seq)
        print(f"No parent loaded. Using uniform n=500, obj={best_score:.10f}")

    n_parent = len(best_seq)
    print(f"Starting search with n={n_parent}, score={best_score:.10f}")

    # ============================================================
    # Phase 1: Quick multi-length exploration (30% of budget)
    # ============================================================
    print("\n=== Phase 1: Multi-length exploration ===")
    phase1_deadline = start_time + float(budget_s) * 0.30

    test_lengths = []
    for dn in range(-200, 350, 50):
        nv = n_parent + dn
        if 250 <= nv <= 1200:
            test_lengths.append(nv)
    test_lengths = sorted(set(test_lengths))

    candidates = []

    for n in test_lengths:
        if time.monotonic() > phase1_deadline - 20:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 15:
            break

        # Resample parent to this length
        a_init = resample_to_n(best_seq, n)
        a_init = normalize_simplex(a_init)
        obj_init = compute_obj(a_init)

        # Quick P-anneal
        alloc = min(remaining * 0.5, max(8, 120.0 / len(test_lengths)))
        a_opt, obj_opt = p_anneal_refine(a_init, time_budget=alloc * 0.6)

        # Quick subgradient
        if time.monotonic() < phase1_deadline - 10:
            a_sg, obj_sg = subgradient_refine(a_opt, max_iters=200,
                                              time_budget=alloc * 0.4)
            if obj_sg < obj_opt:
                a_opt, obj_opt = a_sg, obj_sg

        candidates.append((obj_opt, a_opt.copy(), n))

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n}: {obj_opt:.10f} ** NEW BEST")
        else:
            print(f"  n={n}: {obj_opt:.10f}")

    # Sort candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates: {[f'{c[0]:.8f} (n={c[2]})' for c in candidates[:5]]}")

    # ============================================================
    # Phase 2: Intensive LP optimization of top candidates (45% of budget)
    # ============================================================
    print("\n=== Phase 2: Intensive LP optimization ===")
    phase2_deadline = start_time + float(budget_s) * 0.75

    top_k = min(4, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > phase2_deadline - 30:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 20:
            break

        a_cand = candidates[ci][1].copy()
        obj_cand = candidates[ci][0]
        n_cand = len(a_cand)

        # If this is the current best, give it more time
        if obj_cand == best_score:
            alloc = min(remaining * 0.6, 120.0)
        else:
            alloc = min(remaining * 0.3, 60.0)

        print(f"  Optimizing candidate {ci} (n={n_cand}, obj={obj_cand:.8f}), budget={alloc:.0f}s")

        # Adaptive LP alternating
        if n_cand <= 900:
            a_lp, obj_lp = adaptive_lp_alternate(a_cand, max_iters=40, time_budget=alloc * 0.5)
        else:
            a_lp, obj_lp = p_anneal_refine(a_cand, time_budget=alloc * 0.5)

        if obj_lp < obj_cand:
            obj_cand = obj_lp
            a_cand = a_lp

        # Max-sum LP as complementary
        if n_cand <= 800 and time.monotonic() < phase2_deadline - 20:
            a_ms, obj_ms = maxsum_lp_alternate(a_cand, max_iters=20,
                                               time_budget=alloc * 0.3)
            if obj_ms < obj_cand:
                obj_cand = obj_ms
                a_cand = a_ms

        # P-anneal polish
        if time.monotonic() < phase2_deadline - 15:
            a_pa, obj_pa = p_anneal_refine(a_cand, time_budget=alloc * 0.2)
            if obj_pa < obj_cand:
                obj_cand = obj_pa
                a_cand = a_pa

        # Subgradient final touch
        if time.monotonic() < phase2_deadline - 10:
            a_sg, obj_sg = subgradient_refine(a_cand, max_iters=300,
                                              time_budget=alloc * 0.15)
            if obj_sg < obj_cand:
                obj_cand = obj_sg
                a_cand = a_sg

        if obj_cand < best_score:
            best_score = obj_cand
            best_seq = a_cand.copy()
            print(f"  ** NEW BEST: {best_score:.10f} (n={n_cand})")
        else:
            print(f"  Best from this candidate: {obj_cand:.10f} (no improvement)")

    # ============================================================
    # Phase 3: Multi-start exploration around best (10% of budget)
    # ============================================================
    print(f"\n=== Phase 3: Multi-start perturbation (best: {best_score:.10f}) ===")
    phase3_deadline = start_time + float(budget_s) * 0.90

    if best_seq is not None and time.monotonic() < phase3_deadline:
        n_best = len(best_seq)
        starts = generate_diverse_starts(n_best, best_seq, seed=seed + 777, num=8)

        for si, s in enumerate(starts[:6]):
            if time.monotonic() > phase3_deadline - 10:
                break
            remaining = phase3_deadline - time.monotonic()
            if remaining < 10:
                break

            alloc = min(remaining * 0.4, 15.0)

            # Quick optimize
            a_try, obj_try = p_anneal_refine(s, time_budget=alloc * 0.5)
            if time.monotonic() < phase3_deadline - 8:
                a_try2, obj_try2 = adaptive_lp_alternate(
                    a_try, max_iters=15, time_budget=alloc * 0.5)
                if obj_try2 < obj_try:
                    a_try, obj_try = a_try2, obj_try2

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Start {si}: {obj_try:.10f} ** NEW BEST")

    # ============================================================
    # Phase 4: Final intensive polish (rest of budget)
    # ============================================================
    print(f"\n=== Phase 4: Final polish (best: {best_score:.10f}) ===")

    if best_seq is not None and time.monotonic() < deadline - 10:
        n_final = len(best_seq)
        remaining = deadline - time.monotonic()

        # Alternate LP methods several times
        round_budget = remaining / 6
        for rnd in range(6):
            if time.monotonic() > deadline - 5:
                break

            if n_final <= 850:
                a_r, obj_r = adaptive_lp_alternate(
                    best_seq, max_iters=20, time_budget=round_budget)
            else:
                a_r, obj_r = p_anneal_refine(best_seq, time_budget=round_budget)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Round {rnd}: {best_score:.10f}")
            else:
                # Try the other method
                if n_final <= 800:
                    a_r2, obj_r2 = maxsum_lp_alternate(
                        best_seq, max_iters=15, time_budget=round_budget * 0.7)
                else:
                    a_r2, obj_r2 = subgradient_refine(
                        best_seq, max_iters=200, time_budget=round_budget * 0.7)
                if obj_r2 < best_score:
                    best_score = obj_r2
                    best_seq = a_r2.copy()
                    print(f"  Round {rnd} (alt): {best_score:.10f}")

        # Final subgradient sweep
        if time.monotonic() < deadline - 8:
            remaining_final = deadline - time.monotonic()
            a_fin, obj_fin = subgradient_refine(
                best_seq, max_iters=int(remaining_final * 50),
                time_budget=remaining_final * 0.6)
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"  Final SG: {best_score:.10f}")

    # Final LP if time permits
    if best_seq is not None and len(best_seq) <= 800 and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_final_lp, obj_final_lp = adaptive_lp_alternate(
                best_seq, max_iters=25, time_budget=remaining * 0.7)
            if obj_final_lp < best_score:
                best_score = obj_final_lp
                best_seq = a_final_lp.copy()
                print(f"  Final LP: {best_score:.10f}")

    # ============================================================
    # Validation and return
    # ============================================================
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
