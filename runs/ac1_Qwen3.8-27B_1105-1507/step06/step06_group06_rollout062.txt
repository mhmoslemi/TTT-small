def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:

    return 2.0 * n * max_b / (sum_a * sum_a)

"""

Uses smoothed subgradient with FFT-based gradient computation, LP alternating,
multiplicative updates, and adaptive multi-length exploration.
Key improvements: FFT-accelerated subgradients, top-K peak tracking,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def compute_subgradient_fft(a, conv, tau=0.01):
    """
    Compute smoothed subgradient of max(conv) w.r.t. a using FFT.
    sg[i] = 2 * sum_k p[k] * a[k-i] where p is softmax over conv with temperature tau.
    """
    n = len(a)
    m = len(conv)
    cmax = conv.max()
    if cmax < 1e-15:
        return np.zeros(n)
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -400.0, 400.0)
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        idx = int(np.argmax(conv))
        p = np.zeros(m)
        p[idx] = 1.0
    else:
        p /= p_sum

    N = 1
    while N < m + n - 1:
        N <<= 1
    fp = rfft(p, N)
    fa = rfft(a, N)
    corr = irfft(fp * np.conj(fa), N)
    sg = 2.0 * corr[:n]
    return sg


def topk_subgradient(a, conv, K=20):
    """Compute subgradient using top-K peaks (hard selection)."""
    n = len(a)
    m = len(conv)
    top_idx = np.argpartition(-conv, K - 1)[:K]
    p = np.zeros(m)
    p[top_idx] = 1.0 / K

    N = 1
    while N < m + n - 1:
        N <<= 1
    fp = rfft(p, N)
    fa = rfft(a, N)
    corr = irfft(fp * np.conj(fa), N)
    sg = 2.0 * corr[:n]
    return sg


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns normalized g."""
    n = len(a)
    if n > 1500:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns (g, M_opt)."""
    n = len(a)
    if n > 1500:
        return None, None
    A = build_toeplitz(a)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def gradient_step(a, conv, tau, step_size, momentum_vec=None, momentum=0.85):
    """One step of smoothed subgradient descent with momentum."""
    n = len(a)
    sg = compute_subgradient_fft(a, conv, tau)
    sg_centered = sg - np.mean(sg)
    if momentum_vec is None:
        momentum_vec = np.zeros(n)
    momentum_vec = momentum * momentum_vec - step_size * sg_centered
    a_new = a + momentum_vec
    a_new = project_nonneg_sum1(a_new)
    return a_new, momentum_vec

        if p_sum < 1e-300:

            p = np.zeros(m)
            p[idx] = 1.0
        else:
            p /= p_sum

    N = 1
    while N < m + n - 1:
        N <<= 1
    fp = rfft(p, N)
    fa = rfft(a, N)
    corr = irfft(fp * np.conj(fa), N)

    log_update -= np.mean(log_update)  # Keep sum approximately constant

    a_new = a * np.exp(log_update)
    a_new = project_nonneg_sum1(a_new)

            # Quick optimization (short)
            a_quick, obj_quick = main_optimize(init, time_budget=per_init_budget, rng=rng, use_lp=False)
            if obj_quick < np.inf:
                candidates.append((obj_quick, a_quick))
                if obj_quick < best_score:
                    best_score = obj_quick
                    best_seq = a_quick.copy()
                    print(f"  n={n_try}: new best {best_score:.8f}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: {len(candidates)} candidates, best: {candidates[0][0] if candidates else 'N/A'}")

    # === Phase 2: Intensive optimization on top candidates (55% time) ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    print(f"Best before Phase 2: {best_score:.8f}")
    phase2_budget = min(800.0, (deadline - start_time) * 0.55)
    phase2_start = time.monotonic()

    # Focus on top candidates
    top_cands = [(best_score, best_seq.copy())]
    for obj_c, a_c in candidates[:3]:
        if obj_c < best_score + 0.05:
            top_cands.append((obj_c, a_c.copy()))

    for ci, (obj_init, a_init) in enumerate(top_cands):
        if time.monotonic() > phase2_start + phase2_budget:
            break
        remaining = phase2_start + phase2_budget - time.monotonic()
        if remaining < 20:
            break
        n_cand = len(a_init)
        alloc = min(remaining / max(1, len(top_cands) - ci), remaining * 0.6)
        print(f"  Candidate {ci} (n={n_cand}): obj={obj_init:.8f}, time={alloc:.0f}s")

        a_opt, obj_opt = main_optimize(a_init, time_budget=alloc, rng=rng, use_lp=(n_cand <= 1200))
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** Candidate {ci} improved: {best_score:.8f} **")

        # Additional LP refinement if time allows
        remaining2 = phase2_start + phase2_budget - time.monotonic()
        if remaining2 > 30 and n_cand <= 1000:
            a_lp, obj_lp = lp_alternate(a_opt, max_iters=15, time_budget=min(remaining2 * 0.4, 60.0), lp_time=10)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  ** LP refinement: {best_score:.8f} **")

    # === Phase 3: Multi-length exploration (20% time) ===
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    print(f"Best before Phase 3: {best_score:.8f}")
    phase3_budget = min(300.0, (deadline - time.monotonic()) * 0.25)
    phase3_start = time.monotonic()

    n_best = len(best_seq)
    n_tries = sorted(set([
        n_best - 50, n_best - 100, n_best + 50, n_best + 100,
        n_best + 200, n_best - 200, int(n_best * 1.3), int(n_best * 0.7),
    ]))
    n_tries = [n for n in n_tries if 150 <= n <= 3000]
    n_tries = list(dict.fromkeys(n_tries))

    per_n_budget = max(10.0, phase3_budget / len(n_tries))

    for n_try in n_tries:
        if time.monotonic() > phase3_start + phase3_budget:
            break
        a_resampled = project_nonneg_sum1(resample_sequence(best_seq, n_try))

        # Asymmetric perturbation
        x = np.linspace(0, 1, n_try)
        tilt = 1.0 + 0.2 * rng.random() * (2 * x - 1)
        a_tilted = project_nonneg_sum1(a_resampled * tilt)

        # Optimize both
        a_opt1, obj_opt1 = main_optimize(a_resampled, time_budget=per_n_budget * 0.4, rng=rng, use_lp=(n_try <= 1000))
        a_opt2, obj_opt2 = main_optimize(a_tilted, time_budget=per_n_budget * 0.4, rng=rng, use_lp=(n_try <= 1000))

        best_local_a = a_opt1 if obj_opt1 <= obj_opt2 else a_opt2
        best_local_obj = min(obj_opt1, obj_opt2)

        if best_local_obj < best_score:
            best_score = best_local_obj
            best_seq = best_local_a.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={best_local_obj:.8f}")

    # === Phase 4: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 15 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        n_final = len(a_final)
        polish_start = time.monotonic()

        # Multiple polishing rounds with different strategies
        round_budget = min(remaining4 * 0.5, 120.0)
        a_p, obj_p = main_optimize(a_final, time_budget=round_budget, rng=rng, use_lp=(n_final <= 1200))
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Round 1: {best_score:.8f}")
            a_final = a_p.copy()

        # LP refinement
        remaining_lp = deadline - time.monotonic()
        if remaining_lp > 30 and n_final <= 1200:
            a_lp, obj_lp = lp_alternate(a_final, max_iters=20,
                                        time_budget=min(remaining_lp * 0.5, 90.0), lp_time=12)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_final = a_lp.copy()
                print(f"  LP polish: {best_score:.8f}")

        # Small perturbations + re-optimize
        remaining_pert = deadline - time.monotonic()
        n_perts = min(5, int(remaining_pert / 15))
        for pi in range(n_perts):
            if time.monotonic() > deadline - 5:
                break
            scale = 0.01 + 0.01 * pi
            a_pert = a_final * np.exp(rng.normal(0, scale, n_final))
            a_pert = project_nonneg_sum1(a_pert)
            a_pert, obj_pert = main_optimize(a_pert, time_budget=min(15.0, (deadline - time.monotonic()) * 0.3),
                                             rng=rng, use_lp=False)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                a_final = a_pert.copy()
                print(f"  Perturb {pi}: {best_score:.8f}")

        #Final tiny gradient steps
        remaining_final = deadline - time.monotonic()
        if remaining_final > 5:
            conv = compute_conv_fft(a_final)
            for _ in range(min(50, int(remaining_final * 5))):
                if time.monotonic() > deadline - 2:
                    break
                sg = compute_subgradient_fft(a_final, conv, tau=0.005)
                sg_c = sg - np.mean(sg)
                a_try = project_nonneg_sum1(a_final - 0.01 * sg_c)
                obj_try = compute_obj(a_try)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    a_final = a_try.copy()
                    conv = compute_conv_fft(a_final)

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
