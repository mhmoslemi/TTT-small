"""
Optimized step function search using: (1) smoothed subgradient with Nesterov momentum
and Barzilai-Borwein step size, (2) LP alternating polish, (3) frequency-domain spectral
equalization for escaping local minima, (4) multi-restart with diverse starts, (5) adaptive
length exploration around 700.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def sliding_dot(w, a):
    """Compute g[i] = sum_j a[j] * w[j+i] for i=0..len(a)-1 using FFT."""
    n = len(a)
    N = 1
    while N < len(w) + n - 1:
        N <<= 1
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)
    return corr[:n]


def compute_gradient(a, weights):
    """Gradient of sum_k weights[k]*c[k] where c=conv(a,a). Returns grad of length n."""
    return 2.0 * sliding_dot(weights, a)


def softmax_weights(conv, tau):
    """Compute softmax weights over convolution values with temperature tau."""
    cmax = float(np.max(conv))
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    if s < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
        return w
    return exp_vals / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=20):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    if n > 2500:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def subgradient_optimize(a, max_iters=2000, time_budget=60.0, tau_init=0.02,
                         momentum_init=0.8, lr_init=0.05, verbose=False):

    Smoothed subgradient with Nesterov momentum and BB step size.
    Minimizes 2n*max(conv(a,a)) subject to a>=0, sum(a)=1.

    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    momentum = momentum_init
    lr = lr_init
    tau = tau_init

    prev_a = a.copy()
    prev_grad = None
    bb_count = 0

    stall_counter = 0
    last_improve_obj = best_obj

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        obj = 2.0 * n * cmax  # since sum(a)=1

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall_counter = 0
        else:
            stall_counter += 1

        # Periodically increase tau (annealing)
        tau = tau_init * (1.0 + 0.5 * it / max_iters)

        # Compute softmax weights
        w = softmax_weights(conv, tau)

        # Compute gradient
        grad = compute_gradient(a, w)

        # Center gradient (to preserve sum=1 after projection)
        grad_centered = grad - np.mean(grad)

        # Barzilai-Borwein step size
        if prev_grad is not None and bb_count < 500:
            s_vec = a - prev_a
            y_vec = grad_centered - prev_grad
            sy = float(np.dot(s_vec, y_vec))
            yy = float(np.dot(y_vec, y_vec))
            ss = float(np.dot(s_vec, s_vec))
            if sy > 1e-10 and yy > 1e-10:
                lr_bb1 = sy / yy
                lr_bb2 = ss / sy
                lr = 0.5 * lr_bb1 + 0.5 * lr_bb2
                lr = np.clip(lr, lr_init * 0.1, lr_init * 10.0)
            elif ss > 1e-20:
                lr = lr * 0.9

        prev_a = a.copy()
        prev_grad = grad_centered.copy()
        bb_count += 1

        # Nesterov momentum step
        v = momentum * v - lr * grad_centered

        # Try a few step scalings for robustness
        best_trial_obj = obj
        best_trial_a = a.copy()
        for scale in [1.0, 0.5, 2.0]:
            a_trial = a + scale * v
            a_trial = project(a_trial)
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_trial_obj:
                best_trial_obj = obj_trial
                best_trial_a = a_trial.copy()

        a = best_trial_a
        momentum = min(0.95, momentum * 1.001)  # slowly increase momentum

        # Decay learning rate slightly
        lr *= 0.9995

        if verbose and it % 100 == 0:

        # Restart if stalled

            a = project(a + 0.01 * np.random.randn(n))
            v = np.zeros(n)
            lr = lr_init

def lp_polish(a, max_iters=30, time_budget=40.0, verbose=False):
    """LP alternating with line search for final polishing."""
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_max_sum(a, M, time_limit=min(15, max(5, time_budget - (time.monotonic() - start))))
        if g is None:
            break

        # Line search
        best_t_obj = best_obj
        best_t_a = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = project((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a.copy()
            if verbose:
                print(f"      LP it {it}: obj={best_obj:.8f}")
        else:
            break

    return best_a, best_obj


def freq_domain_equality(a, max_iters=100, time_budget=20.0):
    """
    Frequency-domain approach: equalize the power spectrum to flatten the convolution.
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2

        # We want ifft(power) to be flat -> power should be concentrated at DC
        # So we "amplify" DC and "compress" other frequencies
        c = np.fft.ifft(power).real[:2 * n - 1]
        c_max = np.max(c)
        c_mean = np.mean(c)

        if c_max < 1e-15:
            break

        # Compression factor: reduce the dynamic range of power spectrum
        # Keep DC, compress the rest
        alpha = 0.8 + 0.15 * (it / max_iters)
        power_mod = power.copy()
        power_mod[0] = power[0]  # keep DC
        # Compress non-DC: blend toward mean
        non_dc_mean = np.mean(power_mod[1:])
        power_mod[1:] = power_mod[1:] * (1 - alpha * 0.3) + non_dc_mean * (alpha * 0.3)

        # Reconstruct with original phase
        phase = np.angle(fa)
        mag_new = np.sqrt(power_mod)
        fa_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def active_set_subgradient(a, max_iters=500, time_budget=30.0, peak_tol=0.01):
    """
    Subgradient focusing only on active peaks (where c[k] is within peak_tol of max).
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    momentum = 0.7
    lr = 0.03

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        obj = 2.0 * n * cmax

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Active set: peaks within tolerance of max
        threshold = cmax * (1.0 - peak_tol)
        active_mask = conv >= threshold
        active_frac = float(np.sum(active_mask)) / len(active_mask)

        # Weighted gradient: only active peaks contribute
        weights = np.zeros_like(conv)
        weights[active_mask] = 1.0 / np.sum(active_mask)

        grad = compute_gradient(a, weights)
        grad_centered = grad - np.mean(grad)

        # Step
        v = momentum * v - lr * grad_centered
        a = project(a + v)

        # Adaptive lr
        if it % 50 == 0:
            lr *= 0.95
            momentum = min(0.9, momentum + 0.005)

        # If too many or too few active peaks, adjust
        if active_frac > 0.5:
            peak_tol *= 0.8
        elif active_frac < 0.05:
            peak_tol = min(0.1, peak_tol * 1.5)

    return best_a, best_obj


def multi_perturb_local_opt(a, rng, n_perturbs=5, time_budget=30.0):
    """Apply multiple perturbations and locally optimize each."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    per_perturb_budget = time_budget / n_perturbs

    for p in range(n_perturbs):
        if time.monotonic() - start > time_budget:
            break

        # Different perturbation types
        a_pert = a.copy()
        ptype = p % 4

        if ptype == 0:
            # Gaussian noise
            a_pert = a_pert + rng.normal(0, 0.02, n)
        elif ptype == 1:
            # Window flip
            wsize = max(5, n // 20)
            w1 = rng.integers(0, n - wsize)
            w2 = rng.integers(0, n - wsize)
            a_pert[w1:w1+wsize] = a * (1.0 + 0.3 * rng.normal(0, 1, wsize))
        elif ptype == 2:
            # Frequency domain perturbation
            fa = np.fft.fft(a, 2 * n)
            fa_noisy = fa * np.exp(1j * rng.normal(0, 0.1, 2 * n))
            a_pert = np.fft.ifft(fa_noisy).real[:n]
        else:
            # Sparse perturbation: modify a few random entries
            n_mod = max(5, n // 50)
            idx = rng.choice(n, size=n_mod, replace=False)
            a_pert[idx] *= np.exp(rng.normal(0, 0.3, n_mod))

        a_pert = project(a_pert)

        # Quick local optimization
        a_opt, obj_opt = subgradient_optimize(a_pert, max_iters=100,
                                               time_budget=per_perturb_budget * 0.6,
                                               tau_init=0.03, lr_init=0.03)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization combining smoothed subgradient with
    Nesterov momentum + BB step, LP polishing, frequency-domain equalization,
    active-set subgradient, and multi-perturbation exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent: n={parent.size}, obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700)
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"No parent loaded. Starting with uniform n=700, obj={best_score:.8f}")

    print(f"Initial best: {best_score:.8f}, n={len(best_seq)}")

    # === Phase 1: Quick improvement via subgradient (10% time) ===
    t0 = time.monotonic()
    phase1_budget = min(120.0, (deadline - t0) * 0.08)
    print(f"\n=== Phase 1: Subgradient warmup (budget={phase1_budget:.0f}s) ===")

    a = project(best_seq.copy())
    a, obj = subgradient_optimize(a, max_iters=2000, time_budget=phase1_budget,
                                   tau_init=0.02, momentum_init=0.75, lr_init=0.05,
                                   verbose=True)
    if obj < best_score:
        best_score = obj
        best_seq = a.copy()
        print(f"  Phase 1 result: {best_score:.8f}")

    # === Phase 2: Intensive alternating optimization (40% time) ===
    t0 = time.monotonic()
    phase2_budget = min(600.0, (deadline - t0) * 0.40)
    print(f"\n=== Phase 2: Intensive alternating (budget={phase2_budget:.0f}s) ===")

    a = project(best_seq.copy())
    stall = 0

    while time.monotonic() < t0 + phase2_budget:
        remaining = t0 + phase2_budget - time.monotonic()
        if remaining < 15:
            break

        cycle_budget = min(remaining / 3.0, 60.0)

        # Strategy 1: Subgradient with momentum
        a_new, obj_new = subgradient_optimize(a, max_iters=500,
                                               time_budget=cycle_budget * 0.4,
                                               tau_init=0.015, momentum_init=0.8,
                                               lr_init=0.04)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new.copy()
            stall = 0
            print(f"  SG: {best_score:.8f}")
        else:
            stall += 1

        # Strategy 2: LP polish
        if time.monotonic() < t0 + phase2_budget - 10:
            a_lp, obj_lp = lp_polish(a, max_iters=20,
                                      time_budget=min(30.0, (t0 + phase2_budget - time.monotonic()) * 0.3))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a = a_lp.copy()
                stall = 0
                print(f"  LP: {best_score:.8f}")
            else:
                stall += 1

        # Strategy 3: Active-set subgradient
        if time.monotonic() < t0 + phase2_budget - 10:
            a_as, obj_as = active_set_subgradient(a, max_iters=300,
                                                   time_budget=min(20.0, (t0 + phase2_budget - time.monotonic()) * 0.2),
                                                   peak_tol=0.01)
            if obj_as < best_score:
                best_score = obj_as
                best_seq = a_as.copy()
                a = a_as.copy()
                stall = 0
                print(f"  AS: {best_score:.8f}")
            else:
                stall += 1

        # Strategy 4: Frequency domain equality
        if time.monotonic() < t0 + phase2_budget - 10:
            a_fd, obj_fd = freq_domain_equality(a, max_iters=100,
                                                 time_budget=min(15.0, (t0 + phase2_budget - time.monotonic()) * 0.2))
            if obj_fd < best_score:
                best_score = obj_fd
                best_seq = a_fd.copy()
                a = a_fd.copy()
                stall = 0
                print(f"  FD: {best_score:.8f}")
            else:
                stall += 1

        if stall > 8:
            break

    # === Phase 3: Multi-perturbation exploration (20% time) ===
    t0 = time.monotonic()
    phase3_budget = min(300.0, (deadline - t0) * 0.20)
    print(f"\n=== Phase 3: Multi-perturbation (budget={phase3_budget:.0f}s) ===")
    print(f"  Best before: {best_score:.8f}")

    n_rounds = max(3, int(phase3_budget / 25.0))
    for r in range(n_rounds):
        if time.monotonic() > t0 + phase3_budget:
            break
        remaining = t0 + phase3_budget - time.monotonic()
        if remaining < 10:
            break

        a_pert, obj_pert = multi_perturb_local_opt(best_seq, rng,
                                                    n_perturbs=3,
                                                    time_budget=min(remaining * 0.8, 25.0))
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Round {r}: {best_score:.8f}")

    # === Phase 4: Multi-length exploration (15% time) ===
    t0 = time.monotonic()
    phase4_budget = min(200.0, (deadline - t0) * 0.15)
    print(f"\n=== Phase 4: Multi-length (budget={phase4_budget:.0f}s) ===")
    print(f"  Best before: {best_score:.8f}")

    n_best = len(best_seq)
    n_tries = []
    for delta in [-30, -20, -10, 10, 20, 30, 50, -50]:
        n_try = n_best + delta
        if 100 <= n_try <= 2000:
            n_tries.append(n_try)
    n_tries = list(dict.fromkeys(n_tries))

    per_n_budget = max(10.0, phase4_budget / max(1, len(n_tries)))

    for n_try in n_tries:
        if time.monotonic() > t0 + phase4_budget:
            break
        remaining = t0 + phase4_budget - time.monotonic()
        if remaining < 8:
            break

        a_res = project(resample_sequence(best_seq, n_try))
        a_opt, obj_opt = subgradient_optimize(a_res, max_iters=500,
                                               time_budget=min(per_n_budget * 0.6, remaining * 0.5),
                                               tau_init=0.02, lr_init=0.04)
        if n_try <= 1200:
            a_opt2, obj_opt2 = lp_polish(a_opt, max_iters=10,
                                          time_budget=min(per_n_budget * 0.3, remaining * 0.3))
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    t0 = time.monotonic()
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Best before final: {best_score:.8f}")

    a_final = project(best_seq.copy())
    n_final = len(a_final)

    while time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break

        # LP polish
        if n_final <= 1500:
            a_p, obj_p = lp_polish(a_final, max_iters=15,
                                    time_budget=min(remaining * 0.3, 30.0))
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  Final LP: {best_score:.8f}")

        # Subgradient with small steps
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_s, obj_s = subgradient_optimize(a_final, max_iters=200,
                                               time_budget=min(remaining * 0.3, 20.0),
                                               tau_init=0.01, momentum_init=0.85,
                                               lr_init=0.02)
            if obj_s < best_score:
                best_score = obj_s
                best_seq = a_s.copy()
                a_final = a_s.copy()
                print(f"  Final SG: {best_score:.8f}")

        # Active set
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_as, obj_as = active_set_subgradient(a_final, max_iters=200,
                                                   time_budget=min(remaining * 0.25, 15.0),
                                                   peak_tol=0.005)
            if obj_as < best_score:
                best_score = obj_as
                best_seq = a_as.copy()
                a_final = a_as.copy()
                print(f"  Final AS: {best_score:.8f}")

        # Tiny perturbation + re-optimize
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_tiny = project(a_final * np.exp(rng.normal(0, 0.005, n_final)))
            a_tiny, obj_tiny = subgradient_optimize(a_tiny, max_iters=100,
                                                     time_budget=min(remaining * 0.3, 10.0),
                                                     tau_init=0.01, lr_init=0.02)
            if obj_tiny < best_score:
                best_score = obj_tiny
                best_seq = a_tiny.copy()
                a_final = a_tiny.copy()
                print(f"  Final tiny: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
