"""
Optimized multi-strategy solver for minimizing 2*n*max(conv(a,a))/(sum(a))^2
over non-negative sequences. Key improvements over previous version:
1. Correct gradient computation via FFT (subgradient of center peak = 2*reversed(a))
2. Refined LP alternating with adaptive constraints

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def build_toeplitz_cross(a):
    """Build matrix A where (A @ g)[k] = (conv(a,g))[k] for g of same length as a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        length = k_end - k_start
        A[k_start:k_end, j] = a[:length]
    return A


def solve_lp_givensum(a, M, time_limit=15):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 2500:
        return None
    A = build_toeplitz_cross(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate(a, max_iters=30, time_budget=60.0):
    """LP alternating: find optimal g for current a, blend, repeat."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        rem = time_budget - (time.monotonic() - start)
        if rem < 3:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_givensum(a, M, time_limit=min(rem * 0.4, 15))
        if g is None:
            break

        # Adaptive line search
        candidate_ts = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            stall = 0
        else:
            stall += 1
            if stall > 5:
                break

    return best_a, best_obj


def subgradient_smoothed(a, max_iters=300, time_budget=40.0, tau=0.5, momentum=0.8, step_init=0.15):
    """
    Smoothed subgradient with Nesterov momentum.
    Uses softmax-weighted average of subgradients from top convolution peaks.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    step = step_init
    N_fft = 1
    while N_fft < 2 * n - 1 + n - 1:
        N_fft <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights over convolution values
        tau_it = tau + 0.5 * (it / max_iters)
        shifted = (conv - cmax) / tau_it
        shifted = np.clip(shifted, -50.0, 0.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient via FFT: grad[i] = 2 * sum_k weights[k] * a[k-i] (valid)
        # This is 2 * cross_correlation(weights, a) at appropriate lags
        fw = rfft(weights, N_fft)
        fa = rfft(a, N_fft)
        # cross-correlation: h[i] = sum_k w[k] * a[k-i]
        # = ifft(conj(FFT(w)) * FFT(a)) ... but shifted
        # Actually: sum_k w[k] * a[k-i] = (w * a_flipped_full)[i + n - 1]
        # Let's use: correlate(w, a) = ifft(conj(fft(w)) * fft(a))
        # numpy convention: (f*g)(n) = sum_k f(k)*g(n-k)
        # So ifft(conj(FFT(w)) * FFT(a))[m] = sum_k w[k] * a[m-k] (circular)
        # We want h[i] = sum_k w[k] * a[k-i] = sum_k w[k] * a[-(i-k)]
        # = cross_corr(w, a)[i] = ifft(FFT(w) * conj(FFT(a)))[n-1-i]... 
        # Let me just use the direct relationship:
        # h[i] = sum_k w[k] * a[k-i] for 0 <= k-i < n
        # This equals (full correlation of w and a)[i]
        # In FFT: corr = ifft(fft(w, N) * conj(fft(a, N)), N)
        # corr[m] = sum_k w[k] * conj(a[m-k]) = sum_k w[k] * a[m-k] for real
        # We want h[i] = sum_k w[k] * a[k-i]
        # Set m = k-i => k = m+i: h[i] = sum_m w[m+i] * a[m]
        # This is the cross-correlation (w ⋆ a)[i]
        # In FFT: ifft(conj(FFT(w)) * FFT(a))[i] gives (a ⋆ w)[i] = sum_k a[k] * w[k-i]... 
        # I'm getting confused. Let me just compute it directly for the top few peaks.

        # For efficiency, if weights are concentrated at few peaks, compute directly
        top_k = np.where(weights > 0.01 * weights.max())[0]
        grad = np.zeros(n)
        for k_idx in top_k:
            w_val = weights[k_idx]
            if w_val < 1e-12:
                continue
            # grad[i] += 2 * w_val * a[k_idx - i] for 0 <= k_idx - i < n
            i_min = max(0, k_idx - n + 1)
            i_max = min(n, k_idx + 1)
            if i_max > i_min:
                indices = np.arange(i_min, i_max)
                a_indices = k_idx - indices
                grad[indices] += 2.0 * w_val * a[a_indices]

        # Center gradient (remove mean to preserve sum constraint direction)
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum step
        v = momentum * v - step * grad_centered
        a_new = a + v
        a_new = project_nonneg_sum1(a_new)

        # Backtracking line search
        obj_new = compute_obj(a_new)
        if obj_new > best_obj * 1.001:
            # Try smaller steps
            for shrink in [0.5, 0.25, 0.1]:
                a_try = a + shrink * v
                a_try = project_nonneg_sum1(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    a_new = a_try
                    obj_new = obj_try
                    break
            else:
                a_new = a
                obj_new = best_obj

        a = a_new
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        step *= 0.995

    return best_a, best_obj


def subgradient_center(a, max_iters=200, time_budget=20.0, step_init=0.1):
    """
    Fast subgradient targeting only the center peak (k = n-1).
    Gradient is simply 2 * reversed(a). Very fast per iteration.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    step = step_init
    momentum = 0.85

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Check if peak is still at center
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))

        # Compute subgradient for the peak
        if peak_idx == n - 1:
            grad = 2.0 * a[::-1].copy()
        else:
            grad = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    grad[i] = 2.0 * a[j]

        grad_centered = grad - np.mean(grad)

        v = momentum * v - step * grad_centered
        a_new = a + v
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        elif obj_new < best_obj * 1.01:
            a = a_new
        else:
            # Reset momentum
            v = np.zeros(n)
            step *= 0.8

        step *= 0.998

    return best_a, best_obj


def phase_retrieval_flat(a, max_iters=200, time_budget=30.0):
    """
    Phase retrieval: alternate between
    (1) frequency domain: flatten the spectrum (set |A(f)| = constant)
    (2) time domain: project onto a >= 0, sum(a) = 1
    This drives the autocorrelation toward a delta (flat spectrum = white autocorrelation).
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    target_power = 1.0 / n  # flat spectrum level

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Forward FFT
        A = fft(a, N)
        power = np.abs(A) ** 2

        # Blend current power toward flat
        alpha = 0.3 + 0.4 * (it / max_iters)  # gradually increase flattening
        power_new = (1 - alpha) * power + alpha * target_power * np.sum(power) / n

        # Take square root to get new magnitude
        mag_new = np.sqrt(np.maximum(power_new, 1e-20))
        phase = np.angle(A)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, N))[:n]

        # Project
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def phase_retrieval_weighted(a, max_iters=150, time_budget=25.0):
    """
    Weighted phase retrieval: flatten spectrum but with weight that
    de-emphasizes low frequencies (where the center peak lives).
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    # Frequency weights: de-emphasize DC and very low frequencies
    freqs = np.fft.fftfreq(N)
    freq_abs = np.abs(freqs)
    # Weight: 1 at high freq, less at low freq
    weight = 0.5 + 0.5 * (freq_abs / max(freq_abs, 1e-10))
    weight[0] = 0.1  # heavily de-emphasize DC

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, N)
        power = np.abs(A) ** 2
        mag = np.abs(A)

        # Target: weighted flat spectrum
        target_mag = np.sqrt(weight * np.sum(power) / np.sum(weight))
        alpha = 0.2 + 0.3 * (it / max_iters)
        mag_new = (1 - alpha) * mag + alpha * target_mag

        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, N))[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def chase_peak_fast(a, n_steps=50, step_size=0.05):
    """Fast peak chasing: directly reduce the maximum convolution value."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        grad_centered = grad - np.mean(grad)
        norm_g = np.linalg.norm(grad_centered)
        if norm_g < 1e-15:
            break

        grad_norm = grad_centered / norm_g

        # Try a few step sizes
        for ss in [step_size, step_size * 0.5, step_size * 2.0]:
            a_try = a - ss * grad_norm
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a, best_obj


def simulated_annealing(a, rng, max_iters=200, T_init=0.3, T_min=0.005, time_budget=60.0):
    """SA with diverse move types."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    cur_a = a.copy()
    cur_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = cur_obj
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        T = T_init * (T_min / T_init) ** (it / max_iters)

        move_type = rng.integers(0, 4)
        a_trial = cur_a.copy()

        if move_type == 0:
            # Window noise
            wsize = max(5, n // 20)
            wstart = int(rng.integers(0, n - wsize))
            orig = a_trial[wstart:wstart + wsize].copy()
            scale = max(orig.mean(), 0.01)
            a_trial[wstart:wstart + wsize] += rng.normal(0, 0.4 * scale, wsize)
            a_trial = project_nonneg_sum1(a_trial)

        elif move_type == 1:
            # Window swap
            wsize = max(5, n // 25)
            w1 = int(rng.integers(0, n - wsize))
            w2 = int(rng.integers(0, n - wsize))
            if abs(w1 - w2) >= wsize:
                s = min(w1, w2)
                e = max(w1, w2)
                a_trial[s:e], a_trial[e:e + (e - s)] = a_trial[e:e + (e - s)].copy(), a_trial[s:e].copy()
                a_trial = project_nonneg_sum1(a_trial)

        elif move_type == 2:
            # Mirror perturbation: add a small reversed copy at random offset
            offset = int(rng.integers(-n // 4, n // 4))
            scale = 0.1 * rng.random()
            rev = a_trial[::-1]
            # Add reversed version shifted
            new_a = a_trial.copy()
            for i in range(n):
                j = i + offset
                if 0 <= j < n:
                    new_a[j] += scale * rev[i]
            a_trial = project_nonneg_sum1(new_a)

        else:
            # Random restart of a window
            wsize = max(10, n // 8)
            wstart = int(rng.integers(0, n - wsize))
            new_window = rng.dirichlet(np.ones(wsize) * 0.5) * a_trial[wstart:wstart + wsize].sum()
            a_trial[wstart:wstart + wsize] = new_window
            a_trial = project_nonneg_sum1(a_trial)

        trial_obj = compute_obj(a_trial)
        delta = trial_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            cur_a = a_trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()

    return best_a, best_obj


def destroy_repair(a, rng, destroy_frac=0.2, time_budget=10.0):
    """Destroy a fraction and repair with structured fill."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))

    n_destroy = max(10, int(n * destroy_frac))
    destroy_idx = rng.choice(n, size=n_destroy, replace=False)
    mass = a[destroy_idx].sum()
    a[destroy_idx] = 0

    # Repair: distribute mass using a smooth kernel
    n_fill = max(20, n // 4)
    fill_idx = rng.choice(n, size=n_fill, replace=False)
    # Use a smooth weight for the fill
    positions = np.sort(fill_idx) / n
    weights = np.exp(-((positions - 0.5) ** 2) / 0.1) * (1 + 0.5 * np.sin(4 * np.pi * positions))
    weights = np.maximum(weights, 0.01)
    weights /= weights.sum()
    a[fill_idx] = weights * mass

    a = project_nonneg_sum1(a)
    a, obj = chase_peak_fast(a, n_steps=30, step_size=0.03)
    return a, obj


def variance_subgradient(a, max_iters=200, time_budget=25.0):
    """
    Minimize variance of autocorrelation (not just peak).
    This encourages a flatter autocorrelation overall.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    step = 0.05
    momentum = 0.7

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        cmean = float(np.mean(conv))

        # Combined objective: peak + alpha * variance
        alpha = 0.1
        # Weight: emphasis on peaks and spread
        shifted = (conv - cmax) / (0.5 * cmax + 1e-10)
        weights = np.exp(np.clip(shifted, -30, 0))
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Also add uniform weight for variance reduction
        uniform_w = np.ones(len(conv)) / len(conv)
        combined_w = (1 - alpha) * weights + alpha * uniform_w

        # Compute gradient
        top_idx = np.where(combined_w > 0.005)[0]
        grad = np.zeros(n)
        for k_idx in top_idx:
            w_val = combined_w[k_idx]
            if w_val < 1e-12:
                continue
            i_min = max(0, k_idx - n + 1)
            i_max = min(n, k_idx + 1)
            if i_max > i_min:
                indices = np.arange(i_min, i_max)
                a_indices = k_idx - indices
                grad[indices] += 2.0 * w_val * a[a_indices]

        grad_centered = grad - np.mean(grad)
        v = momentum * v - step * grad_centered
        a_new = project_nonneg_sum1(a + v)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.9

        step *= 0.998

    return best_a, best_obj


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular
    tri = np.abs(np.arange(n) - n / 2) + 0.1
    inits.append(tri / tri.sum())

    # Sine
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    inits.append(rc / rc.sum())

    # Two-sided exponential
    exp2 = np.exp(-np.abs(x - 0.5) * 3)
    inits.append(exp2 / exp2.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Bimodal
    bimodal = np.exp(-((x - 0.3) ** 2) / 0.01) + np.exp(-((x - 0.7) ** 2) / 0.01)
    inits.append(bimodal / bimodal.sum())

    # Parent resampled
    if parent is not None:
        inits.append(resample_sequence(parent, n))
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.15, n))
        inits.append(project_nonneg_sum1(pert))

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization combining LP alternating,
    smoothed subgradient, phase retrieval, and simulated annealing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"No parent loaded, starting from uniform n={len(best_seq)}")

    n_best = len(best_seq)
    print(f"Initial: best={best_score:.8f}, n={n_best}")

    # === Phase 1: Quick screening (8% time) ===
    phase1_budget = min(100.0, (deadline - start_time) * 0.08)
    print(f"\n=== Phase 1: Screening (budget={phase1_budget:.0f}s) ===")

    n_options = [n_best, max(100, n_best - 100), n_best + 50, n_best + 100,
                 min(2000, n_best + 200), max(100, n_best - 200)]
    n_options = list(dict.fromkeys([x for x in n_options if 50 <= x <= 3000]))

    for n_try in n_options:
        if time.monotonic() > start_time + phase1_budget:
            break
        inits = generate_initializations(n_try, rng, parent)
        for init in inits:
            if time.monotonic() > start_time + phase1_budget:
                break
            a_quick, obj_quick = chase_peak_fast(init, n_steps=30, step_size=0.04)
            if obj_quick < best_score:
                best_score = obj_quick
                best_seq = a_quick.copy()
                n_best = len(a_quick)
                print(f"  n={n_try}: improved to {best_score:.8f}")

    print(f"After Phase 1: best={best_score:.8f}, n={len(best_seq)}")

    # === Phase 2: Intensive LP + Subgradient (40% time) ===
    phase2_budget = min(600.0, (deadline - start_time) * 0.40)
    print(f"\n=== Phase 2: LP + Subgradient (budget={phase2_budget:.0f}s) ===")
    phase2_start = time.monotonic()

    a_work = project_nonneg_sum1(best_seq.copy())
    local_best = a_work.copy()
    local_obj = best_score

    for cycle in range(100):
        if time.monotonic() > phase2_start + phase2_budget:
            break
        rem = phase2_start + phase2_budget - time.monotonic()
        if rem < 10:
            break

        # Alternate strategies
        strat = cycle % 5

        if strat == 0 and len(a_work) <= 1500:
            # LP alternating
            a_new, obj_new = lp_alternate(a_work, max_iters=15, time_budget=min(rem * 0.3, 40))
        elif strat == 1:
            # Smoothed subgradient
            a_new, obj_new = subgradient_smoothed(a_work, max_iters=100, time_budget=min(rem * 0.2, 25))
        elif strat == 2:
            # Center peak subgradient (fast)
            a_new, obj_new = subgradient_center(a_work, max_iters=150, time_budget=min(rem * 0.15, 15))
        elif strat == 3:
            # Phase retrieval
            a_new, obj_new = phase_retrieval_flat(a_work, max_iters=80, time_budget=min(rem * 0.2, 20))
        else:
            # Variance subgradient
            a_new, obj_new = variance_subgradient(a_work, max_iters=100, time_budget=min(rem * 0.2, 20))

        if obj_new < local_obj:
            local_obj = obj_new
            local_best = a_new.copy()
            a_work = a_new.copy()
            if local_obj < best_score:
                best_score = local_obj
                best_seq = local_best.copy()
                if cycle % 5 == 0:
                    print(f"  Cycle {cycle} strat {strat}: NEW BEST {best_score:.8f}")
        else:
            # Nudge with small perturbation to escape
            a_pert = a_work * np.exp(rng.normal(0, 0.03, len(a_work)))
            a_pert = project_nonneg_sum1(a_pert)
            a_work = a_pert

    print(f"After Phase 2: best={best_score:.8f}")

    # === Phase 3: Phase retrieval + Weighted (20% time) ===
    phase3_budget = min(300.0, (deadline - start_time) * 0.20)
    print(f"\n=== Phase 3: Phase Retrieval (budget={phase3_budget:.0f}s) ===")
    phase3_start = time.monotonic()

    a_work = project_nonneg_sum1(best_seq.copy())

    for round_idx in range(20):
        if time.monotonic() > phase3_start + phase3_budget:
            break
        rem = phase3_start + phase3_budget - time.monotonic()
        if rem < 5:
            break

        # Alternate flat and weighted phase retrieval
        if round_idx % 2 == 0:
            a_new, obj_new = phase_retrieval_flat(a_work, max_iters=60, time_budget=min(rem * 0.3, 20))
        else:
            a_new, obj_new = phase_retrieval_weighted(a_work, max_iters=60, time_budget=min(rem * 0.3, 20))

        # Follow up with subgradient
        if obj_new < best_score or round_idx % 3 == 0:
            a_new2, obj_new2 = subgradient_smoothed(a_new, max_iters=80, time_budget=min(rem * 0.2, 15))
            if obj_new2 < obj_new:
                a_new, obj_new = a_new2, obj_new2

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_work = a_new.copy()
            print(f"  PR round {round_idx}: NEW BEST {best_score:.8f}")
        else:
            # Small perturbation
            a_work = project_nonneg_sum1(a_work * np.exp(rng.normal(0, 0.02, len(a_work))))

    print(f"After Phase 3: best={best_score:.8f}")

    # === Phase 4: SA + Destroy/Repair (15% time) ===
    phase4_budget = min(220.0, (deadline - start_time) * 0.15)
    print(f"\n=== Phase 4: SA + D/R (budget={phase4_budget:.0f}s) ===")
    phase4_start = time.monotonic()

    # SA on best
    a_sa, obj_sa = simulated_annealing(best_seq, rng, max_iters=150,
                                       T_init=0.2, T_min=0.003,
                                       time_budget=min(phase4_budget * 0.3, 60))
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = a_sa.copy()
        print(f"  SA: NEW BEST {best_score:.8f}")

    # Destroy/repair attempts
    remaining4 = phase4_budget - (time.monotonic() - phase4_start)
    for dr_i in range(min(15, int(remaining4 / 10))):
        if time.monotonic() > phase4_start + phase4_budget:
            break
        frac = 0.1 + 0.03 * dr_i
        a_dr, obj_dr = destroy_repair(best_seq, rng, destroy_frac=frac, time_budget=8.0)
        # Polish
        a_dr2, obj_dr2 = subgradient_center(a_dr, max_iters=80, time_budget=5.0)
        if obj_dr2 < obj_dr:
            a_dr, obj_dr = a_dr2, obj_dr2
        if obj_dr < best_score:
            best_score = obj_dr
            best_seq = a_dr.copy()
            print(f"  D/R {dr_i}: NEW BEST {best_score:.8f}")

    print(f"After Phase 4: best={best_score:.8f}")

    # === Phase 5: Multi-length + Final Polish (remaining time) ===
    remaining5 = deadline - time.monotonic()
    print(f"\n=== Phase 5: Multi-length + Polish (remaining={remaining5:.0f}s) ===")

    if remaining5 > 20:
        # Try nearby lengths
        n_current = len(best_seq)
        n_tries = [n_current - 30, n_current + 30, n_current - 60, n_current + 60,
                   n_current + 100, n_current - 100]
        n_tries = [x for x in n_tries if 50 <= x <= 3000]
        n_tries = list(dict.fromkeys(n_tries))

        per_n_budget = max(5.0, remaining5 * 0.4 / len(n_tries))
        for n_try in n_tries:
            if time.monotonic() > deadline - 30:
                break
            a_res = resample_sequence(best_seq, n_try)
            a_res = project_nonneg_sum1(a_res)
            a_opt, obj_opt = chase_peak_fast(a_res, n_steps=40, step_size=0.03)
            a_opt2, obj_opt2 = subgradient_smoothed(a_opt, max_iters=80,
                                                    time_budget=min(per_n_budget * 0.5, 10))
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")

    # Final intensive polish
    remaining_final = deadline - time.monotonic()
    if remaining_final > 10 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        n_final = len(a_final)

        for pr in range(min(15, int(remaining_final / 8))):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 5:
                break

            if n_final <= 1200:
                a_p, obj_p = lp_alternate(a_final, max_iters=10, time_budget=min(rem * 0.3, 20))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Final LP r{pr}: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem > 3:
                a_p2, obj_p2 = subgradient_center(a_final, max_iters=50,
                                                  time_budget=min(rem * 0.3, 10))
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_final = a_p2.copy()
                    print(f"  Final SG r{pr}: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem > 3:
                a_p3, obj_p3 = phase_retrieval_flat(a_final, max_iters=40,
                                                    time_budget=min(rem * 0.3, 10))
                if obj_p3 < best_score:
                    best_score = obj_p3
                    best_seq = a_p3.copy()
                    a_final = a_p3.copy()
                    print(f"  Final PR r{pr}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
