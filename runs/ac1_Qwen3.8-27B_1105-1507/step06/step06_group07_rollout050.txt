def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:

    sum_a = float(np.sum(a))
    if sum_a < 0.01:

    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    nrows = 2 * n - 1
    A = lil_matrix((nrows, n))
    for k in range(nrows):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1]
    return A.tocsr()


def solve_lp_fixed_M(a, M, time_limit=20):

    try:

                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:

        pass
    return None


def solve_lp_min_M(a, time_limit=20):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    if n > 1200:
        return None, None
    A_conv = build_toeplitz_sparse(a)

    # Minimize: M
    # Constraints: (a*g)[k] - M <= 0 for all k, sum(g) = 1, g >= 0, M free
    # c = [0, 0, ..., 0, 1] (n+1 vars)

    c[-1] = 1.0

def subgradient_polish(a, time_budget=40, lr_init=0.1):
    """
    Subgradient descent with momentum for polishing.
    Uses the subgradient of max_k (a*a)[k] with multiplicative updates.
    """
    start = time.monotonic()
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    best_a = a_norm.copy()
    best_obj = compute_obj(a_norm)
    lr = lr_init
    momentum = np.zeros(n)
    beta_m = 0.85
    no_improve = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        if no_improve > 500:
            break

        conv = compute_conv(a_norm)
        # Find top-3 peaks for a more robust subgradient
        top_peaks = np.argsort(conv)[::-1][:3]
        peak_vals = conv[top_peaks]
        max_val = peak_vals[0]
        # Soft weights for peaks
        weights = np.exp((peak_vals - max_val) * 10.0)
        weights = weights / weights.sum()

        # Compute weighted subgradient
        subgrad = np.zeros(n)
        for wi, ki in zip(weights, top_peaks):
            ki = int(ki)
            if ki == 0:
                subgrad += wi * 2.0 * a_norm
            else:
                for j in range(n):
                    s1 = a_norm[j + ki] if j + ki < n else 0.0
                    s2 = a_norm[j - ki] if j - ki >= 0 else 0.0
                    subgrad[j] += wi * (s1 + s2)

        # Center to maintain sum=1
        subgrad_centered = subgrad - np.mean(subgrad)

        # Momentum
        momentum = beta_m * momentum + subgrad_centered

        # Multiplicative update (mirror descent step)
        update = np.exp(-lr * momentum * 10.0)
        a_try = a_norm * update
        a_try = np.maximum(a_try, 1e-15)
        s_try = np.sum(a_try)
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a_norm = a_try.copy()
            no_improve = 0
        else:
            a_norm = (1 - 0.1) * a_norm + 0.1 * a_try
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.9

    return best_a


def peak_shaving(a, time_budget=30):

        else:

            contributions = np.zeros(n)
            for j in range(j_start, j_end + 1):

            a_try = a_try / s_try

def anneal_restart(a, time_budget=30, temp=0.05):
    """Simulated annealing: random perturbations with temperature-controlled acceptance."""
    start = time.monotonic()
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    best_a = a_norm.copy()
    best_obj = compute_obj(a_norm)
    current_a = a_norm.copy()
    current_obj = best_obj
    rng = np.random.default_rng(999)

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        # Propose perturbation
        a_try = current_a.copy()
        num_moves = rng.integers(1, max(2, n // 100))
        for _ in range(num_moves):
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue
            frac = rng.uniform(0.001, 0.05)
            delta = frac * max(current_a[src], 1e-6)
            if current_a[src] - delta < 0:
                delta = current_a[src] * 0.5
            a_try[src] -= delta
            a_try[dst] += delta

        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        delta_obj = obj_try - current_obj

        # Accept if better or with probability exp(-delta/T)
        if delta_obj < 0 or (temp > 1e-8 and rng.random() < np.exp(-delta_obj / temp)):
            current_a = a_try
            current_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

        # Cool down
        temp *= 0.999

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

    print(f"Phase 1: Advanced LP iteration (n={len(best_seq)})")
    print(f"{'=' * 50}")

    if time.monotonic() < deadline - 60:
        budget = min(240, (deadline - time.monotonic()) * 0.35)
        a_lp = lp_iterate_advanced(best_seq, max_iters=100, time_budget=budget, use_min_M=True)
        obj_lp = compute_obj(a_lp)
        print(f"  LP advanced: {obj_lp:.8f}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  *** UPDATE to {best_score:.8f}")

    # === Phase 2: Peak shaving ===
    print(f"\n{'=' * 50}")
    print(f"Phase 2: Peak shaving")
    print(f"{'=' * 50}")

    if time.monotonic() < deadline - 50:
        budget = min(40, (deadline - time.monotonic()) * 0.1)
        a_ps = peak_shaving(best_seq, time_budget=budget)
        obj_ps = compute_obj(a_ps)
        print(f"  Peak shaving: {obj_ps:.8f}")
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  *** UPDATE to {best_score:.8f}")

    # === Phase 3: Subgradient polish ===
    print(f"\n{'=' * 50}")
    print(f"Phase 3: Subgradient polish")
    print(f"{'=' * 50}")

    if time.monotonic() < deadline - 50:
        budget = min(50, (deadline - time.monotonic()) * 0.12)
        a_sg = subgradient_polish(best_seq, time_budget=budget, lr_init=0.1)
        obj_sg = compute_obj(a_sg)
        print(f"  Subgradient: {obj_sg:.8f}")
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  *** UPDATE to {best_score:.8f}")

    # === Phase 4: Another LP round on improved sequence ===
    print(f"\n{'=' * 50}")
    print(f"Phase 4: Second LP round")
    print(f"{'=' * 50}")

    if time.monotonic() < deadline - 50:
        budget = min(150, (deadline - time.monotonic()) * 0.25)
        a_lp2 = lp_iterate_advanced(best_seq, max_iters=80, time_budget=budget, use_min_M=True)
        obj_lp2 = compute_obj(a_lp2)
        print(f"  LP round 2: {obj_lp2:.8f}")
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  *** UPDATE to {best_score:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n{'=' * 50}")
    print(f"Phase 5: Multi-n exploration")
    print(f"{'=' * 50}")

    n_best = len(best_seq)
    n_options = sorted(set([
        max(300, n_best - 200), max(300, n_best - 100), max(300, n_best - 50),
        n_best,
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        500, 600, 800, 1000, 1200, 1400
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1600]

    for n_try in n_options:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        s = np.sum(a_try)
        if s > 1e-10:
            a_try = a_try / s

        # LP + peak shaving for this n
        alloc_lp = min(remaining * 0.3, 50.0)
        a_opt = lp_iterate_advanced(a_try, max_iters=30, time_budget=alloc_lp, use_min_M=(n_try <= 800))
        obj = compute_obj(a_opt)

        if obj < best_score:
            # Follow up with peak shaving
            if time.monotonic() < deadline - 30:
                a_opt2 = peak_shaving(a_opt, time_budget=min(15, (deadline - time.monotonic()) * 0.2))
                obj2 = compute_obj(a_opt2)
                if obj2 < obj:
                    obj = obj2
                    a_opt = a_opt2

            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  n={n_try}: {obj:.8f}")

    # === Phase 6: Annealing restarts for diversity ===
    print(f"\n{'=' * 50}")
    print(f"Phase 6: Annealing restarts")
    print(f"{'=' * 50}")

    if time.monotonic() < deadline - 30:
        for restart in range(5):
            if time.monotonic() > deadline - 25:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # Perturb best sequence
            rng = np.random.default_rng(seed + restart * 100)
            a_pert = best_seq.copy()
            noise = rng.normal(0, 0.02 + 0.01 * restart, len(a_pert))
            a_pert = a_pert * np.exp(noise)
            a_pert = np.maximum(a_pert, 0)
            s_p = np.sum(a_pert)
            if s_p < 1e-10:
                continue
            a_pert = a_pert / s_p

            # Quick LP + anneal
            alloc = min(remaining * 0.2, 30.0)
            a_opt = lp_iterate_advanced(a_pert, max_iters=20, time_budget=alloc * 0.5, use_min_M=False)
            a_anneal = anneal_restart(a_opt, time_budget=alloc * 0.4, temp=0.03)
            obj = compute_obj(a_anneal)

            if obj < best_score:
                best_score = obj
                best_seq = a_anneal.copy()
                print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart}: {obj:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n{'=' * 50}")
    print(f"Phase 7: Final polish")
    print(f"{'=' * 50}")

    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()

        # Final LP
        budget = min(remaining * 0.4, 60.0)
        a_final_lp = lp_iterate_advanced(best_seq, max_iters=50, time_budget=budget, use_min_M=True)
        obj_f = compute_obj(a_final_lp)
        print(f"  Final LP: {obj_f:.8f}")
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final_lp.copy()

        # Final subgradient
        if time.monotonic() < deadline - 10:
            budget = min((deadline - time.monotonic()) * 0.4, 25.0)
            a_final_sg = subgradient_polish(best_seq, time_budget=budget, lr_init=0.05)
            obj_f2 = compute_obj(a_final_sg)
            print(f"  Final subgrad: {obj_f2:.8f}")
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_final_sg.copy()

        # Final peak shaving
        if time.monotonic() < deadline - 8:
            budget = min((deadline - time.monotonic()) * 0.4, 15.0)
            a_final_ps = peak_shaving(best_seq, time_budget=budget)
            obj_f3 = compute_obj(a_final_ps)
            print(f"  Final peak shave: {obj_f3:.8f}")
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_final_ps.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    result = [float(x) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'=' * 50}")

    return result
