"""
Refined multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Focuses on intensive LP alternating with warm starts, adaptive M reduction,
frequency-domain flatness iteration, and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_max_sum(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 3000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def solve_lp_minpeak(a, time_limit=30):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 3000:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None

def lp_alternate_intensive(a, max_iters=100, time_budget=200.0, m_reduction=0.995):
    """
    Intensive LP alternating with adaptive M reduction and golden section line search.
    At each step, we try to solve for g with a slightly reduced M to force progress.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M_current = float(np.max(conv))
        
        # Try with current M first, then with reduced M
        M_try = M_current * m_reduction
        
        g = solve_lp_max_sum(a, M_try, time_limit=min(15, (time_budget - (time.monotonic() - start)) * 0.3))
        
        if g is None:
            # Try with current M
            g = solve_lp_max_sum(a, M_current, time_limit=min(10, (time_budget - (time.monotonic() - start)) * 0.2))
        
        if g is None:
            break

        # Golden section line search for best blend
        gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
        t_lo, t_hi = 0.0, 1.0
        t1 = t_hi - gr * (t_hi - t_lo)
        t2 = t_lo + gr * (t_hi - t_lo)
        
        a1 = project_nonneg_sum1((1 - t1) * a + t1 * g)
        a2 = project_nonneg_sum1((1 - t2) * a + t2 * g)
        obj1 = compute_obj(a1)
        obj2 = compute_obj(a2)
        
        for _ in range(8):
            if obj1 < obj2:
                t_hi = t2
                t2 = t1
                obj2 = obj1
                t1 = t_hi - gr * (t_hi - t_lo)
                a1 = project_nonneg_sum1((1 - t1) * a + t1 * g)
                obj1 = compute_obj(a1)
            else:
                t_lo = t1
                t1 = t2
                obj1 = obj2
                t2 = t_lo + gr * (t_hi - t_lo)
                a2 = project_nonneg_sum1((1 - t2) * a + t2 * g)
                obj2 = compute_obj(a2)
        
        t_best = (t_lo + t_hi) / 2
        a_new = project_nonneg_sum1((1 - t_best) * a + t_best * g)
        obj_new = compute_obj(a_new)
        
        # Also try t=1 (full replacement)
        obj_full = compute_obj(project_nonneg_sum1(g))
        if obj_full < obj_new:
            a_new = project_nonneg_sum1(g)
            obj_new = obj_full
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Try smaller M reduction
            m_reduction = max(0.99, m_reduction * 0.95)
            if stall > 15:
                break

    return best_a, best_obj

def freq_flat_iteration(a, max_iters=200, time_budget=60.0):
    """
    Frequency domain iteration to flatten the power spectrum.
    Iteratively compress |H(f)|^2 toward uniform while maintaining non-negativity.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2

        # Target: uniform power spectrum
        # Apply iterative compression
        max_p = np.max(power)
        min_p = np.min(power)
        if max_p < 1e-15:
            break

        # Geometric mean as target
        log_p = np.log(np.maximum(power, 1e-30))
        target_log = np.mean(log_p)
        target_p = np.exp(target_log)

        # Blend current power toward target
        alpha = 0.1 + 0.3 * (it / max_iters)  # gradually increase compression
        log_new = (1 - alpha) * log_p + alpha * target_log
        power_new = np.exp(log_new)

        # Normalize to preserve total energy (Parseval)
        power_new = power_new * (np.sum(power) / np.sum(power_new))

        # Reconstruct
        phase = np.angle(fa)
        magnitude = np.sqrt(power_new)
        fa_new = magnitude * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj

def freq_compression_iteration(a, max_iters=200, time_budget=60.0, compress_power=0.7):
    """
    Compress the power spectrum using power law, then reconstruct.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        max_p = np.max(power)
        if max_p < 1e-15:
            break

        # Power compression
        norm_p = power / max_p
        compressed = norm_p ** compress_power

        # Normalize
        compressed = compressed * (np.sum(power) / np.sum(compressed))

        phase = np.angle(fa)
        magnitude = np.sqrt(np.maximum(compressed, 0))
        fa_new = magnitude * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj

def coordinate_descent_polish(a, max_sweeps=5, time_budget=60.0, step_sizes=None):
    """
    Coordinate descent: for each position, try small perturbations and keep improvements.
    Uses efficient incremental convolution updates.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.05]

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            for ss in step_sizes:
                for sign in [-1, 1]:
                    a_try = a.copy()
                    a_try[i] += sign * ss
                    if a_try[i] < 0:
                        continue
                    # Redistribute to maintain sum=1
                    a_try = project_nonneg_sum1(a_try)
                    if np.allclose(a_try, a, atol=1e-10):
                        continue
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_obj

def chase_peak_multi(a, n_steps=100, time_budget=30.0):
    """
    Chase multiple peaks simultaneously using weighted subgradient.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Softmax weighting over all peaks
        tau = cmax * 0.1 + 1e-10
        weights = np.exp((conv - cmax) / tau)
        weights /= np.sum(weights)

        # Compute gradient (subgradient of weighted max)
        grad = np.zeros(n)
        for k in range(len(weights)):
            if weights[k] < 1e-6:
                continue
            # d conv[k] / d a[i] = a[k-i] (if 0 <= k-i < n) 
            # But also the "other" term: when j = i, a[k-i] contributes
            # Actually conv[k] = sum_j a[j]*a[k-j], so d/d a[i] = a[k-i] + a[k-i] = 2*a[k-i]?
            # No: d/d a[i] of sum_j a[j] * a[k-j]
            # = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
            # = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n, since j=k-i)
            # Wait, the second term: delta(k-j, i) means k-j=i, so j=k-i
            # So the second term contributes a[j] = a[k-i] when j=k-i is valid
            # But we're summing over j, so:
            # sum_j delta(j,i) * a[k-j] = a[k-i] (when 0<=k-i<n)
            # sum_j a[j] * delta(k-j,i) = a[k-i] (when 0<=k-i<n, i.e., j=k-i is in [0,n-1])
            # Total: 2 * a[k-i] when 0 <= k-i < n
            
            # Hmm wait, that's only if i != k-i. If i == k-i (i.e., k = 2i), then
            # we get a[i] + a[i] = 2*a[i], which is correct since conv[k] contains a[i]^2.
            
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * weights[k] * a[j]

        # Center and step
        grad_centered = grad - np.mean(grad)
        
        # Adaptive step size with backtracking
        for ss in [0.05, 0.02, 0.01, 0.005, 0.002]:
            a_try = project_nonneg_sum1(a - ss * grad_centered)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a, best_obj

def minpeak_lp_blend(a, time_budget=30.0):
    """Solve the min-peak LP and blend the result with current a."""
    n = len(a)
    if n > 1500:
        return a, compute_obj(a)
    
    g_lp, M_opt = solve_lp_minpeak(a, time_limit=min(time_budget * 0.5, 25))

    best_a = a.copy()

def entropy_regularized_iteration(a, max_iters=100, time_budget=40.0):
    """
    Add entropy regularization to encourage spreading, then project back.
    Iteratively balance between peak reduction and entropy maximization.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Penalty for peaks
        tau = cmax * 0.05 + 1e-10
        peak_penalty = np.exp((conv - cmax) / tau)
        
        # Gradient of peak penalty w.r.t. a
        grad = np.zeros(n)
        for k in range(len(peak_penalty)):
            if peak_penalty[k] < 1e-6:
                continue
            j = k - 0  # We need d conv[k]/d a[i] for each i

"""
Intensive multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over parent: (1) intensive LP alternating with golden-section
line search and adaptive M reduction, (2) log-space frequency flattening,
(3) multi-peak subgradient with proper 2*a[k-i] gradient, (4) cutting-plane LP
for speed, (5) adaptive perturbation-reoptimization cycles.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]. Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_minpeak(a, time_limit=30):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 2000:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(a, M, time_limit=25):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 2000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section_line_search(a, g, n_evals=10):
    """Golden section search for best t in (a, (1-t)*a + t*g)."""
    gr = (np.sqrt(5) - 1) / 2
    t_lo, t_hi = 0.0, 1.0
    t1 = t_hi - gr * (t_hi - t_lo)
    t2 = t_lo + gr * (t_hi - t_lo)
    a1 = project_nonneg_sum1((1 - t1) * a + t1 * g)
    a2 = project_nonneg_sum1((1 - t2) * a + t2 * g)
    obj1 = compute_obj(a1)
    obj2 = compute_obj(a2)
    for _ in range(n_evals):
        if obj1 < obj2:
            t_hi = t2
            t2 = t1
            obj2 = obj1
            t1 = t_hi - gr * (t_hi - t_lo)
            a1 = project_nonneg_sum1((1 - t1) * a + t1 * g)
            obj1 = compute_obj(a1)
        else:
            t_lo = t1
            t1 = t2
            obj1 = obj2
            t2 = t_lo + gr * (t_hi - t_lo)
            a2 = project_nonneg_sum1((1 - t2) * a + t2 * g)
            obj2 = compute_obj(a2)
    t_best = (t_lo + t_hi) / 2.0
    a_best = project_nonneg_sum1((1 - t_best) * a + t_best * g)
    obj_best = compute_obj(a_best)
    # Also check t=1
    obj_full = compute_obj(project_nonneg_sum1(g))
    if obj_full < obj_best:
        a_best = project_nonneg_sum1(g)
        obj_best = obj_full
    return a_best, obj_best


def lp_alternate_intensive(a, max_iters=200, time_budget=300.0, m_factor=0.999):
    """
    Intensive LP alternating: at each step solve min-peak LP for g given a,
    then blend a toward g using golden section search.
    Also periodically tries maxsum LP with reduced M.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Every 5 iterations, try min-peak LP; otherwise maxsum with reduced M
        if it % 5 == 4 or it == 0:
            g, M_opt = solve_lp_minpeak(a, time_limit=min(20, remaining * 0.4))
            if g is None:
                g = solve_lp_maxsum(a, float(np.max(compute_conv(a))), time_limit=min(15, remaining * 0.3))
        else:
            conv = compute_conv(a)
            M_cur = float(np.max(conv))
            M_try = M_cur * m_factor
            g = solve_lp_maxsum(a, M_try, time_limit=min(15, remaining * 0.3))
            if g is None:
                g = solve_lp_maxsum(a, M_cur, time_limit=min(10, remaining * 0.25))

        if g is None:
            stall += 1
            if stall > 8:
                break
            continue

        # Golden section line search for blend
        a_new, obj_new = golden_section_line_search(a, g, n_evals=8)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 12:
                break

    return best_a, best_obj


def freq_log_flat(a, max_iters=200, time_budget=60.0):
    """
    Frequency domain iteration: flatten the power spectrum in log-space.
    Iteratively move |H(f)|^2 toward its geometric mean.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        max_p = np.max(power)
        if max_p < 1e-15:
            break

        # Log-space averaging: move toward geometric mean
        log_p = np.log(np.maximum(power, 1e-30))
        target_log = np.mean(log_p)
        alpha = 0.05 + 0.25 * (it / max_iters)
        log_new = (1 - alpha) * log_p + alpha * target_log
        power_new = np.exp(log_new)

        # Preserve total energy (Parseval's theorem)
        power_new = power_new * (np.sum(power) / np.sum(power_new))

        phase = np.angle(fa)
        magnitude = np.sqrt(power_new)
        fa_new = magnitude * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def freq_power_compress(a, max_iters=150, time_budget=40.0, p=0.7):
    """
    Compress the power spectrum with power law p < 1, then reconstruct.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        max_p = np.max(power)
        if max_p < 1e-15:
            break

        norm_p = power / max_p
        compressed = norm_p ** p
        compressed = compressed * (np.sum(power) / np.sum(compressed))

        phase = np.angle(fa)
        magnitude = np.sqrt(np.maximum(compressed, 0))
        fa_new = magnitude * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def multi_peak_subgradient(a, n_steps=100, time_budget=30.0):
    """
    Multi-peak subgradient descent with softmax weighting.
    Uses the correct gradient: d conv[k]/d a[i] = 2*a[k-i] if 0<=k-i<n.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights over convolution positions
        tau = cmax * 0.05 + 1e-10
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= wsum

        # Compute gradient: grad[i] = sum_k weights[k] * 2*a[k-i] (if 0<=k-i<n)
        # This is 2 * (weights * a) cross-correlation
        # Use FFT for efficiency
        N_fft = 1
        while N_fft < len(weights) + n - 1:
            N_fft <<= 1
        fw = rfft(weights, N_fft)
        fa = rfft(a, N_fft)
        # correlation: sum_k w[k]*a[k-i] = ifft(fft(w) * conj(fft(a))) shifted
        # Actually: sum_k w[k] * a[k-i] for 0<=k-i<n
        # Let's use the convolution formula: conv(w, a_rev)[n-1+i] doesn't work simply
        # Direct computation via Toeplitz-like structure:
        # grad[i] = 2 * sum_{k=i}^{i+n-1} weights[k] * a[k-i]  (clipped to valid range)
        # This is 2 * (weights cross-correlated with a)[i]
        # Using FFT: ifft(fft(weights) * conj(fft(a)))[i] gives sum_k weights[k]*a[k-i+n-1]...
        # Let me just do it with the Toeplitz matrix approach for correctness
        # grad[i] = 2 * sum_k weights[k] * a[k-i] where 0<=k-i<n means i<=k<i+n
        # = 2 * (A^T @ weights)[i] where A is the Toeplitz of a
        # Actually A @ g = conv(a, g), so (A^T @ weights)[i] = sum_k A[k,i]*weights[k]
        # A[k,i] = a[k-i] if 0<=k-i<n, else 0
        # So (A^T @ weights)[i] = sum_k a[k-i] * weights[k] (valid k)
        # This is what we want!
        A = build_toeplitz(a)
        grad = 2.0 * (A.T @ weights)

        # Center the gradient (invariance to constant shifts in sum)
        grad_centered = grad - np.mean(grad)

        # Backtracking line search
        found = False
        for ss in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            a_try = project_nonneg_sum1(a - ss * grad_centered)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                found = True
                break

        if not found and step > 50:
            # Try smaller step
            a_try = project_nonneg_sum1(a - 0.0005 * grad_centered)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()

    return best_a, best_obj


def entropy_spread(a, max_iters=50, time_budget=20.0):
    """
    Iteratively blend toward a flatter distribution to increase entropy,
    which can help escape local minima in peak chasing.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Blend toward uniform (increases entropy)
        alpha = 0.02 + 0.05 * (it / max_iters)
        a_new = (1 - alpha) * a + alpha * (1.0 / n)
        a_new = project_nonneg_sum1(a_new)

        # Also try blending toward the "median level"
        med_level = np.median(a[a > 0]) if np.any(a > 0) else 1.0 / n
        a_alt = (1 - alpha) * a + alpha * med_level
        a_alt = project_nonneg_sum1(a_alt)

        obj_new = compute_obj(a_new)
        obj_alt = compute_obj(a_alt)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        elif obj_alt < best_obj:
            best_obj = obj_alt
            best_a = a_alt.copy()
            a = a_alt.copy()

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    inits.append(np.ones(n) / n)
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())
    gx = np.linspace(-3, 3, n)
    g = np.exp(-gx ** 2 / 3)
    inits.append(g / g.sum())
    bi = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02) + 0.01
    inits.append(bi / bi.sum())
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())
    if parent is not None:
        r = resample_sequence(parent, n)
        inits.append(r)
        p = r * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_nonneg_sum1(p))
    return inits

    """
    Main search: intensive LP alternating with golden-section search,
    log-space frequency flattening, multi-peak subgradient, and adaptive

    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    parent = try_load_parent()

    print(f"Starting: best={best_score:.8f}, n={n_parent}")

    # === Phase 1: Quick screening (5% time) ===
    print("\n=== Phase 1: Quick Screening ===")
    p1_end = start_time + min(60.0, (budget_s - 20) * 0.05)

    for n_t in n_opts:

            break

            a_q, obj_q = multi_peak_subgradient(init, n_steps=20, time_budget=3.0)

    print(f"\n=== Phase 2: Intensive LP ===")
    p2_end = start_time + min(550.0, (budget_s - 20) * 0.35)

        if obj_lp < best_score:
            best_score = obj_lp

        best_score = obj_f1
        best_seq = a_f1.copy()

            a_pert = a_fin * np.exp(rng.normal(0, 0.015, n_fin))
            a_pert = project_nonneg_sum1(a_pert)
            a_p1, obj_p1 = multi_peak_subgradient(a_pert, n_steps=20, time_budget=min(rem * 0.15, 8.0))
            if n_fin <= 800:
                a_p2, obj_p2 = lp_alternate_intensive(a_p1, max_iters=10,
                                                       time_budget=min(rem * 0.15, 8.0),
                                                       m_factor=0.999)
                if obj_p2 < obj_p1:
                    a_p1, obj_p1 = a_p2, obj_p2
            if obj_p1 < best_score:
                best_score = obj_p1
                best_seq = a_p1.copy()
                a_fin = a_p1.copy()
                print(f"  Polish Perturb: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    print(f"\nValidation: n={len(result)}, evaluate_sequence = {val:.8f}")

    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
