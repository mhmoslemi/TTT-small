Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent (KL geometry) with adaptive step size - exploits structure of subgradient
2. Fictitious play with QP best response for global exploration

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto probability simplex in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho_idx = int(np.argmax(~cond))
        if not (~cond).any():
            rho_idx = n - 1
        theta = (cv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def mirror_descent_step(a, k_star, eta):
    """One mirror descent step: a_i <- a_i * exp(-eta * 2*a[k_star-i]) / Z."""
    n = len(a)
    # Compute subgradient: g[i] = 2 * a[k_star - i] (with a[m]=0 for m outside [0,n-1])
    # For efficiency, extract the relevant slice of a
    # g[i] = 2*a[k_star-i] for i where 0 <= k_star-i < n
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    # a[k_star - i] for i in [lo, hi]: a[k_star - lo] down to a[k_star - hi]
    indices = k_star - np.arange(lo, hi + 1)
    g = np.zeros(n)
    g[lo:hi+1] = 2.0 * a[indices]
    
    # Mirror descent update
    log_update = -eta * g
    log_update -= log_update.max()  # for numerical stability
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def run_mirror_descent(a_init, max_iters, deadline, eta0=1.0, decay=0.999):
    """Mirror descent with KL geometry and adaptive step size."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    eta = eta0
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Try multiple subgradient directions (top indices)
        top_k = np.argsort(b)[::-1][:3]
        
        improved = False
        for ks in top_k:
            for t in [eta, eta * 0.5, eta * 2.0]:
                a_new = mirror_descent_step(a, int(ks), t)
                b_new = fast_conv(a_new)
                M_new = b_new.max()
                if M_new < best_M - 1e-15:
                    a = a_new
                    best_M = M_new
                    best_a = a.copy()
                    improved = True
                    eta = t * 1.1  # slightly increase step size if it worked
                    break
            if improved:
                break
        
        if not improved:
            eta *= decay
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0
    
    return best_a

        a_new = mirror_descent_step(a, k_star, eta)
        a = a_new
        
        # Update running average
        weight = 1.0 / (it + 1)
        a_avg = (1 - weight) * a_avg + weight * a
        a_avg = np.maximum(a_avg, 0)
        a_avg = a_avg / a_avg.sum()
        
        # Track best
        if (it + 1) % 10 == 0:
            M_cur = fast_conv(a_avg).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a_avg.copy()
            
            M_cur2 = fast_conv(a).max()
            if M_cur2 < best_M:
                best_M = M_cur2
                best_a = a.copy()

        eta = eta0 / (1.0 + 0.01 * it)
    
    # Final check
    M_avg = fast_conv(a_avg).max()
    if M_avg < best_M:
        best_M = M_avg
        best_a = a_avg.copy()

def run_accelerated_subgrad(a_init, max_iters, deadline, eta0=0.1):
    """Nesterov-accelerated subgradient method with projection."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    eta = eta0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(y)
        k_star = int(np.argmax(b))
        
        # Subgradient at y
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        indices = k_star - np.arange(lo, hi + 1)
        g = np.zeros(n)
        g[lo:hi+1] = 2.0 * y[indices]
        
        # Project gradient to tangent space
        g_tan = g - g.mean()
        
        # Step
        step = eta / np.sqrt(it + 1)
        a_new = y - step * g_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        # Track best
        if (it + 1) % 5 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def fictitious_play_qp(a_init, p_init, max_iters, deadline, qp_iters=50):

    Payoff: (a*a)[k]
    """

    p = p / p.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()

        if time.monotonic() > deadline:
            break

        # Q_p[j,m] = p[j+m] (with p[k]=0 for k >= conv_len)
        # Gradient of a^T Q_p a at a: 2 * Q_p @ a
        # (Q_p @ a)[i] = sum_m p[i+m] * a[m]
        
        # Solve QP using projected gradient
        a_qp = a.copy()
        # Lipschitz constant of 2*Q_p@a: <= 2 * sum_k p[k] * 1 = 2 (rough)
        # More precisely: ||2*Q_p||_2 <= 2 * max_i sum_m p[i+m]
        # For simplicity, use backtracking

        eta_qp = 1.0 / max(lp, 1e-6)
        
        for qp_it in range(qp_iters):
            # Compute gradient: grad[i] = 2 * sum_m p[i+m] * a_qp[m]
            # This is a "cross-correlation" of p (reversed) and a_qp
            # grad[i] = 2 * sum_{m: 0<=i+m<conv_len} p[i+m] * a_qp[m]
            #         = 2 * (p_reversed * a_qp)[i] where p_reversed[k] = p[conv_len-1-k]
            # Actually: sum_m p[i+m]*a[m] is the "anti-convolution" or cross-correlation
            # Let's compute directly using FFT
            # sum_m p[i+m] * a[m] for i in [0, n-1], m in [0, n-1]
            # Let q[k] = p[k] for k in [0, conv_len-1], 0 otherwise
            # Then sum_m q[i+m] * a[m] = cross-correlation of q and a, evaluated at lag i
            # cross_corr[i] = sum_m q[i+m] * conj(a[m]) = sum_m q[i+m] * a[m] (real)
            
            # Using FFT: cross_corr = ifft(conj(FFT(q)) * FFT(a))
            size = 1
            while size < conv_len + n - 1:
                size <<= 1
            fq = np.fft.fft(q, size)  # q = p padded
            fa = np.fft.fft(a_qp, size)
            cross = np.fft.ifft(np.conj(fq) * fa).real[:n]
            
            grad = 2.0 * cross
            
            # Projected gradient step
            a_qp_new = a_qp - eta_qp * grad
            a_qp_new = proj_simplex(a_qp_new)
            a_qp = a_qp_new
        
        # Update a
        a = a_qp
        
        # Player 2's best response: k* = argmax (a*a)
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        
        # Update p (fictitious play)
        alpha = 1.0 / (it + 1)
        p = (1 - alpha) * p + alpha * 0
        p[k_star] += alpha
        # Actually standard FP: p_{t+1} = (t*p_t + e_{k*}) / (t+1)
        # Which is the same as alpha = 1/(t+1)
        
        # Track best
        M_cur = b.max()
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()
    
    return best_a

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent (KL geometry) exploiting subgradient structure

import numpy as np
from scipy.optimize import linprog

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho_idx = int(np.argmax(~cond))
        if not (~cond).any():
            rho_idx = n - 1
        theta = (cv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def get_subgradient(a, k_star):
    """Compute subgradient of max_k (a*a)[k] at index k_star.
    g[i] = 2 * a[k_star - i] (with a[m] = 0 for m outside [0, n-1])
    """
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    if lo <= hi:
        g[lo:hi+1] = 2.0 * a[k_star - lo : k_star - hi + 1][::-1]
    return g
