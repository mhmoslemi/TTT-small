import numpy as np
from scipy.optimize import linprog, minimize

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    if not (~cond).any():
        rho = n - 1
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_grad(a, tau):
    """Compute smoothed gradient of max(conv(a,a)).
    
    Returns (smooth_value, gradient) where gradient = d(smooth_max)/da
    """
    n = len(a)
    b = fast_conv(a)
    # Log-sum-exp
    b_max = b.max()
    log_sum = b_max + np.log(np.sum(np.exp(tau * (b - b_max))))
    smooth_val = log_sum / tau  # = (1/tau) * log(sum(exp(tau*b)))
    
    # Weights: p_k = exp(tau*b_k) / sum(exp(tau*b_k))
    p = np.exp(tau * (b - b_max))
    p = p / p.sum()
    
    # Gradient: d/d a[i] of sum_k p_k * (conv(a,a))[k]
    # = sum_k p_k * 2*a[k-i] (for valid k-i)
    # = 2 * cross_correlation(p, a)[i]
    grad = np.zeros(n)
    for i in range(n):
        # sum_{j=0}^{n-1} p[i+j] * a[j]
        idx = np.arange(i, i + n)
        valid = (idx >= 0) & (idx < 2 * n - 1)
        if valid.any():
            grad[i] = 2.0 * np.dot(p[idx[valid]], a[valid])
    
    return smooth_val, grad


def adam_mirror_descent(a_init, deadline, tau_start=50.0, tau_end=5000.0,
                        lr=0.1, beta1=0.9, beta2=0.999, noise_start=0.05,
                        noise_end=0.0):
    """Adam-style multiplicative weights with adaptive tau and noise annealing."""
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = 2.0 * n * best_M
    
    # Adam moments
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    
    total_iters_estimate = 10000  # rough
    t_start = time.monotonic()
    
    while time.monotonic() < deadline:
        t_step += 1
        frac = min(1.0, t_step / total_iters_estimate)
        
        # Adaptive tau
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute smoothed gradient
        _, grad = smooth_grad(a, tau)
        
        # Project to tangent of simplex
        grad_tan = grad - grad.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_tan
        v = beta2 * v + (1 - beta2) * grad_tan ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        
        # Learning rate schedule
        lr_t = lr * 0.7071 / np.sqrt(t_step + 1)  # Adam-like
        
        # Multiplicative update
        update = lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = a * np.exp(-update)
        a_new = np.maximum(a_new, 1e-15)
        
        # Noise injection (annealing)
        noise_level = noise_start * (1 - frac) ** 2
        if noise_level > 1e-4:
            noise = np.random.dirichlet(np.ones(n) * 10)
            a_new = (1 - noise_level) * a_new + noise_level * noise
            a_new = np.maximum(a_new, 1e-15)
        
        a_new = a_new / a_new.sum()
        a = a_new
        
        # Track best
        M_cur = fast_conv(a).max()
        if M_cur < best_M:
            best_M = M_cur
            best_J = 2.0 * n * M_cur
            best_a = a.copy()
    
    return best_a, best_J


def cosine_param_eval(params, n):
    """Evaluate the cosine-parameterized sequence.
    
    params: (c_1, phi_1, c_2, phi_2, ..., c_M, phi_M)
    a[j] = exp(sum_m c_m * cos(2*pi*m*j/n + phi_m)) / Z
    """
    M = len(params) // 2
    t = np.arange(n) / n
    log_a = np.zeros(n)
    for m_idx in range(M):
        c_m = params[2 * m_idx]
        phi_m = params[2 * m_idx + 1]
        log_a += c_m * np.cos(2 * np.pi * (m_idx + 1) * t + phi_m)
    # Softmax normalization
    log_a -= log_a.max()
    a = np.exp(log_a)
    a = a / a.sum()
    return a


def cosine_param_grad(params, n):
    """Compute J and dJ/dparams for the cosine parameterization."""
    M = len(params) // 2
    t = np.arange(n) / n
    
    log_a = np.zeros(n)
    cos_vals = np.zeros((M, n))
    sin_vals = np.zeros((M, n))
    for m_idx in range(M):
        c_m = params[2 * m_idx]
        phi_m = params[2 * m_idx + 1]
        cos_vals[m_idx] = np.cos(2 * np.pi * (m_idx + 1) * t + phi_m)
        sin_vals[m_idx] = np.sin(2 * np.pi * (m_idx + 1) * t + phi_m)
        log_a += c_m * cos_vals[m_idx]
    
    log_a -= log_a.max()
    a = np.exp(log_a)
    a = a / a.sum()
    
    # Compute smoothed J and gradient wrt a
    tau = 200.0
    b = fast_conv(a)
    b_max = b.max()
    p = np.exp(tau * (b - b_max))
    p = p / p.sum()
    smooth_val = (b_max + np.log(p.sum() * np.exp(tau * b_max) / np.exp(tau * b_max))) / tau
    # Actually: (1/tau) * log(sum(exp(tau*b)))
    log_sum_exp = b_max + np.log(np.sum(np.exp(tau * (b - b_max))))
    smooth_val = log_sum_exp / tau
    
    # d(smooth_val)/da[i] = 2 * sum_k p[k] * a[k-i] (for valid range)
    grad_a = np.zeros(n)
    for i in range(n):
        grad_a[i] = 2.0 * np.dot(p[i:i+n], a)
    
    # dJ/dparams = dJ/da * da/dparams
    # da[j]/dparams[m] = a[j] * (d log_a[j] / dparams[m] - sum_k a[k] * d log_a[k] / dparams[m])
    J = 2.0 * n * smooth_val  # since sum(a) = 1
    
    dJ_dparams = np.zeros(len(params))
    for m_idx in range(M):
        # d log_a / d c_m = cos_vals[m_idx]
        # d log_a / d phi_m = -c_m * sin_vals[m_idx]

        dphi = -params[2 * m_idx] * sin_vals[m_idx]
        
        # da[j]/dc_m = a[j] * (dc[j] - sum_k a[k]*dc[k])
        dc_mean = np.dot(a, dc)
        dphi_mean = np.dot(a, dphi)
        
        da_dc = a * (dc - dc_mean)
        da_dphi = a * (dphi - dphi_mean)
        
        dJ_dparams[2 * m_idx] = np.dot(grad_a, da_dc)
        dJ_dparams[2 * m_idx + 1] = np.dot(grad_a, da_dphi)
    
    return J, dJ_dparams

def mirror_descent_kl(a_init, max_iters, deadline, step_init=0.3, momentum=0.0):
    """Mirror descent with KL geometry using top-k subgradients."""
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)  # momentum
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M_val = b.max()
        
        # Weighted subgradient: use top-5 locations
        top_k = 5
        sorted_idx = np.argsort(b)[::-1][:top_k]
        weights = b[sorted_idx]
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for kk in range(top_k):
            k_star = sorted_idx[kk]
            w = weights[kk]
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] += 2.0 * w * a[j]
        
        grad_tan = grad - grad.mean()
        
        # Momentum
        v = momentum * v + grad_tan
        
        # Exponential tilting
        step = step_init / (1.0 + 0.01 * it)
        log_w = -step * v
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
        
        M_cur = fast_conv(a).max()
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()
    
    return best_a


def lp_best_response(a, n_active=30, time_limit=10):
    """LP best response: find g that minimizes max(a*g)."""
    n = len(a)
    b = fast_conv(a)
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_indices):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_indices), n))
    if A.nnz == 0:
        return None
    
    c = -np.ones(n)
    b_ub = np.full(len(top_indices), M_target)
    bounds = [(0.0, None)] * n
    
    try:

        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:

    return None

Oh! The max of the convolution doesn't have to be at k = n-1. If the sequence is structured so that the convolution is "anti-peaked" at k = n-1 but has peaks elsewhere that are lower, then max < 1/n.

For example, if a[j] = 0 for j in the middle (a "two-hump" distribution), then:
- conv at k = n-1: sum a[j]*a[n-1-j] would be small (because the "overlap" is in the zero region)
- But conv at other k values would be larger (because the non-zero parts overlap more)

Wait, that gives a HIGHER max, not lower.

Let me think more carefully. For a "two-hump" a where a[j] = 2c for j in [0, n/2) and a[j] = 0 for j in [n/2, n):
- sum(a) = n*c, so c = 1/n
- conv at k in [0, n/2): (k+1) * (2c)^2 = 4(k+1)/n^2
- conv at k = n/2-1: 4*(n/2)/n^2 = 2/n
- conv at k = n-1: 0 (no overlap between the two humps)
- conv at k in [n/2, n-1]: decreasing...

So max(conv) = 2/n, giving J = 4. Worse than uniform!

So the "two-hump" is bad. What about a more subtle structure?

Actually, I think the key insight is this: for the convolution to have a low peak, the sequence needs to be "well-spread" in a specific way. The uniform sequence already achieves the minimum peak (1/n) at the center, and the edges are much lower (1/n^2). So the max is at the center.

To beat J = 2, we need to make the center value LESS than 1/n while keeping all other values below 1/n.

The center value is sum_j a[j]*a[n-1-j]. By Cauchy-Schwarz:
sum_j a[j]*a[n-1-j] >= (sum_j sqrt(a[j]*a[n-1-j]))^2 / n

Hmm, that's not directly helpful.

Actually, let's use the Cauchy-Schwarz inequality differently:
(sum_j a[j]*a[n-1-j]) * (sum_j 1^2) >= (sum_j sqrt(a[j]*a[n-1-j]))^2

So: sum_j a[j]*a[n-1-j] >= (sum_j sqrt(a[j]*a[n-1-j]))^2 / n

But (sum_j sqrt(a[j]*a[n-1-j]))^2 <= (sum_j a[j]) * (sum_j a[n-1-j]) = 1 (by Cauchy-Schwarz again)

So we get: sum_j a[j]*a[n-1-j] >= something / n, but the "something" can be as small as 0 (if a[j] and a[n-1-j] have disjoint support).

But if they have disjoint support, then the convolution at other lags would be large.

The key tradeoff: reducing the center value increases other lag values. The optimal balance is where all the "active" lag values are equal (by the min-max principle).

This is exactly what the LP approach captures! The LP finds the "best response" g that minimizes the maximum of (a*g), and the optimal a is a fixed point of this best-response mapping.

I think the issue is that the optimization landscape has many local minima, and the smooth approximation (with finite tau) is not sharp enough to capture the true max.

Let me try a different approach: **direct max computation with subgradient + Nesterov acceleration**.

Instead of the smooth approximation, use the exact subgradient at the max point, but with Nesterov's acceleration for nonsmooth functions.

The key idea from Nesterov's "accelerated gradient for nonsmooth problems":
- Use a "composite" step: y_t = a_t + ((t-1)/(t+1)) * (a_t - a_{t-1})
- Take a subgradient step from y_t
- Project onto the constraint set

This gives O(1/t^2) convergence for the nonsmooth case (vs O(1/t) for plain subgradient).

Let me implement this:

```python
def nesterov_subgradient(a_init, deadline, step_scale=1.0):
    a = a_init.copy()
    a_prev = a.copy()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t = 1
    
    while time.monotonic() < deadline:
        # Nesterov extrapolation
        alpha = (t - 1) / (t + 1)
        y = a + alpha * (a - a_prev)
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        # Subgradient at y
        b = fast_conv(y)
        k_star = int(np.argmax(b))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * y[j]
        
        # Step size: L_i = 2*n (Lipschitz constant of the gradient of each f_k)
        # For the nonsmooth min-max, use step = step_scale / t
        step = step_scale / (t * np.sqrt(n))
        
        # Gradient step in the "dual" space (mirror geometry)
        grad_tan = grad - grad.mean()
        a_new = y - step * grad_tan
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        a_prev = a
        a = a_new
        t += 1
        
        M_cur = fast_conv(a).max()
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()
    
    return best_a
```

Hmm, but this is basically what the mirror descent was doing. The issue is the step size and the fact that we're using only one subgradient direction.

Let me think about this differently. What if we use **multiple subgradients** (from the top-K peaks) with optimal weights?

At the optimum, by the KKT conditions, there exists a distribution q over the active peaks such that:
sum_k q_k * ∇f_k(a*) = lambda * 1 (on the support of a*)

This means the weighted average of the subgradients should be "flat" (proportional to the all-ones vector).

So the algorithm is:
1. At current a, find the top-K peaks
2. Find q (a distribution over the K peaks) that minimizes ||sum_k q_k * ∇f_k(a)||_2 (subject to q >= 0, sum q = 1)
3. Use this as the "effective subgradient"

This is a small QP (K variables, K constraints). For K=10, it's trivial to solve.

Let me implement this:

```python
def weighted_subgradient(a, K=10):
    """Compute optimal weighted subgradient from top-K peaks."""
    n = len(a)
    b = fast_conv(a)
    top_k = np.argsort(b)[::-1][:K]
    
    # Gradients: grad_k[i] = 2*a[k-i] for 0 <= k-i < n
    # Effective grad = sum_k q_k * grad_k
    # We want: sum_k q_k * grad_k = lambda * 1 (approximately)
    # This means: sum_k q_k * (grad_k - mean(grad_k)) = 0
    # i.e., sum_k q_k * grad_k_tan = 0
    
    G = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                G[idx, i] = 2.0 * a[j]
    
    # Remove the mean (tangent to simplex)
    G_tan = G - G.mean(axis=1, keepdims=True)
    
    # Find q that minimizes ||G_tan^T * q||^2 subject to q >= 0, sum(q) = 1
    # This is a QP: minimize q^T (G_tan @ G_tan.T) q s.t. q >= 0, 1^T q = 1
    # For small K, we can solve this analytically or with a simple method
    
    # Simple approach: use the "least squares" solution
    A_mat = G_tan @ G_tan.T  # K x K
    c_vec = np.ones(K)
    
    # Solve: min q^T A q s.t. q >= 0, sum(q) = 1
    # Using the KKT: 2*A*q + lambda*1 + mu = 0, mu >= 0, mu_i * q_i = 0
    # Try: q = -A^{-1} * 1 / (1^T * A^{-1} * 1)
    try:
        A_inv = np.linalg.inv(A_mat + 1e-8 * np.eye(K))
        q = -A_inv @ c_vec
        q_sum = q.sum()
        if q_sum > 0 and np.all(q >= -1e-6):
            q = np.maximum(q, 0)
            q = q / q.sum()
        else:
            # Fall back to uniform
            q = np.ones(K) / K
    except:
        q = np.ones(K) / K
    
    # Effective subgradient
    eff_grad = G_tan.T @ q
    
    return eff_grad, b.max()
```

This is a nice improvement! By finding the optimal weights q, we get a "flatter" effective subgradient, which should lead to better convergence.

The step size should be step = 1/L where L is the "Lipschitz constant" of the mapping a -> weighted_subgradient(a). This is hard to estimate, so we'll use a heuristic.
