import numpy as np

def compute_conv(a):
    """
    Compute linear convolution a*a using FFT.
    
    Parameters:
    a: input sequence
    
    Returns:
    conv: full linear convolution of a with a
    """
    n = len(a)
    size = 1
    while size < 2*n-1:
        size *= 2
        
    fa = np.fft.rfft(a, size)
    conv = np.fft.ifft(fa * np.conj(fa), size).real[:2*n-1]
    return conv

def compute_objective(a):
    """
    Compute the objective: 2*n*max(conv(a,a))/sum(a)^2.
    
    Parameters:
    a: input sequence
    
    Returns:
    float: objective value, inf if invalid.
    """

    if not isinstance(a, (list, np.ndarray)):
        return np.inf
        
    # Convert to numpy array for efficient computation, but must be float
    try:
        a_arr = np.asarray(a, dtype=np.float64)

        a_arr = np.array([float(x) for x in a], dtype=np.float64) if isinstance(a, list) else a

def parametric_sequence_generator(length):
    """
    Create initial parametric sequences for specified lengths.
    
    We generate four distinct prototypes for each length:
        Bell-shaped (exponential decay)
        Uniform (Constant exponent)
        Spike trains (periodic signals with adaptive decay)
        Sidon (low autocorrelation with varying phase)
        
    For each length, we create:
        np.logspace(-4, 4, length) [Bell-like]
        (np.arange(0, length) ** -0.5) [Power law decay]
        [np.exp(-(i - length/2) ** 2 / (2 * length/3)) for i in range(length)] [Gaussian]

    The prototypes must be non-negative and have reasonably high energy (sum >= 0.01).
    
    Parameters:
    length: sequence length to consider
    
    Returns:
    Four candidate sequences (numpy arrays) for this length
    """
    sequences = []
    
    # Type 0: Bell (similar to previous program's exponential with sine modulation) but minimal
    bell_base = np.exp(-(np.arange(0, length) - length/2) ** 2 / (2 * length**2))
    bell_base = bell_base / np.sum(bell_base)
    sequences.append(bell_base)
    
    # Type 1: Bell with adaptive gamma
    gamma = np.logspace(-3, 2, length)
    sequences.append(gamma/np.sum(gamma))
    
    # Type 2: Power-law decay envelope
    power = np.arange(0, length) ** -1.0
    power = power / (np.sum(power) * 10)  # Lower energy for power (avoid extreme self)
    sequences.append(power)
    
    # Type 3: Gaussian with shrinking bandwidth
    bandwidth = np.sqrt(length)
    gaussian = np.exp(-(np.arange(length)-length/2)**2/(2*bandwidth**2)) / np.sum(gaussian)
    sequences.append(gaussian)
    
    return sequences
