```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def solve_lp_phase(fixed_seq, time_limit=5):
    """
    Solve: min t s.t. conv(a, fixed_seq)[k] <= t for all k, sum(a)=1, a>=0.
    Returns optimal a and t.
    """
    n = len(fixed_seq)
    m = 2 * n - 1  # number of convolution lags
    
    # A_ub: each row k has the Toeplitz row for conv(a, fixed_seq)[k]
    # conv(a, fixed_seq)[k] = sum_i a[i] * fixed_seq[k-i] (where fixed_seq[j]=0 for j<0 or j>=n)
    # So row k of A_ub is: [fixed_seq[k-0], fixed_seq[k-1], ..., fixed_seq[k-(n-1)]] (with 0 for out of range)
    A_ub = np.zeros((m, n))
    for k in range(m):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = fixed_seq[j]
    
    # Minimize t: c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub constraints: conv[k] - t <= 0 → A_ub row is [toeplitz_row, -1]
    A_ub_full = np.zeros((m, n + 1))
    A_ub_full[:, :n] = A_ub
    A_ub_full[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a[i] >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': time_limit})
    
    if res.status == 0:
        a_opt = np.maximum(res.x[:n], 0)
        s = np.sum(a_opt)
        if s > 1e-15:
            a_opt = a_opt / s
        t_opt = float(res.x[n])
        return a_opt, t_opt
    return None, None

```python
"""
LP-alternation based optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses alternating LP solves (fixing one side of the convolution), gradient-based
polishing with softmax-smoothed max, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time
from scipy.linalg import toeplitz


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(fixed_seq):
    """Build the matrix M such that M @ a = conv(a, fixed_seq)."""
    n = len(fixed_seq)
    m = 2 * n - 1
    # M[k, i] = fixed_seq[k-i] if 0 <= k-i < n, else 0
    # This is a Toeplitz matrix
    first_col = np.zeros(m)
    first_col[:n] = fixed_seq[:n]
    first_row = np.zeros(n)
    first_row[0] = fixed_seq[0]
    M = toeplitz(first_col, first_row)
    return M


def solve_lp(fixed_seq, regularize=0.0, ref_seq=None, time_limit=5):
    """
    Solve: min t + reg*||a - ref||_1 s.t. conv(a, fixed)[k] <= t, sum(a)=1, a>=0.
    Returns (a_opt, t_opt) or (None, None).
    """
    n = len(fixed_seq)
    m = 2 * n - 1
    
    M = build_conv_matrix(fixed_seq)
    
    if regularize > 0 and ref_seq is not None:
        # min t + reg * sum(|a[i] - ref[i]|)
        # Introduce variables s[i] >= 0 for |a[i] - ref[i]|
        # a[i] - ref[i] <= s[i], -(a[i] - ref[i]) <= s[i]
        # Variables: a[0..n-1], t, s[0..n-1] → 2n+1 variables
        # This doubles the variables, might be slow for large n
        # For simplicity, just use min t with a small perturbation
        pass
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = M
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            a_opt = np.maximum(res.x[:n], 0)
            s = np.sum(a_opt)
            if s > 1e-15:
                a_opt = a_opt / s
            t_opt = float(res.x[n])
            return a_opt, t_opt
    except Exception:
        pass
    return None, None


def lp_alternate(a, max_iters=200, time_budget=100.0, verbose=True):
    """
    LP alternation: alternate between fixing a and fixing g.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    g = a.copy()
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Solve for a given g
        a_new, t1 = solve_lp(g, time_limit=3)
        if a_new is None:
            break
        
        # Solve for g given a_new
        g_new, t2 = solve_lp(a_new, time_limit=3)
        if g_new is None:
            g_new = a_new
        
        # Check the objective of a_new
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t1={t1:.6f}, t2={t2:.6f}")
        
        # Check convergence
        diff = np.max(np.abs(a_new - g_new))
        a = a_new
        g = g_new
        
        if diff < 1e-8:
            if verbose:
                print(f"    LP converged at it{it}: obj={best_obj:.8f}")
            break
        
        # If not improving, try blending
        if it > 20:
            a = 0.7 * a + 0.3 * g
            a = project_nonneg_sum1(a)
    
    return best_a, best_obj


def gradient_polish(a, max_iters=500, time_budget=30.0, tau_init=0.01, tau_end=0.001):
    """
    Gradient descent with softmax-smoothed max and backtracking line search.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = 0.1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        tau = tau_init * (tau_end / tau_init) ** (it / max_iters)
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: grad[i] = sum_k weights[k] * 2 * a[k-i] (where 0<=k-i<n)
        # This is 2 * correlation of weights with a
        # Use FFT for efficiency
        N_fft = 1
        while N_fft < 2 * n - 1:
            N_fft <<= 1
        fw = rfft(weights, N_fft)
        fa = rfft(a, N_fft)
        # We want grad[i] = 2 * sum_k weights[k] * a[k-i] for 0<=k-i<n
        # sum_k weights[k] * a[k-i] = (weights * a)[i] in the "cross" sense
        # Actually this is a correlation: (weights ⊛ a)[i] = sum_k weights[k+i] * a[k]
        # Hmm, let me think more carefully.
        # grad[i] = 2 * sum_{k=i}^{i+n-1} weights[k] * a[k-i]
        #          = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
        # This is a cross-correlation of weights and a, evaluated at lag i.
        # corr[i] = sum_j a[j] * weights[i+j]
        # = (flip(a) * weights)[i + n - 1] ... 
        # Actually, let's just use the convolution:
        # If we define b[j] = a[n-1-j] (reversed a), then
        # (b * weights)[i + n - 1] = sum_j b[j] * weights[i+n-1-j] = sum_j a[n-1-j] * weights[i+n-1-j]
        # Let k = n-1-j: = sum_k a[k] * weights[i+n-1-(n-1-k)] = sum_k a[k] * weights[i+k]
        # That's what we want!
        b = a[::-1]
        fb = rfft(b, N_fft)
        corr_full = irfft(fb * fw, N_fft)
        grad = 2.0 * corr_full[n-1:n-1+n]
        
        # Center gradient (to preserve sum)
        grad = grad - np.mean(grad)
        
        # Backtracking line search
        best_step = step
        best_trial_obj = np.inf
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            s = step * factor
            a_try = a - s * grad
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_trial_obj:
                best_trial_obj = obj_try
                best_step = s
            if obj_try < best_obj:
                break
        
        a = project_nonneg_sum1(a - best_step * grad)
        obj = compute_obj(a)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Decay step size
        step *= 0.999
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_nonneg_sum1(a_new)
