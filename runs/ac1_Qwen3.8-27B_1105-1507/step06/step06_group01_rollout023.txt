"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-constraint LP Frank-Wolfe (conditional gradient) with adaptive step size
2. Adaptive active-set LP for fast iterations
3. Spectral diversification (FFT-based perturbation)
4. Exact 1D coordinate descent polish
5. Multi-length search (640, 768, 896, 1024)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a, n):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)=1."""
    b = fast_conv(a)
    return float(2 * n * b.max())


def lp_full(a, M, n):
    """Solve full LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None)."""
    # Variables: g[0..n-1], M
    # Minimize M
    # Constraints: (a*g)[k] - M <= 0 for k=0..2n-2
    #              sum(g) = 1
    #              g >= 0, M free
    
    n_vars = n + 1  # g[0..n-1] and M
    c = np.zeros(n_vars)
    c[n] = 1.0  # minimize M
    
    # Build constraint matrix for (a*g)[k] - M <= 0
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        # (a*g)[k] = sum_{j} a[k-j] * g[j]
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[m])
        # -M
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n_vars))
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            g = g / g.sum()
            return g, r.x[n]
    except Exception:
        pass
    return None, None


def lp_reduced(a, active_indices, M, n):
    """Solve reduced LP with only active constraints.
    max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0, sum(g) <= 1
    Returns g (normalized) or None."""
    c = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def frank_wolfe_full(a, deadline, max_iters=200):
    """Full Frank-Wolfe using the complete LP. Each iteration solves the
    exact best-response LP."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Solve full LP: find g minimizing max_k (a*g)[k]
        g, M_opt = lp_full(a, M, n)
        if g is None:
            break
        
        if M_opt >= M * (1 - 1e-10):
            # No improvement possible with this a (we're at a fixed point of the LP)
            break
        
        # Line search: find best t in (0,1] such that (1-t)*a + t*g improves
        # We want to minimize max_k ((1-t)*a + t*g) * ((1-t)*a + t*g)[k]
        # This is complicated, so use a simpler approach:
        # The LP guarantees that max_k (a*g)[k] = M_opt < M = max_k (a*a)[k]
        # So for small t, max_k ((1-t)a+tg)*((1-t)a+tg)[k] < M
        
        # Use a bisection/line search on t
        best_t = 0.0
        best_val = M
        for t in [0.5, 0.7, 0.8, 0.9, 0.95, 1.0, 0.3, 0.4, 0.6]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 0.01:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_val:
                best_val = M_trial
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            M_new = fast_conv(a).max()
            if M_new < best_M:
                best_M = M_new
                best_a = a.copy()
                if it % 10 == 0:
                    print(f"    FW iter {it}: M={M_new:.10f} J={2*n*M_new:.8f}", flush=True)
        else:
            break
    
    return best_a


def frank_wolfe_reduced(a, deadline, max_iters=2000, n_active_start=30):
    """Fast Frank-Wolfe using reduced LP with adaptive active set."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    n_active = n_active_start
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Get active set
        active = np.argsort(b)[::-1][:n_active].tolist()
        
        # Solve reduced LP: max sum(g) s.t. (a*g)[k] <= M for active k
        g = lp_reduced(a, active, M, n)
        if g is None:
            n_active = min(n_active * 2, 2 * n - 1)
            if n_active >= 2 * n - 1:
                break
            continue
        
        # Check if improvement is possible
        # The LP maximizes sum(g) subject to (a*g)[k] <= M
        # If the optimal sum(g) is close to 1, then g ≈ a (no improvement)
        
        # Line search
        best_t = 0.0
        best_val = M
        for t in [0.5, 0.7, 0.9, 1.0, 0.3, 0.6, 0.8]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 0.01:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_val:
                best_val = M_trial
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            M_new = fast_conv(a).max()
            if M_new < best_M:
                best_M = M_new
                best_a = a.copy()
                # Can try fewer active constraints
                if n_active > 15:
                    n_active = max(15, int(n_active * 0.9))
            elif it > 10:
                # No improvement, expand active set
                n_active = min(int(n_active * 1.5), 2 * n - 1)
        else:
            n_active = min(n_active * 2, 2 * n - 1)
            if n_active >= 2 * n - 1:
                break
    
    return best_a


def coordinate_descent(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with exact 1D solve."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(seed)
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False
        
        for i in order:
            # Find optimal a[i] given all others
            # (a*a)[k] for k in [i, i+n-1] (clipped to [0, 2n-2])
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Precompute: for each affected k, (a*a)[k] = base[k] + slope[k] * a[i]
            # where base and slope don't depend on a[i]
            # (a*a)[k] = sum_j a[j]*a[k-j]
            # Terms involving a[i]: when j=i -> a[i]*a[k-i], when k-j=i -> a[j]*a[i]
            # So (a*a)[k] = a[i]*(a[k-i] + a[i-k]) + rest[k]
            # (with appropriate boundary handling)
            
            c_vals = []
            for k in range(k_lo, k_hi + 1):
                # (a*a)[k] = sum_j a[j]*a[k-j]
                # = a[i]*a[k-i] (if 0<=k-i<n) + a[k-i]*a[i] (if 0<=k-i<n, same term if k-i==i)
                # Actually: terms with a[i] are when j=i: a[i]*a[k-i], and when k-j=i i.e. j=k-i: a[k-i]*a[i]
                # If i == k-i (i.e., k=2i), these are the same term: a[i]^2
                # Otherwise: 2*a[i]*a[k-i] (if 0 <= k-i < n)
                
                # Let's compute base (without a[i] terms) and the coefficient of a[i]
                # (a*a)[k] = sum_{j != i, k-j != i} a[j]*a[k-j] + a[i]*a[k-i] (i≠k-i) + a[k-i]*a[i] (k-i≠i)
                # = base + 2*a[i]*a[k-i] (if 0<=k-i<n and k-i≠i)
                # = base + a[i]^2 (if k=2i)
                
                coeff = 0.0
                quadratic = False
                j_other = k - i
                if 0 <= j_other < n:
                    if j_other == i:
                        quadratic = True
                    else:
                        coeff = 2.0 * a[j_other]
                
                if quadratic:
                    # (a*a)[k] = base + a[i]^2
                    # base = (a*a)[k] - a[i]^2
                    base = b[k] - a[i] * a[i]
                    c_vals.append((base, coeff, True))
                elif coeff > 0:
                    # (a*a)[k] = base + coeff * a[i]
                    base = b[k] - coeff * a[i]
                    c_vals.append((base, coeff, False))
                else:
                    # (a*a)[k] doesn't depend on a[i]
                    c_vals.append((b[k], 0.0, False))
            
            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            
            # Now find x >= 0 that minimizes max(base_max, max_k (base_k + coeff_k * x or base_k + x^2))
            # This is a 1D convex optimization
            
            def obj_at(x):
                m = base_max
                for base, coeff, quad in c_vals:
                    if quad:
                        v = base + x * x
                    else:
                        v = base + coeff * x
                    if v > m:
                        m = v
                return m
            
            # Golden section search
            lo = 0.0
            hi = max(2.0 * a[i], 0.5)
            
            # Ensure hi is large enough
            for _ in range(5):
                if obj_at(hi) > obj_at(hi * 0.5) and hi < 5.0:
                    hi *= 2.0
                else:
                    break
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_at(a_pt)
            fb = obj_at(b_pt)
            
            for _ in range(30):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_at(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)
            
            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = x_opt
                b = fast_conv(a)  # Recompute (expensive but correct)
                sweep_improved = True
        
        s = a.sum()
        if s > 0:
            a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 10:
            break
    
    return best_a


def spectral_perturb(a, rng, strength=0.1):
    """Add a small spectral perturbation to escape local minima."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    
    a_pad = np.zeros(size)
    a_pad[:n] = a
    
    fa = np.fft.rfft(a_pad)
    mag = np.abs(fa)
    phase = np.angle(fa)
    
    # Perturb the magnitude
    noise = rng.normal(0, 1, len(mag))
    mag_new = mag + strength * mag * noise
    mag_new = np.maximum(mag_new, 0)
    
    fa_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(fa_new, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0.01:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def lp_full_best_response(a, n):
    """Solve the full LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None)."""
    n_vars = n + 1
    c = np.zeros(n_vars)
    c[n] = 1.0

    rows = []
    cols = []
    vals = []

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)

    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n_vars))
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
                return g, float(r.x[n])
    except Exception:
        pass
    return None, None

            s = g.sum()

    except Exception:
        pass
    return None
