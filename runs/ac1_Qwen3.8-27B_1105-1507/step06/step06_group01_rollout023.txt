"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Randomized accelerated mirror descent (subgradient from random active set)
2. Fast exact coordinate descent (O(n) per coordinate via precomputed slopes)

4. Multi-start from diverse initializations
Key: Direct subgradient (no smooth approximation), randomized active set,

import numpy as np
import time

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def subgradient_at_k(a, k):
    """Compute subgradient of b[k] w.r.t. a.
    db[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    g = np.zeros(n)
    # k-i >= 0 => i <= k
    # k-i < n => i > k-n
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        g[lo:hi+1] = 2.0 * a[k - np.arange(lo, hi+1)]
    return g

def run_randomized_mirror_descent(a_init, deadline, n_active=5, lr_init=1.0, use_acceleration=True):
    """Randomized mirror descent with KL geometry.
    
    At each step, pick a random active index (where b is near max),
    compute the subgradient, and do a multiplicative weights update.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 0
    lr = lr_init
    y = a.copy()  # for Nesterov
    a_prev = a.copy()
    nk = 1.0  # Nesterov momentum
    
    while time.monotonic() < deadline:
        t += 1
        # Adaptive learning rate
        lr = lr_init / np.sqrt(1.0 + 0.1 * t)
        
        # Compute convolution
        b = fast_conv(y)
        M = b.max()
        k_max = int(np.argmax(b))
        
        # Find active set: indices where b[k] > (1 - eps) * M
        # Use top-n_active for diversity
        if n_active >= len(b):
            active_indices = np.arange(len(b))
        else:
            # Get top n_active
            top_idx = np.argpartition(b, -n_active)[-n_active:]
            active_indices = top_idx
        
        # Randomly pick from active set
        k_star = int(active_indices[np.random.randint(len(active_indices))])
        
        # Compute subgradient
        g = subgradient_at_k(y, k_star)
        
        # Project gradient to tangent space of simplex (remove mean)
        g_tan = g - g.mean()
        
        # Mirror descent step (multiplicative weights)
        log_weights = -lr * g_tan
        # For stability
        log_weights -= log_weights.max()
        a_new = y * np.exp(log_weights)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Nesterov update
        if use_acceleration:
            nk_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * nk * nk))
            alpha = (nk - 1.0) / nk_new
            y_new = a_new + alpha * (a_new - a_prev)
            y_new = np.maximum(y_new, 1e-15)
            y_new = y_new / y_new.sum()
            a_prev = a
            a = a_new
            y = y_new
            nk = nk_new
        else:
            a = a_new
        
        # Track best (every 10 iterations or so)
        if t % 10 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    
    # Final check
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a

The mirror descent step is: a_new ∝ a * exp(-lr * g_tan)

If g is concentrated at the beginning (k* small), then the beginning of a gets decreased, pushing energy to the end. This creates asymmetry.

If g is at the middle (k* ≈ n-1), then the update is more uniform.

So the "direction" of the subgradient depends on where the max of b is. By randomizing among near-max indices, we get different "directions" and can explore more of the space.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Randomized mirror descent (subgradient from random near-active set) for exploration
2. Fast exact coordinate descent with incremental b updates (O(n^2) per sweep)
3. Periodic FFT recompute to avoid numerical drift
4. Random perturbation restarts to escape local minima
Key improvements: much faster CD than previous (no per-coordinate FFT),
randomized active set for diverse subgradients, and aggressive sweep count.
"""

import numpy as np
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    """Compute max(conv(a,a)) for normalized a (sum=1)."""
    b = fast_conv(a)
    return b.max()


def run_mirror_descent(a_init, deadline, n_active=5, lr_init=0.5, use_accel=True, seed=0):
    """Randomized mirror descent with KL geometry on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    t = 0
    y = a.copy()
    a_prev = a.copy()
    nk = 1.0

    while time.monotonic() < deadline:
        t += 1
        lr = lr_init / np.sqrt(1.0 + 0.05 * t)

        b = fast_conv(y)
        M = b.max()

        # Get active set
        if n_active >= len(b):
            active = np.arange(len(b))
        else:
            active = np.argpartition(b, -n_active)[-n_active:]

        # Random pick from active set
        k_star = int(active[rng.integers(len(active))])

        # Subgradient: g[i] = 2*a[k-i] if 0<=k-i<n
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            g[lo:hi+1] = 2.0 * y[k_star - np.arange(lo, hi+1)]

        g_tan = g - g.mean()

        log_w = -lr * g_tan
        log_w -= log_w.max()
        a_new = y * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        if use_accel:
            nk_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * nk * nk))
            alpha = (nk - 1.0) / nk_new
            y_new = a_new + alpha * (a_new - a_prev)
            y_new = np.maximum(y_new, 1e-15)
            y_new = y_new / y_new.sum()
            a_prev = a
            a = a_new
            y = y_new
            nk = nk_new
        else:
            a = a_new

        if t % 20 == 0:
            M_style = fast_conv(a).max()
            if M_style < best_M:
                best_M = M_style
                best_a = a.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()
    return best_a


def coord_descent_fast(a_init, deadline, max_sweeps=50000, recompute_every=50, seed=0):
    """Fast coordinate descent with incremental b updates.
    
    Each sweep is O(n^2) with small constant.
    Periodically recompute b via FFT to prevent drift.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    b = fast_conv(a)
    best_M = b.max()
    best_a = a.copy()

    order = np.arange(n)
    updates_since_recompute = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        rng.shuffle(order)
        sweep_changed = False

        for i in order:
            x_cur = a[i]

            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            n_aff = k_hi - k_lo + 1

            # Build base (b with a[i] contribution removed) and slopes
            b_base = np.empty(n_aff)
            s_arr = np.zeros(n_aff)
            quad_pos = -1

            for idx in range(n_aff):
                k = k_lo + idx
                j = k - i
                if 0 <= j < n:
                    if j == i:
                        b_base[idx] = b[k] - x_cur * x_cur
                        quad_pos = idx
                    else:
                        b_base[idx] = b[k] - 2.0 * x_cur * a[j]
                        s_arr[idx] = 2.0 * a[j]
                else:
                    b_base[idx] = b[k]

            # Unaffected max
            base_max = 0.0
            if k_lo > 0:
                pm = b[:k_lo].max()
                if pm > base_max:
                    base_max = pm
            if k_hi < 2 * n - 2:
                sm = b[k_hi + 1:].max()
                if sm > base_max:
                    base_max = sm

            # Golden section for x >= 0
            # obj(x) = max(base_max, max(b_base + x*s_arr), b_base[quad_pos] + x^2)

            def _obj(xv):
                m = base_max
                vm = (b_base + xv * s_arr).max()
                if vm > m:
                    m = vm
                if quad_pos >= 0:
                    qv = b_base[quad_pos] + xv * xv
                    if qv > m:
                        m = qv
                return m

            lo = 0.0
            hi = max(x_cur * 2.0, 0.01)
            for _ in range(5):
                if _obj(hi) > _obj(hi * 0.5):
                    hi *= 2.0
                else:
                    break
            hi = min(hi, 5.0)

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            ga = hi - gr * (hi - lo)
            gb = lo + gr * (hi - lo)
            fa = _obj(ga)
            fb = _obj(gb)
            for _ in range(25):
                if fa < fb:
                    hi = gb
                    gb = ga
                    fb = fa
                    ga = hi - gr * (hi - lo)
                    fa = _obj(ga)
                else:
                    lo = ga
                    ga = gb
                    fa = fb
                    gb = lo + gr * (hi - lo)
                    fb = _obj(gb)

            x_opt = (lo + hi) * 0.5
            if _obj(0.0) <= _obj(x_opt) + 1e-14:
                x_opt = 0.0

            if abs(x_opt - x_cur) > 1e-13:
                # Update b incrementally
                a[i] = x_opt
                for idx in range(n_aff):
                    k = k_lo + idx
                    j = k - i
                    if 0 <= j < n:
                        if j == i:
                            b[k] = b_base[idx] + x_opt * x_opt
                        else:
                            b[k] = b_base[idx] + s_arr[idx] * x_opt
                sweep_changed = True
                updates_since_recompute += 1

            # Renormalize: a /= sum, b /= sum^2
            s_sum = a.sum()
            if s_sum < 1e-10:
                a = np.ones(n) / n
                b = fast_conv(a)
                break
            a = a / s_sum
            b = b / (s_sum * s_sum)

            M_cur = b.max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()

        # Periodic recompute to prevent drift
        if updates_since_recompute >= recompute_every:
            b = fast_conv(a)
            updates_since_recompute = 0
            M_cur = b.max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()

        if not sweep_changed:
            break

    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = np.maximum(parent, 0.0)
        ps = p.sum()
        if ps > 1e-10:
            inits.append(p / ps)
        p_pert = np.maximum(p / ps + rng.normal(0, 0.02, n), 1e-6)
        inits.append(p_pert / p_pert.sum())
        p_pert2 = np.maximum(p / ps + rng.normal(0, 0.08, n), 1e-6)
        inits.append(p_pert2 / p_pert2.sum())

    inits.append(np.ones(n) / n)

    # Linear ramp (asymmetric)
    ramp = np.linspace(1.0, 0.1, n)
    inits.append(ramp / ramp.sum())

    # Exponential decay
    exp_d = np.exp(-5.0 * t)
    inits.append(exp_d / exp_d.sum())

    # Cosine
    cos_w = 0.5 + 0.5 * np.cos(np.pi * t)
    inits.append(cos_w / cos_w.sum())

    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src,
                     dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
