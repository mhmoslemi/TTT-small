def compute_conv(a):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

    if n == 0:

    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    return a / s

    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

            t = best_t
        else:
            t = 0.3
        
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a = a_new / s_new
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Check for convergence
        if it > 10 and abs(compute_obj(a) - best_obj) < 1e-10:
            break
    
    return best_a

def gs_iterate(a, n_iters=50, target_flatness=1.0):
    """
    Gerchberg-Saxton style iteration for flat autocorrelation.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(n_iters):
        # Compute FFT of current signal
        A = rfft(a, size)
        
        # Target: flat autocorrelation in the middle
        # Desired autocorrelation: flat = 1/(2n-1) everywhere (ideal)
        # In frequency domain: |A|^2 should be constant
        # So we want |A(ω)| to be constant
        
        # Current power spectrum
        power = np.abs(A) ** 2
        total_power = np.sum(power)
        
        # Target power: uniform (white spectrum)
        target_power = np.ones_like(power) * (total_power / len(power))
        
        # Adjust magnitude
        current_mag = np.abs(A)
        target_mag = np.sqrt(target_power)
        
        # Blend: new_mag = (1-alpha)*current_mag + alpha*target_mag
        alpha = 0.5
        new_mag = (1 - alpha) * current_mag + alpha * target_mag
        new_mag = np.maximum(new_mag, 1e-10)
        
        # New phase: keep the existing phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = irfft(A_new, size)[:n]
        
        # Project onto non-negative
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a = a_new / s_new
    
    return a
