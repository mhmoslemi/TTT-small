import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def grad_cross(a, w):
    """Compute C[i] = 2*sum_j a[j]*w[i+j] for i=0..n-1."""
    n = len(a)
    w_rev = w[::-1]
    size = 1
    while size < n + len(w_rev) - 1:
        size <<= 1
    A = rfft(a, size)
    W = rfft(w_rev, size)
    conv = irfft(A * W, size)
    return 2.0 * conv[n-1:2*n-1][::-1]

def obj_grad_pnorm(a, p):
    """Compute p-norm smoothed objective and gradient."""
    n = len(a)
    conv = compute_conv(a)
    S = float(np.sum(a))
    cmax = float(np.max(conv))
    if cmax < 1e-30:
        return 0.0, np.zeros(n)
    cn = conv / cmax
    cn_p = cn ** p
    T = float(np.sum(cn_p))
    g = cmax * T ** (1.0/p)
    if p > 1:
        w = cn ** (p-1)
    else:
        w = np.ones_like(conv)
    C_raw = grad_cross(a, w)
    dg_da = C_raw / (T ** ((p-1.0)/p))
    f = 2.0 * n * g / (S * S)
    grad = (2.0 * n / (S*S)) * (dg_da - 2.0*g/S)
    return f, grad

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def peak_shave(a, time_budget, n_iters=200):
    """Iteratively reduce peaks by moving mass."""
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        S = np.sum(a)
        # Find top peaks
        peak_positions = np.argsort(conv)[::-1][:3]
        improved = False
        for p in peak_positions:
            # Find contributing pairs
            i_min = max(0, p - (n-1))
            i_max = min(p, n-1)
            # Contributions
            contribs = []
            for i in range(i_min, i_max+1):
                j = p - i
                contribs.append((i, j, a[i] * a[j]))
            # Sort by contribution
            contribs.sort(key=lambda x: -x[2])
            # Move mass from top contributor to a random low-contributing position
            if contribs:
                i_top, j_top, _ = contribs[0]
                # Find a position that doesn't contribute to any top peak
                target_positions = list(range(n))
                # Remove positions that contribute to peaks
                for p2 in peak_positions:
                    for i2 in range(max(0, p2-(n-1)), min(p2, n-1)+1):
                        if i2 in target_positions:
                            target_positions.remove(i2)
                        j2 = p2 - i2
                        if 0 <= j2 < n and j2 in target_positions:
                            target_positions.remove(j2)
                if not target_positions:
                    target_positions = list(range(n))
                i_target = target_positions[rng.integers(0, len(target_positions))]
                # Move small amount
                delta = 0.01 * max(a[i_top], 1e-6)
                a_new = a.copy()
                a_new[i_top] -= delta
                a_new[i_target] += delta
                if a_new[i_top] < 0:
                    continue
                a_new = a_new / np.sum(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
        if not improved and it > 20:
            break
    
    return best_a

def resample(a, new_n):

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
