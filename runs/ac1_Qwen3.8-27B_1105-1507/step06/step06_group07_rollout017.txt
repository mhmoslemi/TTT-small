"""
Nesterov-accelerated optimization with adaptive temperature logsumexp smoothing,
active-set-aware refinement, and diverse initializations for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient.
    Uses logsumexp with temperature beta.
    Returns (smooth_max, gradient_wrt_a)
    """

    conv = compute_conv(a)
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    
    smooth_max = cmax + np.log(e_sum) / beta
    
    # Gradient: d(smooth_max)/d(a[m]) = 2 * sum_k w[k] * (cross_corr(w,a))[m]
    # Actually: d(max_k conv[k])/d(a[m]) = 2 * sum_k w[k] * d(conv[k])/d(a[m])

    # Using FFT: grad = 2 * ifft(FFT(w) * conj(FFT(a)))[:n]
    size = 1

    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    grad = 2.0 * cc[:n]
    
    return smooth_max, grad

    s = np.sum(a)

    x = np.log(np.maximum(a, 1e-15))
    x -= np.max(x)  # center for numerical stability

    y = x.copy()
    v = np.zeros(n)
    
    best_a = a.copy()
    best_obj = compute_obj(a)

        gamma = (t_k - 1) / (it + 1 + t_k)

        y_c = y - np.max(y)
        a_y = np.exp(y_c)
        a_y = a_y / np.sum(a_y)

def make_hann_init(n):
    """Create a Hann-window initialized sequence."""
    x = np.arange(n)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    return hann / np.sum(hann)


def make_gaussian_init(n, sigma_frac=0.3):
    """Create a Gaussian-window initialized sequence."""
    x = np.arange(n)
    center = (n - 1) / 2
    gauss = np.exp(-0.5 * ((x - center) / (sigma_frac * n)) ** 2)
    return gauss / np.sum(gauss)


def make_uniform_init(n):
    """Uniform initialization."""
    return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
Nesterov-accelerated (FISTA) optimization with adaptive logsumexp temperature,
LP refinement, active-set coordinate descent, and diverse initializations for
minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key differences from previous approach:
- FISTA (Nesterov momentum) instead of Adam for faster convergence
- Adaptive temperature based on gap between smooth and true max
- Projected gradient in log-space with careful step-size control
- Multi-phase strategy with increasing precision
- EVoLUTIONARY refinement in final phase for escaping local minima

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_max_and_grad(a, beta):
    """
    Compute logsumexp(beta * conv) as smooth approximation of max(conv),
    and its gradient w.r.t. a.
    Returns (smooth_value, gradient_array_of_length_n).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    smooth_val = cmax + np.log(e_sum) / beta

    # Gradient via FFT cross-correlation
    size = 1
    while size < 2 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad = 2.0 * cc[:n]

    return smooth_val, grad


def nesterov_opt(a_init, time_budget, beta_start=10.0, beta_end=500.0,
                 lr=0.005, max_iters=50000):
    """
    FISTA (Nesterov accelerated gradient) on smoothed objective in log-space.
    Minimizes 2n * LS_beta(conv(a)) / (sum(a))^2 with a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a_init
    a = a / s

    # Log-space parameterization
    x = np.log(np.maximum(a, 1e-15))
    x -= np.max(x)

    # FISTA state
    y = x.copy()
    t = 1.0

    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive temperature
        progress = min(1.0, it / max(1, int(max_iters * 0.7)))
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute a from Nesterov lookahead y
        y_c = y - np.max(y)
        a_y = np.exp(y_c)
        a_y = a_y / np.sum(a_y)

        # Smoothed objective gradient
        smooth_val, grad_a = smooth_max_and_grad(a_y, beta)
        sum_a = np.sum(a_y)

            lp_time = min(15.0, (time_budget - (time.monotonic() - start)) * 0.4)
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': lp_time})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                s_g = float(np.sum(g))
                if s_g > 1e-10:
                    g = g / s_g
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_try = (1 - t) * a_norm + t * g
                        a_try = np.maximum(a_try, 0)
                        s_try = np.sum(a_try)
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
        except Exception:
            pass

    return best_a


def active_set_cd(a_init, time_budget=30.0, n_top=8):
    """
    Coordinate descent focusing on variables contributing to top peaks.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    rng = np.random.default_rng(123)

    for outer in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        top_peaks = np.argsort(conv)[::-1][:n_top]

        improved = False
        for p in top_peaks:
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)

            for i in range(i_min, i_max + 1):
                if time.monotonic() - start > time_budget:
                    break
                j = p - i
                if j < 0 or j >= n:
                    continue
                if j == i:
                    continue

                for delta_frac in [0.01, 0.02, 0.005, -0.01, -0.02]:
                    delta = delta_frac * max(best_a[i], 1e-10)
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break

        if not improved:
            # Random perturbation to escape
            idx = rng.integers(0, n)
            a_try = best_a.copy()
            a_try[idx] *= (1.0 + rng.uniform(-0.03, 0.03))
            if a_try[idx] < 0:
                a_try[idx] = 0.0
            s_try = np.sum(a_try)
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True

        if not improved and outer > 50:
            break

    return best_a


def evolutionary_refine(a_init, time_budget=60.0, pop_size=6, seed=42):
    """
    Simple evolutionary strategy: maintain population, mutate best, evaluate.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)

    # Initialize population
    pop = [a_init.copy() / np.sum(a_init)]
    best_obj = compute_obj(pop[0])

    for i in range(1, pop_size):
        a_mut = pop[0] * (1.0 + rng.normal(0, 0.05, n))
        a_mut = np.maximum(a_mut, 0)
        if np.sum(a_mut) > 1e-10:
            a_mut = a_mut / np.sum(a_mut)
        pop.append(a_mut)

    start = time.monotonic()
    gen = 0

    while time.monotonic() - start < time_budget:
        gen += 1

        # Evaluate all in population
        objs = []
        for a in pop:
            objs.append(compute_obj(a))

        best_idx = int(np.argmin(objs))
        if objs[best_idx] < best_obj:
            best_obj = objs[best_idx]

        # Create next generation from top 2
        sorted_indices = np.argsort(objs)
        top2 = sorted_indices[:2]

        new_pop = []
        for _ in range(pop_size):
            parent = pop[top2[rng.integers(0, 2)]].copy()
            # Mutation
            mut_type = rng.integers(0, 3)
            if mut_type == 0:
                # Small Gaussian noise
                parent = parent * (1.0 + rng.normal(0, 0.02, n))
            elif mut_type == 1:
                # Sparse spike
                idx = rng.integers(0, n)
                parent[idx] += rng.uniform(0.01, 0.05) * np.mean(parent)
            else:
                # Blend with uniform
                alpha = rng.uniform(0.01, 0.05)
                parent = (1 - alpha) * parent + alpha / n

            parent = np.maximum(parent, 0)
            if np.sum(parent) > 1e-10:
                parent = parent / np.sum(parent)
            new_pop.append(parent)

        pop = new_pop

        # Periodic Nesterov polish on best
        if gen % 5 == 0 and time.monotonic() - start < time_budget - 10:
            best_in_pop = pop[int(np.argmin([compute_obj(a) for a in pop]))]
            remaining = time_budget - (time.monotonic() - start)
            if remaining > 15:
                polished = nesterov_opt(best_in_pop, min(10.0, remaining * 0.3),
                                        100.0, 1000.0, 0.003, 10000)
                obj_p = compute_obj(polished)
                if obj_p < best_obj:
                    best_obj = obj_p
                    pop[0] = polished

    return pop[int(np.argmin([compute_obj(a) for a in pop]))]


def make_hann(n):
    """Hann window."""
    x = np.arange(n)
    h = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    return h / np.sum(h)


def make_gaussian(n, sigma=0.3):
    """Gaussian window."""
    x = np.arange(n)
    c = (n - 1) / 2.0
    g = np.exp(-0.5 * ((x - c) / (sigma * n)) ** 2)
    return g / np.sum(g)


def make_triangular(n):
    """Triangular (Bartlett) window."""
    x = np.arange(n)
    t = 1.0 - np.abs(2.0 * x / (n - 1) - 1.0)
    return t / np.sum(t)


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FISTA-based optimization with adaptive temperature,
    LP refinement, active-set CD, evolutionary strategy, and diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent = parent / np.sum(parent)
            obj = compute_obj(parent)
            print(f"Parent: n={parent.size}, obj={obj:.8f}")
            best_seq = parent.copy()
            best_score = obj
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_hann(700)
        best_score = compute_obj(best_seq)
        print(f"Hann init: n={len(best_seq)}, obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n={n_best}, budget={budget_s}s")

    # === Phase 1: Multi-phase Nesterov on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-phase Nesterov (n={n_best})")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.25, 250.0)
    a_new = multi_phase_nesterov(best_seq, alloc)
    obj = compute_obj(a_new)
    if obj < best_score:
        best_score = obj
        best_seq = a_new.copy()
        print(f"  Nesterov: {best_score:.8f} ***")
    else:
        print(f"  Nesterov: {obj:.8f} (no improvement)")

    # === Phase 2: LP refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 2: LP refinement")
    print(f"{'='*60}")

    if len(best_seq) <= 900:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.12, 60.0)
        a_lp = lp_refine(best_seq, alloc)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  LP: {best_score:.8f} ***")
        else:
            print(f"  LP: {obj:.8f} (no improvement)")

        # Nesterov after LP
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.08, 30.0)
            a_post = multi_phase_nesterov(best_seq, alloc)
            obj = compute_obj(a_post)
            if obj < best_score:
                best_score = obj
                best_seq = a_post.copy()
                print(f"  LP+Nesterov: {best_score:.8f} ***")

    # === Phase 3: Active set coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Active set CD")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.08, 40.0)
    a_as = active_set_cd(best_seq, alloc)
    obj = compute_obj(a_as)
    if obj < best_score:
        best_score = obj
        best_seq = a_as.copy()
        print(f"  Active set CD: {best_score:.8f} ***")
    else:
        print(f"  Active set CD: {obj:.8f} (no improvement)")

    # === Phase 4: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_tries = sorted(set([
        n_best, n_best + 50, n_best + 100, n_best + 200,
        n_best + 400, n_best - 50,
    ]))
    n_tries = [n for n in n_tries if 300 <= n <= 2000]
    n_tries = [n for n in n_tries if n != n_best] + [n_best]

    for n_try in n_tries:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample(best_seq, n_try)

        alloc = min(remaining * 0.1, 35.0)
        a_opt = multi_phase_nesterov(a_try, alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  n={n_try}: {obj:.8f}")

    # === Phase 5: Evolutionary refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Evolutionary refinement")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.12, 60.0)
    a_evo = evolutionary_refine(best_seq, alloc, pop_size=5, seed=seed)
    obj = compute_obj(a_evo)
    if obj < best_score:
        best_score = obj
        best_seq = a_evo.copy()
        print(f"  Evolutionary: {best_score:.8f} ***")
    else:
        print(f"  Evolutionary: {obj:.8f} (no improvement)")

    # === Phase 6: Diverse restarts with Nesterov ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Diverse restarts")
    print(f"{'='*60}")

    rng = np.random.default_rng(seed + 1)
    n_final = len(best_seq)
    best_norm = best_seq / np.sum(best_seq)

    restarts = []
    # Gaussian perturbation
    for scale in [0.03, 0.06, 0.1]:
        a_p = best_norm * (1 + rng.normal(0, scale, n_final))
        a_p = np.maximum(a_p, 0)
        if np.sum(a_p) > 1e-10:
            restarts.append(a_p / np.sum(a_p))
    # Blend with structured windows
    for window, alpha in [(make_hann(n_final), 0.2), (make_gaussian(n_final), 0.3),
                          (make_hann(n_final), 0.4), (make_triangular(n_final), 0.3)]:
        restarts.append((1 - alpha) * best_norm + alpha * window)
    # Different lengths
    for n_alt in [n_final + 150, n_final - 100]:
        if 300 <= n_alt <= 2000:
            restarts.append(resample(best_seq, n_alt))

    for r_idx, a_p in enumerate(restarts):
        if time.monotonic() > deadline - 15:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        alloc = min(remaining * 0.12, 20.0)
        a_opt = multi_phase_nesterov(a_p, alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Restart {r_idx}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  Restart {r_idx}: {obj:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()

        # Very high beta Nesterov
        alloc1 = min(remaining * 0.4, 30.0)
        a_final = nesterov_opt(best_seq, alloc1, 500.0, 5000.0, 0.002, 50000)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  High-beta Nesterov: {best_score:.8f} ***")

        # Active set on final
        if time.monotonic() < deadline - 3:
            remaining = deadline - time.monotonic()
            a_as = active_set_cd(best_seq, min(remaining * 0.4, 15.0))
            obj = compute_obj(a_as)
            if obj < best_score:
                best_score = obj
                best_seq = a_as.copy()
                print(f"  Final CD: {best_score:.8f} ***")

    # Final validation
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
