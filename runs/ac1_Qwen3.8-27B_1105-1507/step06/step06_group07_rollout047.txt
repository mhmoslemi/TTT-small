Let me think about the subgradient method more carefully.

The objective (with sum(a) = 1) is: f(a) = 2n * max_k c_k where c_k = conv(a,a)[k].

The subgradient of max_k c_k at a is: for k* = argmax c_k, the subgradient is ∇c_k*(a) = 2 * reverse_padding(a, k*).

Specifically, `c_k = sum_i a[i] * a[k-i]` (with valid indices).
`dc_k/da[j] = a[k-j] + a[j-k]` (when both indices are valid).

Wait, more carefully: `c_k = sum_i a[i] * a[k-i]` where i ranges over [max(0, k-n+1), min(k, n-1)].

`dc_k/da[j] = a[k-j]` (if k-j is in [0,n-1]) `+ a[j-k]` (if j-k is in [0,n-1]).

So `dc_k/da[j] = a[k-j] * 1_{0<=k-j<n} + a[j-k] * 1_{0<=j-k<n}`.

This is essentially the cross-correlation of `a` with a shifted version, evaluated at position j.

For the subgradient method:
- At each step, compute c = conv(a, a), find k* = argmax c.
- Compute subgradient g = ∇c_k*(a).
- Update: a_new = a - eta * g (projected to non-negative, sum=1).

The challenge is choosing eta. Polyak step-size would be eta = (f(a) - f*)/||g||^2, but we don't know f*. However, we know f* >= 2n * sum_a^2/(2n-1) / sum_a^2 = 2n/(2n-1) (the flat bound). Wait, that's the lower bound on the objective, which is 2n/(2n-1) ≈ 1.

Actually, let me reconsider. With sum(a) = 1, the lower bound on max_k c_k is 1/(2n-1) (since sum of all c_k = 1, there are 2n-1 of them, so max >= average = 1/(2n-1)). So f* >= 2n/(2n-1).

For n=700, f* >= 1400/1399 ≈ 1.0007. So the gap between the lower bound and our current solution (1.506) is significant.

Polyak step-size: eta = (f(a) - f_lb) / ||g||^2, where f_lb = 2n/(2n-1).

This should give a good step-size!

But wait, we also need to handle the constraint sum(a) = 1. The subgradient method with projection onto the simplex {a >= 0, sum(a) = 1} would work.

Actually, since the objective is scale-invariant, we can work in the simplex without loss of generality.

then normalize to sum to 1.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def subgradient_at_max(a, k_star):
    """Compute subgradient of conv(a,a)[k_star] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        # dc_k/da[j] = a[k-j] (if valid) + a[j-k] (if valid)
        val = 0.0
        idx1 = k_star - j
        if 0 <= idx1 < n:
            val += a[idx1]
        idx2 = j - k_star
        if 0 <= idx2 < n:
            val += a[idx2]
        g[j] = val
    return g

def subgradient_fft(a, k_star):
    """Compute subgradient using FFT for efficiency."""
    n = len(a)
    # dc_k/da[j] = a[k-j] + a[j-k] (with valid index checks)
    # This is the cross-correlation of a with a, evaluated at specific offsets
    # corr[j] = sum_i a[i] a[i + (k_star - j)] ... hmm, let me think differently
    
    # Actually: dc_k/da[j] = a[k_star - j] * 1[0 <= k_star - j < n] + a[j - k_star] * 1[0 <= j - k_star < n]
    
    # Part 1: a[k_star - j] for j in [k_star - n + 1, k_star]
    # Part 2: a[j - k_star] for j in [k_star, k_star + n - 1]
    
    g = np.zeros(n)
    
    # Part 1: j ranges from max(0, k_star - n + 1) to min(k_star, n - 1)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_start <= j_end:
        # a[k_star - j] for j in [j_start, j_end]
        idx = k_star - np.arange(j_start, j_end + 1)
        g[j_start:j_end + 1] = a[idx]
    
    # Part 2: j ranges from max(k_star, 0) to min(k_star + n - 1, n - 1)
    j_start2 = max(k_star, 0)
    j_end2 = min(k_star + n - 1, n - 1)
    if j_start2 <= j_end2:
        # a[j - k_star] for j in [j_start2, j_end2]
        idx2 = np.arange(j_start2, j_end2 + 1) - k_star
        g[j_start2:j_end2 + 1] += a[idx2]
    
    return g

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def polyak_subgradient(a, f_lb, time_budget=30.0, max_iters=5000):
    """Subgradient method with Polyak step-size for minimizing max conv on simplex."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s  # normalize to simplex
    
    best_a = a.copy()
    best_f = 2.0 * n * float(np.max(compute_conv(a)))  # with sum=1
    best_exact = compute_obj(a)
    
    start = time.monotonic()
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        f_val = 2.0 * n * float(np.max(conv))
        
        if f_val < best_f - 1e-10:
            best_f = f_val
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 200:
            break
        
        # Find argmax
        k_star = int(np.argmax(conv))
        
        # Compute subgradient
        g = subgradient_fft(a, k_star)
        g_norm_sq = float(np.dot(g, g))
        
        if g_norm_sq < 1e-20:
            break
        
        # Polyak step-size
        gap = f_val - f_lb
        if gap < 1e-6:
            break
        eta = gap / g_norm_sq
        
        # Projected subgradient step: a_new = proj_simplex(a - eta * g)
        a_try = project_simplex(a - eta * g)
        
        # Check if it's an improvement (subgradient method doesn't guarantee monotone decrease)
        # Use a backtracking-like approach
        best_step = 0.0
        for frac in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            a_candidate = project_simplex(a - frac * eta * g)
            conv_c = compute_conv(a_candidate)
            f_c = 2.0 * n * float(np.max(conv_c))
            if f_c < f_val:
                best_step = frac
                a = a_candidate
                break
        
        if best_step == 0.0:
            # No improvement with subgradient; try a small random perturbation
            if it % 50 == 0:
                noise = np.random.default_rng(it).normal(0, 0.01, n)
                a = project_simplex(a + noise)
            else:
                # Reduce step and continue
                a = project_simplex(a - 0.01 * eta * g)
    
    return best_a, best_exact

def lp_direction(a, M, n_constraints=None):
    """Find direction g that maximizes sum(g) subject to conv(a,g) <= M, g >= 0.
    Optionally only enforce the top n_constraints constraints."""

        return None
    
    conv = compute_conv(a)

        A_rows = []
        b_ub = []
        for k in top_idx:
            row = np.zeros(n)
            # conv(a,g)[k] = sum_i a[i] g[k-i]
            for i in range(n):
                idx = k - i
                if 0 <= idx < n:
                    row[i] = a[idx]  # Wait, this isn't right
            # Actually conv(a,g)[k] = sum_i a[i] * g[k-i]
            # So the row for constraint k is: row[j] = a[k-j] for valid j
            row = np.zeros(n)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    row[j] = a[idx]
            A_rows.append(row)
            b_ub.append(M)
        A = np.array(A_rows)
        b = np.array(b_ub)
    else:
        # All constraints
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):

                idx = k - j
                if 0 <= idx < n:
                    A[k, j] = a[idx]
        b = np.full(2 * n - 1, M)

                      options={'presolve': True, 'time_limit': 30})

            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:

    except Exception:
        pass
    return None

def build_toeplitz_row(a, k):
    """Build the row of the Toeplitz matrix for constraint k.
    conv(a,g)[k] = sum_j a[k-j] * g[j] (where indices are valid)."""
    n = len(a)
    row = np.zeros(n)
    for j in range(n):
        idx = k - j
        if 0 <= idx < n:
            row[j] = a[idx]
    return row

def lp_direction_fast(a, M, top_k=50):

def multiplicative_update(a, conv, f_lb, time_budget=30.0, max_iters=3000):
    """Multiplicative (exponentiated gradient) update."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_f = 2.0 * n * float(np.max(conv))
    
    start = time.monotonic()
    eta = 1.0 / np.sqrt(n)  # initial step size
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        f_val = 2.0 * n * float(np.max(conv))
        
        if f_val < best_f - 1e-10:
            best_f = f_val
            best_a = a.copy()
        
        k_star = int(np.argmax(conv))
        g = subgradient_fft(a, k_star)
        
        # Multiplicative update: a[i] *= exp(-eta * g[i])
        # This is equivalent to a[i] = a[i] * (1 - eta * g[i] + ...) for small eta * g[i]
        log_update = -eta * g
        # Clip to avoid overflow
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        # Check improvement
        conv_new = compute_conv(a_new)
        f_new = 2.0 * n * float(np.max(conv_new))
        
        if f_new < f_val:
            a = a_new
            conv = conv_new
            # Decrease eta slowly
            eta *= 0.999
        else:
            # Increase eta (we're not making progress, maybe step is too small? No, too large)
            # Actually if not improving, the step was too large. Reduce eta.
            eta *= 0.5
            # Try a smaller step
            log_update_small = -eta * g
            log_update_small = np.clip(log_update_small, -10, 10)
            a_try = a * np.exp(log_update_small)
            s_try = np.sum(a_try)
            if s_try > 1e-10:
                a_try = a_try / s_try
                conv_try = compute_conv(a_try)
                f_try = 2.0 * n * float(np.max(conv_try))
                if f_try < f_val:
                    a = a_try
                    conv = conv_try
        
        # Periodic check
        if it % 100 == 0:
            exact = compute_obj(a)
            if exact < compute_obj(best_a):
                best_a = a.copy()
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def coordinate_descent(a, time_budget=30.0):
    """Coordinate-wise optimization: try adjusting each coordinate."""
    n = len(a)

        # Try each coordinate in random order

                break

            old_val = best_a[j]
            if old_val < 1e-8:
                continue
            
            for frac in [0.5, 0.7, 0.85, 0.95]:

                    best_a[j] = old_val
                    continue
                best_a += removed * (best_a / best_a.sum())  # This isn't quite right
                # Actually let's just normalize

                s_try = best_a.sum()
                if s_try < 1e-10:
