"""
Multi-phase optimizer combining L-BFGS on smoothed logsumexp objective,
LP-based active-set refinement, diverse parametric initializations,
multi-n exploration, and subgradient polishing to minimize
2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: L-BFGS (quasi-Newton) replaces Adam,
better active-set identification, trust-region LP refinement,
and structured parametric diversity.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_and_grad_x(x, beta):
    """
    Compute smoothed objective and gradient w.r.t. log-space x.
    a = softmax(x), objective = 2n * LSE(beta * conv(a)) (since sum(a)=1).
    """
    n = len(x)
    # Numerically stable softmax
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    a = a / np.sum(a)

    # Compute convolution
    conv = compute_conv(a)

    # Log-sum-exp of beta * conv
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum  # softmax weights over lags
    ls = cmax + np.log(e_sum) / beta

    # Objective (S=1 since a is normalized)
    obj = 2.0 * n * ls

    # Gradient of LSE w.r.t. a: grad_a[i] = 2*beta*(w ~~ a)[i] - 2n*ls (chain rule for /S^2 when S=1)
    # Actually for the full objective 2n*ls/S^2 with S=1: grad = 2n * d(ls)/d(a)
    # d(ls)/d(a[i]) = 2*beta*sum_k w_k * a[k-i] = 2*beta*cc[i]
    # where cc[i] = sum_k w_k * a[k-i] (cross-correlation)

    # Compute cc via FFT
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    cc = cc[:n]

    # grad_a[i] = 2n * 2*beta * cc[i] (since S=1, the -ls/S term is -ls)
    # Full: grad_obj/da[i] = (2n/S^2)*(2*beta*cc[i]) - (2n*ls/S^3)*1
    # With S=1: = 2n*2*beta*cc[i] - 2n*ls
    grad_a = 2.0 * n * 2.0 * beta * cc - 2.0 * n * ls

    # Chain rule for softmax: grad_x[j] = a[j]*(grad_a[j] - sum_i a[i]*grad_a[i])
    mean_grad = float(np.dot(a, grad_a))
    grad_x = a * (grad_a - mean_grad)

    return obj, grad_x


def smooth_obj_and_grad_a(a, beta):
    """Compute smoothed objective and gradient w.r.t. a directly (a assumed normalized, sum=1)."""
    n = len(a)
    conv = compute_conv(a)

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    cc = cc[:n]

    grad_a = 2.0 * n * 2.0 * beta * cc - 2.0 * n * ls

    return obj, grad_a, w, ls


def lbfgs_smooth_opt(a, beta, maxiter=200, time_budget=30.0):
    """Run L-BFGS-B on the smoothed objective in log-space."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    a = np.maximum(a, 1e-15)

    x0 = np.log(a)
    x0 -= np.mean(x0)

    start = time.monotonic()

    def fun(x):
        if time.monotonic() - start > time_budget:
            raise RuntimeError("time exceeded")
        obj, grad = smooth_obj_and_grad_x(x, beta)
        if not np.isfinite(obj):
            return 1e10, np.zeros_like(x)
        return obj, grad

    try:
        res = minimize(fun, x0, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': maxiter * 3,
                                'ftol': 1e-12, 'gtol': 1e-8, 'iprint': 0})
    except (RuntimeError, Exception):
        return a

    if res.x is not None and np.all(np.isfinite(res.x)):
        x_new = res.x - np.max(res.x)
        a_new = np.exp(x_new)
        a_new = a_new / np.sum(a_new)
        if compute_obj(a_new) < compute_obj(a):
            return a_new
    return a


def lbfgs_smooth_opt_a(a, beta, maxiter=100, time_budget=20.0):
    """Run L-BFGS-B directly on a (with projection to simplex)."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    a = np.maximum(a, 1e-15)

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    def fun_and_grad(y):
        if time.monotonic() - start > time_budget:
            raise RuntimeError("time exceeded")
        # Project to simplex (a >= 0, sum = 1)
        a_try = np.maximum(y, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            return 1e10, np.zeros_like(y)
        a_try = a_try / s_try
        obj, grad_a, _, _ = smooth_obj_and_grad_a(a_try, beta)
        if not np.isfinite(obj):
            return 1e10, np.zeros_like(y)
        return obj, grad_a / s_try  # adjust for normalization

    try:
        res = minimize(fun_and_grad, a, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': maxiter * 2,
                                'ftol': 1e-12, 'gtol': 1e-8, 'iprint': 0})
    except (RuntimeError, Exception):
        pass

    if res.x is not None and np.all(np.isfinite(res.x)):
        a_new = np.maximum(res.x, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_a = a_new
                best_obj = obj_new

    return best_a


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k] for k=0,...,2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_active_set_refine(a, time_budget=30.0, trust_delta=0.3):
    """
    LP-based refinement using linearized active constraints.
    Solves: min M s.t. linearized (a*g)[k] <= M for active k,
            sum(g)=1, g>=0, ||g-a||_1 <= trust_delta.
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Identify active set: lags within 5% of max
    threshold = max_conv * 0.995
    active_lags = np.where(conv >= threshold)[0]

    if len(active_lags) == 0:
        active_lags = np.array([np.argmax(conv)])

    # Limit active set size
    if len(active_lags) > 50:
        # Take the top 50
        top_idx = np.argsort(conv)[::-1][:50]
        active_lags = top_idx

    # Solve LP
    # Variables: g[0..n-1], M
    # Minimize: M
    # Subject to: for each active lag k: sum_i 2*a[k-i]*(g[i]-a[i]) + conv[k] <= M
    #             sum(g) = 1, g >= 0, sum|g-a| <= trust_delta
    #
    # The trust region constraint sum|g-a| <= delta is handled by:
    # g[i] <= a[i] + delta and g[i] >= a[i] - delta (relaxed, per-coordinate)

    try:
        # c = [0,...,0, 1] (minimize M, last variable)
        c = np.zeros(n + 1)
        c[-1] = 1.0

        # Inequality constraints: A_ub @ vars <= b_ub
        # For each active lag k: 2*sum_i a[k-i]*g[i] - M <= 2*sum_i a[k-i]*a[i] - conv[k]
        # Rearranged: 2*sum_i a[k-i]*g[i] - M <= max_conv (since conv[k] ≈ max_conv for active)
        # Actually more precisely: conv[k] + 2*sum_i a[k-i]*(g[i]-a[i]) <= M
        # => 2*sum_i a[k-i]*g[i] - 2*sum_i a[k-i]*a[i] + conv[k] - M <= 0
        # => 2*sum_i a[k-i]*g[i] - M <= 2*sum_i a[k-i]*a[i] - conv[k]

        A_ub = np.zeros((len(active_lags), n + 1))
        b_ub = np.zeros(len(active_lags))

        for idx, k in enumerate(active_lags):
            # Row: 2*a[k-i] for i=0..n-1 (where valid), -1 for M
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    A_ub[idx, i] = 2.0 * a[j]
            A_ub[idx, -1] = -1.0
            # RHS: 2*sum_i a[k-i]*a[i] - conv[k]
            # But 2*sum_i a[k-i]*a[i] = 2*conv[k] (by the derivative formula)
            # So RHS = 2*conv[k] - conv[k] = conv[k]
            b_ub[idx] = conv[k]

        # Trust region: g[i] <= a[i] + trust_delta, g[i] >= a[i] - trust_delta
        # g[i] - (a[i] + trust_delta) <= 0 => g[i] <= a[i] + trust_delta
        # (a[i] - trust_delta) - g[i] <= 0 => -g[i] <= trust_delta - a[i]

        A_trust1 = np.zeros((n, n + 1))
        for i in range(n):
            A_trust1[i, i] = 1.0
        b_trust1 = a + trust_delta

        A_trust2 = np.zeros((n, n + 1))
        for i in range(n):
            A_trust2[i, i] = -1.0
        b_trust2 = np.maximum(trust_delta - a, 0)

        # Combine constraints
        A_all = np.vstack([A_ub, A_trust1, A_trust2])
        b_all = np.concatenate([b_ub, b_trust1, b_trust2])

        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # Bounds: g[i] >= 0, M >= 0
        bounds = [(0.0, None)] * n + [(0.0, None)]

        res = linprog(c, A_ub=A_all, b_ub=b_all, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_budget * 0.5})

        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s_g = np.sum(g)
            if s_g > 1e-10:
                g = g / s_g
                # Try blending
                for t in [0.3, 0.5, 0.7, 0.8, 1.0]:
                    if time.monotonic() - start > time_budget:
                        break
                    a_try = (1 - t) * a + t * g
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()

    except Exception:
        pass

    return best_a


def subgradient_polish(a, time_budget=20.0, lr_start=0.01, n_iters=3000):
    """
    Subgradient method on the active set for final polish.
    Uses the subgradient of max(conv) at the current point.
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    lr = lr_start

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Active set: lags within 1% of max
        threshold = max_conv * 0.999
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Subgradient: average over active lags
        # d(conv[k])/da[i] = 2*a[k-i] (with a[j]=0 for j outside [0,n-1])
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(active)

        # Full subgradient of objective (with S=1): 2n*grad - 2n*max_conv * grad_S
        # d(S)/da[i] = 1, so full grad = 2n*grad - 2n*max_conv
        # But we're on the simplex, so we project: subtract mean
        full_grad = 2.0 * n * grad - 2.0 * n * max_conv
        full_grad -= np.mean(full_grad)  # project to tangent of simplex

        # Update with projection onto simplex
        lr_t = lr * (0.999 ** it)
        a_new = a - lr_t * full_grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            # Step back if no improvement for a while
            pass
        a = a_new

    return best_a


def pair_swap_refine(a, time_budget=20.0, seed=42):
    """
    Local search: try moving small amounts of mass between pairs (i,j)
    to reduce the maximum convolution value.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    conv = compute_conv(best_a)
    max_lag = int(np.argmax(conv))

    # Elements most contributing to the peak lag
    peak_contrib = np.zeros(n)
    for i in range(n):
        j = max_lag - i
        if 0 <= j < n:
            peak_contrib[i] = best_a[i] * best_a[j]

    # Sort by contribution (highest first)
    top_idx = np.argsort(peak_contrib)[::-1][:min(20, n)]

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        # Try pairs from top contributors
        for _ in range(10):
            i = int(rng.choice(top_idx))
            j = int(rng.integers(0, n))
            if i == j:
                continue

            for frac in [0.01, 0.005, 0.02, 0.001]:
                delta = frac * max(best_a[i], best_a[j], 1e-6)
                if best_a[i] - delta < 0:
                    continue
                a_try = best_a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    conv = compute_conv(best_a)
                    max_lag = int(np.argmax(conv))
                    peak_contrib = np.zeros(n)
                    for ii in range(n):
                        jj = max_lag - ii
                        if 0 <= jj < n:
                            peak_contrib[ii] = best_a[ii] * best_a[jj]
                    top_idx = np.argsort(peak_contrib)[::-1][:min(20, n)]
                    improved = True
                    break

        if not improved and it > 200:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines L-BFGS, LP refinement, subgradient polish,
    pair swaps, parametric diversity, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.8f}, n={len(best_seq) if best_seq is not None else '?'}")

    # === Phase 1: L-BFGS optimization on parent ===
    print("\n" + "=" * 50)
    print("Phase 1: L-BFGS on parent")
    print("=" * 50)

    if best_seq is not None:
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        # Multiple L-BFGS runs with different beta
        beta_schedule = [5, 10, 20, 40, 80, 150, 300]
        for beta in beta_schedule:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.1, 40.0)

            a_new = lbfgs_smooth_opt(a, beta=beta, maxiter=100, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  L-BFGS beta={beta}: {best_score:.8f} ***")
            else:
                print(f"  L-BFGS beta={beta}: {obj_new:.8f}")

        # Also try direct-space L-BFGS
        for beta in [20, 50, 100]:
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.08, 30.0)

            a_new = lbfgs_smooth_opt_a(a, beta=beta, maxiter=80, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  L-BFGS-B direct beta={beta}: {best_score:.8f} ***")
            else:
                print(f"  L-BFGS-B direct beta={beta}: {obj_new:.8f}")

    # === Phase 2: Multi-n exploration ===
    print("\n" + "=" * 50)
    print("Phase 2: Multi-n exploration")
    print("=" * 50)

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = []
        for delta in [-50, -25, -10, 0, 10, 25, 50, 75, 100, 150]:
            n_try = n_best + delta
            if 100 <= n_try <= 2000:
                n_options.append(n_try)
        n_options = sorted(set(n_options))

        for n_try in n_options:
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if n_try == len(best_seq):
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s

            # L-BFGS optimize
            alloc = min(remaining * 0.1, 30.0)
            a_opt = lbfgs_smooth_opt(a_try, beta=30, maxiter=60, time_budget=alloc)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 3: Parametric family exploration ===
    print("\n" + "=" * 50)
    print("Phase 3: Parametric families")
    print("=" * 50)

    if best_seq is not None:
        n_base = len(best_seq)

        # Try various parametric forms
        parametric_starts = []

        # Blend with polynomial profiles
        for p in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
            x = np.arange(n_base) / max(n_base - 1, 1)
            # U-shaped: (x - 0.5)^2 type
            poly = (x - 0.5) ** 2
            poly = poly / np.sum(poly)
            for alpha in [0.1, 0.2, 0.3, 0.5]:
                a_p = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * poly
                parametric_starts.append(a_p / np.sum(a_p))

        # Exponential decay from both ends
        for rate in [0.5, 1.0, 1.5, 2.0]:
            x = np.arange(n_base)
            mid = n_base / 2.0
            exp_profile = np.exp(-rate * np.abs(x - mid) / n_base)
            exp_profile = exp_profile / np.sum(exp_profile)
            for alpha in [0.2, 0.4, 0.6]:
                a_p = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * exp_profile
                parametric_starts.append(a_p / np.sum(a_p))

        # Cosine modulation
        for freq in [1, 2, 3, 4]:
            x = np.arange(n_base)
            cos_mod = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / (n_base - 1))
            cos_mod = cos_mod / np.sum(cos_mod)
            for alpha in [0.1, 0.2, 0.3]:
                a_p = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * cos_mod
                parametric_starts.append(a_p / np.sum(a_p))

        # Limit number of starts
        if len(parametric_starts) > 12:
            # Take evenly spaced subset
            idxs = np.linspace(0, len(parametric_starts) - 1, 12, dtype=int)
            parametric_starts = [parametric_starts[i] for i in idxs]

        for pi, a_p in enumerate(parametric_starts):
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            alloc = min(remaining * 0.05, 15.0)
            a_opt = lbfgs_smooth_opt(a_p, beta=20, maxiter=50, time_budget=alloc)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Param start {pi}: {best_score:.8f} *** UPDATE")

    # === Phase 4: LP active-set refinement ===
    print("\n" + "=" * 50)
    print("Phase 4: LP active-set refinement")
    print("=" * 50)

    if best_seq is not None and len(best_seq) <= 1000:
        for round_num in range(4):
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # Decreasing trust region
            delta = 0.3 * (0.7 ** round_num)
            alloc = min(remaining * 0.15, 30.0)

            a_lp = lp_active_set_refine(best_seq, time_budget=alloc, trust_delta=delta)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  LP round {round_num} (delta={delta:.3f}): {best_score:.8f} ***")
            else:
                print(f"  LP round {round_num} (delta={delta:.3f}): {obj:.8f}")

            # Follow up with L-BFGS
            if obj < best_score + 0.001 and time.monotonic() < deadline - 20:
                a_follow = lbfgs_smooth_opt(a_lp, beta=100, maxiter=50,
                                            time_budget=min(remaining * 0.05, 15.0))
                obj_f = compute_obj(a_follow)
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_follow.copy()
                    print(f"  LP+L-BFGS round {round_num}: {best_score:.8f} ***")

    # === Phase 5: Subgradient polish ===
    print("\n" + "=" * 50)
    print("Phase 5: Subgradient polish")
    print("=" * 50)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.2, 30.0)
            a_sg = subgradient_polish(best_seq, time_budget=alloc, lr_start=0.005, n_iters=5000)
            obj = compute_obj(a_sg)
            if obj < best_score:
                best_score = obj
                best_seq = a_sg.copy()
                print(f"  Subgradient: {best_score:.8f} ***")
            else:
                print(f"  Subgradient: {obj:.8f}")

    # === Phase 6: Pair swap refinement ===
    print("\n" + "=" * 50)
    print("Phase 6: Pair swap refinement")
    print("=" * 50)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.2, 25.0)
            a_ps = pair_swap_refine(best_seq, time_budget=alloc, seed=seed + 1)
            obj = compute_obj(a_ps)
            if obj < best_score:
                best_score = obj
                best_seq = a_ps.copy()
                print(f"  Pair swap: {best_score:.8f} ***")
            else:
                print(f"  Pair swap: {obj:.8f}")

    # === Phase 7: Final L-BFGS polish with high beta ===
    print("\n" + "=" * 50)
    print("Phase 7: Final polish")
    print("=" * 50)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            for beta in [200, 500, 1000]:
                if time.monotonic() > deadline - 10:
                    break
                alloc = min(remaining * 0.15, 15.0)
                a_final = lbfgs_smooth_opt(best_seq, beta=beta, maxiter=100, time_budget=alloc)
                obj = compute_obj(a_final)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_final.copy()
                    print(f"  Final L-BFGS beta={beta}: {best_score:.8f} ***")
                else:
                    print(f"  Final L-BFGS beta={beta}: {obj:.8f}")
                remaining = deadline - time.monotonic()
                if remaining < 10:
                    break

    # === Phase 8: One more shot with different n near the best ===
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            for n_try in [n_best - 5, n_best + 5]:
                if time.monotonic() > deadline - 10:
                    break
                if 100 <= n_try <= 2000:
                    a_try = resample_sequence(best_seq, n_try)
                    s = np.sum(a_try)
                    if s > 1e-10:
                        a_try = a_try / s
                    alloc = min(remaining * 0.1, 10.0)
                    a_opt = lbfgs_smooth_opt(a_try, beta=50, maxiter=50, time_budget=alloc)
                    obj = compute_obj(a_opt)
                    if obj < best_score:
                        best_score = obj
                        best_seq = a_opt.copy()
                        print(f"  Fine n={n_try}: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'=' * 50}")

    return result
