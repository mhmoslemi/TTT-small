"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using iterated full-constraint LP
with perturbation-based local search, coordinate descent polish, and multi-start.
Key innovation: using ALL 2n-1 convolution constraints in the LP (not just top-k),
which gives the exact best response and enables proper convergence analysis.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_full_lp_matrix(a):
    """Build sparse matrix A for full convolution constraints.
    (a*g)[k] = sum_j A[k,j] * g[j]
    """
    n = len(a)
    conv_len = 2 * n - 1
    rows = []
    cols = []
    vals = []
    
    for k in range(conv_len):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            ms = k - js
            rows.extend(range(k, k + len(js)))
            cols.extend(js.tolist())
            vals.extend(a[ms].tolist())
    
    A = csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
    return A


def solve_lp_full(a, M, time_limit=20):
    """Solve full LP: max sum(g) s.t. A*g <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_full_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def iterated_lp_optimize(a_init, deadline, max_iters=500, patience=30):
    """Iterated LP: solve LP, blend, repeat until convergence."""
    a = np.maximum(a_init.astype(np.float64), 0.0)

    else:
        a = a / s

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

        if time.monotonic() > deadline:
            break
        
        M = fast_conv(a).max()
        g = solve_lp_full(a, M, time_limit=15)

            no_improve += 5

                break

        improved = False
        for t in [0.5, 0.3, 0.7, 0.2, 0.8, 0.4, 0.6, 0.1]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()

                    b_pt = a_pt

                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)

def fast_conv(a):

def compute_J(a):

def build_full_lp_matrix(a):
    """Build sparse matrix A for full convolution constraints: (a*g)[k] = sum_j A[k,j]*g[j]."""
    n = len(a)

    rows_list = []
    cols_list = []
    vals_list = []

            cols_list.extend(js.tolist())

def run_subgradient_fast(a_init, deadline, max_iters=10000):

            intercepts = np.zeros(k_hi - k_lo + 1)

                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = eval_obj(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = eval_obj(b_pt)
            
            x_opt = (lo + hi) / 2.0
            if eval_obj(0.0) <= eval_obj(x_opt):
                x_opt = 0.0
            x_opt = max(0.0, x_opt)
            
            if abs(x_opt - x_cur) > 1e-13:
                a[i] = x_opt
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        
        if M < best_M * (1.0 - 1e-14):
            best_M = M
            best_a = a.copy()
        
        if not sweep_improved:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0.0, 1.0, src.size)
    t_new = np.linspace(0.0, 1.0, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    
    # From parent (resampled)
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        
        p_noisy = np.maximum(p + rng.normal(0.0, 0.05, n), 1e-5)
        inits.append(p_noisy / p_noisy.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window (low freq)
    cos1 = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    cos1 = np.maximum(cos1, 0.01)
    inits.append(cos1 / cos1.sum())
    
    # Cosine window (med freq)
    cos2 = 0.5 + 0.5 * np.cos(4.0 * np.pi * t)
    cos2 = np.maximum(cos2, 0.01)
    inits.append(cos2 / cos2.sum())
    
    # Gaussian
    gauss = np.exp(-6.0 * (t - 0.5) ** 2) + 0.03
    inits.append(gauss / gauss.sum())
    
    # U-shaped
    u_shape = (2.0 * t - 1.0) ** 2 + 0.05
    inits.append(u_shape / u_shape.sum())
    
    # Random smooth
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search using iterated full-constraint LP, subgradient methods,
    perturbation-based local search, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0
    
    best_a = None
    best_score = float('inf')
    
    def try_update(a):
        nonlocal best_a, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_a = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            try_update(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512, dtype=np.float64)
        try_update(parent)
        print(f"No parent, uniform n=512, J={best_score:.8f}", flush=True)
    
    n_best = best_a.size
    
    # === Phase 1: Iterated LP from parent (350s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Iterated full-constraint LP from parent", flush=True)
    print(f"{'='*60}", flush=True)
    
    p1_deadline = t_start + 350
    a_lp1 = iterated_lp_optimize(parent, p1_deadline, max_iters=400, patience=30)
    try_update(a_lp1)
    
    # Continue from result
    if time.monotonic() < p1_deadline:
        a_lp2 = iterated_lp_optimize(a_lp1, p1_deadline, max_iters=100, patience=20)
        try_update(a_lp2)
    
    print(f"  End Phase 1: J={best_score:.8f} (t={time.monotonic()-t_start:.0f}s)", flush=True)
    
    # === Phase 2: Perturbation + LP (400s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Perturbation + LP re-optimization", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_deadline = t_start + 750
    eps_list_1 = [0.02, 0.03, 0.05, 0.07, 0.1, 0.15]
    a_pert1 = perturb_and_reoptimize(best_a, rng, p2_deadline, eps_list_1, n_trials_per_eps=2)
    try_update(a_pert1)
    
    # Second round with different eps
    if time.monotonic() < p2_deadline:
        eps_list_2 = [0.2, 0.3, 0.4, 0.5, 0.7]
        a_pert2 = perturb_and_reoptimize(best_a, rng, p2_deadline, eps_list_2, n_trials_per_eps=1)
        try_update(a_pert2)
    
    print(f"  End Phase 2: J={best_score:.8f} (t={time.monotonic()-t_start:.0f}s)", flush=True)
    
    # === Phase 3: Subgradient + LP from diverse inits (300s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-start with subgradient + LP", flush=True)
    print(f"{'='*60}", flush=True)
    
    p3_deadline = t_start + 1050
    inits = generate_diverse_inits(n_best, rng, best_a)
    
    for ii, a0 in enumerate(inits):
        if time.monotonic() > p3_deadline:
            break
        sub_dl_subgrad = min(p3_deadline, time.monotonic() + 20)
        a_sg = run_subgradient_fast(a0, sub_dl_subgrad, max_iters=3000)
        
        if time.monotonic() < p3_deadline:
            sub_dl_lp = min(p3_deadline, time.monotonic() + 40)
            a_opt = iterated_lp_optimize(a_sg, sub_dl_lp, max_iters=40, patience=12)
            try_update(a_opt)
    
    print(f"  End Phase 3: J={best_score:.8f} (t={time.monotonic()-t_start:.0f}s)", flush=True)
    
    # === Phase 4: Coordinate descent polish (150s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent polish + final LP", flush=True)
    print(f"{'='*60}", flush=True)
    
    p4_deadline = t_start + 1250
    a_cd = coordinate_descent_polish(best_a, min(p4_deadline, t_start + 1200),
                                      max_sweeps=25, seed=seed + 1)
    try_update(a_cd)
    
    # Final LP from the CD result
    if time.monotonic() < p4_deadline:
        a_lp_final = iterated_lp_optimize(best_a, p4_deadline, max_iters=50, patience=15)
        try_update(a_lp_final)
    
    print(f"  End Phase 4: J={best_score:.8f} (t={time.monotonic()-t_start:.0f}s)", flush=True)
    
    # === Phase 5: More perturbation rounds with tighter eps (150s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Fine perturbation on best", flush=True)
    print(f"{'='*60}", flush=True)
    
    if time.monotonic() < deadline - 10:
        eps_fine = [0.01, 0.015, 0.02, 0.03, 0.04]
        a_fine = perturb_and_reoptimize(best_a, rng, deadline - 10, eps_fine, n_trials_per_eps=3)
        try_update(a_fine)
    
    # === Phase 6: Try different lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if time.monotonic() < deadline - 20:
        for mult in [1.25, 0.8, 1.5, 0.67]:
            if time.monotonic() > deadline - 20:
                break
            new_n = int(n_best * mult)
            new_n = max(128, min(new_n, 2560))
            new_n = (new_n // 32) * 32
            if new_n == n_best:
                continue
            a_ext = resample_seq(best_a, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 10, time.monotonic() + 25)
            a_opt = iterated_lp_optimize(a_ext, sub_dl, max_iters=30, patience=10)
            try_update(a_opt)
    
    # Final check
    if time.monotonic() < deadline:
        final_lp = iterated_lp_optimize(best_a, deadline, max_iters=30, patience=8)
        try_update(final_lp)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size if best_a is not None else 0} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_a is None:
        best_a = np.ones(512)
        best_score = compute_J(best_a)
    
    return [float(x) for x in best_a]
