"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: Adam log-space with temperature annealing, LP with relaxed constraints,
coordinate descent, multi-n exploration, and random restarts.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_obj_grad(a, beta):
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj

def adam_log_opt(a, max_iters=5000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995):
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    x = np.log(a)
    x -= np.mean(x)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)
        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a
        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        if it % 200 == 0:
            x -= np.mean(x)
        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    return best_a

def lp_refine_relaxed(a, time_budget=30.0, K=20):
    """LP refinement with relaxed constraints: only constrain top K lags."""
    n = len(a)
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    conv = compute_conv(a_norm)
    M = float(np.max(conv))
    
    # Find top K lags
    top_k_indices = np.argsort(conv)[::-1][:K]
    
    # Build constraint matrix for only these K lags
    A_ub = np.zeros((K, n))
    for idx, k in enumerate(top_k_indices):
        # (a*g)[k] = sum_i a[i] * g[k-i] for valid i
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i
            A_ub[idx, j] = a[i]
    
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n

def resample_sequence(a, new_n):

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: Adam log-space with temperature annealing, LP with relaxed constraints,
coordinate descent, multi-n exploration, random restarts, and proximal methods.
Key improvements: relaxed LP (top-K lags only), larger n exploration,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """Compute smoothed objective (logsumexp of beta*conv) and gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def adam_log_opt(a, max_iters=5000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995,
                 momentum_boost=1.0):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    x = np.log(a)
    x -= np.mean(x)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)
        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a
        t += 1
        lr_t = lr_start * (lr_decay ** it) * momentum_boost
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        if it % 200 == 0:
            x -= np.mean(x)
        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    return best_a


def proximal_gradient_opt(a, max_iters=5000, time_budget=50.0, beta=50.0, lr=0.01):
    """Proximal gradient: gradient step + projection onto simplex-like set."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        obj_sm, grad_a = smooth_obj_grad(a, beta)
        a_new = a - lr * grad_a
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    return best_a


def lp_refine(a, time_budget=30.0, K=None):
    """LP refinement: find g maximizing sum(g) s.t. (a*g)_k <= M for top-K lags."""
    n = len(a)
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    conv = compute_conv(a_norm)
    M = float(np.max(conv))

    if K is None:
        if n <= 800:
            K = 2 * n - 1  # All lags
        else:
            K = min(50, 2 * n - 1)

    # Get top K lags (or all lags if K >= 2n-1)
    total_lags = 2 * n - 1
    if K >= total_lags:
        lag_indices = np.arange(total_lags)
    else:
        lag_indices = np.argsort(conv)[::-1][:K]

    # Build constraint matrix
    A_ub = np.zeros((K, n))
    for idx, k in enumerate(lag_indices):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i
            A_ub[idx, j] = a_norm[i]

    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n

    try:
        time_limit = min(time_budget * 0.4, 20)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                g = g / gs
                best_a = a_norm.copy()
                best_obj = compute_obj(a_norm)
                for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
                    if time.monotonic() - start > time_budget:
                        break
                    a_try = (1 - t) * a_norm + t * g
                    a_try = np.maximum(a_try, 0)
                    st = np.sum(a_try)
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a
    except Exception:
        pass
    return a


def coordinate_refine(a, time_budget=20.0, seed=0):
    """Focused coordinate descent on peak-contributing elements."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy().astype(np.float64)
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(seed)

    conv = compute_conv(best_a)
    peaks = np.argsort(conv)[::-1][:min(8, len(conv))]
    no_improve = 0

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        if no_improve > 200:
            break
        improved = False
        for p in peaks:
            if time.monotonic() - start > time_budget:
                break
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            # Compute contributions
            contribs = []
            for i in range(i_min, i_max + 1):
                j = p - i
                if j < n:
                    contribs.append((i, j, best_a[i] * best_a[j]))
            contribs.sort(key=lambda x: -x[2])

            for i, j, cval in contribs[:5]:
                if time.monotonic() - start > time_budget:
                    break
                # Try reducing a[i] (which contributes to peak p)
                for delta_frac in [0.01, 0.02, 0.05, 0.1]:
                    delta = delta_frac * best_a[i]
                    if delta > best_a[i] * 0.5:
                        continue
                    # Move mass to a random location far from peak
                    for _ in range(3):
                        target = rng.integers(0, n)
                        if abs(target - i) < n // 4:
                            target = (i + n // 2 + rng.integers(-n // 10, n // 10)) % n
                        a_try = best_a.copy()
                        a_try[i] -= delta
                        a_try[target] += delta
                        if a_try[i] < 0:
                            continue
                        st = np.sum(a_try)
                        if st < 1e-10:
                            continue
                        a_try = a_try / st
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            conv = compute_conv(a_try)
                            peaks = np.argsort(conv)[::-1][:min(8, len(conv))]
                            improved = True
                            no_improve = 0
                            break
        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def hann_window(n):
    """Generate a normalized Hann window of length n."""
    if n <= 1:
        return np.ones(n)
    t = np.arange(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
    w = np.maximum(w, 1e-10)
    return w / np.sum(w)


def raised_cosine(n, alpha=0.5):
    """Raised cosine window with parameter alpha."""
    if n <= 1:
        return np.ones(n)
    t = np.linspace(-1, 1, n)
    w = np.ones(n)
    mask = np.abs(t) > alpha
    w[mask] = 0.5 * (1 + np.cos(np.pi * (np.abs(t[mask]) - alpha) / (1 - alpha)))
    w = np.maximum(w, 1e-10)
    return w / np.sum(w)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization combining Adam in log-space,
    LP refinement with relaxed constraints, coordinate descent, multi-n exploration,
    and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = hann_window(700)
        best_score = compute_obj(best_seq)
        print(f"No parent, using Hann (n=700): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, budget={budget_s}s")

    # === Phase 1: Intensive Adam at parent length ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Adam log-space optimization (n={n_parent})")
    print(f"{'='*60}")

    a = best_seq.copy()
    s = np.sum(a)
    if s > 1e-10:
        a = a / s

    # Multiple Adam rounds with progressively higher beta
    rounds = [
        (3.0, 20.0, 0.06, 0.99997),
        (8.0, 50.0, 0.04, 0.99998),
        (20.0, 100.0, 0.03, 0.99998),
        (50.0, 200.0, 0.02, 0.99999),
        (100.0, 400.0, 0.015, 0.99999),
        (200.0, 600.0, 0.01, 0.99999),
        (400.0, 800.0, 0.008, 0.99999),
        (600.0, 1200.0, 0.005, 0.99999),
    ]

    for r_idx, (b_s, b_e, lr0, lr_dec) in enumerate(rounds):
        if time.monotonic() > deadline - 400:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        # Allocate more time to earlier rounds
        frac = max(0.05, 0.12 - 0.01 * r_idx)
        alloc = min(remaining * frac, 80.0)
        a_new = adam_log_opt(a, max_iters=30000, time_budget=alloc,
                             beta_start=b_s, beta_end=b_e,
                             lr_start=lr0, lr_decay=lr_dec)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Adam r{r_idx} (beta {b_s:.0f}->{b_e:.0f}): {best_score:.8f} ***")
        else:
            print(f"  Adam r{r_idx} (beta {b_s:.0f}->{b_e:.0f}): {obj:.8f}")

    # Interleaved LP refinement at parent length
    if time.monotonic() < deadline - 300 and n_parent <= 1200:
        for lp_r in range(4):
            if time.monotonic() > deadline - 250:
                break
            remaining = deadline - time.monotonic()
            K_use = min(30, 2 * n_parent - 1)
            a_lp = lp_refine(a, time_budget=min(remaining * 0.08, 25), K=K_use)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score - 1e-12:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a = a_lp.copy()
                print(f"  LP r{lp_r} (K={K_use}): {best_score:.8f} ***")
            else:
                print(f"  LP r{lp_r} (K={K_use}): {obj_lp:.8f}")
            # Quick Adam after LP
            if time.monotonic() < deadline - 200:
                rem2 = deadline - time.monotonic()
                a_post = adam_log_opt(a, max_iters=10000, time_budget=min(rem2 * 0.05, 15),
                                      beta_start=50.0, beta_end=300.0,
                                      lr_start=0.02, lr_decay=0.99999)
                obj_post = compute_obj(a_post)
                if obj_post < best_score - 1e-12:
                    best_score = obj_post
                    best_seq = a_post.copy()
                    a = a_post.copy()
                    print(f"  LP+Adam r{lp_r}: {best_score:.8f} ***")

    # === Phase 2: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Multi-n exploration")
    print(f"{'='*60}")

    n_candidates = sorted(set([n_parent, 900, 1200, 1500, 2000, 2500]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 3000]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        s = np.sum(a_try)
        if s > 1e-10:
            a_try = a_try / s

        # Adam optimization
        alloc = min(remaining * 0.15, 50.0)
        a_opt = adam_log_opt(a_try, max_iters=20000, time_budget=alloc,
                             beta_start=10.0, beta_end=400.0,
                             lr_start=0.03, lr_decay=0.99998)

        # LP if feasible
        if time.monotonic() < deadline - 50 and len(a_opt) <= 1200:
            rem2 = deadline - time.monotonic()
            a_opt = lp_refine(a_opt, time_budget=min(rem2 * 0.08, 20), K=30)

        obj = compute_obj(a_opt)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  n={n_try}: {obj:.8f}")

    # === Phase 3: Aggressive LP + Adam cycles on best ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Aggressive optimization (n={len(best_seq)})")
    print(f"{'='*60}")

    a = best_seq.copy()
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    n_cur = len(a)

    for cycle in range(8):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # LP step
        K_use = min(20 + cycle * 5, max(30, n_cur // 3))
        alloc_lp = min(remaining * 0.1, 25.0)
        a = lp_refine(a, time_budget=alloc_lp, K=K_use)

        # Adam step with increasing beta
        rem2 = deadline - time.monotonic()
        alloc_adam = min(rem2 * 0.15, 35.0)
        b_s = 30 + cycle * 40
        b_e = 150 + cycle * 150
        lr_use = 0.025 / (1.0 + 0.3 * cycle)
        a = adam_log_opt(a, max_iters=15000, time_budget=alloc_adam,
                         beta_start=b_s, beta_end=b_e,
                         lr_start=lr_use, lr_decay=0.99999)

        obj = compute_obj(a)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a.copy()
            print(f"  Cycle {cycle}: {best_score:.8f} ***")
        else:
            print(f"  Cycle {cycle}: {obj:.8f}")

    # === Phase 4: Proximal gradient + coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Proximal + Coordinate descent")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        # Proximal gradient
        a_pg = proximal_gradient_opt(best_seq, max_iters=5000,
                                     time_budget=min(remaining * 0.1, 20),
                                     beta=100.0, lr=0.005)
        obj_pg = compute_obj(a_pg)
        if obj_pg < best_score - 1e-12:
            best_score = obj_pg
            best_seq = a_pg.copy()
            print(f"  Proximal: {best_score:.8f} ***")
        else:
            print(f"  Proximal: {obj_pg:.8f}")

    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        a_cd = coordinate_refine(best_seq, time_budget=min(remaining * 0.15, 30.0),
                                 seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-12:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord descent: {best_score:.8f} ***")
        else:
            print(f"  Coord descent: {obj_cd:.8f}")

    # === Phase 5: Random restarts with diverse strategies ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Random restarts")
    print(f"{'='*60}")

    n_best = len(best_seq)
    rng = np.random.default_rng(seed)
    strategies = ['noise', 'hann_blend', 'cosine_blend', 'resample', 'asym']

    for restart in range(12):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        strat = strategies[restart % len(strategies)]

        if strat == 'noise':
            scale = 0.02 + 0.015 * min(restart, 5)
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
        elif strat == 'hann_blend':
            alpha = rng.uniform(0.1, 0.4)
            hw = hann_window(n_best)
            a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * hw
        elif strat == 'cosine_blend':
            alpha = rng.uniform(0.1, 0.3)
            cw = raised_cosine(n_best, alpha=rng.uniform(0.3, 0.7))
            a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * cw
        elif strat == 'resample':
            new_n = n_best + rng.integers(-200, 300)
            new_n = max(300, min(new_n, 2500))
            a_pert = resample_sequence(best_seq, new_n)
        else:  # asym - asymmetric perturbation
            a_pert = best_seq.copy()
            # Create an asymmetric version
            shift = rng.integers(n_best // 10, n_best // 4)
            a_shifted = np.zeros(n_best)
            if shift > 0:
                a_shifted[shift:] = best_seq[:-shift]
                a_shifted[:shift] = best_seq[-shift:] * 0.3
            a_pert = 0.7 * (best_seq / np.sum(best_seq)) + 0.3 * (a_shifted / max(np.sum(a_shifted), 1e-10))

        a_pert = np.maximum(a_pert, 1e-10)
        s_p = np.sum(a_pert)
        if s_p < 1e-10:
            continue
        a_pert = a_pert / s_p

        alloc = min(remaining * 0.15, 25.0)
        a_opt = adam_log_opt(a_pert, max_iters=15000, time_budget=alloc,
                             beta_start=5.0, beta_end=500.0,
                             lr_start=0.035, lr_decay=0.99998)
        obj = compute_obj(a_opt)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {restart} [{strat}] (n={n_best}): {best_score:.8f} ***")
        else:
            print(f"  Restart {restart} [{strat}]: {obj:.8f}")

    # === Final polish ===
    print(f"\n{'='*60}")
    print(f"Final polish")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        # Very high beta Adam
        a_final = adam_log_opt(best_seq, max_iters=10000,
                               time_budget=min(remaining * 0.5, 20.0),
                               beta_start=300.0, beta_end=1500.0,
                               lr_start=0.004, lr_decay=0.99999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-12:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  High-beta polish: {best_score:.8f} ***")

        # One more coordinate descent
        if time.monotonic() < deadline - 3:
            rem = deadline - time.monotonic()
            a_cd2 = coordinate_refine(best_seq, time_budget=min(rem * 0.4, 10.0),
                                      seed=seed + 1)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score - 1e-12:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  Final CD: {best_score:.8f} ***")

    # Validation and finalization
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    # Ensure sum is large enough
    if np.sum(best_seq) < 0.01:
        best_seq = best_seq / max(np.sum(best_seq), 1e-15) * 1.0

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
