    f = 2n * logsumexp(beta*conv) / (sum(a))^2
    """
    n = len(a)
    conv = compute_conv(a)
    S = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    phi = beta * cmax + np.log(e_sum)

    f = 2.0 * n * phi / (S * S)

    # Gradient of phi w.r.t. a[j] = 2*beta * (w cross-correlate a)[j]
    # (w cross-correlate a)[j] = sum_k w[k] * a[k-j]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    corr = irfft(W * np.conj(A), size)
    corr = corr[:n]

    grad_phi = 2.0 * beta * corr
    grad_f = (2.0 * n / (S * S)) * (grad_phi - phi / S)

    return f, grad_f


def l_bfgsb_opt(a, beta, time_budget=100.0, max_iter=5000):
    """Run L-BFGS-B on the logsumexp-smoothed objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    start = time.monotonic()

    def objective(x):
        x_nonneg = np.maximum(x, 0.0)
        s = np.sum(x_nonneg)
        if s < 1e-10:
            return 1e10, np.zeros_like(x)
        x_norm = x_nonneg / s
        val, grad = smooth_obj_and_grad(x_norm, beta)
        # Chain rule for normalization
        grad_norm = grad - (grad @ x_norm)  # project onto tangent of simplex
        return val, grad_norm

    x0 = a.copy()
    try:
        res = minimize(objective, x0, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-12,
                                'gtol': 1e-9, 'maxls': 50})
        if time.monotonic() - start < time_budget:
            x_best = np.maximum(res.x, 0.0)
            s = np.sum(x_best)
            if s > 1e-10:
                return x_best / s
    except Exception:
        pass
    return a


def nesterov_subgrad(a, time_budget=100.0, lr_init=0.01, momentum=0.9):
    """
    Nesterov-accelerated subgradient method on the exact (non-smoothed) objective.
    Uses the true max for the subgradient.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)

    v = np.zeros(n)
    lr = lr_init
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        # Nesterov lookahead
        a_look = a - momentum * v

        # Compute subgradient at a_look
        conv = compute_conv(a_look)
        S_look = np.sum(a_look)
        peak_idx = np.argmax(conv)
        k = peak_idx

        # Subgradient of conv[k] w.r.t. a: 2*a[k-j] for valid j
        # i.e., d(conv[k])/da[j] = 2*a[k-j] if 0<=k-j<n else 0
        sg_conv = np.zeros(n)
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            sg_conv[j] = 2.0 * a_look[k - j]

        # Subgradient of f = 2n*max_conv/S^2
        obj_val = 2.0 * n * float(conv[k]) / (S_look * S_look)
        sg = (2.0 * n / (S_look * S_look)) * (sg_conv - float(conv[k]) / S_look)

        # Update
        v = momentum * v + lr * sg
        a_new = a - v

        # Project: non-negative and normalized
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        # Check objective
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        # Decay learning rate
        lr = lr_init / np.sqrt(1.0 + it * 0.01)

    return best_a


def peak_shaving(a, time_budget=60.0, n_peaks=3, n_trials_per_peak=20):
    """
    Iteratively shave the peak by reducing elements that contribute most
    to the top convolution peaks and redistributing mass.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    while time.monotonic() - start < time_budget and it < 500:
        it += 1
        conv = compute_conv(best_a)
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        improved = False

        for pk in peak_indices:
            k = int(pk)
            # Elements contributing to conv[k]
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            contributors = list(range(j_min, j_max + 1))

            if not contributors:
                continue

            # Find elements that contribute to this peak but NOT to other top peaks
            other_contributors = set()
            for pp in peak_indices:
                if pp == k:
                    continue
                jp_min = max(0, int(pp) - (n - 1))
                jp_max = int(pp) if int(pp) < n - 1 else n - 1
                for jj in range(jp_min, jp_max + 1):
                    other_contributors.add(jj)

            # Priority: reduce elements only in this peak
            only_here = [j for j in contributors if j not in other_contributors]
            targets = only_here if only_here else contributors

            if not targets:
                continue

            # Try reducing the largest contributing element
            target = max(targets, key=lambda j: best_a[j] * best_a[k - j])

            # How much to reduce
            for frac in [0.01, 0.005, 0.002, 0.02, 0.05]:
                delta = frac * best_a[target]
                a_try = best_a.copy()
                a_try[target] -= delta
                if a_try[target] < 0:
                    continue

                # Redistribute to elements NOT contributing to any top peak
                all_peak_contribs = set()
                for pp in peak_indices:
                    pp = int(pp)
                    pmin = max(0, pp - (n - 1))
                    pmax = min(pp, n - 1)
                    for jj in range(pmin, pmax + 1):
                        all_peak_contribs.add(jj)

                free_elements = [j for j in range(n) if j not in all_peak_contribs and best_a[j] > 0]
                if not free_elements:
                    free_elements = [j for j in range(n) if j != target and best_a[j] > 0]
                if not free_elements:
                    continue

                # Distribute evenly among top free elements by value
                free_elements.sort(key=lambda j: -best_a[j])
                n_redist = min(n_trials_per_peak, len(free_elements))
                per = delta / n_redist
                for j in free_elements[:n_redist]:
                    a_try[j] += per

                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break

        if not improved:
            if it > 50:
                break

    return best_a


def symmetrize(a):
    """Enforce symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    return (a + a[::-1]) / 2.0


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def structured_init(n, kind=0):
    """Generate structured initial sequences."""
    if kind == 0:
        # Hann window
        t = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
    elif kind == 1:
        # Raised cosine with different frequency
        t = np.arange(n)
        a = 0.5 * (1 + np.cos(4 * np.pi * t / (n - 1)))
        a = np.maximum(a, 0)
    elif kind == 2:
        # Two-lobe: high at edges, low in middle
        t = np.arange(n)
        a = 0.5 * (1 - np.cos(2 * np.pi * t / (n - 1)))
    elif kind == 3:
        # Flat with slight taper
        a = np.ones(n)
        taper = 10
        a[:taper] *= np.linspace(0.5, 1, taper)
        a[-taper:] *= np.linspace(1, 0.5, taper)
    elif kind == 4:
        # Block structure
        a = np.zeros(n)
        block_size = max(1, n // 7)
        a[:block_size] = 1.0
        a[n//2 - block_size//2:n//2 + block_size//2] = 0.7
        a[-block_size:] = 1.0
    else:
        a = np.ones(n) / n
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-strategy optimization combining L-BFGS-B,
    Nesterov subgradient, peak shaving, symmetry, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, budget={budget_s}s")

    # === Phase 1: L-BFGS-B on symmetrized parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: L-BFGS-B optimization (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        S = np.sum(a)
        a = a / S
        a = symmetrize(a)
        a = a / np.sum(a)

        beta_schedule = [5, 10, 20, 40, 80, 150, 250]
        for beta in beta_schedule:
            if time.monotonic() > deadline - 120:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.12, 80.0)
            if alloc < 5:
                break

            a_opt = l_bfgsb_opt(a, beta, time_budget=alloc)
            a_opt = symmetrize(a_opt)
            s = np.sum(a_opt)
            if s > 1e-10:
                a_opt = a_opt / s
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                a = a_opt.copy()
                print(f"  L-BFGS-B beta={beta}: {best_score:.8f} ***")
            else:
                print(f"  L-BFGS-B beta={beta}: {obj:.8f} (no imp)")
                # Keep the better of current and previous
                a = a_opt if compute_obj(a_opt) < compute_obj(a) else a

    # === Phase 2: Nesterov subgradient ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Nesterov subgradient (n={len(best_seq) if best_seq is not None else '?'})")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            alloc = min(remaining * 0.15, 120.0)
            a_nest = nesterov_subgrad(best_seq, time_budget=alloc, lr_init=0.005)
            a_nest = symmetrize(a_nest)
            s = np.sum(a_nest)
            if s > 1e-10:
                a_nest = a_nest / s
            obj = compute_obj(a_nest)
            if obj < best_score:
                best_score = obj
                best_seq = a_nest.copy()
                print(f"  Nesterov: {best_score:.8f} ***")
            else:
                print(f"  Nesterov: {obj:.8f} (no imp)")

    # === Phase 3: Peak shaving ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Peak shaving (n={len(best_seq) if best_seq is not None else '?'})")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc = min(remaining * 0.12, 80.0)
            a_shave = peak_shaving(best_seq, time_budget=alloc)
            obj = compute_obj(a_shave)
            if obj < best_score:
                best_score = obj
                best_seq = a_shave.copy()
                print(f"  Peak shaving: {best_score:.8f} ***")
            else:
                print(f"  Peak shaving: {obj:.8f} (no imp)")

    # === Phase 4: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        # Try a range of n values
        n_options = []
        for dn in [-100, -50, 0, 50, 100, 200, 300, 500, 800]:
            n_try = n_best + dn
            if 300 <= n_try <= 2000:
                n_options.append(n_try)
        # Also try some absolute values
        for n_abs in [400, 500, 600, 800, 1000, 1200, 1500]:
            if n_abs not in n_options:
                n_options.append(n_abs)
        n_options = sorted(set(n_options))

        for n_try in n_options:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if n_try == len(best_seq):
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
            S = np.sum(a_try)
            if S > 1e-10:
                a_try = a_try / S
            a_try = symmetrize(a_try)
            a_try = a_try / np.sum(a_try)

            # Quick L-BFGS-B
            alloc = min(remaining * 0.08, 40.0)
            a_opt = l_bfgsb_opt(a_try, beta=20, time_budget=alloc)
            a_opt = symmetrize(a_opt)
            s = np.sum(a_opt)
            if s > 1e-10:
                a_opt = a_opt / s
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 5: Structured initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Structured initializations")
    print(f"{'='*60}")

    n_for_struct = len(best_seq) if best_seq is not None else 700
    for kind in range(5):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        a_struct = structured_init(n_for_struct, kind=kind)
        # Blend with best
        for alpha in [0.0, 0.3, 0.5]:
            if time.monotonic() > deadline - 60:
                break
            if best_seq is not None:
                a_blend = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * a_struct
            else:
                a_blend = a_struct.copy()
            a_blend = symmetrize(a_blend)
            s = np.sum(a_blend)
            if s > 1e-10:
                a_blend = a_blend / s

            alloc = min(remaining * 0.06, 30.0)
            a_opt = l_bfgsb_opt(a_blend, beta=15, time_budget=alloc)
            a_opt = symmetrize(a_opt)
            s = np.sum(a_opt)
            if s > 1e-10:
                a_opt = a_opt / s
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Struct kind={kind} alpha={alpha}: {best_score:.8f} ***")
            else:
                print(f"  Struct kind={kind} alpha={alpha}: {obj:.8f}")

    # === Phase 6: Fine-tuning with high-beta L-BFGS-B ===
    print(f"\n{'='*60}")
    print(f"Phase 6: High-beta fine-tuning")
    print(f"{'='*60}")

    if best_seq is not None:
        for beta in [200, 400, 600, 800, 1000]:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.08, 40.0)
            a_opt = l_bfgsb_opt(best_seq, beta=beta, time_budget=alloc)
            a_opt = symmetrize(a_opt)
            s = np.sum(a_opt)
            if s > 1e-10:
                a_opt = a_opt / s
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Fine-tune beta={beta}: {best_score:.8f} ***")
            else:
                print(f"  Fine-tune beta={beta}: {obj:.8f}")

    # === Phase 7: Final peak shaving + Nesterov polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc1 = min(remaining * 0.3, 40.0)
            a_shave = peak_shaving(best_seq, time_budget=alloc1)
            obj = compute_obj(a_shave)
            if obj < best_score:
                best_score = obj
                best_seq = a_shave.copy()
                print(f"  Final shaving: {best_score:.8f} ***")

            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc2 = min(remaining * 0.3, 30.0)
                a_nest = nesterov_subgrad(best_seq, time_budget=alloc2, lr_init=0.003, momentum=0.8)
                a_nest = symmetrize(a_nest)
                s = np.sum(a_nest)
                if s > 1e-10:
                    a_nest = a_nest / s
                obj = compute_obj(a_nest)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_nest.copy()
                    print(f"  Final Nesterov: {best_score:.8f} ***")

    # === Phase 8: Random restarts with perturbation ===
    print(f"\n{'='*60}")
    print(f"Phase 8: Random restarts")
    print(f"{'='*60}")

    rng = np.random.default_rng(seed)
    if best_seq is not None:
        n_best = len(best_seq)
        for restart in range(6):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # Different perturbations
            scale = 0.02 + 0.01 * restart
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_pert = symmetrize(a_pert)
            a_pert = np.maximum(a_pert, 1e-15)
            s = np.sum(a_pert)
            if s < 1e-10:
                continue
            a_pert = a_pert / s

            alloc = min(remaining * 0.08, 25.0)
            a_opt = l_bfgsb_opt(a_pert, beta=30, time_budget=alloc)
            a_opt = symmetrize(a_opt)
            s = np.sum(a_opt)
            if s > 1e-10:
                a_opt = a_opt / s
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} ***")
            else:
                print(f"  Restart {restart}: {obj:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
