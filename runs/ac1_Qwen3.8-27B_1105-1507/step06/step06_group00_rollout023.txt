2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-based gradient computation, adaptive
temperature cooling, and multi-phase search with diversification.

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def compute_soft_gradient(a, beta):
    """
    Compute gradient of soft-max objective using FFT.
    Returns (grad, conv, weights) where grad[i] = sum_k w[k]*a[k-i].
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    
    if beta < 1e6:
        shifted = beta * (conv - max_conv)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum > 0:
            weights /= w_sum
        else:
            # Fallback to hard max
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
    else:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    
    # Compute grad[i] = sum_j weights[i+j] * a[j] via cross-correlation
    # Using FFT: grad = ifft(FFT(W) * conj(FFT(A)))
    size = 1
    while size < 3 * n:
        size <<= 1
    
    A = np.zeros(size)
    A[:n] = a
    W = np.zeros(size)
    W[:2*n-1] = weights
    
    FFT_A = rfft(A, size)
    FFT_W = rfft(W, size)
    grad = irfft(FFT_W * np.conj(FFT_A), size)[:n]
    
    return grad, conv, weights

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Hybrid optimization: FFT-based soft-max mirror descent + LP alternating.
Uses adaptive temperature cooling, multi-phase search, and diversification
to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def compute_soft_gradient(a, beta):
    """
    Compute gradient of soft-max objective using FFT cross-correlation.
    grad[i] = sum_k w[k] * a[k-i] * [0<=k-i<n]
    Returns (grad, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    if beta < 1e10:
        shifted = beta * (conv - max_conv)
        # Clamp to avoid overflow
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum > 1e-300:
            weights /= w_sum
        else:
            weights = np.zeros_like(conv)
            weights[int(np.argmax(conv))] = 1.0
    else:
        weights = np.zeros_like(conv)
        weights[int(np.argmax(conv))] = 1.0

    # Compute grad[i] = sum_j weights[i+j] * a[j] via FFT cross-correlation
    # grad = ifft(FFT(W) * conj(FFT(A)))[:n]
    size = 1
    while size < 3 * n:
        size <<= 1

    A = np.zeros(size)
    A[:n] = a
    W = np.zeros(size)
    W[:2 * n - 1] = weights

    FFT_A = rfft(A, size)
    FFT_W = rfft(W, size)
    grad = irfft(FFT_W * np.conj(FFT_A), size)[:n]

    return grad, conv


def soft_max_mirror_descent(a, time_budget, beta_start=5.0, beta_end=500.0,
                            eta_start=0.3, eta_end=0.005, verbose=False):
    """
    Soft-max mirror descent on the simplex with adaptive cooling.
    Uses multiplicative updates: a[i] <- a[i] * exp(-eta*(grad[i]-mean(grad)))
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    total_iters = 0
    best_eta = eta_start
    no_improve_count = 0

    while time.monotonic() - start < time_budget:
        progress = min((time.monotonic() - start) / time_budget, 1.0)
        # Exponential cooling for beta
        beta = beta_start * (beta_end / beta_start) ** progress
        # Linear decrease for eta with recovery
        eta_base = eta_start * (1 - 0.9 * progress) + eta_end * 0.9 * progress
        eta = best_eta * (eta_base / eta_start)
        eta = max(eta, eta_end * 0.5)

        grad, conv = compute_soft_gradient(a, beta)

        # Center the gradient (project onto tangent space of simplex)
        grad_centered = grad - np.mean(grad)

        # Multiplicative update with clipping
        log_update = -eta * grad_centered
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        total_iters += 1

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve_count = 0
            if verbose and total_iters % 2000 == 0:
                remaining = time_budget - (time.monotonic() - start)
                print(f"      SDM it{total_iters}: obj={best_obj:.8f}, "
                      f"beta={beta:.1f}, eta={eta:.5f}, rem={remaining:.0f}s")
        else:
            no_improve_count += 1
            # Try half step
            if no_improve_count <= 3:
                log_half = -0.5 * eta * grad_centered
                log_half = np.clip(log_half, -10, 10)
                a_half = a * np.exp(log_half)
                a_half = np.maximum(a_half, 1e-15)
                sh = float(np.sum(a_half))
                if sh > 1e-10:
                    a_half = a_half / sh
                    obj_half = compute_obj(a_half)
                    if obj_half < best_obj - 1e-14:
                        best_obj = obj_half
                        best_a = a_half.copy()
                        a = a_half
                        no_improve_count = 0
                        if verbose and total_iters % 2000 == 0:
                            print(f"      SDM it{total_iters} (half): obj={best_obj:.8f}")
                        total_iters += 1
                        continue

            if no_improve_count > 10:
                # Diversify: add small random perturbation
                rng = np.random.default_rng(total_iters * 31 + 7)
                scale = 0.02 + 0.01 * min(no_improve_count // 20, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                no_improve_count = 0

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget, verbose=False):
    """LP alternating optimization with line search and diversification."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t = 0.5
        best_t_obj = np.inf
        best_a_t = a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_t = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_t.copy()
            a = a_try.copy()
            stall = 0
            if verbose and it % 5 == 0:
                rem = time_budget - (time.monotonic() - start)
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}, rem={rem:.0f}s")
        else:

                rng = np.random.default_rng(it * 104729 + n * 31)

                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                stall = 0

    return best_a


def chase_peak_refine(a, time_budget, verbose=False):
    """Chase-the-peak: iterative subgradient step targeting the maximum convolution bin."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: grad[i] = 2*a[k_star - i] (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        grad_centered = grad - np.mean(grad)
        a_new = a - step_size * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                step_size = min(step_size * 1.05, 0.15)
            else:
                step_size *= 0.7
        else:
            step_size *= 0.5

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0).astype(np.float64)

        # Final soft-max with very high beta
        if remaining > 20:
            alloc = min(40.0, remaining * 0.4)
            a_final2 = soft_max_mirror_descent(best_seq, time_budget=alloc,
                                                beta_start=100, beta_end=5000,
                                                eta_start=0.03, eta_end=0.001)
            obj_final2 = compute_obj(a_final2)
            if obj_final2 < best_score:
                best_score = obj_final2
                best_seq = a_final2.copy()
                print(f"  Final SDM: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        # Final chase peak
        if remaining > 15:
            a_final3 = chase_peak_refine(best_seq, time_budget=min(20.0, remaining * 0.5))
            obj_final3 = compute_obj(a_final3)
            if obj_final3 < best_score:
                best_score = obj_final3
                best_seq = a_final3.copy()
                print(f"  Final Chase: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
