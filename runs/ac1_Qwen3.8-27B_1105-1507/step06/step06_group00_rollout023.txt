"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth subgradient descent (log-sum-exp) with FFT for fast iterations,
LP polishing with CVXPY, multi-n exploration, and adaptive strategies.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def smooth_grad_step(a, tau, eta):
    """
    One step of smooth subgradient descent (log-sum-exp smoothing).
    Uses multiplicative update to maintain non-negativity.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Softmax weights (numerically stable)
    c_max = np.max(conv)
    w = np.exp((conv - c_max) / tau)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        return a
    w = w / w_sum
    
    # Compute gradient: g(i) = sum_j w(i+j) * a(j)
    # = conv(w_rev, a)(2n-2-i) for i = 0,...,n-1
    w_rev = w[::-1]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fw = rfft(w_rev, size)
    fa = rfft(a, size)
    corr = irfft(fw * fa, size)
    
    # g[i] = corr[2n-2-i] for i=0..n-1
    indices = np.arange(2 * n - 2, n - 2, -1)
    g = corr[indices]
    
    # Ensure g is non-negative (it should be since w and a are non-negative)
    g = np.maximum(g, 0)
    
    # Multiplicative update
    a_new = a * np.exp(-eta * g)
    a_new = np.maximum(a_new, 1e-300)  # avoid zeros
    s_new = float(np.sum(a_new))
    if s_new < 1e-15:
        return a
    a_new = a_new / s_new
    return a_new


def l2_grad_step(a, eta):
    """
    One step of gradient descent on the L2 norm of the convolution.
    Minimize sum_k c(k)^2.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Gradient of sum c(k)^2 w.r.t. a(i):
    # 2 * sum_k c(k) * a(k-i) for valid k-i
    # = 2 * conv(c_rev, a)(2n-2-i)
    c_rev = conv[::-1]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fc = rfft(c_rev, size)
    fa = rfft(a, size)
    corr = irfft(fc * fa, size)
    
    indices = np.arange(2 * n - 2, n - 2, -1)
    g = 2.0 * corr[indices]
    
    # Projected gradient step
    a_new = a - eta * g
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-15:
        return a
    a_new = a_new / s_new
    return a_new


def smooth_subgradient_optimize(a, time_budget=100.0, verbose=False):
    """
    Optimize using smooth subgradient descent.
    Adaptive tau and eta.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Start with larger tau (smoother) and decrease
    tau_start = 0.5 * np.max(compute_conv(a))
    tau_current = tau_start
    eta = 0.1
    stall_count = 0
    
    iteration = 0
    while time.monotonic() - start < time_budget:
        # Adaptive tau: decrease over time
        progress = (time.monotonic() - start) / time_budget
        tau_current = tau_start * (1.0 - 0.9 * progress) + 1e-6
        
        # Try the step
        a_new = smooth_grad_step(a, tau_current, eta)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            # Increase eta slightly if making progress
            eta = min(eta * 1.01, 1.0)
        else:
            stall_count += 1
            # Decrease eta
            eta *= 0.9
            if stall_count > 50:

                a = best_a * np.exp(rng.normal(0, 0.05, n))

                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                eta = 0.1

        iteration += 1
        if verbose and iteration % 100 == 0:

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g_normalized or None.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_val = float(np.sum(g_val))
            if s_val > 1e-10:
                return g_val / s_val
    except Exception:
        pass

    return None

def chase_peak(a, time_budget=30.0):
    """Chase-the-peak refinement."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.01

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        grad_centered = grad - np.mean(grad)
        a_new = a - step_size * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                step_size *= 0.8
                if step_size < 1e-6:
                    break
        else:
            step_size *= 0.5

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 0:
        a_new = a_new / s
    return a_new

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)

    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)

def compute_obj(a):
    """Compute objective."""
