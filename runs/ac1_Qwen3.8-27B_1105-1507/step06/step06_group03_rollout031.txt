    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def smooth_gradient_fft(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*C[k])) w.r.t. a.
    Uses FFT for O(n log n) computation.
    Returns (grad, C, w) where w is the softmax weights.
    """
    n = len(a)
    C = fast_conv(a)
    C_shifted = C - C.max()
    exp_C = np.exp(tau * C_shifted)
    w = exp_C / exp_C.sum()

    # grad[j] = 2 * sum_m a[m] * w[j+m] = 2 * (a * w_rev)[2n-2-j]
    w_rev = w[::-1]
    # Convolution of a (len n) and w_rev (len 2n-1) has length 3n-2
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w_rev, size)
    conv_result = np.fft.irfft(A * W, size)[:3 * n - 2]

    # grad[j] = 2 * conv_result[2n-2-j] for j=0,...,n-1
    indices = np.arange(2 * n - 2, n - 2, -1)  # [2n-2, 2n-3, ..., n-1]
    grad = 2.0 * conv_result[indices]

    return grad, C, w


def smooth_gradient_descent(a, deadline, tau_init=50.0, tau_min=5.0,
                            lr_init=0.1, verbose=False):
    """
    Projected smooth gradient descent with adaptive temperature.
    Works on the probability simplex.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    tau = tau_init
    lr = lr_init
    no_improve = 0
    it = 0
    stall_counter = 0

    while time.monotonic() < deadline:
        it += 1

        grad, C, w = smooth_gradient_fft(a, tau)

        # Project gradient to simplex tangent
        grad = grad - np.mean(grad)

        # Backtracking line search
        alpha = lr
        obj_curr = compute_J(a)
        found = False
        for _ in range(10):
            a_try = a - alpha * grad
            a_try = project_simplex(a_try)
            obj_try = compute_J(a_try)
            if obj_try < obj_curr - 1e-10:
                a = a_try
                obj_curr = obj_try
                found = True
                break
            alpha *= 0.5

        if not found:
            # Try smaller steps
            for frac in [0.01, 0.005, 0.001]:
                a_try = a - frac * grad
                a_try = project_simplex(a_try)
                obj_try = compute_J(a_try)
                if obj_try < obj_curr - 1e-10:
                    a = a_try
                    obj_curr = obj_try
                    found = True
                    break

        if obj_curr < best_obj - 1e-12:
            best_obj = obj_curr
            best_a = a.copy()
            no_improve = 0
            stall_counter = 0
        else:
            no_improve += 1
            stall_counter += 1

        # Adaptive tau: decrease when we're making progress, increase when stuck
        if stall_counter > 50:
            # Stuck: try increasing tau to smooth more and escape
            tau = min(tau * 1.3, 200.0)
            lr = lr * 1.5
            stall_counter = 0
        elif no_improve > 100:
            # Very stuck: restart with perturbation
            rng = np.random.default_rng(it)
            a = project_simplex(best_a + rng.normal(0, 0.01, n))
            tau = tau_init
            lr = lr_init
            no_improve = 0
            stall_counter = 0
        else:
            # Making progress: slowly decrease tau
            tau = max(tau_min, tau * 0.999)
            lr = min(0.3, lr * 1.01)

        if verbose and it % 200 == 0:
            print(f"      SGD it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def coord_descent_fast(a, max_sweeps, deadline, seed=42):
    """
    Fast coordinate descent with incremental convolution updates.
    Tries multiple candidate values per coordinate.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 200

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n) if sweep % 3 == 0 else range(n)
        improved = False

        for i in order:
            old = a[i]
            if old < 1e-10:
                candidates = [0.0, 1e-4, 1e-3, 1e-2, 0.05]
            else:
                candidates = np.array([0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0]) * old
                candidates = np.unique(candidates)

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i

            best_val = old
            best_local = 2.0 * n * conv.max()

            for x in candidates:
                x = float(max(0.0, x))
                if x == old:
                    continue
                delta = x - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                conv[kq] += delta * delta

                new_obj = 2.0 * n * conv.max()

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-14:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def coord_descent_fine(a, max_sweeps, deadline, seed=0):
    """
    Fine coordinate descent: tries to find the optimal value for each coordinate
    using ternary search around the current value.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            old = a[i]
            if old < 1e-12:
                continue

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i

            # Ternary search in [0, 3*old]
            lo_x = 0.0
            hi_x = 3.0 * old
            best_val = old
            best_local = 2.0 * n * conv.max()

            for _ in range(12):
                m1 = lo_x + (hi_x - lo_x) / 3
                m2 = hi_x - (hi_x - lo_x) / 3

                # Evaluate at m1
                delta1 = m1 - old
                conv[lo:hi + 1] += 2.0 * delta1 * a_slice
                conv[kq] += delta1 * delta1
                obj1 = 2.0 * n * conv.max()
                conv[lo:hi + 1] -= 2.0 * delta1 * a_slice
                conv[kq] -= delta1 * delta1

                # Evaluate at m2
                delta2 = m2 - old
                conv[lo:hi + 1] += 2.0 * delta2 * a_slice
                conv[kq] += delta2 * delta2
                obj2 = 2.0 * n * conv.max()
                conv[lo:hi + 1] -= 2.0 * delta2 * a_slice
                conv[kq] -= delta2 * delta2

                if obj1 < obj2:
                    hi_x = m2
                else:
                    lo_x = m1

                if m1 < best_local / (2.0 * n) + 0.01 and m2 < best_local / (2.0 * n) + 0.01:
                    break

            # Final evaluation at the best point
            best_x = (lo_x + hi_x) / 2.0
            delta = best_x - old
            conv[lo:hi + 1] += 2.0 * delta * a_slice
            conv[kq] += delta * delta
            final_obj = 2.0 * n * conv.max()

            if final_obj < best_local - 1e-10:
                best_local = final_obj
                best_val = best_x
                a[i] = best_x
                s = s - old + best_x
                improved = True
            else:
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                conv[kq] -= delta * delta

        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def sca_lp_step(a, max_iters, deadline):
    """
    SCA-LP: At each step, linearize conv constraint at current point.
    Optimized for speed with pre-allocated arrays.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    # Pre-allocate row/col/val arrays
    max_nonzero = 2 * n * n  # upper bound
    rows = np.zeros(max_nonzero, dtype=np.int32)
    cols = np.zeros(max_nonzero, dtype=np.int32)
    vals = np.zeros(max_nonzero, dtype=np.float64)
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Build sparse matrix
        nnz = 0
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows[nnz] = k
                cols[nnz] = j
                vals[nnz] = 2.0 * a[k - j]
                nnz += 1
            rows[nnz] = k
            cols[nnz] = n
            vals[nnz] = -1.0
            nnz += 1
            b_ub[k] = c[k]

        A = coo_matrix((vals[:nnz], (rows[:nnz], cols[:nnz])),
                       shape=(2 * n - 1, n + 1)).tocsr()

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        # Equality: sum(g) = 1
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=np.int32),
                                         np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(None, None)]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    J_g = compute_J(g)
                    if J_g < best_obj - 1e-12:
                        a = g.copy()
                        best_obj = J_g
                        best_a = g.copy()
                        no_improve = 0
                    else:
                        found = False
                        for frac in [0.5, 0.3, 0.2, 0.1, 0.05]:
                            a_try = (1 - frac) * a + frac * g
                            a_try = np.maximum(a_try, 0)
                            if a_try.sum() < 1e-10:
                                continue
                            a_try /= a_try.sum()
                            J_try = compute_J(a_try)
                            if J_try < best_obj - 1e-12:
                                a = a_try
                                best_obj = J_try
                                best_a = a_try.copy()
                                no_improve = 0
                                found = True
                                break
                        if not found:
                            no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_obj


def peak_shaving(a, max_iters, deadline):
    """
    Peak shaving: iteratively reduce the maximum convolution value
    by transferring mass from high-contribution positions to low ones.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        C = fast_conv(a)
        k_star = np.argmax(C)
        M = C[k_star]

        # Gradient of C[k_star] w.r.t. a[j] = 2*a[k_star - j] if 0 <= k_star-j < n
        # We want to reduce C[k_star], so move mass from positions with high
        # a[k_star-j] to positions with low a[k_star-j]
        if k_star < n:
            contrib = a[k_star::-1]  # a[k_star], a[k_star-1], ..., a[0]
            pad = np.zeros(n)
            pad[:len(contrib)] = contrib
            contributes = pad
        else:
            # k_star >= n: a[k_star - j] is valid for j >= k_star - n + 1
            contributes = np.zeros(n)
            valid_start = k_star - n + 1
            valid_end = n
            contributes[valid_start:valid_end] = a[n - 1: (n - 1) - (valid_end - valid_start) - 1:-1]

        # Actually let me just compute it directly
        contributes = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                contributes[j] = a[idx]

        # Sort positions by contribution: high contrib positions should lose mass
        # Low contrib positions should gain mass
        sorted_idx = np.argsort(contributes)
        n_transfer = max(1, n // 10)

        # Transfer a small amount from top-contrib to bottom-contrib
        step = 0.01 / n
        for t in range(n_transfer):
            src = sorted_idx[-(t + 1)]  # highest contribution
            dst = sorted_idx[t]  # lowest contribution
            amount = min(a[src], step)
            if amount < 1e-10:
                continue
            a[src] -= amount
            a[dst] += amount

        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            break
        a /= a.sum()

        obj = compute_J(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            a = best_a.copy()
            no_improve += 1

        if no_improve > 50:
            break

    return best_a, best_obj


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences on the simplex."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns (good for flat autocorrelation)
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t + 0.5)))

    # Two-peak patterns (to reduce center peak)
    for c1 in [0.25, 0.33]:
        for c2 in [0.75, 0.67]:
            for width in [0.15, 0.25]:
                sig = np.exp(-((t - c1) / width) ** 2) + np.exp(-((t - c2) / width) ** 2)
                inits.append(project_simplex(sig + 0.05))

    # Edge-concentrated (reduces center convolution)
    for edge_width in [0.1, 0.2, 0.3]:
        sig = np.exp(-((t - 0) / edge_width) ** 2) + np.exp(-((t - 1) / edge_width) ** 2)
        inits.append(project_simplex(sig + 0.01))

    # Random smooth
    ksize = max(5, n // 12)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    return inits


def solve_single(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return a
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Phase 1a: Quick coordinate descent (20% time)
    t1 = time.monotonic() + 0.20 * remaining
    a = coord_descent_fast(a, max_sweeps=5000, deadline=t1, seed=42)
    a = try_update(a)
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Phase 1b: Smooth gradient descent (25% time)
    t2 = time.monotonic() + 0.25 * remaining
    a_sgd, obj_sgd = smooth_gradient_descent(a, deadline=t2, tau_init=40.0,
                                             tau_min=8.0, lr_init=0.05)
    if obj_sgd < best_obj:
        best_a = a_sgd.copy()
        best_obj = obj_sgd
    a = best_a.copy()
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Phase 1c: SCA-LP (30% time)
    t3 = time.monotonic() + 0.30 * remaining
    a_sca, obj_sca = sca_lp_step(a, max_iters=100, deadline=t3)
    if obj_sca < best_obj:
        best_a = a_sca.copy()
        best_obj = obj_sca
    a = best_a.copy()
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Phase 1d: Fine coordinate descent (remaining time)
    t4 = deadline - 2
    a = coord_descent_fine(a, max_sweeps=5000, deadline=t4, seed=123)
    a = try_update(a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Combines FFT-based smooth gradient descent, coordinate descent, SCA-LP,
    and peak shaving with adaptive temperature and multiple restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # ============ PHASE 1: Multi-start at parent length (0-400s) ============
    phase1_deadline = t_start + 400.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-init at n={n_parent} (budget {phase1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    n_init = min(len(inits), 10)
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, n_init) - 3

    for idx in range(n_init):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 20:
            break
        alloc = min(per_init_time, remaining_check - 3)
        if alloc < 10:
            break

        a0 = inits[idx].copy()
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt, obj_opt = solve_single(a0, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.8f}", flush=True)

    # ============ PHASE 2: Multi-length exploration (400-700s) ============
    phase2_deadline = t_start + 700.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidate_ns = []
    for delta in [-64, -32, 32, 64, 96, -128, 128, 192]:
        nn = n_parent + delta
        if 128 <= nn <= 3000:
            candidate_ns.append(nn)
    # Also try some standard lengths
    for nn in [256, 384, 512, 640, 768, 896, 1024, 1280, 1536]:
        if nn != n_parent and 128 <= nn <= 3000:
            candidate_ns.append(nn)
    candidate_ns = list(set(candidate_ns))
    candidate_ns.sort(key=lambda x: abs(x - n_parent))

    for new_n in candidate_ns[:4]:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 40:
            break
        alloc = min(70.0, remaining_check - 10)
        if alloc < 20:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = solve_single(a_ext, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # ============ PHASE 3: Intensive optimization on best (700-1300s) ============
    phase3_deadline = t_start + 1300.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive local optimization (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        a_curr = best_seq.copy()
        a_curr = np.maximum(a_curr, 0)
        a_curr = a_curr / a_curr.sum()
        remaining = phase3_deadline - time.monotonic()

        if remaining > 50:
            # Pass 1: Long coordinate descent
            t_p1 = time.monotonic() + 0.20 * remaining
            print(f"  CD pass (budget {t_p1 - time.monotonic():.0f}s)...", flush=True)
            a_curr = coord_descent_fast(a_curr, max_sweeps=20000, deadline=t_p1, seed=777)
            update_best(a_curr)
            a_curr = best_seq.copy()
            a_curr = np.maximum(a_curr, 0)
            a_curr = a_curr / a_curr.sum()

            remaining = phase3_deadline - time.monotonic()

        if remaining > 50:
            # Pass 2: SCA-LP (most powerful for fine-tuning)
            t_p2 = time.monotonic() + 0.35 * remaining
            print(f"  SCA-LP pass (budget {t_p2 - time.monotonic():.0f}s)...", flush=True)
            a_sca, obj_sca = sca_lp_step(a_curr, max_iters=200, deadline=t_p2)
            update_best(a_sca)
            a_curr = best_seq.copy()
            a_curr = np.maximum(a_curr, 0)
            a_curr = a_curr / a_curr.sum()

            remaining = phase3_deadline - time.monotonic()

        if remaining > 50:
            # Pass 3: Smooth gradient with low tau (sharp optimization)
            t_p3 = time.monotonic() + 0.20 * remaining
            print(f"  SGD pass (budget {t_p3 - time.monotonic():.0f}s)...", flush=True)
            a_sgd, obj_sgd = smooth_gradient_descent(a_curr, deadline=t_p3,
                                                     tau_init=15.0, tau_min=5.0,
                                                     lr_init=0.03)
            update_best(a_sgd)
            a_curr = best_seq.copy()
            a_curr = np.maximum(a_curr, 0)
            a_curr = a_curr / a_curr.sum()

            remaining = phase3_deadline - time.monotonic()

        if remaining > 30:
            # Pass 4: Fine coordinate descent
            t_p4 = deadline - 5
            print(f"  Fine CD pass (budget {t_p4 - time.monotonic():.0f}s)...", flush=True)
            a_fine = coord_descent_fine(a_curr, max_sweeps=20000, deadline=t_p4, seed=999)
            update_best(a_fine)

            # One more SCA-LP on the result
            if time.monotonic() < deadline - 30:
                a_sca2, obj_sca2 = sca_lp_step(a_fine, max_iters=50, deadline=deadline - 5)
                update_best(a_sca2)

    # ============ PHASE 4: Final restarts (1300-1485s) ============
    remaining = deadline - time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Final restarts (budget {remaining:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and remaining > 30:
        n_best = len(best_seq)
        # Try a few random perturbations of the best
        n_restarts = int(remaining // 40)
        for r in range(min(n_restarts, 4)):
            if time.monotonic() > deadline - 10:
                break
            a_pert = best_seq + rng.normal(0, 0.01 * best_seq.mean(), n_best)
            a_pert = project_simplex(a_pert)
            alloc = min(35.0, deadline - time.monotonic() - 5)
            if alloc < 10:
                break
            a_opt, obj_opt = solve_single(a_pert, time.monotonic() + alloc)
            update_best(a_opt)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    # Ensure sum is reasonable for evaluate_sequence
    if best_seq.sum() < 0.01:
        best_seq = best_seq / best_seq.sum() * 1.0

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
