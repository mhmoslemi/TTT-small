def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2. Assumes a >= 0."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2 * len(a) * c.max() / (s * s))


def project_nonneg_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent(a, deadline, seed=42, verbose=False, max_sweeps=200000):
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a).copy()
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 80

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            old = a[i]
            if old < 1e-14:
                candidates = np.array([0.0, 0.5 / n, 1.0 / n, 3.0 / n, 5.0 / n])
            else:
                candidates = np.array([0.0, old * 0.1, old * 0.3, old * 0.5, old * 0.7,
                                       old * 1.0, old * 1.3, old * 1.6, old * 2.0, old * 3.0])

            k_lo = max(0, i - n + 1)
            k_hi = min(2 * n - 2, i + n - 1)
            conv_aff = conv[k_lo:k_hi + 1]

            ks = np.arange(k_lo, k_hi + 1)
            ki = ks - i
            valid = (ki >= 0) & (ki < n)
            alpha = np.zeros(k_hi - k_lo + 1)
            alpha[valid] = 2.0 * a[ki[valid]]
            beta = np.zeros(k_hi - k_lo + 1)
            if 2 * i >= k_lo and 2 * i <= k_hi:
                beta[2 * i - k_lo] = 1.0

            max_unaff = 0.0
            if k_lo > 0:
                max_unaff = max(max_unaff, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                max_unaff = max(max_unaff, float(conv[k_hi + 1:].max()))

            best_x = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                new_s = s + delta
                if new_s < 1e-10:
                    continue
                new_aff = conv_aff + delta * alpha + delta * delta * beta
                new_max = max(float(new_aff.max()), max_unaff)
                new_obj = 2.0 * n * new_max / (new_s * new_s)
                if new_obj < best_local:
                    best_local = new_obj
                    best_x = x

            if best_x != old and abs(best_x - old) > 1e-14:
                delta = best_x - old
                conv[k_lo:k_hi + 1] = conv_aff + delta * alpha + delta * delta * beta
                a[i] = best_x
                s += delta
                improved = True

        if s > 1e-10:
            current_obj = 2.0 * n * conv.max() / (s * s)
        else:
            current_obj = float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 200 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f} t={time.monotonic()-deadline:.0f}s", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def sca_lp(a, deadline, verbose=False, max_iters=200):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), list(range(n)))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.05]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 5 == 0:
            print(f"    SCA it {it}: obj={best_obj:.8f}", flush=True)

        if no_improve > 30:
            break

    return best_a, best_obj


def gradient_descent_smooth(a, deadline, verbose=False, tau_init=20.0, tau_max=150.0,
                            lr_init=0.03, max_iters=50000):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_init + (tau_max - tau_init) * frac

        c = fast_conv(a)
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w /= w.sum()

        a_rev = a[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        conv1 = np.fft.irfft(W_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx = n - 1 + i
            if 0 <= idx < len(conv1):
                grad[i] = conv1[idx] * 2.0

        grad = grad - np.mean(grad)

        t_count += 1
        lr = lr_init / (1.0 + 0.05 * t_count ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it + t_count)
                a = best_a + rng.normal(0, 0.03, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

        if verbose and it % 500 == 0:
            print(f"    Grad it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2, 0.4]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))

    inits.append(np.ones(n) / n)

    # Asymmetric patterns (important: optimum must be asymmetric)
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    # Exponential
    for beta in [1, 2, 3, 5, 8]:
        inits.append(project_simplex(np.exp(beta * t) + 0.01))
        inits.append(project_simplex(np.exp(-beta * t) + 0.01))

    # Gaussian bumps (asymmetric placement)
    for center in [0.1, 0.2, 0.3, 0.7, 0.8, 0.9]:
        for width in [0.05, 0.1, 0.15, 0.25]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.01))

    # Sine/cosine
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(project_simplex(1.0 + amp * np.sin(np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.cos(np.pi * freq * t)))

    # Triangle / tent
    for peak in [0.2, 0.3, 0.5, 0.7, 0.8]:
        inits.append(project_simplex(1.0 - np.abs(t - peak) / max(peak, 1 - peak) + 0.1))

    # Random smooth
    for _ in range(6):
        ksize = max(3, n // 8)
        kernel = np.ones(ksize) / ksize
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    # Power-asymmetric: a[i] ∝ (1 + c * sin(pi*i/n))^p
    for c in [0.3, 0.6, 0.9]:
        for p in [1, 2, 3]:
            inits.append(project_simplex((1 + c * np.sin(np.pi * t)) ** p + 0.01))

    return inits


def try_update(best_state, a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    if a.size == 0 or a.sum() < 0.01:
        return
    s = a.sum()
    a_norm = a / s
    obj = compute_J(a_norm)
    if obj < best_state['obj'] - 1e-12:
        best_state['obj'] = obj
        best_state['seq'] = a_norm.copy()
        return True
    return False


def propose_candidate(seed=42, budget_s=1500, height_sequence_1=None):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0

    best_state = {'obj': float('inf'), 'seq': None}

    def global_update(a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = a.sum()
        a_norm = a / s
        obj = compute_J(a_norm)
        if obj < best_state['obj']:
            best_state['obj'] = obj
            best_state['seq'] = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={obj:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
        return obj

    # Load parent
    parent = None
    if height_sequence_1 is not None:
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
                parent = np.maximum(p, 0.0)
                obj_p = global_update(parent)
                print(f"Parent loaded: n={parent.size}, J={obj_p:.6f}", flush=True)
        except Exception:
            pass

    if parent is None:
        parent = np.ones(512) / 512.0
        global_update(parent)

    n_parent = len(parent)

    # === PHASE 1: CD on parent + diverse inits at parent length ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: CD optimization at n={n_parent}", flush=True)
    print(f"{'='*60}", flush=True)
    phase1_end = t_start + min(300.0, 0.25 * budget_s)

    inits = generate_inits(n_parent, rng, parent)
    # Sort by initial J to try good ones first
    init_objs = [(compute_J(i), idx) for idx, i in enumerate(inits)]
    init_objs.sort()

    n_tries = min(len(inits), 15)
    per_try = (phase1_end - time.monotonic()) / max(1, n_tries)

    for trial in range(n_tries):
        remaining = phase1_end - time.monotonic()
        if remaining < 15:
            break
        idx = init_objs[trial % len(init_objs)][1]
        a0 = inits[idx].copy()
        alloc = min(per_try, remaining - 3)
        if alloc < 8:
            break

        a_opt = coord_descent(a0, time.monotonic() + alloc, seed=trial + 100)
        obj_new = global_update(a_opt)
        print(f"  Trial {trial} (init {idx}): J={obj_new:.8f}", flush=True)

    # === PHASE 2: SCA-LP on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: SCA-LP refinement", flush=True)
    print(f"{'='*60}", flush=True)
    phase2_end = t_start + min(550.0, 0.45 * budget_s)

    if best_state['seq'] is not None:
        remaining = phase2_end - time.monotonic()
        if remaining > 30:
            a_sca, obj_sca = sca_lp(best_state['seq'], time.monotonic() + remaining * 0.7)
            global_update(a_sca)
            print(f"  SCA-LP: J={obj_sca:.8f}", flush=True)

            remaining = phase2_end - time.monotonic()
            if remaining > 20:
                a_cd = coord_descent(best_state['seq'], time.monotonic() + remaining, seed=999)
                global_update(a_cd)

    # === PHASE 3: Gradient descent + perturbation restarts ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Gradient descent + restarts", flush=True)
    print(f"{'='*60}", flush=True)
    phase3_end = t_start + min(850.0, 0.7 * budget_s)

    if best_state['seq'] is not None:
        remaining = phase3_end - time.monotonic()
        if remaining > 30:
            a_grad, obj_grad = gradient_descent_smooth(
                best_state['seq'], time.monotonic() + remaining * 0.5,
                tau_init=15.0, tau_max=100.0, lr_init=0.02)
            global_update(a_grad)
            print(f"  Grad descent: J={obj_grad:.8f}", flush=True)

            # Perturbation restarts
            remaining = phase3_end - time.monotonic()
            if remaining > 30:
                n_restarts = min(8, int(remaining / 25))
                for r in range(n_restarts):
                    rem = phase3_end - time.monotonic()
                    if rem < 20:
                        break
                    sigma = 0.05 + 0.05 * r
                    a_pert = best_state['seq'] + rng.normal(0, sigma, best_state['seq'].size)
                    a_pert = project_simplex(a_pert)
                    alloc = min(20.0, rem - 5)
                    a_opt = coord_descent(a_pert, time.monotonic() + alloc, seed=2000 + r)
                    obj_r = global_update(a_opt)
                    print(f"  Restart {r} (sigma={sigma:.2f}): J={obj_r:.8f}", flush=True)

    # === PHASE 4: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    phase4_end = t_start + min(1100.0, 0.82 * budget_s)

    candidate_ns = []
    for cand_n in [384, 448, 512, 576, 640, 704, 832, 896, 1024, 1152, 1280]:
        if cand_n != n_parent and 128 <= cand_n <= 2048:
            candidate_ns.append(cand_n)
    candidate_ns.sort(key=lambda x: abs(x - n_parent))

    for new_n in candidate_ns[:4]:
        remaining = phase4_end - time.monotonic()
        if remaining < 40:
            break
        print(f"  Trying n={new_n}...", flush=True)
        a_ext = resample_seq(best_state['seq'], new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        alloc = min(80.0, remaining - 10)
        a_opt = coord_descent(a_ext, time.monotonic() + alloc * 0.5, seed=5000 + new_n)
        obj_n1 = global_update(a_opt)

        rem2 = phase4_end - time.monotonic()
        if rem2 > 25:
            a_opt2, _ = sca_lp(a_opt, time.monotonic() + min(rem2 * 0.8, 40.0))
            global_update(a_opt2)

        print(f"  n={new_n}: J={best_state['obj']:.8f}", flush=True)

    # === PHASE 5: Final intensive polishing ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final polishing (budget {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_state['seq'] is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Alternating CD and SCA
            n_rounds = min(4, int(remaining / 30))
            for round_i in range(n_rounds):
                rem = deadline - time.monotonic()
                if rem < 15:
                    break
                # CD pass
                t_cd = time.monotonic() + rem * 0.4
                a_cd = coord_descent(best_state['seq'], t_cd, seed=7000 + round_i)
                global_update(a_cd)

                rem = deadline - time.monotonic()
                if rem < 15:
                    break
                # SCA pass
                t_sca = time.monotonic() + min(rem * 0.4, 30.0)
                a_sca, _ = sca_lp(best_state['seq'], t_sca)
                global_update(a_sca)

                rem = deadline - time.monotonic()
                if rem < 10:
                    break
                # Small perturbation + CD
                a_pert = best_state['seq'] + rng.normal(0, 0.01, best_state['seq'].size)
                a_pert = project_simplex(a_pert)
                a_pert_opt = coord_descent(a_pert, time.monotonic() + min(rem * 0.3, 20.0),
                                           seed=9000 + round_i)
                global_update(a_pert_opt)

            # Final CD with max time
            rem = deadline - time.monotonic()
            if rem > 10:
                a_final = coord_descent(best_state['seq'], deadline - 3, seed=12345)
                global_update(a_final)

    # Fallback
    if best_state['seq'] is None:
        best_state['seq'] = np.ones(256) / 256.0
        best_state['obj'] = compute_J(best_state['seq'])
        print(f"  Fallback: J={best_state['obj']:.6f}", flush=True)

    # Final evaluation
    final_seq = best_state['seq']
    final_obj = compute_J(final_seq)
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={final_obj:.8f} n={final_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in final_seq]
