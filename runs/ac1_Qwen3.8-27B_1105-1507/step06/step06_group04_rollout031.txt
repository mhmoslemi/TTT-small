        grad_a = compute_grad_a(a, w, fft_size)

        # Convert to z-gradient (softmax parameterization)
        # grad_z[i] = a[i] * (grad_a[i] - sum(grad_a * a))
        ga_dot_a = np.dot(grad_a, a)
        grad_z = a * (grad_a - ga_dot_a)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Convert back to a via softmax
        z_shifted = z - np.max(z)
        exp_z = np.exp(z_shifted)
        a = exp_z / np.sum(exp_z)

        # Evaluate
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        # Decay temperature
        T = max(T * T_decay, T_min)

        # Slowly decay learning rate
        if t % 500 == 0:
            lr *= 0.9

    return best_a, best_obj

def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=20):
    """
    Solve: min t subject to (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    """
    n = len(fixed_seq)
    if n > 1000:
        return None, None
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None

def peak_chase_subgradient(a, time_budget=30.0, num_peaks=5, lr=0.02):
    """
    Chase top-K peaks with subgradient, using FFT for gradient computation.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    v = np.zeros(n)
    mu = 0.8

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Top-K peaks
        top_k = np.argsort(conv)[-num_peaks:]
        # Weight by how close to max
        weights = np.zeros(len(conv))
        for k in top_k:
            weights[k] = np.exp(-(cmax - conv[k]) / 0.002)
        w_sum = np.sum(weights)
        if w_sum > 0:
            weights /= w_sum
        else:
            weights[top_k[0]] = 1.0

        # Gradient
        grad_a = compute_grad_a(a, weights, fft_size)
        grad_centered = grad_a - np.mean(grad_a)
        gn = np.linalg.norm(grad_centered)
        if gn < 1e-15:
            break

        v = mu * v - lr * grad_centered / gn
        a_new = normalize_simplex(a + v)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.9
            v *= 0.5

    return best_a, best_obj

Adam-based smooth-max auto-correlation minimization with log-parameterization,
multi-scale optimization, adaptive temperature scheduling, active-set focusing,
and LP polishing. Targets improvement over 1.506547 toward 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def compute_grad_a(a, weights, fft_size):
    """
    Compute gradient of sum_k weights[k]*B[k] wrt a[i].
    grad_a[i] = 2 * sum_j a[j] * weights[i+j] (for valid 0<=i+j<2n-1).
    Uses FFT: corr[i] = sum_j a[j]*w[i+j] = ifft(fft(a)*conj(fft(w)))[i]
    """
    n = len(a)
    A_pad = np.zeros(fft_size)
    A_pad[:n] = a
    W_pad = np.zeros(fft_size)
    W_pad[:len(weights)] = weights

    A_fft = rfft(A_pad, fft_size)
    W_fft = rfft(W_pad, fft_size)
    corr = irfft(A_fft * np.conj(W_fft), fft_size)
    return 2.0 * corr[:n]

    t = 0

    est_steps = max(50, int(time_budget * 12))
    T = T_init
    T_decay = (T_min / T_init) ** (1.0 / est_steps)

    active_idx = None
    last_as_update = -1000

        cmax = float(np.max(conv))

        # Update active set periodically
        if t - last_as_update >= active_set_update:
            gap = cmax - np.sort(conv)[-max(1, int(active_set_frac * (2 * n - 1)))]
            threshold = cmax - max(gap * 2, cmax * 0.01)
            active_idx = np.where(conv >= threshold)[0]
            last_as_update = t

        # Compute smooth max weights
        if active_idx is not None and len(active_idx) > 0:
            T_eff = max(T * 0.3, T_min)
            conv_act = conv[active_idx]
            shifted = np.clip((conv_act - np.max(conv_act)) / T_eff, -50, 50)
            w_act = np.exp(shifted)
            ws = np.sum(w_act)
            if ws < 1e-300:
                w_act = np.zeros(len(conv_act))
                w_act[int(np.argmax(conv_act))] = 1.0
            else:
                w_act /= ws
            w = np.zeros(len(conv))
            w[active_idx] = w_act
        else:
            T_eff = max(T, T_min)
            shifted = np.clip((conv - cmax) / T_eff, -50, 50)
            w = np.exp(shifted)
            ws = np.sum(w)
            if ws < 1e-300:
                w = np.zeros(len(conv))
                w[int(np.argmax(conv))] = 1.0
            else:
                w /= ws

        # Gradient wrt a
        grad_a = compute_grad_a(a, w, fft_size)

        # Convert to z-space gradient (softmax parameterization)
        # grad_z[i] = a[i] * (grad_a[i] - sum(grad_a * a))
        ga_da = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - ga_da)

        # Adam
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Softmax to get new a
        z_shift = z - np.max(z)
        exp_z = np.exp(z_shift)
        a = exp_z / np.sum(exp_z)

        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        T = max(T * T_decay, T_min)
        if t % 1000 == 0:
            lr *= 0.92

    return best_a, best_obj


def peak_chase(a, time_budget=30.0, num_peaks=5, lr=0.015):
    """Subgradient method chasing top-K peaks with momentum."""
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    vel = np.zeros(n)
    mu = 0.85

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        cmax = float(np.max(conv))

        top_k = np.argsort(conv)[-num_peaks:]
        w = np.zeros(len(conv))
        for k in top_k:
            w[k] = np.exp(-(cmax - conv[k]) / 0.003)
        ws = np.sum(w)
        if ws > 1e-30:
            w /= ws
        else:
            w[top_k[0]] = 1.0

        grad_a = compute_grad_a(a, w, fft_size)
        grad_c = grad_a - np.mean(grad_a)
        gn = np.linalg.norm(grad_c)
        if gn < 1e-15:
            break

        vel = mu * vel - lr * grad_c / gn
        a_new = normalize_simplex(a + vel)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.92
            vel *= 0.5

    return best_a, best_obj


def build_toeplitz(a):
    """Build (2n-1)x(n) matrix for convolve(a, .)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def lp_minpeak(fixed_seq, time_limit=15):
    """LP: min t s.t. (fixed_seq * g)[k] <= t, sum(g)=1, g>=0."""
    n = len(fixed_seq)
    if n > 1000:
        return None
    A = build_toeplitz(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_polish(a, time_budget=50.0):
    """LP alternating polish."""
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break
        g = lp_minpeak(a, time_limit=min(12, max(3, time_budget - (time.monotonic() - start) - 5)))
        if g is None:
            break
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = normalize_simplex((1 - t) * a + t * g)
            o = compute_obj(a_try)
            if o < best_obj:
                best_obj = o
                best_a = a_try.copy()
        a = best_a.copy()

    return best_a, best_obj


def resample(a, new_n):
    """Linear interpolation resampling."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def gen_starts(n, seed=0, num=8):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    i = np.arange(n, dtype=float)
    c = (n - 1) / 2.0
    starts = [np.ones(n) / n]
    for sf in [0.15, 0.25, 0.4, 0.6]:
        b = np.exp(-((i - c) ** 2) / (2 * (n * sf) ** 2))
        starts.append(b / b.sum())
    ce = 0.5 * (1 + np.cos(np.pi * (i - c) / max(1, n / 2)))
    starts.append(np.maximum(ce, 0.01) / np.maximum(ce, 0.01).sum())
    for _ in range(3):
        f = 1 + int(rng.integers(0, 5))
        amp = 0.1 + 0.4 * rng.random()
        ph = 2 * np.pi * rng.random()
        base = np.exp(-((i - c) ** 2) / (2 * (n * 0.3) ** 2))
        mod = 1 + amp * np.cos(2 * np.pi * f * (i - c) / n + ph)
        a = np.maximum(base * mod, 0)
        starts.append(a / a.sum() if a.sum() > 0 else np.ones(n) / n)
    return starts[:num]


def try_load_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000.0)
        if p.size > 0 and np.isfinite(p).all():
            o = compute_obj(p)
            print(f"Parent (n={p.size}): obj={o:.8f}")
            return p.copy(), o
    except Exception:
        pass
    return None, np.inf
