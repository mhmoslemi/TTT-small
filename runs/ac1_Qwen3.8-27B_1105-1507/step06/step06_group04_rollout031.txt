"""
Optimized LP-alternating peak reduction for step function autocorrelation minimization.
Combines max-sum LP with golden-section line search, adaptive constraint relaxation,
LP-dual-guided subgradient refinement, and multi-length exploration to push below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_maxsum_lp(a, M, time_limit=30):
    """
    Solve: max sum(g) s.t. (a * g)[k] <= M for all k, g >= 0.
    Returns (g_normalized, dual_y) or (None, None).
    """
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                g_norm = g / s
                dual_y = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
                return g_norm, dual_y
    except Exception:
        pass
    return None, None


def solve_minpeak_lp(a, time_limit=30):
    """
    Solve: min t s.t. (a * g)[k] <= t for all k, sum(g) = 1, g >= 0.
    Returns g_normalized or None.
    """
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section_search(a, g, max_evals=30):
    """
    Golden section search for min_t obj((1-t)*a + t*g) over t in [0, 1].
    Both a and g are normalized to sum=1.
    Returns (best_t, best_obj).
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

    def eval_at(t):
        a_try = (1.0 - t) * a + t * g
        a_try = normalize_simplex(a_try)
        return compute_obj(a_try)

    fc = eval_at(c)
    fd = eval_at(d)

    for _ in range(max_evals):
        if hi - lo < 0.005:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_at(d)

    mid = (lo + hi) / 2.0
    fmid = eval_at(mid)
    best_t = mid
    best_obj = fmid
    for t_check in [c, d, lo, hi, 0.0, 1.0]:
        fc2 = eval_at(t_check)
        if fc2 < best_obj:
            best_obj = fc2
            best_t = t_check
    return best_t, best_obj


def compute_subgradient_fft(a, weights, fft_size):
    """
    Compute subgradient: grad[i] = 2 * sum_k weights[k] * a[k-i] for valid k-i.
    Uses FFT cross-correlation.
    """
    n = len(a)
    w_ext = np.zeros(fft_size)
    w_ext[:len(weights)] = weights
    W_ext = rfft(w_ext, fft_size)
    A_ext = rfft(a, fft_size)
    corr = irfft(A_ext * np.conj(W_ext), fft_size)
    grad = 2.0 * corr[:n]
    return grad


def subgradient_refine(a, max_iters=200, time_budget=15.0, tau=None):
    """
    Softmax-subgradient refinement with Nesterov momentum.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau is None:
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        tau = 0.005 * cmax if cmax > 0 else 0.01

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    v = np.zeros(n)
    mu = 0.8
    step = 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -400.0, 400.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        grad = compute_subgradient_fft(a, weights, fft_size)
        grad_centered = grad - np.mean(grad)
        gn = np.linalg.norm(grad_centered)
        if gn < 1e-15:
            break

        v = mu * v - step * grad_centered / gn
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        step *= 0.997
        tau *= 0.998

    return best_a, best_obj


def lp_alternate_optimize(a, time_budget=200.0, use_smoothing=True, smooth_sigma=1.5,

    """

                kernel /= np.sum(kernel)

def generate_starts(n, parent=None, seed=42, num_starts=10):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []

    if parent is not None and len(parent) == n:
        starts.append(normalize_simplex(parent.copy()))

    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    starts.append(np.ones(n) / n)

    # Gaussian (narrow)
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 8.0) ** 2))
    starts.append(bell / np.sum(bell))

    # Gaussian (wide)
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 3.0) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # Triangular
    tri = 1.0 - np.abs(i - center) / (center + 1)
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
    cos_w = np.maximum(cos_w, 0.01)
    starts.append(cos_w / np.sum(cos_w))

    # Parametric with modulation
    for k in range(4):
        sigma = n * (0.2 + 0.3 * rng.random())
        freq = int(1 + rng.integers(0, 6))
        amp = 0.2 + 0.4 * rng.random()
        phase = 2 * np.pi * rng.random()
        bell_p = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
        a = np.maximum(bell_p * mod, 0)
        s = np.sum(a)
        if s > 1e-10:
            starts.append(a / s)

    # Noisy
    noisy = np.abs(1.0 + rng.normal(0, 0.3, n))
    starts.append(noisy / np.sum(noisy))

    return starts[:num_starts]


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-15:
        a_new = a_new / s
    return a_new
