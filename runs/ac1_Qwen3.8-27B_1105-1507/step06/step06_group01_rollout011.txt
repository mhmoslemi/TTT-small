    # Actually: sum_m p[m] a[m-n]. Let i = m-n, so m = i+n:
    # = sum_i p[i+n] a[i]
    #
    # YES! So ifft(FFT(p, N) * conj(FFT(a, N)))[n] = sum_i p[i+n] * a[i]
    # This is exactly what we want! (the cross-correlation at lag n)
    
    size = 1
    while size < n_p + n_a - 1:
        size <<= 1
    Fp = np.fft.rfft(p, size)
    Fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(Fp * np.conj(Fa), size)
    return corr[:n_a]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    d = np.sort(v)[::-1]
    cv = np.cumsum(d)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (d - (cv - 1.0) / rho) > 0
    rho_star = int(np.max(rho[cond])) if cond.any() else 0
    if rho_star == 0:
        return np.ones(n) / n
    theta = (cv[rho_star - 1] - 1.0) / rho_star
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad_fft(a, tau):
    """Compute smooth max approximation of max(conv(a,a)) and its gradient using FFT.
    Returns (smooth_value, gradient)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Log-sum-exp for stability
    exp_b = np.exp(tau * (b - b_max))
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum
    smooth_val = b_max + np.log(exp_sum) / tau
    
    # Gradient: grad[j] = 2 * sum_i p[i+j] * a[i] = 2 * cross_corr(p, a)[j]
    grad = 2.0 * fast_cross_corr(p, a)
    return smooth_val, grad


def run_fista_fft(a_init, tau_start, tau_end, max_iters, deadline, lr=0.3):
    """FISTA with FFT-accelerated gradients."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_grad_fft(y, tau)
        
        # Project gradient to tangent space of simplex
        grad_tan = grad_y - grad_y.mean()
        
        # Step size: use Lipschitz constant estimate
        # The smooth max has Lipschitz constant approximately tau * max|grad|
        lip_est = np.linalg.norm(grad_tan, np.inf) * tau
        step = min(lr / max(lip_est, 0.01), 0.5)
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        it += 1
        
        # Track best periodically
        if it % 10 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    
    M_a = fast_conv(a).max()
    if M_a < best_M:
        best_M = M_a
        best_a = a.copy()
    
    return best_a


def subgradient_method_with_averaging(a_init, max_iters, deadline, step_scale=0.1):
    """Subgradient method with Polyak step size and running average.
    
    At each step, sample a random (or the max) subgradient and take a step.
    Maintain a running average of iterates (this is the key to convergence of
    subgradient methods for non-smooth problems).
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Running average
    avg_a = a.copy()
    avg_M = best_M
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Get subgradient at the argmax
        k_star = int(np.argmax(b))
        subgrad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                subgrad[i] = 2.0 * a[j]
        
        # Also sample from near-max indices for diversity
        if it % 5 == 0:
            # Use top-5 max indices weighted
            top_idx = np.argsort(b)[::-1][:5]
            top_vals = b[top_idx]
            weights = np.exp((top_vals - top_vals[0]) / (M + 1e-10))
            weights = weights / weights.sum()
            subgrad = np.zeros(n)
            for w, ki in zip(weights, top_idx):
                for i in range(n):
                    j = ki - i
                    if 0 <= j < n:
                        subgrad[i] += w * 2.0 * a[j]
        
        grad_tan = subgrad - subgrad.mean()
        
        # Step size: diminishing
        step = step_scale / np.sqrt(it + 1.0)
        
        a_new = a - step * grad_tan
        a_new = proj_simplex(a_new)
        a = a_new
        
        # Update running average
        avg_a = ((it + 1.0) * a + it * avg_a) / (it + 2.0)
        
        # Track best
        if it % 20 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_avg = fast_conv(avg_a).max()
            if M_avg < best_M:
                best_M = M_avg
                best_a = avg_a.copy()
    
    # Final check
    M_a = fast_conv(a).max()
    if M_a < best_M:
        best_M = M_a
        best_a = a.copy()
    M_avg = fast_conv(avg_a).max()
    if M_avg < best_M:
        best_M = M_avg
        best_a = avg_a.copy()
    
    return best_a


def adaptive_lp_best_response(a, current_M, n_active_start=20, deadline_lp=None):
    """LP best response with adaptive active set based on dual variables.
    
    Given current a, find g that minimizes max_k (a*g)[k] s.t. sum(g)=1, g>=0.
    Uses active set: constraints where (a*g)[k] is close to the optimum.
    """
    n = len(a)
    b = fast_conv(a)
    
    # Initial active set: top n_active constraints
    top_indices = np.argsort(b)[::-1][:n_active_start]
    
    best_g = None
    for attempt in range(3):
        active = top_indices.tolist()
        A = build_lp_matrix(n, a, active)
        if A.nnz == 0:
            break
        
        c = -np.ones(n)
        b_ub = np.full(len(active), current_M)
        bounds = [(0.0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 5, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0.0)
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g = g / g_sum
                    # Check how many constraints are violated
                    ag = fast_conv(a) * 0 + np.convolve(a, g, mode='full')
                    violations = ag > current_M + 1e-8
                    if violations.sum() == 0:
                        best_g = g
                        break
                    else:
                        # Add violated constraints to active set
                        viol_indices = np.where(violations)[0]
                        # Merge with existing active set
                        all_indices = np.unique(np.concatenate([top_indices, viol_indices]))
                        top_indices = all_indices[:min(50, len(all_indices))]
                        # Re-sort by b value
                        top_indices = all_indices[np.argsort(b[all_indices])[::-1]][:n_active_start + len(viol_indices)]
                else:
                    best_g = g
                    break
        except Exception:
            break
    
    return best_g


def build_lp_matrix(n, a, active_indices):
    """Build sparse LP matrix for given active constraints."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def run_lp_iteration(a, deadline, max_iters=100, initial_n_active=30):
    """Iterative LP-based optimization.
    
    At each step:
    1. Compute best response g via LP
    2. Line search: find t that minimizes M of (1-t)*a + t*g
    3. Update a
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    n_active = initial_n_active
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = adaptive_lp_best_response(a, best_M, n_active)
        if g is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        
        # Line search
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                # Reduce active set
                if n_active > 10:
                    n_active = max(10, int(n_active * 0.9))
                break
        
        if not improved:
            no_improve += 1
            n_active = min(int(n_active * 1.3), 2 * n - 1)
            if no_improve > 20:
                break
    
    return best_a


def run_mirror_descent(a_init, max_iters, deadline, step_init=0.2, use_fft_grad=True):
    """Mirror descent (multiplicative weights) with optional FFT-accelerated gradients."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    tau = 50.0  # temperature for smooth approximation
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        if use_fft_grad:
            _, grad = smooth_max_grad_fft(a, tau)
        else:
            b = fast_conv(a)
            k_star = int(np.argmax(b))
            grad = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] = 2.0 * a[j]
        
        grad_tan = grad - grad.mean()
        
        step = step_init / (1.0 + 0.01 * it)
        log_weights = -step * grad_tan
        log_weights -= log_weights.max()
        a_new = a * np.exp(log_weights)
        a_new = np.maximum(a_new, 1e-14)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        if it % 10 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
        
        # Anneal temperature
        tau = max(20.0, tau * 0.999)
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def coordinate_descent_optimize(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with greedy ordering (update the most impactful coordinates first)."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        # Order coordinates by their contribution to the maximum
        # Coordinates that appear in the argmax convolution have highest impact
        k_star = int(np.argmax(b))
        impact = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                impact[i] = a[j]
        
        # Sort by impact (descending), with random tiebreaking
        order = np.array(sorted(range(n), key=lambda i: (-impact[i], rng.random())))
        
        sweep_improved = False
        for i in order:
            if time.monotonic() > deadline:
                break
            x_opt = optimize_coordinate(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-10:
                # Update b incrementally
                old_x = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += x_opt * x_opt - old_x * old_x
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old_x)
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 5
        if no_improve > 30:
            break
    
    return best_a


def optimize_coordinate(a, i, b, n):
    """Find optimal value for a[i] given all others fixed.
    Minimizes max of piecewise-linear (plus one quadratic) function."""
    x_cur = a[i]
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Collect the linear/quadratic pieces
    c_vals = []
    slopes = []
    quad_idx = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            quad_idx = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    # Base max (unaffected coordinates)
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals) if c_vals else np.array([0.0])
    s_arr = np.array(slopes) if slopes else np.array([0.0])
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if quad_idx >= 0:
            qv = c_arr[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    # Golden section search
    lo = 0.0
    hi = max(3.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 1.0, 0.1)
    
    # Ensure hi is large enough
    for _ in range(5):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 50.0:
            hi *= 2.0
        else:
            break
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def two_point_reflection(a_init, max_iters, deadline, mu=0.5):
    """Two-point reflection method for the simplex constraint.
    
    At each step:
    1. Compute subgradient g at current point
    2. Find the point on the hyperplane sum(x)=1 closest to a - step*g
    3. Reflect to get a new point
    4. Take a convex combination
    
    This is related to the "proximal gradient" method for non-smooth objectives.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Reference point (average of iterates)
    avg_a = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sg[i] = 2.0 * a[j]
        
        # Step size based on distance to optimum (Polyak step)
        # We estimate the optimum as 1/(2n-1) (flat autocorrelation)
        M_opt_est = 1.0 / (2 * n - 1)
        gap = M - M_opt_est
        if gap < 1e-6:
            break
        
        # Estimated Lipschitz constant
        L = 2.0 * a.max() + 1e-6
        step = mu * gap / (L * L)
        step = min(step, 0.1)  # Cap the step
        
        # Mirror step (multiplicative)
        a_trial = a * np.exp(-step * (sg - sg.mean()))
        a_trial = np.maximum(a_trial, 1e-14)
        a_trial = a_trial / a_trial.sum()
        
        # Average
        avg_a = ((it + 1.0) * a_trial + it * avg_a) / (it + 2.0)
        
        # Track best
        if it % 10 == 0:
            M_cur = fast_conv(a_trial).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a_trial.copy()
            M_avg = fast_conv(avg_a).max()
            if M_avg < best_M:
                best_M = M_avg
                best_a = avg_a.copy()
        
        a = a_trial
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed versions
        for noise_level in [0.01, 0.05, 0.1, 0.2]:
            p_n = np.maximum(p + rng.normal(0, noise_level, n), 1e-6)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various window functions
    # Triangular
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    inits.append(tri / tri.sum())
    
    # Raised cosine
    rc = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(rc / rc.sum())
    
    # Hanning
    hanning = 0.5 * (1 - np.cos(2 * np.pi * t))
    inits.append(hanning / hanning.sum())
    
    # Gaussian
    for sigma in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Polynomial windows
    for p in [2, 3, 4]:
        poly = (t * (1 - t)) ** (p / 4.0) + 0.01
        inits.append(poly / poly.sum())
    
    # Comb / frequency-modulated
    for freq in [3, 7, 13, 21]:
        comb = 0.7 + 0.3 * np.cos(2 * np.pi * freq * t)
        inits.append(comb / comb.sum())
    
    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.exp(-0.5 * (np.arange(-ksize//2, ksize//2+1) / (ksize/4.0))**2)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.001)
    inits.append(smooth / smooth.sum())
    
    # Anti-structured: try to make autocorrelation flat from the start
    a_flat = np.ones(n) / n
    for _ in range(50):
        b = fast_conv(a_flat)
        # Use gradient of max to push down peaks
        k_star = int(np.argmax(b))
        sg = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sg[i] = a_flat[j]
        a_flat = a_flat - 0.1 * (sg - sg.mean())
        a_flat = np.maximum(a_flat, 0.001)
        a_flat = a_flat / a_flat.sum()
    inits.append(a_flat.copy())
    
    return inits


def run_optimization_pipeline(a0, n, deadline, seed=0):
    """Run the full optimization pipeline on a single initialization."""
    best = a0.copy()
    best_M = fast_conv(a0).max()
    t_start = time.monotonic()
    elapsed_budget = deadline - t_start
    
    # Stage 1: FISTA (25% of budget)
    t1 = t_start + 0.25 * elapsed_budget
    if time.monotonic() < t1:
        a1 = run_fista_fft(best, 50, 200, 5000, min(t1, time.monotonic() + 30), lr=0.2)
        M1 = fast_conv(a1).max()
        if M1 < best_M:
            best_M = M1
            best = a1.copy()
    
    # Stage 2: Mirror descent with FFT (20% of budget)
    t2 = t_start + 0.45 * elapsed_budget
    if time.monotonic() < t2:
        a2 = run_mirror_descent(best, 3000, min(t2, time.monotonic() + 30), step_init=0.15, use_fft_grad=True)
        M2 = fast_conv(a2).max()
        if M2 < best_M:
            best_M = M2
            best = a2.copy()
    
    # Stage 3: LP iterations (30% of budget)
    t3 = t_start + 0.75 * elapsed_budget
    if time.monotonic() < t3:
        a3 = run_lp_iteration(best, min(t3, time.monotonic() + 60), max_iters=50, initial_n_active=25)
        M3 = fast_conv(a3).max()
        if M3 < best_M:
            best_M = M3
            best = a3.copy()
        
        # Second LP pass with tighter active set
        if time.monotonic() < t3:
            a3b = run_lp_iteration(a3, min(t3, time.monotonic() + 30), max_iters=30, initial_n_active=15)
            M3b = fast_conv(a3b).max()
            if M3b < best_M:
                best_M = M3b
                best = a3b.copy()
    
    # Stage 4: Coordinate descent (15% of budget)
    t4 = t_start + 0.90 * elapsed_budget
    if time.monotonic() < t4:
        max_sw = max(5, int((t4 - time.monotonic()) / max(0.5, n / 80.0)))
        a4 = coordinate_descent_optimize(best, max_sw, min(t4, time.monotonic() + 30), seed=seed)
        M4 = fast_conv(a4).max()
        if M4 < best_M:
            best_M = M4
            best = a4.copy()

        M5 = fast_conv(a5).max()

        M5b = fast_conv(a5b).max()
        if M5b < best_M:
            best_M = M5b
            best = a5b.copy()
    
    return best
