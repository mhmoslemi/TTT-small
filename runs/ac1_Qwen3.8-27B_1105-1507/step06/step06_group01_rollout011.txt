def fast_conv(a):
    n = len(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def proj_nonneg_sum1(v):

def lp_best_response(a, M, active_indices):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    if A.nnz == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def coordinate_optimize(a, i, b, n):
    """Find optimal a[i] given all others fixed. Returns optimal value."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals) if c_vals else np.array([0.0])
    s_arr = np.array(slopes) if slopes else np.array([0.0])
    
    def obj_at(x_test):
        if len(c_vals) > 0:
            vals = c_arr + s_arr * x_test
            m = vals.max()
        else:
            m = base_max
        if m < base_max:
            m = base_max
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.3)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(35):

            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)

def simulated_annealing(a_init, max_iters, deadline, seed=0):
    """Simulated annealing with structural moves."""
    rng_sa = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    T = 0.05
    T_min = 0.001
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        T = max(T_min, T * 0.999)
        
        # Random structural move
        move_type = rng_sa.integers(0, 4)
        a_trial = a.copy()
        
        if move_type == 0:
            # Single coordinate change
            i = rng_sa.integers(0, n)
            a_trial[i] *= rng_sa.uniform(0.5, 2.0)
        elif move_type == 1:
            # Block change
            width = rng_sa.integers(3, max(4, n // 10))
            start = rng_sa.integers(0, n - width)
            a_trial[start:start+width] *= rng_sa.uniform(0.5, 2.0, width)
        elif move_type == 2:
            # Mirror perturbation
            a_trial = a + rng_sa.normal(0, T, n)
            a_trial = np.maximum(a_trial, 0.0)
        else:
            # Swap two blocks
            w = rng_sa.integers(5, max(6, n // 20))
            i1 = rng_sa.integers(0, n - w)
            i2 = rng_sa.integers(0, n - w)
            if abs(i1 - i2) > w:
                a_trial[i1:i1+w], a_trial[i2:i2+w] = a_trial[i2:i2+w].copy(), a_trial[i1:i1+w].copy()
        
        s_t = a_trial.sum()
        if s_t < 0.01:
            continue
        a_trial = a_trial / s_t
        M_trial = fast_conv(a_trial).max()
        
        if M_trial <= best_M or rng_sa.random() < np.exp(-(M_trial - best_M) / T):
            a = a_trial
            if M_trial < best_M:
                best_M = M_trial
                best_a = a.copy()
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
