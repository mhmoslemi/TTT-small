# NesterovAcceleration class for handling smooth optimization
class AndersonAcceleration():
    def __init__(self, rate, history, prop):
        self.rate = rate
        self.history = np.array([[history[i][j], prop[i][j]] for i in range(n)])
        self.values = np.zeros(len(history[0]))
        
    def mix(self, x):
        ones = np.full(len(x), self.rate, dtype=float)
        return np.dot(x, ones) / np.sum(x)
    
    def update(self, x):
        for i in range(len(x)):
            self.values[i] = min(1.0, max(0.0, x[i]))
        return self.values
