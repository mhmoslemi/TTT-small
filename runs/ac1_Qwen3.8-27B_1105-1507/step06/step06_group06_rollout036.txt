```python
import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s
```

Now, for the Fisher-Rao gradient flow:
- The "gradient" of `f(a) = max_k conv(a,a)[k]` (or smoothed version) in the Fisher-Rao metric leads to the update:
  `a[i] <- a[i] / (1 + eta * grad[i])` (approximately, for small eta)
  or equivalently: `a[i] <- a[i] * exp(-eta * grad[i])`

This is the multiplicative weight update. After the update, we normalize to sum to 1.

The key advantage of Fisher-Rao over Euclidean gradient descent is that it naturally respects the non-negativity constraint (multiplicative updates never make values negative) and has self-adaptive step size (larger values get larger absolute steps).

```python
"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: (1) KKT-condition-guided fixed-point iteration exploiting
Hankel matrix structure, (2) Fisher-Rao gradient flow with adaptive smoothing,
(3) refined LP alternation, (4) multi-length exploration with resampling.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return np.maximum(conv[:2 * n - 1], 0.0)


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Project onto simplex {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 1500:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def kkt_iteration(a, max_iters=100, time_budget=60.0, active_threshold=0.1):
    """
    KKT-condition-guided fixed-point iteration.
    At each step: identify active set, build Hankel matrix, solve linear system,
    project, and update.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        if cmax < 1e-15:
            break

        # Determine active set: values within (1 - active_threshold) of max
        threshold = cmax * (1.0 - active_threshold)
        active_mask = conv >= threshold
        n_active = np.sum(active_mask)

        if n_active < 2:
            active_mask = np.zeros(len(conv), dtype=bool)
            top_idx = np.argsort(conv)[-max(2, len(conv) // 10):]
            active_mask[top_idx] = True
            n_active = np.sum(active_mask)

        # Build lambda vector
        lam = np.zeros(2 * n - 1)
        lam[active_mask] = 1.0

        # Build Hankel matrix H: H[i,j] = 2 * lam[i+j] if 0 <= i+j < 2n-1
        # Use outer product structure for efficiency
        i_idx = np.arange(n)[:, None]
        j_idx = np.arange(n)[None, :]
        ij = i_idx + j_idx
        H = np.where(ij < 2 * n - 1, 2.0 * lam[ij], 0.0)

        # Solve augmented system: [H  -1; 1^T  0] [a; c] = [0; 1]
        A_aug = np.zeros((n + 1, n + 1))
        A_aug[:n, :n] = H
        A_aug[:n, n] = -1.0
        A_aug[n, :n] = 1.0
        b_aug = np.zeros(n + 1)
        b_aug[n] = 1.0

        # Add small regularization for stability
        A_aug[:n, :n] += 1e-8 * np.eye(n)

        try:
            sol = np.linalg.solve(A_aug, b_aug)
            a_kkt = sol[:n]
        except np.linalg.LinAlgError:
            try:
                a_kkt, _, _, _ = np.linalg.lstsq(A_aug, b_aug, rcond=1e-10)
            except Exception:
                break

        a_kkt = np.maximum(a_kkt, 0)
        s_kkt = np.sum(a_kkt)
        if s_kkt < 1e-10:
            break
        a_kkt = a_kkt / s_kkt

        # Line search: try different blend factors
        best_t_obj = best_obj
        best_t_val = 0.3
        best_a_try = a.copy()

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * a_kkt
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_try = a_try.copy()

        if best_t_obj < best_obj:
            best_obj = best_t_obj
            best_a = best_a_try.copy()
            a = best_a_try.copy()
        else:
            # If no improvement, try a smaller step or break
            a_try = (1.0 - 0.05) * a + 0.05 * a_kkt
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
            else:
                # Reduce active threshold (expand active set)
                active_threshold = max(0.01, active_threshold * 0.8)

    return best_a, best_obj

    start = time.monotonic()

    conv = compute_conv(a)
    cmax = float(np.max(conv))
    tau = cmax * 0.1
    kkt_interval = 50
    active_threshold = 0.15

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Every kkt_interval steps, do a KKT update
        if it % kkt_interval == 0 and n <= 1000:
            # Quick KKT step
            conv = compute_conv(a)
            cmax = float(np.max(conv))
            threshold = cmax * (1.0 - active_threshold)
            active_mask = conv >= threshold
            if np.sum(active_mask) < 2:
                top_idx = np.argsort(conv)[-max(2, len(conv) // 10):]
                active_mask = np.zeros(len(conv), dtype=bool)
                active_mask[top_idx] = True

            lam = np.zeros(2 * n - 1)
            lam[active_mask] = 1.0

            i_idx = np.arange(n)[:, None]
            j_idx = np.arange(n)[None, :]
            ij = i_idx + j_idx
            H = np.where(ij < 2 * n - 1, 2.0 * lam[ij], 0.0)
            H += 1e-8 * np.eye(n)

            A_aug = np.zeros((n + 1, n + 1))
            A_aug[:n, :n] = H
            A_aug[:n, n] = -1.0
            A_aug[n, :n] = 1.0
            b_aug = np.zeros(n + 1)
            b_aug[n] = 1.0

            try:
                sol = np.linalg.solve(A_aug, b_aug)
                a_kkt = np.maximum(sol[:n], 0)
                s_kkt = np.sum(a_kkt)
                if s_kkt > 1e-10:
                    a_kkt = a_kkt / s_kkt
                    a_try = 0.5 * a + 0.5 * a_kkt
                    a_try = project_simplex(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                    a = a_try
            except Exception:
                pass
        else:
            # FR subgradient step
            conv = compute_conv(a)
            cmax = float(np.max(conv))
            tau = max(cmax * 0.01, tau * 0.998)
            shifted = np.clip((conv - cmax) / tau, -300.0, 300.0)
            weights = np.exp(shifted)
            w_sum = weights.sum()
            if w_sum < 1e-300:
                weights = np.zeros(len(conv))
                weights[int(np.argmax(conv))] = 1.0
            else:
                weights /= w_sum

            # Cross-correlation via FFT
            n_conv = 2 * n - 1
            size_fft = 1
            while size_fft < n + n_conv - 1:
                size_fft <<= 1
            fa = np.fft.fft(a, size_fft)
            fw = np.fft.fft(weights, size_fft)
            corr_full = np.fft.ifft(np.conj(fa) * fw).real
            grad = 2.0 * corr_full[:n]
            grad_centered = grad - grad.mean()

            gmax = np.max(np.abs(grad_centered))
            if gmax < 1e-15:
                break
            eta = 0.8 / gmax
            a_new = a * np.exp(-np.clip(eta * grad_centered, -20, 20))
            a_new = project_simplex(a_new)

            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def lp_alternate(a, max_iters=30, time_budget=60.0):
    """LP alternating with adaptive line search."""
    n = len(a)
    if n > 1200:
        return a, compute_obj(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_max_sum(a, M, time_limit=min(10, int(time_budget - (time.monotonic() - start))))
        if g is None:
            break

        # Line search
        best_t_obj = best_obj
        best_a_try = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_try = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_try.copy()
            a = best_a_try.copy()
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break

    return best_a, best_obj


def freq_domain_smooth(a, max_iters=100, time_budget=20.0):
    """Frequency-domain iterative smoothing of power spectrum."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power_spec = np.abs(fa) ** 2
        ps_max = np.max(power_spec)
        if ps_max < 1e-15:
            break

        # Compress dynamic range
        ps_norm = power_spec / ps_max
        alpha = 0.8  # compression exponent
        ps_comp = np.maximum(ps_norm, 1e-10) ** alpha

        # Reconstruct
        phase = np.angle(fa)
        magnitude = ps_max ** 0.5 * ps_comp ** 0.5
        # Actually: new magnitude = sqrt(ps_comp * ps_max)
        magnitude = np.sqrt(ps_comp * ps_max)
        fa_new = magnitude * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def chase_peak(a, n_steps=40, step_size=0.02):
    """Chase the peak: iteratively reduce the maximum convolution value."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient direction at the peak
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
        # Also the symmetric contribution
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                # Already counted; the factor of 2 is implicit
                pass
        # Actually grad[i] = 2*a[k_star-i] if valid, else 0
        # Let me redo:
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        grad_centered = grad - grad.mean()

        for ss in [step_size, step_size * 0.5, step_size * 2.0]:
            a_try = a - ss * grad_centered
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    # Perturbed uniform
    p = np.ones(n) * (1 + rng.normal(0, 0.3, n))
    inits.append(project_simplex(p))

    # Sinusoidal
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Gaussian
    x = np.linspace(-3, 3, n)
    gauss = np.exp(-x ** 2 / 4)
    inits.append(gauss / gauss.sum())

    # Power law
    x = np.linspace(0.1, 1, n)
    pw = x ** rng.uniform(-2, 2)
    inits.append(pw / pw.sum())

    if parent is not None:
        resampled = resample_sequence(parent, n)
        inits.append(project_simplex(resampled))
        pert = resampled * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_simplex(pert))

    return inits


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization combining KKT fixed-point,
    FR subgradient, LP alternation, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = project_simplex(parent.copy())
    else:
        best_seq = np.ones(700) / 700
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    print(f"Starting: best={best_score:.8f}, n={len(best_seq)}")

    # === Phase 1: KKT + FR optimization on parent (40% of time) ===
    print("\n=== Phase 1: KKT + FR on parent ===")
    phase1_budget = min(600.0, (deadline - start_time) * 0.40)
    phase1_start = time.monotonic()

    a1 = best_seq.copy()
    n1 = len(a1)

    # KKT iteration first
    if n1 <= 1000:
        a_kkt, obj_kkt = kkt_iteration(a1, max_iters=80, time_budget=min(phase1_budget * 0.3, 60.0))
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  KKT: {best_score:.8f}")
        a1 = a_kkt

    # FR subgradient
    a_fr, obj_fr = fr_subgradient(a1, max_iters=3000, time_budget=min(phase1_budget * 0.4, 120.0))
    if obj_fr < best_score:
        best_score = obj_fr
        best_seq = a_fr.copy()
        print(f"  FR: {best_score:.8f}")
    a1 = a_fr

    # KKT again (refined)
    if n1 <= 1000:
        a_kkt2, obj_kkt2 = kkt_iteration(a1, max_iters=60, time_budget=min(phase1_budget * 0.2, 40.0))
        if obj_kkt2 < best_score:
            best_score = obj_kkt2
            best_seq = a_kkt2.copy()
            print(f"  KKT2: {best_score:.8f}")
        a1 = a_kkt2

    # LP alternation
    if n1 <= 1200:
        a_lp, obj_lp = lp_alternate(a1, max_iters=20, time_budget=min(phase1_budget * 0.1, 30.0))
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: {best_score:.8f}")

    print(f"  Phase 1 final: {best_score:.8f}")

    # === Phase 2: Hybrid KKT+FR with multiple restarts (30% of time) ===
    print("\n=== Phase 2: Hybrid optimization ===")
    phase2_budget = min(450.0, (deadline - start_time) * 0.30)
    phase2_start = time.monotonic()

    n_best = len(best_seq)

    # Try different initializations at the best length
    inits = generate_initializations(n_best, rng, best_seq)
    n_inits_to_try = min(len(inits), max(3, int(phase2_budget / 60)))

    for ii in range(n_inits_to_try):
        if time.monotonic() > phase2_start + phase2_budget:
            break
        remaining = phase2_start + phase2_budget - time.monotonic()
        if remaining < 30:
            break

        a_init = inits[ii].copy()
        a_h, obj_h = kkt_subgradient_hybrid(a_init, max_iters=2000, time_budget=min(remaining * 0.8, 100.0))
        if obj_h < best_score:
            best_score = obj_h
            best_seq = a_h.copy()
            print(f"  Hybrid init {ii}: {best_score:.8f}")

    # Continue with KKT on best
    remaining = phase2_start + phase2_budget - time.monotonic()
    if remaining > 20 and n_best <= 1000:
        a_kkt3, obj_kkt3 = kkt_iteration(best_seq, max_iters=80, time_budget=min(remaining * 0.7, 60.0))
        if obj_kkt3 < best_score:
            best_score = obj_kkt3
            best_seq = a_kkt3.copy()
            print(f"  KKT3: {best_score:.8f}")

    print(f"  Phase 2 final: {best_score:.8f}")

    # === Phase 3: Multi-length exploration (20% of time) ===
    print(f"\n=== Phase 3: Multi-length exploration ===")
    phase3_budget = min(300.0, (deadline - start_time) * 0.20)
    phase3_start = time.monotonic()

    n_current = len(best_seq)
    n_tries = [n_current - 100, n_current - 50, n_current + 50, n_current + 100,
               n_current + 200, n_current - 200, max(300, n_current - 300), min(2000, n_current + 300)]
    n_tries = list(dict.fromkeys([n for n in n_tries if 200 <= n <= 2500]))
    n_tries.sort(key=lambda x: abs(x - n_current))

    per_n_budget = max(15.0, phase3_budget / max(1, len(n_tries)))

    for n_try in n_tries[:6]:
        if time.monotonic() > phase3_start + phase3_budget:
            break
        remaining = phase3_start + phase3_budget - time.monotonic()
        if remaining < 15:
            break

        a_res = project_simplex(resample_sequence(best_seq, n_try))

        # Quick optimization
        a_opt, obj_opt = fr_subgradient(a_res, max_iters=1500, time_budget=min(remaining * 0.4, 40.0))
        if n_try <= 1000:
            a_kkt_r, obj_kkt_r = kkt_iteration(a_opt, max_iters=40, time_budget=min(remaining * 0.3, 30.0))
            if obj_kkt_r < obj_opt:
                a_opt, obj_opt = a_kkt_r, obj_kkt_r
        if n_try <= 1000:
            a_lp_r, obj_lp_r = lp_alternate(a_opt, max_iters=10, time_budget=min(remaining * 0.2, 20.0))
            if obj_lp_r < obj_opt:
                a_opt, obj_opt = a_lp_r, obj_lp_r

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    print(f"  Phase 3 final: {best_score:.8f}")

    # === Phase 4: Final intensive polish (10% of time) ===
    print(f"\n=== Phase 4: Final polish ===")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 15 and best_seq is not None:
        n_final = len(best_seq)
        a_final = best_seq.copy()

        # Cycle through strategies
        polish_cycles = min(20, int(remaining4 / 10))
        for pc in range(polish_cycles):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            strat = pc % 4

            if strat == 0 and n_final <= 1000:
                a_p, obj_p = kkt_iteration(a_final, max_iters=30, time_budget=min(rem * 0.4, 20.0))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Polish KKT {pc}: {best_score:.8f}")

            elif strat == 1:
                a_p, obj_p = fr_subgradient(a_final, max_iters=1000, time_budget=min(rem * 0.3, 15.0))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Polish FR {pc}: {best_score:.8f}")

            elif strat == 2 and n_final <= 1000:
                a_p, obj_p = lp_alternate(a_final, max_iters=10, time_budget=min(rem * 0.4, 20.0))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Polish LP {pc}: {best_score:.8f}")

            else:
                # Small perturbation + re-optimize
                a_pert = a_final * np.exp(rng.normal(0, 0.02, n_final))
                a_pert = project_simplex(a_pert)
                a_p, obj_p = fr_subgradient(a_pert, max_iters=500, time_budget=min(rem * 0.3, 15.0))
                if n_final <= 1000:
                    a_kk, obj_kk = kkt_iteration(a_p, max_iters=20, time_budget=min(rem * 0.2, 10.0))
                    if obj_kk < obj_p:
                        a_p, obj_p = a_kk, obj_kk
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Polish Pert {pc}: {best_score:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
        print(f"  Evaluation improved: {best_score:.8f}")
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
