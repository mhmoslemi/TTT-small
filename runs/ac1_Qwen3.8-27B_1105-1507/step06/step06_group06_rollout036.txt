"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating, subgradient with softmax peak weighting, spectral shaping,
and multi-length exploration. Key improvements: adaptive tau scheduling for
subgradient, golden-section line search for LP blending, and aggressive
multi-restart with diverse initializations.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def build_toeplitz_conv(a):
    """Build matrix T where (T@g)[k] = (conv(a,g))[k]. Shape (2n-1, n)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        T[k_start:k_end, j] = a[:k_end - k_start]
    return T


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns (g, M_opt)."""
    n = len(a)
    if n > 2500:
        return None, None
    T = build_toeplitz_conv(a)
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def golden_section_line_search(a, g, max_evals=20):
    """Find optimal blend parameter lambda in [0,1] minimizing obj((1-lam)*a + lam*g)."""
    a = project_nonneg_sum1(a)
    g = project_nonneg_sum1(g)
    # Check endpoints and midpoint
    candidates = [0.0, 0.25, 0.5, 0.75, 1.0]
    best_lam = 0.0
    best_obj = compute_obj(a)
    for lam in candidates:
        a_try = (1.0 - lam) * a + lam * g
        a_try = project_nonneg_sum1(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_lam = lam
    # Refine with golden section around best
    if 0 < best_lam < 1:
        gr = (np.sqrt(5) - 1) / 2
        lo = max(0.0, best_lam - 0.25)
        hi = min(1.0, best_lam + 0.25)
        for _ in range(max_evals - 5):
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            a1 = project_nonneg_sum1((1 - x1) * a + x1 * g)
            a2 = project_nonneg_sum1((1 - x2) * a + x2 * g)
            o1 = compute_obj(a1)
            o2 = compute_obj(a2)
            if o1 < o2:
                hi = x2
            else:
                lo = x1
            mid = (lo + hi) / 2
            a_mid = project_nonneg_sum1((1 - mid) * a + mid * g)
            o_mid = compute_obj(a_mid)
            if o_mid < best_obj:
                best_obj = o_mid
                best_lam = mid
            if hi - lo < 0.01:
                break
    a_best = project_nonneg_sum1((1 - best_lam) * a + best_lam * g)
    return a_best, best_obj


def lp_alternate(a, max_iters=100, time_budget=60.0, verbose=False):
    """LP alternating optimization with golden section line search."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        g, M_opt = solve_lp_minpeak(a, time_limit=10)
        if g is None:
            break
        a_new, obj_new = golden_section_line_search(a, g, max_evals=15)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

    return best_a, best_obj


def subgradient_optimize(a, max_iters=2000, time_budget=30.0,
                         tau_init=None, step_init=0.5, momentum=0.9,
                         verbose=False):
    """Subgradient optimization with softmax peak weighting and Nesterov momentum."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_init is None:
        conv = compute_conv(a)
        tau_init = float(np.max(conv)) * 0.05

    v = np.zeros(n)
    step = step_init
    indices = np.arange(n)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Adaptive tau: start small (focus on peak), grow over time
        tau = tau_init + 0.1 * it / max_iters * cmax

        # Softmax weights
        shifted = np.clip((conv - cmax) / tau, -300.0, 0.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient: grad[i] = 2 * sum_j weights[i+j] * a[j]
        # Build the weight matrix W[i,j] = weights[i+j]
        # Since i+j ranges from 0 to 2n-2 = len(weights)-1, all valid
        W = weights[indices[:, None] + indices[None, :]]
        grad = 2.0 * (W @ a)

        # Center gradient (since we're on the simplex, only the centered part matters)
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum step
        v = momentum * v - step * grad_centered
        a_new = a - v
        a_new = project_nonneg_sum1(a_new)

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Adaptive step size
        step *= 0.9995
        if step < 0.001:
            step = 0.001

    return best_a, best_obj


def spectral_shaping(a, max_iters=200, time_budget=20.0, compression=0.8):
    """Spectral shaping: flatten the periodogram to reduce autocorrelation peaks."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        ps_max = np.max(power)
        if ps_max < 1e-20:
            break

        # Adaptive compression: stronger early, weaker later
        compress = compression + 0.2 * (it / max_iters)
        ps_norm = power / ps_max
        ps_compressed = ps_norm ** compress

        # Keep DC bin fixed
        ps_compressed[0] = ps_norm[0]

        # Reconstruct with modified magnitude, keep phase
        phase = np.angle(fa)
        mag = np.sqrt(ps_compressed * ps_max)
        fa_new = mag * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def chase_peak(a, n_steps=100, step_size=0.05, verbose=False):
    """Chase-the-peak: iteratively reduce the maximum convolution peak."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient at the peak
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]

        grad_centered = grad - np.mean(grad)

        # Try multiple step sizes
        improved = False
        for ss in [step_size, step_size * 0.5, step_size * 0.25]:
            a_try = a - ss * grad_centered
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            # Try smaller steps
            for ss in [step_size * 0.1, step_size * 0.05]:
                a_try = a - ss * grad_centered
                a_try = project_nonneg_sum1(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break
            else:
                if verbose and it % 50 == 0:
                    print(f"    chase_peak stuck at iter {it}, obj={best_obj:.8f}")

    return best_a, best_obj


def simulated_annealing_guided(a, rng, max_iters=500, T_init=0.5, T_min=0.01,
                                time_budget=60.0):
    """SA with subgradient-guided moves."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    cur_a = a.copy()
    cur_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = cur_obj
    start = time.monotonic()
    indices = np.arange(n)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T_init * (T_min / T_init) ** (it / max_iters)
        if T < 1e-7:
            T = 1e-7

        a_trial = cur_a.copy()
        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Subgradient-guided perturbation
            conv = compute_conv(cur_a)
            k_star = int(np.argmax(conv))
            grad = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] = 2.0 * cur_a[j]
            grad_centered = grad - np.mean(grad)
            a_trial -= 0.05 * grad_centered
            noise = rng.normal(0, 0.02, n)
            a_trial += noise
            a_trial = project_nonneg_sum1(a_trial)

        elif move_type == 1:
            # Window perturbation
            wsize = max(5, n // 20)
            wstart = int(rng.integers(0, n - wsize))
            a_trial[wstart:wstart + wsize] *= np.exp(
                rng.normal(0, 0.5, wsize))
            a_trial = project_nonneg_sum1(a_trial)

        elif move_type == 2:
            # Random coordinate rescaling
            n_coords = max(10, n // 10)
            coords = rng.choice(n, size=n_coords, replace=False)
            a_trial[coords] *= np.exp(rng.normal(0, 0.5, n_coords))
            a_trial = project_nonneg_sum1(a_trial)

        else:
            # Swap-based move
            wsize = max(5, n // 30)
            w1 = int(rng.integers(0, n - wsize))
            w2 = int(rng.integers(0, n - wsize))
            if abs(w1 - w2) >= wsize:
                temp = a_trial[w1:w1 + wsize].copy()
                a_trial[w1:w1 + wsize] = a_trial[w2:w2 + wsize]
                a_trial[w2:w2 + wsize] = temp
            a_trial = project_nonneg_sum1(a_trial)

        trial_obj = compute_obj(a_trial)

        delta = trial_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_a = a_trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()

    return best_a, best_obj


def generate_diverse_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular
    tri = np.concatenate([np.arange(1, n // 2 + 1),
                          np.arange(n // 2, 0, -1)])[:n]
    inits.append(tri / tri.sum())

    # Sinusoidal
    x = np.linspace(0, 1, n)
    for freq in [1, 2, 3]:
        sine = np.sin(np.pi * freq * x) ** 2 + 0.05
        inits.append(sine / sine.sum())

    # Gaussian center
    x = np.linspace(-3, 3, n)
    gauss = np.exp(-x ** 2 / 2)
    inits.append(gauss / gauss.sum())

    # Bimodal
    bimodal = np.exp(-((x - 1.5) ** 2) / 0.5) + np.exp(-((x + 1.5) ** 2) / 0.5)
    inits.append(bimodal / bimodal.sum())

    # Random
    for _ in range(3):
        rnd = rng.random(n) + 0.01
        inits.append(rnd / rnd.sum())

    # Power law
    power_seq = np.array([1.0 / (i + 1) ** 0.5 for i in range(n)])
    inits.append(power_seq / power_seq.sum())

    # W-shaped
    w_shape = np.exp(-((x - 0.2) ** 2) / 0.02) + np.exp(-((x - 0.8) ** 2) / 0.02) + 0.01
    inits.append(w_shape / w_shape.sum())

    if parent is not None and len(parent) > 0:
        resampled = resample_sequence(parent, n)
        resampled = np.maximum(resampled, 0)
        s = resampled.sum()
        if s > 1e-10:
            inits.append(resampled / s)
            pert = resampled * np.exp(rng.normal(0, 0.1, n))
            pert = project_nonneg_sum1(pert)
            inits.append(pert)

    return inits


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        parent = parent[~np.isnan(parent) & ~np.isinf(parent)]
        if parent.size > 0:
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, subgradient with softmax
    peak weighting, spectral shaping, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"Default start: n={len(best_seq)}, obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}")

    # === Phase 1: Diverse initialization screening (8% time) ===
    print("\n=== Phase 1: Initialization Screening ===")
    phase1_budget = min(100.0, (deadline - start_time) * 0.08)
    n_options = list(dict.fromkeys([
        n_parent, max(100, n_parent - 100), max(100, n_parent - 50),
        n_parent + 50, n_parent + 100, min(2000, n_parent + 200),
        min(2000, n_parent + 400)
    ]))
    n_options = [n for n in n_options if 50 <= n <= 3000]

    candidates = []
    total_inits = len(n_options) * 4  # approximate
    per_init_time = max(2.0, phase1_budget / total_inits)

    for n_try in n_options:
        if time.monotonic() > start_time + phase1_budget:
            break
        inits = generate_diverse_initializations(n_try, rng, parent)
        for init in inits[:4]:  # Limit to 4 per length
            if time.monotonic() > start_time + phase1_budget:
                break
            # Quick optimize: chase peak + brief subgradient
            a_q, obj_q = chase_peak(init, n_steps=30, step_size=0.03)
            a_q, obj_q = subgradient_optimize(a_q, max_iters=100,
                                              time_budget=min(per_init_time * 0.5, 5.0),
                                              step_init=0.3)
            if np.isfinite(obj_q):
                candidates.append((obj_q, a_q, n_try))

    candidates.sort(key=lambda x: x[0])
    if candidates:
        print(f"  Screened {len(candidates)} candidates, best: {candidates[0][0]:.8f}")
        for obj_c, a_c, n_c in candidates[:3]:
            if obj_c < best_score:
                best_score = obj_c
                best_seq = a_c.copy()
                print(f"  New best from screening: n={n_c}, obj={best_score:.8f}")

    # === Phase 2: Intensive LP alternating (55% time) ===
    print(f"\n=== Phase 2: Intensive LP Alternating ===")
    print(f"Best before: {best_score:.8f}")
    phase2_budget = min(600.0, (deadline - start_time) * 0.55)
    phase2_start = time.monotonic()

    a_work = project_nonneg_sum1(best_seq.copy())
    local_best = a_work.copy()
    local_obj = best_score
    iter_count = 0
    stall_count = 0

    while time.monotonic() - phase2_start < phase2_budget:
        remaining = phase2_start + phase2_budget - time.monotonic()
        if remaining < 10:
            break

        n_work = len(a_work)

        # Choose strategy based on iteration and stall
        iter_count += 1
        strategy = iter_count % 7

        if strategy in [0, 1, 2, 3, 4] and n_work <= 2000:
            # LP alternating (primary)
            lp_time = min(remaining * 0.3, 30.0)
            a_new, obj_new = lp_alternate(a_work, max_iters=30, time_budget=lp_time)
        elif strategy == 5:
            # Subgradient with momentum
            sg_time = min(remaining * 0.2, 20.0)
            a_new, obj_new = subgradient_optimize(
                a_work, max_iters=500, time_budget=sg_time,
                tau_init=float(np.max(compute_conv(a_work))) * 0.03,
                step_init=0.3, momentum=0.92)
        else:
            # Spectral shaping
            sp_time = min(remaining * 0.2, 15.0)
            a_new, obj_new = spectral_shaping(a_work, max_iters=150,
                                              time_budget=sp_time, compression=0.75)

        if obj_new < local_obj - 1e-12:
            local_obj = obj_new
            local_best = a_new.copy()
            a_work = a_new.copy()
            stall_count = 0
            if local_obj < best_score:
                best_score = local_obj
                best_seq = local_best.copy()
                if iter_count % 5 == 0:
                    print(f"  LP it{iter_count}: obj={best_score:.8f} (strategy={strategy})")
        else:
            stall_count += 1
            if stall_count > 20:
                # Perturb and restart
                a_pert = a_work * np.exp(rng.normal(0, 0.05, n_work))
                a_pert = project_nonneg_sum1(a_pert)
                a_work = a_pert
                local_obj = compute_obj(a_work)
                stall_count = 0
                print(f"    Perturbed at it{iter_count}, obj={local_obj:.8f}")

    print(f"  Phase 2 final: obj={local_obj:.8f}")

    # === Phase 3: Multi-length exploration (15% time) ===
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    print(f"Best so far: {best_score:.8f}")
    phase3_budget = min(200.0, (deadline - time.monotonic()) * 0.15)
    phase3_start = time.monotonic()
    n_best = len(best_seq)

    n_tries = [n_best - 100, n_best - 50, n_best + 50, n_best + 100,
               n_best + 200, n_best - 200, n_best + 300]
    n_tries = [n for n in n_tries if 50 <= n <= 3000]
    n_tries = list(dict.fromkeys(n_tries))

    per_n_budget = max(5.0, phase3_budget / max(1, len(n_tries)))

    for n_try in n_tries:
        if time.monotonic() > phase3_start + phase3_budget:
            break

        a_resampled = resample_sequence(best_seq, n_try)
        a_resampled = project_nonneg_sum1(a_resampled)

        # Optimize at this length
        a_opt, obj_opt = lp_alternate(a_resampled, max_iters=20,
                                      time_budget=min(per_n_budget * 0.5, 20.0))
        a_opt, obj_opt = subgradient_optimize(a_opt, max_iters=200,
                                              time_budget=min(per_n_budget * 0.3, 10.0),
                                              step_init=0.3)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f} (no improvement)")

    # === Phase 4: SA + diversification (12% time) ===
    print(f"\n=== Phase 4: Simulated Annealing ===")
    print(f"Best so far: {best_score:.8f}")
    phase4_budget = min(150.0, (deadline - time.monotonic()) * 0.12)
    phase4_start = time.monotonic()

    a_sa, obj_sa = simulated_annealing_guided(best_seq, rng,
                                               max_iters=2000,
                                               T_init=0.3, T_min=0.005,
                                               time_budget=phase4_budget * 0.6)
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = a_sa.copy()
        print(f"  SA improved: {best_score:.8f}")

    # Try a few more SA restarts from perturbed best
    remaining4 = phase4_start + phase4_budget - time.monotonic()
    n_restarts = max(1, int(remaining4 / 15))
    for rs in range(min(n_restarts, 5)):
        if time.monotonic() > phase4_start + phase4_budget:
            break
        a_pert = best_seq * np.exp(rng.normal(0, 0.03, len(best_seq)))
        a_pert = project_nonneg_sum1(a_pert)
        a_sa2, obj_sa2 = simulated_annealing_guided(a_pert, rng,
                                                     max_iters=500,
                                                     T_init=0.2, T_min=0.003,
                                                     time_budget=12.0)
        if obj_sa2 < best_score:
            best_score = obj_sa2
            best_seq = a_sa2.copy()
            print(f"  SA restart {rs}: {best_score:.8f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()

    if remaining5 > 15 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        n_final = len(a_final)

        polish_rounds = min(20, int(remaining5 / 10))
        for pr in range(polish_rounds):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            # LP (if size allows)
            if n_final <= 2000:
                a_p, obj_p = lp_alternate(a_final, max_iters=20,
                                          time_budget=min(rem * 0.3, 25.0))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Polish LP r{pr}: {best_score:.8f}")

            # Subgradient with fine step
            remain = deadline - time.monotonic()
            if remain > 5:
                a_p2, obj_p2 = subgradient_optimize(
                    a_final, max_iters=500,
                    time_budget=min(remain * 0.25, 15.0),
                    tau_init=float(np.max(compute_conv(a_final))) * 0.02,
                    step_init=0.15, momentum=0.95)
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_final = a_p2.copy()
                    print(f"  Polish SG r{pr}: {best_score:.8f}")

            # Chase peak (very fine)
            remain = deadline - time.monotonic()
            if remain > 3:
                a_p3, obj_p3 = chase_peak(a_final, n_steps=50, step_size=0.01)
                if obj_p3 < best_score:
                    best_score = obj_p3
                    best_seq = a_p3.copy()
                    a_final = a_p3.copy()

            # Spectral
            remain = deadline - time.monotonic()
            if remain > 5:
                a_p4, obj_p4 = spectral_shaping(a_final, max_iters=100,
                                                 time_budget=min(remain * 0.2, 10.0),
                                                 compression=0.85)
                if obj_p4 < best_score:
                    best_score = obj_p4
                    best_seq = a_p4.copy()
                    a_final = a_p4.copy()
                    print(f"  Polish SP r{pr}: {best_score:.8f}")

            # Tiny perturbation + re-optimize
            remain = deadline - time.monotonic()
            if remain > 10:
                a_pert = a_final * np.exp(rng.normal(0, 0.01, n_final))
                a_pert = project_nonneg_sum1(a_pert)
                a_pert, obj_pert = lp_alternate(a_pert, max_iters=10, time_budget=5.0)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_final = a_pert.copy()
                    print(f"  Polish pert r{pr}: {best_score:.8f}")

    # Final validation using evaluate_sequence
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
