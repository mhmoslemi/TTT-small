import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_obj_and_grad(a, beta):
    """Compute log-sum-exp smoothed objective and gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    # Gradient via FFT
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj

def nesterov_log(a, beta_start, beta_end, lr, max_iters, time_budget):

    start = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)
    
    x = np.log(a)
    x -= np.mean(x)
    y = x.copy()  # Nesterov look-ahead
    v = np.zeros(n)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)

        # Nesterov step
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        y = x + ((it) / gamma) * (x - y + v) if it > 0 else x

        a_t = np.exp(y_c)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using
L-BFGS-B with adaptive log-sum-exp smoothing, Nesterov momentum in log-space,
LP-based peak reduction, peak shaving, multi-n exploration, and structured restarts.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_and_grad(a, beta):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    if e_sum < 1e-300:
        e_sum = 1e-300
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    # Gradient via FFT cross-correlation
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def lbfgs_smooth(a, beta, time_budget=30.0):
    """L-BFGS-B optimization of smoothed objective in log-space."""
    start = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)

    x0 = np.log(a)
    x0 -= np.mean(x0)

    def objective(x):
        if time.monotonic() - start > time_budget:
            return 1e10, np.zeros(len(x))
        x_c = x - np.max(x)
        a_t = np.exp(x_c)
        a_t = a_t / np.sum(a_t)
        obj, grad_a = smooth_obj_and_grad(a_t, beta)
        grad_x = grad_a * a_t
        return obj, grad_x

    try:
        res = minimize(objective, x0, method='L-BFGS-B',
                       options={'maxiter': 10000, 'maxfun': 10000,
                                'ftol': 1e-14, 'gtol': 1e-10,
                                'maxls': 100, 'iprint': -1})
        x_opt = res.x
        x_c = x_opt - np.max(x_opt)
        a_opt = np.exp(x_c)
        a_opt = a_opt / np.sum(a_opt)
        return a_opt
    except Exception:
        return a


def momentum_log(a, beta_start, beta_end, lr, max_iters, time_budget):
    """Momentum-based optimization in log-space with annealing."""
    start = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)

    x -= np.mean(x)
    v = np.zeros(n)
    mu = 0.9

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        x_c = x - np.max(x)
        a_t = np.exp(x_c)
        a_t = a_t / np.sum(a_t)

        _, grad_a = smooth_obj_and_grad(a_t, beta)

        v = mu * v - lr * grad_x
        x = x + v

            true_obj = compute_obj(a_check)

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A

def peak_shave(a, time_budget=20.0):

    n = len(a)

        i_min = max(0, peak_pos - (n - 1))
        i_max = min(peak_pos, n - 1)

        best_improvement = 0
        best_move = None

            j = peak_pos - i
            if j < 0 or j >= n or j == i:
                continue
            for delta_frac in [0.01, 0.02, 0.05, 0.1]:
                # Move mass from the larger of a[i], a[j] to reduce their product
                if best_a[i] > best_a[j]:
                    delta = delta_frac * best_a[i]
                    new_i = best_a[i] - delta
                    new_j = best_a[j] + delta * best_a[j] / max(best_a[i], 1e-15)
                    # Simpler: just reduce the larger one and increase a random other

                    if new_i < 0:
                        continue
                else:
                    delta = delta_frac * best_a[j]

                    if new_j < 0:

                a_try = a_try / s_try

def coordinate_refine(a, time_budget=20.0, seed=0):

    rng = np.random.default_rng(seed)

        i = rng.integers(0, n)
        j = rng.integers(0, n)

            continue

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

    n_options = [n for n in n_options if 100 <= n <= 2500]

    for n_try in n_options:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        if n_try == n_best:
            a_try = a.copy()
        else:
            a_try = resample_sequence(a, n_try)
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / np.sum(a_try)

        # L-BFGS-B on resampled
        alloc = min(remaining * 0.08, 20.0)
        a_opt = lbfgs_smooth(a_try, 50, time_budget=alloc)
        # Follow with higher beta
        if time.monotonic() < deadline - 15:
            a_opt = lbfgs_smooth(a_opt, 500, time_budget=min(remaining * 0.04, 12.0))

        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try
            a = a_opt.copy()
            print(f"  n={n_try:>5}: {best_score:.10f} *** UPDATE")
        else:
            print(f"  n={n_try:>5}: {obj:.10f}")

    # === Phase 6: Structured random restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Structured restarts")
    print(f"{'='*60}")

    n_best = len(best_seq)
    rng = np.random.default_rng(seed)

    for restart in range(12):
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break

        strategy = restart % 5
        if strategy == 0:
            # Sinusoidal perturbation
            num_sines = rng.integers(1, 4)
            pert = np.zeros(n_best)
            for _ in range(num_sines):
                freq = rng.uniform(1, 10)
                phase = rng.uniform(0, 2 * np.pi)
                amp = rng.uniform(0.05, 0.2)
                pert += amp * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
            a_pert = a * (1 + pert)
        elif strategy == 1:
            # Multiplicative noise (Gaussian in log-space)
            a_pert = a * np.exp(rng.normal(0, 0.08 + 0.02 * restart, n_best))
        elif strategy == 2:
            # Blend with periodic comb
            period = rng.integers(max(2, n_best // 20), max(3, n_best // 5))
            comb = np.ones(n_best) * 0.5
            for i in range(0, n_best, period):
                comb[i] = 1.0
            comb = np.maximum(comb, 0.1)
            comb = comb / np.sum(comb)
            alpha = rng.uniform(0.1, 0.4)
            a_pert = (1 - alpha) * a + alpha * comb
        elif strategy == 3:
            # Random spikes at random positions
            a_pert = a.copy()
            num_spikes = max(2, n_best // 50)
            for _ in range(num_spikes):
                idx = rng.integers(0, n_best)
                a_pert[idx] += rng.uniform(0.05, 0.3) * np.mean(a)
        else:
            # Blend with triangular window
            tri = 1.0 - np.abs(np.arange(n_best) - n_best / 2) / (n_best / 2)
            tri = np.maximum(tri, 0.01)
            tri = tri / np.sum(tri)
            alpha = rng.uniform(0.1, 0.3)
            a_pert = (1 - alpha) * a + alpha * tri

        a_pert = np.maximum(a_pert, 1e-12)
        a_pert = a_pert / np.sum(a_pert)

        # Optimize the perturbed sequence
        alloc1 = min(remaining * 0.07, 15.0)
        a_opt = lbfgs_smooth(a_pert, 20, time_budget=alloc1)
        if time.monotonic() < deadline - 10:
            a_opt = lbfgs_smooth(a_opt, 500, time_budget=min(remaining * 0.04, 10.0))

        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  Restart {restart:>2} (strat {strategy}): {best_score:.10f} *** UPDATE")
        else:
            print(f"  Restart {restart:>2} (strat {strategy}): {obj:.10f}")

    # === Phase 7: Final LP + L-BFGS-B cycle ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final LP + LBFGS cycle")
    print(f"{'='*60}")

    n_best = len(best_seq)
    if n_best <= 1000:
        for r in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            alloc = min(remaining * 0.15, 20.0)
            a_lp = lp_refine(best_seq, time_budget=alloc)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  Final LP {r}: {best_score:.10f} ***")
            else:
                print(f"  Final LP {r}: {obj:.10f}")

    # === Phase 8: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"Phase 8: Coordinate descent polish")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        a_cd = coordinate_refine(a, time_budget=min(remaining * 0.3, 25.0), seed=seed + 1)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: {best_score:.10f} ***")
        else:
            print(f"  CD: {obj_cd:.10f}")

    # === Phase 9: Final high-beta L-BFGS-B ===
    if time.monotonic() < deadline - 3:
        remaining = deadline - time.monotonic()
        a_final = lbfgs_smooth(best_seq, 5000, time_budget=max(3.0, remaining * 0.6))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LBFGS: {best_score:.10f} ***")
        else:
            print(f"  Final LBFGS: {obj_final:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
