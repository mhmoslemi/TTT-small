def compute_conv(a):

    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):

    scale = s_p ** (1.0 / p - 1.0)
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (scale * 2.0 * cc[:n] / w_sum * w_sum - norm_p / sum_a)
    # Simplify: grad_obj[i] = 2n/(sum_a^2) * (s_p^(1/p-1) * 2 * sum_k conv[k]^(p-1) * a[k-i] - norm_p/sum_a)
    # The first term: s_p^(1/p-1) * 2 * (w_unnorm cross-correlate a)[i]
    # where w_unnorm[k] = conv[k]^(p-1)
    # = s_p^(1/p-1) * 2 * w_sum * (w cross-correlate a)[i]
    
    # Let me recompute more carefully
    # norm_p = (sum conv^p)^(1/p)
    # d norm_p / d a[i] = (1/p) (sum conv^p)^(1/p-1) * p * sum_k conv[k]^(p-1) * d conv[k]/d a[i]
    # = (sum conv^p)^(1/p-1) * sum_k conv[k]^(p-1) * 2*a[k-i] (approximately)
    # = s_p^(1/p-1) * 2 * (w_unnorm ⋆ a)[i]
    # where w_unnorm[k] = conv[k]^(p-1)
    
    w_unnorm = conv_pos ** (p - 1)
    w_unnorm_sum = float(np.sum(w_unnorm))
    if w_unnorm_sum > 0:
        w_norm = w_unnorm / w_unnorm_sum
    else:
        w_norm = np.ones_like(conv) / len(conv)
    
    W2 = rfft(w_norm, size)
    A2 = rfft(a, size)
    cc2 = irfft(np.conj(W2) * A2, size)
    
    grad_norm = s_p ** (1.0 / p - 1.0) * 2.0 * w_unnorm_sum * cc2[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_norm - norm_p / sum_a)

    return obj, grad_obj


def adam_log_opt(a, max_iters, time_budget, beta_start, beta_end, lr_start, lr_decay=None,
                 lr_cosine=False, reg_strength=0.0):
    """
    Adam in log-space with temperature annealing.
    Optionally uses cosine learning rate schedule and L2 regularization on log(a).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = min(it / max(max_iters - 1, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        if lr_cosine:
            lr = lr_start * 0.5 * (1 + np.cos(np.pi * progress))
        else:
            lr = lr_start
            if lr_decay is not None:
                lr = lr_start * (lr_decay ** it)

        x_centered = x - np.max(x)
        a_curr = np.exp(x_centered)
        a_curr = a_curr / np.sum(a_curr)

        obj_sm, grad_a = ls_obj_grad(a_curr, beta)

        if reg_strength > 0:
            grad_a = grad_a + reg_strength * (a_curr - 1.0 / n)

        grad_x = grad_a * a_curr

        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 100 == 0:
            x -= np.mean(x)

        if it % 50 == 0 or it == max_iters - 1:
            xc = x - np.max(x)
            a_check = np.exp(xc)
            s_check = np.sum(a_check)
            if s_check > 1e-10:
                a_check = a_check / s_check
                true_obj = compute_obj(a_check)
                if true_obj < best_obj:
                    best_obj = true_obj
                    best_a = a_check.copy()

    return best_a


def subgradient_momentum(a, time_budget, lr=0.001, momentum=0.9, top_k=5,
                         step_decay=0.9999):
    """
    Subgradient method with heavy momentum on the probability simplex.
    Randomly samples from top-k active constraints (peaks of convolution).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    velocity = np.zeros(n)
    lr_curr = lr
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_val = float(np.max(conv))

        # Get top-k peaks
        top_indices = np.argsort(conv)[::-1][:top_k]
        # Randomly pick from top-k
        k_star = top_indices[rng.integers(0, top_k)]

        # Subgradient of (a*a)[k_star] w.r.t. a
        # d/d(a[i]) (a*a)[k_star] = 2*a[k_star - i] (if 0 <= k_star-i < n)
        # (or a[i] if 2*i == k_star)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j] if j != i else a[i]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm

        # Momentum update
        velocity = momentum * velocity - lr_curr * grad
        a_new = a + velocity

        # Project onto probability simplex
        a_new = _project_simplex(a_new)

        # Check objective
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Don't update a if worsened (line search or just reject)
            # Try half step
            a_half = _project_simplex(a + 0.5 * velocity)
            obj_half = compute_obj(a_half)
            if obj_half < best_obj:
                best_obj = obj_half
                best_a = a_half.copy()
                a = a_half
                velocity *= 0.5

        lr_curr *= step_decay
        it += 1

    return best_a


def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def pnorm_optimize(a, time_budget, p_start=4, p_end=64, lr_start=0.005, n_restarts=2):
    """
    Optimize using p-norm approximation of max, with increasing p.
    Uses projected gradient on the simplex.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77)

    p_values = np.geomspace(p_start, p_end, n_restarts)

    for p in p_values:
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        lr = lr_start
        a_curr = best_a.copy()
        momentum = np.zeros(n)
        mu = 0.9

        while time.monotonic() - start < time_budget - 5:
            obj_p, grad = pnorm_obj_grad(a_curr, p)

            # Projected gradient with momentum
            momentum = mu * momentum + grad
            a_new = a_curr - lr * momentum
            a_new = _project_simplex(a_new)

            true_obj = compute_obj(a_new)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_new.copy()
                a_curr = a_new
            else:
                lr *= 0.9
                momentum = 0.5 * momentum

            if lr < 1e-8:
                break

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for (a*g)[k] = sum_j a[k-j]*g[j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        length = k_end - k_start
        A[k_start:k_end, j] = a[:length]
    return A


def solve_lp(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, time_budget=30.0):
    """LP-based refinement."""
    n = len(a)
    if n > 600:
        return a

    start = time.monotonic()
    s = np.sum(a)
    a_norm = a / s
    conv = compute_conv(a_norm)
    M = float(np.max(conv))

    g = solve_lp(a_norm, M, time_limit=min(time_budget * 0.5, 12))
    if g is None:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)

    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        if time.monotonic() - start > time_budget:
            break
        a_try = (1 - t) * a_norm + t * g
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def coordinate_refine(a, time_budget=20.0, seed=0):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy().astype(np.float64)
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(seed)

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        # Focus on top peaks
        n_peaks = min(3, len(conv))
        peaks = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for p in peaks:
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            # Randomly pick a coordinate to adjust
            candidates = list(range(i_min, i_max + 1))
            if not candidates:
                continue
            i = candidates[rng.integers(0, len(candidates))]
            j = p - i
            if j < 0 or j >= n or j == i:
                continue

            for delta_frac in [0.002, 0.005, 0.01, 0.02, -0.005, -0.01]:
                delta = delta_frac * max(best_a[i], best_a[j], 1.0 / n)
                a_try = best_a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break

        if not improved and it > 100:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def make_init(n, shape):
    """Create initial sequences of various shapes."""
    x = np.arange(n) / float(n)
    if shape == 'uniform':
        a = np.ones(n)
    elif shape == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif shape == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 1e-10)
    elif shape == 'triangular':
        a = 1.0 - 2.0 * np.abs(x - 0.5)
        a = np.maximum(a, 1e-10)
    elif shape == 'bimodal':
        a = np.exp(-8.0 * (x - 0.25) ** 2) + np.exp(-8.0 * (x - 0.75) ** 2)
    elif shape == 'linear_ramp':
        a = np.linspace(0.1, 1.0, n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    return a


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and np.sum(parent) > 0.01:
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_at_length(a_init, n, time_budget, seed=42):
    """
    Run multiple optimization strategies at a given length.
    Returns the best sequence found.
    """
    start = time.monotonic()
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    print(f"    Initial (n={n}): obj={best_obj:.8f}")

    strategies = [
        ('adam_v1', 0.25),
        ('subgrad', 0.20),
        ('pnorm', 0.20),
        ('adam_v2', 0.15),
        ('adam_v3', 0.20),
    ]

    for name, frac in strategies:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        alloc = remaining * frac

        if name == 'adam_v1':
            a_new = adam_log_opt(best_a, max_iters=50000, time_budget=alloc,
                                 beta_start=5.0, beta_end=100.0,
                                 lr_start=0.04, lr_cosine=True)
        elif name == 'subgrad':
            a_new = subgradient_momentum(best_a, time_budget=alloc,
                                         lr=0.0005, momentum=0.92, top_k=3)
        elif name == 'pnorm':
            a_new = pnorm_optimize(best_a, time_budget=alloc,
                                   p_start=4, p_end=32, lr_start=0.003)
        elif name == 'adam_v2':
            a_new = adam_log_opt(best_a, max_iters=50000, time_budget=alloc,
                                 beta_start=20.0, beta_end=300.0,
                                 lr_start=0.02, lr_cosine=True, reg_strength=0.01)
        elif name == 'adam_v3':
            a_new = adam_log_opt(a_new, max_iters=50000, time_budget=alloc,
                                 beta_start=50.0, beta_end=800.0,
                                 lr_start=0.01, lr_cosine=True)
        else:
            a_new = best_a

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"    {name}: {best_obj:.8f} ***")
        else:
            print(f"    {name}: {obj_new:.8f} (no imp)")

        # If subgrad or pnorm improved, follow up with Adam
        if obj_new < best_obj + 0.001 and name in ('subgrad', 'pnorm'):
            remaining = time_budget - (time.monotonic() - start)
            if remaining > 5:
                a_follow = adam_log_opt(a_new, max_iters=50000,
                                        time_budget=min(remaining * 0.3, 15.0),
                                        beta_start=30.0, beta_end=400.0,
                                        lr_start=0.015, lr_cosine=True)
                obj_follow = compute_obj(a_follow)
                if obj_follow < best_obj:
                    best_obj = obj_follow
                    best_a = a_follow.copy()
                    print(f"    {name}+adam: {best_obj:.8f} ***")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase, multi-length, multi-strategy optimization.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"=== Starting search ===")
    print(f"Best initial: {best_score:.8f} (n={n_parent}), budget={budget_s:.0f}s")

    # === Phase 1: Quick multi-length scan ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Quick multi-length scan")
    print(f"{'='*60}")

    candidate_lengths = [n_parent, n_parent + 50, n_parent + 100,
                         n_parent + 200, n_parent + 300, n_parent + 500]
    candidate_lengths = sorted(set([n for n in candidate_lengths if 200 <= n <= 2000]))
    # Ensure we include some diversity
    candidate_lengths = sorted(set(candidate_lengths + [500, 800, 1000, 1200, 1500]))
    candidate_lengths = [n for n in candidate_lengths if 300 <= n <= 2000]

    results_by_length = {}

    for n_try in candidate_lengths:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Allocate time proportional to preference (more for lengths near parent)
        dist = abs(n_try - n_parent)
        alloc = min(remaining / (len(candidate_lengths) + 1), 40.0)
        if n_try == n_parent:
            alloc = min(alloc * 1.5, 50.0)

        # Create initial sequence
        if best_seq is not None and len(best_seq) == n_try:
            a_init = best_seq.copy()
        elif best_seq is not None:
            a_init = resample_sequence(best_seq, n_try)
        else:
            a_init = make_init(n_try, 'uniform')

        s_init = np.sum(a_init)
        if s_init > 0:
            a_init = a_init / s_init

        obj_init = compute_obj(a_init)
        print(f"  n={n_try}: init_obj={obj_init:.8f}, alloc={alloc:.1f}s")

        a_opt, obj_opt = optimize_at_length(a_init, n_try, alloc, seed=seed)

        results_by_length[n_try] = (a_opt, obj_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: BEST UPDATE -> {best_score:.8f}")

    # === Phase 2: Deep optimization on best length ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Deep optimization (n={len(best_seq) if best_seq is not None else '?'})")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()

        # Multiple rounds of deep optimization
        deep_rounds = [
            (10.0, 200.0, 0.03, True, 0.0),
            (5.0, 80.0, 0.04, True, 0.0),
            (30.0, 500.0, 0.02, True, 0.005),
            (50.0, 1000.0, 0.015, True, 0.0),
            (100.0, 2000.0, 0.01, True, 0.0),
        ]

        for r_idx, (b_s, b_e, lr, cosine, reg) in enumerate(deep_rounds):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.15, 60.0)

            a_new = adam_log_opt(best_seq, max_iters=100000, time_budget=alloc,
                                 beta_start=b_s, beta_end=b_e,
                                 lr_start=lr, lr_cosine=cosine,
                                 reg_strength=reg)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Round {r_idx}: {best_score:.8f} ***")
            else:
                print(f"  Round {r_idx}: {obj_new:.8f}")

        # Subgradient on best
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.15, 40.0)
            a_sg = subgradient_momentum(best_seq, time_budget=alloc,
                                        lr=0.0003, momentum=0.95, top_k=5)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Subgradient: {best_score:.8f} ***")
            else:
                print(f"  Subgradient: {obj_sg:.8f}")

        # p-norm on best
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.1, 30.0)
            a_pn = pnorm_optimize(best_seq, time_budget=alloc,
                                  p_start=8, p_end=64, lr_start=0.002)
            obj_pn = compute_obj(a_pn)
            if obj_pn < best_score:
                best_score = obj_pn
                best_seq = a_pn.copy()
                print(f"  p-norm: {best_score:.8f} ***")
            else:
                print(f"  p-norm: {obj_pn:.8f}")

    # === Phase 3: LP refinement (if n is small enough) ===
    print(f"\n{'='*60}")
    print(f"Phase 3: LP refinement")
    print(f"{'='*60}")

    if best_seq is not None and len(best_seq) <= 500:
        for r in range(3):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.15, 30.0)
            a_lp = lp_refine(best_seq, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP {r}: {best_score:.8f} ***")
            else:
                print(f"  LP {r}: {obj_lp:.8f}")

            # Follow up with Adam
            if time.monotonic() < deadline - 20:
                a_after = adam_log_opt(a_lp, max_iters=50000,
                                       time_budget=min(remaining * 0.08, 15.0),
                                       beta_start=50.0, beta_end=500.0,
                                       lr_start=0.01, lr_cosine=True)
                obj_after = compute_obj(a_after)
                if obj_after < best_score:
                    best_score = obj_after
                    best_seq = a_after.copy()
                    print(f"  LP+Adam {r}: {best_score:.8f} ***")
    else:
        print(f"  Skipped (n={len(best_seq) if best_seq is not None else '?'} > 500)")

    # === Phase 4: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Coordinate descent")
    print(f"{'='*60}")

    if best_seq is not None:
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.1, 25.0)
            a_cd = coordinate_refine(best_seq, time_budget=alloc, seed=seed)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Coord: {best_score:.8f} ***")
            else:
                print(f"  Coord: {obj_cd:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Diverse restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 1)
        shapes = ['gaussian', 'raised_cosine', 'bimodal', 'triangular', 'linear_ramp']

        for r in range(6):
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # Create diverse perturbation
            if r < len(shapes):
                a_struct = make_init(n_best, shapes[r])
                alpha = 0.1 + 0.05 * r
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * a_struct
            else:
                # Multiplicative noise
                scale = 0.05 + 0.02 * (r - len(shapes))
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))

            a_pert = np.maximum(a_pert, 1e-15)
            a_pert = a_pert / np.sum(a_pert)

            alloc = min(remaining * 0.12, 20.0)
            a_opt = adam_log_opt(a_pert, max_iters=50000, time_budget=alloc,
                                 beta_start=10.0, beta_end=300.0,
                                 lr_start=0.03, lr_cosine=True)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Restart {r} ({shapes[r] if r < len(shapes) else 'noise'}): {best_score:.8f} ***")
            else:
                print(f"  Restart {r}: {obj_opt:.8f}")

    # === Phase 6: Try different lengths around best ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Length fine-tuning")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        for n_try in [n_best - 20, n_best + 20, n_best + 50, n_best + 100]:
            if time.monotonic() > deadline - 15:
                break
            if n_try < 200 or n_try > 2000:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            a_try = resample_sequence(best_seq, n_try)
            a_try = a_try / np.sum(a_try)
            alloc = min(remaining * 0.15, 20.0)

            a_opt = adam_log_opt(a_try, max_iters=50000, time_budget=alloc,
                                 beta_start=20.0, beta_end=400.0,
                                 lr_start=0.02, lr_cosine=True)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} ***")
            else:
                print(f"  n={n_try}: {obj_opt:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            # Very high beta Adam for precise peak reduction
            alloc = min(remaining * 0.4, 30.0)
            a_final = adam_log_opt(best_seq, max_iters=100000, time_budget=alloc,
                                   beta_start=200.0, beta_end=2000.0,
                                   lr_start=0.005, lr_cosine=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final high-beta: {best_score:.8f} ***")

            # One more subgradient
            remaining = deadline - time.monotonic()
            if remaining > 10:
                alloc = min(remaining * 0.3, 15.0)
                a_sg2 = subgradient_momentum(best_seq, time_budget=alloc,
                                             lr=0.0002, momentum=0.97, top_k=3)
                obj_sg2 = compute_obj(a_sg2)
                if obj_sg2 < best_score:
                    best_score = obj_sg2
                    best_seq = a_sg2.copy()
                    print(f"  Final subgrad: {best_score:.8f} ***")

            # Final coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 5:
                alloc = min(remaining * 0.3, 10.0)
                a_cd2 = coordinate_refine(best_seq, time_budget=alloc, seed=seed + 99)
                obj_cd2 = compute_obj(a_cd2)
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  Final coord: {best_score:.8f} ***")

    # === Validation and return ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
