import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a, b=None):
    """FFT-based linear convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_J(a):
    """Compute J(a)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

import numpy as np

def fast_conv(a, b=None):
    """FFT-based linear convolution of a and b (or a with itself)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix_sparse(a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<len(a), else 0.
    Shape: (2n-1, n)."""
    n = len(a)
    conv_len = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(conv_len):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
    return A

                             'primal_feasibility_tolerance': 1e-9})

            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, r
    except Exception:
        pass
    return None, None

        if not sweep_improved:
            no_improve_sweeps += 2
        if no_improve_sweeps > 15:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-10)
        inits.append(p / p.sum())
        p2 = np.maximum(p + rng.normal(0, 0.05, n), 1e-6)
        inits.append(p2 / p2.sum())
        p3 = np.maximum(p + rng.normal(0, 0.15, n), 1e-6)
        inits.append(p3 / p3.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Parabolic (concave)
    poly = (4.0 * t * (1.0 - t)) + 0.1
    inits.append(poly / poly.sum())

    # U-shaped (convex)
    poly2 = ((2.0 * t - 1.0) ** 2) + 0.1
    inits.append(poly2 / poly2.sum())

    # Cosine windows
    for freq in [1, 2, 3, 5]:
        cos_w = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())

    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    # Smooth random
    ksize = max(3, n // 15)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal step function sequence.
    Phases: LP mirror descent -> spectral flattening -> coordinate descent
            -> second LP pass -> length exploration -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    # Load parent
    best_a = None
    best_score = float('inf')
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                best_a = parent / parent.sum()
                best_score = compute_J(best_a)
                print(f"Parent: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if best_a is None:
        best_a = np.ones(768) / 768
        best_score = compute_J(best_a)
        print(f"No parent, using uniform n=768, J={best_score:.8f}", flush=True)

    def update_best(candidate):
        nonlocal best_a, best_score
        c = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
        s = c.sum()
        if s < 0.01:
            return
        c = c / s
        score = compute_J(c)
        if score < best_score:
            best_score = score
            best_a = c.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={c.size} t={elapsed:.0f}s", flush=True)

    n_parent = best_a.size
    print(f"Starting main optimization on n={n_parent}", flush=True)

    # === Phase 1: LP Mirror Descent (main workhorse) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent (full constraints)", flush=True)
    print(f"{'='*60}", flush=True)

    deadline_p1 = min(deadline, t_start + 500.0)

    # Run from parent
    a_lp = run_lp_mirror_descent(best_a, deadline_p1, max_iters=50000, perturbation_interval=150)
    update_best(a_lp)
    print(f"  Phase 1 (parent init): J={best_score:.8f}", flush=True)

    # Run from a few diverse inits
    remaining = deadline - time.monotonic() - 250.0
    if remaining > 60:
        n_try = n_parent
        inits = generate_initializations(n_try, rng, best_a)
        # Skip the first (parent-based) inits, use diverse ones
        for init_i, a0 in enumerate(inits[3:6]):  # Skip first 3 (parent-based)
            if time.monotonic() > deadline - 250:
                break
            sub_dl = min(deadline - 250, time.monotonic() + max(30, remaining / 4.0))
            a_res = run_lp_mirror_descent(a0, sub_dl, max_iters=20000, perturbation_interval=100)
            update_best(a_res)
            print(f"  Phase 1 (init {init_i}): J={best_score:.8f}", flush=True)
            remaining = deadline - time.monotonic() - 250.0
            if remaining < 30:
                break

    # === Phase 2: Spectral Flattening ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Spectral Flattening", flush=True)
    print(f"{'='*60}", flush=True)

    deadline_p2 = min(deadline - 200.0, time.monotonic() + 150.0)
    if time.monotonic() < deadline_p2:
        a_spec = spectral_flatten(best_a, deadline_p2, max_iters=5000)
        update_best(a_spec)
        print(f"  Phase 2: J={best_score:.8f}", flush=True)

    # === Phase 3: Coordinate Descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Coordinate Descent", flush=True)
    print(f"{'='*60}", flush=True)

    deadline_p3 = min(deadline - 150.0, time.monotonic() + 120.0)
    if time.monotonic() < deadline_p3:
        a_cd = coordinate_descent_optimize(best_a, deadline_p3, max_sweeps=200)
        update_best(a_cd)
        print(f"  Phase 3: J={best_score:.8f}", flush=True)

    # === Phase 4: Second LP pass (refined) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Second LP Pass", flush=True)
    print(f"{'='*60}", flush=True)

    deadline_p4 = min(deadline - 100.0, time.monotonic() + 200.0)
    if time.monotonic() < deadline_p4:
        a_lp2 = run_lp_mirror_descent(best_a, deadline_p4, max_iters=30000, perturbation_interval=100)
        update_best(a_lp2)
        print(f"  Phase 4: J={best_score:.8f}", flush=True)

    # === Phase 5: Length Exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    deadline_p5 = min(deadline - 50.0, time.monotonic() + 200.0)
    if time.monotonic() < deadline_p5 and best_a is not None:
        n_cur = best_a.size
        for delta in [64, -64, 128, -128, 192, -192]:
            if time.monotonic() > deadline_p5:
                break
            n_new = n_cur + delta
            if n_new < 256 or n_new > 2048:
                continue
            n_new = (n_new // 32) * 32  # Round to multiple of 32

            a_res = resample_seq(best_a, n_new)
            a_res = np.maximum(a_res, 1e-8)
            a_res = a_res / a_res.sum()

            sub_dl = min(deadline_p5, time.monotonic() + 60.0)
            if time.monotonic() >= sub_dl:
                continue

            # Quick LP + spectral
            a_opt = run_lp_mirror_descent(a_res, sub_dl, max_iters=10000, perturbation_interval=80)
            update_best(a_opt)

            if time.monotonic() < deadline_p5 - 10:
                a_sp = spectral_flatten(a_opt, min(deadline_p5 - 10, time.monotonic() + 30), max_iters=2000)
                update_best(a_sp)

            print(f"  n={n_new}: J={best_score:.8f}", flush=True)

    # === Phase 6: Final Polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    deadline_p6 = min(deadline, time.monotonic() + 100.0)
    if time.monotonic() < deadline_p6 and best_a is not None:
        # Coordinate descent
        sub_dl = min(deadline_p6, time.monotonic() + 50.0)
        a_cd2 = coordinate_descent_optimize(best_a, sub_dl, max_sweeps=100)
        update_best(a_cd2)

        # Final LP
        if time.monotonic() < deadline:
            a_lp3 = run_lp_mirror_descent(best_a, deadline, max_iters=10000, perturbation_interval=50)
            update_best(a_lp3)

    # Final report
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} elapsed={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    # Ensure valid return
    if best_a is None:
        best_a = np.ones(512)

    return [float(max(0.0, x)) for x in best_a]
