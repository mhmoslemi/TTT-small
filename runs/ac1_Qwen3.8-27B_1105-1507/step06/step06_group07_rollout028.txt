def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_obj_and_grad(a, beta):
    """
    Compute smoothed objective (2n * logsumexp(beta * conv)) and gradient w.r.t. a.
    Assumes sum(a) = 1.
    Returns (obj, grad_a) where grad_a has length n.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Logsumexp of beta * conv
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum  # softmax weights over conv positions
    
    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls  # since sum(a) = 1
    
    # Gradient: grad_a[i] = 2 * sum_k w[k] * a[k-i] (cross-correlation)
    size = 1
    while size < 3 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_a = 2.0 * cc[:n]
    
    return obj, grad_a

def log_space_fun_and_grad(x, beta):
    """
    Objective and gradient in log-space.
    x = log(a) (with a normalized to sum=1)
    Returns (F(x), dF/dx)
    """
    n = len(x)
    # Compute a from x (softmax normalization)
    xc = x - np.max(x)
    u = np.exp(xc)
    a = u / np.sum(u)
    
    # Compute smoothed objective and gradient w.r.t. a
    obj, grad_a = smooth_obj_and_grad(a, beta)

    c = float(np.dot(a, grad_a))
    grad_projected = grad_a - c  # this is the gradient on the simplex
    
    # Chain rule: dF/dx[j] = a[j] * grad_projected[j]
    grad_x = a * grad_projected
    
    return obj, grad_x

    """Build matrix A such that (A @ g)[k] = conv(a,g)[k]"""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for l in range(n):
            i = k - l
            if 0 <= i < n:

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B in log-space with continuation (beta annealing), LP-based
direction finding with line search, multi-n exploration, diverse random
restarts, and peak-directed coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def logsumexp_conv(a, beta):
    """Returns (L, w, conv) where L = logsumexp(beta*conv(a,a)), w = softmax(beta*conv)."""
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    L = cmax + np.log(e_sum) / beta
    return L, w, conv


def logspace_obj_grad(x, beta):
    """
    Objective and gradient in log-space.
    x = log(a) (a will be normalized to sum=1 via softmax).
    Returns (F(x), dF/dx) where F = 2n * logsumexp(beta * conv(a,a)).
    """
    n = len(x)
    xc = x - np.max(x)
    u = np.exp(xc)
    u_sum = float(np.sum(u))
    a = u / u_sum

    L, w, conv = logsumexp_conv(a, beta)
    obj = 2.0 * n * L

    # dL/da[i] = 2 * sum_k w[k] * a[k-i] (cross-correlation)
    size = 1
    while size < 3 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    dL_da = 2.0 * cc[:n]

    c = float(np.dot(a, dL_da))
    grad_x = 2.0 * n * a * (dL_da - c)

    return obj, grad_x


def lbfgsb_log_opt(a_init, beta, max_iters=300, time_budget=30.0):
    """Single L-BFGS-B run in log-space with fixed beta."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = np.maximum(a, 1e-15)
    x = np.log(a)
    x -= np.mean(x)

    start = time.monotonic()
    state = {'done': False}

    def fun(xk):
        if state['done']:
            return float('inf')
        return logspace_obj_grad(xk, beta)[0]

    def jac(xk):
        if state['done']:
            return np.zeros_like(xk)
        return logspace_obj_grad(xk, beta)[1]

    def callback(xk):
        if time.monotonic() - start > time_budget:
            state['done'] = True
        return state['done']

    try:
        result = minimize(fun, x, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iters, 'gtol': 1e-8,
                                   'ftol': 1e-14, 'maxfun': 5000},
                          callback=callback)
        x = result.x
    except Exception:
        pass

    xc = x - np.max(x)
    a_out = np.exp(xc)
    a_out = a_out / np.sum(a_out)
    return a_out


def continuation_opt(a_init, beta_schedule, time_budget_each, time_total):
    """
    Continuation method: run L-BFGS-B with increasing beta.
    Returns best sequence found.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for beta in beta_schedule:
        if time.monotonic() - start > time_total - 5:
            break
        remaining = time_total - (time.monotonic() - start)
        alloc = min(time_budget_each, remaining * 0.9)
        if alloc < 3:
            break

        a_new = lbfgsb_log_opt(a, beta, max_iters=500, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        # Always continue from the latest result (not just best)
        a = a_new

    return best_a


def build_conv_matrix(a):
    """Build matrix A such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        l_min = max(0, k - (n - 1))
        l_max = min(k, n - 1)
        for l in range(l_min, l_max + 1):
            A[k, l] = a[k - l]
    return A


def lp_refine(a, time_budget=30.0, num_line_search=20):
    """
    LP-based direction finding:
    1. Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0
    2. Line search for best blend parameter t
    """
    n = len(a)
    if n > 1200:
        return a.copy()

    a = np.asarray(a, dtype=np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    M = float(np.max(conv))

    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    start = time.monotonic()
    lp_time = min(time_budget * 0.4, 20.0)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': lp_time})
        if res.status != 0:
            return a
        g = np.maximum(res.x, 0)
    except Exception:
        return a

    s_g = float(np.sum(g))
    if s_g < 1e-10:
        return a
    g = g / s_g

    # Line search for best t
    best_a = a.copy()
    best_obj = compute_obj(a)

    if time.monotonic() - start > time_budget - 5:
        return best_a

    # Golden section search for t in [0, 1]
    inv_phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    t1 = hi - inv_phi * (hi - lo)
    t2 = lo + inv_phi * (hi - lo)

    def eval_t(t):
        a_t = (1 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s_t = float(np.sum(a_t))
        if s_t < 1e-10:
            return np.inf
        a_t = a_t / s_t
        return compute_obj(a_t)

    f1 = eval_t(t1)
    f2 = eval_t(t2)

    if f1 < best_obj:
        best_obj = f1
        best_a = ((1 - t1) * a + t1 * g)
        best_a = best_a / np.sum(best_a)
    if f2 < best_obj:
        best_obj = f2
        best_a = ((1 - t2) * a + t2 * g)
        best_a = best_a / np.sum(best_a)

    for _ in range(num_line_search):
        if time.monotonic() - start > time_budget - 5:
            break
        if f1 < f2:
            hi = t2
            t2 = t1
            f2 = f1
            t1 = hi - inv_phi * (hi - lo)
            f1 = eval_t(t1)
        else:
            lo = t1
            t1 = t2
            f1 = f2
            t2 = lo + inv_phi * (hi - lo)
            f2 = eval_t(t2)

        if f1 < best_obj:
            best_obj = f1
            best_a = ((1 - t1) * a + t1 * g)
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
        if f2 < best_obj:
            best_obj = f2
            best_a = ((1 - t2) * a + t2 * g)
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)

    return best_a


def coordinate_peak_shave(a, time_budget=20.0, seed=0):
    """
    Peak-directed coordinate descent:
    For each top peak, try to reduce it by adjusting contributing elements.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    a = np.asarray(a, dtype=np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(best_a)
    top_k = min(10, len(conv))
    peak_indices = np.argsort(conv)[::-1][:top_k]

    iteration = 0
    while time.monotonic() - start < time_budget - 1:
        iteration += 1
        improved = False

        for p in peak_indices:
            if time.monotonic() - start > time_budget - 1:
                break
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)

            # Try moving mass between pairs that contribute to this peak
            candidates = []
            for i in range(i_min, i_max + 1):
                j = p - i
                if j >= 0 and j < n and i != j:
                    candidates.append((i, j))

            rng.shuffle(candidates)
            for (i, j) in candidates[:20]:
                if time.monotonic() - start > time_budget - 1:
                    break
                # Try moving small amount from larger to smaller (within the pair)
                if best_a[i] > best_a[j]:
                    delta = best_a[i] * np.array([0.01, 0.03, 0.05, 0.1])
                else:
                    delta = -best_a[j] * np.array([0.01, 0.03, 0.05, 0.1])

                for d in delta:
                    a_try = best_a.copy()
                    a_try[i] -= d
                    a_try[j] += d
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        conv = compute_conv(best_a)
                        peak_indices = np.argsort(conv)[::-1][:top_k]
                        improved = True
                        break

        if not improved and iteration > 30:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def make_structured_init(n, kind, rng):
    """Create a structured initial sequence of length n."""
    x = np.arange(n) / max(n - 1, 1)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'gaussian':
        sigma = rng.uniform(0.15, 0.4)
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif kind == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == 'triangle':
        a = 1 - np.abs(2 * x - 1)
    elif kind == 'bimodal':
        center1, center2 = 0.25, 0.75
        sigma = rng.uniform(0.08, 0.15)
        a = np.exp(-0.5 * ((x - center1) / sigma) ** 2) + np.exp(-0.5 * ((x - center2) / sigma) ** 2)
    elif kind == 'comb':
        num_peaks = rng.choice([3, 5, 7, 9])
        positions = np.linspace(0.1, 0.9, num_peaks)
        a = np.zeros(n)
        sigma = 0.05
        for pos in positions:
            a += np.exp(-0.5 * ((x - pos) / sigma) ** 2)
    elif kind == 'random':
        a = rng.uniform(0.1, 1.0, n)
        # Smooth it a bit
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    elif kind == 'sine_mod':
        freq = rng.uniform(2, 8)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    return a

                    parent_resampled = resample_sequence(best_seq, n_r)
                    alpha = rng.uniform(0.2, 0.5)
                    a_init = (1 - alpha) * parent_resampled + alpha * a_init
                    a_init = np.maximum(a_init, 1e-10)
                    a_init = a_init / np.sum(a_init)

            beta_sched = [10, 30, 80, 200, 400]
            alloc = min(restart_budget, (deadline - time.monotonic()) * 0.1)
            if alloc < 5:
                break
            a_opt = continuation_opt(a_init, beta_sched, alloc / len(beta_sched), alloc)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {ri} ({kind}, n={n_r}): {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {ri} ({kind}, n={n_r}): {obj:.8f}")

    # === Phase 5: Peak-directed coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Coordinate peak shaving")
    print(f"{'='*60}")

    if best_seq is not None:
        if time.monotonic() < deadline - 30:
            alloc = min(30.0, (deadline - time.monotonic()) * 0.15)
            a_cd = coordinate_peak_shave(best_seq, time_budget=alloc, seed=seed + 1)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Coord shave: {best_score:.8f} ***")
            else:
                print(f"  Coord shave: {obj_cd:.8f} (no imp)")

    # === Phase 6: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        # High-beta L-BFGS-B
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            for beta in [500, 1000, 2000]:
                if time.monotonic() > deadline - 10:
                    break
                alloc = min(20.0, remaining * 0.2)
                if alloc < 3:
                    break
                a_final = lbfgsb_log_opt(best_seq, beta, max_iters=300, time_budget=alloc)
                obj_final = compute_obj(a_final)
                if obj_final < best_score:
                    best_score = obj_final
                    best_seq = a_final.copy()
                    print(f"  Final beta={beta}: {best_score:.8f} ***")
                else:
                    print(f"  Final beta={beta}: {obj_final:.8f}")
                remaining = deadline - time.monotonic()

        # One more LP pass
        if best_seq is not None and len(best_seq) <= 1000 and time.monotonic() < deadline - 15:
            alloc = min(20.0, (deadline - time.monotonic()) * 0.3)
            a_lp_final = lp_refine(best_seq, time_budget=alloc)
            obj_lp_final = compute_obj(a_lp_final)
            if obj_lp_final < best_score:
                best_score = obj_lp_final
                best_seq = a_lp_final.copy()
                print(f"  Final LP: {best_score:.8f} ***")

        # Final coordinate shave
        if time.monotonic() < deadline - 10:
            alloc = min(15.0, (deadline - time.monotonic()) * 0.3)
            a_cd2 = coordinate_peak_shave(best_seq, time_budget=alloc, seed=seed + 42)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  Final coord: {best_score:.8f} ***")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 200
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
