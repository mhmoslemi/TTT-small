import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_grad(a, beta, top_k=5):
    """Compute gradient of smoothed objective with peak emphasis."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    # Softmax weights with temperature
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    
    # Add emphasis on top-k peaks
    if top_k > 0:
        top_indices = np.argsort(conv)[::-1][:top_k]
        top_sum = np.sum(w[top_indices])
        if top_sum > 1e-10:
            w_emph = w.copy()
            w_emph[top_indices] *= 2.0
            w_emph /= np.sum(w_emph)
        else:
            w_emph = w
    else:
        w_emph = w
    
    ls = cmax + np.log(e_sum) / beta
    
    # Gradient: 2 * cross_correlation(w_emph, a)
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w_emph, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    return obj, grad_obj

def adam_log_opt(a, max_iters=20000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995, top_k=3):
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        # Cosine annealing for beta
        beta = beta_start + (beta_end - beta_start) * 0.5 * (1 - np.cos(np.pi * progress))
        
        x_centered = x - np.max(x)
        a_curr = np.exp(x_centered)
        a_curr = a_curr / np.sum(a_curr)
        
        obj_sm, grad_a = smooth_grad(a_curr, beta, top_k=top_k)
        grad_x = grad_a * a_curr
        
        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 200 == 0:
            x -= np.mean(x)
        
        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a

def peak_directed_refine(a, time_budget=20.0, top_k=3):
    """Directly reduce peak positions by moving mass away."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(500):

        improved = False
        
        for p in peaks:
            # Find positions contributing to this peak
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            
            # Find the largest contributor
            contributors = []
            for i in range(i_min, i_max + 1):
                j = p - i
                if 0 <= j < n:
                    contributors.append((i, best_a[i] * best_a[j]))
            
            if not contributors:
                continue
            
            contributors.sort(key=lambda x: -x[1])

            i_top = contributors[0][0]

            # We want to move mass to a position that doesn't contribute to any top peak
            
            # Simple heuristic: move to position n-1-i_top (mirror)
            j_target = n - 1 - i_top if n - 1 - i_top != i_top else 0

                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break

            break

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

"""
Hybrid optimizer combining Adam in log-space with adaptive peak emphasis,
mirror-descent-inspired weight updates, peak-directed local refinement,
diverse structured initializations, and multi-length exploration to minimize
2*n*max(convolve(a,a))/(sum(a))^2.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def gradient_at_peaks(a, top_indices):
    """Compute exact subgradient at specified peak positions.
    grad[i] = 2 * sum_{p in top_indices} a[p-i] (with bounds checking)
    Returns gradient of sum_k conv[a,a][k] over the given positions."""
    n = len(a)
    grad = np.zeros(n)
    for p in top_indices:
        i_min = max(0, p - (n - 1))
        i_max = min(p, n - 1)
        for i in range(i_min, i_max + 1):
            j = p - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    return grad


def gradient_fft(a, weights):
    """Compute gradient using FFT: 2 * cross_correlation(weights, a)."""
    n = len(a)
    size = 1
    while size < 2 * len(weights):
        size <<= 1
    W = rfft(weights, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    return 2.0 * cc[:n]


def smooth_grad(a, beta, top_k=5, use_hard=False):
    """Compute smoothed objective and gradient with peak emphasis."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    if use_hard and top_k > 0:
        # Use hard max gradient at top K positions
        top_indices = np.argsort(conv)[::-1][:top_k]
        grad_ls = gradient_fft(a, np.zeros_like(conv))  # placeholder
        # Exact subgradient: uniform weight on top K
        weights = np.zeros(len(conv))
        weights[top_indices] = 1.0 / top_k
        grad_ls = gradient_fft(a, weights)
        # Objective is the average of top K (approximation of max)
        ls = float(np.mean(conv[top_indices]))
    else:
        # Softmax gradient
        cmax = float(np.max(conv))
        shifted = beta * (conv - cmax)
        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        e_sum = float(np.sum(e))
        w = e / e_sum
        ls = cmax + np.log(e_sum) / beta

        # Add emphasis on top-k peaks
        if top_k > 0:
            top_indices = np.argsort(conv)[::-1][:top_k]
            w_emph = w.copy()
            w_emph[top_indices] *= 1.5
            w_emph /= np.sum(w_emph)
        else:
            w_emph = w

        grad_ls = gradient_fft(a, w_emph)

    obj = 2.0 * n * ls / (sum_a * sum_a)
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def adam_log_opt(a, max_iters=20000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995,
                 top_k=3, hard_switch=0.6, momentum_reset=800):
    """
    Adam optimization in log-space with adaptive peak emphasis.
    Switches from soft to hard gradient based on progress fraction.
    Periodic momentum resets to escape local minima.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start

    best_a = a.copy()
    best_true_obj = compute_obj(a)
    stale_iters = 0
    improvement_threshold = 1e-9

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        # Cosine annealing for beta
        beta = beta_start + (beta_end - beta_start) * 0.5 * (1 - np.cos(np.pi * progress))

        # Switch to hard gradient in the latter portion
        use_hard = progress > hard_switch

        x_centered = x - np.max(x)
        a_curr = np.exp(x_centered)
        a_curr = a_curr / np.sum(a_curr)

        obj_sm, grad_a = smooth_grad(a_curr, beta, top_k=top_k, use_hard=use_hard)
        grad_x = grad_a * a_curr

        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic momentum reset to escape local minima
        if momentum_reset > 0 and it % momentum_reset == momentum_reset - 1 and it > 0:
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0

        if it % 200 == 0:
            x -= np.mean(x)

        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj - improvement_threshold:
                best_true_obj = true_obj
                best_a = a_check.copy()
                stale_iters = 0
            else:
                stale_iters += 1

    return best_a


def peak_reduction_step(a, time_budget=30.0, top_k=5, n_steps=200):
    """
    Direct peak reduction using gradient projected to simplex tangent space.
    At each step, compute the gradient at the top K peaks, then move in the
    direction that most reduces the maximum peak.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    lr = 0.01

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        peaks = np.argsort(conv)[::-1][:top_k]

        # Compute subgradient at peak positions
        weights = np.zeros(len(conv))
        weights[peaks] = 1.0 / top_k
        grad = gradient_fft(best_a, weights)

        # Project to tangent space of simplex (sum = 0)
        grad_centered = grad - np.mean(grad)

        # Adaptive step size: scale by the relative magnitude
        scale = np.max(np.abs(best_a)) * 0.1
        if scale < 1e-15:
            break

        # Try multiple step sizes
        improved = False
        for lr_try in [lr, lr * 0.5, lr * 0.1, lr * 2.0]:
            a_try = best_a - lr_try * grad_centered * best_a / scale
            a_try = np.maximum(a_try, 1e-15)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                improved = True
                break

        if not improved:
            lr *= 0.5
            if lr < 1e-6:
                break

    return best_a


def mirror_descent_blend(a, time_budget=30.0, eta_start=1.0, eta_end=50.0):
    """
    Mirror descent on the k-distribution, combined with log-space updates.
    Maintains a distribution w over convolution positions and alternates
    between updating a (gradient step) and updating w (multiplicative weights).
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)

    conv = compute_conv(a)
    w = np.exp(conv / max(np.max(conv), 1e-10))
    w = w / np.sum(w)

    best_a = a.copy()
    best_obj = compute_obj(a)
    max_iters = 5000
    lr = 0.02

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        eta = eta_start + (eta_end - eta_start) * progress

        # Gradient of sum_k w[k] * conv[a,a][k]
        grad = gradient_fft(a, w)

        # Log-space update
        x = np.log(a)
        x = x - lr * grad * a
        x = x - np.max(x)
        a_new = np.exp(x)
        a_new = a_new / np.sum(a_new)

        # Update w by multiplicative weights
        conv_new = compute_conv(a_new)
        eta_scaled = eta / max(len(conv_new), 1)
        w *= np.exp(eta_scaled * conv_new)
        w = w / np.sum(w)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def cosine_half_window(n):
    """Half-cosine window: peaks in middle, zero at edges."""
    return np.sin(np.pi * np.arange(n + 1) / (n + 1))[:-1] if n > 1 else np.array([1.0])


def hann_window(n):
    """Hann window."""
    if n == 1:
        return np.array([1.0])
    return 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))


def build_structured(n, window_func):
    """Build a structured initial sequence and normalize."""
    a = window_func(n)
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    return a


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 0.01:
                parent = parent / s
                obj = compute_obj(parent)
                print(f"Loaded parent (n={parent.size}): obj={obj:.8f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines Adam log-space optimization with adaptive hardness,
    mirror descent blending, peak-directed gradient steps, diverse structured
    restarts, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, budget={budget_s}s")

    # === Phase 1: Intensive Adam optimization on parent length ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Adam log-space optimization (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()

        # Progressive rounds with increasing hardness
        rounds = [
            # (beta_start, beta_end, lr, lr_decay, top_k, hard_switch, time_alloc)
            (3.0, 15.0, 0.06, 0.99997, 2, 0.8, 120),
            (5.0, 30.0, 0.05, 0.99998, 3, 0.7, 120),
            (10.0, 60.0, 0.04, 0.99998, 3, 0.6, 120),
            (20.0, 100.0, 0.03, 0.99999, 5, 0.5, 120),
            (40.0, 200.0, 0.025, 0.99999, 5, 0.4, 120),
            (80.0, 400.0, 0.02, 0.99999, 7, 0.3, 120),
            (150.0, 600.0, 0.015, 0.99999, 10, 0.2, 100),
            (300.0, 1000.0, 0.01, 0.99999, 15, 0.1, 80),
            (500.0, 2000.0, 0.007, 0.99999, 20, 0.0, 60),
        ]

        for r_idx, (b_s, b_e, lr0, lr_d, tk, hs, t_alloc) in enumerate(rounds):
            if time.monotonic() > deadline - 180:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.15, t_alloc)
            if alloc < 15:
                break

            a_new = adam_log_opt(a, max_iters=30000, time_budget=alloc,
                                 beta_start=b_s, beta_end=b_e,
                                 lr_start=lr0, lr_decay=lr_d,
                                 top_k=tk, hard_switch=hs)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  R{r_idx} (b{b_s:.0f}-{b_e:.0f}, k={tk}, hs={hs:.1f}): {best_score:.8f} ***")
            else:
                print(f"  R{r_idx} (b{b_s:.0f}-{b_e:.0f}, k={tk}, hs={hs:.1f}): {obj:.8f}")

    # === Phase 2: Peak-directed gradient refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Peak-directed gradient refinement")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        for tk in [3, 5, 8, 12]:
            if time.monotonic() > deadline - 120:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.08, 30.0)
            if alloc < 5:
                break

            a_pd = peak_reduction_step(a, time_budget=alloc, top_k=tk, n_steps=300)
            obj = compute_obj(a_pd)
            if obj < best_score:
                best_score = obj
                best_seq = a_pd.copy()
                a = a_pd.copy()
                print(f"  PeakGrad (k={tk}): {best_score:.8f} ***")
            else:
                print(f"  PeakGrad (k={tk}): {obj:.8f}")

            # Follow up with Adam
            if time.monotonic() < deadline - 60:
                remaining = deadline - time.monotonic()
                a_after = adam_log_opt(a, max_iters=15000,
                                       time_budget=min(remaining * 0.08, 25.0),
                                       beta_start=200.0, beta_end=1000.0,
                                       lr_start=0.015, lr_decay=0.99999,
                                       top_k=tk, hard_switch=0.2)
                obj_after = compute_obj(a_after)
                if obj_after < best_score:
                    best_score = obj_after
                    best_seq = a_after.copy()
                    a = a_after.copy()
                    print(f"  PeakGrad+Adam (k={tk}): {best_score:.8f} ***")

    # === Phase 3: Mirror descent blending ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Mirror descent blending")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 80:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.1, 40.0)
        a_md = mirror_descent_blend(best_seq, time_budget=alloc, eta_start=0.5, eta_end=20.0)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MirrorDescent: {best_score:.8f} ***")
        else:
            print(f"  MirrorDescent: {obj_md:.8f}")

        # Adam after mirror descent
        if time.monotonic() < deadline - 40:
            remaining = deadline - time.monotonic()
            a_post = adam_log_opt(a_md, max_iters=15000,
                                  time_budget=min(remaining * 0.1, 25.0),
                                  beta_start=100.0, beta_end=500.0,
                                  lr_start=0.02, lr_decay=0.99999,
                                  top_k=10, hard_switch=0.3)
            obj_post = compute_obj(a_post)
            if obj_post < best_score:
                best_score = obj_post
                best_seq = a_post.copy()
                print(f"  MD+Adam: {best_score:.8f} ***")

    # === Phase 4: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-length exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 150),
            max(200, n_best - 75),
            n_best + 50,
            n_best + 100,
            n_best + 200,
            n_best + 400,
        ]))
        n_options = [n for n in n_options if 200 <= n <= 2500 and n != n_best]

        for n_try in n_options:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_try = resample_sequence(best_seq, n_try)
            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s

            alloc = min(remaining * 0.15, 50.0)
            a_opt = adam_log_opt(a_try, max_iters=20000, time_budget=alloc,
                                 beta_start=10.0, beta_end=300.0,
                                 lr_start=0.03, lr_decay=0.99998,
                                 top_k=5, hard_switch=0.5)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 5: Diverse structured restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Diverse structured restarts")
    print(f"{'='*60}")

    n_use = len(best_seq) if best_seq is not None else 700
    rng = np.random.default_rng(seed)

    windows = [
        ("cos_half", cosine_half_window),
        ("hann", hann_window),
        ("raised", lambda n: 0.5 * (1 + np.cos(np.pi * (2 * np.arange(n) - (n - 1)) / n))),
        ("tapered", lambda n: 1.0 - 0.4 * ((np.arange(n) / max(n - 1, 1)) - 0.5) ** 2),
        ("bipolar_smooth", lambda n: np.maximum(0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n)), 0.05)),
    ]

    for name, func in windows:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        a_struct = build_structured(n_use, func)

        # Blend with current best at different ratios
        alphas = [0.3, 0.5, 0.7]
        for alpha in alphas:
            if time.monotonic() > deadline - 25:
                break
            a_blend = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * a_struct
            a_blend = np.maximum(a_blend, 1e-15)
            a_blend = a_blend / np.sum(a_blend)

            alloc = min(remaining * 0.1, 25.0)
            a_opt = adam_log_opt(a_blend, max_iters=15000, time_budget=alloc,
                                 beta_start=10.0, beta_end=200.0,
                                 lr_start=0.03, lr_decay=0.99998,
                                 top_k=5, hard_switch=0.5)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  {name} a={alpha:.1f}: {best_score:.8f} ***")

    # === Phase 6: Random perturbations + optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Random perturbations")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        for restart in range(6):
            if time.monotonic() > deadline - 25:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if restart % 3 == 0:
                # Multiplicative lognormal noise
                scale = 0.02 + 0.01 * restart
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            elif restart % 3 == 1:
                # Sparse additive noise
                a_pert = best_seq.copy()
                n_spikes = max(3, n_best // 30)
                idx = rng.choice(n_best, size=n_spikes, replace=False)
                a_pert[idx] += rng.uniform(0.005, 0.05, n_spikes) * np.mean(best_seq)
            else:
                # Swap pairs of values
                a_pert = best_seq.copy()
                for _ in range(n_best // 10):
                    i, j = rng.choice(n_best, size=2, replace=False)
                    a_pert[i], a_pert[j] = a_pert[j], a_pert[i]

            a_pert = np.maximum(a_pert, 1e-15)
            a_pert = a_pert / np.sum(a_pert)

            alloc = min(remaining * 0.12, 20.0)
            a_opt = adam_log_opt(a_pert, max_iters=10000, time_budget=alloc,
                                 beta_start=20.0, beta_end=300.0,
                                 lr_start=0.03, lr_decay=0.99999,
                                 top_k=5, hard_switch=0.5)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} ***")
            else:
                print(f"  Restart {restart}: {obj:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()

        # Very high beta with hard gradient
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            a_final = adam_log_opt(a, max_iters=20000,
                                   time_budget=min(remaining * 0.4, 40.0),
                                   beta_start=500.0, beta_end=3000.0,
                                   lr_start=0.005, lr_decay=0.99999,
                                   top_k=25, hard_switch=0.1)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                a = a_final.copy()
                print(f"  High-beta final: {best_score:.8f} ***")

        # Peak reduction polish
        if time.monotonic() < deadline - 8:
            remaining = deadline - time.monotonic()
            a_pr = peak_reduction_step(a, time_budget=min(remaining * 0.3, 20.0),
                                       top_k=10, n_steps=200)
            obj_pr = compute_obj(a_pr)
            if obj_pr < best_score:
                best_score = obj_pr
                best_seq = a_pr.copy()
                print(f"  Peak reduction final: {best_score:.8f} ***")

        # Last Adam
        if time.monotonic() < deadline - 3:
            remaining = deadline - time.monotonic()
            a_last = adam_log_opt(best_seq, max_iters=10000,
                                  time_budget=min(remaining * 0.8, 12.0),
                                  beta_start=1000.0, beta_end=5000.0,
                                  lr_start=0.003, lr_decay=0.99999,
                                  top_k=30, hard_switch=0.0)
            obj_last = compute_obj(a_last)
            if obj_last < best_score:
                best_score = obj_last
                best_seq = a_last.copy()
                print(f"  Last Adam: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
