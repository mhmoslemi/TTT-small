Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines Adam in log-space, L-BFGS on smoothed objective, FISTA acceleration,

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    size = 1
    while size < 2 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)

    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def topk_subgradient(a, k=5):
    """Compute objective and subgradient using top-k peaks for a denser signal."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))

    obj = 2.0 * n * max_b / (sum_a * sum_a)

    # Get top-k lags by value
    top_indices = np.argsort(conv)[::-1][:k]
    top_vals = conv[top_indices]

    # Weight by softmax of top values (gives more weight to actual peak)
    w_alpha = 5.0
    w_shifted = w_alpha * (top_vals - top_vals[0])
    w_e = np.exp(w_shifted)
    w_sum = np.sum(w_e)
    weights = w_e / w_sum

    w = np.zeros(2 * n - 1)
    for idx, wt in zip(top_indices, weights):
        w[idx] = wt

    size = 1
    while size < 2 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)

    grad_maxb = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_maxb - max_b / sum_a)

    return obj, grad_obj, conv


def adam_log_opt(a, max_iters=20000, time_budget=100.0,
                 beta_start=5.0, beta_end=500.0,
                 lr_start=0.05, lr_decay=0.99995,
                 use_topk=False, topk_k=5):
    """
    Adam optimization in log-space with temperature annealing.
    Optionally uses top-k subgradient for denser signal at high beta.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_true_obj = compute_obj(a)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr_t = lr_start * (lr_decay ** it)

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)

        # Choose between smoothed and top-k subgradient
        if use_topk and beta > 100:
            _, grad_a, _ = topk_subgradient(a, k=topk_k)
        else:
            _, grad_a = smooth_obj_grad(a, beta)

        grad_x = grad_a * a

        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 200 == 0:
            x -= np.mean(x)

        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

    return best_a


def lbfgs_smoothed(a, beta, time_budget=60.0):
    """Use L-BFGS-B on the smoothed objective in log-space."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x0 = np.log(a)
    x0 -= np.mean(x0)

    def func(x):
        xc = x - np.max(x)
        aa = np.exp(xc)
        aa = aa / np.sum(aa)
        obj, _ = smooth_obj_grad(aa, beta)
        return obj

    def grad(x):
        xc = x - np.max(x)
        aa = np.exp(xc)
        aa = aa / np.sum(aa)
        _, g_a = smooth_obj_grad(aa, beta)
        return g_a * aa

    start = time.monotonic()
    try:
        result = minimize(func, x0, jac=grad, method='L-BFGS-B',
                          options={'maxiter': 300, 'maxfun': 1500,
                                   'maxls': 50, 'iprint': -1})
        xc = result.x - np.max(result.x)
        a_opt = np.exp(xc)
        a_opt = np.maximum(a_opt, 1e-15)
        a_opt = a_opt / np.sum(a_opt)
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            print(f"  L-BFGS timed out at {elapsed:.1f}s")
        return a_opt
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        return a


def fista_smoothed(a, beta, time_budget=60.0, n_iters=5000):
    """FISTA (accelerated gradient) on the smoothed objective in log-space."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    # Estimate Lipschitz constant for step size
    # The Hessian of the smoothed objective in log-space is bounded; use conservative estimate
    _, g0 = smooth_obj_grad(a, beta)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-15:
        return a
    # Adaptive step size
    step = 0.1 / g_norm
    step = min(step, 0.5)

    x = np.log(a)
    x -= np.mean(x)
    y = x.copy()
    t_k = 1.0

    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        yc = y - np.max(y)
        aa = np.exp(yc)
        aa = aa / np.sum(aa)

        _, grad_a = smooth_obj_grad(aa, beta)
        grad_x = grad_a * aa

        # Gradient step
        x_new = y - step * grad_x

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        t_k = t_new
        x = x_new

        # Periodic centering
        if it % 500 == 0:
            x -= np.mean(x)
            y -= np.mean(y)

        if it % 200 == 0 or it == n_iters - 1:
            xc = x - np.max(x)
            a_check = np.exp(xc)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()

        # Adaptive step size: reduce if objective increases
        if it > 0 and it % 100 == 0:
            xc = x - np.max(x)
            aa_check = np.exp(xc)
            aa_check = aa_check / np.sum(aa_check)
            obj_now, _ = smooth_obj_grad(aa_check, beta)
            prev_xc = prev_x - np.max(prev_x) if 'prev_x' in dir() else x
            if obj_now > best_obj + 0.01:
                step *= 0.5
        if it % 100 == 0:
            prev_x = x.copy()

    return best_a


def mirror_descent(a, time_budget=60.0, beta_start=5.0, beta_end=500.0,
                   lr_start=0.1, lr_end=0.005, max_iters=20000):
    """
    Mirror descent (multiplicative weight updates) in log-space with annealing.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    best_smooth_obj = np.inf
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        obj_sm, grad_a = smooth_obj_grad(a, beta)

        # Multiplicative weight update
        log_update = -lr * grad_a
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = np.maximum(a_new, 1e-15)
        a = a / np.sum(a)

        if it % 50 == 0 or it == max_iters - 1:
            true_obj = compute_obj(a)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1

        # Early stop if very stagnant
        if stagnant > 50 and beta > 200:
            break

    return best_a


def lp_refine(a, time_budget=30.0):
    """LP-based refinement: find direction reducing peak via LP, then blend."""
    n = len(a)
    start = time.monotonic()
    if n > 1200:
        return a
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s

    conv = compute_conv(a_norm)
    M = float(np.max(conv))

    # Build Toeplitz matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a_norm[:k_end - k_start]

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Try several M fractions (from aggressive to conservative)
    for M_frac in [0.85, 0.90, 0.95, 1.0]:
        if time.monotonic() - start > time_budget:
            break

        M_try = M * M_frac
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_try)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 15})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                sg = float(np.sum(g))
                if sg > 1e-10:
                    g = g / sg
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        if time.monotonic() - start > time_budget:
                            break
                        a_try = (1 - t) * a_norm + t * g
                        a_try = np.maximum(a_try, 0)
                        st = np.sum(a_try)
                        if st < 1e-10:
                            continue
                        a_try = a_try / st
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
        except Exception:
            pass

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)

                                 use_topk=True)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart}: {obj:.8f}")

    # === Phase 8: Final intensive polish ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        # Final high-beta Adam
        a_final = adam_log_opt(a, max_iters=50000,
                               time_budget=min(remaining * 0.4, 20.0),
                               beta_start=300.0, beta_end=2000.0,
                               lr_start=0.005, lr_decay=0.99999,
                               use_topk=True, topk_k=3)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final high-beta Adam: {best_score:.8f} ***")

        # Final L-BFGS polish
        if time.monotonic() < deadline - 5:
            a_lb_final = lbfgs_smoothed(a_final, 100.0, time_budget=min(remaining * 0.3, 15.0))
            obj_lb = compute_obj(a_lb_final)
            if obj_lb < best_score:
                best_score = obj_lb
                best_seq = a_lb_final.copy()
                print(f"  Final L-BFGS: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
