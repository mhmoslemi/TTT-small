import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def cross_correlation(w, a):
    """Compute cross-correlation (w*a)[j] = sum_k w[j+k]*a[k] for j=0..n-1."""
    n = len(a)
    size = 1
    while size < 2 * len(w):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    return cc[:n]


def smooth_obj_grad_corrected(a, beta):
    """
    Compute smoothed objective and CORRECTED gradient w.r.t. a.
    F = 2n * LSE(beta*conv) / (beta * S^2)
    dF/da[j] = (2n/S^2) * [2*(w*a)[j] - 2*ls/S]
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    cc = cross_correlation(w, a)
    # Corrected gradient: (2n/S^2) * (2*cc[j] - 2*ls/S)
    grad_a = (2.0 * n / (sum_a * sum_a)) * (2.0 * cc - 2.0 * ls / sum_a)

    return obj, grad_a


def adam_log_opt_v2(a, max_iters=20000, time_budget=100.0, beta_start=5.0,
                    beta_end=200.0, lr_start=0.05, lr_decay=0.99995):

    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_true_obj = compute_obj(a)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)

        obj_sm, grad_a = smooth_obj_grad_corrected(a, beta)
        grad_x = grad_a * a

        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr_t = lr_start * (lr_decay ** it)

            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

    return best_a


def peak_matching_update(a, top_k=10, eta=0.01, n_iters=100, time_budget=30.0):
    """
    Peak-matching: iteratively reduce top peaks by multiplicatively adjusting a.
    Positions contributing to high peaks are reduced; others are increased.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        max_val = np.max(conv)

        # Find top peaks
        top_indices = np.argsort(conv)[::-1][:top_k]
        top_vals = conv[top_indices]
        avg_top = np.mean(top_vals)

        # Compute responsibility
        R = np.zeros(n)
        for k in top_indices:
            w_k = max(0, conv[k] - avg_top * 0.8)  # weight peaks above threshold
            for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
                if 0 <= k - j < n:
                    R[j] += w_k * best_a[k - j]

        # Multiplicative update
        R_mean = np.mean(R)
        R_normalized = (R - R_mean) / (np.max(np.abs(R - R_mean)) + 1e-10)
        a_new = best_a * np.exp(-eta * R_normalized)
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            # Try smaller eta
            eta *= 0.8
            if eta < 1e-5:
                break
            eta = max(eta, 0.001)

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def make_cosine_window(n):
    """Create a cosine window."""
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.array([1.0])
    s = np.sum(a)
    return a / s


def make_gaussian_window(n, sigma_frac=0.3):
    """Create a Gaussian window."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x**2 / (2 * sigma**2))
    s = np.sum(a)
    return a / s


def make_sine_window(n):
    """Create a sine window (raised cosine variant)."""
    x = np.arange(n)
    a = np.sin(np.pi * (x + 1) / (n + 1))
    s = np.sum(a)
    return a / s


def make_symmetric(a):
    """Make a sequence symmetric (palindromic)."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = np.sum(a_sym)
    return a_sym / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = np.sum(a_new)
    return a_new / s

        n_options = sorted(set([
            n_base - 100, n_base - 50, n_base,
            n_base + 50, n_base + 100, n_base + 200,
            n_base + 300, n_base + 500,
            500, 800, 1000, 1200, 1400
        ]))
    else:
        n_options = [500, 700, 900, 1100, 1300, 1500]
    n_options = [n for n in n_options if 200 <= n <= 2000]

    # Try structured initializations
    init_candidates = []
    if best_seq is not None:
        s = np.sum(best_seq)
        if s > 1e-10:
            init_candidates.append(("parent", best_seq / s))

    for n_try in n_options[:8]:
        if time.monotonic() > deadline - 600:
            break
        for name, make_func in [("cosine", make_cosine_window),
                                ("gaussian", make_gaussian_window),
                                ("sine", make_sine_window)]:
            a_init = make_func(n_try)
            init_candidates.append((f"{name}_n{n_try}", a_init))

    # Quick evaluate all initializations
    init_results = []
    for name, a_init in init_candidates:
        obj = compute_obj(a_init)
        init_results.append((obj, name, a_init.copy()))
        if obj < best_score:
            best_score = obj
            best_seq = a_init.copy()

    init_results.sort()
    print(f"  Best init: {init_results[0][1]} -> {init_results[0][0]:.8f}")
    for obj, name, _ in init_results[:5]:
        print(f"    {name}: {obj:.8f}")

    # === Phase 2: Intensive optimization on top candidates ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Intensive optimization")
    print(f"{'='*60}")

    top_candidates = list(set([(init_results[0][2].copy(),),]))
    # Add parent if different
    if parent is not None:
        top_candidates.append((parent.copy(),))
    # Add a few more diverse candidates
    for i in range(1, min(4, len(init_results))):
        top_candidates.append((init_results[i][2].copy(),))

    for cand_idx, (a_start,) in enumerate(top_candidates):
        if time.monotonic() > deadline - 120:
            break

        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        n = len(a_start)
        s = np.sum(a_start)
        if s < 1e-10:
            continue
        a = a_start / s

        print(f"\n  Candidate {cand_idx} (n={n}), initial obj={compute_obj(a):.8f}")

        # Round 1: Peak matching
        alloc = min(remaining * 0.15, 40.0)
        a_pm = peak_matching_update(a, top_k=20, eta=0.05, n_iters=200,
                                    time_budget=alloc)
        obj_pm = compute_obj(a_pm)
        if obj_pm < best_score:
            best_score = obj_pm
            best_seq = a_pm.copy()
            a = a_pm.copy()
            print(f"    Peak matching: {best_score:.8f} ***")

        # Round 2: Adam with low-to-mid temperature
        if time.monotonic() < deadline - 100:
            alloc = min((deadline - time.monotonic()) * 0.15, 50.0)
            a_adam = adam_log_opt_v2(a, max_iters=30000, time_budget=alloc,
                                     beta_start=5.0, beta_end=100.0,
                                     lr_start=0.04, lr_decay=0.99998)
            obj_adam = compute_obj(a_adam)
            if obj_adam < best_score:
                best_score = obj_adam
                best_seq = a_adam.copy()
                a = a_adam.copy()
                print(f"    Adam (low-beta): {best_score:.8f} ***")

        # Round 3: Peak matching again
        if time.monotonic() < deadline - 80:
            alloc = min((deadline - time.monotonic()) * 0.1, 30.0)
            a_pm2 = peak_matching_update(a, top_k=30, eta=0.03, n_iters=200,
                                         time_budget=alloc)
            obj_pm2 = compute_obj(a_pm2)
            if obj_pm2 < best_score:
                best_score = obj_pm2
                best_seq = a_pm2.copy()
                a = a_pm2.copy()
                print(f"    Peak matching 2: {best_score:.8f} ***")

        # Round 4: Adam with high temperature (finer)
        if time.monotonic() < deadline - 60:
            alloc = min((deadline - time.monotonic()) * 0.15, 50.0)
            a_adam2 = adam_log_opt_v2(a, max_iters=30000, time_budget=alloc,
                                      beta_start=50.0, beta_end=500.0,
                                      lr_start=0.02, lr_decay=0.99999)
            obj_adam2 = compute_obj(a_adam2)
            if obj_adam2 < best_score:
                best_score = obj_adam2
                best_seq = a_adam2.copy()
                a = a_adam2.copy()
                print(f"    Adam (high-beta): {best_score:.8f} ***")

    # === Phase 3: LP refinement on the best ===
    print(f"\n{'='*60}")
    print(f"Phase 3: LP refinement (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None and len(best_seq) <= 800:
        for r_idx in range(6):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            alloc = min(remaining * 0.15, 35.0)
            a_lp = lp_refine_v2(best_seq, time_budget=alloc)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  LP {r_idx}: {best_score:.8f} ***")
            else:
                print(f"  LP {r_idx}: {obj:.8f} (no imp)")

            # Follow up with Adam
            if time.monotonic() < deadline - 30:
                a_after = adam_log_opt_v2(a_lp, max_iters=15000,
                                          time_budget=min((deadline - time.monotonic()) * 0.1, 20.0),
                                          beta_start=100.0, beta_end=600.0,
                                          lr_start=0.015, lr_decay=0.99999)
                obj_after = compute_obj(a_after)
                if obj_after < best_score:
                    best_score = obj_after
                    best_seq = a_after.copy()
                    print(f"  LP+Adam {r_idx}: {best_score:.8f} ***")

    # === Phase 4: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_explore = sorted(set([
            n_best - 100, n_best - 50,
            n_best + 50, n_best + 100, n_best + 200,
            n_best + 400, n_best + 600, n_best + 800
        ]))
        n_explore = [n for n in n_explore if 300 <= n <= 2500 and n != n_best]

        for n_try in n_explore[:6]:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_try = resample_sequence(best_seq, n_try)
            a_try = make_symmetric(a_try)  # Try symmetric version
            s = np.sum(a_try)
            if s < 1e-10:
                continue
            a_try = a_try / s

            alloc = min(remaining * 0.12, 40.0)
            a_opt = adam_log_opt_v2(a_try, max_iters=20000, time_budget=alloc,
                                    beta_start=10.0, beta_end=400.0,
                                    lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try} (sym): {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try} (sym): {obj:.8f}")

            # Also try non-symmetric
            if time.monotonic() < deadline - 40:
                a_try2 = resample_sequence(best_seq, n_try)
                s2 = np.sum(a_try2)
                if s2 > 1e-10:
                    a_try2 = a_try2 / s2
                    alloc2 = min((deadline - time.monotonic()) * 0.08, 25.0)
                    a_opt2 = adam_log_opt_v2(a_try2, max_iters=15000, time_budget=alloc2,
                                             beta_start=20.0, beta_end=400.0,
                                             lr_start=0.03, lr_decay=0.99998)
                    obj2 = compute_obj(a_opt2)
                    if obj2 < best_score:
                        best_score = obj2
                        best_seq = a_opt2.copy()
                        print(f"  n={n_try} (asym): {best_score:.8f} *** UPDATE")
                    else:
                        print(f"  n={n_try} (asym): {obj2:.8f}")

    # === Phase 5: Intensive final optimization on the absolute best ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Final intensive optimization")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        # Extended Adam with very high beta
        if time.monotonic() < deadline - 40:
            remaining = deadline - time.monotonic()
            a_final = adam_log_opt_v2(a, max_iters=50000,
                                      time_budget=min(remaining * 0.25, 80.0),
                                      beta_start=200.0, beta_end=2000.0,
                                      lr_start=0.01, lr_decay=0.999995)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                a = a_final.copy()
                print(f"  High-beta Adam: {best_score:.8f} ***")

        # Peak matching polish
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            a_pm_final = peak_matching_update(a, top_k=50, eta=0.01, n_iters=500,
                                              time_budget=min(remaining * 0.2, 40.0))
            obj_pm_final = compute_obj(a_pm_final)
            if obj_pm_final < best_score:
                best_score = obj_pm_final
                best_seq = a_pm_final.copy()
                a = a_pm_final.copy()
                print(f"  Final peak matching: {best_score:.8f} ***")

        # Coordinate descent
        if time.monotonic() < deadline - 25:
            remaining = deadline - time.monotonic()
            a_cd = coordinate_refine_v2(a, time_budget=min(remaining * 0.25, 50.0),
                                        seed=seed)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                a = a_cd.copy()
                print(f"  Coord descent: {best_score:.8f} ***")

        # One more Adam round after CD
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            a_post = adam_log_opt_v2(a, max_iters=30000,
                                     time_budget=min(remaining * 0.3, 30.0),
                                     beta_start=500.0, beta_end=3000.0,
                                     lr_start=0.005, lr_decay=0.999995)
            obj_post = compute_obj(a_post)
            if obj_post < best_score:
                best_score = obj_post
                best_seq = a_post.copy()
                print(f"  Post-CD Adam: {best_score:.8f} ***")

    # === Phase 6: Random restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Random restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed)

        for restart in range(6):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if restart % 3 == 0:
                scale = 0.02 + 0.01 * restart
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            elif restart % 3 == 1:
                # Blend with structured sequence
                alpha = rng.uniform(0.05, 0.2)
                structured = make_gaussian_window(n_best, sigma_frac=rng.uniform(0.2, 0.5))
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * structured
            else:
                # Symmetrize with noise
                a_pert = best_seq * np.exp(rng.normal(0, 0.03, n_best))
                a_pert = make_symmetric(a_pert)

            a_pert = np.maximum(a_pert, 1e-10)
            s_p = np.sum(a_pert)
            if s_p < 1e-10:
                continue
            a_pert = a_pert / s_p

            alloc = min(remaining * 0.15, 25.0)
            a_opt = adam_log_opt_v2(a_pert, max_iters=15000, time_budget=alloc,
                                    beta_start=50.0, beta_end=800.0,
                                    lr_start=0.02, lr_decay=0.99999)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart}: {obj:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = make_cosine_window(700)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
