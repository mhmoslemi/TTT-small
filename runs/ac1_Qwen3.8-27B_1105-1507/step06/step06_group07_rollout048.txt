Peak-balancing optimization: identifies top convolution peaks and uses
targeted subgradient steps to equalize them, combined with LP direction

from scipy.fft import rfft, irfft

def compute_conv(a):

    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):

    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))

def peak_subgradient(a, max_iters=3000, time_budget=60.0, top_k=3):
    """
    Subgradient descent focusing on the top-k peaks simultaneously.
    At each step, compute the weighted gradient from top-k peaks.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adaptive learning rate
    lr = 0.02
    n_stag = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)

        top_idx = np.argsort(conv)[::-1][:top_k]
        top_vals = conv[top_idx]
        max_val = top_vals[0]

            break

        weights = top_vals / top_vals.sum()

def lp_direction(a, M, max_vars=None):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns g/sum(g) or None.
    """
    n = len(a)
    if max_vars is not None and n > max_vars:
        return None
    
    # Build Toeplitz matrix
    # (a*g)[k] = sum_i a[i]*g[k-i] for valid i,k-i
    # Row k of the matrix: coefficients of g[0],...,g[n-1]
    # (a*g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[k-j] * g[j]
    # So the k-th row has: A[k,j] = a[k-j] if 0<=k-j<n, else 0
    
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

            a_try = (1 - t) * a_norm + t * g

            s_try = np.sum(a_try)
            if s_try < 1e-10:

            obj_try = compute_obj(a_try)

def log_space_adam(a, time_budget=60.0, beta=50.0, lr=0.02):
    """
    Adam in log-space with logsumexp smoothing.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    a = a / np.sum(a)
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    max_iters = int(time_budget * 100)  # aim for ~10ms per iter
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute a from x
        x_c = x - np.max(x)
        a_curr = np.exp(x_c)
        a_curr = a_curr / np.sum(a_curr)
        
        # Smoothed objective
        conv = compute_conv(a_curr)
        cmax = float(np.max(conv))
        shifted = beta * (conv - cmax)
        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        e_sum = float(np.sum(e))
        w = e / e_sum
        ls = cmax + np.log(e_sum) / beta
        
        sum_a = float(np.sum(a_curr))
        obj_sm = 2.0 * n * ls / (sum_a * sum_a)
        
        # Gradient of ls w.r.t. a[i] = 2 * sum_j w[j] * a[i+j] (cross-correlation)
        size = 1
        while size < 2 * len(conv):
            size <<= 1
        W = rfft(w, size)
        A = rfft(a_curr, size)
        cc = irfft(W * np.conj(A), size)
        grad_ls = 2.0 * cc[:n]
        
        grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
        
        # Chain rule for log-space
        grad_x = grad_obj * a_curr
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 500 == 0:
            x -= np.mean(x)

            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:

                best_a = a_check.copy()
    
    return best_a

Peak-equalization optimizer: identifies top convolution peaks and uses
targeted gradient steps to equalize them. Combined with LP direction
finding, adaptive log-space Adam, and careful n exploration.

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf

def compute_grad_at_peak(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    i_range = np.arange(n)
    j = k - i_range
    valid = (j >= 0) & (j < n)
    grad = np.zeros(n)
    grad[valid] = a[j[valid]] + a[i_range[valid]]
    return grad

def peak_equalize(a, time_budget=60.0, top_k=5, lr_init=0.01):
    """
    Directly equalize the top-k peaks of the convolution.
    Uses weighted subgradient where weights are proportional to how much
    each peak exceeds the "target" (average of top-k peaks).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a

    lr = lr_init
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        top_idx = np.argsort(conv)[::-1][:top_k]
        top_vals = conv[top_idx]

        target = top_vals.mean()

def tapered_lp_direction(a, M):
    """
    LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Uses Toeplitz structure.
    """
    n = len(a)
    if n > 1000:
        return None
    
    num_cons = 2 * n - 1
    # Build A matrix: A[k,j] = a[k-j] if 0<=k-j<n
    A = np.zeros((num_cons, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, num_cons)
        A[k_start:k_end, j] = a[:k_end - k_start]
    
    c = -np.ones(n)
    b_ub = np.full(num_cons, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None
