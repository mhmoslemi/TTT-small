"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def cd_fast(a, max_sweeps, deadline, random_order=False, seed=0, n_candidates=10):

def cd_block(a, max_sweeps, deadline, block_size=3, seed=0):
    """
    Block coordinate descent: update groups of coordinates simultaneously.
    For each block, try all 2^block_size on/off patterns plus scaled versions.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for start in range(0, n - block_size + 1, block_size):
            block_idx = order[start:start + block_size]
            if len(block_idx) < block_size:
                continue

            old_vals = a[block_idx].copy()
            old_s = s

            # Try various modifications: scale up/down, zero out, redistribute
            candidates_list = []
            # Scale all by factor
            for f in [0.3, 0.5, 0.7, 1.3, 1.7]:
                candidates_list.append(old_vals * f)
            # Zero out one element
            for j in range(block_size):
                c = old_vals.copy()
                c[j] = 0.0
                candidates_list.append(c)
            # Move mass from one to another
            for j in range(block_size - 1):
                for frac in [0.3, 0.5, 0.7]:
                    c = old_vals.copy()
                    transfer = c[j] * frac
                    c[j] -= transfer
                    c[j + 1] += transfer
                    candidates_list.append(c)

            best_candidate = old_vals
            best_local = best_obj * s * s / (2.0 * n)  # = conv.max()

            for cand in candidates_list:
                cand = np.maximum(cand, 0.0)
                delta = cand - old_vals
                new_s = old_s - old_vals.sum() + cand.sum()
                if new_s < 1e-10:
                    continue

                # Update conv
                for idx in range(block_size):
                    i = block_idx[idx]
                    d = delta[idx]
                    if abs(d) < 1e-15:
                        continue
                    lo = max(0, i)
                    hi = min(2 * n - 2, i + n - 1)
                    a_slice = a[lo - i:hi - i + 1].copy()
                    # But we need to account for other elements in the block being changed too
                    # This is approximate - only account for non-block elements
                    conv[lo:hi + 1] += 2.0 * d * a_slice
                    kq = 2 * i
                    if kq <= 2 * n - 2:
                        conv[kq] += d * d

                new_max = conv.max()
                new_obj = 2.0 * n * new_max / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_candidate = cand.copy()

                # Undo
                for idx in range(block_size):
                    i = block_idx[idx]
                    d = delta[idx]
                    if abs(d) < 1e-15:
                        continue
                    lo = max(0, i)
                    hi = min(2 * n - 2, i + n - 1)
                    a_slice = a[lo - i:hi - i + 1].copy()
                    conv[lo:hi + 1] -= 2.0 * d * a_slice
                    kq = 2 * i
                    if kq <= 2 * n - 2:
                        conv[kq] -= d * d

            if np.any(np.abs(best_candidate - old_vals) > 1e-14):
                # Apply the best candidate
                delta = best_candidate - old_vals
                for idx in range(block_size):
                    i = block_idx[idx]
                    d = delta[idx]
                    if abs(d) < 1e-15:
                        continue
                    lo = max(0, i)
                    hi = min(2 * n - 2, i + n - 1)
                    a_slice = a[lo - i:hi - i + 1].copy()
                    conv[lo:hi + 1] += 2.0 * d * a_slice
                    kq = 2 * i
                    if kq <= 2 * n - 2:
                        conv[kq] += d * d
                a[block_idx] = best_candidate
                s = s - old_vals.sum() + best_candidate.sum()
                improved = True

        if s > 1e-10:
            current_obj = 2.0 * n * conv.max() / (s * s)
        else:
            current_obj = float('inf')

        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max(50, max_sweeps // 8):
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def subgradient_step(a, n_steps, deadline, lr_init=0.01, lr_decay=0.995, seed=0):
    """
    Subgradient method for min max_k b[k] subject to sum(a)=1, a>=0.
    Uses the exact subgradient of the max function.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    rng = np.random.default_rng(seed)
    lr = lr_init

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        # Active set: indices where b is close to max
        tol = 0.05 * b_max
        active = np.where(b >= b_max - tol)[0]

        # Subgradient: sg[i] = (1/len(active)) * sum_{k in active} 2*a[k-i]
        sg = np.zeros(n)
        for k in active:
            # 2*a[k-i] for i=0..n-1, where a[x]=0 if x out of [0,n-1]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                sg[i] += 2.0 * a[k - i]
        sg /= len(active)

        # Projected subgradient step
        a_new = a - lr * sg
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            # Decay learning rate
            lr *= lr_decay

        a = a_new

    return best_a, best_obj


def sca_lp_optimize(a, max_iters, deadline, verbose=False):
    """
    SCA-LP optimization with adaptive step size.
    At each iteration, solve LP to find best response direction,
    then use backtracking line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # LP: minimize t s.t. 2*(a*g)[k] - c[k] <= t, sum(g)=1, g>=0
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=np.array(b_ub), A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 30, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs

                    # Backtracking line search
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 25:
            break

    return best_a, best_obj

        # (w * a_rev)[m] = sum_j w[j] * a_rev[m-j] = sum_j w[j] * a[n-1-(m-j)] = sum_j w[j] * a[n-1-m+j]
        # Set m = n-1+i: sum_j w[j] * a[n-1-(n-1-i)+j] = sum_j w[j] * a[i+j]
        # That's not right either.

        # Let me use a different approach. Define h[j] = w[n-1-j] (reversed w).
        # Then (h * a)[k] = sum_j h[j]*a[k-j] = sum_j w[n-1-j]*a[k-j]
        # Let i' = n-1-j: = sum_{i'} w[i']*a[k-n+1+i']
        # Set k-n+1+i' = k': i' = k'-k+n-1
        # = sum_{k'} w[k'-k+n-1]*a[k']
        # For this to equal sum_k w[k]*a[k-i], we need k'-k+n-1 = k and k' = k-i... 
        # This is getting confusing. Let me just use direct computation for the gradient.

        # Actually, the gradient of (1/tau) * log(sum exp(tau * b[k])) w.r.t. a[i] is:
        # sum_k w[k] * db[k]/da[i] = sum_k w[k] * 2*a[k-i]
        # (where a[k-i] = 0 if k-i not in [0,n-1])

        # This can be computed as: 2 * (w * a_rev)[n-1 + i]
        # where (w * a_rev)[m] = sum_j w[j] * a_rev[m-j] and a_rev[j] = a[n-1-j]
        # = sum_j w[j] * a[n-1-(m-j)] = sum_j w[j] * a[n-1-m+j]
        # For m = n-1-i: sum_j w[j] * a[n-1-(n-1-i)+j] = sum_j w[j] * a[i+j]
        # That gives sum_j w[j]*a[i+j], not what we want.

        # Let me try: we want sum_k w[k]*a[k-i]. Let p = k-i, so k = p+i.
        # = sum_p w[p+i]*a[p]
        # This is (rev(w[i:]) * a)[0] ... no, it's a cross-correlation.
        # = cross_correlation(w, a)[i] where cross_correlation(w,a)[i] = sum_p w[p+i]*a[p]

        # Alternatively, this equals (w * a_rev)[n-1-i]:
        # (w * a_rev)[n-1-i] = sum_j w[j]*a_rev[n-1-i-j] = sum_j w[j]*a[n-1-(n-1-i-j)]
        # = sum_j w[j]*a[i+j]
        # Still not matching.

        # OK let me just do it directly with a convolution trick:
        # sum_k w[k]*a[k-i] = (a * w_rev)[i] where w_rev[k] = w[n-1-k]... 
        # Hmm, let me think differently.
        # sum_k w[k]*a[k-i]: let j = k-i, then k = j+i
        # = sum_j w[j+i]*a[j]
        # Range: j from max(0, -(n-1-i)) to min(n-1, n-1-i) = min(n-1, n-1-i)
        # Actually j from 0 to n-1, and w[j+i] is only non-zero when j+i is in [0, 2n-2]
        # (since w has length 2n-1)
        
        # This is the "cross-correlation" of w and a at lag i.
        # cross_corr[w,a](i) = sum_j w[j+i] * a[j]
        # = (w * flip(a))[n-1+i] ... let me verify
        # flip(a)[k] = a[n-1-k]
        # (w * flip(a))[m] = sum_j w[j] * flip(a)[m-j] = sum_j w[j] * a[n-1-m+j]
        # Set m = 2n-2-i: sum_j w[j] * a[n-1-(2n-2-i)+j] = sum_j w[j] * a[i-1+j+1] = sum_j w[j]*a[i+j]
        # Hmm still not quite.

        # Let me just use the simple approach: for each i, compute sum_k w[k]*a[k-i]
        # by taking the appropriate slice of w.
        # For a fixed i: valid k are those where k-i is in [0,n-1], i.e., k in [i, i+n-1]
        # grad_raw[i] = 2 * sum_{k=i}^{i+n-1} w[k] * a[k-i] = 2 * dot(w[i:i+n], a)
        
        # This can be computed for all i simultaneously using a "diagonal" trick or just
        # iterating (which is O(n^2) but fast for n=768).

        # For n=768, O(n^2) = ~589K operations per iteration, which is fast enough.
        grad_raw = np.zeros(n)
        for i in range(n):
            k_start = i
            k_end = i + n
            if k_end > 2 * n - 1:
                k_end = 2 * n - 1
            grad_raw[i] = 2.0 * np.dot(w[k_start:k_end], a[:k_end - k_start])

        # Project to simplex tangent space (remove mean)
        grad = grad_raw - np.mean(grad_raw)

        # Adam update
        t_count += 1
        lr = lr_init / (1.0 + 0.05 * np.sqrt(t_count))
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Restart with perturbation
                a = best_a + rng.normal(0, 0.02, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for noise_level in [0.02, 0.05, 0.1, 0.2]:
            inits.append(project_simplex(p + rng.normal(0, noise_level, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns (various frequencies and amplitudes)
    for freq in [1, 2, 3, 4, 5, 7]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    # Polynomial shapes
    for p1 in [0.5, 1.0, 2.0, 3.0]:
        for p2 in [0.5, 1.0, 2.0, 3.0]:
            inits.append(project_simplex((t + 0.01) ** p1 * ((1 - t) + 0.01) ** p2))

    # Gaussian bumps
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.05, 0.1, 0.2, 0.3]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.02))

    # Double bump
    for c1 in [0.2, 0.3]:
        for c2 in [0.7, 0.8]:
            inits.append(project_simplex(
                np.exp(-((t - c1) / 0.15) ** 2) + np.exp(-((t - c2) / 0.15) ** 2) + 0.02))

    # Smooth random
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(5):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    # Alternating
    inits.append(project_simplex(1.0 + 0.3 * np.sign(np.sin(2 * np.pi * 3 * t))))
    inits.append(project_simplex(1.0 + 0.5 * np.cos(4 * np.pi * t) * np.exp(-2 * (t - 0.5)**2)))

    return inits


def optimize_single(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 1: Coordinate descent (35% of time)
    t1 = time.monotonic() + 0.35 * remaining
    a = cd_fast(a, max_sweeps=20000, deadline=t1, random_order=True, seed=42)
    try_update(a)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 2: SCA-LP (30% of time)
    t2 = time.monotonic() + 0.30 * remaining
    a_sca, obj_sca = sca_lp_optimize(a, max_iters=200, deadline=t2)
    if obj_sca < best_obj:
        best_a = a_sca.copy()
        best_obj = obj_sca
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 3: Adam (20% of time)
    t3 = time.monotonic() + 0.20 * remaining
    a_adam, obj_adam = smooth_adam(a, max_iters=5000, deadline=t3,
                                    tau_init=15.0, tau_end=120.0, lr_init=0.04)
    if obj_adam < best_obj:
        best_a = a_adam.copy()
        best_obj = obj_adam
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 4: Final CD polish (15% of time)
    t4 = deadline - 3
    a = cd_fast(a, max_sweeps=10000, deadline=t4, random_order=True, seed=123)
    try_update(a)

    return best_a, best_obj
