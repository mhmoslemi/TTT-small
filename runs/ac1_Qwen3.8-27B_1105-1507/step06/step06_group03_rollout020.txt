    """Generate a non-negative sequence with approximately flat spectrum."""
    a = np.ones(n) / n  # start with uniform
    for it in range(n_iters):
        # Forward DFT
        A = np.fft.fft(a)
        # Set flat magnitude (excluding DC)
        target_mag = np.mean(np.abs(A[1:]))
        A_new = np.zeros_like(A)
        A_new[0] = A[0]  # keep DC (sum = 1)
        for k in range(1, n):
            # Maintain phase, set magnitude
            if abs(A[k]) > 1e-15:
                A_new[k] = A[k] / abs(A[k]) * target_mag
            else:
                A_new[k] = target_mag  # random phase
        a = np.fft.ifft(A_new).real
        # Project onto simplex
        a = np.maximum(a, 0)
        a = a / a.sum()
    return a

"""
Optimizes J(a) = 2n*max(conv(a,a))/sum(a)^2 using:
1. Spectral (POCS) initialization for flat-spectrum sequences
2. Adam with FFT-based smooth gradient (log-sum-exp surrogate for max)
3. Periodic LP subproblems (best-response direction)
4. Coordinate descent polish
Multi-length exploration with adaptive time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def spectral_init(n, n_iters=100):
    """
    Generate non-negative sequence with approximately flat spectrum
    using POCS (projection onto convex sets) / Gerchberg-Saxton.
    """
    a = np.ones(n) / n
    for it in range(n_iters):
        A = np.fft.fft(a)
        # Target: flat magnitude (excluding DC)
        mags = np.abs(A)
        # Exclude DC (index 0) and the DC of the real signal
        target_mag = np.mean(mags[1:n//2 + 1])
        # Set magnitudes
        for k in range(1, n):
            if mags[k] > 1e-15:
                A[k] = A[k] / mags[k] * target_mag
            else:
                A[k] = target_mag
        A[0] = 1.0  # sum = 1
        a = np.fft.ifft(A).real
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a /= s
    return a


def compute_grad(a, tau):
    """
    Compute gradient of smooth surrogate J_smooth(a) = (2n/tau)*log(sum exp(tau*b_k))
    using FFT. Returns the gradient projected onto simplex tangent space.
    """
    n = len(a)
    b = fast_conv(a)
    b_shifted = b - b.max()
    exp_b = np.exp(tau * b_shifted)
    p = exp_b / exp_b.sum()
    
    size = 1
    while size < 3 * n:
        size <<= 1
    
    P = np.fft.fft(p, size)
    A = np.fft.fft(a, size)
    grad_full = np.fft.ifft((np.conj(P) + P) * A, size)
    grad = grad_full.real[:n]
    
    # Project to simplex tangent space
    grad = 2.0 * n * grad
    grad -= np.mean(grad)
    
    return grad, b.max()


def adam_smooth(a, max_iters, deadline, tau_init=50.0, tau_end=300.0,
                lr=0.1, verbose=False):
    """
    Adam optimization with log-sum-exp smooth gradient.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal tau
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_init + (tau_end - tau_init) * frac
        
        # Compute gradient
        grad, M = compute_grad(a, tau)
        
        # Adam update
        t += 1
        lr_t = lr / (1.0 + 0.1 * np.sqrt(t))
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        obj = compute_J(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Restart
                a = best_a + np.random.default_rng(it * 137).normal(0, 0.03, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_improve = 0
        
        a = a_new
        
        if verbose and it % 200 == 0:
            print(f"      Adam it {it}: tau={tau:.0f} J={best_obj:.8f}", flush=True)
    
    return best_a, best_obj


def lp_best_response(a, deadline, time_limit=5):
    """
    Solve the LP: maximize sum(g)
    subject to: (a*g)[k] <= M for all k
                g >= 0
    where M = max((a*a)[k])
    
    Returns the normalized best response direction g, or None on failure.
    """
    n = len(a)
    a = np.maximum(a, 0)
    b = fast_conv(a)
    M = b.max()
    
    if M < 1e-10:
        return None
    
    # Build constraint matrix: (a * g)[k] <= M
    # (a * g)[k] = sum_{j} a[j] * g[k-j]
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c_obj = -np.ones(n)  # maximize sum(g) = minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                    method='highs', options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    
    return None


def coord_descent(a, max_sweeps, deadline, random_order=True, seed=42):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        improved = False
        for i in order:
            old = a[i]
            
            # Try several candidate values
            candidates = np.array([0.0, 0.25*old, 0.5*old, 0.75*old,
                                   1.25*old, 1.5*old, 2.0*old, np.inf])
            # Replace inf with a value that makes a[i] the largest
            candidates = [min(c, 10.0) for c in candidates]
            
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i: hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            
            best_val = old
            best_local = 2.0 * n * conv.max()
            
            for x in candidates:
                x = float(max(0.0, min(x, 10.0 / (n - 1))))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                
                new_obj = 2.0 * n * conv.max()
                if new_obj < best_local - 1e-14:
                    best_local = new_obj
                    best_val = x
                
                conv[lo:hi+1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
            
            if best_val != old:
                delta = best_val - old
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                improved = True
        
        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            break
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 0)
        s = p.sum()
        if s > 1e-10:
            p = p / s
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.15, n)))
    
    # Spectral init
    inits.append(spectral_init(n, 100))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine patterns
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3]:
        inits.append(project_simplex(1.0 + 0.5 * np.cos(2 * np.pi * freq * t)))
        inits.append(project_simplex(1.0 + 0.3 * np.sin(2 * np.pi * freq * t)))
    
    # Random smooth
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(project_simplex(smooth + 0.1))
    
    return inits


def optimize_sequence(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence:
    1. Adam with smooth gradient
    2. LP best-response
    3. More Adam
    4. Coordinate descent
    5. LP + CD final
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_J(a)
    
    def try_update(a_new):
        nonlocal best_a, best_obj
        if a_new is None:
            return
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            return
        a_new = a_new / s
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
    
    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj
    
    # Stage 1: Adam (40% of time)
    t1 = time.monotonic() + 0.40 * remaining
    a_adam, obj_adam = adam_smooth(a, max_iters=50000, deadline=t1,
                                    tau_init=30, tau_end=200, lr=0.1)
    try_update(a_adam)
    a = best_a.copy()
    
    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj
    
    # Stage 2: LP + Adam alternating (35% of time)
    t2 = time.monotonic() + 0.35 * remaining
    while time.monotonic() < t2:
        # LP step
        t_lp = time.monotonic() + min(30, (t2 - time.monotonic()) * 0.3)
        g = lp_best_response(best_a, deadline=t_lp, time_limit=20)
        if g is not None:
            for frac in [0.5, 0.3, 0.2, 0.1]:
                a_try = (1 - frac) * best_a + frac * g
                try_update(a_try)
        
        # Adam step
        t_adam = time.monotonic() + min(
            (t2 - time.monotonic()) * 0.7, 60)
        if t_adam > time.monotonic() + 5:
            a_adam, _ = adam_smooth(best_a, max_iters=5000, deadline=t_adam,
                                     tau_init=50, tau_end=200, lr=0.05)
            try_update(a_adam)
    
    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj
    
    # Stage 3: Coordinate descent (25% of time)
    t3 = time.monotonic() + 0.25 * remaining
    a_cd = coord_descent(best_a, max_sweeps=50000, deadline=t3, random_order=True, seed=123)
    try_update(a_cd)
    
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj
    
    # Stage 4: Final LP + CD (10% of time)
    t4 = time.monotonic() + 0.05 * remaining
    g = lp_best_response(best_a, deadline=t4, time_limit=15)
    if g is not None:
        for frac in [0.5, 0.3, 0.2]:
            a_try = (1 - frac) * best_a + frac * g
            try_update(a_try)
    
    if time.monotonic() < deadline - 5:
        a_final = coord_descent(best_a, max_sweeps=20000,
                                deadline=deadline - 5, random_order=True, seed=999)
        try_update(a_final)
    
    return best_a, best_obj
