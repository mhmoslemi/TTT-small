"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: LP alternating with precomputed-convolution line search.
Secondary: subgradient with adaptive step, spectral flattening.
Explores multiple sequence lengths.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def convolve_pair(a, b):
    """Linear convolution a*b via FFT."""
    n_a, n_b = len(a), len(b)
    size = 1
    while size < n_a + n_b - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n_a + n_b - 1]


def compute_obj(a):
    """2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0.
    Shape: (2n-1, n)"""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_minpeak_lp(a, time_limit=10):
    """Solve: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None)."""
    n = len(a)
    if n > 3000:
        return None, None
    
    A = build_conv_matrix(a)
    # Decision variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None

    A = build_conv_matrix(a)

def fast_line_search(a, g, n_t=40):
    """Efficient line search using precomputed convolutions.
    a_t = (1-t)*a + t*g, both sum to 1.
    Returns best (t, a_t, obj_t)."""

    conv_aa = compute_conv(a)

    conv_gg = compute_conv(g)
    
    best_obj = np.inf
    best_t = 0.0

def spectral_flatten(a, time_budget=20.0):
    """Flatten the power spectrum to reduce convolution peaks."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = np.fft.fft(a, N)
        power = np.abs(fa)**2
        p_max = np.max(power)
        if p_max < 1e-20:
            break
        
        # Compress the power spectrum
        ps_norm = power / p_max
        # Apply power law compression (alpha < 1 compresses peaks)
        alpha = 0.8 - 0.3 * (it / 1000)  # gradually reduce compression
        ps_new = np.maximum(ps_norm, 1e-10) ** alpha
        ps_new = ps_new / np.sum(ps_new) * np.sum(power)
        
        # Reconstruct with same phase
        phase = np.angle(fa)
        magnitude = np.sqrt(ps_new)
        fa_new = magnitude * np.exp(1j * phase)
        
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x, a)


def chase_peak_fft(a, n_steps=100, step_size=0.02):
    """Iteratively chase the peak using FFT-computed gradients."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for step in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of max conv at a
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
            j2 = i + (2*n - 2) - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]
        
        grad_centered = grad - np.mean(grad)
        
        # Try a few step sizes
        improved = False
        for ss in [step_size, step_size*0.5, step_size*0.25]:
            a_try = project_simplex(a - ss * grad_centered)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_obj

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sine
    x = np.linspace(0, 1, n)
    s = np.sin(np.pi * x)**2 + 0.05
    inits.append(s / s.sum())
    
    # Gaussian
    x = np.linspace(-3, 3, n)
    g = np.exp(-x**2 / 3)
    inits.append(g / g.sum())
    
    # Bimodal
    x = np.linspace(0, 1, n)
    b = np.exp(-((x-0.25)**2)/0.01) + np.exp(-((x-0.75)**2)/0.01)
    inits.append(b / b.sum())
    
    # Random smooth
    t = rng.random(n)
    # Smooth it
    kernel = np.ones(5) / 5
    t_smooth = np.convolve(t, kernel, mode='same')
    inits.append(project_simplex(t_smooth))
    
    # Parent resampled
    if parent is not None:
        inits.append(resample(parent, n))
        # Perturbed parent
        p = resample(parent, n)
        p = p * np.exp(rng.normal(0, 0.15, n))
        inits.append(project_simplex(p))
    
    return inits

"""
LP-alternation with FFT-accelerated line search, subgradient polishing,
spectral flattening, and multi-length exploration to minimize
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements: precomputed-convolution line search (quadratic in t),
aggressive LP iteration budget, adaptive step sizes, reduced-constraint LP
for speed, and random restarts to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Linear self-convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def convolve_pair(a, b):
    """Linear convolution a*b via FFT."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]


def compute_obj(a):
    """2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Matrix A of shape (2n-1, n) where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_minpeak_lp(a, time_limit=8.0, reduced_k=0):
    """
    Solve: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    If reduced_k > 0, only use the top-k most constraining rows.
    Returns (g, M_opt) or (None, inf).
    """
    n = len(a)
    if n > 3000:
        return None, np.inf

    A = build_conv_matrix(a)
    n_rows = 2 * n - 1

    if reduced_k > 0 and reduced_k < n_rows:
        # Estimate which rows are most constraining using current a as proxy
        # Row k is constraining if A[k,:] @ a is large (i.e., conv(a,a)[k] is large)
        conv_a = A @ a
        top_idx = np.argsort(conv_a)[-reduced_k:]
        A = A[top_idx, :]
        n_rows = reduced_k

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((n_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_rows)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, np.inf


def solve_maxsum_lp(a, M, time_limit=8.0):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g>=0. Returns normalized g."""
    n = len(a)
    if n > 3000:
        return None

    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def fast_line_search(a, g, n_t=50):
    """
    Efficient line search: a_t = (1-t)*a + t*g, both sum to 1.
    Uses precomputed convolutions to evaluate objective in O(1) per t.
    Returns (best_t, best_a, best_obj).
    """
    n = len(a)
    conv_aa = compute_conv(a)
    conv_ag = convolve_pair(a, g)
    conv_gg = compute_conv(g)

    best_obj = np.inf
    best_t = 0.0
    best_a = a.copy()

    ts = np.linspace(0.0, 1.0, n_t)
    for t in ts:
        conv_at = (1.0 - t) ** 2 * conv_aa + 2.0 * t * (1.0 - t) * conv_ag + t ** 2 * conv_gg
        obj = 2.0 * n * float(np.max(conv_at))
        if obj < best_obj:
            best_obj = obj
            best_t = float(t)
            best_a = ((1.0 - t) * a + t * g).copy()

    # Also try a few t > 1 extrapolation points (with direct computation)
    for t in [1.2, 1.5, 2.0]:
        a_try = (1.0 - t) * a + t * g
        a_try = project_simplex(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = float(t)
            best_a = a_try.copy()

    return best_t, best_a, best_obj


def lp_alternate(a, time_budget=300.0, max_lp_time=5.0, verbose=True):
    """
    LP alternating with fast line search and adaptive strategy.
    Alternates between min-peak LP and max-sum LP.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iteration = 0
    last_lp_type = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < max_lp_time + 2:
            break

        iteration += 1

        # Alternate between LP types
        lp_type = (iteration // 3) % 2
        if iteration % 5 == 0:
            # Every 5th iteration, use reduced LP for speed
            reduced = max(200, n)
        else:
            reduced = 0

        if lp_type == 0:
            g, M_opt = solve_minpeak_lp(a, time_limit=max_lp_time, reduced_k=reduced)
        else:
            conv_a = compute_conv(a)
            M_current = float(np.max(conv_a))
            # Use slightly tighter bound to force improvement
            M_target = M_current * 0.995
            g = solve_maxsum_lp(a, M_target, time_limit=max_lp_time)

        if g is None:
            # Try the other LP type
            if lp_type == 0:
                g = solve_maxsum_lp(a, float(np.max(compute_conv(a))), time_limit=max_lp_time)
            else:
                g, M_opt = solve_minpeak_lp(a, time_limit=max_lp_time)
            if g is None:
                break

        # Fast line search
        t_best, a_new, obj_new = fast_line_search(a, g, n_t=40)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            # Try smaller blend
            g_half = 0.5 * g + 0.5 * a
            g_half = project_simplex(g_half)
            t2, a2, obj2 = fast_line_search(a, g_half, n_t=20)
            if obj2 < best_obj - 1e-12:
                best_obj = obj2
                best_a = a2.copy()
                a = a2.copy()
                stall = 0
            else:
                stall += 1
                if stall > 8:
                    # Random perturbation to escape
                    a = a * np.exp(np.random.default_rng(iteration).normal(0, 0.1, n))
                    a = project_simplex(a)
                    obj_now = compute_obj(a)
                    if obj_now < best_obj:
                        best_obj = obj_now
                        best_a = a.copy()
                    stall = 0

        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{iteration}: obj={best_obj:.8f}, t={t_best:.3f}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def subgradient_optimize(a, time_budget=30.0, verbose=False):
    """Subgradient with softmax smoothing and momentum."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    lr = 0.03
    momentum = 0.9

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        tau = 0.0005 + 0.005 * (it / 5000.0)
        shifted = np.clip((conv - cmax) / tau, -200.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # grad[j] = 2 * sum_m a[m] * w[j+m]
        grad = np.zeros(n)
        w_len = len(w)
        for j in range(n):
            m_max = min(n, w_len - j)
            if m_max > 0:
                grad[j] = 2.0 * np.dot(a[:m_max], w[j:j + m_max])

        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = project_simplex(a - v)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(lr * 1.01, 0.15)
        else:
            lr *= 0.8
            if lr < 0.0005:
                break

    return best_a


def spectral_flatten(a, time_budget=20.0):
    """Iteratively flatten the power spectrum to reduce time-domain peaks."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        p_max = np.max(power)
        if p_max < 1e-20:
            break

        ps_norm = power / p_max
        alpha = 0.75 - 0.35 * (it / 2000.0)
        ps_new = np.maximum(ps_norm, 1e-12) ** alpha
        ps_new *= np.sum(power) / np.sum(ps_new)

        phase = np.angle(fa)
        magnitude = np.sqrt(np.maximum(ps_new, 0))
        fa_new = magnitude * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a


def chase_peak_fft(a, n_steps=50, step_size=0.02):
    """Chase the maximum convolution peak using subgradient steps."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)

    for step in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += a[j]
            j2 = i + (2 * n - 2) - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]

        grad_centered = grad - np.mean(grad)

        improved = False
        for ss in [step_size, step_size * 0.5, step_size * 0.25, step_size * 2.0]:
            a_try = project_simplex(a - ss * grad_centered)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_obj


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n)
    x = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x, a)

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Sine
    x = np.linspace(0, 1, n)
    s = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(s / s.sum())

    # Gaussian
    x = np.linspace(-3, 3, n)
    g = np.exp(-x ** 2 / 3.0)
    inits.append(g / g.sum())

    # Bimodal
    x = np.linspace(0, 1, n)
    b = np.exp(-((x - 0.25) ** 2) / 0.01) + np.exp(-((x - 0.75) ** 2) / 0.01)
    inits.append(b / b.sum())

    # Random smoothed
    t = rng.random(n)
    kernel = np.ones(7) / 7
    t_sm = np.convolve(t, kernel, mode='same')
    inits.append(project_simplex(t_sm))

    # Parent
    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n) * np.exp(rng.normal(0, 0.2, n))
        inits.append(project_simplex(p))

    return inits
