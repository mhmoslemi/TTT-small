Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses logsumexp smoothing with adaptive beta, true-max subgradient,
and coordinate descent on active peaks.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def softmax_and_smoothed_obj_grad(x, beta):
    """
    Given log-space x, compute a = softmax(x), the smoothed objective,
    and the gradient w.r.t. x.
    """
    n = len(x)
    # Compute a = softmax(x)
    x_shift = x - np.max(x)
    exp_x = np.exp(x_shift)
    a = exp_x / np.sum(exp_x)
    
    # Compute convolution
    conv = compute_conv(a)
    
    # Logsumexp
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)

    e = np.exp(shifted)
    e_sum = float(np.sum(e))

    # Smoothed objective (since sum(a) = 1): obj = 2*n*ls
    obj = 2.0 * n * ls
    
    # Gradient w.r.t. a: d(obj)/d(a[i]) = 4*n*h[i]

    # = a[j] * (grad_a[j] - sum_i grad_a[i]*a[i])

    return obj, grad_x, a, w, conv

def true_max_grad(x):

    Compute the (sub)gradient of the true objective at the given point.
    Returns (obj, grad_x, a, conv, peak_indices)
    """

def simplex_project(z):
    """Project z onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(z)
    # Sort in descending order
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0)

def resample(a, new_n):

    n = len(a)

    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
