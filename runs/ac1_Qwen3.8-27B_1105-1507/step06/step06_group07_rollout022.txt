def smooth_obj_grad(a, beta):
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def adam_log_opt(a, max_iters=20000, time_budget=100.0, beta_start=5.0,

    v = np.zeros(n)
    t = 0
    eps = 1e-8
    best_a = a.copy()

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)
        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a
        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m = momentum * m + (1 - momentum) * grad_x
        v = vel_mom * v + (1 - vel_mom) * (grad_x * grad_x)
        m_hat = m / (1 - momentum ** t)
        v_hat = v / (1 - vel_mom ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            i = rng.integers(i_min, i_max + 1)
            j_other = p - i
            if j_other < 0 or j_other >= n:
                continue
            # Try to move mass from the larger contributor to a random position
            if best_a[i] > best_a[j_other] and best_a[i] > 1e-10:
                src = i
            else:
                src = j_other
            dst = rng.integers(0, n)
            if dst == src:
                continue
            frac = rng.uniform(0.01, 0.08)
            delta = frac * best_a[src]
            a_try = best_a.copy()
            a_try[src] -= delta
            a_try[dst] += delta
            if a_try[src] < 0:
                continue
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                conv = compute_conv(best_a)
                top_peaks = np.argsort(conv)[::-1][:8]
                improved = True

        if not improved and it > 200:
            break

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def make_oscillating_init(n, freq=1.0, amp=0.3, seed=0):
    """Create an oscillating non-negative initialization."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    base = 1.0 + amp * np.cos(2 * np.pi * freq * x / n)
    # Add a second harmonic for more structure
    base += 0.15 * amp * np.cos(4 * np.pi * freq * x / n + 0.5)
    base = np.maximum(base, 0.01)
    base /= base.sum()
    # Add small noise
    base = base * (1 + 0.02 * rng.normal(0, 1, n))
    base = np.maximum(base, 0.001)
    base /= base.sum()
    return base


def make_window_init(n, kind='hann'):
    """Create a windowed initialization."""
    x = np.arange(n) / max(n - 1, 1)
    if kind == 'hann':
        w = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == 'hamming':
        w = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    elif kind == 'u_shape':
        w = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif kind == 'cosine_sq':
        w = np.cos(np.pi * (x - 0.5)) ** 2
    else:
        w = np.ones(n)
    w = np.maximum(w, 0.001)
    w /= w.sum()
    return w


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining LP refinement, Adam log-space optimization,
    multi-initialization, multi-n exploration, and local search.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, budget={budget_s}s")

    # === Phase 1: LP iterative refinement (250s) ===
    print(f"\n{'=' * 60}")
    print(f"Phase 1: LP iterative refinement (n={n_parent})")
    print(f"{'=' * 60}")

    if best_seq is not None and len(best_seq) <= 900:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 250.0)
        a_lp = lp_iterate_refine(best_seq, time_budget=alloc, lp_time_each=20.0)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  After LP: {best_score:.8f} ***")
        else:
            print(f"  After LP: {obj_lp:.8f} (no improvement)")

    # === Phase 2: Adam log-space from best (300s) ===
    print(f"\n{'=' * 60}")
    print(f"Phase 2: Adam log-space optimization (n={len(best_seq) if best_seq is not None else '?'})")
    print(f"{'=' * 60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        adam_rounds = [
            (10.0, 80.0, 0.04, 0.99998),
            (30.0, 150.0, 0.03, 0.99999),
            (80.0, 400.0, 0.02, 0.99999),
            (150.0, 600.0, 0.015, 0.99999),
            (300.0, 1000.0, 0.01, 0.99999),
            (500.0, 2000.0, 0.008, 0.99999),
        ]

        for r_idx, (b_start, b_end, lr0, lr_dec) in enumerate(adam_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.18, 60.0)
            a_new = adam_log_opt(a, max_iters=30000, time_budget=alloc,
                                 beta_start=b_start, beta_end=b_end,
                                 lr_start=lr0, lr_decay=lr_dec)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  Round {r_idx} (beta {b_start:.0f}-{b_end:.0f}): {best_score:.8f} ***")
            else:
                print(f"  Round {r_idx} (beta {b_start:.0f}-{b_end:.0f}): {obj:.8f}")

    # === Phase 3: Multi-initialization Adam (300s) ===
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Multi-initialization exploration")
    print(f"{'=' * 60}")

    if best_seq is not None:
        n_curr = len(best_seq)
        n_inits = max(200, min(n_curr, 800))

        initializations = []
        # Oscillating inits with different frequencies and amplitudes
        for freq, amp, sd in [(1.0, 0.3, 0), (2.0, 0.2, 1), (1.5, 0.4, 2),
                              (3.0, 0.15, 3), (0.5, 0.5, 4), (2.5, 0.25, 5)]:
            initializations.append(make_oscillating_init(n_inits, freq, amp, sd))
        # Window inits
        initializations.append(make_window_init(n_inits, 'hann'))
        initializations.append(make_window_init(n_inits, 'u_shape'))
        initializations.append(make_window_init(n_inits, 'cosine_sq'))
        # Resampled parent
        initializations.append(resample_sequence(best_seq, n_inits) / resample_sequence(best_seq, n_inits).sum())

        for init_idx, a_init in enumerate(initializations):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.1, 25.0)
            a_opt = adam_log_opt(a_init, max_iters=15000, time_budget=alloc,
                                 beta_start=20.0, beta_end=300.0,
                                 lr_start=0.04, lr_decay=0.99997)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Init {init_idx} (n={n_inits}): {best_score:.8f} *** UPDATE")
            else:
                print(f"  Init {init_idx} (n={n_inits}): {obj:.8f}")

    # === Phase 4: Multi-n exploration (250s) ===
    print(f"\n{'=' * 60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'=' * 60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 50, n_best, n_best + 50, n_best + 100,
            n_best + 200, n_best + 300, n_best + 500
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2000]
        # Put n_best in the middle
        if n_best in n_options:
            n_options.remove(n_best)
        mid = len(n_options) // 2
        n_options = n_options[:mid + 1] + [n_best] + n_options[mid + 1:]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_try = resample_sequence(best_seq, n_try)
            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s

            alloc = min(remaining * 0.12, 35.0)
            a_opt = adam_log_opt(a_try, max_iters=20000, time_budget=alloc,
                                 beta_start=20.0, beta_end=500.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 5: Local search + final polish (200s) ===
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Local search and final polish")
    print(f"{'=' * 60}")

    if best_seq is not None:
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        # LP refinement on the best
        remaining = deadline - time.monotonic()
        if remaining > 30 and len(a) <= 900:
            alloc = min(remaining * 0.2, 60.0)
            a_lp = lp_iterate_refine(a, time_budget=alloc, lp_time_each=15.0)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a = a_lp.copy()
                print(f"  Final LP: {best_score:.8f} ***")

        # Local search
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.2, 40.0)
            a_ls = local_search_refine(a, time_budget=alloc, seed=seed)
            obj_ls = compute_obj(a_ls)
            if obj_ls < best_score:
                best_score = obj_ls
                best_seq = a_ls.copy()
                print(f"  Local search: {best_score:.8f} ***")

        # Final high-beta Adam
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a = best_seq.copy()
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            alloc = min(remaining * 0.4, 30.0)
            a_final = adam_log_opt(a, max_iters=20000, time_budget=alloc,
                                   beta_start=300.0, beta_end=3000.0,
                                   lr_start=0.005, lr_decay=0.99999)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final Adam: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'=' * 60}")

    return result
