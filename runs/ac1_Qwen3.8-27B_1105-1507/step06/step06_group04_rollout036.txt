"""
Multi-method search for minimum autocorrelation peak sequences.
Combines active-set LP, QP energy minimization, spectral flattening,
and smoothed subgradient methods with multi-start exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def normalize_simplex(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        return a / s
    return np.ones(len(a)) / len(a)


def build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = (conv(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        A[j:j + length, j] = a[:length]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=10):
    """
    Solve: min t s.t. (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(fixed_seq)
    A = build_conv_matrix(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_qp_energy(fixed_seq, max_iters=200):
    """
    Solve: min g^T A^T A g s.t. sum(g)=1, g>=0
    using active-set method with conjugate gradient.
    Returns normalized g or None.
    """
    n = len(fixed_seq)
    A = build_conv_matrix(fixed_seq)
    Q = A.T @ A  # n x n, symmetric positive semi-definite
    
    # Active set method
    active = np.zeros(n, dtype=bool)  # True = fixed at 0
    for outer_iter in range(max_iters):
        free = ~active
        n_free = np.sum(free)
        if n_free == 0:
            break
        
        # Solve Q_ff * g_f = c * 1, 1^T g_f = 1
        Q_ff = Q[np.ix_(free, free)]
        # Add small regularization for numerical stability
        Q_ff = Q_ff + 1e-10 * np.eye(n_free)
        
        try:
            x = np.linalg.solve(Q_ff, np.ones(n_free))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(Q_ff, np.ones(n_free), rcond=None)[0]
            except:
                break
        
        denom = np.sum(x)
        if abs(denom) < 1e-15:
            break
        g_f = x / denom
        
        # Check non-negativity
        if np.all(g_f >= -1e-10):
            g = np.zeros(n)
            g[free] = np.maximum(g_f, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
            break
        
        # Some components are negative - fix the most negative at 0
        min_idx = np.argmin(g_f)
        active[np.where(free)[0][min_idx]] = True
    
    return None


def smoothed_subgradient(a, tau, max_iters=500, time_budget=30.0, step=0.05, mu=0.8):
    """
    Subgradient method with softmax-smoothed max.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= wsum
        
        # Gradient via FFT
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]
        grad_centered = grad - np.mean(grad)
        gn = np.linalg.norm(grad_centered)
        if gn < 1e-15:
            break
        
        v = mu * v - step * grad_centered / gn
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        step *= 0.999
        tau *= 0.999
    
    return best_a, best_obj


def spectral_flatten(a, n_iters=50, alpha=0.5):
    """
    Spectral domain flattening: iteratively adjust the amplitude spectrum
    to be more flat (reduce peaks in autocorrelation).
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    for it in range(n_iters):
        # Compute spectrum
        A_spec = rfft(a, fft_size)
        power = np.abs(A_spec) ** 2
        mean_power = np.mean(power[1:])  # exclude DC
        
        if mean_power < 1e-20:
            break
        
        # Flatten: reduce peaks, boost valleys
        ratio = mean_power / (power + 1e-20)
        ratio = np.clip(ratio, 0.1, 10.0)
        
        # Adjust amplitude (preserve phase)
        new_amplitude = np.abs(A_spec) * np.sqrt(alpha + (1 - alpha) * ratio)
        new_phase = np.angle(A_spec)
        A_new = new_amplitude * np.exp(1j * new_phase)
        
        # Reconstruct
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-15:
            break
        a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller alpha
            alpha *= 0.8
    
    return best_a, best_obj


def line_search_blend(a, direction, obj_current, n_cands=15):
    """
    Line search: find best t in (0, 1] for (1-t)*a + t*direction.
    """
    direction = normalize_simplex(direction)
    best_t = 0.0
    best_obj = obj_current
    best_a = a.copy()
    
    # Golden section search
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    # Evaluate at a few points first
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_try = normalize_simplex((1.0 - t) * a + t * direction)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
            best_a = a_try.copy()
    
    # Refine with golden section around the best
    if best_t > 0:
        center = best_t
        width = 0.15
        lo = max(0.0, center - width)
        hi = min(1.0, center + width)
        
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        a_c = normalize_simplex((1.0 - c) * a + c * direction)
        a_d = normalize_simplex((1.0 - d) * a + d * direction)
        fc = compute_obj(a_c)
        fd = compute_obj(a_d)
        
        for _ in range(8):
            if hi - lo < 0.01:
                break
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                a_c = normalize_simplex((1.0 - c) * a + c * direction)
                fc = compute_obj(a_c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                a_d = normalize_simplex((1.0 - d) * a + d * direction)
                fd = compute_obj(a_d)
        
        t_mid = (lo + hi) / 2.0
        a_mid = normalize_simplex((1.0 - t_mid) * a + t_mid * direction)
        obj_mid = compute_obj(a_mid)
        if obj_mid < best_obj:
            best_obj = obj_mid
            best_t = t_mid
            best_a = a_mid.copy()
    
    return best_a, best_obj, best_t

        elif method == 'spec':

def generate_starts(n, num=10, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    starts = []
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Gaussian (narrow)
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 8.0) ** 2))
    starts.append(bell / np.sum(bell))
    
    # Gaussian (wide)
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 3.0) ** 2))
    starts.append(bell_w / np.sum(bell_w))
    
    # Cosine bell
    cos_bell = 0.5 * (1 + np.cos(np.pi * i / (n - 1)))
    starts.append(cos_bell / np.sum(cos_bell))
    
    # Raised cosine with modulation
    for k in range(3):
        freq = 1 + k
        mod = 1.0 + 0.3 * np.cos(2 * np.pi * freq * i / n)
        shape = cos_bell * mod
        shape = np.maximum(shape, 0)
        s = np.sum(shape)
        if s > 0:
            starts.append(shape / s)
    
    # Random smooth
    for k in range(3):
        rough = np.abs(rng.normal(0.5, 0.5, n))
        # Smooth with a moving average
        kernel = np.ones(5) / 5
        smooth = np.convolve(rough, kernel, 'same')
        s = np.sum(smooth)
        if s > 0:
            starts.append(smooth / s)
    
    return starts


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
