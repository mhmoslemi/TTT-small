Uses aggressive LP alternating with multiple M thresholds, log-sum-exp

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale-invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz_conv(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k] for k=0..2n-2."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz_conv(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_multi_M(a, time_limit_per=15):
    """Try LP at multiple M values, return best direction found."""
    conv = compute_conv(a)
    M_peak = float(np.max(conv))
    if M_peak < 1e-15:
        return None

    candidates = []
    # Try peak, 95%, 90%, 85% of peak
    for frac in [1.0, 0.95, 0.90, 0.85]:
        M = M_peak * frac
        g = solve_lp_direction(a, M, time_limit=time_limit_per)
        if g is not None:
            candidates.append((frac, g))

    if not candidates:
        return None

    # Return the most aggressive (smallest M) feasible direction
    return candidates[-1][1]


def line_search_backtrack(a, direction, obj_current, alpha_init=1.0, c=0.5):
    """Backtracking line search. Returns (t, obj_new, a_new)."""
    t = alpha_init
    best_a = a.copy()
    best_obj = obj_current
    for _ in range(20):
        a_try = normalize((1.0 - t) * a + t * direction)
        obj_try = compute_obj(a_try)
        if obj_try <= obj_current - c * t * (obj_current - obj_try) or obj_try < obj_current:
            return t, obj_try, a_try
        t *= 0.5
        if t < 1e-6:
            break
    # If backtracking fails, try golden section
    return golden_section_blend(a, direction, obj_current)


def golden_section_blend(a, direction, obj_current, max_iter=12):
    """Golden section search for optimal blend parameter t."""
    def obj_at_t(t):
        a_try = normalize((1.0 - t) * a + t * direction)
        return compute_obj(a_try)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = obj_at_t(c_pt)
    fd = obj_at_t(d_pt)
    for _ in range(max_iter):
        if hi - lo < 0.02:
            break
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = obj_at_t(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = obj_at_t(d_pt)
    t_opt = (lo + hi) / 2.0
    a_new = normalize((1.0 - t_opt) * a + t_opt * direction)
    obj_new = compute_obj(a_new)
    return t_opt, obj_new, a_new


def lp_alternate_aggressive(a, max_iters=80, time_budget=300.0, seed=0):
    """
    Aggressive LP alternating with multi-M, backtracking line search,

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stale_count = 0

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:
            break
        remaining = time_budget - elapsed
        if remaining < 10:
            break

        # Every 15 iterations, add a perturbation to escape local minima
        if it > 0 and it % 15 == 0:
            perturbation = rng.normal(0, 0.05 + 0.02 * (it // 15), n)
            a = normalize(best_a * np.exp(perturbation))
            best_obj = compute_obj(a)
            if best_obj > compute_obj(best_a):
                a = best_a.copy()
                best_obj = compute_obj(a)

        # Solve LP at multiple M thresholds
        time_per_lp = max(5, int(min(remaining * 0.4, 20)))
        direction = solve_lp_multi_M(a, time_limit_per=time_per_lp)

        if direction is None:
            # Fallback: use subgradient direction
            conv = compute_conv(a)
            k_star = int(np.argmax(conv))
            direction = np.zeros(n)
            for j in range(n):
                k2 = k_star - j
                if 0 <= k2 < n:
                    direction[j] = a[k2]
            s = float(np.sum(direction))
            if s > 1e-10:
                direction = direction / s
            else:
                break

        # Line search
        t, obj_new, a_new = line_search_backtrack(a, direction, best_obj)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

                # Stronger perturbation
                perturbation = rng.normal(0, 0.1, n)
                a = normalize(best_a * np.exp(perturbation))

def log_sum_exp_gradient(a, time_budget=120.0, seed=0):
    """
    Continuation method: minimize log-sum-exp of convolution with
    geometrically decreasing temperature. Gradient computed via FFT.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Start with high temperature (smooth), decrease geometrically
    conv0 = compute_conv(a)
    cmax0 = float(np.max(conv0))
    tau = max(cmax0 / 10.0, 1e-6)
    tau_min = max(cmax0 / (50.0 * n), 1e-10)
    tau_decay = 0.85
    step_size = 0.05
    size = 1
    while size < 3 * n - 1:
        size <<= 1

    no_improve = 0

    while time.monotonic() - start < time_budget:
        if tau < tau_min:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Soft-max weights
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient via FFT: grad[i] = 2 * sum_k w[k] * corr contribution
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Center gradient (projection onto sum=const)
        grad_centered = grad - np.mean(grad)

        # Adaptive step: try decreasing steps
        found = False
        for _ in range(10):
            a_new = normalize(a - step_size * grad_centered)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                found = True
                break
            step_size *= 0.5

        if not found:
            # Try smaller steps or perturb
            step_size *= 0.3
            if step_size < 1e-9:
                # Perturb and reset step
                rng = np.random.default_rng(hash(tau) % (2**31))
                a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))
                step_size = 0.03
                no_improve += 1
                if no_improve > 5:
                    break
        else:
            no_improve = 0

        # Decrease temperature
        tau *= tau_decay

    return best_a


def spectral_shaping(a, time_budget=60.0):
    """
    Frequency-domain shaping: iteratively modify the amplitude spectrum
    to flatten the autocorrelation, then project back to non-negative.
    Uses a softer compression than pure POCS.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Adaptive compression: more aggressive early, gentler later
        progress = it / 100.0
        alpha = 0.3 + 0.5 * progress  # starts at 0.3, ends at 0.8

        P_norm = P / p_max
        # Compress toward uniform but don't go below 0.1
        P_new = np.maximum(P_norm ** alpha, 0.05) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        # Projection: non-negativity + normalization
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend to maintain progress
            a = 0.6 * a + 0.4 * a_new
            a = normalize(a)

    return best_a


def coordinate_descent(a, time_budget=40.0, seed=0):
    """
    Targeted coordinate descent: identify peak contributors and
    redistribute mass to reduce the peak.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - start < time_budget:
        if no_improve > 100:
            # Perturb and reset
            a = normalize(best_a * np.exp(rng.normal(0, 0.04, n)))
            no_improve = 0

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = float(conv[k_star])

        # Find top contributors to the peak
        contrib_indices = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib_indices.append((a[j] * a[k2], j, k2))
        contrib_indices.sort(reverse=True)

        improved = False
        for peak_contrib, j1, j2 in contrib_indices[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            # Try moving small amounts from j1 (or j2) to other positions
            for frac in [0.02, 0.05, 0.1, 0.2]:
                for source in [j1, j2]:
                    if a[source] * frac < 1e-10:
                        continue
                    delta = a[source] * frac

                    # Target: position far from the peak
                    for _ in range(3):
                        target = int(rng.integers(0, n))
                        if target == source:
                            continue
                        # Prefer targets that would spread more uniformly
                        a_try = a.copy()
                        a_try[source] -= delta
                        a_try[target] += delta
                        a_try = normalize(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break

                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def zadoff_chu_init(n, root=1):
    """
    Generate a Zadoff-Chu-like non-negative sequence.
    ZC sequences have flat autocorrelation; we take magnitude and add DC offset.
    """
    t = np.arange(n)
    zc = np.exp(-1j * np.pi * root * t * (t + 1) / n)
    a = np.abs(zc) + 0.5  # magnitude + DC to ensure non-negativity and structure
    # Add some variation
    a = a + 0.3 * np.cos(2 * np.pi * t / n * 2) + 0.1
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase strategy:
    Phase 1: Initialization (50s)
    Phase 2: Aggressive LP alternating (650s) - main workhorse
    Phase 3: Log-sum-exp + spectral (200s)
    Phase 4: Multi-n exploration (250s)
    Phase 5: Final polish (150s)
    Reserve: 200s
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # =========================================================
    # Phase 1: Initialization
    # =========================================================
    print(f"{'='*60}")
    print(f"Phase 1: Initialization")
    print(f"{'='*60}")

    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    # Try alternative initializations
    rng = np.random.default_rng(seed)

    if best_seq is not None:
        n_base = len(best_seq)
    else:
        n_base = 700

    init_candidates = [best_seq.copy() if best_seq is not None else np.ones(n_base) / n_base]

    # Zadoff-Chu inspired
    init_candidates.append(zadoff_chu_init(n_base, root=1))
    init_candidates.append(zadoff_chu_init(n_base, root=3))

    # Random
    init_candidates.append(rng.dirichlet(np.ones(n_base) * 0.3))

    # Triangular
    t = np.linspace(0, 1, n_base)
    tri = 1.0 - np.abs(2 * t - 1)
    init_candidates.append(tri / np.sum(tri))

    # Sine-modulated
    t = np.linspace(0, 1, n_base)
    sine_mod = 0.5 + 0.5 * np.cos(2 * np.pi * t * 3)
    init_candidates.append(sine_mod / np.sum(sine_mod))

    print(f"  Evaluating {len(init_candidates)} initializations...")
    for i, cand in enumerate(init_candidates):
        cand = normalize(cand)
        obj = compute_obj(cand)
        print(f"    Init {i}: n={len(cand)}, obj={obj:.6f}")
        if obj < best_score:
            best_score = obj
            best_seq = cand.copy()

    n_best = len(best_seq)
    print(f"  Best init: n={n_best}, score={best_score:.8f}")

    # =========================================================
    # Phase 2: Aggressive LP Alternating (main workhorse)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP Alternating (n={n_best})")
    print(f"{'='*60}")

    a_work = normalize(best_seq.copy())

    for cycle in range(8):
        if time.monotonic() > deadline - 400:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.15, 80.0)

        a_new = lp_alternate_aggressive(a_work, max_iters=40, time_budget=alloc,
                                         seed=seed + cycle * 100)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            a_work = a_new
            print(f"  LP cycle {cycle}: {best_score:.8f} **")
        else:
            a_work = a_new
            print(f"  LP cycle {cycle}: {obj_new:.8f}")

    # =========================================================
    # Phase 3: Log-sum-exp gradient + Spectral shaping
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Smooth optimization")
    print(f"{'='*60}")

    a_work = normalize(best_seq.copy())

    # Log-sum-exp continuation
    if time.monotonic() < deadline - 300:
        alloc = min((deadline - time.monotonic()) * 0.3, 100.0)
        a_lse = log_sum_exp_gradient(a_work, time_budget=alloc)
        obj_lse = compute_obj(a_lse)
        if obj_lse < best_score:
            best_score = obj_lse
            best_seq = a_lse.copy()
            a_work = a_lse
            print(f"  LogSumExp: {best_score:.8f} **")
        else:
            print(f"  LogSumExp: {obj_lse:.8f}")
            a_work = normalize(0.7 * normalize(a_work) + 0.3 * normalize(a_lse))

    # Spectral shaping
    if time.monotonic() < deadline - 250:
        alloc = min((deadline - time.monotonic()) * 0.2, 60.0)
        a_spec = spectral_shaping(a_work, time_budget=alloc)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            a_work = a_spec
            print(f"  Spectral: {best_score:.8f} **")
        else:
            print(f"  Spectral: {obj_spec:.8f}")
            a_work = normalize(0.7 * normalize(a_work) + 0.3 * normalize(a_spec))

    # Follow-up LP after smooth methods
    if time.monotonic() < deadline - 200:
        alloc = min((deadline - time.monotonic()) * 0.3, 80.0)
        a_work = lp_alternate_aggressive(a_work, max_iters=30, time_budget=alloc,
                                          seed=seed + 500)
        obj_work = compute_obj(a_work)
        if obj_work < best_score:
            best_score = obj_work
            best_seq = a_work.copy()
            print(f"  Post-smooth LP: {best_score:.8f} **")
        else:
            print(f"  Post-smooth LP: {obj_work:.8f}")

    # =========================================================
    # Phase 4: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_cur = len(best_seq)
    n_options = sorted(set([
        n_cur - 100, n_cur - 50, n_cur - 25,
        n_cur, n_cur + 25, n_cur + 50, n_cur + 100,
        n_cur + 200, n_cur + 300, n_cur + 500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]

    for n_try in n_options:
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_try = normalize(resample_sequence(best_seq, n_try))
        alloc = min(remaining * 0.15, 40.0)

        # Quick optimization
        if n_try <= 1000:
            a_try = lp_alternate_aggressive(a_try, max_iters=15, time_budget=alloc * 0.5,
                                             seed=seed + n_try)
        a_try = log_sum_exp_gradient(a_try, time_budget=alloc * 0.35)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 5: Final polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    a_final = normalize(best_seq.copy())
    n_final = len(a_final)

    for polish_round in range(6):
        if time.monotonic() > deadline - 15:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.2, 25.0)

        # LP if n is manageable
        if n_final <= 1000 and alloc > 10:
            a_pol = lp_alternate_aggressive(a_final, max_iters=10, time_budget=alloc * 0.4,
                                             seed=seed + 1000 + polish_round)
        else:
            a_pol = a_final.copy()

        # Coordinate descent
        if alloc > 10:
            a_pol = coordinate_descent(a_pol, time_budget=min(10, alloc * 0.35),
                                        seed=seed + 2000 + polish_round)

        # Log-sum-exp with tight temperature
        if alloc > 8:
            a_pol = log_sum_exp_gradient(a_pol, time_budget=min(12, alloc * 0.25))

        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            a_final = a_pol
            print(f"  Polish {polish_round}: {best_score:.8f} **")
        else:
            a_final = a_pol
            print(f"  Polish {polish_round}: {obj_pol:.8f}")

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate with the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
