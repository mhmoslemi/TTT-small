```python
def subgradient_step(a, step_size, n_active=5):
    n = len(a)
    conv = fast_conv(a)
    M = conv.max()
    
    # Find active indices (top n_active)
    sorted_idx = np.argsort(conv)[::-1]
    active = sorted_idx[:n_active]
    
    # Compute subgradient (average of gradients at active indices)
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            if k == 2*i:
                grad[i] += 2 * a[i]
            else:
                grad[i] += (a[k-i] if 0 <= k-i < n else 0) + (a[i-k] if 0 <= i-k < n else 0)
    grad /= len(active)
    
    # Project to simplex tangent (remove mean)
    grad -= np.mean(grad)
    
    # Update
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0)
    a_new /= a_new.sum()
    
    return a_new
```

```python
def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def resample_seq(src, dest_n):

    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def mirror_descent_subgrad(a, max_iters, deadline, eta_init=0.1, verbose=False):
    """
    Mirror descent (multiplicative weights) using subgradient of max(conv).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = fast_conv(a)
        M = conv.max()
        
        # Find top-k active indices
        top_k = min(10, 2*n-1)
        active = np.argsort(conv)[::-1][:top_k]
        
        # Compute subgradient (average gradient at active k's)
        grad = np.zeros(n)
        for k in active:
            # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (or 2a[i] if k=2i)
            # Vectorized: for each i, the gradient is a[k-i] (if valid) + a[i-k] (if valid)
            # a[k-i] for i in 0..n-1: valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
            # a[i-k] for i in 0..n-1: valid when 0 <= i-k < n, i.e., k <= i < k+n
            
            idx1_lo = max(0, k - n + 1)
            idx1_hi = min(n - 1, k)
            if idx1_lo <= idx1_hi:
                grad[idx1_lo:idx1_hi+1] += a[k - idx1_lo:k - idx1_hi + 1][::-1]
            
            idx2_lo = max(0, k)
            idx2_hi = min(n - 1, k + n - 1)
            if idx2_lo <= idx2_hi:
                grad[idx2_lo:idx2_hi+1] += a[idx2_lo - k:idx2_hi - k + 1]
            
            # Subtract the double-counted term at i = k/2 if k is even
            if k % 2 == 0:
                i_mid = k // 2
                if 0 <= i_mid < n:
                    grad[i_mid] -= a[i_mid]  # We added it twice, remove once
        
        grad /= top_k
        
        # Project to simplex tangent
        grad -= np.mean(grad)
        
        # Anneal eta
        frac = it / max(1, max_iters)
        eta = eta_init * (1.0 - 0.5 * frac)
        
        # Multiplicative weights update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()
        
        obj_new = compute_J(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break
        
        if verbose and it % 100 == 0:
            print(f"      MD it {it}: obj={best_obj:.8f}", flush=True)
    
    return best_a, best_obj
```

```python
"""
Optimized step function search combining fast coordinate descent with adaptive
candidate selection, mirror descent (multiplicative weights), multi-n exploration,
perturbation restarts, and SCA-LP polish. Targets J < 1.5027.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent(a, max_sweeps, deadline, seed=0, verbose=False):
    """
    Fast coordinate descent with adaptive candidate selection.
    Uses parabolic interpolation when 3 candidate points bracket a minimum.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = float(n)
    a = a / s
    s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 80

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            kq = 2 * i
            valid_kq = (0 <= kq <= 2 * n - 2)

            # Generate adaptive candidates
            if old < 1e-10:
                cands = [0.0, 5e-5, 2e-4, 1e-3, 5e-3]
            else:
                # Core candidates: current, zero, mean, and ratios
                cands = [0.0, old * 0.5, old * 0.75, old * 1.25, old * 1.5]
                # Add mean value
                cands.append(1.0 / n)
                # Add a value based on local "pressure" - try to balance
                # Estimate: if conv at k=2i is high, reduce a[i]
                if valid_kq and conv[kq] > conv.mean() * 1.1:
                    cands.append(old * 0.3)
                    cands.append(old * 0.6)
                elif valid_kq and conv[kq] < conv.mean() * 0.9:
                    cands.append(old * 1.8)
                    cands.append(old * 2.5)

            best_x = old
            best_local = 2.0 * n * conv.max() / (s * s)
            eval_points = []  # Store (x, obj) for parabolic interpolation

            for x in cands:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                a_slice = a[lo - i:hi - i + 1].copy()
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 0.01:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                eval_points.append((x, new_obj))
                if new_obj < best_local:
                    best_local = new_obj
                    best_x = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            # Parabolic interpolation if we have 3+ finite points
            if len(eval_points) >= 3:
                finite = [(x, o) for x, o in eval_points if np.isfinite(o)]
                if len(finite) >= 3:
                    # Take the 3 points around the best
                    finite.sort(key=lambda p: p[1])
                    x0, f0 = finite[0]
                    x1, f1 = finite[1]
                    x2, f2 = finite[2]
                    # Ensure we have 3 distinct x values
                    if x0 != x1 and x1 != x2 and x0 != x2:
                        # Fit parabola through 3 points
                        denom = (x0 - x1) * (x0 - x2) * (x1 - x2)
                        if abs(denom) > 1e-30:
                            a_coeff = ((f0 / ((x0 - x1) * (x0 - x2))) +
                                       (f1 / ((x1 - x0) * (x1 - x2))) +
                                       (f2 / ((x2 - x0) * (x2 - x1))))
                            b_coeff = (-(f0 * (x1 + x2) / ((x0 - x1) * (x0 - x2))) -
                                       (f1 * (x0 + x2) / ((x1 - x0) * (x1 - x2))) -
                                       (f2 * (x0 + x1) / ((x2 - x0) * (x2 - x1))))
                            if abs(a_coeff) > 1e-30:
                                x_par = -b_coeff / (2.0 * a_coeff)
                                if 0 <= x_par <= max(old * 3, 1.0 / n * 3):
                                    x_par = float(x_par)
                                    delta_p = x_par - old
                                    if abs(delta_p) > 1e-12:
                                        a_slice = a[lo - i:hi - i + 1].copy()
                                        conv[lo:hi + 1] += 2.0 * delta_p * a_slice
                                        if valid_kq:
                                            conv[kq] += delta_p * delta_p
                                        new_s = s - old + x_par
                                        if new_s > 0.01:
                                            obj_p = 2.0 * n * conv.max() / (new_s * new_s)
                                            if obj_p < best_local:
                                                best_local = obj_p
                                                best_x = x_par
                                        conv[lo:hi + 1] -= 2.0 * delta_p * a_slice
                                        if valid_kq:
                                            conv[kq] -= delta_p * delta_p

            if abs(best_x - old) > 1e-13:
                delta = best_x - old
                a_slice = a[lo - i:hi - i + 1].copy()
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_x
                s = s - old + best_x
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 500 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def mirror_descent_subgrad(a, max_iters, deadline, eta_init=0.05, verbose=False):
    """
    Multiplicative weights (mirror descent) using subgradient of max(conv).
    Naturally lives on the simplex via exponential update.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = conv.max()

        # Find top active indices
        n_active = min(8, 2 * n - 1)
        active = np.argsort(conv)[::-1][:n_active]

        # Compute subgradient efficiently using FFT
        # grad[i] = (1/n_active) * sum_{k in active} (a[k-i] + a[i-k]) with correction
        # This equals cross-correlation of w_active with a
        w_active = np.zeros(2 * n - 1)
        for k in active:
            w_active[k] = 1.0 / n_active

        # cross-correlation: R[j] = sum_k w[k] * a[k-j]
        # Use FFT: R = ifft(fft(w) * conj(fft(a_padded)))
        size = 1
        while size < 3 * n:
            size <<= 1
        w_padded = np.zeros(size)
        w_padded[:2 * n - 1] = w_active
        a_padded = np.zeros(size)
        a_padded[:n] = a

        W = np.fft.rfft(w_padded)
        A = np.fft.rfft(a_padded)
        # Cross-correlation R[j] = sum_k w[k] a[k-j]
        # = ifft(W * conj(A))[j + (2n-2)] ... let me think about this differently
        # Actually: ifft(fft(w) * conj(fft(a))) gives sum_k w[k+l] * a[k] = cross-corr at lag l
        cross1 = np.fft.irfft(W * np.conj(A), size)
        # cross1[m] = sum_k w[k+m] * a[k] for m >= 0
        # We want: sum_k w[k] * a[k-j] = sum_m w[m+j] * a[m] = cross1[j]
        # And: sum_k w[k] * a[j-k] = sum_m w[m] * a[j-m] = cross1[-j] = cross1[size-j]

        # grad[i] = cross1[i] + cross1[size - i] - w[2i]/n_active * a[i] (correction for double count)
        grad = np.zeros(n)
        for i in range(n):
            val1 = cross1[i] if i < len(cross1) else 0.0
            val2 = cross1[size - i] if size - i < len(cross1) else 0.0
            grad[i] = val1 + val2
            if 2 * i <= 2 * n - 2:
                grad[i] -= w_active[2 * i] * a[i]

        # Subtract mean for simplex projection
        grad -= np.mean(grad)

        # Anneal learning rate
        frac = it / max(1, max_iters)
        eta = eta_init * (1.0 - 0.7 * frac * frac)

        # Multiplicative weights update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            # Try a smaller step
            if no_improve < 20:
                a_new2 = a * np.exp(-0.3 * eta * grad)
                a_new2 = np.maximum(a_new2, 1e-15)
                a_new2 /= a_new2.sum()
                obj_new2 = compute_J(a_new2)
                if obj_new2 < best_obj - 1e-14:
                    best_obj = obj_new2
                    best_a = a_new2.copy()
                    a = a_new2
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 60:
            break

        if verbose and it % 200 == 0:
            print(f"    MD it {it}: obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def sca_lp_step(a, max_iters, deadline, max_lp_time=15, verbose=False):
    """
    SCA-LP: Linearize quadratic constraints and solve LP to find best response.
    Only uses active constraints for efficiency.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Use only top constraints (active set + a few more for robustness)
        n_constraints = min(2 * n - 1, max(30, n // 2))
        active_indices = np.argsort(c)[::-1][:n_constraints]

        # LP: minimize t
        # s.t. 2*(a*g)[k] - t <= c[k]  for k in active_indices
        #      sum(g) = 1
        #      g >= 0
        # Variables: g[0..n-1], t (index n)

        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in active_indices:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub))
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(len(b_ub))
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(float(c[k]))

        A = coo_matrix((vals, (rows, cols)), shape=(n_constraints, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': max_lp_time, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-13:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 15:
            break

        if verbose and it % 5 == 0:
            print(f"    SCA it {it}: obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences for optimization."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.03, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.07, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.12, n)))
        # Smoothed perturbation
        ksize = max(3, n // 32)
        kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) / (ksize * 0.5)) ** 2)
        kernel /= kernel.sum()
        smooth_noise = np.convolve(rng.normal(0, 0.1, n), kernel, mode='same')
        inits.append(project_simplex(p + smooth_noise))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns
    for freq in [1, 2, 3]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.25, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.02))

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    # Scaled parent with different shapes
    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        # Tapered versions
        taper = np.linspace(0.5, 1.5, n)
        inits.append(project_simplex(p * taper))
        taper2 = 1.0 + 0.3 * np.sin(np.pi * t)
        inits.append(project_simplex(p * taper2))

    return inits


def refine_sequence(a, deadline, verbose=False):
    """
    Multi-method refinement of a single sequence.
    Alternates between CD, mirror descent, and SCA-LP.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 0.01:
            return a
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            if verbose:
                print(f"      Updated: J={obj:.8f}", flush=True)
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Cycle through methods
    method_budget = min(60.0, remaining / 8.0)  # ~60s per method cycle
    cycle = 0

    while time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        alloc = min(method_budget, remaining - 20)
        if alloc < 15:
            break

        # Method selection: alternate CD -> MD -> CD -> SCA -> CD
        method = cycle % 4
        t_end = time.monotonic() + alloc

        if method == 0:
            # Coordinate descent
            a_new = coord_descent(a, max_sweeps=50000, deadline=t_end,
                                  seed=42 + cycle, verbose=False)
            a = try_update(a_new)
        elif method == 1:
            # Mirror descent
            a_md, obj_md = mirror_descent_subgrad(a, max_iters=2000, deadline=t_end,
                                                   eta_init=0.03, verbose=False)
            a = try_update(a_md)
        elif method == 2:
            # CD with different seed
            a_new = coord_descent(a, max_sweeps=50000, deadline=t_end,
                                  seed=100 + cycle, verbose=False)
            a = try_update(a_new)
        else:
            # SCA-LP (only if n is not too large)
            if n <= 1200:
                a_sca, obj_sca = sca_lp_step(a, max_iters=30, deadline=t_end,
                                              max_lp_time=max(5, alloc // 3))
                a = try_update(a_sca)
            else:
                # Fallback to CD for large n
                a_new = coord_descent(a, max_sweeps=50000, deadline=t_end,
                                      seed=200 + cycle, verbose=False)
                a = try_update(a_new)

        a = best_a.copy()
        cycle += 1

    return best_a, best_obj

    candidate_ns = sorted(candidate_ns)
    # Prioritize: parent first, then nearby
    candidate_ns.sort(key=lambda n: (abs(n - n_parent), n))
    candidate_ns = candidate_ns[:6]  # Try at most 6 different n values

    per_n_time = (phase1_deadline - time.monotonic()) / max(1, len(candidate_ns)) - 10

    n_results = {}
    for new_n in candidate_ns:
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 30:
            break
        alloc = min(per_n_time, remaining_check - 5)
        if alloc < 20:
            break

        print(f"  Quick CD at n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_init = resample_seq(parent, new_n)
        a_init = np.maximum(a_init, 1e-8)
        a_init = a_init / a_init.sum()

        # Quick coordinate descent
        a_opt = coord_descent(a_init, max_sweeps=200, deadline=time.monotonic() + alloc,
                             seed=new_n % 1000)
        update_best(a_opt)
        n_results[new_n] = a_opt.copy()
        print(f"  n={new_n}: J={compute_J(a_opt):.8f}", flush=True)

    # === PHASE 2: Intensive refinement of best (500s) ===
    phase2_deadline = t_start + 750.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Intensive refinement (budget {phase2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Use the best result so far
    a_curr = best_seq.copy() if best_seq is not None else resample_seq(parent, n_parent)

    # Run multi-method refinement
    a_ref, obj_ref = refine_sequence(a_curr, phase2_deadline, verbose=False)
    update_best(a_ref)

    # === PHASE 3: Perturbation restarts (350s) ===
    phase3_deadline = t_start + 1100.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Perturbation restarts (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = phase3_deadline - time.monotonic()
        n_restarts = min(8, max(2, int(remaining / 60)))
        per_restart = remaining / max(1, n_restarts) - 5

        for restart in range(n_restarts):
            remaining_check = phase3_deadline - time.monotonic()
            if remaining_check < 40:
                break
            alloc = min(per_restart, remaining_check - 10)
            if alloc < 25:
                break

            # Generate perturbation
            pert_size = 0.05 + 0.1 * (restart % 3)
            noise = rng.normal(0, pert_size, n_best)
            a_pert = best_seq + noise
            a_pert = project_simplex(a_pert)

            # Also try a smoothed perturbation
            if restart % 2 == 1:
                ksize = max(3, n_best // 32)
                kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) / (ksize * 0.4)) ** 2)
                kernel /= kernel.sum()
                smooth_noise = np.convolve(rng.normal(0, pert_size * 1.5, n_best), kernel, mode='same')
                a_pert = best_seq + smooth_noise
                a_pert = project_simplex(a_pert)

            # Quick refinement
            a_opt, obj_opt = refine_sequence(a_pert, time.monotonic() + alloc, verbose=False)
            update_best(a_opt)
            print(f"  Restart {restart}: J={compute_J(a_opt):.8f} (pert={pert_size:.3f})", flush=True)

    # === PHASE 4: Final polish (remaining time) ===
    phase4_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Final polish (budget {phase4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase4_deadline - time.monotonic()
        if remaining > 60:
            n_best = len(best_seq)

            # Long coordinate descent pass
            t_cd = time.monotonic() + 0.45 * remaining
            print(f"  Long CD pass (budget {t_cd - time.monotonic():.0f}s)...", flush=True)
            a_cd = coord_descent(best_seq, max_sweeps=100000, deadline=t_cd, seed=777)
            update_best(a_cd)

            remaining = phase4_deadline - time.monotonic()
            if remaining > 30:
                # Mirror descent pass
                t_md = time.monotonic() + 0.30 * remaining
                print(f"  MD pass (budget {t_md - time.monotonic():.0f}s)...", flush=True)
                a_md, obj_md = mirror_descent_subgrad(best_seq, max_iters=5000, deadline=t_md,
                                                      eta_init=0.02, verbose=False)
                update_best(a_md)

            remaining = phase4_deadline - time.monotonic()
            if remaining > 30:
                # SCA-LP polish
                t_sca = time.monotonic() + 0.25 * remaining
                if n_best <= 1200:
                    print(f"  SCA-LP polish (budget {t_sca - time.monotonic():.0f}s)...", flush=True)
                    a_sca, obj_sca = sca_lp_step(best_seq, max_iters=30, deadline=t_sca,
                                                  max_lp_time=max(5, remaining // 4))
                    update_best(a_sca)

            # Final tiny CD
            if time.monotonic() < phase4_deadline - 10:
                a_final = coord_descent(best_seq, max_sweeps=50000,
                                        deadline=phase4_deadline - 5, seed=999)
                update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
```
