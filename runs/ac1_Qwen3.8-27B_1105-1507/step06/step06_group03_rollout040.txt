    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v

    cssv = np.cumsum(u) - 1.0

    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

                if 2 * i <= 2 * n - 2:
                    conv[2 * i] += delta * delta

                new_obj = 2.0 * n * conv.max()

                if new_obj < best_local:
                    best_local = new_obj
                    best_x = x

                conv[k_lo:k_hi + 1] -= 2.0 * delta * a_slice
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] -= delta * delta

            if best_x != old:
                delta = best_x - old
                conv[k_lo:k_hi + 1] += 2.0 * delta * a_slice
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] += delta * delta
                a[i] = best_x

        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a, 2.0 * len(a) * fast_conv(a).max()


def power_grad_descent(a, deadline, p_init=10.0, p_max=100.0, lr_init=0.02, seed=42):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    rng = np.random.default_rng(seed)
    no_improve = 0

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0

    size = 1
    while size < 3 * n + 1:
        size <<= 1

    while time.monotonic() < deadline:
        frac = t_count / max(1, int((deadline - time.monotonic()) * 2))
        p = p_init + (p_max - p_init) * min(1.0, t_count / 200.0)

        conv = fast_conv(a)
        conv_clipped = np.maximum(conv, 1e-15)

        w_raw = conv_clipped ** (p - 1)
        S = np.sum(conv_clipped ** p)
        scale = 4.0 * n * S ** ((1.0 - p) / p)

        A_fft = np.fft.rfft(a, size)
        W_fft = np.fft.rfft(w_raw, size)
        corr = np.fft.irfft(np.conj(A_fft) * W_fft, size)
        grad = scale * corr[:n]

        grad = grad - np.mean(grad)

        t_count += 1
        lr = lr_init / (1.0 + 0.05 * np.sqrt(t_count))
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = 2.0 * n * fast_conv(a_new).max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                a = best_a + rng.normal(0, 0.015, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0
                p = p_init

        a = a_new

    return best_a, best_obj


def alternating_optimize(a, deadline, seed=42):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    elapsed_total = 0.0
    stall = 0

    t_start = time.monotonic()
    cycle_time = max(3.0, (deadline - t_start) * 0.15)

    while time.monotonic() < deadline - 5:
        cycle_start = time.monotonic()
        cycle_dl = min(deadline - 5, t_start + elapsed_total + cycle_time)

        a_r, obj_r = coord_descent(best_a, cycle_dl, seed=seed)
        elapsed_total = time.monotonic() - t_start
        if obj_r < best_obj - 1e-10:
            best_obj = obj_r
            best_a = a_r
            stall = 0
        else:
            stall += 1

        if time.monotonic() < deadline - 5:
            pg_dl = min(deadline - 5, t_start + elapsed_total + cycle_time)
            a_p, obj_p = power_grad_descent(best_a, pg_dl, p_init=8.0, p_max=60.0,
                                            lr_init=0.015, seed=seed + 1)
            elapsed_total = time.monotonic() - t_start
            if obj_p < best_obj - 1e-10:
                best_obj = obj_p
                best_a = a_p
                stall = 0
            else:
                stall += 1

        if stall > 6:
            break

        seed += 7

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))

    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3, 5, 7]:
        for amp in [0.3, 0.6, 0.8]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    for center in [0.15, 0.5, 0.85]:
        for width in [0.08, 0.15, 0.3]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.03))

    for alpha in [0.3, 1.0, 2.0, 4.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    for _ in range(6):
        ksize = max(3, n // 20)
        kernel = np.ones(ksize) / ksize
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.02))

    for _ in range(3):
        a_rand = rng.uniform(0, 1, n)
        a_rand = np.convolve(a_rand, np.ones(5) / 5.0, mode='same')
        inits.append(project_simplex(a_rand))

    inits.append(project_simplex(np.maximum(np.sin(np.pi * t) ** 0.5 + 0.1, 0.01)))
    inits.append(project_simplex(np.maximum(np.cos(np.pi * (t - 0.5)) ** 2 + 0.1, 0.01)))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError, TypeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # Phase 1: Polish parent with CD + PGD (200s)
    print(f"\nPhase 1: Polish parent n={n_parent} (200s budget)", flush=True)
    p1_deadline = t_start + 200.0
    a_p1, obj_p1 = alternating_optimize(parent, p1_deadline, seed=seed)
    update_best(a_p1)
    print(f"  Phase 1 done: J={obj_p1:.8f}", flush=True)

    # Phase 2: Diverse initializations at parent length (500s)
    print(f"\nPhase 2: Diverse inits at n={n_parent} (500s budget)", flush=True)
    p2_deadline = t_start + 700.0

    inits = generate_inits(n_parent, rng, parent)
    n_inits = min(len(inits), 20)
    per_init_time = (p2_deadline - time.monotonic()) / max(1, n_inits) - 2

    for idx in range(n_inits):
        remaining = p2_deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(per_init_time, remaining - 3)
        if alloc < 15:
            break

        a0 = inits[idx]
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt = a0
        obj_opt = 2.0 * n_parent * fast_conv(a0).max()

        cd_dl = time.monotonic() + 0.6 * alloc
        a_cd, obj_cd = coord_descent(a0, cd_dl, seed=seed + idx)
        if obj_cd < obj_opt:
            a_opt = a_cd
            obj_opt = obj_cd

        if time.monotonic() < time.monotonic() + 0.4 * alloc - 2:
            pg_dl = time.monotonic() + 0.35 * alloc
            a_pg, obj_pg = power_grad_descent(a_opt, pg_dl, p_init=8.0, p_max=50.0,
                                              lr_init=0.012, seed=seed + idx + 100)
            if obj_pg < obj_opt:
                a_opt = a_pg
                obj_opt = obj_pg

        cd_dl2 = min(time.monotonic() + 0.05 * alloc, time.monotonic() + alloc)
        a_cd2, obj_cd2 = coord_descent(a_opt, cd_dl2, seed=seed + idx + 200)
        if obj_cd2 < obj_opt:
            a_opt = a_cd2
            obj_opt = obj_cd2

        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.8f}", flush=True)

    # Phase 3: Try different lengths (300s)
    print(f"\nPhase 3: Multi-length exploration (300s budget)", flush=True)
    p3_deadline = t_start + 1000.0

    if best_seq is not None:
        ref_n = best_seq.size
        candidate_ns = sorted(set([n_parent - 128, n_parent - 64, n_parent,
                                   n_parent + 64, n_parent + 128, n_parent + 256]),
                              key=lambda x: abs(x - ref_n))
        candidate_ns = [n for n in candidate_ns if n >= 128 and n <= 2048 and n != n_parent]
    else:
        candidate_ns = [384, 512, 640, 896, 1024]

    for new_n in candidate_ns[:5]:
        remaining = p3_deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(55.0, remaining - 10)
        if alloc < 30:
            break

        print(f"  Trying n={new_n} ({alloc:.0f}s)...", flush=True)
        ref = best_seq if best_seq is not None else parent
        a_ext = resample_seq(ref, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = alternating_optimize(a_ext, time.monotonic() + alloc,
                                              seed=seed + new_n)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # Phase 4: Intensive optimization on best (remaining time)
    print(f"\nPhase 4: Intensive optimization (budget {deadline - time.monotonic():.0f}s)", flush=True)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_curr = best_seq.copy()

            # Alternate CD and PGD multiple times
            sub_budget = remaining * 0.4
            a_curr, obj = alternating_optimize(a_curr, time.monotonic() + sub_budget,
                                               seed=seed + 500)
            update_best(a_curr)

            remaining = deadline - time.monotonic()
            if remaining > 20:
                sub_budget = remaining * 0.3
                a_curr, obj = power_grad_descent(best_seq, time.monotonic() + sub_budget,
                                                 p_init=5.0, p_max=40.0,
                                                 lr_init=0.01, seed=seed + 600)
                update_best(a_curr)

            remaining = deadline - time.monotonic()
            if remaining > 15:
                sub_budget = remaining * 0.3
                a_curr, obj = coord_descent(best_seq, time.monotonic() + sub_budget,
                                            seed=seed + 700)
                update_best(a_curr)

            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_curr, obj = alternating_optimize(best_seq, time.monotonic() + remaining * 0.8,
                                                   seed=seed + 800)
                update_best(a_curr)

    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        update_best(best_seq)

    print(f"\nFINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    return [float(x) for x in best_seq]
