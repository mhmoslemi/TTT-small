"""
Focused LP-alternating optimizer with structured initializations (ramp, V-shape, etc.)
to minimize 2*n*max(conv(a,a))/(sum(a))^2. Starts from a simple ramp which already
achieves ~1.34 for n=700, then refines using LP-based alternating direction.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix T where (T@g)[k] = conv(a,g)[k] for k=0,...,2n-2."""
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        T[k_start:k_end, j] = a[:k_end - k_start]
    return T


def solve_lp_max_sum(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    if n > 3000:
        return None
    T = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_min_peak(a, time_limit=10):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    if n > 3000:
        return None, None
    T = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def lp_alternate(a, max_iters=200, time_budget=60.0, verbose=False):
    """LP alternating: iterate between a and LP solution to reduce peak."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_max_sum(a, M, time_limit=8)
        if g is None:
            break

        # Line search over blend parameter
        candidate_ts = [0.05, 0.1, 0.2, last_t * 0.7, last_t, last_t * 1.3, 0.5, 0.7, 1.0]
        candidate_ts = [t for t in candidate_ts if 0 < t <= 1.0]
        candidate_ts = list(dict.fromkeys(candidate_ts))

        best_t_obj = best_obj
        best_t_val = 0.2
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            last_t = best_t_val
            stall = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it={it}: obj={best_obj:.8f}, t={best_t_val:.3f}, elapsed={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 15:
                break

    return best_a, best_obj


def subgradient_momentum(a, max_iters=300, time_budget=30.0, tau_init=0.3, momentum=0.8):
    """Subgradient optimization with momentum on the max convolution peak."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    step = 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        tau = tau_init + 3.0 * it / max_iters
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute subgradient: sg[j] = sum_k weights[k] * (a[k-j] + a[j]*delta_{k,2j})
        # First term: cross-correlation of weights with a
        # sg[j] = sum_k w[k] * a[k-j] (for valid k-j) + w[2j]*a[j] (if 2j in range)
        # Use FFT for the first term
        N_fft = 1
        while N_fft < 2 * n:
            N_fft <<= 1
        fw = rfft(weights, N_fft)
        fa = rfft(a, N_fft)
        # cross-corr: sum_k w[k] * a[k-j]
        # This is ifft(fft(w) * conj(fft(a))) but we need to be careful with indexing
        # Actually: ifft(fft(w) * conj(fft(a)))[j] = sum_k w[k]*a[k-j] (circular)
        # For linear, we need N_fft >= len(w) + n - 1
        corr = irfft(fw * np.conj(fa), N_fft)
        # corr[j] = sum_k w[k] * a[(k-j) mod N_fft]
        # For our case, since weights has length 2n-1 and a has length n,
        # and N_fft >= 2n, the circular correlation should match linear for j=0..n-1
        # But we need to check the indexing carefully.
        # Actually, let me just compute it directly for correctness.
        grad = np.zeros(n)
        for k_idx in range(len(weights)):
            if weights[k_idx] < 1e-14:
                continue
            # grad[j] += w[k] * a[k-j] for 0 <= k-j < n
            j_lo = max(0, k_idx - n + 1)
            j_hi = min(n, k_idx + 1)
            if j_hi > j_lo:
                js = np.arange(j_lo, j_hi)
                a_idx = k_idx - js
                grad[js] += weights[k_idx] * a[a_idx]
            # Add the delta term: w[2j] * a[j] if 2j < len(weights)
            # This is a separate contribution
        # Add diagonal term: for each j, if 2*j < 2*n-1, add w[2j]*a[j]
        diag_idx = 2 * np.arange(n)
        valid_diag = diag_idx < len(weights)
        grad[valid_diag] += weights[diag_idx[valid_diag]] * a[valid_diag]

        grad_centered = grad - np.mean(grad)

        v = momentum * v - step * grad_centered
        a_new = a + v
        a_new = project_nonneg_sum1(a_new)

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        step *= 0.999

    return best_a, best_obj


def generate_ramp(n, direction=1):
    """Generate a ramp sequence normalized to sum=1."""
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def generate_v_shape(n):
    """Generate a V-shaped sequence (high at ends, low in middle)."""
    half = n // 2
    if n % 2 == 0:
        left = np.linspace(1.0, 0.1, half)
        right = np.linspace(0.1, 1.0, half)
    else:
        left = np.linspace(1.0, 0.1, half + 1)
        right = np.linspace(0.1, 1.0, half)
    a = np.concatenate([left, right])
    return a / a.sum()


def generate_inverse_ramp(n):
    """Quadratic ramp: a[i] = (i+1)^2."""
    a = np.arange(1, n + 1, dtype=np.float64) ** 2
    return a / a.sum()


def generate_sine(n):
    """Sinusoidal: a[i] = sin(pi*(i+1)/(n+1))."""
    a = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    a = np.maximum(a, 0) + 1e-6
    return a / a.sum()


def generate_two_triangle(n):
    """Two triangles: high in two separate regions."""
    a = np.zeros(n)
    w = n // 4
    center1 = n // 4
    center2 = 3 * n // 4
    for i in range(max(0, center1 - w), min(n, center1 + w)):
        a[i] = 1.0 - abs(i - center1) / w
    for i in range(max(0, center2 - w), min(n, center2 + w)):
        a[i] = 1.0 - abs(i - center2) / w
    a = np.maximum(a, 0) + 1e-6
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_nonneg_sum1(a_new)

def freq_domain_refine(a, max_iters=100, time_budget=20.0):
    """Frequency domain: compress dynamic range of power spectrum to reduce peak."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power_spec = np.abs(fa) ** 2

        # Temporal domain peak reduction
        conv = np.fft.ifft(fa * fa).real[:2 * n - 1]
        c_max = np.max(conv)
        if c_max < 1e-15:
            break

        # Soft-clip the convolution to reduce peak
        threshold = c_max * 0.85
        conv_clipped = np.where(conv > threshold, threshold + (conv - threshold) * 0.3, conv)
        conv_clipped = np.maximum(conv_clipped, 0)

        # Extract new a via sqrt of FFT
        cf = np.fft.fft(conv_clipped, N)
        fa_new = np.sqrt(cf + 0j)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj

        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and np.sum(parent) > 0.01:
            p_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={p_obj:.6f}")
            if p_obj < best_score:
                best_score = p_obj
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Structured initialization screening ===
    print("\n=== Phase 1: Structured Initialization Screening ===")
    phase1_budget = min(60.0, (deadline - start_time) * 0.04)

    n_base = 700
    n_options = [n_base, 800, 900, 1000, 600, 500]
    n_options = list(dict.fromkeys(n for n in n_options if 100 <= n <= 2500))

    generators = [
        ("ramp_up", lambda n: generate_ramp(n, 1)),
        ("ramp_down", lambda n: generate_ramp(n, -1)),
        ("quad_ramp", lambda n: generate_inverse_ramp(n)),
        ("v_shape", lambda n: generate_v_shape(n)),
        ("sine", lambda n: generate_sine(n)),
        ("two_tri", lambda n: generate_two_triangle(n)),
        ("uniform", lambda n: np.ones(n) / n),
    ]

    candidates = []
    per_test_budget = max(1.0, phase1_budget / (len(n_options) * len(generators)))

    for n_try in n_options:
        if time.monotonic() > start_time + phase1_budget:
            break
        for name, gen_func in generators:
            if time.monotonic() > start_time + phase1_budget:
                break
            a_init = gen_func(n_try)
            # Quick LP polish (just a few iterations)
            a_quick, obj_quick = lp_alternate(a_init, max_iters=5, time_budget=min(per_test_budget, 3.0), verbose=False)
            if obj_quick < np.inf:
                candidates.append((obj_quick, a_quick, n_try, name))

    # Sort and report
    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates evaluated.")
    for obj_c, a_c, n_c, name_c in candidates[:8]:
        print(f"    n={n_c}, {name_c}: obj={obj_c:.6f}")

    # Update best from screening
    for obj_c, a_c, n_c, name_c in candidates:
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  ** Phase 1 best: n={n_c}, {name_c}, obj={best_score:.8f} **")

    if best_seq is None:
        best_seq = generate_ramp(700)
        best_score = compute_obj(best_seq)
        print(f"  Fallback ramp: obj={best_score:.6f}")

    # === Phase 2: Aggressive LP optimization on best candidate ===
    print(f"\n=== Phase 2: Aggressive LP Optimization ===")
    print(f"  Starting obj: {best_score:.8f}")
    phase2_budget = min(600.0, (deadline - time.monotonic()) * 0.45)
    a_current = project_nonneg_sum1(best_seq.copy())

    a_opt, obj_opt = lp_alternate_aggressive(a_current, time_budget=phase2_budget, verbose=True)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  ** After Phase 2 LP: obj={best_score:.8f} **")
    else:
        print(f"  No improvement from LP (obj stayed at {best_score:.8f})")

    # === Phase 3: Try other top candidates with LP ===
    print(f"\n=== Phase 3: Refine Other Candidates ===")
    print(f"  Best so far: {best_score:.8f}")
    phase3_budget = min(300.0, (deadline - time.monotonic()) * 0.25)
    phase3_start = time.monotonic()

    # Try top 3 other candidates
    tried = set()
    for obj_c, a_c, n_c, name_c in candidates[:6]:
        if time.monotonic() - phase3_start > phase3_budget:
            break
        key = (n_c, name_c)
        if key in tried:
            continue
        if obj_c >= best_score * 1.01:
            continue  # Skip if already much worse
        tried.add(key)

        remaining3 = phase3_budget - (time.monotonic() - phase3_start)
        if remaining3 < 10:
            break
        alloc = min(60.0, remaining3 * 0.4)

        a_try = project_nonneg_sum1(a_c.copy())
        a_res, obj_res = lp_alternate_aggressive(a_try, time_budget=alloc, verbose=False)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  ** Phase 3 improvement: n={n_c}, {name_c}, obj={best_score:.8f} **")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    print(f"  Best so far: {best_score:.8f}")
    phase4_budget = min(200.0, (deadline - time.monotonic()) * 0.2)
    phase4_start = time.monotonic()

    n_best = len(best_seq)
    n_explore = [n_best - 50, n_best + 50, n_best - 100, n_best + 100,
                 n_best + 200, n_best + 300]
    n_explore = [n for n in dict.fromkeys(n_explore) if 200 <= n <= 3000]

    for n_try in n_explore:
        if time.monotonic() - phase4_start > phase4_budget:
            break
        remaining4 = phase4_budget - (time.monotonic() - phase4_start)
        if remaining4 < 15:
            break
        alloc = min(40.0, remaining4 * 0.3)

        # Resample best sequence to new length
        a_resampled = resample_sequence(best_seq, n_try)
        # Also try generating fresh at this length
        a_ramp = generate_ramp(n_try)

        for a_init_in in [a_resampled, a_ramp]:
            if time.monotonic() - phase4_start > phase4_budget:
                break
            a_test, obj_test = lp_alternate(a_init_in, max_iters=30,
                                             time_budget=min(alloc, 20.0), verbose=False)
            if obj_test < best_score:
                best_score = obj_test
                best_seq = a_test.copy()
                print(f"  ** Phase 4: n={n_try}, obj={best_score:.8f} **")

    # === Phase 5: Subgradient refinement ===
    print(f"\n=== Phase 5: Subgradient Refinement ===")
    print(f"  Best so far: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 20 and best_seq is not None:
        a_sg = project_nonneg_sum1(best_seq.copy())
        a_sg_opt, obj_sg = subgradient_momentum(a_sg, max_iters=200,
                                                 time_budget=min(remaining5 * 0.4, 60.0))
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg_opt.copy()
            print(f"  ** Subgradient improved: obj={best_score:.8f} **")

    # === Phase 6: Frequency domain refinement ===
    remaining6 = deadline - time.monotonic()
    if remaining6 > 15 and best_seq is not None:
        a_fd = project_nonneg_sum1(best_seq.copy())
        a_fd_opt, obj_fd = freq_domain_refine(a_fd, max_iters=80,
                                               time_budget=min(remaining6 * 0.3, 30.0))
        if obj_fd < best_score:
            best_score = obj_fd
            best_seq = a_fd_opt.copy()
            print(f"  ** Freq domain improved: obj={best_score:.8f} **")

    # === Phase 7: Final LP polish ===
    remaining7 = deadline - time.monotonic()
    print(f"\n=== Phase 7: Final LP Polish (remaining {remaining7:.0f}s) ===")
    if remaining7 > 10 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        a_final_opt, obj_final = lp_alternate_aggressive(a_final,
                                                          time_budget=min(remaining7 - 5, 60.0),
                                                          verbose=True)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final_opt.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = generate_ramp(700)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [float(x) for x in generate_ramp(700)]
        best_score = compute_obj(np.array(result))
        val = evaluate_sequence(result)
        print(f"  Fallback validation: {val}")

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
