"""
Optimized step function search using FISTA (Fistral) accelerated proximal gradient
with adaptive temperature annealing on the softmax-smoothed max-convolution objective,
combined with periodic LP polishing and multi-resolution exploration.
Key improvements: correct FFT-based gradient computation, Nesterov acceleration,
adaptive tau scheduling, and strategic multi-length search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear self-convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    conv = irfft(fa * fa, N)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_gradient_ff(a, p, N_cache=None):
    """
    Compute gradient of smoothed max-convolution.
    grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i in [0,n-1]
    Uses FFT via reversal trick.
    """
    n = len(a)
    a_rev = a[::-1]
    if N_cache is None:
        N = 1
        while N < 3 * n:
            N <<= 1
    else:
        N = N_cache
    P = np.fft.fft(p, N)
    A = np.fft.fft(a_rev, N)
    conv = np.fft.ifft(P * A).real
    c = conv[n - 1: n - 1 + n]
    return 2.0 * c


def fista_optimize(a, max_iters=2000, tau_start=0.5, tau_end=0.01,
                   step_start=0.5, step_end=0.05, time_budget=100.0,
                   lp_every=200, n_lp_budget=5.0, verbose=False):
    """
    FISTA (Nesterov accelerated proximal gradient) with adaptive temperature.
    a should be normalized to sum=1, non-negative.
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute FFT size
    N = 1
    while N < 3 * n:
        N <<= 1

    y = a.copy()
    t = 1.0
    k = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive temperature
        frac = it / max(max_iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        if tau < 0.005:
            tau = 0.005

        # Adaptive step size
        step = step_start * (step_end / step_start) ** frac

        # Compute convolution at y
        conv_y = compute_conv(y)
        cmax = float(np.max(conv_y))
        if cmax < 1e-15:
            break

        # Softmax weights
        shifted = (conv_y - cmax) / tau
        shifted = np.clip(shifted, -500.0, 500.0)
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            idx = int(np.argmax(conv_y))
            p = np.zeros(len(conv_y))
            p[idx] = 1.0
        else:
            p /= p_sum

        # Gradient via FFT
        grad = compute_gradient_ff(y, p, N)

        # FISTA update
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t_old - 1.0) / t

        # Gradient step
        a_new = y - step * grad
        a_new = project(a_new)

        # Objective check
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose and it % 100 == 0:
                print(f"        FISTA it{it}: obj={best_obj:.8f}, tau={tau:.4f}")

        # Update y for next iteration
        y = a_new + momentum * (a_new - a)
        a = a_new
        k += 1

        # Periodic LP polishing
        if lp_every > 0 and k % lp_every == 0 and n <= 1200:
            remaining = time_budget - (time.monotonic() - start)
            if remaining > n_lp_budget * 2:
                a_lp, obj_lp = lp_polish(a, time_budget=n_lp_budget)
                if obj_lp < best_obj:
                    best_obj = obj_lp
                    best_a = a_lp.copy()
                    a = a_lp.copy()
                    y = a.copy()
                    if verbose:
                        print(f"        LP polish it{k}: obj={best_obj:.8f}")

    return best_a, best_obj


def lp_polish(a, time_budget=10.0):
    """LP-based polishing: solve max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 1500:
        return a, compute_obj(a)

    a = project(a.copy())
    conv = compute_conv(a)
    M = float(np.max(conv))

    # Build the constraint matrix ( Toeplitz-like )
    # (a*g)[k] = sum_i a[i]*g[k-i] for k = 0,...,2n-2
    # We need: sum_i a[i]*g[k-i] <= M for all k
    # This is A @ g <= M where A[k,i] = a[k-i] if 0<=k-i<n else 0
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n)
        A[k_start:k_end, i] = a[:k_end - k_start]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    t_lp_start = time.monotonic()
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True,
                               'time_limit': max(1.0, time_budget - (time.monotonic() - t_lp_start))})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                # Line search on blend
                best_a = a.copy()
                best_obj = compute_obj(a)
                for t_blend in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_try = (1.0 - t_blend) * a + t_blend * g
                    a_try = project(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a, best_obj
    except Exception:
        pass
    return a, compute_obj(a)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n (normalized to sum=1)."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.1
    inits.append(rc / rc.sum())

    # Sinc-squared (windowed)
    xi = (x - 0.5) * n
    sinc = np.sinc(xi / (n * 0.3))
    sinc = np.maximum(sinc, 0)
    if sinc.sum() > 1e-10:
        inits.append(sinc / sinc.sum())

    # Gaussian centered
    gauss = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
    inits.append(gauss / gauss.sum())

    # Two-lobe
    two = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
    inits.append(two / two.sum())

    # Random smoothed
    rnd = rng.random(n)
    # Simple boxcar smooth
    k = max(3, n // 50)
    kernel = np.ones(k) / k
    rnd_s = np.convolve(rnd, kernel, mode='same')
    rnd_s = np.maximum(rnd_s, 0)
    if rnd_s.sum() > 1e-10:
        inits.append(rnd_s / rnd_s.sum())

    # Random fine structure + smooth
    rnd_fine = rng.random(n)
    rnd_coarse = np.interp(x, np.linspace(0, 1, 10), rng.random(10) + 0.1)
    combo = rnd_coarse * (1 + 0.3 * rnd_fine)
    combo = np.maximum(combo, 0)
    inits.append(combo / combo.sum())

    # Bump with oscillation
    base = np.exp(-((x - 0.5) ** 2) / 0.1)
    osc = 1 + 0.2 * np.sin(20 * np.pi * x)
    bump_osc = base * osc
    bump_osc = np.maximum(bump_osc, 0)
    inits.append(bump_osc / bump_osc.sum())

    # Flat top with tapered edges
    flat = np.ones(n)
    taper = max(5, n // 20)
    for i in range(taper):
        scale = 0.5 * (1 + np.cos(np.pi * i / taper))
        flat[i] = scale
        flat[n - 1 - i] = scale
    inits.append(flat / flat.sum())

    if parent is not None:
        p_resampled = resample_sequence(parent, n)
        inits.append(project(p_resampled))
        # Perturbed parent
        pert = p_resampled * np.exp(rng.normal(0, 0.15, n))
        inits.append(project(pert))
        # More aggressive perturbation
        pert2 = p_resampled * np.exp(rng.normal(0, 0.4, n))
        inits.append(project(pert2))

    return inits


def multi_resolution_optimize(a, rng, time_budget=200.0, verbose=False):
    """
    Multi-resolution: downsample, optimize, upsample, optimize at full res.
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    if n < 100:
        return a, best_obj

    # Downsample to n/4
    n_coarse = max(50, n // 4)
    a_coarse = resample_sequence(a, n_coarse)
    a_coarse = project(a_coarse)

    # Optimize coarse
    a_c, obj_c = fista_optimize(a_coarse, max_iters=500, tau_start=0.3,
                                 tau_end=0.02, step_start=0.4, step_end=0.1,
                                 time_budget=time_budget * 0.3, lp_every=0)
    if obj_c < compute_obj(a_coarse):
        if verbose:
            print(f"      Coarse (n={n_coarse}): {obj_c:.8f}")

    # Upsample back
    a_up = resample_sequence(a_c, n)
    a_up = project(a_up)

    # Fine optimize
    remaining = time_budget - (time.monotonic() - (time.monotonic() - time_budget))  # approximate
    a_f, obj_f = fista_optimize(a_up, max_iters=500, tau_start=0.2,
                                 tau_end=0.01, step_start=0.3, step_end=0.05,
                                 time_budget=time_budget * 0.6, lp_every=200)
    if obj_f < best_obj:
        best_obj = obj_f
        best_a = a_f.copy()

    return best_a, best_obj


def perturb_and_reopt(a, rng, time_budget=30.0):
    """Perturb the sequence and re-optimize."""
    n = len(a)
    a_pert = a * np.exp(rng.normal(0, 0.1, n))
    a_pert = project(a_pert)

    # Try different perturbation scales
    best_a = a.copy()
    best_obj = compute_obj(a)

    for scale in [0.05, 0.1, 0.2, 0.3]:
        if time.monotonic() > time_budget - 5:
            break
        a_p = a * np.exp(rng.normal(0, scale, n))
        a_p = project(a_p)
        a_opt, obj_opt = fista_optimize(a_p, max_iters=200, tau_start=0.3,
                                         tau_end=0.02, step_start=0.3, step_end=0.05,
                                         time_budget=max(5.0, time_budget / 4), lp_every=0)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FISTA with adaptive tau + LP polishing + multi-resolution.
    Explores multiple sequence lengths and initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"No parent. Starting uniform n=700: obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # === Phase 1: Screen initializations at multiple lengths (15% time) ===
    print("\n=== Phase 1: Initialization Screening ===")
    phase1_budget = min(150.0, (deadline - start_time) * 0.12)
    n_options = sorted(set([
        max(200, n_parent - 100), n_parent,
        n_parent + 100, n_parent + 200,
        min(3000, n_parent + 400),
        500, 1000, 1500, 2000
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000]
    n_options = list(dict.fromkeys(n_options))  # preserve order, remove dupes

    candidates = []
    per_n_budget = max(5.0, phase1_budget / len(n_options))

    for n_try in n_options:
        if time.monotonic() > start_time + phase1_budget:
            break
        inits = generate_inits(n_try, rng, parent)
        per_init = max(2.0, per_n_budget / len(inits))

        for init in inits:
            if time.monotonic() > start_time + phase1_budget:
                break
            # Quick FISTA
            a_q, obj_q = fista_optimize(init, max_iters=min(100, int(per_init * 10)),
                                         tau_start=0.3, tau_end=0.05,
                                         step_start=0.4, step_end=0.1,
                                         time_budget=max(2.0, per_init), lp_every=0)
            if obj_q < np.inf:
                candidates.append((obj_q, a_q, n_try))

    candidates.sort(key=lambda x: x[0])
    if candidates:
        print(f"  Screened {len(candidates)} candidates, best: {candidates[0][0]:.8f} (n={candidates[0][2]})")
        for obj_c, a_c, n_c in candidates[:3]:
            if obj_c < best_score:
                best_score = obj_c
                best_seq = a_c.copy()
                print(f"  New best from screening: n={n_c}, obj={best_score:.8f}")

    # === Phase 2: Intensive FISTA on best candidate (50% time) ===
    print(f"\n=== Phase 2: Intensive FISTA (best score: {best_score:.8f}) ===")
    phase2_budget = min(700.0, (deadline - start_time) * 0.50)
    phase2_start = time.monotonic()

    a_work = project(best_seq.copy().astype(np.float64))
    n_work = len(a_work)

    # Long FISTA with LP polishing
    remaining2 = phase2_budget - (time.monotonic() - phase2_start)
    a_f, obj_f = fista_optimize(a_work, max_iters=5000,
                                 tau_start=0.5, tau_end=0.005,
                                 step_start=0.5, step_end=0.02,
                                 time_budget=remaining2 * 0.6,
                                 lp_every=150, n_lp_budget=8.0,
                                 verbose=True)
    if obj_f < best_score:
        best_score = obj_f
        best_seq = a_f.copy()
        print(f"  After long FISTA: {best_score:.8f}")

    # Multi-resolution on the best
    remaining2 = phase2_budget - (time.monotonic() - phase2_start)
    if remaining2 > 30 and len(best_seq) > 300:
        a_mr, obj_mr = multi_resolution_optimize(best_seq, rng,
                                                  time_budget=min(remaining2 * 0.5, 120.0),
                                                  verbose=True)
        if obj_mr < best_score:
            best_score = obj_mr
            best_seq = a_mr.copy()
            print(f"  After multi-res: {best_score:.8f}")

    # === Phase 3: LP polishing + perturbation (20% time) ===
    print(f"\n=== Phase 3: LP + Perturbation (best: {best_score:.8f}) ===")
    phase3_budget = min(300.0, (deadline - time.monotonic()) * 0.25)
    phase3_start = time.monotonic()

    # Intensive LP cycling
    n_best = len(best_seq)
    if n_best <= 1500:
        a_cur = project(best_seq.copy())
        for lp_round in range(20):
            if time.monotonic() > phase3_start + phase3_budget * 0.5:
                break
            remaining_lp = phase3_budget * 0.5 - (time.monotonic() - phase3_start)
            if remaining_lp < 5:
                break
            a_lp, obj_lp = lp_polish(a_cur, time_budget=min(20.0, remaining_lp))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_cur = a_lp.copy()
                print(f"  LP round {lp_round}: {best_score:.8f}")
            else:
                break

    # Perturbation + re-optimization
    remaining3 = phase3_budget - (time.monotonic() - phase3_start)
    for pert_round in range(min(5, int(remaining3 / 30))):
        if time.monotonic() > phase3_start + phase3_budget:
            break
        a_p, obj_p = perturb_and_reopt(best_seq, rng,
                                        time_budget=min(remaining3 * 0.2, 40.0))
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Perturb {pert_round}: {best_score:.8f}")

    # === Phase 4: Multi-length exploration (15% time) ===
    print(f"\n=== Phase 4: Multi-Length (best: {best_score:.8f}) ===")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 30:
        phase4_budget = min(remaining4 * 0.6, 200.0)
        phase4_start = time.monotonic()
        n_cur = len(best_seq)

        # Try nearby lengths
        n_tries = sorted(set([
            max(200, n_cur - 50), max(200, n_cur - 100),
            n_cur + 50, n_cur + 100, n_cur + 200,
            max(200, n_cur - 200)
        ]))
        n_tries = [n for n in n_tries if 200 <= n <= 3000]
        n_tries = list(dict.fromkeys(n_tries))

        per_n4 = max(5.0, phase4_budget / len(n_tries))

        for n_try in n_tries:
            if time.monotonic() > phase4_start + phase4_budget:
                break
            a_res = resample_sequence(best_seq, n_try)
            a_res = project(a_res)

            a_o, obj_o = fista_optimize(a_res, max_iters=300,
                                         tau_start=0.3, tau_end=0.01,
                                         step_start=0.4, step_end=0.05,
                                         time_budget=max(5.0, per_n4 * 0.7),
                                         lp_every=100 if n_try <= 1200 else 0)
            if obj_o < best_score:
                best_score = obj_o
                best_seq = a_o.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_o:.8f}")

    # === Phase 5: Final polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish (best: {best_score:.8f}) ===")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 15 and best_seq is not None:
        a_final = project(best_seq.copy().astype(np.float64))
        n_final = len(a_final)

        # One more long FISTA with very small tau
        a_p, obj_p = fista_optimize(a_final, max_iters=3000,
                                     tau_start=0.1, tau_end=0.003,
                                     step_start=0.2, step_end=0.01,
                                     time_budget=min(remaining5 * 0.5, 120.0),
                                     lp_every=100 if n_final <= 1200 else 0,
                                     n_lp_budget=5.0, verbose=True)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Final FISTA: {best_score:.8f}")

        # Final LP if applicable
        remaining5 = deadline - time.monotonic()
        if remaining5 > 10 and n_final <= 1200:
            a_lp, obj_lp = lp_polish(best_seq, time_budget=min(remaining5 * 0.5, 30.0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
