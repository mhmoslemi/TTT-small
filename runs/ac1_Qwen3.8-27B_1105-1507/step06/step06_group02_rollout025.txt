import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a, b=None):
    """Full convolution via FFT. If b is None, auto-convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: flatten power spectrum while maintaining non-negativity."""
    a = np.maximum(init.astype(np.float64), 1e-8)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_max = mag.max() + 1e-30
        mag_mean = mag.mean()
        alpha = beta * (0.5 + 0.5 * it / n_iters)
        mag_new = (1.0 - alpha) * mag + alpha * mag_mean
        mag_new = np.maximum(mag_new, 0.1 * mag_mean)
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 1e-8)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def smooth_max_grad(a, tau):
    """Compute gradient of smooth approximation to max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, p, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.4, lr_end=0.003):
    """Nesterov-accelerated gradient descent on smooth max of auto-convolution."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta_ns = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_ns * beta_ns)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta_ns * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new
    return best_a

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 30})
            if not r.success or r.x is None:
                return None
            g = np.maximum(r.x, 0.0)
            if g.sum() < 1e-15:
                return None

            b_ag = fast_conv(a, g)
            violations = b_ag - M
            max_viol = float(np.max(violations))

            if max_viol > tol * max(M, 1e-10):
                violated_idx = np.where(violations > tol * max(M, 1e-10))[0]
                not_in_set = set(violated_idx.tolist()) - constraint_set
                if not_in_set:
                    add_candidates = sorted(not_in_set, key=lambda k: -violations[k])
                    n_add = min(len(add_candidates), max(5, n // 16))
                    constraint_set.update(add_candidates[:n_add])
                    continue
            return g
        except Exception:
            pass

    # Fallback: solve with all constraints
    k_list = list(range(2 * n - 1))
    A = build_sparse_conv_matrix(n, a, k_list)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 60})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def lp_full(a, M):
    """Solve the full LP with all constraints (for final polish)."""
    n = len(a)
    k_list = list(range(2 * n - 1))
    A = build_sparse_conv_matrix(n, a, k_list)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 120})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=100, t_base=0.3, min_t=0.01,
                  use_cg=True, initial_K=None):
    """LP mirror descent with constraint generation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break

        if use_cg:
            g = lp_constraint_gen(a, M, max_rounds=30, initial_K=initial_K)
        else:
            g = lp_full(a, M)

        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        if it < 3:
            t = t_base
        else:
            t = max(min_t, t_base / (1.0 + 0.08 * it))

        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


def run_lp_mirror_adaptive(a, deadline, max_iters=80, use_cg=True):
    """LP mirror descent with adaptive step size (backtracking)."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break

        if use_cg:
            g = lp_constraint_gen(a, M, max_rounds=25, initial_K=None)
        else:
            g = lp_full(a, M)

        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        improved = False
        for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break

        if not improved:
            a_new = np.maximum(0.7 * a + 0.3 * g, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a = a_new / s

    return best_a


def resample_seq(src, n):
    """Resample a sequence to length n using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.04, n), 0.005))
        inits.append(np.maximum(p * (1.0 + 0.1 * rng.normal(0, 1, n)), 0.005))
        inits.append(gs_flat_spectral(p, n_iters=100, beta=0.9))

    inits.append(np.ones(n) / n)

    # Power-law edge shapes
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        w = t**alpha + (1.0 - t)**alpha
        inits.append(np.maximum(w / w.sum(), 0.001))

    # Cosine bumps
    for f in [1, 2, 3]:
        for c in [0.3, 0.5, 0.7]:
            w = 0.5 + 0.5 * np.cos(2.0 * np.pi * f * (t - c))
            inits.append(np.maximum(w / w.sum(), 0.001))

    # Gaussian bumps
    for c in [0.25, 0.5, 0.75]:
        for sig in [0.1, 0.2, 0.3]:
            w = np.exp(-0.5 * ((t - c) / sig)**2) + 0.02
            inits.append(w / w.sum())

    # Asymmetric shapes (important for breaking the center-peak)
    for alpha in [1.0, 2.0, 3.0]:
        w = t**alpha + 0.1
        inits.append(w / w.sum())
        w2 = (1.0 - t)**alpha + 0.1
        inits.append(w2 / w2.sum())

    # Random smooth
    k = max(3, n // 30)
    kern = np.ones(k) / k
    r = rng.uniform(0.2, 1.0, n)
    inits.append(np.maximum(np.convolve(r, kern, 'same') + 0.05, 0.01))

    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_flat_spectral(a_rand, n_iters=100, beta=0.85))

    # Alternating-modulated (positive version)
    for f in [2, 4, 8]:
        w = 0.7 + 0.3 * np.cos(2.0 * np.pi * f * t)
        inits.append(w / w.sum())

    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print("  ** J={:.8f} n={} t={:.0f}s".format(s, a.size, el), flush=True)

    def report(self, label):
        if self.seq is not None:
            print("  {}: J={:.8f} n={}".format(label, self.score, self.seq.size), flush=True)


def run_phase_smooth(lengths, best_tracker, deadline, budget):
    """Phase 1: Smooth optimization on diverse initializations."""
    start = time.monotonic()
    best_per_length = {}
    time_per = budget / max(1, len(lengths))

    for n in lengths:
        t0 = time.monotonic()
        remaining = deadline - 200 - time.monotonic()
        if remaining < 30:
            break
        allot = min(time_per, remaining * 0.4)
        dl = time.monotonic() + allot
        if time.monotonic() >= dl:
            continue

        parent = best_tracker.seq if best_tracker.seq is not None else None
        inits = make_inits(n, np.random.default_rng(42 + n), parent)

        best_n = None
        best_M = float('inf')
        n_iters_base = max(300, int(6000 * (256.0 / n)))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()
            frac = 1.0 / (1.0 + idx * 0.4)
            sub_dl = time.monotonic() + (dl - time.monotonic()) * frac
            a = run_nesterov_smooth(a0, sub_dl, max_iters=n_iters_base,
                                    tau_start=1.0, tau_end=250.0,
                                    lr_init=0.35, lr_end=0.002)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print("  n={:4d}: M={:.7f} J={:.6f} ({:.1f}s)".format(
                n, best_M, J, time.monotonic() - t0), flush=True)
            best_tracker.update(best_n)
            best_per_length[n] = best_n.copy()

    print("Phase smooth time: {:.1f}s".format(time.monotonic() - start), flush=True)
    return best_per_length


def run_phase_lp(a, deadline, label="LP"):
    """Phase 2: LP mirror descent on best sequence."""
    start = time.monotonic()
    n = len(a)
    best_local = a.copy()
    best_M = fast_conv(a).max()
    print("  {} start: n={}, M={:.8f}".format(label, n, best_M), flush=True)

    # Constraint generation LP
    a1 = run_lp_mirror_adaptive(a, min(time.monotonic() + 300, deadline - 60),
                                max_iters=100, use_cg=True)
    M1 = fast_conv(a1).max()
    if M1 < best_M:
        best_M = M1
        best_local = a1.copy()
        print("  {} after CG-LP: M={:.8f}".format(label, best_M), flush=True)

    # Full LP polish
    if time.monotonic() < deadline - 30:
        a2 = run_lp_mirror(a1, deadline - 20, max_iters=50, t_base=0.2,
                           min_t=0.01, use_cg=False)
        M2 = fast_conv(a2).max()
        if M2 < best_M:
            best_M = M2
            best_local = a2.copy()
            print("  {} after full-LP: M={:.8f}".format(label, best_M), flush=True)

    print("  {} done: M={:.8f} ({:.1f}s)".format(label, best_M, time.monotonic() - start), flush=True)
    return best_local


def perturb_sequence(a, rng, strength=0.05):
    """Add structured noise to escape local minima."""
    n = len(a)
    t = np.linspace(0, 1, n, endpoint=False)
    # Smooth noise
    k = max(3, n // 20)
    kern = np.ones(k) / k
    noise = np.convolve(rng.normal(0, 1, n), kern, 'same')
    a_pert = a + strength * noise
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1500, **kw):

    # Fallback
    if best.seq is None:
