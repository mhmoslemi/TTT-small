import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build the sparse matrix A where (A @ g)[k] = conv(a, g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(A, c, b_ub, bounds):
    """Solve the LP: min c^T x s.t. A x <= b_ub, bounds."""
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                             'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception as e:
        print(f"  LP error: {e}", flush=True)
    return None


def smooth_gradient(a, tau=50.0):
    """Compute gradient of smoothed max of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    # Softmax gradient
    shifted = b - b_max
    exp_b = np.exp(tau * shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(2 * n - 1)
    # grad[k] = d/dc[k] (sum p_k * c_k) = p_k
    # But we need d/da of c, which is the convolution structure
    # d(c[k])/da[j] = a[k-j] (if valid)
    result = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            result[j] += p[k] * a[k - j]
    return result, b, b_max

def run_smooth_refine(a, deadline, max_iters=200, tau_start=10, tau_end=200):
    """Smooth gradient refinement with Nesterov momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    
    y = a.copy()
    momentum = 0.9
    lr = 0.5
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr_t = lr * (0.1) ** frac  # Decay learning rate
        
        grad, b, b_max = smooth_gradient(y, tau)
        
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        
        # Nesterov step
        a_new = y - lr_t * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new
    
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gs_spectral_v2(a, n_iters=200, beta=0.9):
    """Improved GS: flatten the SQUARE of the FFT (not just magnitude)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        F_sq = F * F  # This is the DFT of the convolution (padded)
        
        # We want F_sq to be concentrated at DC (index 0)
        # Current: F_sq[0] should be (sum a)^2 = 1
        # We want other F_sq[k] to be small
        
        # Strategy: modify the phase of F to make F^2 more concentrated at DC
        # F_sq[k] = F[k]^2, so |F_sq[k]| = |F[k]|^2 and phase(F_sq[k]) = 2*phase(F[k])
        # To minimize |F_sq[k]| for k != 0, we want |F[k]| to be small for k != 0
        # But we also want phase(F_sq[k]) to be such that they cancel in the ifft
        
        # Simpler approach: just flatten |F| (same as before) but with phase adjustment
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Flatten magnitude
        target_mag = mag.mean()
        mag_new = (1 - beta) * mag + beta * target_mag
        
        # Try to align phases: for the convolution, we want F[k]^2 to have zero phase
        # i.e., phase(F[k]) should be 0 or pi for all k
        # Adjust phase to be close to 0 or pi
        phase_adj = np.arctan2(np.sin(2 * phase), np.cos(2 * phase)) / 2
        # This maps phase to the nearest multiple of pi/2... not quite what we want
        # Let's just keep the original phase for now
        F_new = mag_new * np.exp(1j * phase)
        
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        
        M_c = float(fast_conv(a_candidate).max())
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    
    return best_a

Let me also think about the gradient computation more carefully.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Gerchberg-Saxton spectral initialization (adapted for convolution, not autocorrelation)
2. LP mirror descent with adaptive line search for coarse optimization
3. Nesterov-accelerated smooth gradient descent for fine-tuning
4. Perturbation-based exploration to escape local minima
5. Multi-length exploration around parent length
The convolution peak (not autocorrelation) is the target, requiring
phase-aware spectral methods.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(A, b_ub, n):
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def smooth_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    result = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            result[j] += p[k] * a[k - j]
    return result, b, b_max


def run_lp_adaptive(a, deadline, max_iters=500):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        
        A = build_conv_matrix(n, a)
        b_ub = np.full(2 * n - 1, M)
        g = solve_lp(A, b_ub, n)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Line search
        best_t = 0.1
        best_t_M = float('inf')
        for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
            a_t = (1 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = float(fast_conv(a_t).max())
            if M_t < best_t_M:
                best_t_M = M_t
                best_t = t
        
        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        if it % 5 == 0:
            print(f"    LP[{it}] M={M:.8f} best={best_M:.8f} t={best_t:.3f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    return best_a


def run_nesterov(a, deadline, max_iters=1000, tau0=5, tau1=300, lr0=0.5, lr1=0.01):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = (it - 1) / max_iters
        tau = tau0 * (tau1 / tau0) ** (frac ** 0.5)
        lr = lr0 * (lr1 / lr0) ** frac
        mom = 0.9
        
        grad, b, b_max = smooth_grad(y, tau)
        
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * mom * mom)) / 2.0
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new
    
    return best_a


def gs_conv(a, n_iters=150, beta=0.85):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # For convolution: we want F^2 to be concentrated at DC
        # |F^2[k]| = |F[k]|^2, so flattening |F| helps
        # Phase: we want 2*phase(F[k]) to be close to 0 for k != 0
        # This means phase(F[k]) should be close to 0 or pi
        
        # Flatten magnitude
        mag_target = mag.mean()
        mag_new = (1 - beta) * mag + beta * mag_target
        
        # Phase adjustment: pull phase toward 0 or pi
        # sin(2*phase) tells us how far from 0 or pi
        phase_signal = np.sin(2 * phase)
        # Reduce the "off-axis" phase component
        phase_adj = phase - 0.1 * beta * phase_signal
        phase_adj = np.angle(np.exp(1j * phase_adj))
        
        F_new = mag_new * np.exp(1j * phase_adj)
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = float(fast_conv(a_candidate).max())
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Phase-aware Gerchberg-Saxton for convolution (not autocorrelation)
2. LP mirror descent with vectorized matrix construction and adaptive line search
3. Nesterov-accelerated smooth gradient descent for fine-tuning
4. Stochastic perturbation to escape local minima
5. Multi-length exploration
Key: DFT(conv(a,a)) = DFT(a)^2, so phase of DFT(a) matters (unlike autocorrelation).

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_matrix_structure(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def make_conv_matrix(n, a, rows, cols):
    vals = a[rows - cols]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(A, b_ub, n):
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def smooth_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / (exp_b.sum() + 1e-30)
    result = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            result[j] += p[k] * a[k - j]
    return result, b, b_max

def run_nesterov(a, deadline, max_iters=5000, tau0=5, tau1=300, lr0=0.4, lr1=0.005):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()
    mom = 0.9
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = (it - 1) / max_iters
        tau = tau0 * (tau1 / tau0) ** (frac ** 0.4)
        lr = lr0 * (lr1 / lr0) ** frac
        
        grad, b, b_max = smooth_grad(y, tau)
        
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * mom * mom)) / 2.0
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new
    
    return best_a


def gs_conv(a, n_iters=150, beta=0.85):
    """Phase-aware GS for convolution: flatten |F| and align phases of F^2 toward 0."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Flatten magnitude (equal energy across frequencies)
        mag_target = mag.mean()
        mag_new = (1 - beta) * mag + beta * mag_target
        
        # Phase alignment: want F[k]^2 to have phase ~ 0 (mod 2*pi)
        # i.e., 2*phase(F[k]) ~ 0, so phase(F[k]) ~ 0 or pi
        # Measure "misalignment": sin(2*phase) is 0 when phase is 0 or pi/2 or pi
        # We want phase close to 0 or pi, so we penalize sin(phase)*cos(phase) = 0.5*sin(2*phase)
        # Actually we want 2*phase(F[k]) to be close to 0 (mod 2*pi)
        # phase(F^2[k]) = 2*phase(F[k])
        # We want this to be 0, so phase(F[k]) should be 0 (mod pi)
        # sin(2*phase) measures deviation from 0 (mod pi) for phase
        phase_penalty = np.sin(2 * phase)
        # Reduce the phase that causes F^2 to have non-zero phase
        phase_adj = phase - 0.15 * beta * phase_penalty
        phase_adj = np.angle(np.exp(1j * phase_adj))
        
        F_new = mag_new * np.exp(1j * phase_adj)
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = float(fast_conv(a_candidate).max())
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    
    return best_a


def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
