def compute_conv(a):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))

def solve_active_set(a, active_lags, M_bound):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for k in active_lags, sum(g)=1, g>=0
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix for only active lags
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[idx, i] = a_norm[j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

        except:

    size = 1
    while size < 2*n - 1:

def line_search(a, direction, n_trials=30):
    """Find best blend of a and direction."""
    best_t = 0.0
    best_obj = obj_val(a)
    
    # Coarse search
    for t in np.linspace(0.05, 1.0, n_trials):
        a_try = np.maximum((1-t)*a + t*direction, 0)
        s = a_try.sum()

        a_try /= s
        o = obj_val(a_try)
        if o < best_obj:
            best_obj = o
            best_t = t
    
    # Fine search around best
    if 0 < best_t < 1:

def subgradient_polish(a, n_iters=500, time_budget=60.0):
    """Polish with subgradient steps on the peak lag."""
    n = len(a)
    a = a.copy().astype(np.float64) / a.sum()
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    step = 0.005
    v = np.zeros(n)
    beta = 0.9
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = fft_conv(a)
        k_star = int(np.argmax(conv))
        
        # Gradient of conv[k_star] w.r.t. a: grad[m] = 2*a[k_star - m] if valid
        grad = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                grad[m] = 2.0 * a[j]
        
        # Center to maintain sum
        gc = grad - grad.mean()
        
        # Momentum
        v = beta * v - step * gc
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
            o = obj_val(a_new)
            if o < best_obj:
                best_obj = o
                best_a = a_new.copy()
                a = a_new
            else:
                step *= 0.95
                v *= 0.5
        else:
            step *= 0.5
            v = np.zeros(n)
        
        # Decay step over time
        if it > 100:
            step = max(1e-5, step * 0.999)
    
    return best_a

"""
LP-alternating optimization with HiGHS solver, subgradient polishing,
active-set awareness, and multi-n exploration for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: fast sparse LP via HiGHS, momentum-based subgradient,
adaptive stepping, and efficient time allocation across phases.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import scipy.sparse as sp
import time


def fft_conv(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_val(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fft_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_sparse_B(a):
    """Build sparse matrix B (2n-1 x n) where B[k,i] = a[k-i] if 0<=k-i<n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(a, time_limit=8.0):
    """
    Solve LP: min t s.t. conv(a,g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g_normalized, t_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    B = build_sparse_B(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = sp.hstack([B, -sp.identity(2 * n - 1, format='csr')], format='csc')
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g_val = np.maximum(res.x[:n], 0)
            s = g_val.sum()
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None

def subgradient_polish(a, n_iters=3000, time_budget=60.0):
    """
    Momentum-based subgradient polish targeting the peak lag.
    Uses adaptive step size and gradient averaging.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()

    step = 0.003
    v = np.zeros(n)
    beta = 0.92
    grad_avg = np.zeros(n)
    n_avg = 0
    patience = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = fft_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient of conv[k_star] w.r.t. a: grad[m] = 2*a[k_star-m] if valid
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        for m in range(j_start, j_end + 1):
            grad[m] = 2.0 * a[k_star - m]

        # Center gradient (maintain sum=1)
        gc = grad - grad.mean()

        # Average gradient for smoother direction
        n_avg = min(n_avg + 1, 50)
        grad_avg = (grad_avg * (n_avg - 1) + gc) / n_avg if n_avg > 1 else gc

        # Momentum update
        v = beta * v - step * grad_avg
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
            o = obj_val(a_new)
            if o < best_obj:
                best_obj = o
                best_a = a_new.copy()
                a = a_new
                patience = 0
            else:
                patience += 1
                if patience > 10:
                    step *= 0.9
                    v *= 0.3
                    patience = 0
        else:
            step *= 0.5
            v = np.zeros(n)

        # Adaptive step decay
        if it > n_iters // 3:
            step = max(1e-6, step * 0.9995)

    return best_a


def coordinate_polish(a, time_budget=30.0):
    """
    Coordinate descent: adjust pairs to reduce peak.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = fft_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from peak-contributing positions to others
        contrib = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                contrib[m] = a[j]

        # Sort by contribution (high contrib = more to reduce)
        order = np.argsort(contrib)[::-1]

        for idx in range(min(20, n)):
            i = order[idx]
            # Find position with low contribution
            j = order[-1 - (idx % 10)] if n > 10 else (i + 1) % n
            if i == j:
                continue
            if a[i] < 1e-10:
                continue

            for frac in [0.02, 0.05, 0.1, 0.15]:
                delta = frac * a[i]
                a[i] -= delta
                a[j] += delta
                o = obj_val(a)
                if o < best_obj - 1e-12:
                    best_obj = o
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    break

        if not improved:
            break

    return best_a


def multi_start_lp(n, n_restarts=4, time_budget=100.0, seed=0):
    """Try multiple random initializations and LP-optimize each."""
    rng = np.random.default_rng(seed)
    best_a = None
    best_obj = np.inf

    for r in range(n_restarts):
        remaining = time_budget - (r / n_restarts) * time_budget
        if remaining < 10:
            break
        alloc = remaining * 0.9 / (n_restarts - r)

        # Different initialization strategies
        mode = r % 4
        if mode == 0:
            a = np.ones(n) / n
        elif mode == 1:
            a = rng.uniform(0.1, 1.0, n)
            a = a / a.sum()
        elif mode == 2:
            # Smooth bump
            center = n * rng.uniform(0.3, 0.7)
            width = n * rng.uniform(0.15, 0.4)
            x = np.arange(n)
            a = np.exp(-0.5 * ((x - center) / width) ** 2)
            a = a + 0.1 * rng.uniform(0, 1, n)
            a = a / a.sum()
        else:
            # Cosine window
            x = np.linspace(0, np.pi, n)
            a = 0.5 + 0.5 * np.cos(x)
            a = a + 0.05 * rng.uniform(0, 1, n)
            a = a / a.sum()

        a_opt = lp_alternate(a, time_budget=alloc)
        o = obj_val(a_opt)
        if o < best_obj:
            best_obj = o
            best_a = a_opt.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function.
    Uses LP alternating (HiGHS), subgradient polishing, coordinate search,
    multi-n exploration, and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Load parent ---
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s > 0.01:
            parent = parent / s
            best_score = obj_val(parent)
            best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        best_seq = np.ones(800) / 800
        best_score = obj_val(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    # --- Phase 1: Intensive LP alternating on parent ---
    print(f"\n{'='*60}")
    print(f"Phase 1: LP Alternating (n={len(best_seq)})")
    print(f"{'='*60}")
    remaining = deadline - time.monotonic()
    alloc = min(350.0, remaining * 0.30)
    print(f"  Time budget: {alloc:.0f}s")
    a = lp_alternate(best_seq.copy(), time_budget=alloc, verbose=True)
    o = obj_val(a)
    if o < best_score:
        best_score = o
        best_seq = a.copy()
        print(f"  Result: obj={best_score:.10f}")
    else:
        print(f"  No improvement (obj={o:.10f})")

    # --- Phase 2: Subgradient polish ---
    print(f"\n{'='*60}")
    print(f"Phase 2: Subgradient Polish")
    print(f"{'='*60}")
    remaining = deadline - time.monotonic()
    alloc = min(120.0, remaining * 0.10)
    print(f"  Time budget: {alloc:.0f}s")
    a = subgradient_polish(best_seq.copy(), n_iters=10000, time_budget=alloc)
    o = obj_val(a)
    if o < best_score:
        best_score = o
        best_seq = a.copy()
        print(f"  Result: obj={best_score:.10f}")
    else:
        print(f"  No improvement (obj={o:.10f})")

    # --- Phase 3: More LP alternating ---
    print(f"\n{'='*60}")
    print(f"Phase 3: Continued LP Alternating")
    print(f"{'='*60}")
    remaining = deadline - time.monotonic()
    alloc = min(200.0, remaining * 0.18)
    print(f"  Time budget: {alloc:.0f}s")
    a = lp_alternate(best_seq.copy(), time_budget=alloc, verbose=True)
    o = obj_val(a)
    if o < best_score:
        best_score = o
        best_seq = a.copy()
        print(f"  Result: obj={best_score:.10f}")
    else:
        print(f"  No improvement (obj={o:.10f})")

    # --- Phase 4: Coordinate polish ---
    print(f"\n{'='*60}")
    print(f"Phase 4: Coordinate Polish")
    print(f"{'='*60}")
    remaining = deadline - time.monotonic()
    alloc = min(60.0, remaining * 0.05)
    a = coordinate_polish(best_seq.copy(), time_budget=alloc)
    o = obj_val(a)
    if o < best_score:
        best_score = o
        best_seq = a.copy()
        print(f"  Result: obj={best_score:.10f}")
    else:
        print(f"  No improvement (obj={o:.10f})")

    # --- Phase 5: Multi-n exploration ---
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-n Exploration")
    print(f"{'='*60}")
    n_parent = len(best_seq)
    n_options = sorted(set([
        max(400, n_parent - 200),
        max(400, n_parent - 100),
        n_parent,
        n_parent + 100,
        n_parent + 200,
        n_parent + 400,
        600, 800, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000]
    # Remove duplicates and sort
    n_options = sorted(set(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s > 1e-10:
                a_try = a_try / s

        alloc = min(80.0, remaining * 0.2)
        a_opt = lp_alternate(a_try, time_budget=alloc)
        o = obj_val(a_opt)
        if o < best_score:
            best_score = o
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={o:.10f}")

    # --- Phase 6: Random restarts ---
    print(f"\n{'='*60}")
    print(f"Phase 6: Random Restarts")
    print(f"{'='*60}")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_restarts = min(6, int(remaining / 40))

    for r in range(max(1, n_restarts)):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(50.0, remaining * 0.3)

        mode = r % 3
        if mode == 0:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        elif mode == 1:
            # Different n
            n_r = n_best + int(rng.integers(-200, 200))
            n_r = max(400, min(2000, n_r))
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_r)
            a_r = np.interp(x_new, x_old, best_seq)
            a_r = a_r * np.exp(rng.normal(0, 0.05, n_r))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        else:
            # Fresh random
            n_r = n_best
            a_r = rng.uniform(0.1, 1.0, n_r)
            a_r = a_r / a_r.sum()

        a_opt = lp_alternate(a_r, time_budget=alloc)
        o = obj_val(a_opt)
        if o < best_score:
            best_score = o
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}, n={len(a_opt)}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={o:.10f}")

    # --- Phase 7: Final intensive polish ---
    print(f"\n{'='*60}")
    print(f"Phase 7: Final Polish")
    print(f"{'='*60}")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        # Subgradient
        alloc = min(remaining * 0.25, 80.0)
        a = subgradient_polish(best_seq.copy(), n_iters=50000, time_budget=alloc)
        o = obj_val(a)
        if o < best_score:
            best_score = o
            best_seq = a.copy()
            print(f"  SG polish: obj={best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # LP
        alloc = min(remaining * 0.35, 100.0)
        a = lp_alternate(best_seq.copy(), time_budget=alloc, verbose=True)
        o = obj_val(a)
        if o < best_score:
            best_score = o
            best_seq = a.copy()
            print(f"  Final LP: obj={best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Coordinate
        alloc = min(remaining * 0.3, 30.0)
        a = coordinate_polish(best_seq.copy(), time_budget=alloc)
        o = obj_val(a)
        if o < best_score:
            best_score = o
            best_seq = a.copy()
            print(f"  Coord: obj={best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        # One more subgradient with very small steps
        alloc = min(remaining * 0.5, 20.0)
        a = subgradient_polish(best_seq.copy(), n_iters=100000, time_budget=alloc)
        o = obj_val(a)
        if o < best_score:
            best_score = o
            best_seq = a.copy()
            print(f"  Final SG: obj={best_score:.10f}")

    # --- Final validation ---
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
