In game theory, the "shapley-folkman" or "averaging" trick says that the average of the iterates converges to the value of the game, even when the iterates themselves don't converge. 2. Average all a-iterates (or the last T/2 iterates)
3. Use the average as the final solution

"""
LP alternating optimization with iterate averaging for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses game-theoretic best response
with running averages for convergence acceleration, subgradient
polishing, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def build_shift_vectors(g, n):
    """Build matrix where row k is S_k * g (the k-th shifted version of g)."""
    # (S_k g)[j] = g[k-j] if 0 <= k-j < n, else 0
    # So row k of the matrix has entry g[k-j] at column j
    # This is equivalent to the convolution matrix
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = g[idx]
    return A

def solve_lp_given_other(other, n):
    """
    Solve LP: min M s.t. conv(x, other)[k] <= M for all k,
    sum(x) = 1, x >= 0.
    Returns optimal x (normalized) and M.
    """
    # Build constraint matrix
    # conv(x, other)[k] = sum_j x[j] * other[k-j]
    # So the k-th constraint is: sum_j x[j] * other[k-j] - M <= 0
    # Let c_k[j] = other[k-j] if valid, else 0
    
    # For efficiency, build the matrix using the convolution structure
    A_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                row[j] = other[idx]
        A_rows.append(row)
    A_conv = np.array(A_rows)  # shape (2n-1, n)
    
    # Full LP matrix: (A_conv, -1) * [x; M] <= 0
    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(x) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Bounds
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status == 0:
        x = np.maximum(result.x[:n], 0)
        s = float(np.sum(x))
        if s > 1e-10:
            x = x / s
        M = float(result.x[n])
        return x, M
    return None, None

def subgradient_refine(a_init, n, max_iters=10000, time_budget=60.0):
    """
    Subgradient method on the normalized objective.
    """
    a = a_init / np.sum(a_init)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    eta = 0.1  # initial step size
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute convolution and find peak
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of max_k conv(a,a)[k] at a:
        # = 2 * S_{k_star} a, where (S_k a)[i] = a[k-i] if valid
        grad = np.zeros(n)
        for i in range(n):
            idx = k_star - i
            if 0 <= idx < n:
                grad[i] = 2.0 * a[idx]
        
        # Center the gradient (for the sum constraint)
        grad_centered = grad - np.mean(grad)
        
        # Take step
        eta = max(eta * 0.999, 0.001)
        a_new = a - eta * grad_centered
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
    
    return best_a

"""
LP alternating optimization with iterate averaging for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses game-theoretic best response LPs
solved via HiGHS, running averages for convergence acceleration,
subgradient polishing, and multi-n exploration.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(other, n):
    """Build matrix A (2n-1 x n) where A[k,j] = other[k-j] if valid else 0."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            A[k, j_start:j_end + 1] = other[idx]
    return A


def solve_lp_player(other, n):
    """
    Solve LP: min M s.t. conv(x, other)[k] <= M for all k,
    sum(x) = 1, x >= 0.
    Returns (x_normalized, M) or (None, None).
    """
    A_conv = build_conv_matrix(other, n)

            break

        # Compute convolution and find all peak lags
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Subgradient: use the argmax lag
        k_star = int(np.argmax(conv))

        # Gradient of conv(a,a)[k_star] w.r.t. a
        # = 2 * a[k_star - i] for valid i
        grad = np.zeros(n)
        for i in range(n):
            idx = k_star - i
            if 0 <= idx < n:
                grad[i] = 2.0 * a[idx]

        # Center gradient for sum constraint
        grad_centered = grad - np.mean(grad)

        # Step size: diminishing
        eta = base_eta / np.sqrt(it + 1)

        # Take step
        a_new = a - eta * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new

        if time.monotonic() - last_report > 20:
            print(f"    Subgrad it{it}: best={best_obj:.8f}")
            last_report = time.monotonic()

    return best_a


def momentum_subgradient(a_init, n, max_iters=30000, time_budget=60.0):
    """
    Subgradient with momentum (like heavy ball method).
    """
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    velocity = np.zeros(n)
    momentum = 0.9

    start = time.monotonic()
    base_eta = 0.02

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            idx = k_star - i
            if 0 <= idx < n:
                grad[i] = 2.0 * a[idx]

        grad_centered = grad - np.mean(grad)
        eta = base_eta / np.sqrt(it + 1)

        velocity = momentum * velocity - eta * grad_centered
        a_new = a - velocity
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
            else:
                # Revert velocity
                velocity *= 0.5

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.01)
