- Keeping all individual `a[i]` small (to reduce edge peaks like `a[0]^2`)

"""
Hybrid peak-aware optimization: combines logsumexp-smoothed Adam with
exact subgradient steps, top-k peak focusing, multi-n exploration,
and intensive final polish to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def topk_logsumexp_obj_grad(a, beta, k_frac=0.1):
    """
    Compute top-k logsumexp smoothed objective and gradient.
    Only considers the top k_frac fraction of convolution values.
    This gives a sharper focus on actual peaks while maintaining smoothness.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    m = len(conv)
    
    # Select top k indices
    k = max(1, int(m * k_frac))
    top_idx = np.argpartition(conv, -k)[-k:]
    top_vals = conv[top_idx]
    
    # Logsumexp on top values
    cmax = float(np.max(top_vals))
    shifted = beta * (top_vals - cmax)
    shifted = np.clip(shifted, -400, 400)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w_top = e / e_sum  # weights on top values
    
    ls = cmax + np.log(e_sum) / beta
    
    # Build full weight vector
    w = np.zeros(m)
    w[top_idx] = w_top
    
    # Objective
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    # Gradient: d(ls)/d(a[i]) = sum_k w[k] * d(conv[k])/d(a[i])
    # d(conv[k])/d(a[i]) = 2*a[k-i] (if valid and i != k-i), or a[k-i] if i == k-i
    # Using FFT: cross_correlation(w, a)
    size = 1
    while size < 2 * m:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    # cc[i] = sum_k w[k] * a[k-i] (approximately, with circular effects)
    # The gradient of ls w.r.t. a[i] is 2*cc[i] minus diagonal correction
    # For simplicity, use 2*cc[i] (the diagonal correction is small for large n)
    grad_ls = 2.0 * cc[:n]
    
    # Full gradient
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj


def subgradient_step(a, step_size, k_peaks=5):
    """
    Take a subgradient step on the exact max objective.
    Returns updated a (projected onto simplex).
    """
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    
    # Find peaks within a small tolerance of max
    tolerance = 1e-6 * max_b + 1e-12
    peak_mask = conv >= max_b - tolerance
    peak_indices = np.where(peak_mask)[0]
    
    # If too many peaks, take the top k
    if len(peak_indices) > k_peaks:
        top_k = np.argsort(conv[peak_indices])[-k_peaks:]
        peak_indices = peak_indices[top_k]
    
    # Compute subgradient: average of gradients at active peaks
    # grad_k[i] = 2*a[k-i] (if valid, i != k-i) or a[k-i] (if i == k-i)
    grad = np.zeros(n)
    for k in peak_indices:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                if i == j:
                    grad[i] += 0.5 * a[j]  # factor of 0.5 for the diagonal
                else:
                    grad[i] += a[j]
    
    # Wait, let me recompute. conv(a,a)[k] = sum_i a[i]*a[k-i]
    # d/d(a[m]) conv(a,a)[k] = sum_i [delta_{i,m}*a[k-i] + a[i]*delta_{k-i,m}]
    # = a[k-m] + a[k-m] (if both terms valid)
    # = 2*a[k-m] (if 0 <= k-m < n and m != k-m)
    # = a[k-m] (if m = k-m, i.e., k even and m = k/2)
    
    # So for a single peak k: grad_k[m] = (1 + delta_{m,k-m}) * a[k-m] if 0<=k-m<n
    # = 2*a[k-m] normally, a[k-m] on the diagonal
    
    # For multiple peaks, sum them:
    grad = np.zeros(n)
    for k in peak_indices:
        m = 0
        while m < n:
            j = k - m
            if 0 <= j < n:
                if m == j:
                    grad[m] += a[j]
                else:
                    grad[m] += a[j]  # This counts once for the "a[i]*a[k-i]" term
                    # But we also need the term from "a[k-i]*a[i]" which is the same
                    # So total is 2*a[j] for m != j
                m += 1
            else:
                m += 1
    
    # Hmm, I'm confusing myself. Let me just use the FFT-based approach.
    # d(conv(a,a)[k])/d(a) is the "cross-correlation" of a with a delta at position k
    # Which is just a shifted (and possibly flipped) version of a.
    
    # For the sum over peaks: sum_k w_k * d(conv[k])/d(a)
    # = sum_k w_k * (a shifted by k, flipped)
    # This can be computed as: ifft(FFT(w) * conj(FFT(a))) where w is the peak weight vector
    
    w = np.zeros(2*n-1)
    w[peak_indices] = 1.0 / len(peak_indices)
    
    size = 1
    while size < 2 * (2*n-1):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    # cc[i] = sum_k w[k] * a[k-i] (with circular wraparound, but for our purposes ok)
    # The gradient of sum_k w[k]*conv[k] w.r.t. a[i] is 2*cc[i] (approximately)
    grad = 2.0 * cc[:n]
    
    # Projected gradient step on simplex
    a_new = a - step_size * grad
    # Project onto simplex: min sum (a_new - x)^2 s.t. sum x = 1, x >= 0
    a_new = project_simplex(a_new)
    
    return a_new


def project_simplex(v):
    """Project v onto the probability simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fista_peak_opt(a, max_iters=2000, time_budget=50.0, k_peaks=5, 
                   lr_start=0.01, lr_decay=0.9999):
    """
    FISTA (accelerated projected gradient) on the exact max objective.
    Uses subgradient at top-k peaks.
    """
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    # FISTA variables
    y = a.copy()
    t = 1.0
    lr = lr_start
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute y on simplex
        y = project_simplex(y)
        
        # Compute gradient at y
        conv_y = compute_conv(y)
        max_b = float(np.max(conv_y))
        tolerance = 1e-8 * max_b + 1e-14
        peak_mask = conv_y >= max_b - tolerance
        peak_indices = np.where(peak_mask)[0]
        
        if len(peak_indices) > k_peaks:
            top_k = np.argsort(conv_y[peak_indices])[-k_peaks:]
            peak_indices = peak_indices[top_k]
        
        # Compute gradient via FFT
        w = np.zeros(2*n-1)
        w[peak_indices] = 1.0 / len(peak_indices)
        
        size = 1
        while size < 2 * (2*n-1):
            size <<= 1
        W = rfft(w, size)
        Y = rfft(y, size)
        cc = irfft(W * np.conj(Y), size)
        grad = 2.0 * cc[:n]
        
        # Update x
        x = project_simplex(y - lr * grad)
        
        # Update y for next iteration
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = x + (t - 1) / t_new * (x - a)
        t = t_new
        a = x
        
        # Decay learning rate
        lr *= lr_decay
        
        # Check objective periodically
        if it % 50 == 0 or it == max_iters - 1:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a


def adam_log_opt_v2(a, max_iters=10000, time_budget=100.0, beta_start=5.0,
                    beta_end=500.0, lr_start=0.05, lr_decay=0.99999,
                    k_frac=0.05):
    """
    Improved Adam in log-space with top-k logsumexp.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr_t = lr_start * (lr_decay ** it)
        
        # Compute a from x
        x_centered = x - np.max(x)
        a_curr = np.exp(x_centered)
        a_curr = a_curr / np.sum(a_curr)
        
        # Compute top-k logsumexp objective and gradient
        obj_sm, grad_a = topk_logsumexp_obj_grad(a_curr, beta, k_frac=k_frac)
        
        # Chain rule for log-space
        grad_x = grad_a * a_curr
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 200 == 0:
            x -= np.mean(x)
        
        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a


def coordinate_refine_v2(a, time_budget=30.0, seed=0):
    """
    Improved coordinate descent: pair-swaps focused on peak reduction.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

        iterations += 1
        conv = compute_conv(best_a)
        max_b = float(np.max(conv))
        
        # Find the peak position
        peak_pos = int(np.argmax(conv))
        
        # Find elements contributing to this peak
        i_min = max(0, peak_pos - (n - 1))
        i_max = min(peak_pos, n - 1)
        
        # For each contributing pair, try shifting mass
        found_improvement = False
        num_tried = 0
        
        # Pick a random contributing element
        candidates = list(range(i_min, i_max + 1))
        rng.shuffle(candidates)
        
        for i in candidates[:min(10, len(candidates))]:
            if time.monotonic() - start > time_budget:
                break
            j = peak_pos - i
            if j < 0 or j >= n:
                continue
            
            # Try moving mass from the larger of (i, j) to a random other position
            if best_a[i] >= best_a[j]:
                src = i
            else:
                src = j
            
            # Target: a random position that doesn't contribute to the peak
            all_pos = set(range(n)) - set(candidates)
            if not all_pos:
                continue
            dst = int(rng.choice(list(all_pos)))
            
            # Try different fractions
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                delta = frac * best_a[src]
                a_try = best_a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                if a_try[src] < 0:
                    continue
                s_try = np.sum(a_try)

                    found_improvement = True
                    num_tried += 1
                    break

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
