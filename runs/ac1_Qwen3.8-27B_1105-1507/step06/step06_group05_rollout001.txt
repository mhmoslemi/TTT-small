import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz_matrix(a):
    """Build matrix T where (T@g)[k] = (conv(a,g))[k]. Shape (2n-1, n)."""
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        T[k_start:k_end, j] = a[:k_end - k_start]
    return T


def solve_lp_max_sum(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    T = build_toeplitz_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def softmax_gradient_v2(a, tau_init=0.0008, step_size=0.05, max_steps=300, time_budget=50.0,

def spectral_equalization_v2(a, max_iters=80, time_budget=50.0, alpha_range=(0.3, 0.8)):
    """
    POCS-style spectral equalization with adaptive compression.
    Alternates between flattening the power spectrum and enforcing non-negativity.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Adaptive compression: start gentle, become more aggressive
        progress = it / max(max_iters, 1)
        alpha = alpha_range[0] + (alpha_range[1] - alpha_range[0]) * progress

        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)

        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend to maintain progress
            blend = 0.3 + 0.4 * (1.0 - progress)
            a = normalize((1.0 - blend) * a + blend * a_new)

    return best_a

                break
            alpha *= 0.5

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step_size_decay = 0.8
            alpha *= step_size_decay
            if alpha < 1e-9:
                # Random escape
                rng = np.random.default_rng(step * 48271 + 1)
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a


def coordinate_descent_v3(a, time_budget=25.0):
    """
    Pairwise transfer coordinate descent with peak targeting.
    Moves mass from high-contributing positions to low-contributing ones.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Find top 3 peaks
        peak_indices = np.argsort(conv)[-3:][::-1]

        improved = False
        for k_star in peak_indices:
            if time.monotonic() - start > time_budget:
                break

            # Find contributors to this peak
            contrib = []
            for j in range(n):
                k2 = int(k_star) - j
                if 0 <= k2 < n:
                    contrib.append((a[j] * a[k2], j, k2))
            contrib.sort(reverse=True)

            for peak_val, j1, j2 in contrib[:4]:
                if time.monotonic() - start > time_budget:
                    break
                if j1 == j2:
                    continue

                # Try transferring mass to a random position far from both
                for _ in range(4):
                    j_target = int(rng.integers(0, n))
                    if j_target == j1 or j_target == j2:
                        continue
                    # Also check if target position contributes to another peak
                    delta = 0.03 * a[j1]
                    if a[j1] - delta < 0:
                        continue

                    a_try = a.copy()
                    a_try[j1] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)

                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved and rng.random() < 0.15:
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def symmetric_projection(a):
    """Project a onto symmetric sequences: a_i = a_{n-1-i}."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return a_sym


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase strategy focusing on:
    1. LP alternation with adaptive tightening
    2. Target-flattening gradient (new method)
    3. Multi-n exploration
    4. Structured random restarts
    5. Intensive final polish
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # Also try symmetric version
    if best_seq is not None:
        a_sym = normalize(symmetric_projection(best_seq.copy()))
        obj_sym = compute_obj(a_sym)
        print(f"Symmetric version: n={n_parent}, score={obj_sym:.8f}")
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()

    # =========================================================
    # Phase 1: Quick multi-n exploration (200s)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-n exploration")
    print(f"{'='*60}")

    phase1_end = start_time + 200.0
    n_best = len(best_seq) if best_seq is not None else 700

    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best - 20,
        n_best, n_best + 20, n_best + 50, n_best + 100,
        n_best + 200, n_best + 300, n_best + 500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]

    for n_try in n_options:
        if time.monotonic() > phase1_end - 10:
            break
        remaining = min(phase1_end - time.monotonic(), 25.0)
        if remaining < 5:
            break

        if best_seq is not None:
            if n_try == n_best:
                a_try = normalize(best_seq.copy())
            else:
                a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try

        # Quick optimization: gradient + short LP
        a_try = softmax_gradient_v2(a_try, tau_init=0.001, step_size=0.04,
                                     max_steps=50, time_budget=remaining * 0.4)
        if n_try <= 1000 and remaining > 10:
            a_try = lp_alternation_v2(a_try, max_iters=5,
                                       time_budget=remaining * 0.4,
                                       tighten_factor=0.9)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 2: Intensive LP alternation (500s)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Intensive LP alternation (n={n_best})")
    print(f"{'='*60}")

    phase2_end = start_time + 700.0
    if best_seq is not None:
        a = normalize(best_seq.copy())

        for cycle in range(8):
            if time.monotonic() > phase2_end - 20:
                break
            remaining = phase2_end - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.15, 60.0)

            # Vary tighten factor for diversity
            tf = 0.92 + 0.06 * (cycle % 3)
            a_new = lp_alternation_v2(a, max_iters=20, time_budget=alloc,
                                       tighten_factor=tf)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  LP cycle {cycle} (tf={tf:.2f}): {best_score:.8f}")
            else:
                a = normalize(0.8 * a + 0.2 * a_new)
                print(f"  LP cycle {cycle} (tf={tf:.2f}): {obj_new:.8f} (no improvement)")

    # =========================================================
    # Phase 3: Target-flattening + spectral + gradient (400s)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Target-flattening + spectral + gradient")
    print(f"{'='*60}")

    phase3_end = start_time + 1100.0
    if best_seq is not None:
        a = normalize(best_seq.copy())

        # 3a: Target flattening gradient
        if time.monotonic() < phase3_end - 200:
            a_tf = target_flattening_gradient(a, max_steps=150, time_budget=50.0)
            obj_tf = compute_obj(a_tf)
            if obj_tf < best_score:
                best_score = obj_tf
                best_seq = a_tf.copy()
                a = a_tf
                print(f"  Target-flattening: {best_score:.8f}")
            else:
                print(f"  Target-flattening: {obj_tf:.8f} (no improvement)")
                a = normalize(0.7 * a + 0.3 * a_tf)

        # 3b: Spectral equalization
        if time.monotonic() < phase3_end - 150:
            a_spec = spectral_equalization_v2(a, max_iters=60, time_budget=45.0,
                                              alpha_range=(0.4, 0.7))
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                a = a_spec
                print(f"  Spectral: {best_score:.8f}")
            else:
                print(f"  Spectral: {obj_spec:.8f} (no improvement)")
                a = normalize(0.7 * a + 0.3 * a_spec)

        # 3c: Soft-max gradient
        if time.monotonic() < phase3_end - 100:
            a_soft = softmax_gradient_v2(a, tau_init=0.0005, step_size=0.04,
                                         max_steps=200, time_budget=45.0)
            obj_soft = compute_obj(a_soft)
            if obj_soft < best_score:
                best_score = obj_soft
                best_seq = a_soft.copy()
                a = a_soft
                print(f"  SoftMax gradient: {best_score:.8f}")
            else:
                print(f"  SoftMax gradient: {obj_soft:.8f} (no improvement)")

        # 3d: Follow-up LP
        if time.monotonic() < phase3_end - 50:
            a_new = lp_alternation_v2(a, max_iters=15, time_budget=50.0,
                                       tighten_factor=0.93)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  Post Phase 3 LP: {best_score:.8f}")

    # =========================================================
    # Phase 4: Structured random restarts (200s)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Structured random restarts")
    print(f"{'='*60}")

    phase4_end = start_time + 1300.0
    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed)

        restart_strategies = [
            "gaussian", "sinusoidal", "block_reverse", "dirichlet_blend",
            "taper", "bump", "dual_peak", "random_symmetric"
        ]

        for restart in range(10):
            if time.monotonic() > phase4_end - 10:
                break
            remaining = phase4_end - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.13, 20.0)

            strategy = restart_strategies[restart % len(restart_strategies)]
            idx = np.arange(n_cur) / n_cur

            if strategy == "gaussian":
                a_try = best_seq * np.exp(rng.normal(0, 0.06, n_cur))
            elif strategy == "sinusoidal":
                freq = rng.uniform(1.0, 8.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * idx + phase)
                a_try = best_seq * np.maximum(mod, 0.1)
            elif strategy == "block_reverse":
                a_try = best_seq.copy()
                block_len = int(rng.integers(n_cur // 5, n_cur // 3))
                start_idx = int(rng.integers(0, n_cur - block_len))
                a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]
            elif strategy == "dirichlet_blend":
                a_rand = rng.dirichlet(np.ones(n_cur) * 0.3)
                mix = rng.uniform(0.15, 0.4)
                a_try = (1 - mix) * normalize(best_seq) + mix * a_rand
            elif strategy == "taper":
                taper = np.ones(n_cur)
                taper_width = int(rng.integers(n_cur // 10, n_cur // 3))
                taper[:taper_width] = np.linspace(0.3, 1.0, taper_width)
                taper[-taper_width:] = np.linspace(1.0, 0.3, taper_width)
                a_try = best_seq * taper
            elif strategy == "bump":
                center = int(rng.integers(n_cur // 4, 3 * n_cur // 4))
                width = int(rng.integers(n_cur // 8, n_cur // 4))
                bump = 1.0 - 0.3 * np.exp(-0.5 * ((np.arange(n_cur) - center) / width) ** 2)
                a_try = best_seq * bump
            elif strategy == "dual_peak":
                c1 = int(rng.integers(n_cur // 5, n_cur // 3))
                c2 = int(rng.integers(2 * n_cur // 3, 4 * n_cur // 5))
                w = int(n_cur // 10)
                mod = 1.0
                mod -= 0.25 * np.exp(-0.5 * ((np.arange(n_cur) - c1) / w) ** 2)
                mod -= 0.25 * np.exp(-0.5 * ((np.arange(n_cur) - c2) / w) ** 2)
                a_try = best_seq * np.maximum(mod, 0.1)
            else:  # random_symmetric
                a_rand = rng.random(n_cur)
                a_rand = symmetric_projection(a_rand)
                a_try = 0.6 * normalize(best_seq) + 0.4 * normalize(a_rand)

            a_try = normalize(a_try)

            # Optimize
            if n_cur <= 1000 and alloc > 8:
                a_try = lp_alternation_v2(a_try, max_iters=8, time_budget=alloc * 0.4,
                                           tighten_factor=0.92)
            a_try = softmax_gradient_v2(a_try, tau_init=0.001, step_size=0.03,
                                        max_steps=60, time_budget=alloc * 0.4)
            a_try = target_flattening_gradient(a_try, max_steps=50, time_budget=alloc * 0.2)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart} ({strategy}): {best_score:.8f} **")
            else:
                print(f"  Restart {restart} ({strategy}): {obj_try:.8f}")

    # =========================================================
    # Phase 5: Final intensive polish (150s)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final intensive polish")
    print(f"{'='*60}")

    if best_seq is not None:
        n_final = len(best_seq)
        a_final = normalize(best_seq.copy())

        for polish_round in range(5):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.22, 30.0)

            # LP
            if n_final <= 1000 and alloc > 12:
                a_pol = lp_alternation_v2(a_final, max_iters=10, time_budget=alloc * 0.35,
                                           tighten_factor=0.94)
            else:
                a_pol = a_final.copy()

            # Target flattening
            a_pol = target_flattening_gradient(a_pol, max_steps=80, time_budget=alloc * 0.3)

            # Soft-max
            a_pol = softmax_gradient_v2(a_pol, tau_init=0.0003, step_size=0.02,
                                        max_steps=60, time_budget=alloc * 0.2)

            # Coordinate descent
            if alloc > 8:
                a_pol = coordinate_descent_v3(a_pol, time_budget=min(8, alloc * 0.15))

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish_round}: {best_score:.8f}")
            else:
                a_final = a_pol
                print(f"  Polish {polish_round}: {obj_pol:.8f} (no improvement)")

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Ensure sum constraint
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate with the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
