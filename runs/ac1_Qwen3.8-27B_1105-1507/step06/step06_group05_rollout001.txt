"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP alternation, spectral equalization,
subgradient descent, and multi-n exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

# Helper functions (all top-level, no closures)

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz_active(a, active_indices):
    """Build Toeplitz matrix only for active constraint rows."""
    n = len(a)
    m = len(active_indices)
    A = np.zeros((m, n), dtype=np.float64)
    for i, k in enumerate(active_indices):
        # conv(k) = sum_j a[j] * g[k-j] for valid j
        # So the row for constraint k has entries: A[k,j] = a[k-j] if 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[i, j] = a[k - j]
    return A

def solve_lp_active(a, M, active_indices, time_limit=10):
    """Solve LP with only active constraints."""
    n = len(a)
    A = build_toeplitz_active(a, active_indices)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def backtracking_line_search(a, direction, obj_current, max_step=1.0, min_step=1e-6):
    """Backtracking line search for t in [0, max_step]."""
    a_norm = normalize(a)
    d_norm = normalize(direction)
    t = max_step
    best_t = 0.0
    best_obj = obj_current
    for _ in range(20):
        a_try = normalize((1.0 - t) * a_norm + t * d_norm)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
        t *= 0.7
        if t < min_step:
            break
    return best_t, best_obj

def spectral_equalization(a, max_iters=40, time_budget=30.0):
    """POCS-style spectral equalization."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        alpha = 0.3 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.7 * a + 0.3 * a_new
            a = normalize(a)
    return best_a

def subgradient_descent(a, max_steps=500, time_budget=60.0):
    """Subgradient descent using top-K peaks."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05
    K = 10  # number of peaks to consider
    
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Get top K peak indices
        top_k = np.argsort(conv)[::-1][:K]
        
        # Compute subgradient: weighted sum of shifted a's
        grad = np.zeros(n)
        for k in top_k:
            weight = (conv[k] - cmax * 0.9) / (cmax * 0.1 + 1e-15)
            weight = max(0, weight)
            for j in range(n):
                k2 = k - j
                if 0 <= k2 < n:
                    grad[j] += weight * a[k2]
        
        # Center the gradient
        grad_centered = grad - np.mean(grad)
        
        a_try = normalize(a - step_size * grad_centered)
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            step_size = min(step_size * 1.1, 0.1)
        else:
            step_size *= 0.7
            if step_size < 1e-8:
                break
    
    return best_a

def lp_alternate_active(a, max_iters=80, K=50, time_budget=100.0):
    """LP alternation with active set (top K peaks)."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Get active set: top K peaks
        top_k = np.argsort(conv)[::-1][:K]
        
        # Solve LP with active constraints
        lp_time = min(remaining * 0.4, 8.0)
        g = solve_lp_active(a, M, top_k, time_limit=int(max(2, lp_time)))
        if g is None:
            break
        
        # Backtracking line search
        t, obj_try = backtracking_line_search(a, g, best_obj)
        
        if obj_try < best_obj - 1e-10:
            best_obj = obj_try
            a_new = normalize((1.0 - t) * normalize(a) + t * g)
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step or perturb
            if it % 10 == 0:

                a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))
            else:

    return best_a

def resample_sequence(a, new_n):
    n = len(a)
