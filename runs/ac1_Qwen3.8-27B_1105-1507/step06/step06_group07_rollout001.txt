            x -= np.mean(x)

        if it % 100 == 0 or it == max_iters - 1:
            x_c2 = x - np.max(x)
            a_check = np.exp(x_c2)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()

    return best_a


def build_toeplitz_matrix(a):
    """Build matrix T where (T @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        T[k_start:k_end, j] = a[:k_end - k_start]
    return T


def lp_peak_reduction(a, M, time_limit=20):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    if n > 1200:
        return None
    T = build_toeplitz_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True,
                                                'time_limit': time_limit,
                                                'mip_rel_gap': 0.0})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine_loop(a, time_budget=60.0, max_rounds=10):
    """
    Iterative LP-based refinement: repeatedly find g that increases sum
    without increasing peak, blend, and repeat.
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for r in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        alloc = min((time_budget - (time.monotonic() - start)) * 0.5, 15.0)
        if alloc < 2.0:
            break

        g = lp_peak_reduction(a, M, time_limit=alloc)
        if g is None:
            break

        # Try multiple blend factors
        improved = False
        for t_blend in [0.3, 0.5, 0.7, 1.0]:
            if time.monotonic() - start > time_budget - 5:
                break
            a_try = (1 - t_blend) * a + t_blend * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            # Try smaller M (tighter constraint) to find a better direction
            M_tight = M * 0.99
            g = lp_peak_reduction(a, M_tight, time_limit=min(alloc, 10.0))
            if g is not None:
                for t_blend in [0.2, 0.4, 0.6]:
                    a_try = (1 - t_blend) * a + t_blend * g
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
            if not improved:
                break

    return best_a


def subgradient_peak_chase(a, time_budget=30.0, lr_start=0.01, lr_end=0.001):
    """
    Direct subgradient method on the max of conv(a,a) in log-space.
    Focuses entirely on reducing the peak.
    """
    n = len(a)
    start = time.monotonic()
    a = np.maximum(a.copy().astype(np.float64), 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    x = np.log(a)
    x -= np.mean(x)

    best_a = a.copy()
    best_obj = compute_obj(a)
    momentum = np.zeros(n)
    beta_m = 0.9
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        progress = min(it / 5000.0, 1.0)
        lr = lr_start * (lr_end / lr_start) ** progress

        x_c = x - np.max(x)
        a_cur = np.exp(x_c)
        a_cur = a_cur / np.sum(a_cur)

        conv = compute_conv(a_cur)
        cmax = float(np.max(conv))
        sum_a = np.sum(a_cur)

        # Weight: concentrated on top values
        scale = max(cmax * 0.005, 1e-10)
        weights = np.exp((conv - cmax) / scale)
        weights /= weights.sum()

        size = 1
        while size < 2 * len(conv):
            size <<= 1
        W = rfft(weights, size)
        A = rfft(a_cur, size)
        cc = irfft(W * np.conj(A), size)
        grad_c = 2.0 * cc[:n]
        grad_obj = (2.0 * n / (sum_a ** 2)) * (grad_c - cmax / sum_a)
        grad_x = grad_obj * a_cur

        # Momentum update
        momentum = beta_m * momentum + grad_x
        x = x - lr * momentum

        if it % 50 == 0:
            x -= np.mean(x)

        if it % 200 == 0:
            x_c2 = x - np.max(x)
            a_check = np.exp(x_c2)
            a_check = a_check / np.sum(a_check)
            obj_check = compute_obj(a_check)
            if obj_check < best_obj:
                best_obj = obj_check
                best_a = a_check.copy()

    return best_a


def coordinate_refine(a, time_budget=20.0, seed=42):
    """Coordinate descent focusing on top peak contributors."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy().astype(np.float64)
    s = np.sum(best_a)
    if s < 1e-10:
        return a
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(seed)

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        # Find top peaks
        top_k = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]

        improved = False
        for p in peak_indices:
            if time.monotonic() - start > time_budget:
                break
            # Find contributors to this peak
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            # Try a random subset of contributors
            indices = list(range(i_min, i_max + 1))
            rng.shuffle(indices)
            for i in indices[:10]:
                j = p - i
                if j < 0 or j >= n or j == i:
                    continue
                # Try shifting mass
                for frac in [0.01, 0.005, 0.02]:
                    delta = frac * max(best_a[i], 1e-6)
                    if best_a[i] - delta < 0:
                        continue
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    s_try = np.sum(a_try)
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and it > 200:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Loaded parent: n={parent.size}, obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining adaptive Adam, LP refinement, subgradient chase,
    multi-n exploration, and coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_seq = parent.copy()
        best_score = parent_obj

    if best_seq is None:
        # Fallback: uniform sequence
        best_seq = np.ones(700)
        best_score = compute_obj(best_seq)
        print(f"Fallback uniform n=700: obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Initial: n={n_parent}, score={best_score:.8f}, budget={budget_s}s")

    # === Phase 1: Intensive Adam with adaptive beta (45% of time) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Adaptive Adam optimization")
    print(f"{'='*60}")

    phase1_budget = (deadline - time.monotonic()) * 0.45
    a = best_seq.copy()
    s = np.sum(a)
    if s > 1e-10:
        a = a / s

    # Multiple rounds with progressively higher beta
    rounds = [
        # (beta_start, beta_end, lr_start, lr_decay, frac_of_phase1)
        (5.0, 50.0, 0.05, 0.99997, 0.20),
        (10.0, 100.0, 0.04, 0.99997, 0.20),
        (20.0, 200.0, 0.03, 0.99998, 0.20),
        (50.0, 400.0, 0.02, 0.99998, 0.20),
        (100.0, 800.0, 0.015, 0.99999, 0.20),
    ]

    for r_idx, (bs, be, lr0, lr_dec, frac) in enumerate(rounds):
        if time.monotonic() > deadline - 60:
            break
        alloc = phase1_budget * frac
        if time.monotonic() + alloc > deadline - 30:
            alloc = deadline - 30 - time.monotonic()
        if alloc < 5:
            break

        a_new = adam_log_opt(a, max_iters=50000, time_budget=alloc,
                             beta_start=bs, beta_end=be,
                             lr_start=lr0, lr_decay=lr_dec,
                             adaptive_beta=True,
                             switch_to_subgrad_at=0.7)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Round {r_idx} (beta {bs:.0f}-{be:.0f}): {best_score:.8f} ***")
        else:
            print(f"  Round {r_idx} (beta {bs:.0f}-{be:.0f}): {obj:.8f}")

    # === Phase 2: LP refinement (25% of time) ===
    print(f"\n{'='*60}")
    print(f"Phase 2: LP-based peak reduction")
    print(f"{'='*60}")

    phase2_budget = (deadline - time.monotonic()) * 0.25
    if len(best_seq) <= 1000:
        for r in range(8):
            if time.monotonic() > deadline - 60:
                break
            alloc = min(phase2_budget / 8.0, (deadline - 60 - time.monotonic()))
            if alloc < 3:
                break

            a_lp = lp_refine_loop(best_seq, time_budget=alloc,
                                  max_rounds=5)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {r}: {best_score:.8f} ***")
            else:
                print(f"  LP round {r}: {obj_lp:.8f} (no imp)")

            # Follow up with Adam
            if time.monotonic() < deadline - 40:
                alloc2 = min(15.0, deadline - 40 - time.monotonic())
                if alloc2 > 3:
                    a_post = adam_log_opt(a_lp, max_iters=20000,
                                          time_budget=alloc2,
                                          beta_start=50.0, beta_end=500.0,
                                          lr_start=0.02, lr_decay=0.99998,
                                          adaptive_beta=True,
                                          switch_to_subgrad_at=0.6)
                    obj_post = compute_obj(a_post)
                    if obj_post < best_score:
                        best_score = obj_post
                        best_seq = a_post.copy()
                        print(f"  LP+Adam {r}: {best_score:.8f} ***")

    # === Phase 3: Subgradient peak chase (15% of time) ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Subgradient peak chase")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 50:
        alloc = min(60.0, deadline - 50 - time.monotonic())
        if alloc > 5:
            a_sg = subgradient_peak_chase(best_seq, time_budget=alloc,
                                          lr_start=0.008, lr_end=0.001)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Subgradient: {best_score:.8f} ***")
            else:
                print(f"  Subgradient: {obj_sg:.8f}")

    # === Phase 4: Multi-n exploration (10% of time) ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq)
    n_options = sorted(set([
        max(300, n_best - 100),
        max(300, n_best - 50),
        n_best,
        min(1500, n_best + 50),
        min(1500, n_best + 100),
        min(1500, n_best + 200),
    ]))
    n_options = [n for n in n_options if n != n_best]

    phase4_budget = min((deadline - time.monotonic()) * 0.12, 80.0)
    for n_try in n_options:
        if time.monotonic() > deadline - 50:
            break
        alloc = phase4_budget / max(len(n_options), 1)
        if alloc < 3:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = np.sum(a_try)
        if s_try > 1e-10:
            a_try = a_try / s_try
        a_opt = adam_log_opt(a_try, max_iters=20000, time_budget=alloc,
                             beta_start=20.0, beta_end=300.0,
                             lr_start=0.03, lr_decay=0.99997,
                             adaptive_beta=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  n={n_try}: {obj:.8f}")

    # === Phase 5: Final polish (remaining time) ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    a = best_seq.copy()
    s = np.sum(a)
    if s > 1e-10:
        a = a / s

    # Coordinate descent
    if time.monotonic() < deadline - 30:
        alloc = min(25.0, deadline - 30 - time.monotonic())
        if alloc > 3:
            a_cd = coordinate_refine(a, time_budget=alloc, seed=seed)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                a = a_cd.copy()
                print(f"  Coord descent: {best_score:.8f} ***")

    # High-beta Adam
    if time.monotonic() < deadline - 20:
        alloc = min(20.0, deadline - 20 - time.monotonic())
        if alloc > 3:
            a_hb = adam_log_opt(a, max_iters=20000, time_budget=alloc,
                                beta_start=200.0, beta_end=1500.0,
                                lr_start=0.008, lr_decay=0.99999,
                                adaptive_beta=True,
                                switch_to_subgrad_at=0.5)
            obj_hb = compute_obj(a_hb)
            if obj_hb < best_score:
                best_score = obj_hb
                best_seq = a_hb.copy()
                print(f"  High-beta Adam: {best_score:.8f} ***")

    # Another LP round if time permits
    if time.monotonic() < deadline - 15 and len(best_seq) <= 1000:
        alloc = min(15.0, deadline - 15 - time.monotonic())
        if alloc > 3:
            a_final_lp = lp_refine_loop(best_seq, time_budget=alloc,
                                        max_rounds=3)
            obj_fl = compute_obj(a_final_lp)
            if obj_fl < best_score:
                best_score = obj_fl
                best_seq = a_final_lp.copy()
                print(f"  Final LP: {best_score:.8f} ***")

    # Random perturbation + quick optimize
    if time.monotonic() < deadline - 10:
        rng = np.random.default_rng(seed + 1)
        n = len(best_seq)
        for attempt in range(3):
            if time.monotonic() > deadline - 10:
                break
            a_pert = best_seq * np.exp(rng.normal(0, 0.02, n))
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / np.sum(a_pert)
            alloc = min(10.0, deadline - 10 - time.monotonic())
            if alloc < 2:
                break
            a_opt = adam_log_opt(a_pert, max_iters=5000, time_budget=alloc,
                                 beta_start=50.0, beta_end=500.0,
                                 lr_start=0.02, lr_decay=0.99998,
                                 adaptive_beta=True)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Perturb {attempt}: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
