import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_at_peak(a, k_star):

    n = len(a)

    hi = min(n, k_star + 1)

"""
Multi-phase gradient-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max gradient descent with Adam optimizer, hard subgradient refinement,
chase-the-peak coordinate descent, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_fft_size(n):
    """Get next power of 2 >= n."""
    size = 1
    while size < n:
        size <<= 1
    return size


def soft_max_gradient_conv(a, conv, temperature, fft_size):
    """
    Compute soft-max subgradient: 2 * cross_correlation(p, a)[:n]
    where p = softmax(conv / temperature).
    """
    n = len(a)
    shifted = conv - np.max(conv)
    exp_vals = np.exp(shifted / temperature)
    p = exp_vals / np.sum(exp_vals)

    fp = rfft(p, fft_size)
    fa = rfft(a, fft_size)
    corr = irfft(fp * np.conj(fa), fft_size)[:n]
    return 2.0 * corr


def hard_subgradient(a, k_star):
    """Compute subgradient of conv(a,a)[k_star] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    # grad[j] = 2*a[k_star - j] for j in [lo, hi)
    indices = np.arange(lo, hi)
    grad[indices] = 2.0 * a[k_star - indices]
    return grad


def adam_soft_max_optimize(a, time_budget, T_init=None, T_min=None, lr=0.01, beta1=0.9, beta2=0.999):
    """
    Adam optimizer with soft-max gradient and temperature annealing.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv_size = get_fft_size(2 * n - 1)
    grad_size = get_fft_size(3 * n)

    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    if T_init is None:
        T_init = max_conv / 3.0
    if T_min is None:
        T_min = max_conv / 200.0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Number of iterations to anneal over
    anneal_iters = 8000

    while time.monotonic() - start < time_budget:
        t += 1

        # Anneal temperature
        if t <= anneal_iters:
            T = T_init * (T_min / T_init) ** (t / float(anneal_iters))
        else:
            T = T_min

        # Compute convolution
        fa = rfft(a, conv_size)
        conv = irfft(fa * fa, conv_size)[:2 * n - 1]

        # Compute soft-max gradient
        grad = soft_max_gradient_conv(a, conv, T, grad_size)

        # Center gradient (since we're on the simplex, only tangential component matters)
        grad -= np.mean(grad)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_new = a - step
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a = best_a.copy()
            break
        a_new = a_new / s_new
        a = a_new

        # Track best (every 100 iterations to save time)
        if t % 100 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    # Final check
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv_size = get_fft_size(2 * n - 1)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

        # Compute convolution
        fa = rfft(a, conv_size)
        conv = irfft(fa * fa, conv_size)[:2 * n - 1]
        k_star = int(np.argmax(conv))
        peak_val = float(conv[k_star])

        hi = min(n, k_star + 1)

        i_src = indices[max_idx_local]
        j_partner = k_star - i_src

        # Find a good destination: position d where a[d] * a[k_star - d] is minimal
        # and d is not i_src and k_star - d is not j_partner (to avoid just swapping)
        # Actually, we want to move mass from i_src to some d that doesn't increase other peaks
        # Simple heuristic: move to a position where the product is smallest

        all_products = np.zeros(n)
        for d in range(n):
            k2 = k_star - d
            if 0 <= k2 < n:
                all_products[d] = a[d] * a[k2]

        # Exclude source and its partner
        all_products[i_src] = np.inf
        if 0 <= j_partner < n and j_partner != i_src:
            all_products[j_partner] = np.inf

        d_dst = int(np.argmin(all_products))
        if all_products[d_dst] == np.inf:
            break

        # Try moving different fractions of mass
        moved = False
        for frac in [0.3, 0.15, 0.05, 0.02]:
            delta = frac * a[i_src]
            a[i_src] -= delta
            a[d_dst] += delta
            s = float(np.sum(a))
            a = a / s

            fa = rfft(a, conv_size)
            new_conv = irfft(fa * fa, conv_size)[:2 * n - 1]
            obj = 2.0 * n * float(np.max(new_conv))

            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                stall = 0
                moved = True
                break
            else:
                # Revert
                a[i_src] += delta
                a[d_dst] -= delta
                a = a / s

        if not moved:
            stall += 1
            if stall > 50:
                # Perturb
                rng = np.random.default_rng(hash((stall, n)) % (2**32))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0

    return best_a


def multi_peak_gradient_optimize(a, time_budget, n_peaks=5, lr=0.003):
    """
    Gradient optimization using weighted average of top-k peak subgradients.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv_size = get_fft_size(2 * n - 1)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    while time.monotonic() - start < time_budget:
        t += 1

        fa = rfft(a, conv_size)
        conv = irfft(fa * fa, conv_size)[:2 * n - 1]

        # Get top-k peaks
        top_k = np.argsort(conv)[::-1][:n_peaks]
        # Weight by how close to max each is (exponential weighting)
        weights = np.exp(-(conv[top_k[0]] - conv[top_k]) / (conv[top_k[0]] * 0.1 + 1e-10))
        weights = weights / np.sum(weights)

        # Weighted gradient
        grad = np.zeros(n)
        for idx, k in enumerate(top_k):
            grad += weights[idx] * hard_subgradient(a, int(k))
        grad -= np.mean(grad)

        # Adam
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_new = a - step
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            a = a_new

        if t % 50 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, mode=0):
    """Generate structured initial sequences."""
    if mode == 0:
        # Raised cosine
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
    elif mode == 1:
        # Gaussian
        x = np.arange(n)
        center = n / 2.0
        a = np.exp(-0.5 * ((x - center) / (n / 4.0)) ** 2)
    elif mode == 2:
        # Bimodal
        x = np.arange(n)
        c1 = n * 0.3
        c2 = n * 0.7
        a = np.exp(-0.5 * ((x - c1) / (n / 6.0)) ** 2) + np.exp(-0.5 * ((x - c2) / (n / 6.0)) ** 2)
    elif mode == 3:
        # Sinc-like (absolute value)
        x = np.arange(n) - n / 2.0
        a = np.abs(np.sinc(x / (n / 3.0)))
    elif mode == 4:
        # Random
        rng = np.random.default_rng(42)
        a = rng.uniform(0, 1, n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase gradient-based optimization
    combining soft-max Adam, hard subgradient Adam, chase-the-peak,
    multi-n exploration, and final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(800, 0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Soft-max Adam with annealing ===
    print(f"\n=== Phase 1: Soft-max Adam (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(400.0, remaining * 0.35)

    a = best_seq.copy()
    a_opt = adam_soft_max_optimize(a, time_budget=alloc)
    obj = compute_obj(a_opt)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  Soft-max: obj={best_score:.8f}")
    else:
        print(f"  Soft-max: obj={obj:.8f} (no improvement)")

    # === Phase 2: Hard subgradient Adam ===
    print(f"\n=== Phase 2: Hard Subgradient Adam ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(300.0, remaining * 0.3)
        a_opt = adam_hard_subgradient_optimize(best_seq, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Hard subgrad: obj={best_score:.8f}")
        else:
            print(f"  Hard subgrad: obj={obj:.8f} (no improvement)")

    # === Phase 3: Multi-peak gradient ===
    print(f"\n=== Phase 3: Multi-Peak Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(200.0, remaining * 0.25)
        a_opt = multi_peak_gradient_optimize(best_seq, time_budget=alloc, n_peaks=5)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Multi-peak: obj={best_score:.8f}")
        else:
            print(f"  Multi-peak: obj={obj:.8f} (no improvement)")

    # === Phase 4: Chase the peak ===
    print(f"\n=== Phase 4: Chase Peak ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(200.0, remaining * 0.25)
        a_opt = chase_peak_optimize(best_seq, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Chase peak: obj={best_score:.8f}")
        else:
            print(f"  Chase peak: obj={obj:.8f} (no improvement)")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_candidates = sorted(set([n_best - 50, n_best - 20, n_best, n_best + 20, n_best + 50, n_best + 100]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1200]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick Adam optimization
        alloc = min(100.0, remaining * 0.3)
        a_opt = adam_soft_max_optimize(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj:.8f}")

    # === Phase 6: Second round of refinement on best ===
    print(f"\n=== Phase 6: Second Round Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Soft-max again with lower temperature
        alloc = min(150.0, remaining * 0.35)
        a_opt = adam_soft_max_optimize(best_seq, time_budget=alloc, T_init=None, T_min=None)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Soft-max 2: obj={best_score:.8f}")
        else:
            print(f"  Soft-max 2: obj={obj:.8f}")

        # Hard subgradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(100.0, remaining * 0.35)
            a_opt = adam_hard_subgradient_optimize(best_seq, time_budget=alloc)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Hard 2: obj={best_score:.8f}")
            else:
                print(f"  Hard 2: obj={obj:.8f}")

        # Chase peak
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(100.0, remaining * 0.35)
            a_opt = chase_peak_optimize(best_seq, time_budget=alloc)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Chase 2: obj={best_score:.8f}")
            else:
                print(f"  Chase 2: obj={obj:.8f}")

    # === Phase 7: Random restarts from best ===
    print(f"\n=== Phase 7: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    n_restarts = min(6, max(2, int(remaining / 60)))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        alloc = min(50.0, remaining * 0.4)

        # Different perturbation strategies
        if r % 3 == 0:
            scale = 0.05 + 0.03 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_final))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r % 3 == 1:
            # Flip a random subset
            a_r = best_seq.copy()
            mask = rng.random(n_final) < 0.3
            a_r[mask] *= rng.uniform(0.5, 2.0, mask.sum())
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            # Mix with structured init
            a_struct = structured_init(n_final, mode=r % 5)
            a_r = 0.7 * best_seq + 0.3 * a_struct
            a_r = a_r / np.sum(a_r)

        a_opt = adam_soft_max_optimize(a_r, time_budget=alloc * 0.6)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj:.8f}")

        # Quick hard subgradient on the result
        remaining = deadline - time.monotonic()
        if remaining > 20 and obj < best_score + 0.01:
            a_opt2 = adam_hard_subgradient_optimize(a_opt, time_budget=min(30.0, remaining * 0.3))
            obj2 = compute_obj(a_opt2)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_opt2.copy()
                print(f"  Restart {r} + hard: obj={best_score:.8f} **")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Very fine soft-max
        alloc = min(60.0, remaining * 0.4)
        a_opt = adam_soft_max_optimize(best_seq, time_budget=alloc, T_init=None, T_min=None)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Final soft: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            # Hard subgradient
            alloc = min(40.0, remaining * 0.4)
            a_opt = adam_hard_subgradient_optimize(best_seq, time_budget=alloc)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final hard: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            # Chase peak
            alloc = min(40.0, remaining * 0.4)
            a_opt = chase_peak_optimize(best_seq, time_budget=alloc)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final chase: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
