def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):

    M = cp.Variable()

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]

        try:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

        A_ub[:, n] = -1.0

        A_eq = np.zeros((1, n + 1))

            s = float(np.sum(g_val))

    # d obj / d a_j = 2n * tau * exp(tau * conv[k*]) / sum(exp(tau*conv)) * d conv / d a_j

import numpy as np

from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.optimize import minimize, linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

    m = np.max(conv)

        mean_c = np.dot(c, a)
        grad_a = 4.0 * n * (c - mean_c)
        
        # Project gradient: subtract mean to stay on hyperplane sum=1
        grad_projected = grad_a - np.mean(grad_a)
        
        a_new = a - step_size * grad_projected
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                step_size = min(step_size * 1.1, 1.0)
            else:
                step_size *= 0.9
        else:
            step_size *= 0.7
    
    return best_a, best_obj


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(300):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose and it % 10 == 0:
                print(f"        LP alt it{it}: obj={best_obj:.8f}")
        else:
            # Perturb to escape
            if it % 10 == 0:
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
    
    return best_a


def coordinate_descent(a, time_budget=30.0):
    """Coordinate descent targeting the peak bin."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False
        
        # Try moving mass away from peak contributors
        for i in range(0, n, max(1, n // 200)):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i:
                for frac in [0.01, 0.03, 0.05]:
                    delta = frac * a[i]
                    if delta > a[i]:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break
        
        if not improved:
            # Try random moves
            rng = np.random.default_rng(it * 31 + n)
            for _ in range(50):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.01 * a[i] * rng.uniform(0.1, 1.0)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
            if not improved:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines smoothed L-BFGS-B, LP alternating, projected gradient,
    and coordinate descent with multi-n exploration and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    n_parent = len(best_seq)
    
    # Phase 1: Smoothed L-BFGS-B (primary new method)
    print(f"\n=== Phase 1: Smoothed L-BFGS-B ===")
    for trial in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        if trial == 0:
            a_init = best_seq.copy()
        elif trial == 1:
            # Perturbed
            a_init = best_seq * np.exp(rng.normal(0, 0.05, n_parent))
            a_init = np.maximum(a_init, 0)
            a_init = a_init / np.sum(a_init)
        else:
            # More asymmetric perturbation
            shift = rng.uniform(0.1, 0.3)
            ramp = np.linspace(1 - shift, 1 + shift, n_parent)
            a_init = best_seq * ramp
            a_init = np.maximum(a_init, 0)
            a_init = a_init / np.sum(a_init)
        
        alloc = min(remaining * 0.35, 150.0)
        a_opt, obj_opt = smooth_lbfgs(a_init, tau_start=10.0, tau_end=500.0,
                                       time_budget=alloc, max_iter=150)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_parent = len(best_seq)
            print(f"  Trial {trial}: obj={best_score:.8f} **")
        else:
            print(f"  Trial {trial}: obj={obj_opt:.8f}")
    
    # Phase 2: Projected gradient on simplex (complementary to L-BFGS-B)
    print(f"\n=== Phase 2: Projected Gradient (Simplex) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for tau_pg in [20.0, 50.0, 100.0]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_pg, obj_pg = projected_gradient_simplex(best_seq, tau=tau_pg,
                                                      steps=150, time_budget=min(remaining * 0.2, 25.0))
            if obj_pg < best_score:
                best_score = obj_pg
                best_seq = a_pg.copy()
                print(f"  PG tau={tau_pg}: obj={best_score:.8f} **")
            else:
                print(f"  PG tau={tau_pg}: obj={obj_pg:.8f}")
    
    # Phase 3: LP alternating
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_cur = len(best_seq)
        alloc_lp = min(remaining * 0.25, 120.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc_lp, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")
    
    # Phase 4: Multi-n exploration
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([
        max(400, n_cur - 100), n_cur - 50, n_cur,
        n_cur + 50, n_cur + 100, n_cur + 200,
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500]
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        
        if n_try == n_cur:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        # Quick smoothed optimization
        alloc = min(remaining * 0.2, 40.0)
        a_opt, obj_opt = smooth_lbfgs(a_try, tau_start=20.0, tau_end=300.0,
                                       time_budget=alloc, max_iter=80)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # Phase 5: Random restarts with diverse initializations
    print(f"\n=== Phase 5: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        init_type = r % 3
        if init_type == 0:
            # Linear ramp (strong asymmetry)
            slope = rng.uniform(0.5, 2.0)
            a_r = np.linspace(1.0, slope, n_best)
            a_r = a_r / np.sum(a_r)
        elif init_type == 1:
            # Exponential asymmetry
            rate = rng.uniform(0.5, 3.0)
            a_r = np.exp(np.linspace(0, rate, n_best))
            a_r = a_r / np.sum(a_r)
        else:
            # Blend of parent with ramp
            blend = rng.uniform(0.3, 0.7)
            ramp = np.linspace(0.5, 1.5, n_best)
            a_r = blend * best_seq + (1 - blend) * ramp
            a_r = a_r / np.sum(a_r)
        
        alloc = min(remaining * 0.15, 30.0)
        a_opt, obj_opt = smooth_lbfgs(a_r, tau_start=15.0, tau_end=400.0,
                                       time_budget=alloc, max_iter=100)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (type={init_type}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_opt:.8f}")
    
    # Phase 6: Final intensive polish
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 30:
        # LP alternating
        alloc = min(remaining * 0.3, 60.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Coordinate descent
        a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.3, 30.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # One more smoothed optimization
        alloc = min(remaining * 0.4, 40.0)
        a_opt, obj_opt = smooth_lbfgs(best_seq, tau_start=50.0, tau_end=800.0,
                                       time_budget=alloc, max_iter=100)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final Smooth: obj={best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
