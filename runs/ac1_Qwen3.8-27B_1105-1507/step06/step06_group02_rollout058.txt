import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based full convolution."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective value."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def smooth_grad(a, tau):
    """Compute smooth max gradient using log-sum-exp trick."""
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / (exp_b.sum() + 1e-300)
    n = len(a)
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[lo:hi + 1][::-1] if k >= n - 1 else 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, b_max


def smooth_grad_fft(a, tau):
    """Faster smooth gradient using FFT for the weighted convolution."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / (exp_b.sum() + 1e-300)
    # grad = 2 * conv(p, a) truncated to n elements (symmetric since conv(a,a) is symmetric)
    # Actually grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # This is 2 * (p * a)[i] where * is cross-correlation... 
    # Let's use: grad = 2 * ifft(fft(p_full) * conj(fft(a_full)))... 
    # Simpler: the gradient of (1/2)*max_k sum_i a[i]*a[k-i] w.r.t. a
    # is sum_k p[k] * (e_k * a) where (e_k * a)[i] = a[k-i] if valid
    # = sum_k p[k] * flip_a_shifted
    # This equals: conv(p, a_flipped) where a_flipped[i] = a[n-1-i], truncated
    # Actually it's simpler: grad = 2 * ifft(conj(fft(p, 2n-1)) * fft(a, 2n-1))[:n]
    # But p has length 2n-1 and a has length n. Let's just do it directly with the loop
    # for simplicity (it's at most 2n-1 terms but most p[k] are tiny)
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        # Contribution: 2 * p[k] * a[j] where j = k - i, i.e., a[k-i] for i in range
        # grad[i] += 2*p[k]*a[k-i] for valid i
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2*p[k] * a[k-hi : k-lo+1][::-1]
        # a index: when i=lo, a[k-lo]; when i=hi, a[k-hi]
        # So a indices go from k-hi to k-lo, reversed
        a_slice = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * a_slice
    return grad, b_max


def nesterov_smooth(a, deadline, max_iters=3000, tau_start=2.0, tau_end=500.0,
                    lr_init=0.5, lr_end=0.01):
    """Nesterov accelerated smooth gradient descent with mirror (exponential) steps."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = normalize(a)
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()
    t_k = 1.0
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b_max = smooth_grad_fft(y, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Mirror descent step (exponential family)
        step = lr / (gnorm + 1e-10)
        a_new = y * np.exp(-step * grad)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = (t_k - 1.0) / t_new
        y = np.maximum(alpha * a_new + (1.0 - alpha) * y, 1e-15)
        sy = y.sum()
        y = y / sy

        a = a_new
        t_k = t_new

        if b_max < best_M - 1e-12:
            best_M = b_max
            best_a = a.copy()

    return best_a


def nesterov_euclidean(a, deadline, max_iters=3000, tau_start=2.0, tau_end=300.0,
                       lr_init=0.3, lr_end=0.005):
    """Nesterov accelerated smooth gradient descent with Euclidean projection."""
    a = np.maximum(a.astype(np.float64), 1e-10)

    n = len(a)

        frac = min(1.0, it / max_iters)

        grad, b_max = smooth_grad_fft(y, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Euclidean step + project to simplex

        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-10:
            x_new = a.copy()
        else:
            x_new /= s

        # Nesterov momentum
        t = 1.0 + np.sqrt(1.0 + 4.0 * it * it) 
        # Actually use proper Nesterov: beta_k = (1 - t_k) / (1 + t_{k+1})
        # Simplified: use increasing momentum
        momentum = min(0.9, 0.9 * it / (it + 50))
        y = np.maximum(x_new + momentum * (x_new - x_prev), 0.0)

        if sy < 1e-10:

        x_prev = a

def build_conv_matrix_indices(n):
    """Build sparse indices for the convolution matrix (a * g = A @ g)."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return rows, cols


def lp_solve(a, M_target):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M_target, g >= 0.
    Returns (g, active_set) where active_set are indices where constraint is tight."""
    n = len(a)
    rows, cols = build_conv_matrix_indices(n)
    # Values of A: A[k, j] = a[k - j]
    vals = np.array([a[k - j] for k, j in zip(rows, cols)])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 10})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            # Determine active set
            Ag = A @ g
            active = np.where(Ag > M_target * 0.999)[0]
            return g, active
    except Exception:
        pass
    return None, None


def lp_step_line_search(a, M, max_ls=8):
    """One LP step with discrete line search. Returns (new_a, improved)."""
    n = len(a)
    g, active = lp_solve(a, M)
    if g is None or g.sum() < 1e-10:
        return a, False
    g = normalize(g)

    # Check if g is too similar to a
    cos_sim = np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-15)
    if cos_sim > 0.9999:
        return a, False

    # Line search: try different step sizes
    best_a = a.copy()
    best_M = M
    for t in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        st = a_t.sum()
        if st < 1e-10:
            continue
        a_t /= st
        M_t = float(fast_conv(a_t).max())
        if M_t < best_M - 1e-14:
            best_M = M_t
            best_a = a_t.copy()
    return best_a, best_M < M - 1e-14

def resample_seq(src, n):
    """Resample a sequence to a new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gs_flat(a, n_iters=200, beta=0.85):
    """Gerchberg-Saxton spectral flattening."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    for _ in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_min = mag.min()
        mag_max = mag.max()
        if mag_max - mag_min < 1e-10:
            break
        # Compress magnitude
        mag_norm = (mag - mag_min) / (mag_max - mag_min + 1e-30)
        mag_new = (1.0 - beta) * mag + beta * ((1.0 - beta) * mag_min + beta * mag_max * 0.5)
        # Alternative: raise to a power less than 1
        mag_alt = mag ** 0.7 * (mag_max / (mag ** 0.7).max() + 1e-30)
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        sc = a_new.sum()
        if sc < 1e-15:
            break
        a_new /= sc
        M_c = float(fast_conv(a_new).max())
        if M_c < best_M:
            best_M = M_c
            best_a = a_new.copy()
        a = a_new
    return best_a


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(128)
        best.update(parent)
        print("No parent, using ones(128)", flush=True)

    n_parent = parent.size
    print(f"Starting optimization, budget={budget_s}s, deadline in {1500-30:.0f}s", flush=True)

    # === PHASE 1: Nesterov smooth optimization (250s) ===
    p1_deadline = time.monotonic() + 250.0
    p1_deadline = min(p1_deadline, deadline - 800.0)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Nesterov Smooth ({min(250.0, deadline - time.monotonic() - 800.0):.0f}s budget)", flush=True)
    print(f"{'='*50}", flush=True)

    best_per_length = {}

    # Try parent length and a few others
    lengths_to_try = sorted(set([
        n_parent,
        max(256, n_parent - 64),
        n_parent + 64,
        n_parent + 128,
        n_parent + 256,
    ]))
    # Filter reasonable lengths
    lengths_to_try = [n for n in lengths_to_try if 256 <= n <= 3072]
    print(f"Lengths: {lengths_to_try}", flush=True)

    for n in lengths_to_try:
        if time.monotonic() > p1_deadline:
            break
        a0 = resample_seq(parent, n) if n != n_parent else parent.copy()
        a0 = normalize(a0)

        # Multiple initializations
        inits = [a0.copy()]
        # Perturbed
        inits.append(normalize(a0 + 0.03 * rng.normal(0, 1, n)))
        # GS flattened
        inits.append(gs_flat(a0, n_iters=100, beta=0.85))
        # Cosine blend
        t = np.linspace(0, 1, n, endpoint=False)
        inits.append(normalize(0.5 * a0 + 0.5 * np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * t), 0.01)))

        best_n = None
        best_M = float('inf')

        for idx, a_init in enumerate(inits):
            sub_dl = min(p1_deadline, time.monotonic() + 60.0 * (1.0 / (1.0 + idx * 0.5)))
            if time.monotonic() >= sub_dl:
                break
            a_res = nesterov_smooth(a_init, sub_dl, max_iters=2000,
                                    tau_start=2.0, tau_end=300.0,
                                    lr_init=0.4, lr_end=0.01)
            M = float(fast_conv(a_res).max())
            if M < best_M:
                best_M = M
                best_n = a_res.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-best.t0:.0f}s total)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report = lambda label: print(f"  {label}: J={best.score:.8f} n={best.seq.size if best.seq is not None else 0}", flush=True)
    print(f"  Phase 1 best: J={best.score:.8f}", flush=True)

    # === PHASE 2: LP optimization with line search (600s) ===
    p2_deadline = time.monotonic() + 600.0
    p2_deadline = min(p2_deadline, deadline - 150.0)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP with Line Search ({min(600.0, p2_deadline - time.monotonic()):.0f}s budget)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        a = best.seq.copy()
        # Run LP phase
        a_lp = lp_phase(a, min(p2_deadline, deadline - 200.0), max_iters=300, use_adaptive_M=True)
        best.update(a_lp)

        # Second LP pass from the result
        if time.monotonic() < p2_deadline - 30:
            a_lp2 = lp_phase(a_lp, p2_deadline, max_iters=200, use_adaptive_M=True)
            best.update(a_lp2)

    print(f"  Phase 2 best: J={best.score:.8f}", flush=True)

    # === PHASE 3: Multi-length exploration (300s) ===
    p3_deadline = time.monotonic() + 300.0
    p3_deadline = min(p3_deadline, deadline - 100.0)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Multi-length ({min(300.0, p3_deadline - time.monotonic()):.0f}s budget)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        # Try nearby lengths
        extra_lengths = sorted(set([
            n_best - 32, n_best + 32, n_best - 96, n_best + 96,
            n_best + 128, n_best + 256,
        ]))
        extra_lengths = [n for n in extra_lengths if 256 <= n <= 3072 and n != n_best]
        extra_lengths = extra_lengths[:4]  # Limit to 4 extra lengths

        for n_try in extra_lengths:
            if time.monotonic() > p3_deadline:
                break
            a_cand = resample_seq(best.seq, n_try)
            a_cand = normalize(a_cand)
            # Quick smooth pass
            if time.monotonic() < p3_deadline - 20:
                a_cand = nesterov_smooth(a_cand, min(time.monotonic() + 20, p3_deadline - 15),
                                         max_iters=500, tau_start=5.0, tau_end=200.0,
                                         lr_init=0.2, lr_end=0.01)
            best.update(a_cand)
            # Quick LP pass
            if time.monotonic() < p3_deadline - 15:
                a_lp = lp_phase(a_cand, p3_deadline, max_iters=50, use_adaptive_M=False)
                best.update(a_lp)

    print(f"  Phase 3 best: J={best.score:.8f}", flush=True)

    # === PHASE 4: Final fine polish (remaining time) ===
    p4_deadline = deadline
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Final Polish ({p4_deadline - time.monotonic():.0f}s budget)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < p4_deadline - 5:
        a_final = lp_phase_fine(best.seq, p4_deadline, max_iters=100)
        best.update(a_final)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    total_time = time.monotonic() - best.t0
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={total_time:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in best.seq]
