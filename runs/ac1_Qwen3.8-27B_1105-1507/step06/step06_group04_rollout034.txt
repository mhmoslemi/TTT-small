    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv_fft(a)
    return 2.0 * n * float(np.max(conv)) / (s*s)

def normalize(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a):
    """A[j, i] = a[j-i] if 0<=j-i<n else 0, so (A@g)[j] = (a*g)[j]"""
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # For output position k, contribution from g[j] is a[k-j]
        # So A[k, j] = a[k-j] if 0 <= k-j < n
        for k in range(j, j+n):
            A[k, j] = a[k-j]
    return A

    b_ub = np.zeros(m)

"""
Advanced peak-reduction search for non-negative step functions minimizing
2*n*max(conv(a,a))/(sum(a))^2. Combines LP-based alternating minimization with
dual-informed subgradient, golden-section line search, multi-length exploration,
and structured perturbation for escaping local optima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build matrix A where (A@g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_minpeak_lp(a, time_limit=15):
    """
    Solve: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt, dual_weights) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                t_opt = float(res.x[n])
                # Extract dual variables for the inequality constraints
                dual = None
                if hasattr(res, 'unequalityMarginals') and res.unequalityMarginals is not None:
                    dual = np.abs(res.unequalityMarginals)
                    if np.sum(dual) > 1e-15:
                        dual = dual / np.sum(dual)
                    else:
                        dual = None
                return g, t_opt, dual
    except Exception:
        pass
    return None, None, None


def compute_gradient_fft(a, weights):
    """
    Compute subgradient of max_k weights[k]*conv(a,a)[k] with respect to a.
    grad[i] = 2 * sum_k weights[k] * a[k-i] (with appropriate bounds).
    Uses FFT for efficiency.
    """
    n = len(a)
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    w_ext = np.zeros(fft_size)
    w_ext[:len(weights)] = weights
    W = rfft(w_ext, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    grad = 2.0 * corr[:n]
    return grad


def subgradient_optimize(a, max_iters=500, time_budget=60.0, tau_init=0.01, tau_decay=0.999):
    """
    Subgradient method with adaptive temperature and momentum.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    mu = 0.82
    step = 0.05
    tau = tau_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights = weights / w_sum

        grad = compute_gradient_fft(a, weights)
        grad_centered = grad - float(np.mean(grad))
        grad_norm = float(np.linalg.norm(grad_centered))
        if grad_norm < 1e-15:
            break

        v = mu * v - step * grad_centered / grad_norm
        a_new = normalize(a - v)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = best_a.copy()
            step *= 0.98

        step = max(0.01, step * 0.9995)
        tau = max(0.001, tau * tau_decay)

    return best_a, best_obj


def golden_line_search(a, g, obj_current, max_evals=12):
    """
    Golden section search for optimal blend t: minimize obj((1-t)*a + t*g).
    Returns (best_t, best_a, best_obj).
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.5
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

    def eval_blend(t):
        x = normalize((1.0 - t) * a + t * g)
        return compute_obj(x)

    fc = eval_blend(c)
    fd = eval_blend(d)

    for _ in range(max_evals - 2):
        if hi - lo < 0.01:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_blend(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_blend(d)

    t_mid = (lo + hi) / 2.0
    candidates = [0.0, c, t_mid, d, 1.0, 1.5]
    best_t = 0.0
    best_obj = obj_current
    best_x = a.copy()
    for t in candidates:
        x = normalize((1.0 - t) * a + t * g)
        o = compute_obj(x)
        if o < best_obj:
            best_obj = o
            best_t = t
            best_x = x
    return best_t, best_x, best_obj

def diverse_starts(n, num_starts=10, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    starts.append(np.ones(n) / n)

    # Triangular
    tri = 1.0 - np.abs(i - center) / center
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    # Gaussian (narrow)
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 5.0) ** 2))
    starts.append(bell / np.sum(bell))

    # Gaussian (wide)
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 2.5) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * i / n))
    starts.append(rc / np.sum(rc))

    # Flat-top
    ft = np.ones(n)
    edge = int(n * 0.1)
    ft[:edge] = 0.5 * (1 + np.cos(np.pi * np.arange(edge) / edge))
    ft[-edge:] = 0.5 * (1 + np.cos(np.pi * np.arange(edge) / edge))
    starts.append(ft / np.sum(ft))

    # Parametric with random parameters
    for _ in range(num_starts - 6):
        sigma = n * (0.1 + 0.4 * rng.random())
        freq = int(1 + rng.integers(0, 6))
        amp = 0.1 + 0.6 * rng.random()
        phase = 2 * np.pi * rng.random()
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
        a = np.exp(-((i - center) ** 2) / (2 * sigma ** 2)) * mod
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-15:
            starts.append(a / s)

    return starts


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)

    # Phase 1: Multi-start exploration
    print("\n=== Phase 1: Multi-start exploration ===")
    t_phase1_end = start_time + budget_s * 0.3

    n_values = sorted(set([
        n_parent,
        max(400, n_parent - 50),
        max(400, n_parent - 100),
        n_parent + 50,
        n_parent + 100,
        n_parent + 150,
    ]))
    n_values = [x for x in n_values if 400 <= x <= 1200]

    candidates = []

    for n in n_values:
        if time.monotonic() > t_phase1_end:
            break
        remaining = t_phase1_end - time.monotonic()
        if remaining < 20:
            break

        if n == n_parent and best_seq is not None:
            starts = [normalize(best_seq.copy())]
            starts.extend(diverse_starts(n, num_starts=6, seed=seed))
        else:
            starts = diverse_starts(n, num_starts=5, seed=seed + n)

        alloc_per_start = max(5.0, min(remaining / len(starts), 20.0))

        for si, s in enumerate(starts):
            if time.monotonic() > t_phase1_end - 5:
                break

            t_alloc = min(alloc_per_start, 15.0)
            if t_alloc < 3:
                break

            # Quick optimization
            a_opt, obj_opt = subgradient_optimize(s, max_iters=int(t_alloc * 10), time_budget=t_alloc * 0.5)
            a_opt2, obj_opt2 = peak_targeting(a_opt, time_budget=t_alloc * 0.3)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {obj_opt:.8f} ** NEW BEST **")

            candidates.append((obj_opt, a_opt.copy(), n))

    candidates.sort(key=lambda x: x[0])
    if candidates:
        print(f"\nTop candidates: {[f'{c[0]:.6f}' for c in candidates[:5]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 2: Intensive LP optimization on top candidates
    print("\n=== Phase 2: Intensive LP optimization ===")
    t_phase2_end = start_time + budget_s * 0.65

    top_k = min(4, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > t_phase2_end:
            break
        remaining = t_phase2_end - time.monotonic()
        if remaining < 30:
            break

        obj_cand, a_cand, n_cand = candidates[ci]
        alloc = min(remaining * 0.35, 120.0)

        print(f"  Candidate {ci} (n={n_cand}, obj={obj_cand:.8f}): LP with {alloc:.0f}s")
        a_lp, obj_lp = lp_alternate(a_cand, max_iters=int(alloc / 5), time_budget=alloc)

        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** UPDATE: {obj_lp:.8f} **")
        else:
            print(f"  No LP improvement (got {obj_lp:.8f})")

    # Phase 3: Full polish on the best
    print(f"\n=== Phase 3: Full polish (best: {best_score:.8f}) ===")
    t_phase3_end = start_time + budget_s * 0.85

    if best_seq is not None:
        remaining = t_phase3_end - time.monotonic()
        if remaining > 20:
            a_pol, obj_pol = full_polish(best_seq, time_budget=min(remaining, 200.0))
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Polish result: {best_score:.8f}")
            else:
                print(f"  Polish result: {obj_pol:.8f} (no improvement)")

    # Phase 4: Try different lengths with resampling
    print(f"\n=== Phase 4: Length exploration (best: {best_score:.8f}) ===")
    t_phase4_end = start_time + budget_s * 0.93

    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = sorted(set([
            max(500, n_best - 100),
            max(500, n_best - 50),
            n_best + 50,
            n_best + 100,
            n_best + 150,
        ]))
        test_lengths = [x for x in test_lengths if 500 <= x <= 1500]

        for n_test in test_lengths:
            if time.monotonic() > t_phase4_end:
                break
            remaining = t_phase4_end - time.monotonic()
            if remaining < 15:
                break

            a_res = resample_sequence(best_seq, n_test)
            a_res = normalize(a_res)
            obj_init = compute_obj(a_res)

            alloc = min(remaining * 0.3, 30.0)
            a_pol, obj_pol = full_polish(a_res, time_budget=alloc)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** UPDATE to {obj_pol:.8f} **")
            else:
                print(f"  n={n_test}: {obj_pol:.8f} (no improvement)")

    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final polish (best: {best_score:.8f}) ===")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_final, obj_final = full_polish(best_seq, time_budget=min(remaining * 0.7, 100.0))
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: {best_score:.8f}")
            else:
                print(f"  Final: {obj_final:.8f}")

        # Last-resort perturbations
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_final = len(best_seq)
            rng = np.random.default_rng(seed + 77777)
            for pi in range(5):
                if time.monotonic() > deadline - 5:
                    break
                scale = 0.01 + 0.008 * pi
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
                a_pert = normalize(a_pert)
                t_alloc = min(remaining * 0.15, 10.0)
                if t_alloc < 2:
                    break
                a_pert, obj_pert = subgradient_optimize(a_pert, max_iters=int(t_alloc * 12), time_budget=t_alloc)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb {pi}: ** {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
