    s = float(np.sum(a))

        return np.ones(len(a)) / len(a)
    return a / s


def build_band_matrix(a, ks):
    """Build sparse constraint rows for given k values. Returns dense (len(ks), n)."""
    n = len(a)
    A = np.zeros((len(ks), n))
    for idx, k in enumerate(ks):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[idx, j_start:j_end + 1] = a[j_start:j_end + 1]
    return A


def sparse_lp_direction(a, M, max_constraints=300, max_rounds=15, global_time_limit=20.0):
    """
    Sparse LP with constraint generation.
    max sum(g) s.t. conv(a,g)_k <= M for k in active set, g >= 0.
    """
    n = len(a)
    start_t = time.monotonic()

    # Initialize active set with top constraints
    conv_aa = compute_conv(a)
    top_k = np.argsort(-conv_aa)[:min(max_constraints, 2 * n - 1)]
    active_ks = sorted(int(k) for k in top_k)

    g = None
    for round_num in range(max_rounds):
        if time.monotonic() - start_t > global_time_limit - 2:
            break

        # Build constraint matrix
        A_sub = build_band_matrix(a, active_ks)

        # Solve LP
        c = -np.ones(n)
        b_ub = np.full(len(active_ks), M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
        except Exception:
            break

        if g is None:
            break

        # Check for violated constraints
        conv_ag = compute_conv_diff(a, g)
        violation = conv_ag - M
        max_viol = float(np.max(violation))

        if max_viol < 1e-8 * max(M, 1e-10):
            break  # All satisfied

        worst_k = int(np.argmax(violation))
        if worst_k in active_ks:
            break  # Numerical issue
        active_ks.append(worst_k)

        if len(active_ks) > max_constraints * 2:
            break

    if g is None:
        return None

    s = float(np.sum(g))
    if s < 1e-10:
        return None
    return g / s


def multi_level_lp_direction(a, M, top_k_count=10, ratio=0.7, global_time_limit=25.0):
    """
    LP with multi-level constraints: top peaks get stricter bounds.
    """
    n = len(a)
    start_t = time.monotonic()

    conv_aa = compute_conv(a)
    sorted_ks = np.argsort(-conv_aa)

    # Top peaks get bound M, next tier gets M*ratio
    tier1_ks = sorted_ks[:top_k_count]
    tier2_ks = sorted_ks[top_k_count:top_k_count * 3]

    all_ks = list(tier1_ks) + list(tier2_ks)
    all_ks = [int(k) for k in all_ks]

    A = build_band_matrix(a, all_ks)
    b_ub = np.zeros(len(all_ks))
    b_ub[:len(tier1_ks)] = M
    b_ub[len(tier1_ks):] = M * ratio

    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def soft_max_gradient_step(a, time_budget=40.0, tau_init=0.002, max_steps=300):
    """
    Gradient descent with soft-max smoothing and backtracking line search.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 3 * n:
        size <<= 1

    step_size = 0.05
    tau = tau_init

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Adaptive temperature
        tau = max(0.0001, tau_init * (0.995 ** step) + 0.3 * cmax / (n * np.sqrt(step + 1)))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient via cross-correlation
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Project onto sum=const
        grad = grad - np.mean(grad)

        # Backtracking line search
        alpha = step_size
        for _ in range(10):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                step_size = min(alpha * 1.2, 0.5)
                break
            alpha *= 0.5

        else:
            # No improvement, perturb and reduce step
            step_size *= 0.7
            if step_size < 1e-8:
                break
            # Small random perturbation to escape
            rng = np.random.default_rng(step)
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a


def spectral_shaping(a, time_budget=30.0, max_iters=80):
    """
    Spectral domain shaping: flatten the power spectrum while maintaining non-negativity.
    Uses proper handling of linear vs circular convolution.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Adaptive compression
        progress = it / max_iters
        alpha = 0.3 + 0.5 * progress

        # Shape: compress towards uniform
        P_norm = P / p_max
        # Keep DC component, compress the rest
        P_new = P.copy()
        P_new[1:] = (P_norm[1:] ** alpha) * p_max[1:] if p_max[1] > 0 else P_new[1:]

        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            blend = 0.7 + 0.2 * (it / max_iters)
            a = normalize(blend * a + (1 - blend) * a_new)

    return best_a


def lp_gradient_alternation(a, time_budget=120.0, seed=42):
    """
    Main optimization loop: alternate between LP direction and gradient steps.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    stall_count = 0
    best_this_phase = best_obj

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Decide strategy based on progress
        conv = compute_conv(a)
        M = float(np.max(conv))

        # Try LP direction
        lp_alloc = min(15.0, remaining * 0.3)
        g = None
        if n <= 1000:
            g = multi_level_lp_direction(a, M, top_k_count=8, ratio=0.75,
                                         global_time_limit=lp_alloc)
        if g is None and n <= 800:
            g = sparse_lp_direction(a, M, max_constraints=200, max_rounds=10,
                                    global_time_limit=lp_alloc)

        if g is not None:
            # Line search for optimal blend
            best_t = 0.0
            best_obj_t = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = normalize((1 - t) * a + t * g)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_t:
                    best_obj_t = obj_try
                    best_t = t

            if best_t > 0:
                a = normalize((1 - best_t) * a + best_t * g)
                obj_now = compute_obj(a)
                if obj_now < best_obj - 1e-12:
                    best_obj = obj_now
                    best_a = a.copy()
                    best_this_phase = obj_now
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                stall_count += 1
        else:
            stall_count += 1

        # Gradient step
        grad_alloc = min(15.0, remaining * 0.3)
        if stall_count >= 2 or g is None:
            a_grad = soft_max_gradient_step(a, time_budget=grad_alloc,
                                            tau_init=0.002, max_steps=50)
            obj_grad = compute_obj(a_grad)
            if obj_grad < best_obj:
                best_obj = obj_grad
                best_a = a_grad.copy()
                a = a_grad
                stall_count = 0
                best_this_phase = obj_grad
            else:
                # Partial blend
                a = normalize(0.8 * a + 0.2 * a_grad)

        # Occasional spectral shaping
        if iter_count % 5 == 4 and remaining > 30:
            a_spec = spectral_shaping(a, time_budget=min(10, remaining * 0.1), max_iters=20)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_obj:
                best_obj = obj_spec
                best_a = a_spec.copy()
                a = a_spec
                stall_count = 0
            else:
                a = normalize(0.85 * a + 0.15 * a_spec)

        # Perturbation to escape local minima
        if stall_count >= 5:
            perturbation = 0.03 + 0.02 * (stall_count // 5)
            a = normalize(a * np.exp(rng.normal(0, perturbation, n)))
            stall_count = 0

        iter_count += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(x_new, x_old, a)
    return np.maximum(a_interp, 0.0)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def diverse_initialization(n, seed):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)
    strategy = seed % 5

    if strategy == 0:
        # Uniform with noise
        return normalize(np.ones(n) * (1 + 0.1 * rng.normal(size=n)))
    elif strategy == 1:
        # Raised cosine window
        t = np.arange(n) / (n - 1)
        w = 0.5 * (1 + np.cos(2 * np.pi * t))
        return normalize(w + 0.05 * rng.normal(size=n))
    elif strategy == 2:
        # Gaussian window
        t = np.linspace(-1, 1, n)
        w = np.exp(-t ** 2 / 0.3)
        return normalize(w + 0.02 * rng.normal(size=n))
    elif strategy == 3:
        # Dirichlet random
        return normalize(rng.dirichlet(np.ones(n) * 0.5))
    else:
        # Piecewise constant blocks
        n_blocks = rng.integers(3, 10)
        block_sizes = rng.multinomial(n, np.ones(n_blocks) / n_blocks)
        amplitudes = rng.uniform(0.5, 2.0, n_blocks)
        a = np.repeat(amplitudes, block_sizes)[:n]
        return normalize(a + 0.05 * rng.normal(size=n))


def coordinate_refine(a, time_budget=15.0):
    """
    Targeted coordinate descent: move mass from peak contributors to reduce max.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_start > j_end:
            break

        contribs = []
        for j in range(j_start, j_end + 1):
            k2 = k_star - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contribs[:8]:
            if time.monotonic() - start > time_budget:
                break

            for attempt in range(5):
                j_target = int(rng.integers(0, n))
                if j_target in (j1, j2):
                    continue

                for frac in [0.03, 0.05, 0.08, 0.12]:
                    delta = frac * a[j1]
                    if a[j1] - delta < 0:
                        continue
                    a_try = a.copy()
                    a_try[j1] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)

                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved:
            # Try moving mass from j2
            for peak_val, j1, j2 in contribs[:5]:
                j_target = int(rng.integers(0, n))
                if j_target in (j1, j2):
                    continue
                delta = 0.05 * a[j2]
                if a[j2] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[j2] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function with multi-phase strategy.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # =========================================================
    # Phase 1: Core optimization at parent length (55% of time)
    # =========================================================
    phase1_budget = (deadline - start_time) * 0.55
    print(f"\n{'='*60}")
    print(f"Phase 1: Core LP+Gradient optimization (budget={phase1_budget:.0f}s)")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())

        # Run multiple cycles with increasing aggressiveness
        for cycle in range(3):
            if time.monotonic() > deadline - phase1_budget * 0.4:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            cycle_budget = min(remaining * 0.35, phase1_budget / 3.0)
            print(f"  Cycle {cycle}: budget={cycle_budget:.0f}s")

            a_new = lp_gradient_alternation(a, time_budget=cycle_budget, seed=seed + cycle * 100)
            obj_new = compute_obj(a_new)

            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  Cycle {cycle} improved: {best_score:.8f}")
            else:
                print(f"  Cycle {cycle}: {obj_new:.8f} (no improvement)")
                # Try a perturbed version
                rng = np.random.default_rng(seed + cycle * 200)
                a = normalize(a * np.exp(rng.normal(0, 0.05, len(a))))

    # =========================================================
    # Phase 2: Spectral + targeted refinement (20% of time)
    # =========================================================
    phase2_budget = (deadline - start_time) * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 2: Spectral + coordinate refinement")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a = normalize(best_seq.copy())

            # Spectral shaping
            spec_budget = min(25.0, remaining * 0.4)
            a_spec = spectral_shaping(a, time_budget=spec_budget, max_iters=60)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                print(f"  Spectral improved: {best_score:.8f}")
            else:
                # Blend and continue
                a = normalize(0.7 * normalize(best_seq) + 0.3 * a_spec)
                print(f"  Spectral: {obj_spec:.8f} (no improvement)")

            # Coordinate refinement
            coord_budget = min(20.0, remaining * 0.3)
            if time.monotonic() < deadline - 20:
                a_coord = coordinate_refine(a, time_budget=coord_budget)
                obj_coord = compute_obj(a_coord)
                if obj_coord < best_score:
                    best_score = obj_coord
                    best_seq = a_coord.copy()
                    print(f"  Coordinate improved: {best_score:.8f}")
                else:
                    print(f"  Coordinate: {obj_coord:.8f}")

            # Follow-up LP on the best
            if time.monotonic() < deadline - 60:
                follow_budget = min(40.0, deadline - time.monotonic() - 40)
                a_follow = lp_gradient_alternation(best_seq, time_budget=follow_budget,
                                                   seed=seed + 500)
                obj_follow = compute_obj(a_follow)
                if obj_follow < best_score:
                    best_score = obj_follow
                    best_seq = a_follow.copy()
                    print(f"  Follow-up improved: {best_score:.8f}")

    # =========================================================
    # Phase 3: Multi-n exploration (15% of time)
    # =========================================================
    phase3_budget = (deadline - start_time) * 0.15
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 200), max(200, n_best - 100),
            n_best,
            n_best + 100, n_best + 200, n_best + 400
        ]))
        n_options = [n for n in n_options if 200 <= n <= 1500]

        for n_try in n_options:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            if n_try == n_best:
                a_try = normalize(best_seq.copy())
            else:
                a_try = normalize(resample_sequence(best_seq, n_try))

            alloc = min(remaining * 0.25, 40.0)
            print(f"  n={n_try}: starting (budget={alloc:.0f}s)")

            a_opt = lp_gradient_alternation(a_try, time_budget=alloc, seed=seed + n_try)
            obj_try = compute_obj(a_opt)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  ** n={n_try}: {best_score:.8f} IMPROVED **")
            else:
                print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 4: Random restarts with diverse initializations (7% of time)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Diverse restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        remaining = deadline - time.monotonic()

        restarts = min(6, max(2, int(remaining / 30)))
        for r in range(restarts):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            alloc = min(remaining * 0.2, 30.0)
            strat_seed = seed + 1000 + r * 7

            # Diverse initialization
            if r % 3 == 0:
                a_try = diverse_initialization(n_cur, strat_seed)
            elif r % 3 == 1:
                # Blend parent with diverse
                a_div = diverse_initialization(n_cur, strat_seed)
                a_try = normalize(0.7 * normalize(best_seq) + 0.3 * a_div)
            else:
                # Modulated parent
                t = np.arange(n_cur) / n_cur
                freq = np.random.default_rng(strat_seed).uniform(1, 4)
                mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * t)
                a_try = normalize(best_seq * mod)

            a_opt = lp_gradient_alternation(a_try, time_budget=alloc, seed=strat_seed)
            obj_opt = compute_obj(a_opt)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Restart {r}: {best_score:.8f} ** IMPROVED **")
            else:
                print(f"  Restart {r}: {obj_opt:.8f}")

    # =========================================================
    # Phase 5: Final intensive polish (8% of time)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_final = normalize(best_seq.copy())

            # Multiple short polish rounds
            polish_rounds = min(4, max(1, int(remaining / 20)))
            for pr in range(polish_rounds):
                if time.monotonic() > deadline - 10:
                    break
                round_budget = min(20.0, remaining * 0.3)

                # Gradient polish
                a_pol = soft_max_gradient_step(a_final, time_budget=round_budget * 0.5,
                                               tau_init=0.0005, max_steps=100)

                # Coordinate polish
                if time.monotonic() < deadline - 10:
                    a_pol = coordinate_refine(a_pol, time_budget=min(8, round_budget * 0.3))

                obj_pol = compute_obj(a_pol)
                if obj_pol < best_score:
                    best_score = obj_pol
                    best_seq = a_pol.copy()
                    a_final = a_pol
                    print(f"  Polish {pr}: {best_score:.8f}")
                else:
                    a_final = normalize(0.9 * a_final + 0.1 * a_pol)
                    print(f"  Polish {pr}: {obj_pol:.8f} (no improvement)")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
