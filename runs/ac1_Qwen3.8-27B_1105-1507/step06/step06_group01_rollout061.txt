```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_M(a):
    """Compute max of self-convolution."""
    return fast_conv(a).max()

def proj_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (1.0 + u * rho) > cv
    if not cond.any():
        return np.ones(n) / n
    rho_val = int(np.argmax(~cond))
    theta = (cv[rho_val] - 1.0) / (rho_val + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

    b = fast_conv(a)

    # (a*a)_k = sum_j a[j]*a[k-j]
    # d/d a[i] (a*a)_k = a[k-i] + a[k-i] = 2*a[k-i] (if i != k-i)
```

```python
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv

        return np.ones(n) / n
    rho_val = int(np.argmax(~cond))

    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s
```

```python
    # grad[i] = 2 * sum_j p[i+j] * a[j] for i=0,...,n-1
    # This is 2 * (p ⊛ a)[i]
    
    L = 1
    while L < 3*n:
        L <<= 1
    
    p_ext = np.zeros(L)
    p_ext[:2*n-1] = p
    a_ext = np.zeros(L)
    a_ext[:n] = a
    
    # Cross-correlation: IFFT(FFT(p) * conj(FFT(a)))
    P = np.fft.rfft(p_ext)
    A = np.fft.rfft(a_ext)
    cc = np.fft.irfft(P * np.conj(A), L)
    
    grad = 2.0 * cc[:n]
    return smooth_val, grad
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. FFT-based FISTA on log-sum-exp smooth approximation with very high tau (up to 3000)
2. LP best-response with full active set (all constraints within 95% of max)
3. Adaptive blending with dual-based stopping certificate
4. Targeted coordinate descent on peak-contributing coordinates via golden section
5. Multi-restart with perturbation to escape local minima
Key differences from previous: much higher tau, full active set LP, FFT gradient,
adaptive step sizes, and spectral perturbation restarts.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv
    if not cond.any():
        return np.ones(n) / n
    rho_val = int(np.argmax(~cond))
    theta = (cv[rho_val] - 1.0) / (rho_val + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad_fft(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = b - b_max
    exp_b = np.exp(tau * shifted)
    sum_exp = exp_b.sum()
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(b))] = 1.0
        smooth_val = b_max
    else:
        p = exp_b / sum_exp
        smooth_val = b_max + np.log(sum_exp) / tau

    # grad[i] = 2 * sum_j p[i+j] * a[j] for i=0,...,n-1
    # Use FFT-based cross-correlation
    L = 1
    while L < 3 * n:
        L <<= 1
    p_ext = np.zeros(L)
    p_ext[:2 * n - 1] = p
    a_ext = np.zeros(L)
    a_ext[:n] = a
    P = np.fft.rfft(p_ext)
    A = np.fft.rfft(a_ext)
    cc = np.fft.irfft(P * np.conj(A), L)
    grad = 2.0 * cc[:n]
    return smooth_val, grad


def run_fista(a_init, tau_start, tau_end, max_iters, deadline, lr=0.3):
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)

    t = 1.0
    y = a.copy()
    a_prev = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(1, max_iters - 1))
        tau = tau_start * (tau_end / tau_start) ** frac

        _, grad_y = smooth_max_grad_fft(y, tau)
        grad_tan = grad_y - grad_y.mean()

        step = lr / (1.0 + 0.02 * np.sqrt(tau))
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t = t_new

        if it % 10 == 0 or it == max_iters - 1:
            M_y = compute_M(y)
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
            M_a = compute_M(a)
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()

    M_final = compute_M(a)
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    return best_a


def build_lp_matrix(a, active_indices):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def solve_lp_best_response(a, M, active_indices):
    n = len(a)
    if len(active_indices) == 0:
        return None, 0.0
    A = build_lp_matrix(a, active_indices)
    if A.nnz == 0:
        return None, 0.0
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, 0.0


def run_lp_mirror(a_init, deadline, frac_active=0.95, min_active=10, max_active=100):
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()

        M_cur = b.max()

        # Active set: all k where b[k] > frac_active * M_cur

        active = active[:max_active]

        g, g_sum = solve_lp_best_response(a, M_cur, active)
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            # Try with more relaxed threshold
            frac_active = max(0.8, frac_active - 0.02)
            continue

        # If dual value <= 1, we're at (approximate) optimum
        if g_sum <= 1.0 + 1e-6:
            break

        # Adaptive step size: try decreasing t until improvement
        improved = False
        t_candidates = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = compute_M(a_trial)
            if M_trial < best_M - 1e-14:

                best_M = M_trial
                best_a = a.copy()
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1
            frac_active = min(1.0, frac_active + 0.01)
            if no_improve > 15:
                # Reset frac_active and try again with different step
                frac_active = 0.95
                if no_improve > 30:
                    break

    M_final = compute_M(a)
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    return best_a


def coord_optimize(a, i, b, n):
    """Find optimal a[i] given all others fixed, using golden section on 1D objective."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if quad_pos >= 0:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(3.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.3)
    for _ in range(8):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 50.0:
            hi *= 2.0
        else:
            break

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coord_descent(a_init, deadline, seed=0):
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for sweep in range(50000):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        sweep_improved = False
        # Focus on coordinates that affect the top convolution values
        top_k = np.argsort(b)[::-1][:min(50, 2 * n - 1)]
        relevant_coords = set()
        for k in top_k:
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                relevant_coords.add(j)
        relevant_coords = list(relevant_coords)
        rng_cd.shuffle(relevant_coords)

        for i in relevant_coords:
            x_opt = coord_optimize(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                delta = x_opt - a[i]
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += x_opt * x_opt - a[i] * a[i]
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * delta
                a[i] = x_opt
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = compute_M(a)
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 2
        if no_improve > 20:
            break

    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.02, n), 1e-5)
        inits.append(p_n / p_n.sum())

    inits.append(np.ones(n) / n)

    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())

    # Spectral flattening init
    a_spec = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_spec)
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.05 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_full_pipeline(a0, deadline, seed=0):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a0)
    best = a0.copy()
    best_M = compute_M(a0)
    t_start = time.monotonic()
    dur = max(1, deadline - t_start)

    # Stage 1: FISTA (40% of time)
    t1 = t_start + 0.4 * dur
    if time.monotonic() < t1:
        for tau_s in [50, 200, 800]:
            if time.monotonic() > t1:
                break
            a_f = run_fista(best, tau_s, tau_s * 4, 5000, t1, lr=0.25)
            M_f = compute_M(a_f)
            if M_f < best_M:
                best_M = M_f
                best = a_f.copy()
    print(f"    After FISTA: M={best_M:.12f}", flush=True)

    # Stage 2: LP (30% of time)
    t2 = t_start + 0.7 * dur
    if time.monotonic() < t2:
        a_lp = run_lp_mirror(best, t2, frac_active=0.95, min_active=10, max_active=80)
        M_lp = compute_M(a_lp)
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
    print(f"    After LP: M={best_M:.12f}", flush=True)

    # Stage 3: Coordinate descent (30% of time)
    if time.monotonic() < deadline:
        a_cd = run_coord_descent(best, deadline, seed=seed)
        M_cd = compute_M(a_cd)
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
    print(f"    After CD: M={best_M:.12f}", flush=True)

    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size

    # === Phase 1: Multi-length, multi-init search (60% of time) ===
    lengths_to_try = [n_parent]
    for delta in [256, -256, 512]:
        nl = n_parent + delta
        if 384 <= nl <= 2560 and nl not in lengths_to_try:
            lengths_to_try.append(nl)

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length search on {lengths_to_try}", flush=True)
    print(f"{'='*60}", flush=True)

    t_phase1_end = time.monotonic() + 0.55 * (deadline - time.monotonic())
    n_lengths = len(lengths_to_try)
    time_per_length = (t_phase1_end - time.monotonic()) / n_lengths

    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > t_phase1_end:
            break
        t0 = time.monotonic()
        time_alloc = min(time_per_length, t_phase1_end - time.monotonic())
        if time_alloc < 30:
            break
        print(f"\n  Length n={n}, time_alloc={time_alloc:.0f}s", flush=True)

        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_M_for_n = float('inf')

        n_inits = len(inits)
        time_per_init = time_alloc / max(1, n_inits)

        for ii, a0 in enumerate(inits):
            if time.monotonic() > t_phase1_end:
                break
            sub_deadline = min(t0 + time_alloc, t_phase1_end)
            if time.monotonic() >= sub_deadline - 5:
                break

            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            a_opt = run_full_pipeline(a0, sub_deadline, seed=seed + li * 100 + ii)
            M_opt = compute_M(a_opt)

            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best M={best_M_for_n:.12f} J={J_val:.6f} "
                  f"(elapsed={time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("After Phase 1")

    # === Phase 2: Intensive LP polish on best (15% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive LP polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 60:
        t_lp_deadline = min(deadline - 60, time.monotonic() + 0.15 * (deadline - time.monotonic()))
        a_lp = run_lp_mirror(tracker.best, t_lp_deadline, frac_active=0.97, min_active=10, max_active=120)
        tracker.update(a_lp)

        if time.monotonic() < deadline - 60:
            a_lp2 = run_lp_mirror(a_lp, min(deadline - 50, time.monotonic() + 60),
                                  frac_active=0.99, min_active=10, max_active=60)
            tracker.update(a_lp2)

    tracker.report("After Phase 2")

    # === Phase 3: High-tau FISTA refinement (10% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: High-tau FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 60:
        t3_deadline = min(deadline - 50, time.monotonic() + 0.10 * (deadline - time.monotonic()))
        for tau in [1000, 2000, 3000]:
            if time.monotonic() > t3_deadline:
                break
            a_f = run_fista(tracker.best, tau, tau * 1.5, 8000, t3_deadline, lr=0.15)
            tracker.update(a_f)

    tracker.report("After Phase 3")

    # === Phase 4: Coordinate descent polish (10% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        t4_deadline = min(deadline - 20, time.monotonic() + 0.10 * (deadline - time.monotonic()))
        a_cd = run_coord_descent(tracker.best, t4_deadline, seed=99999)
        tracker.update(a_cd)

    tracker.report("After Phase 4")

    # === Phase 5: Restart with perturbation (remaining time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Perturbation restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_best = tracker.best.size
        for restart in range(5):
            if time.monotonic() > deadline - 30:
                break
            noise_scale = 0.01 * (1 + restart * 0.5)
            a_pert = tracker.best + rng.normal(0, noise_scale, n_best)
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert = a_pert / a_pert.sum()
            sub_dl = min(deadline - 20, time.monotonic() + min(60, (deadline - 30 - time.monotonic()) / (5 - restart)))
            a_opt = run_full_pipeline(a_pert, sub_dl, seed=seed + 500 + restart)
            tracker.update(a_opt)

    tracker.report("After Phase 5")

    # Final validation
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
```
