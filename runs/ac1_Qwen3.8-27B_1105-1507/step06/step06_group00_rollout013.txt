"""
Smoothed subgradient / mirror descent for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines fast O(n log n) gradient steps with LP polishing and multi-n search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smoothed_subgradient_step(a, tau, eta):

def mirror_descent_step(a, tau, eta):
    """
    One step of mirror descent (multiplicative weights) with KL divergence.
    a is a probability vector.
    """
    n = len(a)
    conv = compute_conv(a)
    
    tc = tau * conv
    p = np.exp(tc - tc.max())
    p /= p.sum()
    
    # Same gradient computation
    p_ext = np.zeros(2 * n - 1 + n)
    p_ext[:2*n-1] = p
    p_cumsum = np.zeros(len(p_ext) + 1)
    p_cumsum[1:] = np.cumsum(p_ext)
    
    S = np.zeros(n)
    for j in range(n):
        k_end = min(2 * n - 2, j + n - 1)
        S[j] = p_cumsum[k_end + 1] - p_cumsum[j]
    
    if n <= 3000:
        row_idx = np.arange(n)[:, None]
        col_idx = np.arange(n)[None, :]
        P_indices = row_idx + col_idx
        P = np.where(P_indices < 2*n-1, p[P_indices.clip(0, 2*n-2)], 0.0)
        C = P @ a
    else:
        C = np.zeros(n)
        for j in range(n):
            m_max = min(n - 1, 2*n-2-j)
            C[j] = np.dot(a[:m_max+1], p[j:j+m_max+1])
    
    grad = a * S + C
    
    # Mirror descent update (multiplicative weights)
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-30)
    a_new /= a_new.sum()
    
    return a_new, p


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
        for solver in solvers:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget=60.0, n_iters=100):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Line search
        best_t_obj = best_obj
        best_t_val = 0.5
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t

        if best_t_obj < best_obj - 1e-12:
            a_new = (1.0 - best_t_val) * a + best_t_val * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break

    return best_a


def smoothed_optimize(a, time_budget=60.0, verbose=False):
    """
    Smoothed subgradient / mirror descent optimization.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adaptive step size and tau
    tau = 2.0  # smoothing parameter
    eta = 0.1  # step size
    tau_increase = 1.05
    eta_decay = 0.98
    stall = 0

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute smoothed subgradient step
        a_new, p = smoothed_subgradient_step(a, tau, eta)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = best_a.copy()  # revert
            stall += 1
        
        # Anneal: increase tau (sharper), decrease eta
        if stall > 10:
            tau *= tau_increase
            eta *= eta_decay
            stall = 0
            
        if verbose and it % 100 == 0:
            print(f"    smoothed it{it}: obj={best_obj:.8f}, tau={tau:.3f}, eta={eta:.5f}")

    # Try mirror descent as well
    a_md = best_a.copy()
    eta_md = 0.05
    tau_md = tau
    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break
        a_new, p = mirror_descent_step(a_md, tau_md, eta_md)
        obj_new = compute_obj(a_new)
        if obj_new < compute_obj(best_a) - 1e-12:
            a_md = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
        else:
            a_md = best_a.copy()
            eta_md *= 0.95
            tau_md *= 1.02

    return best_a


def random_init(n, rng):
    """Generate random non-negative sequence initialized to simplex."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.exponential(1.0, n)
        a /= a.sum()
    elif mode == 2:
        # Bimodal
        center = n // 2
        width = n // 4
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = a + 0.5 * (np.exp(-((x - center*0.3) ** 2) / (2 * (width//2) ** 2)) + 
                       np.exp(-((x - center*1.7) ** 2) / (2 * (width//2) ** 2)))
        a /= a.sum()
    elif mode == 3:
        # Monotone
        a = np.random.dirichlet(np.ones(n) * 0.5)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = (np.cos(x) ** 2 + 0.3)
        a /= a.sum()
    else:
        # Random smooth
        freqs = rng.integers(1, 5, 4)
        phases = rng.uniform(0, 2*np.pi, 4)
        amps = rng.uniform(0.5, 1.0, 4)
        x = np.linspace(0, 1, n)
        a = 1.0
        for f, ph, am in zip(freqs, phases, amps):
            a += am * np.cos(2*np.pi*f*x + ph)
        a = np.maximum(a, 0.01)
        a /= a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    return a_new


def coordinate_descent(a, time_budget=30.0):
    """
    Coordinate descent: iteratively adjust the peak-contributing coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Find coordinates that contribute to the peak
        # grad[j] = a[j] + a[k_star - j] for valid j
        for i in range(0, n, max(1, n // 200)):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j > i:
                # Try moving mass from i to j (or vice versa)
                for frac in [0.01, 0.03, 0.05]:
                    for direction in [-1, 1]:
                        delta = direction * frac * a[i]
                        if a[i] + delta < 0 or a[j] + delta < 0:
                            continue
                        a[i] += delta
                        a[j] -= delta
                        s_try = a.sum()
                        if s_try < 1e-10:
                            a[i] -= delta
                            a[j] += delta
                            continue
                        a_try = a / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                        else:
                            a[i] -= delta
                            a[j] += delta
                    if improved:
                        break

        if not improved and it > 20:
            break

    return best_a
