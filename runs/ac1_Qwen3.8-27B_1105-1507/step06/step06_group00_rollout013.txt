"""
Improved LP-alternating optimization with successive linearization,
multi-anchor constraints, peak shaving, and systematic n exploration.
Key innovations: (1) successive linearization LP as alternative to min-peak LP,
(2) over-relaxation and momentum in the alternating scheme,
(3) multi-anchor constraints for robustness, (4) peak-shaving coordinate descent,
(5) systematic exploration of sequence lengths.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp_minpeak(a, M_init=None):
    """Solve min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_init is not None:
        constraints.append(M <= M_init)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_successive_lin(a_curr):
    """
    Successive linearization LP:
    min t s.t. 2*(a_curr*a)[k] - f_k(a_curr) <= t, sum(a)=1, a>=0
    where f(a_curr) = a_curr * a_curr.
    """
    n = len(a_curr)
    a_curr = a_curr / np.sum(a_curr)
    A = build_conv_matrix(a_curr)
    f = A @ a_curr  # a_curr * a_curr

    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    # Constraint: 2*A@a - t <= f  =>  2*A@a - t*ones <= f
    constraints = [2.0 * (A @ a) - t <= f, cp.sum(a) == 1]

    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if a.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.maximum(a.value, 0)
                s = float(np.sum(a_val))
                if s > 1e-10:
                    return a_val / s, float(t.value)
        except Exception:
            continue
    return None, None


def solve_lp_multi_anchor(anchors, M_target):
    """Solve max sum(g) s.t. (a_i*g)[k] <= M_target for all anchors, g>=0."""
    n = len(anchors[0])
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for a in anchors:
        a = a / np.sum(a)
        A = build_conv_matrix(a)
        constraints.append(A @ g <= M_target)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def alternating_optimize(a, time_budget=100.0, verbose=False, method='minpeak'):
    """
    LP alternating with over-relaxation.
    method: 'minpeak' for original LP, 's.lin' for successive linearization.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    t_relax = 1.0  # relaxation parameter
    momentum = np.zeros(n)

    max_iters = 1000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        if method == 's.lin':
            g_lp, M_lp = solve_successive_lin(a)
        else:
            g_lp, M_lp = solve_lp_minpeak(a)

        if g_lp is None:
            # Fallback
            conv_a = compute_conv(a)
            M_cur = float(np.max(conv_a))
            g_lp = _solve_lp_maxsum(a, M_cur)
            if g_lp is None:
                break

        # Over-relaxation: a_new = a + t*(g - a) = (1-t)*a + t*g, can have t>1
        best_t_obj = np.inf
        best_t_val = 1.0
        best_a_local = best_a.copy()

        candidate_ts = [0.5, 0.7, 0.9, 1.0, 1.1, 1.2, 1.3, 1.5]
        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            # Apply momentum
            velocity = best_a_local - a
            momentum = 0.8 * momentum + 0.2 * velocity
            a = best_a_local + 0.1 * momentum
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
                obj_check = compute_obj(a)
                if obj_check > best_obj:
                    a = best_a_local  # revert momentum if it hurt
                    best_a = a.copy()
                else:
                    best_a = a.copy()
                    best_obj = obj_check
            else:
                a = best_a_local.copy()
                best_a = a.copy()
            t_relax = best_t_val
            stall = 0
            if verbose and it % 10 == 0:
                print(f"        it{it}: obj={best_obj:.8f}, t={best_t_val:.2f}")
        else:
            stall += 1
            if stall > 30:
                # Perturb to escape local minimum
                rng = np.random.default_rng(hash((it, n, int(time.monotonic()*1000))) % (2**32))
                scale = 0.02 + 0.02 * min(stall // 30, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                stall = 0
            elif stall > 10:
                # Try switching method
                method = 's.lin' if method == 'minpeak' else 'minpeak'

    return best_a


def _solve_lp_maxsum(a, M_target):
    """Solve max sum(g) s.t. (a*g)[k] <= M_target, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M_target]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def multi_anchor_optimize(a, time_budget=100.0, n_anchors=3, verbose=False):
    """
    Multi-anchor LP: maintain a set of anchor sequences and solve
    max sum(g) s.t. (a_i * g)[k] <= M for all anchors.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Initialize anchor set
    anchors = [a.copy()]
    rng = np.random.default_rng(12345)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        # Compute M from current best
        conv_a = compute_conv(a)
        M_target = float(np.max(conv_a)) * 0.95  # slightly tighter

        # Solve multi-anchor LP
        if len(anchors) > 1:
            g_lp = solve_lp_multi_anchor(anchors, M_target)
        else:
            g_lp = _solve_lp_maxsum(a, M_target)

        if g_lp is None:
            # Relax M target
            M_target = float(np.max(conv_a)) * 1.0
            g_lp = _solve_lp_maxsum(a, M_target)
            if g_lp is None:
                break

        # Update
        a_try = (1 - 0.5) * a + 0.5 * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                # Update anchor set
                if len(anchors) < n_anchors:
                    anchors.append(a_try.copy() * np.exp(rng.normal(0, 0.05, n)))
                    anchors[-1] = np.maximum(anchors[-1], 0)
                    anchors[-1] = anchors[-1] / np.sum(anchors[-1])
                else:
                    # Replace random anchor
                    idx = rng.integers(0, n_anchors)
                    anchors[idx] = a_try.copy()

        if verbose and it % 10 == 0:
            print(f"        multi-anchor it{it}: obj={best_obj:.8f}")

    return best_a


def peak_shave(a, time_budget=30.0, verbose=False):
    """
    Peak shaving: iteratively reduce the top peaks by adjusting
    contributing coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Find top peaks
        top_k = np.argsort(conv)[::-1][:5]
        k_star = int(top_k[0])

        # Compute gradient for the top peak
        grad = np.zeros(n)
        for k_idx in top_k:
            k = int(k_idx)
            weight = conv[k] - conv[top_k[-1]]  # weight by excess
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += weight * a[j]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Try multiple step sizes
        improved = False
        for step in [0.01, 0.02, 0.03, 0.05, 0.08]:
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            # Try targeted coordinate moves
            k_star = int(np.argmax(compute_conv(a)))
            coord_improved = False
            for i in range(0, n, 2):
                j = k_star - i
                if 0 <= j < n and j != i and a[i] > 0.01 * a.max():
                    for frac in [0.02, 0.05]:
                        delta = frac * a[i]
                        a[i] -= delta
                        a[j] += delta
                        obj_try = compute_obj(a)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a.copy()
                            coord_improved = True
                            break
                        else:
                            a[i] += delta
                            a[j] -= delta
                            break
                if coord_improved:
                    break
            if not coord_improved:
                # Perturb
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_obj = compute_obj(a)
                if best_obj > 1.51:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Raised cosine (Hann-like)
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Flat top
        x = np.linspace(0, 1, n)
        a = np.where(np.abs(x - 0.5) < 0.3, 1.0, 0.5)
        a = a + 0.1 * rng.normal(0, 1, n)
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    elif mode == 4:
        # Bimodal
        centers = n * rng.uniform(0.2, 0.8, 2)
        widths = n * rng.uniform(0.05, 0.2, 2)
        x = np.arange(n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-(x - c) ** 2 / (2 * w ** 2))
        a = a / np.sum(a)
    else:
        # Smooth random
        a = rng.uniform(0, 1, n)
        # Smooth with a simple filter
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = a + 0.1
        a = a / np.sum(a)
    return a


def combined_optimize(a, time_budget=100.0, verbose=False):
    """
    Combined optimization: alternate between min-peak LP, successive linearization,
    and peak shaving for maximum effect.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase A: min-peak LP alternating
    alloc_a = time_budget * 0.35
    a_new = alternating_optimize(a, time_budget=alloc_a, verbose=verbose, method='minpeak')
    obj_new = compute_obj(a_new)
    if obj_new < best_obj:
        best_obj = obj_new
        best_a = a_new.copy()
        a = a_new.copy()
    if verbose:
        print(f"  After min-peak LP: obj={best_obj:.8f}")

    # Phase B: successive linearization
    remaining = deadline_remaining(start, time_budget)
    if remaining > 10:
        alloc_b = remaining * 0.35
        a_new = alternating_optimize(a, time_budget=alloc_b, verbose=verbose, method='s.lin')
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        if verbose:
            print(f"  After s.lin: obj={best_obj:.8f}")

    # Phase C: peak shaving
    remaining = deadline_remaining(start, time_budget)
    if remaining > 10:
        alloc_c = remaining * 0.3
        a_new = peak_shave(a, time_budget=alloc_c)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        if verbose:
            print(f"  After peak shave: obj={best_obj:.8f}")

    # Phase D: another round of LP
    remaining = deadline_remaining(start, time_budget)
    if remaining > 15:
        a_new = alternating_optimize(best_a, time_budget=remaining * 0.4, method='minpeak')
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a


def deadline_remaining(start, budget):
    return max(0, (start + budget) - time.monotonic())


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # Phase 1: Intensive combined optimization on parent
    print(f"\n=== Phase 1: Combined Optimization (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc1 = min(remaining * 0.3, 350.0)
        a_opt = combined_optimize(best_seq, time_budget=alloc1, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1 result: obj={best_score:.8f}")
        else:
            print(f"  Phase 1: no improvement (obj={obj_opt:.8f})")

    # Phase 2: Multi-anchor optimization
    print(f"\n=== Phase 2: Multi-Anchor Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc2 = min(remaining * 0.2, 150.0)
        a_opt = multi_anchor_optimize(best_seq, time_budget=alloc2, n_anchors=3, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 2 result: obj={best_score:.8f}")
        else:
            print(f"  Phase 2: no improvement (obj={obj_opt:.8f})")

    # Phase 3: Systematic n exploration
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(400, n_best - 100), max(400, n_best - 50),
        n_best,
        n_best + 50, n_best + 100, n_best + 200,
        600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500]
    # Remove duplicates and sort
    n_options = sorted(set(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.2, 60.0)
        a_opt = alternating_optimize(a_try, time_budget=alloc, method='minpeak')
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 4: Random restarts with diverse initializations
    print(f"\n=== Phase 4: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        init_type = r % 5
        if init_type == 0:
            a_r = random_init(n_best, rng)
        elif init_type == 1:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            # Different n with perturbation
            n_r = int(n_best + rng.integers(-150, 150))
            n_r = max(400, min(1500, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.04, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 3:
            # Cosine window variants
            n_r = n_best
            x = np.linspace(0, np.pi, n_r)
            a_r = np.cos(x / 2) ** (1 + r * 0.3) + 0.05
            a_r = a_r / np.sum(a_r)
        else:
            # Structural: two-lobe
            n_r = n_best
            x = np.arange(n_r)
            c1 = n_r * 0.3
            c2 = n_r * 0.7
            w = n_r * 0.15
            a_r = np.exp(-(x - c1) ** 2 / (2 * w ** 2)) + np.exp(-(x - c2) ** 2 / (2 * w ** 2)) + 0.1
            a_r = a_r / np.sum(a_r)

        alloc_r = min(remaining * 0.15, 50.0)
        a_opt = combined_optimize(a_r, time_budget=alloc_r)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_r:.8f}")

    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_final = best_seq.copy()

        # Combined optimization with remaining time
        alloc_final = remaining * 0.6
        a_opt = combined_optimize(a_final, time_budget=alloc_final, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final combined: obj={best_score:.8f}")

        # Peak shaving
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_ps = peak_shave(best_seq, time_budget=min(remaining * 0.3, 30.0))
            obj_ps = compute_obj(a_ps)
            if obj_ps < best_score:
                best_score = obj_ps
                best_seq = a_ps.copy()
                print(f"  Final peak shave: obj={best_score:.8f}")

        # One more LP round
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_last = alternating_optimize(best_seq, time_budget=min(remaining * 0.5, 40.0), method='s.lin')
            obj_last = compute_obj(a_last)
            if obj_last < best_score:
                best_score = obj_last
                best_seq = a_last.copy()
                print(f"  Final s.lin: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
