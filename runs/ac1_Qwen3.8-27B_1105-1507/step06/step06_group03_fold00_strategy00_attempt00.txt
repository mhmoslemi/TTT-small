                if new_obj < best_obj:
                    a_curr[idx] = scaled_new
                    improved = True
                    best_obj = new_obj
                    break
            # Case divergence: change_midpoint
            if no_improve > 60:
                # Modify the sequence's midpoint region
                break
        if improved:
            # Compute new objective
            current_conv = np.convolve(a_curr, a_curr, mode='full')
            current_sum = sum(a_curr)
            current_obj = 2.0 * len(a_curr) * current_conv.max() / (current_sum ** 2)
            if current_obj < best_obj:
                best_obj = current_obj
                best_seq = a_curr.copy()
            no_improve = 0
        else:
            no_improve += 1
    return best_seq.tolist(), best_obj

def smooth_adam_step(init_seq, max_iters, deadline, verbose=True):
    """Adam optimization with smoothed gradient descent."""
    n = len(init_seq)
    init_seq = np.array(init_seq)
    total_sum = init_seq.sum()
    if total_sum < 1e-10:
        init_seq = np.ones(n)
        total_sum = 1.0
    init_seq = init_seq / total_sum
    best_seq = init_seq.copy()
    obj_val = 2.0 * n * np.convolve(init_seq, init_seq, mode='full').max() / total_sum**2
    best_obj = obj_val
    no_improve = 0
    for it in range(int(max_iters)+int(np.ceil(0.1*max_iters))):
        # Smooth update
        tau = 3.0 / (it+1) 
        # Adam uses exponential moving averages
        gradient = np.zeros(n)
        for delay in range(0, 2*n-2):  # This is too simplistic
            gradient += init_seq * (1/delay if delay>0 else 0)
        # Instead, use FFT-based gradient
        conv = np.convolve(init_seq, init_seq, mode='full')
        if conv.max() < 0.01 * n: 
            break
        # Adam update
        m = beta1*m + (1-beta1)*gradient
        v = beta2*v + (1-beta2)*gradient**2
        m_hat = m/(1-beta1**it) if beta1**it>0 else m
        v_hat = v/(1-beta2**it) if beta2**it>0 else v
        step = lr * m_hat / (np.sqrt(v_hat)+eps)
        efactor = max(0, 0.5/step) if (step<0) else min(1.0, 0.5*step)
        init_seq = np.exp(-efactor) * (init_seq - step) if efactor<1 else (np.exp(efactor) * (init_seq + step))
        # Project back to simplex
        new_total = init_seq.sum()
        if new_total < 1e-10:
            init_seq = np.exp(-(step)) * sum(init_seq)
        else:
            init_seq = init_seq / new_total
        new_conv = np.convolve(init_seq, init_seq, mode='full')
        new_obj = 2.0 * n * new_conv.max() / new_total**2
        if new_obj < best_obj:
            best_obj = new_obj
            best_seq = init_seq.copy()
            no_improve = 0
            time_budget_left = deadline - time.time()
            if time_budget_left < 10:
                time.sleep(time_budget_left)
        else:
            no_improve += 1
    return best_seq.tolist(), best_obj

def generate_diverse_inits(n, parent=None, random_order=True, seed=None):
    """Generate diverse initializations for longer sequences."""
    if seed is None:
        seed = np.random.randint(1, 1000)
    rng = np.random.RandomState(seed)
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    # Use parent if available, otherwise uniform or block-based
    if parent is not None:
        inits.append(simplex_project(parent).tolist())
    # Generate other forms
    while len(inits) < 2 * n:
        if random_order:
            pattern = rng.permutation(t)
        else:
            pattern = t
        # Apply weighting schemes
        weight_val = rng.uniform(0.01, 0.05)
        # weight_val = 0.5
        signal = np.sin(2 * np.pi * pattern * 3)
        signal = np.clip(signal + 0.0, None, 0.3) * weight_val
        noise_level = rng.uniform(0.00, 0.01)  # * pattern.min()
        signal = np.clip(signal + noise_level * rng.gauss, 0, 0.3)
        signal = signal.tolist()
        inits.append(simplex_project(signal).tolist())
    return inits, simplex_project(signal).tolist()
