    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a = a / s
    c = compute_conv(a)
    return 2.0 * n * np.max(c)

    n = len(v)
    # Scale to sum = s
    u = v / s * n  # work with sum = n for simplicity? No, let's do it properly.

    # Algorithm: find tau such that sum(max(v_i - tau, 0)) = s.
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.find_last((u - (cv - s) / np.arange(1, n + 1)) > 0)
    if rho is None:
        return np.full(n, s / n)
    tau = (cv[rho] - s) / (rho + 1)
    return np.maximum(v - tau, 0)

def subgradient(a, c):

    n = len(a)
    max_val = np.max(c)
    # Find all k achieving the max (within tolerance)
    tol = 1e-12
    peak_indices = np.where(c >= max_val - tol)[0]

    for k in peak_indices:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += 2.0 * a[j]
    g /= len(peak_indices)
    return g

Combined optimization: LSE-smoothed L-BFGS + LP alternating + projected subgradient.
Uses softmax parameterization for LSE, and direct simplex projection for subgradient.
"""

from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time

# --- Core functions ---

def softmax_fn(x):
    x_max = np.max(x)
    e = np.exp(x - x_max)
    return e / np.sum(e)

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    c = compute_conv(a)
    return 2.0 * n * float(np.max(c))

def compute_b_fft(p, a):
    n = len(a)
    m = len(p)
    size = 1
    while size < m + n - 1:
        size <<= 1
    P = rfft(p, size)
    A = rfft(a, size)
    corr = irfft(np.conj(P) * A, size)
    return corr[:n]

def project_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.find_last((u - (cv - s) / np.arange(1, n + 1)) > 0)
    if rho is None:
        return np.full(n, s / n)
    tau = (cv[rho] - s) / (rho + 1)
    return np.maximum(v - tau, 0)

# --- LSE optimizer ---

    def __init__(self, eps):
        self.eps = eps
    def __call__(self, x):
        a = softmax_fn(x)
        c = compute_conv(a)
        c_scaled = c / self.eps
        c_max = np.max(c_scaled)
        exp_c = np.exp(c_scaled - c_max)
        log_sum_exp = c_max + np.log(np.sum(exp_c))
        f = self.eps * log_sum_exp
        p = exp_c / np.sum(exp_c)
        b = compute_b_fft(p, a)
        pc = float(np.dot(p, c))
        grad = 2.0 * a * (b - pc)
        return f, grad

# ─── Core utilities ───────────────────────────────────────────────

def softmax_fn(x):
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    c = compute_conv(a)
    return 2.0 * n * float(np.max(c))

def compute_b_fft(p, a):
    """b[i] = sum_k p[k] * a[k-i] for i=0..n-1 via FFT cross-correlation."""
    n = len(a)
    m = len(p)
    size = 1
    while size < m + n - 1:
        size <<= 1
    P = rfft(p, size)
    A = rfft(a, size)
    corr = irfft(np.conj(P) * A, size)
    return corr[:n]

def project_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.find_last((u - (cv - s) / np.arange(1, n + 1)) > 0)
    if rho is None:
        return np.full(n, s / n)
    tau = (cv[rho] - s) / (rho + 1)
    return np.maximum(v - tau, 0)
