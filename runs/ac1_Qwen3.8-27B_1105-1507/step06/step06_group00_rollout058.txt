def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a, M_target=None):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0
    OR: max sum(g) s.t. conv(a,g) <= M_target, g>=0"""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    if M_target is None:
        # Min-peak LP
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = np.sum(g_val)
                    if s > 1e-10:

            except:
                continue
    else:
        # Max-sum LP
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M_target]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
            try:

"""
Combined LP-alternating and smooth-gradient optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Uses CLARABEL for LP solves, L-BFGS for smooth surrogate, and spectral
parameterization for diverse starting points.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a,g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    # More efficient construction
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        A[j:j + end, j] = a[:end]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating optimization with golden section line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Golden section line search parameters
    inv_phi = (np.sqrt(5) - 1) / 2  # ~0.618
    
    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break
        
        g, M_lp = solve_lp_minpeak(a)
        if g is None:
            break
        
        # Check convergence: distance between a and g
        dist = np.linalg.norm(a - g) / np.linalg.norm(a)
        if dist < 1e-6:
            break
        
        # Line search: find best t in (0, 1]
        # Use golden section search on f(t) = obj((1-t)*a + t*g)
        lo, hi = 0.0, 1.0
        for _ in range(20):
            if time.monotonic() - start > time_budget:
                break
            t1 = lo + inv_phi * (hi - lo)
            t2 = hi - inv_phi * (hi - lo)
            
            a1 = (1 - t1) * a + t1 * g
            a1 = a1 / np.sum(a1)
            f1 = compute_obj(a1)
            
            a2 = (1 - t2) * a + t2 * g
            a2 = a2 / np.sum(a2)
            f2 = compute_obj(a2)
            
            if f1 < f2:
                hi = t2
            else:
                lo = t1
        
        t_best = (lo + hi) / 2
        a = (1 - t_best) * a + t_best * g
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            break
        a = a / s
        obj = compute_obj(a)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose and it % 10 == 0:
                print(f"  LP-alt it{it}: obj={obj:.8f}, dist={dist:.6f}, M_lp={M_lp:.6f}")
    
    return best_a

def smooth_gradient_obj(a, c=None):
    """
    Smooth surrogate: sum_k ((a*a)_k - c)^2 where c = 1/(2n-1) (target flat level).
    Returns (value, gradient).
    """
    n = len(a)
    a = a / np.sum(a)  # normalize
    conv = compute_conv(a)
    if c is None:
        c = 1.0 / (2 * n - 1)
    residual = conv - c
    value = np.sum(residual ** 2)
    
    # Gradient: d/da_i sum_k residual_k^2 = 2 * sum_k residual_k * a_{k-i}
    # = 2 * (residual cross-correlate a)_i
    # residual has length 2n-1, a has length n
    # (residual star a)_i = sum_k residual_k * a_{k-i}
    # This is the convolution of residual (reversed) with a, evaluated at appropriate lag
    # More precisely: sum_{k} residual[k] * a[k-i] where 0<=k-i<n
    # = sum_{j=0}^{n-1} residual[j+i] * a[j]  (substituting k=j+i)
    # = cross-correlation
    
    # Use FFT for efficiency
    size = 1
    while size < 3 * n:
        size <<= 1
    # cross-correlation of residual (length 2n-1) with a (length n)
    # (residual star a)_i = sum_k residual[k] * a[k-i]
    # = sum_j a[j] * residual[j+i]
    # This is the cross-correlation, which can be computed as:
    # IFFT(FFT(residual) * conj(FFT(a)))
    # But we need to be careful about the indexing for linear cross-correlation.
    
    # Actually, let's just do it directly for clarity (n is at most ~1200)
    grad = np.zeros(n)
    for i in range(n):
        # sum_k residual[k] * a[k-i] where 0<=k<2n-1 and 0<=k-i<n
        # k-i >= 0 => k >= i
        # k-i < n => k < i+n
        # k < 2n-1
        k_min = i
        k_max = min(i + n, 2 * n - 1)
        if k_min < k_max:
            grad[i] = 2.0 * np.dot(residual[k_min:k_max], a[0:k_max - k_min])
    
    return value, grad

def smooth_optimize(a, time_budget=100.0, verbose=False):
    """Optimize the smooth surrogate using L-BFGS."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()
    
    def objective_and_grad(x):
        if time.monotonic() - start > time_budget:
            return 0, np.zeros(n)
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 1e-10:
            return 0, np.zeros(n)
        x = x / s
        val, grad = smooth_gradient_obj(x)
        return val, grad
    
    # Use L-BFGS-B
    try:
        result = minimize(objective_and_grad, a, method='L-BFGS-B',
                         options={'maxiter': 200, 'gtol': 1e-8, 'maxls': 50})
        if result.x is not None:
            a_new = np.maximum(result.x, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                true_obj = compute_obj(a_new)
                if true_obj < best_true_obj:
                    best_a = a_new
                    best_true_obj = true_obj
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    return best_a

def spectral_init(n, rng, n_harmonics=10):
    """Generate a spectral initialization: sum of cosines."""
    x = np.arange(n) / n
    a = np.ones(n)
    for k in range(1, n_harmonics + 1):
        freq = rng.integers(1, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-0.3, 0.3)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
Combined LP-alternating, smooth-gradient, and spectral optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses CLARABEL solver, L-BFGS polishing,
diverse initializations, and multi-n exploration to beat 1.5027 target.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternating(a, time_budget=100.0, verbose=False):
    """
    LP alternating optimization with golden section line search and
    convergence detection via a-g distance.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    inv_phi = (np.sqrt(5) - 1) / 2.0
    stall_count = 0
    last_improvement = best_obj

    max_iters = 2000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g, M_lp = solve_lp_minpeak(a)
        if g is None:

        dist = float(np.linalg.norm(a - g))
        if verbose and it % 20 == 0:
            print(f"    LP it{it}: obj={compute_obj(a):.8f}, dist={dist:.6f}, M_lp={M_lp:.6f}")

        if dist < 1e-7:

        # Golden section line search
        lo, hi = 0.0, 1.0
        for _ in range(25):
            if time.monotonic() - start > time_budget:
                break
            t1 = lo + inv_phi * (hi - lo)
            t2 = hi - inv_phi * (hi - lo)

            a1 = (1.0 - t1) * a + t1 * g
            a1 = np.maximum(a1, 0)
            s1 = float(np.sum(a1))
            f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf

            a2 = (1.0 - t2) * a + t2 * g
            a2 = np.maximum(a2, 0)

            else:
                lo = t1

        a_new = (1.0 - t_best) * a + t_best * g

def smooth_gradient_and_value(a):
    """
    Compute smooth surrogate F(a) = sum_k ((a*a)_k - c)^2 and its gradient.
    c = 1/(2n-1) is the target flat level.
    Returns (value, gradient w.r.t. normalized a).
    """
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    c = 1.0 / (2.0 * n - 1)
    residual = conv - c
    value = float(np.sum(residual ** 2))

    # grad_hat[i] = 4 * sum_j=0^{min(n-1, 2n-2-i)} residual[i+j] * a[j]
    grad_hat = np.zeros(n)
    for i in range(n):
        k_max = min(n, 2 * n - 1 - i)
        if k_max > 0:
            grad_hat[i] = 4.0 * float(np.dot(residual[i:i + k_max], a[:k_max]))

    # Convert to gradient w.r.t. raw a (accounting for normalization)
    # grad_a_i = (1/S) * (grad_hat_i - <grad_hat, a>)
    S = float(np.sum(a))
    weighted_sum = float(np.dot(grad_hat, a))
    grad_a = (grad_hat - weighted_sum) / S

            return 0.0, np.zeros(n)
        x = np.maximum(x, 0)
        s_x = float(np.sum(x))
        if s_x < 1e-10:
            return 0.0, np.zeros(n)
        x_n = x / s_x
        val, grad = smooth_gradient_and_value(x_n)
        return val, grad

    try:
        result = minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': 300, 'gtol': 1e-9, 'maxls': 50})
        if result.x is not None:
            a_new = np.maximum(result.x, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                true_obj = compute_obj(a_new)
                if verbose:
                    print(f"    L-BFGS: surrogate={result.fun:.8f}, true_obj={true_obj:.8f}")
                if true_obj < best_true_obj:
                    best_a = a_new.copy()
                    best_true_obj = true_obj
    except Exception as e:
        if verbose:
            print(f"    L-BFGS error: {e}")

    return best_a


def spectral_init(n, rng, n_harmonics=8):
    """Generate a diverse spectral initialization."""
    x = np.arange(n) / n
    a = np.ones(n)
    for k in range(n_harmonics):
        freq = int(rng.integers(1, max(2, n // 3)))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(-0.4, 0.4)) / (k + 1)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def peak_targeted_refine(a, time_budget=50.0):
    """
    Targeted refinement: identify top peaks in convolution and move mass
    from contributing positions to non-contributing ones.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Find top 3 peaks
        top_indices = np.argsort(conv)[::-1][:3]
        improved = False

        for k_star in top_indices:
            if time.monotonic() - start > time_budget:
                break

            # Gradient of conv[k_star] w.r.t. a[i] is 2*a[k_star - i]
            # We want to reduce conv[k_star], so decrease a[i] where a[k_star-i] is large
            # and increase a[j] where a[k_star-j] is small
            contrib = np.zeros(n)
            for i in range(n):
                j = int(k_star) - i
                if 0 <= j < n:
                    contrib[i] = 2.0 * a[j]

            # Find positions with high contribution (decrease) and low (increase)
            high_idx = np.argsort(contrib)[::-1][:5]
            low_idx = np.argsort(contrib)[:5]

            for hi in high_idx:
                for lo in low_idx:
                    if hi == lo:
                        continue
                    if a[hi] < 1e-8:
                        continue
                    for frac in [0.01, 0.02, 0.05]:
                        delta = frac * a[hi]
                        a[hi] -= delta
                        a[lo] += delta
                        s_check = float(np.sum(a))
                        if s_check < 1e-10:
                            a[hi] += delta
                            a[lo] -= delta
                            break
                        a_n = a / s_check
                        obj_try = compute_obj(a_n)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_n.copy()
                            a = a_n
                            improved = True
                            break
                        else:
                            a[hi] += delta
                            a[lo] -= delta
                            break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, smooth gradient,
    peak-targeted refinement, spectral initialization, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent_arr = parent_arr / s
                best_seq = parent_arr.copy()
                best_score = compute_obj(parent_arr)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Intensive LP Alternating ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP Alternating (n={n_best})")
    print(f"{'='*60}")

    a = best_seq.copy()
    total_phase1 = min(500.0, (deadline - start_time) * 0.35)
    n_rounds = 4
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = total_phase1 / (n_rounds - rnd)
        a = lp_alternating(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            n_best = len(best_seq)
            print(f"  Round {rnd}: obj={best_score:.8f} ***")
        else:
            print(f"  Round {rnd}: obj={obj:.8f} (no improvement)")

    # === Phase 2: Smooth Gradient Polishing ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Smooth Gradient (n={n_best})")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    if remaining > 50:
        a_sm = smooth_optimize(best_seq, time_budget=min(150, remaining * 0.2), verbose=True)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Smooth: obj={best_score:.8f} ***")
        else:
            print(f"  Smooth: obj={obj_sm:.8f}")

    # LP polish after smooth
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_pol = lp_alternating(best_seq, time_budget=min(100, remaining * 0.15), verbose=True)
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            n_best = len(best_seq)
            print(f"  Post-smooth LP: obj={best_score:.8f} ***")

    # === Phase 3: Peak Targeted Refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Peak Targeted Refinement (n={n_best})")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_pt = peak_targeted_refine(best_seq, time_budget=min(60, remaining * 0.1))
        obj_pt = compute_obj(a_pt)
        if obj_pt < best_score:
            best_score = obj_pt
            best_seq = a_pt.copy()
            print(f"  Peak: obj={best_score:.8f} ***")
        else:
            print(f"  Peak: obj={obj_pt:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n Exploration")
    print(f"{'='*60}")

    n_options = sorted(set([
        max(300, n_best - 100),
        max(400, n_best - 50),
        n_best,
        min(1400, n_best + 50),
        min(1500, n_best + 100),
        500, 600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)

        alloc = min(70, remaining * 0.15)
        a_opt = lp_alternating(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  ** n={n_try}: obj={best_score:.8f} ***")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Diverse Restarts")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    n_restarts = 12

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        init_type = r % 6
        if init_type == 0:
            a_r = spectral_init(n_best, rng, n_harmonics=int(rng.integers(3, 12)))
        elif init_type == 1:
            a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            n_r = max(300, min(1500, n_best + int(rng.integers(-150, 150))))
            a_r = resample_seq(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.02, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 3:
            x = np.linspace(0, np.pi, n_best)
            a_r = 1.0 + 0.4 * np.cos(x) + 0.15 * np.cos(2 * x) - 0.1 * np.cos(3 * x)
            a_r = a_r + rng.normal(0, 0.01, n_best)
            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / np.sum(a_r)
        elif init_type == 4:
            # Superposition of Gaussians
            n_r = n_best
            a_r = np.zeros(n_r)
            for _ in range(int(rng.integers(2, 5))):
                center = float(rng.uniform(0.1, 0.9))
                width = float(rng.uniform(0.05, 0.2))
                height = float(rng.uniform(0.5, 1.5))
                a_r += height * np.exp(-((np.arange(n_r) / n_r - center) ** 2) / (2 * width ** 2))
            a_r = a_r + 0.1
            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / np.sum(a_r)
        else:
            # Linear ramp + noise
            a_r = np.linspace(1.0, 1.0, n_best)
            a_r = a_r + 0.3 * (1.0 - 2.0 * np.abs(np.arange(n_best) / n_best - 0.5))
            a_r = a_r + rng.normal(0, 0.02, n_best)
            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / np.sum(a_r)

        alloc = min(50, remaining * 0.12)
        a_opt = lp_alternating(a_r, time_budget=alloc)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (type={init_type}, n={n_best}): obj={best_score:.8f} ***")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_r:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Intensive Polish (n={n_best})")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    polish_rounds = 6
    for i in range(polish_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.35, 80)
        a_final = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Polish {i}: obj={best_score:.8f} ***")
        else:
            print(f"  Polish {i}: obj={obj_f:.8f}")
            if obj_f - best_score < 1e-10:
                break

    # Smooth + peak on final
    remaining = deadline - time.monotonic()
    if remaining > 25:
        a_s = smooth_optimize(best_seq, time_budget=min(40, remaining * 0.4), verbose=True)
        obj_s = compute_obj(a_s)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_s.copy()
            print(f"  Final smooth: obj={best_score:.8f} ***")
        else:
            print(f"  Final smooth: obj={obj_s:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_p = peak_targeted_refine(best_seq, time_budget=min(30, remaining * 0.5))
        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Final peak: obj={best_score:.8f} ***")
        else:
            print(f"  Final peak: obj={obj_p:.8f}")

    # One final LP round
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.8, 40))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP: obj={best_score:.8f} ***")
        else:
            print(f"  Final LP: obj={obj_last:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
