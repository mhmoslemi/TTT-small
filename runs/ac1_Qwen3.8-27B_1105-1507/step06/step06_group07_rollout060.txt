def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:

    return 2.0 * n * max_b / (sum_a * sum_a)

def gs_spectral_flatten(a, n_iters=100, alpha=0.5, taper=True):
    """
    Gerchberg-Saxton style spectral flattening.
    Compresses the magnitude spectrum to make it flatter.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    # Zero-pad to next power of 2 >= 2n
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(n_iters):
        A = fft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress magnitude
        mag_c = mag ** alpha
        # Normalize to preserve energy
        mag_c = mag_c / np.sqrt(np.mean(mag_c**2)) * np.sqrt(np.mean(mag**2))
        
        # Reconstruct
        A_new = mag_c * np.exp(1j * phase)
        a_new = np.real(ifft(A_new)[:n])
        
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0)
        
        # Normalize
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        a = a_new
    
    return a

    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)

        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_centered = x - np.max(x)
        a_cur = np.exp(x_centered)
        a_cur = a_cur / np.sum(a_cur)
        
        # Compute smoothed gradient
        conv = compute_conv(a_cur)

        size = 1
        while size < 2 * len(conv):
            size <<= 1

        A_ft = rfft(a_cur, size)
        cc = irfft(W * np.conj(A_ft), size)
        grad_ls = 2.0 * cc[:n]
        
        grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
        grad_x = grad_obj * a_cur
        
        # Adam update
        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            x -= np.mean(x)

            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a

def active_set_optimize(a, time_budget=30.0, max_iters=100):
    """
    Optimize by focusing on active (peak) convolution values.
    Uses a quadratic penalty on the peaks.
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_b = np.max(conv)
        sum_a = np.sum(a)
        
        # Find active set: peaks within 5% of max
        threshold = max_b * 0.95
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Build the subgradient from active peaks
        # d(conv[k])/d(a[i]) = a[k-i] if valid
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j] / len(active)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        grad /= grad_norm
        
        # Line search: move in direction -grad while maintaining non-negativity
        obj_prev = compute_obj(a)
        found_improvement = False
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                found_improvement = True
                break
        
        if not found_improvement and it > 20:
            break
    
    return best_a

def differential_evolution_search(a_init, n_evals_max=500, time_budget=120.0, 
                                   population_size=20, F=0.8, CR=0.5, seed=42):
    """
    Simple differential evolution for the normalized sequence.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    
    # Initialize population
    pop = np.zeros((population_size, n))
    for i in range(population_size):
        if i == 0:
            pop[i] = a_init
        else:
            pop[i] = a_init * np.exp(rng.normal(0, 0.1, n))
            pop[i] = np.maximum(pop[i], 0)
            s = np.sum(pop[i])
            if s > 1e-10:
                pop[i] /= s
            else:
                pop[i] = a_init.copy()
    
    # Evaluate
    fitness = np.array([compute_obj(pop[i]) for i in range(population_size)])
    
    best_fitness = np.min(fitness)
    best_a = pop[np.argmin(fitness)].copy()
    
    n_evals = population_size
    gen = 0
    
    while n_evals < n_evals_max:
        if time.monotonic() - start > time_budget:
            break
        gen += 1
        
        # Find current best in population
        best_idx = np.argmin(fitness)
        
        for i in range(population_size):
            if n_evals >= n_evals_max:
                break
            if time.monotonic() - start > time_budget:
                break
            
            # Select three distinct random individuals
            r1, r2, r3 = rng.choice(population_size, 3, replace=False)
            while r1 == i or r2 == i or r3 == i:
                r1, r2, r3 = rng.choice(population_size, 3, replace=False)
            
            # Mutation
            v = pop[r1] + F * (pop[r2] - pop[r3])
            v = np.maximum(v, 0)
            s_v = np.sum(v)
            if s_v < 1e-10:
                continue
            v = v / s_v
            
            # Crossover
            trial = v.copy()
            j_rand = rng.integers(0, n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v[j]
                else:
                    trial[j] = pop[i][j]
            
            # Evaluate
            n_evals += 1
            f_trial = compute_obj(trial)
            
            # Selection
            if f_trial <= fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_fitness:
                    best_fitness = f_trial
                    best_a = trial.copy()
        
        # Print progress
        if gen % 10 == 0:
            print(f"    DE gen {gen}: best={best_fitness:.8f}, evals={n_evals}")
    
    return best_a

"""
Multi-strategy search combining spectral flattening (GS-like), Adam log-space
optimization, active-set peak reduction, differential evolution, and diverse
initializations to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def gs_spectral_flatten(a, n_iters=150, alpha=0.5):
    """
    Gerchberg-Saxton style spectral flattening with magnitude compression.
    Aims to make |FFT(a)| as flat as possible, which flattens the circular
    autocorrelation. Works best when boundaries are small.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(n_iters):
        A = fft(a, size)
        mag = np.abs(A).copy()
        phase = A / (mag + 1e-30)  # keep phase as complex unit vector

        # Compress magnitude to flatten spectrum
        mag_safe = np.maximum(mag, 1e-15)
        mag_c = mag_safe ** alpha
        # Preserve total energy
        energy_orig = np.mean(mag_safe ** 2)
        energy_new = np.mean(mag_c ** 2)
        if energy_new > 1e-30:
            mag_c = mag_c * np.sqrt(energy_orig / energy_new)

        A_new = mag_c * phase
        a_new = np.real(ifft(A_new)[:n])

        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)

        # Normalize
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a = a_new / s_new

    return a


def smooth_obj_grad(a, beta):
    """
    Compute smoothed objective (logsumexp of beta*conv) and its gradient w.r.t. a.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A_ft = rfft(a, size)
    cc = irfft(W * np.conj(A_ft), size)
    grad_ls = 2.0 * cc[:n]

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def adam_log_opt(a, max_iters=5000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        x_centered = x - np.max(x)
        a_cur = np.exp(x_centered)
        a_cur = a_cur / np.sum(a_cur)

        obj_sm, grad_a = smooth_obj_grad(a_cur, beta)
        grad_x = grad_a * a_cur

        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 200 == 0:
            x -= np.mean(x)

        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

    return best_a


def peak_reduction_step(a, n_steps=200, time_budget=30.0):
    """
    Directly target reduction of the maximum convolution value.
    Identifies which (i,j) pairs contribute to the peak and redistributes mass.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_b = np.max(conv)
        sum_a = np.sum(a)
        obj = 2.0 * n * max_b / (sum_a * sum_a)

        # Find the peak location
        peak_k = int(np.argmax(conv))

        # Find the top contributors to this peak
        # conv[peak_k] = sum_i a[i] * a[peak_k - i] where valid
        contribs = []
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                contribs.append((i, j, a[i] * a[j]))
        contribs.sort(key=lambda x: -x[2])

        # Try to reduce the top contributor by moving mass
        improved = False
        for i, j, val in contribs[:5]:
            if val < max_b * 0.1:
                break
            # Move mass from the larger of a[i], a[j] to a less critical position
            src = i if a[i] >= a[j] else j
            dst = j if a[i] >= a[i] else i  # the other one in the pair

            # Find a destination that doesn't contribute to the peak
            # Try moving to a position far from the peak
            candidates = []
            for idx in range(n):
                # This position contributes to peak_k if peak_k - idx is valid
                if not (0 <= peak_k - idx < n):
                    candidates.append((idx, 0))  # no contribution to peak
                else:
                    other = peak_k - idx
                    candidates.append((idx, a[idx] * a[other]))
            candidates.sort(key=lambda x: x[1])

            for dest_idx, dest_contrib in candidates[:3]:
                if dest_idx == src:
                    continue
                for frac in [0.05, 0.02, 0.01]:
                    delta = frac * a[src]
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[dest_idx] += delta
                    if a_try[src] < -1e-10 or a_try[dest_idx] < -1e-10:
                        continue
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
        if not improved and step > 50:
            break

    return best_a


def differential_evolution_search(a_init, n_evals_max=400, time_budget=120.0,
                                   population_size=15, F=0.7, CR=0.6, seed=42):
    """Simple differential evolution for normalized non-negative sequences."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    pop = np.zeros((population_size, n))
    for i in range(population_size):
        if i == 0:
            pop[i] = a_init.copy()
        else:
            noise_scale = 0.05 + 0.03 * (i // 5)
            pop[i] = a_init * np.exp(rng.normal(0, noise_scale, n))
            pop[i] = np.maximum(pop[i], 0)
            s = np.sum(pop[i])
            if s > 1e-10:
                pop[i] /= s
            else:
                pop[i] = a_init.copy()

    fitness = np.array([compute_obj(pop[i]) for i in range(population_size)])
    best_fitness = float(np.min(fitness))
    best_a = pop[np.argmin(fitness)].copy()
    n_evals = population_size
    gen = 0

    while n_evals < n_evals_max:
        if time.monotonic() - start > time_budget:
            break
        gen += 1

        for i in range(population_size):
            if n_evals >= n_evals_max or time.monotonic() - start > time_budget:
                break

            idxs = list(range(population_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)

            v = pop[r1].astype(np.float64) + F * (pop[r2].astype(np.float64) - pop[r3].astype(np.float64))
            v = np.maximum(v, 0)
            s_v = np.sum(v)
            if s_v < 1e-10:
                continue
            v = v / s_v

            trial = pop[i].copy()
            j_rand = int(rng.integers(0, n))
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v[j]

            n_evals += 1
            f_trial = compute_obj(trial)

            if f_trial <= fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_fitness:
                    best_fitness = f_trial
                    best_a = trial.copy()

        if gen % 5 == 0:
            print(f"    DE gen {gen}: best={best_fitness:.8f}, evals={n_evals}")

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_refine(a, time_budget=30.0):
    """LP-based refinement: find direction reducing peak via LP, then blend."""
    n = len(a)
    if n > 1000:
        return a
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s

    conv = compute_conv(a_norm)
    M = float(np.max(conv))

    A = build_toeplitz(a_norm)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': min(time_budget * 0.5, 20)})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s_g = float(np.sum(g))
            if s_g > 1e-10:
                g = g / s_g
                best_a = a.copy()
                best_obj = compute_obj(a)
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                    if time.monotonic() - start > time_budget:
                        break
                    a_try = (1 - t) * a_norm + t * g
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a
    except Exception:
        pass
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):

    Multi-strategy search combining spectral flattening, Adam optimization,
    peak reduction, LP refinement, differential evolution, and diverse
    initializations to minimize the convolution peak objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    if best_seq is None:
        n = 700
        best_seq = np.ones(n) / n
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n}): obj={best_score:.8f}")

    # === Strategy 1: Spectral flattening on parent ===
    print(f"\n{'='*50}")
    print("Strategy 1: Spectral flattening (GS)")
    print(f"{'='*50}")

    if time.monotonic() < deadline - 60:
        for alpha in [0.3, 0.5, 0.7]:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.1, 50.0)

            a_sf = gs_spectral_flatten(best_seq, n_iters=200, alpha=alpha)
            obj_sf = compute_obj(a_sf)

            # Follow up with Adam
            if time.monotonic() < deadline - 30:
                a_sf_opt = adam_log_opt(a_sf, max_iters=10000,
                                        time_budget=min(alloc * 0.4, 25.0),
                                        beta_start=20.0, beta_end=500.0,
                                        lr_start=0.03, lr_decay=0.99998)
                obj_after = compute_obj(a_sf_opt)
                if obj_after < obj_sf:
                    a_sf = a_sf_opt
                    obj_sf = obj_after

            if obj_sf < best_score:
                best_score = obj_sf
                best_seq = a_sf.copy()
                print(f"  GS alpha={alpha}: {best_score:.8f} *** NEW BEST")
            else:
                print(f"  GS alpha={alpha}: {obj_sf:.8f}")

    # === Strategy 2: Intensive Adam with temperature annealing ===
    print(f"\n{'='*50}")
    print("Strategy 2: Adam log-space (various temperatures)")
    print(f"{'='*50}")

    if time.monotonic() < deadline - 60:
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        adam_configs = [
            (5.0, 50.0, 0.05, 0.99997),
            (10.0, 100.0, 0.04, 0.99997),
            (20.0, 200.0, 0.03, 0.99998),
            (50.0, 500.0, 0.02, 0.99998),
            (100.0, 1000.0, 0.015, 0.99999),
            (200.0, 2000.0, 0.01, 0.99999),
            (500.0, 5000.0, 0.005, 0.99999),
            (1000.0, 10000.0, 0.003, 0.99999),
        ]

        for b_s, b_e, lr, lr_d in adam_configs:
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.1, 80.0)

            a_new = adam_log_opt(a, max_iters=30000, time_budget=alloc,
                                 beta_start=b_s, beta_end=b_e,
                                 lr_start=lr, lr_decay=lr_d)
            obj_new = compute_obj(a_new)

            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  Adam beta={b_s:.0f}/{b_e:.0f}: {best_score:.8f} *** NEW BEST")
            else:
                print(f"  Adam beta={b_s:.0f}/{b_e:.0f}: {obj_new:.8f}")

    # === Strategy 3: Peak reduction ===
    print(f"\n{'='*50}")
    print("Strategy 3: Peak reduction")
    print(f"{'='*50}")

    if time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.08, 40.0)
        a_pr = peak_reduction_step(best_seq, n_steps=500, time_budget=alloc)
        obj_pr = compute_obj(a_pr)
        if obj_pr < best_score:
            best_score = obj_pr
            best_seq = a_pr.copy()
            print(f"  Peak reduction: {best_score:.8f} *** NEW BEST")
        else:
            print(f"  Peak reduction: {obj_pr:.8f}")

        # Adam after peak reduction
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            a_post = adam_log_opt(a_pr, max_iters=10000,
                                  time_budget=min(remaining * 0.05, 20.0),
                                  beta_start=100.0, beta_end=2000.0,
                                  lr_start=0.01, lr_decay=0.99999)
            obj_post = compute_obj(a_post)
            if obj_post < best_score:
                best_score = obj_post
                best_seq = a_post.copy()
                print(f"  Post-peak Adam: {best_score:.8f} *** NEW BEST")

    # === Strategy 4: LP refinement ===
    print(f"\n{'='*50}")
    print("Strategy 4: LP refinement")
    print(f"{'='*50}")

    if len(best_seq) <= 900 and time.monotonic() < deadline - 40:

                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.1, 40.0)

            a_lp = lp_refine(best_seq, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP {r_idx}: {best_score:.8f} *** NEW BEST")
            else:
                print(f"  LP {r_idx}: {obj_lp:.8f}")

            # Adam after LP
            if time.monotonic() < deadline - 20:
                remaining = deadline - time.monotonic()
                a_after = adam_log_opt(a_lp, max_iters=10000,
                                       time_budget=min(remaining * 0.05, 15.0),
                                       beta_start=50.0, beta_end=1000.0,
                                       lr_start=0.02, lr_decay=0.99999)
                obj_after = compute_obj(a_after)
                if obj_after < best_score:
                    best_score = obj_after
                    best_seq = a_after.copy()
                    print(f"  LP+Adam {r_idx}: {best_score:.8f} *** NEW BEST")

    # === Strategy 5: Differential Evolution ===
    print(f"\n{'='*50}")
    print("Strategy 5: Differential Evolution")
    print(f"{'='*50}")

    if time.monotonic() < deadline - 50:
        n = len(best_seq)
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 150.0)

        a_de = differential_evolution_search(best_seq, n_evals_max=250,
                                              time_budget=alloc, population_size=12,
                                              F=0.6, CR=0.5, seed=seed)
        obj_de = compute_obj(a_de)
        if obj_de < best_score:
            best_score = obj_de
            best_seq = a_de.copy()
            print(f"  DE: {best_score:.8f} *** NEW BEST")
        else:
            print(f"  DE: {obj_de:.8f}")

        # Adam polish after DE
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            a_de_polish = adam_log_opt(a_de, max_iters=10000,
                                       time_budget=min(remaining * 0.05, 15.0),
                                       beta_start=20.0, beta_end=500.0,
                                       lr_start=0.02, lr_decay=0.99998)
            obj_de_polish = compute_obj(a_de_polish)
            if obj_de_polish < best_score:
                best_score = obj_de_polish
                best_seq = a_de_polish.copy()
                print(f"  DE+Adam: {best_score:.8f} *** NEW BEST")

    # === Strategy 6: Multi-n exploration ===
    print(f"\n{'='*50}")
    print("Strategy 6: Multi-n exploration")
    print(f"{'='*50}")

    if best_seq is not None and time.monotonic() < deadline - 50:
        n_best = len(best_seq)
        n_options = [n_best - 100, n_best - 50, n_best + 50, n_best + 100,
                     n_best + 200, n_best + 300]
        n_options = sorted(set([n for n in n_options if 200 <= n <= 2000 and n != n_best]))

        for n_try in n_options:
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_try = resample_sequence(best_seq, n_try)
            s = np.sum(a_try)
            if s < 1e-10:
                continue
            a_try = a_try / s

            alloc = min(remaining * 0.1, 50.0)
            a_opt = adam_log_opt(a_try, max_iters=15000, time_budget=alloc,
                                 beta_start=10.0, beta_end=500.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** NEW BEST")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Strategy 7: Diverse new initializations ===
    print(f"\n{'='*50}")
    print("Strategy 7: Diverse new initializations")
    print(f"{'='*50}")

    if time.monotonic() < deadline - 40:
        n = len(best_seq)
        rng = np.random.default_rng(seed + 200)
        inits = []

        # Hann window
        hann = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
        inits.append(('hann', hann / np.sum(hann)))

        # Blackman
        x = np.arange(n) / (n - 1)
        blm = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        blm = np.maximum(blm, 0)
        inits.append(('blackman', blm / np.sum(blm)))

        # Asymmetric ramp (key: asymmetry reduces center peak)
        ramp = np.linspace(0.01, 1.0, n)
        inits.append(('ramp', ramp / np.sum(ramp)))

        # Bimodal
        c1, c2 = n // 4, 3 * n // 4
        sig = n // 8
        bim = (np.exp(-(np.arange(n) - c1) ** 2 / (2 * sig ** 2)) +
                np.exp(-(np.arange(n) - c2) ** 2 / (2 * sig ** 2)))
        inits.append(('bimodal', bim / np.sum(bim)))

        # Random smooth
        rand_s = np.exp(rng.normal(0, 0.8, n))
        inits.append(('rand', rand_s / np.sum(rand_s)))

        # Power law
        pwr = np.arange(1, n + 1) ** 1.5
        inits.append(('power', pwr / np.sum(pwr)))

        for name, a_init in inits:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # Spectral flatten as preprocessing
            a_sf = gs_spectral_flatten(a_init, n_iters=80, alpha=0.5)

            alloc = min(remaining * 0.07, 35.0)
            a_opt = adam_log_opt(a_sf, max_iters=10000, time_budget=alloc,
                                 beta_start=10.0, beta_end=500.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  {name}: {best_score:.8f} *** NEW BEST")
            else:
                print(f"  {name}: {obj:.8f}")

    # === Strategy 8: Final intensive polish ===
    print(f"\n{'='*50}")
    print("Strategy 8: Final polish")
    print(f"{'='*50}")

    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()

        # Very high beta Adam (near-exact max targeting)
        alloc = min(remaining * 0.25, 50.0)
        a_fin = adam_log_opt(best_seq, max_iters=20000, time_budget=alloc,
                             beta_start=500.0, beta_end=20000.0,
                             lr_start=0.003, lr_decay=0.99999)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final high-beta Adam: {best_score:.8f} *** NEW BEST")

        # Peak reduction
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            a_pr2 = peak_reduction_step(best_seq, n_steps=300,
                                        time_budget=min(remaining * 0.3, 25.0))
            obj_pr2 = compute_obj(a_pr2)
            if obj_pr2 < best_score:
                best_score = obj_pr2
                best_seq = a_pr2.copy()
                print(f"  Final peak reduction: {best_score:.8f} *** NEW BEST")

        # One more Adam pass
        if time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic()
            a_last = adam_log_opt(best_seq, max_iters=10000,
                                  time_budget=min(remaining * 0.5, 15.0),
                                  beta_start=1000.0, beta_end=50000.0,
                                  lr_start=0.002, lr_decay=0.99999)
            obj_last = compute_obj(a_last)
            if obj_last < best_score:
                best_score = obj_last
                best_seq = a_last.copy()
                print(f"  Final Adam: {best_score:.8f} *** NEW BEST")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
