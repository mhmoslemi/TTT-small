FISTA-based optimization with adaptive log-sum-exp smoothing,
combined with LP alternating and spectral projection for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to have sum 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    corr = irfft(Wf * Af, size)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    
    # The full gradient of tau * LSE / s^2 is (2*tau/s^2) * grad - 2*tau*LSE*a/s^3
    # But for simplicity, we work with the "direction" which is grad - mean(grad)
    # (projecting onto sum=const)
    grad_centered = grad - np.mean(grad)
    
    return grad_centered, tau * log_sum_exp


def fista_optimize(a, time_budget=100.0, tau_init=0.002, use_backtrack=True):
    """
    FISTA (Nesterov accelerated gradient) on the smoothed objective.
    Minimizes 2*n*tau*log-sum-exp(conv(a,a)/tau) / (sum(a))^2
    as a smoothed approximation to the original objective.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # FISTA state
    y = a.copy()
    t_k = 1.0
    
    # Adaptive parameters
    tau = tau_init
    step_size = 0.05
    start = time.monotonic()
    
    # Precompute FFT size
    size = 1
    while size < 3 * n:
        size <<= 1
    
    stagnant = 0

        if time.monotonic() - start > time_budget:
            break
            
        # Compute gradient at y
        grad_y, _ = gradient_smooth_max(y, tau)

        # L ≈ 2*n / (sum(y))^2 * max(y) / tau
        s_y = float(np.sum(y))
        L_est = 2.0 * n * float(np.max(y)) / (s_y * s_y * tau) + 1.0
        
        # Step size with backtracking
        alpha = 1.0 / max(L_est, 1.0)
        # Cap step size
        alpha = min(alpha, 0.1)
        
        # Gradient step
        x_new = y - alpha * grad_y
        x_new = normalize(x_new)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = x_new + ((t_k - 1.0) / t_new) * (x_new - y)
        y_new = normalize(y_new)
        
        # Check improvement
        obj_new = compute_obj(x_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = x_new.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                # Reduce step size and adjust tau
                alpha *= 0.5
                tau *= 0.95
                stagnant = 0
                if alpha < 1e-8:
                    break
        
        y = y_new
        t_k = t_new
        
        # Adaptively decrease tau
        if step % 50 == 0:
            tau *= 0.97
        
        # Adaptively increase step size if making progress
        if stagnant == 0 and step % 20 == 0:
            alpha = min(alpha * 1.05, 0.1)
    
    return best_a

"""
FISTA + LP alternating optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive log-sum-exp smoothing, momentum, symmetric constraint,
multi-n exploration, and spectral projection.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def symmetrize(a):
    """Enforce a[j] = a[n-1-j]."""
    n = len(a)
    return 0.5 * (a + a[::-1])


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def gradient_lse(a, tau):
    """Compute gradient of LSE(conv(a,a)/tau) w.r.t. a, centered."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights = exp_vals / w_sum
    
    # grad_j = 2 * sum_k weights[k] * a[k-j]
    size = 1
    while size < 3 * n:
        size <<= 1
    a_flip = a[::-1]
    Wf = rfft(weights, size)
    Af = rfft(a_flip, size)
    corr = irfft(Wf * Af, size)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    
    return grad - np.mean(grad)


def fista_smooth(a, time_budget=60.0, tau_init=0.003, use_sym=True):
    """FISTA with adaptive log-sum-exp smoothing."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    if use_sym:
        a = normalize(symmetrize(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    y = a.copy()
    t_k = 1.0
    tau = tau_init
    lr = 1.0  # learning rate
    start = time.monotonic()
    stagnant = 0
    
    while time.monotonic() - start < time_budget:
        # Compute gradient
        grad = gradient_lse(y, tau)
        
        # Adaptive step size
        s_y = float(np.sum(y))
        # Lipschitz constant estimate: L ≈ 2*n*max(y)/(s_y^2 * tau)
        L = 2.0 * n * float(np.max(y)) / (s_y * s_y * tau)
        step = 1.0 / max(L, 10.0)
        step = min(step, lr)
        
        # Gradient step
        x_new = y - step * grad
        x_new = normalize(x_new)
        if use_sym:
            x_new = normalize(symmetrize(x_new))
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = x_new + ((t_k - 1.0) / t_new) * (x_new - y)
        y_new = normalize(y_new)
        if use_sym:
            y_new = normalize(symmetrize(y_new))
        
        # Evaluate
        obj_new = compute_obj(x_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = x_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 30:
            lr *= 0.8
            tau *= 0.93
            stagnant = 0
            if lr < 1e-10:
                break
        
        # Cool tau slowly
        tau *= 0.999
        
        y = y_new
        t_k = t_new
    
    return best_a


def lp_alternate(a, max_iters=30, time_budget=60.0, use_sym=True):
    """LP alternating method."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    if use_sym:
        a = normalize(symmetrize(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 15:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp_max_sum(a, M, time_limit=max(3, int(remaining * 0.7)))
        if g is None:
            break
        
        if use_sym:
            g = symmetrize(g)
        
        # Try different blend factors
        best_t = 0.0
        best_blend_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = normalize((1.0 - t) * a + t * g)
            if use_sym:
                a_try = normalize(symmetrize(a_try))
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t
        
        a_new = normalize((1.0 - best_t) * a + best_t * g)
        if use_sym:
            a_new = normalize(symmetrize(a_new))
        obj_new = compute_obj(a_new)

            a = a_new

            a = normalize(a * np.exp(rng.normal(0, 0.03, n)))
            if use_sym:
                a = normalize(symmetrize(a))
    
    return best_a


def spectral_project(a, time_budget=30.0, use_sym=True):
    """Spectral flattening via POCS."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    if use_sym:
        a = normalize(symmetrize(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_mean = float(np.mean(P))
        p_max = float(np.max(P))
        
        if p_max < 1e-30:
            break
        
        # Compress spectrum toward flat
        # Target: P_new = p_mean + (P - p_mean) * decay
        decay = 0.7 + 0.2 * (it / 200.0)
        P_target = p_mean + (P - p_mean) * decay
        P_target = np.maximum(P_target, 1e-30)
        
        # Preserve phase
        phase = np.angle(A)
        mag = np.sqrt(P_target)
        A_new = mag * np.exp(1j * phase)
        
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        if use_sym:
            a_new = normalize(symmetrize(a_new))
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.5 * a + 0.5 * a_new)
            if use_sym:
                a = normalize(symmetrize(a))
    
    return best_a


def coordinate_polish(a, time_budget=20.0, use_sym=True):
    """Targeted coordinate descent on peak contributors."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    if use_sym:
        a = normalize(symmetrize(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find top contributors
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)
        
        improved = False
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            
            for _ in range(5):
                j_target = int(rng.integers(0, n))
                if j_target in (j1, j2):
                    continue
                
                a_try = a.copy()
                for delta_frac in [0.02, 0.05, 0.1]:
                    delta = delta_frac * a_try[j1]
                    if a_try[j1] - delta < 0:
                        continue
                    a_try[j1] -= delta
                    if use_sym:
                        a_try[n-1-j1] -= delta
                    a_try[j_target] += delta
                    if use_sym:
                        a_try[n-1-j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
        
        if not improved:
            # Try moving mass from high to low
            for _ in range(3):
                j_from = int(rng.integers(0, n))
                j_to = int(rng.integers(0, n))
                if j_from == j_to:
                    continue
                a_try = a.copy()
                delta = 0.03 * a[j_from]
                if a_try[j_from] - delta < 0:
                    continue
                a_try[j_from] -= delta
                if use_sym:
                    a_try[n-1-j_from] -= delta
                a_try[j_to] += delta
                if use_sym:
                    a_try[n-1-j_to] += delta
                a_try = normalize(a_try)
                if compute_obj(a_try) < best_obj:
                    best_obj = compute_obj(a_try)
                    best_a = a_try.copy()
                    a = a_try

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

"""
Multi-strategy optimization: FISTA with adaptive log-sum-exp smoothing,
LP alternating, spectral projection, and coordinate descent.
Enforces symmetry constraint and explores multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def sym(a):
    """Enforce a[j] = a[n-1-j]."""
    return 0.5 * (a + a[::-1])


def build_toeplitz(a):
    """Toeplitz matrix for conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        ke = min(j + n, m)
        A[j:ke, j] = a[:ke - j]
    return A


def lp_solve(a, M, tl=20):
    """LP: max sum(g) s.t. conv(a,g)<=M, g>=0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                    bounds=[(0, None)]*n, method='highs',
                    options={'presolve': True, 'time_limit': tl})
        if r.status == 0:
            g = np.maximum(r.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def grad_lse(a, tau):
    """Gradient of LSE(conv(a,a)/tau), centered. Uses FFT."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    sh = np.clip((conv - cmax) / tau, -500.0, 0.0)
    ev = np.exp(sh)
    ws = float(np.sum(ev))
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w = ev / ws
    # grad_j = 2 * sum_k w_k * a_{k-j}; compute via conv(w, a_reversed)
    sz = 1
    while sz < 3 * n:
        sz <<= 1
    corr = irfft(rfft(w, sz) * rfft(a[::-1], sz), sz)
    g = 2.0 * corr[n-1:2*n-1]
    return g - np.mean(g)


def fista(a, tb=60.0, tau0=0.002):
    """FISTA on smoothed objective with adaptive tau and restart."""
    n = len(a)
    a = normalize(sym(a.copy()))
    best_a, best_f = a.copy(), compute_obj(a)
    y, tk, tau, lr = a.copy(), 1.0, tau0, 1.0
    st = 0
    t0 = time.monotonic()
    while time.monotonic() - t0 < tb:
        gy = grad_lse(y, tau)
        L = 2.0 * n * float(np.max(y)) / (float(np.sum(y))**2 * tau) + 1.0
        step = min(1.0 / max(L, 5.0), lr, 0.15)
        x = normalize(sym(y - step * gy))
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = normalize(sym(x + ((tk - 1.0) / tn) * (x - y)))
        tk = tn
        f = compute_obj(x)
        if f < best_f - 1e-14:
            best_f, best_a, st = f, x.copy(), 0
        else:
            st += 1
        if st > 40:
            lr *= 0.7
            tau *= 0.9
            st = 0
            if lr < 1e-10:
                break
        tau *= 0.998
    return best_a


def lp_alt(a, mi=25, tb=60.0):
    """LP alternating with line search."""
    n = len(a)
    a = normalize(sym(a.copy()))
    best_a, best_f = a.copy(), compute_obj(a)
    t0 = time.monotonic()
    for it in range(mi):
        if time.monotonic() - t0 > tb - 10:
            break
        rem = tb - (time.monotonic() - t0)
        if rem < 5:
            break
        M = float(np.max(compute_conv(a)))
        g = lp_solve(a, M, max(3, int(rem * 0.6)))
        if g is None:
            break
        g = sym(g)
        # Line search
        bf, bt = best_f, 0.0
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            at = normalize(sym((1-t)*a + t*g))
            ft = compute_obj(at)
            if ft < bf:
                bf, bt = ft, t
        an = normalize(sym((1-bt)*a + bt*g))
        fn = compute_obj(an)
        if fn < best_f - 1e-13:
            best_f, best_a, a = fn, an.copy(), an
        else:
            rng = np.random.default_rng(it*31+7)
            a = normalize(sym(a * np.exp(rng.normal(0, 0.03, n))))
    return best_a


def spectral(a, tb=30.0):
    """Spectral flattening via POCS."""
    n = len(a)
    a = normalize(sym(a.copy()))
    best_a, best_f = a.copy(), compute_obj(a)
    t0 = time.monotonic()
    sz = 1
    while sz < 2*n:
        sz <<= 1
    for it in range(300):
        if time.monotonic() - t0 > tb:
            break
        A = fft(a, sz)
        P = np.abs(A)**2
        pm = float(np.mean(P))
        if pm < 1e-30:
            break
        dec = 0.6 + 0.3*(it/300.0)
        Pt = np.maximum(pm + (P-pm)*dec, 1e-30)
        Ph = np.angle(A)
        an = np.real(ifft(np.sqrt(Pt)*np.exp(1j*Ph), sz))[:n]
        an = normalize(sym(an))
        fn = compute_obj(an)
        if fn < best_f:
            best_f, best_a, a = fn, an.copy(), an
        else:
            a = normalize(sym(0.5*a + 0.5*an))
    return best_a


def coord(a, tb=15.0):
    """Coordinate descent targeting peak."""
    n = len(a)
    a = normalize(sym(a.copy()))
    best_a, best_f = a.copy(), compute_obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    while time.monotonic() - t0 < tb:
        conv = compute_conv(a)
        ks = int(np.argmax(conv))
        contrib = sorted([(a[j]*a[ks-j], j, ks-j) for j in range(n) if 0 <= ks-j < n], reverse=True)
        imp = False
        for _, j1, j2 in contrib[:6]:
            if j1 == j2:
                continue
            for _ in range(4):
                jt = int(rng.integers(0, n))
                if jt in (j1, j2, n-1-j1, n-1-j2):
                    continue
                for df in [0.02, 0.05, 0.08]:
                    d = df * a[j1]
                    if a[j1] - d < 0:
                        continue
                    at = a.copy()
                    at[j1] -= d
                    at[n-1-j1] -= d
                    at[jt] += d
                    at[n-1-jt] += d
                    at = normalize(at)
                    ft = compute_obj(at)
                    if ft < best_f - 1e-14:
                        best_f, best_a, a, imp = ft, at.copy(), at, True
                        break
        if not imp and rng.random() < 0.1:
            a = normalize(sym(a * np.exp(rng.normal(0, 0.02, n))))
    return best_a


def resample(a, nn):
    n = len(a)
    if n == nn:
        return a.copy()
    return np.interp(np.linspace(0, 1, nn), np.linspace(0, 1, n), a)
