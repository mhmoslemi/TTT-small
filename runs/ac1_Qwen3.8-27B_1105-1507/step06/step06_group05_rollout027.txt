import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=30):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def nesterov_smoothed_gradient(a, tau_init=0.0005, max_steps=300, time_budget=60.0, momentum=0.9):
    """
    Nesterov-accelerated gradient descent on the log-sum-exp smoothed objective.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Nesterov variables
    y = a.copy()
    v = np.zeros(n)
    momentum = 0.5  # Start with lower momentum, increase over time
    
    size = 1
    while size < 3 * n:
        size <<= 1
    
    # For gradient computation
    a_flip = a[::-1]
    
    tau = tau_init
    step_size = 0.05
    prev_grad = None
    prev_a = None
    
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov lookahead
        y = a + momentum * v
        
        # Compute smoothed objective gradient at y
        conv_y = compute_conv(y)
        cmax = float(np.max(conv_y))
        
        # Adaptive temperature
        tau = tau_init * (0.995 ** step) + 0.1 * cmax / (n * (step + 1))
        shifted = np.clip((conv_y - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv_y))
            weights = np.zeros(len(conv_y))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: grad[i] = sum_k w[k] * d(conv[k])/d(a[i])
        # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (with bounds)
        # = (a flipped and correlated with weights)
        y_flip = y[::-1]
        Wf = rfft(weights, size)
        Af = rfft(y_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        
        # Center the gradient (remove component along the all-ones direction)
        grad_centered = grad - np.mean(grad)
        
        # BB step size
        if prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad_centered - prev_grad
            sy = float(np.dot(s_vec, y_vec))
            if abs(sy) > 1e-15:
                ss = float(np.dot(s_vec, s_vec))
                yy = float(np.dot(y_vec, y_vec))
                if ss > 0 and yy > 0:
                    alpha_bb = ss / sy
                    if 1e-7 < alpha_bb < 1.0:
                        step_size = 0.5 * (step_size + alpha_bb)
        
        # Update
        v = momentum * v - step_size * grad_centered
        a_new = normalize(a + v)
        
        prev_a = a.copy()
        prev_grad = grad_centered.copy()
        a = a_new
        
        # Increase momentum
        momentum = min(0.95, 0.5 + 0.001 * step)
        
        obj_new = compute_obj(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
            step_size = min(step_size * 1.05, 0.1)
        else:
            step_size *= 0.7
            a = normalize(0.5 * a + 0.5 * best_a)
    
    return best_a

"""
Advanced multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: Nesterov-accelerated smoothed gradient with BB step sizes,
multi-M LP directions with extended line search, LP-dual subgradient,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

        a_try = (1.0 - t) * a + t * g

        a_try = normalize(a_try)
        return compute_obj(a_try)

    # Coarse grid search
    t_grid = np.linspace(0.0, max_t, 15)

        if val < best_val:
            best_val = val
            best_t = t

    lo = max(0.0, best_t - 0.25)
    hi = min(max_t, best_t + 0.25)

    v = np.zeros(n)
    momentum = 0.5
    tau = tau_init
    step_size = 0.05
    prev_grad = None
    prev_a = None
    no_improve_count = 0

    size = 1
    while size < 3 * n:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov lookahead
        y = a + momentum * v

        # Compute smoothed gradient at y
        conv_y = compute_conv(y)
        cmax = float(np.max(conv_y))

        # Adaptive temperature
        tau_eff = max(tau_init * (0.997 ** step), 1e-8) + 0.05 * cmax / (n * (step + 1))
        shifted = np.clip((conv_y - cmax) / tau_eff, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv_y))
            weights = np.zeros(len(conv_y))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient via correlation
        y_flip = y[::-1]
        Wf = rfft(weights, size)
        Af = rfft(y_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        grad_centered = grad - np.mean(grad)

        # BB step size
        if prev_grad is not None and prev_a is not None:
            s_vec = a - prev_a
            y_vec = grad_centered - prev_grad
            sy = float(np.dot(s_vec, y_vec))
            if sy > 1e-12:
                ss = float(np.dot(s_vec, s_vec))
                if ss > 1e-15:
                    alpha_bb = ss / sy
                    if 1e-6 < alpha_bb < 0.5:
                        step_size = 0.6 * step_size + 0.4 * alpha_bb

        # Update
        v = momentum * v - step_size * grad_centered
        a_new = normalize(a + v)
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)

        prev_a = a.copy()
        prev_grad = grad_centered.copy()
        a = a_new
        momentum = min(0.93, 0.5 + 0.002 * step)

        obj_new = compute_obj(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
            no_improve_count = 0
            step_size = min(step_size * 1.02, 0.12)
        else:
            no_improve_count += 1
            if no_improve_count > 30:
                # Re-center toward best
                a = normalize(0.5 * a + 0.5 * best_a)
                v = np.zeros(n)
                no_improve_count = 0
            step_size *= 0.85
            if step_size < 1e-7:
                break

    return best_a


def lp_optimize_v2(a, max_iters=40, time_budget=90.0):
    """LP alternating with multi-M and extended line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        best_t = 0.0
        best_val = best_obj
        best_g = None

        M_factors = [1.0, 0.92, 1.08]
        for M_f in M_factors:
            if time.monotonic() - start > time_budget - 3:
                break
            M_try = M * M_f
            lp_time = max(3, int((time_budget - (time.monotonic() - start)) * 0.2))
            g = solve_lp(a, M_try, time_limit=lp_time)
            if g is None:
                continue

            t, val = extended_line_search(a, g, max_t=1.3)
            if val < best_val:
                best_val = val
                best_t = t
                best_g = g

        if best_g is None:
            break

        a_new = normalize((1.0 - best_t) * a + best_t * best_g)

        if best_val < best_obj - 1e-13:
            best_obj = best_val
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 5:
                rng = np.random.default_rng((it * 6271 + 4242) % (2**31))
                a = normalize(best_a * np.exp(rng.normal(0, 0.04 + 0.02 * stall_count, n)))
                stall_count = 0
            else:
                a = a_new

        if stall_count > 20:
            break

    return best_a


def lp_dual_gradient_step(a, max_iters=30, time_budget=60.0):
    """Use LP dual solution as exact subgradient direction with line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.03

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Get dual weights
        lp_time = max(3, int(remaining * 0.3))
        y = get_lp_dual_weights(a, M, time_limit=lp_time)
        if y is None:
            break

        # Subgradient direction: A^T @ y
        A = build_toeplitz(a)
        grad = A.T @ y
        grad_centered = grad - np.mean(grad)

        # Try descending
        a_try = normalize(a - step_size * grad_centered)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            step_size = min(step_size * 1.1, 0.1)
        else:
            # Try smaller step
            a_try2 = normalize(a - step_size * 0.3 * grad_centered)
            obj_try2 = compute_obj(a_try2)
            if obj_try2 < best_obj:
                best_obj = obj_try2
                best_a = a_try2.copy()
                a = a_try2
            else:
                step_size *= 0.6
                if step_size < 1e-7:
                    break

    return best_a


def spectral_flatten_v2(a, max_iters=80, time_budget=40.0):
    """Spectral flattening with adaptive blending."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        alpha = 0.25 + 0.5 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        blend = 0.2 + 0.6 * (it / max_iters)
        a_try = normalize(blend * a_new + (1 - blend) * a)

        obj_new = compute_obj(a_try)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_try.copy()
            a = a_try
        else:
            a = normalize(0.85 * a + 0.15 * a_try)

    return best_a


def resample_sequence(a, new_n):
    """Resample via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization + Nesterov screening
    Phase 2: Intensive LP with multi-M and extended line search
    Phase 3: Nesterov gradient + LP-dual gradient polishing
    Phase 4: Multi-n exploration
    Phase 5: Final combined polish with adaptive method selection
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 35.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # ============================================================
    # Phase 1: Diverse initialization and screening
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Diverse initialization")
    print(f"{'='*60}")

    rng = np.random.default_rng(seed)
    n_init = n_parent
    candidates = []

    if best_seq is not None:
        candidates.append(("parent", normalize(best_seq.copy())))

    # Structured initializations
    x = np.arange(n_init) / n_init
    candidates.append(("uniform", np.ones(n_init)))
    candidates.append(("gauss", np.exp(-5 * (x - 0.5) ** 2)))
    candidates.append(("sin_mod", 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x)))
    candidates.append(("tri", 1.0 - 2.0 * np.abs(x - 0.5)))
    candidates.append(("bimodal", np.exp(-10*(x-0.3)**2) + np.exp(-10*(x-0.7)**2)))
    candidates.append(("dirichlet", rng.dirichlet(np.ones(n_init) * 0.2) * n_init))

    # Smooth random
    a_rand = 1.0 + 0.25 * rng.normal(0, 1, n_init)
    kernel = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
    a_rand = np.convolve(a_rand, kernel, mode='same')
    candidates.append(("smooth_rand", a_rand))

    # Quick screen with Nesterov
    for name, a_cand in candidates:
        if time.monotonic() > deadline - 700:
            break
        a_cand = normalize(a_cand)
        a_screen = nesterov_smoothed_gradient(a_cand, tau_init=0.001, max_steps=25, time_budget=7.0)
        obj_screen = compute_obj(a_screen)
        print(f"  {name}: obj={obj_screen:.8f}")
        if obj_screen < best_score:
            best_score = obj_screen
            best_seq = a_screen.copy()
            n_init = len(best_seq)

    # ============================================================
    # Phase 2: Intensive LP optimization
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Intensive LP (n={len(best_seq)})")
    print(f"{'='*60}")

    a = normalize(best_seq.copy())

    for cycle in range(10):
        if time.monotonic() > deadline - 350:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.14, 55.0)

        a_new = lp_optimize_v2(a, max_iters=25, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP cycle {cycle}: {best_score:.8f} **")
        else:
            a = normalize(0.8 * normalize(a) + 0.2 * normalize(a_new))
            print(f"  LP cycle {cycle}: {obj_new:.8f}")

    # ============================================================
    # Phase 3: Gradient polishing (Nesterov + LP-dual + spectral)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Gradient polishing")
    print(f"{'='*60}")

    a = normalize(best_seq.copy())

    # Nesterov with different temperatures
    for tau_val in [0.0003, 0.001, 0.003]:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.13, 45.0)

        a_new = nesterov_smoothed_gradient(a, tau_init=tau_val, max_steps=200, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  Nesterov(tau={tau_val}): {best_score:.8f} **")
        else:
            a = normalize(0.7 * normalize(a) + 0.3 * normalize(a_new))
            print(f"  Nesterov(tau={tau_val}): {obj_new:.8f}")

    # LP-dual gradient
    if time.monotonic() < deadline - 150 and len(a) <= 1200:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.12, 40.0)
        a_new = lp_dual_gradient_step(a, max_iters=20, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP-dual grad: {best_score:.8f} **")
        else:
            print(f"  LP-dual grad: {obj_new:.8f}")

    # Spectral
    if time.monotonic() < deadline - 120:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.1, 30.0)
        a_new = spectral_flatten_v2(a, max_iters=50, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  Spectral: {best_score:.8f} **")
        else:
            a = normalize(0.7 * normalize(a) + 0.3 * normalize(a_new))
            print(f"  Spectral: {obj_new:.8f}")

    # LP follow-up after gradient methods
    if time.monotonic() < deadline - 80 and len(a) <= 1000:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.12, 35.0)
        a_new = lp_optimize_v2(a, max_iters=15, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  Post-Phase3 LP: {best_score:.8f} **")
        else:
            print(f"  Post-Phase3 LP: {obj_new:.8f}")

    # ============================================================
    # Phase 4: Multi-n exploration
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700
    # Generate diverse n options
    n_options = sorted(set([
        max(200, n_best - 300), max(200, n_best - 200), max(200, n_best - 100),
        max(200, n_best - 50),
        n_best,
        n_best + 50, n_best + 100, n_best + 200,
        n_best + 300, n_best + 500, n_best + 700, n_best + 1000
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2500]

    for n_try in n_options:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if n_try == n_best:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample_sequence(best_seq, n_try))

        alloc = min(remaining * 0.08, 20.0)

        # For larger n, skip LP (too slow) and use gradient only
        if n_try <= 1000 and alloc > 10:
            a_try = lp_optimize_v2(a_try, max_iters=8, time_budget=alloc * 0.5)
        a_try = nesterov_smoothed_gradient(a_try, tau_init=0.001, max_steps=40, time_budget=alloc * 0.4)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # ============================================================
    # Phase 5: Final intensive polish
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    a_final = normalize(best_seq.copy())
    n_final = len(a_final)

    for polish_round in range(8):
        if time.monotonic() > deadline - 3:
            break
        remaining = deadline - time.monotonic()
        if remaining < 6:
            break
        alloc = min(remaining * 0.28, 25.0)

        # Alternate methods based on round
        if polish_round % 3 == 0 and n_final <= 1000 and alloc > 10:
            a_pol = lp_optimize_v2(a_final, max_iters=8, time_budget=alloc * 0.5)
        elif polish_round % 3 == 1:
            a_pol = nesterov_smoothed_gradient(a_final, tau_init=0.0002, max_steps=80, time_budget=alloc * 0.6)
        else:
            a_pol = nesterov_smoothed_gradient(a_final, tau_init=0.0005, max_steps=60, time_budget=alloc * 0.5)
            if n_final <= 1000 and alloc > 8:
                a_pol = lp_optimize_v2(a_pol, max_iters=5, time_budget=alloc * 0.3)

        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            a_final = a_pol
            print(f"  Polish {polish_round}: {best_score:.8f} **")
        else:
            a_final = normalize(0.6 * normalize(a_final) + 0.4 * normalize(a_pol))
            print(f"  Polish {polish_round}: {obj_pol:.8f}")

    # ============================================================
    # Final validation
    # ============================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
