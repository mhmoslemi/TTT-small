from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fft_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

    conv = fft_conv(a)
    s = a.sum()
    
    # Log-sum-exp
    cmax = conv.max()
    exp_shifted = np.exp((conv - cmax) / tau)
    lse = cmax + tau * np.log(exp_shifted.sum())
    
    F = 2.0 * n * lse / (s*s)

    # Gradient of LSE via FFT
    fft_size = 2 * (2*n - 1)  # oversize to avoid wrap

    fft_size = 1
    while fft_size < 2*n - 1:

    A = rfft(a, fft_size)

    grad_lse = (2.0 / tau) * corr[:n]

def fft_conv_linear(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute the objective function."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = fft_conv_linear(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def smooth_obj_grad(a, tau):
    """Smooth objective and its gradient using log-sum-exp."""
    n = len(a)
    conv = fft_conv_linear(a)
    s = float(np.sum(a))
    
    # Numerically stable log-sum-exp
    cmax = float(np.max(conv))
    shifted = (conv - cmax) / tau
    exp_shifted = np.exp(np.clip(shifted, -500, 500))
    lse = cmax + tau * float(np.log(np.sum(exp_shifted)))
    
    F = 2.0 * n * lse / (s * s)
    
    # Weights for the "soft max"
    w = exp_shifted / float(np.sum(exp_shifted))
    
    # Cross-correlation via FFT: corr[i] = sum_j a[j] * w[i+j]
    fft_size = 1
    while fft_size < 2*n - 1:

    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    
    # Gradient
    grad_lse = (2.0 / tau) * corr[:n]
    grad = 2.0 * n / (s*s) * grad_lse - 4.0 * n * lse / (s*s*s) * a
    
    return F, grad

    """Flatten the magnitude spectrum to reduce autocorrelation peak."""
    a = a.astype(np.complex128)
    n = len(a)

        size <<= 1

        frac = it / max(n_iters - 1, 1)
        p = p_start + (p_end - p_start) * frac

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using spectral magnitude flattening,
smooth L-BFGS optimization with log-sum-exp temperature annealing, and
multi-start exploration across sequence lengths. Exploits the FFT structure
of the autocorrelation gradient for O(n log n) per iteration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time


def fft_conv_linear(a):
    """Linear convolution a*a via real FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = fft_conv_linear(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def normalize_positive(a):
    """Project to a >= 0 and normalize to sum=1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_obj_grad(a, tau):
    """
    Smooth objective using log-sum-exp and its analytic gradient.
    F(a) = 2*n*logsumexp(conv(a,a)/tau) / (sum(a))^2
    """
    n = len(a)
    conv = fft_conv_linear(a)
    s = float(np.sum(a))

    # Numerically stable log-sum-exp
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    log_sum = float(np.sum(exp_vals))
    lse = cmax + tau * np.log(log_sum)

    F = 2.0 * n * lse / (s * s)

    # Soft-max weights
    w = exp_vals / log_sum

    # Cross-correlation via FFT: corr[i] = sum_j a[j] * w[i+j]
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)

    # dLSE/da[i] = (2/tau) * corr[i]
    grad_lse = (2.0 / tau) * corr[:n]

    # Full gradient: dF/da[i] = 2n/s^2 * grad_lse[i] - 4n*lse*a[i]/s^3
    grad = (2.0 * n / (s * s)) * grad_lse - (4.0 * n * lse / (s * s * s)) * a

    return F, grad


def lbfbs_optimize(a, tau_list, max_iter_per_tau=300):
    """
    L-BFGS-B optimization with temperature annealing.
    Each tau level refines the solution from the previous.
    """
    a = np.maximum(a, 1e-10).astype(np.float64)
    n = len(a)
    bounds = [(1e-10, 1000.0)] * n

    for tau in tau_list:
        # Define fun and jac that capture tau
        def _fun(x, _tau=tau):
            return smooth_obj_grad(x, _tau)[0]

        def _jac(x, _tau=tau):
            return smooth_obj_grad(x, _tau)[1]

        res = minimize(_fun, a, jac=_jac, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter_per_tau,
                                'ftol': 1e-14, 'gtol': 1e-10,
                                'maxfun': max_iter_per_tau * 3})
        a = np.maximum(res.x, 1e-10)

    return a


def spectral_flatten(a, n_iters=80, p_start=0.8, p_end=0.05):
    """
    Flatten the magnitude spectrum to reduce autocorrelation peak.
    Iteratively compresses |FFT(a)| toward uniform.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(n_iters):
        frac = it / max(n_iters - 1, 1)
        p = p_start + (p_end - p_start) * frac
        p = max(p, 0.02)

        A = fft(a, size)
        mag = np.abs(A)
        # Avoid division by zero
        mag_safe = np.maximum(mag, 1e-30)
        phase = A / mag_safe

        # Compress magnitude: mag_new = (mag^p + eps)^(1/p)
        mag_p = mag ** p
        mag_new = np.power(mag_p + 1e-20, 1.0 / p)

        # Preserve total energy
        energy_orig = np.sum(mag ** 2)
        energy_new = np.sum(mag_new ** 2)
        if energy_new > 1e-30:
            mag_new *= np.sqrt(energy_orig / energy_new)

        A_new = phase * mag_new

        # Enforce conjugate symmetry for real output
        A_new[size // 2] = np.real(A_new[size // 2])
        for k in range(1, size // 2):
            A_new[size - k] = np.conj(A_new[k])

        a = np.real(ifft(A_new, size)[:n])
        a = np.maximum(a, 0)

    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
