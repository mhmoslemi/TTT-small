```python
def solve_inner_smoothed(a, tau=0.01, n_iters=100):
    """Solve min sum(exp(conv(a,g)/tau)) s.t. sum(g)=1, g>=0 using gradient descent."""
    n = len(a)
    A = build_conv_matrix(a)  # (2n-1) x n
    
    g = np.ones(n) / n
    lr = 0.1 / n
    
    for it in range(n_iters):
        # Gradient of sum(exp(A@g / tau)) w.r.t. g
        conv_val = A @ g  # (2n-1,)
        exp_val = np.exp(conv_val / tau)
        grad = A.T @ (exp_val / tau)
        
        # Projected gradient step
        g = g - lr * grad
        # Project onto simplex {g >= 0, sum(g) = 1}
        g = project_simplex(g)
    
    return g

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
Optimized step function search using fast smoothed alternating optimization.

1. FFT/np.correlate based gradient computation (O(n^2) but in C, very fast)
2. FISTA acceleration for the inner convex subproblem

4. Multiple outer iterations with adaptive blending

import numpy as np
from scipy.fft import rfft, irfft
import time

# --- Core computation functions ---

def compute_conv(a, b=None):
    """Full linear convolution a * b via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def inner_gradient(a, g, tau):
    """
    Compute gradient of sum_k exp(conv(a,g)[k]/tau) w.r.t. g.
    Uses np.correlate for efficiency.
    """
    n = len(a)

    u = np.exp(conv_ag / tau) / tau  # length 2n-1
    # grad[j] = sum_m a[m] * u[m+j]
    grad = np.correlate(u, a, mode='valid')  # length 2n-1 - n + 1 = n

def solve_inner_fista(a, tau, n_iters=50, verbose=False):
    """

    best_val = None

        y_new = g_new + ((t_k - 1.0) / t_new) * (g_new - g)
        y = project_simplex(y_new)  # Ensure y stays in simplex
        t_k = t_new
        g = g_new

        grad_g, conv_g = inner_gradient(a, g, tau)

        if best_val is None or val_g < best_val:

            best_g = g.copy()
        
        # Backtracking if objective increased significantly
        if best_val is not None and val_g > best_val * 1.5:
            step *= 0.7
        
    return best_g
```

```python
import numpy as np

import time


def compute_conv(a, b=None):
    """Full linear convolution a*b via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def inner_gradient(a, g, tau):
    """
    Compute gradient of sum_k exp(conv(a,g)[k]/tau) w.r.t. g.
    Returns (grad, conv_ag, log_sum_exp_val).
    """
    n = len(a)
    conv_ag = compute_conv(a, g)  # length 2n-1
    # Log-sum-exp trick for numerical stability
    max_cv = float(np.max(conv_ag))
    shifted = (conv_ag - max_cv) / tau
    exp_shifted = np.exp(shifted)
    lse = max_cv / tau + np.log(np.sum(exp_shifted))
    # Gradient: grad[j] = sum_m a[m] * u[m+j] where u[k] = exp((conv[k]-max)/tau)/tau
    u = exp_shifted / tau
    grad = np.correlate(u, a, mode='valid')  # length n
    return grad, conv_ag, lse


def solve_inner_fista(a, tau, n_iters=40, verbose=False):
    """
    Solve min sum_k exp(conv(a,g)[k]/tau) s.t. g >= 0, sum(g) = 1
    using FISTA (accelerated projected gradient descent).
    """
    n = len(a)
    g = np.ones(n) / n
    y = g.copy()
    t_k = 1.0

    # Estimate initial step size
    grad_init, _, _ = inner_gradient(a, g, tau)
    max_grad = float(np.max(np.abs(grad_init)))
    if max_grad < 1e-15:
        return g.copy()
    step = 0.5 / max_grad

    best_g = g.copy()
    best_val = np.inf

    for k in range(n_iters):
        grad_y, _, val_y = inner_gradient(a, y, tau)

        # FISTA gradient step
        g_new = project_simplex(y - step * grad_y)

        # Check convergence
        if np.max(np.abs(g_new - g)) < 1e-10:
            g = g_new
            break

        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = g_new + ((t_k - 1.0) / t_new) * (g_new - g)
        y = project_simplex(y_new)
        t_k = t_new
        g = g_new

        # Track best
        _, _, val_g = inner_gradient(a, g, tau)
        if val_g < best_val:
            best_val = val_g
            best_g = g.copy()

    return best_g

def chase_peak_refine(a, n_steps=200, time_budget=30.0):
    """
    Subgradient method targeting the maximum convolution bin with momentum.

    step_size = 0.02

    beta = 0.9

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient of max_conv w.r.t. a[i]
        grad = np.zeros(n)

                else:
                    grad[i] = 2.0 * a[j]

        # Project gradient onto tangent space (sum = 0)
        grad_centered = grad - np.mean(grad)

        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad_centered

        a_new = a - step_size * momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:

                step_size = min(step_size * 1.05, 0.05)
            else:
                step_size *= 0.85
        else:
            step_size *= 0.5

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def random_init(n, rng):

    elif mode == 2:

        a = a + 0.2
        a = a / np.sum(a)
    else:
        # Bimodal

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak_limited(a, n_solvers=2):

        c[-1] = 1.0

        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

            s = float(np.sum(g_val))

        a_opt = alternating_smoothed(a_r, tau_start=0.05, tau_end=0.005,
                                     n_outer=100, inner_iters=25,
                                     time_budget=restart_budget)
        obj_r = compute_objective(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            a_work = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # One more round of smoothed alternating with tight tau
        alloc = min(remaining * 0.3, 80.0)
        a_final = alternating_smoothed(a_work, tau_start=0.01, tau_end=0.001,
                                       n_outer=200, inner_iters=40,
                                       time_budget=alloc, verbose=True)
        obj_f = compute_objective(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            a_work = a_final.copy()
            print(f"  Final smooth: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Multi-peak polish
        a_mp2 = multi_peak_refine(a_work, n_steps=200, time_budget=min(remaining * 0.2, 30.0))
        obj_mp2 = compute_objective(a_mp2)
        if obj_mp2 < best_score:
            best_score = obj_mp2
            best_seq = a_mp2.copy()
            a_work = a_mp2.copy()
            print(f"  Final multi-peak: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Chase peak polish
        a_cp2 = chase_peak_refine(a_work, n_steps=300, time_budget=min(remaining * 0.2, 25.0))
        obj_cp2 = compute_objective(a_cp2)
        if obj_cp2 < best_score:
            best_score = obj_cp2
            best_seq = a_cp2.copy()
            a_work = a_cp2.copy()
            print(f"  Final chase: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # LP polish (limited, since LP is slow)
        n_lp = min(len(a_work), 1000)  # Cap for LP speed
        if n_lp == len(a_work):
            a_lp = lp_polish(a_work, n_rounds=3, time_budget=min(remaining * 0.3, 30.0))
            obj_lp = compute_objective(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_objective(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
