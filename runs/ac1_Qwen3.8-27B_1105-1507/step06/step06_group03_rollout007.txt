def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)

    """Compute grad[i] = 2 * sum_j w[i+j] * a[j] for i=0..n-1."""
    n = len(a)
    len_w = len(w)
    # We need sum_j w[i+j] * a[j] for i = 0, ..., n-1
    # This equals (w * rev_a)[(len_w - 1) - i + n - 1]... 
    # Let me just use correlate
    # np.correlate(w, a, mode='full')[k] = sum_j w[j+k] * a[j] (for appropriate range)
    # Actually: np.correlate(w, a, 'full')[i] = sum_j w[j + offset] * conj(a[j])
    # For real: np.correlate(w, a, 'full')[i] = sum_j w[j + (n-1-i)] * a[j]
    # Hmm, the indexing is tricky.
    
    # Let me just do: result[i] = sum_j w[i+j] * a[j]
    # = conv(w_extended, a)[i + len_w - 1]... no
    # 
    # Actually the simplest: use FFT
    # result = ifft(fft(w, N) * fft(a, N)) gives circular convolution (w*a)[k] = sum_j w[j]*a[k-j]
    # We want sum_j w[i+j]*a[j]
    # Let's define b[j] = a[n-1-j] (reversed a)
    # Then sum_j w[i+j]*a[j] = sum_j w[i+j]*b[n-1-j]
    # Let m = n-1-j: sum_m w[i+n-1-m]*b[m] = (w*b)[i+n-1]... 
    # (w*b)[k] = sum_j w[j]*b[k-j] = sum_j w[j]*b[k-j]
    # We want sum_m w[i+n-1-m]*b[m] = sum_m b[m]*w[i+n-1-m] = (b*w)[i+n-1]
    # So: result[i] = (b*w)[i+n-1] where b = a reversed, and * is linear convolution
    
    b = a[::-1]
    N = 1
    while N < len(w) + n:
        N <<= 1
    W = np.fft.rfft(w, N)
    B = np.fft.rfft(b, N)
    conv_bw = np.fft.irfft(W * B, N)
    # result[i] = conv_bw[i + n - 1] for i = 0, ..., n-1
    result = conv_bw[n-1 : 2*n-1]
    return 2.0 * result
