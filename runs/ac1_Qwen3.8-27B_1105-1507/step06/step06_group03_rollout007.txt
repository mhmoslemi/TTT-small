"""
Precision optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))².
Uses exact 1D coordinate minimization with peak-aware ordering, aggressive LP mirror descent,
multi-length exploration, and simulated-annealing-style restarts to beat 1.5027.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def precise_coord_descent(a, max_sweeps, deadline, seed=42, verbose=False):
    """
    Coordinate descent with precise 1D minimization per coordinate.
    Uses dense sampling + golden section for each coordinate.
    Peak-aware coordinate ordering for faster convergence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    else:
        a /= S
        S = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (S * S)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 50

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Peak-aware ordering: prioritize coordinates contributing to top peaks
        if sweep % 2 == 0 and sweep > 0:
            # Get top 5 peak positions
            peak_positions = np.argsort(conv)[::-1][:5]
            contrib = np.zeros(n)
            for pk in peak_positions:
                for j in range(n):
                    k2 = pk - j
                    if 0 <= k2 < n:
                        contrib[j] += a[j] * a[k2]
            order = np.argsort(-contrib + rng.uniform(0, 0.05, n))
        else:
            order = rng.permutation(n)

        improved = False

        for i in order:
            if time.monotonic() > deadline + 5:
                break

            old = a[i]
            # Determine search range
            if old < 1e-12:
                upper = max(S / n * 2.0, 1e-3)
            else:
                upper = max(old * 4.0, S / n * 2.0)

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            kq = 2 * i
            has_kq = (kq <= 2 * n - 2)
            a_slice = a[lo - i:hi - i + 1].copy()

            def J_of_x(x):
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-18:
                    return 2.0 * n * conv.max() / (S * S)
                new_S = S + delta
                if new_S < 1e-10:
                    return float('inf')
                # Update conv in-place (we'll restore)
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if has_kq:
                    conv[kq] += delta * delta
                val = 2.0 * n * conv.max() / (new_S * new_S)
                # Restore
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if has_kq:
                    conv[kq] -= delta * delta
                return val

            # Stage 1: Coarse grid
            n_grid = 25
            x_grid = np.linspace(0.0, upper, n_grid)
            J_grid = np.array([J_of_x(x) for x in x_grid])
            idx_best = np.argmin(J_grid)
            best_x = x_grid[idx_best]
            best_local = J_grid[idx_best]

            # Stage 2: Golden section refinement
            if 0 < idx_best < n_grid - 1:
                lo_x = x_grid[idx_best - 1]
                hi_x = x_grid[idx_best + 1]
            elif idx_best == 0:
                lo_x = 0.0
                hi_x = x_grid[1]
            else:
                lo_x = x_grid[idx_best - 1]
                hi_x = upper

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            for _ in range(20):
                if hi_x - lo_x < 1e-13:
                    break
                x1 = hi_x - gr * (hi_x - lo_x)
                x2 = lo_x + gr * (hi_x - lo_x)
                f1 = J_of_x(x1)
                f2 = J_of_x(x2)
                if f1 < f2:
                    hi_x = x2
                else:
                    lo_x = x1
            x_gold = (lo_x + hi_x) / 2.0
            J_gold = J_of_x(x_gold)
            if J_gold < best_local:
                best_x = x_gold
                best_local = J_gold

            # Apply update
            if abs(best_x - old) > 1e-13 and best_local < best_obj + 1e-12:
                delta = best_x - old
                if S + delta > 1e-10:
                    conv[lo:hi + 1] += 2.0 * delta * a_slice
                    if has_kq:
                        conv[kq] += delta * delta
                    a[i] = best_x
                    S += delta
                    improved = True

        current_obj = 2.0 * n * conv.max() / (S * S) if S > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 10 == 0:
            print(f"      CD sweep {sweep}: obj={best_obj:.9f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a, best_obj


def lp_mirror_descent(a, max_iters, deadline, verbose=False):
    """
    LP mirror descent: solve LP for best response direction, blend with current.
    More aggressive: tries multiple blend fractions and uses tighter time limits.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Build LP: minimize t
        # s.t. 2*(a * g)[k] - t <= c[k] for all k
        #      sum(g) = 1
        #      g >= 0
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A_ub = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 60, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
                        a_try = (1.0 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-14:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 3 == 0:
            print(f"      LP it {it}: obj={best_obj:.9f}", flush=True)

        if no_improve > 30:
            break

    return best_a, best_obj


def smooth_gradient_descent(a, max_iters, deadline, seed=42, verbose=False):
    """
    Adam-style gradient descent with log-sum-exp smoothing of max convolution.
    Uses softmax parameterization for automatic positivity.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    rng = np.random.default_rng(seed)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0
    no_improve = 0
    tau = 50.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = 30.0 + 150.0 * frac

        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()

        # Gradient via FFT
        a_rev = a[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        Wrev_fft = np.fft.rfft(w[::-1], size)
        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wrev_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad[i] += conv2[idx2]

        # Project to tangent space of simplex
        grad = grad - np.mean(grad)

        t_count += 1
        lr = 0.03 / (1.0 + 0.05 * t_count ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                a = best_a + rng.normal(0, 0.01, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

        if verbose and it % 100 == 0:
            print(f"      Adam it {it}: tau={tau:.0f} obj={best_obj:.9f}", flush=True)

    return best_a, best_obj


def peak_shaving(a, max_iters, deadline, verbose=False):
    """
    Direct peak shaving: identify top convolution peaks and reduce contributing coordinates.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    a /= a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = conv.max()
        
        # Find top peaks
        peak_threshold = M * 0.95
        peak_positions = np.where(conv >= peak_threshold)[0]
        
        if len(peak_positions) == 0:
            peak_positions = [np.argmax(conv)]
        
        reduced = False
        for pk in peak_positions[:5]:
            # Find top contributing pairs
            contribs = []
            for j in range(n):
                k2 = pk - j
                if 0 <= k2 < n:
                    contribs.append((a[j] * a[k2], j, k2))
            contribs.sort(reverse=True)
            
            # Reduce the largest contributors
            for _, j, k2 in contribs[:3]:
                # Reduce the larger one
                if a[j] > a[k2]:
                    a[j] *= 0.9
                else:
                    a[k2] *= 0.9
                reduced = True
        
        if not reduced:
            break
        
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            break
        a /= a.sum()
        
        obj = compute_J(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                # Revert and try different strategy
                a = best_a.copy()
                no_improve = 0
        
        if verbose and it % 10 == 0:
            print(f"      PeakShave it {it}: obj={best_obj:.9f}", flush=True)

    return best_a, best_obj

            update_best(a_f3)

        # Final short CD
        remaining = phase5_end - time.monotonic()
        if remaining > 15:
            a_f4, obj_f4 = precise_coord_descent(best_seq, max_sweeps=5000,
                                                 deadline=phase5_end - 5, seed=7777)
            update_best(a_f4)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.9f} n={best_seq.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
