import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def projected_subgradient_descent(a, time_budget=60.0, verbose=False):
    """
    Projected subgradient descent with Nesterov momentum for minimizing
    max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 0.01  # Initial step size
    momentum = 0.9  # Momentum parameter (or Nesterov beta)
    v = np.zeros(n)  # Momentum buffer
    
    it = 0
    no_improve = 0
    evals_since_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Check time every few iterations
        if it % 100 == 0:
            elapsed = time.monotonic() - start
            if elapsed > time_budget:
                break
            if verbose and it % 1000 == 0:
                print(f"    PGD it={it}: obj={best_obj:.8f}, eta={eta:.6f}, elapsed={elapsed:.1f}s")
        
        # Compute convolution and find peak
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        
        # Subgradient: g[j] = 2*a[k_star - j] if 0 <= k_star - j < n, else 0
        g = np.zeros(n)
        j_start = max(0, k_star - (n - 1))
        j_end = min(n - 1, k_star)
        for j in range(j_start, j_end + 1):
            g[j] = 2.0 * a[k_star - j]

Optimized step function search using smooth projected gradient descent
with FFT-based convolution, adaptive temperature annealing, Nesterov momentum,
and periodic LP refinement. Explores non-symmetric sequences to reduce peak
autocorrelation below the symmetric bound.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def next_pow2(n):
    size = 1
    while size < n:
        size <<= 1
    return size


def compute_grad_softmax(a, weights, fft_size):
    """
    Compute gradient: grad[j] = 2 * sum_k weights[k] * a[k-j]
    Using FFT cross-correlation.
    """
    corr = irfft(rfft(weights, fft_size) * np.conj(rfft(a, fft_size)), fft_size)
    return 2.0 * corr[:len(a)]

    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=1.

                print(f"      SPGD it={it}: obj={best_obj:.8f}, T={T:.6f}, eta={eta:.6f}, t={elapsed:.1f}s")
            T = max(T * 0.98, 1e-8)
        
        conv = compute_conv(a)
        conv_max = float(np.max(conv))

        conv_shifted = (conv - conv_max) / T
        weights = np.exp(conv_shifted)
        w_sum = float(np.sum(weights))

            peak_idx = int(np.argmax(conv))
            weights[peak_idx] = 1.0

                eta *= 0.85
                v *= 0.5
            if no_improve > 500:
                # Perturb to escape
                rng = np.random.default_rng(seed + it)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                eta = 5.0 / n
                no_improve = 0
                best_obj = float(2.0 * n * np.max(compute_conv(a)))
        
        it += 1
    
    return best_a


def lp_refine(a, time_budget=60.0, verbose=False):
    """
    LP-based refinement: solve min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Then blend.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Build convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    
    for it in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        # Solve min M s.t. A@g <= M, sum(g) = 1, g >= 0
        g_val = None
        M_opt = None
        
        # Try CVXPY solvers
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            
            for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
                try:
                    prob.solve(solver=solver)
                    if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        g_val = np.maximum(g.value, 0)
                        M_opt = float(M.value)
                        break
                except Exception:
                    continue
        except Exception:
            pass
        
        # Fallback to scipy linprog
        if g_val is None:
            try:
                c = np.zeros(n + 1)
                c[-1] = 1.0
                A_ub = np.zeros((2 * n - 1, n + 1))
                A_ub[:, :n] = A
                A_ub[:, n] = -1.0
                A_eq = np.zeros((1, n + 1))
                A_eq[0, :n] = 1.0
                b_eq = np.array([1.0])
                bounds = [(0, None)] * n + [(0, None)]
                res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1),
                             A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
                if res.status == 0:
                    g_val = np.maximum(res.x[:n], 0)
                    M_opt = float(res.x[n])
            except Exception:
                pass
        
        if g_val is None:
            break
        
        s_g = float(np.sum(g_val))
        if s_g < 1e-10:
            break
        g_val = g_val / s_g
        
        # Line search
        best_t = 0
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                best_t = t
        
        if best_t == 0:
            break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
