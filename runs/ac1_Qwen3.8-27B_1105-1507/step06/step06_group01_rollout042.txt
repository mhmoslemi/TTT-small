- x in R^n (unconstrained)
- a = softmax(x) = exp(x) / sum(exp(x))
- f(x) = 2n * smooth_max(conv(softmax(x), softmax(x)))  [since sum(a)=1]
- df/dx[i] = (da/dx[i])^T * df/da

da/dx[i] = a[i] * (e_i - a) (Jacobian of softmax)

So df/dx[i] = sum_j (da/dx[i])_j * df/da[j]
= a[i] * (df/da[i] - sum_j a[j] * df/da[j])
= a[i] * (df/da[i] - mean_a(grad_f))

where grad_f = df/da (the gradient w.r.t. a).

So: grad_x[i] = a[i] * (grad_a[i] - dot(a, grad_a))

This is the "natural gradient" or "mirror gradient" for the simplex.

```python
def softmax_obj_grad(x, tau):
    """Compute f(x) and grad_x for the log-sum-exp smoothed objective.
    
    x: n-vector (unconstrained)
    Returns: (f_value, grad_x)
    """
    n = len(x)
    # Compute a = softmax(x)
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    
    # Compute b = conv(a, a)
    b = fast_conv(a)
    
    # Log-sum-exp
    b_max = b.max()
    tau_b = tau * (b - b_max)
    exp_tb = np.exp(tau_b)
    log_sum_exp = b_max + np.log(exp_tb.sum()) / tau
    f = 2.0 * n * log_sum_exp  # since sum(a) = 1
    
    # Gradient w.r.t. a
    p = exp_tb / exp_tb.sum()
    grad_a = np.zeros(n)
    for i in range(n):
        grad_a[i] = (2.0 / tau) * np.dot(a, p[i:i+n])
    
    # Gradient w.r.t. x (natural gradient)
    mean_grad = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_grad)
    
    return f, grad_x
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B on log-sum-exp smoothed objective with softmax parameterization
2. Frank-Wolfe (LP-based) with full constraint set
3. Efficient coordinate descent with O(n^2) per sweep
4. Multi-length exploration
5. Final polishing with alternating methods

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_M(a):
    """Compute max of self-convolution."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(b.max() / (s * s))  # This is M/s^2, and J = 2n * this

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv
    if cond.any():
        rho_idx = np.where(cond)[0][-1]
    else:
        rho_idx = 0
    theta = (cv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def softmax(x):
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    return exp_x / exp_x.sum()

def smooth_obj_grad_x(x, tau):
    """Objective and gradient w.r.t. x for log-sum-exp smoothed problem.
    a = softmax(x), minimize 2n * (1/tau) * log(sum(exp(tau * conv(a,a))))
    (sum(a) = 1 by construction)
    """
    n = len(x)
    a = softmax(x)
    b = fast_conv(a)
    
    b_max = b.max()
    tau_b = tau * (b - b_max)
    exp_tb = np.exp(tau_b)
    z = exp_tb.sum()
    f = 2.0 * n * (b_max + np.log(z) / tau)
    
    p = exp_tb / z
    # grad_a[i] = (2/tau) * sum_{j=0}^{n-1} p[i+j] * a[j]
    grad_a = np.zeros(n)
    for i in range(n):
        grad_a[i] = (2.0 / tau) * np.dot(a, p[i:i + n])
    
    # grad_x[i] = a[i] * (grad_a[i] - sum_j a[j] * grad_a[j])
    mean_g = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_g)
    
    return f, grad_x

def run_lbfgsb(x0, tau, maxiter=5000, deadline=None):
    """Run L-BFGS-B on the smoothed problem."""
    def objective(x):
        f, _ = smooth_obj_grad_x(x, tau)
        return f
    
    def grad_fn(x):
        _, g = smooth_obj_grad_x(x, tau)
        return g
    
    try:
        result = minimize(objective, x0, jac=grad_fn, method='L-BFGS-B',
                         options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-10})
        a = softmax(result.x)
        return a, result
    except Exception as e:
        print(f"    L-BFGS-B error: {e}")
        return x0, None

def run_fw_lp(a_init, max_iters, deadline, n_active=None):
    """Frank-Wolfe with LP using active constraints.
    If n_active is None, use all constraints.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    b_full_len = 2 * n - 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    no_improve = 0
    base_n_active = n_active if n_active is not None else b_full_len
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M_cur = b.max()
        
        # Select active constraints
        if base_n_active >= b_full_len:
            active = list(range(b_full_len))
        else:
            # Top-k by value, plus some random for diversity
            top_k = np.argsort(b)[::-1][:base_n_active]
            active = top_k.tolist()
        
        # Build LP: max sum(g) s.t. (a*g)[k] <= M_cur for k in active, g >= 0
        # (a*g)[k] = sum_j a[k-j] * g[j] (where a[m]=0 outside [0,n-1])
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active):
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    rows.append(idx)
                    cols.append(j)
                    vals.append(a[m])
        
        A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
        
        c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
        b_ub = np.full(len(active), M_cur)
        bounds = [(0.0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                       options={'time_limit': 10, 'presolve': True})
            if not r.success or r.x is None:
                no_improve += 1
                if no_improve > 10:
                    break
                # Try with fewer constraints
                if base_n_active < b_full_len:
                    base_n_active = max(10, base_n_active // 2)
                continue
            
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                no_improve += 1
                if no_improve > 10:
                    break
                continue
            g = g / g_sum
        except Exception:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Line search
        improved = False
        for t in [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed
        for sigma in [0.02, 0.05, 0.1]:
            p_pert = np.maximum(p + rng.normal(0, sigma, n), 1e-4)
            inits.append(p_pert / p_pert.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Concave (parabola)
    par = (1.0 - (2*t - 1)**2) + 0.05
    inits.append(np.maximum(par, 0.01) / np.maximum(par, 0.01).sum())
    
    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5)**2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Cosine
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(np.maximum(cos_w, 0.01) / np.maximum(cos_w, 0.01).sum())
    
    # Bi-cosine
    cos2 = 0.3 + 0.4 * np.cos(2 * np.pi * t) + 0.3 * np.cos(4 * np.pi * t)
    inits.append(np.maximum(cos2, 0.01) / np.maximum(cos2, 0.01).sum())
    
    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B on log-sum-exp smoothed objective with softmax parameterization
2. Frank-Wolfe (conditional gradient) with full LP oracle
3. Efficient coordinate descent with O(n^2) per sweep (maintaining b explicitly)
4. Multi-length exploration
5. Final alternating polish (FW + CD + L-BFGS)
Key improvements: proper O(n^2) coordinate descent, full-constraint LP,
softmax-reparameterized L-BFGS-B for constraint-free optimization,
and adaptive multi-phase time allocation.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv
    if cond.any():
        rho_idx = np.where(cond)[0][-1]
    else:
        rho_idx = 0
    theta = (cv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def softmax_fn(x):
    """Stable softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    return exp_x / exp_x.sum()


def smooth_obj_grad_x(x, tau):
    """
    Objective and gradient w.r.t. x for log-sum-exp smoothed problem.
    a = softmax(x), minimize 2n * (1/tau) * log(sum(exp(tau * conv(a,a))))
    """
    n = len(x)
    a = softmax_fn(x)
    b = fast_conv(a)

    b_max = b.max()
    tau_b = tau * (b - b_max)
    exp_tb = np.exp(tau_b)
    z = exp_tb.sum()
    f = 2.0 * n * (b_max + np.log(z) / tau)

    p = exp_tb / z
    # grad_a[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    grad_a = np.zeros(n)
    for i in range(n):
        grad_a[i] = 2.0 * np.dot(a, p[i:i + n])

    # grad_x[i] = a[i] * (grad_a[i] - sum_j a[j] * grad_a[j])
    mean_g = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_g)

    return f, grad_x


def run_lbfgsb(x0, tau, maxiter=5000, max_time=60.0):
    """Run L-BFGS-B on the smoothed problem."""
    t_start = time.monotonic()

    def objective(x):
        f, _ = smooth_obj_grad_x(x, tau)
        return f

    def grad_fn(x):
        if time.monotonic() - t_start > max_time:
            raise TimeoutError()
        _, g = smooth_obj_grad_x(x, tau)
        return g

    try:
        result = minimize(objective, x0, jac=grad_fn, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                                   'maxfun': maxiter * 3})
        a = softmax_fn(result.x)
        return a
    except (TimeoutError, Exception):
        return None


def build_lp_matrix(a, active_indices, n):
    """Build sparse LP constraint matrix for given active indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def run_fw_lp(a_init, max_iters, deadline, n_active=None):
    """Frank-Wolfe with LP oracle. n_active=None means use all constraints."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    b_full_len = 2 * n - 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M_cur = b.max()

        # Select active constraints
        if n_active is None or n_active >= b_full_len:
            active = list(range(b_full_len))
        else:
            top_k = np.argsort(b)[::-1][:n_active]
            active = top_k.tolist()

        # Build and solve LP: max sum(g) s.t. (a*g)[k] <= M_cur, g >= 0
        A = build_lp_matrix(a, active, n)

        c = -np.ones(n)
        b_ub = np.full(len(active), M_cur)
        bounds = [(0.0, None)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if not r.success or r.x is None:
                no_improve += 1
                if no_improve > 15:
                    break
                continue

            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                no_improve += 1
                if no_improve > 15:
                    break
                continue
            g = g / g_sum
        except Exception:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        # Geometric line search
        improved = False
        for t in [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.0039]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    return best_a


def coordinate_descent_efficient(a_init, max_sweeps, deadline, seed=0):
    """
    Efficient coordinate descent maintaining b = conv(a,a) explicitly.
    O(n^2) per sweep.
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    b = fast_conv(a)
    best_a = a.copy()
    best_M = b.max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = np.arange(n)
        rng.shuffle(order)

        sweep_changed = False

        for i in order:
            if time.monotonic() > deadline:
                break

            x_cur = a[i]

            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            # Unaffected max
            base_max = 0.0
            if k_lo > 0:
                base_max = max(base_max, b[:k_lo].max())
            if k_hi < 2 * n - 2:
                base_max = max(base_max, b[k_hi + 1:].max())

            # Build c and s arrays for affected positions
            k_range = np.arange(k_lo, k_hi + 1)
            num_affected = len(k_range)
            c_vals = np.zeros(num_affected)
            s_vals = np.zeros(num_affected)
            quad_pos = -1

            for idx in range(num_affected):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = b[k] - x_cur * x_cur
                    s_vals[idx] = 0.0
                    quad_pos = idx
                else:
                    j = k - i
                    if 0 <= j < n:
                        c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                        s_vals[idx] = 2.0 * a[j]
                    else:
                        c_vals[idx] = b[k]
                        s_vals[idx] = 0.0

            has_quad = (quad_pos >= 0)

            def obj_at(x_test):
                vals = c_vals + s_vals * x_test
                m = base_max
                vm = vals.max()
                if vm > m:
                    m = vm
                if has_quad:
                    qv = c_vals[quad_pos] + x_test * x_test
                    if qv > m:
                        m = qv
                return m

            # Golden section search
            lo = 0.0
            hi = max(2.0 * x_cur + 0.05, np.sqrt(max(best_M, 0.0)) + 0.05, 0.05)

            # Expand hi if needed
            for _ in range(5):
                if obj_at(hi * 0.5) < obj_at(hi) and hi < 10.0:
                    hi *= 2.0
                else:
                    break

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            ga = hi - gr * (hi - lo)
            gb = lo + gr * (hi - lo)
            fa = obj_at(ga)
            fb = obj_at(gb)

            for _ in range(28):
                if fa < fb:
                    hi = gb
                    gb = ga
                    fb = fa
                    ga = hi - gr * (hi - lo)
                    fa = obj_at(ga)
                else:
                    lo = ga
                    ga = gb
                    fa = fb
                    gb = lo + gr * (hi - lo)
                    fb = obj_at(gb)

            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0
            x_opt = max(0.0, x_opt)

            if abs(x_opt - x_cur) > 1e-13:
                # Update b values
                for idx in range(num_affected):
                    k = k_lo + idx
                    if k == 2 * i:
                        b[k] = c_vals[idx] + x_opt * x_opt
                    else:
                        b[k] = c_vals[idx] + s_vals[idx] * x_opt
                a[i] = x_opt
                sweep_changed = True

        # Renormalize and recompute b for consistency
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        b = fast_conv(a)

        M_cur = b.max()
        if M_cur < best_M * (1.0 - 1e-11):
            best_M = M_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_changed:
            no_improve += 5
        if no_improve > 20:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            p_pert = np.maximum(p + rng.normal(0, sigma, n), 1e-4)
            inits.append(p_pert / p_pert.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Concave (parabola)
    par = (1.0 - (2 * t - 1) ** 2) + 0.05
    par = np.maximum(par, 0.01)
    inits.append(par / par.sum())

    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    # Cosine
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())

    # Bi-cosine
    cos2 = 0.3 + 0.4 * np.cos(2 * np.pi * t) + 0.3 * np.cos(4 * np.pi * t)
    cos2 = np.maximum(cos2, 0.01)
    inits.append(cos2 / cos2.sum())

    # Triangular
    tri = 1.0 - 2.0 * np.abs(t - 0.5) + 0.05
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_J = float('inf')
        self.t0 = time.monotonic()

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        J = float(evaluate_sequence(a.tolist()))
        elapsed = time.monotonic() - self.t0
        if np.isfinite(J) and J < self.best_J:
            self.best = a.copy()
            self.best_J = J
            print(f"  ** NEW BEST ({label}): J={J:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
        elif label:
            print(f"    ({label}): J={J:.8f} n={a.size} (no improvement, best={self.best_J:.8f})", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [{label}] best: J={self.best_J:.8f} n={self.best.size}", flush=True)
        else:
            print(f"  [{label}] no valid solution yet", flush=True)


def run_optimization_pipeline(a0, n, deadline, seed=0):
    """Run a full optimization pipeline: L-BFGS -> FW -> CD."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 1e-6)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t_start = time.monotonic()
    phase_dur = max(1.0, deadline - t_start)

    # Stage 1: L-BFGS (40% time)
    t1 = t_start + 0.4 * phase_dur
    if time.monotonic() < t1:
        x0 = np.log(np.maximum(a, 1e-10))
        for tau in [50, 150, 400]:
            if time.monotonic() > t1:
                break
            time_left = t1 - time.monotonic()
            if time_left < 5:
                break
            a_lb = run_lbfgsb(x0, tau, maxiter=2000, max_time=time_left - 1)
            if a_lb is not None:
                M_lb = fast_conv(a_lb).max()
                if M_lb < best_M:
                    best_M = M_lb
                    best_a = a_lb.copy()
                x0 = np.log(np.maximum(best_a, 1e-10))

    # Stage 2: Frank-Wolfe (30% time)
    t2 = t_start + 0.7 * phase_dur
    if time.monotonic() < t2:
        for n_act in [40, 2 * n - 1]:
            if time.monotonic() > t2:
                break
            a_fw = run_fw_lp(best_a, max_iters=100, deadline=time.monotonic() + 30,
                            n_active=min(n_act, 2 * n - 1))
            M_fw = fast_conv(a_fw).max()
            if M_fw < best_M:
                best_M = M_fw
                best_a = a_fw.copy()

    # Stage 3: Coordinate Descent (30% time)
    if time.monotonic() < deadline:
        max_sw = max(5, int((deadline - time.monotonic()) / max(0.5, n / 80.0)))
        a_cd = coordinate_descent_efficient(best_a, max_sw, deadline, seed=seed + 1)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best_a = a_cd.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: L-BFGS-B + FW + CD from multiple inits (main lengths)
    Phase 2: Intensive FW polish with expanding active sets
    Phase 3: Coordinate descent deep polish
    Phase 4: Length exploration
    Phase 5: Final alternating refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "parent")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_J:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768)
        parent = parent / parent.sum()
        tracker.update(parent, "uniform")
        print(f"No parent, using uniform n=768, J={tracker.best_J:.8f}", flush=True)

    n_parent = parent.size

    # Determine lengths to explore
    lengths_main = [n_parent]
    if 768 not in lengths_main:
        lengths_main.append(768)
    if 640 not in lengths_main:
        lengths_main.append(640)
    if 896 not in lengths_main:
        lengths_main.append(896)

    # === PHASE 1: Main optimization ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Main optimization (lengths: {lengths_main})", flush=True)
    print(f"{'='*60}", flush=True)

    t_p1_end = time.monotonic() + 400
    num_lengths = len(lengths_main)
    time_per_length = (t_p1_end - time.monotonic()) / max(1, num_lengths)
    time_per_length = max(time_per_length, 60)

    for li, n in enumerate(lengths_main):
        if time.monotonic() > t_p1_end - 30:
            break

        t0 = time.monotonic()
        remaining = t_p1_end - 30 - time.monotonic()
        if remaining < 30:
            break

        time_alloc = min(time_per_length, remaining / max(1, num_lengths - li))
        print(f"\n  Length n={n}, alloc={time_alloc:.0f}s", flush=True)

        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        inits = inits[:6]  # Limit to 6 inits

        best_for_n = None
        best_M_n = float('inf')
        time_per_init = time_alloc / max(1, len(inits))

        for ii, a0 in enumerate(inits):
            if time.monotonic() > t_p1_end - 30:
                break
            sub_deadline = min(t0 + time_alloc, t_p1_end - 30)
            if time.monotonic() >= sub_deadline - 5:
                break

            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            a_opt = run_optimization_pipeline(a0, n, sub_deadline, seed=seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()

            if M_opt < best_M_n:
                best_M_n = M_opt
                best_for_n = a_opt.copy()
                J_est = 2 * n * best_M_n
                print(f"    init={ii}: M={best_M_n:.10f} J~{J_est:.6f}", flush=True)

        if best_for_n is not None:
            tracker.update(best_for_n, f"Phase1 n={n}")

    tracker.report("After Phase 1")

    # === PHASE 2: Intensive FW polish ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: Intensive FW polish", flush=True)
    print(f"{'='*60}", flush=True)

    t_p2_end = time.monotonic() + 300

    if tracker.best is not None and time.monotonic() < t_p2_end:
        n = tracker.best.size
        # Progressive active set expansion
        for n_act in [30, 50, 100, 200, 2 * n - 1]:
            if time.monotonic() > t_p2_end:
                break
            time_left = t_p2_end - time.monotonic()
            if time_left < 20:
                break
            a_fw = run_fw_lp(tracker.best, max_iters=150, deadline=time.monotonic() + time_left - 5,
                            n_active=min(n_act, 2 * n - 1))
            tracker.update(a_fw, f"FW2 active={n_act}")

    tracker.report("After Phase 2")

    # === PHASE 3: Deep coordinate descent ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Deep coordinate descent", flush=True)
    print(f"{'='*60}", flush=True)

    t_p3_end = time.monotonic() + 250

    if tracker.best is not None and time.monotonic() < t_p3_end:
        n = tracker.best.size
        max_sweeps = max(20, int((t_p3_end - time.monotonic()) / max(0.3, n / 100.0)))
        a_cd = coordinate_descent_efficient(tracker.best, max_sweeps, t_p3_end, seed=42)
        tracker.update(a_cd, f"CD n={n}")

        # Second CD pass with different seed
        if time.monotonic() < t_p3_end - 10:
            a_cd2 = coordinate_descent_efficient(tracker.best, max_sweeps // 2,
                                                t_p3_end - 5, seed=999)
            tracker.update(a_cd2, f"CD2 n={n}")

    tracker.report("After Phase 3")

    # === PHASE 4: Length exploration ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    t_p4_end = time.monotonic() + 250

    if tracker.best is not None and time.monotonic() < t_p4_end:
        n_cur = tracker.best.size
        # Try nearby lengths
        candidate_lengths = []
        for delta in [-128, -64, 64, 128, -256, 256]:
            nl = n_cur + delta
            if 192 <= nl <= 2048:
                nl = (nl // 32) * 32
                if nl != n_cur:
                    candidate_lengths.append(nl)

        # Also try some standard lengths
        for nl in [512, 640, 768, 896, 1024, 1280, 1536]:
            if nl != n_cur and nl not in candidate_lengths and 192 <= nl <= 2048:
                candidate_lengths.append(nl)

        candidate_lengths = sorted(set(candidate_lengths))

        for nl in candidate_lengths[:5]:  # Try at most 5
            if time.monotonic() > t_p4_end - 10:
                break

            a_ext = resample_seq(tracker.best, nl)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            # Quick pipeline
            sub_dl = min(t_p4_end - 5, time.monotonic() + 40)
            a_opt = run_optimization_pipeline(a_ext, nl, sub_dl, seed=seed + 500 + nl)
            tracker.update(a_opt, f"LenExp n={nl}")

    tracker.report("After Phase 4")

    # === PHASE 5: Final alternating polish ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 5: Final alternating polish", flush=True)
    print(f"{'='*60}", flush=True)

    t_final = deadline

    if tracker.best is not None:
        for rnd in range(30):
            if time.monotonic() > t_final - 5:
                break

            time_step = min(25.0, (t_final - 5 - time.monotonic()) / 3.0)
            if time_step < 5:
                break

            # FW step
            a_fw = run_fw_lp(tracker.best, max_iters=40, deadline=time.monotonic() + time_step,
                            n_active=min(60, 2 * tracker.best.size - 1))
            tracker.update(a_fw, f"Polish5 FW r{rnd}")

            # CD step
            a_cd = coordinate_descent_efficient(tracker.best, max_sweeps=5,
                                               deadline=time.monotonic() + time_step,
                                               seed=5000 + rnd)
            tracker.update(a_cd, f"Polish5 CD r{rnd}")

            # L-BFGS step (high tau for fine-tuning)
            x0 = np.log(np.maximum(tracker.best, 1e-10))
            a_lb = run_lbfgsb(x0, tau=500, maxiter=1000, max_time=time_step)
            if a_lb is not None:
                tracker.update(a_lb, f"Polish5 LB r{rnd}")

    tracker.report("After Phase 5")

    # Ensure valid return
    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_J = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_J:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
```
