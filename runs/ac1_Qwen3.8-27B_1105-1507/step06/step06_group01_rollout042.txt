        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-8,
                             'primal_feasibility_tolerance': 1e-8})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def run_lp_iteration(a, max_iters=200, deadline=None, time_per_lp=10):
    """Run LP best-response iterations with line search."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # If M has decreased since we started, update
        if M < best_M:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
        
        g = solve_lp_full(a, M, time_limit=time_per_lp)
        if g is None:
            break
        
        # Line search: find best t
        best_t = 0.0
        best_M_trial = M
        for t in [0.3, 0.5, 0.7, 1.0, 0.1, 0.2, 0.4, 0.6, 0.8, 0.9]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            st = a_t.sum()
            if st < 0.01:
                continue
            a_t = a_t / st
            Mt = fast_conv(a_t).max()
            if Mt < best_M_trial:
                best_M_trial = Mt
                best_t = t
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            no_improve += 5
            if no_improve > 30:
                break
    
    # Final update
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def coordinate_optimize(a, i):
    """Find optimal a[i] to minimize max convolution. Returns optimal value."""
    n = len(a)
    x = a[i]
    b = fast_conv(a)
    
    # Determine affected k values
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Precompute base values and coefficients
    # For k in [k_lo, k_hi]:
    #   if k == 2*i: (a*a)[k] = x^2 + base
    #   else if 0 <= k-i < n: (a*a)[k] = 2*x*a[k-i] + base
    # For k outside [k_lo, k_hi]: (a*a)[k] = base (constant)
    
    # Compute base = (a*a)[k] - contribution from a[i]
    bases = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        bases[k] = b[k]
        # Subtract contribution from a[i]
        if 0 <= k - i < n:
            if k == 2 * i:
                bases[k] -= x * x
            else:
                bases[k] -= 2.0 * x * a[k - i]
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = bases[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    def obj(xv):
        """Compute max conv value when a[i] = xv."""
        m = base_max
        for k in range(k_lo, k_hi + 1):
            if k == 2 * i:
                val = bases[k] + xv * xv
            else:
                j = k - i
                if 0 <= j < n:
                    val = bases[k] + 2.0 * xv * a[j]
                else:
                    val = bases[k]
            if val > m:
                m = val
        return m
    
    # Golden section search
    lo = 0.0
    hi = max(2.0 * x, 1.0, np.sqrt(max(b.max(), 0.0)) + 0.5)
    
    # Expand hi if needed
    for _ in range(20):
        if obj(hi) > obj(hi * 0.5):
            hi *= 2.0
        else:
            break
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    fc1 = obj(c1)
    fc2 = obj(c2)
    
    for _ in range(50):
        if fc1 < fc2:
            hi = c2
            c2 = c1
            fc2 = fc1
            c1 = hi - gr * (hi - lo)
            fc1 = obj(c1)
        else:
            lo = c1
            c1 = c2
            fc1 = fc2
            c2 = lo + gr * (hi - lo)
            fc2 = obj(c2)
    
    x_opt = (lo + hi) / 2.0
    # Check boundary
    if obj(0.0) <= obj(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coord_descent(a, max_sweeps, deadline, seed=0):
    """Run coordinate descent."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        b = fast_conv(a)
        sweep_changed = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            x_opt = coordinate_optimize(a, i)
            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
                sweep_changed = True
        
        if not sweep_changed:
            no_improve += 1
        else:
            no_improve = 0

            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
        
        if no_improve > 30:
            break
    
    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        # Resample parent
        if parent.size != n:
            t_old = np.linspace(0, 1, parent.size)
            p = np.maximum(np.interp(t, t_old, parent), 0.0)
        else:
            p = np.maximum(parent, 0.0).copy()
        ps = p.sum()
        if ps > 1e-10:
            inits.append(p / ps)
            # Perturbed versions
            for sigma in [0.01, 0.03, 0.05, 0.1, 0.2]:
                pn = np.maximum(p / ps + rng.normal(0, sigma, n), 0.0)
                psn = pn.sum()
                if psn > 1e-10:
                    inits.append(pn / psn)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine
    for freq in [1, 2, 3, 5]:
        c = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Gaussian
    g = np.exp(-2.0 * (t - 0.5) ** 2) + 0.05
    inits.append(g / g.sum())
    
    # U-shaped
    u = (2 * t - 1) ** 2 + 0.1
    inits.append(u / u.sum())
    
    # Random smooth
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    r = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    r = np.maximum(r, 0.01)
    inits.append(r / r.sum())
    
    # Flat-top with rounded edges
    w = np.ones(n)
    edge = n // 8
    for i in range(edge):
        frac = i / edge
        w[i] = frac
        w[n-1-i] = frac
    inits.append(w / w.sum())
    
    return inits

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv
    if not cond.any():
        theta = 0.0
    else:
        neg_idx = np.where(~cond)[0]
        if len(neg_idx) == 0:
            rho_val = n - 1
        else:
            rho_val = neg_idx[0]
        theta = (cv[rho_val] - 1.0) / (rho_val + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (A @ g)[k] = sum_j a[k-j]*g[j]."""
    conv_len = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(conv_len):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            ms = k - js
            mask = a[ms] > 1e-12
            if mask.any():
                rows.extend([k] * int(mask.sum()))
                cols.extend(js[mask].tolist())
                vals.extend(a[ms[mask]].tolist())
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
    return A


def solve_lp_full(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a@g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    if A is None:
        return None
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-8,
                             'primal_feasibility_tolerance': 1e-8})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def run_coord_descent(a_init, max_sweeps, deadline, seed=0):

    for sweep in range(max_sweeps):

            break

        b = fast_conv(a)
        sweep_changed = False

        for i in order:
            if time.monotonic() > deadline:
                break
            x_cur = a[i]

            mask_not_sq = affected != 2 * i

            if len(lin_ks) > 0:
                js = lin_ks - i
                valid = (js >= 0) & (js < n)
                lin_ks = lin_ks[valid]
                js = js[valid]
                coeffs = 2.0 * a[js]

            else:

                lin_bases = np.array([])

                base_max_unaff = b[all_mask].max()
            else:
                base_max_unaff = 0.0

            def obj_val(xv):
                if coeffs.size > 0:
                    vals = lin_bases + coeffs * xv
                    m = max(base_max_unaff, vals.max())
                else:
                    m = base_max_unaff
                qv = quad_base + xv * xv
                if qv > m:
                    m = qv
                return m

            lo = 0.0
            hi = max(2.0 * x_cur, 1.0)

                if obj_val(hi) > obj_val(mid) and hi < 50.0:

            c2 = lo + gr * (hi - lo)
            fc1 = obj_val(c1)
            fc2 = obj_val(c2)

                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    fc1 = obj_val(c1)
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    fc2 = obj_val(c2)

            x_opt = (lo + hi) / 2.0
            if obj_val(0.0) <= obj_val(x_opt):
                x_opt = 0.0

            if abs(x_opt - x_cur) > 1e-10:
                a[i] = max(0.0, x_opt)
                sweep_changed = True

        if not sweep_changed:
            no_improve += 1
        else:
            no_improve = 0

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()

        if no_improve > 40:
            break

    return best_a


def spectral_init(n, rng, iters=50):
    """Gerchberg-Saxton style: try to flatten the circular autocorrelation."""
    a = np.ones(n) / n
    for _ in range(iters):
        size = 1
        while size < 2 * n:
            size <<= 1
        A = np.fft.rfft(a, size)
        mag2 = np.abs(A) ** 2
        mag2 = np.maximum(mag2, 1e-20)
        mag_target = mag2 ** 0.25
        mag_target = mag_target / np.sqrt(mag_target.sum()) if mag_target.sum() > 0 else 1.0
        phase = np.exp(1j * np.angle(A))
        A_new = mag_target * phase
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a = a_new / s
    return a


def smooth_max_grad(a, tau):
    """Compute smoothed max of conv(a,a) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted - shifted.max())
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau

    grad = np.zeros(n)
    for i in range(n):
        idx = np.arange(i, min(i + n, 2 * n - 1))
        grad[i] = 2.0 * np.dot(p[idx], a[:len(idx)])
    return smooth_val, grad


def run_fista_smooth(a_init, deadline, max_iters=2000, lr=0.15, tau_start=100, tau_end=500):
    """FISTA on log-sum-exp smoothed objective."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_M = fast_conv(a).max()
    t_param = 1.0
    y = a.copy()
    a_prev = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        _, grad_y = smooth_max_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        step = lr / (1.0 + 0.01 * it)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        alpha = (t_param - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        a_prev = a
        a = a_new
        y = proj_simplex(y_new)
        t_param = t_new

        if it % 10 == 0:
            M_chk = fast_conv(a).max()
            if M_chk < best_M:
                best_M = M_chk
                best_a = a.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        if parent.size != n:
            p = resample_seq(parent, n)
        else:
            p = np.maximum(parent, 0.0).copy()
        ps = p.sum()
        if ps > 1e-10:
            p_norm = p / ps
            inits.append(p_norm.copy())
            for sigma in [0.02, 0.05, 0.1, 0.15, 0.25]:
                pn = np.maximum(p_norm + rng.normal(0, sigma, n), 0.0)
                pns = pn.sum()
                if pns > 1e-10:
                    inits.append(pn / pns)

    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3]:
        c = 0.3 + 0.7 * np.cos(2 * np.pi * freq * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    g = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(g / g.sum())

    u = (2 * t - 1) ** 2 + 0.1
    inits.append(u / u.sum())

    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    r = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    r = np.maximum(r, 0.01)
    inits.append(r / r.sum())

    spec = spectral_init(n, rng, iters=30)
    inits.append(spec.copy())

    w = np.ones(n)
    edge = max(1, n // 10)
    for i in range(edge):
        frac = (i + 1) / edge
        w[i] = frac
        w[n - 1 - i] = frac
    inits.append(w / w.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining LP-FISTA, coordinate descent, FISTA-smooth,
    and perturbation-based local minimum escape across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    deadline = t_global_start + float(budget_s) - 20.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size

    lengths = list(dict.fromkeys([n_parent,
                                   max(256, n_parent - 128),
                                   max(256, n_parent - 64),
                                   min(2048, n_parent + 64),
                                   min(2048, n_parent + 128),
                                   min(2048, n_parent + 256)]))
    lengths.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    lengths = lengths[:5]
    print(f"Lengths: {lengths}", flush=True)

    # === Phase 1: Multi-length, multi-init search ===
    print(f"\n{'='*60}\nPHASE 1: Multi-length search\n{'='*60}", flush=True)

    reserved = 200
    time_p1 = max(60, deadline - t_global_start - reserved)
    time_per_len = time_p1 / len(lengths)

    for li, n in enumerate(lengths):
        rem = deadline - t_global_start - reserved
        if rem < 30:
            break
        alloc = min(time_per_len * (1.5 if n == n_parent else 1.0), rem / (len(lengths) - li))
        alloc = max(alloc, 20)
        print(f"\n  n={n}, alloc={alloc:.0f}s", flush=True)

        inits = generate_inits(n, rng, parent)
        n_inits = min(len(inits), 8)
        tpi = alloc / n_inits
        best_n = None
        best_M_n = float('inf')

        for ii in range(n_inits):
            sub_dl = min(t_global_start + t_global_start * 0 + time.monotonic() + tpi, deadline - reserved)
            if time.monotonic() >= sub_dl:
                break
            a0 = np.maximum(inits[ii], 1e-8)
            a0 = a0 / a0.sum()

            t_lp = min(sub_dl, time.monotonic() + 0.5 * tpi)
            a1 = run_lp_fista(a0, max_iters=200, deadline=t_lp, time_per_lp=min(10, 0.3 * tpi))
            M1 = fast_conv(a1).max()

            t_cd = min(sub_dl, time.monotonic() + 0.35 * tpi)
            a2 = run_coord_descent(a1, max_sweeps=80, deadline=t_cd, seed=seed + ii * 7)
            M2 = fast_conv(a2).max()

            t_lp2 = min(sub_dl, time.monotonic() + 0.15 * tpi)
            a3 = run_lp_fista(a2, max_iters=50, deadline=t_lp2, time_per_lp=5, use_momentum=False)
            M3 = fast_conv(a3).max()

            for aa, MM in [(a1, M1), (a2, M2), (a3, M3)]:
                if MM < best_M_n:
                    best_M_n = MM
                    best_n = aa.copy()
                if MM < best_M_n:
                    pass
            print(f"    init {ii}: best_M={best_M_n:.10f} J_est={2.0*n*best_M_n:.6f}", flush=True)

        if best_n is not None:
            tracker.update(best_n)

    tracker.report("After Phase 1")

    # === Phase 2: Intensive polish ===
    print(f"\n{'='*60}\nPHASE 2: Intensive LP polish\n{'='*60}", flush=True)
    if tracker.best is not None and time.monotonic() < deadline - 40:
        t2 = min(deadline - 20, time.monotonic() + 100)
        a_p = run_lp_fista(tracker.best, max_iters=300, deadline=t2, time_per_lp=8, use_momentum=True)
        tracker.update(a_p)

        if time.monotonic() < deadline - 15:
            t2b = min(deadline - 10, time.monotonic() + 50)
            a_p2 = run_coord_descent(a_p, max_sweeps=150, deadline=t2b, seed=12345)
            tracker.update(a_p2)

        if time.monotonic() < deadline - 8:
            t2c = min(deadline - 5, time.monotonic() + 40)
            a_p3 = run_lp_fista(a_p2, max_iters=100, deadline=t2c, time_per_lp=6)
            tracker.update(a_p3)

    tracker.report("After Phase 2")

    # === Phase 3: FISTA smooth refinement ===
    print(f"\n{'='*60}\nPHASE 3: FISTA smooth refinement\n{'='*60}", flush=True)
    if tracker.best is not None and time.monotonic() < deadline - 30:
        for tau_s in [200, 400]:
            if time.monotonic() > deadline - 20:
                break
            t3 = min(deadline - 15, time.monotonic() + 40)
            a_f = run_fista_smooth(tracker.best, deadline=t3, max_iters=3000, lr=0.1, tau_start=tau_s, tau_end=tau_s * 2)
            tracker.update(a_f)

    tracker.report("After Phase 3")

    # === Phase 4: Perturbation escape ===
    print(f"\n{'='*60}\nPHASE 4: Perturbation escape\n{'='*60}", flush=True)
    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_b = tracker.best.size
        a_norm = tracker.best / tracker.best.sum()
        for trial in range(6):
            if time.monotonic() > deadline - 15:
                break
            sigma = 0.03 + 0.03 * trial
            pert = np.maximum(a_norm + rng.normal(0, sigma, n_b), 0.0)
            pert = pert / pert.sum()
            sub_dl = min(deadline - 10, time.monotonic() + 15)
            a_pt = run_lp_fista(pert, max_iters=60, deadline=sub_dl, time_per_lp=5)
            tracker.update(a_pt)

    tracker.report("After Phase 4")

    # === Phase 5: Final CD + LP ===
    print(f"\n{'='*60}\nPHASE 5: Final polish\n{'='*60}", flush=True)
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t5 = min(deadline - 3, time.monotonic() + 30)
        a_fin = run_coord_descent(tracker.best, max_sweeps=50, deadline=t5, seed=77777)
        tracker.update(a_fin)
        if time.monotonic() < deadline - 2:
            t5b = min(deadline - 1, time.monotonic() + 20)
            a_fin2 = run_lp_fista(a_fin, max_iters=30, deadline=t5b, time_per_lp=5)
            tracker.update(a_fin2)

    tracker.report("After Phase 5")

    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)

    total_time = time.monotonic() - t_global_start
    print(f"\n{'='*60}\nFINAL: J={tracker.best_score:.8f} n={tracker.best.size} t={total_time:.0f}s\n{'='*60}", flush=True)
    return [float(x) for x in tracker.best]
