        momentum = min(0.99, momentum * 0.9995 + 0.001)
        y = a + momentum * (a - y + a * (1 - momentum))  # simplified momentum
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-12:
            y = a.copy()
        else:
            y = y / sy
    return best_a


def run_nesterov_simple(a_init, deadline, max_iters=10000,
                        tau_start=3.0, tau_max=3000.0):
    """Simpler Nesterov with fixed schedule (reliable fallback)."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    y = a.copy()
    beta_nest = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_max / tau_start) ** (frac ** 0.7)
        lr = 0.4 * (0.005 / 0.4) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        a_new = np.maximum(y - lr * (grad + beta_nest * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_nest * beta_nest)) / 2.0
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-12:
            y = a_new
        else:
            y = y / sy
        if b_max < best_M:
            best_M = b_max
            best_a = a_new.copy()
        a = a_new
    return best_a


def build_A_sparse(n, a, active_ks=None):
    """Build sparse matrix for LP constraints. If active_ks given, use only those."""
    rows = []
    cols = []
    vals = []
    if active_ks is None:
        ks = range(2 * n - 1)
    else:
        ks = active_ks
    for k in ks:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(float(a[k - j]))
    n_constraints = 2 * n - 1
    A = csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))
    return A


def lp_best_response(a, M, active_ks=None):
    """Solve the best-response LP."""
    n = len(a)
    A = build_A_sparse(n, a, active_ks)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'time_limit': 30})
        if r.success and r.x is not None:
            x = np.maximum(r.x, 0)
            if x.sum() > 1e-12:
                return x
    except Exception:
        pass
    return None


def anderson_mix(history_x, history_Fx, m, extrapolate=1.0):
    """Anderson mixing with optional extrapolation."""
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = float(ones @ G_inv_ones)
        if abs(denom) < 1e-14:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    # Allow mild extrapolation
    alpha = np.clip(alpha, -extrapolate, 1.0 + extrapolate)
    s = alpha.sum()
    if abs(s) < 1e-14:
        alpha = ones / m
        s = 1.0
    alpha = alpha / s
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def run_lp_anderson(a_init, deadline, max_iters=500, window=7,
                    t_base=0.4, t_decay=0.995, t_min=0.02):
    """LP mirror descent with Anderson acceleration and adaptive step size."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    hist_x = []
    hist_Fx = []
    t = t_base
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_imp = 0
            t = min(t_base, t * 1.2)  # increase step on improvement
        else:
            no_imp += 1
            t *= t_decay
            if no_imp > 40:
                break
        # Find active constraints (peaks within 2% of max)
        threshold = M * 0.98
        active_ks = list(np.where(b >= threshold)[0])
        if len(active_ks) > 200:
            active_ks = list(np.where(b >= M * 0.95)[0])
        g = lp_best_response(a, M, active_ks if len(active_ks) < (2 * n - 1) else None)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-12:
            break
        g = g / gs
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        sa = a_new.sum()
        if sa < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / sa
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        if len(hist_x) >= 4:
            a_and = anderson_mix(hist_x, hist_Fx, len(hist_x), extrapolate=0.5)
            saa = a_and.sum()
            if saa > 1e-12:
                a_and = a_and / saa
            M_and = compute_M(a_and)
            M_new = compute_M(a_new)
            if M_and < M_new:
                a = a_and
            else:
                a = a_new
        else:
            a = a_new
    return best_a


def run_lp_backtrack(a_init, deadline, max_iters=300):
    """LP mirror descent with backtracking step size (no Anderson)."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break
        threshold = M * 0.97
        active_ks = list(np.where(b >= threshold)[0])
        g = lp_best_response(a, M, active_ks if len(active_ks) < (2 * n - 1) else None)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-12:
            break
        g = g / gs
        improved = False
        for t in [0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            st = a_t.sum()
            if st < 1e-12:
                continue
            a_t = a_t / st
            M_t = compute_M(a_t)
            if M_t < best_M - 1e-15:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if not improved:
            no_imp += 1
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_perturbations(a, rng, n_pert=5):
    """Generate diverse perturbations of a solution."""
    n = len(a)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    perts = [a.copy()]
    # Symmetrized
    sym = (a + a[::-1]) / 2.0
    perts.append(np.maximum(sym, 1e-10))
    # Small noise
    perts.append(np.maximum(a + rng.normal(0, 0.01, n), 1e-10))
    # Medium noise
    perts.append(np.maximum(a + rng.normal(0, 0.03, n), 1e-10))
    # Scaled regions
    for _ in range(n_pert - 3):
        lo = rng.integers(0, n - 50)
        hi = lo + rng.integers(10, min(100, n - lo))
        scale = rng.uniform(0.7, 1.3)
        pert = a.copy()
        pert[lo:hi] *= scale
        perts.append(np.maximum(pert, 1e-10))
    for p in perts:
        s = p.sum()
        if s > 1e-10:
            p /= s
    return perts


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
    except (NameError, Exception):
        pass

    if parent is None:
        parent = np.ones(1024)

    parent = parent / parent.sum()
    n_parent = parent.size
    J_parent = compute_J(parent)
    print(f"Parent: n={n_parent}, J={J_parent:.8f}", flush=True)

    best_seq = parent.copy()
    best_J = J_parent
    best_n = n_parent

    def update_best(a):
        nonlocal best_seq, best_J, best_n
        a = np.clip(a, 0, 1000)
        if a.size == 0 or a.sum() < 0.01:
            return
        J = compute_J(a)
        if np.isfinite(J) and J < best_J:
            best_J = J
            best_seq = a.copy()
            best_n = a.size
            el = time.monotonic() - t_start
            print(f"  ** J={J:.8f} n={a.size} t={el:.0f}s", flush=True)

    update_best(parent)

    # === PHASE 1: Smooth optimization at parent length (350s) ===
    p1_dl = t_start + 350.0
    print(f"\nPhase 1: Smooth Nesterov at n={n_parent} (budget 350s)", flush=True)

    # Try parent and a few perturbations
    perts = make_perturbations(parent, rng, n_pert=4)
    for idx, p in enumerate(perts):
        sub_dl = min(p1_dl, time.monotonic() + (p1_dl - time.monotonic()) / (len(perts) - idx))
        if time.monotonic() > sub_dl:
            break
        if idx == 0:
            a_opt = run_nesterov_simple(p, sub_dl, max_iters=15000, tau_start=3.0, tau_max=3000.0)
        else:
            a_opt = run_nesterov_simple(p, sub_dl, max_iters=8000, tau_start=5.0, tau_max=2000.0)
        update_best(a_opt)

    # Also try adaptive tau version on best so far
    if time.monotonic() < p1_dl - 30:
        a_opt = run_nesterov_bt(best_seq, p1_dl - 5, max_iters=10000, tau_start=10.0, tau_max=5000.0)
        update_best(a_opt)

    print(f"Phase 1 done: J={best_J:.8f} (t={time.monotonic()-t_start:.0f}s)", flush=True)

    # === PHASE 2: LP mirror descent with Anderson (600s) ===
    p2_dl = t_start + 950.0
    print(f"\nPhase 2: LP+Anderson at n={best_n} (budget 600s)", flush=True)

    a_lp = run_lp_anderson(best_seq, min(p2_dl, time.monotonic() + 400),
                           max_iters=500, window=7, t_base=0.4, t_decay=0.995, t_min=0.02)
    update_best(a_lp)

    # Second LP pass with different params
    if time.monotonic() < p2_dl - 60:
        a_lp2 = run_lp_anderson(best_seq, min(p2_dl, time.monotonic() + 180),
                                max_iters=300, window=5, t_base=0.25, t_decay=0.997, t_min=0.01)
        update_best(a_lp2)

    # LP backtracking pass
    if time.monotonic() < p2_dl - 20:
        a_lp3 = run_lp_backtrack(best_seq, p2_dl - 5, max_iters=200)
        update_best(a_lp3)

    print(f"Phase 2 done: J={best_J:.8f} (t={time.monotonic()-t_start:.0f}s)", flush=True)

    # === PHASE 3: Multi-length exploration (300s) ===
    p3_dl = t_start + 1250.0
    print(f"\nPhase 3: Multi-length exploration (budget 300s)", flush=True)

    try_lengths = []
    for delta in [64, -64, 128, -128, 192, -192, 96, -96]:
        nl = best_n + delta
        if 384 <= nl <= 2048 and nl != best_n:
            try_lengths.append(nl)
    # Remove duplicates, keep sorted by proximity to best_n
    try_lengths = sorted(set(try_lengths), key=lambda x: abs(x - best_n))
    try_lengths = try_lengths[:4]  # Limit to 4 lengths

    for nl in try_lengths:
        if time.monotonic() > p3_dl - 30:
            break
        a_resample = resample_seq(best_seq, nl)
        a_resample = np.maximum(a_resample, 1e-8)
        a_resample = a_resample / a_resample.sum()
        # Quick smooth
        sub_dl = min(time.monotonic() + 40, p3_dl - 60)
        a_s = run_nesterov_simple(a_resample, sub_dl, max_iters=3000, tau_start=5.0, tau_max=1500.0)
        update_best(a_s)
        # Quick LP
        if time.monotonic() < p3_dl - 30:
            a_l = run_lp_anderson(a_s, min(time.monotonic() + 40, p3_dl - 20),
                                  max_iters=80, window=5, t_base=0.3, t_decay=0.99, t_min=0.02)
            update_best(a_l)

    print(f"Phase 3 done: J={best_J:.8f} (t={time.monotonic()-t_start:.0f}s)", flush=True)

    # === PHASE 4: Final intensive LP polish on overall best (200s) ===
    p4_dl = deadline
    print(f"\nPhase 4: Final LP polish at n={best_n} (budget 200s)", flush=True)

    # Multiple LP passes with decreasing step sizes
    for t_b in [0.3, 0.15, 0.08, 0.04]:
        if time.monotonic() > p4_dl - 15:
            break
        sub_budget = min(50.0, (p4_dl - time.monotonic()) * 0.4)
        a_f = run_lp_anderson(best_seq, time.monotonic() + sub_budget,
                              max_iters=100, window=5, t_base=t_b, t_decay=0.995, t_min=0.005)
        update_best(a_f)

    # Final backtracking LP
    if time.monotonic() < p4_dl - 10:
        a_f = run_lp_backtrack(best_seq, p4_dl - 3, max_iters=100)
        update_best(a_f)

    # Very final: try a smooth restart from best
    if time.monotonic() < p4_dl - 20:
        a_smooth = run_nesterov_simple(best_seq, p4_dl - 5, max_iters=5000,
                                       tau_start=50.0, tau_max=5000.0)
        update_best(a_smooth)
        if time.monotonic() < p4_dl - 5:
            a_lp_final = run_lp_anderson(best_seq, p4_dl, max_iters=30,
                                         window=4, t_base=0.1, t_decay=0.99, t_min=0.005)
            update_best(a_lp_final)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best_J:.8f} n={best_n} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    # Verify with evaluate_sequence
    final_attempt = best_seq.tolist()
    eval_score = evaluate_sequence(final_attempt)
    if np.isfinite(eval_score) and eval_score < best_J:
        print(f"  evaluate_sequence gives {eval_score:.8f} (using this)", flush=True)
        best_J = eval_score

    if best_seq is None or best_seq.size == 0:
        return [1.0] * 256

    return [float(x) for x in best_seq]
