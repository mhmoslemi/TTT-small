```python
def multiplicative_flatten(a, eta=0.1, p=1.0):
    a = np.maximum(a, 1e-10)
    b = fast_conv(a)
    M = b.max()
    # For each i, compute its "contribution" to the peaks
    # contribution[i] = sum over active k of p_k * a[k-i] (where the product is a[i]*a[k-i])
    n = len(a)
    active = np.where(b > M * 0.9)[0]  # near-active constraints
    contrib = np.zeros(n)
    for k in active:
        p_k = (b[k] / M) ** 2  # weight by how close to max
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # conv[k] = sum_{i=lo}^{hi} a[i] * a[k-i]
        # d(conv[k])/d(a[i]) = a[k-i] for i in [lo, hi]
        contrib[lo:hi+1] += p_k * a[k - (np.arange(lo, hi+1))]
    # Multiplicative update: reduce elements that contribute to peaks
    a_new = a * (1.0 + eta * (1.0 - contrib / (contrib.mean() + 1e-15))) ** p
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new
```

```python
3. Anderson acceleration with adaptive window

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lp_truncated(a, M, num_constraints):
    """Solve LP best response using only the top num_constraints active constraints."""
    n = len(a)
    b = fast_conv(a)
    # Find the top constraints
    top_idx = np.argsort(b)[-num_constraints:]
    top_idx = np.sort(top_idx)
    
    # Build sparse constraint matrix for just these constraints
    rows = []
    cols = []
    vals = []
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    # Actually, for speed, let's build the full matrix but only use top constraints
    # The matrix A has A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # For truncated: only rows in top_idx
    
    # For efficiency, build sparse matrix directly
    A = lil_matrix((len(top_idx), n))
    for ii, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[ii, j] = a[k - j]
    A = A.tocsr()
    
    c = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    bounds = [(0, None)] * n
    
    try:
```

For the fixed-point iteration x_{k+1} = F(x_k), Anderson acceleration maintains a history of (x_k, F(x_k)) pairs and computes the next point as:
x_{k+1} = sum_i alpha_i * F(x_i)
where alpha minimizes ||sum_i alpha_i * (x_i - F(x_i))||^2 subject to sum alpha_i = 1.

```python
def anderson_step(hist_x, hist_F, window=5):
    """Anderson acceleration step.
    hist_x: list of x_k values
    hist_F: list of F(x_k) values
    Returns the accelerated next iterate.
    """
    m = min(window, len(hist_x))
    if m < 2:
        return hist_F[-1]
    
    # Get the last m pairs
    xs = np.array(hist_x[-m:])  # (m, n)
    Fs = np.array(hist_F[-m:])  # (m, n)
    
    # Residuals
    Rs = xs - Fs  # (m, n)
    
    # Solve: min ||R^T alpha||^2 s.t. 1^T alpha = 1
    # G = R R^T (m x m)
    G = Rs @ Rs.T
    G += 1e-10 * np.eye(m)  # regularization
    
    # Solve KKT system:
    # [G  1] [alpha]   [0]
    # [1^T 0] [beta ] = [1]
    KKT = np.zeros((m+1, m+1))
    KKT[:m, :m] = G
    KKT[:m, m] = 1
    KKT[m, :m] = 1
    rhs = np.zeros(m+1)
    rhs[m] = 1
    
    try:
        sol = np.linalg.solve(KKT, rhs)
        alpha = sol[:m]
    except np.linalg.LinAlgError:
        alpha = np.ones(m) / m
    
    # Clamp alpha for stability (allow mild extrapolation)
    alpha = np.clip(alpha, -0.5, 2.0)
    alpha = alpha / alpha.sum()
    
    # Mix the F values
    x_next = Fs.T @ alpha  # sum_i alpha_i * F(x_i)
    return np.maximum(x_next, 0)
```

```python
                    (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                   shape=(m, n))

    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def solve_lp_full(a, M, row_ranges, n, time_limit=10):
    """Solve LP with all constraints."""
    active_idx = list(range(2 * n - 1))
    return solve_lp_with_active(a, M, active_idx, row_ranges, n, time_limit)


def anderson_accelerate(hist_x, hist_F, window=5):
    """Anderson acceleration for fixed-point iteration x = F(x)."""
    m = min(window, len(hist_x))
    if m < 3:
        return hist_F[-1]

    xs = np.array(hist_x[-m:])
    Fs = np.array(hist_F[-m:])
    Rs = xs - Fs

    G = Rs @ Rs.T
    G += 1e-12 * np.eye(m)

    KKT = np.zeros((m + 1, m + 1))
    KKT[:m, :m] = G
    KKT[:m, m] = 1.0
    KKT[m, :m] = 1.0
    rhs = np.zeros(m + 1)
    rhs[m] = 1.0

    try:
        sol = np.linalg.solve(KKT, rhs)
        alpha = sol[:m]
    except np.linalg.LinAlgError:
        alpha = np.ones(m) / m

    alpha = np.clip(alpha, -0.5, 2.0)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = np.ones(m) / m
    else:
        alpha /= s

    x_next = Fs.T @ alpha
    return np.maximum(x_next, 0)


def subgradient_step(a, b, M, n, lr):
    """Take a subgradient step on max(conv(a,a)) using the active peaks."""
    active = np.where(b >= M * 0.99)[0]
    p = np.zeros(2 * n - 1)
    p[active] = 1.0 / len(active)
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        grad[js] += p[k] * a[k - js]
    a_new = np.maximum(a - lr * grad, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def run_lp_mirror_descent(a, deadline, row_ranges, n, max_iters=10000,
                          init_active=80, max_active=400, grow_every=5,
                          t_init=0.3, t_min=0.01, anderson_window=6):
    """Main LP mirror descent loop with adaptive active set and Anderson acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    best_a = a.copy()
    best_b = fast_conv(a)
    best_M = float(best_b.max())

    b = fast_conv(a)
    M = float(b.max())
    active_idx = sorted(np.argsort(b)[-init_active:].tolist())

    hist_x = []
    hist_F = []
    no_improve = 0
    it = 0

    while it < max_iters and time.monotonic() < deadline:
        it += 1

        # Solve LP
        g = solve_lp_with_active(a, M, active_idx, row_ranges, n,
                                 time_limit=max(1, min(5, (deadline - time.monotonic()) * 0.01)))
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Line search: try several step sizes
        t_candidates = [t_init / (1 + 0.02 * it)]
        t_candidates = list(set([max(t_min, max(0.005, t_init / (1 + 0.02 * it))),
                                 max(t_min, 0.1), max(t_min, 0.2),
                                 max(t_min, 0.05), max(t_min, 0.15)]))
        t_candidates = sorted(t_candidates, reverse=True)

        best_t = t_min
        best_val = float('inf')
        best_a_try = None

        for t in t_candidates[:4]:
            a_try = np.maximum((1 - t) * a + t * g, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            M_try = fast_conv(a_try).max()
            if M_try < best_val:
                best_val = M_try
                best_t = t
                best_a_try = a_try

        if best_a_try is None:
            a_try = np.maximum((1 - t_min) * a + t_min * g, 0)
            s = a_try.sum()
            if s < 1e-10:
                break
            a_try /= s
            best_a_try = a_try
            best_val = fast_conv(a_try).max()

        # Anderson acceleration
        hist_x.append(a.copy())
        hist_F.append(best_a_try.copy())
        if len(hist_x) > anderson_window + 1:
            hist_x.pop(0)
            hist_F.pop(0)

        if len(hist_x) >= 4:
            a_anderson = anderson_accelerate(hist_x, hist_F, anderson_window)
            sa = a_anderson.sum()
            if sa > 1e-10:
                a_anderson /= sa
                M_and = fast_conv(a_anderson).max()
                if M_and <= best_val * 1.02:
                    best_a_try = a_anderson
                    best_val = M_and

        a = best_a_try

        # Track best
        if best_val < best_M * (1 - 1e-12):
            best_M = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

        # Update active set periodically
        if it % grow_every == 0:
            b = fast_conv(a)
            M = float(b.max())
            # Grow active set: use top constraints plus any that are close to max
            threshold = M - (M - b.min()) * 0.1
            candidate_idx = np.where(b >= threshold)[0]
            # Also include the current active set
            all_idx = set(active_idx)
            all_idx.update(candidate_idx.tolist())
            # Keep at most max_active, prioritizing those closest to max
            if len(all_idx) > max_active:
                sorted_idx = sorted(all_idx, key=lambda k: -b[k])
                active_idx = sorted_idx[:max_active]
            else:
                active_idx = sorted(all_idx)

    return best_a


def run_full_lp_polish(a, deadline, row_ranges, n, max_iters=200):
    """Polish with full-constraint LP (all 2n-1 constraints)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = solve_lp_full(a, M, row_ranges, n, time_limit=max(1, int(deadline - time.monotonic())))
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        for t in [0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002]:
            a_t = np.maximum((1 - t) * a + t * g, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-15:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break

    return best_a


def run_subgradient_polish(a, deadline, n, max_iters=5000):
    """Subgradient method with Polyak-Ruppert averaging."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()

    # Running average for Polyak-Ruppert
    a_avg = a.copy()
    count = 0
    lr = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

        # Subgradient step
        a = subgradient_step(a, b, M, n, lr)
        a_avg = a_avg + (a - a_avg) / (count + 1)
        count += 1

        # Decrease learning rate
        lr = 0.1 / (1 + 0.01 * it)
        if lr < 0.001:
            lr = 0.001

        # Check averaged iterate
        if it % 100 == 0 and count > 100:
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_M = M_avg
                best_a = a_avg.copy()

    return best_a


def gs_flat_init(n, rng, n_iters=200, beta=0.85):
    """Gerchberg-Saxton initialization for flat autocorrelation."""
    a = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    a /= a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    for _ in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        power = 0.5 + 0.5 * beta
        mag_p = mag ** power
        mag_new = (1 - beta) * mag + beta * mag.mean()
        mag_p = mag_p / (mag_p.max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_p
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a = np.maximum(a_new, 1e-6)
        a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1470):
```
