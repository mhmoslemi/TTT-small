    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a

def solve_lp_active(a, M, active_indices):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_indices, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in active_indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    if not rows:
        return np.ones(n) / n
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.zeros(2 * n - 1)
    for k in active_indices:
        b_ub[k] = M
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def get_active_indices(b, M, epsilon_frac):
    """Get indices where b[k] >= M * (1 - epsilon_frac)."""
    threshold = M * (1.0 - epsilon_frac)
    return np.where(b >= threshold)[0]

def run_lp_active_descent(a, deadline, max_iters=500,

        if g is None:
            # Fallback: use full LP or just break

            if g is None:
                break
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        t = t_start * (t_end / t_start) ** frac
        best_t = t

            M_try = fast_conv(a_try).max()

                best_t = t_try

        s = a.sum()
        if s < 1e-10:

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def gs_flat_spectral(a, n_iters=200, beta=0.85):
    """Gerchberg-Saxton for flattening the convolution spectrum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress magnitude to flatten
        mag_max = mag.max() + 1e-30
        mag_mean = mag.mean() + 1e-30
        # Soft compression
        power = 0.3 + 0.3 * beta
        mag_comp = (mag / mag_max) ** power * mag_max
        mag_final = (1 - beta) * mag + beta * mag_comp
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 1e-10)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new
    return best_a

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        seg = a[k - j_hi: k - j_lo + 1][::-1]
        grad[j_lo:j_hi + 1] += 2.0 * p[k] * seg
    return grad, p, b, b_max

def solve_lp_active(a, M, active_indices, time_limit=10):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in active_indices:
        j_lo = max(0, k - n + 1)

    return np.where(b >= threshold)[0]


def run_lp_momentum(a, deadline, max_iters=3000,
                    epsilon_start=0.3, epsilon_end=0.0005,
                    t_start=0.35, t_end=0.005,
                    momentum=0.5, verify_every=50):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 80:
                break
        
        frac = it / max_iters
        epsilon = epsilon_start * (epsilon_end / epsilon_start) ** (frac ** 0.8)
        
        active = get_active_indices(b, M, epsilon)
        if len(active) < 5:
            active = get_active_indices(b, M, max(epsilon * 5, 0.02))
        if len(active) < 3:
            active = np.arange(2 * n - 1)
        
        # Occasional full LP verification
        if verify_every > 0 and (it + 1) % verify_every == 0:
            g = solve_lp_active(a, M, np.arange(2 * n - 1), time_limit=20)
        else:
            g = solve_lp_active(a, M, active, time_limit=5)
        
        if g is None:
            g = solve_lp_active(a, M, np.arange(2 * n - 1), time_limit=20)
            if g is None:
                break
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Momentum step
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 1e-10)
        y = y / y.sum()
        
        # Line search on step size
        t_base = t_start * (t_end / t_start) ** frac
        best_t = t_base
        best_M_t = fast_conv(np.maximum((1 - t_base) * y + t_base * g, 0)).max()
        
        candidates = [t_base * 0.5, t_base, t_base * 1.5, t_base * 2.0]
        for t_try in candidates:
            if t_try <= 0 or t_try > 1.0:
                continue
            a_try = np.maximum((1 - t_try) * y + t_try * g, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try /= s_try
            M_try = fast_conv(a_try).max()
            if M_try < best_M_t:
                best_M_t = M_try
                best_t = t_try
        
        a_prev = a.copy()
        a = np.maximum((1 - best_t) * y + best_t * g, 0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
        else:
            a /= s
    
    return best_a


def run_lp_full_polish(a, deadline, max_iters=200, t_start=0.15, t_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        
        g = solve_lp_active(a, M, np.arange(2 * n - 1), time_limit=30)
        if g is None:
            break
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        frac = it / max_iters
        t = t_start * (t_end / t_start) ** frac
        
        a_new = np.maximum((1 - t) * a + t * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            a = best_a.copy()
        else:
            a = a_new / s
    
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gs_flat_spectral(a, n_iters=200, beta=0.85):
    a = np.maximum(a.astype(np.float64), 1e-10)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_max = mag.max() + 1e-30
        power = 0.3 + 0.4 * beta
        mag_comp = (mag / mag_max) ** power * mag_max
        mag_final = (1 - beta) * mag + beta * mag_comp
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 1e-10)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new
    return best_a


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(1024)
        best.update(parent)

    n = parent.size

    # === PHASE 1: Initialization (120s) ===
    p1_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Initialization (n={n})", flush=True)
    print(f"{'='*50}", flush=True)

    a_init = np.maximum(parent, 1e-8)
    a_init /= a_init.sum()

    # GS from parent
    a_gs = gs_flat_spectral(a_init, n_iters=200, beta=0.88)
    best.update(a_gs)

    # Nesterov smooth from GS
    a_sm = run_nesterov_smooth(a_gs, time.monotonic() + 60,
                               max_iters=3000, tau_start=2, tau_end=400,
                               lr_init=0.35, lr_end=0.003)
    best.update(a_sm)

    # Try diverse inits
    inits = [a_sm.copy()]
    for trial in range(3):
        t = np.linspace(0, 1, n, endpoint=False)
        if trial == 0:
            a_try = np.maximum(0.3 + 0.7 * np.sin(np.pi * t) + rng.normal(0, 0.1, n), 0.01)
        elif trial == 1:
            a_try = np.maximum(0.2 + 0.8 * (1 - (2 * t - 1) ** 2) + rng.normal(0, 0.15, n), 0.01)
        else:
            a_try = np.maximum(rng.uniform(0.2, 1.0, n), 0.05)
            kern = np.ones(max(3, n // 50)) / max(3, n // 50)
            a_try = np.convolve(a_try, kern, 'same')

        a_try = np.maximum(a_try, 1e-8)
        a_try /= a_try.sum()
        a_gs2 = gs_flat_spectral(a_try, n_iters=100, beta=0.9)
        a_sm2 = run_nesterov_smooth(a_gs2, time.monotonic() + 15,
                                    max_iters=800, tau_start=5, tau_end=300,
                                    lr_init=0.25, lr_end=0.01)
        inits.append(a_sm2)
        best.update(a_sm2)

    # Pick best init
    best_init = min(inits, key=lambda x: fast_conv(x).max() / x.sum() ** 2)
    best.update(best_init)
    a_current = best_init.copy()

    print(f"Phase 1: {time.monotonic()-p1_start:.1f}s, J={best.score:.6f}", flush=True)

    # === PHASE 2: Active-set LP (main, ~900s) ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: Active-Set LP Descent", flush=True)
    print(f"{'='*50}", flush=True)

    # Estimate LP time
    b = fast_conv(a_current)
    M_est = float(b.max())
    active_est = get_active_indices(b, M_est, 0.15)
    t0_lp = time.monotonic()
    _ = solve_lp_active(a_current, M_est, active_est, time_limit=5)
    lp_time = time.monotonic() - t0_lp
    print(f"  Est LP time (active={len(active_est)}): {lp_time*1000:.0f}ms", flush=True)

    # Coarse phase: loose epsilon, larger steps
    coarse_end = min(time.monotonic() + 400, deadline - 400)

                                momentum=0.4, verify_every=30)

    print(f"  Coarse LP: {time.monotonic()-p2_start:.1f}s, J={best.score:.6f}", flush=True)

    if best.seq is None:
        best.seq = np.ones(1024)
        best.score = compute_J(best.seq)

    return [float(x) for x in best.seq]
