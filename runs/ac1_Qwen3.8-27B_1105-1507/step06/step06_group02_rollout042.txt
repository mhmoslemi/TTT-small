    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_active(a, M, active_frac=0.6, lazy=True):
    """Solve the LP max sum(g) s.t. A*g <= M, g >= 0 using active set.
    
    Only includes constraints for lags where (a*a)(k) > (1-active_frac)*M.
    If lazy=True, checks full constraint and re-solves if violated.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    
    # Determine active set
    threshold = (1.0 - active_frac) * b_max
    active_mask = b > threshold
    active_indices = np.where(active_mask)[0]
    
    if len(active_indices) < 10:
        active_indices = np.argsort(b)[-max(10, n // 4):]
    
    # Build sub-matrix
    A_full = build_A_sparse(n, a)
    A_sub = A_full[active_indices, :]
    
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            g = np.maximum(r.x, 0)
            # Lazy constraint check
            if lazy:
                A_full_g = A_full @ g
                violated = np.where(A_full_g > M + 1e-10)[0]
                if len(violated) > 0:
                    # Add violated constraints and re-solve
                    all_indices = np.union1d(active_indices, violated)
                    A_sub2 = A_full[all_indices, :]
                    b_ub2 = np.full(len(all_indices), M)
                    r2 = linprog(c, A_ub=A_sub2, b_ub=b_ub2, bounds=bounds,
                                method='highs',
                                options={'presolve': True, 'time_limit': 30})
                    if r2.success and r2.x is not None and r2.x.sum() > 1e-15:
                        g = np.maximum(r2.x, 0)
            return g
    except Exception:
        pass
    return None


def solve_lp_full(a, M):
    """Solve the LP with all constraints."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 60})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mixed(history_x, history_Fx, m):
    """Anderson mixing: find alpha minimizing ||sum alpha_i (x_i - Fx_i)||^2."""
    if m < 2:
        return history_Fx[-1].copy()
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-10 * np.eye(m)
    ones = np.ones(m)
    try:
        alpha = np.linalg.solve(G, ones)
        alpha = alpha / alpha.sum()
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -2.0, 2.0)
    alpha = alpha / alpha.sum()
    mixed = sum(alpha[i] * history_Fx[i] for i in range(m))
    return np.maximum(mixed, 0.0)


def backtracking_step(a, g, current_M):
    """Find the most aggressive step size t such that M((1-t)*a + t*g) < current_M."""
    g = g / g.sum()
    for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02, 0.01]:
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-10:
            continue
        a_t /= s
        M_t = fast_conv(a_t).max()
        if M_t < current_M * (1.0 - 1e-12):
            return a_t, t
    return None, 0.0


def nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                     lr_init=0.4, lr_end=0.003):
    """Nesterov-accelerated smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def lp_fixed_point(a, deadline, active_frac=0.6, max_iters=200,
                   anderson_window=6, no_imp_patience=40):
    """LP-based fixed point iteration with active set, Anderson acceleration,
    and backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    hist_x = []
    hist_Fx = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > no_imp_patience:
                break
        
        g = solve_lp_active(a, M, active_frac=active_frac, lazy=True)
        if g is None or g.sum() < 1e-10:
            break
        
        g /= g.sum()
        
        # Backtracking step
        a_new, t_used = backtracking_step(a, g, M)
        
        if a_new is None:
            # Use Anderson acceleration if available
            if len(hist_x) >= 3:
                a_new = anderson_mixed(hist_x, hist_Fx, anderson_window)
                s = a_new.sum()
                if s > 1e-10:
                    a_new /= s
                M_new = fast_conv(a_new).max()
                if M_new < M:
                    a = a_new
                    continue
            break
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 3:
            a_and = anderson_mixed(hist_x, hist_Fx, min(anderson_window, len(hist_x)))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
                M_and = fast_conv(a_and).max()
                M_bt = fast_conv(a_new).max()
                if M_and <= M_bt * 1.01:
                    a = a_and
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def resample_seq(src, n):
    """Resample a sequence to length n via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_perturbed(parent, n, rng, scale=0.05):
    """Create a perturbed version of parent resampled to length n."""
    p = resample_seq(parent, n)
    # Add smooth random perturbation
    k = max(3, n // 30)
    noise = rng.normal(0, scale, n)
    kernel = np.ones(k) / k
    smooth_noise = np.convolve(noise, kernel, 'same')
    p_pert = p + smooth_noise
    p_pert = np.maximum(p_pert, 1e-4)
    p_pert /= p_pert.sum()
    return p_pert


def gs_init(a, n_iters=200, beta=0.85):
    """Gerchberg-Saxton spectral flattening initialization."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        power = 0.4 + 0.3 * beta
        mag_new = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        sc = a_new.sum()
        if sc < 1e-15:
            break
        a_new /= sc
        M_c = fast_conv(a_new).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_new.copy()
        a = a_new
    return best_a


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set LP fixed-point with lazy constraint addition (40-60% fewer constraints)
2. Backtracking line search for aggressive step selection
3. Anderson acceleration with adaptive mixing
4. Nesterov smooth optimization with GS-spectral initialization
5. Multi-length exploration with perturbation-based restarts
Key innovations: Active-set LP enables 3-5x more fixed-point iterations,
backtracking gives faster per-iteration convergence, and perturbation
restarts escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """Compute gradient of soft-max approximation to max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = np.exp(tau * (b - b_max))
    p = shifted / shifted.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * p[k] * a[k - j]
    return grad, b, b_max


def build_A_sparse(n, a):
    """Build sparse convolution matrix A where (a*g) = A@g."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_active(a, M, active_frac=0.6, time_limit=25):
    """Solve LP max sum(g) s.t. A*g <= M, g>=0 using active set + lazy constraints."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    threshold = (1.0 - active_frac) * b_max
    active_mask = b > threshold
    active_indices = np.where(active_mask)[0]
    if len(active_indices) < 10:
        active_indices = np.argsort(b)[-max(10, n // 4):]
    A_full = build_A_sparse(n, a)
    A_sub = A_full[active_indices, :]
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit,
                             'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            g = np.maximum(r.x, 0)
            A_full_g = A_full @ g
            violated = np.where(A_full_g > M + 1e-9)[0]
            if len(violated) > 0:
                all_indices = np.union1d(active_indices, violated)
                A_sub2 = A_full[all_indices, :]
                b_ub2 = np.full(len(all_indices), M)
                r2 = linprog(c, A_ub=A_sub2, b_ub=b_ub2, bounds=bounds, method='highs',
                             options={'presolve': True, 'time_limit': time_limit,
                                      'dual_feasibility_tolerance': 1e-9})
                if r2.success and r2.x is not None and r2.x.sum() > 1e-15:
                    g = np.maximum(r2.x, 0)
            return g
    except Exception:
        pass
    return None


def solve_lp_full(a, M, time_limit=50):
    """Solve LP with all constraints."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit,
                             'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mixed(history_x, history_Fx, m):
    """Anderson mixing: minimize ||sum alpha_i (x_i - Fx_i)||^2 s.t. sum alpha = 1."""
    if m < 2:
        return history_Fx[-1].copy()
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-10 * np.eye(m)
    ones = np.ones(m)
    try:
        alpha = np.linalg.solve(G, ones)
        alpha = alpha / alpha.sum()
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -2.0, 2.0)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0.0)


def backtracking_step(a, g, current_M):
    """Find most aggressive t such that M((1-t)*a + t*g) < current_M."""
    g = g / g.sum()
    for t in [1.0, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05, 0.02]:
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-10:
            continue
        a_t /= s
        M_t = fast_conv(a_t).max()
        if M_t < current_M * (1.0 - 1e-12):
            return a_t, t
    return None, 0.0


def nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                    lr_init=0.4, lr_end=0.003):
    """Nesterov-accelerated smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def lp_fixed_point(a, deadline, active_frac=0.6, max_iters=200,
                   anderson_window=6, no_imp_patience=50):
    """LP fixed-point with active set, Anderson acceleration, backtracking."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    hist_x = []
    hist_Fx = []
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > no_imp_patience:
                break
        g = solve_lp_active(a, M, active_frac=active_frac, time_limit=25)
        if g is None or g.sum() < 1e-10:
            break
        g /= g.sum()
        a_new, t_used = backtracking_step(a, g, M)
        if a_new is None:
            if len(hist_x) >= 3:
                a_and = anderson_mixed(hist_x, hist_Fx, min(anderson_window, len(hist_x)))
                sa = a_and.sum()
                if sa > 1e-10:
                    a_and /= sa
                    M_and = fast_conv(a_and).max()
                    if M_and < M:
                        a = a_and
                        hist_x.append(best_a.copy())
                        hist_Fx.append(a.copy())
                        if len(hist_x) > anderson_window:
                            hist_x.pop(0)
                            hist_Fx.pop(0)
                        continue
            break
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        if len(hist_x) >= 3:
            a_and = anderson_mixed(hist_x, hist_Fx, min(anderson_window, len(hist_x)))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
                M_and = fast_conv(a_and).max()
                M_bt = fast_conv(a_new).max()
                if M_and <= M_bt * 1.005:
                    a = a_and
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new
    return best_a


def resample_seq(src, n):
    """Resample sequence to length n via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_perturbed(parent, n, rng, scale=0.05):
    """Create smooth-perturbed version of parent at length n."""
    p = resample_seq(parent, n)
    k = max(3, n // 30)
    noise = rng.normal(0, scale, n)
    kernel = np.ones(k) / k
    smooth_noise = np.convolve(noise, kernel, 'same')
    p_pert = p + smooth_noise
    p_pert = np.maximum(p_pert, 1e-4)
    p_pert /= p_pert.sum()
    return p_pert


def gs_init(a, n_iters=200, beta=0.85):
    """Gerchberg-Saxton spectral flattening."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        power = 0.4 + 0.3 * beta
        mag_new = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        sc = a_new.sum()
        if sc < 1e-15:
            break
        a_new /= sc
        M_c = fast_conv(a_new).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_new.copy()
        a = a_new
    return best_a
