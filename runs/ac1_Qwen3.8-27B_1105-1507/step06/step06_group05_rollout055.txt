import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def convolution_equalization(a, num_iters=100, time_budget=60.0):

    start = time.monotonic()

    tau = 1e-4  # initial temperature

        if time.monotonic() - start > time_budget:

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Multi-scale temperature
        tau = max(1e-6, 1e-4 * np.exp(-0.02 * step))

        shifted = np.clip(shifted, -100, 0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum > 1e-300:
            weights /= w_sum
        else:
            weights = np.zeros(len(conv))
            weights[np.argmax(conv)] = 1.0

        grad = 2.0 * corr[n - 1:2 * n - 1]

            obj_try = compute_obj(a_try)

                best_a = a_try.copy()
                a = a_try
                step_size = min(step_size * 1.2, 0.1)

            if step_size < 1e-10:

            # Occasional random kick
            rng = np.random.default_rng(step * 31 + 7)
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

def freq_domain_peak_shave(a, num_iters=80, time_budget=40.0):
    """
    Frequency-domain peak shaving: iteratively reduce the magnitude of
    DFT bins that contribute most to the time-domain convolution peak.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        t_domain = irfft(P, size)  # This is the circular convolution, approx linear for large size
        t_domain = t_domain[:2*n-1]  # Take linear part
        
        t_max = float(np.max(t_domain))
        
        # Compute per-frequency contribution to peak
        # The contribution of bin k to time-domain sample m is |A_k|^2 * exp(2j*pi*k*m/size)
        # We want to reduce bins that make t_domain[m*] large
        m_star = int(np.argmax(t_domain))
        
        # Phase alignment: contribution of bin k to peak is |A_k|^2 * cos(2*pi*k*m_star/size)
        # We reduce bins with positive contribution
        phase = 2.0 * np.pi * np.arange(size) * m_star / size
        contribution = P * np.cos(phase)
        
        # Scale factor
        alpha = 0.3 + 0.3 * (1.0 - it / float(num_iters))
        
        # Reduce contributing bins, leave others
        P_new = P.copy()
        reduce_mask = contribution > 0.2 * np.mean(contribution)
        P_new[reduce_mask] *= (1.0 - alpha * 0.1)
        
        # Reconstruct
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.7 * a + 0.3 * a_new
            a = normalize(a)

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

**Idea: Use the "descent direction" from the LP more effectively**

The LP gives us a direction g that maximizes the sum while keeping conv(a,g) <= M. This direction is "good" in the sense that it doesn't increase the convolution peak (for the fixed a). But we want a direction that doesn't increase the peak of conv(a_new, a_new), not just conv(a, a_new).

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def soft_max_equalize(a, num_iters=120, time_budget=50.0):
    """
    Soft-max gradient descent targeting multiple convolution peaks.
    Uses adaptive temperature and backtracking line search.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))

        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        tau = max(1e-6, tau_base * np.exp(-0.015 * step))

        # Soft weights

        weights = np.exp(shifted)

            weights /= w_sum
        else:
            weights = np.zeros(len(conv))
            weights[np.argmax(conv)] = 1.0

        # Gradient via FFT cross-correlation
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        # Backtracking line search
        improved = False

            if obj_try < best_obj - 1e-14:

                best_a = a_try.copy()
                a = a_try

                improved = True
                break

            rng = np.random.default_rng(step * 31 + 7)
            if rng.random() < 0.4:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:

    return best_a


def freq_peak_shave(a, num_iters=60, time_budget=35.0):
    """
    Frequency-domain peak shaving: reduce DFT magnitudes that contribute
    most to the time-domain convolution peak.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        # Circular autocorrelation (equals linear for first n lags with sufficient zero-padding)
        t_auto = irfft(P, size)[:n]

        t_max = float(np.max(t_auto))
        m_star = int(np.argmax(t_auto))

        # Phase alignment for peak at m_star
        phase = 2.0 * np.pi * np.arange(size) * m_star / size
        contribution = P * np.cos(phase)

        # Aggressiveness decreases over time
        alpha = 0.04 + 0.02 * (1.0 - it / float(num_iters))

        # Reduce bins that positively contribute to the peak
        P_new = P.copy()
        threshold = 0.1 * np.max(contribution)
        reduce_mask = contribution > threshold
        P_new[reduce_mask] *= (1.0 - alpha)

        # Reconstruct
        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.6 * a + 0.4 * a_new
            a = normalize(a)

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

                a_try = 0.5 * normalize(best_seq) + 0.5 * normalize(a_try)

            a_try = normalize(a_try)
            if n_cur <= 1000 and remaining > 25:
                a_try = lp_optimize(a_try, num_iters=8, time_budget=min(18, remaining * 0.3))
            if remaining > 18:
                a_try = soft_max_equalize(a_try, num_iters=35, time_budget=min(12, remaining * 0.25))
            if remaining > 12:
                a_try = pairwise_swaps(a_try, time_budget=min(8, remaining * 0.15))

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart}: {best_score:.8f} **")
            else:
                print(f"  Restart {restart}: {obj_try:.8f}")

    # Phase 8: Final intensive polish
    print(f"\n{'='*60}\nPhase 8: Final polish\n{'='*60}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        for polish in range(5):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            if remaining > 25:
                a_final = lp_optimize(a_final, num_iters=10, time_budget=min(18, remaining * 0.3))
            if remaining > 18:
                a_final = soft_max_equalize(a_final, num_iters=50, time_budget=min(12, remaining * 0.3))
            if remaining > 10:
                a_final = freq_peak_shave(a_final, num_iters=25, time_budget=min(8, remaining * 0.2))
            if remaining > 6:
                a_final = pairwise_swaps(a_final, time_budget=min(5, remaining * 0.2))

            obj_pol = compute_obj(a_final)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_final.copy()
                print(f"  Polish {polish}: {best_score:.8f}")
            else:
                print(f"  Polish {polish}: {obj_pol:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Ensure sum constraint
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100
        best_score = compute_obj(np.array(result))

    # Validate with actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"Target: 1.5027, Difference: {best_score - 1.5027:.6f}")
    print(f"{'='*60}")

    return result
