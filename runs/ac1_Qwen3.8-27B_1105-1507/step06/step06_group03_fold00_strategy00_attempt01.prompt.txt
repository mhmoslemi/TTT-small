import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent(a, max_sweeps, deadline, random_order=False, seed=0, verbose=False):

    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)

        else:

            if old < 1e-12:

            else:
                ratios = np.array([0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7])
                candidates = np.unique(np.concatenate([[0.0], ratios * old]))

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:

                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:

    if a.sum() > 1e-10:

def sca_lp_step(a, max_iters, deadline, verbose=False):
    """
    SCA-LP: Successive Convex Approximation with LP subproblems.
    At each step, linearize the quadratic constraint and solve an LP.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Build LP: minimize t
        # s.t. 2*(a * g)[k] - c[k] <= t  for k=0,...,2n-2
        #      sum(g) = 1
        #      g >= 0
        # Variables: g[0..n-1], t (index n)
        # Objective: minimize t -> c_vec = [0,...,0, 1]

        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            # 2*(a*g)[k] - t <= c[k]
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)  # t variable
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t

        bounds = [(0, None)] * n + [(None, None)]

        # Add equality constraint: sum(g) = 1
        A_eq_rows = [0] * n
        A_eq_cols = list(range(n))
        A_eq = coo_matrix((np.ones(n), (A_eq_rows, A_eq_cols)), shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 30, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                if g.sum() > 1e-10:
                    g /= g.sum()
                    # Try full step first
                    J_g = compute_J(g)
                    if J_g < best_obj:
                        a = g.copy()
                        best_obj = J_g
                        best_a = g.copy()
                        no_improve = 0
                    else:
                        # Line search: try fractions
                        found = False
                        for frac in [0.5, 0.3, 0.2, 0.1]:
                            a_try = (1 - frac) * a + frac * g
                            a_try = np.maximum(a_try, 0)
                            a_try /= a_try.sum()
                            J_try = compute_J(a_try)
                            if J_try < best_obj:
                                a = a_try
                                best_obj = J_try
                                best_a = a_try.copy()
                                no_improve = 0
                                found = True
                                break
                        if not found:
                            no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 5 == 0:
            print(f"    SCA it {it}: obj={best_obj:.8f}", flush=True)

        if no_improve > 20:
            break

    return best_a, best_obj


def smooth_adam_descent(a, max_iters, deadline, tau_init=30.0, tau_max=200.0,
                        lr_init=0.05, verbose=False):
    """
    Adam optimization with log-sum-exp smoothing of max(conv(a,a)).
    Parameterizes on the probability simplex.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_count = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal tau
        frac = it / max(1, max_iters)
        tau = tau_init + (tau_max - tau_init) * frac

        # Compute smooth gradient
        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()

        # Compute gradient: g[i] = sum_k w[k] * (a[k-i] + a[i-k])
        # Using FFT: g = (w * a_rev)[n-1+i] + (w_rev * a_rev)[3n-3-i]
        a_rev = a[::-1]
        w_rev = w[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        Wrev_fft = np.fft.rfft(w_rev, size)

        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wrev_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad[i] += conv2[idx2]

        # Project gradient to simplex tangent space
        grad = grad - np.mean(grad)

        # Adam update
        t_count += 1
        lr = lr_init / (1.0 + 0.1 * t_count ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Restart with perturbation
                a = best_a + np.random.default_rng(it).normal(0, 0.02, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

        if verbose and it % 100 == 0:
            print(f"    Adam it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def lp_mirror_descent(a, max_iters, deadline, verbose=False):
    """LP mirror descent: at each step solve LP for best response direction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 20, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [0.5, 0.3, 0.2, 0.15, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1

                no_improve += 1

def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.03, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.08, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.15, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine/sine patterns
    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    # Gaussian bumps
    for center in [0.2, 0.5, 0.8]:
        for width in [0.1, 0.2, 0.35]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.05))

    # Power-law
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    # Alternating patterns
    inits.append(project_simplex(1.0 + 0.3 * np.sign(np.sin(2 * np.pi * 3 * t))))

    return inits


def solve_full(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence:
    1. Coordinate descent
    2. SCA-LP
    3. Smooth Adam
    4. Final CD + LP polish
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return a
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            if verbose:
                print(f"      Updated: J={obj:.8f}", flush=True)
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 1: Coordinate descent (30% of time)
    t1 = time.monotonic() + 0.30 * remaining
    if verbose:
        print(f"    Stage 1: Coord Descent (budget {t1 - time.monotonic():.0f}s)", flush=True)
    a = coord_descent(a, max_sweeps=10000, deadline=t1, random_order=True, seed=42)
    a = try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 2: SCA-LP (30% of time)
    t2 = time.monotonic() + 0.30 * remaining
    if verbose:
        print(f"    Stage 2: SCA-LP (budget {t2 - time.monotonic():.0f}s)", flush=True)
    a_sca, obj_sca = sca_lp_step(a, max_iters=200, deadline=t2)
    if obj_sca < best_obj:
        best_a = a_sca.copy()
        best_obj = obj_sca
        if verbose:
            print(f"      SCA improved: J={obj_sca:.8f}", flush=True)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 3: Smooth Adam (25% of time)
    t3 = time.monotonic() + 0.25 * remaining
    if verbose:
        print(f"    Stage 3: Smooth Adam (budget {t3 - time.monotonic():.0f}s)", flush=True)
    a_adam, obj_adam = smooth_adam_descent(a, max_iters=5000, deadline=t3,
                                           tau_init=20.0, tau_max=100.0, lr_init=0.03)
    if obj_adam < best_obj:
        best_a = a_adam.copy()
        best_obj = obj_adam
        if verbose:
            print(f"      Adam improved: J={obj_adam:.8f}", flush=True)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 4: Final polish - CD then LP (15% of time)
    t4a = time.monotonic() + 0.08 * remaining
    a = coord_descent(a, max_sweeps=5000, deadline=t4a, random_order=True, seed=123)
    a = try_update(a)

    t4b = deadline - 5
    a_lp, obj_lp = lp_mirror_descent(best_a, max_iters=100, deadline=t4b)
    if obj_lp < best_obj:
        best_a = a_lp.copy()
        best_obj = obj_lp
        if verbose:
            print(f"      Final LP improved: J={obj_lp:.8f}", flush=True)

    return best_a, best_obj
