def gauss_newton_convolve(a, target, max_step=0.3, line_search=True):
    """

        conv_norm = conv_pos / c_max

        # Power compression
        conv_power = conv_norm ** alpha * c_max

        # Blend
        conv_mod = (1.0 - blend) * conv + blend * conv_power
        conv_mod = np.maximum(conv_mod, 0.0)

        # Ensure sum is preserved (approximately)
        conv_max = np.max(conv_mod)
        if conv_max > 0:
            conv_mod = conv_mod / conv_max * c_max

        # Extract new a: FFT(a_new) = sqrt(FFT(conv_mod))
        cf = fft(conv_mod, N)
        cf_mag = np.abs(cf)
        cf_phase = np.angle(cf)
        # Handle near-zero magnitudes
        cf_mag_safe = np.maximum(cf_mag, 1e-30)
        fa_new = np.sqrt(cf_mag_safe) * np.exp(0.5j * cf_phase)
        a_new = ifft(fa_new).real[:n]

        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            # Try with less blend
            conv_mod2 = (1.0 - blend * 0.5) * conv + blend * 0.5 * conv_power
            conv_mod2 = np.maximum(conv_mod2, 0.0)
            cf2 = fft(conv_mod2, N)
            cf2_mag = np.maximum(np.abs(cf2), 1e-30)
            cf2_phase = np.angle(cf2)
            fa_new2 = np.sqrt(cf2_mag) * np.exp(0.5j * cf2_phase)
            a_new2 = ifft(fa_new2).real[:n]
            a_new2 = project_nonneg_sum1(a_new2)
            obj_new2 = compute_obj(a_new2)
            if obj_new2 < best_obj:
                best_obj = obj_new2
                best_a = a_new2.copy()

    return best_a, best_obj


def gauss_newton_iter(a, time_budget=40.0, gamma=1.5):
    """
    Iterative Gauss-Newton for convolution matching.
    Target: compressed version of current convolution.
    """
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Pre-build A for this n (reuse across iterations if n doesn't change)
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv_curr = compute_conv(a)
        c_max = np.max(conv_curr)

        # Build target: compress peaks toward mean
        conv_mean = 1.0 / (2 * n - 1)  # since sum(a)=1, sum(conv)=1
        # Target: blend between conv and flat
        target = conv_curr ** gamma
        target = target / np.sum(target)  # normalize to sum = 1

        a_new, obj_new = gauss_newton_convolve(a, target, max_step=0.5)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try different gamma
            for g_try in [1.2, 2.0, 3.0]:
                target2 = conv_curr ** g_try
                target2 = target2 / np.sum(target2)
                a_try, obj_try = gauss_newton_convolve(a, target2, max_step=0.3)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break
            else:
                # No improvement, try smaller step
                target3 = 0.7 * conv_curr + 0.3 * conv_mean * np.ones(2 * n - 1)
                target3 = target3 / np.sum(target3)
                a_try3, obj_try3 = gauss_newton_convolve(a, target3, max_step=0.1)
                if obj_try3 < best_obj:
                    best_obj = obj_try3
                    best_a = a_try3.copy()
                    a = a_try3.copy()
                else:
                    break

    return best_a, best_obj

def log_param_ascent(a, max_iters=200, time_budget=25.0):
    """
    Optimize in log-space: a = softmax(b). Use subgradient of the objective w.r.t. b.
    """
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    # Initialize b from a
    b = np.log(np.maximum(a, 1e-30))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.05
    momentum_v = np.zeros(n)
    momentum_coeff = 0.8

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        a_curr = np.exp(b - np.max(b))
        a_curr = a_curr / np.sum(a_curr)

        conv = compute_conv(a_curr)
        c_max = np.max(conv)
        k_star = int(np.argmax(conv))

        # Gradient of objective w.r.t. a: d(2n*max(conv))/d(sum^2) * ...
        # Simplified: gradient of max(conv) w.r.t. a[j] = 2*a[k_star - j] if valid
        grad_a = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        if j_max >= j_min:
            grad_a[j_min:j_max + 1] = 2.0 * a_curr[k_star - j_max:k_star - j_min - 1:-1]

        # Convert to gradient w.r.t. b: da/db = diag(a) - a*a^T (for softmax)
        # grad_b[j] = sum_i grad_a[i] * da_i/db_j = grad_a[j]*a[j] - a[j]*sum_i(grad_a[i]*a[i])
        ga_dot_a = np.dot(grad_a, a_curr)
        grad_b = a_curr * (grad_a - ga_dot_a)

        # Normalize
        gb_norm = np.linalg.norm(grad_b)
        if gb_norm > 1e-15:
            grad_b = grad_b / gb_norm

        # Momentum step (descent, so negative)
        momentum_v = momentum_coeff * momentum_v - lr * grad_b
        b = b + momentum_v
        b = b - np.max(b)  # for numerical stability

        a_new = np.exp(b)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        # Adaptive learning rate
        lr *= 0.995

    return best_a, best_obj


def generate_asymmetric_inits(n, rng):
    """Generate diverse asymmetric initial sequences."""
    inits = []

    # Linear ramp (increasing)
    ramp = np.arange(1, n + 1, dtype=float)
    inits.append(ramp / ramp.sum())

    # Linear ramp (decreasing)
    inits.append(ramp[::-1] / ramp.sum())

    # Quadratic (convex)
    x = np.linspace(0, 1, n)
    quad = x ** 2
    inits.append(quad / quad.sum())

    # Quadratic (concave)
    quad2 = (x - 0.5) ** 2 + 0.1
    inits.append(quad2 / quad2.sum())

    # Off-center Gaussian
    for center in [0.3, 0.4, 0.6, 0.7]:
        g = np.exp(-((x - center) ** 2) / 0.05)
        inits.append(g / g.sum())

    # Sum of two off-center Gaussians
    g1 = np.exp(-((x - 0.25) ** 2) / 0.03)
    g2 = np.exp(-((x - 0.75) ** 2) / 0.03)
    inits.append((g1 + g2) / (g1 + g2).sum())

    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        pw = (x + 0.01) ** p
        inits.append(pw / pw.sum())

    # Sawtooth
    saw = (x * n - np.floor(x * n)).clip(0.01, None)
    inits.append(saw / saw.sum())

    # Sinusoidal (one full period)
    sine = np.sin(2 * np.pi * x) ** 2 + 0.1
    inits.append(sine / sine.sum())

    # Cosine (half period, peaked in middle)
    cos_half = np.cos(np.pi * x) ** 2 + 0.1
    inits.append(cos_half / cos_half.sum())

    # Random
    inits.append((rng.random(n) + 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()

    # Exponential decay
    exp_decay = np.exp(-x * 3)
    inits.append(exp_decay / exp_decay.sum())
    exp_decay2 = np.exp(-(1 - x) * 3)
    inits.append(exp_decay2 / exp_decay2.sum())

    return inits


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def optimize_sequence(a, time_budget=60.0):
    """Run a cycle of optimization strategies on a sequence."""
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    strategies = []

    # Add strategies based on n
    if n <= 1200:
        strategies.append(('gn', lambda: gauss_newton_iter(a, time_budget=min(30.0, (time_budget - (time.monotonic() - start)) * 0.4))))
    strategies.append(('freq', lambda: freq_compress(a, max_iters=150, time_budget=min(20.0, (time_budget - (time.monotonic() - start)) * 0.3))))
    strategies.append(('chase', lambda: chase_peak_multi(a, n_steps=50, time_budget=min(15.0, (time_budget - (time.monotonic() - start)) * 0.25))))
    strategies.append(('log', lambda: log_param_ascent(a, max_iters=100, time_budget=min(15.0, (time.monotonic() - start) if False else (time_budget - (time.monotonic() - start)) * 0.2))))

    for cycle in range(int(time_budget / 10) + 1):
        if time.monotonic() - start > time_budget - 2:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        for name, fn in strategies:
            if time.monotonic() - start > time_budget - 2:
                break
            a_new, obj_new = fn()
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization with asymmetric initializations,
    Gauss-Newton convolution matching, frequency-domain compression,
    and adaptive multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"No parent, starting uniform n=700: obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # === Phase 1: Asymmetric initialization screening ===
    print("\n=== Phase 1: Asymmetric Initialization Screening ===")
    phase1_budget = min(150.0, (deadline - start_time) * 0.10)
    n_options = [n_parent, max(200, n_parent - 100), n_parent + 50, n_parent + 100,
                 min(1500, n_parent + 200), max(200, n_parent - 200)]
    n_options = list(dict.fromkeys([n for n in n_options if 100 <= n <= 2000]))

    all_candidates = []

    for n_try in n_options:
        if time.monotonic() > start_time + phase1_budget:
            break

        inits = generate_asymmetric_inits(n_try, rng)
        # Also add resampled parent
        if parent is not None:
            inits.append(resample_sequence(parent, n_try))
            inits.append(project_nonneg_sum1(resample_sequence(parent, n_try) * np.exp(rng.normal(0, 0.1, n_try))))

        per_init_budget = max(2.0, phase1_budget / (len(n_options) * len(inits)))

        for init in inits:
            if time.monotonic() > start_time + phase1_budget:
                break
            init = project_nonneg_sum1(init)
            # Quick optimization
            a_opt, obj_opt = chase_peak_multi(init, n_steps=30, time_budget=min(per_init_budget, 5.0))
            if obj_opt < np.inf:
                all_candidates.append((obj_opt, a_opt, n_try))
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  n={n_try}: NEW BEST {best_score:.8f}")

    # Sort and keep top
    all_candidates.sort(key=lambda x: x[0])
    print(f"  {len(all_candidates)} candidates, best from phase 1: {all_candidates[0][0] if all_candidates else 'N/A'}")

    # === Phase 2: Intensive optimization on top candidates ===
    print(f"\n=== Phase 2: Intensive Optimization (best so far: {best_score:.8f}) ===")
    phase2_budget = min(600.0, (deadline - start_time) * 0.45)
    phase2_start = time.monotonic()

    # Focus on top 3-4 candidates
    top_cands = [(best_score, best_seq.copy(), len(best_seq))]
    for obj_c, a_c, n_c in all_candidates[:4]:
        if abs(obj_c - top_cands[0][0]) > 1e-12:
            top_cands.append((obj_c, a_c.copy(), n_c))

    for ci, (obj_init, a_init, n_init) in enumerate(top_cands):
        if time.monotonic() > phase2_start + phase2_budget:
            break
        remaining_total = phase2_start + phase2_budget - time.monotonic()
        alloc = min(150.0, remaining_total / max(1, len(top_cands) - ci))
        if alloc < 15:
            break

        print(f"  Candidate {ci} (n={n_init}): obj={obj_init:.8f}, budget={alloc:.0f}s")
        a_opt, obj_opt = optimize_sequence(a_init, time_budget=alloc)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()

        else:
            print(f"    Best: {obj_opt:.8f} (no improvement)")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length Exploration (best: {best_score:.8f}) ===")
    remaining3 = deadline - time.monotonic()
    if remaining3 > 30:
        phase3_budget = min(300.0, remaining3 * 0.30)
        phase3_start = time.monotonic()
        n_current = len(best_seq)

        n_tries = sorted(set([
            n_current - 50, n_current - 100, n_current - 150,
            n_current + 50, n_current + 100, n_current + 150, n_current + 250,

        ]))
        n_tries = [n for n in n_tries if 100 <= n <= 2500]

                continue

            a_res = resample_sequence(best_seq, n_try)
            a_res = project_nonneg_sum1(a_res)

            # Quick optimize
            a_opt, obj_opt = chase_peak_multi(a_res, n_steps=30, time_budget=min(per_n_budget * 0.3, 10.0))
            if n_try <= 1000:
                a_opt2, obj_opt2 = gauss_newton_iter(a_opt, time_budget=min(per_n_budget * 0.4, 15.0))
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2
            else:
                a_opt2, obj_opt2 = freq_compress(a_opt, max_iters=80, time_budget=min(per_n_budget * 0.3, 10.0))
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: ** NEW BEST {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_opt:.8f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish (best: {best_score:.8f}) ===")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 20 and best_seq is not None:
        phase4_budget = min(remaining4, 300.0)
        phase4_start = time.monotonic()
        n_final = len(best_seq)

        # Multiple rounds of the best strategies
        for round_idx in range(int(phase4_budget / 20) + 1):
            if time.monotonic() > phase4_start + phase4_budget - 5:
                break
            rem = phase4_start + phase4_budget - time.monotonic()
            if rem < 15:
                break

            # Gauss-Newton (most effective for local optimization)
            if n_final <= 1200:
                a_p, obj_p = gauss_newton_iter(best_seq, time_budget=min(rem * 0.4, 40.0))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Round {round_idx} GN: {best_score:.8f}")

            # Frequency compression
            rem = phase4_start + phase4_budget - time.monotonic()
            if rem > 10:
                a_p2, obj_p2 = freq_compress(best_seq, max_iters=100,
                                             time_budget=min(rem * 0.3, 20.0),
                                             alpha_init=0.7, blend=0.4)
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    print(f"  Round {round_idx} Freq: {best_score:.8f}")

            # Chase peak
            rem = phase4_start + phase4_budget - time.monotonic()
            if rem > 8:
                a_p3, obj_p3 = chase_peak_multi(best_seq, n_steps=40,
                                                time_budget=min(rem * 0.3, 15.0))
                if obj_p3 < best_score:
                    best_score = obj_p3
                    best_seq = a_p3.copy()
                    print(f"  Round {round_idx} Chase: {best_score:.8f}")

            # Small perturbation + re-optimize
            rem = phase4_start + phase4_budget - time.monotonic()
            if rem > 15:
                perturbation = np.exp(rng.normal(0, 0.01, n_final))
                a_pert = project_nonneg_sum1(best_seq * perturbation)
                a_pert, obj_pert = chase_peak_multi(a_pert, n_steps=20, time_budget=min(rem * 0.2, 8.0))
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Round {round_idx} Perturb: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
