def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_gradient_fft(a, p, conv=None):
    """Compute gradient of sum(conv^p) w.r.t. a using FFT. O(n log n)."""
    n = len(a)
    if conv is None:
        conv = compute_conv(a)
    w = np.maximum(conv, 1e-30) ** (p - 1)
    # h[i] = 2 * sum_k w[k] * a[k-i] (cross-correlation)
    N = 1
    while N < 3 * n:
        N <<= 1
    fw = np.fft.fft(w.astype(np.complex128), N)
    fa = np.fft.fft(a.astype(np.complex128), N)
    h_full = np.fft.ifft(fw * np.conj(fa), N).real
    h = 2.0 * h_full[:n]
    return h

def multiplicative_update(a, grad, eta):
    """Multiplicative update: a_new = a * exp(-eta * (grad - mean(grad)))."""
    centered = grad - np.mean(grad)
    # Clip to prevent overflow
    centered = np.clip(centered, -50.0, 50.0)
    a_new = a * np.exp(-eta * centered)
    return a_new

def additive_update(a, grad, eta):

    centered = grad - np.mean(grad)
    a_new = a - eta * centered
    return project(a_new)

def spectral_shaping(a, max_iters=100, time_budget=30.0):

    Spectral shaping: iteratively reshape the power spectrum to flatten autocorrelation.
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()

        N <<= 1

            break
        
        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        phase = np.angle(fa)
        
        # Target: make autocorrelation flat
        # Current autocorrelation:
        ac = np.fft.ifft(power).real[:2*n-1]
        ac_max = np.max(ac)

        pad[:2*n-1] = ac_target
        # But we need to handle the wrapping: for linear autocorrelation,
        # the DFT of the padded version gives the power spectrum
        power_target = np.fft.fft(pad)

        # But it might not be, so let's use a different approach

        power_new[1:] = power[1:] * (1 - alpha * 0.5)
        # But this changes the total energy... let's use a different approach
        
        # Better: apply a "peak reduction" in the autocorrelation domain
        # then go back to frequency domain
        ac_mod = ac.copy()
        # Reduce peaks
        threshold = ac_mean * 2.0
        peak_mask = ac_mod > threshold
        if np.any(peak_mask):
            ac_mod[peak_mask] = ac_mod[peak_mask] * 0.7 + threshold * 0.3
        
        # Go back to power spectrum
        ac_padded = np.zeros(N)
        # For linear autocorrelation r[0], r[1], ..., r[2n-2]:
        # The power spectrum is FFT of r[0], r[1], ..., r[2n-2], r[2n-2], ..., r[1]
        # (i.e., make it even/circular)
        ac_padded[:2*n-1] = ac_mod
        ac_padded[1:2*n-1] = ac_mod[2*n-2:0:-1]  # mirror
        power_new = np.fft.fft(ac_padded)
        # Ensure non-negative
        power_new = np.maximum(power_new.real, 0)
        
        if np.max(power_new) < 1e-30:
            break

        fa_new = magnitude * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

def equi_ripple_approach(a, max_iters=200, time_budget=40.0):
    """
    Try to make all convolution values equal (equi-ripple).
    Use iterative proportional fitting.
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        conv_max = np.max(conv)
        
        if conv_max < 1e-15:
            break
        
        # Target: all conv values should be 1/(2n-1)
        target = 1.0 / (2 * n - 1)
        
        # Compute "error" at each convolution lag
        error = conv - target
        # We want to reduce positive errors (peaks above target)
        # Gradient of sum(max(conv-k, 0)^2) w.r.t. a[i]:
        # = 2 * sum_{k: conv[k]>target} (conv[k]-target) * 2*a[k-i]
        
        pos_error = np.maximum(conv - target, 0)
        w = 2.0 * pos_error  # weight for each lag
        
        # h[i] = sum_k w[k] * a[k-i] (same as before)
        N = 1
        while N < 3 * n:
            N <<= 1
        fw = np.fft.fft(w.astype(np.complex128), N)
        fa = np.fft.fft(a.astype(np.complex128), N)
        h = 2.0 * np.fft.ifft(fw * np.conj(fa), N).real[:n]
        
        # Also add a small component to push values below target up
        neg_error = np.maximum(target - conv, 0)
        w_neg = 0.1 * neg_error  # smaller weight for "valleys"
        fw_neg = np.fft.fft(w_neg.astype(np.complex128), N)
        h_neg = 2.0 * np.fft.ifft(fw_neg * np.conj(fa), N).real[:n]
        
        grad = h - h_neg  # net gradient: push peaks down, valleys up
        
        # Multiplicative update
        centered = grad - np.mean(grad)
        centered = np.clip(centered, -30.0, 30.0)
        eta = 0.5 / (np.mean(np.abs(grad)) + 1e-10)
        a_new = a * np.exp(-eta * centered)
        a_new = project(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            eta = eta * 0.5
            a_new = a * np.exp(-eta * centered)
            a_new = project(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def generate_barker_like(n, rng):
    """Generate a Barker-like non-negative sequence."""
    # Start with random binary sequence, then smooth
    binary = (rng.random(n) > 0.5).astype(float)
    # Smooth with a Gaussian kernel
    kernel_size = max(3, n // 20)
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size//2)**2 / (kernel_size//4)**2)
    kernel = kernel / kernel.sum()
    smoothed = np.convolve(binary, kernel, mode='same')
    return project(smoothed)

def generate_golay_like(n, rng):
    """Generate a Golay-like sequence."""
    # Golay complementary pairs have the property that their aperiodic autocorrelations sum to zero
    # We'll generate a random sequence and try to minimize peak sidelobe
    a = rng.random(n)
    # Apply a few iterations of peak reduction
    for _ in range(50):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        # Reduce elements that contribute to the peak
        for i in range(min(n, k_star+1)):
            j = k_star - i
            if 0 <= j < n:
                a[i] *= 0.95
                a[j] *= 0.95
        a = project(a)
    return a

- If a is concentrated (like a delta): conv is a delta, max = a[0]^2 = 1/n^2... wait no.

If a = [1, 0, 0, ..., 0] (one element is 1, rest are 0), then:
- sum = 1, n = n
- conv = [1, 0, 0, ..., 0] (length 2n-1, only conv[0] = 1)
- max = 1
- obj = 2*n*1/1 = 2n

That's terrible. So concentrated is bad.

If a is spread uniformly: obj ≈ 2.

To do better than 2, we need the convolution to be "flatter" than the triangular shape of the uniform case.

For a "flat-top" window (like a Fejér kernel), the autocorrelation is more peaked, which is worse. We want the opposite: a signal whose autocorrelation is as flat as possible.

In the frequency domain, a flat autocorrelation means the power spectrum is concentrated at DC. But for a non-negative signal, the DC component is the largest, so this is "natural." The question is how concentrated we can make it.

For a non-negative signal a with sum=1, the power spectrum is S(f) = |sum_k a[k] exp(-2πifk/N)|^2. At f=0, S(0) = 1. At other frequencies, S(f) = |sum_k a[k] exp(-2πifk/N)|^2 <= (sum_k a[k])^2 = 1 by triangle inequality, with equality iff all a[k] exp(-2πifk/N) have the same phase, i.e., iff a is concentrated at one point.

So S(0) = 1 is the maximum, and we want S(f) for f≠0 to be as small as possible. This is the "concentration at DC" or "narrowband" design.

The autocorrelation r(k) = ifft(S(f)). We want r(k) ≈ 1/(2n-1) for all k (flat). This means S(f) should be approximately the DFT of a flat sequence of length 2n-1 padded to N:
S_target(f) = DFT([1/(2n-1)] * (2n-1) padded to N)

But S(f) must be non-negative and S(0) = 1. The target S_target(0) = (2n-1)/(2n-1) = 1. ✓
S_target(f) for f≠0: This is the Dirichlet kernel D(N, 2n-1, f) / (2n-1), which oscillates and can be negative! So it's not a valid power spectrum.

This means a perfectly flat autocorrelation is not achievable. The best we can do is make the sidelobes as small as possible.

This is essentially the "minimum peak sidelobe" problem, which is well-studied in radar/communications. For non-negative sequences, the optimal solutions tend to have a specific structure.

One known result: for a non-negative sequence of length n, the minimum possible peak sidelobe of the autocorrelation (relative to the peak at lag 0) is related to the number of non-zero elements.

For the uniform sequence, r(k) = (n-|k|)/n^2, which peaks at k=0 with value 1/n. The "sidelobes" are at other k values.

For a "good" sequence, we want r(0) to be small (which means spread out) AND the other r(k) to be small too. But r(0) = sum a[i]^2, which is minimized when a is uniform (r(0) = 1/n). """
FFT-accelerated Lp gradient descent with adaptive parameters, equi-ripple
targeting, spectral shaping, and multi-length exploration to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Key innovations: O(n log n) gradient via FFT cross-correlation, multiplicative
updates for natural non-negativity, adaptive p-scheduling, equi-ripple
equalization, and aggressive iteration counts enabled by fast per-iter cost.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lp_gradient_fft(a, p, conv=None):
    """Compute gradient of sum(conv^p) w.r.t. a using FFT. O(n log n)."""
    n = len(a)
    if conv is None:
        conv = compute_conv(a)
    w = np.maximum(conv, 1e-30) ** (p - 1)
    N = 1
    while N < 3 * n:
        N <<= 1
    fw = np.fft.fft(w.astype(np.complex128), N)
    fa = np.fft.fft(a.astype(np.complex128), N)
    h_full = np.fft.ifft(fw * np.conj(fa), N).real
    h = 2.0 * h_full[:n]
    return h


def optimize_lp(a, max_iters=500, time_budget=60.0, p_start=4, p_end=30,
                use_multiplicative=True, verbose=False):
    """Main Lp optimization with adaptive p and step size."""
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.5
    eta_min = 1e-6
    patience = 0
    max_patience = 100

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(max_iters - 1, 1)
        p = p_start + (p_end - p_start) * frac

        conv = compute_conv(a)
        obj_now = 2.0 * n * float(np.max(conv))

        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            eta = max(eta_min, eta * 0.7)
            patience = 0
            if eta < eta_min * 2:
                break

        grad = lp_gradient_fft(a, p, conv)

        grad_scale = np.mean(np.abs(grad))
        if grad_scale < 1e-15:
            break
        eta_eff = eta / grad_scale

        if use_multiplicative:
            centered = grad - np.mean(grad)
            centered = np.clip(centered, -50.0, 50.0)
            a_new = a * np.exp(-eta_eff * centered)
            a_new = project(a_new)
        else:
            centered = grad - np.mean(grad)
            a_new = a - eta_eff * centered
            a_new = project(a_new)

        conv_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(conv_new))

        if obj_new < obj_now:
            a = a_new
            eta = min(eta * 1.02, 5.0)
        else:
            eta = max(eta_min, eta * 0.8)

    return best_a, best_obj


def equi_ripple_optimize(a, max_iters=300, time_budget=50.0, verbose=False):
    """
    Equi-ripple optimization: drive all convolution values toward the target
    1/(2n-1) using gradient of the squared overshoot.
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n:
        N <<= 1

    target = 1.0 / (2 * n - 1)
    eta = 0.3

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        obj_now = 2.0 * n * float(np.max(conv))

        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()

        # Focus on peaks above target
        overshoot = np.maximum(conv - target, 0)
        # Also gently push valleys up
        undershoot = np.maximum(target - conv, 0) * 0.05

        w = 2.0 * overshoot - undershoot
        if np.max(np.abs(w)) < 1e-15:
            break

        fw = np.fft.fft(w.astype(np.complex128), N)
        fa = np.fft.fft(a.astype(np.complex128), N)
        h = 2.0 * np.fft.ifft(fw * np.conj(fa), N).real[:n]

        grad_scale = np.mean(np.abs(h))
        if grad_scale < 1e-15:
            break
        eta_eff = eta / grad_scale

        centered = h - np.mean(h)
        centered = np.clip(centered, -50.0, 50.0)
        a_new = a * np.exp(-eta_eff * centered)
        a_new = project(a_new)

        conv_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(conv_new))

        if obj_new < obj_now:
            a = a_new
            eta = min(eta * 1.03, 2.0)
        else:
            eta = max(1e-5, eta * 0.85)

    return best_a, best_obj


def spectral_shaping(a, max_iters=200, time_budget=40.0):
    """
    Spectral shaping: modify the power spectrum to reduce autocorrelation peaks.
    Uses autocorrelation-domain peak clipping followed by spectral reconstruction.
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        phase = np.angle(fa)

        ac = np.fft.ifft(power).real[:2 * n - 1]
        ac_max = float(np.max(ac))
        ac_mean = float(np.mean(ac))

        # Clip peaks in autocorrelation
        blend = 0.3 + 0.4 * (it / max(max_iters - 1, 1))
        threshold = ac_mean * 1.5 + (ac_max - ac_mean) * blend
        ac_mod = ac.copy()
        ac_mod[ac_mod > threshold] = threshold + (ac_mod[ac_mod > threshold] - threshold) * 0.1

        # Reconstruct power spectrum from modified autocorrelation
        ac_padded = np.zeros(N)
        ac_padded[:2 * n - 1] = ac_mod
        ac_padded[1:2 * n - 1] = ac_mod[2 * n - 2:0:-1]
        power_new = np.fft.fft(ac_padded).real
        power_new = np.maximum(power_new, 0)

        if np.max(power_new) < 1e-30:
            break

        magnitude = np.sqrt(power_new)
        fa_new = magnitude * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def chase_peak_fft(a, n_steps=200, time_budget=20.0):
    """Chase the peak using FFT-accelerated subgradient."""
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n:
        N <<= 1

    step = 0.1

    for _ in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient at peak: h[i] = 2*a[k_star - i] (if valid)
        # This is just a shifted version of a
        h = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                h[i] = 2.0 * a[j]

        h_centered = h - np.mean(h)
        h_scale = np.mean(np.abs(h))
        if h_scale < 1e-15:
            break
        eta = step / h_scale

        centered = np.clip(h_centered, -50.0, 50.0)
        a_new = a * np.exp(-eta * centered)
        a_new = project(a_new)

        conv_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(conv_new))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.9

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Sinusoidal
    x = np.linspace(0, 1, n)
    inits.append(project(np.sin(np.pi * x) ** 2 + 0.05))

    # Gaussian
    x = np.linspace(-3, 3, n)
    inits.append(project(np.exp(-x ** 2 / 3)))

    # Random
    inits.append(project(rng.random(n) + 0.05))

    # Two-peak
    p1, p2 = int(n * 0.3), int(n * 0.7)
    two = np.zeros(n)
    two[max(0, p1 - n//10):p1 + n//10] = 1.0
    two[max(0, p2 - n//10):p2 + n//10] = 1.0
    inits.append(project(two))

    # Bimodal smooth
    bimodal = np.exp(-((x - 0.35) ** 2) / 0.03) + np.exp(-((x - 0.65) ** 2) / 0.03)
    inits.append(project(bimodal + 0.01))

    # Flat-top
    frac = 0.6
    start_idx = int(n * (1 - frac) / 2)
    end_idx = int(n * (1 + frac) / 2)
    fl = np.zeros(n)
    fl[start_idx:end_idx] = 1.0
    inits.append(project(fl))

    # Cosine taper
    cos_taper = 0.5 * (1 + np.cos(2 * np.pi * x)) 
    inits.append(project(cos_taper + 0.05))

    if parent is not None:
        # Resampled parent
        inits.append(project(resample_sequence(parent, n)))
        # Perturbed parent
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.15, n))
        inits.append(project(pert))
        # More perturbed
        pert2 = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.4, n))
        inits.append(project(pert2))

    return inits

        if obj_opt2 < best_score:
            best_score = obj_opt2
            best_seq = a_opt2.copy()
            print(f"  Alt n={n_c}: {best_score:.8f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    print(f"Best before: {best_score:.8f}")
    phase3_budget = min(300.0, (deadline - time.monotonic()) * 0.25)
    phase3_start = time.monotonic()

    n_best = len(best_seq)
    n_tries = list(dict.fromkeys([
        n_best - 20, n_best + 20, n_best + 50, n_best - 50,
        n_best + 100, n_best + 150, n_best - 100,
        300, 450, 550, 650, 750, 850, 950, 1100, 1300
    ]))
    n_tries = [n for n in n_tries if 30 <= n <= 3000]
    per_n_alloc = max(5.0, phase3_budget / max(len(n_tries), 1))

    for n_try in n_tries:
        if time.monotonic() > phase3_start + phase3_budget:
            break
        a_res = project(resample_sequence(best_seq, n_try))
        alloc = min(40.0, per_n_alloc)

        # Quick optimization: Lp + equi-ripple
        a_opt, obj_opt = optimize_lp(a_res, max_iters=300, time_budget=alloc * 0.5,
                                      p_start=2, p_end=15)
        if obj_opt < best_score:
            a_opt2, obj_opt2 = equi_ripple_optimize(a_opt, max_iters=200,
                                                     time_budget=alloc * 0.4)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"Best before: {best_score:.8f}")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 20 and best_seq is not None:

        # Run full pipeline again on the best
        a_final, obj_final = full_pipeline(project(best_seq.copy()),
                                           min(remaining4 * 0.5, 200.0), rng)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Full pipeline polish: {best_score:.8f}")

        # High-p Lp
        remaining4 = deadline - time.monotonic()
        if remaining4 > 15:
            a_hp, obj_hp = optimize_lp(best_seq, max_iters=2000,
                                        time_budget=remaining4 * 0.3,
                                        p_start=20, p_end=100)
            if obj_hp < best_score:
                best_score = obj_hp
                best_seq = a_hp.copy()
                print(f"  High-p Lp: {best_score:.8f}")

        # Equi-ripple
        remaining4 = deadline - time.monotonic()
        if remaining4 > 10:
            a_er, obj_er = equi_ripple_optimize(best_seq, max_iters=500,
                                                 time_budget=remaining4 * 0.4)
            if obj_er < best_score:
                best_score = obj_er
                best_seq = a_er.copy()
                print(f"  Equi-ripple: {best_score:.8f}")

        # Small perturbation + re-optimize
        remaining4 = deadline - time.monotonic()
        if remaining4 > 15:
            for pert_scale in [0.02, 0.05, 0.1]:
                remaining4 = deadline - time.monotonic()
                if remaining4 < 10:
                    break
                a_pert = best_seq * np.exp(rng.normal(0, pert_scale, len(best_seq)))
                a_pert = project(a_pert)
                a_p, obj_p = optimize_lp(a_pert, max_iters=300,
                                          time_budget=remaining4 * 0.3,
                                          p_start=5, p_end=25)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Perturb ({pert_scale}): {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
