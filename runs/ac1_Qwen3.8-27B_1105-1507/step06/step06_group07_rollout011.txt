        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """
    Compute smoothed objective (using weighted average of top peaks) and gradient.
    Uses a tempered softmax to focus on the top few peaks rather than all.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    if e_sum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        ls = cmax
    else:
        w = e / e_sum
        ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    # Gradient of soft_max w.r.t. a[i]: 2 * sum_j w[j+i] * a[j]
    # Use FFT cross-correlation
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def adam_log_opt(a, max_iters=5000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995,
                 momentum_schedule='cosine'):
    """
    Adam optimization in log-space with adaptive beta and optional cosine momentum.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start

    best_a = a.copy()
    best_true_obj = compute_obj(a)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        if momentum_schedule == 'cosine':
            beta = beta_start + (beta_end - beta_start) * (1 - np.cos(np.pi * progress)) / 2
        else:
            beta = beta_start * (beta_end / beta_start) ** progress

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)

        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a

        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 100 == 0:
            x -= np.mean(x)

        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

    return best_a


def peak_surgery(a, time_budget=20.0, seed=0):
    """
    Directly target the peak of the autocorrelation by reducing elements
    that contribute most to the maximum convolution value.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy().astype(np.float64)
    s = np.sum(best_a)
    if s < 1e-10:
        return best_a
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(seed)

    for iteration in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        # Find top peaks
        num_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:num_peaks]
        peak_value = conv[peak_indices[0]]

        improved = False
        for p in peak_indices:
            if time.monotonic() - start > time_budget:
                break
            # Find contributing elements
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            contributors = []
            for i in range(i_min, i_max + 1):
                j = p - i
                if j < 0 or j >= n:
                    continue
                contributors.append((i, j, best_a[i] * best_a[j]))

            if not contributors:
                continue

            # Sort by contribution
            contributors.sort(key=lambda x: -x[2])
            # Try reducing the top contributor
            for i, j, contrib in contributors[:5]:
                # Reduce the larger element
                if best_a[i] >= best_a[j]:
                    idx_reduce = i
                    idx_add = j
                else:
                    idx_reduce = j
                    idx_add = i

                for frac in [0.01, 0.02, 0.05, 0.1]:
                    delta = frac * best_a[idx_reduce]
                    if best_a[idx_reduce] - delta < 0:
                        continue
                    a_try = best_a.copy()
                    a_try[idx_reduce] -= delta
                    a_try[idx_add] += delta * 0.5
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random perturbation focused on high-value elements
            top_k = max(5, n // 10)
            top_elements = np.argsort(best_a)[::-1][:top_k]
            idx = top_elements[rng.integers(0, len(top_elements))]
            for delta_frac in [0.01, 0.02, -0.01, -0.02]:
                a_try = best_a.copy()
                a_try[idx] *= (1 + delta_frac)
                if a_try[idx] < 0:
                    continue
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break
            if not improved and iteration > 20:
                break

    return best_a


def flat_spectrum_iteration(a, num_iters=50, time_budget=10.0):
    """
    Frequency-domain iteration to flatten the power spectrum.
    This minimizes the peak of the autocorrelation by making |FFT(a)|^2 uniform.
    Uses Gerchberg-Saxton style projections.
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute FFT
        A = fft(a, 2 * n)

        # Target: flatten |A|^2
        # Use a soft flattening: new_amp = |A|^alpha for alpha < 1
        alpha = 0.5 + 0.3 * (it / max(num_iters - 1, 1))  # 0.5 to 0.8
        amp = np.abs(A)
        phase = np.angle(A)

        # Soft flatten: interpolate between current and flat
        flat_amp = np.ones_like(amp)
        flat_amp[0] = amp[0]  # Keep DC to maintain the sum
        # Blend
        blend = 0.3 + 0.4 * (it / max(num_iters - 1, 1))
        new_amp = (1 - blend) * amp + blend * flat_amp

        # Apply new amplitude with original phase
        A_new = new_amp * np.exp(1j * phase)

        # Inverse FFT
        a_new = np.real(ifft(A_new, 2 * n)[:n])

        # Project onto non-negative
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_refine(a, time_budget=20.0):
    """LP-based refinement: find g maximizing sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 800:
        return a
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s

    conv = compute_conv(a_norm)
    M = float(np.max(conv))

    A = build_toeplitz(a_norm)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        tl = max(1, min(time_budget * 0.5, 10))
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': tl})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            sg = float(np.sum(g))
            if sg > 1e-10:
                g = g / sg
                best_a = a.copy()
                best_obj = compute_obj(a)
                for t_blend in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    if time.monotonic() - start > time_budget:
                        break
                    a_try = (1 - t_blend) * a_norm + t_blend * g
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a
    except Exception:
        pass
    return a


def coordinate_refine(a, time_budget=15.0, seed=0):
    """Fine-tune individual coordinates with adaptive steps."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy().astype(np.float64)
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(seed)

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        # Pick random element to perturb
        idx = rng.integers(0, n)
        for delta_frac in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]:
            a_try = best_a.copy()
            a_try[idx] *= (1 + delta_frac)
            if a_try[idx] < 0:
                continue
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                improved = True
                break

        if not improved and it > 50:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def generate_structured_init(n, family, seed=42):
    """Generate structured initial sequences from various families."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)

    if family == 'uniform':
        a = np.ones(n)
    elif family == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    elif family == 'gaussian':
        center = (n - 1) / 2
        sigma = n / 4
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif family == 'modulated':
        carrier_freq = rng.uniform(0.5, 3.0)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * carrier_freq * x / n)
        a = np.maximum(a, 0.01)
    elif family == 'modulated2':
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 5.0)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x / n) + 0.2 * np.cos(2 * np.pi * f2 * x / n)
        a = np.maximum(a, 0.01)
    elif family == 'cyclostationary':
        period = rng.integers(3, 10)
        pattern = rng.uniform(0.5, 1.5, period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif family == 'barker_like':
        # Pseudo-random with equal energy
        a = rng.uniform(0.5, 1.5, n)
    elif family == 'triangular':
        a = 1 - np.abs(x - (n - 1) / 2) / ((n - 1) / 2)
        a = np.maximum(a, 0.01)
    elif family == 'sinc_like':
        center = (n - 1) / 2
        a = np.sinc((x - center) / (n / 6))
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-6)
    a = a / np.sum(a)
    return a


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multiple optimization strategies:
    1. Diverse structured initialization
    2. Adam log-space with adaptive smoothing
    3. Peak-directed surgery
    4. Flat spectrum iteration
    5. LP refinement
    6. Multi-n exploration
    7. Coordinate descent polishing
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    def time_left():
        return deadline - time.monotonic()

    def update_best(a):
        nonlocal best_seq, best_score
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            return True
        return False

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        s = np.sum(parent)
        if s > 1e-10:
            parent = parent / s
        update_best(parent)

    print(f"Initial best: {best_score:.8f}")

    # === Phase 1: Diverse initialization and quick optimization ===
    print(f"\n{'='*50}")
    print(f"Phase 1: Diverse initialization (time left: {time_left():.0f}s)")
    print(f"{'='*50}")

    n_base = len(best_seq) if best_seq is not None else 700
    n_options_init = [n_base]
    for n_try in [300, 400, 500, 600, 700, 800, 900, 1000, 1200]:
        if n_try != n_base:
            n_options_init.append(n_try)
    n_options_init = sorted(set([n for n in n_options_init if 100 <= n <= 1500]))

    candidates = []
    families = ['uniform', 'hann', 'gaussian', 'modulated', 'modulated2',
                'cyclostationary', 'barker_like', 'triangular', 'sinc_like']

    for n_try in n_options_init[:5]:  # Limit initial exploration
        if time_left() < 60:
            break
        for fam in families[:4]:  # Subset of families per n
            if time_left() < 60:
                break
            a_init = generate_structured_init(n_try, fam, seed=seed + n_try + len(fam))
            # Quick Adam
            alloc = min(15.0, time_left() * 0.1)
            if alloc < 3:
                break
            a_opt = adam_log_opt(a_init, max_iters=3000, time_budget=alloc,
                                 beta_start=5.0, beta_end=100.0,
                                 lr_start=0.05, lr_decay=0.99995)
            obj = compute_obj(a_opt)
            candidates.append((obj, a_opt))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}, {fam}: {best_score:.8f} ***")

    # Sort candidates by objective
    candidates.sort(key=lambda x: x[0])
    print(f"  Top candidate: {candidates[0][0]:.8f}")

    # === Phase 2: Intensive optimization on best candidates ===
    print(f"\n{'='*50}")
    print(f"Phase 2: Intensive optimization (time left: {time_left():.0f}s)")
    print(f"{'='*50}")

    # Work on top candidates
    work_list = []
    if best_seq is not None:
        work_list.append(best_seq.copy())
    for obj_c, a_c in candidates[:3]:
        if not any(np.allclose(a_c, w) for w in work_list):
            work_list.append(a_c.copy())

    for w_idx, a_work in enumerate(work_list[:3]):
        if time_left() < 60:
            break
        n_w = len(a_work)
        print(f"  Working on candidate {w_idx} (n={n_w}), obj={compute_obj(a_work):.8f}")

        # Adam with progressive annealing
        rounds = [
            (5.0, 50.0, 0.05, 0.99997),
            (20.0, 200.0, 0.03, 0.99998),
            (50.0, 500.0, 0.02, 0.99999),
            (100.0, 1000.0, 0.01, 0.99999),
        ]

        for r_idx, (b_s, b_e, lr0, lr_d) in enumerate(rounds):
            if time_left() < 50:
                break
            alloc = min(40.0, time_left() * 0.15)
            if alloc < 5:
                break
            a_new = adam_log_opt(a_work, max_iters=10000, time_budget=alloc,
                                 beta_start=b_s, beta_end=b_e,
                                 lr_start=lr0, lr_decay=lr_d)
            if compute_obj(a_new) < compute_obj(a_work):
                a_work = a_new
            if update_best(a_work):
                print(f"    Round {r_idx}: {best_score:.8f} ***")

        # Peak surgery
        if time_left() > 30:
            alloc = min(20.0, time_left() * 0.1)
            a_surg = peak_surgery(a_work, time_budget=alloc, seed=seed + w_idx)
            if compute_obj(a_surg) < compute_obj(a_work):
                a_work = a_surg
            if update_best(a_work):
                print(f"    Peak surgery: {best_score:.8f} ***")

        # Flat spectrum
        if time_left() > 20:
            alloc = min(10.0, time_left() * 0.08)
            a_fs = flat_spectrum_iteration(a_work, num_iters=30, time_budget=alloc)
            if compute_obj(a_fs) < compute_obj(a_work):
                a_work = a_fs
            if update_best(a_work):
                print(f"    Flat spectrum: {best_score:.8f} ***")

        # LP refinement
        if len(a_work) <= 800 and time_left() > 20:
            alloc = min(25.0, time_left() * 0.1)
            a_lp = lp_refine(a_work, time_budget=alloc)
            if compute_obj(a_lp) < compute_obj(a_work):
                a_work = a_lp
            if update_best(a_work):
                print(f"    LP: {best_score:.8f} ***")

            # Adam after LP
            if time_left() > 15:
                alloc2 = min(15.0, time_left() * 0.08)
                a_post = adam_log_opt(a_work, max_iters=5000, time_budget=alloc2,
                                      beta_start=50.0, beta_end=500.0,
                                      lr_start=0.02, lr_decay=0.99999)
                if update_best(a_post):
                    print(f"    LP+Adam: {best_score:.8f} ***")
                    a_work = a_post

    # === Phase 3: Multi-n exploration ===
    print(f"\n{'='*50}")
    print(f"Phase 3: Multi-n exploration (time left: {time_left():.0f}s)")
    print(f"{'='*50}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_to_try = sorted(set([
            n_best, n_best - 50, n_best + 50,
            n_best - 100, n_best + 100, n_best + 200,
            n_best + 300, n_best + 500
        ]))
        n_to_try = [n for n in n_to_try if 100 <= n <= 2000]
        n_to_try = [n for n in n_to_try if n != n_best] + [n_best]

        for n_try in n_to_try[:8]:
            if time_left() < 40:
                break
            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
                s = np.sum(a_try)
                if s > 1e-10:
                    a_try = a_try / s

            # Adam optimization
            alloc = min(30.0, time_left() * 0.12)
            if alloc < 5:
                break
            a_opt = adam_log_opt(a_try, max_iters=8000, time_budget=alloc,
                                 beta_start=10.0, beta_end=300.0,
                                 lr_start=0.03, lr_decay=0.99998)

            # Peak surgery on result
            if time_left() > 20:
                a_surg = peak_surgery(a_opt, time_budget=min(10.0, time_left() * 0.05),
                                      seed=seed + n_try)
                if compute_obj(a_surg) < compute_obj(a_opt):
                    a_opt = a_surg

            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 4: Intensive polishing ===
    print(f"\n{'='*50}")
    print(f"Phase 4: Intensive polishing (time left: {time_left():.0f}s)")
    print(f"{'='*50}")

    if best_seq is not None:
        a = best_seq.copy()

        # Cycle of polish operations
        for cycle in range(3):
            if time_left() < 30:
                break

            # Coordinate descent
            if time_left() > 20:
                alloc = min(15.0, time_left() * 0.1)
                a_cd = coordinate_refine(a, time_budget=alloc, seed=seed + cycle * 100)
                if update_best(a_cd):
                    print(f"  CD cycle {cycle}: {best_score:.8f} ***")
                    a = a_cd.copy()

            # High-beta Adam (targeted at the peak)
            if time_left() > 20:
                alloc = min(20.0, time_left() * 0.12)
                a_hb = adam_log_opt(a, max_iters=8000, time_budget=alloc,
                                    beta_start=200.0, beta_end=2000.0,
                                    lr_start=0.008, lr_decay=0.99999,
                                    momentum_schedule='cosine')
                if update_best(a_hb):
                    print(f"  High-beta Adam cycle {cycle}: {best_score:.8f} ***")
                    a = a_hb.copy()

            # Peak surgery
            if time_left() > 15:
                alloc = min(15.0, time_left() * 0.1)
                a_ps = peak_surgery(a, time_budget=alloc, seed=seed + 500 + cycle)
                if update_best(a_ps):
                    print(f"  Peak surgery cycle {cycle}: {best_score:.8f} ***")
                    a = a_ps.copy()

            # Flat spectrum
            if time_left() > 15:
                alloc = min(10.0, time_left() * 0.08)
                a_fs = flat_spectrum_iteration(a, num_iters=50, time_budget=alloc)
                if update_best(a_fs):
                    print(f"  Flat spectrum cycle {cycle}: {best_score:.8f} ***")
                    a = a_fs.copy()

            # LP (if size allows)
            if len(a) <= 700 and time_left() > 20:
                alloc = min(20.0, time_left() * 0.1)
                a_lp = lp_refine(a, time_budget=alloc)
                if update_best(a_lp):
                    print(f"  LP cycle {cycle}: {best_score:.8f} ***")
                    a = a_lp.copy()

    # === Phase 5: Random restarts with diverse perturbations ===
    print(f"\n{'='*50}")
    print(f"Phase 5: Random restarts (time left: {time_left():.0f}s)")
    print(f"{'='*50}")

    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 999)

        for restart in range(10):
            if time_left() < 20:
                break

            # Different perturbation strategies
            strategy = restart % 5
            if strategy == 0:
                # Multiplicative lognormal noise
                scale = 0.02 + 0.02 * (restart // 5)
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            elif strategy == 1:
                # Blend with structured sequence
                fam = families[rng.integers(0, len(families))]
                a_struct = generate_structured_init(n_best, fam, seed=seed + restart * 7)
                alpha = rng.uniform(0.1, 0.4)
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * a_struct
            elif strategy == 2:
                # Sparse spike addition
                a_pert = best_seq.copy()
                num_spikes = max(1, n_best // 100)
                for _ in range(num_spikes):
                    idx = rng.integers(0, n_best)
                    a_pert[idx] += rng.uniform(0.05, 0.2) * np.mean(best_seq)
            elif strategy == 3:
                # Frequency-domain perturbation
                A = fft(a_pert if 'a_pert' in dir() else best_seq, 2 * n_best)
                amp = np.abs(A) * (1 + rng.normal(0, 0.1, len(A)))
                phase = np.angle(A) + rng.normal(0, 0.05, len(A))
                a_pert = np.real(ifft(amp * np.exp(1j * phase), 2 * n_best)[:n_best])
                a_pert = np.maximum(a_pert, 0)
            else:
                # Directional perturbation targeting the peak
                conv = compute_conv(best_seq)
                peak_idx = np.argmax(conv)
                a_pert = best_seq.copy()
                # Reduce elements contributing to the peak
                i_min = max(0, peak_idx - (n_best - 1))
                i_max = min(peak_idx, n_best - 1)
                for i in range(i_min, i_max + 1):
                    a_pert[i] *= (1 - 0.02 * rng.random())
                a_pert = np.maximum(a_pert, 0)

            a_pert = np.maximum(a_pert, 1e-10)
            s_p = np.sum(a_pert)
            if s_p < 1e-10:
                continue
            a_pert = a_pert / s_p

            alloc = min(20.0, time_left() * 0.1)
            if alloc < 3:
                break
            a_opt = adam_log_opt(a_pert, max_iters=6000, time_budget=alloc,
                                 beta_start=10.0, beta_end=400.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart} (str={strategy}): {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart} (str={strategy}): {obj:.8f}")

    # === Phase 6: Final intensive push ===
    print(f"\n{'='*50}")
    print(f"Phase 6: Final push (time left: {time_left():.0f}s)")
    print(f"{'='*50}")

    if best_seq is not None and time_left() > 15:
        a = best_seq.copy()

        # Very high beta Adam (essentially targeting the single peak)
        alloc = min(20.0, time_left() * 0.3)
        if alloc > 5:
            a_final = adam_log_opt(a, max_iters=10000, time_budget=alloc,
                                   beta_start=500.0, beta_end=5000.0,
                                   lr_start=0.005, lr_decay=0.999995,
                                   momentum_schedule='cosine')
            if update_best(a_final):
                print(f"  Final high-beta: {best_score:.8f} ***")

        # One more peak surgery
        if time_left() > 10:
            alloc = min(10.0, time_left() * 0.3)
            a_ps = peak_surgery(a, time_budget=alloc, seed=seed + 777)
            if update_best(a_ps):
                print(f"  Final peak surgery: {best_score:.8f} ***")

        # Coordinate refinement
        if time_left() > 10:
            alloc = min(10.0, time_left() * 0.3)
            a_cd = coordinate_refine(a, time_budget=alloc, seed=seed + 888)
            if update_best(a_cd):
                print(f"  Final CD: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
