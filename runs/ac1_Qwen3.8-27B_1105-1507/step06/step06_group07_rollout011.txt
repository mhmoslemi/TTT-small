import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """
    Compute smoothed objective (logsumexp of beta*conv) and its CORRECTED gradient.
    The gradient accounts for both (w*a) and (w xa a) terms.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    # FFT size for gradient: need linear convolution of w (len 2n-1) with a (len n)
    size = 1
    while size < 4 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    
    # (w * a)[i] = sum_j w[j]*a[i-j]
    conv_term = irfft(W * A, size)[:n]
    # (w ⊛ a)[i] = sum_j w[i+j]*a[j]
    cc_term = irfft(W * np.conj(A), size)[:n]
    
    grad_ls = conv_term + cc_term
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj


def adam_log_opt(a, max_iters=10000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995, seed=0):
    """
    Adam optimization in log-space (x = log(a)) with temperature annealing.
    Uses corrected gradient for asymmetric sequences.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    # Initialize in log-space
    x = np.log(a)
    x -= np.mean(x)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Temperature annealing
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute a from x
        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)
        
        # Compute smoothed objective and CORRECTED gradient
        obj_sm, grad_a = smooth_obj_grad(a, beta)
        
        # Chain rule: gradient w.r.t. x is grad_a * a
        grad_x = grad_a * a
        
        # Adam update
        t_count += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic centering
        if it % 100 == 0:
            x -= np.mean(x)
        
        # Check true objective periodically
        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, time_budget=30.0):
    """LP-based refinement."""
    n = len(a)
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    
    conv = compute_conv(a_norm)
    M = float(np.max(conv))
    
    g = solve_lp(a_norm, M, time_limit=min(time_budget * 0.4, 15))
    if g is None:
        return a
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for t in np.linspace(0.05, 1.0, 10):
        if time.monotonic() - start > time_budget:
            break
        a_try = (1 - t) * a_norm + t * g
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
    
    return best_a


def build_ramp(n, p=1.0):
    """Build a power-law ramp: a[i] = ((i+1)/n)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p / n
    return a / np.sum(a)


def build_cosine_ramp(n):
    """Build a cosine-tapered ramp."""
    i = np.arange(n, dtype=np.float64)
    t = i / n
    a = np.sin(np.pi * t / 2) ** 2  # Rises from 0 to 1
    return a / np.sum(a)


def build_exponential(n, rate=2.0):
    """Build an exponential ramp."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(rate * i / n)
    return a / np.sum(a)


def build_bump(n, width=1.0):
    """Build a Gaussian bump in the center."""
    i = np.arange(n, dtype=np.float64)
    center = n / 2
    a = np.exp(-((i - center) / (width * n / 4)) ** 2)
    return a / np.sum(a)


def build_rect(n, frac=0.7):
    """Build a rectangular window of given fraction."""
    a = np.zeros(n)
    start = int(n * (1 - frac) / 2)
    end = n - start
    a[start:end] = 1.0
    return a / np.sum(a)


def build_three_part(n, left_ramp=0.3, plateau=0.4, right_decay=0.0):
    """Build a three-part sequence: ramp up, plateau, decay."""
    a = np.zeros(n)
    n_left = int(n * left_ramp)
    n_plat = int(n * plateau)
    n_right = n - n_left - n_plat
    
    if n_left > 0:
        a[:n_left] = np.linspace(0, 1, n_left)
    if n_plat > 0:
        a[n_left:n_left+n_plat] = 1.0
    if n_right > 0:
        a[n_left+n_plat:] = np.linspace(1 - right_decay, right_decay, n_right) if right_decay < 1 else np.zeros(n_right)
    
    if n_right > 0 and right_decay == 0:
        a[n_left+n_plat:] = np.linspace(1.0, 0.0, n_right)
    
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

            i1 = rng.integers(i_min, i_max + 1)
            i2 = rng.integers(0, n)
            if i1 == i2:
                continue
            
            delta_frac = rng.choice([0.002, 0.005, 0.01, 0.02]) * rng.choice([1, -1])
            delta = delta_frac * max(best_a[i1], best_a[i2])
            
            a_try = best_a.copy()
            a_try[i1] -= delta
            a_try[i2] += delta
            if a_try[i1] < 0 or a_try[i2] < 0:
                continue
            
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                conv = compute_conv(a_try)
                peak_idx = np.argmax(conv)
                improved = True
                break
        
        if not improved and it > 20:
            break
    
    return best_a


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < np.inf:
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf
