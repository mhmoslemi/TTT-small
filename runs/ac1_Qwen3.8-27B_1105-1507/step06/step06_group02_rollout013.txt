def fast_conv(a):
    """Compute linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective value J(a)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_convolution(a, n_iters=300, beta=0.85):
    """
    Gerchberg-Saxton style iteration to flatten the self-convolution.
    Works in frequency domain: compress |FFT(a)| to make convolution flatter.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Target: make |F|^2 as flat as possible (inverse of flat convolution)
        # Actually, |F|^2 flat means convolution is a delta, which is what we want
        # But we want convolution to be flat, which means |F|^2 should be... 
        # Let me think: conv(a,a) has DFT = F * conj(F) * some_factor
        # Actually, FFT(conv(a,a)) = F(a) * F(a) (in the appropriate padded sense)
        # So to make conv flat, we want F(a)*F(a) to have a specific form
        # This is tricky. Let me use a different approach.
        
        # Alternative: directly work with the convolution
        # Compute current convolution, find where it's too peaked, adjust
        b = fast_conv(a)
        b_max = b.max()
        # Create a "desired" convolution that's flatter
        b_target = np.exp(-0.5 * (b - b_max) / (b_max * 0.1 + 1e-10))
        # Project: find a_new such that conv(a_new, a_new) ≈ b_target
        # This requires solving a quadratic, which is hard
        # Instead, use the frequency domain hack:
        # If b = conv(a,a), then FFT(b) = F^2 (padded)
        # We want a new a' with conv(a',a') = b_target
        # So FFT(b_target) = F'^2, meaning F' = sqrt(FFT(b_target))
        B = np.fft.rfft(b_target, size)
        B = np.maximum(B.real, 0) + 1j * B.imag  # Keep complex
        # Actually B should be real and non-negative if b_target is symmetric
        # But b_target might not be symmetric, so let's just use magnitude
        mag_target = np.abs(B)
        # Scale to match energy
        mag_target = mag_target * (mag / (mag_target + 1e-30)).mean()
        # Blend
        mag_new = (1 - beta) * mag + beta * mag_target
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def smooth_max_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using softmax."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    # grad[j] = 2 * sum_k p[k] * a[k-j] (where valid)
    # This is 2 * (p * a_reversed) convolved appropriately
    # More efficiently: grad = 2 * (ifft(2*real(ifft(p)*conj(ifft(a))))... ) 
    # Actually, the gradient of (conv(a,a))[k] with respect to a[j] is:
    # a[k-j] + a[k-j] = 2*a[k-j] (if we're differentiating w.r.t. a[j] and the conv is symmetric)
    # Wait, conv(a,a)[k] = sum_i a[i]*a[k-i]
    # d/d a[j] conv(a,a)[k] = a[k-j] + (1 if k-j=j else 0 * a[k-j])
    # = a[k-j] + a[k-j] = 2*a[k-j] (for j ≠ k-j)
    # = 2*a[k-j] (the a[j] appears in two terms: i=j gives a[j]*a[k-j], i=k-j gives a[k-j]*a[j])
    # Unless j = k-j, in which case it's 2*a[j] (same)
    # So d/d a[j] conv(a,a)[k] = 2*a[k-j] for all valid j
    
    # So grad[j] = 2 * sum_k p[k] * a[k-j]
    # = 2 * (p * a)[j] where * is cross-correlation... 
    # Actually = 2 * (reversed_p * a)[j] = 2 * (p_reversed conv a)[j+n-1]... 
    # Let me just do it directly with FFT
    # grad[j] = 2 * sum_{k: valid} p[k] * a[k-j]
    # Let k' = k - j, then k = k' + j, so:
    # grad[j] = 2 * sum_{k'} p[k'+j] * a[k']
    # This is 2 * (cross-correlation of p and a)[j]
    # = 2 * ifft(conj(fft(p)) * fft(a))
    size = 1
    while size < 2 * n:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    grad_complex = np.fft.irfft(np.conj(P) * A, size)
    grad = 2.0 * grad_complex[:n]
    return grad, b, b_max


def nesterov_smooth_descent(a, deadline, max_iters=10000, tau_start=1.0, tau_end=300.0,
                            lr_init=0.4, lr_end=0.003):
    """Nesterov accelerated smooth gradient descent on the softmax-smoothed objective."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta_ns = 0.9  # Nesterov momentum
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, b, b_max = smooth_max_gradient(y, tau)
        if b_max < best_M * (1 - 1e-9):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_ns * beta_ns)) / 2.0
        momentum = ((t_k - 1.0) / (t_k + 1.0))
        a_new = np.maximum(y - lr * (grad + beta_ns * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new
    
    return best_a


def build_conv_matrix(a):
    """
    Build the sparse matrix A such that (A @ g) = conv(a, g).
    Shape: (2n-1, n)
    A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            v = a[k - j]
            if v > 0:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def solve_best_response(a, M_margin=0.0):
    """
    Solve: min M s.t. (a*g)(k) <= M for all k, g >= 0, sum(g) = 1
    Returns (g, M_optimal) or (None, None) if infeasible.
    """
    n = len(a)
    A = build_conv_matrix(a)
    # Minimize M (the last variable)
    # Variables: g_0, ..., g_{n-1}, M (total n+1)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # Minimize M
    
    # Inequality constraints: A*g - M <= 0, i.e., [A, -1] * [g; M] <= 0
    # Build [A, -1]
    A_ext_rows = list(A.rows)
    A_ext_cols = list(A.cols)
    A_ext_vals = list(A.data)
    for k in range(2 * n - 1):
        A_ext_rows.append(k)
        A_ext_cols.append(n)
        A_ext_vals.append(-1.0)
    
    A_ub = csr_matrix((A_ext_vals, (A_ext_rows, A_ext_cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 60})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            M_opt = float(r.x[n])
            return g, M_opt
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None, None

def anderson_mix(history_x, history_Fx, m):
    """Anderson mixing."""
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 2.0)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)

        M = float(b.max())
        
        # Update running average
        running_avg = running_avg + (a - running_avg) / (count + 1)
        count += 1
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                # Try from running average
                M_avg = fast_conv(running_avg).max()
                if M_avg < best_M:
                    best_M = M_avg
                    best_a = running_avg.copy()
                    a = running_avg.copy()
                    no_imp = 0
                else:
                    break
        
        # Check running average
        M_avg = fast_conv(running_avg).max()
        if M_avg < best_M:
            best_M = M_avg
            best_a = running_avg.copy()
        
        # Solve LP
        if use_minimal_lp:
            g, M_opt = solve_best_response(a)
            if g is None:
                # Fallback to fixed-M LP
                g = solve_best_response_simple(a, M * 1.01)
                if g is None:
                    break
            gs = g.sum()
            if gs < 1e-10:
                break
            g = g / gs
        else:
            g = solve_best_response_simple(a, M * 1.01)
            if g is None:
                break
            gs = g.sum()
            if gs < 1e-10:
                break
            g = g / gs
        
        # Step size
        if it < 10:
            t = t_start
        else:
            t = max(t_min, t_start / (1 + 0.03 * it))
        
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
                M_and = fast_conv(a_and).max()
                M_new = fast_conv(a_new).max()
                if M_and <= M_new * 1.02:
                    a = a_and
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def frequency_polish(a, deadline, n_iters=200):
    """Final frequency domain polish: adjust phase to reduce convolution peak."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        F = np.fft.rfft(a, size)
        b = fast_conv(a)
        b_max = b.max()
        
        # Compute "pressure" at each frequency: where the convolution is peaked
        # We want to adjust the phase of F to spread out the convolution
        # Use the gradient of the smoothed max in freq domain
        tau = 10 + 50 * it / n_iters
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # The convolution peak locations tell us which "interference" to reduce
        # P = FFT(p) tells us the "error spectrum"
        P = np.fft.rfft(p, size)
        # Adjust: F_new = F / (1 + alpha * |P|/|F| * phase_correction)
        # Simpler: just damp the frequencies where |P| is large relative to |F|
        mag_F = np.abs(F)
        mag_P = np.abs(P)
        alpha = 0.1
        damping = 1.0 - alpha * mag_P / (mag_P + mag_F * 0.1 + 1e-15)
        F_new = F * damping
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    
    return best_a


def resample_seq(src, n):
    """Resample a sequence to length n using interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent=None):
    """Create diverse initializations for length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.01))
        inits.append(np.maximum(p + 0.05 * rng.normal(0, 1, n), 0.01))
    
    inits.append(np.ones(n))
    # Various shapes
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        inits.append(0.5 + 0.5 * (t**alpha + (1-t)**alpha) / (1 + (0.5**alpha + 0.5**alpha) / 2))
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.7 + 0.3 * np.cos(2*np.pi*f*t), 0.05))
    inits.append(np.exp(-3*(t-0.5)**2) + 0.05)
    # Random smooth
    k = max(3, n // 16)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # Random (less smooth)
    k2 = max(3, n // 32)
    kern2 = np.ones(k2) / k2
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern2, 'same') + 0.02, 0.01))
    
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)
    
    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
