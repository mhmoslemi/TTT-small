import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def smooth_max_grad_fft(a, tau):
    """Compute log-sum-exp(tau * conv(a,a)) and its gradient using FFT."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerically stable log-sum-exp
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted - shifted.max())
    log_sum_exp = b_max + (shifted.max() + np.log(exp_shifted.sum())) / tau
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i]
    # This is 2 * cross_correlation(p, a)[i] for i = 0, ..., n-1
    size = 1
    while size < len(p) + n:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = 2.0 * corr
    
    return log_sum_exp, grad

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based fast smooth gradient (log-sum-exp approximation) for O(n log n) per iteration
2. FISTA with adaptive temperature, restarts, and backtracking
3. LP best-response mirror descent with warm-started HiGHS solver
4. Exact coordinate descent with golden-section search
5. Multi-start with asymmetric initializations (key: optimal must be asymmetric for J<2)
Key improvement: FFT gradient enables 100x more FISTA iterations than O(n^2) approach.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad_fft(a, tau):
    """Compute log-sum-exp(tau*conv(a,a)) and gradient using FFT. O(n log n)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted_max = shifted.max()
    exp_shifted = np.exp(shifted - shifted_max)
    p = exp_shifted / exp_shifted.sum()
    log_sum_exp = b_max + (shifted_max + np.log(exp_shifted.sum())) / tau

    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation)
    size = 1
    while size < len(p) + n:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = 2.0 * corr
    return log_sum_exp, grad


def run_fista_fast(a_init, tau_start, tau_end, max_iters, deadline, lr_init=0.5):
    """FISTA with FFT gradient, adaptive tau, and restarts."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    step = lr_init
    stagnant = 0

    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac

        _, grad_y = smooth_max_grad_fft(y, tau)
        grad_tan = grad_y - grad_y.mean()

        # Adaptive step: reduce if gradient is large
        g_norm = np.max(np.abs(grad_tan))
        if g_norm > 1e-8:
            step = min(step * 1.05, 2.0 / max(g_norm, 1e-8))
        else:
            step = lr_init

        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)

        # Nesterov momentum with restart
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        # Check for restart
        M_y = fast_conv(y_new).max()
        if M_y > best_M * 1.001 and it > 100:
            stagnant += 1
        else:
            stagnant = 0

        if stagnant > 50:
            # Restart: reset momentum
            y_new = a_new.copy()
            t_new = 1.0
            stagnant = 0

        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        it += 1

        if it % 20 == 0:
            M_a = fast_conv(a).max()
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    return best_a


def build_active_lp_matrix(n, a, active_indices):
    """Build sparse matrix A where A[k,j] = a[k-j] if valid, for k in active_indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response(a, M, n_active, time_per_lp=5.0):
    """Solve LP best response using top n_active convolution coefficients."""
    n = len(a)
    b = fast_conv(a)
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    active = top_indices.tolist()

    A = build_active_lp_matrix(n, a, active)
    if A.nnz == 0:
        return None

    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_per_lp, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def coordinate_opt_1d(a, i, b, n):
    """Find optimal a[i] given all others, using golden section on max of affected conv values."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.5)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with golden-section 1D optimization."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)

        sweep_improved = False
        for i in order:
            x_opt = coordinate_opt_1d(a, i, b, n)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences emphasizing asymmetry."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.02, n), 1e-4)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.08, n), 1e-4)
        inits.append(p_n2 / p_n2.sum())

    # Asymmetric: linear ramp (key insight: optimal must be asymmetric)
    ramp = t + 0.1
    inits.append(ramp / ramp.sum())

    # Asymmetric: exponential
    exp_seq = np.exp(2.0 * t) + 0.05
    inits.append(exp_seq / exp_seq.sum())

    # Asymmetric: quadratic (convex)
    quad = t * t + 0.1
    inits.append(quad / quad.sum())

    # Cosine (slightly asymmetric due to endpoint=False)
    cos_w = 0.3 + 0.7 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.05)
    inits.append(cos_w / cos_w.sum())

    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.05)
    inits.append(smooth / smooth.sum())

    # "Half" concentration: more weight on first half
    half = np.exp(-2.0 * t) + 0.1
    inits.append(half / half.sum())

    return inits

                a_cur = a_f.copy()
                M_cur = M_f

        if M_cur < best_M_p1:
            best_M_p1 = M_cur
            best_for_phase1 = a_cur.copy()
            print(f"    Init {ii}: M={M_cur:.10f} J={2*target_n*M_cur:.6f}", flush=True)

    if best_for_phase1 is not None:
        tracker.update(best_for_phase1)
    else:
        best_for_phase1 = parent / parent.sum() if parent.sum() > 0 else np.ones(target_n) / target_n

    print(f"  Phase 1 done: J={tracker.best_score:.8f} (elapsed {time.monotonic()-t_phase1_start:.0f}s)", flush=True)

    # ===== PHASE 2: LP Mirror Descent (main optimization) =====
    print(f"\n--- Phase 2: LP Mirror Descent ---", flush=True)
    t_phase2_end = min(deadline - 200, time.monotonic() + min(500, (deadline - time.monotonic()) * 0.45))

    a_lp = run_lp_mirror_descent(tracker.best, max_iters=5000, deadline=t_phase2_end,
                                 n_active_start=20, n_active_max=250)
    tracker.update(a_lp)

    # Second LP pass with tighter settings
    if time.monotonic() < deadline - 150:
        t_lp2 = min(deadline - 150, time.monotonic() + 100)
        a_lp2 = run_lp_mirror_descent(tracker.best, max_iters=2000, deadline=t_lp2,
                                      n_active_start=15, n_active_max=100)
        tracker.update(a_lp2)

    print(f"  Phase 2 done: J={tracker.best_score:.8f}", flush=True)

    # ===== PHASE 3: Coordinate Descent =====
    print(f"\n--- Phase 3: Coordinate Descent ---", flush=True)
    t_phase3_end = min(deadline - 100, time.monotonic() + min(200, (deadline - time.monotonic()) * 0.25))

    max_sw = max(10, int((t_phase3_end - time.monotonic()) / max(0.5, target_n / 40.0)))
    a_cd = run_coordinate_descent(tracker.best, max_sw, t_phase3_end, seed=seed + 1)
    tracker.update(a_cd)

    print(f"  Phase 3 done: J={tracker.best_score:.8f}", flush=True)

    # ===== PHASE 4: LP + CD Alternation (final polish) =====
    print(f"\n--- Phase 4: Final LP + CD polish ---", flush=True)
    remaining = deadline - 30 - time.monotonic()
    if remaining > 30:
        for round_i in range(5):
            if time.monotonic() > deadline - 30:
                break
            t_lp = min(deadline - 30, time.monotonic() + remaining / (2 * (5 - round_i)))
            a_r = run_lp_mirror_descent(tracker.best, max_iters=500, deadline=t_lp,
                                        n_active_start=10, n_active_max=80)
            tracker.update(a_r)

            if time.monotonic() > deadline - 30:
                break
            t_cd = min(deadline - 30, time.monotonic() + remaining / (2 * (5 - round_i)))
            max_sw = max(5, int((t_cd - time.monotonic()) / max(0.5, target_n / 50.0)))
            a_r2 = run_coordinate_descent(tracker.best, max_sw, t_cd, seed=seed + 100 + round_i)
            tracker.update(a_r2)

    print(f"  Phase 4 done: J={tracker.best_score:.8f}", flush=True)

    # ===== PHASE 5: Try one alternative length =====
    print(f"\n--- Phase 5: Alternative length ---", flush=True)
    if time.monotonic() < deadline - 50:
        alt_n = int(target_n * 1.2)
        alt_n = min(max(alt_n, 256), 2048)
        alt_n = (alt_n // 64) * 64
        if alt_n != target_n and alt_n >= 128:
            a_alt = resample_seq(tracker.best, alt_n)
            a_alt = np.maximum(a_alt, 1e-6)
            a_alt = a_alt / a_alt.sum()
            sub_dl = min(deadline - 20, time.monotonic() + 60)
            a_opt = run_lp_mirror_descent(a_alt, max_iters=500, deadline=sub_dl,
                                          n_active_start=20, n_active_max=100)
            tracker.update(a_opt)

    # ===== Final =====
    if tracker.best is None:
        tracker.best = np.ones(target_n) / target_n
        tracker.best_score = compute_J(tracker.best)

    # Ensure valid output
    final_seq = np.clip(tracker.best, 0.0, 1000.0)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={final_seq.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in final_seq]
