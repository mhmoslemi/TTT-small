def proj_simplex(v):

    # Then IFFT(P * conj(A))[i] gives sum_k p[k] * conj(a[(k-i) mod size])
    # For real signals and appropriate zero-padding, this gives the cross-correlation
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(np.conj(P) * A, size)
    # corr[i] = sum_k p[k] * a[(k-i) mod size]
    # For i in 0..n-1, we want sum_k p[k]*a[k-i] where a is zero outside [0,n-1]
    # With zero-padding to size >= 2n-1, this should work for i in 0..n-1
    grad = 2.0 * corr[:n]
    
    return smooth_val, grad

def solve_lp_best_response(a, M, active_indices):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0.
    Returns (g_normalized, dual_variables) or (None, None)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    if len(rows) == 0:
        return None, None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                # Get dual variables (shadow prices)
                duals = r.ineqlin.marginals if hasattr(r, 'ineqlin') and r.ineqlin is not None else None
                if duals is not None and len(duals) == len(active_indices):
                    duals = np.array([max(0.0, d) for d in duals])
                else:
                    duals = None
                return g / g_sum, duals
    except Exception:
        pass
    return None, None

def greedy_active_set(a, b, k_star, max_size=50, threshold_frac=0.8):
    """Build active set greedily: start with argmax, add constraints violated by LP solution."""
    n = len(a)
    M = b.max()
    active = [int(k_star)]
    
    # Add top values
    sorted_idx = np.argsort(b)[::-1]
    for idx in sorted_idx:
        if len(active) >= 5:
            break
        idx = int(idx)
        if idx not in active:
            active.append(idx)
    
    # Greedy expansion: solve LP, check violations, add most violated
    for _ in range(max_size - len(active)):
        g, _ = solve_lp_best_response(a, M, active)
        if g is None:
            break
        # Check all constraints
        ag = fast_conv(a, ) # no, we need a*g not a*a
        # Actually we need (a*g)[k] for all k
        # (a*g)[k] = conv(a, g)[k]
        ag = np.convolve(a, g, mode='full')
        violations = ag - M
        # Find most violated constraint not in active
        max_viol = -np.inf
        best_idx = -1
        for k in range(len(ag)):
            if k not in active:
                if violations[k] > max_viol:
                    max_viol = violations[k]
                    best_idx = k
        if best_idx == -1 or max_viol < 1e-10 * M:
            break
        active.append(best_idx)
    
    return active[:max_size]

def run_lp_descent(a_init, deadline, max_iters=200, init_active=10, max_active=80, seed=0):
    """LP-based descent with greedy active set and line search."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    current_active = init_active
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Build active set
        if it % 5 == 0:
            active = greedy_active_set(a, b, k_star, max_size=current_active)
        else:
            # Use top-k for speed
            top_idx = np.argsort(b)[::-1][:current_active]
            active = [int(x) for x in top_idx]
        
        g, duals = solve_lp_best_response(a, M, active)
        if g is None:
            no_improve += 2
            if no_improve > 10:
                break
            continue
        
        # Line search on step size
        best_t = None
        best_trial_J = best_J
        for t in [0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_t = t
                best_trial = a_trial.copy()
        
        if best_t is not None:
            a = best_trial
            best_J = best_trial_J
            best_a = a.copy()
            no_improve = 0
            # Reduce active set if improving
            if current_active > init_active:
                current_active = max(init_active, int(current_active * 0.9))
        else:
            no_improve += 1
            current_active = min(max_active, int(current_active * 1.3))
            if no_improve > 20:
                break
    
    return best_a, best_J

def run_mirror_descent_dual(a_init, deadline, max_iters=5000, step_init=0.3, seed=0):
    """Mirror descent using LP-dual smoothed subgradient."""
    a = np.maximum(a_init.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Get active set (top 20)
        top_idx = np.argsort(b)[::-1][:min(20, 2*n-1)]
        active = [int(x) for x in top_idx]
        
        g, duals = solve_lp_best_response(a, M, active)
        
        if g is not None and duals is not None and duals.sum() > 1e-10:
            # Use dual-based gradient
            # grad_j = 2 * sum_k duals[idx_k] * a[k-j]
            grad = np.zeros(n)
            for idx, k in enumerate(active):
                if duals[idx] > 1e-12:
                    for j in range(n):
                        m = k - j
                        if 0 <= m < n:
                            grad[j] += duals[idx] * a[m]
            grad = 2.0 * grad
        else:
            # Fall back to single-subgradient at argmax
            k_star = int(np.argmax(b))
            grad = np.zeros(n)
            for j in range(n):
                m = k_star - j
                if 0 <= m < n:
                    grad[j] = 2.0 * a[m]
        
        # Center gradient (tangent to simplex)
        grad_tan = grad - grad.mean()
        
        # Multiplicative update
        step = step_init / (1.0 + 0.01 * it)
        log_update = -step * grad_tan
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        a = a_new
        J_cur = compute_J(a)
        if J_cur < best_J:
            best_J = J_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_a, best_J

def coordinate_descent_1d(a, i, b, n):
    """Find optimal a[i] given all others fixed. Returns optimal value (unnormalized)."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if quad_pos >= 0:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, np.sqrt(max(b.max(), 0.0)) + 0.1, 0.3)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

def run_coordinate_descent(a_init, deadline, max_sweeps=200, seed=0):
    """Coordinate descent with golden section 1D optimization."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()

        a = a / s

    best_a = a.copy()

    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        
        b = fast_conv(a)
        b_changed = False
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                a[i] = x_opt
                # Update b
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = b[k] - old * old + x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] = b[k] - 2.0 * old * a[j] + 2.0 * x_opt * a[j]
                b_changed = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        b = fast_conv(a)  # Recompute for accuracy
        
        J_cur = compute_J(a)
        if J_cur < best_J * (1 - 1e-12):
            best_J = J_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            no_improve += 5
        if no_improve > 30:

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.02, 0.05, 0.1]:
            pn = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(pn / pn.sum())
    
    inits.append(np.ones(n) / n)
    
    # Parabolic
    poly = (t * (1 - t)) + 0.1
    inits.append(poly / poly.sum())
    
    # Cosine windows
    for freq in [1, 2, 3, 4]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    # Gaussian
    gauss = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Spectral flat optimization (few iterations)
    a_spec = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_spec)
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.15 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    
    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using an improved multi-phase hybrid:
1. LP best-response with greedy active set and dual-based gradient smoothing
2. Mirror descent (multiplicative weights) with LP-dual subgradient
3. FISTA on adaptive log-sum-exp smooth approximation with FFT-based gradients
4. Exact 1D coordinate optimization via golden section

Key improvements: FFT-accelerated gradients, greedy active set expansion,

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    if not (~cond).any():
        rho = n - 1
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def lse_smooth_grad(a, tau):
    """Log-sum-exp smooth max(conv(a,a)) and gradient via FFT cross-correlation."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    max_shifted = shifted.max()
    exp_b = np.exp(shifted - max_shifted)
    log_sum = max_shifted + np.log(exp_b.sum())
    smooth_val = b_max + log_sum / tau
    p = exp_b / exp_b.sum()

    # Need size >= (2n-1) + n - 1 = 3n-2 for correct circular correlation
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    P = np.fft.fft(p, size)
    A = np.fft.fft(a, size)
    corr = np.fft.ifft(P * np.conj(A)).real[:n]
    grad = 2.0 * corr
    return smooth_val, grad


def solve_lp(a, M, active_indices, time_limit=8):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0.
    Returns (g_normalized, duals) or (None, None)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        k = int(k)
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])

    if len(rows) == 0:
        return None, None

    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                duals = None
                if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                    margs = r.ineqlin.marginals
                    if margs is not None:
                        duals = np.array([max(0.0, float(d)) for d in margs[:len(active_indices)]])
                return g / g_sum, duals
    except Exception:
        pass
    return None, None

        grad_tan = grad - grad.mean()

        # Multiplicative update

        if best_J_new >= best_J:

                log_u = -step * mult * grad_tan
                log_u -= log_u.max()
                a_try = a * np.exp(log_u)
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                J_try = compute_J(a_try)

                    best_a_new = a_try.copy()
            a_new = best_a_new

        J_cur = compute_J(a)
        if J_cur < best_J:

            best_a = a.copy()
            no_improve = 0
        else:

        grad_tan = grad_y - grad_y.mean()

        step = lr / (1.0 + 0.01 * it)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t = t_new

        if it % 20 == 0:
            J_a = compute_J(a)
            if J_a < best_J:
                best_J = J_a
                best_a = a.copy()
            J_y = compute_J(y)
            if J_y < best_J:
                best_J = J_y
                best_a = y.copy()

    J_final = compute_J(a)
    if J_final < best_J:
        best_J = J_final
        best_a = a.copy()

    return best_a, best_J


def coord_opt_1d(a, i, b, n):
    """Find optimal a[i] given all others. Golden section search."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_fn(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if quad_pos >= 0:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, np.sqrt(max(b.max(), 0.0)) + 0.1, 0.3)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_fn(a_pt)
    fb = obj_fn(b_pt)

    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_fn(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_fn(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_fn(0.0) <= obj_fn(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coordinate_descent(a_init, deadline, max_sweeps=300, seed=0):
    """Coordinate descent with golden section 1D optimization."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)

        b = fast_conv(a)
        b_changed = False
        for i in order:
            x_opt = coord_opt_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = b[k] - old * old + x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] = b[k] - 2.0 * old * a[j] + 2.0 * x_opt * a[j]
                b_changed = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        J_cur = compute_J(a)
        if J_cur < best_J * (1.0 - 1e-12):
            best_J = J_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not b_changed:
            no_improve += 5
        if no_improve > 30:
            break

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            pn = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(pn / pn.sum())

    inits.append(np.ones(n) / n)

    poly = (t * (1 - t)) + 0.1
    inits.append(poly / poly.sum())

    for freq in [1, 2, 3, 4, 5]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())

    gauss = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    gauss2 = np.exp(-8.0 * (t - 0.5) ** 2) + 0.02
    inits.append(gauss2 / gauss2.sum())

    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # Spectral flat init
    a_spec = np.ones(n) / n
    for _ in range(40):
        bs = fast_conv(a_spec)
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.15 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())

    return inits


def run_phase_combined(a0, deadline, seed):
    """Run all optimization phases in sequence."""
    best = a0.copy()
    best_J = compute_J(a0)
    n = len(a0)
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best, best_J

    # Phase 1: LP descent (45%)
    t1 = time.monotonic() + 0.45 * remaining
    a_lp, J_lp = run_lp_descent(best, t1, max_iters=400, init_active=10, max_active=80)
    if J_lp < best_J:
        best_J = J_lp
        best = a_lp.copy()

    # Phase 2: Mirror descent with dual (20%)
    t2 = time.monotonic() + 0.2 * remaining
    a_md, J_md = run_mirror_descent(best, t2, max_iters=5000, step_init=0.2)
    if J_md < best_J:
        best_J = J_md
        best = a_md.copy()

    # Phase 3: FISTA (15%)
    t3 = time.monotonic() + 0.15 * remaining
    a_f, J_f = run_fista(best, t3, tau_start=50, tau_end=500, max_iters=4000)
    if J_f < best_J:
        best_J = J_f
        best = a_f.copy()

    # Phase 4: Coordinate descent (10%)
    a_cd, J_cd = run_coordinate_descent(best, deadline - 2, max_sweeps=100, seed=seed + 1)
    if J_cd < best_J:
        best_J = J_cd
        best = a_cd.copy()

    # Phase 5: Final LP polish (10%)
    if time.monotonic() < deadline - 2:
        a_lp2, J_lp2 = run_lp_descent(best, deadline, max_iters=100, init_active=20, max_active=100)
        if J_lp2 < best_J:
            best_J = J_lp2
            best = a_lp2.copy()

    return best, best_J
