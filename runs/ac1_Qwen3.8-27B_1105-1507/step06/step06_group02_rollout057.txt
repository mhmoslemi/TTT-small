"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth gradient descent with Nesterov momentum and periodic symmetrization
2. LP-based best response with cutting-plane acceleration and adaptive step size
3. Multi-start with diverse initializations and length exploration
4. Careful time management to maximize optimization within budget
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fast_cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def symmetrize(a):
    return (a + a[::-1]) / 2.0

def smooth_grad(a, tau):
    """Compute gradient of soft-max approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # grad_j = 2 * sum_d a[d] * p[d+j] (cross-correlation)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = 2.0 * cross
    return grad, b_max, p

def run_smooth(a, deadline, max_iters=5000, tau_start=2.0, tau_end=300.0,
               lr_init=0.3, lr_end=0.005, symm_every=5):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = symmetrize(a)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    momentum = 0.9
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, b_max, p = smooth_grad(y, tau)
        
        a_new = np.maximum(y - lr * (grad + momentum * (y - a)), 0.0)
        if it % symm_every == 0:
            a_new = symmetrize(a_new)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M * (1 - 1e-12):
            best_M = M_new
            best_a = a_new.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        beta_k = (t_k - 1.0) / (t_k + 1.0)
        y = a_new + beta_k * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        if it % symm_every == 0:
            y = symmetrize(y)
        
        a = a_new
    
    return best_a

def solve_lp_restricted(a, M, active_set):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for k in active_set, g >= 0."""
    n = len(a)
    k_count = len(active_set)
    if k_count == 0:
        return None
    
    rows = []
    cols = []
    vals = []
    row_offset = 0
    for k_val in active_set:
        j_min = max(0, k_val - n + 1)
        j_max = min(n - 1, k_val)
        for j in range(j_min, j_max + 1):
            rows.append(row_offset)
            cols.append(j)
            vals.append(a[k_val - j])
        row_offset += 1
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(k_count, n))
    c = -np.ones(n)
    b_ub = np.full(k_count, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

    """LP best response using cutting-plane method."""
    n = len(a)
    b = fast_conv(a)
    # Start with top-k most violated constraints
    top_idx = np.argsort(b)[-initial_topk:]
    active = list(top_idx.astype(int))
    
    for round_num in range(max_rounds):
        g = solve_lp_restricted(a, M, active)
        if g is None:
            break
        
        # Check all constraints

def run_lp_fixedpoint(a, deadline, max_iters=500, t_init=0.4, t_min=0.02, decay=0.005):
    """Fixed-point iteration: a <- (1-t)*a + t*g where g = LP best response."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_imp_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        J_curr = 2.0 * n * M
        s = a.sum()
        
        if J_curr / (s * s) < best_J * (1 - 1e-10):
            best_J = J_curr / (s * s)
            best_a = a.copy()
            no_imp_count = 0
        else:
            no_imp_count += 1
            if no_imp_count > 50:
                break
        
        t = max(t_min, t_init / (1.0 + decay * it))
        
        g = lp_best_response_cutting(a, M)
        if g is None:
            break
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        
        a = a_new
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth gradient descent with Nesterov momentum and periodic symmetrization
2. LP-based best response with full constraint set and adaptive step size
3. Cutting-plane accelerated LP for faster iterations
4. Multi-start with diverse initializations and length exploration
5. Multi-pass LP polishing with decreasing step sizes

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def symmetrize(a):
    return (a + a[::-1]) / 2.0


def smooth_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = 2.0 * cross
    return grad, b_max, p


def run_smooth(a, deadline, max_iters=5000, tau_start=2.0, tau_end=300.0,
               lr_init=0.3, lr_end=0.005, symm_every=4):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = symmetrize(a)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    momentum = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b_max, p = smooth_grad(y, tau)

        a_new = np.maximum(y - lr * (grad + momentum * (y - a)), 0.0)
        if it % symm_every == 0:
            a_new = symmetrize(a_new)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s

        M_new = fast_conv(a_new).max()
        if M_new < best_M * (1.0 - 1e-12):
            best_M = M_new
            best_a = a_new.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        beta_k = (t_k - 1.0) / (t_k + 1.0)
        y = a_new + beta_k * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        if it % symm_every == 0:
            y = symmetrize(y)

        a = a_new

    return best_a


def build_A_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response_full(a, M):
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def solve_lp_rows(a, M, row_indices):

    k_count = len(row_indices)
    if k_count == 0:
        return None
    rows = []
    cols = []
    vals = []
    for idx, k_val in enumerate(row_indices):
        j_min = max(0, k_val - n + 1)
        j_max = min(n - 1, k_val)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k_val - j])
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(k_count, n))

        pass

def lp_best_response_cutting(a, M, max_rounds=12, initial_topk=128):

    b = fast_conv(a)
    top_idx = np.argsort(b)[-initial_topk:]

        viol_indices = np.where(viol_mask)[0]
        worst = int(viol_indices[np.argmax(ag[viol_indices])])
        if worst not in active:

        else:
            break

    if g is not None:
        ag = fast_cross_conv(a, g)
        if ag.max() <= M * (1.0 + 0.01):

    return None

    k = max(3, n // 30)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))

    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.2, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)

    # Anti-peak: emphasize edges
    inits.append(np.maximum(0.1 + 0.9 * (t * (1.0 - t) ** 0.5 + (1.0 - t) * (t) ** 0.5), 0.01))

    return inits


def update_tracker(tracker, a, label=""):
    if a is None or len(a) == 0:
        return
    a_arr = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a_arr.sum() < 0.01:
        return
    s = float(evaluate_sequence(a_arr.tolist()))
    if np.isfinite(s) and s < tracker['score']:
        tracker['score'] = s
        tracker['seq'] = a_arr.copy()
        el = time.monotonic() - tracker['t0']
        print(f"  ** NEW BEST: J={s:.8f} n={a_arr.size} t={el:.0f}s {label}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0

    tracker = {'seq': None, 'score': float('inf'), 't0': t_start}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_tracker(tracker, parent, "(parent)")
            print(f"Parent: n={parent.size}, J={tracker['score']:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        update_tracker(tracker, parent, "(default)")

    # === PHASE 1: Smooth optimization ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Smooth Gradient Optimization", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = t_start + 380.0
    lengths = [896, 1024, 1152, 1280, 1408, 1536, 1792]

    per_length_best = {}

    for n in lengths:
        if time.monotonic() > p1_deadline - 25:
            break

        n_start = time.monotonic()
        inits = make_initializations(n, rng, tracker['seq'] if tracker['seq'] is not None else parent)

        best_n_seq = None
        best_n_M = float('inf')

        max_inits = max(3, min(len(inits), int(12 * (1024.0 / n))))

        for idx in range(min(max_inits, len(inits))):
            if time.monotonic() > p1_deadline - 15:
                break
            a0 = np.maximum(inits[idx].astype(np.float64), 1e-8)
            a0 = symmetrize(a0)
            a0 /= a0.sum()

            remaining_inits = max_inits - idx
            sub_budget = min(45.0, (p1_deadline - 15 - time.monotonic()) / remaining_inits)
            sub_dl = time.monotonic() + sub_budget

            a = run_smooth(a0, sub_dl, max_iters=4000,
                           tau_start=2.0, tau_end=250.0,
                           lr_init=0.25, lr_end=0.003, symm_every=3)

            M = fast_conv(a).max()
            if M < best_n_M:
                best_n_M = M
                best_n_seq = a.copy()

        if best_n_seq is not None:
            J = 2.0 * n * best_n_M / (best_n_seq.sum() ** 2)
            print(f"  n={n:4d}: M={best_n_M:.8f} J={J:.6f} ({time.monotonic()-n_start:.1f}s)", flush=True)
            update_tracker(tracker, best_n_seq, f"(smooth n={n})")
            per_length_best[n] = best_n_seq.copy()

    print(f"Phase 1 done. Best J={tracker['score']:.6f}", flush=True)

    # === PHASE 2: LP Fixed-Point Polishing (main phase) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: LP Fixed-Point Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker['seq'] is not None and time.monotonic() < deadline - 80:
        a = np.maximum(tracker['seq'].astype(np.float64), 1e-10)
        a /= a.sum()
        n_best = a.size
        print(f"  Starting LP: n={n_best}", flush=True)

        # Pass 1: aggressive with cutting plane
        p2a_dl = time.monotonic() + min(250.0, deadline - time.monotonic() - 60)
        a = run_lp_fixedpoint(a, p2a_dl, max_iters=200, t_init=0.45,
                             t_min=0.05, decay=0.003, use_cutting=True, topk=192)
        update_tracker(tracker, a, "(LP cut pass1)")

        # Pass 2: medium with full LP for precision
        if time.monotonic() < deadline - 50:
            p2b_dl = time.monotonic() + min(350.0, deadline - time.monotonic() - 40)
            a = run_lp_fixedpoint(a, p2b_dl, max_iters=150, t_init=0.25,
                                  t_min=0.02, decay=0.002, use_cutting=False)
            update_tracker(tracker, a, "(LP full pass2)")

        # Pass 3: fine tuning with cutting plane
        if time.monotonic() < deadline - 40:
            p2c_dl = time.monotonic() + min(300.0, deadline - time.monotonic() - 30)
            a = run_lp_fixedpoint(a, p2c_dl, max_iters=200, t_init=0.15,
                                  t_min=0.01, decay=0.001, use_cutting=True, topk=256)
            update_tracker(tracker, a, "(LP cut pass3)")

        # Pass 4: very fine with full LP
        if time.monotonic() < deadline - 20:
            p2d_dl = deadline - 10
            a = run_lp_fixedpoint(a, p2d_dl, max_iters=100, t_init=0.08,
                                  t_min=0.005, decay=0.001, use_cutting=False)
            update_tracker(tracker, a, "(LP full pass4)")

    print(f"Phase 2 done. Best J={tracker['score']:.6f}", flush=True)

    # === PHASE 3: Length exploration ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker['seq'] is not None and time.monotonic() < deadline - 40:
        n_best = tracker['seq'].size
        deltas = [-128, -64, 64, 128, 256]
        for delta in deltas:
            if time.monotonic() > deadline - 25:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 2560:
                continue
            a_ext = resample_seq(tracker['seq'], n_try)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = symmetrize(a_ext)
            a_ext /= a_ext.sum()

            a_ext = run_smooth(a_ext, min(time.monotonic() + 25, deadline - 20),
                               max_iters=2000, tau_start=5.0, tau_end=150.0,
                               lr_init=0.2, lr_end=0.01, symm_every=5)
            update_tracker(tracker, a_ext, f"(resample n={n_try})")

            if time.monotonic() < deadline - 15:
                a_lp = run_lp_fixedpoint(a_ext, min(time.monotonic() + 20, deadline - 10),
                                         max_iters=40, t_init=0.3, t_min=0.05,
                                         decay=0.01, use_cutting=True, topk=128)
                update_tracker(tracker, a_lp, f"(LP resample n={n_try})")

    # === PHASE 4: Final polish ===
    if tracker['seq'] is not None and time.monotonic() < deadline - 5:
        a_final = np.maximum(tracker['seq'].astype(np.float64), 1e-10)
        a_final /= a_final.sum()
        a_final = run_lp_fixedpoint(a_final, deadline, max_iters=50,
                                    t_init=0.05, t_min=0.005, decay=0.001,
                                    use_cutting=False)
        update_tracker(tracker, a_final, "(final)")

    # Fallback
    if tracker['seq'] is None:
        tracker['seq'] = np.ones(128)
        tracker['score'] = compute_J(tracker['seq'])

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker['score']:.8f} n={tracker['seq'].size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker['seq']]
