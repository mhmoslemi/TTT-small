def fast_conv(a):

    n = len(a)
    if n == 0:

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_J(a):

    no_improve = 0

    # Precompute constraint matrix structure
    # Row k: columns j where max(0,k-n+1) <= j <= min(n-1,k) have coefficient a[k-j]
    # Plus column n (t) has coefficient -1

    if warm_start is not None:
        g_prev = np.maximum(warm_start, 0).copy()
        if g_prev.sum() > 1e-10:
            g_prev = g_prev / g_prev.sum()
        else:
            g_prev = a.copy()
    else:
        g_prev = a.copy()

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        c = fast_conv(a)

        # Build LP constraint matrix (values change with a)
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(0, None)]
        A_eq = coo_matrix((np.ones(n), (list(range(n)), list(range(n)))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs

                    # Adaptive step size: try multiple fractions
                    fracs = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]
                    found = False
                    for frac in fracs:
                        a_try = (1.0 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        sts = a_try.sum()
                        if sts < 1e-10:
                            continue
                        a_try = a_try / sts
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-13:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            g_prev = g.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 80:
            break

        if verbose and iteration % 50 == 0:
            print(f"    SCA it {iteration}: obj={best_obj:.8f} no_improve={no_improve}", flush=True)

    return best_a, best_obj, g_prev


def spectral_perturb(a, rng, strength=0.1):
    """
    Perturb using spectral domain: modify power spectrum to encourage flatter autocorrelation.
    """
    n = len(a)
    a = a.copy()
    A = np.fft.rfft(a)
    power = np.abs(A) ** 2

    # Boost low-power frequency bins to flatten the spectrum
    power_std = power.std()
    if power_std > 1e-12:
        power_normalized = (power - power.min()) / (power.max() - power.min() + 1e-12)
        # Flatten: reduce dynamic range
        power_flat = 1.0 + strength * (1.0 - power_normalized)
        power_new = power * power_flat

        # Reconstruct with same phases
        phase = np.angle(A)
        A_new = np.sqrt(power_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, 2 * n - 1)[:n]
        a_new = np.maximum(a_new, 0)
        if a_new.sum() > 1e-10:
            a_new = a_new / a_new.sum()
            return a_new

    return a


def gaussian_perturb(a, rng, sigma=0.05):
    """Add correlated Gaussian noise."""
    n = len(a)
    noise = rng.normal(0, sigma, n)
    # Smooth the noise
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    noise = np.convolve(noise, kernel, mode='same')
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    if a_new.sum() > 1e-10:
        a_new = a_new / a_new.sum()
    return a_new


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: CD -> SCA-LP -> perturbation restarts -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={time.monotonic()-t_start:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        update_best(parent)
    else:
        parent = parent / parent.sum()

    n = len(parent)
    print(f"\n{'='*60}", flush=True)
    print(f"Starting optimization at n={n}", flush=True)
    print(f"{'='*60}", flush=True)

    # === PHASE 1: Fast coordinate descent polish (200s) ===
    t1 = t_start + 200.0
    print(f"\n[Phase 1] Coordinate Descent (budget {t1 - time.monotonic():.0f}s)", flush=True)
    a1 = coord_descent_fast(parent, deadline=t1, random_order=True, seed=42)
    update_best(a1)
    print(f"  Phase 1 done: J={compute_J(a1):.8f}", flush=True)

    # === PHASE 2: SCA-LP intensive (400s) ===
    t2 = t_start + 600.0
    print(f"\n[Phase 2] SCA-LP (budget {t2 - time.monotonic():.0f}s)", flush=True)
    a2, obj2, g_warm = solve_lp_sca(a1, deadline=t2)
    update_best(a2)
    print(f"  Phase 2 done: J={obj2:.8f}", flush=True)

    # === PHASE 3: CD + SCA-LP refinement cycle (200s) ===
    t3 = t_start + 800.0
    print(f"\n[Phase 3] Refinement cycle (budget {t3 - time.monotonic():.0f}s)", flush=True)
    a3 = coord_descent_fast(best_seq, deadline=min(t3 - 50, time.monotonic() + 80), random_order=True, seed=123)
    update_best(a3)

    remaining_p3 = t3 - time.monotonic()
    if remaining_p3 > 30:
        a3b, obj3b, g_warm2 = solve_lp_sca(a3, deadline=t3 - 5, warm_start=g_warm)
        update_best(a3b)
        print(f"  Phase 3 done: J={compute_J(best_seq):.8f}", flush=True)

    # === PHASE 4: Multi-restart with perturbations (250s) ===
    t4 = t_start + 1050.0
    print(f"\n[Phase 4] Multi-restart perturbations (budget {t4 - time.monotonic():.0f}s)", flush=True)
    n_restarts = 8
    per_restart = (t4 - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > t4:
            break
        alloc = min(per_restart, t4 - time.monotonic() - 10)
        if alloc < 20:
            break

        # Different perturbation strategies
        rng_r = np.random.default_rng(seed + 1000 * r)
        if r % 3 == 0:
            a_seed = gaussian_perturb(best_seq, rng_r, sigma=0.03 + 0.01 * r)
        elif r % 3 == 1:
            a_seed = spectral_perturb(best_seq, rng_r, strength=0.15 + 0.05 * (r % 4))
        else:
            # Random restart from parent with noise
            a_seed = parent + rng_r.normal(0, 0.05, n)
            a_seed = np.maximum(a_seed, 0)
            a_seed = a_seed / a_seed.sum()

        # Quick local optimization
        sub_deadline = time.monotonic() + alloc * 0.7
        a_r = coord_descent_fast(a_seed, deadline=sub_deadline, random_order=True, seed=2000 + r)
        update_best(a_r)

        # SCA polish
        sub_deadline2 = time.monotonic() + alloc * 0.3
        if time.monotonic() < sub_deadline2:
            a_r2, obj_r2, _ = solve_lp_sca(a_r, deadline=sub_deadline2)
            update_best(a_r2)

        print(f"  Restart {r}: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Different length exploration (100s) ===
    t5 = t_start + 1150.0
    print(f"\n[Phase 5] Length exploration (budget {t5 - time.monotonic():.0f}s)", flush=True)
    candidate_ns = [512, 640, 768, 896, 1024, 1280]
    candidate_ns = [cn for cn in candidate_ns if cn != n][:3]

    for cn in candidate_ns:
        if time.monotonic() > t5:
            break
        alloc = min(50.0, t5 - time.monotonic() - 10)
        if alloc < 20:
            break

        a_ext = resample_seq(best_seq, cn)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        sub_d = time.monotonic() + alloc * 0.6
        a_ext = coord_descent_fast(a_ext, deadline=sub_d, random_order=True, seed=5000)
        update_best(a_ext)

        sub_d2 = time.monotonic() + alloc * 0.4
        if time.monotonic() < sub_d2:
            a_ext2, _, _ = solve_lp_sca(a_ext, deadline=sub_d2)
            update_best(a_ext2)

        print(f"  n={cn}: J={compute_J(np.clip(a_ext, 0, 1000)):.8f}", flush=True)

    # === PHASE 6: Final intensive optimization (remaining time) ===
    t6 = deadline
    remaining = t6 - time.monotonic()
    print(f"\n[Phase 6] Final polish (budget {remaining:.0f}s)", flush=True)

    if remaining > 100:
        # Long CD
        t6a = time.monotonic() + 0.3 * remaining
        a6a = coord_descent_fast(best_seq, deadline=t6a, random_order=True, seed=9999)
        update_best(a6a)

        remaining = t6 - time.monotonic()
        if remaining > 50:
            # SCA-LP with lots of iterations
            t6b = time.monotonic() + 0.5 * remaining
            a6b, obj6b, _ = solve_lp_sca(a6a, deadline=t6b)
            update_best(a6b)

            remaining = t6 - time.monotonic()
            if remaining > 30:
                # Final CD
                a6c = coord_descent_fast(best_seq, deadline=t6 - 5, random_order=True, seed=7777)
                update_best(a6c)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
