"""
Optimizer for minimizing J(a) = 2n * max(conv(a,a)) / (sum(a))^2.
Uses: GS (Gerchberg-Saxton) initialization, improved coordinate descent with
breakpoint analysis, active-set LP subproblems, smooth subgradient methods,
and random restarts for escaping local minima.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gs_init(n, n_iter=300, seed=42):
    """Generate initialization using Gerchberg-Saxton algorithm for flat autocorrelation."""
    rng = np.random.default_rng(seed)
    L = 1
    while L < 2 * n - 1:
        L *= 2

    a = rng.uniform(0.5, 1.5, n)
    a /= a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    for it in range(n_iter):
        A = np.fft.rfft(a, L)
        mag = np.abs(A)
        energy = np.sum(a ** 2)
        target_mag = np.sqrt(energy)
        A_new = target_mag * A / np.maximum(mag, 1e-15)
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a = rng.uniform(0.5, 1.5, n)
            a /= a.sum()
            continue
        a = a_new / s
        obj = compute_J(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def coord_descent(a, max_sweeps, deadline, random_order=False, seed=0, n_candidates=12):
    """Coordinate descent with improved candidate selection."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 500

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order and sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved = False

        for i in order:
            old = a[i]
            S_prime = s - old

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            two_i = 2 * i
            two_i_in_range = (k_min <= two_i <= k_max)

            # Build candidate list
            if old < 1e-10:
                candidates = np.array([0.0, 1e-5, 1e-4, 1e-3, 5e-3, 1e-2, 5e-2, 0.1])
            else:
                # Include ratio-based candidates and breakpoint estimates
                ratios = [0.0, 0.1, 0.25, 0.4, 0.55, 0.7, 0.85, 1.0, 1.15, 1.35, 1.6, 2.0]
                candidates = np.array([r * old for r in ratios])

                # Find the dominant lag and estimate breakpoint
                conv_max_idx = np.argmax(conv)
                if conv_max_idx != two_i and 0 <= conv_max_idx - i < n:
                    ai_k = a[conv_max_idx - i]
                    if ai_k > 1e-10:
                        # Estimate where reducing a[i] would help
                        # conv[k](x) = conv[k] + 2*(x-old)*a[k-i]
                        # We want to find x where this equals the second-highest conv
                        other_vals = conv.copy()
                        other_vals[conv_max_idx] = -np.inf
                        second_max = np.max(other_vals)
                        # conv[k](x) = second_max => x = old + (second_max - conv[k]) / (2*a[k-i])
                        x_bp = old + (second_max - conv[conv_max_idx]) / (2.0 * ai_k)
                        if 0 <= x_bp <= 3 * old:
                            candidates = np.append(candidates, x_bp)

            candidates = np.unique(np.maximum(candidates, 0.0))

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                dx = x - old
                if abs(dx) < 1e-15:
                    continue

                if two_i_in_range:
                    # Apply linear update to all affected k except 2i
                    ks = np.arange(k_min, k_max + 1)
                    conv[ks] += 2.0 * dx * a[ks - i]
                    # Correct: at k=2i, the linear update should be dx*(a[0] + ...) 
                    # Actually the linear formula gives 2*dx*a[k-i], but at k=2i, a[k-i]=a[i]=old
                    # The correct change is x^2 - old^2 = (x-old)(x+old), not 2*dx*old
                    # So we need to adjust: subtract 2*dx*old and add (x^2 - old^2)
                    conv[two_i] -= 2.0 * dx * old
                    conv[two_i] += x * x - old * old
                else:
                    ks = np.arange(k_min, k_max + 1)
                    conv[ks] += 2.0 * dx * a[ks - i]

                new_s = S_prime + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                # Undo
                if two_i_in_range:
                    conv[ks] -= 2.0 * dx * a[ks - i]
                    conv[two_i] += 2.0 * dx * old
                    conv[two_i] -= (x * x - old * old)
                else:
                    conv[ks] -= 2.0 * dx * a[ks - i]

            if best_val != old and abs(best_val - old) > 1e-14:
                dx = best_val - old
                ks = np.arange(k_min, k_max + 1)
                if two_i_in_range:
                    conv[ks] += 2.0 * dx * a[ks - i]
                    conv[two_i] -= 2.0 * dx * old
                    conv[two_i] += best_val * best_val - old * old
                else:
                    conv[ks] += 2.0 * dx * a[ks - i]
                a[i] = best_val
                s = S_prime + best_val
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    return best_a


def active_set_lp(a, M, max_iters, deadline):
    """
    Active-set LP: only constrain the top M lags of the convolution.
    Much faster than full LP for large n.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        # Find top M lags
        top_M = np.argsort(c)[::-1][:M]
        M_val = c[top_M[0]]

        # LP: minimize t
        # s.t. 2*(a*g)[k] - t <= c[k] for k in top_M
        #      sum(g) = 1, g >= 0
        rows = []
        cols = []
        vals = []
        b_ub = []

        for idx, k in enumerate(top_M):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(idx)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(M, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs', options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [1.0, 0.6, 0.4, 0.25, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_obj


def smooth_subgradient(a, max_iters, deadline, tau_init=50.0, tau_max=300.0, lr_init=0.02):
    """Smooth subgradient method with log-sum-exp approximation of max."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_init + (tau_max - tau_init) * frac

        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()

        # Gradient of smooth max: w[k] gives weight to each conv[k]
        # d/dx_i of conv[k] = a[k-i] + a[i-k] (with bounds)
        # Use FFT for efficiency
        a_rev = a[::-1]
        size = 1
        while size < 2 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        conv_w = np.fft.irfft(W_fft * A_fft, size)
        grad = conv_w[:n].copy()

        # Project gradient to simplex tangent space
        grad = grad - np.mean(grad)

        t_count += 1
        lr = lr_init / (1.0 + 0.05 * t_count ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:

                a = best_a + rng.normal(0, 0.03, n)
                a = project_simplex(a)

                v = np.zeros(n)
                t_count = 0

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    # GS initialization
    try:
        gs = gs_init(n, n_iter=200, seed=rng.integers(10000))
        inits.append(gs)
    except Exception:
        pass

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine/sine
    for freq in [1, 2, 3, 5, 7]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    # Gaussian bumps
    for center in [0.1, 0.3, 0.5, 0.7, 0.9]:
        for width in [0.05, 0.1, 0.2, 0.3]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.02))

    # Power-law
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    # Wigner semicircle
    theta = np.arccos(1 - 2 * t)
    inits.append(project_simplex(np.sin(theta) ** 2 + 0.01))

    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.02))

    # Bell-shaped
    inits.append(project_simplex(1.0 / (1.0 + ((t - 0.5) * 4) ** 4)))

    return inits


def optimize_single(a, deadline):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 1: Coordinate descent (40% time)
    t1 = time.monotonic() + 0.40 * remaining
    a_cd = coord_descent(best_a, max_sweeps=50000, deadline=t1, random_order=True, seed=42)
    try_update(a_cd)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 2: Active-set LP (30% time)
    M_lags = min(30, 2 * n - 1)
    t2 = time.monotonic() + 0.30 * remaining
    a_lp, obj_lp = active_set_lp(best_a, M_lags, max_iters=200, deadline=t2)
    if obj_lp < best_obj:
        best_a = a_lp.copy()
        best_obj = obj_lp

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 3: Smooth subgradient (20% time)
    t3 = time.monotonic() + 0.20 * remaining
    a_sg, obj_sg = smooth_subgradient(best_a, max_iters=3000, deadline=t3,
                                      tau_init=30.0, tau_max=200.0, lr_init=0.015)
    if obj_sg < best_obj:
        best_a = a_sg.copy()
        best_obj = obj_sg

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 4: Final CD polish (10% time)
    a_final = coord_descent(best_a, max_sweeps=30000, deadline=deadline - 1,
                            random_order=True, seed=999)
    try_update(a_final)

    return best_a, best_obj

            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Multi-init at parent length (35% time) ===
    phase1_deadline = t_start + 0.35 * (budget_s - 15)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-init optimization at n={n_parent}", flush=True)
    print(f"  Deadline: {phase1_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    # Prioritize: parent-based inits first, then GS, then structural
    # Reorder: parent, parent+noise, GS, then others
    n_init = min(len(inits), 15)
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, n_init) - 2

    for idx in range(n_init):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 20:
            break
        alloc = min(per_init_time, remaining_check - 2)
        if alloc < 10:
            break

        a0 = inits[idx]
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt, obj_opt = optimize_single(a0, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Multi-length exploration (30% time) ===
    phase2_deadline = t_start + 0.65 * (budget_s - 15)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"  Deadline: {phase2_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Try various lengths, prioritizing those near parent
    candidate_ns = set()
    for base in [n_parent]:
        for mult in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 0.6, 1.4, 1.8]:
            nn = int(round(base * mult))
            if 64 <= nn <= 4096:
                candidate_ns.add(nn)
    # Also try some "interesting" lengths
    for nn in [128, 192, 256, 320, 384, 448, 512, 576, 640, 704, 768, 832, 896, 1024, 1280, 1536]:
        if 64 <= nn <= 4096:
            candidate_ns.add(nn)

    candidate_ns = sorted([n for n in candidate_ns if n != n_parent], key=lambda x: abs(x - n_parent))

    for new_n in candidate_ns[:8]:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 40:
            break
        alloc = min(80.0, remaining_check - 15)
        if alloc < 25:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)

        # Resample best sequence to new length
        if best_seq is not None:
            a_ext = resample_seq(best_seq, new_n)
        else:
            a_ext = np.ones(new_n) / new_n
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = optimize_single(a_ext, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive local optimization on best (30% time) ===
    phase3_deadline = deadline
    remaining = phase3_deadline - time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive optimization (budget {remaining:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and remaining > 20:
        a_curr = best_seq.copy()

        # Pass 1: Long CD
        t_p1 = time.monotonic() + 0.20 * remaining
        print(f"  CD pass 1 (budget {t_p1 - time.monotonic():.0f}s)...", flush=True)
        a_curr = coord_descent(a_curr, max_sweeps=50000, deadline=t_p1,
                               random_order=True, seed=12345)
        update_best(a_curr)

        remaining = phase3_deadline - time.monotonic()
        if remaining < 20:
            pass

        # Pass 2: Active-set LP with more lags
        t_p2 = time.monotonic() + 0.25 * remaining
        M_lags2 = min(50, 2 * len(best_seq) - 1)
        print(f"  LP pass (M={M_lags2}, budget {t_p2 - time.monotonic():.0f}s)...", flush=True)
        a_lp, obj_lp = active_set_lp(best_seq, M_lags2, max_iters=300, deadline=t_p2)
        update_best(a_lp)

        remaining = phase3_deadline - time.monotonic()
        if remaining < 20:
            pass

        # Pass 3: Smooth subgradient with different params
        t_p3 = time.monotonic() + 0.20 * remaining
        print(f"  Subgradient pass (budget {t_p3 - time.monotonic():.0f}s)...", flush=True)
        a_sg, obj_sg = smooth_subgradient(best_seq, max_iters=5000, deadline=t_p3,
                                          tau_init=20.0, tau_max=150.0, lr_init=0.01)
        update_best(a_sg)

        remaining = phase3_deadline - time.monotonic()
        if remaining < 20:
            pass

        # Pass 4: Perturbation + CD (try to escape local minimum)
        t_p4 = time.monotonic() + 0.15 * remaining
        print(f"  Perturbation + CD (budget {t_p4 - time.monotonic():.0f}s)...", flush=True)
        rng2 = np.random.default_rng(seed + 777)
        for pert_frac in [0.02, 0.05, 0.1, 0.15]:
            if time.monotonic() > t_p4:
                break
            a_pert = best_seq + rng2.normal(0, pert_frac * best_seq.mean(), len(best_seq))
            a_pert = project_simplex(a_pert)
            a_pert_cd = coord_descent(a_pert, max_sweeps=10000,
                                      deadline=min(time.monotonic() + 30, t_p4),
                                      random_order=True, seed=rng2.integers(10000))
            update_best(a_pert_cd)

        remaining = phase3_deadline - time.monotonic()
        if remaining > 15:
            # Final CD polish
            print(f"  Final CD (budget {remaining:.0f}s)...", flush=True)
            a_final = coord_descent(best_seq, max_sweeps=50000,
                                    deadline=phase3_deadline - 2,
                                    random_order=True, seed=424242)
            update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
