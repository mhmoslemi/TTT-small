Optimized step function search using sparse LP alternating, spectral shaping,

Key improvements: sparse Toeplitz LP, parabolic line search, peak-shaving gradient,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft, rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix
import time

def compute_conv(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = fft(a, N)
    conv = ifft(fa * fa).real[:2 * n - 1]
    return conv

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_sparse_toeplitz(a):

    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A.tocsc()

def solve_lp_max_sum(a, M, time_limit=20):

def parabolic_line_search(a, g, a_current, obj_current, proj=project_nonneg_sum1):

    Uses parabolic interpolation after initial grid search.
    """

    best_t = 0.0
    best_obj = obj_current
    best_a = a_current.copy()
    
    objs = []
    for t in ts:
        a_t = proj((1 - t) * a_current + t * g)
        obj_t = compute_obj(a_t)

        if obj_t < best_obj:

            best_t = t
            best_a = a_t

def peak_shaving_gradient(a, n_steps=100, time_budget=20.0):
    """
    Gradient step that specifically targets reducing the peak of conv(a,a).
    Uses a smoothed max (log-sum-exp) for differentiability.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Log-sum-exp weighting
        tau = 0.01 * cmax + 0.001
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -50, 0)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient of weighted conv
        grad = np.zeros(n)
        n_conv = len(weights)
        # Efficient computation using FFT
        # grad[i] = sum_k weights[k] * a[k-i] (for valid k-i)
        # This is cross-correlation of weights and a
        N_fft = 1
        while N_fft < n_conv + n - 1:
            N_fft <<= 1
        # Use the fact that grad[i] = sum_k w[k] * a[k-i]
        # = (w_rev * a)[n-1-i + offset]... 
        # Simpler: just compute directly for the significant weights
        significant = np.where(weights > 1e-6)[0]
        for k in significant:
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_end > i_start:
                idx = np.arange(i_start, i_end)
                a_idx = k - idx
                grad[idx] += weights[k] * a[a_idx]
        
        # Center
        grad = grad - grad.mean()
        
        # Line search
        for ss in [step_size, step_size * 0.5, step_size * 2]:
            a_try = a - ss * grad
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break
        
        step_size *= 0.99
    
    return best_a, best_obj

def spectral_flatten(a, n_iters=50, time_budget=20.0):
    """
    Frequency-domain approach: iteratively modify the amplitude spectrum
    to flatten the auto-correlation.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = fft(a, N)
        # Power spectrum
        ps = np.abs(fa) ** 2
        ps_max = ps.max()
        if ps_max < 1e-15:
            break
        
        # Target: flat power spectrum (which would give flat autocorrelation)
        # But we need to maintain the DC component (sum constraint)
        # Compress dynamic range
        ps_norm = ps / ps_max
        alpha = 0.8 + 0.15 * (it / n_iters)  # gradually reduce compression
        ps_compressed = np.maximum(ps_norm, 1e-10) ** alpha
        
        # Reconstruct with modified amplitude, keep phase
        phase = np.angle(fa)
        new_amp = np.sqrt(ps_compressed) * np.sqrt(ps_max)
        fa_new = new_amp * np.exp(1j * phase)
        
        # Also: enforce Hermitian symmetry (a is real)
        # fft of real signal has conjugate symmetry
        # Since we're working with full FFT, this should be automatic if we
        # handle it correctly. Let's just take the real part after IFFT.
        
        a_new = ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_obj

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0/new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangle
    x = np.linspace(0, 1, n)
    tri = 1 - np.abs(2*x - 1)
    inits.append(tri / tri.sum())
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    rc = np.maximum(rc, 0.01)
    inits.append(rc / rc.sum())
    
    # Sine
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())
    
    # Gaussian (centered)
    g = np.exp(-((x - 0.5) ** 2) / 0.08)
    inits.append(g / g.sum())
    
    # Wide Gaussian
    gw = np.exp(-((x - 0.5) ** 2) / 0.2)
    inits.append(gw / gw.sum())
    
    # Tukey-like (flat middle, cosine edges)
    w = 0.5  # width of cosine taper
    tukey = np.ones(n)
    taper = int(n * w / 2)
    for i in range(min(taper, n//2)):
        t = 0.5 * (1 + np.cos(np.pi * (i + 0.5) / taper))
        tukey[i] = t
        tukey[-1-i] = t
    inits.append(tukey / tukey.sum())
    
    # Random
    rnd = rng.random(n) + 0.1
    inits.append(rnd / rnd.sum())
    
    # Power law
    pl = x ** 0.5 + (1-x) ** 0.5
    inits.append(pl / pl.sum())
    
    if parent is not None:
        # Resampled parent
        inits.append(project_nonneg_sum1(resample(parent, n)))
        # Perturbed parent
        p = resample(parent, n)
        pert = p * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_nonneg_sum1(pert))
        # Scaled parent
        for s in [0.8, 1.2]:
            p_scaled = resample(parent, n) * s
            inits.append(project_nonneg_sum1(p_scaled))
    
    return inits

"""
Optimized step function search using sparse LP alternating with parabolic line search,
peak-shaving gradient descent, spectral amplitude shaping, and adaptive multi-length
exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Key improvements over prior approach: sparse Toeplitz LP for larger n, parabolic
line search, log-sum-exp smoothed gradient, and focused time allocation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft
from scipy.sparse import lil_matrix, csc_matrix
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n <= 1:
        return np.array([float(a[0] * a[0])]) if n == 1 else np.array([0.0])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = fft(a, N)
    conv = ifft(fa * fa).real[:2 * n - 1]
    return conv


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_sparse_toeplitz(a):
    """Build sparse matrix T where (T@g)[k] = sum_j a[k-j]*g[j] (cross-conv)."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A.tocsc()

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def parabolic_line_search(a_cur, g, obj_cur):
    """Find best t: a(t) = (1-t)*a_cur + t*g using parabolic interpolation."""
    ts = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
    best_t = 0.0
    best_obj = obj_cur
    best_a = a_cur.copy()
    objs = []
    for t in ts:
        a_t = project((1.0 - t) * a_cur + t * g)
        obj_t = compute_obj(a_t)
        objs.append(obj_t)
        if obj_t < best_obj:
            best_obj = obj_t
            best_t = t
            best_a = a_t
    # Parabolic refinement
    if 0 < best_t < 1.0:
        idx = ts.index(best_t)
        if 0 < idx < len(ts) - 1:
            t0, t1, t2 = ts[idx - 1], ts[idx], ts[idx + 1]
            f0, f1, f2 = objs[idx - 1], objs[idx], objs[idx + 1]
            denom = f0 - 2.0 * f1 + f2
            if abs(denom) > 1e-15:
                t_star = t1 - 0.5 * ((t1 - t0) ** 2 * (f1 - f2) - (t1 - t2) ** 2 * (f1 - f0)) / denom
                if t0 < t_star < t2:
                    a_star = project((1.0 - t_star) * a_cur + t_star * g)
                    obj_star = compute_obj(a_star)
                    if obj_star < best_obj:
                        best_obj = obj_star
                        best_t = t_star
                        best_a = a_star
    return best_a, best_obj


def lp_alternate(a, max_iters=80, time_budget=50.0):
    """LP alternating with parabolic line search."""
    n = len(a)
    a = project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        lp_tl = min(15.0, remaining * 0.5)
        g = solve_lp_for_g(a, M, time_limit=lp_tl)
        if g is None:
            break

        a_new, obj_new = parabolic_line_search(a, g, best_obj)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break

    return best_a, best_obj


def peak_shaving(a, n_steps=80, time_budget=15.0):
    """
    Gradient step targeting the peak of conv(a,a) using log-sum-exp smoothing.
    Uses FFT for efficient gradient computation.
    """
    n = len(a)
    a = project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.08

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        if cmax < 1e-15:
            break

        # Log-sum-exp smoothing
        tau = max(0.005 * cmax, 1e-6)
        shifted = np.clip((conv - cmax) / tau, -40.0, 0.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= wsum

        # Gradient via cross-correlation (FFT-based)
        N_fft = 1
        while N_fft < len(weights) + n - 1:
            N_fft <<= 1
        fw = fft(weights, N_fft)
        fa = fft(a, N_fft)
        # grad[i] = sum_k weights[k] * a[k-i]  (cross-correlation)
        # = ifft(fft(w) * conj(fft(a))) with appropriate shift
        corr = ifft(fw * np.conj(fa)).real
        # corr[m] = sum_k w[k] * a[k+m-n+1]... need to be careful
        # Actually: ifft(fft(w)*conj(fft(a)))[m] = sum_k w[k] * a[(k-m) mod N]
        # We want grad[i] = sum_k w[k] * a[k-i] for 0<=k-i<n
        # Let's just use the Toeplitz approach for correctness
        grad = np.zeros(n)
        # Only process significant weights
        thresh = weights.max() * 1e-4
        sig_idx = np.where(weights > thresh)[0]
        for k in sig_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n, k + 1)
            if i_hi > i_lo:
                idx = np.arange(i_lo, i_hi)
                a_idx = k - idx
                grad[idx] += weights[k] * a[a_idx]

        grad = grad - grad.mean()
        gnrm = np.linalg.norm(grad)
        if gnrm < 1e-15:
            break

        # Try a few step sizes
        improved = False
        for ss in [step, step * 0.5, step * 1.5, step * 0.25]:
            a_try = project(a - ss * grad / gnrm * cmax)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        if not improved:
            step *= 0.7
            if step < 0.001:
                break
        else:
            step = min(step * 1.02, 0.15)

    return best_a, best_obj


def spectral_shape(a, n_iters=40, time_budget=15.0):
    """
    Iteratively compress the power spectrum to flatten autocorrelation.
    """
    n = len(a)
    a = project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = fft(a, N)
        ps = np.abs(fa) ** 2
        ps_max = ps.max()
        if ps_max < 1e-15:
            break

        ps_norm = ps / ps_max
        # Progressive compression: start strong, weaken over iterations
        alpha = 0.75 + 0.2 * (it / max(n_iters, 1))
        ps_comp = np.maximum(ps_norm, 1e-12) ** alpha

        # Keep DC component fixed (preserves sum)
        phase = np.angle(fa)
        new_mag = np.sqrt(ps_comp) * np.sqrt(ps_max)
        new_mag[0] = np.abs(fa[0])  # preserve DC
        fa_new = new_mag * np.exp(1j * phase)

        # Enforce Hermitian symmetry for real output
        fa_new[0] = np.abs(fa_new[0])
        if N % 2 == 0:
            fa_new[N // 2] = np.abs(fa_new[N // 2])
        for k in range(1, (N // 2) + 1):
            fa_new[N - k] = np.conj(fa_new[k])

        a_new = ifft(fa_new).real[:n]
        a_new = project(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, float(a[0]) if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangle
    tri = 1.0 - np.abs(2.0 * x - 1.0)
    inits.append(tri / tri.sum())

    # Sine
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    # Gaussian centered
    g = np.exp(-((x - 0.5) ** 2) / 0.1)
    inits.append(g / g.sum())

    # Wide Gaussian
    gw = np.exp(-((x - 0.5) ** 2) / 0.25)
    inits.append(gw / gw.sum())

    # Tukey window
    w_frac = 0.4
    tukey = np.ones(n)
    taper = max(1, int(n * w_frac / 2))
    for i in range(min(taper, n // 2)):
        t = 0.5 * (1.0 + np.cos(np.pi * (i + 0.5) / taper))
        tukey[i] = t
        tukey[n - 1 - i] = t
    inits.append(tukey / tukey.sum())

    # Raised cosine
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    rc = np.maximum(rc, 0.01)
    inits.append(rc / rc.sum())

    # Power-symmetric
    ps = x ** 0.7 + (1.0 - x) ** 0.7
    inits.append(ps / ps.sum())

    # Random
    rnd = rng.random(n) + 0.1
    inits.append(rnd / rnd.sum())

    if parent is not None:
        p = project(resample_seq(parent, n))
        inits.append(p)
        # Mild perturbation
        pert = p * np.exp(rng.normal(0, 0.03, n))
        inits.append(project(pert))
        # Stronger perturbation
        pert2 = p * np.exp(rng.normal(0, 0.1, n))
        inits.append(project(pert2))

    return inits


def try_load_parent():
    """Try to load the parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization with sparse LP alternating,
    gradient peak-shaving, spectral shaping, and adaptive n exploration.
    """
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        po = compute_obj(parent)
        print(f"Parent loaded: n={len(parent)}, obj={po:.8f}")
        if po < best_score:
            best_score = po
            best_seq = parent.copy()
    else:
        best_seq = np.ones(700)
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"No parent. Uniform n=700: obj={best_score:.8f}")

    # ===== Phase 1: Multi-length screening (10% budget) =====
    print("\n--- Phase 1: Multi-length Screening ---")
    p1_deadline = t0 + (deadline - t0) * 0.10
    n_values = sorted(set([500, 600, 700, 800, 1000, 1200, 1500] + [len(parent)]))
    n_values = [n for n in n_values if 100 <= n <= 2500]

    cands = []
    for nv in n_values:
        if time.monotonic() > p1_deadline - 3:
            break
        inits = make_inits(nv, rng, parent)
        for init in inits:
            if time.monotonic() > p1_deadline - 2:
                break
            a_q, o_q = peak_shaving(init, n_steps=25, time_budget=2.0)
            if o_q < np.inf:
                cands.append((o_q, a_q, nv))

    cands.sort(key=lambda x: x[0])
    if cands:
        print(f"  {len(cands)} candidates. Top: {[(round(c[0],6), c[2]) for c in cands[:4]]}")
    for o_c, a_c, n_c in cands[:5]:
        if o_c < best_score:
            best_score = o_c
            best_seq = a_c.copy()
            print(f"  Screening update: n={n_c}, obj={best_score:.8f}")

    # ===== Phase 2: Intensive optimization (60% budget) =====
    print(f"\n--- Phase 2: Intensive LP + Gradient (best={best_score:.8f}) ---")
    p2_deadline = t0 + (deadline - t0) * 0.70

    # Build optimization list: best + diverse top candidates
    opt_list = [(best_score, best_seq.copy(), len(best_seq))]
    seen = {len(best_seq)}
    for o_c, a_c, n_c in cands[:6]:
        if n_c not in seen and n_c != len(best_seq):
            opt_list.append((o_c, a_c.copy(), n_c))
            seen.add(n_c)
            if len(opt_list) >= 4:
                break

    for ci, (o_init, a_init, n_init) in enumerate(opt_list):
        rem_total = p2_deadline - time.monotonic()
        if rem_total < 20:
            break
        alloc = max(15.0, min(rem_total / max(1, len(opt_list) - ci), 150.0))
        print(f"  [{ci}] n={n_init}, init={o_init:.8f}, budget={alloc:.0f}s")

        a = project(a_init)
        local_best = a.copy()
        local_obj = compute_obj(a)

        # Cycle strategies until time runs out
        cycle = 0
        while time.monotonic() < p2_deadline - 5:
            rem = p2_deadline - time.monotonic()
            if rem < 5:
                break
            cycle += 1
            strat = (cycle + ci) % 4

            if strat == 0:
                a_new, o_new = lp_alternate(a, max_iters=min(60, int(rem // 0.3)),
                                            time_budget=min(rem * 0.45, 50.0))
            elif strat == 1:
                a_new, o_new = peak_shaving(a, n_steps=min(80, int(rem * 4)),
                                            time_budget=min(rem * 0.2, 15.0))
            elif strat == 2:
                a_new, o_new = spectral_shape(a, n_iters=min(40, int(rem * 2)),
                                              time_budget=min(rem * 0.2, 12.0))
            else:
                # LP followed by gradient
                a_new, o_new = lp_alternate(a, max_iters=25,
                                            time_budget=min(rem * 0.3, 20.0))
                a_new, o_new = peak_shaving(a_new, n_steps=40,
                                            time_budget=min(rem * 0.15, 10.0))

            if o_new < local_obj - 1e-12:
                local_obj = o_new
                local_best = a_new.copy()
                a = a_new.copy()
                if local_obj < best_score:
                    best_score = local_obj
                    best_seq = local_best.copy()
                    print(f"    ** [{ci}] n={n_init} c{cycle} s{strat}: {best_score:.8f} **")

        print(f"  [{ci}] n={n_init} done: {local_obj:.8f}")

    # ===== Phase 3: Focused refinement on global best (20% budget) =====
    print(f"\n--- Phase 3: Focused Refinement (best={best_score:.8f}) ---")
    p3_deadline = t0 + (deadline - t0) * 0.90
    a_best = project(best_seq.copy())

    n_rounds = max(3, int((p3_deadline - time.monotonic()) / 8))
    for r in range(n_rounds):
        if time.monotonic() > p3_deadline - 3:
            break
        rem = p3_deadline - time.monotonic()
        if rem < 5:
            break

        # LP alternating
        a_p, o_p = lp_alternate(a_best, max_iters=min(40, int(rem)),
                                time_budget=min(rem * 0.35, 25.0))
        if o_p < best_score:
            best_score = o_p
            best_seq = a_p.copy()
            a_best = a_p.copy()
            print(f"  r{r} LP: {best_score:.8f}")

        # Gradient
        rem = p3_deadline - time.monotonic()
        if rem > 5:
            a_g, o_g = peak_shaving(a_best, n_steps=50,
                                    time_budget=min(rem * 0.25, 12.0))
            if o_g < best_score:
                best_score = o_g
                best_seq = a_g.copy()
                a_best = a_g.copy()
                print(f"  r{r} Grad: {best_score:.8f}")

        # Spectral
        rem = p3_deadline - time.monotonic()
        if rem > 5:
            a_s, o_s = spectral_shape(a_best, n_iters=30,
                                      time_budget=min(rem * 0.25, 10.0))
            if o_s < best_score:
                best_score = o_s
                best_seq = a_s.copy()
                a_best = a_s.copy()
                print(f"  r{r} Spec: {best_score:.8f}")

    # ===== Phase 4: Perturbation search + final validation (10% budget) =====
    print(f"\n--- Phase 4: Perturbation + Validate (best={best_score:.8f}) ---")
    n_best = len(best_seq)

    for pidx in range(8):
        rem = deadline - time.monotonic()
        if rem < 8:
            break
        sigma = 0.005 + 0.005 * pidx
        a_pert = best_seq * np.exp(rng.normal(0, sigma, n_best))
        a_pert = project(a_pert)
        a_pert, o_pert = lp_alternate(a_pert, max_iters=20, time_budget=min(rem * 0.3, 12.0))
        if o_pert < best_score:
            best_score = o_pert
            best_seq = a_pert.copy()
            print(f"  Pert {pidx}: {best_score:.8f}")

    # Final validation using evaluate_sequence
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 200
        best_score = compute_obj(np.array(result))
        print("  WARNING: fallback used")

    elapsed = time.monotonic() - t0
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
