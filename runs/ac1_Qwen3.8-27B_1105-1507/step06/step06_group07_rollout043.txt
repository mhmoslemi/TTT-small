"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """
    Compute smoothed objective (LSE of beta*conv) and gradient w.r.t. a.
    Returns (obj_smoothed, grad_a) where grad_a = d(obj)/d(a).
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    if sum_a < 1e-15:
        return np.inf, np.zeros(n)

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum

    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)

    # Gradient of LSE w.r.t. a[j] = 2 * cross_corr(w, a)[j]
    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def adam_log_opt(a, max_iters=10000, time_budget=60.0, beta_start=10.0,
                 beta_end=300.0, lr_start=0.04, lr_end=0.003,
                 nesterov=True, seed=0):
    """
    Nesterov Adam optimization in log-space with cosine LR decay.
    Works on normalized (sum=1) non-negative sequences.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    s = np.sum(a)
    if s < 1e-15:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_true_obj = compute_obj(a)

    start = time.monotonic()
    x_prev = x.copy()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr_t = lr_start + 0.5 * (lr_end - lr_start) * (1 + np.cos(np.pi * progress))

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)

        obj_sm, grad_a = smooth_obj_grad(a, beta)

        grad_x = grad_a * a

        if nesterov:
            x_prev = x.copy()
            x = x - lr_t * beta1 * m / (np.sqrt(v) + eps)
            x_centered2 = x - np.max(x)
            a2 = np.exp(x_centered2)
            a2 = a2 / np.sum(a2)
            obj_sm2, grad_a2 = smooth_obj_grad(a2, beta)
            grad_x = grad_a2 * a2

        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        if not nesterov:
            x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 100 == 0:
            x -= np.mean(x)

        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

    return best_a

def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (a*g)[k] for full convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': min(time_budget * 0.5, 10)})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    if time.monotonic() - start > time_budget:
                        break
                    a_try = (1 - t) * a_norm + t * g
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a
    except Exception:
        pass
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def build_asymmetric(n, shape='exp', seed=42):
    """Build an asymmetric initial sequence of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)

    if shape == 'exp_inc':
        a = np.exp(0.5 * x / n)
    elif shape == 'exp_dec':
        a = np.exp(-0.5 * x / n)
    elif shape == 'power_inc':
        a = (x + 1) ** 0.5
    elif shape == 'power_dec':
        a = (n - x) ** 0.5
    elif shape == 'ramp_right':
        a = x + 1
    elif shape == 'ramp_left':
        a = n - x
    elif shape == 'asym_bump':
        center = 0.6 * n
        a = np.exp(-((x - center) / (0.3 * n)) ** 2)
    elif shape == 'asym_bump2':
        center = 0.7 * n
        a = np.exp(-((x - center) / (0.25 * n)) ** 2)
    elif shape == 'hann_asym':
        t = x / max(n - 1, 1)
        a = 0.5 * (1 + np.cos(2 * np.pi * (t - 0.2)))
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    return a


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 0.01:
                parent = parent / s
                obj = compute_obj(parent)
                print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multiple strategies:
    1. Try several asymmetric initializations
    2. Adam optimization with Nesterov momentum and cosine LR
    3. Peak-targeted coordinate descent
    4. LP-based refinement
    5. Multi-length exploration
    6. Random restarts with structural perturbations
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: best={best_score:.8f}, n={n_parent}, budget={budget_s}s")

    # === Phase 1: Try multiple asymmetric initializations at parent length ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Asymmetric initializations (n={n_parent})")
    print(f"{'='*60}")

    shapes = ['exp_inc', 'exp_dec', 'power_inc', 'power_dec',
              'ramp_right', 'ramp_left', 'asym_bump', 'asym_bump2']

    if time.monotonic() < deadline - 200:
        for shape in shapes:
            if time.monotonic() > deadline - 200:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.05, 15.0)
            if alloc < 3:
                break

            a_init = build_asymmetric(n_parent, shape, seed=seed + hash(shape) % 1000)
            a_opt = adam_log_opt(a_init, max_iters=5000, time_budget=alloc,
                                 beta_start=5.0, beta_end=100.0,
                                 lr_start=0.05, lr_end=0.01)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  {shape}: {best_score:.8f} *** NEW BEST")
            else:
                print(f"  {shape}: {obj:.8f}")

    # === Phase 2: Intensive Adam on best sequence ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Intensive Adam optimization (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        a = a / np.sum(a)

        rounds = [
            (5.0, 50.0, 0.05, 0.015),
            (15.0, 150.0, 0.04, 0.01),
            (40.0, 300.0, 0.03, 0.008),
            (80.0, 500.0, 0.02, 0.005),
            (150.0, 800.0, 0.015, 0.003),
            (300.0, 1200.0, 0.01, 0.002),
        ]

        for r_idx, (b_s, b_e, lr_s, lr_e) in enumerate(rounds):
            if time.monotonic() > deadline - 120:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.12, 60.0)
            if alloc < 5:
                break

            a_new = adam_log_opt(a, max_iters=20000, time_budget=alloc,
                                 beta_start=b_s, beta_end=b_e,
                                 lr_start=lr_s, lr_end=lr_e)
            obj = compute_obj(a_new)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  Round {r_idx} (beta {b_s:.0f}-{b_e:.0f}): {best_score:.8f} ***")
            else:
                print(f"  Round {r_idx} (beta {b_s:.0f}-{b_e:.0f}): {obj:.8f}")
                a = a_new.copy() if compute_obj(a_new) < compute_obj(a) else a

    # === Phase 3: Peak-targeted coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Peak-targeted refinement (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 80.0)
        if alloc > 5:
            a_pt = peak_target_refine(best_seq, time_budget=alloc, seed=seed)
            obj = compute_obj(a_pt)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_pt.copy()
                print(f"  Peak target: {best_score:.8f} ***")
            else:
                print(f"  Peak target: {obj:.8f}")

            # Follow-up Adam after peak targeting
            if time.monotonic() < deadline - 60:
                remaining2 = deadline - time.monotonic()
                alloc2 = min(remaining2 * 0.1, 30.0)
                a_post = adam_log_opt(best_seq, max_iters=10000, time_budget=alloc2,
                                      beta_start=100.0, beta_end=600.0,
                                      lr_start=0.02, lr_end=0.003)
                obj_post = compute_obj(a_post)
                if obj_post < best_score - 1e-12:
                    best_score = obj_post
                    best_seq = a_post.copy()
                    print(f"  Post-PT Adam: {best_score:.8f} ***")
                else:
                    print(f"  Post-PT Adam: {obj_post:.8f}")

    # === Phase 4: LP refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 4: LP refinement (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None and len(best_seq) <= 600:
        for r_idx in range(4):
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.1, 30.0)
            if alloc < 5:
                break

            a_lp = lp_refine(best_seq, time_budget=alloc)
            obj = compute_obj(a_lp)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  LP {r_idx}: {best_score:.8f} ***")
            else:
                print(f"  LP {r_idx}: {obj:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-length exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 200),
            max(200, n_best - 100),
            n_best,
            n_best + 100,
            n_best + 200,
            n_best + 400,
            n_best + 600,
        ]))
        n_options = [n for n in n_options if 200 <= n <= 2500]

        for n_try in n_options:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
            a_try = a_try / np.sum(a_try)

            alloc = min(remaining * 0.1, 40.0)
            a_opt = adam_log_opt(a_try, max_iters=10000, time_budget=alloc,
                                 beta_start=10.0, beta_end=400.0,
                                 lr_start=0.04, lr_end=0.005)
            obj = compute_obj(a_opt)

            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 6: Random restarts with structural perturbations ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Random restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        for restart in range(10):
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            if restart % 4 == 0:
                # Multiplicative log-normal noise
                scale = 0.05 + 0.03 * restart
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            elif restart % 4 == 1:
                # Reverse + noise (exploits asymmetry)
                a_pert = best_seq[::-1].copy()
                a_pert *= np.exp(rng.normal(0, 0.05, n_best))
            elif restart % 4 == 2:
                # Blend with asymmetric structure
                alpha = rng.uniform(0.2, 0.5)
                struct = build_asymmetric(n_best, 'asym_bump', seed=seed + restart)
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * struct
            else:
                # Sparse spike perturbation
                a_pert = best_seq.copy()
                n_spikes = max(3, n_best // 30)
                for _ in range(n_spikes):
                    idx = int(rng.integers(0, n_best))
                    a_pert[idx] += rng.uniform(0.05, 0.3) * np.mean(best_seq)

            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / np.sum(a_pert)

            alloc = min(remaining * 0.1, 20.0)
            a_opt = adam_log_opt(a_pert, max_iters=8000, time_budget=alloc,
                                 beta_start=20.0, beta_end=500.0,
                                 lr_start=0.03, lr_end=0.005)
            obj = compute_obj(a_opt)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart}: {obj:.8f}")

    # === Phase 7: Final high-beta polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.5, 40.0)
        if alloc > 3:
            a_final = adam_log_opt(best_seq, max_iters=15000, time_budget=alloc,
                                   beta_start=200.0, beta_end=2000.0,
                                   lr_start=0.01, lr_end=0.001)
            obj_final = compute_obj(a_final)
            if obj_final < best_score - 1e-12:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final Adam: {best_score:.8f} ***")
            else:
                print(f"  Final Adam: {obj_final:.8f}")

        # One more peak-target pass
        if time.monotonic() < deadline - 5:
            remaining2 = deadline - time.monotonic()
            a_pt2 = peak_target_refine(best_seq, time_budget=min(remaining2 * 0.5, 15.0),
                                       seed=seed + 999)
            obj_pt2 = compute_obj(a_pt2)
            if obj_pt2 < best_score - 1e-12:
                best_score = obj_pt2
                best_seq = a_pt2.copy()
                print(f"  Final peak-target: {best_score:.8f} ***")
            else:
                print(f"  Final peak-target: {obj_pt2:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 700
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
