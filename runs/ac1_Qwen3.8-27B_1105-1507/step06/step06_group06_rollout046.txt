For this to equal sum_k w[k] * a[k-i], we need n-1-m+k = k-i, so n-1-m = -i, i.e., m = n-1+i.

So: (w ⊛ a)[i] = conv(w, a_flipped)[n-1+i]

Where conv is the linear convolution.

So grad[i] = 2 * conv(w, a[::-1])[n-1+i]

I can compute this via FFT:
- Zero-pad w and a_flipped to length N >= (2n-1) + n - 1 = 3n-2
- Compute conv via FFT
- Extract elements [n-1 : n-1+n]

Let me implement this:

```python
def compute_grad(c, a, tau):
    n = len(a)
    c_shifted = (c - c.max()) * tau
    c_shifted = np.clip(c_shifted, -50, 50)
    w = np.exp(c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        idx = np.argmax(c)
        w = np.zeros(len(c))
        w[idx] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    # = 2 * conv(w, a[::-1])[n-1 : n-1+n]
    a_flipped = a[::-1]
    len_conv = len(w) + n - 1  # = (2n-1) + n - 1 = 3n-2
    N = 1
    while N < len_conv:
        N <<= 1
    wf = fft(w, N)
    af = fft(a_flipped, N)
    conv_result = ifft(wf * af).real
    grad = 2.0 * conv_result[n-1:n-1+n]
    
    return grad
```

def spectral_smooth(a, alpha=0.3):
    """Blend the power spectrum with a "flat convolution" target."""
    n = len(a)
    N = 1
    while N < 2*n-1:
        N <<= 1
    
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    power = np.abs(A)**2
    
    # Target: we want IFFT(power) to be flat
    # FFT of flat signal of length 2n-1:
    flat = np.zeros(N)
    flat[:2*n-1] = 1.0
    target_power = np.abs(fft(flat))**2
    target_power = target_power / target_power.sum() * power.sum()
    
    # Blend
    new_power = (1-alpha) * power + alpha * target_power
    new_power = np.maximum(new_power, 0)
    
    # Reconstruct: keep phase, new magnitude
    new_A = np.sqrt(new_power) * np.exp(1j * np.angle(A))
    a_new = np.real(ifft(new_A))[:n]
    a_new = project_simplex(a_new)
    
    return a_new
