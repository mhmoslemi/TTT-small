Actually wait - it's `2*n*max_b/sum_a^2`. If we scale `a` by factor `c`, then `max_b` scales by `c^2` and `sum_a` scales by `c`, so the ratio `max_b/sum_a^2` is invariant. So the objective only depends on the shape (normalized to sum=1 or any other normalization).

2. **The key quantity to minimize**: `n * max(conv(a,a)) / sum(a)^2` (ignoring the factor of 2). With `sum(a) = 1`, this becomes `n * max(conv(a,a))`. So we want to minimize the peak of the autocorrelation for a unit-sum non-negative sequence of length n.

3. **What makes a good sequence?**: We want the convolution `a*a` to be as flat as possible (low peak). This is related to Barker sequences, CAZAC sequences, and other low-peak-cessation-autocorrelation (low-PACF) sequences. For non-negative sequences, this is different from complex sequences.

4. **Theoretical lower bound**: For a non-negative sequence of length n with sum 1, the convolution has sum = 1 (since sum(conv(a,a)) = (sum(a))^2 = 1). The convolution has 2n-1 terms. import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a

def fft_correlation(a, w):
    """
    Compute cross-correlation: corr[i] = sum_j a[j] * w[i+j]
    where w is zero-padded to length >= 2n-1.
    Uses FFT for efficiency.
    """
    n = len(a)
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    return corr[:n]

def soft_peak_subgradient(a, max_iters=5000, time_budget=60.0, tau0=0.005, tau_min=0.0005, step0=0.05, decay=0.9995):
    """
    FFT-accelerated subgradient with adaptive temperature and multi-peak weighting.
    Different from before: uses exponential soft-max weighting across peaks,
    more aggressive temperature decay, and adaptive step size.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    tau = tau0
    step = step0
    v = np.zeros(n)
    mu = 0.8  # momentum
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Soft-max weighting on the convolution
        shifted = np.clip((conv - cmax) / tau, -300.0, 300.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Compute gradient via FFT
        w_padded = np.zeros(fft_size)
        w_padded[:len(weights)] = weights
        A_freq = rfft(a, fft_size)
        W_freq = rfft(w_padded, fft_size)
        corr = irfft(A_freq * np.conj(W_freq), fft_size)
        grad = 2.0 * corr[:n]
        
        # Center the gradient (since we're on the simplex)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break
        
        # Nesterov momentum
        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        
        # Check if we made progress
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive temperature and step
        tau = max(tau * decay, tau_min)
        step *= 0.9999
        
    return best_a, best_obj

def peak_pair_elimination(a, max_iters=500, time_budget=30.0):
    """
    Target the specific coordinate pairs that contribute most to the peak.
    At each step, find the peak position k*, identify pairs (i, k*-i) with
    high products a[i]*a[k*-i], and reduce them.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = 0.02
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        
        # Find pairs contributing to the peak
        # conv[k_star] = sum_i a[i] * a[k_star - i] for valid i
        # For coordinate i, its contribution to the peak is a[i] * a[k_star - i]
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Also consider near-peak positions (top 5)
        top_positions = np.argsort(conv)[-5:]
        all_contrib = np.zeros(n)
        for k in top_positions:
            for i in range(n):
                j = int(k) - i
                if 0 <= j < n:
                    all_contrib[i] += a[i] * a[j]
        
        # Gradient: increase coordinates with low contribution, decrease those with high
        # But we also need to maintain sum = 1
        # Simple approach: reduce high-contribution coordinates, redistribute to low ones
        mean_contrib = np.mean(all_contrib)
        if np.std(all_contrib) < 1e-15:
            break
        
        # Create a direction: -sign(contribution - mean)
        direction = -(all_contrib - mean_contrib)
        direction = direction / (np.linalg.norm(direction) + 1e-15)
        
        a_new = a + step * direction
        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.95
            if step < 0.001:
                break
    
    return best_a, best_obj

def coordinate_descent_peak(a, max_iters=2000, time_budget=30.0):
    """
    Coordinate descent: at each step, find the coordinate that most reduces the peak
    when changed, and adjust it.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute correlation structure
    # For coordinate i, changing a[i] by delta changes conv[k] by 2*a[i]*delta (approximately, for the pairs involving i)
    # More precisely: conv[k] = sum_j a[j]*a[k-j], so d(conv[k])/d(a[i]) = a[k-i] + a[i-(k)] (if both valid)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        peak = float(np.max(conv))
        peak_pos = int(np.argmax(conv))
        
        # For each coordinate i, compute sensitivity to peak
        # d(peak)/d(a[i]) ≈ a[peak_pos - i] + a[i - peak_pos] (whichever is valid)
        # We want to reduce coordinates with high positive sensitivity to the peak
        
        sensitivity = np.zeros(n)
        for i in range(n):
            j = peak_pos - i
            if 0 <= j < n:
                sensitivity[i] = a[j]
            j2 = i - peak_pos
            if 0 <= j2 < n:
                sensitivity[i] = sensitivity[i] + a[j2]  # actually this would be wrong for linear conv
        
        # Actually for linear conv: conv[k] = sum_{i} a[i]*a[k-i] where both indices valid
        # d(conv[k])/d(a[i]) = a[k-i] (if 0<=k-i<n)
        # So for the peak at position k*:
        # d(peak)/d(a[i]) = a[k*-i] if 0 <= k*-i < n, else 0
        
        k_star = peak_pos
        sensitivity = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sensitivity[i] = a[j]
        
        # Reduce the coordinate with highest sensitivity (and positive value)
        # But this is too greedy; let's use a smoother approach
        # Find the coordinate with highest sensitivity * value
        impact = sensitivity * a
        max_impact_idx = int(np.argmax(impact))
        
        if impact[max_impact_idx] < 1e-10:
            break
        
        # Reduce this coordinate and redistribute
        delta = a[max_impact_idx] * 0.1  # reduce by 10%
        a[max_impact_idx] -= delta
        # Redistribute to coordinates with low sensitivity
        low_sens = np.where(sensitivity < np.median(sensitivity))[0]
        if len(low_sens) > 0:
            low_sens_mask = np.zeros(n)
            low_sens_mask[low_sens] = 1.0
            # Distribute proportionally to (1 - sensitivity/ max_sensitivity)
            weights = np.maximum(1.0 - sensitivity / (max(sensitivity) + 1e-15), 0.01)
            weights = weights * low_sens_mask
            w_sum = np.sum(weights)
            if w_sum > 0:
                a += delta * weights / w_sum
        
        a = normalize_simplex(a)
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

def lp_minpeak(a, time_limit=15):
    """
    Min-peak LP: min t s.t. (a*g)[k] <= t, sum(g)=1, g>=0.
    """
    n = len(a)
    if n > 800:
        return None, None
    
    # Build the convolution matrix
    # (a*g)[k] = sum_j a[k-j]*g[j] for valid k-j
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, j + n):
            A[k, j] = a[k - j]
    
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except:
        pass
    return None, None

def alternating_lp_subgradient(a, max_rounds=20, time_budget=200.0):

    start = time.monotonic()

        if time.monotonic() - start > time_budget:
            break
        
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Phase A: LP step (if n is small enough)
        if n <= 700 and remaining > 30:
            lp_time = min(15.0, remaining * 0.3)

def simulated_annealing_step(a, temperature, rng):
    """
    Perform a simulated annealing style perturbation.
    """
    n = len(a)
    # Perturb: multiply by random exp factors
    perturbation = np.exp(rng.normal(0, temperature, n))
    a_pert = a * perturbation
    return normalize_simplex(a_pert)

def annealing_search(a, max_iters=100, time_budget=60.0):
    """
    Simulated annealing: start with large perturbations, cool down.
    Accept worse solutions with probability exp(-delta/T).
    """
    n = len(a)
    rng = np.random.default_rng(12345)
    a = normalize_simplex(a.copy())
    curr_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = curr_obj
    start = time.monotonic()
    
    T0 = 0.05
    Tend = 0.005
    cooling = (Tend / T0) ** (1.0 / max_iters)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        T = T0 * cooling ** it
        
        a_new = simulated_annealing_step(a, T * 0.5, rng)
        obj_new = compute_obj(a_new)
        
        delta = obj_new - curr_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            curr_obj = obj_new
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def generate_structural_sequence(n, rng):
    """
    Generate a sequence with a specific structural form.
    Combines a base shape with harmonic modulation.
    """
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    # Random base shape
    shape_type = rng.integers(0, 4)
    if shape_type == 0:  # Uniform with edge tapering
        edge_width = n * (0.1 + 0.2 * rng.random())
        a = np.ones(n)
        taper = np.minimum(i, n - 1 - i) / (edge_width + 1e-10)
        a[:] = np.minimum(taper, 1.0)
    elif shape_type == 1:  # Gaussian
        sigma = n * (0.2 + 0.3 * rng.random())
        a = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
    elif shape_type == 2:  # Raised cosine
        width = n * (0.3 + 0.4 * rng.random())
        a = np.maximum(0.5 * (1 + np.cos(np.pi * (i - center) / width)), 0)
    else:  # Flat top with smooth edges
        flat_width = n * (0.3 + 0.3 * rng.random())
        a = np.ones(n)
        dist = np.abs(i - center)
        edge = np.maximum(dist - flat_width/2, 0)
        a[dist > flat_width/2] = np.exp(-edge[dist > flat_width/2] / (n * 0.1))
    
    # Add harmonic modulation
    num_harmonics = rng.integers(1, 4)
    for h in range(num_harmonics):
        freq = rng.integers(1, 10)
        amp = 0.05 + 0.15 * rng.random()
        phase = 2 * np.pi * rng.random()
        a = a * (1 + amp * np.cos(2 * np.pi * freq * i / n + phase))
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

"""
Peak-reduction search for step function coefficients minimizing
2*n*max(conv(a,a))/(sum(a))^2.

Strategy differs from prior approach by:
1. Multi-temperature parallel subgradient with adaptive restart
2. Geometry-informed structural perturbations (edge-emphasis)

6. Iterative peak-pair targeted coordinate reduction

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def get_fft_size(n):
    """Get next power of 2 >= 2n-1."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def soft_peak_subgradient(a, max_iters=5000, time_budget=60.0,
                          tau0=0.005, tau_min=0.0005, step0=0.05,
                          decay=0.9995, momentum=0.82, restart_patience=200):
    """
    FFT-accelerated subgradient with adaptive temperature, momentum,
    and automatic restart on stagnation.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    fft_size = get_fft_size(n)

    tau = tau0
    step = step0
    v = np.zeros(n)
    mu = momentum
    stall = 0
    last_improve = best_obj

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Soft-max weighting
        shifted = np.clip((conv - cmax) / tau, -300.0, 300.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient via FFT cross-correlation
        # corr[m] = sum_j a[j] * w[m+j]
        w_padded = np.zeros(fft_size)
        w_padded[:len(weights)] = weights
        A_freq = rfft(a, fft_size)
        W_freq = rfft(w_padded, fft_size)
        corr = irfft(A_freq * np.conj(W_freq), fft_size)
        grad = 2.0 * corr[:n]

        # Center gradient (tangent to simplex)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        # Nesterov momentum
        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            if obj_new < last_improve - 1e-10:
                stall = 0
                last_improve = obj_new
            else:
                stall += 1
        else:
            stall += 1

        a = a_new

        # Adaptive decay
        tau = max(tau * decay, tau_min)
        step *= 0.9999

        # Restart on stagnation
        if stall > restart_patience:
            tau = max(tau * 3.0, tau0 * 0.5)
            step = max(step * 1.5, step0 * 0.5)
            v = np.zeros(n)
            stall = 0
            # Small random perturbation
            rng = np.random.default_rng(it * 7919)
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = normalize_simplex(a)

    return best_a, best_obj


def peak_pair_reduction(a, max_iters=300, time_budget=25.0):
    """
    Directly target coordinate pairs contributing to the peak.
    At each step, find the peak position k*, compute per-coordinate
    contribution, and apply a targeted reduction.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.015

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Weight top 10 positions for smoother signal
        top_k = 10
        peak_positions = np.argsort(conv)[-top_k:]
        peak_weights = np.zeros(len(conv))
        for rank, pos in enumerate(peak_positions):
            peak_weights[pos] = (top_k - rank) / (top_k * (top_k + 1) / 2.0)

        # Compute per-coordinate contribution to weighted peak
        contributions = np.zeros(n)
        for k in peak_positions:
            k = int(k)
            w_k = peak_weights[k]
            for i in min(n, k + 1):
                j = k - i
                if 0 <= j < n:
                    contributions[i] += w_k * a[i] * a[j]
                    if i != j:
                        contributions[j] += w_k * a[j] * a[i] * 0.5

        # Direction: reduce high-contribution coordinates
        mean_c = np.mean(contributions)
        direction = -(contributions - mean_c)
        norm_d = np.linalg.norm(direction)
        if norm_d < 1e-15:
            break
        direction /= norm_d

        a_new = a + step * direction
        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.93
            if step < 0.001:
                # Try a different approach: increase low-contribution coords
                direction = contributions - mean_c
                norm_d = np.linalg.norm(direction)
                if norm_d > 1e-15:
                    a_try = a + 0.005 * direction / norm_d
                    a_try = normalize_simplex(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                else:
                    break

    return best_a, best_obj


def build_conv_mat(n):
    """Build the convolution matrix A where (A@g)[k] = (a*g)[k]."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        hi = min(j + n, 2 * n - 1)
        A[j:hi, j] = np.arange(hi - j)
    return A


def solve_minpeak_lp(a, time_limit=12):
    """
    Solve: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0.
    Uses the structure that A[k,j] = a[k-j] for valid indices.
    """
    n = len(a)
    if n > 800:
        return None, None

    # Build constraint matrix: [A_a, -1] where A_a[k,j] = a[k-j] if valid
    A_a = np.zeros((2 * n - 1, n))
    for j in range(n):
        j_start = j
        j_end = min(j + n, 2 * n - 1)
        A_a[j_start:j_end, j] = a[:j_end - j_start]

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.hstack([A_a, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None

                a, max_iters=int(sg_time * 25), time_budget=sg_time,
                tau0=0.004, tau_min=0.0004, step0=0.04, decay=0.9993)
            if obj_sg < best_obj - 1e-13:
                best_obj = obj_sg
                best_a = a_sg.copy()
                a = a_sg.copy()

    return best_a, best_obj


def edge_emphasis_perturbation(a, rng, strength=0.3):
    """
    Perturb towards edge-heavy sequences.
    The boundary convolution bins have fewer contributing pairs,
    so emphasizing edges can help flatten the profile.
    """
    n = len(a)
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # Edge emphasis: boost values near boundaries
    edge_boost = 1.0 + strength * np.cos(np.pi * i / (n - 1 + 1e-10))
    a_pert = a * edge_boost
    return normalize_simplex(a_pert)


def oscillatory_perturbation(a, rng, num_modes=3):
    """
    Add oscillatory structure to break symmetry.
    """
    n = len(a)
    i = np.arange(n, dtype=np.float64)
    mod = np.ones(n)
    for _ in range(num_modes):
        freq = rng.integers(1, 12)
        amp = 0.05 + 0.1 * rng.random()
        phase = 2 * np.pi * rng.random()
        mod = mod * (1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase))
    a_pert = a * mod
    return normalize_simplex(a_pert)


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def simulated_annealing(a, max_iters=80, time_budget=45.0, seed=42):
    """
    Simulated annealing with problem-aware proposals.
    Proposals combine random perturbations with structural moves.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    a = normalize_simplex(a.copy())
    curr_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = curr_obj
    start = time.monotonic()

    T0 = 0.04
    Tend = 0.003
    cooling = (Tend / T0) ** (1.0 / max(max_iters, 1))

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        T = T0 * cooling ** it

        # Choose proposal type
        r = rng.random()
        if r < 0.3:
            # Random multiplicative
            a_new = a * np.exp(rng.normal(0, T * 0.7, n))
        elif r < 0.5:
            # Edge emphasis
            a_new = edge_emphasis_perturbation(a, rng, strength=T * 3)
        elif r < 0.7:
            # Oscillatory
            a_new = oscillatory_perturbation(a, rng, num_modes=1 + int(rng.integers(0, 3)))
        elif r < 0.85:
            # Pair swap: move mass from one region to another
            i1 = rng.integers(0, n // 3)
            i2 = rng.integers(2 * n // 3, n)
            i3 = rng.integers(n // 3, 2 * n // 3)
            a_new = a.copy()
            amount = a[i1] * 0.3 + a[i2] * 0.3
            a_new[i1] -= amount * 0.5
            a_new[i2] -= amount * 0.5
            a_new[i3] += amount
        else:
            # Local blur
            kernel = np.ones(3) / 3.0
            a_new = np.convolve(a, kernel, mode='same')

        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)

        delta = obj_new - curr_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            curr_obj = obj_new
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()

    return best_a, best_obj


def multi_config_subgradient(a, time_budget=80.0):
    """
    Try multiple subgradient configurations and return the best.
    Different temperature/step/momentum settings explore different
    regions of the landscape.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    configs = [
        # (tau0, tau_min, step0, decay, momentum)
        (0.003, 0.0003, 0.04, 0.9993, 0.80),
        (0.007, 0.0007, 0.05, 0.9996, 0.85),
        (0.001, 0.0001, 0.02, 0.9988, 0.75),
        (0.012, 0.0012, 0.06, 0.9998, 0.88),
        (0.005, 0.0005, 0.035, 0.9994, 0.82),
        (0.002, 0.0002, 0.03, 0.9991, 0.78),
    ]

    per_config_time = (time_budget - (time.monotonic() - start)) / len(configs)

    for cfg in configs:
        if time.monotonic() - start > time_budget:
            break
        tau0, tau_min, step0, decay, mu = cfg
        t_alloc = min(per_config_time * 1.2, 25.0)

        a_try, obj_try = soft_peak_subgradient(
            best_a, max_iters=int(t_alloc * 30), time_budget=t_alloc,
            tau0=tau0, tau_min=tau_min, step0=step0, decay=decay, momentum=mu)

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy peak reduction for step function coefficients.
    Combines multi-config subgradient, LP alternating, peak-pair reduction,
    geometry-informed perturbations, and length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # === Phase 0: Load parent ===
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent = normalize_simplex(parent)
            parent_obj = compute_obj(parent)
            best_score = parent_obj
            best_seq = parent.copy()
            print(f"Parent: n={parent.size}, obj={parent_obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 500
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Fallback uniform: n={n_init}, obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting: n={n_parent}, obj={best_score:.8f}, budget={budget_s-25:.0f}s")

    # === Phase 1: Quick multi-config subgradient on parent ===
    print("\n=== Phase 1: Multi-config subgradient ===")
    t1 = time.monotonic()
    remaining = deadline - time.monotonic()
    alloc1 = min(remaining * 0.25, 200)

    a1, obj1 = multi_config_subgradient(best_seq, time_budget=alloc1)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        print(f"  Multi-config SG: {obj1:.8f} ** NEW BEST **")
    else:
        print(f"  Multi-config SG: {obj1:.8f}")

    # Peak pair reduction
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a1b, obj1b = peak_pair_reduction(best_seq, max_iters=400, time_budget=min(30, remaining * 0.15))
        if obj1b < best_score:
            best_score = obj1b
            best_seq = a1b.copy()
            print(f"  Peak-pair: {obj1b:.8f} ** NEW BEST **")
        else:
            print(f"  Peak-pair: {obj1b:.8f}")

    print(f"  Phase 1 done: {best_score:.8f} ({time.monotonic()-t1:.1f}s)")

    # === Phase 2: LP alternating ===
    print(f"\n=== Phase 2: LP alternating (n={len(best_seq)}) ===")
    t2 = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 60 and len(best_seq) <= 750:
        alloc2 = min(remaining * 0.25, 180)
        a2, obj2 = alternating_lp_sg(best_seq, max_rounds=12, time_budget=alloc2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  LP+SG: {obj2:.8f} ** NEW BEST **")
        else:
            print(f"  LP+SG: {obj2:.8f}")
    else:
        print("  Skipped (n too large or insufficient time)")

    print(f"  Phase 2 done: {best_score:.8f} ({time.monotonic()-t2:.1f}s)")

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length exploration ===")
    t3 = time.monotonic()

    n_candidates = sorted(set([
        n_parent - 150, n_parent - 100, n_parent - 50,
        n_parent, n_parent + 50, n_parent + 100, n_parent + 150,
        350, 400, 450, 500, 550, 600, 650, 700, 750, 800, 850, 900, 1000
    ]))
    n_candidates = [x for x in n_candidates if 250 <= x <= 1200]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == n_parent:
            a_init = best_seq.copy()
        else:
            a_init = resample_sequence(best_seq, n_try)
            a_init = normalize_simplex(a_init)

        # Quick optimization
        t_alloc = min(remaining * 0.08, 15.0)
        a_res, obj_res = soft_peak_subgradient(
            a_init, max_iters=int(t_alloc * 25), time_budget=t_alloc,
            tau0=0.005, tau_min=0.0005, step0=0.04, decay=0.9993)

        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  n={n_try}: {obj_res:.8f} ** NEW BEST **")
        else:
            print(f"  n={n_try}: {obj_res:.8f}")

    print(f"  Phase 3 done: {best_score:.8f} ({time.monotonic()-t3:.1f}s)")

    print(f"\n=== Phase 4: Simulated annealing ===")

                print(f"  Anneal+SG: {obj4b:.8f} ** NEW BEST **")

    # === Phase 5: Structural perturbation exploration ===
    print(f"\n=== Phase 5: Structural perturbations ===")

            elif r < 0.66:

            else:
                a_pert = best_seq * np.exp(rng.normal(0, 0.03 + 0.02 * rng.random(), len(best_seq)))
                a_pert = normalize_simplex(a_pert)

            t_alloc = min(remaining * 0.06, 12.0)
            a_res, obj_res = soft_peak_subgradient(

            if obj_res < best_score:

        # Multi-config subgradient

            best_score = obj6
            best_seq = a6.copy()
            print(f"  Final multi-config: {obj6:.8f} ** NEW BEST **")

        # Peak pair
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a6b, obj6b = peak_pair_reduction(best_seq, max_iters=300, time_budget=min(remaining * 0.2, 25))
            if obj6b < best_score:
                best_score = obj6b

        rng2 = np.random.default_rng(seed + 999)
        n_best = len(best_seq)
        for p in range(12):
            if time.monotonic() > deadline - 10:
                break
            scale = 0.004 + 0.002 * p
            a_pert = best_seq * np.exp(rng2.normal(0, scale, n_best))

            t_alloc = min(remaining * 0.05, 4.0)
