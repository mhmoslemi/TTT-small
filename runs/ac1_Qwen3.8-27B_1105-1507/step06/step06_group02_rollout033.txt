                k_end = min(2*n-2, i+n-1)  # k such that k-i <= n-1, i.e., k <= i+n-1
                # For k in [k_start, k_end], partner = k-i
                # But we need to exclude k = 2*i

                # All these k have valid partners

                b[ks] += linear_change
                # Quadratic correction at k = 2*i
                if k_start <= 2*i <= k_end:
                    # We already added 2*d*a[i] to b[2*i] (using the NEW a[i] = old_ai + d)
                    # But the correct change at k=2i is 2*old_ai*d + d^2
                    # What we added was 2*d*a_new_partner where a_new_partner = a[2i-i] = a[i] = old_ai + d
                    # So we added 2*d*(old_ai + d) = 2*old_ai*d + 2*d^2
                    # But the correct change is 2*old_ai*d + d^2
                    # So we need to subtract d^2
                    b[2*i] -= d * d
                improved = True
        
        M = b.max()
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a
