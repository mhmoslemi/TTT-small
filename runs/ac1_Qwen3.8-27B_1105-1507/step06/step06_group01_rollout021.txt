"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. FFT-accelerated FISTA on log-sum-exp smooth approximation with adaptive tau
2. LP best-response with cycling active sets (primal and dual)
3. Exact 1D coordinate descent via golden section search
4. Peak-aware perturbation for escaping local minima
5. Multi-start from parent and diverse initializations
Key improvements: FFT-based gradient (O(n log n)), dual LP formulation for speed,
cycling active sets, and adaptive time allocation between phases.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of conv(a,a) for normalized a (sum=1)."""
    b = fast_conv(a)
    return float(b.max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        val = max(float(v[0]), 0.0)
        return np.array([val])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad_fft(a, tau):
    """Compute smooth max of conv(a,a) and gradient using FFT. O(n log n)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    log_w = tau * (b - b_max)
    log_w = np.minimum(log_w, 500.0)
    exp_w = np.exp(log_w)
    p = exp_w / exp_w.sum()
    smooth_val = b_max + np.log(exp_w.sum()) / tau

    # Gradient via FFT: grad[i] = 2 * sum_{m=0}^{n-1} p[i+m] * a[m]
    # This is cross-correlation (p cross-corr a)[i] for i = 0,...,n-1
    size = 1
    while size < 3 * n:
        size <<= 1
    P = np.zeros(size)
    P[:2 * n - 1] = p
    A = np.zeros(size)
    A[:n] = a
    FFT_P = np.fft.rfft(P)
    FFT_A = np.fft.rfft(A)
    corr = np.fft.irfft(FFT_P * np.conj(FFT_A), size)
    grad = 2.0 * corr[:n]

    return smooth_val, grad


def run_fista(a_init, tau_start, tau_end, max_iters, deadline, lr=None):
    """FISTA with adaptive tau and FFT-based gradients."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    if lr is None:
        lr = 0.5

    best_a = a.copy()
    best_M = compute_M(a)

    t = 1.0
    y = a.copy()
    a_prev = a.copy()

    it = 0
    eval_count = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = it / max(1, max_iters)
        # Adaptive tau: start low for smooth landscape, increase for sharpness
        tau = tau_start * (tau_end / tau_start) ** frac

        _, grad_y = smooth_max_grad_fft(y, tau)
        grad_tan = grad_y - grad_y.mean()

        # Adaptive step size
        grad_norm = np.linalg.norm(grad_tan)
        if grad_norm < 1e-12:
            step = lr
        else:
            # Backtracking-inspired: scale by gradient magnitude
            step = lr / max(1.0, tau * grad_norm / (10.0 * n))
            step = min(step, lr * 2.0)

        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        it += 1

        if it % 3 == 0 or it == max_iters - 1:
            M_cur = compute_M(a)
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            if it % 10 == 0:
                M_y = compute_M(y)
                if M_y < best_M:
                    best_M = M_y
                    best_a = y.copy()

    M_final = compute_M(a)
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def build_lp_matrix(n, a, active_indices):
    """Build sparse constraint matrix for LP. A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response(a, M_target, active_indices, time_limit=10):
    """Solve LP best response. Returns normalized g or None."""
    n = len(a)
    A = build_lp_matrix(n, a, active_indices)
    if A.nnz == 0:
        return None

    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M_target)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_optimize(a_init, max_iters, deadline, base_active=20, cycle_len=8):
    """LP optimization with cycling active sets."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        # Cycling active sets: different strategies each iteration
        conv_len = 2 * n - 1
        strategy = it % cycle_len

        if strategy == 0:
            # Top base_active
            active = np.argsort(b)[::-1][:base_active].tolist()
        elif strategy == 1:
            # Top 2*base_active
            active = np.argsort(b)[::-1][:min(2 * base_active, conv_len)].tolist()
        elif strategy == 2:
            # Top 10
            active = np.argsort(b)[::-1][:10].tolist()
        elif strategy == 3:
            # Top base_active + bottom base_active/2 (boundary effects)
            top = np.argsort(b)[::-1][:base_active]
            bot = np.argsort(b)[:base_active // 2]
            active = list(set(top.tolist()) | set(bot.tolist()))
        elif strategy == 4:
            # Evenly spaced
            step = max(1, conv_len // base_active)
            active = list(range(0, conv_len, step))[:base_active]
        elif strategy == 5:
            # Top 50% of significant (above 0.5 * max)
            thresh = 0.5 * b.max()
            active = np.where(b >= thresh)[0][:base_active * 2].tolist()
            if len(active) < 5:
                active = np.argsort(b)[::-1][:base_active].tolist()
        elif strategy == 6:
            # All (if small enough)
            if conv_len <= 200:
                active = list(range(conv_len))
            else:
                active = np.argsort(b)[::-1][:base_active].tolist()
        else:
            # Random subset biased toward high values
            probs = b / b.sum()
            if probs.sum() > 0:
                active = np.random.choice(conv_len, size=min(base_active, conv_len),
                                          replace=False, p=probs).tolist()
            else:
                active = np.argsort(b)[::-1][:base_active].tolist()

        g = lp_best_response(a, best_M, active, time_limit=5)
        if g is None:
            no_improve += 2
            if no_improve > 20:
                break
            continue

        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = compute_M(a_trial)
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a


def coord_1d_opt(a, i, b, n):
    """Find optimal a[i] minimizing max convolution, given all others fixed."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        if x_test < 0:
            x_test = 0.0
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if quad_pos >= 0:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    # Find bracket
    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.1)

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(50):
        if fabs_check(fa, fb):
            pass
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def fabs_check(fa, fb):
    return abs(fa - fb) < 1e-15


def run_coord_descent(a_init, max_sweeps, deadline, seed=42):
    """Coordinate descent with golden section 1D optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = np.arange(n)
        rng.shuffle(order)

        b = fast_conv(a).copy()
        sweep_improved = False

        for i in order:
            x_opt = coord_1d_opt(a, i, b, n)
            x_old = a[i]
            if abs(x_opt - x_old) > 1e-12:
                # Update b incrementally
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - x_old * x_old)
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - x_old)
                a[i] = x_opt
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        # Recompute b from scratch to avoid accumulation errors
        M = compute_M(a)
        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 5
        if no_improve > 15:
            break

    return best_a


def run_mirror_descent(a_init, max_iters, deadline, step_init=0.2):
    """Mirror descent (multiplicative weights) on peak constraint."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))

        # Subgradient at active constraint
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        grad_tan = grad - grad.mean()
        step = step_init / (1.0 + 0.01 * it)

        # Multiplicative update
        log_w = -step * grad_tan
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new

        M_new = compute_M(a)
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

    return best_a


def peak_aware_perturb(a, rng, magnitude=0.05):
    """Perturb coordinates that contribute to near-peak convolution values."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()

    # Find peak location
    k_star = int(np.argmax(b))

    # Find all k where b[k] > 0.8 * M (near-peaks)
    near_peaks = np.where(b > 0.8 * M)[0]

    # For each near-peak, the contributing coordinates are i where k-i is in [0, n-1]
    # Perturb these coordinates
    a_pert = a.copy()
    for k in near_peaks[:10]:  # Limit to top 10 near-peaks
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        contrib = range(j_lo, j_hi + 1)
        for i in contrib:
            # Scale down if contributes to peak
            a_pert[i] *= (1.0 - magnitude * rng.uniform(0, 1))

    # Scale up some non-contributing coordinates
    all_coords = set(range(n))
    peak_coords = set()
    for k in near_peaks[:10]:
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        peak_coords.update(range(j_lo, j_hi + 1))
    non_peak = list(all_coords - peak_coords)
    if non_peak:
        for _ in range(min(len(non_peak), 20)):
            i = rng.choice(non_peak)
            a_pert[i] *= (1.0 + magnitude * rng.uniform(0.5, 1.5))

    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 0.01:
        return a.copy()
    return a_pert / s


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Parent
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed parent
        for sigma in [0.02, 0.05, 0.1]:
            p_pert = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(p_pert / p_pert.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())

    # Raised cosine (Hann)
    hann = 0.5 * (1 + np.cos(2 * np.pi * t))
    hann = np.maximum(hann, 0.01)
    inits.append(hann / hann.sum())

    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    # Flat top
    flat = np.ones(n) * 0.8
    edge = int(0.1 * n)
    flat[:edge] = np.linspace(0.3, 0.8, edge)
    flat[-edge:] = np.linspace(0.8, 0.3, edge)
    inits.append(flat / flat.sum())

    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimization: FISTA + LP + coordinate descent + perturbation.
    Returns list of non-negative floats minimizing the evaluation function.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768) / 768
        tracker.update(parent)
        print(f"No parent, uniform n=768, J={tracker.best_score:.8f}", flush=True)

    n = parent.size  # Use parent's length (768)

    print(f"\n{'='*60}", flush=True)
    print(f"Starting hybrid optimization: n={n}", flush=True)
    print(f"Deadline in {float(budget_s)-30:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    # ===================== PHASE 1: FISTA (25% time) =====================
    t_phase1_end = time.monotonic() + 0.25 * (float(budget_s) - 60)
    print(f"\n--- PHASE 1: FISTA (until t={t_phase1_end - time.monotonic():.0f}s) ---", flush=True)

    inits = generate_diverse_inits(n, rng, parent)
    best_phase1 = None
    best_M_phase1 = float('inf')

    n_inits = min(len(inits), 6)  # Limit to 6 inits
    time_per_init = max(10.0, (t_phase1_end - time.monotonic()) / n_inits)

    for ii in range(n_inits):
        if time.monotonic() > t_phase1_end:
            break
        a0 = inits[ii].copy()
        a0 = np.maximum(a0, 1e-8)
        a0 = a0 / a0.sum()

        sub_deadline = min(t_phase1_end, time.monotonic() + time_per_init)

        # Multi-tau FISTA
        best_local = a0.copy()
        best_M_local = compute_M(a0)

        for tau_s in [10, 30, 100]:
            if time.monotonic() > sub_deadline:
                break
            a_f = run_fista(best_local, tau_s, tau_s * 5, 2000,
                            min(sub_deadline, time.monotonic() + time_per_init / 3),
                            lr=0.3)
            M_f = compute_M(a_f)
            if M_f < best_M_local:
                best_M_local = M_f
                best_local = a_f.copy()

        if best_M_local < best_M_phase1:
            best_M_phase1 = best_M_local
            best_phase1 = best_local.copy()
            tracker.update(best_local)
            print(f"  Init {ii}: M={best_M_local:.10f}", flush=True)

    if best_phase1 is not None:
        print(f"  Phase 1 best: M={best_M_phase1:.10f}", flush=True)
    current_best = best_phase1 if best_phase1 is not None else parent.copy()
    current_best = np.maximum(current_best, 0.0)
    current_best = current_best / current_best.sum()

    # ===================== PHASE 2: LP Optimization (35% time) =====================
    t_phase2_end = time.monotonic() + 0.35 * (float(budget_s) - 60)
    print(f"\n--- PHASE 2: LP Optimization (until t={t_phase2_end - time.monotonic():.0f}s) ---", flush=True)

    # Multiple LP passes with different active set strategies
    a_lp = current_best.copy()
    M_lp = compute_M(a_lp)

    # Long LP run
    a_lp = run_lp_optimize(a_lp, 200, min(t_phase2_end - 60, time.monotonic() + 0.6 * (t_phase2_end - time.monotonic())),
                           base_active=25, cycle_len=8)
    M_lp = compute_M(a_lp)
    if M_lp < compute_M(current_best):
        current_best = a_lp.copy()
    tracker.update(current_best)
    print(f"  After LP: M={compute_M(current_best):.10f}", flush=True)

    # Mirror descent between LP passes
    if time.monotonic() < t_phase2_end - 10:
        a_md = run_mirror_descent(current_best, 3000, min(t_phase2_end - 5, time.monotonic() + 60), step_init=0.15)
        M_md = compute_M(a_md)
        if M_md < compute_M(current_best):
            current_best = a_md.copy()
        tracker.update(current_best)
        print(f"  After Mirror: M={compute_M(current_best):.10f}", flush=True)

    # Another LP pass
    if time.monotonic() < t_phase2_end - 10:
        a_lp2 = run_lp_optimize(current_best, 100, t_phase2_end, base_active=15, cycle_len=6)
        M_lp2 = compute_M(a_lp2)
        if M_lp2 < compute_M(current_best):
            current_best = a_lp2.copy()
        tracker.update(current_best)
        print(f"  After LP2: M={compute_M(current_best):.10f}", flush=True)

    # ===================== PHASE 3: Coordinate Descent (30% time) =====================
    t_phase3_end = time.monotonic() + 0.30 * (float(budget_s) - 60)
    t_phase3_end = min(t_phase3_end, deadline - 60)
    print(f"\n--- PHASE 3: Coordinate Descent (until t={t_phase3_end - time.monotonic():.0f}s) ---", flush=True)

    a_cd = run_coord_descent(current_best, 50, t_phase3_end, seed=seed + 1)
    M_cd = compute_M(a_cd)
    if M_cd < compute_M(current_best):
        current_best = a_cd.copy()
    tracker.update(current_best)
    print(f"  After CD: M={compute_M(current_best):.10f}", flush=True)

    # ===================== PHASE 4: Final Polish (remaining time) =====================
    t_phase4_end = deadline - 10
    print(f"\n--- PHASE 4: Final Polish (until t={t_phase4_end - time.monotonic():.0f}s) ---", flush=True)

    # 4a: High-tau FISTA
    if time.monotonic() < t_phase4_end - 30:
        a_f = run_fista(current_best, 200, 2000, 5000,
                        min(t_phase4_end - 30, time.monotonic() + 80), lr=0.1)
        M_f = compute_M(a_f)
        if M_f < compute_M(current_best):
            current_best = a_f.copy()
        tracker.update(current_best)

    # 4b: Short LP
    if time.monotonic() < t_phase4_end - 20:
        a_lp3 = run_lp_optimize(current_best, 50, min(t_phase4_end - 20, time.monotonic() + 60),
                                base_active=30, cycle_len=10)
        M_lp3 = compute_M(a_lp3)
        if M_lp3 < compute_M(current_best):
            current_best = a_lp3.copy()
        tracker.update(current_best)

    # 4c: Peak-aware perturbation + short CD
    if time.monotonic() < t_phase4_end - 15:
        for attempt in range(3):
            if time.monotonic() > t_phase4_end - 15:
                break
            a_pert = peak_aware_perturb(current_best, rng, magnitude=0.03)
            a_pert_opt = run_coord_descent(a_pert, 5, min(t_phase4_end - 10, time.monotonic() + 30),
                                           seed=seed + 100 + attempt)
            M_p = compute_M(a_pert_opt)
            if M_p < compute_M(current_best):
                current_best = a_pert_opt.copy()
            tracker.update(current_best)
            print(f"  Perturb attempt {attempt}: M={compute_M(current_best):.10f}", flush=True)

    # 4d: Final CD sweep
    if time.monotonic() < t_phase4_end - 5:
        a_cd_final = run_coord_descent(current_best, 10, t_phase4_end, seed=99999)
        M_final = compute_M(a_cd_final)
        if M_final < compute_M(current_best):
            current_best = a_cd_final.copy()
        tracker.update(current_best)

    # Final check
    if tracker.best is not None:
        final_score = float(evaluate_sequence(tracker.best.tolist()))
        if np.isfinite(final_score) and final_score < tracker.best_score:
            tracker.best_score = final_score

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={current_best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is None:
        tracker.best = current_best
        tracker.best_score = compute_J(tracker.best)

    return [float(x) for x in np.maximum(tracker.best, 0.0)]
