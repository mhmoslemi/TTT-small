def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full linearized LP (exact steepest descent direction in linearized problem)
2. Backtracking line search with the true (nonlinear) objective
3. Hinge-based smooth optimization as a complement
4. 2D coordinate descent for escaping 1D saddle points
5. Multi-restart with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def solve_full_linearized_lp(a, b):
    """
    Solve the full linearized LP:
    min t
    s.t. 2*(a*delta)[k] - t <= -b[k]  for k = 0, ..., 2n-2
         sum(delta) = 0
         delta[i] >= -a[i]
    
    Returns the optimal delta (direction) and t.
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # Build constraint matrix A_ub and b_ub
    # Row k: [2*a[k-i] for i=0..n-1 (where 0<=k-i<n), -1] <= -b[k]
    rows = []
    cols = []
    vals = []
    
    for k in range(conv_len):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(2.0 * a[j])
        # Coefficient of t is -1
        rows.append(k)
        cols.append(n)  # t is variable index n
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(conv_len, n + 1))
    b_ub = -b
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: delta[i] >= -a[i], no upper bound; t >= -inf (or 0)
    bounds = [(-a[i], None) for i in range(n)]
    bounds.append((0.0, None))  # t >= 0
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 30, 'presolve': True})
        if result.success:
            delta = result.x[:n]
            t_val = result.x[n]
            return delta, t_val
    except Exception:
        pass
    return None, None


def run_lp_descent(a_init, max_iters, deadline, line_search_min=0.01, line_search_max=1.0):
    """
    LP-based descent: at each step, solve the linearized LP to get a direction,
    then do a line search on the true (nonlinear) objective.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()

        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    
    n = len(a)

    best_M = fast_conv(a).max()
    no_improve = 0

        b = fast_conv(a)
        M_cur = b.max()

        delta, t_lp = solve_full_linearized_lp(a, b)
        if delta is None:

            k_star = int(np.argmax(b))
            delta = np.zeros(n)

                if 0 <= j < n:
                    delta[i] = -2.0 * a[j]
            delta = delta - delta.mean()  # project to sum=0
            t_lp = (M_cur + 1.0) / n  # rough estimate

        improved = False

        # If t_lp < M_cur, the direction should improve things

def run_hinge_descent(a_init, max_iters, deadline, eps_init=0.01):
    """
    Hinge-based smooth descent: minimize (1/eps)*sum_k max(0, b[k]-M0)^2
    where M0 is a target (slightly below current max).
    Uses Adam optimizer with simplex projection.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, adam_eps = 0.8, 0.999, 1e-8
    lr = 0.1
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M_cur = b.max()
        
        # Adaptive target: slightly below current max
        M_target = M_cur - 0.001 * M_cur  # try to reduce by 0.1%
        eps = max(0.001, eps_init / (1 + 0.01 * it))
        
        # Compute gradient of hinge loss
        # f = (1/eps) * sum_k max(0, b[k] - M_target + eps)^2 / 2
        # grad_a[i] = (1/eps) * sum_k max(0, b[k] - M_target + eps) * 2*a[k-i]
        
        violations = np.maximum(b - M_target + eps, 0.0)
        
        # grad[i] = (1/eps) * sum_k violations[k] * 2*a[k-i] (if 0<=k-i<n)
        # This is (2/eps) * (violations * a_reversed) convolved appropriately
        # Actually: grad[i] = (2/eps) * sum_k violations[k] * a[k-i]
        # = (2/eps) * (violations_flip * a)[i]... 
        
        # Let me compute it directly using the convolution structure
        # grad[i] = (2/eps) * sum_{k: 0<=k-i<n} violations[k] * a[k-i]
        # Let j = k-i, so k = i+j, and we need 0 <= i+j < 2n-1 and 0 <= j < n
        # grad[i] = (2/eps) * sum_{j=0}^{n-1} violations[i+j] * a[j] (where i+j < 2n-1)
        
        grad = np.zeros(n)
        for i in range(n):
            # sum_{j=0}^{min(n-1, 2n-2-i)} violations[i+j] * a[j]
            max_j = min(n, 2*n-1-i)
            if max_j > 0:
                grad[i] = (2.0/eps) * np.dot(violations[i:i+max_j], a[:max_j])
        
        # Project gradient to tangent of simplex (remove mean)
        grad_tan = grad - grad.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_tan
        v = beta2 * v + (1 - beta2) * grad_tan ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        a_new = proj_simplex(a_new)
        
        # Check improvement
        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

                if fa < fb:
                    hi, b_pt, fb = b_pt, a_pt, fa
                    a_pt = hi - gr*(hi-lo)
                    fa = obj_at(a_pt)
                else:
                    lo, a_pt, fa = a_pt, b_pt, fb
                    b_pt = lo + gr*(hi-lo)
                    fb = obj_at(b_pt)
            
            x_opt = (lo+hi)/2
            if obj_at(0) <= obj_at(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - a[i]) > 1e-12:
                # Update b
                for k in range(k_lo, k_hi+1):
                    j = k - i
                    if 0 <= j < n:
                        if k == 2*i:
                            b[k] = c_vals[k-k_lo] + x_opt*x_opt
                        else:
                            b[k] = c_vals[k-k_lo] + s_arr[k-k_lo]*x_opt
                a[i] = x_opt
                improved = True
        
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 15:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_freq_init(n, rng):
    """
    Generate initial sequence by designing the frequency response.
    Try to make |FFT(a)|^2 as flat as possible (minus DC).
    """
    # Start with a random phase in frequency domain
    N = 1
    while N < 2*n - 1:
        N <<= 1
    
    # Design |A|^2 to be flat (except DC)
    # |A[0]|^2 = (sum a)^2 = 1 (if normalized)
    # We want |A[k]|^2 ≈ (n-1)/N for k != 0 (roughly flat)
    # Then a = IFFT(A) where A[k] = sqrt(|A[k]|^2) * exp(i*phase[k])
    # We need a >= 0, which constrains the phases.
    
    # Simple approach: use random phases and check if a >= 0
    best_a = np.ones(n) / n
    best_obj = compute_J(best_a)
    
    for trial in range(50):
        # Random phases with conjugate symmetry
        phases = rng.uniform(0, 2*np.pi, N//2 + 1)
        phases[0] = 0  # DC is real and positive
        
        # Magnitude: flat except DC
        mags = np.ones(N//2 + 1) * np.sqrt(n / N)
        mags[0] = np.sqrt(n)  # DC bin: |A[0]| = sum(a) = 1... actually sqrt of sum of a^2
        
        # Actually, let's use Parseval: sum |A[k]|^2 = N * sum a[k]^2
        # If a is uniform: sum a[k]^2 = n * (1/n)^2 = 1/n
        # So sum |A[k]|^2 = N/n
        # |A[0]| = sum a = 1, so |A[0]|^2 = 1
        # Remaining: sum_{k!=0} |A[k]|^2 = N/n - 1
        # Per bin (N/2 - 1 non-DC positive freq bins + their conjugates):
        # (N-1) bins total non-DC, each gets (N/n - 1)/(N-1)
        
        target_energy = N / n - 1  # total non-DC energy
        per_bin = target_energy / (N - 1) if N > 1 else 0
        mags = np.full(N//2 + 1, np.sqrt(per_bin))
        mags[0] = 1.0  # |A[0]| = sum(a) = 1
        
        # Build A with conjugate symmetry
        A = np.zeros(N, dtype=np.complex128)
        A[0] = mags[0]
        for k in range(1, N//2 + 1):
            if k < N - k:  # avoid double counting
                A[k] = mags[k] * np.exp(1j * phases[k])
                A[N-k] = np.conj(A[k])
            else:
                A[k] = mags[k] * np.exp(1j * phases[k])
        
        a_candidate = np.fft.irfft(A, N)[:n]
        if np.all(a_candidate >= -1e-10):
            a_candidate = np.maximum(a_candidate, 0)
            if a_candidate.sum() > 0.01:
                a_candidate = a_candidate / a_candidate.sum()
                obj = compute_J(a_candidate)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_candidate.copy()
    
    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.02, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various parametric shapes
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        shape = t**alpha + (1-t)**alpha + 0.05
        inits.append(shape / shape.sum())
    
    # Cosine windows
    for freq in [1, 2, 3]:
        cos_w = 0.5 + 0.3 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    # Frequency-domain init
    freq_init = generate_freq_init(n, rng)
    inits.append(freq_init)
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_combined_optimization(a0, deadline):
    """Run multiple optimization methods and keep the best."""
    best = a0.copy()
    best_M = fast_conv(a0).max()
    
    n = len(a0)
    
    # Stage 1: LP-based descent (most promising for exact improvement)
    t1 = min(deadline, time.monotonic() + 300)
    if time.monotonic() < t1:
        print(f"    Running LP descent (n={n})...", flush=True)
        a_lp = run_lp_descent(best, 50, t1)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
        print(f"    After LP: M={best_M:.10f}", flush=True)
    
    # Stage 2: Hinge descent
    t2 = min(deadline, time.monotonic() + 200)
    if time.monotonic() < t2:
        print(f"    Running hinge descent...", flush=True)
        a_h = run_hinge_descent(best, 3000, t2, eps_init=0.005)
        M_h = fast_conv(a_h).max()
        if M_h < best_M:
            best_M = M_h
            best = a_h.copy()
        print(f"    After hinge: M={best_M:.10f}", flush=True)
    
    # Stage 3: Coordinate descent
    if time.monotonic() < deadline:
        max_sweeps = max(5, int((deadline - time.monotonic()) / max(1.0, n/100.0)))
        print(f"    Running coordinate descent (max {max_sweeps} sweeps)...", flush=True)
        a_cd = run_2d_coordinate_descent(best, max_sweeps, deadline)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
        print(f"    After CD: M={best_M:.10f}", flush=True)
    
    return best
