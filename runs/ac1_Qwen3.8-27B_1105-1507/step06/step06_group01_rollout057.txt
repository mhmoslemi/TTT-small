        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def solve_lp_fast(y, M, active, n):
    """Solve LP: max sum(g) s.t. conv(y,g)[k] <= M for k in active, g >= 0.
    Returns normalized g or None."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            m_idx = k - j
            yv = y[m_idx]
            if yv > 1e-14:
                rows.append(idx)
                cols.append(j)
                vals.append(yv)
    if len(rows) == 0:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 3, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-8,
                             'primal_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a_init, deadline, n_active_start=8, max_n_active=60,
                          step_init=0.3, momentum=True):
    """LP mirror descent with adaptive active set and optional momentum."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    n_active = n_active_start
    step = step_init
    no_improve = 0
    a_prev = a.copy()
    it = 0
    t_mom = 1.0

    while it < 500000 and time.monotonic() < deadline:
        # Momentum extrapolation
        if momentum and it > 0:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
            y = a + alpha * 0.5 * (a - a_prev)
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys < 1e-10:
                y = a.copy()
            else:
                y = y / ys
            t_mom = t_new
        else:
            y = a.copy()

        a_prev = a.copy()

        # Compute conv at y
        b_y = fast_conv(y)
        M_y = b_y.max()

        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()

        # Active set: top n_active
        top_k = min(n_active, 2 * n - 1)
        top_idx = np.argpartition(b_y, -top_k)[-top_k:]
        active = top_idx.tolist()

        # Solve LP
        g = solve_lp_fast(y, M_y, active, n)

        if g is None:
            n_active = min(int(n_active * 1.5) + 2, max_n_active)
            no_improve += 2
            if no_improve > 30:
                break
            it += 1
            continue

        # Try step sizes
        improved = False
        for t_s in [step, 0.5 * step, 0.25 * step, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t_s) * a + t_s * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial = a_trial / st
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
            if n_active > n_active_start:
                n_active = max(n_active_start, int(n_active * 0.85))
        else:
            no_improve += 1
            n_active = min(int(n_active * 1.4) + 1, max_n_active)
            if no_improve > 3:
                step *= 0.85
            if no_improve > 40:
                break

        it += 1

    return best_a


def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and gradient via log-sum-exp."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    p = exp_shifted / Z
    smooth_val = b_max + np.log(Z) / tau

    # Gradient: d/d a_i [sum_k p_k * conv(a,a)_k] = 2 * sum_k p_k * (conv(a, e_i)_k)
    # conv(a, e_i)_k = a[k - i] (if 0 <= k-i < n)
    # So grad[i] = 2 * sum_k p_k * a[k-i] = 2 * (p * a)[i] in the relevant range
    # More precisely: grad[i] = 2 * sum_{k} p_k * a[k-i] where 0<=k-i<n and 0<=k<2n-1
    # = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]  (where p[j]=0 for j>=2n-1)
    grad = np.zeros(n)
    conv_len = 2 * n - 1
    for i in range(n):
        # p[i : i+n] dotted with a
        end = min(i + n, conv_len)
        if i < conv_len:
            grad[i] = 2.0 * np.dot(p[i:end], a[:end - i])
    return smooth_val, grad


def run_fista_smooth(a_init, deadline, tau=500, lr=0.1):
    """FISTA on log-sum-exp smooth approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    t = 1.0
    y = a.copy()
    a_prev = a.copy()

    it = 0
    while it < 20000 and time.monotonic() < deadline:
        _, grad_y = smooth_max_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()

        step = lr / (1.0 + 0.01 * tau)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        it += 1

        if it % 10 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def coordinate_optimize(a, i, b, n):
    """Find optimal a[i] given all others fixed. Golden section on convex function."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(3.0 * x_cur, np.sqrt(max(b.max(), 0.0)) * 1.5, 0.1)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coordinate_descent(a_init, deadline, seed=0):
    """Coordinate descent with random ordering."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(200):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False

        for i in order:
            x_opt = coordinate_optimize(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - old * old)
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old)
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_improved:
            no_improve += 2
        if no_improve > 10:
            break

    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.02, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(p_n / p_n.sum())

    inits.append(np.ones(n) / n)

    # Concave parabola
    poly = (t * (1 - t)) + 0.1
    inits.append(poly / poly.sum())

    # Cosine windows
    for freq in [1, 2, 3, 5]:
        cos_w = 0.5 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())

    # Asymmetric: skewed
    skew = np.exp(-2.0 * (t - 0.35) ** 2 / 0.05) + 0.05
    inits.append(skew / skew.sum())

    # Modulated: base + high freq
    base = 0.5 + 0.3 * np.cos(2 * np.pi * t)
    mod = base + 0.15 * np.cos(2 * np.pi * 7 * t + 0.5)
    mod = np.maximum(mod, 0.01)
    inits.append(mod / mod.sum())

    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


def run_full_optimization(a0, n, deadline, seed=0):
    """Run all optimization phases on a single starting point."""
    a = a0.copy()
    best_a = a.copy()
    best_M = fast_conv(a).max()

    # Phase A: LP mirror descent (60% of time)
    t_lp = time.monotonic() + 0.6 * (deadline - time.monotonic())
    a_lp = run_lp_mirror_descent(a, t_lp, n_active_start=8, max_n_active=50,
                                 step_init=0.3, momentum=True)
    M_lp = fast_conv(a_lp).max()
    if M_lp < best_M:
        best_M = M_lp
        best_a = a_lp.copy()

    # Phase B: FISTA smooth (25% of time)
    t_fista = time.monotonic() + 0.25 * (deadline - time.monotonic())
    if time.monotonic() < t_fista:
        a_f = run_fista_smooth(best_a, t_fista, tau=300, lr=0.15)
        M_f = fast_conv(a_f).max()
        if M_f < best_M:
            best_M = M_f
            best_a = a_f.copy()

        a_f2 = run_fista_smooth(best_a, min(deadline - 0.05 * (deadline - time.monotonic()),
                                            time.monotonic() + 30), tau=800, lr=0.08)
        M_f2 = fast_conv(a_f2).max()
        if M_f2 < best_M:
            best_M = M_f2
            best_a = a_f2.copy()

    # Phase C: More LP (10% of time)
    t_lp2 = time.monotonic() + 0.1 * (deadline - time.monotonic())
    if time.monotonic() < t_lp2:
        a_lp2 = run_lp_mirror_descent(best_a, t_lp2, n_active_start=15, max_n_active=40,
                                      step_init=0.15, momentum=False)
        M_lp2 = fast_conv(a_lp2).max()
        if M_lp2 < best_M:
            best_M = M_lp2
            best_a = a_lp2.copy()

    # Phase D: Coordinate descent (5% of time)
    if time.monotonic() < deadline:
        a_cd = run_coordinate_descent(best_a, min(deadline, time.monotonic() + 15), seed=seed + 1)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best_a = a_cd.copy()

    return best_a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: diverse inits -> LP mirror descent -> FISTA -> CD.
    Tries multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size

    # Determine lengths to try
    lengths_to_try = [n_parent]
    for delta in [-128, -64, 64, 128, 256, -256]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl not in lengths_to_try:
            lengths_to_try.append(nl)
    lengths_to_try.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    lengths_to_try = lengths_to_try[:4]  # Max 4 lengths

    print(f"\nLengths to try: {lengths_to_try}", flush=True)

    # Allocate time
    reserve_time = 150.0  # Reserve for final polish
    avail_time = deadline - time.monotonic() - reserve_time
    time_per_length = max(60.0, avail_time / len(lengths_to_try))

    # Phase 1: Multi-length, multi-init search
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length search", flush=True)
    print(f"{'='*60}", flush=True)

    all_best = {}

    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - reserve_time:
            break
        t0 = time.monotonic()
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 30:
            break

        time_alloc = min(time_per_length, remaining / max(1, len(lengths_to_try) - li))
        print(f"\n  n={n}, time_alloc={time_alloc:.0f}s", flush=True)

        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        n_inits = min(len(inits), max(2, int(time_alloc / 60)))  # ~60s per init
        n_inits = min(n_inits, 8)

        best_for_n = None
        best_M_for_n = float('inf')

        for ii in range(n_inits):
            if time.monotonic() > deadline - reserve_time:
                break
            sub_deadline = min(t0 + time_alloc, deadline - reserve_time)
            if time.monotonic() >= sub_deadline:
                break

            a0 = inits[ii].copy()
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            a_opt = run_full_optimization(a0, n, sub_deadline, seed=seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()

            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()

            tracker.update(a_opt)

        if best_for_n is not None:
            all_best[n] = best_for_n
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best M={best_M_for_n:.10f} J={J_val:.6f} "
                  f"(elapsed={time.monotonic()-t0:.0f}s)", flush=True)

    tracker.report("After Phase 1")

    # Phase 2: Intensive LP on global best
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive LP polish on best", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        t2_deadline = min(deadline - 10, time.monotonic() + 120)
        a_lp = run_lp_mirror_descent(tracker.best, t2_deadline,
                                     n_active_start=20, max_n_active=80,
                                     step_init=0.2, momentum=False)
        tracker.update(a_lp)

    tracker.report("After Phase 2")

    # Phase 3: Fine-tuning with high-tau FISTA
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: High-tau FISTA fine-tuning", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 20:
        for tau in [500, 1000, 2000]:
            if time.monotonic() > deadline - 15:
                break
            t3 = min(deadline - 10, time.monotonic() + 30)
            a_f = run_fista_smooth(tracker.best, t3, tau=tau, lr=0.05)
            tracker.update(a_f)

    tracker.report("After Phase 3")

    # Phase 4: Final LP + CD
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final LP + CD polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        t4a = min(deadline - 5, time.monotonic() + 40)
        a_final_lp = run_lp_mirror_descent(tracker.best, t4a,
                                           n_active_start=10, max_n_active=40,
                                           step_init=0.1, momentum=False)
        tracker.update(a_final_lp)

        if time.monotonic() < deadline - 3:
            a_final_cd = run_coordinate_descent(tracker.best, deadline - 1, seed=9999)
            tracker.update(a_final_cd)

    tracker.report("After Phase 4")

    # Phase 5: Try one more length variation
    if tracker.best is not None and time.monotonic() < deadline - 15:
        n_cur = tracker.best.size
        for mult in [1.2, 0.8]:
            new_n = int(n_cur * mult)
            new_n = max(256, min(new_n, 2048))
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 5, time.monotonic() + 20)
            a_opt = run_full_optimization(a_ext, new_n, sub_dl, seed=seed + 5555)
            tracker.update(a_opt)
            break

    tracker.report("After Phase 5")

    # Final
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
