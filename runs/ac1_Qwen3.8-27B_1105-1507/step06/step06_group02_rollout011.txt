"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with adaptive tau for initialization
2. LP mirror descent with adaptive constraint selection, warm-starting,
   and backtracking line search

4. Multi-length exploration
Key improvements over previous: adaptive LP constraints (solve smaller LPs),
warm-started LP, line search for step size, and active set refinement phase.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, p, b, b_max


def run_smooth_descent(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                       lr_init=0.4, lr_end=0.003, nesterov=True):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        v = grad + beta * (y - a) if nesterov else grad
        a_new = np.maximum(y - lr * v, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        if nesterov:
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
            y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-10:
                y = a_new
            else:
                y /= sy
        else:
            y = a_new
        a = a_new
    return best_a


def build_A_sparse(n, a, active_indices=None):
    """Build sparse matrix for LP constraints, optionally only for active indices."""
    if active_indices is None:
        active_indices = range(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals, list(active_indices)


def lp_solve_active(a, M, active_k, warm_start=None):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for k in active_k, g >= 0."""
    n = len(a)
    rows, cols, vals, k_list = build_A_sparse(n, a, active_k)
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(k_list), n))
    c = -np.ones(n)
    b_ub = np.full(len(k_list), M)
    bounds = [(0, None)] * n
    try:
        options = {'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                   'primal_feasibility_tolerance': 1e-9}
        if warm_start is not None:
            options['time_limit'] = 10
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
        else:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_adaptive(a, deadline, max_iters=300, t_base=0.3, min_t=0.005,
                    active_frac=0.3, min_active=5):
    """LP mirror descent with adaptive constraint selection and line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_imp = 0
    hist_x = []
    hist_Fx = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        J = compute_J(a)
        if J < best_J * (1 - 1e-10):
            best_J = J
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        
        # Select active constraints
        n_active = max(min_active, int(active_frac * (2 * n - 1)))
        b_sorted = np.argsort(b)[::-1]
        active_k = b_sorted[:n_active]
        
        g = lp_solve_active(a, M, active_k)
        if g is None:
            # Try with more constraints
            n_active2 = max(2 * n_active, (2 * n - 1) // 2)
            active_k2 = b_sorted[:n_active2]
            g = lp_solve_active(a, M, active_k2)
            if g is None:
                break
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Line search for step size
        improved = False
        for t in [t_base, t_base * 0.7, t_base * 0.5, t_base * 0.3, 
                  t_base * 0.2, t_base * 0.1, t_base * 0.05, t_base * 0.02]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            J_t = compute_J(a_t)
            if J_t < best_J - 1e-12:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_imp += 2
            if no_imp > 40:
                break
            # Try smaller step
            t = min_t
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                break
            a_t /= s
            a = a_t
        
        # Track history for potential Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_t.copy())
        if len(hist_x) > 6:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        # Anderson acceleration if available
        if len(hist_x) >= 4:
            a_and = anderson_mix(hist_x, hist_Fx)
            if a_and is not None:
                sa = a_and.sum()
                if sa > 1e-10:
                    a_and /= sa
                    J_and = compute_J(a_and)
                    if J_and <= best_J * 1.01:
                        a = a_and
                        if J_and < best_J:
                            best_J = J_and
                            best_a = a.copy()
    
    return best_a


def anderson_mix(hist_x, hist_Fx):
    """Anderson mixing."""
    m = len(hist_x)
    residuals = np.array([hist_x[i] - hist_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        return None
    alpha = np.clip(alpha, -2.0, 2.0)
    if abs(alpha.sum()) < 1e-15:
        return None
    alpha = alpha / alpha.sum()
    mixed = np.zeros(len(hist_x[0]))
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    result = np.maximum(mixed, 0)
    if result.sum() < 1e-10:
        return None
    return result


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with Nesterov acceleration for initialization
2. LP mirror descent with adaptive constraint selection, warm-starting,
   and backtracking line search for the main optimization
3. Multi-length exploration with resampling
4. Active set refinement and final polish
Key improvements: persistent active constraint set that grows over time,
line search for optimal step size, Anderson acceleration, and more
aggressive LP phase allocation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, p, b, b_max


def run_smooth_descent(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                       lr_init=0.4, lr_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        v = grad + beta * (y - a)
        a_new = np.maximum(y - lr * v, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def build_and_solve_lp(n, a, M, k_indices):
    """Build and solve LP: max sum(g) s.t. (a*g)(k) <= M for k in k_indices, g >= 0."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(k_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9, 'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(hist_x, hist_Fx):
    """Anderson mixing for fixed-point acceleration."""
    m = len(hist_x)
    if m < 3:
        return None
    residuals = np.array([hist_x[i] - hist_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        return None
    alpha = np.clip(alpha, -2.0, 2.0)
    if abs(alpha.sum()) < 1e-15:
        return None
    alpha = alpha / alpha.sum()
    mixed = np.zeros(len(hist_x[0]))
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    result = np.maximum(mixed, 0)
    if result.sum() < 1e-10:
        return None
    return result


def run_lp_adaptive(a, deadline, max_iters=300, t_base=0.3, min_t=0.003,

        else:
            no_imp += 1
            if no_imp > 50:
                break
        
        # Dynamically select which constraints to include in LP

        b_sorted = np.argsort(b)[::-1]
        near_active = b_sorted[:target_n_active]
        for k in near_active:

        # Cap the persistent set size
        max_persistent = int(end_frac * (2 * n - 1))
        if len(persistent_set) > max_persistent:
            # Keep only the most active ones
            k_list = sorted(persistent_set)
            b_vals = b[k_list]
            k_sorted = sorted(k_list, key=lambda x: b[x], reverse=True)
            persistent_set = set(k_sorted[:max_persistent])

        g = build_and_solve_lp(n, a, M, k_indices)

            g = build_and_solve_lp(n, a, M, top_k.tolist())

        g /= gs

                t = min_t
            a_t = (1.0 - t) * a + t * g

            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            J_t = compute_J(a_t)
            if J_t < best_J - 1e-12:
                # Check if reduced LP solution satisfies all constraints
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                break

            no_imp += 1
            if no_imp > 50:

            # Take smallest step anyway to make progress

            a_t = np.maximum(a_t, 0.0)

            if s < 1e-10:

            a = a_t

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        for sigma in [0.005, 0.01, 0.02, 0.04]:
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.03 * rng.normal(0, 1, n)), 0.01))
    inits.append(np.ones(n))
    for alpha in [0.4, 0.7, 1.0, 1.3, 1.8, 2.5]:
        inits.append(np.maximum(0.15 + 0.85 * ((t**alpha + (1 - t)**alpha) / 2), 0.01))
    for f in [1, 2, 3, 5, 7]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-3 * (t - 0.5)**2) + 0.03)
    inits.append(np.exp(-6 * (t - 0.25)**2) + np.exp(-6 * (t - 0.75)**2) + 0.02)
    k = max(3, n // 8)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.03, 0.01))
    # Symmetric
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    return inits


def track_best(seq, score):
    """Returns (seq, score) if seq is better, else (None, score)."""
    return seq, score


def evaluate_and_track(a, tracker_score):
    """Evaluate sequence and return (new_score, improved) if valid."""
    a_clipped = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a_clipped.size == 0 or a_clipped.sum() < 0.01:
        return tracker_score, False
    s = float(evaluate_sequence(a_clipped.tolist()))
    if np.isfinite(s):
        return s, True
    return tracker_score, False


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = time.monotonic() + budget_s - 25.0
    
    best_seq = None
    best_score = float('inf')
    
    def update_best(a):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a.copy()
            el = time.monotonic() - t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)
    
    # Try to use parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass
    
    if parent is None:
        parent = np.ones(512)
        update_best(parent)
    
    n_parent = parent.size
    print(f"Starting optimization, parent n={n_parent}, score={best_score:.6f}", flush=True)
    
    # === PHASE 1: Smooth optimization (100s) ===
    p1_start = time.monotonic()
    p1_budget = min(100.0, (deadline - time.monotonic()) * 0.08)
    print(f"\nPhase 1: Smooth optimization (budget {p1_budget:.0f}s)", flush=True)
    
    n_main = n_parent
    inits = make_inits(n_main, rng, parent)
    best_n_smooth = None
    best_M_smooth = float('inf')
    
    p1_deadline = time.monotonic() + p1_budget
    n_inits = len(inits)
    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        a0 = np.maximum(a0.astype(np.float64), 1e-6)
        a0 /= a0.sum()
        remaining = p1_deadline - time.monotonic()
        if remaining < 10:
            break
        # Allocate time proportionally, more to earlier inits
        frac_time = 1.0 / (1.0 + 0.5 * idx)
        sub_dl = time.monotonic() + remaining * frac_time
        n_iters = max(100, int(4000 * (256.0 / n_main)))
        a = run_smooth_descent(a0, sub_dl, max_iters=n_iters,
                               tau_start=1.0, tau_end=150.0,
                               lr_init=0.3, lr_end=0.002)
        M = fast_conv(a).max()
        if M < best_M_smooth:
            best_M_smooth = M
            best_n_smooth = a.copy()
        update_best(a)
    
    if best_n_smooth is not None:
        update_best(best_n_smooth)
    print(f"  Phase 1 done ({time.monotonic()-p1_start:.1f}s), J={best_score:.6f}", flush=True)
    
    # === PHASE 2: Main LP optimization (75% of remaining time) ===
    p2_start = time.monotonic()
    remaining_after_p1 = deadline - time.monotonic()
    p2_budget = remaining_after_p1 * 0.70
    print(f"\nPhase 2: Adaptive LP (budget {p2_budget:.0f}s)", flush=True)
    
    a_start = best_seq.copy() if best_seq is not None else best_n_smooth
    p2_deadline = time.monotonic() + p2_budget
    
    # Pass 1: Moderate constraints, larger steps
    est_n = a_start.size
    max_iters_p2a = max(100, int(p2_budget * 0.45 / 2.0))  # est ~2s per LP iter
    a2a = run_lp_adaptive(a_start, min(time.monotonic() + p2_budget * 0.45, p2_deadline),
                          max_iters=max_iters_p2a, t_base=0.35, min_t=0.005,
                          start_frac=0.1, end_frac=0.5, min_active=10, use_anderson=True)
    update_best(a2a)
    print(f"  LP pass 1 done, J={best_score:.6f} ({time.monotonic()-p2_start:.1f}s)", flush=True)
    
    # Pass 2: Tighter constraints, smaller steps
    if time.monotonic() < p2_deadline - 10:
        rem = p2_deadline - time.monotonic()
        max_iters_p2b = max(50, int(rem * 0.5 / 2.5))
        a2b = run_lp_adaptive(a2a, time.monotonic() + rem * 0.5,
                              max_iters=max_iters_p2b, t_base=0.2, min_t=0.003,
                              start_frac=0.3, end_frac=0.7, min_active=20, use_anderson=True)
        update_best(a2b)
        print(f"  LP pass 2 done, J={best_score:.6f} ({time.monotonic()-p2_start:.1f}s)", flush=True)
    
    # Pass 3: Very tight constraints, finest steps
    if time.monotonic() < p2_deadline - 5:
        rem = p2_deadline - time.monotonic()
        max_iters_p2c = max(30, int(rem / 3.0))
        a2c = run_lp_adaptive(a2b if best_score < float('inf') else a2a,
                              p2_deadline,
                              max_iters=max_iters_p2c, t_base=0.1, min_t=0.001,
                              start_frac=0.5, end_frac=0.9, min_active=40, use_anderson=False)
        update_best(a2c)
        print(f"  LP pass 3 done, J={best_score:.6f} ({time.monotonic()-p2_start:.1f}s)", flush=True)
    
    print(f"  Phase 2 done ({time.monotonic()-p2_start:.1f}s)", flush=True)
    
    # === PHASE 3: Multi-length exploration (15% of remaining) ===
    p3_start = time.monotonic()
    remaining_after_p2 = deadline - time.monotonic()
    p3_budget = max(60.0, remaining_after_p2 * 0.25)
    if p3_budget < 40:
        p3_budget = 0
    print(f"\nPhase 3: Multi-length exploration (budget {p3_budget:.0f}s)", flush=True)
    
    if p3_budget > 30 and best_seq is not None:
        n_best = best_seq.size
        candidate_lengths = []
        for delta in [-128, -64, -32, 32, 64, 128, 192, 256]:
            n_try = n_best + delta
            if 256 <= n_try <= 2560:
                candidate_lengths.append(n_try)
        for n_fix in [640, 800, 1024, 1280, 1536, 1792, 2048]:
            if n_fix not in candidate_lengths and 256 <= n_fix <= 2560:
                candidate_lengths.append(n_fix)
        
        p3_deadline = time.monotonic() + p3_budget
        for n_try in sorted(candidate_lengths):
            if time.monotonic() > p3_deadline:
                break
            remaining = p3_deadline - time.monotonic()
            if remaining < 30:
                break
            a_cand = resample_seq(best_seq, n_try)
            a_cand = np.maximum(a_cand, 1e-6)
            a_cand /= a_cand.sum()
            
            # Quick smooth
            a_cand = run_smooth_descent(a_cand, time.monotonic() + min(20, remaining * 0.3),
                                        max_iters=200, tau_start=10, tau_end=100,
                                        lr_init=0.1, lr_end=0.005)
            update_best(a_cand)
            
            # Quick LP
            if time.monotonic() < p3_deadline - 15:
                a_lp = run_lp_adaptive(a_cand, time.monotonic() + min(30, p3_deadline - time.monotonic() - 5),
                                       max_iters=25, t_base=0.25, min_t=0.01,
                                       start_frac=0.15, end_frac=0.5, min_active=10,
                                       use_anderson=False)
                update_best(a_lp)
    
    print(f"  Phase 3 done ({time.monotonic()-p3_start:.1f}s), J={best_score:.6f}", flush=True)
    
    # === PHASE 4: Final polish (remaining time) ===
    p4_start = time.monotonic()
    remaining_final = deadline - time.monotonic()
    print(f"\nPhase 4: Final polish (budget {remaining_final:.0f}s)", flush=True)
    
    if best_seq is not None and remaining_final > 10:
        # High-tau smooth polish to sharpen
        a_polish = run_smooth_descent(best_seq, time.monotonic() + min(40, remaining_final * 0.3),
                                      max_iters=800, tau_start=50, tau_end=500,
                                      lr_init=0.03, lr_end=0.0005)
        update_best(a_polish)
        
        # Final LP with all constraints
        if time.monotonic() < deadline - 5:
            a_final = run_lp_adaptive(a_polish, deadline,
                                      max_iters=60, t_base=0.08, min_t=0.001,
                                      start_frac=0.6, end_frac=1.0, min_active=80,
                                      use_anderson=False)
            update_best(a_final)
    
    print(f"  Phase 4 done ({time.monotonic()-p4_start:.1f}s)", flush=True)
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = compute_J(best_seq)
    
    print(f"\n{'='*50}", flush=True)
    print(f"RESULT: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best_seq]
