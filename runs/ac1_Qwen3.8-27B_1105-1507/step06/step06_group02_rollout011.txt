from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix_pattern(n):
    """Build the sparse pattern for the convolution matrix.
    (A g)[k] = sum_i A[k,i] * g[i] = conv(a, g)[k]
    So A[k,i] = a[k-i] if 0 <= k-i < n
    The pattern (which entries are non-zero) depends on k and the bounds:
    i ranges from max(0, k-n+1) to min(k, n-1)
    """
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)

def make_conv_matrix(a, n, row_pattern, col_pattern):
    """Fill in the values of the convolution matrix given current a."""
    k_vals = row_pattern
    i_vals = col_pattern
    vals = a[k_vals - i_vals]
    return csr_matrix((vals, (k_vals, i_vals)), shape=(2 * n - 1, n))

def gs_flat(a, n_iters=200, beta=0.85):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_comp = mag ** 0.5
        mag_comp = mag_comp / mag_comp.max() * mag.max()
        F_new = np.exp(1j * phase) * mag_comp
        a_new = np.fft.irfft(F_new, size)[:n]

        if sc < 1e-15:

        if M_c < best_M:
            best_M = M_c
            best_a = a_new.copy()
        a = a_new
    return best_a

def smooth_grad_step(a, tau, n, size):
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(len(b)):
        if p[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, b

    a = a / a.sum()
    n = len(a)

        size <<= 1

    best_M = fast_conv(a).max()
    y = a.copy()
    momentum = 0.9
    for it in range(max_iters):

        frac = it / max_iters
        tau = 5.0 * (tau_end / 5.0) ** (frac ** 0.5)

        if b.max() < best_M * (1 - 1e-10):
            best_M = b.max()
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum ** 2)) / 2.0
        a_new = np.maximum(y - lr * (grad + momentum * (y - a)), 0.0)

        else:

        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)

        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
