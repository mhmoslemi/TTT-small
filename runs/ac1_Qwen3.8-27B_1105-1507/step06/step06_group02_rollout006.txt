import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Fast auto-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.sqrt(1.0) * np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective value J(a)."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_grad(a, tau):
    """
    Compute gradient of LSE approximation to max(conv(a,a)).
    Returns (grad, lse_value).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    shifted_max = shifted.max()
    exp_shifted = np.exp(shifted - shifted_max)
    w = exp_shifted / exp_shifted.sum()
    
    grad = np.zeros(n)
    # For each lag k: d b[k]/d a[i] = 2*a[k-i] for i in [lo,hi]
    # We process only lags with significant weight
    active = np.where(w > 1e-10)[0]
    for k in active:
        wk = w[k] * 2.0
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # a[k-hi], a[k-hi+1], ..., a[k-lo] corresponds to i=lo,...,hi
        start = k - hi
        grad[lo:hi+1] += wk * a[start:start + (hi - lo + 1)]
    
    lse_val = b_max + (shifted_max + np.log(exp_shifted.sum() + 1e-300)) / tau
    return grad, lse_val, b, b_max


def run_smooth_descent(a0, deadline, max_iters=10000, tau_start=5.0, tau_end=500.0,
                       lr_init=0.5, lr_end=0.01, backtrack=True):
    """
    Nesterov-accelerated gradient descent on LSE-smoothed max autocorrelation.
    Uses backtracking line search and adaptive tau.
    """
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    t_k = 1.0
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max_iters)
        # Adaptive tau: increase as we converge (peak becomes sharper)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        
        grad, lse_val, b, b_max = lse_grad(y, tau)
        
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Step size schedule with optional backtracking
        lr = lr_init * (lr_end / lr_init) ** frac
        
        # Nesterov lookahead
        momentum = (t_k - 1.0) / (t_k + 1.0)
        a_new = np.maximum(y - lr * (grad + momentum * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Backtracking: check if M decreased
        if backtrack and lr > lr_end * 10:
            M_new = fast_conv(a_new).max()
            if M_new > best_M * 1.05:
                # Try smaller step
                lr_half = lr * 0.5
                a_try = np.maximum(y - lr_half * (grad + momentum * (y - a)), 0.0)
                st = a_try.sum()
                if st > 1e-10:
                    a_try /= st
                    M_try = fast_conv(a_try).max()
                    if M_try < M_new:
                        a_new = a_try
                        lr = lr_half
        
        y_new = a_new + momentum * (a_new - a)
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y = y_new / sy
        a = a_new
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
    
    # Also track best during iterations via b_max
    # Final check
    M_final = fast_conv(best_a).max()
    if M_final < best_M:
        best_M = M_final
    
    return best_a


def run_smooth_descent_fista(a0, deadline, name=""):
    """FISTA (Fast Iterative Shrinkage-Thresholding) with adaptive L."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    t_k = 1.0
    L = 2.0 * n * a.max() ** 2 * 5.0  # Initial Lipschitz estimate
    tau_base = 20.0
    
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        
        tau = tau_base * (1.0 + 0.01 * it)
        grad, lse_val, b, b_max = lse_grad(y, tau)
        
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        # Backtracking for L
        step = 1.0 / (tau * L)
        a_new = np.maximum(y - step * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Check if step was too aggressive
        M_new = fast_conv(a_new).max()
        if M_new > best_M * 1.1:
            L *= 2.0
            step = 1.0 / (tau * L)
            a_new = np.maximum(y - step * grad, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
        
        if L < 10.0 * n * 0.01:
            L *= 0.95  # Slowly reduce L if steps are working
        
        momentum = (t_k - 1.0) / (t_k + 1.0)
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-10:
            y = y / sy
        a = a_new
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
    
    return best_a


def build_lp_matrix(n, a):
    """Build sparse convolution matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return np.array(rows), np.array(cols), np.array(vals)


def lp_best_response(a, M):
    """Solve the LP: max sum(g) s.t. (a*g)(k) <= M, g >= 0."""
    n = len(a)
    rows, cols, vals = build_lp_matrix(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'time_limit': 5.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_descent(a0, deadline, max_iters=500, t_init=0.4):
    """
    LP mirror descent with adaptive step size.
    At each step: compute best response g via LP, then blend a = (1-t)*a + t*g.
    Step size adapts based on improvement.
    """
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t = t_init
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Try multiple step sizes
        improved = False
        for t_try in [t, t * 0.7, t * 0.5, t * 0.3, t * 0.2, t * 0.1]:
            a_new = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            M_new = fast_conv(a_new).max()
            if M_new < best_M - 1e-14:
                a = a_new
                best_M = M_new
                best_a = a.copy()
                t = t_try  # Keep successful step size
                improved = True
                break
            elif M_new < fast_conv(a).max():
                # Still better than current a, even if not best
                a = a_new
                t = t_try * 0.5
                improved = True
                break
        
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            t *= 0.7
            if no_imp > 20:
                break
    
    return best_a


def coordinate_descent(a0, deadline, max_rounds=50):
    """
    Targeted coordinate descent: at each round, find the lag k with max autocorrelation,
    then reduce coordinates that contribute most to that peak.
    """
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_peak = int(np.argmax(b))
        M = b[k_peak]
        
        # Find positions that contribute most to b[k_peak]
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        contributions = np.abs(a[lo:hi+1] * a[k_peak - hi:k_peak - lo + 1][::-1])
        
        # Reduce the largest contributors (but keep non-negative)
        idx = np.argsort(contributions)[::-1]
        n_reduce = max(1, (hi - lo + 1) // 8)
        
        a_new = a.copy()
        for i in range(min(n_reduce, len(idx))):
            j = lo + idx[i]
            # Reduce this coordinate slightly
            a_new[j] *= 0.95
            # Compensate by increasing a coordinate far from the peak
            # Find the position farthest from the peak
            dist = np.abs(np.arange(n) - j)
            far = int(np.argmax(dist * a))  # Far and large
            a_new[far] *= 1.05
        
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            M_new = fast_conv(a_new).max()
            if M_new < best_M - 1e-14:
                a = a_new
                best_M = M_new
                best_a = a.copy()
            else:
                # Revert this round
                pass
    
    return best_a


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton for flat spectral initialization."""
    a = np.maximum(init.astype(np.float64), 0.01)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress dynamic range
        power = 0.4 + 0.3 * beta
        mag_new = (mag ** power) / ((mag ** power).max() + 1e-30)
        mag_new = mag_new * (mag.max() + mag.mean()) / 2
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0.001)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    
    return best_a


def resample_seq(src, n):
    """Resample sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initializations for length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(gs_flat_spectral(p, n_iters=100, beta=0.9))
    
    # Raised cosine
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    # Gaussian
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.05)
    # Double Gaussian
    inits.append(np.exp(-6 * (t - 0.3) ** 2) + np.exp(-6 * (t - 0.7) ** 2) + 0.03)
    # Power law edges
    for alpha in [0.3, 0.7, 1.5]:
        inits.append(np.maximum(t ** alpha * (1 - t) ** alpha + 0.05, 0.01))
    # Cosine window (Hann)
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * t)), 0.01))
    # Blackman-like
    inits.append(np.maximum(0.42 + 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t), 0.01))
    # Random smooth
    k = max(3, n // 30)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_flat_spectral(a_rand, n_iters=80, beta=0.85))
    
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            s = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)

                                   max_iters=40, t_init=0.2)
            best.update(a_fin)
        
        # Coordinate
        if time.monotonic() < deadline - 5:
            a_fin = coordinate_descent(a_fin, deadline, max_rounds=50)
            best.update(a_fin)
    
    best.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)
    
    # === PHASE 6: Try nearby lengths of best ===
    if best.seq is not None and time.monotonic() < deadline - 15:
        n_best = best.seq.size
        for delta in [-32, 32, -64, 64, -128, 128]:
            if time.monotonic() > deadline - 10:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 4096:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext = run_smooth_descent(a_ext, min(time.monotonic() + 15, deadline - 8),
                                       max_iters=300, tau_start=10, tau_end=80,
                                       lr_init=0.15, lr_end=0.01)
            best.update(a_ext)
            if time.monotonic() < deadline - 8:
                a_lp = run_lp_descent(a_ext, min(time.monotonic() + 20, deadline - 5),
                                      max_iters=20, t_init=0.25)
                best.update(a_lp)
    
    # Fallback
    if best.seq is None:
        best.seq = np.ones(1024)
        best.score = compute_J(best.seq)
    
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    
    return [float(x) for x in best.seq]
