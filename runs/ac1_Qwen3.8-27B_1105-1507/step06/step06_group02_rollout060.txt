import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum(exp(tau * b))) where b = a*a."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-13)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[lo:hi + 1]
        # grad contribution: 2 * p[k] * a[k - m] for m in [lo, hi]
        # a[k-m] for m in [lo,hi] is a[lo:hi+1] reversed
        grad[lo:hi + 1] += 2.0 * p[k] * seg[::-1]
    grad /= tau
    return grad, b, b_max, p


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=0.5, tau_end=500.0,
                        lr_init=0.3, lr_end=0.001, backtracking=True):
    """Nesterov-accelerated smooth optimization with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max, p = smooth_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        if backtracking:
            # Try to ensure improvement
            M_new = fast_conv(a_new).max()
            if M_new > b_max * 1.01:
                # Backtrack
                a_new = np.maximum(y - 0.5 * lr * (grad + beta * (y - a)), 0.0)

                else:

        sy = y.sum()
        if sy < 1e-10:

def run_lp_best_response(a, M, n):

        else:
            no_imp += 1
            if no_imp > 25:
                break

        # Get LP best response
        g = run_lp_best_response(a, M, n)
        if g is None:
            break
        g = g / g.sum()

        # Get linearized LP direction
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        n_rows = (2 * n - 1) + n
        rows = []
        cols = []
        vals = []
        b_ub = np.zeros(n_rows)
        for k in range(2 * n - 1):
            row_idx = len(b_ub)
            b_ub[row_idx] = -b[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                rows.append(row_idx)
                cols.append(m)
                vals.append(2.0 * a[k - m])
            rows.append(row_idx)
            cols.append(n)
            vals.append(-1.0)
        for m in range(n):
            row_idx = len(b_ub)
            b_ub[row_idx] = a[m]
            rows.append(row_idx)
            cols.append(m)
            vals.append(-1.0)
        A_ub_mat = csr_matrix((vals, (rows, cols)), shape=(n_rows, n + 1))
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(-a[m], 1.0) for m in range(n)] + [(None, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub_mat, b_ub=b_ub, A_eq=A_eq, b_eq=[0],
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 3})
            d = res.x[:n] if (res.success and res.x is not None) else np.zeros(n)
        except Exception:
            d = np.zeros(n)

        # Combine directions
        # Weight: more weight to LP best response early, more to linearized LP later
        w_br = 1.0 / (1.0 + 0.1 * it)
        w_lp = 1.0 - w_br
        direction = w_br * (g - a) + w_lp * d

        # Line search
        improved = False
        for alpha in [0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_try = np.maximum(a + alpha * direction, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            M_try = float(fast_conv(a_try).max())
            if M_try < best_M - 1e-14:
                a = a_try
                best_M = M_try
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_imp += 1
            if no_imp > 15:
                break

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.005))
    # Uniform
    inits.append(np.ones(n) / n)
    # Various shapes
    for alpha in [0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum((t**alpha + (1 - t)**alpha) / 2 + 0.01, 0.005))
    # Cosine bumps
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t) + 0.1, 0.01))
    # Gaussian
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.05)
    inits.append(np.exp(-6 * (t - 0.3)**2) + np.exp(-6 * (t - 0.7)**2) + 0.03)
    # Two-peak
    inits.append(np.exp(-20 * (t - 0.25)**2) + np.exp(-20 * (t - 0.75)**2) + 0.02)
    # Linear ramp (asymmetric)
    inits.append(np.maximum(0.1 + 0.9 * t, 0.01))
    inits.append(np.maximum(0.1 + 0.9 * (1 - t), 0.01))
    # Symmetric bump
    half = np.maximum(rng.uniform(0.3, 1.0, n // 2 + 1), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    # Random smooth
    k = max(3, n // 30)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + budget_s - 10.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    # === PHASE 1: Smooth optimization to refine parent ===
    p1_deadline = time.monotonic() + 200
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth Optimization (200s)", flush=True)
    print(f"{'='*50}", flush=True)

    n_parent = parent.size
    a_smooth = run_smooth_nesterov(parent, p1_deadline, max_iters=2000,
                                   tau_start=0.5, tau_end=300.0,
                                   lr_init=0.3, lr_end=0.002)
    best.update(a_smooth)

    # Try a few different initializations at the same length
    inits = make_initializations(n_parent, rng, parent)
    for idx, a0 in enumerate(inits[:5]):
        if time.monotonic() > p1_deadline:
            break
        a0 = np.maximum(a0, 1e-6)
        a0 /= a0.sum()
        a_try = run_smooth_nesterov(a0, min(time.monotonic() + 30, p1_deadline),
                                    max_iters=500, tau_start=1.0, tau_end=200.0,
                                    lr_init=0.25, lr_end=0.01)
        best.update(a_try)

    best.report("Phase 1")

    # === PHASE 2: Linearized LP refinement ===
    p2_deadline = time.monotonic() + 400
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: Linearized LP Refinement (400s)", flush=True)
    print(f"{'='*50}", flush=True)

    a_lp = run_linearized_lp(best.seq, p2_deadline, max_iters=100)
    best.update(a_lp)

    # Second pass with smaller steps
    if time.monotonic() < p2_deadline - 30:
        a_lp2 = run_linearized_lp(a_lp, min(time.monotonic() + 150, p2_deadline),
                                  max_iters=60,
                                  step_scales=[0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01])
        best.update(a_lp2)

    best.report("Phase 2")

    # === PHASE 3: Alternating method (LP best response + linearized LP) ===
    p3_deadline = time.monotonic() + 300
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Alternating LP (300s)", flush=True)
    print(f"{'='*50}", flush=True)

    a_alt = run_alternating(best.seq, p3_deadline, max_iters=100)
    best.update(a_alt)

    best.report("Phase 3")

    # === PHASE 4: Try different lengths ===
    p4_deadline = time.monotonic() + 300
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Multi-length Exploration (300s)", flush=True)
    print(f"{'='*50}", flush=True)

    n_best = best.seq.size
    candidate_ns = []
    for delta in [-64, -32, -16, 16, 32, 64, 128, 256]:
        n_try = n_best + delta
        if 256 <= n_try <= 2048:
            candidate_ns.append(n_try)
    # Also add some fixed lengths
    for n_fixed in [512, 768, 1024, 1280, 1536, 2048]:
        if n_fixed not in candidate_ns:
            candidate_ns.append(n_fixed)
    candidate_ns = sorted(set(candidate_ns))

    # Budget per length
    time_per = 25.0
    for n_try in candidate_ns:
        if time.monotonic() > p4_deadline - 20:
            break
        a_ext = resample_seq(best.seq, n_try)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext /= a_ext.sum()
        # Quick smooth
        dl_smooth = min(time.monotonic() + time_per * 0.6, p4_deadline - 30)
        a_smoothed = run_smooth_nesterov(a_ext, dl_smooth, max_iters=400,
                                         tau_start=2.0, tau_end=100.0,
                                         lr_init=0.2, lr_end=0.01)
        J_s = compute_J(a_smoothed)
        print(f"  n={n_try}: smooth J={J_s:.6f}", flush=True)
        best.update(a_smoothed)
        # Quick LP
        if time.monotonic() < p4_deadline - 20:
            dl_lp = min(time.monotonic() + time_per * 0.4, p4_deadline - 10)
            a_lped = run_linearized_lp(a_smoothed, dl_lp, max_iters=30)
            J_l = compute_J(a_lped)
            print(f"  n={n_try}: LP J={J_l:.6f}", flush=True)
            best.update(a_lped)

    best.report("Phase 4")

    # === PHASE 5: Final intensive LP on best ===
    p5_deadline = total_deadline
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Final Intensive Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < p5_deadline - 10:
        # Multiple passes of linearized LP with decreasing step sizes
        a_final = best.seq.copy()
        for pass_idx in range(4):
            if time.monotonic() > p5_deadline - 5:
                break
            remaining = p5_deadline - 5 - time.monotonic()
            scales = [0.3 / (pass_idx + 1)] + [0.1 / (pass_idx + 1)] + [0.03 / (pass_idx + 1)] + [0.01 / (pass_idx + 1)]
            a_final = run_linearized_lp(a_final, time.monotonic() + remaining / (4 - pass_idx),
                                        max_iters=50, step_scales=scales)
            best.update(a_final)

        # One more alternating pass
        if time.monotonic() < p5_deadline - 5:
            a_final2 = run_alternating(a_final, p5_deadline - 2, max_iters=30)
            best.update(a_final2)

    best.report("Phase 5")

    # === PHASE 6: Final smooth touch-up ===
    if best.seq is not None and time.monotonic() < total_deadline - 2:
        a_touch = run_smooth_nesterov(best.seq, total_deadline, max_iters=200,
                                      tau_start=50.0, tau_end=500.0,
                                      lr_init=0.05, lr_end=0.001)
        best.update(a_touch)

    best.report("FINAL")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"RESULT: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
