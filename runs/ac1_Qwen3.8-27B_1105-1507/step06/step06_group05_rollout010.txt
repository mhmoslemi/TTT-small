import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, objective=None, time_limit=20):
    """Solve LP: max c@g s.t. conv(a,g) <= M, g >= 0"""
    n = len(a)
    A = build_toeplitz(a)
    if objective is None:
        c = -np.ones(n)
    else:
        c = -objective
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def nesterov_subgradient(a, max_iters=200, time_budget=60.0):
    """Nesterov-accelerated subgradient method for min max(conv(a,a)) s.t. sum(a)=1, a>=0"""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    y = a.copy()
    t_k = 1.0
    step_base = 0.5 / np.sqrt(n)
    
    for k in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(y)
        cmax = float(np.max(conv))
        
        # Subgradient of max_k (a*a)[k] w.r.t. a
        # (a*a)[k] = sum_i a[i]*a[k-i]
        # grad_i = sum_k w_k * (a[k-i] + a[i+k-(n-1)]) ... 
        # Actually for linear conv: (a*a)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
        # d/d a[j] (a*a)[k] = a[k-j] (if 0<=k-j<n) + a[j+k-(k)] ... 
        # Wait, (a*a)[k] = sum_i a[i] * a[k-i] where 0<=i<n and 0<=k-i<n
        # d/d a[j] = a[k-j] (if 0<=k-j<n) + a[j] * (d/d a[j] of a[k-i] where k-i=j, i=k-j)
        # Actually it's just: d/d a[j] (a*a)[k] = a[k-j] if 0 <= k-j < n
        # (each a[j] appears exactly once in the sum for each k where k-j is valid)
        
        # For the max, subgradient is at the argmax position
        # But we can use a smoothed version with soft-max weights
        
        # Soft-max weighting
        tau = 0.01 * cmax / (n + 1)
        shifted = (conv - cmax) / tau
        weights = np.exp(np.clip(shifted, -500, 500))
        w_sum = float(np.sum(weights))
        weights /= w_sum
        
        # Gradient: grad[j] = sum_k weights[k] * a[k-j] (where valid)
        # This is a cross-correlation
        a_flip = a[::-1]
        size = 1
        while size < 2 * n:
            size <<= 1
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = corr[n-1:2*n-1]
        
        # Update with Nesterov acceleration
        t_new = (1 + np.sqrt(1 + 4 * t_k**2)) / 2
        beta = (t_k - 1) / t_new
        
        step = step_base / np.sqrt(k + 1)
        y_new = y + beta * (y - a)
        y_new = normalize(y_new - step * (grad - np.mean(grad)))
        a = y_new
        t_k = t_new
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a

def direct_peak_reduction(a, time_budget=30.0):
    """Directly target the convolution peak by reducing contributing pairs."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find all pairs contributing to this peak
        pairs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                pairs.append((a[j] * a[k2], j, k2))
        pairs.sort(reverse=True)
        
        improved = False
        for _, j1, j2 in pairs[:10]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                # Move mass from a[j1] to a random position far from peak
                for attempt in range(5):
                    target = int(rng.integers(0, n))
                    if target == j1:
                        continue
                    # Check if moving to target creates a worse peak
                    a_try = a.copy()
                    delta = 0.02 * a_try[j1]
                    a_try[j1] -= delta
                    a_try[target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
            else:
                # Transfer mass from the larger of the two
                if a[j1] > a[j2]:
                    src, dst_potential = j1, j2
                else:
                    src, dst_potential = j2, j1
                
                for attempt in range(5):
                    target = int(rng.integers(0, n))
                    if target == src or target == dst_potential:

                    a_try = a.copy()

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: non-symmetric optimization, homotopy from

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, objective=None, time_limit=20):
    """Solve LP: max c^T g s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    if objective is None:
        c = -np.ones(n)
    else:
        c = -objective
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section(f, lo, hi, max_iter=20):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 0.01:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def lp_alternate_adaptive(a, max_iters=40, time_budget=80.0):

def subgradient_nesterov(a, max_iters=300, time_budget=60.0):
    """Nesterov-accelerated soft subgradient descent."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    t_k = 1.0
    size = 1
    while size < 2 * n:
        size <<= 1

    for k in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv_y = compute_conv(y)
        cmax = float(np.max(conv_y))

        # Soft-max weights (adaptive temperature)
        tau = max(0.005, 0.02 * cmax / n) * (1.0 + k / float(max_iters))
        shifted = np.clip((conv_y - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv_y))
            weights = np.zeros(len(conv_y))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient via cross-correlation
        a_flip = y[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = corr[n - 1:2 * n - 1]

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        step = 1.0 / np.sqrt(k + 1) * 0.3

        y_momentum = y + beta * (y - a)
        a_new = normalize(y_momentum - step * (grad - np.mean(grad)))
        a = a_new
        t_k = t_new

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def peak_reduction(a, time_budget=30.0):

                # Find target that minimizes new max conv

def homotopy_optimize(a_start, time_budget=60.0):
    """
    Homotopy: interpolate from a simple shape to the current best,

        1.0 - x,                    # linear decay

        np.exp(-0.5 * ((x - 0.7) / 0.2) ** 2),  # off-center gaussian (right)

    for shape in shapes:

        if remaining < 15:
            break

        # Homotopy: blend from shape to current best
        best_blend_obj = best_obj
        best_blend_a = best_a.copy()

        for alpha in [0.3, 0.5, 0.7]:

        if best_blend_obj < best_obj:
            best_obj = best_blend_obj
            best_a = best_blend_a.copy()

def population_search(a_init, pop_size=5, time_budget=120.0):
    """Maintain a small population of candidates with elitism."""

    rng = np.random.default_rng(2024)
    start = time.monotonic()

            mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
            p = normalize(a_init * mod)

    best_obj = objs[best_idx]

    # Iterate
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)

            child = normalize(child)
        else:
            # LP-based mutation
            conv = compute_conv(population[sel])
            M = float(np.max(conv))
            g = solve_lp(population[sel], M, None, time_limit=max(3, per_individual // 3))
            if g is not None:
                t = rng.uniform(0.3, 0.8)
                child = normalize((1 - t) * population[sel] + t * g)
            else:
                child = normalize(population[sel] * np.exp(rng.normal(0, 0.08, n)))

        # Optimize child
        child = lp_alternate_adaptive(child, max_iters=6, time_budget=per_individual)
        child_obj = compute_obj(child)

        # Replace worst if improved
        worst_idx = int(np.argmax(objs))
        if child_obj < objs[worst_idx]:
            population[worst_idx] = child
            objs[worst_idx] = child_obj
            if child_obj < best_obj:
                best_obj = child_obj
                best_a = child.copy()
                best_idx = worst_idx

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phases:
      1. LP alternating with adaptive step at parent n
      2. Subgradient Nesterov + peak reduction
      3. Homotopy from analytical shapes
      4. Population-based search
      5. Systematic n exploration
      6. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")
    rng = np.random.default_rng(seed)

    # =========================================================
    # Phase 1: Core LP optimization at parent length
    # =========================================================
    print(f"\n--- Phase 1: Adaptive LP (n={n_parent}) ---")
    if best_seq is not None:
        a = normalize(best_seq.copy())
        for cycle in range(4):
            if time.monotonic() > deadline - 500:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.12, 70.0)
            a_new = lp_alternate_adaptive(a, max_iters=30, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  LP cycle {cycle}: {best_score:.8f} ***")
            else:
                print(f"  LP cycle {cycle}: {obj_new:.8f}")
            a = a_new

    # =========================================================
    # Phase 2: Nesterov subgradient + peak reduction
    # =========================================================
    print(f"\n--- Phase 2: Nesterov + peak reduction ---")
    if best_seq is not None:
        a = normalize(best_seq.copy())

        if time.monotonic() < deadline - 100:
            a_new = subgradient_nesterov(a, max_iters=300, time_budget=50.0)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Nesterov: {best_score:.8f} ***")
            else:
                print(f"  Nesterov: {obj_new:.8f}")
                a = normalize(0.7 * normalize(a) + 0.3 * normalize(a_new))

        if time.monotonic() < deadline - 60:
            a_new2 = peak_reduction(a, time_budget=35.0)
            obj_new2 = compute_obj(a_new2)
            if obj_new2 < best_score:
                best_score = obj_new2
                best_seq = a_new2.copy()
                print(f"  Peak reduction: {best_score:.8f} ***")
            else:
                print(f"  Peak reduction: {obj_new2:.8f}")

        # Follow up with LP
        if time.monotonic() < deadline - 50:
            a_new3 = lp_alternate_adaptive(normalize(best_seq.copy()), max_iters=20, time_budget=40.0)
            obj_new3 = compute_obj(a_new3)
            if obj_new3 < best_score:
                best_score = obj_new3
                best_seq = a_new3.copy()
                print(f"  Post-Phase2 LP: {best_score:.8f} ***")

    # =========================================================
    # Phase 3: Homotopy from analytical shapes
    # =========================================================
    print(f"\n--- Phase 3: Homotopy optimization ---")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a = normalize(best_seq.copy())
        a_hom = homotopy_optimize(a, time_budget=min(60.0, (deadline - time.monotonic()) * 0.2))
        obj_hom = compute_obj(a_hom)
        if obj_hom < best_score:
            best_score = obj_hom
            best_seq = a_hom.copy()
            print(f"  Homotopy: {best_score:.8f} ***")
        else:
            print(f"  Homotopy: {obj_hom:.8f}")
            a = normalize(0.8 * normalize(a) + 0.2 * normalize(a_hom))

        # Final LP after homotopy
        if time.monotonic() < deadline - 50:
            a_pol = lp_alternate_adaptive(normalize(best_seq.copy()), max_iters=15, time_budget=35.0)
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Post-homotopy LP: {best_score:.8f} ***")

    # =========================================================
    # Phase 4: Population-based search
    # =========================================================
    print(f"\n--- Phase 4: Population search ---")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a = normalize(best_seq.copy())
        a_pop = population_search(a, pop_size=5, time_budget=min(80.0, (deadline - time.monotonic()) * 0.25))
        obj_pop = compute_obj(a_pop)
        if obj_pop < best_score:
            best_score = obj_pop
            best_seq = a_pop.copy()
            print(f"  Population: {best_score:.8f} ***")
        else:
            print(f"  Population: {obj_pop:.8f}")

    # =========================================================
    # Phase 5: Systematic n exploration
    # =========================================================
    print(f"\n--- Phase 5: n exploration ---")
    n_best = len(best_seq) if best_seq is not None else 700

    n_values = sorted(set([
        400, 500, 550, 600, 650, 700, 750, 800, 850, 900, 950,
        1000, 1100, 1200, 1300, 1500, 1800
    ]))
    # Add neighbors of current best
    for delta in [-40, -20, 20, 40, 80, 150]:
        n_values.append(n_best + delta)
    n_values = sorted(set([n for n in n_values if 200 <= n <= 2500]))

    # Sort: nearest to parent first, but also include diverse lengths
    n_values.sort(key=lambda x: abs(x - n_parent) if x != n_best else 0)

    for n_try in n_values:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if n_try == n_best:
            continue

        if best_seq is not None:
            a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try

        alloc = min(remaining * 0.07, 20.0)
        if n_try <= 1000:
            a_try = lp_alternate_adaptive(a_try, max_iters=8, time_budget=alloc * 0.5)
        a_try = peak_reduction(a_try, time_budget=alloc * 0.4)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} ***")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 6: Final intensive polish
    # =========================================================
    print(f"\n--- Phase 6: Final polish ---")
    if best_seq is not None:
        a_final = normalize(best_seq.copy())

        for polish_round in range(5):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.25, 35.0)

            # LP
            if len(a_final) <= 1000 and alloc > 15:
                a_pol = lp_alternate_adaptive(a_final, max_iters=12, time_budget=alloc * 0.4)
            else:
                a_pol = a_final.copy()

            # Peak reduction
            a_pol = peak_reduction(a_pol, time_budget=alloc * 0.3)

            # Nesterov
            a_pol = subgradient_nesterov(a_pol, max_iters=150, time_budget=alloc * 0.25)

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish_round}: {best_score:.8f} ***")
            else:
                a_final = a_pol
                print(f"  Polish {polish_round}: {obj_pol:.8f}")

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate with actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'=' * 60}")

    return result
