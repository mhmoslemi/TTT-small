Improved multi-phase optimizer for minimum peak autocorrelation.
Uses momentum-augmented ALM, adaptive soft-gradient, and intensive coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]. Shape: (2n-1, n)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def alm_momentum(a, max_iters=100, time_budget=120.0, beta=0.3):
    """ALM with momentum and adaptive line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)
    start = time.monotonic()
    last_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Solve LP
        lp_time = max(2, min(8, remaining * 0.3))
        g = solve_lp_direction(a, M, time_limit=int(lp_time))
        if g is None:
            break
        
        # Momentum update: direction is (g - a) projected to sum=0
        direction = g - a
        direction = direction - np.mean(direction)  # center to maintain sum
        
        v = beta * v + (1.0 - beta) * direction
        
        # Line search: try multiple step sizes
        best_t = 0.0
        best_t_obj = best_obj
        t_candidates = [0.1, 0.25, 0.5, 0.75, 1.0]
        for t in t_candidates:
            a_try = normalize(a + t * v)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
