def anderson_acceleration(x_hist, f_hist, tol=1e-3):
    """
    Perform Anderson acceleration to mix history and predict next point.
    x_hist: list of arrays that were evaluated
    f_hist: list of corresponding function values
    tol: tolerance to determine early termination
    """
    m = len(f_hist)
    if m < 3:
        # Need at least 3 points for effective Anderson mixing
        return None
    
    # Build the matrix for mixed residuals
    M = np.zeros((m, m))
    for i in range(m):
        for j in range(m):
            # Compute mixed residual (adjusting for phases)
            if i == 0:
                continue  # Avoid redundant computation
                
            residual = x_hist[i] - x_hist[-1]
            M[i-1, j] = residual @ (x_hist[j+1] - x_hist[-1])
    
    if np.allclose(M, np.inf):
        return -np.ones(len(x_hist[0]))
        
    try:
        # Solve the QP problem for Anderson mix
        # Minimize 0.5*(z)^T (x_i - x_{last})^2 subject to sum alpha = 1
        # Using projected gradient for minimization
        def objective(z):
            # z is the candidate update vector
            if len(z) != m:
                return np.inf
                
            weighted_residuals = []
            for i in range(m):
                # Create residual for each mixture candidate
                residual_i = [ (xt - x) for xt, x in zip(x_hist[i], x_hist[-1]) ]
                # Only active residual if of significance
