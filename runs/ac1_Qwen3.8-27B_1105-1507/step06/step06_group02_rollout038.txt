"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Power-mean smooth objective with FFT-based O(n log n) gradient
2. Nesterov-accelerated projected gradient descent with backtracking line search
3. Simplex projection for exact constraint handling
4. LP-based mirror descent polish with Anderson acceleration
5. Multi-start with diverse initializations and length exploration
Key improvements over previous: power mean (better conditioned than softmax),
backtracking line search (adapts to landscape), and more aggressive multi-start.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Compute full auto-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def power_mean_grad(a, p):
    """
    Compute gradient of M_p(a) = (sum_k C_k^p)^(1/p) where C = conv(a,a).
    Returns (gradient, M_p_value, C).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    C = fast_conv(a)
    C_p = C ** p
    M_p = C_p.sum() ** (1.0 / p)
    
    if p == 1:
        # M_1 = sum(C) = (sum a)^2, gradient is just 2*sum(a) for all components
        g = 2.0 * np.full(n, C.sum())  # This isn't right for p=1
        # Actually for p=1: M_1 = sum_k C_k = (sum a)^2
        # dM_1/da[j] = 2*sum(a) for all j
        # But that's the gradient of the sum, not useful for minimizing max.
        # For p=1, let's just use p slightly above 1.
        g = 2.0 * np.ones(n) * a.sum()
        return g, M_p, C
    
    C_pw = C ** (p - 1)
    
    # Cross-correlation: g_raw[j] = sum_{k'=0}^{n-1} a[k'] * C_pw[j+k']
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    Fa = np.fft.fft(a, L)
    Fc = np.fft.fft(C_pw, L)
    g_raw = np.real(np.fft.ifft(Fa * np.conj(Fc)))[:n]
    
    g = 2.0 * (M_p ** (1 - p)) * g_raw
    return g, M_p, C


def project_simplex(v, S):
    """Project v onto the simplex {a >= 0, sum(a) = S}."""
    n = len(v)
    # Binary search for lambda
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu - (cv - S) / rho > 0
    if not np.any(cond):
        # All zero, distribute evenly
        return np.full(n, S / n)
    lam_idx = np.max(np.where(cond)[0])
    theta = (cv[lam_idx] - S) / (lam_idx + 1)
    return np.maximum(v - theta, 0)


def nesterov_power_mean(a, deadline, p_start=2.0, p_end=50.0, 
                         max_iters=10000, eta_init=0.5):
    """
    Nesterov-accelerated projected gradient descent on the power mean.
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S  # normalize to sum 1 for optimization, rescale at end
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    t_k = 1.0
    beta = 0.9
    eta = eta_init
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        p = p_start * (p_end / p_start) ** (frac ** 0.5)
        
        g, M_p, C = power_mean_grad(y, p)
        M_exact = C.max()
        
        if M_exact < best_M * (1 - 1e-12):
            best_M = M_exact
            best_a = a.copy()
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        
        # Backtracking line search
        for _ in range(20):
            a_new = y - eta * g
            a_new = project_simplex(a_new, 1.0)
            # Check Armijo condition (approximate: just check M_p decreased)
            g_new, M_p_new, _ = power_mean_grad(a_new, p)
            if M_p_new < M_p * (1 - 1e-12) or eta < 1e-10:
                break
            eta *= 0.5
        else:
            # If line search fails, just take a tiny step
            a_new = project_simplex(y - 1e-10 * g, 1.0)
        
        # Prevent eta from growing too large
        eta = min(eta * 1.05, 2.0)  # slight growth for efficiency
        
        a = a_new
        y = a + momentum * (a - y)  # Nesterov extrapolation
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-10:
            y /= sy
        else:
            y = a.copy()
        
        t_k = t_new
    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix for conv(a, g) as A @ g."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'mip_rel_gap': 1e-6})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception as e:
        print(f"  LP error: {e}", flush=True)
    return None


def anderson_mix(history_x, history_Fx, m):
    """Anderson mixing: find alpha minimizing ||sum alpha_i (x_i - Fx_i)||^2 s.t. sum alpha = 1."""
    if m < 2:
        return history_Fx[-1].copy()
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        alpha = np.linalg.solve(G, ones)
        alpha = alpha / alpha.sum()
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 2.0)
    alpha = alpha / alpha.sum()
    mixed = sum(alpha[i] * history_Fx[i] for i in range(m))
    return np.maximum(mixed, 0)


def run_lp_anderson(a, deadline, max_iters=100, window=5, t_base=0.3, min_t=0.01):
    """LP mirror descent with Anderson acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    hist_x = []
    hist_Fx = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        t = max(min_t, t_base * (0.95 ** it))
        a_new = (1.0 - t) * a + t * g
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
            M_and = fast_conv(a_and).max()
            M_new = fast_conv(a_new).max()
            if M_and <= M_new * 1.02:
                a = a_and
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def run_lp_plain(a, deadline, max_iters=50):
    """Plain LP descent with backtracking step sizes."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g /= g.sum()
        
        improved = False
        for t in [0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 10:
                break
    
    return best_a


def gs_init(n, rng, n_iters=200, beta=0.85):
    """Gerchberg-Saxton initialization to flatten spectrum."""
    a = np.maximum(rng.uniform(0.3, 1.0, n), 0.01)
    a /= a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for _ in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress magnitude
        mag_norm = mag / (mag.max() + 1e-30)
        mag_target = mag_norm ** 0.5  # square root compression
        mag_target = beta * mag_target + (1 - beta) * (1.0 / np.sqrt(n))
        F_new = np.exp(1j * phase) * mag_target
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new
    return best_a


def resample(src, n):
    """Resample sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.01))
    
    inits.append(np.ones(n) / n)
    # GS initialized
    inits.append(gs_init(n, rng, n_iters=150))
    # Various parametric forms
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t**alpha + (1-t)**alpha) / 2, 0.01))
    for f in [1, 2, 3, 4, 6]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t + rng.uniform(0, 2*np.pi)), 0.01))
    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        inits.append(np.exp(-8*((t-center)/0.2)**2) + 0.02)
    # Smooth random
    k = max(3, n // 16)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same'), 0.01))
    # Symmetric random
    half = np.maximum(rng.uniform(0.3, 1.0, n // 2 + 1), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            s = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)
    
    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
