"""

4. LP mirror descent with Anderson acceleration for precise polish

Key innovation: The game-theoretic QP phase exploits the Hankel structure of the

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def hankel_matvec(p, a):
    """Compute B(p) @ a where B(p)_{ij} = p[i+j], using FFT cross-correlation. O(n log n)."""
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fp * fa.conj(), size)
    return result[:n]


def gs_flat_spectral(init, n_iters=400, beta=0.85):
    """Gerchberg-Saxton: iteratively flatten the power spectrum while maintaining non-negativity."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_max = mag.max() + 1e-30
        mag_mean = mag.mean() + 1e-30
        # Soft compression: push magnitudes toward the mean
        mag_new = (1.0 - beta) * mag + beta * mag_mean
        # Power-law compression
        power = 0.4 + 0.4 * beta
        mag_p = (mag + 1e-30) ** power
        mag_alt = mag_p / (mag_p.max() + 1e-30) * mag_max
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def smooth_max_grad(a, tau):
    """Compute smooth approximation of the max gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, p, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.4, lr_end=0.003):
    """Nesterov accelerated smooth descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        beta_nes = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) * (it - 1)))
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        a_new = np.maximum(y - lr * (grad + (beta_nes - 1.0) * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new
        y = a
    return best_a


def solve_qp_hankel(p_dual, a_init, S, max_iters=5000, deadline=None):
    """Solve min a^T B(p) a s.t. a >= 0, sum(a) = S using Nesterov projected gradient.
    B(p)_{ij} = p[i+j] (Hankel matrix). Gradient computed via FFT in O(n log n).
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n * S
    else:
        a = a * S / s
    best_a = a.copy()
    best_val = a @ hankel_matvec(p_dual, a)
    # Estimate Lipschitz constant
    g0 = hankel_matvec(p_dual, a)
    lip_est = 2.0 * np.linalg.norm(g0, np.inf) / (np.linalg.norm(a, np.inf) + 1e-15)
    if lip_est < 1e-10:
        lip_est = 10.0
    lr = 1.0 / (lip_est + 1e-10)
    y = a.copy()
    v = np.zeros(n)
    for it in range(1, max_iters + 1):
        if deadline is not None and time.monotonic() > deadline:
            break
        alpha = 2.0 / (it + 3.0)  # step size for Nesterov
        lr_it = min(500.0, it) * alpha * lr / 10.0
        lr_it = max(lr_it, lr * 0.01)
        grad = 2.0 * hankel_matvec(p_dual, y)
        y_new = y - lr_it * grad
        y_new = np.maximum(y_new, 0.0)
        sn = y_new.sum()
        if sn < 1e-15:
            y_new = y.copy()
            sn = s
        else:
            y_new = y_new * S / sn
        a_new = (1.0 - alpha) * y + alpha * y_new
        a_new = np.maximum(a_new, 0.0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new * S / sa
        val = a_new @ hankel_matvec(p_dual, a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        y = y_new
        a = a_new
        # Reduce step size if objective increasing
        if val > best_val * 1.1:
            lr *= 0.9
    return best_a, best_val


def run_game_theoretic(a_init, deadline, max_outer=200, max_inner=2000):
    """Game-theoretic approach: alternate between dual (softmax) and primal (QP) updates."""
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    S = 1.0
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t_temp = 5.0  # initial temperature for softmax
    no_improve = 0
    for outer_it in range(max_outer):
        if time.monotonic() > deadline:
            break
        # Compute convolution
        b = fast_conv(a)
        b_max = float(b.max())
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
        # Dual update: softmax over convolution values
        # Temperature anneals: start high (spread), end low (focused)
        frac = outer_it / max(1, max_outer)
        t_cur = t_temp * (1.0 - frac) + 200.0 * frac
        exp_b = np.exp(-t_cur * (b_max - b) / (b_max + 1e-30))
        p_dual = exp_b / exp_b.sum()
        # Primal update: solve QP
        a_qp, qp_val = solve_qp_hankel(p_dual, a, S, max_iters=max_inner,
                                        deadline=deadline - 5.0)
        a = a_qp
        # Check if game is close to equilibrium
        b_new = fast_conv(a)
        b_new_max = b_new.max()
        gap = b_new_max - b_new[b_new < b_new_max * 0.99].mean() if (b_new < b_new_max * 0.99).any() else 0
        if gap < 1e-8:
            break
    return best_a


def build_A_rows(n, a):
    """Build sparse matrix for LP: (a*g)(k) = sum_j a[k-j]*g[j]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    n = len(a)
    rows, cols, vals = build_A_rows(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                             'primal_feasibility_tolerance': 1e-10,
                             'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(hist_x, hist_Fx, m):
    """Anderson acceleration mixing."""
    residuals = np.array([hist_x[i] - hist_Fx[i] for i in range(m)])
    G = residuals @ residuals.T + 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 2.0)
    alpha_sum = alpha.sum()
    if abs(alpha_sum) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / alpha_sum
    mixed = np.zeros(len(hist_x[0]))
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    return np.maximum(mixed, 0.0)

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        allot = min(15.0, (p1_end - 10) - time.monotonic())
        if allot < 3:
            continue
        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')
        for a0 in inits:
            if time.monotonic() > p1_end - 5:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            a = run_nesterov_smooth(a0, min(time.monotonic() + allot * 0.7, p1_end - 5),
                                    max_iters=3000, tau_start=1.0, tau_end=200.0,
                                    lr_init=0.35, lr_end=0.003)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()
        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")

    # === PHASE 2: Game-theoretic QP refinement (500s) ===
    p2_end = time.monotonic() + 500.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Game-Theoretic QP Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    # Focus on the best length and a couple of neighbors
    if best.seq is not None:
        n_best = best.seq.size
        # Try the best and neighbors
        game_lengths = sorted(set([n_best] + [n_best - 64, n_best + 64, n_best - 32, n_best + 32]))
        game_lengths = [n for n in game_lengths if 128 <= n <= 2560]
        for n_g in game_lengths:
            if time.monotonic() > p2_end - 30:
                break
            if n_g in best_per_length:
                a_start = best_per_length[n_g]
            elif best.seq is not None:
                a_start = resample_seq(best.seq, n_g)
            else:
                continue
            a_start = np.maximum(a_start, 1e-8)
            a_start /= a_start.sum()
            # Run game-theoretic refinement
            allot = min(100.0, (p2_end - 20) - time.monotonic())
            if allot < 10:
                continue
            t0 = time.monotonic()
            a_game = run_game_theoretic(a_start, time.monotonic() + allot,
                                        max_outer=100, max_inner=3000)
            M_g = fast_conv(a_game).max()
            J_g = compute_J(a_game)
            print(f"  Game n={n_g}: J={J_g:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(a_game)
            if n_g in best_per_length:
                if J_g < compute_J(best_per_length[n_g]):
                    best_per_length[n_g] = a_game.copy()
            else:
                best_per_length[n_g] = a_game.copy()

    best.report("Phase 2")

    # === PHASE 3: LP polish with tight tolerances (450s) ===
    p3_end = time.monotonic() + 450.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Polish (Anderson + Backtracking)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_lp = best.seq.size
        est_lp_time = 2.0 * (n_lp / 128.0)**2  # seconds per LP iteration
        print(f"  n={n_lp}, est LP time={est_lp_time:.1f}s/iter", flush=True)
        # First: Anderson-accelerated LP
        allot1 = min(200.0, (p3_end - 100) - time.monotonic())
        max_iters = max(30, int(allot1 / max(0.05, est_lp_time)))
        t0 = time.monotonic()
        a_lp1 = run_lp_anderson(best.seq, time.monotonic() + allot1,
                                max_iters=max_iters, anderson_window=5,
                                t_base=0.35, min_t=0.003)
        print(f"  LP Anderson: J={compute_J(a_lp1):.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
        best.update(a_lp1)

        # Second: Backtracking LP for final precision
        if time.monotonic() < p3_end - 30:
            allot2 = min(150.0, p3_end - 30 - time.monotonic())
            t0 = time.monotonic()
            a_lp2 = run_lp_backtrack(a_lp1, time.monotonic() + allot2, max_iters=100)
            print(f"  LP Backtrack: J={compute_J(a_lp2):.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(a_lp2)

        # Third: One more Anderson pass with smaller step
        if time.monotonic() < p3_end - 10:
            t0 = time.monotonic()
            a_lp3 = run_lp_anderson(best.seq, time.monotonic() + min(100.0, p3_end - 10 - time.monotonic()),
                                    max_iters=50, anderson_window=6,
                                    t_base=0.15, min_t=0.002)
            print(f"  LP Fine: J={compute_J(a_lp3):.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(a_lp3)

    best.report("Phase 3")

    # === PHASE 4: Game + LP hybrid on best (150s) ===
    p4_end = time.monotonic() + 150.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Game+LP Hybrid", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < p4_end - 20:
        # Alternate game and LP for a few rounds
        a_cur = best.seq.copy()
        for round_idx in range(4):
            if time.monotonic() > p4_end - 10:
                break
            # Game step
            allot_g = min(30.0, (p4_end - 20) - time.monotonic())
            if allot_g > 5:
                a_cur = run_game_theoretic(a_cur, time.monotonic() + allot_g,
                                           max_outer=30, max_inner=2000)
                best.update(a_cur)
            # LP step
            if time.monotonic() < p4_end - 10:
                a_cur = run_lp_backtrack(a_cur, p4_end - 5, max_iters=20)
                best.update(a_cur)

    best.report("Phase 4")

    # === PHASE 5: Final length exploration (80s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        for delta in [-48, 48, -96, 96, -16, 16]:
            if time.monotonic() > deadline - 5:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2560:
                continue
            a_try = resample_seq(best.seq, n_try)
            a_try = np.maximum(a_try, 1e-6)
            a_try /= a_try.sum()
            # Quick smooth
            a_try = run_nesterov_smooth(a_try, min(time.monotonic() + 10, deadline - 10),
                                        max_iters=1000, tau_start=5, tau_end=100,
                                        lr_init=0.2, lr_end=0.01)
            best.update(a_try)
            # Quick LP if time
            if time.monotonic() < deadline - 15:
                a_try = run_lp_backtrack(a_try, deadline - 10, max_iters=15)
                best.update(a_try)

    best.report("Phase 5")

    # === FIX: apply clamping and final check ===
    if best.seq is not None:
        best.seq = np.clip(best.seq, 0.0, 1000.0)
        final_score = evaluate_sequence(best.seq.tolist())
        if np.isfinite(final_score) and final_score < best.score:
            best.score = final_score
        print(f"\nFINAL: J={best.score:.8f} n={best.seq.size} total_time={time.monotonic()-best.t0:.0f}s", flush=True)
    else:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"{'='*60}", flush=True)
    return [float(x) for x in best.seq]
