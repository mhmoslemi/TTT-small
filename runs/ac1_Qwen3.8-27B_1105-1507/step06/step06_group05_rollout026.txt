def get_fft_size(n):

    if n in _fft_size_cache:

def compute_conv(a):

    conv = irfft(fa * fa, size)

def compute_obj(a):

            if t < 1e-8:
                break
            a = normalize(a - t * grad_centered)
        
        # Cool down
        tau_init *= 0.98
        momentum = min(0.99, momentum * 1.001)  # slowly increase momentum

def non_palindromic_spectral(a, max_iters=50, time_budget=40.0):
    """
    Spectral equalization that explicitly breaks palindromic symmetry.
    Key insight: non-palindromic sequences have lower convolution peaks.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    # Add a non-palindromic bias
    bias = np.linspace(0, 0.1, n)  # slight linear bias to break symmetry
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Compress spectrum
        alpha = 0.3 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        # Non-negativity + non-palindromic bias + normalize
        a_new = np.maximum(a_new, 0.0)
        # Add small non-palindromic perturbation
        a_new = a_new + 0.01 * bias
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.6 * a + 0.4 * a_new
            a = normalize(a)

    return best_a

def simulated_annealing(a, time_budget=60.0, T_init=0.1, T_final=0.001):
    """
    Simulated annealing with targeted moves that reduce convolution peaks.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)
    
    num_iter = 0
    T = T_init
    total_iter = max(100, int(time_budget * 10))  # assume ~10 iters/sec

    while time.monotonic() - start < time_budget:
        num_iter += 1
        T = T_init * (T_final / T_init) ** (num_iter / total_iter)
        
        # Random move: transfer mass between two positions
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        
        # Transfer amount
        transfer = rng.uniform(0.01, 0.05) * max(a[i], a[j])
        
        a_try = a.copy()
        a_try[i] = max(0.0, a_try[i] - transfer)
        a_try[j] = a_try[j] + transfer
        a_try = normalize(a_try)
        
        obj_try = compute_obj(a_try)
        delta = obj_try - compute_obj(a)
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a

def resample_sequence(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

"""
Optimized step function for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous:
1. Nesterov momentum with adaptive backtracking
2. Direct peak-minimization LP (for n <= threshold)
3. Multi-scale peak targeting (not just the global max, but top-k peaks)
4. Log-parameterization for better non-negativity handling
5. Targeted n exploration around optimal
6. Periodic re-initialization to escape basins
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def lp_minimize_peak(a, time_limit=15):
    """
    Solve LP: min M s.t. conv(a,g) <= M*e, sum(g) = 1, g >= 0, M >= 0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    A = build_toeplitz(a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Inequality: A@g - M <= 0 (2n-1 constraints)
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit,
                              'dual_feasibility_tolerance': 1e-12,
                              'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                return g, float(res.x[-1])
    except Exception:
        pass
    return None, None

def nesterov_multi_peak(a, tau_init=0.003, step_size=0.03, max_steps=200, 
                         time_budget=60.0, momentum=0.9, top_k=3):
    """
    Nesterov gradient descent targeting top-k peaks of the convolution.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 3 * n - 1:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        mean_conv = float(np.mean(conv))
        
        # Multi-peak targeting: weight peaks above a threshold
        threshold = mean_conv + (top_k - 1) * (cmax - mean_conv) / (2 * top_k)
        # Actually, use softmax with adaptive temperature
        tau = max(tau_init + 0.2 * (cmax - mean_conv) / (n * (0.5 + step / max_steps)), 1e-10)
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: d/max(conv) / d(a[i]) ≈ 2 * sum_j weights[j+i] * a[j] (correlation)
        # More precisely: grad[i] = 2 * sum_k weights[k] * d(conv[k])/d(a[i])
        # d(conv[k])/d(a[i]) = a[k-i] if 0<=k-i<n, else 0
        # So grad[i] = 2 * sum_k weights[k] * a[k-i] * [0<=k-i<n]
        # = 2 * (weights * a_reversed) correlated with a
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        
        # Nesterov momentum
        v = momentum * v + grad
        v_centered = v - np.mean(v)
        
        # Backtracking line search
        t = step_size
        improved = False
        for _ in range(8):
            a_new = normalize(a - t * v_centered)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            t *= 0.5
            if t < 1e-10:
                break
        
        if not improved:
            # Stuck: perturb
            a = normalize(a + 0.01 * np.random.default_rng(step + 1).normal(0, 0.02, n))
            v *= 0.5
        
        # Adaptive momentum
        if step % 20 == 0 and step > 0:
            momentum = min(0.99, momentum * 1.005)
        tau_init *= 0.985

    return best_a
