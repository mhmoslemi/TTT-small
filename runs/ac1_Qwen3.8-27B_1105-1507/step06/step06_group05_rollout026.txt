"""
Optimized step function search using adaptive soft-max gradient descent with
FFT-based gradient computation, spectral repair, and multi-length exploration.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_soft_gradient(a, w, N):
    """
    Compute gradient g[i] = 2 * (a cross w)[i] using FFT.
    a has length n, w has length 2n-1, N is the FFT size (>= 3n-2).
    Returns array of length n.
    """
    A = fft(a, N)
    W = fft(w, N)
    cross = ifft(A * np.conj(W)).real
    return 2.0 * cross[:len(a)]


def soft_max_gradient_optimize(a, time_budget=60.0, tau_init=0.01, tau_min=0.0001):
    """
    Adaptive soft-max gradient descent with backtracking line search.
    Minimizes the peak of conv(a,a) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # FFT size for gradient computation
    N = 1
    while N < 3 * n - 1:
        N <<= 1

    # Adaptive temperature
    tau = tau_init
    step_size = 0.1
    stall_count = 0
    patience = 50

    for step in range(50000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        cmean = float(np.mean(conv))

        # Adaptive temperature based on peakedness
        peakedness = cmax / cmean if cmean > 1e-30 else 1.0
        tau = max(tau_min, 0.5 / (n * np.log1p(peakedness)))

        # Compute soft-max weights
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500.0, 500.0)
        w = np.exp(shifted)
        w_sum = float(np.sum(w))
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Compute gradient via FFT
        g = compute_soft_gradient(a, w, N)

        # Center gradient (remove mean for sum preservation)
        g_centered = g - np.mean(g)

        # Backtracking line search
        accepted = False
        for shrink in range(10):
            eta = step_size * (0.5 ** shrink)
            a_try = a - eta * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                accepted = True
                step_size = min(step_size * 1.2, 1.0)
                stall_count = 0
                break

        if not accepted:
            stall_count += 1
            step_size *= 0.7
            if step_size < 1e-10:
                # Reset with perturbation
                rng = np.random.default_rng(step)
                a = normalize(best_a * np.exp(rng.normal(0, 0.05, n)))
                step_size = 0.05
                stall_count = 0

            if stall_count > patience:
                # Spectral repair: modify frequency content
                a = spectral_repair(a)
                a = normalize(a)
                obj_repair = compute_obj(a)
                if obj_repair < best_obj:
                    best_obj = obj_repair
                    best_a = a.copy()
                step_size = 0.05
                stall_count = 0
                patience = int(patience * 0.7)
                if patience < 10:
                    patience = 10

    return best_a


def spectral_repair(a):
    """
    Spectral repair: flatten the power spectrum to reduce autocorrelation peaks.
    Uses iterative amplitude modification in frequency domain.
    """
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1

    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)

    # Compute current power spectrum
    P = mag ** 2
    P_max = float(np.max(P))
    P_mean = float(np.mean(P))

    if P_max < 1e-30:
        return a

    # Compress the power spectrum: reduce peaks, boost valleys
    # Use a power-law compression
    alpha = 0.6  # compression strength
    P_norm = P / P_max
    P_compressed = P_norm ** alpha * P_max

    # Reconstruct with modified magnitude
    new_mag = np.sqrt(P_compressed)
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, N))[:n]

    # Project to non-negative
    a_new = np.maximum(a_new, 0.0)

    # Blend with original to maintain progress
    a_out = 0.6 * a + 0.4 * a_new
    return a_out


def spectral_repair_aggressive(a, n_iters=20):
    """
    More aggressive spectral repair with multiple iterations.
    """
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1

    A = fft(a, N)

    for it in range(n_iters):
        mag = np.abs(A)
        phase = np.angle(A)
        P = mag ** 2
        P_max = float(np.max(P))
        if P_max < 1e-30:
            break

        # Progressive compression
        alpha = 0.5 + 0.3 * (it / n_iters)
        P_norm = P / P_max
        P_compressed = P_norm ** alpha * P_max

        new_mag = np.sqrt(P_compressed)
        A = new_mag * np.exp(1j * phase)

        a_tmp = np.real(ifft(A, N))[:n]
        a_tmp = np.maximum(a_tmp, 0.0)
        s = float(np.sum(a_tmp))
        if s < 1e-10:
            break
        a = a_tmp / s
        A = fft(a, N)

    return a


def coordinate_descent(a, time_budget=20.0):
    """
    Focused coordinate descent: transfer mass from high-autocorrelation
    contributors to low-autocorrelation positions.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Compute gradient at the peak (subgradient)
        # g[i] = 2*a[k_star - i] for valid i
        g = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = 2.0 * a[j]

        g_centered = g - np.mean(g)

        # Try multiple step sizes
        improved = False
        for eta in [0.02, 0.01, 0.005, 0.001]:
            a_try = a - eta * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Try moving mass from high-gradient to low-gradient positions
            idx_high = int(np.argmax(g))
            idx_low = int(np.argmin(g))
            if idx_high != idx_low:
                delta = 0.01 * a[idx_high]
                a_try = a.copy()
                a_try[idx_high] -= delta
                a_try[idx_low] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True

        if not improved:
            # Random perturbation
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a


def lp_refine(a, time_budget=60.0):
    """
    LP-based refinement: solve max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Then blend the solution with current a.
    """
    from scipy.optimize import linprog

    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(30):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Build constraint matrix: A_ub @ g <= M
        # (conv(a, g))[k] = sum_j a[j] * g[k-j] (with bounds)
        # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
        m = 2 * n - 1
        A_ub = np.zeros((m, n))
        for j in range(n):
            for k in range(j, j + n):
                if k < m:
                    A_ub[k, j] = a[k - j]

        c = -np.ones(n)
        b_ub = np.full(m, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(2, time_budget - (time.monotonic() - start))})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g

            # Golden section search for blend parameter
            lo, hi = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = compute_obj(normalize((1 - c_pt) * a + c_pt * g))
            fd = compute_obj(normalize((1 - d_pt) * a + d_pt * g))

            for _ in range(12):
                if hi - lo < 0.02:
                    break
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = compute_obj(normalize((1 - c_pt) * a + c_pt * g))
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = compute_obj(normalize((1 - d_pt) * a + d_pt * g))

            t_opt = (lo + hi) / 2.0
            a_new = normalize((1 - t_opt) * a + t_opt * g)
            obj_new = compute_obj(a_new)

            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Perturb to escape
                rng = np.random.default_rng(it * 31 + 7)
                a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))

        except Exception:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

            a_try = spectral_repair_aggressive(a_try, n_iters=10)
            a_try = normalize(a_try)
            a_try = soft_max_gradient_optimize(a_try, time_budget=alloc * 0.2,
                                               tau_init=0.003, tau_min=0.0002)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart} ({strat}): {best_score:.8f} **")
            else:
                print(f"  Restart {restart} ({strat}): {obj_try:.8f}")

    # =========================================================
    # Phase 6: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        n_pol = len(a_final)

        for polish in range(6):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.3, 30.0)

            # Gradient
            a_pol = soft_max_gradient_optimize(a_final, time_budget=alloc * 0.4,
                                               tau_init=0.002, tau_min=0.0001)
            # Coordinate descent
            if alloc > 15:
                a_pol = coordinate_descent(a_pol, time_budget=min(10, alloc * 0.25))
            # Spectral
            a_pol = spectral_repair(a_pol)
            a_pol = normalize(a_pol)
            a_pol = soft_max_gradient_optimize(a_pol, time_budget=alloc * 0.2,
                                               tau_init=0.001, tau_min=0.0001)

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish}: {best_score:.8f} **")
            else:
                a_final = a_pol
                print(f"  Polish {polish}: {obj_pol:.8f}")

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 200

    # Validate with the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
