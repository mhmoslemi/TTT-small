```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-constraint LP best-response (exact for the game)
2. Optimal 1D mixing via ternary search on quadratic max
3. Smooth log-sum-exp minimax with Nesterov acceleration
4. Coordinate descent with exact 1D golden-section
5. Multiplicative weights (game-theoretic mirror descent)
6. Multi-start, multi-length exploration with diverse inits
Key novelty: full LP (all 2n-1 constraints) for exact best response,
and analytical optimal mixing ratio via ternary search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    sz = _next_pow2(size)
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:size]


def _fast_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    sz = _next_pow2(size)
    fa = np.fft.rfft(a, sz)
    fb = np.fft.rfft(b, sz)
    return np.fft.irfft(fa * fb, sz)[:size]


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    b = _fast_conv(a)
    return 2.0 * len(a) * b.max()


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not (~cond).any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def _lp_full(a):
    """Solve full LP best response: min t s.t. (a*q)[k]<=t, sum(q)=1, q>=0.
    Returns (q, t) or (None, None).
    """
    n = len(a)
    nk = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(nk):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    A = csr_matrix((vals, (rows, cols)), shape=(nk, n + 1))
    c = np.zeros(n + 1)
    c[n] = 1.0
    b_ub = np.zeros(nk)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            q = np.maximum(r.x[:n], 0.0)
            qs = q.sum()
            if qs > 1e-10:
                q = q / qs
            else:
                q = np.ones(n) / n
            return q, float(r.x[n])
    except Exception:
        pass
    return None, None

    A_eq = np.zeros((1, n + 1))

def _optimal_alpha(baa, baq, bbq):
    """Find alpha in [0,1] minimizing max_k [(1-a)^2*baa[k] + 2a(1-a)*baq[k] + a^2*bbq[k]].
    Uses ternary search.
    """
    def f(alpha):
        vals = (1.0 - alpha) ** 2 * baa + 2.0 * alpha * (1.0 - alpha) * baq + alpha ** 2 * bbq
        return vals.max()

    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = f(m1)
        f2 = f(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = _fast_conv(a).max()
    tol = 1e-12

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        if use_full:
            q, t_lp = _lp_full(a)
        else:
            q, t_lp = _lp_active(a, n_active)

        if q is None:
            if use_full:
                q, t_lp = _lp_active(a, n_active)
            if q is None:
                break

        baa = _fast_conv(a)
        baq = _fast_cross(a, q)
        bbq = _fast_conv(q)

        alpha = _optimal_alpha(baa, baq, bbq)

            break

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        M_new = _fast_conv(a_new).max()

        if M_new < best_M * (1 - 1e-14):
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new
            if M_new >= best_M * (1 + 1e-10):
                alpha /= 2.0
                if alpha < 1e-6:
                    break
                a_new = (1.0 - alpha) * best_a + alpha * q
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
                a = a_new
                M_new = _fast_conv(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()

        gap = best_M - t_lp / (best_M if best_M > 0 else 1.0)
        if it % 50 == 0:
            print(f"    LP iter {it}: M={best_M:.10f} t_lp={t_lp:.10f} alpha={alpha:.4f}", flush=True)

        if t_lp > 0 and abs(best_M - t_lp) / best_M < 1e-6:
            break

    return best_a


def _smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    b = _fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(max(exp_b.sum(), 1e-300)) / tau
    grad = np.zeros(n)
    for i in range(n):
        j_lo = max(0, i - (n - 1))
        j_hi = min(n - 1, (2 * n - 2) - i)
        if j_hi >= j_lo:
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            grad[i] = 2.0 * np.dot(p[i:i + n], a)
    return smooth_val, grad


def _fista_smooth(a, tau, max_iters, deadline, lr=0.3):
    """FISTA on smooth max with simplex projection."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = _fast_conv(a).max()

    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        _, grad_y = _smooth_max_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()

        step = lr / (1.0 + 0.01 * tau * it * 0.001)
        a_new = y - step * grad_tan
        a_new = _proj_simplex(a_new)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha_m = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = a_new + alpha_m * (a_new - a_prev)
        y_new = _proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new

        if it % 10 == 0:
            M_a = _fast_conv(a).max()
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()
            M_y = _fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()

    M_final = _fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def _multiplicative_weights(a, eta, max_iters, deadline):
    """Game-theoretic multiplicative weights on the dual variable."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    nk = 2 * n - 1

    q = np.ones(nk) / nk
    best_a = a.copy()
    best_M = _fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = _fast_conv(a)
        M = b.max()

        # Update q (dual variable) via multiplicative weights
        # q_new[k] ∝ q[k] * exp(eta * b[k])
        log_q = np.log(np.maximum(q, 1e-300)) + eta * b
        log_q -= log_q.max()
        q = np.exp(log_q)
        q = q / q.sum()

        # Update a: minimize sum_k q[k] * (a*a)[k] s.t. a>=0, sum(a)=1
        # Gradient: 2 * sum_k q[k] * a[k-i] for coordinate i
        np.fill(a, 0.0) if False else None
        grad = np.zeros(n)
        for i in range(n):
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            j_lo = 0
            j_hi = n - 1
            # grad[i] = 2 * sum_j a[j] * q[i+j]... wait
            # (a*a)[k] = sum_i a[i]a[k-i], d/d a[i] = sum_k q[k] * (a[k-i] + a[k-i] if i != k-i)
            # = 2 * sum_{k: 0<=k-i<n} q[k] * a[k-i] - q[2i]*a[i] if 2i < nk (diagonal correction)
            # Actually: d/d a[i] (a*a)[k] = a[k-i] + (a[i] if k-2i==0 else 0)... 
            # More carefully: (a*a)[k] = sum_j a[j]a[k-j], d/d a[i] = a[k-i] + a[i]*delta_{k,2i}... 
            # No: d/d a[i] sum_j a[j]a[k-j] = a[k-i] (from j=i term) + (if k-i != i, then also from j=k-i term: a[i])
            # = a[k-i] + a[k-i] if i != k-i, else a[k-i] = 2*a[k-i] if i != k-i, a[k-i] if i == k-i
            # So d/d a[i] sum_k q[k](a*a)[k] = sum_k q[k] * (2*a[k-i] - a[k-i]*I(k==2i))
            # = 2*sum_k q[k]*a[k-i] - q[2i]*a[i] (if 2i < nk)

        # Simpler: just use the subgradient at the current max
        k_star = int(np.argmax(b))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
            if 2 * i == k_star:
                grad[i] = a[i]

        grad_tan = grad - grad.mean()
        step = 0.5 / (1.0 + 0.01 * it)
        a = a - step * grad_tan
        a = np.maximum(a, 1e-12)
        a = a / a.sum()

        M = _fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()

    return best_a


def _coordinate_descent(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with golden section on each coordinate."""
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    b = _fast_conv(a).base_max if hasattr(b, 'base_max') else None
    b = _fast_conv(a)

    best_a = a.copy()
    best_M = b.max()
    no_improve = 0
    order = np.arange(n)

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)

        for i in order:
            x_cur = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            c_vals = []
            slopes = []
            has_quad = False
            quad_pos = -1

            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    c = b[k] - x_cur * x_cur
                    c_vals.append(c)
                    slopes.append(0.0)
                    has_quad = True
                    quad_pos = len(c_vals) - 1
                else:
                    j = k - i
                    if 0 <= j < n:
                        c = b[k] - 2.0 * x_cur * a[j]
                        c_vals.append(c)
                        slopes.append(2.0 * a[j])
                    else:
                        c_vals.append(b[k])
                        slopes.append(0.0)

            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0

            c_arr = np.array(c_vals)
            s_arr = np.array(slopes)

            def obj_at(x_test):
                vals = c_arr + s_arr * x_test
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if has_quad:
                    qv = c_arr[quad_pos] + x_test * x_test
                    if qv > m:
                        m = qv
                return m

            lo_x = 0.0
            hi_x = max(3.0 * x_cur, 0.5, 2.0 / n)

            a_pt = hi_x - gr * (hi_x - lo_x)
            b_pt = lo_x + gr * (hi_x - lo_x)
            fa = obj_at(a_pt)
            fb = obj_at(b_pt)

            for _ in range(35):
                if fa < fb:
                    hi_x = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi_x - gr * (hi_x - lo_x)
                    fa = obj_at(a_pt)
                else:
                    lo_x = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo_x + gr * (hi_x - lo_x)
                    fb = obj_at(b_pt)

            x_opt = (lo_x + hi_x) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0

            if abs(x_opt - a[i]) > 1e-10:
                # Update b
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = b[k] - x_cur * x_cur + x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] = b[k] + 2.0 * a[j] * (x_opt - x_cur)
                a[i] = x_opt

        a = a / a.sum()
        M = _fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 15:
            break

    return best_a


def _resample(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def _make_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = _resample(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.02, n), 1e-6)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.08, n), 1e-6)
        inits.append(p_n2 / p_n2.sum())

    inits.append(np.ones(n) / n)

    # Tapered (cosine window)
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * t))
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())

    # Parabolic
    poly = (t * (1 - t)) * 4 + 0.1
    inits.append(poly / poly.sum())

    # Beta-like (more mass in middle)
    beta_shape = np.exp(-2.0 * (t - 0.5) ** 2 / (0.1 + 0.1 * t * (1 - t))) + 0.05
    inits.append(beta_shape / beta_shape.sum())

    # Sine taper (stronger)
    sine_t = np.sin(np.pi * t) ** 1.5 + 0.05
    inits.append(sine_t / sine_t.sum())

    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # Spectral init: iterate to flatten convolution
    a_spec = np.ones(n) / n
    for _ in range(30):
        bs = _fast_conv(a_spec)
        M_s = bs.max()
        # Reduce where convolution is high
        correction = np.zeros(n)
        for i in range(n):
            contribution = 0.0
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                j = k - i
                if 0 <= j < n:
                    contribution += bs[k] / (2 * (bs.max() + 1e-15)) * a_spec[j]
            correction[i] = contribution
        a_spec = a_spec - 0.3 * (correction / correction.max() - correction.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())

    return inits


def run_combined_phase(a0, deadline_phase, seed=0):
    """Run combined optimization: LP -> FISTA -> CD."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    t_start = time.monotonic()
    phase_dur = max(1.0, deadline_phase - t_start)

    best = a.copy()
    best_M = _fast_conv(a).max()

    # Stage 1: LP (50% of time)
    t1 = t_start + 0.5 * phase_dur
    if time.monotonic() < t1:
        a_lp = _lp_iterations(best, 5000, t1, use_full=True)
        M_lp = _fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()

    # Stage 2: FISTA smooth (25% of time)
    t2 = t_start + 0.75 * phase_dur
    if time.monotonic() < t2:
        for tau in [100, 300]:
            if time.monotonic() > t2:
                break
            a_f = _fista_smooth(best, tau, 2000, min(t2, time.monotonic() + (t2 - time.monotonic()) / 2), lr=0.2)
            M_f = _fast_conv(a_f).max()
            if M_f < best_M:
                best_M = M_f
                best = a_f.copy()

    # Stage 3: LP again (15% of time)
    t3 = t_start + 0.90 * phase_dur
    if time.monotonic() < t3:
        a_lp2 = _lp_iterations(best, 3000, t3, use_full=True)
        M_lp2 = _fast_conv(a_lp2).max()
        if M_lp2 < best_M:
            best_M = M_lp2
            best = a_lp2.copy()

    # Stage 4: Coordinate descent (10% of time)
    if time.monotonic() < deadline_phase:
        max_sw = max(3, int((deadline_phase - time.monotonic()) / max(0.1, n / 80.0)))
        a_cd = _coordinate_descent(best, max_sw, deadline_phase, seed=seed)
        M_cd = _fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()

    return best
```
