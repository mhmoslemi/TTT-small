"""
Hybrid optimization: LP alternating + smooth-max Nesterov gradient descent
+ coordinate descent for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import fftconvolve
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.MOSEK]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def smooth_max_gradient(a, T):
    """Compute gradient of smooth max approximation of max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Softmax weights (shifted for numerical stability)
    shifted = T * (conv - max_conv)
    # Clip to avoid underflow issues
    shifted = np.maximum(shifted, -50)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.zeros(2*n-1)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum
    
    # Compute grad[i] = 2 * sum_j a[j] * w[i+j]
    # Using FFT: result[i] = (a * w_rev)[2n-2-i]
    w_rev = w[::-1]
    conv_full = fftconvolve(a, w_rev, mode='full')
    # conv_full has length n + (2n-1) - 1 = 3n-2
    # We need entries at positions 2n-2-0, 2n-2-1, ..., 2n-2-(n-1) = n-1, n, ..., 2n-2
    indices = 2*n-2 - np.arange(n)
    grad = 2.0 * conv_full[indices]
    
    return grad, conv, w


def nesterov_smooth_max(a_init, time_budget=100.0, T=50.0, lr=None, beta=0.9, verbose=False):
    """
    Nesterov accelerated gradient descent on smooth max of convolution peak.
    Works with a normalized (sum=1) non-negative sequence.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if lr is None:
        lr = 0.01
    
    v = np.zeros(n)  # momentum
    no_improve = 0
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient
        grad, conv, w = smooth_max_gradient(a, T)
        
        # Center the gradient (project out all-ones component)
        grad_centered = grad - np.mean(grad)
        
        # Nesterov step
        v = beta * v - lr * grad_centered
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            v = np.zeros(n)
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.8
            v = np.zeros(n)
            no_improve = 0
            if lr < 1e-8:
                break
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start

def anneal_smooth_max(a_init, time_budget=300.0, verbose=False):
    """
    Smooth max gradient descent with temperature annealing.
    Start with low T (smooth) and increase to high T (sharp).
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = float(np.sum(a))

    T_min = 5.0

    n_levels = 12
    T_values = np.geomspace(T_min, T_max, n_levels)

def fast_alternating(a, time_budget=60.0, max_iters=200):
    """
    Fast LP alternating with aggressive blending.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    last_lr = 0.5
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try multiple blending factors
        best_t_obj = best_obj
        best_t_val = last_lr
        best_a_local = best_a.copy()

        t_candidates = [0.3, 0.5, 0.7, 0.9, 1.0]
        if 0 < last_lr < 1:
            t_candidates = sorted(set(t_candidates + [last_lr * 0.7, last_lr * 1.3]))
            t_candidates = [t for t in t_candidates if 0 < t <= 1.0]

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_lr = best_t_val
            stall = 0
        else:
            stall += 1
            if stall > 15:

                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
                last_lr = 0.5

def coordinate_descent_refine(a, time_budget=30.0):
    """
    Targeted coordinate descent: move mass away from peak-contributing positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        # Find positions contributing to the peak
        # c[k_star] = sum_i a[i]*a[k_star-i]
        # Positions i and k_star-i both contribute
        for i in range(0, n, max(1, n // 200)):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j > i and a[i] * a[j] > 0:
                # Try reducing the product a[i]*a[j] by moving mass
                # Move some mass from the larger to a less contributing position
                if a[i] >= a[j]:
                    src = i
                    dst = j
                else:
                    src = j
                    dst = i

                for frac in [0.01, 0.03, 0.05]:
                    delta = frac * a[src]
                    # Find a target position that doesn't contribute to the peak
                    # Try positions far from the peak
                    target = (src + n // 2) % n
                    if abs(target - k_star) > n // 3:  # far from peak
                        a[src] -= delta
                        a[target] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-12:
                            best_obj = new_obj
                            best_a = a.copy()
                            conv = new_conv
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            break

        if not improved:
            # Try global perturbation
            rng = np.random.default_rng(hash((it, n)) % (2**32))
            for scale in [0.01, 0.02, 0.05]:
                a_try = best_a * np.exp(rng.normal(0, scale, n))
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        conv = compute_conv(a)
                        improved = True
                        break
            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Asymmetric: large on one side
        t = np.arange(n) / n
        a = np.exp(-(t - 0.3) ** 2 / 0.05)
        a = a / np.sum(a)
    elif mode == 3:
        # Random smooth
        freqs = rng.integers(1, 5, 3)
        phases = rng.uniform(0, 2*np.pi, 3)
        x = np.arange(n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.cos(2*np.pi*f*x/n + p)
        a = np.maximum(a, 0)
        a = a / np.sum(a) if np.sum(a) > 0 else np.ones(n)/n
    else:
        # Bimodal asymmetric
        c1, c2 = 0.25, 0.7
        w = n // 10
        a = np.exp(-((np.arange(n)/n - c1)**2) / (2*(w/n)**2)) + \
            0.7 * np.exp(-((np.arange(n)/n - c2)**2) / (2*(w/n)**2))
        a = a / np.sum(a)
    return a

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
- LP alternating for structured updates
- Smooth-max Nesterov accelerated gradient descent (fast, many iterations)

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.MOSEK]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def compute_smooth_gradient(a, T):
    """Compute gradient of smooth max approximation of max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    shifted = T * (conv - max_conv)
    shifted = np.maximum(shifted, -50)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.zeros(2 * n - 1)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum

    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # Using FFT: result[i] = (a * w_rev)[2n-2-i]
    w_rev = w[::-1]
    needed = n + (2 * n - 1) - 1  # = 3n-2
    size = 1
    while size < needed:
        size <<= 1
    A = rfft(a, size)
    W = rfft(w_rev, size)
    conv_full = irfft(A * W, size)[:needed]

    indices = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv_full[indices]

    return grad

def anneal_smooth_max(a_init, time_budget=300.0, verbose=False):
    """Smooth max gradient descent with temperature annealing."""
    n = len(a_init)
