import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    c = compute_conv(a_n)
    return 2.0 * n * float(np.max(c))

def subgradient(c, a):
    """Compute subgradient of max(c) w.r.t. a given c = conv(a,a)."""
    n = len(a)
    k_star = int(np.argmax(c))
    g = np.zeros(n)
    for i in range(n):
        v1 = a[k_star - i] if 0 <= k_star - i < n else 0.0
        v2 = a[i - k_star] if 0 <= i - k_star < n else 0.0
        g[i] = v1 + v2
    return g, k_star

def softmax_params(a):
    """Convert a (non-negative, sum=1) to log-parameterization."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)

def softmax(x):
    x_max = np.max(x)
    e = np.exp(x - x_max)
    return e / np.sum(e)

def adam_subgrad(a, n_steps=10000, time_budget=100.0):

    Adam-optimized subgradient descent in softmax parameterization.

    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Adam hyperparameters

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    start = time.monotonic()
    patience = 0
    best_obj_so_far = best_obj
    
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient in a-space
        c = compute_conv(a)
        k_star = int(np.argmax(c))
        g_a = np.zeros(n)
        for i in range(n):
            v1 = a[k_star - i] if 0 <= k_star - i < n else 0.0
            v2 = a[i - k_star] if 0 <= i - k_star < n else 0.0
            g_a[i] = v1 + v2
        
        # Transform to x-space: d f / d x[j] = a[j] * (g_a[j] - sum(g_a * a))
        weighted_mean = np.dot(g_a, a)
        g_x = a * (g_a - weighted_mean)
        
        # Adam update

        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * g_x ** 2

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back

        obj = 2.0 * n * float(np.max(compute_conv(a)))
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 500:
            break
    
    return best_a

def solve_lp(a, M_init=None):
    """Solve: min M s.t. conv(a,g) <= M*1, sum(g)=1, g>=0"""
    n = len(a)
    an = a / a.sum()
    
    # Build sparse constraint matrix
    # A[k, i] = an[k-i] if 0 <= k-i < n else 0
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(max(0, i), min(2*n-1, i+n)):
            rows.append(k)
            cols.append(i)
            vals.append(an[k-i])

    A_ub = np.hstack([A.toarray(), -np.ones((2*n-1, 1))])

    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])

    if res.status == 0:

        if s > 0:

        M = res.x[-1]

    return None, None

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Adam subgradient in softmax parameterization, (2) LP alternating
with adaptive blending, (3) symmetry enforcement, (4) multi-n exploration,
(5) convex combination mixing, (6) Fourier-based initialization.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fft_conv(a):
    """Full linear convolution a*a via real FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    an = a / s
    return 2.0 * n * float(np.max(fft_conv(an)))

def subgrad_a(a):
    """Compute subgradient of max(conv(a,a)) w.r.t. a. Returns (g, k_star)."""
    n = len(a)
    c = fft_conv(a)
    k_star = int(np.argmax(c))
    i_vals = np.arange(n)
    g = np.zeros(n)
    mask1 = (k_star - i_vals >= 0) & (k_star - i_vals < n)
    g[mask1] += a[k_star - i_vals[mask1]]
    mask2 = (i_vals - k_star >= 0) & (i_vals - k_star < n)
    g[mask2] += a[i_vals[mask2] - k_star]
    return g, k_star, c

def softmax(x):
    x = x - np.max(x)
    e = np.exp(x)
    return e / e.sum()

def adam_subgrad(a, time_budget=100.0, lr=0.005, verbose=False):
    """Adam-optimized subgradient in log-parameterization."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    x = np.log(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    patience = 0
    
    while time.monotonic() - start < time_budget:
        g_a, k_star, c = subgrad_a(a)
        weighted_mean = np.dot(g_a, a)
        g_x = a * (g_a - weighted_mean)
        
        t += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * g_x**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a = softmax(x)
        obj = 2.0 * n * float(np.max(fft_conv(a)))
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if verbose and t % 1000 == 0:
            print(f"    adam t={t}: obj={best_obj:.8f}")
        
        if patience > 2000:
            break
    
    return best_a

def solve_lp_conv(a, M_bound=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS.
    """
    n = len(a)
    an = a / a.sum()
    
    # Build the convolution matrix A (2n-1 x n)
    # A[k, i] = an[k-i] if 0 <= k-i < n
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        lo = max(0, i)
        hi = min(2 * n - 1, i + n)
        A[lo:hi, i] = an[:hi - lo]
    
    # LP: minimize M
    # Variable: [g_0, ..., g_{n-1}, M]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    # Constraints: A@g - M <= 0  =>  [A | -1] @ [g; M] <= 0
    A_ub = np.column_stack([A, -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    from scipy.optimize import linprog
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = g.sum()
        if s > 1e-15:
            g = g / s
        return g, float(res.x[-1])
    return None, None

def lp_alternate(a, time_budget=100.0, verbose=False, use_symmetry=False):
    """LP alternating with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_conv(a)
        if g_lp is None:
            break
        
        if use_symmetry:
            g_lp = (g_lp + g_lp[::-1]) / 2.0
            g_lp = g_lp / g_lp.sum()
        
        # Adaptive line search: try multiple t values
        best_t_obj = best_obj
        best_t = 0.5
        best_a_new = best_a.copy()
        
        # Binary search style: first try t=1, then narrow
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            if use_symmetry:
                a_try = (a_try + a_try[::-1]) / 2.0
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_new = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a
            iter_count += 1
            if verbose and iter_count % 10 == 0:
                print(f"    lp_alt it={iter_count}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:

            rng = np.random.default_rng(hash((iter_count, n)) & 0xFFFFFFFF)
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            if use_symmetry:
                a = (a + a[::-1]) / 2.0
            a = a / a.sum()
            iter_count += 1
            
            if iter_count > 20:

    return best_a

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def fourier_init(n, rng, n_modes=10):
    """Initialize with Fourier series."""
    a = np.ones(n) / n
    for k in range(1, n_modes + 1):

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Adam subgradient in softmax parameterization for fast global moves,
(2) LP alternating with adaptive line search for local refinement,
(3) palindromic symmetry enforcement, (4) multi-n exploration,
(5) Fourier-based diverse initialization, (6) convex combination mixing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fft_conv(a):
    """Full linear convolution a*a via real FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    an = a / s
    return 2.0 * n * float(np.max(fft_conv(an)))


def subgrad_a(a):
    """Compute subgradient of max(conv(a,a)) w.r.t. a. Returns (g, k_star)."""
    n = len(a)
    c = fft_conv(a)
    k_star = int(np.argmax(c))
    i_vals = np.arange(n)
    g = np.zeros(n)
    mask1 = (k_star - i_vals >= 0) & (k_star - i_vals < n)
    g[mask1] += a[k_star - i_vals[mask1]]
    mask2 = (i_vals - k_star >= 0) & (i_vals - k_star < n)
    g[mask2] += a[i_vals[mask2] - k_star]
    return g, k_star


def softmax_fn(x):
    """Numerically stable softmax."""
    xm = x - np.max(x)
    e = np.exp(xm)
    return e / e.sum()


def adam_subgrad(a, time_budget=100.0, lr=0.005, verbose=False):
    """Adam-optimized subgradient descent in log-parameterization (softmax)."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()

    v = np.zeros(n)

    start = time.monotonic()
    patience = 0
    best_patience_obj = best_obj

    while time.monotonic() - start < time_budget:
        g_a, k_star = subgrad_a(a)
        weighted_mean = float(np.dot(g_a, a))
        g_x = a * (g_a - weighted_mean)

        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax_fn(x)
        obj = 2.0 * n * float(np.max(fft_conv(a)))

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if verbose and t % 2000 == 0:
            print(f"    adam t={t}: obj={best_obj:.8f}")

        if patience > 3000:

def solve_lp_conv(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k=0..2n-2, sum(g)=1, g>=0.
    Uses HiGHS via scipy linprog. Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    an = a / a.sum()

    # Build dense convolution matrix A: A[k,i] = an[k-i] if 0<=k-i<n
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = an

    # LP variables: [g_0,...,g_{n-1}, M], minimize M
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0

    # Inequality: A@g - M <= 0
    A_ub = np.column_stack([A, -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    from scipy.optimize import linprog
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})

    if res.status == 0:
        g = np.maximum(res.x[:n], 0.0)
        s = g.sum()
        if s > 1e-15:
            g = g / s
            return g, float(res.x[-1])
    return None, None


def lp_alternate(a, time_budget=100.0, verbose=False, use_symmetry=False,

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_init(n, rng, n_modes=12):
    """Initialize with low-frequency Fourier series perturbation of uniform."""
    a = np.ones(n)
    for k in range(1, n_modes + 1):
        amp = rng.normal(0, 0.15 / k)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * np.arange(n) / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def simplex_project(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

            stall = 0
        else:
            lr *= 0.7
            stall += 1
            if stall > 50:
                break

        if verbose:
            print(f"    proj_subgrad: obj={best_obj:.8f}, lr={lr:.5f}")

    return best_a

        if o4 < best_score:
            best_score = o4
            best_seq = a4.copy()
            print(f"  Phase 4 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase 4: obj={o4:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_options = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50),
        n_best,
        n_best + 50, n_best + 100, n_best + 200,
        500, 700, 1000, 1200, 1400
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1600]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if n_try == n_best:
            continue

        a_try = resample_seq(best_seq, n_try)
        alloc = min(remaining * 0.2, 50.0)

        a_opt = lp_alternate(a_try, time_budget=alloc)
        o_try = compute_obj(a_opt)

        if o_try < best_score:
            best_score = o_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={o_try:.8f}")

    # === Phase 6: Random restarts ===
    print(f"\n=== Phase 6: Random Restarts ===")
    remaining = deadline - time.monotonic()

    for r in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        init_type = r % 4
        if init_type == 0:
            a_r = fourier_init(n_best, rng, n_modes=15)
        elif init_type == 1:
            a_r = best_seq * np.exp(rng.normal(0, 0.06, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        elif init_type == 2:
            # Raised cosine window
            x = np.linspace(0, np.pi, n_best)
            a_r = 0.5 * (1 + np.cos(x)) + 0.2
            a_r = a_r / a_r.sum()
        else:
            # Perturbed resample to different n
            n_r = n_best + int(rng.integers(-100, 100))
            n_r = max(200, min(1600, n_r))
            a_r = resample_seq(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.04, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 35.0)
        a_opt = lp_alternate(a_r, time_budget=alloc)
        o_r = compute_obj(a_opt)

        if o_r < best_score:
            best_score = o_r
            best_seq = a_opt.copy()
            n_best = len(a_opt)
            print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={o_r:.8f}")

    # === Phase 7: Convex combination mixing ===
    print(f"\n=== Phase 7: Convex Mixing ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Mix with uniform
        for t in np.linspace(0.05, 0.5, 10):
            a_mix = (1 - t) * best_seq + t * (np.ones(n_best) / n_best)
            a_mix = a_mix / a_mix.sum()
            o_mix = compute_obj(a_mix)
            if o_mix < best_score:
                best_score = o_mix
                best_seq = a_mix.copy()
                print(f"  Mix t={t:.2f}: obj={best_score:.8f} **")

        # Mix with symmetric version
        a_sym = (best_seq + best_seq[::-1]) / 2.0
        a_sym = a_sym / a_sym.sum()
        for t in [0.3, 0.5, 0.7]:
            a_mix = (1 - t) * best_seq + t * a_sym
            a_mix = a_mix / a_mix.sum()
            o_mix = compute_obj(a_mix)
            if o_mix < best_score:
                best_score = o_mix
                best_seq = a_mix.copy()
                print(f"  SymMix t={t:.2f}: obj={best_score:.8f} **")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.5, 200.0)
        a_f = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        o_f = compute_obj(a_f)
        if o_f < best_score:
            best_score = o_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            # One more Adam pass with smaller lr
            alloc2 = min(remaining * 0.3, 60.0)
            a_a = adam_subgrad(best_seq, time_budget=alloc2, lr=0.002)
            o_a = compute_obj(a_a)
            if o_a < best_score:
                best_score = o_a
                best_seq = a_a.copy()
                print(f"  Final Adam: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            # Symmetric polish
            a_s = (best_seq + best_seq[::-1]) / 2.0
            a_s = a_s / a_s.sum()
            a_sf = lp_alternate(a_s, time_budget=min(remaining * 0.4, 40.0),
                                use_symmetry=True)
            o_sf = compute_obj(a_sf)
            if o_sf < best_score:
                best_score = o_sf
                best_seq = a_sf.copy()
                print(f"  Final SymLP: obj={best_score:.8f}")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
