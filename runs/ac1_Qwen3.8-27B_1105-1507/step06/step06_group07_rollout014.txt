"""
Combined Riemann spectral peak reduction, Adam log-space optimization, 
LP refinement, parametric CMA-ES, and subgradient polish for minimizing
2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def riemann_reduce(a, max_iters=500, step=0.05, num_peaks=1, time_budget=30.0):
    """
    Riemann method: iteratively reduce autocorrelation peaks by modifying
    the power spectrum while preserving phase.
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    a_padded = np.zeros(N)
    a_padded[:n] = a
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a_padded)
        S = np.abs(A)**2
        r = ifft(S).real
        
        peak_val = float(np.max(r))
        if peak_val <= 0:
            break
        
        # Find top peaks to reduce
        peaks = np.argsort(r)[::-1][:num_peaks]
        
        # Modify spectrum to reduce peaks
        S_new = S.copy()
        for k_star in peaks:
            # Gradient of r[k*] w.r.t. S[m] is cos(2*pi*m*k*/N)/N
            # Subtract step * gradient to reduce r[k*]
            grad = np.cos(2.0 * np.pi * np.arange(N) * k_star / N) / N
            S_new -= step * grad
        
        S_new = np.maximum(S_new, 0)
        
        # Reconstruct with phase preservation
        mag = np.sqrt(S_new)
        # Handle zero-magnitude components
        phase = np.ones(N, dtype=complex)
        nonzero = np.abs(A) > 1e-15
        phase[nonzero] = A[nonzero] / np.abs(A)[nonzero]
        
        A_new = mag * phase
        a_new_full = ifft(A_new).real
        a_new = np.maximum(a_new_full[:n], 0)
        
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        a_padded[:n] = a_new
        a_padded[n:] = 0
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        # Adaptive step: reduce if oscillating
        if it > 50 and obj > best_obj:
            step *= 0.95
        elif obj < best_obj - 1e-10:
            step = np.minimum(step * 1.01, 0.5)
    
    return best_a


def smooth_obj_grad(a, beta):
    """Compute smoothed objective (logsumexp) and gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    
    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj


def adam_log_opt(a, max_iters=10000, time_budget=60.0, beta_start=10.0,
                 beta_end=300.0, lr_start=0.03, lr_decay=0.99999):

    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_centered = x - np.max(x)
        a_curr = np.exp(x_centered)
        a_curr = a_curr / np.sum(a_curr)
        
        obj_sm, grad_a = smooth_obj_grad(a_curr, beta)
        grad_x = grad_a * a_curr
        
        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 200 == 0:
            x -= np.mean(x)
        
        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, time_budget=30.0):
    """LP-based refinement."""
    n = len(a)
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    
    conv = compute_conv(a_norm)
    M = float(np.max(conv))
    
    g = solve_lp(a_norm, M, time_limit=min(time_budget * 0.5, 20))
    if g is None:
        return a
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for t in np.linspace(0.05, 1.0, 20):
        if time.monotonic() - start > time_budget:
            break
        a_try = (1 - t) * a_norm + t * g
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
    
    return best_a


def subgradient_polish(a, max_iters=2000, time_budget=30.0):
    """Subgradient descent on the true (non-smooth) objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = 0.01
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_b = float(np.max(conv))
        S = float(np.sum(a))
        
        # Find active set (indices where conv is near max)
        threshold = max_b * 0.999
        active = np.where(conv >= threshold)[0]
        
        # Subgradient: for each i, grad_i = (4n/S^2) * [mean_{k in active} a[k-i] - max_b/S]
        # where a[k-i] = 0 if k-i out of [0, n-1]
        a_padded = np.zeros(n + n - 1)
        a_padded[:n] = a
        
        grad = np.zeros(n)
        for k in active:
            # a[k-i] for i in 0..n-1, valid when 0 <= k-i < n
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            grad[i_min:i_max+1] += a_padded[k - np.arange(i_max+1 - i_min)]
        
        grad /= len(active)
        grad = (4.0 * n / (S * S)) * (grad - max_b / S)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Step with adaptive size
        a_try = a - step * grad / grad_norm
        a_try = np.maximum(a_try, 1e-12)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            step = np.minimum(step * 1.05, 0.1)
        else:
            step *= 0.9
        
        if step < 1e-6:
            break
    
    return best_a


def parametric_search(n, time_budget=30.0):
    """
    Search over a parametric family: a[i] = exp(c0 + c1*t + c2*t^2 + ... + cK*t^K)
    where t = i/(n-1) in [0,1]. Optimize coefficients with scipy minimize.
    """
    K = 6  # polynomial degree
    rng = np.random.default_rng(123)
    
    best_a = None
    best_obj = np.inf
    
    for attempt in range(10):
        if time.monotonic() > time_budget:
            break
        
        # Random initial coefficients
        c_init = np.zeros(K + 1)
        c_init[0] = 0
        for k in range(1, K + 1):
            c_init[k] = rng.uniform(-5, 5) / k
        
        t_vals = np.linspace(0, 1, n)
        
        def param_obj(c):
            a_p = np.exp(c[0] + c[1]*t_vals + c[2]*t_vals**2 + 
                         c[3]*t_vals**3 + c[4]*t_vals**4 +
                         c[5]*t_vals**5 + (c[6]*t_vals**6 if K >= 6 else 0))
            a_p = np.maximum(a_p, 1e-15)
            a_p = a_p / np.sum(a_p)
            return compute_obj(a_p)
        
        res = minimize(param_obj, c_init, method='Nelder-Mead',
                      options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
        
        if res.fun < best_obj:
            best_obj = res.fun
            c_opt = res.x
            a_opt = np.exp(c_opt[0] + c_opt[1]*t_vals + c_opt[2]*t_vals**2 + 
                           c_opt[3]*t_vals**3 + c_opt[4]*t_vals**4 +
                           c_opt[5]*t_vals**5 + (c_opt[6]*t_vals**6 if K >= 6 else 0))
            a_opt = np.maximum(a_opt, 1e-15)
            a_opt = a_opt / np.sum(a_opt)
            best_a = a_opt
    
    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
