import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_obj_grad(a, beta):
    """Compute smoothed objective (logsumexp) and gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))

    e_sum = float(np.sum(e))

    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    size = 1

        size <<= 1

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

"""
Improved Adam log-space optimization with adaptive peak targeting,
multi-scale length exploration, and aggressive high-beta annealing
for minimizing 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """
    Compute smoothed logsumexp objective and gradient w.r.t. a.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)

    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def topk_smooth_grad(a, beta, k=3):
    """
    Compute a weighted logsumexp focusing on top-k peaks with gradient.
    Uses a two-stage approach: first find top-k, then compute weighted smoothing.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    # Find top-k peak indices
    topk_idx = np.argsort(conv)[::-1][:k]
    topk_vals = conv[topk_idx]

    # Weighted logsumexp over top peaks with high beta, plus low-beta over all
    cmax = float(np.max(conv))

    # High concentration on top peaks
    shifted_top = beta * (topk_vals - cmax)
    shifted_top = np.clip(shifted_top, -500, 500)
    e_top = np.exp(shifted_top)
    e_top_sum = float(np.sum(e_top))
    w_top = e_top / e_top_sum

    # Build full weight vector
    w = np.zeros_like(conv)
    for idx_i, tidx in enumerate(topk_idx):
        w[tidx] += w_top[idx_i] * 0.5

    # Add low-contribution from all
    shifted_all = 5.0 * (conv - cmax)
    shifted_all = np.clip(shifted_all, -500, 500)
    e_all = np.exp(shifted_all)
    w_all = e_all / float(np.sum(e_all))
    w = 0.5 * w + 0.5 * w_all

    # Effective logsumexp
    ls = cmax - np.log(np.sum(np.exp(beta * (conv - cmax) - np.log(len(conv))))) / beta
    # Use the weighted version for gradient
    effective_val = float(np.sum(w * conv))

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)

    grad = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad - effective_val / sum_a)

    obj = 2.0 * n * effective_val / (sum_a * sum_a)
    return obj, grad_obj

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

                                 anneal_power=0.8, use_topk=True)
        obj = compute_obj(a_post)
        if obj < best_score:
            best_score = obj
            best_seq = a_post.copy()
            print(f"  Post-local Adam: {best_score:.10f} ***")
        else:
            print(f"  Post-local Adam: {obj:.10f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-length exploration")
    print(f"{'='*60}")

    n_best = len(best_seq)
    n_options = []
    for delta in [-150, -100, -75, -50, -25, 25, 50, 75, 100, 150, 200]:
        n_try = n_best + delta
        if 300 <= n_try <= 2500 and n_try not in n_options:
            n_options.append(n_try)
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_try = resample_sequence(best_seq, n_try)
        s = np.sum(a_try)
        if s > 1e-10:
            a_try = a_try / s

        alloc = min(remaining * 0.1, 50.0)
        a_opt = adam_log_opt_v2(a_try, max_iters=50000, time_budget=alloc,
                                beta_start=20.0, beta_end=800.0,
                                lr_start=0.03, lr_decay=0.99999,
                                anneal_power=0.6, use_topk=True)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  n={n_try}: {best_score:.10f} ***")
        else:
            print(f"  n={n_try}: {obj:.10f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()

        # Very high beta Adam
        alloc = min(remaining * 0.3, 60.0)
        a_f1 = adam_log_opt_v2(best_seq, max_iters=100000, time_budget=alloc,
                               beta_start=2000.0, beta_end=10000.0,
                               lr_start=0.003, lr_decay=0.99999,
                               anneal_power=1.0, use_topk=True)
        obj = compute_obj(a_f1)
        if obj < best_score:
            best_score = obj
            best_seq = a_f1.copy()
            print(f"  Final high-beta: {best_score:.10f} ***")
        else:
            print(f"  Final high-beta: {obj:.10f}")

        # Peak equalize again
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.3, 30.0)
            a_f2 = peak_equalize(best_seq, time_budget=alloc, seed=seed + 100)
            obj = compute_obj(a_f2)
            if obj < best_score:
                best_score = obj
                best_seq = a_f2.copy()
                print(f"  Final PE: {best_score:.10f} ***")
            else:
                print(f"  Final PE: {obj:.10f}")

        # One last Adam
        if time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.5, 20.0)
            a_f3 = adam_log_opt_v2(best_seq, max_iters=50000, time_budget=alloc,
                                   beta_start=5000.0, beta_end=20000.0,
                                   lr_start=0.001, lr_decay=0.99999,
                                   anneal_power=1.0, use_topk=True)
            obj = compute_obj(a_f3)
            if obj < best_score:
                best_score = obj
                best_seq = a_f3.copy()
                print(f"  Ultra-beta: {best_score:.10f} ***")
            else:
                print(f"  Ultra-beta: {obj:.10f}")

    # === Phase 7: Random restarts with structured perturbations ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Random restarts")
    print(f"{'='*60}")

    n_final = len(best_seq)
    for restart in range(10):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        # Different perturbation strategies
        if restart % 4 == 0:
            # Multiplicative noise (small)
            scale = 0.01 + 0.005 * restart
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
        elif restart % 4 == 1:
            # Blend with Hann window
            alpha = 0.05 + 0.02 * restart
            hann = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n_final) / (n_final - 1)))
            hann = hann / np.sum(hann)
            a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * hann
        elif restart % 4 == 2:
            # Blend with raised cosine
            alpha = 0.05 + 0.02 * restart
            rc = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n_final) / n_final))
            rc = np.maximum(rc, 1e-6)
            rc = rc / np.sum(rc)
            a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * rc
        else:
            # Sparse perturbation
            a_pert = best_seq.copy()
            num_spikes = max(1, n_final // 100)
            for _ in range(num_spikes):
                idx = int(rng.integers(0, n_final))
                a_pert[idx] += rng.uniform(0.005, 0.03) * np.mean(best_seq)

        a_pert = np.maximum(a_pert, 1e-10)
        s_p = np.sum(a_pert)
        if s_p < 0.01:
            continue
        a_pert = a_pert / s_p

        alloc = min(remaining * 0.1, 20.0)
        a_opt = adam_log_opt_v2(a_pert, max_iters=30000, time_budget=alloc,
                                beta_start=50.0, beta_end=2000.0,
                                lr_start=0.02, lr_decay=0.99999,
                                anneal_power=0.7, use_topk=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Restart {restart}: {best_score:.10f} ***")
        else:
            print(f"  Restart {restart}: {obj:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif np.isinf(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
