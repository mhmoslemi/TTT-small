"""
Optimized step function height search using:
- FFT-based smoothed gradient descent (logsumexp surrogate) with adaptive beta
- LP-based peak reduction (maximizing sum subject to conv bounds)
- Multi-length exploration with diverse initializations
- Simulated annealing restarts
- Final high-precision polish

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_obj_grad(a, beta):
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj

def adam_log_opt(a, max_iters=10000, time_budget=50.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995, 
                 floor=1e-10):
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_start
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        # Use smoothstep for better annealing
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)
        
        x_centered = x - np.max(x)
        # Add floor to prevent log(0) issues
        a = np.exp(x_centered)
        a = a / np.sum(a)
        
        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a
        
        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 100 == 0:
            x -= np.mean(x)
        
        if it % 200 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(np.maximum(x_c, np.log(floor)))
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=10):
    n = len(a)
    if n > 1200:
        return None
    try:
        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

    n = len(a)

    if s < 1e-10:

    conv = compute_conv(a_norm)
    M = float(np.max(conv))
    
    # Try multiple M values (slightly above current max)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for m_factor in [0.95, 0.97, 0.99, 1.0, 1.01]:

        if M_try <= 0:

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def make_gaussian_init(n, sigma_frac=0.3, seed=42):
    rng = np.random.default_rng(seed)
    x = np.arange(n) / n
    # Multi-Gaussian
    n_gauss = rng.integers(3, 8)
    a = np.zeros(n)
    for _ in range(n_gauss):
        center = rng.uniform(0.1, 0.9) * n
        sigma = rng.uniform(0.05, sigma_frac) * n
        amp = rng.uniform(0.5, 1.0)
        a += amp * np.exp(-0.5 * ((x * n - center) / sigma) ** 2)
    a = np.maximum(a, 1e-8)
    return a / a.sum()

def make_pseudo_random_init(n, seed=42):
    rng = np.random.default_rng(seed)
    # Use a hash-like pattern
    a = rng.uniform(0.1, 1.0, n)
    # Add some structure
    freq = rng.uniform(0.5, 5.0)
    phase = rng.uniform(0, 2*np.pi)
    mod = 0.3 * np.cos(2*np.pi*freq*np.arange(n)/n + phase)
    a = a + mod
    a = np.maximum(a, 1e-8)
    return a / a.sum()

def make_triangle_init(n, seed=42):
    # Triangle with noise
    x = np.arange(n) / n
    a = 1.0 - np.abs(2*x - 1)
    rng = np.random.default_rng(seed)
    a += 0.1 * rng.normal(0, 1, n)
    a = np.maximum(a, 1e-8)
    return a / a.sum()

def make_sparse_init(n, density=0.5, seed=42):
    rng = np.random.default_rng(seed)
    mask = rng.random(n) < density
    a = np.where(mask, rng.uniform(0.5, 1.0, n), 0.0)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()

"""
Optimized step function height search combining:
- FFT-based smoothed gradient descent (logsumexp) with adaptive beta annealing
- LP-based peak reduction via Toeplitz constraint solving
- Multi-length exploration with diverse structured initializations (chirp, Gaussian, sparse)
- Simulated annealing restarts with strategy diversity
- Coordinate descent fine-tuning on peak-contributing elements
- Final high-beta polynomial-time polish

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """Smoothed objective via logsumexp and its gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)

    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def adam_log_opt(a, max_iters=10000, time_budget=50.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995):
    """Adam in log-space with geometric beta annealing and smoothstep progress."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    lr = lr_start

    start = time.monotonic()

        if time.monotonic() - start > time_budget:
            break

        smooth_p = progress * progress * (3.0 - 2.0 * progress)
        beta = beta_start * (beta_end / beta_start) ** smooth_p

        a = np.exp(x_centered)
        a = a / np.sum(a)

def build_toeplitz(a):
    """Build Toeplitz matrix for linear convolution: (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M elementwise, g >= 0."""
    n = len(a)
    if n > 1200:
        return None
    try:
        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def coordinate_refine(a, time_budget=15.0, seed=0):

def make_gaussian_init(n, seed=42):
    """Multi-Gaussian initialization with random centers and widths."""
    rng = np.random.default_rng(seed)
    x = np.arange(n) / n
    n_gauss = rng.integers(3, 8)
    a = np.zeros(n)
    for _ in range(n_gauss):
        center = rng.uniform(0.1, 0.9) * n
        sigma = rng.uniform(0.05, 0.35) * n
        amp = rng.uniform(0.5, 1.0)
        a += amp * np.exp(-0.5 * ((x * n - center) / sigma) ** 2)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def make_chirp_init(n, seed=42):
    """Real chirp sequence (approximately flat DFT magnitude)."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    rate = rng.uniform(0.5, 2.0)
    base = 1.0 + 0.5 * np.cos(np.pi * rate * x ** 2 / n)
    a = np.maximum(base, 0.1)
    return a / a.sum()


def make_pseudo_random_init(n, seed=42):
    """Random initialization with low-frequency modulation."""
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.2, 1.0, n)
    for freq in rng.uniform(0.3, 5.0, size=3):
        phase = rng.uniform(0, 2 * np.pi)
        a += 0.2 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def make_bipolar_init(n, seed=42):
    """Two-hump asymmetric initialization."""
    rng = np.random.default_rng(seed)
    split = int(n * rng.uniform(0.3, 0.7))
    a = np.zeros(n)
    a[:split] = rng.uniform(0.8, 1.2)
    a[split:] = rng.uniform(0.3, 0.7)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def make_sparse_init(n, density=0.6, seed=42):
    """Sparse initialization with random on/off pattern."""
    rng = np.random.default_rng(seed)
    mask = rng.random(n) < density
    a = np.where(mask, rng.uniform(0.5, 1.5, n), 0.0)
    if a.sum() < 1e-10:

    return a / a.sum()

                best_seq = a.copy()
                print(f"    Round {r} (beta {b_s:.0f}-{b_e:.0f}): {best_score:.8f} ***")

        # LP refinement (if size allows)
        if len(a) <= 1100 and time.monotonic() < deadline - 35:
            remaining = deadline - time.monotonic()
            a_lp = lp_refine(a, time_budget=min(remaining * 0.12, 18.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a = a_lp.copy()
                print(f"    LP: {best_score:.8f} ***")

            # Adam after LP
            if time.monotonic() < deadline - 15:
                remaining = deadline - time.monotonic()
                a_post = adam_log_opt(a, max_iters=8000,
                                      time_budget=min(remaining * 0.08, 12.0),
                                      beta_start=100.0, beta_end=600.0,
                                      lr_start=0.02, lr_decay=0.99999)
                obj_post = compute_obj(a_post)
                if obj_post < best_score:
                    best_score = obj_post
                    best_seq = a_post.copy()
                    print(f"    LP+Adam: {best_score:.8f} ***")

    # === Phase 3: Coordinate refinement ===
    print(f"\nPhase 3: Coordinate refinement")

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        a_cd = coordinate_refine(best_seq, time_budget=min(remaining * 0.1, 15.0), seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord: {best_score:.8f} ***")

        for cs in [seed + 100, seed + 200]:
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            a_cd2 = coordinate_refine(best_seq, time_budget=min(remaining * 0.07, 10.0), seed=cs)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  Coord (seed {cs}): {best_score:.8f} ***")

    # === Phase 4: Annealing restarts with diverse strategies ===
    print(f"\nPhase 4: Annealing restarts")

    if best_seq is not None:
        n_best = len(best_seq)
        for restart in range(12):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            strategy = restart % 5
            if strategy == 0:
                scale = 0.04 + 0.01 * restart
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            elif strategy == 1:
                a_pert = best_seq.copy()
                n_swaps = max(2, n_best // 200)
                for _ in range(n_swaps):
                    i1 = rng.integers(0, n_best)
                    i2 = rng.integers(0, n_best)
                    a_pert[i1], a_pert[i2] = a_pert[i2], a_pert[i1]
            elif strategy == 2:
                alpha = rng.uniform(0.1, 0.35)
                other = make_chirp_init(n_best, seed=seed + restart * 13)
                a_pert = (1 - alpha) * best_seq + alpha * other
            elif strategy == 3:
                a_pert = best_seq.copy()
                n_spikes = max(2, n_best // 80)
                for _ in range(n_spikes):
                    idx = rng.integers(0, n_best)
                    a_pert[idx] += rng.uniform(0.05, 0.2) * np.mean(best_seq)
            else:
                a_pert = best_seq.copy()
                # Local smoothing
                w = 3
                a_pert = np.convolve(a_pert, np.ones(w) / w, mode='same')
                a_pert += 0.05 * rng.normal(0, 1, n_best)

            a_pert = np.maximum(a_pert, 1e-12)
            a_pert = a_pert / np.sum(a_pert)

            alloc = min(remaining * 0.08, 10.0)
            a_opt = adam_log_opt(a_pert, max_iters=8000, time_budget=alloc,
                                 beta_start=20.0, beta_end=500.0,
                                 lr_start=0.03, lr_decay=0.99999)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart} (strat {strategy}): {best_score:.8f} ***")

    # === Phase 5: Final high-beta polish ===
    print(f"\nPhase 5: Final polish")

    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()

        # Very high beta for precise peak minimization
        a_final = adam_log_opt(best_seq, max_iters=10000,
                               time_budget=min(remaining * 0.4, 12.0),
                               beta_start=500.0, beta_end=2000.0,
                               lr_start=0.008, lr_decay=0.99999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  High-beta: {best_score:.8f} ***")

        # LP polish if size allows
        if len(best_seq) <= 1000 and time.monotonic() < deadline - 3:
            remaining = deadline - time.monotonic()
            a_final_lp = lp_refine(best_seq, time_budget=min(remaining * 0.4, 10.0))
            obj_final_lp = compute_obj(a_final_lp)
            if obj_final_lp < best_score:
                best_score = obj_final_lp
                best_seq = a_final_lp.copy()
                print(f"  Final LP: {best_score:.8f} ***")

        # One last coordinate pass
        if time.monotonic() < deadline - 2:
            remaining = deadline - time.monotonic()
            a_last = coordinate_refine(best_seq, time_budget=min(remaining * 0.5, 5.0), seed=seed + 999)
            obj_last = compute_obj(a_last)
            if obj_last < best_score:
                best_score = obj_last
                best_seq = a_last.copy()
                print(f"  Final coord: {best_score:.8f} ***")

    # === Fallback and validation ===
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
