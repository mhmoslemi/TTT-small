"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses IFTA (Iterative Fourier Transform), SLM (Successive Linearization with LP),
Adam in log-space, multi-scale approach, and coordinate descent.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def ifta_optimize(a, max_iters=5000, time_budget=60.0, gamma=0.5, eps=1e-10):
    """
    IFTA (Iterative Fourier Transform Algorithm) to minimize max(conv(a,a)).
    Works in the Fourier domain to flatten the self-convolution.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Forward: compute A = fft(a), B = A^2, b = ifft(B) = conv(a,a)
        A = fft(a, size)
        B = A * A
        b = ifft(B).real[:2*n-1]  # this is conv(a, a)
        
        # Modify b to reduce peak
        b_mean = np.mean(b)
        b_max = np.max(b)
        
        # Flat-top: clip values above a threshold
        # threshold = mean + gamma * (max - mean)
        threshold = b_mean + (1.0 - gamma) * (b_max - b_mean)
        b_mod = np.minimum(b, threshold)
        
        # Reconstruct
        B_new = fft(b_mod, size)
        # Ensure B_new is conjugate symmetric (it should be since b_mod is real)
        A_new = np.sqrt(np.maximum(B_new, 0) + eps)
        # Actually, we need to be more careful: B_new should be conjugate symmetric
        # A_new should be the "square root" in the sense that A_new^2 = B_new
        # But we also need A_new to be conjugate symmetric for a_new to be real
        # Since B_new is conjugate symmetric, its square root is also conjugate symmetric
        
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        a = a_new
        
        # Check objective periodically
        if it % 50 == 0 or it == max_iters - 1:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a


def slm_optimize(a, max_iters=200, time_budget=60.0, lp_time_limit=10):
    """
    SLM (Successive Linearization Method): iterate LP to minimize peak of conv(a,a).
    At each step, linearize conv(a,a) around current a and solve LP.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        if n > 1000:
            break  # LP too slow for large n
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Build linearized constraint matrix
        # (a0 * a)[k] = sum_i a0[i] * a[k-i] for valid i, k-i
        # This is a Toeplitz-like matrix
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            # a[j] contributes to (a0*a)[k] for k = j, j+1, ..., j+n-1 with coefficient a0[k-j]
            for k in range(j, min(j + n, 2 * n - 1)):
                A_ub[k, j] = a[k - j]
        
        c = -np.ones(n)  # maximize sum(a)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': lp_time_limit})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                s_g = float(np.sum(g))
                if s_g > 1e-10:
                    g = g / s_g
                    obj_g = compute_obj(g)
                    if obj_g < best_obj:
                        best_obj = obj_g
                        best_a = g.copy()
                    # Blend for smoother convergence
                    # a = 0.5 * a + 0.5 * g (or full step)
                    a = g
        except Exception:
            break
    
    return best_a


def build_toeplitz_fast(a0):
    """Build the matrix for (a0 * a) efficiently."""
    n = len(a0)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a0[:end - j]
    return A


def adam_log_opt(a, max_iters=5000, time_budget=60.0, beta_start=10.0,
                 beta_end=300.0, lr_start=0.04, lr_decay=0.99998):
    """Adam in log-space with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m_ = np.zeros(n)
    v_ = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()
    
    size_a = 1
    while size_a < 2 * n - 1:
        size_a <<= 1
    size_cc = 1
    while size_cc < 2 * (2 * n - 1):
        size_cc <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)
        
        # Compute smoothed gradient
        conv = compute_conv(a)
        sum_a = float(np.sum(a))
        cmax = float(np.max(conv))
        shifted = beta * (conv - cmax)
        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        e_sum = float(np.sum(e))
        w = e / e_sum
        ls = cmax + np.log(e_sum) / beta
        
        # Gradient of soft_max w.r.t. a[i]
        W = rfft(w, size_cc)
        A = rfft(a, size_cc)
        cc = irfft(W * np.conj(A), size_cc)
        grad_ls = 2.0 * cc[:n]
        
        grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
        grad_x = grad_obj * a
        
        # Adam update
        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m_ = beta1 * m_ + (1 - beta1) * grad_x
        v_ = beta2 * v_ + (1 - beta2) * (grad_x * grad_x)
        m_hat = m_ / (1 - beta1 ** t)
        v_hat = v_ / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 200 == 0:
            x -= np.mean(x)
        
        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def coordinate_refine(a, time_budget=15.0, seed=0):
    """Fine-tune by adjusting pairs of coordinates near peak locations."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()

    rng = np.random.default_rng(seed)

        conv = compute_conv(best_a)

        improved = False
        
        for p in peaks:
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            for i in range(i_min, i_max + 1):
                if time.monotonic() - start > time_budget:
                    break
                j = p - i
                if j < 0 or j >= n:
                    continue
                for delta_frac in [0.005, 0.01, 0.02, -0.01, -0.02]:
                    delta = delta_frac * max(best_a[i], best_a[j])
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
            if improved:
                break
        if not improved and it > 30:
            break
    
    return best_a

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines IFTA (Iterative Fourier Transform Algorithm), SLM (Successive Linearization
with LP), Adam in log-space with temperature annealing, multi-scale approach,
and coordinate descent. The IFTA approach works in the Fourier domain to flatten
the self-convolution, providing a different optimization landscape than gradient methods.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def ifta_optimize(a, max_iters=5000, time_budget=60.0, gamma=0.5, strategy="clip"):
    """
    IFTA (Iterative Fourier Transform Algorithm) to minimize max(conv(a,a)).
    Works in the Fourier domain to flatten the self-convolution.
    
    Strategy options:
      "clip" - hard clip values above threshold
      "soft" - soft compression using power law
      "whiten" - reduce contrast iteratively
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute current convolution
        A = fft(a, size)
        B = A * A
        b = ifft(B).real[:2 * n - 1]

        b_mean = np.mean(b)
        b_max = np.max(b)

        if strategy == "clip":
            threshold = b_mean + (1.0 - gamma) * (b_max - b_mean)
            b_mod = np.minimum(b, threshold)
        elif strategy == "soft":
            # Power compression: b_new = sign(b) * |b|^alpha
            alpha = 0.5 + 0.3 * gamma  # between 0.5 and 0.8
            b_mod = b_mean + (b - b_mean) * np.abs((b - b_mean) / max(b_max - b_mean, 1e-15)) ** (alpha - 1) * (b_max - b_mean)
            b_mod = np.maximum(b_mod, 0)
        else:  # whiten
            # Reduce contrast: b_new = b - beta*(b - mean)
            beta = gamma
            b_mod = b - beta * (b - b_mean)
            b_mod = np.maximum(b_mod, 0)

        # Reconstruct a from modified b
        B_new = fft(b_mod, size)
        # Add small epsilon for numerical stability
        A_new = np.sqrt(B_new + 1e-10)
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Check objective periodically
        if it % 30 == 0 or it == max_iters - 1:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

        a = a_new

    return best_a


def slm_optimize(a, max_iters=100, time_budget=60.0, lp_time_limit=10):
    """
    SLM (Successive Linearization Method): iterate LP to minimize peak of conv(a,a).
    At each step, linearize conv(a,a) around current a and solve LP.
    """
    n = len(a)
    if n > 1000:
        return a.copy()

    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Build linearized constraint: (a0 * a)[k] <= M
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            end = min(j + n, 2 * n - 1)
            A_ub[j:end, j] = a[:end - j]

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': lp_time_limit})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                s_g = float(np.sum(g))
                if s_g > 1e-10:
                    g = g / s_g
                    obj_g = compute_obj(g)
                    if obj_g < best_obj:
                        best_obj = obj_g
                        best_a = g.copy()
                    a = g
            else:
                break
        except Exception:
            break

    return best_a


def adam_log_opt(a, max_iters=5000, time_budget=60.0, beta_start=10.0,
                 beta_end=300.0, lr_start=0.04, lr_decay=0.99998):
    """Adam in log-space with temperature annealing for smooth max approximation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m_ = np.zeros(n)
    v_ = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()

    size_a = 1
    while size_a < 2 * n - 1:
        size_a <<= 1
    size_cc = 1
    while size_cc < 2 * (2 * n - 1):
        size_cc <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)

        # Smoothed gradient via logsumexp
        conv = compute_conv(a)
        sum_a = float(np.sum(a))
        cmax = float(np.max(conv))
        shifted = beta * (conv - cmax)
        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        e_sum = float(np.sum(e))
        if e_sum < 1e-300:
            w = np.full(len(conv), 1.0 / len(conv))
            ls = cmax
        else:
            w = e / e_sum
            ls = cmax + np.log(e_sum) / beta

        W = rfft(w, size_cc)
        A_f = rfft(a, size_cc)
        cc = irfft(W * np.conj(A_f), size_cc)
        grad_ls = 2.0 * cc[:n]

        grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
        grad_x = grad_obj * a

        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m_ = beta1 * m_ + (1 - beta1) * grad_x
        v_ = beta2 * v_ + (1 - beta2) * (grad_x * grad_x)
        m_hat = m_ / (1 - beta1 ** t)
        v_hat = v_ / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 200 == 0:
            x -= np.mean(x)

        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def coordinate_refine(a, time_budget=15.0, seed=0):
    """Fine-tune by adjusting pairs of coordinates near peak locations."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        peaks = np.argsort(conv)[::-1][:3]
        improved = False

        for p in peaks:
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            for i in range(i_min, i_max + 1):
                if time.monotonic() - start > time_budget:
                    break
                j = p - i
                if j < 0 or j >= n:
                    continue
                for delta_frac in [0.003, 0.008, 0.015, -0.005, -0.01]:
                    delta = delta_frac * max(best_a[i], best_a[j], 1e-10)
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved and it > 50:
            break

    return best_a


def subgradient_project(a, max_iters=2000, time_budget=30.0):
    """
    Subgradient method on the max function with projection.
    At each step, compute subgradient of max(conv(a,a)) and take a step.
    Uses Polyak step size based on gap to theoretical lower bound.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Theoretical lower bound is 1 (flat convolution)
    lb = 1.0
    gap = best_obj - lb

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_idx = np.argmax(conv)
        M = conv[max_idx]
        sum_a = np.sum(a)
        obj = 2.0 * n * M / (sum_a * sum_a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Subgradient of max(conv(a,a)) w.r.t. a
        # d/d(a[i]) of (a*a)[k] = a[k-i] + a[k-i] (for i != k-i) or 2*a[k-i] (for i == k-i)
        # Actually: (a*a)[k] = sum_{i+j=k} a[i]*a[j]
        # d/d(a[i]) (a*a)[k] = a[k-i] + a[k-i] if we consider both terms
        # = 2*a[k-i] if i = k-i, else a[k-i] + a[k-i] = 2*a[k-i]
        # Wait: (a*a)[k] = sum_i a[i]*a[k-i] (where both indices are valid)
        # d/d(a[i]) = a[k-i] + a[k-i] if i != k-i, or a[k-i] if i = k-i
        # No: d/d(a[i]) of sum_j a[j]*a[k-j] = a[k-i] + a[k-i] (chain rule, i appears in two places)
        # = 2*a[k-i] (if i and k-i are distinct) or a[k-i] (if i = k-i, i.e., k=2i)

        sg = np.zeros(n)
        k = max_idx
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                if i == j:
                    sg[i] += a[j]  # appears once
                else:
                    sg[i] += a[j]  # from term a[i]*a[j]
                    # and a[j] term: d/d(a[i]) of a[j]*a[k-j] where k-j = i means j = k-i
                    # This is already counted
        # Actually simpler: sg[i] = a[k-i] if 0<=k-i<n, plus a[k-i] again if i != k-i
        # Let me just use: sg[i] = (a[k-i] if 0<=k-i<n else 0) + (a[k-i] if 0<=k-i<n and i!=k-i else 0)
        # = 2*a[k-i] if i != k-i and valid, a[k-i] if i == k-i and valid

        # Simpler formulation:
        sg = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                sg[i] = 2.0 * a[j]
        # Fix the diagonal: if k is even and i = k/2 is valid, sg[i] should be a[i] not 2*a[i]
        if k % 2 == 0:
            i_diag = k // 2
            if 0 <= i_diag < n:
                sg[i_diag] = a[i_diag]

        # Normalize gradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm

        # Polyak step size
        gap = max(best_obj - lb, 1e-6)
        step = gap / (sg_norm + 1e-10)
        step = min(step, 0.1)  # cap step size

        # Update in log-space to maintain positivity
        a_new = a * np.exp(-step * sg / a)
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / np.sum(a_new)

        a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-strategy search combining IFTA, SLM-LP, Adam log-space,
    subgradient, multi-scale, and coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            p_obj = compute_obj(parent)
            if p_obj < best_score:
                best_score = p_obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={p_obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 700
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Initialized uniform n={n_init}: obj={best_score:.8f}")

    print(f"Starting: best={best_score:.8f}, n={len(best_seq)}")

    # === Phase 1: IFTA with multiple strategies ===
    print(f"\n{'='*60}")
    print(f"Phase 1: IFTA optimization")
    print(f"{'='*60}")

    n_best = len(best_seq)
    strategies = [
        ("clip", 0.3), ("clip", 0.5), ("clip", 0.7),
        ("soft", 0.4), ("soft", 0.6),
        ("whiten", 0.3), ("whiten", 0.5),
    ]

    for strat, gam in strategies:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        t_alloc = min(remaining * 0.15, 35.0)
        a_ifta = ifta_optimize(best_seq, max_iters=30000, time_budget=t_alloc,
                              gamma=gam, strategy=strat)
        obj_ifta = compute_obj(a_ifta)
        print(f"  IFTA {strat}(gamma={gam}): {obj_ifta:.8f}")

        if obj_ifta < best_score:
            best_score = obj_ifta
            best_seq = a_ifta.copy()
            print(f"  ** Updated best: {best_score:.8f}")

    # === Phase 2: Adam log-space on best ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Adam log-space optimization")
    print(f"{'='*60}")

    for round_idx in range(4):
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        t_alloc = min(remaining * 0.15, 40.0)
        b_start = 5.0 * (1 + round_idx)
        b_end = 100.0 * (1 + round_idx)
        a_adam = adam_log_opt(best_seq, max_iters=30000, time_budget=t_alloc,
                             beta_start=b_start, beta_end=b_end,
                             lr_start=0.04 / (1 + 0.3 * round_idx), lr_decay=0.99998)
        obj_adam = compute_obj(a_adam)
        print(f"  Adam round {round_idx} (beta {b_start:.0f}-{b_end:.0f}): {obj_adam:.8f}")

        if obj_adam < best_score:
            best_score = obj_adam
            best_seq = a_adam.copy()
            print(f"  ** Updated best: {best_score:.8f}")

    # === Phase 3: SLM with LP ===
    print(f"\n{'='*60}")
    print(f"Phase 3: SLM LP optimization (n={len(best_seq)})")
    print(f"{'='*60}")

    if len(best_seq) <= 900:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t_slm = min(remaining * 0.3, 90.0)
            a_slm = slm_optimize(best_seq, max_iters=150, time_budget=t_slm,
                                lp_time_limit=min(8, t_slm / 15))
            obj_slm = compute_obj(a_slm)
            print(f"  SLM: {obj_slm:.8f}")
            if obj_slm < best_score:
                best_score = obj_slm
                best_seq = a_slm.copy()
                print(f"  ** Updated best: {best_score:.8f}")

            # Adam after SLM
            if time.monotonic() < deadline - 30:
                remaining = deadline - time.monotonic()
                a_post = adam_log_opt(a_slm, max_iters=20000,
                                     time_budget=min(remaining * 0.15, 25.0),
                                     beta_start=50.0, beta_end=400.0,
                                     lr_start=0.02, lr_decay=0.99999)
                obj_post = compute_obj(a_post)
                print(f"  SLM+Adam: {obj_post:.8f}")
                if obj_post < best_score:
                    best_score = obj_post
                    best_seq = a_post.copy()
                    print(f"  ** Updated best: {best_score:.8f}")

    # === Phase 4: Subgradient method ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Subgradient optimization")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 50:
        remaining = deadline - time.monotonic()
        t_sub = min(remaining * 0.2, 30.0)
        a_sub = subgradient_project(best_seq, max_iters=10000, time_budget=t_sub)
        obj_sub = compute_obj(a_sub)
        print(f"  Subgradient: {obj_sub:.8f}")
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  ** Updated best: {best_score:.8f}")

    # === Phase 5: Multi-scale approach ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-scale optimization")
    print(f"{'='*60}")

    n_target = len(best_seq)
    a_work = best_seq.copy()
    s = np.sum(a_work)
    if s > 1e-10:
        a_work = a_work / s

    # Build scales
    scales = []
    n_cur = max(20, n_target // 16)
    while n_cur < n_target:
        scales.append(n_cur)
        n_cur = min(int(n_cur * 1.8), n_target)
    scales.append(n_target)
    scales = sorted(set(scales))

    for scale_idx, n_scale in enumerate(scales):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_scale >= n_target:
            a_cur = a_work.copy()
        else:
            a_cur = resample_sequence(a_work, n_scale)
            s = np.sum(a_cur)
            if s > 1e-10:
                a_cur = a_cur / s

        # IFTA at this scale
        t1 = min(remaining * 0.25, 25.0)
        a_s = ifta_optimize(a_cur, max_iters=20000, time_budget=t1, gamma=0.5, strategy="clip")
        obj_s = compute_obj(a_s)

        # Adam at this scale
        t2 = min(remaining * 0.25, 25.0)
        a_s = adam_log_opt(a_s, max_iters=20000, time_budget=t2,
                          beta_start=10.0, beta_end=200.0, lr_start=0.03, lr_decay=0.99998)
        obj_s2 = compute_obj(a_s)
        a_s = a_s if obj_s2 < obj_s else a_s

        a_work = a_s

        if n_scale >= n_target:
            final_obj = compute_obj(a_work)
            if final_obj < best_score:
                best_score = final_obj
                best_seq = a_work.copy()
                print(f"  Multi-scale final: {best_score:.8f} **")
            else:
                print(f"  Multi-scale final: {final_obj:.8f}")
        else:
            print(f"  Scale n={n_scale}: {min(obj_s, obj_s2):.8f}")

    # === Phase 6: IFTA + Adam alternation ===
    print(f"\n{'='*60}")
    print(f"Phase 6: IFTA + Adam alternation")
    print(f"{'='*60}")

    for cycle in range(4):
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # IFTA with varying strategy
        strat = ["clip", "soft", "whiten", "clip"][cycle]
        gam = [0.4, 0.5, 0.4, 0.6][cycle]
        t1 = min(remaining * 0.2, 25.0)
        a1 = ifta_optimize(best_seq, max_iters=20000, time_budget=t1, gamma=gam, strategy=strat)
        obj1 = compute_obj(a1)

        # Adam
        t2 = min(remaining * 0.2, 25.0)
        a2 = adam_log_opt(a1, max_iters=20000, time_budget=t2,
                         beta_start=30.0 + 30 * cycle, beta_end=300.0 + 200 * cycle,
                         lr_start=0.025, lr_decay=0.99999)
        obj2 = compute_obj(a2)

        best_cycle = a2 if obj2 < obj1 else a1
        obj_best_cycle = min(obj1, obj2)
        print(f"  Cycle {cycle} ({strat},{gam}): IFTA={obj1:.8f}, Adam={obj2:.8f}")

        if obj_best_cycle < best_score:
            best_score = obj_best_cycle
            best_seq = best_cycle.copy()
            print(f"  ** Updated best: {best_score:.8f}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq)
    n_options = sorted(set([
        max(100, n_best - 150), max(100, n_best - 80), n_best,
        n_best + 80, n_best + 150, n_best + 300
    ]))
    n_options = [n for n in n_options if 100 <= n <= 2000 and n != n_best]

    for n_try in n_options:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_try = resample_sequence(best_seq, n_try)
        s = np.sum(a_try)
        if s > 1e-10:
            a_try = a_try / s

        t1 = min(remaining * 0.15, 20.0)
        a_t = ifta_optimize(a_try, max_iters=10000, time_budget=t1, gamma=0.5, strategy="clip")
        t2 = min(remaining * 0.15, 20.0)
        a_t = adam_log_opt(a_t, max_iters=10000, time_budget=t2,
                          beta_start=10.0, beta_end=300.0, lr_start=0.03, lr_decay=0.99998)
        obj_t = compute_obj(a_t)
        print(f"  n={n_try}: {obj_t:.8f}")

        if obj_t < best_score:
            best_score = obj_t
            best_seq = a_t.copy()
            print(f"  ** Updated best: {best_score:.8f}")

    # === Phase 8: SLM on final best ===
    if len(best_seq) <= 900 and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        if remaining > 25:
            print(f"\n{'='*60}")
            print(f"Phase 8: Final SLM")
            print(f"{'='*60}")
            t_slm2 = min(remaining * 0.35, 50.0)
            a_slm2 = slm_optimize(best_seq, max_iters=50, time_budget=t_slm2,
                                 lp_time_limit=min(8, t_slm2 / 8))
            obj_slm2 = compute_obj(a_slm2)
            print(f"  SLM: {obj_slm2:.8f}")
            if obj_slm2 < best_score:
                best_score = obj_slm2
                best_seq = a_slm2.copy()
                print(f"  ** Updated best: {best_score:.8f}")

    # === Phase 9: Coordinate refinement ===
    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        a_cd = coordinate_refine(best_seq, time_budget=min(remaining * 0.25, 15.0), seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord refine: {best_score:.8f} ***")

    # === Phase 10: Final high-beta Adam polish ===
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        a_final = adam_log_opt(best_seq, max_iters=10000,
                              time_budget=min(remaining * 0.5, 15.0),
                              beta_start=300.0, beta_end=1500.0,
                              lr_start=0.005, lr_decay=0.99999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final polish: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
