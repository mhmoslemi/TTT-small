def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

    """

    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Use scipy linprog with HiGHS (fastest for this size)
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:

        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    
    # Fallback to CVXPY
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]

Let me implement Anderson acceleration for the alternating method:

The iteration is: `a_{k+1} = T(a_k)` where `T` is the "solve LP and blend" operator. Anderson acceleration maintains a history of iterates and residuals, and extrapolates to get a better next iterate.

For Anderson acceleration of order m:
- Maintain `X = [a_0, ..., a_{m-1}]` and `G = [a - T(a), ..., a_{m-1} - T(a_{m-1})]`
- Solve `min ||G @ beta||^2` s.t. `sum(beta) = 1` to get `beta`
- `a_new = X @ beta` (extrapolated point)
- `g_new = T(a_new)`
- Update history

This can give superlinear convergence in practice.

However, implementing Anderson acceleration correctly with the LP inside is tricky because:
- The extrapolated point `a_new` might not be in the feasible set (sum=1, a>=0)

    """Anderson-accelerated LP alternating."""
    n = len(a)
    a = a / np.sum(a)
    
    X = []  # History of a
    G = []  # History of residuals (a - T(a))
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # T(a) = g_lp (full step)
        residual = a - g_lp
        X.append(a.copy())
        G.append(residual)
        
        # Keep only last m iterates
        if len(X) > m:
            X = X[-m:]
            G = G[-m:]
        
        # Anderson extrapolation
        if len(X) >= 2:
            G_arr = np.array(G)  # (m, n)
            # Solve min ||G @ beta||^2 s.t. sum(beta) = 1
            # Using the standard Anderson formulation
            m_cur = len(X)
            if m_cur >= 2:
                GMG = G_arr.T @ G_arr  # (m, m)
                GM1 = G_arr.T @ np.ones(m_cur)  # (m,)
                # Solve the KKT system
                size = m_cur + 1
                KKT = np.zeros((size, size))
                KKT[:m_cur, :m_cur] = GMG
                KKT[:m_cur, m_cur] = GM1
                KKT[m_cur, :m_cur] = GM1
                rhs = np.zeros(size)
                rhs[m_cur] = 1.0
                try:
                    beta_ext = np.linalg.solve(KKT, rhs)
                    beta = beta_ext[:m_cur]
                    # Clip beta to be non-negative (for stability)
                    beta = np.maximum(beta, 0)
                    bsum = np.sum(beta)
                    if bsum > 0:
                        beta = beta / bsum
                    # Extrapolate
                    a_new = sum(b * x for b, x in zip(beta, X))
                    # Project to feasible set
                    a_new = np.maximum(a_new, 0)
                    s = np.sum(a_new)
                    if s > 1e-10:
                        a_new = a_new / s
                        a = a_new
                except np.linalg.LinAlgError:
                    a = g_lp.copy()
        else:
            a = g_lp.copy()
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a

Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: LP alternating with Anderson acceleration, smoothed gradient descent,
multi-n exploration, and final polish.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):

    Returns (g_normalized, M) or (None, None).
    """

    a = a / np.sum(a)
    A = build_conv_matrix(a)

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        # Solve LP: T(a) = g*
        g_star, M_val = solve_lp(a)
        if g_star is None:
            break
        
        # Residual for Anderson
        residual = a - g_star
        X_hist.append(a.copy())
        G_hist.append(residual)
        
        # Keep only last m
        if len(X_hist) > m:
            X_hist = X_hist[-m:]
            G_hist = G_hist[-m:]
        
        # Anderson extrapolation
        if len(X_hist) >= 2:
            G_arr = np.array(G_hist)  # (m_cur, n)
            m_cur = len(X_hist)
            
            # Solve: min ||G @ beta||_2^2 s.t. sum(beta) = 1
            GMG = G_arr.T @ G_arr  # (m_cur, m_cur)
            GM1 = G_arr.T @ np.ones(m_cur)  # (m_cur,)
            
            # KKT system:
            # [GMG  1] [beta]   [0]
            # [1^T  0] [nu  ] = [1]
            size = m_cur + 1
            KKT = np.zeros((size, size))
            KKT[:m_cur, :m_cur] = GMG + 1e-10 * np.eye(m_cur)  # regularization
            KKT[:m_cur, m_cur] = 1.0
            KKT[m_cur, :m_cur] = 1.0
            rhs = np.zeros(size)
            rhs[m_cur] = 1.0
            
            try:
                sol = np.linalg.solve(KKT, rhs)
                beta = sol[:m_cur]
                # Project beta onto simplex (optional, for stability)
                if np.any(beta < 0):
                    beta = np.maximum(beta, 0)
                    bsum = np.sum(beta)
                    if bsum > 1e-10:
                        beta = beta / bsum
                    else:
                        beta = np.ones(m_cur) / m_cur
                
                # Extrapolate
                a_new = np.zeros(n)
                for b, x in zip(beta, X_hist):
                    a_new += b * x
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a = a_new / s_new
                else:
                    a = g_star.copy()
            except np.linalg.LinAlgError:
                a = g_star.copy()
        else:
            a = g_star.copy()
        
        # Evaluate and track best
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose and it % 10 == 0:
                print(f"    it{it}: obj={best_obj:.8f}")
    
    return best_a


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=0.01, tau_final=0.001, verbose=False):
    """
    Gradient descent on f_smooth(a) = tau * log(sum exp(conv(a,a)/tau)).
    With projection onto simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Anneal tau from tau_init to tau_final
    n_steps = 500
    eta = 0.1  # initial step size
    
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature
        progress = step / n_steps
        tau = tau_init * (tau_final / tau_init) ** progress
        
        # Compute convolution
        conv = compute_conv(a)
        
        # Compute softmax weights (with scaling to avoid overflow)
        conv_shifted = conv / tau - np.max(conv) / tau
        p = np.exp(conv_shifted)
        p = p / np.sum(p)
        
        # Compute gradient: g[j] = 2 * sum_k p[k] * a[k-j] (with bounds)
        # This is 2 * cross_correlation(p, a)[j]
        # Using FFT: ifft(conj(fft(p, N)) * fft(a, N))[:n]
        size = 1
        while size < 2 * n:
            size <<= 1
        P = np.fft.rfft(p, size)
        A_fft = np.fft.rfft(a, size)
        corr = np.fft.irfft(np.conj(P) * A_fft, size)[:n]
        grad = 2.0 * corr
        
        # Step size adaptation
        # Use a simple backtracking scheme
        for _ in range(10):
            a_trial = a - eta * grad
            a_trial = np.maximum(a_trial, 0)
            s_trial = float(np.sum(a_trial))
            if s_trial < 1e-10:
                eta *= 0.5
                continue
            a_trial = a_trial / s_trial
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_obj or obj_trial < compute_obj(a):
                a = a_trial
                eta = min(eta * 1.05, 1.0)  # slightly increase step
                if obj_trial < best_obj:
                    best_obj = obj_trial
                    best_a = a.copy()
                    if verbose and step % 50 == 0:
                        print(f"    sgd step{step}: obj={best_obj:.8f}, tau={tau:.6f}")
                break
            else:
                eta *= 0.5
        else:
            # Couldn't improve, just take a small step
            a_trial = a - eta * grad
            a_trial = np.maximum(a_trial, 0)
            s_trial = float(np.sum(a_trial))
            if s_trial > 1e-10:
                a = a_trial / s_trial
    
    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent_refine(a, time_budget=30.0):
    """
    Coordinate descent: for each pair (i,j) contributing to the peak,
    try to rebalance.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False
        
        # For each i, j = k_star - i, try to move mass
        step = max(1, n // 200)
        for i in range(0, n, step):
            j = k_star - i
            if 0 <= j < n and j != i:
                # Try moving a small amount from max(a[i],a[j]) to min
                if a[i] > a[j]:
                    for frac in [0.05, 0.02, 0.01]:
                        delta = frac * (a[i] - a[j])
                        if delta <= 0:
                            break
                        a[i] -= delta
                        a[j] += delta
                        new_obj = compute_obj(a)
                        if new_obj < best_obj:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[j] -= delta
                elif a[j] > a[i]:
                    for frac in [0.05, 0.02, 0.01]:
                        delta = frac * (a[j] - a[i])
                        if delta <= 0:
                            break
                        a[j] -= delta
                        a[i] += delta
                        new_obj = compute_obj(a)
                        if new_obj < best_obj:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[j] += delta
                            a[i] -= delta
        
        if not improved:
            break
    
    return best_a

We have `f_smooth(a, tau) = tau * log(sum_k exp((a*a)[k] / tau))`.

`∂f_smooth/∂a_j = tau * (1/sum_k exp((a*a)[k]/tau)) * sum_k exp((a*a)[k]/tau) * (1/tau) * ∂(a*a)[k]/∂a_j`
`= (sum_k exp((a*a)[k]/tau) * ∂(a*a)[k]/∂a_j) / sum_k exp((a*a)[k]/tau)`
`= sum_k p_k * ∂(a*a)[k]/∂a_j`

where `p_k = exp((a*a)[k]/tau) / sum_l exp((a*a)[l]/tau)`.

And `∂(a*a)[k]/∂a_j = 2*a[k-j]` if `0 <= k-j < n`, else 0.

This is `2 * (p * a)[j]` where `(p * a)[j] = sum_k p_k * a[k-j]` (with `a[m]=0` for `m` out of range).
