def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

Multi-phase optimization for minimizing peak autocorrelation of non-negative sequences.
Combines LP alternating, log-sum-exp gradient descent, coordinate refinement,
and multi-length exploration with diverse initializations.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a, n):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, n=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    if n is None:
        n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm, n)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.PDLP, cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for sol in solvers:
        try:
            prob.solve(solver=sol)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

    Smoothed obj: 2n/T * log(sum_k exp(T*(a*a)[k])), with sum(a)=1, a>=0.
    """

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    lr_max = 0.1

        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)

        # Compute softmax weights: w_k = exp(T*conv_k) / sum exp(T*conv)
        Tconv = T * conv
        Tconv_max = np.max(Tconv)
        exp_Tconv = np.exp(Tconv - Tconv_max)
        sum_exp = np.sum(exp_Tconv)
        weights = exp_Tconv / sum_exp

        # Using cross-correlation via FFT for efficiency
        # cross_corr[i] = sum_m weights[m+i] * a[m] for valid m

        grad = np.zeros(n)

            lr *= 0.5
            continue

    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eps = 0.5

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # New a proportional to 1/(1 + eps*conv) but we need to be careful
        # since conv has length 2n-1 and a has length n
        # We want to reduce a[i] where it contributes to peaks
        # Simple approach: a_new[i] = a[i] / (1 + eps * max_k conv[k] * weight)
        # Better: use the subgradient at the peak

        k_star = int(np.argmax(conv))
        # Subgradient direction: increase a[i] where it doesn't contribute to peak
        # The peak conv[k_star] involves pairs (i, k_star-i)
        # We want to decrease a[i] for i that contribute to the peak

        # Compute contribution of each a[i] to the peak
        contrib = np.zeros(n)
        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        for i in range(i_lo, i_hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[j]

        # New a: reduce high-contributing positions
        a_new = a.copy()
        max_contrib = np.max(contrib) if np.max(contrib) > 0 else 1.0
        a_new = a * (1.0 - eps * contrib / max_contrib)
        a_new = np.maximum(a_new, 0.01 * np.max(a))  # keep all positive
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eps = min(eps * 1.02, 3.0)
        else:
            eps = max(eps * 0.8, 0.05)

    return best_a


def coordinate_refine(a, time_budget=30.0):
    """Coordinate descent targeting peak reduction."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Top peaks
        num_peaks = min(3, len(conv))
        top_indices = np.argsort(conv)[-num_peaks:]

        improved = False
        for k in top_indices:
            if time.monotonic() - start > time_budget:
                break
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            if i_hi <= i_lo:
                continue

            # Find position contributing most to this peak
            # Contribution of a[i] to conv[k] is proportional to a[k-i]
            i_center = k // 2
            # Try positions away from center (they have more "freedom")
            step = max(1, (i_hi - i_lo) // 20)
            for i in range(i_center, i_hi + 1, step):
                if time.monotonic() - start > time_budget:
                    break
                j = k - i
                if j < 0 or j >= n or j == i:
                    continue
                if a[i] < 1e-8:
                    continue

                # Move mass from i to a position that doesn't affect peak k
                # Position p doesn't affect peak k if p > k or k-p >= n or k-p < 0
                # i.e., p > k or p < k-n+1 (but p>=0) so p > k
                # Or k-p >= n, i.e., p <= k-n
                candidates = []
                # Positions p where p > k (won't pair with anything in [0,n-1] to give k)
                # Actually p pairs with k-p, which is valid if 0<=k-p<n
                # So p contributes to conv[k] if 0<=k-p<n, i.e., k-n+1 <= p <= k
                # Non-contributing: p < k-n+1 or p > k
                for p in [0, n//2, n-1]:
                    if p != i and not (max(0, k-n+1) <= p <= k):
                        candidates.append(p)

                if not candidates:
                    # Pick a random position far from the peak
                    rng_p = np.random.default_rng((it * 1000 + i) % (2**32))
                    candidates = [int(rng_p.integers(0, n))]

                p = candidates[0]
                if p == i:
                    continue

                for frac in [0.03, 0.07, 0.12]:
                    delta = frac * a[i]
                    if delta <= 0:
                        break
                    a[i] -= delta
                    a[p] += delta
                    s = float(np.sum(a))
                    if s < 1e-10:
                        a[i] += delta
                        a[p] -= delta
                        break
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[p] -= delta
                        break

        if not improved and it > 20:
            break

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def init_uniform(n):
    return np.ones(n) / n


def init_random(n, rng):
    a = rng.uniform(0.1, 1.0, n)
    return a / np.sum(a)


def init_cosine(n):
    x = np.linspace(0, np.pi, n)
    a = np.cos(x) ** 2 + 0.3
    return a / np.sum(a)


def init_triangular(n):
    a = 1.0 - np.abs(2.0 * np.arange(n) / (n - 1) - 1.0)
    a = a + 0.1
    return a / np.sum(a)


def init_bimodal(n):
    center = n // 2
    w1 = max(1, n // 6)
    w2 = max(1, n // 10)
    a = np.exp(-((np.arange(n) - center) ** 2) / (2 * w1 ** 2))
    a += 0.4 * np.exp(-((np.arange(n) - center * 0.3) ** 2) / (2 * w2 ** 2))
    a += 0.4 * np.exp(-((np.arange(n) - center * 1.7) ** 2) / (2 * w2 ** 2))
    return a / np.sum(a)


def init_perturbed(base, n, rng, scale=0.1):
    if len(base) != n:
        base = resample_seq(base, n)
    a = base * np.exp(rng.normal(0, scale, n))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return init_uniform(n)
    return a / s


def init_window(n, rng):
    """Random window shape."""
    mode = int(rng.integers(0, 3))
    if mode == 0:
        # Raised cosine
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
    elif mode == 1:
        # Hann-like
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    else:
        # Flat top
        edge = n // 8
        a = np.ones(n)
        t = np.arange(edge)
        a[:edge] = 0.5 - 0.5 * np.cos(np.pi * t / edge)
        a[-edge:] = a[:edge][::-1]
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining LP alternating, log-sum-exp gradient descent,
    repulsion refinement, coordinate descent, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = init_uniform(800)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n0 = len(best_seq)

    # Phase 1: Intensive LP alternating on parent
    print(f"\n=== Phase 1: LP Alternating (n={n0}) ===")
    t1 = time.monotonic()
    b1 = min(250.0, (deadline - t1) * 0.22)
    a1 = lp_alternating(best_seq.copy(), time_budget=b1, verbose=True)
    obj1 = compute_obj(a1)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        print(f"  Phase 1: obj={best_score:.8f}")
    else:
        print(f"  Phase 1: obj={obj1:.8f} (no improvement)")

    # Phase 2: Log-sum-exp gradient descent
    print(f"\n=== Phase 2: LogSumExp GD ===")
    t2 = time.monotonic()
    b2 = min(100.0, (deadline - t2) * 0.08)
    a2 = logsumexp_gd(best_seq.copy(), n_steps=300, time_budget=b2)
    obj2 = compute_obj(a2)
    if obj2 < best_score:
        best_score = obj2
        best_seq = a2.copy()
        print(f"  Phase 2: obj={best_score:.8f}")
    else:
        print(f"  Phase 2: obj={obj2:.8f} (no improvement)")

    # Phase 3: Another LP round
    print(f"\n=== Phase 3: LP Alternating Round 2 ===")
    t3 = time.monotonic()
    b3 = min(200.0, (deadline - t3) * 0.15)
    a3 = lp_alternating(best_seq.copy(), time_budget=b3, verbose=True)
    obj3 = compute_obj(a3)
    if obj3 < best_score:
        best_score = obj3
        best_seq = a3.copy()
        print(f"  Phase 3: obj={best_score:.8f}")
    else:
        print(f"  Phase 3: obj={obj3:.8f} (no improvement)")

    # Phase 4: Repulsion + coordinate refine
    print(f"\n=== Phase 4: Repulsion + Coordinate Refine ===")
    t4 = time.monotonic()
    b4a = min(40.0, (deadline - t4) * 0.04)
    a4a = repulsion_refine(best_seq.copy(), n_steps=200, time_budget=b4a)
    obj4a = compute_obj(a4a)
    if obj4a < best_score:
        best_score = obj4a
        best_seq = a4a.copy()
        print(f"  Repulsion: obj={best_score:.8f}")

    b4b = min(40.0, (deadline - time.monotonic()) * 0.04)
    a4b = coordinate_refine(best_seq.copy(), time_budget=b4b)
    obj4b = compute_obj(a4b)
    if obj4b < best_score:
        best_score = obj4b
        best_seq = a4b.copy()
        print(f"  Coord refine: obj={best_score:.8f}")

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    t5 = time.monotonic()
    n_options = sorted(set([
        n0 - 100, n0 - 50, n0 - 25, n0,
        n0 + 25, n0 + 50, n0 + 75, n0 + 100, n0 + 150, n0 + 200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1400]

    per_n_time = max(8.0, min(40.0, (deadline - t5) * 0.18 / max(1, len(n_options))))

    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n0:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)

        a_opt = lp_alternating(a_try, time_budget=per_n_time, verbose=False)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 6: Diverse initializations with LP
    print(f"\n=== Phase 6: Diverse Initializations ===")
    t6 = time.monotonic()
    n_best = len(best_seq)
    num_restarts = 14
    per_restart = max(10.0, min(40.0, (deadline - t6) * 0.18 / max(1, num_restarts)))

    for r in range(num_restarts):
        if time.monotonic() > deadline - 50:
            break

        # Choose initialization
        n_r = n_best
        init_type = r % 7

        if r >= 6:
            n_r = n_best + int(rng.integers(-75, 75))
            n_r = max(400, min(1400, n_r))

        if init_type == 0:
            a_r = init_uniform(n_r)
        elif init_type == 1:
            a_r = init_random(n_r, rng)
        elif init_type == 2:
            a_r = init_cosine(n_r)
        elif init_type == 3:
            a_r = init_triangular(n_r)
        elif init_type == 4:
            a_r = init_bimodal(n_r)
        elif init_type == 5:
            a_r = init_perturbed(best_seq, n_r, rng, scale=0.15)
        else:
            a_r = init_window(n_r, rng)

        a_opt = lp_alternating(a_r, time_budget=per_restart, verbose=False)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (type={init_type}, n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}, n={n_r}): obj={obj_r:.8f}")

    # Phase 7: Final intensive polish
    print(f"\n=== Phase 7: Final Polish ===")
    t7 = time.monotonic()

    # LP round
    alloc = min(120.0, (deadline - t7) * 0.35)
    if alloc > 15:
        a_lp = lp_alternating(best_seq.copy(), time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")

    # Log-sum-exp GD
    remaining = deadline - time.monotonic()
    alloc = min(50.0, remaining * 0.25)
    if alloc > 10:
        a_gd = logsumexp_gd(best_seq.copy(), n_steps=200, time_budget=alloc)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Final GD: obj={best_score:.8f}")

    # Coordinate refine
    remaining = deadline - time.monotonic()
    alloc = min(30.0, remaining * 0.25)
    if alloc > 8:
        a_cd = coordinate_refine(best_seq.copy(), time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f}")

    # Repulsion
    remaining = deadline - time.monotonic()
    alloc = min(25.0, remaining * 0.25)
    if alloc > 8:
        a_rep = repulsion_refine(best_seq.copy(), n_steps=150, time_budget=alloc)
        obj_rep = compute_obj(a_rep)
        if obj_rep < best_score:
            best_score = obj_rep
            best_seq = a_rep.copy()
            print(f"  Final Repulsion: obj={best_score:.8f}")

    # One more LP
    remaining = deadline - time.monotonic()
    alloc = min(40.0, remaining * 0.6)
    if alloc > 10:
        a_last = lp_alternating(best_seq.copy(), time_budget=alloc, verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = init_uniform(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
