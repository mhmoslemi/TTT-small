import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def solve_lp_min_peak(a, time_limit=20, M_init=None):
    """
    Solve LP: find g >= 0, sum(g) = 1, minimizing max(conv(a,g)).
    Equivalent to: minimize M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    m = 2 * n - 1
    
    # Build Toeplitz-like matrix: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    
    # Variables: g[0], ..., g[n-1], M
    # Minimize M
    # s.t. A @ g - M <= 0 (for each k)
    #      sum(g) = 1
    #      g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Inequality constraints: A@g - M <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, max_iters=30, time_budget=60.0):
    """
    LP-based refinement: alternate between finding optimal direction g
    and taking a step toward it.
    """
    n = len(a)

        g = solve_lp_min_peak(a, time_limit=max(5, int(remaining * 0.5)))
        if g is None:
            break
        
        # Find optimal blend t using golden section
        best_t = 0.0
        best_f = best_obj
        # Try a few values of t
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_try = normalize_simplex((1 - t) * a + t * g)
            f_try = compute_obj(a_try)
            if f_try < best_f:

                best_t = t

            if obj < best_obj:

                best_a = a.copy()
            else:
                a = best_a.copy()
        else:
            break

def multi_peak_gradient_step(a, peak_fraction=0.1, step_size=0.3):
    """
    Gradient step that targets the top `peak_fraction` of convolution peaks.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    threshold = cmax * (1 - peak_fraction)
    
    # Weight peaks that are close to maximum
    above = conv > threshold
    if not np.any(above):
        above = conv >= cmax * 0.9
    
    weights = np.where(above, (conv - threshold) / (cmax - threshold + 1e-10), 0.0)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-10:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
        w_sum = 1.0
    else:
        weights /= w_sum
    
    # Compute gradient via cross-correlation
    size = 1
    while size < 3 * n:
        size <<= 1
    
    Wf = fft(weights, size)
    Af = fft(a, size)
    corr = irfft(np.conj(Wf) * Af, size)[:2*n-1]
    grad = 2.0 * corr
    
    a_new = a - step_size * grad
    a_new = normalize_simplex(a_new)
    return a_new

        grad = np.zeros(n)
        for i in range(n):
            k_diff = k_star - i
            if 0 <= k_diff < n:
                grad[i] = 2.0 * a[k_diff]
        
        # Projected subgradient step
        step = 2.0 * c_peak / (float(np.dot(grad, grad)) + 1e-10)
        step = min(step, 0.5)
        
        a_new = a - step * grad
        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            step = step * 0.5
            a_new = a - step * grad
            a_new = normalize_simplex(a_new)
            if compute_obj(a_new) < best_obj:
                best_obj = compute_obj(a_new)
                best_a = a_new.copy()
                a = a_new
            else:
                break
    
    return best_a


def spectral_flatness_optimization(a, time_budget=50.0, max_iters=100):
    """
    Optimize by flattening the power spectrum.
    Key insight: peak of autocorrelation relates to how "peaked" the spectrum is.
    A flat spectrum (white noise) has low peak autocorrelation.
    """
    n = len(a)
    a = normalize_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        
        # Compress the spectrum toward flatness
        # Use power compression: P_new = P^alpha * P^(1-alpha) * mean(P)
        # where alpha < 1 compresses
        alpha = 0.3 + 0.2 * (it / max_iters)  # start more aggressive
        P_mean = float(np.mean(P))
        P_norm = P / (p_max + 1e-30)
        P_new = P_norm ** alpha
        
        # Preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = normalize_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            a = 0.6 * a + 0.4 * a_new
            a = normalize_simplex(a)
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    # Use spline-like interpolation (cubic for smoothness)
    from scipy.interpolate import interp1d
    f = interp1d(x_old, a, kind='cubic', bounds_error=False, fill_value=0.0)
    result = f(x_new)
    result = np.maximum(result, 0.0)
    return normalize_simplex(result)


def symmetric_init(n):
    """Create a symmetric initial sequence."""
    half = n // 2
    # Create a smooth bump
    x = np.linspace(0, 1, half + 1)
    bump = np.sin(np.pi * x) ** 2
    if n % 2 == 0:
        a = np.concatenate([bump, bump[::-1][1:]])
    else:
        a = np.concatenate([bump, bump[::-1][1:]])
    return normalize_simplex(a)

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale-invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_simplex(a):
    """Exact Euclidean projection onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(a)
    if n == 0:
        return a
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0.0)


def normalize_simplex(a):
    """Quick normalize (clip and scale) - not exact projection but fast."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def gradient_conv_max(a, weights=None):
    """
    Compute gradient of weighted sum of convolution values w.r.t. a.
    If weights is None, uses subgradient of max (only argmax position).

    grad[i] = 2 * sum_j weights[i+j] * a[j]  (for valid indices)
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    if weights is None:
        conv = compute_conv(a)
        idx = int(np.argmax(conv))
        weights = np.zeros(conv_len)
        weights[idx] = 1.0
    else:
        weights = np.asarray(weights, dtype=np.float64)
    
    # Cross-correlation via FFT: grad[i] = 2 * sum_j weights[i+j] * a[j]
    size = 1
    while size < conv_len + n - 1:

    Wf = fft(weights, size)
    Af = fft(a, size)
    corr = np.real(ifft(np.conj(Wf) * Af, size))[:n]

def solve_lp_min_peak(a, time_limit=20):
    """
    LP: minimize M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g (normalized) or None.
    """
    n = len(a)
    m = 2 * n - 1
    
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    # Smooth interpolation
    result = np.interp(x_new, x_old, a)
    result = np.maximum(result, 0.0)
    s = float(np.sum(result))
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return result / s

    deadline = start_time + budget_s - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < np.inf:
        best_score = parent_obj
        best_seq = parent.copy()
    
    if best_seq is None:
        n_init = 700
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Initialized uniform: n={n_init}, score={best_score:.8f}")
    
    n_parent = len(best_seq)
    print(f"Starting: n={n_parent}, score={best_score:.8f}")
    
    # =========================================================
    # Phase 1: Smoothed gradient descent (primary optimizer)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Smoothed projected gradient (n={n_parent})")
    print(f"{'='*60}")
    
    a = project_simplex(best_seq.copy())
    
    for cycle in range(8):
        if time.monotonic() > deadline - 450:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.13, 55.0)
        
        tau_init = 0.008 * (1.0 + 0.2 * cycle)
        a = smooth_gradient_descent(a, time_budget=alloc, tau_init=tau_init,
                                    tau_final=0.00003, step_init=0.4, max_steps=300)
        obj = compute_obj(a)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a.copy()
            print(f"  Grad cycle {cycle}: {best_score:.8f} **")
        else:
            print(f"  Grad cycle {cycle}: {obj:.8f}")
            # Perturb to escape
            rng = np.random.default_rng(seed + cycle * 7)
            a = project_simplex(a * np.exp(rng.normal(0, 0.04 + 0.01 * cycle, len(a))))
    
    # =========================================================
    # Phase 2: LP refinement
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP refinement (n={len(best_seq)})")
    print(f"{'='*60}")
    
    n_cur = len(best_seq)
    if n_cur <= 850:
        # Min-peak LP
        a = project_simplex(best_seq.copy())
        for cycle in range(4):
            if time.monotonic() > deadline - 350:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.13, 45.0)
            
            a = lp_refine(a, max_iters=15, time_budget=alloc)
            obj = compute_obj(a)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP-min-peak {cycle}: {best_score:.8f} **")
            else:
                print(f"  LP-min-peak {cycle}: {obj:.8f}")
        
        # Max-sum LP
        if time.monotonic() < deadline - 250:
            a = project_simplex(best_seq.copy())
            for cycle in range(3):
                if time.monotonic() > deadline - 250:
                    break
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                alloc = min(remaining * 0.13, 40.0)
                
                a = lp_maxsum_refine(a, max_iters=12, time_budget=alloc)
                obj = compute_obj(a)
                if obj < best_score - 1e-12:
                    best_score = obj
                    best_seq = a.copy()
                    print(f"  LP-maxsum {cycle}: {best_score:.8f} **")
                else:
                    print(f"  LP-maxsum {cycle}: {obj:.8f}")
    
    # =========================================================
    # Phase 3: Spectral + multi-peak
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Spectral + multi-peak (n={len(best_seq)})")
    print(f"{'='*60}")
    
    if time.monotonic() < deadline - 200:
        a = project_simplex(best_seq.copy())
        a_spec = spectral_flatness_opt(a, time_budget=45.0, max_iters=70)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score - 1e-12:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: {best_score:.8f} **")
        else:
            print(f"  Spectral: {obj_spec:.8f}")
            # Try blend
            a_bl = project_simplex(0.5 * project_simplex(best_seq) + 0.5 * project_simplex(a_spec))
            obj_bl = compute_obj(a_bl)
            if obj_bl < best_score - 1e-12:
                best_score = obj_bl
                best_seq = a_bl.copy()
                print(f"  Spectral blend: {best_score:.8f} **")
    
    if time.monotonic() < deadline - 180:
        a = project_simplex(best_seq.copy())
        a_mp = multi_peak_descent(a, time_budget=35.0, max_iters=80, peak_frac=0.08)
        obj_mp = compute_obj(a_mp)
        if obj_mp < best_score - 1e-12:
            best_score = obj_mp
            best_seq = a_mp.copy()
            print(f"  Multi-peak: {best_score:.8f} **")
        else:
            print(f"  Multi-peak: {obj_mp:.8f}")
    
    # Follow-up gradient after spectral/multi-peak
    if time.monotonic() < deadline - 150:
        a = project_simplex(best_seq.copy())
        a = smooth_gradient_descent(a, time_budget=35.0, tau_init=0.003,
                                    tau_final=0.00002, step_init=0.3, max_steps=200)
        obj = compute_obj(a)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a.copy()
            print(f"  Post Phase 3 gradient: {best_score:.8f} **")
    
    # =========================================================
    # Phase 4: Multi-length exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-length exploration")
    print(f"{'='*60}")
    
    n_best = len(best_seq)
    # Generate candidate lengths
    n_options = sorted(set([
        max(150, n_best - 200), max(150, n_best - 100),
        max(150, n_best - 50), max(150, n_best - 25),
        n_best + 25, n_best + 50, n_best + 100,
        n_best + 150, n_best + 200, n_best + 300
    ]))
    n_options = [n for n in n_options if 150 <= n <= 1500]
    # Remove duplicates and the current best (we already optimized that)
    n_options = [n for n in n_options if n != n_best]
    
    for n_try in n_options:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 28.0)
        
        # Quick optimization
        a_try = smooth_gradient_descent(a_try, time_budget=alloc * 0.6,
                                        tau_init=0.005, tau_final=0.0001,
                                        step_init=0.3, max_steps=80)
        a_try = multi_peak_descent(a_try, time_budget=alloc * 0.3,
                                   max_iters=30, peak_frac=0.1)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score - 1e-12:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")
    
    # =========================================================
    # Phase 5: Structure-aware random restarts
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Structured restarts (n={len(best_seq)})")
    print(f"{'='*60}")
    
    n_cur = len(best_seq)
    rng = np.random.default_rng(seed + 9999)
    
    strategies = [
        "gaussian", "sinusoidal", "block_reverse", "antispikes",
        "lowpass", "dirichlet_blend", "peak_shave", "phase_noise",
        "bimodal", "tapered"
    ]
    
    for si, strat in enumerate(strategies):
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.12, 28.0)
        
        a_base = project_simplex(best_seq.copy())
        
        if strat == "gaussian":
            a_try = a_base * np.exp(rng.normal(0, 0.08, n_cur))
        elif strat == "sinusoidal":
            freq = rng.uniform(1.0, 6.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
            a_try = a_base * mod
        elif strat == "block_reverse":
            a_try = a_base.copy()
            blen = int(rng.integers(n_cur // 5, max(n_cur // 2, n_cur // 5 + 1)))
            bstart = int(rng.integers(0, max(1, n_cur - blen)))
            a_try[bstart:bstart + blen] = a_try[bstart:bstart + blen][::-1]
        elif strat == "antispikes":
            a_try = a_base.copy()
            top_k = max(1, int(n_cur * 0.1))
            idx = np.argsort(a_try)[-top_k:]
            a_try[idx] *= rng.uniform(0.3, 0.7)
        elif strat == "lowpass":
            sz = max(1, 1 << (2 * n_cur - 1).bit_length())
            A = fft(a_base, sz)
            nz = int(len(A) * 0.25)
            A[nz:len(A) - nz] = 0
            a_try = np.real(ifft(A))[:n_cur]
        elif strat == "dirichlet_blend":
            a_rand = rng.dirichlet(np.ones(n_cur) * 0.3)
            mix = rng.uniform(0.2, 0.5)
            a_try = (1 - mix) * a_base + mix * a_rand
        elif strat == "peak_shave":
            a_try = a_base.copy()
            # Smooth estimate of "background"
            k = max(3, n_cur // 15)
            kernel = np.ones(k) / k
            smooth = np.convolve(a_try, kernel, mode='same')
            excess = np.maximum(a_try - smooth, 0)
            a_try = a_try - 0.4 * excess
        elif strat == "phase_noise":
            sz = max(1, 1 << (2 * n_cur - 1).bit_length())
            A = fft(a_base, sz)
            phase = np.angle(A) + rng.normal(0, 0.2, len(A))
            mag = np.abs(A)
            A_new = mag * np.exp(1j * phase)
            # Enforce conjugate symmetry
            A_new[0] = np.abs(A_new[0])
            half = len(A_new) // 2
            for i in range(1, half + 1):
                if len(A_new) - i >= 0:
                    A_new[len(A_new) - i] = np.conj(A_new[i])
            a_try = np.real(ifft(A_new))[:n_cur]
        elif strat == "bimodal":
            # Create a bimodal perturbation
            center = n_cur // 2
            width = int(n_cur * 0.15)
            bump1 = np.exp(-0.5 * ((np.arange(n_cur) - center * 0.4) / width) ** 2)
            bump2 = np.exp(-0.5 * ((np.arange(n_cur) - center * 1.6) / width) ** 2)
            a_try = a_base * (1.0 + 0.15 * (bump1 - bump2))
        elif strat == "tapered":
            # Apply a different taper
            x = np.linspace(0, 1, n_cur)
            taper = 1.0 + 0.3 * np.cos(2 * np.pi * x) * rng.uniform(0.5, 1.5)
            a_try = a_base * taper
        
        a_try = project_simplex(a_try)
        
        # Optimize
        a_try = smooth_gradient_descent(a_try, time_budget=alloc * 0.5,
                                        tau_init=0.004, tau_final=0.00005,
                                        step_init=0.3, max_steps=60)
        if n_cur <= 750 and alloc > 15:
            a_try = lp_refine(a_try, max_iters=6, time_budget=alloc * 0.3)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score - 1e-12:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  {strat}: {best_score:.8f} **")
        else:
            print(f"  {strat}: {obj_try:.8f}")
    
    # =========================================================
    # Phase 6: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish (n={len(best_seq)})")
    print(f"{'='*60}")
    
    n_final = len(best_seq)
    a_final = project_simplex(best_seq.copy())
    
    for pr in range(8):
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * 0.3, 35.0)
        
        # Alternating: gradient, multi-peak, (LP if possible)
        if pr % 3 == 0:
            a_pol = smooth_gradient_descent(a_final, time_budget=alloc * 0.5,
                                            tau_init=0.001, tau_final=0.00001,
                                            step_init=0.2, max_steps=80)
        elif pr % 3 == 1:
            a_pol = multi_peak_descent(a_final, time_budget=alloc * 0.5,
                                       max_iters=60, peak_frac=0.05)
        else:
            a_pol = a_final.copy()
            # Spectral nudge
            a_spec = spectral_flatness_opt(a_pol, time_budget=min(10, alloc * 0.2),
                                           max_iters=20)
            a_pol = project_simplex(0.6 * a_pol + 0.4 * a_spec)
            a_pol = smooth_gradient_descent(a_pol, time_budget=alloc * 0.3,
                                            tau_init=0.001, tau_final=0.00002,
                                            step_init=0.2, max_steps=50)
        
        # LP if feasible
        if n_final <= 700 and alloc > 15 and pr % 2 == 0:
            a_pol = lp_refine(a_pol, max_iters=5, time_budget=min(12, alloc * 0.3))
        
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score - 1e-13:
            best_score = obj_pol
            best_seq = a_pol.copy()
            a_final = a_pol
            print(f"  Polish {pr}: {best_score:.8f} **")
        else:
            # Gentle blend to maintain progress
            a_final = project_simplex(0.7 * a_final + 0.3 * project_simplex(a_pol))
            print(f"  Polish {pr}: {obj_pol:.8f}")
    
    # =========================================================
    # Final validation
    # =========================================================
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100
    
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    
    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")
    
    return result
