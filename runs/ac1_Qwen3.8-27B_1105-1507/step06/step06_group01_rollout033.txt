import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.max(np.where(cond, rho_arr, 0)))
        theta = (cv[rho - 1] - 1.0) / rho
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_grad_fft(a, p):
    """Compute gradient of log-sum-exp(tau * conv(a,a)) using FFT.
    a: length n, p: length 2n-1 (normalized softmax weights)
    Returns gradient of length n.
    """
    n = len(a)
    # term1[i] = sum_j a[j] * p[i+j] = correlate(p, a, 'valid')[i]
    term1 = np.correlate(p, a, 'valid')
    # term2[i] = sum_j a[j] * p[i-j] = convolve(p, a, 'full')[i]
    term2 = np.convolve(p, a, 'full')[:n]
    return term1 + term2


def mirror_descent_fast(a_init, max_iters, deadline, n_active=5, alpha_init=0.15):
    """Fast mirror descent (multiplicative weights) with multi-active subgradients.
    Works on simplex (sum=1, a>=0).
    """
    a = np.maximum(a_init, 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    alpha = alpha_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            # Adaptive: slightly increase n_active or decrease alpha
            alpha *= 0.95
            if no_improve > 20000:
                break

        # Get top n_active indices
        if n_active >= len(b):
            top_k = np.arange(len(b))
        else:
            top_k = np.argpartition(b, -n_active)[-n_active:]

        # Compute averaged subgradient
        s_grad = np.zeros(n)
        for k in top_k:
            k = int(k)
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            s_grad[valid] += a[idx[valid]]
        s_grad /= float(len(top_k))
        # Project to tangent of simplex
        s_grad -= s_grad.mean()

        # Multiplicative update (mirror descent step)
        # a_new = a * exp(-alpha * s_grad) / Z
        log_update = -alpha * s_grad
        log_update -= log_update.max()  # numerical stability
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-15:
            alpha *= 0.5
            continue
        a_new /= s_new

        a = a_new

    return best_a, best_M


def mirror_descent_adaptive(a_init, deadline, n_active_start=3, n_active_max=30):
    """Mirror descent with adaptive n_active and alpha."""
    a = np.maximum(a_init, 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    alpha = 0.2
    n_active = n_active_start
    total_iters = 0

    while time.monotonic() < deadline:
        # Run a batch of iterations
        batch_end = min(deadline, time.monotonic() + 5.0)
        a_cur, M_cur = mirror_descent_fast(a, 50000, batch_end, n_active, alpha)

        if M_cur < best_M:
            best_M = M_cur
            best_a = a_cur.copy()
        a = a_cur

        # Adapt parameters
        if M_cur < best_M + 1e-10:
            # Making progress: increase aggressiveness
            if n_active < n_active_max:
                n_active = min(n_active + 2, n_active_max)
        else:
            # Stuck: reduce alpha, maybe change n_active
            alpha *= 0.9
            if n_active > 3:
                n_active = max(3, n_active - 2)

        total_iters += 1
        if total_iters % 20 == 0:
            print(f"      MD iter={total_iters}, n_active={n_active}, alpha={alpha:.4f}, M={best_M:.10f}", flush=True)

    return best_a, best_M


def lp_best_response(a, M, n_active, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for k in active, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    b = fast_conv(a)

    # Active set: top n_active
    if n_active >= len(b):
        active = np.arange(len(b))
    else:
        active = np.argpartition(b, -n_active)[-n_active:]
    active = np.sort(active)

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        k = int(k)
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])

    if not vals:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-8,
                             'primal_feasibility_tolerance': 1e-8})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def lp_mirror_descent(a_init, deadline, n_active_start=20, n_active_max=100):
    """LP-based mirror descent with adaptive active set."""
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    n_active = n_active_start
    no_improve = 0

    while time.monotonic() < deadline:
        g = lp_best_response(a, best_M, n_active, time_limit=8)
        if g is None:
            no_improve += 2
            n_active = min(n_active * 2, n_active_max)
            if no_improve > 10:
                break
            continue

        improved = False
        # Line search
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial /= s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
            if n_active > 10:
                n_active = max(10, int(n_active * 0.85))
        else:
            no_improve += 1
            n_active = min(n_active + 5, n_active_max)
            if no_improve > 15:
                break

    return best_a, best_M


def coordinate_descent(a_init, deadline, seed=0):
    """Coordinate descent with golden-section 1D optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(1000):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)

        b = fast_conv(a)
        sweep_changed = False

        for i in order:
            if time.monotonic() > deadline:
                break

            x_cur = a[i]
            # Get affected convolution indices
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            # Compute base values and slopes
            c_vals = []
            slopes = []
            has_quad = False
            quad_pos = -1

            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    c = b[k] - x_cur * x_cur
                    c_vals.append(c)
                    slopes.append(0.0)
                    has_quad = True
                    quad_pos = len(c_vals) - 1
                else:
                    j = k - i
                    if 0 <= j < n:
                        c = b[k] - 2.0 * x_cur * a[j]
                        c_vals.append(c)
                        slopes.append(2.0 * a[j])
                    else:
                        c_vals.append(b[k])
                        slopes.append(0.0)

            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0

            c_arr = np.array(c_vals)
            s_arr = np.array(slopes)

            def obj_at(x_test):
                vals = c_arr + s_arr * x_test
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if has_quad:
                    qv = c_arr[quad_pos] + x_test * x_test
                    if qv > m:
                        m = qv
                return m

            # Golden section search
            lo = 0.0
            hi = max(3.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.1)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_at(a_pt)
            fb = obj_at(b_pt)

            for _ in range(30):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_at(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)

            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0
            x_opt = max(0.0, x_opt)

            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
                sweep_changed = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-14):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_changed:
            no_improve += 5
        if no_improve > 30:
            break

    return best_a, best_M


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.02, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-4)
            inits.append(p_n / p_n.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Concave parabola
    poly = (t * (1 - t)) + 0.1
    inits.append(poly / poly.sum())

    # Cosine windows
    for freq in [1, 2, 3]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())

    # Gaussian
    gauss = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    # Smooth random
    ksize = max(3, n // 15)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # Random with bias toward center
    center_bias = 1.0 + 0.5 * np.exp(-8.0 * (t - 0.5) ** 2)
    rand_init = rng.uniform(0.5, 1.5, n) * center_bias
    inits.append(rand_init / rand_init.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def fista_smoothed(a_init, tau, deadline, lr=0.5):
    """FISTA on log-sum-exp smoothed objective."""
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    t = 1.0
    y = a.copy()
    a_prev = a.copy()

    it = 0
    while time.monotonic() < deadline:
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        grad = smooth_grad_fft(y, p)
        grad_tan = grad - grad.mean()

        step = lr / (1.0 + 0.01 * it)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        it += 1

        if it % 20 == 0:
            M_a = fast_conv(a).max()
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()

    return best_a, best_M


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining fast mirror descent, LP best-response,
    FISTA, and coordinate descent across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size

    # Define lengths to try
    lengths = set()
    lengths.add(n_parent)
    for delta in [-128, -64, 64, 128, 256]:
        nl = n_parent + delta
        if 256 <= nl <= 2048:
            lengths.add(nl)
    for base in [512, 768, 1024, 1536]:
        lengths.add(base)
    lengths = sorted([l for l in lengths if 256 <= l <= 2048])

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Fast mirror descent on multiple lengths", flush=True)
    print(f"Lengths: {lengths}", flush=True)
    print(f"{'='*60}", flush=True)

    # Phase 1: Mirror descent on each length
    phase1_deadline = deadline - 400  # reserve 400s for later phases
    time_per_length = max(30, (phase1_deadline - time.monotonic()) / len(lengths))

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()
        remaining = phase1_deadline - time.monotonic()
        time_alloc = min(time_per_length, remaining * 0.9)
        if time_alloc < 20:
            continue

        print(f"\n  Length n={n}, time_alloc={time_alloc:.0f}s", flush=True)

        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)

        best_for_n = None
        best_M_for_n = float('inf')

        time_per_init = time_alloc / max(1, len(inits))

        for ii, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            sub_dl = min(t0 + time_alloc, phase1_deadline)
            if time.monotonic() >= sub_dl:
                break

            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            # Run adaptive mirror descent
            a_opt, M_opt = mirror_descent_adaptive(a0, sub_dl, n_active_start=3, n_active_max=20)

            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best M={best_M_for_n:.10f} J={J_val:.6f} "
                  f"(elapsed={time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("After Phase 1 (Mirror Descent)")

    # Phase 2: LP polish on best
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP best-response polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 200:
        t_lp_deadline = min(deadline - 200, time.monotonic() + 180)
        a_lp, M_lp = lp_mirror_descent(tracker.best, t_lp_deadline, n_active_start=15, n_active_max=80)
        tracker.update(a_lp)
        print(f"  After LP: M={M_lp:.10f}", flush=True)

        # Second LP pass with different params
        if tracker.best is not None and time.monotonic() < deadline - 150:
            t_lp2 = min(deadline - 150, time.monotonic() + 80)
            a_lp2, M_lp2 = lp_mirror_descent(tracker.best, t_lp2, n_active_start=30, n_active_max=60)
            tracker.update(a_lp2)
            print(f"  After LP2: M={M_lp2:.10f}", flush=True)

    tracker.report("After Phase 2 (LP)")

    # Phase 3: FISTA with high tau for fine-tuning
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 100:
        for tau in [100, 300, 800]:
            if time.monotonic() > deadline - 80:
                break
            t_f = min(deadline - 80, time.monotonic() + 40)
            a_f, M_f = fista_smoothed(tracker.best, tau, t_f, lr=0.3)
            tracker.update(a_f)
            print(f"  FISTA tau={tau}: M={M_f:.10f}", flush=True)

    tracker.report("After Phase 3 (FISTA)")

    # Phase 4: Coordinate descent polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        t_cd = min(deadline - 20, time.monotonic() + 60)
        a_cd, M_cd = coordinate_descent(tracker.best, t_cd, seed=12345)
        tracker.update(a_cd)
        print(f"  After CD: M={M_cd:.10f}", flush=True)

        # Final LP pass
        if tracker.best is not None and time.monotonic() < deadline - 10:
            t_final = deadline - 5
            a_fin, M_fin = lp_mirror_descent(tracker.best, t_final, n_active_start=25, n_active_max=50)
            tracker.update(a_fin)
            print(f"  Final LP: M={M_fin:.10f}", flush=True)

    tracker.report("After Phase 4 (CD + Final LP)")

    # Phase 5: Try resampling to different lengths
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 20:
        n_cur = tracker.best.size
        for mult in [1.25, 0.8, 1.5]:
            if time.monotonic() > deadline - 15:
                break
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 2048)
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 10, time.monotonic() + 25)
            a_opt, M_opt = mirror_descent_adaptive(a_ext, sub_dl, n_active_start=5, n_active_max=15)
            # Quick LP polish
            if time.monotonic() < deadline - 10:
                a_opt, M_opt = lp_mirror_descent(a_opt, min(deadline - 5, time.monotonic() + 10),
                                                  n_active_start=20, n_active_max=40)
            tracker.update(a_opt)
            print(f"  n={new_n}: M={M_opt:.10f} J={compute_J(a_opt):.6f}", flush=True)

    tracker.report("After Phase 5 (Length exploration)")

    # Final validation
    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
