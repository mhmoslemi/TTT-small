"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """FFT-based linear cross-convolution of a with b."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def solve_lp_cutting_plane(a, M, max_iters=50, timeout=10.0):
    """Solve LP best response using cutting-plane method.
    
    Iteratively adds violated constraints until LP solution satisfies all.
    Returns (g, converged) where g is normalized best response.
    """
    n = len(a)
    t0 = time.monotonic()
    
    # Start with top constraints
    b = fast_conv(a)
    top_k = min(20, 2 * n - 1)
    active = set(np.argsort(b)[::-1][:top_k].tolist())
    
    prev_g = None
    for it in range(max_iters):
        if time.monotonic() - t0 > timeout:
            break
        
        # Build LP matrix
        rows = []
        cols = []
        vals = []
        active_list = sorted(active)
        for idx, k in enumerate(active_list):
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    rows.append(idx)
                    cols.append(j)
                    vals.append(a[m])
        
        if not rows:
            return None, False
        
        A = csr_matrix((vals, (rows, cols)), shape=(len(active_list), n))
        c = -np.ones(n)
        b_ub = np.full(len(active_list), M)
        bounds = [(0.0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': timeout - (time.monotonic() - t0), 
                                 'presolve': True})
            if not r.success or r.x is None:
                return prev_g, (prev_g is not None)
        except Exception:
            return prev_g, (prev_g is not None)
        
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            return prev_g, (prev_g is not None)
        
        g_norm = g / g_sum
        prev_g = g_norm.copy()
        
        # Check for violated constraints
        cross = fast_cross_conv(a, g)  # conv(a, g) has length 2n-1
        violations = np.where(cross > M + 1e-10 * M)[0]
        
        if len(violations) == 0:
            return g_norm, True
        else:
            # Add violated constraints
            for v in violations:
                active.add(int(v))
    
    return prev_g, False


def run_lp_momentum(a_init, max_iters, deadline, n_active_start=15, momentum=0.7):
    """LP mirror descent with momentum and cutting-plane LP.
    
    Uses momentum to accelerate convergence.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Momentum state
    v = np.zeros(n)
    beta = momentum
    
    no_improve = 0
    timeout_lp = 8.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Solve LP with cutting plane

        if g is None:
            no_improve += 2
            if no_improve > 10:
                break
            continue
        
        # Momentum update
        v = beta * v + (1 - beta) * (g - a)
        
        # Try multiple step sizes with momentum
        improved = False
        base_t = 0.5 / (1.0 + 0.1 * it)

            a_trial = np.maximum(a_trial, 0.0)

            if s_t < 0.01:

            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try without momentum (direct LP direction)
            for t in [0.4, 0.2, 0.1, 0.05]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 0.01:
                    continue
                a_trial = a_trial / s_t

                if M_trial < best_M - 1e-14:
                    a = a_trial.copy()
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    break

    if M_final < best_M:

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Priority: coordinates with high values first (they contribute most to max)
        priority = a.copy()
        # Add randomness
        priority += rng_cd.uniform(0, 0.1 * a.max(), n)
        order = np.argsort(priority)[::-1]
        
        b = fast_conv(a)
        sweep_improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            x_cur = a[i]
            if x_cur < 1e-10:
                continue
            
            # Affected k range
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Precompute for affected k
            c_vals = []
            slopes = []
            has_quad = False
            quad_pos = -1
            
            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    c = b[k] - x_cur * x_cur
                    c_vals.append(c)
                    slopes.append(0.0)
                    has_quad = True
                    quad_pos = len(c_vals) - 1
                else:
                    j = k - i
                    if 0 <= j < n:
                        c = b[k] - 2.0 * x_cur * a[j]
                        c_vals.append(c)
                        slopes.append(2.0 * a[j])
                    else:
                        c_vals.append(b[k])
                        slopes.append(0.0)

            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0

            s_arr = np.array(slopes)
            
            def obj_at(x_test, c_arr=c_arr, s_arr=s_arr, base_max=base_max, 
                      has_quad=has_quad, quad_pos=quad_pos):
                vals = c_arr + s_arr * x_test
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if has_quad:
                    qv = c_arr[quad_pos] + x_test * x_test
                    if qv > m:
                        m = qv
                return m
            
            # Golden section search
            lo = 0.0
            hi = max(2.0 * x_cur, 0.5)
            for _ in range(5):
                if obj_at(hi) > obj_at(hi * 0.5) and hi < 5.0:
                    hi *= 2.0
                else:
                    break

            a_pt = hi - gr * (hi - lo)

            fa = obj_at(a_pt)
            fb = obj_at(b_pt)
            
            for _ in range(35):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_at(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)
            
            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - x_cur) > 1e-10:
                a[i] = x_opt
                # Update b
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - x_cur)
                sweep_improved = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-11):

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 3
        if no_improve > 15:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # From parent
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.02, n), 1e-5)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.08, n), 1e-5)
        inits.append(p_n2 / p_n2.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine (good autocorrelation properties)
    rc = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    rc = np.maximum(rc, 0.01)
    inits.append(rc / rc.sum())
    
    # Triangular
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())
    
    # Hann window
    hann = 0.5 * (1 + np.cos(2 * np.pi * t))
    hann = np.maximum(hann, 0.01)
    inits.append(hann / hann.sum())
    
    # Blackman window
    blackman = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    blackman = np.maximum(blackman, 0.01)
    inits.append(blackman / blackman.sum())
    
    # Gaussian
    gauss = np.exp(-6.0 * (t - 0.5) ** 2) + 0.03
    inits.append(gauss / gauss.sum())
    
    # Spectral init: iterative reduction of peak autocorrelation
    a_spec = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_spec)
        k_star = int(np.argmax(bs))
        # Subgradient direction
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        corr -= corr.mean()
        a_spec = a_spec - 0.05 * corr
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    
    # Multi-frequency spectral init
    a_spec2 = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_spec2)
        # Focus on top 3 peaks
        top3 = np.argsort(bs)[::-1][:3]
        corr = np.zeros(n)
        for k_star in top3:
            for i in range(n):
                j = int(k_star) - i
                if 0 <= j < n:
                    corr[i] += a_spec2[j]
        corr -= corr.mean()
        a_spec2 = a_spec2 - 0.03 * corr
        a_spec2 = np.maximum(a_spec2, 0.001)
        a_spec2 = a_spec2 / a_spec2.sum()
    inits.append(a_spec2.copy())
    
    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Cutting-plane LP best-response (iteratively adds violated constraints for exact best response)
2. Momentum-augmented LP mirror descent with adaptive step decay
3. Log-sum-exp smoothed subgradient with Nesterov acceleration
4. Priority-ordered coordinate descent with golden section 1D optimization
5. Multi-start, multi-length search with structured initializations
Key improvement over previous: cutting-plane LP gives the exact best response
direction, and momentum accelerates convergence toward the optimum.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """FFT-based linear cross-convolution of a with b. Result has length len(a)+len(b)-1."""
    na = len(a)
    nb = len(b)
    if na == 0 or nb == 0:
        return np.array([])
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def solve_lp_cutting_plane(a, M, max_iters=40, timeout=8.0):
    """Solve LP best response using cutting-plane method.
    
    Iteratively adds violated constraints until LP solution satisfies all.
    Returns (g_normalised, converged) or (None, False).
    """
    n = len(a)
    t0 = time.monotonic()
    
    # Start with top constraints by value
    b = fast_conv(a)
    top_k = min(15, 2 * n - 1)
    active = set(np.argsort(b)[::-1][:top_k].tolist())
    
    prev_g = None
    for it in range(max_iters):
        if time.monotonic() - t0 > timeout:
            break
        
        # Build LP constraint matrix
        rows = []
        cols = []
        vals = []
        active_list = sorted(active)
        for idx, k in enumerate(active_list):
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    rows.append(idx)
                    cols.append(j)
                    vals.append(a[m])
        
        if not rows:
            return prev_g, (prev_g is not None)
        
        A_mat = csr_matrix((vals, (rows, cols)), shape=(len(active_list), n))
        c_vec = -np.ones(n)
        b_ub = np.full(len(active_list), M)
        bounds = [(0.0, None)] * n
        remaining_time = timeout - (time.monotonic() - t0)
        if remaining_time < 0.5:
            break
        
        try:
            r = linprog(c_vec, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                        method='highs',
                        options={'time_limit': remaining_time, 'presolve': True})
            if not r.success or r.x is None:
                return prev_g, (prev_g is not None)
        except Exception:
            return prev_g, (prev_g is not None)
        
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            return prev_g, (prev_g is not None)
        
        g_norm = g / g_sum
        prev_g = g_norm.copy()
        
        # Check for violated constraints among all 2n-1
        cross = fast_cross_conv(a, g)
        # cross has length 2n-1, check if any exceed M
        tol = 1e-10 * max(M, 1.0)
        violations = np.where(cross > M + tol)[0]
        
        if len(violations) == 0:
            return g_norm, True
        else:
            for v in violations:
                active.add(int(v))
    
    return prev_g, False


def run_lp_momentum(a_init, max_iters, deadline, momentum=0.5, lp_timeout=6.0):
    """LP mirror descent with momentum and cutting-plane LP.
    
    Uses momentum on the update direction to accelerate convergence.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Momentum state
    v = np.zeros(n)
    beta = momentum
    
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Solve LP with cutting plane
        g, converged = solve_lp_cutting_plane(a, best_M, max_iters=30, timeout=lp_timeout)
        
        if g is None:
            no_improve += 2
            if no_improve > 15:
                break
            continue
        
        # Compute momentum direction
        direction = g - a
        v = beta * v + (1.0 - beta) * direction
        
        # Try step sizes with momentum direction
        improved = False
        base_t = 0.6 / (1.0 + 0.05 * it)
        step_candidates = [base_t, base_t * 0.5, base_t * 0.25, 0.1, 0.05, 0.02]
        
        for t in step_candidates:
            a_trial = a + t * v
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try direct LP direction (no momentum)
            for t in [0.5, 0.3, 0.15, 0.08, 0.04]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 0.01:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-14:
                    a = a_trial.copy()
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            no_improve = 0
            beta = max(0.3, beta * 0.99)
        else:
            no_improve += 1
            v *= 0.5
            if no_improve > 25:
                break
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def run_smooth_subgradient(a_init, max_iters, deadline, tau=100.0, lr=0.04):
    """Nesterov-accelerated subgradient with log-sum-exp smoothing."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute smooth gradient at y
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * y[j]
        grad = np.zeros(n)
        for i in range(n):
            j_max = min(n, len(p) - i)
            if j_max > 0:
                grad[i] = 2.0 * np.dot(p[i:i + j_max], y[:j_max])
        
        grad_tan = grad - grad.mean()
        
        # Adaptive step
        step = lr / (1.0 + 0.005 * it)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y = a_new + alpha * (a_new - a_prev)
        y = proj_simplex(y)
        
        a_prev = a
        a = a_new
        t_mom = t_new
        
        if it % 10 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with priority ordering and golden section 1D optimization."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Priority: high values first + noise
        priority = a.copy() + rng_cd.uniform(0, 0.05 * a.max(), n)
        order = np.argsort(priority)[::-1]
        
        b = fast_conv(a)
        sweep_improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            x_cur = a[i]
            if x_cur < 1e-12:
                continue
            
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            c_vals = []
            slopes = []
            has_quad = False
            quad_pos = -1
            
            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    c = b[k] - x_cur * x_cur
                    c_vals.append(c)
                    slopes.append(0.0)
                    has_quad = True
                    quad_pos = len(c_vals) - 1
                else:
                    j = k - i
                    if 0 <= j < n:
                        c = b[k] - 2.0 * x_cur * a[j]
                        c_vals.append(c)
                        slopes.append(2.0 * a[j])
                    else:
                        c_vals.append(b[k])
                        slopes.append(0.0)
            
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            
            c_arr = np.array(c_vals) if c_vals else np.array([0.0])
            s_arr = np.array(slopes) if slopes else np.array([0.0])
            
            # Golden section search for min of max(affected, base_max)
            lo = 0.0
            hi = max(2.5 * x_cur, 0.3)
            
            for _ in range(5):
                obj_hi = _cd_obj(c_arr, s_arr, base_max, hi, has_quad, quad_pos)
                obj_mid = _cd_obj(c_arr, s_arr, base_max, hi * 0.5, has_quad, quad_pos)
                if obj_hi > obj_mid and hi < 10.0:
                    hi *= 2.0
                else:
                    break
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = _cd_obj(c_arr, s_arr, base_max, a_pt, has_quad, quad_pos)
            fb = _cd_obj(c_arr, s_arr, base_max, b_pt, has_quad, quad_pos)
            
            for _ in range(35):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = _cd_obj(c_arr, s_arr, base_max, a_pt, has_quad, quad_pos)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = _cd_obj(c_arr, s_arr, base_max, b_pt, has_quad, quad_pos)
            
            x_opt = (lo + hi) / 2.0
            if _cd_obj(c_arr, s_arr, base_max, 0.0, has_quad, quad_pos) <= _cd_obj(c_arr, s_arr, base_max, x_opt, has_quad, quad_pos):
                x_opt = 0.0
            
            if abs(x_opt - x_cur) > 1e-10:
                old_val = a[i]
                a[i] = x_opt
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old_val)
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 3
        if no_improve > 15:
            break
    
    return best_a


def _cd_obj(c_arr, s_arr, base_max, x_test, has_quad, quad_pos):
    """Objective for coordinate descent: max of affine terms and base_max."""
    vals = c_arr + s_arr * x_test
    m = base_max
    if vals.size > 0:
        vm = vals.max()
        if vm > m:
            m = vm
    if has_quad:
        qv = c_arr[quad_pos] + x_test * x_test
        if qv > m:
            m = qv
    return m


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.02, n), 1e-5)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.06, n), 1e-5)
        inits.append(p_n2 / p_n2.sum())
    
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    rc = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    rc = np.maximum(rc, 0.01)
    inits.append(rc / rc.sum())
    
    # Triangular
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())
    
    # Hann
    hann = 0.5 * (1.0 + np.cos(2 * np.pi * t))
    hann = np.maximum(hann, 0.01)
    inits.append(hann / hann.sum())
    
    # Blackman
    blackman = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    blackman = np.maximum(blackman, 0.01)
    inits.append(blackman / blackman.sum())
    
    # Gaussian
    gauss = np.exp(-5.0 * (t - 0.5) ** 2) + 0.03
    inits.append(gauss / gauss.sum())
    
    # Spectral init: iterative peak reduction
    a_spec = np.ones(n) / n
    for _ in range(40):
        bs = fast_conv(a_spec)
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        corr -= corr.mean()
        a_spec = a_spec - 0.04 * corr
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    
    # Multi-peak spectral init
    a_spec2 = np.ones(n) / n
    for _ in range(40):
        bs = fast_conv(a_spec2)
        top_k = np.argsort(bs)[::-1][:3]
        corr = np.zeros(n)
        for k_star in top_k:
            for i in range(n):
                j = int(k_star) - i
                if 0 <= j < n:
                    corr[i] += a_spec2[j]
        corr -= corr.mean()
        a_spec2 = a_spec2 - 0.02 * corr
        a_spec2 = np.maximum(a_spec2, 0.001)
        a_spec2 = a_spec2 / a_spec2.sum()
    inits.append(a_spec2.copy())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_phase_combined(a0, n, deadline_phase, seed):
    """Combined optimization pipeline: subgrad -> LP momentum -> CD -> final LP."""
    best = a0.copy()
    best_M = fast_conv(a0).max()
    t_start = time.monotonic()
    phase_duration = max(1.0, deadline_phase - t_start)
    
    # Stage 1: Smooth subgradient (15% of time)
    t1 = t_start + 0.15 * phase_duration
    if time.monotonic() < t1:
        for tau in [80, 300]:
            if time.monotonic() > t1:
                break
            a_s = run_smooth_subgradient(best, 3000, t1, tau=tau, lr=0.03)
            M_s = fast_conv(a_s).max()
            if M_s < best_M:
                best_M = M_s
                best = a_s.copy()
        print(f"    After subgrad: M={best_M:.10f}", flush=True)
    
    # Stage 2: LP with momentum (50% of time)
    t2 = t_start + 0.65 * phase_duration
    if time.monotonic() < t2:
        est_per_iter = 4.0
        max_lp_iters = max(5, int((t2 - time.monotonic()) / est_per_iter))
        a_lp = run_lp_momentum(best, max_lp_iters, t2, momentum=0.5, lp_timeout=5.0)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
        print(f"    After LP+momentum: M={best_M:.10f}", flush=True)
    
    # Stage 3: Coordinate descent (20% of time)
    t3 = t_start + 0.85 * phase_duration
    if time.monotonic() < t3:
        max_sw = max(3, int((t3 - time.monotonic()) / max(0.5, n / 60.0)))
        a_cd = run_coordinate_descent(best, max_sw, t3, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
        print(f"    After CD: M={best_M:.10f}", flush=True)
    
    # Stage 4: Final LP polish (10% of time)
    if time.monotonic() < deadline_phase:
        est_per_iter = 4.0
        max_lp_iters = max(3, int((deadline_phase - time.monotonic()) / est_per_iter))
        a_lp2 = run_lp_momentum(best, max_lp_iters, deadline_phase, momentum=0.3, lp_timeout=4.0)
        M_lp2 = fast_conv(a_lp2).max()
        if M_lp2 < best_M:
            best_M = M_lp2
            best = a_lp2.copy()
        print(f"    After final LP: M={best_M:.10f}", flush=True)
    
    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search using cutting-plane LP with momentum,
    smooth subgradient, and coordinate descent across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)
    
    # === Phase 1: Multi-length search ===
    n_parent = parent.size
    lengths_to_try = [n_parent]
    for delta in [-64, 64, -128, 128, -256, 256, -512, 512]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl not in lengths_to_try:
            lengths_to_try.append(nl)
    if 768 not in lengths_to_try:
        lengths_to_try.append(768)
    
    lengths_to_try.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    n_lengths = min(len(lengths_to_try), 4)
    lengths_to_try = lengths_to_try[:n_lengths]
    
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length search on {lengths_to_try}", flush=True)
    print(f"{'='*60}", flush=True)
    
    reserve_s = 300
    time_per_length = max(60.0, (deadline - time.monotonic() - reserve_s) / n_lengths)
    
    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - reserve_s:
            break
        t0 = time.monotonic()
        remaining = deadline - reserve_s - time.monotonic()
        if remaining < 20:
            break
        
        time_alloc = min(time_per_length, remaining / max(1, n_lengths - li))
        print(f"\n  Length n={n}, time_alloc={time_alloc:.0f}s", flush=True)
        
        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        
        best_for_n = None
        best_M_for_n = float('inf')
        
        max_inits = min(len(inits), max(2, int(time_alloc / 50)))
        
        for ii in range(max_inits):
            if time.monotonic() > deadline - reserve_s:
                break
            sub_deadline = min(t0 + time_alloc, deadline - reserve_s)
            if time.monotonic() >= sub_deadline - 5:
                break
            
            a0 = np.maximum(inits[ii], 1e-6)
            a0 = a0 / a0.sum()
            
            a_opt = run_phase_combined(a0, n, sub_deadline, seed=seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()
            
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best M={best_M_for_n:.10f} J={J_val:.6f} "
                  f"(elapsed={time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_for_n)
    
    tracker.report("After Phase 1")
    
    # === Phase 2: Intensive cutting-plane LP polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive cutting-plane LP polish", flush=True)
    print(f"{'='*60}", flush=True)

        est_iter = 4.0

        a_lp = run_lp_momentum(tracker.best, max_iters, t_lp_deadline,
                               momentum=0.5, lp_timeout=5.0)
        tracker.update(a_lp)
        
        # Second pass with lower momentum for fine-tuning
        if time.monotonic() < deadline - 30:
            a_lp2 = run_lp_momentum(a_lp, 30, min(deadline - 20, time.monotonic() + 60),
                                    momentum=0.3, lp_timeout=4.0)
            tracker.update(a_lp2)

    print(f"\n{'='*60}", flush=True)

        a_cd = run_coordinate_descent(tracker.best, max_sweeps, deadline - 15, seed=99999)
        tracker.update(a_cd)
    
    tracker.report("After Phase 3")
    
    # === Phase 4: Final refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final refinement", flush=True)

            tracker.update(a_fin)

    # === Final ===

    print(f"\n{'='*60}", flush=True)

          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
