            s = float(np.sum(g))

    h = h_full[:n]

def adam_smooth_step(a, active_set, beta, lr):
    """One Adam step using smoothed objective gradient."""
    n = len(a)
    s = float(np.sum(a))
    
    # Compute gradient
    conv = compute_conv(a)
    active_conv = np.array([conv[k] for k in active_set])
    cmax = float(np.max(active_conv))
    shifted = beta * (active_conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    # Gradient of the logsumexp
    # d(ls)/d(a_i) = (1/beta) * sum_k w_k * d(a*a)_k/d(a_i) / ls... 
    # Actually: ls = (1/beta) * log(sum_k exp(beta * (a*a)_k))
    # d(ls)/d(a_i) = (1/beta) * sum_k exp(beta*(a*a)_k) * beta * d(a*a)_k/d(a_i) / sum_j exp(beta*(a*a)_j)
    # = sum_k w_k * d(a*a)_k/d(a_i)
    
    # d(a*a)_k/d(a_i) ≈ 2 * a_{k-i} (for k-i valid)
    w_full = np.zeros(2 * n - 1)
    for idx, k in enumerate(active_set):
        w_full[k] = w[idx]
    
    size = 1
    while size < 4 * n:
        size <<= 1
    W = rfft(w_full, size)
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    A_pad = rfft(a_pad, size)
    h_full = irfft(np.conj(A_pad) * W, size)
    h = h_full[:n]
    
    # Objective: f = 2n * ls / s^2
    # d(f)/d(a_i) = (2n/s^2) * (d(ls)/d(a_i) - ls/s)
    # = (2n/s^2) * (2*h_i - ls/s)
    grad_f = (2.0 * n / (s * s)) * (2.0 * h - ls / s)
    
    # Project gradient to sum-zero subspace (to maintain normalization)
    grad_f = grad_f - np.mean(grad_f)
    
    # Simple gradient step with clip
    a_new = a - lr * grad_f
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a
    a_new = a_new / s_new
    return a_new

"""
Active-set LP iterations with progressive tightening for minimizing
2n*max(conv(a,a))/(sum(a))^2. Uses FFT-based convolution, HiGHS LP solver,
gradient refinement, multi-n exploration, and coordinate descent polish.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def get_top_k_indices(conv, K):
    """Get sorted list of top K indices of conv."""
    idx = np.argsort(conv)[::-1][:K]
    return sorted(idx.tolist())


def build_constraint_matrix(a, active_set):
    """Build LP constraint matrix: A[r, i] = a[k - i] where k = active_set[r]."""
    n = len(a)
    K = len(active_set)
    A = np.zeros((K, n))
    for r, k in enumerate(active_set):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            A[r, i] = a[k - i]
    return A


def solve_lp_active(a, M, active_set, time_limit=8.0):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_set, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    A = build_constraint_matrix(a, active_set)
    c = -np.ones(n)
    b_ub = np.full(len(active_set), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def gradient_step_a(a, active_set, beta, lr):
    """
    One gradient descent step on the smoothed objective (logsumexp over active set).
    Returns updated a (normalized, non-negative).
    """
    n = len(a)
    s = float(np.sum(a))
    conv = compute_conv(a)
    active_conv = np.array([conv[k] for k in active_set])

    cmax = float(np.max(active_conv))
    shifted = beta * (active_conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    # Compute h_i = sum_k w_k * a_{k-i} via FFT cross-correlation
    w_full = np.zeros(2 * n - 1)
    for idx, k in enumerate(active_set):
        w_full[k] = w[idx]

    size = 1
    while size < 4 * n:
        size <<= 1
    W = rfft(w_full, size)
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    A_pad = rfft(a_pad, size)
    h_full = irfft(np.conj(A_pad) * W, size)
    h = h_full[:n]

    # d(objective)/d(a_i) = (2n/s^2) * (2*h_i - ls/s)
    grad = (2.0 * n / (s * s)) * (2.0 * h - ls / s)
    # Project to sum-zero to maintain normalization

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a
    return a_new / s_new

    """

    if s < 1e-10:

    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    M = float(np.max(conv))

    active_set = get_top_k_indices(conv, min(K, 2 * n - 1))

    it = 0
    stall_count = 0
    prev_obj = best_obj

    while it < max_iters:
        if time.monotonic() - start > time_budget:
            break

        g = solve_lp_active(a, M, active_set, time_limit=10.0)

            break

def coordinate_descent(a, time_budget=30.0, seed=42):
    """Fine-tune using coordinate descent on the top peaks."""
    start = time.monotonic()
    n = len(a)

            rng.shuffle(candidates)

                    continue
                for frac in [0.01, 0.005, 0.02, -0.005, -0.01]:

                        best_obj = obj_try

def resample_sequence(a, new_n):

            a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.15, 40.0),
                                      seed=seed)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  CD: {best_score:.8f} *** IMPROVED")
            else:
                print(f"  CD: {obj_cd:.8f}")

            # LP after CD
            if time.monotonic() < deadline - 20:
                remaining = deadline - time.monotonic()
                a = best_seq.copy()
                s = np.sum(a)
                a = a / s
                a_lp = lp_iterate(a, 20, max_iters=2000,
                                  time_budget=min(remaining * 0.15, 30.0), verbose=False)
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  Post-CD LP: {best_score:.8f} *** IMPROVED")

    # === Phase 6: Final LP push ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final LP push")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        a = best_seq.copy()
        s = np.sum(a)
        a = a / s

        for K in [15, 10, 8]:
            if time.monotonic() > deadline - 10:
                break
            alloc = min(remaining * 0.2, 30.0)
            a_new = lp_iterate(a, K, max_iters=3000, time_budget=alloc, verbose=False)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()
                s = np.sum(a)
                a = a / s
                print(f"  Final K={K}: {best_score:.8f} *** IMPROVED")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
