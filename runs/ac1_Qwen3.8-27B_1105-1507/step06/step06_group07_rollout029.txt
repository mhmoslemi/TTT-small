```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a**2)

def subgradient_conv_peak(a, conv=None):
    """
    Compute subgradient of max_k (a*a)[k] w.r.t. a.
    Returns the subgradient vector of length n.
    """
    n = len(a)
    if conv is None:
        conv = compute_conv(a)
    max_val = np.max(conv)
    # Find all peaks (with tolerance)
    tol = 1e-10 * max_val
    peak_mask = conv >= max_val - tol
    w = peak_mask.astype(float)
    w /= w.sum()  # Average over all peaks
    
    # Gradient of (a*a)[k] w.r.t. a[i] = 2 * a[k-i] (if k-i in [0,n-1])
    # Sum over peaks: 2 * sum_k w[k] * a[k-i]
    # This is 2 * cross_correlation(w, a)[i]
    size = 1
    while size < 2*n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad = 2.0 * cc[:n]
    return grad, max_val, peak_mask
```

def compute_obj(a):

    sum_a = float(np.sum(a))

def subgradient_conv_peak(a, conv=None):

def adam_log_opt(a, max_iters=10000, time_budget=100.0, beta_start=10.0,
                 beta_end=300.0, lr_start=0.03, lr_decay=0.99998):

        return a

    x -= np.mean(x)

    v = np.zeros(n)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    start = time.monotonic()

        beta = beta_start * (beta_end / beta_start) ** progress

        x_centered = x - np.max(x)
        a_curr = np.exp(x_centered)
        a_curr = a_curr / np.sum(a_curr)

        cmax = float(np.max(conv))
        shifted = beta * (conv - cmax)
        shifted = np.clip(shifted, -500, 500)

        e_sum = float(np.sum(e))

        grad_a = (2.0 * n) * (grad_ls - ls)
        grad_x = grad_a * a_curr

        m_hat = m / (1 - beta1 ** t)

        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 900:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

    a_norm = a / s

    conv = compute_conv(a_norm)
    M = float(np.max(conv))

    g = solve_lp_max_sum(a_norm, M, time_limit=min(time_budget * 0.4, 12))
    if g is None:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)

    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        if time.monotonic() - start > time_budget:
            break
        a_try = (1 - t) * a_norm + t * g
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def cosine_basis_seq(n, n_basis, coeffs):
    """Generate sequence from cosine basis with given coefficients."""
    t = np.arange(n) / max(n - 1, 1)
    a = np.zeros(n)
    for j in range(n_basis):
        a += coeffs[j] * np.cos(np.pi * j * t)
    return np.maximum(a, 0)


def cma_es_cosine(n, time_budget=120.0, n_basis=16, seed=42):
    """
    CMA-ES optimization of cosine-basis coefficients.
    Minimize the objective over the low-dimensional coefficient space.
    """
    rng = np.random.default_rng(seed)

    # Initialize coefficients: try a "dome" shape
    best_coeffs = np.zeros(n_basis)
    best_coeffs[0] = 1.0  # Constant term
    for j in range(1, n_basis):
        best_coeffs[j] = -0.1 / j  # Slight negative higher harmonics (dome)

    best_obj = np.inf
    a_init = cosine_basis_seq(n, n_basis, best_coeffs)
    if np.sum(a_init) > 1e-10:
        best_obj = compute_obj(a_init)

    # Simple CMA-ES implementation
    mean = best_coeffs.copy()
    sigma = 0.3
    pop_size = max(4 + int(3 * np.log(n_basis)), 8)
    # Covariance matrix (diagonal initially)
    C = np.eye(n_basis)
    # Evolution paths
    p_sigma = np.zeros(n_basis)
    p_C = np.zeros(n_basis)
    c_sigma = 1.0 / (pop_size + n_basis)
    d_sigma = 1 + 2 * max(0, (n_basis - 1) / (pop_size + n_basis) - 1) + c_sigma
    c_c = (4 + n_basis / pop_size) / (pop_size + 4 + 2 * n_basis / pop_size)

    # Cholesky of C
    L = np.linalg.cholesky(C)
    eigvals, eigvecs = np.linalg.eigh(C)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        # Generate population
        population = []
        for i in range(pop_size):
            z = rng.standard_normal(n_basis)
            x = mean + sigma * (L @ z)
            population.append(x)

        # Evaluate
        fitness = []
        for x in population:
            a = cosine_basis_seq(n, n_basis, x)
            s = np.sum(a)
            if s < 1e-10:
                fitness.append(np.inf)
            else:
                fitness.append(compute_obj(a))

        # Sort
        indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in indices]
        sorted_fit = [fitness[i] for i in indices]

        # Update best
        if sorted_fit[0] < best_obj:
            best_obj = sorted_fit[0]
            best_coeffs = sorted_pop[0].copy()

        # Recombination
        weights = np.exp(-np.arange(pop_size) / (pop_size / 2.0))
        weights /= weights.sum()
        mu_eff = 1.0 / np.sum(weights**2)
        new_mean = np.zeros(n_basis)
        for i in range(pop_size):
            new_mean += weights[i] * sorted_pop[i]
        mean = new_mean

        # Update covariance
        diff = np.zeros((n_basis, pop_size))
        for i in range(pop_size):
            diff[:, i] = (sorted_pop[i] - mean) / sigma
        C = (1 - c_c) * C + c_c * (diff @ (weights * diff.T) + (2 * c_sigma / c_c) * (p_sigma @ (p_sigma.T)))
        # Ensure PSD
        C = (C + C.T) / 2
        eigvals, eigvecs = np.linalg.eigh(C)
        eigvals = np.maximum(eigvals, 1e-20)
        C = (eigvecs * eigvals) @ eigvecs.T
        L = np.linalg.cholesky(C + 1e-10 * np.eye(n_basis))

        # Update sigma
        p_sigma = (1 - c_sigma) * p_sigma + c_sigma * np.sqrt(mu_eff / (pop_size)) * (diff.T @ np.diag(weights))[:, 0]
        sigma *= np.exp(c_sigma / d_sigma * (np.linalg.norm(p_sigma) / np.sqrt(mu_eff) - 1))
        sigma = np.clip(sigma, 1e-6, 10.0)

        it += 1

    # Final evaluation
    a_best = cosine_basis_seq(n, n_basis, best_coeffs)
    s = np.sum(a_best)
    if s > 1e-10:
        a_best = a_best / s
    return a_best, best_obj


def simulated_annealing_conv(a, time_budget=200.0, seed=42):
    """
    Simulated annealing with convolution-aware moves.
    Moves: (1) swap mass between two positions, (2) smooth a local region,
    (3) move mass away from peak contributors.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_f = compute_obj(a)
    start = time.monotonic()

    T = 0.001
    cooling = 0.9997
    min_T = 1e-7
    it = 0

    cur_f = best_f
    while time.monotonic() - start < time_budget and T > min_T:
        conv = compute_conv(a)
        max_b = float(np.max(conv))
        peak_proxies = np.argsort(conv)[::-1][:3]

        # Choose move type
        move_type = rng.integers(0, 3)

        if move_type == 0:
            # Swap mass between two random positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = rng.uniform(0.001, 0.02) * max(a[i], a[j], 1e-10)
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            if a_try[i] < 0 or a_try[j] < 0:
                continue
            a_try = a_try / np.sum(a_try)
            delta_f = compute_obj(a_try) - cur_f

        elif move_type == 1:
            # Smooth a local region (diffusion step)
            i = rng.integers(1, n - 1)
            radius = rng.integers(1, 4)
            a_try = a.copy()
            avg = np.mean(a_try[max(0, i - radius):i + radius + 1])
            a_try[max(0, i - radius):i + radius + 1] = avg
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            delta_f = compute_obj(a_try) - cur_f

        else:
            # Move mass away from peak contributor
            p = peak_proxies[rng.integers(0, len(peak_proxies))]
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            if i_max <= i_min:
                i_min, i_max = i_max, i_min
            i = rng.integers(i_min, i_max + 1)
            # Move to a random position far from the peak
            j = rng.integers(0, n)
            if abs(i - j) < n // 4:
                j = (j + n // 2) % n
            delta = rng.uniform(0.001, 0.01) * a[i]
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            if a_try[i] < 0:
                continue
            a_try = a_try / np.sum(a_try)
            delta_f = compute_obj(a_try) - cur_f

        # Accept/reject
        if delta_f < 0 or rng.random() < np.exp(-delta_f / T):
            a = a_try
            cur_f += delta_f
            if cur_f < best_f:
                best_f = cur_f
                best_a = a.copy()

        T *= cooling
        it += 1

        if it % 500 == 0 and it % 5000 == 0:
            print(f"    SA: T={T:.6f}, best={best_f:.8f}, it={it}")

    return best_a


def coordinate_refine(a, time_budget=30.0, seed=0):
    """Fine-tune by adjusting individual coordinates."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    conv = compute_conv(best_a)
    peaks = np.argsort(conv)[::-1][:min(5, len(conv))]

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        p = peaks[rng.integers(0, len(peaks))]
        i_min = max(0, p - (n - 1))
        i_max = min(p, n - 1)
        if i_max <= i_min:
            i_min, i_max = i_max, i_min
        i = rng.integers(i_min, i_max + 1)
        j = rng.integers(0, n)
        if i == j:
            continue

        for delta_frac in [0.003, 0.007, 0.015, -0.003, -0.007, -0.015]:
            delta = delta_frac * max(best_a[i], best_a[j], 1e-10)
            a_try = best_a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            if a_try[i] < 0 or a_try[j] < 0:
                continue
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                conv = compute_conv(a_try)
                peaks = np.argsort(conv)[::-1][:min(5, len(conv))]
                improved = True
                break

        if not improved and it > 100:
            # Try random pair
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = rng.uniform(-0.01, 0.01) * max(best_a[i], best_a[j])
                a_try = best_a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] >= 0 and a_try[j] >= 0:
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining multiple strategies:
    1. CMA-ES on cosine basis (parametric optimization)
    2. Polyak subgradient descent with averaging
    3. LP refinement
    4. Multi-n exploration with Adam
    5. Convolution-aware simulated annealing
    6. Coordinate descent polish
    7. Final high-beta Adam
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, budget={budget_s}s")

    # === Phase 1: CMA-ES on cosine basis (new) ===
    print(f"\n{'='*50}")
    print(f"Phase 1: CMA-ES cosine basis optimization")
    print(f"{'='*50}")

    n_cma = n_parent
    if time.monotonic() < deadline - 300:
        # Try different n values for CMA-ES
        for n_try in [n_cma - 50, n_cma, n_cma + 50, n_cma + 100]:
            if n_try < 100 or n_try > 1500:
                continue
            if time.monotonic() > deadline - 300:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.08, 80.0)

            a_cma, obj_cma = cma_es_cosine(n_try, time_budget=alloc,
                                            n_basis=16, seed=seed)
            if obj_cma < best_score:
                best_score = obj_cma
                best_seq = a_cma.copy()
                n_parent = n_try
                print(f"  CMA-ES n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  CMA-ES n={n_try}: {obj_cma:.8f}")

    # === Phase 2: Polyak subgradient descent (new) ===
    print(f"\n{'='*50}")
    print(f"Phase 2: Polyak subgradient descent")
    print(f"{'='*50}")

    if best_seq is not None and time.monotonic() < deadline - 200:
        a = best_seq.copy()
        s = np.sum(a)
        a = a / s
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 150.0)

        a_sg = polyak_subgradient(a, time_budget=alloc, f_lb=1.40)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgradient: {best_score:.8f} *** UPDATE")
        else:
            print(f"  Subgradient: {obj_sg:.8f} (no imp)")

    # === Phase 3: LP refinement ===
    print(f"\n{'='*50}")
    print(f"Phase 3: LP refinement")
    print(f"{'='*50}")

    if best_seq is not None and len(best_seq) <= 900 and time.monotonic() < deadline - 150:
        for r_idx in range(4):
            if time.monotonic() > deadline - 150:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.12, 35.0)
            a_lp = lp_refine(best_seq, time_budget=alloc)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  LP {r_idx}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  LP {r_idx}: {obj:.8f} (no imp)")

            # Follow up with Adam after LP
            if time.monotonic() < deadline - 100:
                a_after = adam_log_opt(a_lp, max_iters=8000,
                                       time_budget=min((deadline - time.monotonic()) * 0.08, 20.0),
                                       beta_start=50.0, beta_end=400.0,
                                       lr_start=0.02, lr_decay=0.99999)
                obj_after = compute_obj(a_after)
                if obj_after < best_score:
                    best_score = obj_after
                    best_seq = a_after.copy()
                    print(f"  LP+Adam {r_idx}: {best_score:.8f} *** UPDATE")

    # === Phase 4: Multi-n exploration with Adam ===
    print(f"\n{'='*50}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*50}")

    if best_seq is not None and time.monotonic() < deadline - 150:
        n_best = len(best_seq)
        n_options = sorted(set([
            max(100, n_best - 150), max(100, n_best - 75), n_best,
            n_best + 75, n_best + 150, n_best + 300
        ]))
        n_options = [n for n in n_options if 200 <= n <= 1500 and n != n_best] + [n_best]

        for n_try in n_options:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if n_try == len(best_seq):
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s

            alloc = min(remaining * 0.12, 45.0)
            a_opt = adam_log_opt(a_try, max_iters=12000, time_budget=alloc,
                                 beta_start=10.0, beta_end=300.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 5: Convolution-aware simulated annealing (new) ===
    print(f"\n{'='*50}")
    print(f"Phase 5: Simulated annealing (conv-aware)")
    print(f"{'='*50}")

    if best_seq is not None and time.monotonic() < deadline - 100:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 120.0)
        a_sa = simulated_annealing_conv(best_seq, time_budget=alloc, seed=seed)
        obj_sa = compute_obj(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA: {best_score:.8f} *** UPDATE")
        else:
            print(f"  SA: {obj_sa:.8f} (no imp)")

        # Also try SA from a perturbed version
        if time.monotonic() < deadline - 60:
            rng = np.random.default_rng(seed + 1)
            a_pert = best_seq * np.exp(rng.normal(0, 0.03, len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / np.sum(a_pert)
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.1, 60.0)
            a_sa2 = simulated_annealing_conv(a_pert, time_budget=alloc2, seed=seed + 1)
            obj_sa2 = compute_obj(a_sa2)
            if obj_sa2 < best_score:
                best_score = obj_sa2
                best_seq = a_sa2.copy()
                print(f"  SA perturbed: {best_score:.8f} *** UPDATE")
            else:
                print(f"  SA perturbed: {obj_sa2:.8f} (no imp)")

    # === Phase 6: Coordinate descent + final Adam polish ===
    print(f"\n{'='*50}")
    print(f"Phase 6: Coordinate descent + final polish")
    print(f"{'='*50}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        # Coordinate descent
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            a_cd = coordinate_refine(best_seq, time_budget=min(remaining * 0.15, 30.0),
                                     seed=seed)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Coord descent: {best_score:.8f} *** UPDATE")
            else:
                print(f"  Coord descent: {obj_cd:.8f} (no imp)")

        # Final high-beta Adam
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            a_final = adam_log_opt(best_seq, max_iters=10000,
                                   time_budget=min(remaining * 0.4, 25.0),
                                   beta_start=200.0, beta_end=1000.0,
                                   lr_start=0.005, lr_decay=0.99999)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final Adam: {best_score:.8f} *** UPDATE")

        # One more subgradient pass
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            a_sg2 = polyak_subgradient(best_seq, time_budget=min(remaining * 0.3, 15.0),
                                       f_lb=best_score - 0.05)
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                print(f"  Final subgradient: {best_score:.8f} *** UPDATE")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
