```python
def project_simplex(x):
    """Project x onto {a >= 0, sum(a) = 1}."""
    n = len(x)
    if x.sum() <= 0:
        return np.ones(n) / n
    # Find tau such that sum(max(x[i] - tau, 0)) = 1
    # Sort x in descending order
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(n, 0, -1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(x - theta, 0)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient descent with
softmax peak weighting, polynomial square root iteration, and coordinate descent.
Explores multiple lengths and uses adaptive step sizes for efficient convergence.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0] ** 2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def obj_val(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def proj_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(n, 0, -1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(x - theta, 0)


def poly_sqrt(c, n):
    size = 1
    while size < 4 * n:
        size <<= 1
    C = np.fft.fft(c, size)
    A_fft = np.sqrt(C)
    if A_fft[0].real < 0:
        A_fft[0] = -A_fft[0]
        for k in range(1, size // 2 + 1):
            if k < size - k:
                A_fft[size - k] = -A_fft[size - k]
            else:
                A_fft[k] = -A_fft[k]
    A = np.fft.ifft(A_fft).real[:n]
    return A


def compute_grad(a, w, N_fft):
    fa = np.fft.fft(a, N_fft)
    fw = np.fft.fft(w, N_fft)
    corr = np.fft.ifft(np.conj(fa) * fw).real
    return 2.0 * corr[:len(a)]


def optimize_grad(a, time_budget, max_iters=500000):
    n = len(a)
    a = proj_simplex(a.copy())
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    alpha = 0.5
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        tau = 0.01 * cmax + 1e-12
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -50.0, 0.0)
        w = np.exp(shifted)
        ws = float(w.sum())
        if ws < 1e-300:
            k_star = int(np.argmax(conv))
            w = np.zeros(len(conv))
            w[k_star] = 1.0
        else:
            w /= ws

        grad = compute_grad(a, w, N_fft)
        grad = grad - float(np.sum(grad))

        improved = False
        for ss_mult in [1.0, 0.5, 0.25, 0.1, 2.0]:
            ss = alpha * ss_mult
            a_try = proj_simplex(a - ss * grad)
            obj_try = obj_val(a_try)
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                stall = 0
                improved = True
                alpha = min(alpha * 1.03, 5.0)
                break
        if not improved:
            alpha *= 0.85
            stall += 1
            if stall > 500:
                alpha = 0.1
                stall = 0

    return best_a, best_obj


def optimize_psr(a, time_budget, max_iters=500):
    n = len(a)
    a = proj_simplex(a.copy())
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        total = float(np.sum(conv))

        improved_this_iter = False
        for q in [0.90, 0.95, 0.98, 0.99, 0.995, 0.999]:
            threshold = float(np.quantile(conv, q))

            c_shaved = np.minimum(conv, threshold)
            c_sum = float(np.sum(c_shaved))
            if c_sum < 1e-15:
                continue
            c_shaved = c_shaved * (total / c_sum)

            a_new = poly_sqrt(c_shaved, n)
            a_new[np.abs(a_new) < 1e-15] = 0
            a_new = proj_simplex(a_new)

def resample_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    if n_old == 1:
        return np.full(n_new, a[0])
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, a)

    noise = rng.normal(0.0, scale, len(a))
```
