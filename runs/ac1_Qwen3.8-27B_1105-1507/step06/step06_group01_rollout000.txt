def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def build_conv_matrix(a, active_indices):
    """Build sparse matrix A where A[k,j] = a[k-j] for k in active_indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A

def lp_best_response(a, M, active_indices):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a, active_indices)
    if A.nnz == 0:
        return None, None
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
                # Get dual variables for constraint importance
                duals = None
                if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                    duals = r.ineqlin.marginals
                return g, duals
    except Exception:
        pass
    return None, None

def golden_section_minimize_1d(f, lo, hi, max_iter=60):
    """Golden section search for minimum of 1D function."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = hi - gr * (hi - lo)
    b1 = lo + gr * (hi - lo)
    fa = f(a1)
    fb = f(b1)
    for _ in range(max_iter):
        if fa < fb:
            hi = b1
            b1 = a1
            fb = fa
            a1 = hi - gr * (hi - lo)
            fa = f(a1)
        else:
            lo = a1
            a1 = b1
            fa = fb
            b1 = lo + gr * (hi - lo)
            fb = f(b1)
    return (lo + hi) / 2.0, f((lo + hi) / 2.0)

        t_opt, f_opt = golden_section_minimize_1d(obj, 0.0, 1.0, 50)

    return 0.0, f0

            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, 0.1)
    
    # Expand hi if needed
    for _ in range(5):
        if obj_at(hi) > obj_at(hi * 0.5) + 1e-12:
            hi *= 2.0
        else:
            break
    
    t_opt, _ = golden_section_minimize_1d(obj_at, lo, hi, 45)
    
    # Check boundary
    if obj_at(0.0) <= obj_at(t_opt) + 1e-12:
        t_opt = 0.0
    
    return max(0.0, t_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Run coordinate descent with random coordinate ordering."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = np.random.default_rng(sweep + seed).permutation(n)
        b = fast_conv(a)
        sweep_improved = False
        
        for i in order:
            x_opt = coordinate_descent_1d_fast(a, int(i), b, n)
            if abs(x_opt - a[i]) > 1e-12:
                delta = x_opt - a[i]
                # Update b
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - a[i] * a[i])
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * delta
                a[i] = x_opt
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 5
        if no_improve > 15:
            break
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def run_dual_subgradient(a_init, max_iters, deadline, step_init=0.1):
    """
    Nesterov's dual subgradient method for min max_k f_k(a) s.t. a >= 0, sum(a)=1.
    Maintains a probability distribution q over the constraints.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Dual: probability distribution over convolution indices
    q = np.ones(conv_len) / conv_len
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Update dual: tilt q towards high-convolution indices
        step = step_init / (1.0 + 0.01 * it)
        # w_i = b_i (the "loss" for constraint i)
        # q_i proportional to q_i * exp(step * b_i)
        log_w = step * b
        log_w -= log_w.max()
        q_new = q * np.exp(log_w)
        q = q_new / q_new.sum()
        
        # Primal update: a proportional to exp(-step * weighted_gradient)
        # weighted_gradient[i] = 2 * sum_k q[k] * a[k-i]
        # This is 2 * (q convolved with a)[i]
        # Use FFT for efficiency
        size = 1
        while size < conv_len:
            size <<= 1
        fq = np.fft.rfft(q, size)
        fa = np.fft.rfft(a, size)
        wgrad = np.fft.irfft(fq * fa, size)[:n] * 2.0
        
        # Projected gradient descent step
        wgrad_tan = wgrad - wgrad.mean()
        a_new = a - step * wgrad_tan
        a_new = proj_simplex(a_new)
        a = a_new
        
        M_new = fast_conv(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    
    return best_a


def run_combined_optimization(a0, n, deadline, seed=0):
    """Combined optimization pipeline."""
    best = np.maximum(a0.astype(np.float64), 0.0)
    s = best.sum()
    if s < 1e-10:
        best = np.ones(n) / n
    else:
        best = best / s
    
    best_M = fast_conv(best).max()
    t_start = time.monotonic()
    remaining = deadline - t_start
    
    # Stage 1: Cutting-plane LP (50% of time)
    t1 = t_start + 0.5 * remaining
    if time.monotonic() < t1:
        a_lp = run_cutting_plane_lp(best, 200, t1, n_active_init=30, n_active_max=300)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
        print(f"    After LP: M={best_M:.12f}", flush=True)
    
    # Stage 2: Dual subgradient (20% of time)
    t2 = t_start + 0.7 * remaining
    if time.monotonic() < t2:
        a_ds = run_dual_subgradient(best, 3000, t2, step_init=0.05)
        M_ds = fast_conv(a_ds).max()
        if M_ds < best_M:
            best_M = M_ds
            best = a_ds.copy()
        print(f"    After DualSG: M={best_M:.12f}", flush=True)
    
    # Stage 3: FISTA (20% of time)
    t3 = t_start + 0.9 * remaining
    if time.monotonic() < t3:
        for tau in [50, 100, 200]:
            if time.monotonic() > t3:
                break
            a_f = run_fista_with_backtracking(best, tau, 2000, 
                                               min(t3, time.monotonic() + 20))
            M_f = fast_conv(a_f).max()
            if M_f < best_M:
                best_M = M_f
                best = a_f.copy()
        print(f"    After FISTA: M={best_M:.12f}", flush=True)
    
    # Stage 4: Coordinate descent (10% of time)
    if time.monotonic() < deadline:
        a_cd = run_coordinate_descent(best, 30, deadline, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
        print(f"    After CD: M={best_M:.12f}", flush=True)
    
    return best


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
            return True
        return False

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)
