import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    max_shifted = shifted.max()
    exp_b = np.exp(shifted - max_shifted)
    sum_exp = exp_b.sum()
    soft_max = b_max + (max_shifted + np.log(sum_exp)) / tau
    p = exp_b / sum_exp
    
    # Gradient: grad[i] = 2 * sum_j p[i+j] * a[j] for j=0..n-1
    # = 2 * (conv(p, a_rev))[n-1+i]
    a_rev = a[::-1]
    size2 = 1
    while size2 < 3 * n:
        size2 <<= 1
    Fp = np.fft.rfft(p, size2)
    Fa = np.fft.rfft(a_rev, size2)
    conv_pa = np.fft.irfft(Fp * Fa, size2)
    grad = 2.0 * conv_pa[n-1:2*n-1]
    
    return soft_max, grad, b, b_max


def bb_step(prev_a, prev_grad, a, grad):
    """Barzilai-Borwein step size."""
    s = a - prev_a
    y = grad - prev_grad
    ss = s @ s
    yy = y @ y
    sy = s @ y
    if ss < 1e-15:
        return 0.01
    if sy > 1e-10:
        return ss / sy  # BB1
    elif -sy > 1e-10:
        return yy / (-sy)  # BB2
    else:
        return 0.01


def run_smooth_bb(a, deadline, max_iters=5000, tau_start=1.0, tau_end=500.0):
    """Smooth optimization with Barzilai-Borwein step size and adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    prev_a = a.copy()
    prev_grad = None
    no_imp = 0
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        
        soft_max, grad, b, b_max = smooth_max_grad(a, tau)
        
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200 and tau > 100:
                break
        
        # Determine step size
        if prev_grad is not None:
            lr = bb_step(prev_a, prev_grad, a, grad)
            lr = np.clip(lr, 1e-6, 1.0)
        else:
            lr = 0.1 / (tau + 1)
        
        # Gradient step with projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        prev_a = a.copy()
        prev_grad = grad.copy()
        a = a_new
    
    return best_a


def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                         lr_init=0.4, lr_end=0.003):
    """Nesterov accelerated smooth descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    no_imp = 0
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        soft_max, grad, b, b_max = smooth_max_grad(y, tau)
        
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 300 and tau > 50:
                break
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a


def lp_best_response_topk(a, M, K=100):
    """LP best response using only top-K constraints."""
    n = len(a)
    b = fast_conv(a)
    # Get top-K indices
    if K < 2 * n - 1:
        top_k = np.argpartition(b, -K)[-K:]
    else:
        top_k = np.arange(2 * n - 1)
    
    # Build sparse matrix for only these constraints
    rows = []
    cols = []
    vals = []
    for idx in range(len(top_k)):
        k = top_k[idx]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_k), n))
    c = -np.ones(n)
    b_ub = np.full(len(top_k), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_best_response_full(a, M):
    """LP best response using all constraints."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=200, use_topk=True, K=100,
                  t_base=0.3, min_t=0.005):
    """LP mirror descent with top-K or full constraints."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
        
        if use_topk:
            g = lp_best_response_topk(a, M, K=K)
        else:
            g = lp_best_response_full(a, M)
        
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        if it < 5:
            t = t_base
        else:
            t = max(min_t, t_base / (1 + 0.03 * it))
        
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new
    
    return best_a


def local_perturbation_search(a, deadline, n_tries=500, scale=0.01):
    """Try random perturbations focused on peak reduction."""
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(12345)
    
    for it in range(n_tries):
        if time.monotonic() > deadline:
            break
        
        # Find peak position
        b = fast_conv(a)
        peak_idx = int(b.argmax())
        
        # Create perturbation focused on reducing the peak
        # The peak is at lag peak_idx, which involves a[j]*a[j+peak_idx] pairs
        # Perturb elements that contribute to the peak
        j_min = max(0, peak_idx - n + 1)
        j_max = min(n - 1, peak_idx)
        
        # Random perturbation
        if rng.random() < 0.5:
            # Global random perturbation
            pert = rng.normal(0, scale, n)
        else:
            # Focused perturbation on peak-contributing elements
            pert = np.zeros(n)
            pert[j_min:j_max+1] = rng.normal(0, scale * 2, j_max - j_min + 1)
            # Also perturb the "complementary" part
            comp_min = max(0, n - 1 - peak_idx)
            comp_max = min(n - 1, n - 1 - j_min)
            if comp_max >= comp_min:
                pert[comp_min:comp_max+1] += rng.normal(0, scale, comp_max - comp_min + 1)
        
        a_new = np.maximum(a + pert, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new /= s
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new.copy()
            # Decrease scale as we improve
            scale *= 0.98
    
    return best_a


def gs_flat_spectral(init, n_iters=300, beta=0.8):
    """Gerchberg-Saxton for flat autocorrelation."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        # Power law compression
        power = 0.5 + 0.3 * beta
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_alt
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.005))
    
    inits.append(np.ones(n))
    for alpha in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t**alpha + (1 - t)**alpha) / 2, 0.01))
    for f in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-3 * (t - 0.5)**2) + 0.1)
    inits.append(0.5 + 0.3 * np.sin(np.pi * t))
    # Raised cosine
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    # Random smooth
    k = max(3, n // 10)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_flat_spectral(a_rand, n_iters=80, beta=0.85))
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(gs_flat_spectral(p, n_iters=100, beta=0.9))
    return inits
