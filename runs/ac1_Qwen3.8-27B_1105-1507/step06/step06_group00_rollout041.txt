"""
FFT-based chase-peak optimization with periodic LP refinement for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses multiple shape initializations, adaptive
step sizes, mirror descent, and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def chase_peak_optimize(a, n_iter=10000, time_budget=60.0, lr_init=0.03,
                        decay=0.9999, verbose=False, start_time=None,
                        use_mirror=False, top_k=1):
    """
    Chase-the-peak subgradient descent with adaptive step size.
    Optionally uses mirror (multiplicative) descent.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    if start_time is None:
        start_time = time.monotonic()

    lr = lr_init
    prev_conv = None
    stall_count = 0

    for it in range(n_iter):
        if time.monotonic() - start_time > time_budget:
            break

        conv = compute_conv(a)
        cur_obj = 2.0 * n * float(np.max(conv))

        # Find top-k active lags
        if top_k > 1:
            top_idx = np.argsort(conv)[-top_k:]
        else:
            top_idx = [int(np.argmax(conv))]

        # Compute subgradient (average over top lags)
        g = np.zeros(n)
        for k_star in top_idx:
            # g[j] += 2 * a[k_star - j] for valid range
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            if lo <= hi:
                j_range = np.arange(lo, hi + 1)
                m_range = k_star - j_range
                g[j_range] += 2.0 * a[m_range]
        g /= len(top_idx)
        g -= g.mean()  # center to maintain sum

        if use_mirror:
            # Multiplicative update (mirror descent with entropy)
            a = a * np.exp(-lr * g)
            a = np.maximum(a, 1e-15)
        else:
            a = a - lr * g
            a = np.maximum(a, 0)

        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        obj = 2.0 * n * float(np.max(compute_conv(a)))
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 200:
                #小型反弹
                a = best_a * np.exp(np.random.default_rng(it).normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall_count = 0

        lr *= decay
        if lr < 1e-6:
            lr = 1e-6

        if verbose and it % 2000 == 0:
            print(f"      chase it{it}: obj={cur_obj:.8f}, best={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def lp_minpeak_step(a, use_highs=True):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    a = a / float(np.sum(a))

    # Build constraint matrix: (2n-1) x (n+1)
    # A_ub @ [g; M] <= 0 where A_ub[k, j] = a[k-j] (if valid), A_ub[k, n] = -1
    n_rows = 2 * n - 1
    A_ub = np.zeros((n_rows, n + 1))
    A_ub[:, n] = -1.0

    for j in range(n):
        # A_ub[k, j] = a[k-j] for k-j in [0, n-1]
        lo = max(0, j)
        hi = min(n_rows, j + n)
        if lo < hi:
            k_range = np.arange(lo, hi)
            A_ub[k_range, j] = a[k_range - j]

    b_ub = np.zeros(n_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    c = np.zeros(n + 1)
    c[n] = 1.0

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_maxsum_step(a, M_target, use_highs=True):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_target, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    a = a / float(np.sum(a))

    n_rows = 2 * n - 1
    A_ub = np.zeros((n_rows, n))
    for j in range(n):
        lo = max(0, j)
        hi = min(n_rows, j + n)
        if lo < hi:
            k_range = np.arange(lo, hi)
            A_ub[k_range, j] = a[k_range - j]

    b_ub = np.full(n_rows, M_target)
    c = -np.ones(n)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_polish(a, time_budget=30.0, start_time=None, verbose=False):
    """
    Alternating LP optimization: iteratively solve min-peak LP and blend.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    if start_time is None:
        start_time = time.monotonic()

    iter_count = 0
    while time.monotonic() - start_time < time_budget:
        iter_count += 1
        g_lp = lp_minpeak_step(a)
        if g_lp is None:
            # Fallback: max-sum LP at current peak
            conv_a = compute_conv(a)
            M_cur = float(np.max(conv_a))
            g_lp = lp_maxsum_step(a, M_cur)
            if g_lp is None:
                break

        # Line search
        improved = False
        for t in [0.3, 0.5, 0.7, 1.0, 0.2, 0.4, 0.6, 0.8]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Try smaller steps
            for t in [0.1, 0.15, 0.25]:
                a_try = (1 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, perturb
                break

        if verbose and iter_count % 5 == 0:
            print(f"      LP iter{iter_count}: obj={best_obj:.8f}")

    return best_a


def coordinate_descent_polish(a, time_budget=20.0, start_time=None):
    """
    Coordinate descent: pairwise mass transfer to reduce peak convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    if start_time is None:
        start_time = time.monotonic()

    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    for it in range(200):
        if time.monotonic() - start_time > time_budget:
            break

        improved = False
        # Try transferring mass from peak contributors to non-contributors
        for i in range(0, n, max(1, n // 200)):
            if time.monotonic() - start_time > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i and a[i] > 1e-10:
                for frac in [0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        if not improved:
            # Try moving mass away from peak contributors to edges
            for i in range(0, n, max(1, n // 100)):
                m = k_star - i
                if 0 <= m < n and a[i] > 0:
                    for target in [0, n - 1, i // 2, (i + n) // 2]:
                        if target == i or target == m or a[target] >= a[i]:
                            continue
                        delta = 0.05 * a[i]
                        a[i] -= delta
                        a[target] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-12:
                            best_obj = new_obj
                            best_a = a.copy()
                            conv = new_conv
                            k_star = int(np.argmax(conv))
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[target] -= delta
                    if improved:
                        break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        return a / a.sum()
    elif mode == 3:

        return a / a.sum()
