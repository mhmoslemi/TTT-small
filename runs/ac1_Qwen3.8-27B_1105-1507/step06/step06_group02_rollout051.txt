"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP mirror descent with Anderson acceleration (primary method)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_init(n, rng, n_iters=200, beta=0.85):
    """Gerchberg-Saxton flat spectral initialization."""
    a = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    a /= a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_power = mag ** 0.5
        mag_target = np.ones_like(mag) * mag.mean()
        mag_new = (1.0 - beta) * mag_power + beta * mag_target * (mag_power.max() / (mag_target.max() + 1e-30))
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_cand = np.maximum(a_new, 1e-8)
        sc = a_cand.sum()
        if sc < 1e-15:
            break
        a_cand /= sc
        M_c = fast_conv(a_cand).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_cand.copy()
        a = a_cand
    return best_a


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (A*g)(k) = (a*g)(k)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, constraint_fraction=1.0):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M, g >= 0. Optionally use subset of constraints."""
    n = len(a)
    b = fast_conv(a)
    if constraint_fraction < 1.0:
        n_constraints = int((2 * n - 1) * constraint_fraction)
        idx = np.argsort(b)[-n_constraints:]
    else:
        idx = np.arange(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for k in idx:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(idx), n))
    c = -np.ones(n)
    b_ub = np.full(len(idx), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 5.0})
        if r.success and r.x is not None:
            gx = np.maximum(r.x, 0.0)
            if gx.sum() > 1e-12:
                return gx
    except Exception:
        pass
    return None


def anderson_mix(history_x, history_g, m, clip_range=2.0):
    """Anderson mixing: find best convex combination of history_g."""
    if m < 2:
        return history_g[-1].copy()
    residuals = np.array([history_x[i] - history_g[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -clip_range, clip_range)
    s_a = alpha.sum()
    if abs(s_a) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s_a
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_g[i]
    return np.maximum(mixed, 0.0)


def run_lp_anderson(a_init, deadline, max_iters=5000, anderson_window=10,

    """Main LP mirror descent with Anderson acceleration."""
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = 2.0 * n * best_M
    no_imp = 0
    hist_x = []
    hist_g = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if it % 500 == 0 and it > 0:
            print(f"    iter {it}: J={best_J:.8f}, M={best_M:.8f}, t={t_initial * (t_decay**it):.4f}", flush=True)
        
        b = fast_conv(a)
        M = float(b.max())
        J_curr = 2.0 * n * M
        if J_curr < best_J * (1.0 - 1e-12):
            best_J = J_curr
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200 and it > 100:
                break
        
        cf = 1.0 if not use_reduced_constraints else constraint_frac
        g = lp_best_response(a, M, constraint_fraction=cf)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        g = np.maximum(g, 1e-8)
        g = g / g.sum()
        
        t = max(t_min, t_initial * (t_decay ** it))
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        hist_x.append(a.copy())
        hist_g.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_g.pop(0)

                M_new = fast_conv(a_new).max()

                    a = a_and
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def smooth_polish(a_init, deadline, max_iters=2000):
    """Quick smooth gradient descent using softmax approximation of max."""
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta_mom = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        b = fast_conv(y)
        b_max = float(b.max())
        tau = 5.0 + 50.0 * (it / max_iters)
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / (exp_b.sum() + 1e-30)
        grad = np.zeros(n)
        active = np.where(p > 1e-12)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * p[k] * seg
        lr = 0.3 * (0.98 ** it)
        a_new = np.maximum(y - lr * (grad + beta_mom * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_mom * beta_mom)) / 2.0
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new
        if b_max < best_M * (1.0 - 1e-10):
            best_M = b_max
            best_a = a.copy()
    return best_a

                                                t_initial=0.1, t_min=0.001, t_decay=0.9999)
            best.update(a_final)
            print(f"  Final Anderson: J={J_final:.8f}", flush=True)

    # === Phase 6: Random restarts if time remains ===
    if best.seq is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Random Restarts", flush=True)
        print(f"{'='*60}", flush=True)
        n_best = best.seq.size
        for restart in range(5):
            if time.monotonic() > deadline - 15:
                break
            # Perturb the best
            noise_scale = 0.05 / (restart + 1)
            a_pert = np.maximum(best.seq + rng.normal(0, noise_scale, n_best), 1e-6)
            a_pert /= a_pert.sum()
            # Quick smooth + LP
            a_pert = smooth_polish(a_pert, min(time.monotonic() + 20, deadline - 10), max_iters=500)
            best.update(a_pert)
            if time.monotonic() < deadline - 10:
                a_pert_lp, _ = run_lp_anderson(a_pert, deadline - 8,
                                                max_iters=2000, anderson_window=8,
                                                t_initial=0.2, t_min=0.003, t_decay=0.9998)
                best.update(a_pert_lp)

    best.report("FINAL")

    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    final_eval = evaluate_sequence(best.seq.tolist())
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL ANSWER: J={final_eval:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best.seq]
