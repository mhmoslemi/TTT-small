```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def topk_smooth_grad(a, beta, K=20):
    """Compute gradient using top-K focused smoothing."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    # Get top K indices
    if K >= len(conv):
        K = len(conv)
    top_k = np.argsort(conv)[::-1][:K]
    
    # Compute weighted sum focusing on top K
    # w_k = exp(beta * (conv[k] - max)) / sum(exp(beta * (conv[k] - max)))
    # but only over top K
    cmax = float(conv[top_k[0]])
    shifted = beta * (conv[top_k] - cmax)
    e = np.exp(shifted)
    w_top = e / e.sum()
    
    # Create full weight vector (sparse)
    w = np.zeros(len(conv))
    w[top_k] = w_top
    
    # Gradient: 2 * cross_correlation(w, a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    grad_ls = 2.0 * cc[:n]
    
    # Smoother value (for tracking)
    ls_val = cmax + np.log(np.sum(e)) / beta
    
    # Full gradient with normalization
    S = sum_a
    grad_obj = (2.0 * n / (S * S)) * (grad_ls - ls_val / S)
    
    return ls_val, grad_obj
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Strategy: Top-K peak-focused Adam in log-space, linearized min-max LP for
peak reduction, L-BFGS-B on smoothed objective, multi-n exploration,
diverse initializations, and coordinate descent polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def topk_smooth_grad(a, beta, K=20):
    """Compute smoothed objective and gradient using top-K peak focusing."""
    n = len(a)
    conv = compute_conv(a)
    S = float(np.sum(a))
    max_conv = float(np.max(conv))

    K = min(K, len(conv))
    top_idx = np.argpartition(conv, -K)[-K:]

    cmax = float(conv[top_idx].max())
    shifted = beta * (conv[top_idx] - cmax)
    e = np.exp(shifted)
    w_top = e / e.sum()

    w = np.zeros(len(conv))
    w[top_idx] = w_top

    # Cross-correlation: grad_ls[i] = 2 * sum_j w[j+i] * a[j]
    # = 2 * ifft(FFT(w) * conj(FFT(a)))[:n]
    fft_size = 1
    while fft_size < (2 * n - 1) + n - 1:
        fft_size <<= 1
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cc = irfft(W * np.conj(A), fft_size)
    grad_ls = 2.0 * cc[:n]

    ls_val = cmax + np.log(e.sum()) / beta
    grad_obj = (2.0 * n / (S * S)) * (grad_ls - ls_val / S)

    return ls_val, grad_obj, max_conv, conv


def full_smooth_grad(a, beta):
    """Compute smoothed objective and gradient using full log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    S = float(np.sum(a))
    max_conv = float(np.max(conv))

    cmax = max_conv
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum

    fft_size = 1
    while fft_size < (2 * n - 1) + n - 1:
        fft_size <<= 1
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cc = irfft(W * np.conj(A), fft_size)
    grad_ls = 2.0 * cc[:n]

    ls_val = cmax + np.log(e_sum) / beta
    grad_obj = (2.0 * n / (S * S)) * (grad_ls - ls_val / S)

    return ls_val, grad_obj, max_conv, conv


def adam_peak_opt(a, max_iters=20000, time_budget=100.0,
                  beta_start=10.0, beta_end=500.0,
                  lr_start=0.05, lr_decay=0.99995, K=20,
                  use_topk=True):
    """Adam optimization in log-space with peak-focused smoothing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        # Cosine annealing
        beta = beta_end - (beta_end - beta_start) * 0.5 * (1 + np.cos(np.pi * progress))

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)

        if use_topk:
            _, grad_a, _, _ = topk_smooth_grad(a, beta, K)
        else:
            _, grad_a, _, _ = full_smooth_grad(a, beta)

        grad_x = grad_a * a

        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m = b1 * m + (1 - b1) * grad_x
        v = b2 * v + (1 - b2) * (grad_x * grad_x)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 200 == 0:
            x -= np.mean(x)

        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()

    return best_a


def build_toeplitz_conv(a):
    """Build matrix T where (T @ d)[k] = (a*d)[k] for linear convolution."""
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        T[k_start:k_end, j] = a[:k_end - k_start]
    return T

def lbfgs_smooth_opt(a, beta, time_budget=60.0, K=20):

    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s

    x0 = np.log(a_norm)
    x0 -= np.mean(x0)

    start = time.monotonic()

    def obj_fn(x):
        if time.monotonic() - start > time_budget:
            return np.inf
        x_c = x - np.max(x)
        a_try = np.exp(x_c)
        a_try = a_try / np.sum(a_try)
        if K >= len(compute_conv(a_try)):
            val, _, _, _ = full_smooth_grad(a_try, beta)
        else:
            val, _, _, _ = topk_smooth_grad(a_try, beta, K)
        S = np.sum(a_try)
        return 2.0 * n * val / (S * S)

    def grad_fn(x):
        if time.monotonic() - start > time_budget:
            return np.zeros(n)
        x_c = x - np.max(x)
        a_try = np.exp(x_c)
        a_try = a_try / np.sum(a_try)
        if K >= len(compute_conv(a_try)):
            _, g, _, _ = full_smooth_grad(a_try, beta)
        else:
            _, g, _, _ = topk_smooth_grad(a_try, beta, K)
        return g * a_try

    try:

                       options={'maxiter': 500, 'maxfun': 2000,

        if res.success or res.fun < obj_fn(x0):
            x_opt = res.x
            x_c = x_opt - np.max(x_opt)
            a_opt = np.exp(x_c)
            a_opt = a_opt / np.sum(a_opt)
            return a_opt
    except Exception:
        pass
    return a_norm


def coordinate_refine(a, time_budget=20.0, seed=0):
    """Fine-tune by adjusting pairs of coordinates to reduce top peaks."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy().astype(np.float64)
    s = np.sum(best_a)
    if s < 1e-10:
        return a
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(seed)

    conv = compute_conv(best_a)
    top_peaks = np.argsort(conv)[::-1][:min(5, len(conv))]

    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for p in top_peaks[:3]:
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            if i_max <= i_min:
                continue
            # Pick two random indices contributing to this peak
            candidates = list(range(i_min, i_max + 1))
            if len(candidates) < 2:
                continue
            i, j = rng.choice(candidates, size=2, replace=False)
            j2 = p - j
            if j2 < 0 or j2 >= n:
                continue

            for delta_frac in [0.002, 0.005, 0.01, 0.02, -0.005, -0.01]:
                delta = delta_frac * max(best_a[i], best_a[j2])
                a_try = best_a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    conv = compute_conv(a_try)
                    top_peaks = np.argsort(conv)[::-1][:min(5, len(conv))]
                    improved = True
                    break

        if not improved and it > 100:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def build_plateau(n, frac=0.5):
    """Build a plateau sequence: constant in middle, linear ramps to edges."""
    n_plateau = max(1, int(round(frac * n)))
    peak_start = (n - n_plateau) // 2
    peak_end = peak_start + n_plateau
    candidate = np.zeros(n)
    candidate[peak_start:peak_end] = 1.0
    # Linear ramps
    left_len = peak_start
    right_len = n - peak_end
    if left_len > 0:
        candidate[:left_len] = np.linspace(0, 1, left_len)
    if right_len > 0:
        candidate[peak_end:] = np.linspace(1, 0, right_len)[::-1]
    s = np.sum(candidate)
    if s > 0:
        candidate /= s
    return candidate


def build_cosine(n):
    """Raised cosine window."""
    x = np.arange(n)
    candidate = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    candidate = np.maximum(candidate, 1e-10)
    s = np.sum(candidate)
    return candidate / s


def build_super_gaussian(n, p=6):
    """Super-Gaussian window."""
    center = (n - 1) / 2.0
    sigma = n / 4.0
    x = np.arange(n)
    candidate = np.exp(-np.abs(x - center) / sigma * 1.0) ** p
    candidate = np.maximum(candidate, 1e-15)
    s = np.sum(candidate)
    return candidate / s


def build_uniform(n):
    return np.ones(n) / n


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 0.01:
                parent_norm = parent / s
                obj = compute_obj(parent_norm)
                print(f"Loaded parent (n={parent.size}): obj={obj:.8f}")
                return parent_norm.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines:
    1. Diverse initializations (plateau, cosine, super-Gaussian, parent)
    2. Top-K peak-focused Adam with cosine beta annealing
    3. Linearized min-max LP for peak reduction
    4. L-BFGS-B on smoothed objective
    5. Multi-n exploration
    6. Coordinate descent and random restarts
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Load parent and generate diverse initializations ---
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_base = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: n={n_base}, score={best_score:.8f}")

    # Generate diverse initializations
    init_candidates = []
    if best_seq is not None:
        init_candidates.append(('parent', best_seq.copy()))

    inits_to_try = [
        ('plateau_0.4', build_plateau(n_base, 0.4)),
        ('plateau_0.5', build_plateau(n_base, 0.5)),
        ('plateau_0.6', build_plateau(n_base, 0.6)),
        ('cosine', build_cosine(n_base)),
        ('super_gauss_4', build_super_gaussian(n_base, 4)),
        ('super_gauss_6', build_super_gaussian(n_base, 6)),
        ('super_gauss_8', build_super_gaussian(n_base, 8)),
        ('uniform', build_uniform(n_base)),
    ]

    for name, init in inits_to_try:
        obj_init = compute_obj(init)
        init_candidates.append((name, init))
        if obj_init < best_score:
            best_score = obj_init
            best_seq = init.copy()
            print(f"  Better init: {name} (n={n_base}) -> {best_score:.8f}")
        else:
            print(f"  Init {name} (n={n_base}): {obj_init:.8f}")

    # Sort by objective
    init_candidates.sort(key=lambda x: compute_obj(x[1]))
    print(f"\nBest init score: {best_score:.8f}")

    # === Phase 1: Top-K Adam optimization on best initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Top-K Adam optimization")
    print(f"{'='*60}")

    # Try top 3 initializations
    for init_idx in range(min(3, len(init_candidates))):
        if time.monotonic() > deadline - 200:
            break
        name, a_init = init_candidates[init_idx]
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        alloc = min(remaining * 0.15, 120.0)
        print(f"  Adam on '{name}' (n={len(a_init)}), budget={alloc:.0f}s")

        # Multiple K values
        for K in [10, 20, 50]:
            if time.monotonic() > deadline - 180:
                break
            a_opt = adam_peak_opt(a_init, max_iters=30000, time_budget=alloc / 3,
                                  beta_start=5.0, beta_end=200.0,
                                  lr_start=0.04, lr_decay=0.99997, K=K)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"    K={K}: {best_score:.8f} *** UPDATE")
            else:
                print(f"    K={K}: {obj:.8f}")

    # === Phase 2: LP-based peak reduction ===
    print(f"\n{'='*60}")
    print(f"Phase 2: LP peak reduction (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None:
        for lp_iter in range(4):
            if time.monotonic() > deadline - 200:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            lp_time = min(remaining * 0.1, 40.0)
            print(f"  LP iter {lp_iter}, time={lp_time:.0f}s")
            a_lp = lp_peak_reduce(best_seq, time_limit=lp_time)

            if a_lp is not None:
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"    LP result: {best_score:.8f} *** UPDATE")
                else:
                    print(f"    LP result: {obj_lp:.8f} (no imp)")

                # Follow with Adam
                if time.monotonic() < deadline - 100:
                    a_after = adam_peak_opt(a_lp, max_iters=20000,
                                            time_budget=min(remaining * 0.05, 30.0),
                                            beta_start=50.0, beta_end=400.0,
                                            lr_start=0.02, lr_decay=0.99999, K=10)
                    obj_after = compute_obj(a_after)
                    if obj_after < best_score:
                        best_score = obj_after
                        best_seq = a_after.copy()
                        print(f"    LP+Adam: {best_score:.8f} *** UPDATE")
            else:
                print(f"    LP: no improvement found")

    # === Phase 3: L-BFGS-B on smoothed objective ===
    print(f"\n{'='*60}")
    print(f"Phase 3: L-BFGS-B optimization")
    print(f"{'='*60}")

    if best_seq is not None:
        for beta_try in [30, 100, 300]:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            alloc = min(remaining * 0.12, 60.0)
            a_lb = lbfgs_smooth_opt(best_seq, beta=beta_try,
                                    time_budget=alloc, K=20)
            obj_lb = compute_obj(a_lb)
            if obj_lb < best_score:
                best_score = obj_lb
                best_seq = a_lb.copy()
                print(f"  L-BFGS-B (beta={beta_try}): {best_score:.8f} *** UPDATE")
            else:
                print(f"  L-BFGS-B (beta={beta_try}): {obj_lb:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = sorted(set([n_best - 100, n_best - 50, n_best,
                                n_best + 50, n_best + 100, n_best + 200,
                                n_best + 300]))
        n_options = [n for n in n_options if 300 <= n <= 1500]

        rng_local = np.random.default_rng(seed + 1)
        for n_try in n_options:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if n_try == len(best_seq):
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s

            alloc = min(remaining * 0.15, 60.0)
            a_opt = adam_peak_opt(a_try, max_iters=20000, time_budget=alloc,
                                  beta_start=10.0, beta_end=300.0,
                                  lr_start=0.03, lr_decay=0.99998, K=20)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 5: Coordinate descent + high-beta Adam ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        # Coordinate descent
        if time.monotonic() < deadline - 40:
            remaining = deadline - time.monotonic()
            a_cd = coordinate_refine(best_seq,
                                     time_budget=min(remaining * 0.15, 30.0),
                                     seed=seed)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Coord descent: {best_score:.8f} *** UPDATE")
            else:
                print(f"  Coord descent: {obj_cd:.8f}")

        # High-beta Adam (exploitation)
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            a_hb = adam_peak_opt(best_seq, max_iters=20000,
                                 time_budget=min(remaining * 0.2, 40.0),
                                 beta_start=200.0, beta_end=1000.0,
                                 lr_start=0.01, lr_decay=0.99999, K=5)
            obj_hb = compute_obj(a_hb)
            if obj_hb < best_score:
                best_score = obj_hb
                best_seq = a_hb.copy()
                print(f"  High-beta Adam: {best_score:.8f} *** UPDATE")
            else:
                print(f"  High-beta Adam: {obj_hb:.8f}")

        # Full smoothing Adam (broader view)
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            a_fs = adam_peak_opt(best_seq, max_iters=15000,
                                 time_budget=min(remaining * 0.15, 30.0),
                                 beta_start=5.0, beta_end=50.0,
                                 lr_start=0.03, lr_decay=0.99998,
                                 K=40, use_topk=True)
            obj_fs = compute_obj(a_fs)
            if obj_fs < best_score:
                best_score = obj_fs
                best_seq = a_fs.copy()
                print(f"  Full-smooth Adam: {best_score:.8f} *** UPDATE")

    # === Phase 6: Random restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Random restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        rng_r = np.random.default_rng(seed + 100)

        for restart in range(6):
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if restart % 3 == 0:
                # Multiplicative noise
                scale = 0.02 + 0.01 * restart
                a_pert = best_seq * np.exp(rng_r.normal(0, scale, n_best))
            elif restart % 3 == 1:
                # Blend with cosine
                alpha = rng_r.uniform(0.05, 0.2)
                cos_init = build_cosine(n_best)
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * cos_init
            else:
                # Blend with super-Gaussian
                alpha = rng_r.uniform(0.05, 0.2)
                sg_init = build_super_gaussian(n_best, int(rng_r.integers(3, 9)))
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * sg_init

            a_pert = np.maximum(a_pert, 1e-12)
            s_p = np.sum(a_pert)
            if s_p < 1e-10:
                continue
            a_pert = a_pert / s_p

            alloc = min(remaining * 0.15, 25.0)
            a_opt = adam_peak_opt(a_pert, max_iters=15000, time_budget=alloc,
                                  beta_start=20.0, beta_end=400.0,
                                  lr_start=0.03, lr_decay=0.99998, K=15)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart}: {obj:.8f}")

    # === Final: LP + Adam on absolute best ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        # One more LP attempt
        a_final_lp = lp_peak_reduce(best_seq, time_limit=min(remaining * 0.3, 20.0))
        if a_final_lp is not None:
            obj_fl = compute_obj(a_final_lp)
            if obj_fl < best_score:
                best_score = obj_fl
                best_seq = a_final_lp.copy()
                print(f"\n  Final LP: {best_score:.8f} *** UPDATE")

        # Final Adam
        if time.monotonic() < deadline - 10:
            remaining2 = deadline - time.monotonic()
            a_final = adam_peak_opt(best_seq, max_iters=15000,
                                    time_budget=min(remaining2 * 0.4, 20.0),
                                    beta_start=50.0, beta_end=500.0,
                                    lr_start=0.015, lr_decay=0.99999, K=10)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final Adam: {best_score:.8f} *** UPDATE")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
```
