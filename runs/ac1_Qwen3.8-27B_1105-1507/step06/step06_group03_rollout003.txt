def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Step function optimizer using pressure-based multiplicative updates (KKT-inspired),
combined with adaptive coordinate descent, SCA-LP, and multi-scale annealing.
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def pressure_update(a, tau, alpha, line_search=True):
    """
    One step of pressure-based multiplicative update.
    Returns updated a and the objective.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s  # normalize to sum=1
    
    conv = fast_conv(a)
    M = conv.max()
    
    # Softmax weights (shifted for numerical stability)
    shifted = tau * (conv - M)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Cross-correlation: cross_corr[i] = sum_j a[j] * w[i+j]
    # This is convolve(w, a, mode='valid')
    cross_corr = np.convolve(w, a, mode='valid')[:n]
    
    # Pressure
    pressure = 2.0 * cross_corr
    
    # Avoid division by zero
    pressure = np.maximum(pressure, 1e-15)
    
    # Target: equalize pressure
    p_mean = pressure.mean()
    
    # Multiplicative update
    a_new = a * (p_mean / pressure) ** alpha
    
    # Project onto simplex
    a_new = project_simplex(a_new)
    
    return a_new, compute_J(np.maximum(a_new, 0))

def pressure_optimize(a, deadline, tau_init=5.0, tau_final=50.0, 
                      alpha=0.5, restart_threshold=500, verbose=False):
    """
    Pressure-based optimization with adaptive temperature and restarts.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        # Adaptive temperature
        frac = min(1.0, it / 5000)
        tau = tau_init + (tau_final - tau_init) * frac
        
        # Pressure update
        a_new, obj_new = pressure_update(a, tau, alpha)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it += 1
        
        # Restart if stuck
        if no_improve > restart_threshold:
            # Add noise and restart
            rng = np.random.default_rng(it)
            noise = rng.normal(0, 0.05, n)
            a_restart = project_simplex(best_a + noise)
            a = a_restart
            no_improve = 0
            # Reset temperature
            it = 0
            
            if verbose and it % 1000 == 0:
                print(f"      Pressure restart: obj={best_obj:.8f}", flush=True)
        
        if verbose and it % 10000 == 0:
            print(f"      Pressure it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)
    
    return best_a, best_obj

def coord_descent_fast(a, deadline, max_sweeps=50000, verbose=False):
    """
    Fast coordinate descent with incremental convolution update.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    no_improve = 0
    max_no_improve = 500

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved = False

        for i in range(n):
            old = a[i]
            
            # Try a few candidate values
            if old < 1e-10:
                candidates = [0.0, 1e-4, 1e-3, 1e-2, 0.05]
            else:
                candidates = np.array([0.0, 0.5, 0.75, 1.25, 1.5, 2.0, 3.0]) * old
                candidates = np.unique(candidates)

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i

            best_val = old
            best_local = 2.0 * n * conv.max()

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta

                new_conv_max = conv.max()
                new_obj = 2.0 * n * new_conv_max / (s * s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] -= delta * delta

            if abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a

def sca_lp_optimize(a, deadline, max_iters=500, verbose=False):
    """
    SCA-LP optimization.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Build LP
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))), 
                         shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                       bounds=bounds, method='highs',
                       options={'time_limit': 20, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                if g.sum() > 1e-10:
                    g /= g.sum()
                    found = False
                    for frac in [1.0, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() > 1e-10:
                            a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 50:
            break

    return best_a, best_obj

def generate_inits(n, rng, parent=None):

    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.03, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    inits.append(np.ones(n) / n)
    
    # Cosine patterns
    for freq in [1, 2, 3, 5, 7, 11]:
        for amp in [0.3, 0.6, 0.9]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
    
    # Gaussian mixtures
    for _ in range(5):
        centers = rng.uniform(0.1, 0.9, size=rng.integers(1, 4))
        widths = rng.uniform(0.05, 0.3, size=len(centers))
        sig = np.zeros(n)
        for c, w in zip(centers, widths):
            sig += np.exp(-((t - c) / w) ** 2)
        inits.append(project_simplex(sig + 0.05))
    
    # Power laws
    for alpha in [0.3, 0.7, 1.5, 2.5]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))
    
    return inits
