**The algorithm (Iteratively Reweighted Quadratic Optimization - IRQO):**

1. Initialize a (e.g., from parent or uniform)
2. Repeat:
   a. Compute conv = a*a
   b. Set weights w[k] = conv[k]^p for some p (or w[k] = exp(conv[k]/tau) for some temperature tau)
   c. Build Hankel matrix W[i,j] = w[min(i+j, 2n-2)] (or just w[i+j] for valid indices)
   d. Solve QP: min a^T W a s.t. sum(a)=1, a>=0
   e. Update a
3. Return best a found

The parameter p (or tau) controls how aggressively we focus on the peak. Large p focuses more on the peak; small p is more balanced.

This is similar to the "majorization-minimization" (MM) or "iteratively reweighted least squares" ideas from robust statistics.

Let me think about why this should work: the QP solution minimizes a weighted version of the "energy" of the convolution. By putting more weight on the peak, we push the peak down. The solution of the QP will have the property that the weighted sum is minimized, which tends to flatten the convolution.

            # Let m = i+j, j = m-i: sum_m w[m]*a[m-i]
            # = correlation of w and a at lag i

            corr = np.fft.irfft(np.fft.rfft(w, N2) * np.conj(np.fft.rfft(a, N2)), N2)

            # Actually, the standard result is:
            # ifft(FFT(x) * conj(FFT(y)))[n] = sum_k x[k]*y[k-n] (cross-correlation)

            # We want sum_j w[i+j]*a[j]. Let k = i+j, j = k-i:
            # = sum_k w[k]*a[k-i] = corr[i] ✓ (for i >= 0 and k-i >= 0, i.e., k >= i)
            # But we also need k < 2n-1 (valid w index) and k-i < n (valid a index)
            # With zero-padding, this should be automatic.
            
            Wa = corr[:n]
        else:

            Wa = np.array([np.dot(w[i:i+n], a) for i in range(n)])

        obj_val = float(np.dot(a, Wa))
        return obj_val, grad
    
    # Constraints: sum(a) = 1, a >= 0
    constraints = [{'type': 'eq', 'fun': lambda a: np.sum(a) - 1.0, 'jac': lambda a: np.ones(n)}]
    bounds = [(0, None)] * n
    
    # Initial point
    a0 = np.ones(n) / n
    
    result = minimize(obj_and_grad, a0, method='SLSQP', 
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 1000, 'ftol': 1e-10})
    
    if result.success:
        return project_nonneg_sum1(result.x)
    return None

"""
Iteratively Reweisted Quadratic Optimization (IRQO) for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses Hankel matrix structure for efficient QP subproblems, combined
with subgradient polishing and multi-length exploration.
"""
import numpy as np
from scipy.fft import rfft, irfft
import time
import sys

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def hankel_matvec(w, a):
    """Compute W@a where W[i,j] = w[i+j] (Hankel matrix).
    Result: (W@a)[i] = sum_j w[i+j]*a[j] for i=0,...,n-1
    """
    n = len(a)
    # Direct computation (fast enough for n <= ~2000)
    # (W@a)[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # We can use FFT: this is the cross-correlation
    N = 1
    while N < 2 * 2 * n:  # enough for linear correlation
        N <<= 1
    w_pad = np.zeros(N)
    w_pad[:2*n-1] = w
    a_pad = np.zeros(N)
    a_pad[:n] = a
    
    # Cross-correlation: (w ⋆ a)[i] = sum_j w[i+j]*a[j]
    # Using: ifft(FFT(w) * conj(FFT(a)))[i] = sum_j w[j]*a[(j-i) mod N]
    # We want sum_j w[i+j]*a[j]. Let k=i+j, j=k-i: sum_k w[k]*a[k-i]
    # = ifft(FFT(w) * conj(FFT(a)))[i]? Let's check:
    # ifft(FFT(w) * conj(FFT(a)))[m] = sum_k w[k] * conj(a[k-m])... 
    # For real a: = sum_k w[k] * a[k-m]
    # So we want m = -i, or equivalently m = N-i (circular)
    # Hmm, this is getting confusing. Let me just use the "flip" trick.
    
    # sum_j w[i+j]*a[j] = sum_j w[i+j]*a[j]
    # Let b[j] = a[n-1-j] (reverse of a)
    # Then sum_j w[i+j]*a[j] = sum_j w[i+j]*b[n-1-j]
    # = (w * b)[i+n-1] (convolution at position i+n-1)
    
    a_rev = a[::-1]
    N2 = 1
    while N2 < (2*n-1) + n:
        N2 <<= 1
    conv_wb = irfft(rfft(w, N2) * rfft(a_rev, N2), N2)
    # (w * a_rev)[k] = sum_j w[j]*a_rev[k-j] = sum_j w[j]*a[n-1-k+j]
    # We want sum_j w[i+j]*a[j]. Let j' = n-1-j: sum_j w[i+j]*a[n-1-j']
    # Hmm, this substitution is getting messy.
    
    # Let me just do it directly for correctness
    result = np.zeros(n)
    for i in range(n):
        result[i] = np.dot(w[i:i+n], a)
    return result

def hankel_matvec_fast(w, a):
    """Fast Hankel matrix-vector product using FFT."""
    n = len(a)
    # (W@a)[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is equivalent to the "anti-diagonal" sum of the outer product
    # Use FFT-based approach:
    # Let's define c[k] = w[k] for k=0,...,2n-2 (padded to 0 elsewhere)
    # Then (W@a)[i] = sum_j c[i+j]*a[j]
    # This is the cross-correlation (c ⋆ a)[i]
    # In FFT: ifft(FFT(c) * conj(FFT(a)))[i] gives the circular cross-correlation
    # With zero-padding, it gives the linear cross-correlation
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    cf = rfft(w, N)  # w has length 2n-1
    af = rfft(a, N)  # a has length n
    
    # Cross-correlation: ifft(FFT(c) * conj(FFT(a)))
    corr = irfft(cf * np.conj(af), N)
    
    # corr[i] = sum_k c[k] * a[k-i] (for linear, with zero padding)
    # We want sum_j w[i+j]*a[j] = sum_k w[k]*a[k-i] (substituting k=i+j)
    # = corr[i] for i >= 0 ✓
    
    return corr[:n]

def irqo_optimize(a, max_outer=20, max_inner=50, tau_init=0.1, power_mode=False, power_p=2.0, time_budget=60.0):
    """
    Iteratively Reweighted Quadratic Optimization.
    Outer loop: update weights based on current conv
    Inner loop: projected gradient on the QP objective
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for outer in range(max_outer):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        
        # Update weights
        if power_mode:
            conv_max = np.max(conv)
            if conv_max < 1e-15:
                break
            w = (conv / conv_max) ** power_p
        else:
            conv_max = np.max(conv)
            tau = tau_init * (1 + outer * 0.1)
            w = np.exp(conv / tau - conv_max / tau)  # shift for numerical stability
            w = w / np.max(w)  # normalize
        
        # Inner loop: projected gradient on min a^T W a s.t. sum(a)=1, a>=0
        step = 1.0 / (2.0 * np.max(w) * n)  # initial step size based on Lipschitz constant
        # Actually, the Lipschitz constant of the gradient 2*W*a is 2*lambda_max(W)
        # lambda_max(W) <= max_row_sum(W) = max_i sum_j w[i+j] <= n * max(w)
        # So step <= 1/(2 * n * max(w))
        
        a_inner = a.copy()
        for inner in range(max_inner):
            if time.monotonic() - start > time_budget:
                break
            
            Wa = hankel_matvec_fast(w, a_inner)
            grad = 2.0 * Wa
            grad_centered = grad - np.mean(grad)
            
            a_new = a_inner - step * grad_centered
            a_new = project_nonneg_sum1(a_new)
            
            # Simple backtracking line search
            obj_try = compute_obj(a_new)
            if obj_try < compute_obj(a_inner):
                a_inner = a_new
                step = min(step * 1.1, 1.0 / (2.0 * np.max(w) * n) * 2)
            else:
                step *= 0.7
            
            if time.monotonic() - start > time_budget:
                break
        
        obj_inner = compute_obj(a_inner)
        if obj_inner < best_obj:
            best_obj = obj_inner
            best_a = a_inner.copy()
        else:
            # If no improvement, try with current a
            a = a_inner  # still use it as starting point for next outer iteration
    
    return best_a, best_obj

def subgradient_optimize(a, max_iters=200, time_budget=30.0, tau_schedule='linear'):
    """
    Subgradient optimization with momentum on the max-conv objective.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum_buf = np.zeros(n)
    momentum = 0.7
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of max(conv) at a: grad[j] = 2*a[k_star - j] if 0 <= k_star-j < n
        # Using the formula: d/d a[j] (a*a)[k*] = 2*a[k*-j] if 0 <= k*-j < n
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
        
        grad_centered = grad - np.mean(grad)
        
        # Adaptive step size
        if tau_schedule == 'linear':
            step = 0.3 / (1.0 + 0.01 * it)
        elif tau_schedule == 'diminishing':
            step = 0.5 / np.sqrt(1.0 + it)
        else:
            step = 0.2
        
        # Momentum
        momentum_buf = momentum * momentum_buf - step * grad_centered
        a_new = a + momentum_buf
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Accept based on the max-conv value (not the full objective, for stability)
        conv_new = compute_conv(a_new)
        conv_old = compute_conv(a)
        if np.max(conv_new) <= np.max(conv_old) * 1.001:
            a = a_new
        else:
            # Roll back
            momentum_buf *= 0.5
    
    return best_a, best_obj

def smooth_anneal_objective(a, max_iters=100, time_budget=20.0):
    """
    Optimize the smoothed objective: 2*n*softmax_beta(conv)/(sum(a))^2
    where softmax_beta(x) = log(sum(exp(x/beta))/len(x)) * beta (smooth max)
    This gives a differentiable objective for gradient-based optimization.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.01  # smoothing parameter (smaller = closer to true max)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        sum_a = np.sum(a)
        
        # Smooth max: m = beta * log(sum(exp(conv/beta))/len(conv))
        # Gradient of smooth max: softmax weights
        conv_shifted = conv - np.max(conv)
        weights = np.exp(conv_shifted / beta)
        weights /= np.sum(weights)
        
        # Gradient of smoothed objective w.r.t. a
        # d/d a[j] of sum_k weights[k] * (a*a)[k] = sum_k weights[k] * 2*a[k-j] (if valid)
        grad = np.zeros(n)
        for k in range(len(weights)):
            if weights[k] < 1e-10:
                continue
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * weights[k] * a[idx]
        
        grad_centered = grad - np.mean(grad)
        
        # Adaptive step
        step = 0.1 / (1.0 + 0.05 * it)
        
        a_new = a - step * grad_centered
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Anneal beta (make it smaller over time to approach true max)
        beta = max(0.001, beta * 0.95)
    
    return best_a, best_obj

def chase_peak_ff(a, n_steps=50, step_size=0.02):

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0/new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

"""
Iteratively Reweighted Quadratic Optimization (IRQO) combined with smooth
annealing and subgradient polishing to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: uses Hankel matrix structure for efficient QP subproblems,
iteratively reweighting convolution peaks to drive them down.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def hankel_matvec(w, a):
    """Compute W@a where W[i,j] = w[i+j] (Hankel matrix).
    (W@a)[i] = sum_{j=0}^{n-1} w[i+j]*a[j] for i=0,...,n-1
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    cf = rfft(w, N)
    af = rfft(a, N)
    corr = irfft(cf * np.conj(af), N)
    return corr[:n]


def irqo_optimize(a, max_outer=20, max_inner=50, tau_init=0.1, power_mode=False,
                  power_p=2.0, time_budget=60.0):
    """
    Iteratively Reweighted Quadratic Optimization.
    Outer loop: update weights from current conv
    Inner loop: projected gradient on min a^T W a s.t. sum(a)=1, a>=0
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for outer in range(max_outer):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        conv_max = np.max(conv)
        if conv_max < 1e-15:
            break

        if power_mode:
            w = (conv / conv_max) ** power_p
        else:
            tau = tau_init * (1.0 + 0.15 * outer)
            w = np.exp((conv - conv_max) / tau)
            w = w / max(np.max(w), 1e-30)

        # Inner loop: projected gradient on QP
        w_max = np.max(w)
        # Lipschitz constant of 2*W is 2*lambda_max(W) <= 2*n*max(w)
        lip = 2.0 * n * w_max
        step = 1.0 / lip
        a_inner = a.copy()
        prev_obj = compute_obj(a_inner)

        for inner in range(max_inner):
            if time.monotonic() - start > time_budget:
                break

            Wa = hankel_matvec(w, a_inner)
            grad = 2.0 * Wa
            grad_centered = grad - np.mean(grad)

            a_new = a_inner - step * grad_centered
            a_new = project_nonneg_sum1(a_new)

            obj_new = compute_obj(a_new)
            if obj_new < prev_obj - 1e-12:
                a_inner = a_new
                prev_obj = obj_new
                # Reduce step slightly to be safe
                step = step * 0.98
            else:
                step *= 0.6
                if step < 1e-10:
                    break

        obj_inner = compute_obj(a_inner)
        if obj_inner < best_obj - 1e-12:
            best_obj = obj_inner
            best_a = a_inner.copy()
        a = a_inner

    return best_a, best_obj


def smooth_anneal_objective(a, max_iters=150, time_budget=25.0, beta_init=0.02):
    """
    Optimize smoothed objective using softmax weights on convolution peaks.
    Gives a smooth, differentiable approximation to the max-conv objective.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = beta_init
    momentum_buf = np.zeros(n)
    momentum = 0.75

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        conv_max = np.max(conv)

        # Softmax weights
        shifted = (conv - conv_max) / beta
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient: grad[j] = sum_k weights[k] * 2*a[k-j] (if 0<=k-j<n)
        grad = np.zeros(n)
        for k in range(len(weights)):
            if weights[k] < 1e-10:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * weights[k] * a[idx]

        grad_centered = grad - np.mean(grad)

        # Adaptive step with momentum
        step = 0.5 / (1.0 + 0.02 * it)
        momentum_buf = momentum * momentum_buf - step * grad_centered
        a_new = a + momentum_buf
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        # Accept if not too bad
        conv_new = compute_conv(a_new)
        if np.max(conv_new) <= conv_max * 1.005:
            a = a_new
        else:
            momentum_buf *= 0.5

        # Anneal beta
        beta = max(0.002, beta * 0.97)

    return best_a, best_obj


def subgradient_momentum(a, max_iters=300, time_budget=30.0):
    """Subgradient method with heavy-ball momentum on max-conv."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum_buf = np.zeros(n)
    momentum = 0.8

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: grad[j] = 2*a[k_star-j] if 0<=k_star-j<n
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)
        for j in range(j_start, j_end):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]

        grad_centered = grad - np.mean(grad)

        # Diminishing step
        step = 0.4 / (1.0 + 0.015 * it)

        momentum_buf = momentum * momentum_buf - step * grad_centered
        a_new = a + momentum_buf
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        conv_new = compute_conv(a_new)
        if np.max(conv_new) <= np.max(conv) * 1.002:
            a = a_new
        else:
            momentum_buf *= 0.5

    return best_a, best_obj


def chase_peak(a, n_steps=50, step_size=0.02):
    """Chase-the-peak: iteratively reduce the maximum convolution peak."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)
        for j in range(j_start, j_end):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]

        grad_centered = grad - np.mean(grad)

        a_try = a - step_size * grad_centered
        a_try = project_nonneg_sum1(a_try)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_obj


def multi_peak_subgradient(a, max_iters=200, time_budget=25.0, n_peaks=3):
    """Subgradient using top-k peaks with softmax weighting."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum_buf = np.zeros(n)
    momentum = 0.7

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        conv_max = np.max(conv)

        # Use top peaks with exponential weighting
        tau = 0.01 + 0.02 * (it / max_iters)
        shifted = (conv - conv_max) / tau
        weights = np.exp(shifted)
        weights /= np.sum(weights)

        # Only keep top n_peaks significant contributors
        top_idx = np.argsort(weights)[-n_peaks * (2 * n - 1) // 10:]
        if len(top_idx) > 0:
            w_mask = np.zeros(len(weights))
            w_mask[top_idx] = 1.0
            weights = weights * w_mask
            weights /= np.sum(weights)

        grad = np.zeros(n)
        for k in range(len(weights)):
            if weights[k] < 1e-8:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * weights[k] * a[idx]

        grad_centered = grad - np.mean(grad)
        step = 0.3 / (1.0 + 0.01 * it)

        momentum_buf = momentum * momentum_buf - step * grad_centered
        a_new = a + momentum_buf
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        if compute_conv(a_new).max() <= conv_max * 1.003:
            a = a_new
        else:
            momentum_buf *= 0.5

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def perturbed_restart(a, rng, noise_level=0.05):
    """Create a perturbed version of the sequence for diversification."""
    n = len(a)
    a_new = a * np.exp(rng.normal(0, noise_level, n))
    return project_nonneg_sum1(a_new)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization combining IRQO (iteratively
    reweighted QP via Hankel matrix), smooth annealing, subgradient methods,
    and multi-length exploration to minimize the peak autocorrelation ratio.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}", flush=True)
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700)
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}", flush=True)

    print(f"Starting: best={best_score:.8f}, n={len(best_seq)}", flush=True)

    # === Phase 1: IRQO with various weightings ===
    print(f"\n=== Phase 1: IRQO Optimization ===")
    phase1_end = start_time + min(350.0, (deadline - start_time) * 0.28)

    irqo_configs = [
        ("power_p1.5", dict(power_mode=True, power_p=1.5, max_outer=12, max_inner=50)),
        ("power_p2", dict(power_mode=True, power_p=2.0, max_outer=12, max_inner=50)),
        ("power_p3", dict(power_mode=True, power_p=3.0, max_outer=12, max_inner=50)),
        ("power_p4", dict(power_mode=True, power_p=4.0, max_outer=10, max_inner=60)),
        ("exp_tau003", dict(power_mode=False, tau_init=0.03, max_outer=12, max_inner=50)),
        ("exp_tau005", dict(power_mode=False, tau_init=0.05, max_outer=12, max_inner=50)),
        ("exp_tau01", dict(power_mode=False, tau_init=0.1, max_outer=12, max_inner=50)),
    ]

    for cfg_name, kwargs in irqo_configs:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 15:
            break
        t_alloc = min(remaining * 0.4, 80.0)
        n_cur = len(best_seq)
        print(f"  IRQO {cfg_name} (n={n_cur}, t={t_alloc:.0f}s)...", flush=True)
        a_new, obj_new = irqo_optimize(best_seq, time_budget=t_alloc, **kwargs)
        if obj_new < best_score - 1e-10:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"    ** IMPROVED: {best_score:.8f} **", flush=True)
        else:
            print(f"    obj={obj_new:.8f}", flush=True)

    # === Phase 2: Smooth anneal + multi-peak subgradient ===
    print(f"\n=== Phase 2: Smooth Anneal + Multi-Peak Subgrad ===")
    print(f"Best: {best_score:.8f}")
    phase2_end = start_time + min(650.0, (deadline - start_time) * 0.45)

    phase2_methods = [
        ("smooth_b002", lambda a, t: smooth_anneal_objective(a, max_iters=100, time_budget=t, beta_init=0.02)),
        ("smooth_b005", lambda a, t: smooth_anneal_objective(a, max_iters=100, time_budget=t, beta_init=0.05)),
        ("multi_peak3", lambda a, t: multi_peak_subgradient(a, max_iters=200, time_budget=t, n_peaks=3)),
        ("multi_peak5", lambda a, t: multi_peak_subgradient(a, max_iters=200, time_budget=t, n_peaks=5)),
        ("subgrad_hb", lambda a, t: subgradient_momentum(a, max_iters=250, time_budget=t)),
    ]

    for m_name, m_func in phase2_methods:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 15:
            break
        t_alloc = min(remaining * 0.45, 80.0)
        print(f"  {m_name} (n={len(best_seq)}, t={t_alloc:.0f}s)...", flush=True)
        a_new, obj_new = m_func(best_seq, t_alloc)
        if obj_new < best_score - 1e-10:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"    ** IMPROVED: {best_score:.8f} **", flush=True)
        else:
            print(f"    obj={obj_new:.8f}", flush=True)

    # === Phase 3: Chase peak + IRQO refinement ===
    print(f"\n=== Phase 3: Chase Peak + IRQO Refinement ===")
    phase3_end = start_time + min(900.0, (deadline - start_time) * 0.62)

    # Chase peak with fine steps
    a_cp, obj_cp = chase_peak(best_seq, n_steps=60, step_size=0.01)
    if obj_cp < best_score - 1e-10:
        best_score = obj_cp
        best_seq = a_cp.copy()
        print(f"  Chase peak: {best_score:.8f}", flush=True)

    remaining3 = phase3_end - time.monotonic()
    if remaining3 > 20:
        a_r, obj_r = irqo_optimize(best_seq, max_outer=10, max_inner=50,
                                    power_mode=True, power_p=5.0,
                                    time_budget=min(remaining3 * 0.5, 60))
        if obj_r < best_score - 1e-10:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  IRQO p5 refine: {best_score:.8f}", flush=True)

        remaining3b = phase3_end - time.monotonic()
        if remaining3b > 15:
            a_sa, obj_sa = smooth_anneal_objective(best_seq, max_iters=80,
                                                    time_budget=min(remaining3b * 0.6, 40),
                                                    beta_init=0.01)
            if obj_sa < best_score - 1e-10:
                best_score = obj_sa
                best_seq = a_sa.copy()
                print(f"  Smooth b001: {best_score:.8f}", flush=True)

    # === Phase 4: Diversification + multi-length ===
    print(f"\n=== Phase 4: Diversification + Multi-Length ===")
    print(f"Best: {best_score:.8f} (n={len(best_seq)})")
    phase4_end = min(deadline - 100.0, start_time + (deadline - start_time) * 0.80)
    remaining4 = phase4_end - time.monotonic()

    if remaining4 > 30:
        # First: try perturbed restarts on current best
        for restart in range(3):
            if time.monotonic() > phase4_end:
                break
            a_pert = perturbed_restart(best_seq, rng, noise_level=0.03 + 0.02 * restart)
            a_opt, obj_opt = irqo_optimize(a_pert, max_outer=8, max_inner=40,
                                            power_mode=True, power_p=2.5,
                                            time_budget=20)
            a_opt2, obj_opt2 = chase_peak(a_opt, n_steps=30, step_size=0.01)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2
            if obj_opt < best_score - 1e-10:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Perturb {restart}: {best_score:.8f}", flush=True)

        # Multi-length exploration
        remaining4b = phase4_end - time.monotonic()
        if remaining4b > 20:
            n_current = len(best_seq)
            n_tries = list(dict.fromkeys([
                n_current - 50, n_current - 100, n_current + 50,
                n_current + 100, n_current + 200, n_current - 200,
                max(100, n_current - 300), min(2000, n_current + 300)
            ]))
            n_tries = [n for n in n_tries if 100 <= n <= 2500]
            per_n_time = max(8.0, remaining4b / len(n_tries))

            for n_try in n_tries:
                if time.monotonic() > phase4_end:
                    break
                a_res = project_nonneg_sum1(resample_sequence(best_seq, n_try))
                a_opt, obj_opt = irqo_optimize(a_res, max_outer=6, max_inner=35,
                                                power_mode=True, power_p=2.0,
                                                time_budget=min(per_n_time * 0.6, 25))
                a_opt2, obj_opt2 = chase_peak(a_opt, n_steps=25, step_size=0.01)
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

                if obj_opt < best_score - 1e-10:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  n={n_try}: NEW BEST {best_score:.8f}", flush=True)
                else:
                    print(f"  n={n_try}: obj={obj_opt:.8f}", flush=True)

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()

    if remaining5 > 15 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())

        polish_configs = [
            ("irqo_p3", lambda a, t: irqo_optimize(a, max_outer=10, max_inner=45,
                                                    power_mode=True, power_p=3.0, time_budget=t)),
            ("irqo_p6", lambda a, t: irqo_optimize(a, max_outer=8, max_inner=50,
                                                    power_mode=True, power_p=6.0, time_budget=t)),
            ("smooth_tight", lambda a, t: smooth_anneal_objective(a, max_iters=60,
                                                                  time_budget=t, beta_init=0.008)),
            ("subgrad_final", lambda a, t: subgradient_momentum(a, max_iters=150, time_budget=t)),
            ("chase_final", lambda a, t: chase_peak(a, n_steps=min(40, int(t * 3)), step_size=0.008)),
            ("multi_pk2", lambda a, t: multi_peak_subgradient(a, max_iters=100,
                                                              time_budget=t, n_peaks=2)),
        ]

        for round_idx in range(min(6, int(remaining5 / 12))):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 10:
                break

            for m_name, m_func in polish_configs:
                if time.monotonic() > deadline - 5:
                    break
                t_all = min(rem * 0.25, 35.0)
                if t_all < 6:
                    continue

                a_new, obj_new = m_func(a_final, t_all)
                if obj_new < best_score - 1e-10:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_final = a_new.copy()
                    print(f"  R{round_idx} {m_name}: {best_score:.8f}", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
