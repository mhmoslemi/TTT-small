```python
def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # We want to find theta such that sum(max(v - theta, 0)) = target_sum
    # Use binary search
    sorted_v = np.sort(v)[::-1]
    cv = np.cumsum(sorted_v)
    rho = np.where(sorted_v > 0)[0]
    # Find largest rho such that sorted_v[rho] > (cv[rho] - target_sum) / (rho + 1)
    cond = sorted_v[rho] > (cv[rho] - target_sum) / (rho + 1)
    if len(cond) == 0:
        theta = (np.sum(v) - target_sum) / n
    else:
        rho_val = rho[np.max(np.where(cond)[0])] + 1
        theta = (cv[rho_val - 1] - target_sum) / rho_val
    result = np.maximum(v - theta, 0)
    # Adjust for floating point
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        # Distribute the difference
        idx = np.argmax(result)
        result[idx] += diff
    return result
```

```python
"""
Optimization of non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth gradient descent with Adam, LP-based alternating, and multi-length search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

# --- Helper functions ---

def compute_conv(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    """Project to {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax_weights(c, tau):
    """Compute softmax weights over convolution values."""
    c_shifted = c / tau
    c_shifted -= np.max(c_shifted)
    w = np.exp(c_shifted)
    w /= w.sum()
    return w

def compute_smooth_gradient(weights, a):
    """Compute gradient of weighted sum of convolution values."""
    n = len(a)
    # grad[i] = 2 * sum_j weights[i+j] * a[j]
    grad = np.zeros(n)
    for i in range(n):
        # j ranges from 0 to n-1, weights index is i+j (valid if < 2n-1)
        valid_end = min(n, 2*n-1-i)
        if valid_end > 0:
            grad[i] = 2.0 * np.dot(weights[i:i+valid_end], a[:valid_end])
    return grad

def smooth_gradient_descent(a, max_iters=2000, time_budget=120.0, 
                            tau_start=0.05, tau_end=0.001):
    """
    Adam-based smooth gradient descent on the smoothed max autocorrelation.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    t = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Animate temperature
        progress = it / max(max_iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute convolution and weights
        conv = compute_conv(a)
        c_max = np.max(conv)
        weights = softmax_weights(conv, tau * c_max)
        
        # Compute gradient
        grad = compute_smooth_gradient(weights, a)
        
        # Center gradient (to maintain sum constraint approximately)
        grad = grad - np.mean(grad)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_nonneg_sum1(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive learning rate: reduce if no improvement for a while
        # (handled by Adam's adaptivity)
    
    return best_a, best_obj

def lp_alternate(a, max_iters=30, time_budget=60.0):
    """LP-based alternating optimization."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Build constraint matrix
        # (a * g)[k] <= M for k = 0, ..., 2n-2
        # (a*g)[k] = sum_j a[j] * g[k-j] (for valid indices)
        # This is a Toeplitz-like system
        A = np.zeros((2*n-1, n))
        for j in range(n):
            for k in range(j, j+n):
                if k < 2*n-1:
                    A[k, j] = a[j]  # wait, this isn't right
        # Actually: (conv(a,g))[k] = sum_j a[j] * g[k-j]
        # For a fixed k: sum over j where 0<=j<n and 0<=k-j<n
        # = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * g[k-j]
        # Let i = k-j, so j = k-i:
        # = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[k-i] * g[i]
        # So A[k,i] = a[k-i] if 0 <= k-i < n, else 0
        
        # Rebuild more carefully
        A = np.zeros((2*n-1, n))
        for i in range(n):
            for k in range(i, i+n):
                if k < 2*n-1:
                    A[k, i] = a[k-i]
```

```python
"""
Optimization of non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth gradient descent with Adam optimizer, LP-based alternating,
multi-length exploration, and iterative peak-chasing for targeted refinement.
Key innovations: (1) principled softmax-smoothed gradient with annealed temperature,
(2) efficient O(n^2) gradient via Hankel structure, (3) LP min-peak formulation,
(4) cosine-annealed learning rate, (5) targeted peak reduction moves.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project to {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_weights(c, tau):
    """Compute softmax weights over convolution values with temperature tau."""
    c_scaled = c / tau
    c_scaled -= np.max(c_scaled)
    w = np.exp(c_scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        idx = int(np.argmax(c))
        w = np.zeros(len(c))
        w[idx] = 1.0
    else:
        w /= w_sum
    return w


def compute_smooth_gradient(weights, a):
    """
    Compute gradient of sum_k weights[k]*conv(a,a)[k] with respect to a.
    Formula: grad[i] = 2 * sum_j weights[i+j] * a[j]
    """
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        valid_n = min(n, 2 * n - 1 - i)
        if valid_n > 0:
            grad[i] = 2.0 * np.dot(weights[i:i + valid_n], a[:valid_n])
    return grad


def compute_peak_gradient(k_star, a):
    """Compute gradient of conv(a,a)[k_star] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def smooth_gradient_descent(a, max_iters=2000, time_budget=120.0,
                            tau_start=0.05, tau_end=0.002, lr_init=0.005):
    """
    Adam-based smooth gradient descent on smoothed max autocorrelation.
    Uses cosine-annealed learning rate and annealed temperature.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    stale_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        # Anneal temperature
        tau = tau_start * (tau_end / tau_start) ** progress
        # Cosine-annealed learning rate
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, lr_init * 0.01)

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        if c_max < 1e-15:
            break
        tau_eff = tau * c_max
        weights = softmax_weights(conv, tau_eff)

        grad = compute_smooth_gradient(weights, a)
        # Center gradient to approximately preserve sum
        grad = grad - np.mean(grad)

        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_nonneg_sum1(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stale_count = 0
        else:
            stale_count += 1
            if stale_count > 200:
                # Reset Adam state to escape potential local minima
                m *= 0.5
                v *= 0.5
                stale_count = 0

        a = a_new

    return best_a, best_obj

            best_a = ((1.0 - best_t_val) * a + best_t_val * g)
            best_a = project_nonneg_sum1(best_a)
            a = best_a.copy()
            # Update A for next iteration
            for i in range(n):
                A[i:i + n, i] = a.copy()
        else:
            break

    return best_a, best_obj


def chase_peak_sgd(a, n_steps=100, step_size=0.02):
    """
    Targeted peak chasing: iteratively reduce the maximum convolution value
    using subgradient steps with adaptive step size.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        grad = compute_peak_gradient(k_star, a)
        grad = grad - np.mean(grad)

        # Adaptive step size: try several
        improved = False
        for ss in [step_size, step_size * 0.5, step_size * 0.25, step_size * 2.0]:
            a_try = a - ss * grad
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] / new_n if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def spectral_shaping(a, n_iters=50, time_budget=20.0):
    """
    Frequency-domain spectral shaping: modify the power spectrum to reduce
    peak autocorrelation, then reconstruct via spectral factorization.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2

        # Target: flatten the power spectrum (reduce peaks away from DC)
        p_max = np.max(power)
        if p_max < 1e-15:
            break

        # Apply log-compression to reduce dynamic range
        p_norm = power / p_max
        p_norm = np.maximum(p_norm, 1e-10)
        alpha = 0.6 + 0.3 * (it / n_iters)
        p_modified = p_norm ** alpha

        # Reconstruct: keep phase, modify magnitude
        phase = np.angle(fa)
        magnitude = np.sqrt(p_modified) * np.sqrt(p_max)
        fa_new = magnitude * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function using multi-strategy optimization:
    1. Smooth gradient descent with Adam (annealed temperature + cosine LR)
    2. LP min-peak alternating
    3. Targeted peak chasing
    4. Spectral shaping
    5. Multi-length exploration
    6. Final polishing
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_norm = project_nonneg_sum1(parent)
        parent_obj = compute_obj(parent_norm)
        print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent_norm.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    n_parent = len(best_seq)
    print(f"Initial best: {best_score:.8f} (n={n_parent})")

    # === Phase 1: Initial SGD (12% of time) ===
    print("\n=== Phase 1: Smooth Gradient Descent ===")
    phase1_budget = min(180.0, (deadline - start_time) * 0.12)
    a1, obj1 = smooth_gradient_descent(best_seq, max_iters=5000,
                                       time_budget=phase1_budget,
                                       tau_start=0.08, tau_end=0.005,
                                       lr_init=0.008)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        print(f"  SGD improved: {best_score:.8f}")

    # === Phase 2: LP Alternating (12% of time) ===
    print("\n=== Phase 2: LP Min-Peak Alternating ===")
    phase2_budget = min(180.0, (deadline - start_time) * 0.12)
    a2, obj2 = lp_minpeak_alternate(best_seq, max_iters=30,
                                    time_budget=phase2_budget)
    if obj2 < best_score:
        best_score = obj2
        best_seq = a2.copy()
        print(f"  LP improved: {best_score:.8f}")

    # === Phase 3: Peak Chasing + SGD (12% of time) ===
    print("\n=== Phase 3: Peak Chasing ===")
    phase3_budget = min(180.0, (deadline - start_time) * 0.12)
    a3, obj3 = chase_peak_sgd(best_seq, n_steps=200, step_size=0.015)
    if obj3 < best_score:
        best_score = obj3
        best_seq = a3.copy()
        print(f"  Peak chase improved: {best_score:.8f}")

    # Follow up with SGD
    a3b, obj3b = smooth_gradient_descent(best_seq, max_iters=3000,
                                         time_budget=phase3_budget * 0.5,
                                         tau_start=0.03, tau_end=0.002,
                                         lr_init=0.005)
    if obj3b < best_score:
        best_score = obj3b
        best_seq = a3b.copy()
        print(f"  SGD after chase: {best_score:.8f}")

    # === Phase 4: Spectral Shaping (8% of time) ===
    print(f"\n=== Phase 4: Spectral Shaping (best={best_score:.8f}) ===")
    phase4_budget = min(120.0, (deadline - start_time) * 0.08)
    a4, obj4 = spectral_shaping(best_seq, n_iters=100, time_budget=phase4_budget)
    if obj4 < best_score:
        best_score = obj4
        best_seq = a4.copy()
        print(f"  Spectral improved: {best_score:.8f}")
    else:
        print(f"  Spectral: {obj4:.8f} (no improvement)")

    # === Phase 5: Multi-length exploration (25% of time) ===
    print(f"\n=== Phase 5: Multi-length Exploration (best={best_score:.8f}) ===")
    phase5_budget = min(375.0, (deadline - start_time) * 0.25)
    n_options = [n_parent - 75, n_parent - 150, n_parent + 75, n_parent + 150,
                 n_parent + 300, n_parent - 300]
    n_options = [n for n in n_options if 150 <= n <= 2500]
    n_options = list(dict.fromkeys(n_options))

    per_n_budget = max(25.0, phase5_budget / len(n_options))
    print(f"  Trying lengths: {n_options}, budget each: {per_n_budget:.0f}s")

    for n_try in n_options:
        if time.monotonic() > start_time + (deadline - start_time) * 0.72:
            break
        a_res = resample_sequence(best_seq, n_try)
        a_res = project_nonneg_sum1(a_res)

        # SGD
        a_opt, obj_opt = smooth_gradient_descent(a_res, max_iters=3000,
                                                 time_budget=per_n_budget * 0.4,
                                                 tau_start=0.05, tau_end=0.003,
                                                 lr_init=0.006)
        # LP (if n is small enough)
        if n_try <= 1500:
            a_opt2, obj_opt2 = lp_minpeak_alternate(a_opt, max_iters=10,
                                                     time_budget=per_n_budget * 0.3)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2
        # Peak chase
        a_opt3, obj_opt3 = chase_peak_sgd(a_opt, n_steps=50, step_size=0.01)
        if obj_opt3 < obj_opt:
            a_opt, obj_opt = a_opt3, obj_opt3

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        elif time.monotonic() < start_time + (deadline - start_time) * 0.7:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 6: Final intensive polishing (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining time: {remaining:.0f}s")

    if remaining > 20 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        a_final_obj = compute_obj(a_final)

        # Multiple rounds of alternating strategies
        round_budget = max(15.0, remaining / 8)
        n_rounds = min(8, int(remaining / round_budget))

        for r in range(n_rounds):
            if time.monotonic() > deadline - 10:
                break
            rem = deadline - time.monotonic()
            if rem < 15:
                break

            # SGD with increasingly low temperature
            tau_s = 0.02 / (r + 1)
            tau_e = 0.001 / (r + 1)
            a_p, obj_p = smooth_gradient_descent(a_final, max_iters=2000,
                                                 time_budget=min(rem * 0.3, 40.0),
                                                 tau_start=tau_s, tau_end=tau_e,
                                                 lr_init=0.003)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  SGD r{r}: {best_score:.8f}")

            # LP
            rem = deadline - time.monotonic()
            if rem > 10 and len(a_final) <= 1500:
                a_p2, obj_p2 = lp_minpeak_alternate(a_final, max_iters=8,
                                                     time_budget=min(rem * 0.25, 25.0))
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_final = a_p2.copy()
                    print(f"  LP r{r}: {best_score:.8f}")

            # Peak chase
            rem = deadline - time.monotonic()
            if rem > 5:
                a_p3, obj_p3 = chase_peak_sgd(a_final, n_steps=50, step_size=0.005)
                if obj_p3 < best_score:
                    best_score = obj_p3
                    best_seq = a_p3.copy()
                    a_final = a_p3.copy()
                    print(f"  Chase r{r}: {best_score:.8f}")

            # Small perturbation + re-optimize
            rem = deadline - time.monotonic()
            if rem > 15:
                pert = a_final * np.exp(rng.normal(0, 0.01, len(a_final)))
                pert = project_nonneg_sum1(pert)
                a_p4, obj_p4 = smooth_gradient_descent(pert, max_iters=1000,
                                                       time_budget=min(rem * 0.2, 20.0),
                                                       tau_start=0.01, tau_end=0.001,
                                                       lr_init=0.003)
                if obj_p4 < best_score:
                    best_score = obj_p4
                    best_seq = a_p4.copy()
                    a_final = a_p4.copy()
                    print(f"  Pert r{r}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    # Ensure all values are valid (non-negative, finite, bounded)
    result_arr = np.asarray(best_seq, dtype=np.float64)
    result_arr = np.clip(result_arr, 0.0, 1000.0)
    result_arr = np.where(np.isfinite(result_arr), result_arr, 0.0)
    if result_arr.sum() < 0.01:
        result_arr = np.ones(100)

    result = [float(x) for x in result_arr]

    # Validate via evaluate_sequence
    val = evaluate_sequence(result)
    if np.isinf(val):
        result = [1.0] * 100
        best_score = 2.0
    else:
        best_score = min(best_score, val)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
```
