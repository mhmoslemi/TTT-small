```python
def active_set_solve(a, tol=1e-6):
    """Given a near-equilibrium, identify active set and solve for dual variables."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    M = b_max
    # Active set: indices where b_k is within tol of M
    active = np.where(b >= M - tol * M)[0]
    m = len(active)
    
    if m == 0:
        return None, None
    if m > n:
        # Too many active, use top-n
        idx = np.argsort(b)[-n:]
        active = idx
        m = n
    
    # Build the system: for each k where a_k > 0, we need (A^T y)_k = 1
    # (A^T y)_k = sum_{k' in active} y_{k'} * A[k', k] = sum_{k' in active} y_{k'} * a[k'-k]
    # where a[k'-k] is defined as 0 if k'-k is out of [0, n-1]
    
    # We want: sum_{k' in active} y_{k'} * a[k'-k] = 1 for k = 0, ..., n-1
    # This is n equations in m unknowns.
    # If m < n: underdetermined, use minimum norm solution
    # If m = n: square system, solve directly
    # If m > n: overdetermined, use least squares
    
    # Build matrix C where C[k, i] = a[active[i] - k] if 0 <= active[i] - k < n
    C = np.zeros((n, m))
    for i, k_prime in enumerate(active):
        for k in range(n):
            diff = k_prime - k
            if 0 <= diff < n:
                C[k, i] = a[diff]
    
    rhs = np.ones(n)
    
    # Solve C @ y = rhs
    if m <= n:
        # Underdetermined or square: use lstsq (gives min norm solution)
        y, residuals, rank, sv = np.linalg.lstsq(C, rhs, rcond=None)
    else:
        y, residuals, rank, sv = np.linalg.lstsq(C, rhs, rcond=None)
    
    # Ensure y >= 0 (clip negative values)
    y = np.maximum(y, 0)
    
    return active, y
```

3. LP mirror descent with Polyak averaging and backtracking on J

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def conv_grad_active(a, active, weights):
    """Compute subgradient sum_k w_k * 2 * shift(a, k) efficiently."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(active, weights):
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        a_lo = k - i_hi
        a_hi = k - i_lo
        grad[i_lo:i_hi + 1] += w * 2.0 * a[a_lo:a_hi + 1][::-1]
    return grad


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: iteratively flatten the power spectrum."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress dynamic range
        mag = (1.0 - beta) * mag + beta * mag.mean()
        F_new = np.exp(1j * phase) * mag
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def subgradient_nesterov(a0, deadline, max_iters=10000, tau=50.0):
    """
    Nesterov-accelerated subgradient method on the normalized problem.
    Maintains sum(a) = 1 at all times.
    Uses softmax-based active set for smooth gradient, transitioning to hard active set.
    """
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = compute_J(a)
    
    y = a.copy()  # Nesterov momentum
    beta = 0.9
    lr = 0.3
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        b = fast_conv(y)
        b_max = float(b.max())
        
        # Softmax active set (smooth approximation)
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Compute gradient
        grad = conv_grad_active(y, np.arange(len(b)), p)
        
        # But we only need the active ones for efficiency
        # For now, compute all (the active set is usually small near optimum)
        active_mask = p > 1e-8
        if active_mask.sum() < len(p):
            grad = conv_grad_active(y, np.where(active_mask)[0], p[active_mask])
        
        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        y = y / max(y.sum(), 1e-15)
        a = a_new
        
        # Check improvement
        M_new = fast_conv(a).max()
        if M_new < best_M * (1 - 1e-12):
            best_M = M_new
            best_a = a.copy()
            best_J = compute_J(a)
        
        # Decay learning rate
        lr = 0.3 / (1 + 0.01 * it)
        # Increase tau (sharpen active set)
        tau = min(500.0, tau * 1.002)
    
    return best_a


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (Ag)_k = (conv(a,g))_k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    """Solve the LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    rows, cols, vals = build_conv_matrix(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def mirror_descent_lp(a0, deadline, max_iters=300):
    """
    LP mirror descent with Polyak averaging and backtracking on J.
    """
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = compute_J(a)
    
    # Polyak average
    a_avg = a.copy()
    
    no_imp = 0
    t_base = 0.4
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Try to improve
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            best_J = compute_J(a)
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break
        
        # LP step
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Backtracking on J
        improved = False
        for t in [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            J_t = compute_J(a_t)
            if J_t < best_J * (1 - 1e-12):
                a = a_t
                best_J = J_t
                best_M = fast_conv(a).max()
                best_a = a.copy()
                # Update Polyak average
                a_avg = (a_avg + a) / 2
                improved = True
                break
            elif t == 0.01:
                # Even the smallest step doesn't help; try maintaining a
                a_t = a
        
        if not improved:
            # Use the LP direction with a very small step
            t = 0.005
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s > 1e-10:
                a = a_t / s
            a_avg = (a_avg + a) / 2
    
    return best_a


def mirror_descent_anderson(a0, deadline, max_iters=300, window=5):
    """
    LP mirror descent with Anderson acceleration.
    """
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = compute_J(a)
    
    no_imp = 0
    hist_x = []
    hist_Fx = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            best_J = compute_J(a)
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Determine step size
        if it < 10:
            t = 0.4
        else:
            t = max(0.02, 0.4 / (1 + 0.03 * it))
        
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 3:
            a_anderson = anderson_mixed(hist_x, hist_Fx)
            if a_anderson is not None:
                sa = a_anderson.sum()
                if sa > 1e-10:
                    a_anderson = a_anderson / sa
                M_and = fast_conv(a_anderson).max()
                M_new = fast_conv(a_new).max()
                if M_and <= M_new * 1.02:
                    a = a_anderson
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def anderson_mixed(hist_x, hist_Fx):
    """Anderson mixing."""
    m = len(hist_x)
    if m < 2:
        return hist_Fx[-1].copy()
    residuals = []
    for i in range(m):
        residuals.append(hist_x[i] - hist_Fx[i])
    residuals = np.array(residuals)
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 1.5)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(hist_x[0]))
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    return np.maximum(mixed, 0)


def active_set_polish(a0, deadline, tol=1e-8):
    """
    Identify active constraints and solve KKT system for precise polish.
    """
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(200):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = float(b.max())
        
        # Identify active set
        active = np.where(b >= b_max * (1 - tol))[0]
        m = len(active)
        
        if m == 0:
            break
        if m > n:
            # Keep only the top n
            idx = np.argsort(b)[-n:]
            active = idx
            m = n
        
        # Solve for dual variables: C @ y = 1 where C[k, i] = a[active[i] - k]
        C = np.zeros((n, m))
        for i, k_p in enumerate(active):
            i_lo = max(0, k_p - (n - 1))
            i_hi = min(n - 1, k_p)
            if i_lo > i_hi:
                continue
            a_lo = k_p - i_hi
            a_hi = k_p - i_lo
            C[i_lo:i_hi + 1, i] = a[a_lo:a_hi + 1][::-1]
        
        rhs = np.ones(n)
        try:
            y, _, _, _ = np.linalg.lstsq(C, rhs, rcond=None)
        except np.linalg.LinAlgError:
            break
        
        y = np.maximum(y, 0)
        if y.sum() < 1e-15:
            break
        
        # Compute exact subgradient
        grad = conv_grad_active(a, active, y)
        
        # Projected gradient step with backtracking
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        improved = False
        for alpha in [0.01, 0.005, 0.001, 0.0005, 0.0001]:
            a_new = np.maximum(a - alpha * grad, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            J_new = compute_J(a_new)
