from scipy.optimize import linprog
from scipy.sparse import csr_matrix, diags
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(n, a):
    """Build the sparse convolution matrix for conv(a, .)"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                             'primal_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

    # For each row k (0 to 2n-2), the non-zero entries are:

    cols = []

def resample_seq(src, n):
    """Resample a sequence to length n using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gs_spectral_init(a, n_iters=200, beta=0.85):
    """Gerchberg-Saxton initialization to flatten the power spectrum."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_min = mag.min()
        mag_max = mag.max()
        if mag_max - mag_min < 1e-10:
            break
        # Compress dynamic range
        mag_norm = (mag - mag_min) / (mag_max - mag_min + 1e-15)
        mag_new = mag_min + (mag_max - mag_min) * (1.0 - beta) + \
                  beta * (mag_max - mag_min) * mag_norm ** 0.3
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        sc = a_new.sum()
        if sc < 1e-15:
            break
        a_new /= sc
        M_c = fast_conv(a_new).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_new.copy()
        a = a_new
    return best_a


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(1024)
        best.update(parent)
        print(f"No parent found, using uniform n=1024, J={best.score:.6f}", flush=True)

    n_parent = parent.size

    # === PHASE 1: LP fixed-point on parent length ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: LP Fixed-Point on parent (n={n_parent})", flush=True)
    print(f"{'=' * 60}", flush=True)
    p1_start = time.monotonic()

    a_p1 = parent.copy()
    a_p1 = np.maximum(a_p1, 1e-6)
    a_p1 /= a_p1.sum()

    # Quick gradient warmup to get out of any bad spots
    p1_warmup_deadline = min(time.monotonic() + 60, deadline - 600)
    a_p1 = gradient_descent_warmup(a_p1, p1_warmup_deadline, max_iters=200,
                                    tau_start=5.0, tau_end=50.0)
    best.update(a_p1)

    # LP fixed-point iteration (main workhorse)
    p1_lp_deadline = deadline - 300
    a_p1 = lp_fixed_point(a_p1, p1_lp_deadline, max_iters=5000, verbose=True)
    best.update(a_p1)

    # Aitken acceleration pass
    if time.monotonic() < deadline - 100:
        a_p1_ait = lp_fixed_point_aitken(a_p1, deadline - 80, max_iters=300)
        best.update(a_p1_ait)
        if best.seq is not None and best.score < compute_J(a_p1):
            a_p1 = best.seq.copy()

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic() - p1_start:.1f}s", flush=True)

    # === PHASE 2: Multi-length exploration ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)
    p2_start = time.monotonic()

    # Try lengths around the parent and some larger lengths
    lengths_to_try = []
    for delta in [-64, -32, -16, 16, 32, 64, 128, 256]:
        n_try = n_parent + delta
        if 256 <= n_try <= 3000:
            lengths_to_try.append(n_try)
    # Also try some specific larger lengths
    for n_try in [1280, 1536, 1792, 2048, 2560]:
        if n_try not in lengths_to_try and n_try >= 512:
            lengths_to_try.append(n_try)

    lengths_to_try = sorted(set(lengths_to_try))
    print(f"Lengths to try: {lengths_to_try[:10]}...", flush=True)

    # Allocate time based on expected LP cost
    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        # Estimate time per LP solve
        est_lp_time = 0.02 * (n_try / 1024.0) ** 2
        # Budget for this length: enough for ~200-500 LP iterations
        n_iters_est = min(500, max(50, int(60.0 / est_lp_time)))
        time_budget = est_lp_time * n_iters_est * 1.5 + 30  # + warmup
        # Cap at available time
        remaining = deadline - time.monotonic() - 30
        time_budget = min(time_budget, remaining * 0.4)
        if time_budget < 30:
            continue

        print(f"\n  Trying n={n_try} (budget={time_budget:.0f}s, est {n_iters_est} LP iters)", flush=True)
        t0 = time.monotonic()

        # Initialize from parent (resampled)
        a_init = resample_seq(best.seq if best.seq is not None else parent, n_try)
        a_init = np.maximum(a_init, 1e-6)
        a_init /= a_init.sum()

        # GS spectral refinement
        a_init = gs_spectral_init(a_init, n_iters=100, beta=0.85)

        # Gradient warmup
        sub_deadline = time.monotonic() + min(30, time_budget * 0.2)
        a_init = gradient_descent_warmup(a_init, sub_deadline, max_iters=150,
                                         tau_start=5.0, tau_end=50.0)
        best.update(a_init)

        # LP fixed-point
        lp_deadline = time.monotonic() + min(time_budget * 0.7, remaining - 40)
        a_opt = lp_fixed_point(a_init, lp_deadline, max_iters=n_iters_est)
        best.update(a_opt)

        print(f"    n={n_try}: J={compute_J(a_opt):.8f} ({time.monotonic() - t0:.1f}s)", flush=True)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic() - p2_start:.1f}s", flush=True)

    # === PHASE 3: Perturbation and re-optimize ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Perturbation-based diversification", flush=True)
    print(f"{'=' * 60}", flush=True)
    p3_start = time.monotonic()

    if best.seq is not None and time.monotonic() < deadline - 60:
        n_best = best.seq.size
        perturbation_sizes = [0.01, 0.03, 0.05, 0.08, 0.12]
        for ps in perturbation_sizes:
            if time.monotonic() > deadline - 40:
                break
            a_pert = best.seq + rng.normal(0, ps, n_best)
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert /= a_pert.sum()

            # Quick LP polish
            sub_dl = min(time.monotonic() + 60, deadline - 35)
            a_pol = lp_fixed_point(a_pert, sub_dl, max_iters=200)
            best.update(a_pol)

            if best.score < 1.503:
                print(f"  Pert ps={ps}: J={best.score:.8f} - below 1.503!", flush=True)
                break

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic() - p3_start:.1f}s", flush=True)

    # === PHASE 4: Final intensive LP on best found ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Final intensive LP optimization", flush=True)
    print(f"{'=' * 60}", flush=True)
    p4_start = time.monotonic()

    if best.seq is not None and time.monotonic() < deadline - 10:
        a_final = best.seq.copy()
        a_final = np.maximum(a_final, 1e-6)
        a_final /= a_final.sum()
        a_final = lp_fixed_point(a_final, deadline - 5, max_iters=3000)
        best.update(a_final)

        # One more Aitken pass
        if time.monotonic() < deadline - 5:
            a_final2 = lp_fixed_point_aitken(a_final, deadline, max_iters=200)
            best.update(a_final2)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic() - p4_start:.1f}s", flush=True)

    # === PHASE 5: Try slightly different normalizations/scales ===
    if best.seq is not None and time.monotonic() < deadline - 5:
        # Sometimes a small scale change + re-optimization helps
        for scale in [0.9, 0.95, 1.05, 1.1]:
            if time.monotonic() > deadline - 3:
                break
            a_scaled = best.seq * scale
            a_scaled = np.maximum(a_scaled, 1e-6)
            a_scaled /= a_scaled.sum()
            sub_dl = min(time.monotonic() + 20, deadline - 2)
            a_opt = lp_fixed_point(a_scaled, sub_dl, max_iters=200)
            best.update(a_opt)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic() - best.t0:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)
    return [float(x) for x in best.seq]
