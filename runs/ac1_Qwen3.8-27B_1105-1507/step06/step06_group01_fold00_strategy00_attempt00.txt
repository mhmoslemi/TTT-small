def propose_candidate(budget=1500, seed=42):
    import numpy as np
    import time
    
    # If parent sequence exists, use it as a base
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        parent = None
    
    deadline = time.time() + budget
    tracker = ScoreTracker()
    best_sequence = None
    best_score = float('inf')
    
    # Set default base functions if no parent
    if parent is None:
        base_functions = ["linear", "quadratic", "polynomial", "threshold"]
    else:
        base_functions = [parent.tolist()]
    
    # Allocate time to phases
    time_phases = {
        "initialization": budget * 0.15,
        "fista": budget * 0.25,
        "mirror_descent": budget * 0.25,
        "lp_polish": budget * 0.15,
        "coordinate_descent": budget * 0.10
    }
    
    # Show progress and warnings
    def print_progress(label, time_used, time_remaining):
        elapsed = time.time() - deadline + budget - time_used
        time_left = time_remaining
        print(f"\n{label}: elapsed={elapsed:.0f}s ofbudget={time_left:.0f}s", flush=True)
        print("\tStarting newiteration", flush=True)
    
    # Build diverse initial sequences
    inits = []
    for base in base_functions:
        if base is not None:
            # Act aggressively on parent by extending its strengths
            n0 = 2 if parent is None else parent.size
            
            # Phase 1: Initialization - scaled by time allocation
            if time_phases["initialization"] > 0:
                # Uniform initialization
                a_uniform = np.ones(n0) / n0
                tracker.update(a_uniform)
                inits.append((a_uniform, "uniform", time_left=time_phases["initialization"]))
                
                # Polynomial (parabolic) initialization
                x = np.linspace(0, 1, n0)
                a_poly = (x*(1-x)) + 0.05
                tracker.update(a_uniform)
                inits.append((a_poly, "polynomialsame", time_left=time_phases["initialization"]))
                
                # Sine wave initialization (oscillatory)
                freqs = [0.1, 0.2, 0.3, 0.4]
                for freq in freqs:
                    a_sine = 0.5 + 0.5*np.cos(2*np.pi*freq*x)
                    tracker.update(a_sine)
                    inits.append((a_sine, "sinefreq={}".format(freq), time_left=time_phases["initialization"]))
                
                # Threshold-based initialization (staircase)
                thresholds = [int(n0/4) for _ in range(4)]
                for step in thresholds:
                    a_threshold = np.zeros(n0)
                    for i in range(n0):
                        if x[i] < step:
                            a_threshold[i] = 0.1
                        else:
                            a_threshold[i] = min(0.9, 0.1+0.5*(1/i)) if i!=0 else 0
                    tracker.update(a_threshold)
                    inits.append((a_threshold, "thresholdstep={}".format(step), time_left=time_phases["initialization"]))

    # Sort inits by expected improvement (heuristic from previous program)
    inits.sort(key=lambda x: x[1], reverse=True)  # This is a placeholder - actual sorting based on objective would be better
    
    # Optimization phases with explicit time checks
    optimal_length = 1024 if parent is None else parent.size
    length_candidates = [optimal_length//2, optimal_length, optimal_length*2]
    length_candidates = [int(x) for x in length_candidates if 256<=x<=2048]
    scores = []
    for n_val in length_candidates:
        time_start = time.time()
        for init_vector, init_type, time_left in inits:
            time_used_init = time.time() - time_start
            if time_used_init > time_left*0.5:  # Twice the time of initialization phase
                continue
            candidate = refit_sequence(init_vector, n_val)
            try:
                score = evaluate_sequence(candidate)
                if score < best_score:
                    best_score = score
                    best_sequence = candidate
                    tracker.update(candidate)
            except Exception as e:

                continue

    if best_sequence is None:
        uniform_512 = np.ones(512)/512
        tracker.update(uniform_512)
        best_sequence = uniform_512
        best_score = evaluate_sequence(best_sequence)
    
    return best_sequence
