def fast_conv(a):

def run_lp_mirror(a, deadline, max_iters=200, active_thresh=0.85):

    best_a = a.copy()
    best_M = float(fast_conv(a).max())

            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
        
        # Active set: lags where b is close to M
        active = np.where(b >= active_thresh * M)[0].tolist()
        g = lp_best_response(a, M, active)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Line search
        best_t = 0
        best_M_t = M
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_t = np.maximum((1-t)*a + t*g, 0)
            st = a_t.sum()
            if st < 1e-10:

            a_t /= st
            Mt = float(fast_conv(a_t).max())
            if Mt < best_M_t:
                best_M_t = Mt
                best_t = t
        if best_t == 0:
            break
        a = np.maximum((1-best_t)*a + best_t*g, 0)
        a /= a.sum()
    
    return best_a

def gs_flat(a, n_iters=200, beta=0.8):
    """Gerchberg-Saxton spectral flattening."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_target = np.ones_like(mag) * mag.mean()
        mag_new = (1-beta)*mag + beta*mag_target
        F_new = np.exp(1j*phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        sc = a_new.sum()
        if sc < 1e-10:
            break
        a_new /= sc
        M_new = float(fast_conv(a_new).max())
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new
    return best_a

Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set QP via cvxpy (exact convex optimization) - the key innovation
2. LP mirror descent with active-set constraint reduction for fast iteration

4. Multi-length exploration with adaptive time budget

The active-set QP directly solves the convex problem:
  minimize t
  subject to (a*a)[k] <= t for all k, sum(a)=1, a>=0
which guarantees finding the global optimum (unlike the LP fixed-point method).

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_Q_k(n, k):
    """Build sparse symmetric PSD matrix Q_k such that a^T Q_k a = (a*a)[k]."""
    rows = []
    cols = []
    vals = []
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    for i in range(i_min, i_max + 1):
        j = k - i
        if i == j:
            rows.append(i)
            cols.append(i)
            vals.append(1.0)
        elif i < j:
            rows.append(i)
            cols.append(j)
            vals.append(0.5)
            rows.append(j)
            cols.append(i)
            vals.append(0.5)
    Q = csr_matrix((vals, (rows, cols)), shape=(n, n))
    return Q

    """
    Solve the convex QP using active-set method.
    minimize t s.t. (a*a)[k] <= t for active k, sum(a)=1, a>=0

    a_init = np.maximum(a_init.astype(np.float64), 0)
    s = a_init.sum()
    if s < 1e-10:
        return None
    a_init = a_init / s
    n = len(a_init)

    best_t = t0
    best_J = compute_J(a_init)

            obj = cp.Minimize(t_var)

                best_t = actual_max

                # Add most violated constraints
                idx = np.where(violations > 1e-9)[0]
                for k in sorted(idx.tolist(), key=lambda x: -violations[x])[:5]:
                    if k not in active_lags:
                        active_lags.append(k)
                active_lags = sorted(set(active_lags))
            else:
                # The solver's solution doesn't satisfy the active constraints perfectly
                # This can happen with inexact solvers; add the worst violators
                idx = np.where(b_new > t_new * 1.001)[0]
                for k in sorted(idx.tolist(), key=lambda x: -b_new[x])[:5]:
                    if k not in active_lags:
                        active_lags.append(k)
                active_lags = sorted(set(active_lags))
                
        except Exception as e:
            if verbose:
                print(f"    QP iter {it} error: {e}", flush=True)
            break
    
    return best_a
