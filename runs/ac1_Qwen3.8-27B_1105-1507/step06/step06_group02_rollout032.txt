import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def softmax(x):
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    return exp_x / exp_x.sum()

def cross_corr_fft(p, a):
    """Compute g[i] = sum_{m=0}^{n-1} p[i+m] * a[m] for i=0,...,n-1 using FFT."""
    n = len(a)
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    fp = np.fft.rfft(p, L)
    fa = np.fft.rfft(a, L)
    corr = np.fft.irfft(fp * np.conj(fa), L)
    return corr[:n]

def obj_and_grad_x(x, tau, n):
    """Objective and gradient w.r.t. x (softmax parameterization)."""
    a = softmax(x)
    b = fast_conv(a)
    # Numerically stable logsumexp
    b_max = b.max()
    shifted = tau * (b - b_max)
    log_sum_exp = b_max + np.log(np.exp(shifted).sum()) / tau
    f = 2.0 * n * log_sum_exp
    
    # p_k = exp(tau * b_k) / sum(exp(tau * b_k))
    p = np.exp(shifted) / np.exp(shifted).sum()
    
    # grad_f w.r.t. a: (2/tau) * cross_corr(p, a)
    g = cross_corr_fft(p, a)
    grad_a = 2.0 / tau * g
    
    # grad w.r.t. x: a * (grad_a - dot(a, grad_a))
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    
    return f, grad_x

def run_lbfgs_continuation(a_init, tau_values, max_iter_per_stage, deadline):
    """Run L-BFGS with continuation on tau."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / a_init.sum()
    
    # Initialize x from a_init: x_i = log(a_i)
    x0 = np.log(a_init + 1e-30)
    
    best_a = a_init.copy()
    best_f = float('inf')
    
    for tau in tau_values:
        if time.monotonic() > deadline:
            break
        # Use current a as initialization
        a_cur = softmax(x0)
        result = minimize(
            lambda x: obj_and_grad_x(x, tau, n)[0],
            x0,
            jac=lambda x: obj_and_grad_x(x, tau, n)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter_per_stage, 'gtol': 1e-8, 'maxls': 50}
        )
        x0 = result.x
        a_cur = softmax(x0)
        f_cur = 2.0 * n * (fast_conv(a_cur).max())  # actual (non-smooth) objective
        if f_cur < best_f:
            best_f = f_cur
            best_a = a_cur.copy()
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Direct L-BFGS-B with tau-continuation on the smooth max approximation
2. Gerchberg-Saxton spectral initialization for diversity
3. Anderson-accelerated LP mirror descent for high-precision polish
4. Multi-start with diverse initializations, length exploration, and restarts
Key improvement: second-order L-BFGS converges much faster near optimum vs first-order.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_corr_fft(p, a):
    """g[i] = sum_{m=0}^{n-1} p[i+m] * a[m] for i=0..n-1 via FFT."""
    n = len(a)
    L = 1
    while L < 3 * n:
        L <<= 1
    fp = np.fft.rfft(p, L)
    fa = np.fft.rfft(a, L)
    corr = np.fft.irfft(fp * np.conj(fa), L)
    return corr[:n]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def obj_grad_a(a, tau):
    """Smooth objective and gradient for L-BFGS-B on a directly."""
    a = np.maximum(a, 0.0)
    n = len(a)
    s = float(a.sum())
    if s < 1e-12:
        return 0.0, np.zeros(n)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    L_val = (b_max + np.log(sum_exp)) / tau
    f = 2.0 * n * L_val / (s * s)
    p = exp_shifted / sum_exp
    grad_L = (2.0 / tau) * cross_corr_fft(p, a)
    grad_f = 2.0 * n * (s * grad_L - 2.0 * L_val) / (s * s * s)
    return f, grad_f


def run_lbfgs_continuation(a_init, deadline, tau_list=None, max_iter=200):
    """L-BFGS-B with continuation on tau. Returns best a found."""
    a = np.maximum(a_init.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    if tau_list is None:
        tau_list = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0, 5000.0]

    best_a = a.copy()
    best_J = compute_J(a)

    for tau in tau_list:
        if time.monotonic() > deadline:
            break
        if time.monotonic() > deadline - 5.0:
            break
        # Use current a as starting point
        try:
            res = minimize(
                lambda x: obj_grad_a(x, tau)[0],
                a.copy(),
                jac=lambda x: obj_grad_a(x, tau)[1],
                method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': max_iter, 'gtol': 1e-10, 'maxls': 30,
                         'ftol': 1e-14}
            )
            a_new = np.maximum(res.x, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new /= s
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                a = a_new
        except Exception as e:
            print(f"    L-BFGS error at tau={tau}: {e}", flush=True)
            break
        print(f"    tau={tau:.0f}: J={best_J:.8f} ({time.monotonic():.0f}s elapsed)", flush=True)

    return best_a


def run_lbfgs_multi_tau(a_init, deadline, tau_stages=16):
    """More aggressive L-BFGS with many tau stages and restarts."""
    a = np.maximum(a_init.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    # Log-spaced tau from 1 to 20000
    tau_list = np.logspace(0, np.log10(20000), tau_stages)

    for tau in tau_list:
        if time.monotonic() > deadline - 10.0:
            break
        for attempt in range(2):
            if time.monotonic() > deadline - 10.0:
                break
            start = a.copy() if attempt == 0 else np.maximum(a + 0.01 * np.random.randn(n), 1e-10)
            start /= start.sum()
            try:
                res = minimize(
                    lambda x: obj_grad_a(x, tau)[0],
                    start,
                    jac=lambda x: obj_grad_a(x, tau)[1],
                    method='L-BFGS-B',
                    bounds=[(0.0, None)] * n,
                    options={'maxiter': 150, 'gtol': 1e-10, 'maxls': 20,
                             'ftol': 1e-15}
                )
                a_new = np.maximum(res.x, 0.0)
                s = a_new.sum()
                if s > 1e-10:
                    a_new /= s
                    J_new = compute_J(a_new)
                    if J_new < best_J:
                        best_J = J_new
                        best_a = a_new.copy()
                    a = a_new
            except Exception:
                break
    return best_a


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: flatten power spectrum while maintaining non-negativity."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        # Compress dynamic range
        power = 0.4 + 0.3 * beta
        mag_new = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0.0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: (a*g)(k) <= M for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(history_x, history_Fx, m):
    """Anderson mixing: find alpha minimizing ||sum alpha_i * r_i||^2 s.t. sum alpha = 1."""
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = float(ones @ G_inv_ones)
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -2.0, 2.0)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def run_lp_anderson(a, deadline, max_iters=150, window=5, t_base=0.3, min_t=0.005):
    """LP mirror descent with Anderson acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    hist_x = []
    hist_Fx = []

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break

        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-12:
            break
        g = g / gs

        t = max(min_t, t_base / (1.0 + 0.03 * it))
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > window:
            hist_x.pop(0)
            hist_Fx.pop(0)

        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and = a_and / sa
                M_and = float(fast_conv(a_and).max())
                M_new = float(fast_conv(a_new).max())
                if M_and <= M_new * 1.02:
                    a = a_and
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new

    return best_a


def run_lp_linesearch(a, deadline, max_iters=80):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-12:
            break
        g = g / gs

        improved = False
        for t in [0.4, 0.25, 0.15, 0.1, 0.05, 0.03, 0.01, 0.005]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = float(fast_conv(a_t).max())
            if M_t < best_M - 1e-15:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initializations for length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.005))
        # GS from parent
        inits.append(gs_flat_spectral(p, n_iters=150, beta=0.9))
    inits.append(np.ones(n) / n)
    # Power-law shapes
    for alpha in [0.3, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t ** alpha + (1 - t) ** alpha) / 2, 0.005))
    # Cosine
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.005))
    # Gaussian bumps
    inits.append(np.exp(-6 * (t - 0.5) ** 2) + 0.05)
    inits.append(np.exp(-10 * (t - 0.3) ** 2) + np.exp(-10 * (t - 0.7) ** 2) + 0.03)
    # Random smooth
    k = max(3, n // 15)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_flat_spectral(a_rand, n_iters=120, beta=0.85))
    return inits

    # === PHASE 6: Quick L-BFGS polish after LP ===
    if best.seq is not None and time.monotonic() < deadline - 10:
        a_final = run_lbfgs_continuation(best.seq, deadline,
                                         tau_list=[100, 500, 2000, 10000],
                                         max_iter=100)
        best.update(a_final)
        # One more LP
        if time.monotonic() < deadline - 5:
            a_final2 = run_lp_anderson(a_final, deadline, max_iters=20,
                                       window=4, t_base=0.15, min_t=0.002)
            best.update(a_final2)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(256)
        best.score = compute_J(best.seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-start_time:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best.seq]
