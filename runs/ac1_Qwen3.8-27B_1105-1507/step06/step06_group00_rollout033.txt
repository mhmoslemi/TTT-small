from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses progressive soft-max gradient descent, LP alternating, 
chase-peak subgradient, and coordinate refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def softmax_gradient_descent(a, betas, iters_per_beta, lr, momentum=0.9, time_budget=60.0):
    """Progressive soft-max gradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)
    start = time.monotonic()
    
    for beta in betas:
        for it in range(iters_per_beta):
            if time.monotonic() - start > time_budget:
                return best_a
            
            conv = compute_conv(a)
            max_conv = float(np.max(conv))
            
            if beta < 500:
                # Soft gradient
                w = np.exp(beta * (conv - max_conv))
                grad = 2.0 * beta * np.correlate(w, a, mode='valid')
            else:
                # Hard subgradient
                k_star = int(np.argmax(conv))
                grad = np.zeros(n)
                for m in range(n):
                    j = k_star - m
                    if 0 <= j < n:
                        grad[m] = 2.0 * a[j]
            
            # Project to sum=0
            grad = grad - np.mean(grad)
            
            # Nesterov momentum
            v = momentum * v + grad
            a_new = a - lr * v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
                # Adaptive lr
                if it % 10 == 0:
                    lr = max(0.001, lr * 0.99)
    
    return best_a

def chase_peak(a, max_iters=500, lr_init=0.02, time_budget=30.0):
    """Chase-the-peak subgradient method with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    lr = lr_init
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient
        grad = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                grad[m] = 2.0 * a[j]
        
        # Project
        grad = grad - np.mean(grad)
        
        # Multiple step sizes
        improved = False
        for lr_try in [lr, lr * 0.5, lr * 0.1, lr * 0.05]:
            a_new = a - lr_try * grad
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    lr = lr_try
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            lr *= 0.7
            if stall > 50:
                break
    
    return best_a

    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)

        if time.monotonic() - start > time_budget:

        if g_lp is None:

        best_t_val = 0.5
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            a_try = (1 - best_t_val) * a + best_t_val * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
            best_a = a_try.copy()
            a = a_try
        else:
            # Perturb
            rng = np.random.default_rng(it * 7919 + n)

            a = np.maximum(a, 0)

def coordinate_refine(a, time_budget=20.0):
    """Coordinate descent targeting peak reduction."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 0.01:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False
        
        # Try moving mass from peak contributors to low-conv positions
        contrib = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                contrib[m] = a[j]
        
        # Source: high contribution positions
        # Target: positions where conv is low
        conv_norm = conv / (np.max(conv) + 1e-15)
        
        for src in np.argsort(contrib)[-10:][::-1]:
            for tgt in np.argsort(conv_norm)[:10]:
                if src == tgt:
                    continue
                for frac in [0.01, 0.03, 0.05]:
                    delta = frac * a[src]
                    a[src] -= delta
                    a[tgt] += delta
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[tgt] -= delta
                        break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def random_init(n, rng):
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Smooth
        freq = rng.uniform(0.5, 3.0)
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Two-frequency
        f1, f2 = rng.uniform(0.5, 2.0), rng.uniform(2.0, 5.0)
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.cos(2*np.pi*f1*x) + 0.2 * np.cos(2*np.pi*f2*x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Exponential taper
        x = np.arange(n)
        center = n / 2
        a = np.exp(-0.5 * ((x - center) / (n * 0.3)) ** 2)
        a = a + 0.1
        a = a / a.sum()
    else:
        # Asymmetric
        x = np.arange(n)
        a = np.exp(-x / (n * 0.4))
        a = a + 0.1
        a = a / a.sum()
    return a

Wait, I need to double-check my gradient computation. Let me verify:

(a*a)[k] = sum_{i: valid} a[i] * a[k-i

The derivative with respect to a[m]:
- a[m] appears in the sum when i = m (giving a[m]*a[k-m]) or when k-i = m (i.e., i = k-m, giving a[k-m]*a[m])
- These are the same term when m = k-m, i.e., 2m = k

For the soft-max objective f(a) = sum_k exp(beta * (a*a)[k]):
df/da[m] = sum_k exp(beta*(a*a)[k]) * beta * d(a*a)[k]/da[m]
= sum_k exp(beta*(a*a)[k]) * beta * 2*a[k-m] (for valid k-m)
= 2*beta * sum_k w[k] * a[k-m] where w[k] = exp(beta*(a*a)[k])

Let j = k-m: = 2*beta * sum_j w[j+m] * a[j]

So df/da[m] = 2*beta * sum_j a[j] * w[j+m] for m = 0, ..., n-1

This is 2*beta * correlate(w, a, 'valid')[m] where w has length 2n-1 and a has length n.

np.correlate(w, a, 'valid') gives:

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def soft_gradient_descent(a, betas, iters_per_beta, lr_init, momentum, time_budget):
    """Progressive soft-max gradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)
    lr = lr_init
    start = time.monotonic()

    for beta in betas:
        for it in range(iters_per_beta):
            if time.monotonic() - start > time_budget:
                return best_a

            conv = compute_conv(a)
            max_conv = float(np.max(conv))

            if beta < 500:
                shifted = beta * (conv - max_conv)
                max_shift = np.max(shifted)
                w = np.exp(shifted - max_shift)
                p = w / np.sum(w)
                grad = 4.0 * n * np.correlate(p, a, mode='valid')
            else:
                k_star = int(np.argmax(conv))
                grad_arr = np.zeros(n)
                for m in range(n):
                    j = k_star - m
                    if 0 <= j < n:
                        grad_arr[m] = 2.0 * a[j]
                grad = 2.0 * n * grad_arr

            # Project onto sum=0
            grad = grad - np.mean(grad)

            # Nesterov momentum
            v = momentum * v + grad
            a_new = a - lr * v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
            else:
                v *= 0.5
                lr *= 0.5

            # Decay
            if it % 20 == 19:
                lr *= 0.95
                lr = max(lr, 0.0001)

    return best_a


def chase_peak(a, max_iters, lr_init, time_budget):
    """Chase-the-peak subgradient with adaptive step and multi-peak awareness."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    lr = lr_init
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Find all near-max positions (within 1% of max) for smoother gradient
        threshold = max_conv * 0.99
        peak_indices = np.where(conv >= threshold)[0]

        # Compute combined subgradient from all peak positions
        grad = np.zeros(n)
        for k_star in peak_indices:
            k_star = int(k_star)
            for m in range(n):
                j = k_star - m
                if 0 <= j < n:
                    grad[m] += 2.0 * a[j]
        grad /= len(peak_indices)

        # Scale to match objective gradient (2n * ...)
        grad *= 2.0 * n

        # Project to sum=0
        grad = grad - np.mean(grad)

        # Try multiple step sizes
        improved = False
        for lr_try in [lr, lr * 0.5, lr * 0.2, lr * 0.1]:
            a_new = a - lr_try * grad
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    lr = lr_try
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            lr *= 0.8
            if stall > 80:
                # Perturb to escape
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
                lr = lr_init * 0.5

    return best_a


def lp_alternating(a, max_iters, time_budget):
    """LP alternating with adaptive blending and perturbation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Line search with more candidates
        best_t_obj = best_obj
        best_t_val = 0.5
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            a = (1.0 - best_t_val) * a + best_t_val * g_lp
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
        else:
            # Perturb
            rng = np.random.default_rng(it * 7919 + n + 1)
            scale = 0.03 + 0.01 * min(it // 10, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s

    return best_a


def coordinate_refine(a, time_budget):
    """Targeted coordinate descent: move mass from peak contributors to valleys."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 0.01:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Contribution to peak
        contrib = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                contrib[m] = a[j]

        # Valley positions (low convolution)
        conv_min = float(np.min(conv))
        conv_range = float(np.max(conv)) - conv_min + 1e-15
        valley_score = 1.0 - (conv - conv_min) / conv_range

        # Sort sources (high contrib) and targets (high valley)
        src_order = np.argsort(contrib)[-8:][::-1]
        tgt_order = np.argsort(valley_score)[:8]

        for src in src_order:
            for tgt in tgt_order:
                if src == tgt:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    delta = frac * a[src]
                    if delta < 1e-10:
                        break
                    a[src] -= delta
                    a[tgt] += delta
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[tgt] -= delta
                        break
            if improved:
                break

        if not improved:
            # Try smaller fracs
            for src in src_order[:3]:
                for tgt in tgt_order[:3]:
                    if src == tgt:
                        continue
                    a[src] *= 0.95
                    a[tgt] *= 1.95
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_new = np.linspace(0, 1, new_n)
        return np.maximum(a[0] + (a[1] - a[0]) * x_new, 0)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random non-negative sequences."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / max(a.sum(), 1e-10)
    elif mode == 2:
        # Single cosine modulation
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) / n
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Multi-frequency
        x = np.arange(n) / n
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.3)
            a += amp * np.cos(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Asymmetric exponential
        x = np.arange(n)
        rate = rng.uniform(0.5, 3.0)
        a = np.exp(-rate * np.abs(x - n * rng.uniform(0.2, 0.8)) / n)
        a = a + 0.05
        a = a / a.sum()
    elif mode == 5:
        # Piecewise linear
        breakpoints = np.sort(rng.uniform(0, 1, rng.integers(3, 8)))
        values = rng.uniform(0.1, 1.0, len(breakpoints))
        a = np.interp(np.linspace(0, 1, n), breakpoints, values)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Chirp-like
        x = np.arange(n) / n
        chirp_rate = rng.uniform(1.0, 5.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * chirp_rate * x**2)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        # Smooth random walk
        steps = rng.normal(0, 0.3, n)
        a = np.cumsum(steps)
        a = a - np.min(a) + 0.1
        a = a / a.sum()
    return a


def anneal_and_cool(a, time_budget, rng):
    """Heat up (add noise) then cool down (optimize) to escape local minima."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    temps = [0.02, 0.05, 0.1, 0.15, 0.1, 0.05, 0.02]
    for temp in temps:
        if time.monotonic() - start > time_budget:
            break
        a_heated = a * np.exp(rng.normal(0, temp, n))
        a_heated = np.maximum(a_heated, 0)
        a_heated = a_heated / a_heated.sum()

        # Cool with a few gradient steps
        a_cooled = soft_gradient_descent(a_heated, [10.0, 50.0],
                                         iters_per_beta=20, lr_init=0.005,
                                         momentum=0.9,
                                         time_budget=min(10.0, (time_budget - (time.monotonic() - start)) * 0.5))
        a_cooled = chase_peak(a_cooled, max_iters=100, lr_init=0.005,
                              time_budget=min(5.0, (time_budget - (time.monotonic() - start)) * 0.5))
        obj = compute_obj(a_cooled)
        if obj < best_obj:
            best_obj = obj
            best_a = a_cooled.copy()
            a = a_cooled.copy()

    return best_a

        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # Phase 1: Soft-max gradient descent (progressive beta)
    print(f"\n=== Phase 1: Soft-max GD (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(remaining * 0.15, 150.0)
        betas = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0]
        a1 = soft_gradient_descent(best_seq, betas, iters_per_beta=40,
                                   lr_init=0.008, momentum=0.9, time_budget=alloc)
        obj1 = compute_obj(a1)
        print(f"  Soft-max GD: obj={obj1:.8f}")
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  IMPROVED to {best_score:.8f}")

    # Phase 2: Chase peak
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 2: Chase Peak ===")
        alloc = min(remaining * 0.1, 80.0)
        a2 = chase_peak(best_seq, max_iters=2000, lr_init=0.008, time_budget=alloc)
        obj2 = compute_obj(a2)
        print(f"  Chase peak: obj={obj2:.8f}")
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  IMPROVED to {best_score:.8f}")

    # Phase 3: LP Alternating (multiple rounds)
    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        print(f"\n=== Phase 3.{round_idx}: LP Alternating ===")
        alloc = min(remaining * 0.18, 120.0)
        a3 = lp_alternating(best_seq, max_iters=300, time_budget=alloc)
        obj3 = compute_obj(a3)
        print(f"  LP round {round_idx}: obj={obj3:.8f}")
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  IMPROVED to {best_score:.8f}")
        else:
            # Anneal after LP stall
            if remaining > 60:
                a_ann = anneal_and_cool(best_seq, time_budget=min(remaining * 0.1, 40.0), rng=rng)
                obj_ann = compute_obj(a_ann)
                if obj_ann < best_score:
                    best_score = obj_ann
                    best_seq = a_ann.copy()
                    print(f"  Anneal: IMPROVED to {best_score:.8f}")

    # Phase 4: Chase peak again
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Chase Peak 2 ===")
        alloc = min(remaining * 0.08, 50.0)
        a4 = chase_peak(best_seq, max_iters=1000, lr_init=0.004, time_budget=alloc)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  IMPROVED to {best_score:.8f}")
        else:
            print(f"  obj={obj4:.8f}")

    # Phase 5: Coordinate refinement
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 5: Coordinate Refine ===")
        alloc = min(remaining * 0.08, 30.0)
        a5 = coordinate_refine(best_seq, time_budget=alloc)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  IMPROVED to {best_score:.8f}")
        else:
            print(f"  obj={obj5:.8f}")

    # Phase 6: Multi-n exploration
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Multi-n Exploration ===")
        n_options = sorted(set([
            n_parent - 75, n_parent - 50, n_parent - 25,
            n_parent, n_parent + 25, n_parent + 50, n_parent + 75,
            n_parent + 150, 500, 600, 700, 900, 1000, 1100
        ]))
        n_options = [n for n in n_options if 300 <= n <= 1300]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 35:
                break
            if n_try == n_parent:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)

            alloc = min(remaining * 0.2, 50.0)
            # Quick pipeline: soft GD + chase
            a_opt = soft_gradient_descent(a_try, [10.0, 50.0, 200.0],
                                          iters_per_beta=25, lr_init=0.005,
                                          momentum=0.9, time_budget=alloc * 0.5)
            a_opt = chase_peak(a_opt, max_iters=300, lr_init=0.005,
                               time_budget=alloc * 0.5)
            obj_try = compute_obj(a_opt)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 7: Random restarts
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Random Restarts ===")
        n_best = len(best_seq)
        n_restarts = min(12, max(4, int(remaining / 25)))

        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            if r % 4 == 0:
                a_r = random_init(n_best, rng)
            elif r % 4 == 1:
                a_r = best_seq * np.exp(rng.normal(0, 0.04, n_best))
                a_r = np.maximum(a_r, 0)
                a_r = a_r / a_r.sum()
            elif r % 4 == 2:
                # Reversed
                a_r = best_seq[::-1].copy()
                a_r = a_r * np.exp(rng.normal(0, 0.02, n_best))
                a_r = np.maximum(a_r, 0)
                a_r = a_r / a_r.sum()
            else:
                # Different n
                n_r = n_best + int(rng.integers(-100, 100))
                n_r = max(300, min(1300, n_r))
                a_r = resample_sequence(best_seq, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.03, len(a_r)))
                a_r = np.maximum(a_r, 0)
                a_r = a_r / a_r.sum()

            alloc = min(remaining * 0.2, 25.0)
            a_opt = soft_gradient_descent(a_r, [5.0, 20.0, 100.0],
                                          iters_per_beta=20, lr_init=0.004,
                                          momentum=0.9, time_budget=alloc * 0.5)
            a_opt = chase_peak(a_opt, max_iters=200, lr_init=0.004,
                               time_budget=alloc * 0.5)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r} (n={len(a_opt)}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # Phase 8: Final intensive polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 8: Final Polish ===")

        # LP
        alloc = min(remaining * 0.35, 100.0)
        if alloc > 15:
            a_final = lp_alternating(best_seq, max_iters=200, time_budget=alloc)
            obj_f = compute_obj(a_final)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.8f}")
            else:
                print(f"  Final LP: obj={obj_f:.8f}")

        # Chase peak with small steps
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.25, 40.0)
            a_final = chase_peak(best_seq, max_iters=1000, lr_init=0.002,
                                 time_budget=alloc)
            obj_f = compute_obj(a_final)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_final.copy()
                print(f"  Final Chase: obj={best_score:.8f}")
            else:
                print(f"  Final Chase: obj={obj_f:.8f}")

        # Soft-max with very high beta
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.25, 30.0)
            a_final = soft_gradient_descent(best_seq, [200.0, 500.0, 2000.0],
                                            iters_per_beta=40, lr_init=0.001,
                                            momentum=0.95, time_budget=alloc)
            obj_f = compute_obj(a_final)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_final.copy()
                print(f"  Final Soft: obj={best_score:.8f}")
            else:
                print(f"  Final Soft: obj={obj_f:.8f}")

        # Coordinate
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.2, 20.0)
            a_final = coordinate_refine(best_seq, time_budget=alloc)
            obj_f = compute_obj(a_final)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_final.copy()
                print(f"  Final Coord: obj={best_score:.8f}")

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_final = lp_alternating(best_seq, max_iters=100,
                                     time_budget=min(remaining * 0.5, 30.0))
            obj_f = compute_obj(a_final)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_final.copy()
                print(f"  Final LP2: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = compute_obj(np.array(result))

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
