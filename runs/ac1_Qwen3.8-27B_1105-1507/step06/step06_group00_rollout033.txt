"""
Hybrid subgradient + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based convolution, top-K subgradient smoothing, momentum-based acceleration,
top-K constrained LP refinement, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)


def subgradient_smoothed(a, K=10, temperature=0.05):
    """
    Compute smoothed subgradient of max_k conv(a,a)[k] using top-K lags.
    Returns (gradient, conv_array).
    Assumes sum(a) = 1.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Get top K lags
    top_K_idx = np.argsort(conv)[-K:][::-1]
    top_K_vals = conv[top_K_idx]
    
    # Softmax weights (temperature controls how "peaked" the weight is)
    max_v = top_K_vals[0]
    scaled = (top_K_vals - max_v) / (max_v * temperature + 1e-15)
    scaled -= np.max(scaled)
    weights = np.exp(scaled)
    weights /= np.sum(weights)
    
    # Compute gradient: grad = sum_i w_i * 2 * a_shifted(k_i)
    grad = np.zeros(n)
    for i in range(len(top_K_idx)):
        k = int(top_K_idx[i])
        w = weights[i]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        # grad[lo:hi+1] += w * 2 * a[k-hi : k-lo+1][::-1]
        grad[lo:hi+1] += w * 2.0 * a[k-hi : k-lo+1][::-1]
    
    return grad, conv


def projected_step(a, grad, lr):
    """
    Projected gradient step maintaining sum(a)=1 and a>=0.
    """
    # Center gradient to maintain sum constraint
    grad_centered = grad - np.mean(grad)
    a_new = a - lr * grad_centered
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def backtracking_line_search(a, grad, lr_init=0.1, max_iter=20):
    """
    Backtracking line search for projected gradient.
    Returns (best_a, best_obj, best_lr).
    """
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    best_lr = 0.0
    
    grad_centered = grad - np.mean(grad)
    
    for i in range(max_iter):
        lr = lr_init * (0.7 ** i)
        if lr < 1e-8:
            break
        a_try = a - lr * grad_centered
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            best_lr = lr
    
    return best_a, best_obj, best_lr


def lp_minpeak_topk(a, K=50):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for top-K k, sum(g)=1, g>=0.
    Returns g (normalized) or None.
    """
    n = len(a)
    a = a / float(np.sum(a))
    conv_a = compute_conv(a)
    top_K_idx = np.argsort(conv_a)[-K:][::-1]
    
    # Build constraint matrix A (K x n): A[i,m] = a[k_i - m] if valid
    A = np.zeros((K, n))
    for i, k in enumerate(top_K_idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[i, lo:hi+1] = a[k-hi : k-lo+1][::-1]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    return None


def subgradient_optimize(a, time_budget, K=10, temperature=0.05, momentum=0.9, verbose=False):
    """
    Momentum-based smoothed subgradient optimization.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    # Momentum variables
    v = np.zeros(n)
    lr = 0.05
    lr_decay = 0.9995
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        grad, conv = subgradient_smoothed(a, K=K, temperature=temperature)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        v_centered = v - np.mean(v)
        
        # Line search
        a_new, obj_new, used_lr = backtracking_line_search(a, v, lr_init=lr)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Restart with smaller K (sharper gradient)
                K = max(1, K // 2)
                temperature = max(0.01, temperature * 0.5)
                no_improve = 0
                # Small perturbation
                rng = np.random.default_rng(it * 12345 + n)
                a_pert = a * np.exp(rng.normal(0, 0.02, n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / np.sum(a_pert)
                obj_pert = compute_obj_normalized(a_pert)
                if obj_pert < best_obj:
                    a = a_pert
                    best_a = a.copy()
                    best_obj = obj_pert
        
        lr *= lr_decay
        it += 1
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"  SG it={it}: obj={best_obj:.8f}, lr={lr:.6f}, K={K}, elapsed={elapsed:.1f}s")
    
    return best_a


def lp_refine(a, time_budget, K_lp=50, verbose=False):
    """
    LP-based refinement: alternate between LP direction and line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall = 0
    
    max_iters = 100
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp = lp_minpeak_topk(a, K=K_lp)
        if g_lp is None:
            break
        
        # Line search over blending factor
        best_t_obj = best_obj
        best_t = 0
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            a_try = (1 - best_t) * a + best_t * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
            best_a = a_try.copy()
            a = a_try
            stall = 0
            if verbose and it % 5 == 0:
                print(f"  LP it={it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 10:
                break
    
    return best_a


def coordinate_descent_polish(a, time_budget=10.0):
    """
    Pairwise coordinate descent: move mass between pairs to reduce peak conv.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    it = 0
    while time.monotonic() - start < time_budget:
        improved = False
        # Try moving mass from high-value positions that contribute to the peak
        step = max(1, n // 400)
        for i in range(0, n, step):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if j < 0 or j >= n or j == i:
                continue
            if a[i] <= a[j]:
                continue
            
            # Try moving a small fraction from i to j
            for frac in [0.01, 0.05, 0.1]:
                delta = frac * a[i]
                a[i] -= delta
                a[j] += delta
                conv_new = compute_conv(a)
                obj_new = 2.0 * n * float(np.max(conv_new))
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    conv = conv_new
                    k_star = int(np.argmax(conv))
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    break
        
        if not improved:
            break
        it += 1
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0).astype(np.float64)

"""
Hybrid subgradient + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based convolution, top-K subgradient smoothing with adaptive temperature,
momentum-based acceleration, top-K constrained LP refinement, coordinate descent polish,
and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)


def subgradient_smoothed(a, K=10, temperature=0.05):
    """
    Compute smoothed subgradient of max_k conv(a,a)[k] using top-K lags.
    Returns (gradient, conv_array, top_K_indices).
    Assumes sum(a) = 1.
    """
    n = len(a)
    conv = compute_conv(a)

    # Get top K lags
    K_actual = min(K, len(conv))
    top_K_idx = np.argsort(conv)[-K_actual:][::-1]
    top_K_vals = conv[top_K_idx]

    # Softmax weights
    max_v = top_K_vals[0]
    if max_v < 1e-15:
        weights = np.ones(K_actual) / K_actual
    else:
        scaled = (top_K_vals - max_v) / (max_v * temperature + 1e-15)
        scaled -= np.max(scaled)
        weights = np.exp(scaled)
        weights /= np.sum(weights)

    # Compute gradient: grad[j] = sum_i w_i * 2 * a[k_i - j] (where valid)
    grad = np.zeros(n)
    for i in range(K_actual):
        k = int(top_K_idx[i])
        w = weights[i]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]

    return grad, conv, top_K_idx


def projected_step(a, grad, lr):
    """Projected gradient step maintaining sum(a)=1 and a>=0."""
    grad_centered = grad - np.mean(grad)
    a_new = a - lr * grad_centered
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def line_search(a, grad, lr_init=0.1, max_iter=30):
    """Backtracking line search. Returns (best_a, best_obj, best_lr)."""
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    best_lr = 0.0

    grad_centered = grad - np.mean(grad)
    grad_norm = float(np.linalg.norm(grad_centered))
    if grad_norm < 1e-15:
        return best_a, best_obj, 0.0

    for i in range(max_iter):
        lr = lr_init * (0.7 ** i)
        if lr < 1e-10:
            break
        a_try = a - lr * grad_centered
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_obj - 1e-15:
            best_obj = obj_try
            best_a = a_try.copy()
            best_lr = lr

    return best_a, best_obj, best_lr


def lp_minpeak_topk(a, K=50):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for top-K k, sum(g)=1, g>=0.
    Returns g (normalized) or None.
    """
    n = len(a)
    a = a / float(np.sum(a))
    conv_a = compute_conv(a)
    K_actual = min(K, len(conv_a))
    top_K_idx = np.argsort(conv_a)[-K_actual:][::-1]

    # Build constraint matrix A (K x n): A[i,m] = a[k_i - m] if valid
    A = np.zeros((K_actual, n))
    for i, k in enumerate(top_K_idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((K_actual, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(K_actual)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None

        grad, conv, top_K_idx = subgradient_smoothed(a, K=K_cur, temperature=temp)

def coordinate_descent_polish(a, time_budget=10.0):
    """
    Pairwise coordinate descent: move mass between pairs contributing to peak conv.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    while time.monotonic() - start < time_budget:
        improved = False
        step = max(1, n // 500)
        for i in range(0, n, step):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if j < 0 or j >= n or j == i:
                continue
            if a[i] <= a[j] * 1.01:
                continue

            for frac in [0.005, 0.02, 0.05]:
                delta = frac * a[i]
                if delta < 1e-12:
                    break
                a[i] -= delta
                a[j] += delta
                conv_new = compute_conv(a)
                obj_new = 2.0 * n * float(np.max(conv_new))
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    conv = conv_new
                    k_star = int(np.argmax(conv))
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0).astype(np.float64)


def asymmetric_perturb(a, rng, scale=0.05):
    """Break symmetry by perturbing one half differently."""
    n = len(a)
    half = n // 2
    a_new = a.copy()
    pert_left = np.exp(rng.normal(0, scale, half))
    pert_right = np.exp(rng.normal(0, scale * 0.5, n - half))
    a_new[:half] *= pert_left
    a_new[half:] *= pert_right
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new

            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Asym[{trial}]: obj={best_score:.8f} **")
            else:
                print(f"  Asym[{trial}]: obj={obj:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_options = sorted(set([
        n_parent,
        n_parent - 50, n_parent + 50,
        n_parent - 100, n_parent + 100,
        n_parent - 200, n_parent + 200,
        n_parent - 300, n_parent + 300,
        600, 700, 900, 1000, 1200, 1400,
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1600]
    n_options.sort(key=lambda x: abs(x - n_parent))

    tried_n = set()
    for n_try in n_options[:10]:
        if n_try in tried_n:
            continue
        tried_n.add(n_try)

        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Subgradient
        alloc_sg = min(20.0, remaining * 0.35)
        if alloc_sg > 5:
            a_opt = subgradient_optimize(a_try, time_budget=alloc_sg,
                                         K=10, temperature=0.05)
        else:
            a_opt = a_try

        # Quick LP
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc_lp = min(15.0, remaining * 0.3)
            a_opt = lp_refine(a_opt, time_budget=alloc_lp, K_lp=50)

        obj = compute_obj_normalized(a_opt)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_final = len(best_seq)

        # Best subgradient config
        alloc = min(remaining * 0.25, 50.0)
        a_opt = subgradient_optimize(best_seq, time_budget=alloc,
                                     K=5, temperature=0.02, momentum=0.9, lr_init=0.02)
        obj = compute_obj_normalized(a_opt)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

        # LP
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.25, 30.0)
            a_opt = lp_refine(best_seq, time_budget=alloc, K_lp=60, verbose=False)
            obj = compute_obj_normalized(a_opt)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 8:
            alloc = min(remaining * 0.2, 15.0)
            a_opt = coordinate_descent_polish(best_seq, time_budget=alloc)
            obj = compute_obj_normalized(a_opt)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

        # One more round: subgradient with very small lr
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.3, 30.0)
            a_opt = subgradient_optimize(best_seq, time_budget=alloc,
                                         K=2, temperature=0.01, momentum=0.95, lr_init=0.005)
            obj = compute_obj_normalized(a_opt)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final SG2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    best_seq = np.clip(best_seq, 0, 1000)
    result = [float(x) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        print(f"\n=== FINAL: n={len(result)}, score={val:.8f} ===")
    else:
        print(f"WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s")

    return result
