```python
def solve_best_response_lp(a, n):
    """Solve: min M s.t. conv(g,a)[k] <= M for all k, sum(g)=1, g>=0.
    
    Returns (g, M) where g is the optimal best response.
    """
    # Build constraint matrix A where conv(g,a)[k] = sum_j g[j] * a[k-j]
    # A has shape (2n-1, n)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: min M
    # Variables: [g_0, ..., g_{n-1}, M] (n+1 variables)
    # Objective: c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: A@g - M <= 0, i.e., [A, -1] * [g; M] <= 0
    A_ub = np.hstack([A.toarray(), -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                bounds=bounds, method='highs')
    
    if r.success:
        g = r.x[:n]
        M = r.x[n]
        g = np.maximum(g, 0)
        g = g / g.sum()
        return g, M
    return None, None
```

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best response
iteration (exploiting convexity of the problem), combined with mirror descent
polish and multi-length search. Key insight: the problem is convex (max of PSD
quadratic forms over the simplex), so LP best-response converges to global optimum.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, preserve positivity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def build_lp_matrices(n):
    """Precompute sparsity pattern and dense matrix structure for the LP.
    
    The LP is: min M s.t. sum_j A[k,j]*g[j] <= M for k=0..2n-2, sum g=1, g>=0, M>=0.
    A[k,j] = a[k-j] if 0 <= k-j < n else 0.
    Returns (rows, cols) for the sparsity pattern.
    """
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_best_response(a, rows, cols, time_limit_ms=50):
    """Solve LP: min M s.t. conv(g,a)[k] <= M, sum(g)=1, g>=0, M>=0.
    
    Returns (g, M_value) or (None, None) on failure.
    """
    n = len(a)
    nz = len(rows)
    
    # Build sparse A (2n-1 x n)
    vals = np.zeros(nz)
    for idx in range(nz):
        k = rows[idx]
        j = cols[idx]
        m = k - j
        vals[idx] = a[m]
    
    A_sp = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Build dense A_ub: [A_sp | -1] with shape (2n-1, n+1)
    A_d = A_sp.toarray()
    A_ub = np.hstack([A_d, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M (last variable)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0.0, None)] * (n + 1)
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': time_limit_ms / 1000.0,
                             'dual_feasibility_tolerance': 1e-8,
                             'primal_feasibility_tolerance': 1e-8,
                             'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-20:
                g = g / gs
            M_val = r.x[n]
            return g, M_val
    except Exception:
        pass
    return None, None


def solve_best_response_sparse(a, rows, cols, active_fract=1.0):

    Uses the top constraints (those with highest conv(a,a) values).

    b = fast_conv(a)
    num_constraints = max(5, int(len(b) * active_fract))
    top_k = np.argsort(b)[::-1][:num_constraints]
    
    # Filter rows/cols to only active constraints
    k_set = set(top_k.tolist())
    mask = np.array([rows[i] in k_set for i in range(len(rows))])
    r_active = rows[mask]
    c_active = cols[mask]
    
    nz = len(r_active)
    if nz == 0:
        return None, None
    
    vals = np.zeros(nz)
    for idx in range(nz):
        k = r_active[idx]
        j = c_active[idx]
        vals[idx] = a[k - j]

    A_ub = np.hstack([A_d, -np.ones((num_act, 1))])
    b_ub = np.zeros(num_act)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

def mirror_descent_step(a, eta):
    """One mirror descent (multiplicative weights) step.
    
    a <- a * exp(-eta * h) / sum, where h is subgradient of max conv.
    """
    n = len(a)
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    
    # Subgradient: grad[i] = 2 * a[k_star - i] (for valid indices)
    h = np.zeros(n)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)
    if i_lo <= i_hi:
        h[i_lo:i_hi + 1] = 2.0 * a[k_star - i_hi:k_star - i_lo - 1:-1] if i_hi > i_lo else 2.0 * a[k_star - i_lo]
    
    i_lo2 = max(0, k_star)
    i_hi2 = min(n - 1, k_star + n - 1)
    if i_lo2 <= i_hi2:
        h[i_lo2:i_hi2 + 1] += 2.0 * a[i_lo2 - k_star:i_hi2 - k_star + 1]
    
    # Away from the max, we don't need the gradient (it's not active)
    # Mirror descent: a_new = a * exp(-eta * h)
    log_update = -eta * h
    log_update -= log_update.max()
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / a_new.sum()
    return a_new


def run_mirror_descent(a_init, max_iters, deadline, eta0=0.5):
    """Run mirror descent with adaptive step size."""
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Adaptive eta: decrease over time
        eta = eta0 / (1.0 + 0.01 * it)
        
        a = mirror_descent_step(a, eta)
        
        if it % 10 == 0:
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a

        if M_lp >= M_cur * (1 - 1e-10):
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Geometric interpolation: a_new = a^alpha * g^(1-alpha)
        # Try different alpha values
        improved = False
        best_trial_M = M_cur
        best_trial_a = None
        
        alphas = [0.3, 0.5, 0.7, 0.8, 0.9]
        for alpha in alphas:
            # a_new = a^alpha * g^(1-alpha), normalized
            log_a = np.log(np.maximum(a, 1e-30))
            log_g = np.log(np.maximum(g, 1e-30))
            log_new = alpha * log_a + (1 - alpha) * log_g
            a_new = np.exp(log_new)
            a_new = np.maximum(a_new, 0)
            a_new = normalize(a_new)
            
            M_new = fast_conv(a_new).max()
            if M_new < best_trial_M:
                best_trial_M = M_new
                best_trial_a = a_new
        
        if best_trial_a is not None and best_trial_M < M_cur:
            a = best_trial_a
            M_cur = best_trial_M
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve = 1
            improved = True
        else:
            # Try linear interpolation
            for t in [0.3, 0.5, 0.7]:
                a_lin = a + t * (g - a)
                a_lin = normalize(a_lin)
                M_lin = fast_conv(a_lin).max()
                if M_lin < M_cur:
                    a = a_lin
                    M_cur = M_lin
                    if M_cur < best_M:
                        best_M = M_cur
                        best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        
        # Periodically try with full constraints
        if it % 50 == 0 and n_active_frac < 0.99:
            n_active_frac = min(1.0, n_active_frac * 1.1)
        
        # Track timing
        if it % 100 == 0:
            elapsed = time.monotonic() - (deadline - 100)  # rough
            print(f"    LP iter {it}: M={M_cur:.10f}, best={best_M:.10f}", flush=True)
    
    return best_a


def coordinate_descent_step(a, i, b, n):
    """Optimize a single coordinate a[i] while keeping others fixed.
    
    The objective is max_k conv(a,a)[k] which as a function of x=a[i] is:
    max of (c_k + s_k * x + (quad_k * x^2)) over k, plus the max over unaffected k.
    """
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Compute c and slope for each affected k
    c_arr = []
    s_arr = []
    has_quad = False
    quad_idx = -1
    
    for k in range(k_lo, k_hi + 1):
        j = k - i
        if 0 <= j < n:
            if j == i:  # k = 2i
                c_arr.append(b[k] - x_cur * x_cur)
                s_arr.append(0.0)
                has_quad = True
                quad_idx = len(c_arr) - 1
            else:
                c_arr.append(b[k] - 2.0 * x_cur * a[j])
                s_arr.append(2.0 * a[j])
        else:
            c_arr.append(b[k])
            s_arr.append(0.0)
    
    c_arr = np.array(c_arr)
    s_arr = np.array(s_arr)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    def obj_at(x):
        vals = c_arr + s_arr * x
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_idx] + x * x
            if qv > m:
                m = qv
        return m
    
    # Golden section search
    lo = 0.0
    hi = max(3.0 * x_cur, 0.5 / n, 1e-6)
    
    for _ in range(15):
        if obj_at(hi * 0.5) < obj_at(hi) and hi < 10.0 / n:
            hi *= 2.0
        else:
            break
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline):
    """Run coordinate descent."""
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    rng = np.random.default_rng(12345)
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        
        b = fast_conv(a)
        sweep_improved = False
        
        for i in order:
            x_opt = coordinate_descent_step(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                # Update b efficiently
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                delta = x_opt - a[i]
                a[i] = x_opt
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if j == i:
                            b[k] = x_opt * x_opt
                        else:
                            b[k] += 2.0 * a[j] * delta
                sweep_improved = True
        
        a = normalize(a)
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 5
        if no_improve > 15:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        p = normalize(p)
        inits.append(p)
        for sigma in [0.01, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 0)
            inits.append(normalize(p_n))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various structured inits
    # Concave (parabola)
    w = 4 * t * (1 - t) + 0.01
    inits.append(normalize(w))
    
    # Convex (U-shape)
    w = (2 * t - 1) ** 2 + 0.01
    inits.append(normalize(w))
    
    # Cosine
    for freq in [1, 2, 3]:
        w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        w = np.maximum(w, 0.001)
        inits.append(normalize(w))
    
    # Gaussian
    w = np.exp(-8 * (t - 0.5) ** 2) + 0.01
    inits.append(normalize(w))
    
    # Beta-like distributions
    for alpha, beta in [(0.5, 0.5), (2, 2), (0.5, 2), (2, 0.5)]:
        w = t ** (alpha - 0.5) * (1 - t) ** (beta - 0.5) + 0.001
        inits.append(normalize(w))
    
    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / (ksize / 4) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0.1, 1, n), kernel, mode='same')
    inits.append(normalize(smooth))
    
    return inits


def evaluate_and_track(a, tracker):
    """Evaluate a sequence and update tracker."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < tracker[0]:
        tracker[0] = score
        tracker[1] = a.copy()
        print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={time.monotonic()-tracker[2]:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase LP-based optimization for minimizing J(a) = 2n*max(conv(a,a))/S^2.
    Uses LP best-response (exploiting convexity), mirror descent, and coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)
    
    # Tracker: [best_score, best_sequence, start_time]
    tracker = [float('inf'), None, t_start]
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            evaluate_and_track(parent, tracker)
            print(f"Parent loaded: n={parent.size}, J={tracker[0]:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(768)
        evaluate_and_track(parent, tracker)
    
    # Determine lengths to try
    n_parent = parent.size
    lengths = []
    for nl in [512, 640, 768, 896, 1024, 1200]:
        if 256 <= nl <= 2048:
            lengths.append(nl)
        if n_parent in [512, 640, 768, 896, 1024, 1200]:
            break  # Keep parent length in the list
    if n_parent not in lengths and 256 <= n_parent <= 2048:
        lengths.append(n_parent)
    lengths = sorted(set(lengths))
    
    # Phase 1: LP-based optimization for each length
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP best-response optimization", flush=True)
    print(f"Lengths to try: {lengths}", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Allocate time: 60% to Phase 1, 20% to mirror descent, 15% to CD, 5% buffer
    phase1_deadline = t_start + 0.55 * float(budget_s)
    phase2_deadline = t_start + 0.75 * float(budget_s)
    phase3_deadline = t_start + 0.90 * float(budget_s)
    phase4_deadline = deadline
    
    # Try each length
    time_per_length = max(20, (phase1_deadline - time.monotonic()) / len(lengths))
    
    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        
        print(f"\n  --- n={n} ---", flush=True)
        t0 = time.monotonic()
        
        # Get initializations
        inits = generate_inits(n, rng, parent)
        
        best_for_n = None
        best_M_for_n = float('inf')
        
        # For each init, run LP optimization (time-limited)
        time_per_init = max(10, time_per_length / len(inits))
        
        for ii, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline - 10:
                break
            
            sub_deadline = min(t0 + time_per_length, 
                              time.monotonic() + time_per_init)
            
            a0 = normalize(a0)
            
            # Run LP optimization
            a_opt = run_lp_optimization(a0, sub_deadline)
            M_opt = fast_conv(a_opt).max()
            
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
            
            # Quick check
            J_val = 2.0 * n * M_opt
            if J_val < tracker[0] + 0.01:
                evaluate_and_track(a_opt, tracker)
        
        if best_for_n is not None:
            J_val = 2.0 * n * best_M_for_n
            print(f"  n={n}: best M={best_M_for_n:.10f} J={J_val:.8f} "
                  f"(elapsed={time.monotonic()-t0:.1f}s)", flush=True)
            evaluate_and_track(best_for_n, tracker)
    
    print(f"\n  Phase 1 done. Best: J={tracker[0]:.8f} n={tracker[1].size if tracker[1] is not None else 0}", flush=True)
    
    # Phase 2: Extended LP on the best
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Extended LP on best sequence", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker[1] is not None and time.monotonic() < phase2_deadline:
        n_best = tracker[1].size
        a_ext = run_lp_optimization(tracker[1], phase2_deadline, n_active_frac=1.0)
        evaluate_and_track(a_ext, tracker)
        # Also try with tight active set
        if time.monotonic() < phase2_deadline - 10:
            a_tight = run_lp_optimization(a_ext, phase2_deadline, n_active_frac=0.3)
            evaluate_and_track(a_tight, tracker)
    
    print(f"  Phase 2 done. Best: J={tracker[0]:.8f}", flush=True)
    
    # Phase 3: Mirror descent polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Mirror descent polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker[1] is not None and time.monotonic() < phase3_deadline:
        a_md = run_mirror_descent(tracker[1], 50000, phase3_deadline, eta0=0.1)
        evaluate_and_track(a_md, tracker)
        
        # Try small step size for fine tuning
        if time.monotonic() < phase3_deadline - 5:
            a_md2 = run_mirror_descent(a_md, 20000, phase3_deadline, eta0=0.01)
            evaluate_and_track(a_md2, tracker)
    
    print(f"  Phase 3 done. Best: J={tracker[0]:.8f}", flush=True)
    
    # Phase 4: Coordinate descent final polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker[1] is not None and time.monotonic() < phase4_deadline:
        max_sweeps = max(5, int((phase4_deadline - time.monotonic()) / max(0.1, tracker[1].size / 200.0)))
        a_cd = run_coordinate_descent(tracker[1], max_sweeps, phase4_deadline)
        evaluate_and_track(a_cd, tracker)
    
    print(f"  Phase 4 done. Best: J={tracker[0]:.8f}", flush=True)
    
    # Final verification
    if tracker[1] is None:
        tracker[1] = np.ones(768)
        tracker[0] = compute_J(tracker[1])
    
    final_score = float(evaluate_sequence(tracker[1].tolist()))
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={tracker[1].size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    if final_score < tracker[0]:
        tracker[0] = final_score
    
    return [float(x) for x in tracker[1]]
