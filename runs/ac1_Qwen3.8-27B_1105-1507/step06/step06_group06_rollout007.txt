```python
"""
ALM (Alternating Linear Minimization) based optimizer for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses alternating LP solves as the primary driver, combined with
FFT-based projected subgradient for fine-tuning.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.array(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return a
    # Water-filling algorithm
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(a - theta, 0)


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(a_new)


def build_conv_matrix(b):
    """Build matrix A where A[k, i] = b[k-i] if 0<=k-i<len(b), else 0."""
    n = len(b)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_min = i
        k_max = min(i + n - 1, 2 * n - 2)
        A[k_min:k_max + 1, i] = b[:k_max - k_min + 1]
    return A


def solve_lp_minpeak(b_fixed, time_limit=5):
    """
    Solve: min M s.t. conv(a, b_fixed)[k] <= M for all k, sum(a)=1, a>=0.
    Returns (a_opt, M_opt) or (None, None).
    """
    n = len(b_fixed)
    if n > 2000:
        return None, None
    
    A = build_conv_matrix(b_fixed)
    
    # Variables: [a_0, ..., a_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A @ a - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            a_opt = np.maximum(res.x[:n], 0)
            s = np.sum(a_opt)
            if s > 1e-10:
                a_opt = a_opt / s
            M_opt = float(res.x[n])
            return a_opt, M_opt
    except Exception:
        pass
    return None, None


def alm_optimize(a_init, max_iters=300, time_budget=300.0):
    """
    ALM: Alternating LP optimization.
    Alternates between:
      a <- argmin max_k conv(a, b)[k] s.t. a in simplex
      b <- argmin max_k conv(a, b)[k] s.t. b in simplex
    """
    n = len(a_init)
    a = project_simplex(a_init.copy())
    b = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    print(f"  ALM start: obj={best_obj:.8f}")
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Optimize a given b
        a_new, M_a = solve_lp_minpeak(b, time_limit=min(5.0, time_budget - (time.monotonic() - start)))
        if a_new is None:
            break
        
        # Optimize b given a_new
        b_new, M_b = solve_lp_minpeak(a_new, time_limit=min(5.0, time_budget - (time.monotonic() - start)))
        if b_new is None:
            break
        
        # Update (with damping for stability)
        a = a_new
        b = b_new
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if it % 10 == 0 or obj < 1.503:
                print(f"  ALM it{it}: obj={best_obj:.8f}, M_a={M_a:.8f}, M_b={M_b:.8f}")
        
        # Check convergence
        if it > 0 and abs(M_a - M_b) < 1e-8:
            print(f"  ALM converged at it{it}: M={M_a:.10f}")
            break
    
    print(f"  ALM done: best_obj={best_obj:.8f}")
    return best_a, best_obj


def fft_subgradient_refine(a_init, max_iters=5000, time_budget=300.0):
    """
    Projected subgradient with FFT-based gradient computation.
    Uses adaptive softmax temperature and momentum.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # FFT size for correlation: need N >= (2n-1) + n - 1 = 3n-2
    N = 1
    while N < 3 * n:
        N <<= 1
    
    momentum = np.zeros(n)
    base_step = 0.02
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute convolution via FFT
        fa = np.fft.rfft(a, N)
        conv = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        
        cmax = float(np.max(conv))
        
        # Adaptive temperature
        progress = it / max_iters
        tau = 0.005 + 0.05 * (1 - progress) ** 2
        
        # Softmax weights
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500, 0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
        
        # Compute subgradient via cross-correlation (FFT)
        # g[i] = 2 * sum_k weights[k] * a[k-i] for valid k-i
        # = 2 * corr(weights, a)[i] where corr is cross-correlation
        fw = np.fft.rfft(weights, N)
        fa_full = np.fft.rfft(a, N)
        corr = np.fft.irfft(fw * np.conj(fa_full), N)[:n]
        grad = 2.0 * corr
        
        # Center gradient (optimization on simplex)
        grad = grad - np.mean(grad)
        
        # Step with momentum
        decay = 0.95
        momentum = decay * momentum - base_step * grad
        a_new = a + momentum
        a_new = project_simplex(a_new)
        
        # Computed objective
        fa_new = np.fft.rfft(a_new, N)
        conv_new = np.fft.irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(conv_new))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if it % 500 == 0:
                print(f"  Subgrad it{it}: obj={best_obj:.8f}")
        
        a = a_new
        
        # Reduce step size over time
        base_step *= 0.9995
    
    return best_a, best_obj

        # So subgradient = (2/|S|) sum_{k in S} a[k-i]

        fa_new = np.fft.rfft(a_new, N)
        conv_new = np.fft.irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(conv_new))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        base_step *= 0.999
    
    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sinusoidal
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.1
    inits.append(sine / sine.sum())
    
    # Multi-sine (designed to flatten autocorrelation)
    x = np.linspace(0, 1, n)
    multi = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x) ** 2
    multi = np.maximum(multi, 0.01)
    inits.append(multi / multi.sum())
    
    # Flat-top with edges
    flat = np.ones(n)
    edge = max(5, n // 20)
    flat[:edge] = np.linspace(0.3, 1.0, edge)
    flat[-edge:] = np.linspace(1.0, 0.3, edge)
    inits.append(flat / flat.sum())
    
    # Random
    rnd = rng.random(n) + 0.05
    inits.append(rnd / rnd.sum())
    
    if parent is not None:
        inits.append(resample_sequence(parent, n))
        # Perturbed parent
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_simplex(pert))
    
    return inits
