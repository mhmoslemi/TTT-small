"""
Coordinate-descent and Frank-Wolfe hybrid optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Novel elements: exact 1D coordinate optimization exploiting self-convolution
structure, Frank-Wolfe conditional gradient steps, multi-length exploration
with adaptive allocation, and LP-based polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent(a, n_passes=50, time_budget=60.0, order=None):
    """
    Exact 1D coordinate descent on max conv(a,a).
    For each coordinate i, find x >= 0 minimizing max_k conv[k](x)
    where conv[k](x) is the convolution with a[i] replaced by x.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    conv = compute_conv(a)
    best_obj = compute_obj(a)
    start = time.monotonic()

    if order is None:
        order = np.arange(n)

    for p in range(n_passes):
        if time.monotonic() - start > time_budget:
            break

        any_change = False
        for i in order:
            if time.monotonic() - start > time_budget:
                break

            x_old = a[i]
            S = float(np.sum(a))

            # Build conv[k] as function of x = a[i]
            # conv[k](x) = C[k] + B[k]*x + A[k]*x^2
            A_k = np.zeros(2 * n - 1)
            B_k = np.zeros(2 * n - 1)
            C_k = conv.copy()

            j_valid_start = max(0, i)  # k - i >= 0 -> k >= i
            j_valid_end = min(2 * n - 1, (n - 1) + i + 1)  # k - i <= n-1 -> k <= n-1+i

            for k in range(max(0, i), min(2 * n - 1, n - 1 + i + 1)):
                j = k - i
                if 0 <= j < n:
                    if j == i:
                        A_k[k] = 1.0
                        C_k[k] = conv[k] - x_old * x_old
                    else:
                        B_k[k] = 2.0 * a[j]
                        C_k[k] = conv[k] - 2.0 * x_old * a[j]

            # Find x >= 0 minimizing max_k (A_k[k]*x^2 + B_k[k]*x + C_k[k])
            # Use golden section search with bracketing
            def max_conv_at_x(x):
                vals = A_k * x * x + B_k * x + C_k
                return float(np.max(vals))

            # Bracket
            x_lo = 0.0
            x_hi = max(x_old * 4.0, 2.0 / n)

            # Check if f is decreasing at x_lo
            f_lo = max_conv_at_x(0.0)
            f_mid = max_conv_at_x(x_hi * 0.25)
            if f_mid < f_lo:
                # Minimum is not at 0, extend bracket
                while f_mid < max_conv_at_x(x_hi) and x_hi < 100.0 / n:
                    x_lo = x_hi
                    x_hi *= 2.0
                    f_mid = max_conv_at_x(x_hi * 0.25)
            else:
                # Minimum is near 0
                x_hi = x_hi * 0.5
                if x_hi < x_old * 0.1:
                    x_hi = max(1.0 / n, x_old * 0.1)

            # Also try x_old as a candidate
            x_candidates = [0.0, x_old, x_lo, x_hi]

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            ga = x_lo
            gb = x_hi
            if gb - ga > 1e-12:
                gc = gb - gr * (gb - ga)
                gd = ga + gr * (gb - ga)
                fc = max_conv_at_x(gc)
                fd = max_conv_at_x(gd)
                for _ in range(40):
                    if gb - ga < 1e-12:
                        break
                    if fc < fd:
                        gb = gd
                        gd = gc
                        fd = fc
                        gc = gb - gr * (gb - ga)
                        fc = max_conv_at_x(gc)
                    else:
                        ga = gc
                        gc = gd
                        fc = fd
                        gd = ga + gr * (gb - ga)
                        fd = max_conv_at_x(gd)
                x_candidates.append((ga + gb) / 2.0)

            # Evaluate all candidates
            best_x = x_old
            best_peak = max_conv_at_x(x_old)
            for xc in x_candidates:
                xc = max(0.0, xc)
                pk = max_conv_at_x(xc)
                if pk < best_peak - 1e-15:
                    best_peak = pk
                    best_x = xc

            if best_x != x_old:
                # Update convolution efficiently
                a[i] = best_x
                # Recompute affected conv entries
                for k in range(max(0, i), min(2 * n - 1, n - 1 + i + 1)):
                    j = k - i
                    if j == i:
                        conv[k] = C_k[k] + A_k[k] * best_x * best_x
                    elif 0 <= j < n:
                        conv[k] = C_k[k] + B_k[k] * best_x
                any_change = True

        if not any_change:
            break

        # Shuffle order for next pass
        order = rng.permutation(n) if p > 0 else order

    # Final projection
    a = project_nonneg_sum1(a)
    obj = compute_obj(a)
    return a, obj


def frank_wolfe(a, n_iters=200, time_budget=30.0, tau=0.1):
    """
    Frank-Wolfe (conditional gradient) method for minimizing max conv(a,a)
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient: grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        for i in range(j_lo, j_hi + 1):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # FW step: find j_min = argmin grad[i], set v = e_{j_min}
        j_min = int(np.argmin(grad))

        # Step size: t = 2/(it+2) or line search
        t_fw = 2.0 / (it + 2.0)
        # Try a few step sizes
        best_t = t_fw
        best_t_obj = np.inf
        for t_try in [t_fw * 0.5, t_fw, t_fw * 1.5, 0.05, 0.1, 0.2]:
            t_try = min(1.0, max(0.01, t_try))
            a_try = (1.0 - t_try) * a + t_try * np.zeros(n)
            a_try[j_min] += t_try
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t_try

        if best_t_obj < best_obj:
            a[j_min] += best_t
            a = project_nonneg_sum1(a)
            best_obj = best_t_obj
            best_a = a.copy()

    return best_a, best_obj


def softmax_gradient_descent(a, n_iters=500, time_budget=60.0, tau_init=0.05, momentum=0.8):
    """
    Subgradient method with softmax smoothing of the max and Nesterov momentum.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()

    v = np.zeros(n)
    step = 0.05

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        tau = tau_init + 0.3 * it / max(n_iters, 1)

        # Softmax weights
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -300.0, 0.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum

        # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (for valid k-i)
        grad = np.zeros(n)
        for i in range(n):
            j_lo = max(0, i)
            j_hi = min(2 * n - 1, i + n - 1)
            # sum_k weights[k] * a[k-i] for k in [i, i+n-1]
            seg = weights[j_lo:j_hi + 1]
            a_seg = a[0:len(seg)]
            grad[i] = 2.0 * float(np.sum(seg * a_seg))

        # Center gradient
        grad = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - step * grad
        a_new = a - v
        a = project_nonneg_sum1(a_new)

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        step *= 0.997

    return best_a, best_obj


def lp_alternate(a, max_iters=30, time_budget=40.0):
    """LP alternating method."""
    n = len(a)
    if n > 2500:
        return a, compute_obj(a)

    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Build Toeplitz matrix
    A_toep = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_toep[k_start:k_end, j] = a[:k_end - k_start]

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Solve: max sum(g) s.t. A_toep @ g <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A_toep, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                   'time_limit': min(15, time_budget * 0.5)})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                s = float(np.sum(g))
                if s > 1e-10:
                    g = g / s
                    # Line search
                    best_t_obj = best_obj
                    best_t_a = best_a
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_try = (1.0 - t) * a + t * g
                        a_try = project_nonneg_sum1(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_t_obj:
                            best_t_obj = obj_try
                            best_t_a = a_try.copy()
                    if best_t_obj < best_obj - 1e-12:
                        a = best_t_a
                        best_obj = best_t_obj
                        best_a = a.copy()
                        # Update Toeplitz
                        for j in range(n):
                            k_start = j
                            k_end = min(j + n, 2 * n - 1)
                            A_toep[k_start:k_end, j] = a[:k_end - k_start]
                    else:
                        if it > 5:
                            break
        except Exception:
            break

    return best_a, best_obj


def simulated_annealing(a, rng, max_iters=200, T_init=0.3, T_min=0.005, time_budget=60.0):
    """Simulated annealing with diverse move types."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    cur_a = a.copy()
    cur_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = cur_obj
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T_init * (T_min / T_init) ** (it / max_iters)
        T = max(T, 1e-6)

        a_trial = cur_a.copy()
        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Window perturbation
            wsize = max(5, n // 20)
            wstart = int(rng.integers(0, n - wsize))
            noise = rng.normal(0, 0.2, wsize) * (np.mean(a_trial[wstart:wstart+wsize]) + 0.01)
            a_trial[wstart:wstart + wsize] += noise
        elif move_type == 1:
            # Coordinate swap
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2:
                a_trial[i1], a_trial[i2] = a_trial[i2], a_trial[i1]
        elif move_type == 2:
            # Spike: concentrate mass
            wsize = max(5, n // 30)
            wstart = int(rng.integers(0, n - wsize))
            mass = float(a_trial[wstart:wstart + wsize].sum())
            a_trial[wstart:wstart + wsize] = 0
            target = int(rng.integers(0, n))
            a_trial[target] += mass
        else:
            # Smooth: spread mass from a peak
            peak = int(np.argmax(a_trial))
            wsize = max(5, n // 15)
            lo = max(0, peak - wsize // 2)
            hi = min(n, peak + wsize // 2)
            mass = a_trial[peak] * 0.5
            a_trial[peak] -= mass
            a_trial[lo:hi] += mass / (hi - lo)

        a_trial = project_nonneg_sum1(a_trial)
        trial_obj = compute_obj(a_trial)

        delta = trial_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_a = a_trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Sine
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    # Gaussian
    x = np.linspace(-3, 3, n)
    gauss = np.exp(-x ** 2 / 3.0)
    inits.append(gauss / gauss.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Cosine window
    x = np.linspace(0, np.pi, n)
    cosw = 0.5 * (1 + np.cos(x))
    inits.append(cosw / cosw.sum())

    # Two-lobe
    x = np.linspace(0, 1, n)
    two_lobe = np.exp(-((x - 0.3) ** 2) / 0.03) + np.exp(-((x - 0.7) ** 2) / 0.03)
    inits.append(two_lobe / two_lobe.sum())

    # Oscillatory
    x = np.linspace(0, 1, n)
    osc = 1.0 + 0.3 * np.sin(4 * np.pi * x)
    inits.append(osc / osc.sum())

    if parent is not None:
        resampled = resample_sequence(parent, n)
        inits.append(resampled / resampled.sum())
        perturbed = resampled * np.exp(rng.normal(0, 0.1, n))
        inits.append(project_nonneg_sum1(perturbed))

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining coordinate descent,
    Frank-Wolfe, softmax gradient descent, LP alternating, and SA.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000.0)
        if p.size > 0 and np.isfinite(p).all():
            parent = p.copy()
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={best_score:.8f}")

    if best_seq is None:
        best_seq = np.ones(700)
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"Initialized uniform n=700: obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Multi-length exploration (15% time) ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    phase1_budget = 0.15 * (deadline - start_time)
    phase1_deadline = start_time + phase1_budget

    n_options = sorted(set([
        n_parent,
        max(200, n_parent - 100),
        n_parent + 100,
        n_parent + 200,
        n_parent + 400,
        min(2500, n_parent + 800),
        min(3000, n_parent + 1000),
    ]))
    n_options = [n for n in n_options if 100 <= n <= 3000]

    phase1_candidates = []
    per_n_time = phase1_budget / max(1, len(n_options))

    for n_try in n_options:
        if time.monotonic() > phase1_deadline:
            break
        inits = generate_initializations(n_try, rng, parent)
        per_init_time = per_n_time / max(1, len(inits))

        for init in inits:
            if time.monotonic() > phase1_deadline:
                break
            # Quick coordinate descent
            a_q, obj_q = coordinate_descent(init, n_passes=3, time_budget=min(per_init_time, 5.0))
            if obj_q < np.inf:
                phase1_candidates.append((obj_q, a_q, n_try))

    phase1_candidates.sort(key=lambda x: x[0])
    print(f"  {len(phase1_candidates)} candidates, best: {phase1_candidates[0][0]:.8f}" if phase1_candidates else "  No candidates")

    for obj_c, a_c, n_c in phase1_candidates[:5]:
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  Phase1 update: n={n_c}, obj={best_score:.8f}")

    # === Phase 2: Intensive optimization (50% time) ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    print(f"Best: {best_score:.8f}")
    phase2_budget = 0.50 * (deadline - start_time)
    phase2_deadline = start_time + phase2_budget

    # Focus on top 3 candidates from phase 1 + current best
    top_list = [(best_score, best_seq.copy(), len(best_seq))]
    seen_ns = {len(best_seq)}
    for obj_c, a_c, n_c in phase1_candidates[:4]:
        if n_c not in seen_ns:
            top_list.append((obj_c, a_c.copy(), n_c))
            seen_ns.add(n_c)
        if len(top_list) >= 3:
            break

    for ci, (obj_init, a_init, n_init) in enumerate(top_list):
        if time.monotonic() > phase2_deadline:
            break
        alloc = min(100.0, (phase2_deadline - time.monotonic()) / max(1, len(top_list) - ci))
        if alloc < 15:
            break

        a = project_nonneg_sum1(a_init.copy())
        local_obj = compute_obj(a)
        local_best = a.copy()
        print(f"  Candidate {ci} (n={n_init}): obj={local_obj:.8f}, budget={alloc:.0f}s")

        # Cycle through strategies
        n_strategies = 7
        cycle_count = int(alloc / 12)
        for cycle in range(cycle_count):
            if time.monotonic() > phase2_deadline:
                break
            rem = phase2_deadline - time.monotonic()
            if rem < 8:
                break

            strat = (cycle + ci) % n_strategies

            if strat == 0:
                # Coordinate descent (the main novel method)
                a_new, obj_new = coordinate_descent(a, n_passes=10, time_budget=min(rem * 0.4, 20.0))
            elif strat == 1:
                # Frank-Wolfe
                a_new, obj_new = frank_wolfe(a, n_iters=100, time_budget=min(rem * 0.3, 15.0))
            elif strat == 2:
                # Softmax gradient descent
                a_new, obj_new = softmax_gradient_descent(a, n_iters=200, time_budget=min(rem * 0.3, 20.0))
            elif strat == 3 and n_init <= 1500:
                # LP alternating
                a_new, obj_new = lp_alternate(a, max_iters=10, time_budget=min(rem * 0.4, 20.0))
            elif strat == 4:
                # Coordinate descent with different order
                order = rng.permutation(n_init)
                a_new, obj_new = coordinate_descent(a, n_passes=10, time_budget=min(rem * 0.4, 20.0), order=order)
            elif strat == 5:
                # SA with small perturbation
                a_pert = a * np.exp(rng.normal(0, 0.03, n_init))
                a_pert = project_nonneg_sum1(a_pert)
                a_new, obj_new = simulated_annealing(a_pert, rng, max_iters=50,
                                                     T_init=0.1, T_min=0.005,
                                                     time_budget=min(rem * 0.3, 15.0))
            else:
                # Combined: short CD + short FW
                a_temp, _ = coordinate_descent(a, n_passes=3, time_budget=min(rem * 0.2, 8.0))
                a_new, obj_new = frank_wolfe(a_temp, n_iters=50, time_budget=min(rem * 0.2, 8.0))

            if obj_new < local_obj - 1e-12:
                local_obj = obj_new
                local_best = a_new.copy()
                a = a_new.copy()
                if local_obj < best_score:
                    best_score = local_obj
                    best_seq = local_best.copy()
                    print(f"    ** Update ci={ci} cycle={cycle} s{strat}: obj={best_score:.8f} **")

        print(f"  Candidate {ci} final: {local_obj:.8f}")

    # === Phase 3: SA + perturbation (20% time) ===
    print(f"\n=== Phase 3: SA + Perturbation ===")
    print(f"Best: {best_score:.8f}")
    phase3_budget = 0.20 * (deadline - start_time)
    phase3_deadline = start_time + phase3_budget
    n_best = len(best_seq)

    # SA on best
    a_sa, obj_sa = simulated_annealing(best_seq, rng, max_iters=200,
                                       T_init=0.2, T_min=0.003,
                                       time_budget=min(phase3_budget * 0.4, 80.0))
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = a_sa.copy()
        print(f"  SA: {best_score:.8f}")

    # CD on SA result
    if time.monotonic() < phase3_deadline:
        a_cd, obj_cd = coordinate_descent(a_sa if obj_sa < best_score else best_seq,
                                          n_passes=15,
                                          time_budget=min(phase3_deadline - time.monotonic(), 40.0))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD after SA: {best_score:.8f}")

    # Multiple perturbation rounds
    rem3 = phase3_deadline - time.monotonic()
    n_rounds = max(1, int(rem3 / 15))
    for r in range(n_rounds):
        if time.monotonic() > phase3_deadline:
            break
        a_pert = best_seq * np.exp(rng.normal(0, 0.02 + 0.01 * r, n_best))
        a_pert = project_nonneg_sum1(a_pert)
        a_opt, obj_opt = coordinate_descent(a_pert, n_passes=8,
                                            time_budget=min(rem3 * 0.3, 12.0))
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Perturb {r}: {best_score:.8f}")

    # === Phase 4: LP polishing (10% time) ===
    print(f"\n=== Phase 4: LP Polishing ===")
    print(f"Best: {best_score:.8f}")
    phase4_deadline = min(deadline, start_time + 0.93 * (deadline - start_time))
    n_cur = len(best_seq)

    if n_cur <= 2000 and time.monotonic() < phase4_deadline:
        a_lp, obj_lp = lp_alternate(best_seq, max_iters=20,
                                    time_budget=min(phase4_deadline - time.monotonic(), 60.0))
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: {best_score:.8f}")

    # === Phase 5: Final intensive CD polish (remaining time) ===
    print(f"\n=== Phase 5: Final CD Polish ===")
    print(f"Best: {best_score:.8f}")
    rem5 = deadline - time.monotonic()
    if rem5 > 10:
        # Multiple CD passes with shuffle
        a_final = project_nonneg_sum1(best_seq.copy())
        for fp in range(min(20, int(rem5 / 8))):
            if time.monotonic() > deadline - 5:
                break
            rem_fp = deadline - time.monotonic() - 5
            if rem_fp < 5:
                break
            order = rng.permutation(len(a_final))
            a_new, obj_new = coordinate_descent(a_final, n_passes=5,
                                                time_budget=rem_fp * 0.6,
                                                order=order)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_final = a_new.copy()
                if fp % 5 == 0:
                    print(f"  CD polish {fp}: {best_score:.8f}")

            # Interleave with FW
            rem_fw = deadline - time.monotonic() - 5
            if rem_fw > 5:
                a_fw, obj_fw = frank_wolfe(a_final, n_iters=50,
                                           time_budget=rem_fw * 0.3)
                if obj_fw < best_score:
                    best_score = obj_fw
                    best_seq = a_fw.copy()
                    a_final = a_fw.copy()

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
