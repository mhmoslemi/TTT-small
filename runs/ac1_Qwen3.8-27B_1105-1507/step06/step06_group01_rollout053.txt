"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based fast convolution
2. LP best-response with adaptive active set management
3. Targeted pairwise mass-transfer moves attacking peak indices
4. Coordinate descent with exact 1D optimization
5. Population-based diversity (multiple candidates with periodic recombination)
6. Multi-length exploration with spectral initialization
7. Subgradient descent with momentum for continuous refinement

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def lp_best_response(a, n_active=30, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= max(conv(a,a)) for k in active, g >= 0.
    Returns normalized g or None."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    
    # Active set: top n_active
    top_idx = np.argsort(b)[::-1][:min(n_active, 2*n-1)]
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_idx):
        k = int(k)
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                if a[m] > 1e-12:
                    rows.append(idx)
                    cols.append(j)
                    vals.append(a[m])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_idx), n))
    c = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def targeted_pair_moves(a, n_moves=100, deadline=None):
    """Move mass from peak-contributing pairs to valley indices."""
    n = len(a)
    a = a.copy()
    b = fast_conv(a)
    best_M = b.max()
    best_a = a.copy()

            break

        k_star = int(np.argmax(b))
        M = b[k_star]

        i_max = 0
        max_prod = 0
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                prod = a[i] * a[j]
                if prod > max_prod:
                    max_prod = prod
                    i_max = i
        
        i1 = i_max

        if i1 == i2:

            a[j_dest] += delta
        else:

                src = i1
                dst = i2
            else:
                src = i2
                dst = i1
            # Move some mass from src to a random "valley" index

            a[src] -= delta
            a[j_dest] += delta
        
        a = np.maximum(a, 0.0)
        a = normalize(a)
        
        M_new = fast_conv(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    
    return best_a

def coordinate_descent_1d(a, i, b):
    """Exact 1D optimization for coordinate i."""
    n = len(a)
    x_cur = a[i]
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    # Unaffected max
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_lo:k_hi + 1] = False
    base_max = b[mask].max() if mask.any() else 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur, 1.0 / n * 3.0, 0.1)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

def run_cd(a_init, max_sweeps, deadline):
    """Coordinate descent with random ordering."""
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(42)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = np.arange(n)
        rng.shuffle(order)
        
        b = fast_conv(a)
        imps = 0
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                a[i] = x_opt
                # Update b
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * (x_opt - old) * a[j]
                imps += 1
        
        a = normalize(a)
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
        
        if imps == 0:
            break
    
    return best_a

def subgradient_descent(a_init, max_iters, deadline, mu=0.9):
    """Heavy-ball subgradient method on max(conv(a,a))."""
    a = normalize(a_init.copy())
    n = len(a)
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_M = fast_conv(a).max()
    step = 1.0 / n
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Active set: indices where b is close to max
        active = np.where(b >= M * 0.99)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient: average of gradients at active indices
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(active)
        
        # Project gradient to tangent of simplex
        grad_tan = grad - grad.mean()
        
        # Adaptive step size
        step = step * (1.0 + 50.0 / (it + 50.0))
        step = min(step, 2.0 / (np.linalg.norm(grad_tan) + 1e-10))
        
        v = mu * v - step * grad_tan
        a_new = a + v
        a_new = proj_simplex(a_new)
        a = a_new
        
        M_new = fast_conv(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    
    return best_a

def spectral_init(n, rng):
    """Initialize by trying to flatten the spectrum."""
    a = np.ones(n) / n
    for _ in range(50):
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        # Reduce contribution to peak
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a[j]
        a = a - 0.05 * (corr - corr.mean())
        a = np.maximum(a, 1e-8)
        a /= a.sum()
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_diverse_inits(n, rng, parent=None):

def run_lp_mirror(a_init, max_iters, deadline, n_active_start=30, n_active_max=100):
    """LP mirror descent with adaptive active set."""
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    n_active = n_active_start
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = lp_best_response(a, n_active, time_limit=5)
        if g is None:
            n_active = min(int(n_active * 1.5), n_active_max)
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Line search
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = normalize((1 - t) * a + t * g)
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-12):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
            if n_active > 15:
                n_active = max(15, int(n_active * 0.9))
        else:
            no_improve += 1
            n_active = min(int(n_active * 1.3), n_active_max)
            if no_improve > 15:
                break
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based fast convolution for efficiency
2. LP best-response with adaptive active set management
3. Heavy-ball subgradient descent with momentum
4. Coordinate descent with exact golden-section 1D optimization
5. Targeted pairwise mass-transfer moves attacking peak indices
6. Multi-length exploration with diverse spectral/structured initializations
7. Population recombination for diversity maintenance

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def proj_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def lp_best_response(a, n_active=30, time_limit=8):
    """Solve LP best response: max sum(g) s.t. conv(a,g)[k] <= M for active k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()

    top_idx = np.argsort(b)[::-1][:min(n_active, 2 * n - 1)]

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_idx):
        k = int(k)
        for j in range(n):
            m = k - j
            if 0 <= m < n and a[m] > 1e-12:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(top_idx), n))
    c = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def targeted_pair_moves(a, n_moves=200, deadline=None, seed=0):
    """Move mass from peak-contributing pairs to reduce the max convolution."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = a.copy()
    best_M = fast_conv(a).max()
    best_a = a.copy()

    for move in range(n_moves):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_star = int(np.argmax(b))
        M = b[k_star]

        # Find largest contributing pair
        i_max = 0
        max_prod = 0
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                prod = a[i] * a[j]
                if prod > max_prod:
                    max_prod = prod
                    i_max = i

        i1 = i_max
        i2 = k_star - i1

        if i1 == i2 or i2 < 0 or i2 >= n:
            # Quadratic term or invalid: reduce a[i1], add elsewhere
            j_dest = rng.integers(n)
            delta = a[i1] * 0.08
            a[i1] -= delta
            a[j_dest] += delta
        else:
            # Transfer from larger to a valley location
            if a[i1] >= a[i2]:
                src = i1
            else:
                src = i2
            # Pick destination from low-convolution region
            low_floor = max(0, k_star - n // 2)
            low_ceil = min(2 * n - 2, k_star + n // 2)
            region = b[low_floor:low_ceil + 1]
            dest_offsets = np.argsort(region)[:max(1, len(region) // 5)]
            k_dest = low_floor + int(dest_offsets[0])
            # Map k_dest back to a valid index
            j_dest = rng.integers(n)
            delta = a[src] * 0.05
            a[src] -= delta
            a[j_dest] += delta

        a = np.maximum(a, 0.0)
        a = normalize(a)
        M_new = fast_conv(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


def coordinate_descent_1d(a, i, b):
    """Exact 1D golden-section optimization for coordinate i."""
    n = len(a)
    x_cur = a[i]

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_lo:k_hi + 1] = False
    base_max = b[mask].max() if mask.any() else 0.0

    c_arr = np.array(c_vals) if c_vals else np.array([0.0])
    s_arr = np.array(slopes) if slopes else np.array([0.0])

    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(2.0 * x_cur + 1e-6, 1.0 / n * 5.0, 0.01)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_cd(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with random ordering per sweep."""
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = np.arange(n)
        rng.shuffle(order)

        b = fast_conv(a)
        imps = 0
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * (x_opt - old) * a[j]
                imps += 1

        a = normalize(a)
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()

        if imps == 0:
            break

    return best_a


def subgradient_descent(a_init, max_iters, deadline, mu=0.85):
    """Heavy-ball subgradient method on max(conv(a,a))."""
    a = normalize(a_init.copy())
    n = len(a)
    v = np.zeros(n)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        # Active set with soft threshold
        threshold = M * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient: average gradient at active indices
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(active)

        grad_tan = grad - grad.mean()
        grad_norm = np.linalg.norm(grad_tan)
        if grad_norm < 1e-15:
            break

        # Adaptive step: Polyak-like
        step = 2.0 / (it + 10.0) * (1.0 / n)
        step = min(step * 10.0, 1.0 / (grad_norm + 1e-10))

        v = mu * v - step * grad_tan
        a_new = a + v
        a_new = proj_simplex(a_new)
        a = a_new

        M_new = fast_conv(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


def spectral_init(n, rng, n_iters=80):
    """Initialize by iteratively reducing peak convolution (flat-top spectrum)."""
    a = np.ones(n) / n
    for _ in range(n_iters):
        b = fast_conv(a)
        # Use multiple active indices for better flattening
        M = b.max()
        active = np.where(b >= M * 0.9)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        corr = np.zeros(n)
        for k in active:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    corr[i] += a[j]
        corr /= len(active)

        a = a - 0.03 * (corr - corr.mean())
        a = np.maximum(a, 1e-8)
        a /= a.sum()

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
