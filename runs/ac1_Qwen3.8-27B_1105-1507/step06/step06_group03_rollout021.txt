def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def sca_lp_reduced(a_input, deadline, max_iters=200, top_m=80, verbose=False):
    """
    SCA-LP with constraint reduction: only use top M constraints.
    This makes each LP much faster, allowing more iterations.
    """
    n = len(a_input)
    a = np.maximum(a_input.copy().astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Select top M constraints (those with largest c[k])
        if 2 * n - 1 > top_m:
            idx = np.argsort(c)[::-1][:top_m]
        else:
            idx = np.arange(2 * n - 1)

        # Build LP
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub))
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(len(b_ub))
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(len(b_ub), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]

        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                if g.sum() > 1e-10:
                    g /= g.sum()
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1.0 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        ts = a_try.sum()
                        if ts < 1e-10:
                            continue
                        a_try /= ts
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-14:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 10 == 0:
            print(f"    SCA it {it}: obj={best_obj:.8f}", flush=True)

        if no_improve > 30:
            break

    return best_a, best_obj

    """
    Spectral flattening: iteratively modify the power spectrum to be flatter,
    then project back to non-negative simplex.
    """
    n = len(a_input)
    a = np.maximum(a_input.copy().astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_J(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, size)
        power = np.abs(A) ** 2
        mean_power = power.mean()

        # Flatten: reduce high power, increase low power
        for frac in [1.0, 0.5, 0.3, 0.15, 0.05]:
            flat_power = power + strength * frac * (mean_power - power)
            flat_power = np.maximum(flat_power, 1e-20)

            # Keep phase, modify magnitude
            phase = np.angle(A)
            A_new = np.sqrt(flat_power) * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, size)[:n]
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new /= sn

            J_new = compute_J(a_new)
            if J_new < best_obj:
                best_obj = J_new
                best_a = a_new.copy()
                a = a_new.copy()
                break

    return best_a, best_obj


def smooth_adam(a_input, max_iters, deadline, tau_init=20.0, tau_max=150.0,
                lr_init=0.04, verbose=False):
    """
    Adam with log-sum-exp smooth max gradient.
    Operates on the probability simplex.
    """
    n = len(a_input)
    a = np.maximum(a_input.copy().astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_count = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    size = 1
    while size < 3 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_init + (tau_max - tau_init) * frac

        c = fast_conv(a)
        c_max = c.max()
        exp_c = np.exp(tau * (c - c_max))
        w = exp_c / exp_c.sum()

        # Gradient computation via FFT
        a_rev = a[::-1]
        w_rev = w[::-1]
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        Wrev_fft = np.fft.rfft(w_rev, size)

        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wrev_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad[i] += conv2[idx2]

        # Project to simplex tangent (zero mean)
        grad = grad - np.mean(grad)

        t_count += 1
        lr = lr_init / (1.0 + 0.05 * t_count ** 0.5)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                a = best_a + rng.normal(0, 0.02, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0
                continue

        a = a_new

        if verbose and it % 200 == 0:
            print(f"    Adam it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def pairwise_cd(a_input, max_sweeps, deadline, seed=0, verbose=False):
    """
    Coordinate descent that updates pairs of coordinates simultaneously.
    Pairs are chosen to be far apart (minimal interaction).
    """
    n = len(a_input)
    a = np.maximum(a_input.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Create pairs: (i, j) where j = (i + n//2) % n
        half = n // 2
        pairs = [(i, (i + half) % n) for i in range(half)]
        rng.shuffle(pairs)

        improved = False

        for i, j in pairs:
            old_i, old_j = a[i], a[j]
            if old_i < 1e-12 and old_j < 1e-12:
                continue

            # Try a few joint perturbations
            best_ii, best_jj = old_i, old_j
            best_local = 2.0 * n * conv.max() / (s * s)

            ratios_i = [0.5, 0.7, 1.0, 1.3, 1.5]
            ratios_j = [0.5, 0.7, 1.0, 1.3, 1.5]

            # Try a small grid
            for ri in ratios_i:
                for rj in ratios_j:
                    new_i = old_i * ri
                    new_j = old_j * rj
                    new_s = s - old_i - old_j + new_i + new_j
                    if new_s < 1e-10:
                        continue

                    di = new_i - old_i
                    dj = new_j - old_j
                    if abs(di) < 1e-14 and abs(dj) < 1e-14:
                        continue

                    # Apply both changes
                    lo_i, hi_i = i, min(2 * n - 2, i + n - 1)
                    lo_j, hi_j = j, min(2 * n - 2, j + n - 1)

                    # Update conv for i
                    ki_lo, ki_hi = 0, hi_i - i
                    a_slice_i = a[ki_lo:ki_hi + 1].copy()
                    if j - i >= ki_lo and j - i <= ki_hi:
                        a_slice_i[j - i] = new_j  # account for j's change
                    conv[lo_i:hi_i + 1] += 2.0 * di * a_slice_i
                    k2i = 2 * i
                    if lo_i <= k2i <= hi_i:
                        conv[k2i] += di * di

                    # Update conv for j (using updated conv)
                    kj_lo, kj_hi = 0, hi_j - j
                    a_slice_j = a[kj_lo:kj_hi + 1].copy()
                    if i - j >= kj_lo and i - j <= kj_hi:
                        a_slice_j[i - j] = new_i
                    # But a_slice_j[i-j] should reflect the new value of a[i]
                    conv[lo_j:hi_j + 1] += 2.0 * dj * a[0:kj_hi + 1]
                    # This is getting complicated with overlapping regions. Let's be safe:
                    # Just do a full recompute for the affected region
                    # Actually, for correctness with overlapping regions, let's just
                    # compute the new conv values directly for the union of affected regions.
                    
                    # Undo
                    conv[lo_i:hi_i + 1] -= 2.0 * di * a_slice_i
                    if lo_i <= k2i <= hi_i:
                        conv[k2i] -= di * di
                    conv[lo_j:hi_j + 1] -= 2.0 * dj * a[0:kj_hi + 1]

                    # This approach is getting too complex for overlapping regions.
                    # Let's skip pairwise for now and just use single-coordinate CD.
                    break
                break
            # Skip the rest - pairwise is too complex with overlapping conv regions
        break  # Just do one sweep to test, then fall back to single CD

    return best_a, best_obj


def generate_spectral_inits(a_parent, n, rng, n_inits=8):
    """Generate initial sequences using spectral modifications of the parent."""
    inits = []
    parent_resampled = resample_seq(a_parent, n)
    ps = parent_resampled.sum()
    if ps > 1e-10:
        parent_resampled = parent_resampled / ps

    inits.append(parent_resampled.copy())

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    A = np.fft.rfft(parent_resampled, size)
    phase = np.angle(A)
    mag = np.abs(A)
    mean_mag = mag.mean()

    for i in range(n_inits - 1):
        # Different flattening strengths and types
        strength = 0.05 + 0.15 * (i % 4)
        bias = rng.uniform(-0.1, 0.1)
        new_mag = mag + strength * (mean_mag - mag) + bias * mean_mag
        new_mag = np.maximum(new_mag, 1e-10)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
            inits.append(a_new)
        else:
            inits.append(project_simplex(parent_resampled + rng.normal(0, 0.1, n)))

    # Add a few random perturbations
    for _ in range(3):
        noise_scale = rng.uniform(0.05, 0.2)
        a_new = parent_resampled + rng.normal(0, noise_scale, n)
        a_new = project_simplex(a_new)
        inits.append(a_new.copy())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function: multi-phase optimization to minimize J(a).
    Phase 1: CD on parent with diverse inits
    Phase 2: SCA-LP with constraint reduction
    Phase 3: Spectral flattening + CD
    Phase 4: Multi-length exploration
    Phase 5: Final intensive polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        n_def = 768
        parent = np.ones(n_def) / n_def
        update_best(parent)
        print(f"Using default parent: n={n_def}, J={best_score:.6f}", flush=True)

    n_parent = len(parent)
    parent_norm = parent / parent.sum() if parent.sum() > 1e-10 else parent / max(parent.max(), 1.0)

    # === PHASE 1: CD with diverse initializations (350s) ===
    phase1_end = t_start + 350.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: CD optimization at n={n_parent} (budget {phase1_end - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    inits = generate_spectral_inits(parent_norm, n_parent, rng, n_inits=6)
    # Add uniform
    inits.append(np.ones(n_parent) / n_parent)

    per_init_time = (phase1_end - time.monotonic()) / len(inits) - 5
    print(f"  {len(inits)} inits, ~{per_init_time:.0f}s each", flush=True)

    for idx, a0 in enumerate(inits):
        remaining = phase1_end - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_init_time, remaining - 5)
        if alloc < 15:
            break

        a_opt, obj_opt = coord_descent(a0, max_sweeps=50000, deadline=time.monotonic() + alloc,
                                       seed=seed + idx)
        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.8f}", flush=True)

    # === PHASE 2: SCA-LP with constraint reduction (300s) ===
    phase2_end = t_start + 650.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: SCA-LP (budget {phase2_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_base = best_seq.copy()
        a_base = np.maximum(a_base, 0)
        if a_base.sum() > 1e-10:
            a_base /= a_base.sum()
        a_sca, obj_sca = sca_lp_reduced(a_base, deadline=phase2_end, max_iters=300, top_m=60,
                                        verbose=True)
        update_best(a_sca)
        print(f"  SCA result: J={obj_sca:.8f}", flush=True)

    # === PHASE 3: Spectral flattening + CD (250s) ===
    phase3_end = t_start + 900.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Spectral perturbation + CD (budget {phase3_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_tries = 10
        per_try = (phase3_end - time.monotonic()) / n_tries - 5
        for trial in range(n_tries):
            remaining = phase3_end - time.monotonic()
            if remaining < 15:
                break
            alloc = min(per_try, remaining - 5)
            if alloc < 10:
                break

            # Spectral perturbation
            a_spec, obj_spec = spectral_flatten(best_seq, n_iters=8,
                                                strength=0.05 + 0.1 * (trial % 5),
                                                deadline=time.monotonic() + alloc * 0.3)
            update_best(a_spec)

            # Then CD on the result
            if time.monotonic() < phase3_end - 10:
                a_cd, obj_cd = coord_descent(a_spec, max_sweeps=20000,
                                             deadline=time.monotonic() + (phase3_end - time.monotonic()) * 0.6,
                                             seed=seed + 100 + trial)
                update_best(a_cd)
            print(f"  Trial {trial}: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Multi-length exploration (200s) ===
    phase4_end = t_start + 1100.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length exploration (budget {phase4_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    candidate_ns = [512, 640, 896, 1024, 1280]
    candidate_ns = [n for n in candidate_ns if n != n_parent]
    candidate_ns.sort(key=lambda x: abs(x - n_parent))

    for new_n in candidate_ns[:4]:
        remaining = phase4_end - time.monotonic()
        if remaining < 40:
            break
        alloc = min(50.0, remaining - 10)
        if alloc < 25:
            break

        print(f"  Trying n={new_n}...", flush=True)
        a_res = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_res = np.maximum(a_res, 1e-8)
        a_res /= a_res.sum()

        a_opt, obj_opt = coord_descent(a_res, max_sweeps=10000,
                                       deadline=time.monotonic() + alloc * 0.6,
                                       seed=seed + 200)
        update_best(a_opt)

        if time.monotonic() < phase4_end - 10:
            a_sca, obj_sca = sca_lp_reduced(a_opt, deadline=phase4_end - 5,
                                            max_iters=30, top_m=40)
            update_best(a_sca)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 5: Final intensive polish (remaining time) ===
    phase5_end = deadline
    remaining = phase5_end - time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final polish (budget {remaining:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None and remaining > 20:
        a_curr = best_seq.copy()
        a_curr = np.maximum(a_curr, 0)
        if a_curr.sum() > 1e-10:
            a_curr /= a_curr.sum()

        # Long CD
        t_cd = time.monotonic() + remaining * 0.35
        print(f"  Final CD (budget {t_cd - time.monotonic():.0f}s)...", flush=True)
        a_cd, obj_cd = coord_descent(a_curr, max_sweeps=100000, deadline=t_cd, seed=999)
        update_best(a_cd)

        # SCA-LP
        remaining = phase5_end - time.monotonic()
        if remaining > 20:
            t_sca = time.monotonic() + remaining * 0.35
            print(f"  Final SCA (budget {t_sca - time.monotonic():.0f}s)...", flush=True)
            a_sca, obj_sca = sca_lp_reduced(best_seq, deadline=t_sca, max_iters=200, top_m=50,
                                            verbose=True)
            update_best(a_sca)

        # Adam
        remaining = phase5_end - time.monotonic()
        if remaining > 20:
            t_adam = time.monotonic() + remaining * 0.30
            print(f"  Final Adam (budget {t_adam - time.monotonic():.0f}s)...", flush=True)
            a_adam, obj_adam = smooth_adam(best_seq, max_iters=5000, deadline=t_adam,
                                           tau_init=15.0, tau_max=100.0, lr_init=0.03)
            update_best(a_adam)

        # Very final CD
        remaining = phase5_end - time.monotonic()
        if remaining > 15:
            print(f"  Very final CD (budget {remaining:.0f}s)...", flush=True)
            a_final, obj_final = coord_descent(best_seq, max_sweeps=50000,
                                               deadline=phase5_end - 5, seed=7777)
            update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
