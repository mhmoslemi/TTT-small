"""
Optimized step function search minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses subgradient-guided coordinate descent, SCA-LP polishing, spectral initialization,
and multi-length exploration with random restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent_fast(a, deadline, seed=0, verbose=False, min_time_per_sweep=0.001):
    """
    Fast coordinate descent with incremental convolution updates.
    Uses subgradient-guided ordering and adaptive candidates.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)

    # Precompute affected ranges
    lo_bounds = np.maximum(0, np.arange(n))
    hi_bounds = np.minimum(2 * n - 2, np.arange(n) + n - 1)

    no_improve = 0
    max_no_improve = 500
    sweep = 0

    while time.monotonic() < deadline:
        if no_improve > max_no_improve:
            break

        # Determine update order: prioritize coordinates with large subgradient
        c_max = conv.max()
        k_max = np.argmax(conv)
        # Subgradient magnitude for each coordinate i:
        # proportional to |a[k_max - i]| + |a[i - k_max]| (when in range)
        sg = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n and j != i:
                sg[i] = 2.0 * a[j]
            elif 0 <= j < n and j == i:
                sg[i] = 2.0 * a[i]
        # Add small random component for diversity
        sg += rng.random(n) * 0.01 * (c_max if c_max > 0 else 1.0)

        # Sort by subgradient magnitude (descending)
        order = np.argsort(-sg)

        # Shuffle within groups to avoid systematic bias
        for start in range(0, n, max(1, n // 50)):
            end = min(start + max(1, n // 50), n)
            perm = rng.permutation(end - start)
            order[start:end] = order[start:end][perm]

        improved = False

        for i in order:
            if time.monotonic() > deadline:
                return best_obj, best_a

            old = a[i]
            lo = lo_bounds[i]
            hi = hi_bounds[i]
            a_slice = a[lo - i:hi - i + 1].copy()
            k_self = 2 * i

            if old < 1e-12:
                candidates = np.array([0.0, 1e-4, 1e-3, 1e-2, 1e-1, 0.3])
            else:
                # Adaptive candidates: include values that could equalize peaks
                c_max = conv.max()
                c_second = np.partition(conv, -2)[-2]
                # Value that would reduce the peak
                if hi - lo + 1 > 0:
                    peak_contrib = 2.0 * old  # self contribution at k=2i
                    target = (c_second - (conv[k_self] - peak_contrib)) / (2.0 * old) if old > 1e-15 else 0
                    target = max(0.0, min(target, 3.0 * old))
                else:
                    target = 0.0

                candidates = np.array([0.0, 0.2 * old, 0.5 * old, 0.8 * old,
                                       old, 1.2 * old, 1.5 * old, 2.0 * old, target])
                candidates = np.unique(candidates)
                candidates = candidates[candidates >= 0]

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue

                # Apply update
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if 0 <= k_self < 2 * n - 1:
                    conv[k_self] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                # Revert
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if 0 <= k_self < 2 * n - 1:
                    conv[k_self] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if 0 <= k_self < 2 * n - 1:
                    conv[k_self] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        sweep += 1
        if verbose and sweep % 100 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}", flush=True)

    return best_obj, best_a


def sca_lp_step(a, deadline, max_iters=200, verbose=False):
    """
    SCA-LP: Solve LP subproblem to find good update direction.
    LP: minimize t s.t. 2*(a*g)[k] - t <= conv(a,a)[k], sum(g)=1, g>=0
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # For efficiency, only use constraints where conv is in the top 30%
        threshold = np.percentile(c, 70)
        active = np.where(c >= threshold)[0]

        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub))
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(len(b_ub))
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        if len(b_ub) == 0:
            break

        A = coo_matrix((vals, (rows, cols)), shape=(len(b_ub), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 20, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_obj


def smooth_adam_descent(a, deadline, tau_init=30.0, tau_max=200.0,
                        lr_init=0.05, max_iters=5000, verbose=False):
    """Adam optimization with log-sum-exp smoothing of max(conv(a,a))."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_count = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_init + (tau_max - tau_init) * frac

        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()

        # Compute gradient via FFT
        a_rev = a[::-1]
        w_rev = w[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        Wrev_fft = np.fft.rfft(w_rev, size)

        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wrev_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad[i] += conv2[idx2]

        grad = grad - np.mean(grad)

        t_count += 1
        lr = lr_init / (1.0 + 0.1 * t_count ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                rng = np.random.default_rng(it * 1000 + 42)
                a = best_a + rng.normal(0, 0.03, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.12, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Spectral-based: modulated sinusoids
    for freq in [1, 2, 3]:
        for phase in [0, np.pi / 3, 2 * np.pi / 3]:
            sig = 1.0 + 0.6 * np.cos(2 * np.pi * freq * t + phase)
            inits.append(project_simplex(sig))

    # Asymmetric patterns (key insight: non-symmetry helps)
    inits.append(project_simplex(np.linspace(0.1, 1.0, n)))
    inits.append(project_simplex(np.linspace(1.0, 0.1, n)))
    inits.append(project_simplex(np.abs(np.sin(np.pi * t)) ** 2 + 0.1))
    inits.append(project_simplex(np.abs(np.sin(np.pi * t * 1.5)) ** 2 + 0.1))

    # Random smooth
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    return inits


def solve_one(a, deadline, verbose=False):

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 4: Final CD polish (10% of time)
    t4 = deadline - 1
    obj4, a4 = coord_descent_fast(best_a, t4, seed=999, verbose=False)
    try_update(a4)

    return best_a, best_obj


def sce_lp_polish(a, deadline, max_iters=100):
    """SCA-LP polish with progressive constraint tightening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Use all constraints for small n, top constraints for large n
        if n <= 512:
            active = np.arange(2 * n - 1)
        else:
            threshold = np.percentile(c, 60)
            active = np.where(c >= threshold)[0]

        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub))
                cols.append(j)
                vals.append(a[k - j])
            rows.append(len(b_ub))
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k] * 0.95)  # Slightly tighter for progress

        if len(b_ub) == 0:
            break

        A = coo_matrix((vals, (rows, cols)), shape=(len(b_ub), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 25:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses spectral-inspired initialization, subgradient-guided CD, SCA-LP, Adam,
    and multi-length exploration with random restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Optimize parent at native length (350s) ===
    phase1_deadline = t_start + 350.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Optimize parent at n={n_parent}", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Try multiple initializations at parent length
    inits = generate_inits(n_parent, rng, parent)
    n_init = min(len(inits), 4)
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, n_init) - 3

    for idx in range(n_init):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 20:
            break
        alloc = min(per_init_time, remaining_check - 3)
        if alloc < 15:
            break

        a0 = inits[idx].copy()
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt, obj_opt = solve_one(a0, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Multi-length exploration (400s) ===
    phase2_deadline = t_start + 750.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidate_ns = sorted(set([256, 384, 512, 640, 768, 896, 1024, 1280, 1536]),
                          key=lambda x: abs(x - n_parent))
    # Remove parent length (already done)
    candidate_ns = [n for n in candidate_ns if n != n_parent][:5]

    for new_n in candidate_ns:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 50:
            break
        alloc = min(120.0, remaining_check - 10)
        if alloc < 30:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = solve_one(a_ext, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive optimization on best (400s) ===
    phase3_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive optimization (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase3_deadline - time.monotonic()

        if remaining > 60:
            # Pass 1: Long CD
            t_pass1 = time.monotonic() + 0.30 * remaining
            print(f"  CD pass 1 (budget {t_pass1 - time.monotonic():.0f}s)...", flush=True)
            _, a_curr = coord_descent_fast(best_seq.copy(), t_pass1, seed=777)
            update_best(a_curr)

            remaining = phase3_deadline - time.monotonic()

        if remaining > 60:
            # Pass 2: SCA-LP
            t_pass2 = time.monotonic() + 0.25 * remaining
            print(f"  SCA-LP pass (budget {t_pass2 - time.monotonic():.0f}s)...", flush=True)
            a_sca, obj_sca = sce_lp_polish(best_seq.copy(), t_pass2, max_iters=150)
            update_best(a_sca)

            remaining = phase3_deadline - time.monotonic()

        if remaining > 60:
            # Pass 3: Adam
            t_pass3 = time.monotonic() + 0.20 * remaining
            print(f"  Adam pass (budget {t_pass3 - time.monotonic():.0f}s)...", flush=True)
            a_adam, obj_adam = smooth_adam_descent(best_seq.copy(), t_pass3,
                                                   tau_init=15, tau_max=100,
                                                   lr_init=0.02, max_iters=5000)
            update_best(a_adam)

            remaining = phase3_deadline - time.monotonic()

        if remaining > 30:
            # Pass 4: Random restart + re-optimize
            print(f"  Restart pass (budget {remaining:.0f}s)...", flush=True)
            for restart in range(3):
                if time.monotonic() > phase3_deadline - 20:
                    break
                pert = best_seq + rng.normal(0, 0.02 * (restart + 1), len(best_seq))
                pert = project_simplex(pert)
                a_r, obj_r = solve_one(pert, min(time.monotonic() + 40, phase3_deadline - 10))
                update_best(a_r)

        # Final CD polish
        if time.monotonic() < phase3_deadline - 10:
            _, a_final = coord_descent_fast(best_seq.copy(), phase3_deadline - 5, seed=12345)
            update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
