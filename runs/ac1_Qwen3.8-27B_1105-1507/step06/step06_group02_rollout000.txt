"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Fast full convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective value J."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse matrix A such that (A @ g) = conv(a, g) truncated to 2n-1 elements."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M):
    """Solve the LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 60})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def backtracking_step(a, g, M_current, max_backtracks=10):
    """Find step size via backtracking: minimize max(conv(a_new, a_new))."""
    n = len(a)
    best_t = 0.01
    best_M = fast_conv(a).max()
    
    t = 0.5
    for _ in range(max_backtracks):
        a_new = np.maximum((1 - t) * a + t * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        a_new /= s
        M_new = fast_conv(a_new).max()
        if M_new < M_current - 1e-12:
            return a_new, M_new
        best_t = t
        best_M = min(best_M, M_new)
        t *= 0.5
    
    # Return the best we found
    a_new = np.maximum((1 - best_t) * a + best_t * g, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new, fast_conv(a_new).max()


def anderson_mingle(history_x, history_Fx, m, gamma=0.5):
    """Anderson mixing: find optimal linear combination of history."""
    if m < 2:
        return history_Fx[-1].copy()
    
    # Residuals
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    # G matrix: G[i,j] = <r_i, r_j>
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    
    try:
        # Solve for alpha: minimize ||sum alpha_i r_i||^2 s.t. sum alpha_i = 1
        # Gradient: 2 G alpha = lambda * 1
        # => G alpha = (lambda/2) * 1
        # With constraint: 1^T alpha = 1
        # Augmented system: [G, 1; 1^T, 0] [alpha; lambda/2] = [0; 1]
        m_aug = m + 1
        A_aug = np.zeros((m_aug, m_aug))
        A_aug[:m, :m] = G
        A_aug[:m, m] = 1.0
        A_aug[m, :m] = 1.0
        b_aug = np.zeros(m_aug)
        b_aug[m] = 1.0
        
        sol = np.linalg.solve(A_aug, b_aug)
        alpha = sol[:m]
    except np.linalg.LinAlgError:
        alpha = np.ones(m) / m
    
    # Clip and renormalize
    alpha = np.clip(alpha, -0.5, 2.0)
    alpha_sum = alpha.sum()
    if alpha_sum < 1e-10:
        alpha = np.ones(m) / m
    else:
        alpha /= alpha_sum
    
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)

        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
    
    return best_a


def gs_spectral(a, n_iters=100, beta=0.85):
    """Gerchberg-Saxton spectral flattening."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for _ in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress dynamic range
        power = 0.5 + 0.5 * beta
        mag_p = mag ** power
        mag_p = mag_p / (mag_p.max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_p
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 1e-10)
        sc = a_candidate.sum()
        if sc < 1e-10:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    
    return best_a


def make_symmetric(a):
    """Make a sequence symmetric."""
    n = len(a)
    half = (a[:n//2+1] + a[n-1:n//2-1:-1]) / 2.0
    if n % 2 == 0:
        return np.concatenate([a[:n//2], a[:n//2-1:-1]])
    else:
        return np.concatenate([a[:n//2], [a[n//2]], a[n//2-1::-1]])


def resample(src, n):
    """Resample a sequence to a new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, max(src.size, 2))
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
