**Approach: Direct optimization of the spectral flatness**

The autocorrelation is the IFFT of the power spectrum. So:
(a*a)[k] = IFFT(|FFT(a)|^2)[k]

We want to minimize max_k IFFT(|FFT(a)|^2)[k].

This is equivalent to: minimize the peak of the inverse DFT of the periodogram.

In the frequency domain, this is: minimize max_k sum_f |A(f)|^2 * exp(2πifk/N) / N.

This is a max over k of a sum of exponentials weighted by |A(f)|^2. It's like wanting the periodogram to be as "white" as possible.

We can parameterize a in terms of its DFT coefficients (with the constraint that a >= 0). The constraint a >= 0 becomes: IFFT(A) >= 0, which is a linear constraint on A.

So the problem becomes:
minimize max_k sum_f |A(f)|^2 * exp(2πifk/N) / N
subject to IFFT(A) >= 0, sum(IFFT(A)) = 1, A is conjugate symmetric.

This is still non-convex (the objective is the max of linear functions of |A|^2, which is convex in |A|^2 but not in A).

However, if we fix |A(f)| and only optimize the phase, the problem becomes:
minimize max_k sum_f |A(f)|^2 * exp(i*phase(f) + 2πifk/N) / N (taking real part)

This is still non-convex in the phases.

"""
Peak-reduction optimizer using adaptive subgradient + sparse LP hybrid.
Uses FFT-based convolutions, top-K peak targeting, and periodic LP-based
direction finding. Explores multiple sequence lengths and initializations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_maxsum_lp_sparse(a, M, K, time_limit=10):
    """
    Solve: max sum(g) subject to (a*g)[k] <= M for top-K most critical k, g >= 0.
    Uses only K constraints (the ones where (a*a)[k] is highest).
    """
    n = len(a)
    conv_a = compute_conv(a)
    # Get top-K lags
    top_k = np.argsort(conv_a)[-K:]
    A = build_toeplitz_convolve_mat(a)
    A_sub = A[top_k, :]
    b_sub = np.full(K, M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_maxsum_lp_full(a, M, time_limit=10):
    """Solve: max sum(g) subject to (a*g)[k] <= M for ALL k, g >= 0."""
    n = len(a)
    A = build_toeplitz_convolve_mat(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def compute_subgradient(a, weights):
    """
    Compute subgradient of sum_k weights[k] * (a*a)[k] with respect to a.
    grad[i] = 2 * sum_k weights[k] * a[(i+k) mod ...] (with bounds)
    Uses FFT for efficiency.
    """
    n = len(a)
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    w_ext = np.zeros(fft_size)
    w_ext[:len(weights)] = weights
    W_ext = rfft(w_ext, fft_size)
    A_ext = rfft(a, fft_size)
    corr = irfft(A_ext * np.conj(W_ext), fft_size)
    grad = 2.0 * corr[:n]
    return grad


def adaptive_subgradient(a, max_iters, time_budget, top_k=5, use_lp_every=20):
    """
    Adaptive subgradient with periodic LP direction finding.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = 0.1
    v = np.zeros(n)
    mu = 0.8
    stall = 0
    tau = 0.005  # temperature for softmax weights
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Compute weights: softmax over top-K with temperature
        idx = np.argsort(conv)[-top_k:]
        weights = np.zeros(len(conv))
        top_vals = conv[idx]
        shifted = (top_vals - top_vals.max()) / tau
        w_vals = np.exp(shifted)
        w_vals /= w_vals.sum()
        for i, idx_i in enumerate(idx):
            weights[idx_i] = w_vals[i]
        
        # Compute subgradient
        grad = compute_subgradient(a, weights)
        grad_centered = grad - np.mean(grad)
        gn = np.linalg.norm(grad_centered)
        
        if gn < 1e-15:
            break
        
        # Nesterov step
        v = mu * v - step * grad_centered / gn
        a_new = a + v  # note: we add because we're minimizing, and gradient points to increase
        # Wait, we want to DECREASE the objective, so we should go AGAINST the gradient
        # The gradient of max(conv) points in the direction that INCREASES the peak
        # So we should subtract: a_new = a - step * grad
        a_new = a - v  # v already has the negative sign from the update
        # Actually let me be more careful:
        # We want to minimize f(a) = max_k (a*a)[k]
        # Subgradient of f at a: g = (1/|S|) * sum_{k in S} d(a*a)[k]/da where S = argmax
        # Descent direction: -g
        # With Nesterov: v = mu*v_prev - step*g, then a_new = a + v
        # So v accumulates the negative gradient direction
        
        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Try smaller step
            if stall % 5 == 0:
                step *= 0.9
        
        # Periodic LP direction
        if use_lp_every > 0 and it % use_lp_every == use_lp_every - 1 and n <= 1000:
            remaining = time_budget - (time.monotonic() - start)
            if remaining > 15:
                M = float(np.max(compute_conv(a)))
                K = min(50, 2 * n - 1)
                g_dir = solve_maxsum_lp_sparse(a, M, K, time_limit=min(10, remaining - 5))
                if g_dir is not None:
                    # Blend
                    for t_blend in [0.3, 0.5, 0.7, 1.0]:
                        a_try = (1 - t_blend) * a + t_blend * g_dir
                        a_try = normalize_simplex(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            stall = 0
                            break
        
        # Adapt step size
        step = max(0.01, min(0.2, step))
        tau = max(0.001, tau * 0.999)
    
    return best_a, best_obj


def hamming_window(n):
    """Generate a Hamming window normalized to sum=1."""
    n = int(n)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    return w / np.sum(w)


def blackman_window(n):
    """Generate a Blackman window normalized to sum=1."""
    n = int(n)
    w = 0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1)) + \
        0.08 * np.cos(4 * np.pi * np.arange(n) / (n - 1))
    w = np.maximum(w, 0)
    s = np.sum(w)
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def raised_cosine(n, beta=0.5):
    """
    Raised cosine window: high at edges, dip in middle.
    This is intended to reduce the central autocorrelation peak.
    """
    n = int(n)
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = np.abs(i - center) / center
    w = 1.0 - beta * np.cos(np.pi * np.clip(x, 0, 1))
    w = np.maximum(w, 0.01)
    return w / np.sum(w)


def w_shape(n, num_peaks=3):
    """Generate a W-shaped sequence with multiple peaks."""
    n = int(n)
    i = np.arange(n, dtype=np.float64)
    w = np.zeros(n)
    peak_positions = np.linspace(0, n - 1, num_peaks)
    sigma = n / (2 * num_peaks)
    for p in peak_positions:
        w += np.exp(-((i - p) ** 2) / (2 * sigma ** 2))
    w = np.maximum(w, 0.01)
    return w / np.sum(w)

"""
Peak-reduction optimizer using direct subgradient with backtracking line search,
periodic sparse/full LP direction finding, diverse window-based initialization,
and length exploration. Targets improvement below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def compute_grad_direct(a, weights):
    """
    Compute subgradient of sum_k weights[k]*(a*a)[k] w.r.t. a.
    weights has length 2n-1 (the convolution length).
    Only non-zero entries of weights are processed.
    """
    n = len(a)
    grad = np.zeros(n)
    nz = np.nonzero(np.abs(weights) > 1e-15)[0]
    for k in nz:
        wk = weights[k]
        if k == 0:
            grad += 2.0 * wk * a
        else:
            if k < n:
                grad[:n - k] += wk * a[k:]
                grad[k:] += wk * a[:n - k]
    return grad


def solve_maxsum_lp(a, M, K, time_limit=10.0):
    """
    Solve: max sum(g) subject to (a*g)[k] <= M for top-K lags, g >= 0.
    K = None means use all constraints.
    """
    n = len(a)
    conv_a = compute_conv(a)
    if K is None or K >= 2 * n - 1:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.argsort(conv_a)[-K:]
    A = build_toeplitz_convolve_mat(a)
    A_sub = A[idx, :]
    b_sub = np.full(len(idx), M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def backtracking_search(a, grad, step_init=0.1, max_iters=30):
    """
    Backtracking line search for f((1-t)*a + t*(1 - grad)*a) or similar.
    Here we search for the best t in [0,1] for a_new = a - t*grad normalized.
    Returns (best_a, best_obj, best_t).
    """
    n = len(a)
    best_obj = compute_obj(a)
    best_a = a.copy()
    best_t = 0.0
    step = step_init
    for _ in range(max_iters):
        a_try = a - step * grad
        a_try = normalize_simplex(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            best_t = step
            step = min(step * 1.2, 0.5)
        else:
            step *= 0.6
            if step < 0.001:
                break
    return best_a, best_obj, best_t

            stall = 0

        # Slowly decrease temperature
        tau = max(0.0005, tau * 0.9995)

    return best_a, best_obj


def hamming_window(n):
    w = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / max(n - 1, 1))
    return w / np.sum(w)


def blackman_window(n):
    i = np.arange(n)
    denom = max(n - 1, 1)
    w = 0.42 - 0.5 * np.cos(2 * np.pi * i / denom) + 0.08 * np.cos(4 * np.pi * i / denom)
    w = np.maximum(w, 0)
    s = np.sum(w)
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def nuttall_window(n):
    i = np.arange(n)
    denom = max(n - 1, 1)
    w = (0.36358 - 0.48917 * np.cos(2 * np.pi * i / denom)
         + 0.13659 * np.cos(4 * np.pi * i / denom)
         - 0.01064 * np.cos(6 * np.pi * i / denom))
    w = np.maximum(w, 0)
    s = np.sum(w)
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def raised_cosine_window(n, beta=0.5):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = np.abs(i - center) / center
    w = 1.0 - beta * np.cos(np.pi * np.clip(x, 0, 1))
    w = np.maximum(w, 0.01)
    return w / np.sum(w)


def multi_gauss_window(n, num_peaks=3, spread=0.3):
    i = np.arange(n, dtype=np.float64)
    w = np.zeros(n)
    positions = np.linspace(0, n - 1, num_peaks)
    sigma = n * spread
    for p in positions:
        w += np.exp(-((i - p) ** 2) / (2 * sigma ** 2))
    w = np.maximum(w, 0.01)
    return w / np.sum(w)


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(budget_s=1500):
    """
    Main search: multi-start with diverse windows, adaptive subgradient + LP,
    length exploration, and final polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Phase 0: Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # Phase 1: Multi-start exploration
    print("\n=== Phase 1: Multi-start exploration ===")

    n_values = sorted(set([
        n_parent - 100, n_parent - 50, n_parent, n_parent + 50,
        n_parent + 100, n_parent + 200,
        400, 500, 600, 800, 900, 1000, 1200
    ]))
    n_values = [x for x in n_values if 300 <= x <= 1400]

    candidates = []

    for n in n_values:
        if time.monotonic() > deadline - 250:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        starts = []
        starts.append(np.ones(n) / n)
        starts.append(hamming_window(n))
        starts.append(blackman_window(n))
        starts.append(nuttall_window(n))
        starts.append(raised_cosine_window(n, 0.5))
        starts.append(raised_cosine_window(n, 0.8))
        starts.append(multi_gauss_window(n, 3, 0.3))
        starts.append(multi_gauss_window(n, 5, 0.2))

        if best_seq is not None:
            if len(best_seq) == n:
                starts.append(normalize_simplex(best_seq.copy()))
            else:
                starts.append(normalize_simplex(resample_sequence(best_seq, n)))

        rng = np.random.default_rng(42 + n)
        for gi in range(3):
            r = np.abs(rng.normal(0, 1, n)) + 0.05
            starts.append(r / np.sum(r))

        alloc_per = min(remaining / len(starts), 18.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(alloc_per, 15.0)
            if t_alloc < 3:
                break

            a_opt, obj_opt = optimize_subgradient_lp(s, t_alloc, top_k=5,
                                                     lp_every=20, lp_K=50)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {obj_opt:.8f} ** NEW BEST **")
            else:
                print(f"  n={n}, start {si}: {obj_opt:.8f}")
            candidates.append((obj_opt, a_opt.copy(), n))

    candidates.sort(key=lambda x: x[0])
    print(f"\nTop 5: {[f'{c[0]:.6f}' for c in candidates[:5]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 2: Intensive optimization of top candidates
    print("\n=== Phase 2: Intensive optimization ===")

    top_k = min(3, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand = candidates[ci][1].copy()
        obj_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        alloc = min(remaining * 0.35, 100.0)
        print(f"  Cand {ci} (n={n_cand}, obj={obj_cand:.8f}): {alloc:.0f}s")

        a_pol, obj_pol = optimize_subgradient_lp(a_cand, alloc, top_k=7,
                                                  lp_every=15, lp_K=80)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {obj_pol:.8f} **")
        else:
            print(f"  No improvement ({obj_pol:.8f})")

    # Phase 3: Full LP optimization on the best
    print(f"\n=== Phase 3: LP polish (best={best_score:.8f}) ===")

    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        a = normalize_simplex(best_seq.copy())
        a_obj = compute_obj(a)

        for lp_iter in range(40):
            if time.monotonic() > deadline - 25:
                break
            rem = deadline - time.monotonic()
            if rem < 12:
                break

            conv = compute_conv(a)
            M = float(np.max(conv))

            if n_best <= 1000:
                K_use = min(80, 2 * n_best - 1)
            else:
                K_use = 100

            g = solve_maxsum_lp(a, M, K_use, time_limit=min(12.0, rem - 5))
            if g is None:
                if n_best <= 800:
                    g = solve_maxsum_lp(a, M, None, time_limit=min(12.0, rem - 5))
                if g is None:
                    break

            improved = False
            for t in [0.4, 0.3, 0.5, 0.2, 0.6, 0.7, 0.8, 0.1, 1.0]:
                a_try = (1.0 - t) * a + t * g
                a_try = normalize_simplex(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < a_obj:
                    a = a_try
                    a_obj = obj_try
                    improved = True
                    break

            if not improved:
                rng = np.random.default_rng(lp_iter * 4253 + 7)
                a = a * np.exp(rng.normal(0, 0.025, n_best))
                a = normalize_simplex(a)
                a_obj = compute_obj(a)

            if a_obj < best_score:
                best_score = a_obj
                best_seq = a.copy()
                print(f"  LP {lp_iter}: {best_score:.8f}")

    # Phase 4: Final perturbations and re-optimization
    print(f"\n=== Phase 4: Final perturbations (best={best_score:.8f}) ===")

    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        rng = np.random.default_rng(99999)

        for p in range(10):
            if time.monotonic() > deadline - 8:
                break
            t_alloc = min(remaining * 0.1, 12.0)
            if t_alloc < 3:
                break

            scale = 0.015 + 0.015 * p
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_pert = normalize_simplex(a_pert)

            a_opt, obj_opt = optimize_subgradient_lp(a_pert, t_alloc, top_k=7,
                                                     lp_every=15, lp_K=60)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Pert {p} (s={scale:.3f}): ** {best_score:.8f} **")

    # Phase 5: One final LP round if time remains
    if best_seq is not None and time.monotonic() < deadline - 15:
        rem = deadline - time.monotonic()
        n_best = len(best_seq)
        if n_best <= 900 and rem > 15:
            a = normalize_simplex(best_seq.copy())
            for _ in range(5):
                if time.monotonic() > deadline - 10:
                    break
                rem2 = deadline - time.monotonic()
                if rem2 < 8:
                    break
                M = float(np.max(compute_conv(a)))
                g = solve_maxsum_lp(a, M, min(60, 2 * n_best - 1),
                                    time_limit=min(8.0, rem2 - 3))
                if g is None:
                    break
                for t in [0.3, 0.5, 0.7]:
                    a_try = (1 - t) * a + t * g
                    a_try = normalize_simplex(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_score:
                        best_score = obj_try
                        best_seq = a_try.copy()
                        a = a_try
                        break

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
