18. **Cutting plane / ellipsoid approach**:

Actually, for the QP: min t s.t. a^T Q_k a <= t for k in S, 1^T a = 1, a >= 0

The Lagrangian is: L = t + sum_{k in S} λ_k (a^T Q_k a - t) - μ(1^T a - 1) + sum_i ν_i a_i

KKT conditions:

- λ_k (a^T Q_k a - t) = 0 (complementary slackness)
- ν_i a_i = 0 (complementary slackness for non-negativity)
- a^T Q_k a <= t, λ_k >= 0
- 1^T a = 1, a >= 0, t free

This is complex to solve directly. But if the active set S is small (say |S| <= 20), we can use scipy's SLSQP or trust-constr.

And the "Hessian" is the matrix H with H[i,j] = p[i+j].

This matrix has a special structure! It's a "Hankel-like" matrix (constant along anti-diagonals). Actually, it's the opposite of Hankel: H[i,j] depends on i+j, not i-j. This is sometimes called a "flip-Hankel" or "anti-Hankel" matrix.

a = H^{-1} 1 / (1^T H^{-1} 1)

The value at the optimum (ignoring non-negativity) is 1/(1^T H^{-1} 1).

This is very computable! If we can efficiently compute H^{-1} 1 for Hankel-structured H, we can solve the inner QP quickly.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated smoothed fictitious play (O(n log n) per iteration)
   - Solves the max-min game: max_p min_a a^T H(p) a where H(p)[i,j] = p[i+j]
   - Uses softmax smoothing with adaptive temperature schedule
2. Nesterov-accelerated FISTA on the smooth max objective (FFT gradients)
3. LP-based best response with adaptive active sets
4. Efficient coordinate descent with exact 1D optimization
5. Multi-length, multi-initialization search with diverse starting points
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_corr(a, b):
    """Compute cross-correlation: result[i] = sum_j a[j] * b[i+j] for i=0..len(a)-1.
    Uses FFT. b should have length >= len(a)+len(b)-1... actually b just needs length >= 2*len(a)-1.
    """
    n = len(a)
    m = len(b)
    b_rev = b[::-1].copy()
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b_rev, size)
    conv = np.fft.irfft(fa * fb, size)[:n + m - 1]
    # result[i] = conv[m-1-i] for i=0..n-1 (derived from index analysis)
    # Actually: conv[k] = sum_j a[j] * b_rev[k-j] = sum_j a[j] * b[m-1-k+j]
    # We want result[i] = sum_j a[j] * b[i+j], so m-1-k+j = i+j => k = m-1-i
    indices = (m - 1) - np.arange(n)
    return conv[indices]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_fictitious_play(a_init, tau_start, tau_end, max_iters, deadline, lr_decay=0.999, n_samples=1):
    """
    Smoothed fictitious play / mirror descent game:
    - Player p (over lags k) tries to maximize a^T H(p) a
    - Player a (over positions) tries to minimize a^T H(p) a
    - H(p)[i,j] = p[i+j]
    - Updates are mirror descent steps with exponential tilting
    
    The key: H(p) @ a = cross_correlation(a, p) which is O(n log n) via FFT.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    b_len = 2 * n - 1
    
    # Initialize p as uniform
    p = np.ones(b_len) / b_len
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    step = 1.0  # step size for a-update
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
            
        # Adaptive temperature
        frac = min(1.0, it / max(1, max_iters * 0.8))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute H(p) @ a = cross_correlation(a, p)
        # This gives: (Ha)[i] = sum_j p[i+j] * a[j]
        Ha = fast_cross_corr(a, p)
        
        # Update a: mirror descent step
        # Natural gradient direction: Ha (the "expected" contribution of each a[i])
        # Step: a_new[i] = a[i] * exp(-lr * (Ha[i] - mean(Ha)))
        Ha_tan = Ha - Ha.mean()
        log_a = np.log(np.maximum(a, 1e-300))
        log_a = log_a - log_a.max()  # numerical stability
        log_a_new = log_a - step * Ha_tan
        log_a_new = log_a_new - log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        # Nesterov momentum (simple version)
        if it > 0 and hasattr(fast_fictitious_play, '_a_prev'):
            pass  # keep simple, no momentum for now
        
        a = a_new
        
        # Compute b = conv(a, a) and update p
        b = fast_conv(a)
        M_cur = b.max()
        
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()
        
        # Update p: softmax of b with temperature tau
        b_shifted = b - b.max()
        p = np.exp(b_shifted / tau)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(b_len) / b_len
        else:
            p = p / p_sum
        
        # Decay step size
        step *= lr_decay
        step = max(step, 0.01)
    
    # Final check
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a, best_M


def fista_fft(a_init, tau, max_iters, deadline, lr=0.5):
    """FISTA with FFT-based smooth max gradient."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    b_len = 2 * n - 1
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute softmax distribution p over convolution values
        b_y = fast_conv(y)
        b_max = b_y.max()
        exp_b = np.exp(tau * (b_y - b_max))
        p = exp_b / exp_b.sum()
        
        # Compute gradient via FFT: grad[i] = 2 * sum_j p[i+j] * a[j]
        # But we need it w.r.t. y, so: grad[i] = 2 * sum_j p[i+j] * y[j]
        grad = 2.0 * fast_cross_corr(y, p)
        
        # Project gradient to tangent of simplex
        grad_tan = grad - grad.mean()
        
        # Adaptive step size
        step = lr / (1.0 + 0.01 * tau * it / max(1, max_iters))
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        # Track best
        if it % 10 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def lp_best_response(a, n_active=30, time_limit=10):
    """Solve LP best response: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    active = top_indices.tolist()
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_iterations(a_init, max_iters, deadline, n_active_init=30):
    """LP-based mirror descent iterations."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    current_na = n_active_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = lp_best_response(a, current_na, time_limit=8)
        if g is None:
            current_na = min(current_na * 2, 2 * n - 1)
            no_improve += 1
            if no_improve > 8:
                break
            continue
        
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
            if current_na > 15:
                current_na = max(15, int(current_na * 0.85))
        else:
            no_improve += 1
            current_na = min(int(current_na * 1.5), 2 * n - 1)
            if no_improve > 12:
                break
    
    return best_a


def coordinate_descent_1d_fast(a, i, b, n):
    """Find optimal a[i] given all others. Returns optimal value."""
    x_cur = a[i]
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Collect affected convolution values
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    # Unaffected max
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_lo:k_hi + 1] = False
    base_max = b[mask].max() if mask.any() else 0.0
    
    c_arr = np.array(c_vals) if c_vals else np.array([0.0])
    s_arr = np.array(slopes) if slopes else np.array([0.0])
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, np.sqrt(max(b.max(), 0.0)) * 1.5, 0.5)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with random ordering."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        
        b = fast_conv(a)
        sweep_improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            x_opt = coordinate_descent_1d_fast(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                old_val = a[i]
                a[i] = x_opt
                # Update b
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = c_arr_val = b[k] - old_val * old_val + x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old_val)
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 2
        if no_improve > 15:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-4)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Concave (bell-shaped)
    for width in [2, 4, 8]:
        bell = np.exp(-width * (t - 0.5) ** 2) + 0.05
        inits.append(bell / bell.sum())
    
    # Convex (U-shaped)
    for width in [2, 4]:
        u_shape = ((2 * t - 1) ** 2) * width + 0.1
        inits.append(u_shape / u_shape.sum())
    
    # Cosine windows
    for freq in [1, 2, 3, 5]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    # Power law
    for alpha in [0.5, 1.0, 2.0]:
        pl = np.maximum(t ** alpha + (1-t) ** alpha, 0.01)
        inits.append(pl / pl.sum())
    
    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # "Anti-peak" init: minimize center mass
    anti = np.ones(n)
    center = n // 2
    anti[center - n//8:center + n//8] = 0.3
    inits.append(anti / anti.sum())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def optimize_sequence(a0, deadline, seed=0):
    """Full optimization pipeline for a single sequence."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 1e-8)
    a0 = a0 / a0.sum()
    
    best = a0.copy()
    best_M = fast_conv(a0).max()
    t_start = time.monotonic()
    duration = max(1.0, deadline - t_start)
    
    # Stage 1: Fictitious play with decreasing tau (50% of time)
    t1 = t_start + 0.4 * duration
    if time.monotonic() < t1:
        # Multi-tau schedule
        tau_schedule = [(50, 5), (20, 2), (10, 1), (5, 0.5)]
        time_per_tau = (t1 - time.monotonic()) / len(tau_schedule)
        for tau_s, tau_e in tau_schedule:
            if time.monotonic() > t1:
                break
            sub_dl = min(t1, time.monotonic() + time_per_tau)
            max_it = max(100, int((sub_dl - time.monotonic()) * 200))  # estimate ~5ms per iter for n=768
            a_opt, M_opt = smooth_fictitious_play(best, tau_s, tau_e, max_it, sub_dl)
            if M_opt < best_M:
                best_M = M_opt
                best = a_opt.copy()
    
    print(f"    After fictitious play: M={best_M:.10f}", flush=True)
    
    # Stage 2: FISTA refinement (25% of time)
    t2 = t_start + 0.65 * duration
    if time.monotonic() < t2:
        for tau in [20, 50, 100]:
            if time.monotonic() > t2:
                break
            sub_dl = min(t2, time.monotonic() + (t2 - time.monotonic()) / 3)
            a_fista = fista_fft(best, tau, 5000, sub_dl, lr=0.4)
            M_f = fast_conv(a_fista).max()
            if M_f < best_M:
                best_M = M_f
                best = a_fista.copy()
    
    print(f"    After FISTA: M={best_M:.10f}", flush=True)
    
    # Stage 3: LP best response (15% of time)
    t3 = t_start + 0.85 * duration
    if time.monotonic() < t3:
        a_lp = run_lp_iterations(best, 50, t3, n_active_init=25)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
    
    print(f"    After LP: M={best_M:.10f}", flush=True)
    
    # Stage 4: Coordinate descent (10% of time)
    if time.monotonic() < deadline:
        max_sw = max(5, int((deadline - time.monotonic()) / max(0.05, n / 200.0)))
        a_cd = run_coordinate_descent(best, max_sw, deadline, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
    
    print(f"    After CD: M={best_M:.10f}", flush=True)
    
    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining FFT-accelerated fictitious play, FISTA,
    LP best response, and coordinate descent across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)
    
    n_parent = parent.size
    
    # Determine lengths to try
    lengths_to_try = [n_parent]
    for delta in [-128, -64, 64, 128, 256, -256, 384, -384]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl not in lengths_to_try:
            lengths_to_try.append(nl)
    # Also try some fixed good lengths
    for fixed_n in [512, 640, 768, 1024, 1280]:
        if 256 <= fixed_n <= 2048 and fixed_n not in lengths_to_try:
            lengths_to_try.append(fixed_n)
    
    # Sort: parent first, then by proximity
    lengths_to_try.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    
    # Limit to at most 6 lengths
    lengths_to_try = lengths_to_try[:6]
    
    print(f"\n{'='*60}", flush=True)
    print(f"Search lengths: {lengths_to_try}", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Phase 1: Multi-length, multi-init search
    # Reserve time for later polishing phases
    reserve = 400  # seconds for polishing
    phase1_end = deadline - reserve
    n_lengths = len(lengths_to_try)
    time_per_length = max(30, (phase1_end - time.monotonic()) / n_lengths)
    
    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > phase1_end:
            break
        
        t0 = time.monotonic()
        remaining = phase1_end - time.monotonic()
        if remaining < 20:
            break
        
        time_alloc = min(time_per_length, remaining / max(1, n_lengths - li))
        print(f"\n  === Length n={n}, time_alloc={time_alloc:.0f}s ===", flush=True)
        
        # Generate inits (limit to avoid too many)
        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        # Limit number of inits based on time
        max_inits = max(3, min(len(inits), int(time_alloc / 20)))
        inits = inits[:max_inits]
        
        best_for_n = None
        best_M_for_n = float('inf')
        
        for ii, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            sub_deadline = min(t0 + time_alloc, phase1_end)
            if time.monotonic() >= sub_deadline:
                break
            
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            
            a_opt = optimize_sequence(a0, sub_deadline, seed=seed + li * 1000 + ii)
            M_opt = compute_J(a_opt)
            
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best J={J_val:.6f} (elapsed={time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_for_n)
    
    tracker.report("After Phase 1")
    
    # Phase 2: Intensive polishing on the best
    if tracker.best is not None and time.monotonic() < deadline - 60:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 2: Intensive polish", flush=True)
        print(f"{'='*60}", flush=True)
        
        n_best = tracker.best.size
        
        # 2a: Long fictitious play with very low tau
        t_lp = min(deadline - 30, time.monotonic() + 120)
        a_fp, M_fp = smooth_fictitious_play(tracker.best, 5, 0.1, 10000, t_lp, lr_decay=0.9995)
        tracker.update(a_fp)
        
        # 2b: LP with growing active set
        if time.monotonic() < deadline - 30:
            t_lp2 = min(deadline - 20, time.monotonic() + 100)
            a_l = run_lp_iterations(tracker.best, 80, t_lp2, n_active_init=20)
            tracker.update(a_l)
        
        # 2c: FISTA with high tau
        if time.monotonic() < deadline - 20:
            t_f = min(deadline - 15, time.monotonic() + 60)
            for tau in [100, 200, 500]:
                if time.monotonic() > t_f:
                    break
                a_f = fista_fft(tracker.best, tau, 3000,
                               min(t_f, time.monotonic() + 20), lr=0.2)
                tracker.update(a_f)
        
        # 2d: Coordinate descent
        if time.monotonic() < deadline - 10:
            max_sw = max(20, int((deadline - 10 - time.monotonic()) / max(0.1, n_best / 100.0)))
            a_cd = run_coordinate_descent(tracker.best, max_sw, deadline - 5, seed=99999)
            tracker.update(a_cd)
    
    tracker.report("After Phase 2")
    
    # Phase 3: Try length variations from best
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 3: Length exploration from best", flush=True)
        print(f"{'='*60}", flush=True)
        
        n_cur = tracker.best.size
        for mult in [1.125, 1.25, 0.875, 0.75, 1.5, 0.667]:
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 2048)
            new_n = (new_n // 16) * 16
            if new_n == n_cur or new_n < 128 or time.monotonic() > deadline - 15:
                continue
            
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            
            sub_dl = min(deadline - 5, time.monotonic() + 40)
            a_opt = optimize_sequence(a_ext, sub_dl, seed=seed + 5555)
            tracker.update(a_opt)
    
    tracker.report("After Phase 3")
    
    # Final safety check
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
