"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated FISTA on top-k log-sum-exp smooth approximation
2. LP mirror descent with iterative active set refinement

4. Multi-scale, multi-start search with adaptive time allocation
Key innovations: O(n log n) gradient via FFT, top-k max for conditioning,
iterative LP active set, and aggressive multi-phase polish.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if cond.any():
        rho = int(np.max(np.where(cond, np.arange(n), 0)))
    else:
        rho = n - 1
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_topk_and_grad(a, tau, k):
    """Compute top-k log-sum-exp smooth max of conv(a,a) and its gradient.
    
    Uses FFT for O(n log n) gradient computation.
    Returns (smooth_value, gradient_of_length_n).
    """
    n = len(a)
    b = fast_conv(a)
    conv_len = 2 * n - 1
    
    if k < conv_len:
        top_idx = np.argpartition(b, -k)[-k:]
    else:
        top_idx = np.arange(conv_len)
    
    b_top = b[top_idx]
    b_max = b_top.max()
    
    # For numerical stability
    shifted = tau * (b_top - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    exp_sum = exp_b.sum()
    if exp_sum < 1e-300:
        p_top = np.ones(len(b_top)) / len(b_top)
    else:
        p_top = exp_b / exp_sum
    
    smooth_val = b_max + np.log(exp_sum) / tau
    
    # Build full p array
    p = np.zeros(conv_len)
    p[top_idx] = p_top
    
    # Gradient: g[i] = 2 * sum_j p[j] * a[j-i]
    # This is 2 * (p ⊛ a)[i] = 2 * ifft(conj(FFT(p)) * FFT(a))
    size = 1
    while size < conv_len + n - 1:
        size <<= 1
    
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    g = 2.0 * np.fft.irfft(np.conj(P) * A, size)[:n]
    
    return smooth_val, g


def run_fista_topk(a_init, tau_start, tau_end, k_start, k_end, max_iters, deadline, lr=None):
    """FISTA with top-k log-sum-exp smooth approximation and FFT gradient."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if lr is None:
        lr = 0.5
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters - 1))
        tau = tau_start * (tau_end / tau_start) ** frac
        k_cur = int(k_start + (k_end - k_start) * frac)
        k_cur = max(1, k_cur)
        
        _, grad_y = smooth_topk_and_grad(y, tau, k_cur)
        
        # Step size: Lipschitz constant of the smooth part
        # The smooth max has Lipschitz constant ~ tau * 2 * max(a) * 2 (rough)
        # Use a safe step size
        step = lr / (1.0 + tau * max(a.max() * 2.0, 0.01))
        step = min(step, 2.0 * n)  # cap for safety
        
        # Projected gradient step
        grad_tan = grad_y - grad_y.mean()
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv_cross(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if cond.any():
        rho = int(np.max(np.where(cond, np.arange(n), 0)))
    else:
        rho = n - 1
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_topk_and_grad(a, tau, k):
    """Compute top-k log-sum-exp smooth max of conv(a,a) and its gradient via FFT.
    
    Returns (smooth_value, gradient_of_length_n).
    """
    n = len(a)
    b = fast_conv(a)
    conv_len = 2 * n - 1

    if k < conv_len and k > 0:
        top_idx = np.argpartition(b, -k)[-k:]
    else:
        top_idx = np.arange(conv_len)
        k = conv_len

    b_top = b[top_idx]
    b_max = b_top.max()

    shifted = tau * (b_top - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    exp_sum = exp_b.sum()
    if exp_sum < 1e-300:
        p_top = np.ones(len(b_top)) / len(b_top)
    else:
        p_top = exp_b / exp_sum

    smooth_val = b_max + np.log(exp_sum) / tau

    # Build full p array
    p = np.zeros(conv_len)
    p[top_idx] = p_top

    # Gradient: g[i] = 2 * sum_j p[j] * a[j-i] = 2 * (p cross-correlate a)[i]
    # Using FFT: cross_corr = ifft(conj(FFT(p)) * FFT(a))
    size = 1
    while size < conv_len + n - 1:
        size <<= 1

    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    g = 2.0 * np.fft.irfft(np.conj(P) * A, size)[:n]

    return smooth_val, g


def run_fista_topk(a_init, tau_start, tau_end, k_start, k_end, max_iters, deadline, lr=0.5):
    """FISTA with top-k log-sum-exp smooth approximation and FFT gradient."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters - 1))
        tau = tau_start * (tau_end / tau_start) ** frac
        k_cur = int(k_start + (k_end - k_start) * frac)
        k_cur = max(1, min(k_cur, 2 * n - 1))

        _, grad_y = smooth_topk_and_grad(y, tau, k_cur)

        # Adaptive step size based on gradient norm
        grad_norm = np.linalg.norm(grad_y)
        if grad_norm < 1e-12:
            break
        step = lr / (tau * max(grad_norm, 0.1))
        step = min(step, 5.0 * n)
        step = max(step, 1e-6)

        # Projected gradient step on simplex
        grad_tan = grad_y - grad_y.mean()
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_mom + 1.0

        # Periodic exact evaluation
        if it % 20 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()

    if M_a < best_M:

        best_a = a.copy()

def run_lp_mirror_descent(a_init, max_iters, deadline, n_active=25, max_expansions=3):

def coordinate_optimize_1d(a, i, b, n):
    """Find optimal value for a[i] given all others via golden section search."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        if x_test < 0:
            x_test = 0.0
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, np.sqrt(max(b.max(), 0.0)) + 0.5)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Cyclic coordinate descent with golden section 1D optimization."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)

        b = fast_conv(a)
        sweep_improved = False

        for i in order:
            x_opt = coordinate_optimize_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                old_val = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt - old_val) * (x_opt + old_val)
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old_val)
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        if sweep % 3 == 0 or sweep == max_sweeps - 1:
            M = fast_conv(a).max()
            if M < best_M * (1 - 1e-13):
                best_M = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if not sweep_improved:
            no_improve += 3
        if no_improve > 25:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for noise_level in [0.02, 0.05, 0.1, 0.2]:
            p_n = np.maximum(resample_seq(parent, n) + rng.normal(0, noise_level, n), 1e-4)
            inits.append(p_n / p_n.sum())

    inits.append(np.ones(n) / n)

    # Various structured inits
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        poly = (t ** alpha) + 0.01
        inits.append(poly / poly.sum())

    for freq in [1, 2, 3, 5]:
        cos_w = 0.5 + 0.4 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())

    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    # Spectral init: iterate to flatten autocorrelation
    a_spec = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_spec)
        M_s = bs.max()
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            corr[i] = a_spec[k_star - i]
        a_spec = a_spec - 0.15 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())

    # Random smooth
    ksize = max(3, n // 15)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_combined_optimization(a0, deadline_phase, seed=0):
    """Run multi-stage optimization: FISTA -> Mirror -> LP -> CD."""
    best = np.maximum(a0.astype(np.float64), 0.0)
    s = best.sum()
    if s < 1e-10:
        best = np.ones_like(best) / len(best)
    else:
        best = best / s
    n = len(best)
    best_M = fast_conv(best).max()
    t_start = time.monotonic()
    phase_duration = max(1, deadline_phase - t_start)

    # Stage 1: Multi-tau FISTA (45% of time)
    t1 = t_start + 0.45 * phase_duration
    if time.monotonic() < t1:
        # Coarse-to-fine: large k, small tau -> small k, large tau
        configs = [
            (50, 200, max(20, n // 4), min(5, 2 * n - 1)),
            (100, 500, max(10, n // 8), 3),
            (200, 1000, 5, 2),
        ]
        sub_time_each = (t1 - time.monotonic()) / max(1, len(configs))
        for tau_s, tau_e, k_s, k_e in configs:
            if time.monotonic() > t1:
                break
            sub_dl = min(t1, time.monotonic() + max(5, sub_time_each))
            a_f = run_fista_topk(best, tau_s, tau_e, k_s, max(1, k_e),
                                  50000, sub_dl, lr=0.3)
            M_f = fast_conv(a_f).max()
            if M_f < best_M:
                best_M = M_f
                best = a_f.copy()

    # Stage 2: Mirror descent (15% of time)
    t2 = t_start + 0.60 * phase_duration
    if time.monotonic() < t2:
        a_md = run_mirror_descent(best, 100000, t2, step_init=0.2)
        M_md = fast_conv(a_md).max()
        if M_md < best_M:
            best_M = M_md
            best = a_md.copy()

    # Stage 3: LP mirror descent (25% of time)
    t3 = t_start + 0.85 * phase_duration
    if time.monotonic() < t3:
        a_lp = run_lp_mirror_descent(best, 200, t3, n_active=30, max_expansions=3)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()

    # Stage 4: Coordinate descent (15% of time)
    if time.monotonic() < deadline_phase:
        max_sw = max(5, int((deadline_phase - time.monotonic()) / max(0.05, n / 200.0)))
        a_cd = run_coordinate_descent(best, max_sw, deadline_phase, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()

    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid search: FFT-FISTA, LP mirror descent, coordinate descent.
    Focuses on parent length with exploration of nearby lengths.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    deadline = t_global_start + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size
    normalized_parent = parent / parent.sum()

    # === PHASE 1: Multi-start FISTA + Mirror at parent length ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-start optimization at n={n_parent}", flush=True)
    print(f"{'='*60}", flush=True)

    t_p1_end = t_global_start + 0.45 * (budget_s - 20)
    inits = generate_inits(n_parent, rng, normalized_parent)

    # Limit number of inits based on time
    time_per_init = (t_p1_end - time.monotonic()) / max(1, len(inits))
    max_inits = max(3, min(len(inits), int((t_p1_end - time.monotonic()) / 30)))
    inits = inits[:max_inits]

    for ii, a0 in enumerate(inits):
        if time.monotonic() > t_p1_end:
            break
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()
        sub_dl = min(t_p1_end, time.monotonic() + max(15, time_per_init))
        print(f"  Init {ii+1}/{max_inits}, sub_dl={max(0, sub_dl - time.monotonic()):.0f}s", flush=True)
        a_opt = run_combined_optimization(a0, sub_dl, seed=seed + ii)
        tracker.update(a_opt)

    tracker.report("After Phase 1")

    # === PHASE 2: Intensive LP polish on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive LP polish", flush=True)
    print(f"{'='*60}", flush=True)

    t_p2_end = t_global_start + 0.65 * (budget_s - 20)
    if tracker.best is not None and time.monotonic() < t_p2_end:
        # Long LP with large active set
        a_lp1 = run_lp_mirror_descent(tracker.best, 300, t_p2_end, n_active=50, max_expansions=4)
        tracker.update(a_lp1)

        # Second pass with smaller active set for fine-tuning
        if time.monotonic() < t_p2_end:
            a_lp2 = run_lp_mirror_descent(a_lp1, 100,
                                          min(t_p2_end, time.monotonic() + 60),
                                          n_active=15, max_expansions=2)
            tracker.update(a_lp2)

    tracker.report("After Phase 2")

    # === PHASE 3: High-tau FISTA refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: High-tau FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)

    t_p3_end = t_global_start + 0.78 * (budget_s - 20)
    if tracker.best is not None and time.monotonic() < t_p3_end:
        # Very high tau for near-exact max approximation
        a_f1 = run_fista_topk(tracker.best, 500, 2000, 3, 1,
                               100000, min(t_p3_end, time.monotonic() + 60), lr=0.2)
        tracker.update(a_f1)

        # Mirror descent from refined point
        if time.monotonic() < t_p3_end:
            a_md = run_mirror_descent(a_f1, 50000, t_p3_end, step_init=0.1)
            tracker.update(a_md)

    tracker.report("After Phase 3")

    # === PHASE 4: Coordinate descent + final LP ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent + LP finish", flush=True)
    print(f"{'='*60}", flush=True)

    t_p4_end = t_global_start + 0.90 * (budget_s - 20)
    if tracker.best is not None and time.monotonic() < t_p4_end:
        n_best = tracker.best.size
        max_sw = max(10, int((t_p4_end - time.monotonic()) / max(0.02, n_best / 500.0)))
        a_cd = run_coordinate_descent(tracker.best, max_sw, t_p4_end, seed=99999)
        tracker.update(a_cd)

        # Final LP polish
        if time.monotonic() < t_p4_end:
            a_final_lp = run_lp_mirror_descent(a_cd, 50, t_p4_end, n_active=25, max_expansions=3)
            tracker.update(a_final_lp)
            # One more CD pass
            if time.monotonic() < t_p4_end:
                a_cd2 = run_coordinate_descent(a_final_lp, 20, t_p4_end, seed=88888)
                tracker.update(a_cd2)

    tracker.report("After Phase 4")

    # === PHASE 5: Length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    t_p5_end = deadline - 10
    if tracker.best is not None and time.monotonic() < t_p5_end:
        n_cur = tracker.best.size
        candidates = [int(n_cur * 1.2), int(n_cur * 0.85), int(n_cur * 1.5), int(n_cur * 0.7)]
        candidates = [c for c in candidates if 256 <= c <= 2048 and c != n_cur]
        # Deduplicate
        seen = set()
        unique_candidates = []
        for c in candidates:
            c_rounded = (c // 32) * 32
            if c_rounded != n_cur and c_rounded not in seen and 128 <= c_rounded <= 2048:
                seen.add(c_rounded)
                unique_candidates.append(c_rounded)

        time_per = max(10, (t_p5_end - time.monotonic()) / max(1, len(unique_candidates)))
        for nc in unique_candidates[:3]:
            if time.monotonic() > t_p5_end:
                break
            a_ext = resample_seq(tracker.best, nc)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(t_p5_end, time.monotonic() + time_per)
            a_opt = run_combined_optimization(a_ext, sub_dl, seed=seed + 777)
            tracker.update(a_opt)

    tracker.report("After Phase 5")

    # === Final safety: ensure valid output ===
    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)

    # Final verification
    final_score = evaluate_sequence([float(x) for x in tracker.best])
    if np.isfinite(final_score):
        tracker.best_score = min(tracker.best_score, final_score)

    elapsed = time.monotonic() - t_global_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
