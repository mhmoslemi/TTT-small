"""
Multi-scale optimization with targeted peak shaving, LP refinement,
and diverse asymmetric initializations for minimizing the peak autocorrelation
ratio 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """
    Compute smoothed objective (logsumexp of beta*conv) and its gradient w.r.t. a.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)

    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def adam_log_opt(a, max_iters=5000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995, seed=0):
    """
    Adam optimization in log-space with temperature annealing.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start

    best_a = a.copy()
    best_true_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        # Cosine annealing for beta
        beta = beta_end + 0.5 * (beta_start - beta_end) * (1 + np.cos(np.pi * progress))

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)

        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a

        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

            true_obj = compute_obj(a_check)

                best_a = a_check.copy()

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, time_budget=30.0):
    """LP-based refinement."""
    n = len(a)
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s

    conv = compute_conv(a_norm)
    M = float(np.max(conv))

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Try multiple M values
    for m_factor in [1.0, 1.01, 1.02, 1.05]:
        if time.monotonic() - start > time_budget:
            break
        M_try = M * m_factor
        g = solve_lp_max_sum(a_norm, M_try, time_limit=min((time_budget - (time.monotonic() - start)) * 0.3, 10))
        if g is None:
            continue

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            if time.monotonic() - start > time_budget:
                break
            a_try = (1 - t) * a_norm + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a

            break

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def gen_linear_ramp(n, c=0.5, seed=0):
    """Generate an asymmetric linear ramp."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = 1 + c * (2 * x - 1)
    a = np.maximum(a, 1e-6)
    # Add small random perturbation
    a *= np.exp(rng.normal(0, 0.1, n))
    return a / np.sum(a)


def gen_quadratic(n, seed=0):
    """Generate a quadratic asymmetric sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = (x - 0.3) ** 2 + 0.1
    a = np.maximum(a, 1e-6)
    a *= np.exp(rng.normal(0, 0.1, n))
    return a / np.sum(a)


def gen_sine_mod(n, freq=1.0, seed=0):
    """Generate a sine-modulated asymmetric sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = 1 + 0.3 * np.sin(2 * np.pi * freq * x) * x  # asymmetric modulation
    a = np.maximum(a, 1e-6)
    a *= np.exp(rng.normal(0, 0.1, n))
    return a / np.sum(a)


def gen_exponential(n, rate=1.0, seed=0):
    """Generate an exponentially decaying sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.exp(-rate * x)
    a = np.maximum(a, 1e-6)
    a *= np.exp(rng.normal(0, 0.05, n))
    return a / np.sum(a)


def multi_scale_optimize(a_init, n_target, time_budget, seed=0):
    """
    Multi-scale optimization: start at low resolution, gradually increase.
    """
    # Start at n/4
    scales = [max(50, n_target // 4), max(100, n_target // 2), n_target]
    scales = sorted(set([s for s in scales if s <= n_target]))

    a = a_init.copy()
    s = np.sum(a)
    if s > 0:
        a = a / s

    for scale_n in scales:
        if time_budget <= 5:
            break

        # Resample to current scale
        if len(a) != scale_n:
            a = resample_sequence(a, scale_n)
            s = np.sum(a)
            if s > 0:
                a = a / s

        # Optimize at this scale
        alloc = time_budget * 0.6
        time_budget -= alloc

        a = adam_log_opt(a, max_iters=10000, time_budget=alloc,
                         beta_start=5.0, beta_end=100.0,
                         lr_start=0.05, lr_decay=0.99998, seed=seed)

        seed += 1

    return a

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key insights: symmetric sequences have obj >= 2, so strong asymmetry is required.
Combines: diverse asymmetric initializations, multi-scale Adam in log-space,
LP refinement, targeted peak shaving, block-swap perturbations, and
simulated annealing restarts with plateau detection.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """Smoothed objective (logsumexp of beta*conv) and gradient w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)

    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def adam_log_opt(a, max_iters=5000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995, seed=0):
    """Adam optimization in log-space with cosine beta annealing and noise injection."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_true_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    stall_count = 0
    last_improve = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        # Cosine annealing for beta
        beta = beta_end + 0.5 * (beta_start - beta_end) * (1 + np.cos(np.pi * progress))

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)

        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a

        # Periodic noise injection to escape saddle points
        if it % 300 == 0 and it > 100:
            noise_scale = 0.02 * (1.0 - progress) * (1.0 + stall_count * 0.5)
            grad_x += noise_scale * rng.standard_normal(n)

        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 200 == 0:
            x -= np.mean(x)

        # Plateau detection
        if it % 150 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj - 1e-10:
                best_true_obj = true_obj
                best_a = a_check.copy()
                stall_count = 0
                last_improve = it
            else:
                stall_count += 1
                # If stalled for a while, inject a larger perturbation
                if stall_count >= 4 and it < max_iters - 200:
                    x += 0.05 * rng.standard_normal(n)
                    x -= np.mean(x)
                    stall_count = 0

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, time_budget=30.0):
    """LP-based refinement with multiple M values."""
    n = len(a)
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s

    conv = compute_conv(a_norm)
    M = float(np.max(conv))

    best_a = a.copy()
    best_obj = compute_obj(a)

    for m_factor in [1.0, 1.01, 1.03, 1.06]:
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        M_try = M * m_factor
        g = solve_lp_max_sum(a_norm, M_try, time_limit=min(remaining * 0.35, 10))
        if g is None:
            continue

        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            if time.monotonic() - start > time_budget:
                break
            a_try = (1 - t) * a_norm + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a


def peak_shaving(a, time_budget=30.0, seed=0):
    """Targeted peak shaving: reduce top convolution peaks by moving mass."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        sorted_idx = np.argsort(conv)[::-1]
        top_peaks = sorted_idx[:min(8, len(conv))]

        improved = False
        for p in top_peaks:
            if time.monotonic() - start > time_budget:
                break
            peak_val = conv[p]
            if peak_val < 0.1 * conv[max(sorted_idx)]:
                break

            # Find contributing pairs
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)

            # Get contributors sorted by contribution
            contribs = []
            for i in range(i_min, i_max + 1):
                j = p - i
                if 0 <= j < n:
                    contribs.append((best_a[i] * best_a[j], i, j))
            contribs.sort(reverse=True)

            # Try reducing top contributors
            for _, i, j in contribs[:5]:
                if time.monotonic() - start > time_budget:
                    break
                # Try moving mass from i to random positions
                for trial in range(8):
                    target = rng.integers(0, n)
                    if target == i or target == j:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.03]:
                        delta = frac * best_a[i]
                        if delta < 1e-12:
                            continue
                        a_try = best_a.copy()
                        a_try[i] -= delta
                        a_try[target] += delta
                        if a_try[i] < 0:
                            continue
                        s_try = np.sum(a_try)
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            improved = True
                            break

        if not improved and it > 50:
            # Try a random coordinate swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = 0.01 * abs(best_a[i] - best_a[j])
                if delta > 0:
                    a_try = best_a.copy()
                    if best_a[i] > best_a[j]:
                        a_try[i] -= delta
                        a_try[j] += delta
                    else:
                        a_try[i] += delta
                        a_try[j] -= delta
                    if a_try[i] >= 0 and a_try[j] >= 0:
                        a_try = a_try / np.sum(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            improved = True

        if not improved and it > 100:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def gen_linear_ramp(n, c=0.5, seed=0):
    """Asymmetric linear ramp initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = 1 + c * (2 * x - 1)
    a = np.maximum(a, 1e-6)
    a *= np.exp(rng.normal(0, 0.08, n))
    return a / np.sum(a)


def gen_quadratic_asym(n, seed=0):
    """Asymmetric quadratic initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    center = rng.uniform(0.2, 0.4)
    a = (x - center) ** 2 + 0.05
    a = np.maximum(a, 1e-6)
    a *= np.exp(rng.normal(0, 0.08, n))
    return a / np.sum(a)


def gen_sine_asym(n, freq=1.0, seed=0):
    """Sine-modulated asymmetric initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = 1 + 0.4 * np.sin(2 * np.pi * freq * x) * (0.5 + x)
    a = np.maximum(a, 1e-6)
    a *= np.exp(rng.normal(0, 0.05, n))
    return a / np.sum(a)


def gen_exponential(n, rate=2.0, seed=0):
    """Exponentially decaying initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.exp(-rate * x)
    a = np.maximum(a, 1e-6)
    a *= np.exp(rng.normal(0, 0.05, n))
    return a / np.sum(a)


def gen_bimodal(n, seed=0):
    """Two-peaked asymmetric initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    c1 = rng.uniform(0.2, 0.35)
    c2 = rng.uniform(0.65, 0.8)
    s1 = rng.uniform(0.05, 0.1)
    s2 = rng.uniform(0.05, 0.1)
    a = np.exp(-(x - c1) ** 2 / (2 * s1 ** 2)) + 0.7 * np.exp(-(x - c2) ** 2 / (2 * s2 ** 2)) + 0.01
    a = np.maximum(a, 1e-6)
    a *= np.exp(rng.normal(0, 0.03, n))
    return a / np.sum(a)


def multi_scale_optimize(a_init, n_target, time_budget, seed=0):
    """Multi-scale: optimize at n/4, n/2, then n_target."""
    scales = sorted(set([
        max(50, n_target // 4),
        max(100, n_target // 2),
        n_target
    ]))
    scales = [s for s in scales if s <= n_target]

    a = a_init.copy()
    s = np.sum(a)
    if s > 0:
        a = a / s

    for scale_n in scales:
        if time_budget <= 5:
            break
        if len(a) != scale_n:
            a = resample_sequence(a, scale_n)
            s = np.sum(a)
            if s > 0:
                a = a / s

        # Allocate time proportional to scale
        alloc = time_budget * (scale_n / n_target) * 0.7
        time_budget -= alloc

        a = adam_log_opt(a, max_iters=10000, time_budget=max(alloc, 3.0),
                         beta_start=5.0, beta_end=150.0,
                         lr_start=0.05, lr_decay=0.99998, seed=seed)
        seed += 1

    return a

                rng.shuffle(block_starts)
                for i in range(len(block_starts) - 1):
                    b1 = block_starts[i]
                    b2 = block_starts[i + 1]
                    if b2 > b1 + block_size:
                        tmp = a_pert[b1:b1+block_size].copy()
                        a_pert[b1:b1+block_size] = a_pert[b2:b2+block_size]
                        a_pert[b2:b2+block_size] = tmp
            elif strategy == 3:
                # Blend with structured asymmetric
                alpha = rng.uniform(0.15, 0.4)
                x = np.linspace(0, 1, n_best)
                structured = 0.5 + rng.uniform(0.3, 0.8) * x
                structured = structured / np.sum(structured)
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * structured
            else:
                # Sparse spikes
                a_pert = best_seq.copy()
                num_spikes = rng.integers(3, 10)
                for _ in range(num_spikes):
                    idx = rng.integers(0, n_best)
                    a_pert[idx] += rng.uniform(0.05, 0.2) * np.mean(best_seq)

            a_pert = np.maximum(a_pert, 1e-10)
            s_p = np.sum(a_pert)
            if s_p < 1e-10:
                continue
            a_pert = a_pert / s_p

            # Simulated annealing: accept with probability based on "energy"
            obj_pert = compute_obj(a_pert)
            if obj_pert > best_score:
                # Metropolis criterion
                delta = obj_pert - best_score
                accept_prob = np.exp(-delta / (current_temp * 0.01))
                if rng.random() > accept_prob:
                    # Still try to optimize it, but with less time
                    alloc = min(remaining * 0.05, 5.0)
                else:
                    alloc = min(remaining * 0.08, 10.0)
            else:
                alloc = min(remaining * 0.08, 10.0)

            a_opt = adam_log_opt(a_pert, max_iters=8000, time_budget=alloc,
                                 beta_start=10.0, beta_end=500.0,
                                 lr_start=0.03, lr_decay=0.99998, seed=seed)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart} (strat {strategy}): {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart} (strat {strategy}): {obj:.8f}")

            # Cool down
            current_temp *= 0.85
            seed += 3

    # === Phase 7: n exploration on best structure ===
    print(f"\n{'='*60}")
    print(f"Phase 7: n-exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_tries = [n for n in [n_best - 30, n_best - 15, n_best + 15, n_best + 30]
                   if 300 <= n <= 1500 and n != n_best]

        for n_try in n_tries:
            if time.monotonic() > deadline - 25:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_try = resample_sequence(best_seq, n_try)
            s_t = np.sum(a_try)
            if s_t < 1e-10:
                continue
            a_try = a_try / s_t

            alloc = min(remaining * 0.15, 20.0)
            a_opt = adam_log_opt(a_try, max_iters=10000, time_budget=alloc,
                                 beta_start=20.0, beta_end=500.0,
                                 lr_start=0.03, lr_decay=0.99998, seed=seed)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")
            seed += 2

    # === Phase 8: Final high-precision polish ===
    print(f"\n{'='*60}")
    print(f"Phase 8: Final polish")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()

        # Very high beta Adam
        a_final = adam_log_opt(best_seq, max_iters=15000,
                               time_budget=min(remaining * 0.35, 20.0),
                               beta_start=1000.0, beta_end=10000.0,
                               lr_start=0.003, lr_decay=0.99999, seed=seed)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  High-beta final: {best_score:.8f} ***")
        else:
            print(f"  High-beta final: {obj_final:.8f}")

        # One more LP
        if time.monotonic() < deadline - 8 and len(best_seq) <= 800:
            remaining2 = deadline - time.monotonic()
            a_lp = lp_refine(best_seq, time_budget=min(remaining2 * 0.4, 12.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.8f} ***")

        # Final peak shave
        if time.monotonic() < deadline - 6:
            remaining3 = deadline - time.monotonic()
            a_ps = peak_shaving(best_seq, time_budget=min(remaining3 * 0.5, 10.0),
                                seed=seed + 100)
            obj_ps = compute_obj(a_ps)
            if obj_ps < best_score:
                best_score = obj_ps
                best_seq = a_ps.copy()
                print(f"  Final shave: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        s = np.sum(best_seq)
        best_seq = best_seq / s

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
