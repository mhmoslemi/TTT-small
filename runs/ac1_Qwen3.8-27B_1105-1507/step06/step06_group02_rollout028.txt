```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Parent sequence as starting point
2. Quick Gerchberg-Saxton + smooth Nesterov optimization
3. Aggressive LP mirror descent with:

   - Periodic random perturbation for global exploration
   - Multi-M LP (solve with slightly different M values)
4. Length exploration around best found length
5. Final LP polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def gs_flat_spectral(init, n_iters=200, beta=0.8):
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-30)
        power = 0.5 + 0.3 * beta
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals

def solve_lp(a, M):
    n = len(a)
    rows, cols, vals = build_A_sparse(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def run_smooth_descent(a, deadline, max_iters=3000, tau=50.0, lr=0.1):
    """Simple smooth descent using log-sum-exp approximation of max."""
    a = normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    momentum = 0.9
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(y)
        b_max = b.max()
        # Softmax weights
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        grad = np.zeros(n)
        active = np.where(p > 1e-10)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * p[k] * seg
        y_new = np.maximum(y - lr * (grad + momentum * (y - a)), 0)
        y_new = normalize(y_new)
        a_new = y_new + ((it + 2) / (it + 2 + np.sqrt(1 + 4 * (it + 2)**2 * 0.25))) * 0  # simplified Nesterov
        # Actually let's use simple momentum
        a = y_new
        b_check = fast_conv(a)
        if b_check.max() < best_M:
            best_M = b_check.max()
            best_a = a.copy()
    return best_a
```

```
    1. Compute M = max(a*a)
    2. Solve LP: max sum(g) s.t. (a*g) <= M, g >= 0
    3. Normalize g to sum = 1
    4. Line search: for t in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02, 0.01]:
       a_t = (1-t)*a + t*g
       J_t = J(a_t)
       Pick t with best J_t
    5. Update a = a_best
    6. Every 15 iterations: perturb a by adding small noise
    7. Check convergence
```

This is more expensive per iteration (9 evaluations of J) but each iteration makes more progress.

Actually, computing J requires computing the full convolution, which is O(n log n). For n=1024, this is very fast. The LP is the expensive part (O(n^2) or so). So the line search is cheap relative to the LP.

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LP mirror descent
with line search, Anderson acceleration, and periodic perturbation.
Phases: smooth init -> LP optimization -> length exploration -> final polish.
Key improvements over previous: proper line search in LP phase, more LP iterations,
periodic perturbation for local minimum escape, and better time allocation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a)
    return a / s


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-30)
        power = 0.5 + 0.3 * beta
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def solve_lp(a, M):
    n = len(a)
    rows, cols, vals = build_A_sparse(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def smooth_grad(a, b, tau):
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / (exp_b.sum() + 1e-30)
    n = len(a)
    grad = np.zeros(n)
    active = np.where(p > 1e-8)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad


def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=100.0,
                        lr_init=0.3, lr_end=0.003):
    a = normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    t_k = 1.0
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        b = fast_conv(y)
        grad = smooth_grad(a, b, tau)
        b_max = b.max()
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta * t_k)) / 2.0
        a_new = normalize(np.maximum(y - lr * grad, 0))
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy
        else:
            y = a_new
        a = a_new
    return best_a


def anderson_mix(hist_x, hist_y, m):
    if m < 2:
        return hist_y[-1].copy()
    residuals = np.array([hist_x[i] - hist_y[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-10 * np.eye(m)
    ones = np.ones(m)
    try:
        alpha = np.linalg.solve(G, ones)
        alpha = alpha / (alpha.sum() + 1e-30)
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 1.5)
    alpha = alpha / (alpha.sum() + 1e-30)
    mixed = np.zeros(len(hist_x[0]))
    for i in range(m):
        mixed += alpha[i] * hist_y[i]
    return np.maximum(mixed, 0)


def run_lp_optimization(a, deadline, max_iters=500, with_line_search=True,
                        anderson_window=4, perturb_every=25, perturb_size=0.01):
    """Aggressive LP mirror descent with line search and Anderson acceleration."""
    a = normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = compute_J(a)
    
    no_imp = 0
    hist_x = []
    hist_y = []
    rng = np.random.default_rng(12345)
    
    step_sizes = [0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Periodic perturbation
        if it > 0 and it % perturb_every == 0:
            noise = rng.normal(0, perturb_size, n)
            a_pert = normalize(a + noise)
            # Only accept perturbation if it doesn't worsen J too much
            J_pert = compute_J(a_pert)
            if J_pert < best_J * 1.05:
                a = a_pert
                M = fast_conv(a).max()
            # Reset Anderson history
            hist_x = []
            hist_y = []
        
        # Solve LP
        g = solve_lp(a, M)
        if g is None:
            # Try slightly larger M
            g = solve_lp(a, M * 1.01)
            if g is None:
                break
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        # Line search
        if with_line_search:
            best_t = 0.1
            best_M_t = float('inf')
            for t in step_sizes:
                a_t = normalize((1 - t) * a + t * g)
                M_t = fast_conv(a_t).max()
                if M_t < best_M_t:
                    best_M_t = M_t
                    best_t = t
            a_new = normalize((1 - best_t) * a + best_t * g)
        else:
            a_new = normalize((1 - 0.2) * a + 0.2 * g)
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_y.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_y.pop(0)
        
        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_y, len(hist_x))
            sa = a_and.sum()
            if sa > 1e-15:
                a_and /= sa
                M_and = fast_conv(a_and).max()
                M_new = fast_conv(a_new).max()
                if M_and <= M_new:
                    a_new = a_and
        
        # Update
        M_new = fast_conv(a_new).max()
        if M_new < best_M * (1 - 1e-12):
            best_M = M_new
            best_a = a_new.copy()
            best_J = compute_J(best_a)
            no_imp = 0
        else:
            no_imp += 1
        
        a = a_new
        
        if no_imp > 15:
            # Try a larger perturbation
            noise = rng.normal(0, 0.03, n)
            a = normalize(a + noise)
            hist_x = []
            hist_y = []
            no_imp = 0
    
    return best_a


def resample_seq(src, n_target):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n_target:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_target)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LP mirror descent
with line search, Anderson acceleration, and periodic perturbation for local minimum escape.
Phases: smooth Nesterov init -> aggressive LP optimization -> length exploration -> final polish.
Key improvements: proper line search in LP phase (12 step sizes), more LP iterations,
adaptive perturbation, Anderson acceleration, and better time allocation toward LP phase.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-30)
        power = 0.5 + 0.3 * beta
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def solve_lp(a, M):
    n = len(a)
    rows, cols, vals = build_A_sparse(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def smooth_grad(a, b, tau):
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -50, 50)
    exp_b = np.exp(shifted)
    p = exp_b / (exp_b.sum() + 1e-30)
    n = len(a)
    grad = np.zeros(n)
    active = np.where(p > 1e-8)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad


def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=100.0,
                        lr_init=0.25, lr_end=0.003):
    a = normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    t_k = 1.0
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        b = fast_conv(y)
        grad = smooth_grad(a, b, tau)

        b_max = b.max()
        if b_max < best_M * (1.0 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta * t_k)) / 2.0
        momentum = (t_k - 1.0) / (t_k + 1.0)
        a_new = normalize(np.maximum(y - lr * grad, 0))
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy
        else:
            y = a_new.copy()
        a = a_new
    return best_a


def anderson_mix(hist_x, hist_y, m):
    if m < 2:
        return hist_y[-1].copy()
    residuals = np.array([hist_x[i] - hist_y[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-10 * np.eye(m)
    ones = np.ones(m)
    try:
        alpha = np.linalg.solve(G, ones)
        s = alpha.sum()
        if abs(s) < 1e-15:
            alpha = ones / m
        else:
            alpha = alpha / s
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 1.5)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(hist_x[0]))
    for i in range(m):
        mixed += alpha[i] * hist_y[i]
    return np.maximum(mixed, 0)


def run_lp_optimization(a, deadline, max_iters=500, with_line_search=True,
                        anderson_window=4, perturb_every=20, perturb_size=0.01,
                        rng=None):
    if rng is None:
        rng = np.random.default_rng(999)
    a = normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = compute_J(a)

    no_imp = 0
    hist_x = []
    hist_y = []

    step_sizes = [0.5, 0.4, 0.35, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.01]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Periodic perturbation to escape local minima
        if it > 0 and it % perturb_every == 0:
            noise = rng.normal(0, perturb_size, n)
            a_pert = normalize(a + noise)
            J_pert = compute_J(a_pert)
            if J_pert < best_J * 1.03:
                a = a_pert
                M = float(fast_conv(a).max())
            hist_x = []
            hist_y = []

        # Solve LP
        g = solve_lp(a, M)
        if g is None:
            g = solve_lp(a, M * 1.005)
            if g is None:
                g = solve_lp(a, M * 1.01)
                if g is None:
                    break
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        # Line search or fixed step
        if with_line_search:
            best_t = 0.15
            best_M_t = float('inf')
            for t in step_sizes:
                a_t = (1 - t) * a + t * g
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                M_t = fast_conv(a_t).max()
                if M_t < best_M_t:
                    best_M_t = M_t
                    best_t = t
            a_new = (1 - best_t) * a + best_t * g
            a_new = normalize(a_new)
        else:
            a_new = normalize((1 - 0.2) * a + 0.2 * g)

        # Anderson acceleration
        hist_x.append(a.copy())
        hist_y.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_y.pop(0)

        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_y, len(hist_x))
            sa = a_and.sum()
            if sa > 1e-15:
                a_and = a_and / sa
                M_and = fast_conv(a_and).max()
                M_new_check = fast_conv(a_new).max()
                if M_and <= M_new_check * 1.001:
                    a_new = a_and

        # Track best
        M_new = fast_conv(a_new).max()
        if M_new < best_M * (1.0 - 1e-12):
            best_M = M_new
            best_a = a_new.copy()
            best_J = compute_J(best_a)
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

        if no_imp > 12:
            noise = rng.normal(0, max(0.02, perturb_size * 3), n)
            a = normalize(a + noise)
            hist_x = []
            hist_y = []
            no_imp = 0

    return best_a


def resample_seq(src, n_target):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n_target:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_target)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        print(f"\nPhase 4: Final LP polish", flush=True)
        a_final = run_lp_optimization(best_a, deadline, max_iters=500,
                                      with_line_search=True, anderson_window=5,
                                      perturb_every=25, perturb_size=0.003, rng=rng)
        J_final = compute_J(a_final)
        if J_final < best_J:
            best_J = J_final
            best_a = a_final.copy()
        print(f"  Final: J={best_J:.8f}", flush=True)

    # Verify with evaluate_sequence
    try:
        verified = evaluate_sequence([float(x) for x in best_a])
        if np.isfinite(verified) and verified < best_J:
            print(f"  Verified: {verified:.8f} < {best_J:.8f} (using verified)", flush=True)
            best_J = verified
        elif np.isfinite(verified):
            print(f"  Verified: {verified:.8f} (computed: {best_J:.8f})", flush=True)
    except Exception as e:
        print(f"  Verification error: {e}", flush=True)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best_J:.8f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in best_a]
```
