At each step, the LP solution g is the "best possible" sequence given the current a and the current M. As M decreases, g is forced to be "flatter," which is what we want.

The schedule could be: M_t = M_0 * (1 - t/T)^alpha, where T is the total number of iterations and alpha controls the rate of decrease.

Actually, this doesn't quite make sense because M should always be >= max(a*a) for the current a (otherwise the LP might be infeasible... wait, no, the LP constraint is on (a*g), not (a*a). So M can be anything > 0, and the LP is always feasible since g=0 is always feasible.)

Hmm, but if M is too small, the LP solution g will be very small (close to 0), and after normalization it might not be a good direction. So there's a trade-off.

Let me just use M = max(a*a) and rely on the line search for good convergence.

One final thought: **symmetry exploitation**. If the optimal sequence is symmetric (a[i] = a[n-1-i]), we can halve the number of variables. But I don't know if the optimal is symmetric, so let's not impose this constraint.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with quadratic line search (exploiting bilinear convolution structure)
2. Gerchberg-Saxton spectral initialization for diverse starting points
3. Multi-length exploration with resampling
4. Adaptive M-schedule and multi-restart strategy
Key innovation: vectorized line search using precomputed cross-convolutions,
enabling ~100 LP iterations per second for n~1024.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a, b=None):
    """Fast convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute the objective function value."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse matrix A where (A @ g)[k] = (a*g)(k) = sum_j a[k-j]*g[j]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def line_search_convex(a, g):
    """Find optimal t in [0,1] minimizing max_k conv((1-t)*a + t*g, (1-t)*a + t*g).
    Uses precomputed cross-convolutions for O(n) per evaluation."""
    aa = fast_conv(a)
    ag = fast_conv(a, g)
    gg = fast_conv(g)

    # f_t(k) = (1-t)^2 * aa[k] + 2*t*(1-t) * ag[k] + t^2 * gg[k]
    #       = aa[k] + t*(2*ag[k] - 2*aa[k]) + t^2*(gg[k] - 2*ag[k] + aa[k])
    A0 = aa
    B0 = 2.0 * (ag - aa)
    C0 = gg - 2.0 * ag + aa

    best_t = 0.0
    best_val = float(A0.max())

    # Coarse grid
    t_grid = np.linspace(0.0, 1.0, 51)
    for t in t_grid:
        f_t = A0 + t * B0 + t * t * C0
        val = float(f_t.max())
        if val < best_val:
            best_val = val
            best_t = t

    # Fine grid around best
    if 0 < best_t < 1:
        lo = max(0.0, best_t - 0.02)
        hi = min(1.0, best_t + 0.02)
        t_fine = np.linspace(lo, hi, 31)
        for t in t_fine:
            f_t = A0 + t * B0 + t * t * C0
            val = float(f_t.max())
            if val < best_val:
                best_val = val
                best_t = t

    a_new = (1.0 - best_t) * a + best_t * g
    return a_new, best_t, best_val


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: flatten power spectrum while maintaining non-negativity."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_mean = mag.mean() + 1e-30
        mag_new = (1.0 - beta) * mag + beta * mag_mean
        power = 1.0 - 0.5 * beta
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def resample_seq(src, n):
    """Resample a sequence to length n using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    if src.size < 2:
        return np.full(n, src.mean() if src.size > 0 else 1.0)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_lp_optimize(a, deadline, max_iters=5000):
    """Main LP best-response optimization with line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
            stall_count = 0
        else:
            no_improve += 1
            if best_M - M < best_M * 1e-10:
                stall_count += 1
            else:
                stall_count = 0

        if stall_count > 40:
            break
        if no_improve > 100:
            break

        # Solve LP
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Check if g is already close to a (converged)
        if np.abs(g - a).max() < 1e-8:
            break

        # Line search
        a_new, t_used, new_M = line_search_convex(a, g)

        if new_M >= best_M * (1 + 1e-10):
            # Line search didn't help, try smaller step
            for t_try in [0.5, 0.3, 0.1, 0.05]:
                a_try = (1.0 - t_try) * a + t_try * g
                M_try = float(fast_conv(a_try).max())
                if M_try < best_M:
                    a_new = a_try
                    new_M = M_try
                    break
            else:
                break

        a = np.maximum(a_new, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        if (it + 1) % 50 == 0:
            J = 2 * n * best_M / 1.0  # since sum=1
            print(f"    iter {it+1}: M={best_M:.8f} J={J:.6f} t={t_used:.3f}", flush=True)

    return best_a


def run_lp_optimize_adaptive(a, deadline, max_iters=5000):
    """LP optimization with adaptive M (start loose, tighten over time)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M_current = float(b.max())

        if M_current < best_M * (1.0 - 1e-12):
            best_M = M_current
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                break

        # Adaptive M: use current max, but allow slight tightening
        # This helps the LP find more "aggressive" directions
        M_lp = M_current

        g = lp_best_response(a, M_lp)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        if np.abs(g - a).max() < 1e-9:
            break

        a_new, t_used, new_M = line_search_convex(a, g)

        if new_M >= M_current * (1 + 1e-12):
            # Try direct step
            a_try = g  # full step to LP solution
            M_try = float(fast_conv(a_try).max())
            if M_try < M_current:
                a_new = a_try
            else:
                # Try half step
                a_try = 0.5 * a + 0.5 * g
                M_try = float(fast_conv(a_try).max())
                if M_try < M_current:
                    a_new = a_try
                else:
                    no_improve += 10
                    if no_improve > 80:
                        break
                    continue

        a = np.maximum(a_new, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

    return best_a


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations for a given length."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.001))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.001))

    # Uniform
    inits.append(np.ones(n) / n)

    # Concentrated center
    inits.append(np.exp(-8 * (t - 0.5) ** 2) / np.exp(-8 * (t - 0.5) ** 2).sum())

    # Two peaks
    inits.append((np.exp(-10 * (t - 0.3) ** 2) + np.exp(-10 * (t - 0.7) ** 2)) /
                 (np.exp(-10 * (t - 0.3) ** 2) + np.exp(-10 * (t - 0.7) ** 2)).sum())

    # Cosine window
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()

    # Random smooth
    k = max(3, n // 16)
    kern = np.ones(k) / k
    r = np.convolve(rng.uniform(0, 1, n), kern, 'same')
    inits.append(np.maximum(r, 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()

    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_flat_spectral(a_rand, n_iters=80, beta=0.85))

    # GS from parent
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(gs_flat_spectral(p, n_iters=100, beta=0.9))

    return inits
