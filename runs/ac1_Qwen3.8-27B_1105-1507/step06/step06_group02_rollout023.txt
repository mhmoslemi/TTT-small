def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def spectral_flat_init(n, rng, n_iters=500):
    """Initialize via spectral flattening: start with random non-negative, 
    iteratively flatten the power spectrum."""
    a = np.maximum(rng.normal(1.0, 0.3, n), 0.01)
    a /= a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        # Target: flat spectrum (|F| constant)
        target_mag = np.full_like(np.abs(F), np.abs(F).mean())
        # Interpolate between current and target
        alpha = 0.5 + 0.4 * np.sin(np.pi * it / n_iters)  # oscillate
        new_mag = alpha * target_mag + (1 - alpha) * np.abs(F)
        F_new = (F / (np.abs(F) + 1e-30)) * new_mag
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        M = fast_conv(a_new).max()
        if M < best_M:
            best_M = M
            best_a = a_new.copy()
        a = a_new
    return best_a


def smooth_grad(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using softmax."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    # Softmax weights
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    # Gradient: sum_k p_k * 2 * a[k-:][reversed]
    grad = np.zeros(n)
    # Only process significant p_k
    threshold = 1e-10
    active = np.where(p > threshold)[0]
    for k in active:
        # (conv(a,a))_k = sum_i a_i * a[k-i]
        # d/d(a_j) = 2 * a[k-j] (when 0 <= k-j < n)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        # For j in [j_min, j_max], contribution is 2 * a[k-j]
        # a[k-j] for j=j_min,...,j_max is a[k-j_max],...,a[k-j_min] = a[n-1-... ]
        # Let's just do it directly
        a_reversed = a[k - j_max : k - j_min + 1][::-1]  # a[k-j_max], ..., a[k-j_min]
        grad[j_min:j_max+1] += 2.0 * p[k] * a_reversed
    return grad, b_max, p


def adam_smooth(a, deadline, max_iters=3000, tau_start=0.5, tau_end=500.0,
                lr_init=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam-style optimization with adaptive smoothing."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iters)
        # SMoothing schedule: start moderate, grow aggressively
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_init / (1 + 0.5 * frac)
        
        grad, b_max, p = smooth_grad(a, tau)
        
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new
    
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def fast_conv(a):
    n = len(a)

    while size < 2 * n - 1:
        size <<= 1

    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')

def smooth_objective(a, tau):
    """Smoothed objective using log-sum-exp approximation of max."""
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 0.01:
        return 1e10
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -100, 100)
    log_sum_exp = b_max + np.log(np.sum(np.exp(shifted))) / tau
    return 2 * len(a) * log_sum_exp / (s * s)


def smooth_grad_lse(a, tau):
    """Gradient of the log-sum-exp smoothed objective."""
    a = np.maximum(a, 1e-10)
    s = a.sum()
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -100, 100)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    # d/da [log(sum exp(tau*(b-bmax)))/tau] = sum_k p_k * db_k/da
    # db_k/da_j = 2*a[k-j] (when valid)
    grad_b = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        a_seg = a[k - j_max : k - j_min + 1][::-1]
        grad_b[j_min:j_max + 1] += 2.0 * p[k] * a_seg
    # Full gradient of J = 2n*smooth_max/s^2
    # dJ/da = 2n*(s^2 * d(smooth_max)/da - smooth_max * 2s * ds/da) / s^4

    smooth_max_val = b_max + np.log(np.sum(exp_b)) / tau
    grad = 2.0 * n * (s * grad_b - 2.0 * smooth_max_val) / (s ** 3)

        f_val = float(b_max)

        # Nesterov restart: if f increased from y to where we were heading, reset momentum
        if prev_f is not None and f_val > prev_f * 1.001:
            momentum = 0.0
        prev_f = f_val

        new_momentum = (1 + np.sqrt(1 + 4 * momentum ** 2)) / 2
        a_new = np.maximum(y - lr * (grad + momentum * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((new_momentum - 1) / (new_momentum + 1)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
        momentum = new_momentum

    return best_a


def lbfgs_smooth(a, deadline, tau=30.0, max_iter=500):
    """Use L-BFGS-B on the smoothed log-sum-exp objective."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    def obj(x):
        if time.monotonic() > deadline:
            return 1e10
        x = np.maximum(x, 1e-10)
        return smooth_objective(x, tau)

    def grad(x):
        x = np.maximum(x, 1e-10)
        return smooth_grad_lse(x, tau)

    bounds = [(0.0, None)] * n
    try:
        res = minimize(obj, a, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iter,
                                               'ftol': 1e-15, 'gtol': 1e-8})
        if res.x is not None:
            a_res = np.maximum(res.x, 0)
            s = a_res.sum()
            if s > 1e-10:
                a_res /= s
                M = fast_conv(a_res).max()
                if M < best_M:
                    best_M = M
                    best_a = a_res.copy()
    except Exception:
        pass

    return best_a


def build_lp_sparse(n, a, active_k):
    """Build sparse constraint matrix for LP with given active lags."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
    return A


def lp_solve(a, M, active_k):
    """Solve the LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_lp_sparse(n, a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                    method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_mirror(a, deadline, max_iters=200, t_base=0.3, start_frac=0.4, end_frac=1.0):
    """LP mirror descent with progressive constraint tightening."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    all_k = np.arange(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break

        # Progressive constraint selection
        frac = start_frac + (end_frac - start_frac) * min(1.0, it / max_iters)
        if frac >= 0.99:
            active_k = all_k
        else:
            threshold = frac * M
            active_k = np.where(b >= threshold)[0]
            if len(active_k) < 5:
                active_k = all_k

        g = lp_solve(a, M, active_k)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        t = t_base / (1 + 0.05 * it)
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new

    return best_a


def lp_backtrack(a, deadline, max_iters=100, start_frac=0.8):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    all_k = np.arange(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, all_k)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        for t in [0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 15:
                break

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def perturb_seq(a, rng, sigma):
    a_p = a + rng.normal(0, sigma, len(a))
    a_p = np.maximum(a_p, 0)
    s = a_p.sum()
    if s < 1e-10:
        return a.copy()
    return a_p / s


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            s = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def make_inits(n, rng, parent):
    """Generate diverse initializations for sequence of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(perturb_seq(p, rng, 0.02))
        inits.append(perturb_seq(p, rng, 0.05))
        inits.append(perturb_seq(p, rng, 0.1))

    inits.append(np.ones(n))
    # Raised cosine
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    # Gaussian bump
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.05)
    # Double bump
    inits.append(np.exp(-6 * (t - 0.3) ** 2) + np.exp(-6 * (t - 0.7) ** 2) + 0.03)
    # Power-law taper
    inits.append(np.maximum(0.1 + 0.9 * (1 - np.abs(2 * t - 1) ** 0.5), 0.01))
    # Spectral init
    inits.append(spectral_init(n, rng, n_iters=150))
    # Smooth random
    k = max(3, n // 15)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    # Symmetric
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)

    return inits


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + budget_s - 20.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    # === PHASE 1: L-BFGS + Nesterov on multiple lengths ===
    print(f"\n{'=' * 50}", flush=True)
    print("PHASE 1: Fast Optimization (L-BFGS + Nesterov)", flush=True)
    print(f"{'=' * 50}", flush=True)
    p1_dl = time.monotonic() + 250

    n_base = parent.size
    lengths = sorted(set([n_base, 896, 960, 1024, 1088, 1152, 1280]))
    # Remove very large ones if time is tight
    lengths = [n for n in lengths if 128 <= n <= 1536]

    best_per_length = {}

    for n in lengths:
        if time.monotonic() > p1_dl - 20:
            break
        t0 = time.monotonic()
        sub_dl = min(time.monotonic() + 40, p1_dl - 10)

        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n_M = float('inf')
        best_n_a = None

        for idx, a0 in enumerate(inits):
            if time.monotonic() > sub_dl - 5:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 /= a0.sum()

            # L-BFGS first (fast global-ish)
            remaining = sub_dl - time.monotonic()
            if remaining > 10:
                a_lb = lbfgs_smooth(a0, time.monotonic() + remaining * 0.5,
                                    tau=20.0, max_iter=200)
            else:
                a_lb = a0

            # Then Nesterov for refinement
            if time.monotonic() < sub_dl - 5:
                a_ns = nesterov_smooth(a_lb, sub_dl, max_iters=1500,
                                       tau_start=1.0, tau_end=300.0,
                                       lr_init=0.3, lr_end=0.01)
            else:
                a_ns = a_lb

            M = fast_conv(a_ns).max()
            if M < best_n_M:
                best_n_M = M
                best_n_a = a_ns.copy()

        if best_n_a is not None:
            J = compute_J(best_n_a)
            print(f"  n={n:4d}: M={best_n_M:.8f} J={J:.6f} ({time.monotonic() - t0:.1f}s)", flush=True)
            best.update(best_n_a)
            best_per_length[n] = best_n_a.copy()

    best.report("Phase 1")

    # === PHASE 2: LP Mirror Descent (main workhorse) ===
    print(f"\n{'=' * 50}", flush=True)
    print("PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'=' * 50}", flush=True)
    p2_dl = min(time.monotonic() + 500, total_deadline - 200)

    if best.seq is not None:
        n = best.seq.size
        t0 = time.monotonic()
        print(f"  LP on n={n}", flush=True)

        # Pass 1: relaxed constraints, larger steps
        a = lp_mirror(best.seq, min(time.monotonic() + 250, p2_dl - 200),
                      max_iters=150, t_base=0.4, start_frac=0.3, end_frac=0.7)
        best.update(a)
        print(f"  Pass 1 done: M={fast_conv(a).max():.8f} ({time.monotonic() - t0:.0f}s)", flush=True)

        # Pass 2: tighter constraints
        if time.monotonic() < p2_dl - 100:
            a = lp_mirror(a, min(time.monotonic() + 200, p2_dl - 100),
                          max_iters=120, t_base=0.25, start_frac=0.6, end_frac=0.9)
            best.update(a)
            print(f"  Pass 2 done: M={fast_conv(a).max():.8f} ({time.monotonic() - t0:.0f}s)", flush=True)

        # Pass 3: full constraints
        if time.monotonic() < p2_dl - 50:
            a = lp_mirror(a, min(time.monotonic() + 150, p2_dl - 50),
                          max_iters=80, t_base=0.2, start_frac=0.8, end_frac=1.0)
            best.update(a)
            print(f"  Pass 3 done: M={fast_conv(a).max():.8f} ({time.monotonic() - t0:.0f}s)", flush=True)

    best.report("Phase 2")

    # === PHASE 3: Perturbation + re-optimize ===
    print(f"\n{'=' * 50}", flush=True)
    print("PHASE 3: Perturbation & Re-optimize", flush=True)
    print(f"{'=' * 50}", flush=True)
    p3_dl = min(time.monotonic() + 200, total_deadline - 100)

    if best.seq is not None:
        for sigma in [0.005, 0.01, 0.015, 0.02, 0.03]:
            if time.monotonic() > p3_dl - 20:
                break
            a_p = perturb_seq(best.seq, rng, sigma)
            # Quick Nesterov
            a_p = nesterov_smooth(a_p, min(time.monotonic() + 20, p3_dl - 15),
                                  max_iters=500, tau_start=5, tau_end=100, lr_init=0.15)
            best.update(a_p)
            # LP refine
            if time.monotonic() < p3_dl - 15:
                a_p = lp_mirror(a_p, min(time.monotonic() + 30, p3_dl - 10),
                                max_iters=40, t_base=0.25, start_frac=0.6, end_frac=0.9)
                best.update(a_p)

    best.report("Phase 3")

    # === PHASE 4: Multi-length LP polish ===
    print(f"\n{'=' * 50}", flush=True)
    print("PHASE 4: Multi-length polish", flush=True)
    print(f"{'=' * 50}", flush=True)
    p4_dl = min(time.monotonic() + 150, total_deadline - 50)

    if best_per_length and best.seq is not None:
        n_best = best.seq.size
        for n_try in sorted(best_per_length.keys(),
                            key=lambda x: abs(x - n_best)):
            if time.monotonic() > p4_dl - 10:
                break
            if n_try == n_best:
                continue
            a_c = best_per_length[n_try]
            # Quick refine
            a_c = nesterov_smooth(a_c, min(time.monotonic() + 10, p4_dl - 8),
                                  max_iters=300, tau_start=5, tau_end=80, lr_init=0.1)
            best.update(a_c)
            if time.monotonic() < p4_dl - 8:
                a_c = lp_mirror(a_c, min(time.monotonic() + 20, p4_dl - 5),
                                max_iters=30, t_base=0.25, start_frac=0.6, end_frac=0.9)
                best.update(a_c)

    best.report("Phase 4")

    # === PHASE 5: Final backtracking LP ===
    print(f"\n{'=' * 50}", flush=True)
    print("PHASE 5: Final Backtracking LP", flush=True)
    print(f"{'=' * 50}", flush=True)

    if best.seq is not None and time.monotonic() < total_deadline - 10:
        a_final = lp_backtrack(best.seq, total_deadline, max_iters=50, start_frac=0.9)
        best.update(a_final)

        # One more Nesterov polish
        if time.monotonic() < total_deadline - 5:
            a_polish = nesterov_smooth(a_final, total_deadline, max_iters=200,
                                       tau_start=20, tau_end=100, lr_init=0.05)
            best.update(a_polish)

    best.report("Phase 5")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'=' * 50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic() - best.t0:.0f}s", flush=True)
    print(f"{'=' * 50}", flush=True)
    return [float(x) for x in best.seq]
