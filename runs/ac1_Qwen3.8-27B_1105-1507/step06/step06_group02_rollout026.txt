def gs_flat_spectral(init, n_iters=200, beta=0.85):
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()

    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        # Compression: push magnitude toward mean
        target = mag.mean()
        mag_new = (1.0 - beta) * mag + beta * target
        # Also try power compression
        power = 0.6
        mag_alt = (mag ** power)
        mag_alt = mag_alt / (mag_alt.max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a

def smooth_max_grad(a, tau):
    """Compute gradient of smooth max approximation of the autocorrelation."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    # Log-sum-exp for numerical stability
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    exp_b_sum = exp_b.sum()
    if exp_b_sum < 1e-300:
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
    else:
        p = exp_b / exp_b_sum
    grad = np.zeros(n)
    # For each k with significant weight, add contribution to grad
    # b_k = sum_j a_j * a_{k-j}
    # db_k/da_j = a_{k-j} + a_{k-j} (if k-j != j) or 2*a_{k-j} (always, since conv is sum of products)
    # Actually: db_k/da_j = a_{k-j} (from the term a_j*a_{k-j}) + a_{j} * d(a_{k-j})/da_j
    # Wait, for convolution: b_k = sum_{j'} a_{j'} * a_{k-j'}
    # db_k/da_j: the term where j'=j gives a_{k-j} (if 0<=k-j<=n-1)
    #              the term where k-j'=j, i.e., j'=k-j, gives a_{k-j} (if 0<=k-j<=n-1 and k-j!=j)
    #              if k=2j: gives 2*a_j
    # So db_k/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j} (when k!=2j and 0<=k-j<=n-1)
    #               = 2*a_j (when k=2j)
    # In both cases: 2 * a_{k-j} (with the understanding that a_{k-j}=0 if out of range)
    # Wait, that's not right. Let me redo.
    # b_k = sum_{j'=0}^{n-1} a_{j'} * a_{k-j'} (where a_{m}=0 if m not in [0,n-1])
    # d(b_k)/d(a_j) = sum_{j'} [delta_{j',j} * a_{k-j'} + a_{j'} * delta_{k-j',j}]
    #                = a_{k-j} + a_{k-j}  (first term: j'=j gives a_{k-j}; second term: k-j'=j gives j'=k-j, so a_{k-j})
    #                = 2 * a_{k-j} (if 0 <= k-j <= n-1)
    #                = 0 (otherwise)
    # Wait, but if k-j = j (i.e., k = 2j), then both terms give a_j, so total is 2*a_j. ✓
    # If k-j != j, first term gives a_{k-j}, second term: j'=k-j, gives a_{k-j}. Total: 2*a_{k-j}. ✓
    # So in all cases: d(b_k)/d(a_j) = 2 * a_{k-j} if 0 <= k-j <= n-1, else 0.
    
    # For efficiency, for each j: grad[j] = sum_k p[k] * 2 * a[k-j] (where 0 <= k-j <= n-1)
    # This is 2 * (p * a)[j] where (p * a) is the cross-correlation of p and a... 
    # Actually it's 2 * sum_k p[k] * a[k-j] for k-j in [0, n-1]
    # = 2 * (p reversed) convolved with a, evaluated at position... 
    # Let me just do it directly for the active k's.
    
    active = np.where(p > 1e-10)[0]
    for k in active:
        wk = 2.0 * p[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For j in [lo, hi]: k-j is in [0, n-1], so contribution is wk * a[k-j]
        # j goes from lo to hi, k-j goes from k-lo to k-hi
        # a[k-lo : k-hi+1] is the slice we need, reversed
        grad[lo:hi+1] += wk * a[k-hi:k-lo+1][::-1]
    return grad, p, b, b_max

def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0, lr_init=0.4, lr_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    momentum = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        # Nesterov
        a_new = np.maximum(y - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        beta_k = (t_k - 1.0) / t_k
        y = a_new + beta_k * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a

def build_conv_matrix(n, a):
    """Build sparse matrix A where (A @ g)_k = (a * g)_k (cross-convolution)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, prev_x=None):
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    x0 = prev_x if prev_x is not None else None
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, x0=x0, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def lp_mirror_descent(a, deadline, max_iters=300, t_base=0.3, eta=0.03):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    prev_x = None
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        g = lp_best_response(a, M, prev_x)
        if g is None:
            break
        prev_x = g
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        # Adaptive step size: try multiple t values
        improved = False
        for t in [t_base, t_base*0.6, t_base*0.3, t_base*0.1, 0.02, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if not improved:
            # Try a very small step anyway to avoid getting stuck
            t_small = 0.005
            a_t = (1.0 - t_small) * a + t_small * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s > 1e-10:

                M_t = fast_conv(a_t).max()

def deflation_phase(a, deadline, n_rounds=50):
    """Targeted deflation: identify peak autocorrelation values and reduce them."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Find indices where b_k is close to max
        threshold = M * 0.99
        active_k = np.where(b >= threshold)[0]
        if len(active_k) == 0:
            break
        # For each active k, find which a_j contributes most
        # and try to reduce it
        reduced = False
        for k in active_k[:10]:  # Limit to top 10 peaks
            # Find j that maximizes a_j * a_{k-j} (the dominant contribution)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            contrib = a[lo:hi+1] * a[k-hi:k-lo+1][::-1]
            j_local = np.argmax(contrib)
            j = lo + j_local
            partner = k - j
            if partner < 0 or partner >= n:
                continue
            # Try reducing a[j] (or a[partner]) slightly
            for target_j in [j, partner]:
                for frac in [0.1, 0.05, 0.02]:
                    a_try = a.copy()
                    reduction = a_try[target_j] * frac
                    if reduction < 1e-10:
                        continue
                    a_try[target_j] -= reduction
                    # Redistribute to the coordinate with smallest impact
                    # (i.e., the one that's most "flat" in the autocorrelation)
                    # Simple approach: add to the coordinate with minimum a
                    min_j = np.argmin(a_try)
                    a_try[min_j] += reduction
                    s = a_try.sum()
                    if s < 1e-10:
                        continue
                    a_try /= s
                    M_try = fast_conv(a_try).max()
                    if M_try < M - 1e-14:
                        a = a_try
                        M = M_try
                        reduced = True
                        break
                if reduced:
                    break
            if reduced:
                break
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
        if not reduced:
            break
    return best_a

def resample_seq(src, n):

One more idea: **Use the dual variables from the LP to guide the step size.**

The LP has dual variables y_k >= 0 for each constraint (A@g)_k <= M. At the optimum, complementary slackness gives:
- If (A@g)_k < M, then y_k = 0
- If y_k > 0, then (A@g)_k = M

The dual variables y_k tell us which constraints are "binding." The gradient of the objective with respect to a (in the context of the outer optimization) is related to y. Specifically, if we think of the LP value as a function of a:
V(a) = max {sum(g) : A(a)@g <= M, g >= 0}

Then dV/da_j = -sum_k y_k * a_{k-j} (by the envelope theorem, since A depends on a).

So the "true" gradient for the outer problem is -sum_k y_k * a_{k-j}, and the descent direction is in the direction of -grad.

This is essentially what the LP mirror descent is doing: the LP solution g gives the primal, and the dual y gives the gradient information. But the previous code doesn't explicitly use the dual; it just uses the primal to form the update direction.

Can we use the dual to get a better step size? The key is:

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton spectral initialization
2. Nesterov-accelerated smooth gradient descent
3. LP mirror descent with warm starts, adaptive line search

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: flatten power spectrum while maintaining non-negativity."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_max = mag.max() + 1e-30
        mean_mag = mag.mean() + 1e-30
        # Linear compression toward mean
        mag_lin = (1.0 - beta) * mag + beta * mean_mag
        # Power compression
        power = 0.5 + 0.3 * beta
        mag_pow = (mag / mag_max) ** power
        mag_pow = mag_pow * mag_max
        # Mix
        mag_final = 0.5 * mag_lin + 0.5 * mag_pow
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def smooth_max_grad(a, tau):
    """Gradient of smooth max (log-sum-exp) approximation of autocorrelation."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_b = np.exp(shifted)
    exp_b_sum = exp_b.sum()
    if exp_b_sum < 1e-300:
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
    else:
        p = exp_b / exp_b_sum
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        wk = 2.0 * p[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += wk * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.4, lr_end=0.003):
    """Nesterov-accelerated gradient descent on smooth max approximation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    t_k = 1.0
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        a_new = np.maximum(y - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        t_k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        beta_k = (t_k - 1.0) / t_k_new
        y = a_new + beta_k * (a_new - a)
        t_k = t_k_new
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def build_conv_matrix(n, a):
    """Build sparse matrix A where (A @ g)_k = (a * g)_k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            rows.extend(range(k, k + (j_max - j_min + 1)))
            cols.extend(range(j_min, j_max + 1))
            vals.extend(a[k - j_max:k - j_min + 1][::-1])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, A_cache=None, prev_x=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    if A_cache is None:
        A_cache = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        kwargs = {'presolve': True, 'dual_feasibility_tolerance': 1e-9}
        if prev_x is not None:
            r = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                        x0=prev_x, method='highs', options=kwargs)
        else:
            r = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                        method='highs', options=kwargs)
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, max_iters=300, t_base=0.35, use_line_search=True):
    """LP mirror descent with warm starts and adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    prev_x = None
    A_cache = None
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break
        A_cache = build_conv_matrix(n, a)
        g = lp_best_response(a, M, A_cache, prev_x)
        if g is None:
            break
        prev_x = g.copy()
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        # Line search for step size
        improved = False
        if use_line_search:
            for t in [t_base, 0.6 * t_base, 0.3 * t_base, 0.15 * t_base, 0.05, 0.02, 0.01]:
                a_t = (1.0 - t) * a + t * g
                a_t = np.maximum(a_t, 0)
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t /= s
                M_t = fast_conv(a_t).max()
                if M_t < best_M - 1e-14:
                    a = a_t
                    best_M = M_t
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            # Small step to avoid stalling
            t = 0.005
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s > 1e-10:
                a_t /= s
                M_t = fast_conv(a_t).max()
                if M_t <= best_M + 1e-12:
                    a = a_t
    return best_a


def deflation_phase(a, deadline, max_rounds=200):
    """Targeted deflation: reduce peak autocorrelation by adjusting specific coordinates."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Find top peaks
        threshold = M * 0.995
        active_k = np.where(b >= threshold)[0]
        if len(active_k) == 0:
            break
        reduced = False
        for k in active_k[:5]:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # Contribution of each j to b_k
            contrib = a[lo:hi + 1] * a[k - hi:k - lo + 1][::-1]
            # Find the dominant contributor
            j_local = np.argmax(contrib)
            j = lo + j_local
            partner = k - j
            if partner < 0 or partner >= n:
                continue
            # Try reducing the larger of a[j] and a[partner]
            candidates = []
            if a[j] >= a[partner]:
                candidates.append(j)
            if a[partner] > a[j]:
                candidates.append(partner)
            for target_j in candidates:
                for frac in [0.05, 0.02, 0.01, 0.005]:
                    reduction = a[target_j] * frac
                    if reduction < 1e-10:
                        continue
                    a_try = a.copy()
                    a_try[target_j] -= reduction
                    # Redistribute: add to coordinate with least impact on peaks
                    # Simple heuristic: add to the minimum a
                    min_j = np.argmin(a_try)
                    a_try[min_j] += reduction
                    s = a_try.sum()
                    if s < 1e-10:
                        continue
                    a_try /= s
                    M_try = fast_conv(a_try).max()
                    if M_try < M - 1e-14:
                        a = a_try
                        M = M_try
                        reduced = True
                        break
                if reduced:
                    break
            if reduced:
                break
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
        if not reduced:
            # Try a different strategy: reduce ALL coordinates that contribute to top peaks
            top_k = active_k[0]
            lo = max(0, top_k - n + 1)
            hi = min(n - 1, top_k)
            contrib = a[lo:hi + 1] * a[top_k - hi:top_k - lo + 1][::-1]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                break
            # Reduce proportionally
            a_try = a.copy()
            scale = 0.02
            a_try[lo:hi + 1] -= scale * contrib
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                break
            a_try /= s
            M_try = fast_conv(a_try).max()
            if M_try < best_M - 1e-14:
                a = a_try
                best_M = M_try
                best_a = a.copy()
            else:
                break
    return best_a

def resample_seq(src, n):
    """Resample sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
