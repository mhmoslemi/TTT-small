from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response_reduced(a, M, active_indices=None, tol=1e-6):
    """Solve LP with only active constraints for speed."""
    n = len(a)
    b = fast_conv(a)
    if active_indices is None:
        # Find active constraints
        threshold = M - tol
        active_indices = np.where(b >= threshold)[0]
        if len(active_indices) == 0:
            active_indices = np.where(b >= M * 0.99)[0]
        if len(active_indices) == 0:
            active_indices = np.argmax(b).reshape(1)
    
    # Build constraint matrix only for active constraints
    rows = []
    cols = []
    vals = []
    for k in active_indices:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(j - j_min)  # 0-indexed row
            cols.append(j)
            vals.append(a[k - j])
    
    if not rows:
        # Fallback: just return a uniform vector
        return np.ones(n) / n
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def run_mirror_descent(a, deadline, max_iters=500, use_line_search=True,
                       active_tol=1e-6, verbose=True):
    """LP mirror descent with active set reduction and optional line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    t = 0.3
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
            if verbose and it % 50 == 0:
                print(f"    iter {it}: M={M:.10f}, active_tol={active_tol}")
        else:
            no_imp += 1
            if no_imp > 50:
                break
        
        # Solve LP with reduced constraints
        g = lp_best_response_reduced(a, M, tol=active_tol)
        if g is None:
            # Fallback: full LP
            g = lp_best_response_full(a, M)
            if g is None:
                break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Line search for step size
        if use_line_search:
            best_t = 0
            best_M_t = M
            for t_try in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05, 0.02]:
                a_t = np.maximum((1 - t_try) * a + t_try * g, 0)
                s_t = a_t.sum()
                if s_t < 1e-10:

                a_t = a_t / s_t
                M_t = fast_conv(a_t).max()
                if M_t < best_M_t:
                    best_M_t = M_t
                    best_t = t_try
                    best_a_t = a_t.copy()
            if best_t > 0:
                a = best_a_t
            else:
                a = np.maximum((1 - 0.05) * a + 0.05 * g, 0)
                a = a / a.sum()
        else:
            a = np.maximum((1 - t) * a + t * g, 0)
            s = a.sum()
            if s < 1e-10:
                a = a.copy()  # shouldn't happen
            else:
                a = a / s
            # Decay step size
            t = max(0.01, t * 0.995)
    
    return best_a

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    """Compute gradient of sum_k conv(a,a)_k^2 with respect to a."""
    n = len(a)
    b = fast_conv(a)

    # = sum_{k'} b[k'+j] * a[k']

    # f(a) = sum_k (a^T T_k a)^2 where T_k is the k-th off-diagonal
    # grad f = 4 * sum_k (a^T T_k a) * T_k a

    # So (grad f)_j = 4 * sum_k b_k * a_{k-j}
    # 
    # This is 4 * (b * a_rev)[j + n - 1] where a_rev[i] = a[n-1-i] and * is full conv
    # Let me verify: (b * a_rev)[m] = sum_k b[k] * a_rev[m-k] = sum_k b[k] * a[n-1-m+k]
    # Setting m = j + n - 1: sum_k b[k] * a[n-1-j-n+1+k] = sum_k b[k] * a[k-j] ✓

    # We want d[j + n - 1] for j = 0, ..., n-1

    return grad

    """Minimize sum_k conv(a,a)_k^2 using gradient descent."""

    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = np.sum(fast_conv(a)**2)

        grad = variance_grad(a)
        # Line search
        a_new = a.copy()
        best_lr = lr
        for lr_try in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:
            a_try = np.maximum(a - lr_try * grad, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = np.sum(fast_conv(a_try)**2)
            if obj_try < best_obj:
                best_obj = obj_try
                a_new = a_try.copy()
                best_lr = lr_try
                break
        if a_new is not a:  # if we found an improvement
            a = a_new
            max_val = fast_conv(a).max()
            if max_val < best_max:
                best_max = max_val
                best_a = a.copy()
            lr = min(lr * 1.05, 1.0)  # slowly increase lr
        else:
            lr = lr * 0.5
            if lr < 1e-8:
                break
    
    return best_a

    """Solve the LP: max sum(g) s.t. conv(a,g)[k] <= M for k in active_k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    
    if active_k is None:

        threshold = M - tol
        active_k = np.where(b >= threshold)[0]

    cols = []

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Variance minimization (quartic smooth objective) as warm start
2. LP mirror descent with adaptive active set reduction (10-100x faster LP)
3. Line search for optimal step size per iteration
4. Symmetric LP variant for additional speedup
5. Multi-length and perturbation-based multi-start
Key improvements over previous: active set LP gives 10x more iterations,
line search ensures monotone progress, variance warm start avoids bad regions.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lp_solve_active(a, M, active_k, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for k in active_k, g >= 0.
    Uses only the specified active constraints for speed."""
    n = len(a)
    if len(active_k) == 0:
        return np.ones(n) / n

    cols = []
    vals = []
    for k in active_k:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            cols.append(j)
            vals.append(a[k - j])

    if not cols:
        return np.ones(n) / n

    A = csr_matrix((vals, (list(active_k), cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_solve_full(a, M, time_limit=10):
    """Solve the full LP with all 2n-1 constraints."""
    n = len(a)
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            cols.append(j)
            vals.append(a[k - j])

    A = csr_matrix((vals, (list(range(2 * n - 1)), cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def get_active_set(b, M, n, top_k=30, rel_tol=0.005):
    """Determine active constraint indices: those where b_k is close to M."""
    # Include constraints within relative tolerance of M
    threshold = M * (1 - rel_tol)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.where(b >= M - 1e-10)[0]
    # Also always include the top few
    top = np.argsort(b)[-top_k:]
    active = np.unique(np.concatenate([active, top]))
    return active


def mirror_descent_fast(a, deadline, max_iters=2000, verbose=True):
    """Fast LP mirror descent with adaptive active set and line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_imp = 0
            if verbose and (it < 20 or it % 50 == 0):
                J = 2.0 * n * M
                print(f"    MD it={it:4d}: J={J:.8f}, M={M:.10f}, "
                      f"active~{np.sum(b >= M*0.995)}", flush=True)
        else:
            no_imp += 1
            if no_imp > 80:
                break

        # Adaptive active set: wider early, tighter later
        if it < 20:
            rel_tol = 0.02
            top_k = 50
        elif it < 100:
            rel_tol = 0.01
            top_k = 30
        else:
            rel_tol = 0.005
            top_k = 20

        active = get_active_set(b, M, n, top_k=top_k, rel_tol=rel_tol)
        g = lp_solve_active(a, M, active, time_limit=3)
        if g is None:
            # Fallback: full LP
            g = lp_solve_full(a, M, time_limit=8)
            if g is None:
                break

        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Line search
        best_t = 0.0
        best_M_t = M
        best_a_t = a.copy()
        for t_try in [0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05, 0.02]:
            a_t = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
            s_t = a_t.sum()
            if s_t < 1e-10:
                continue
            a_t = a_t / s_t
            M_t = float(fast_conv(a_t).max())
            if M_t < best_M_t:
                best_M_t = M_t
                best_t = t_try
                best_a_t = a_t.copy()

        if best_t > 0:
            a = best_a_t
        else:
            # Tiny step
            a = np.maximum((1.0 - 0.005) * a + 0.005 * g, 0.0)
            a = a / a.sum()

    return best_a


def variance_grad(a):
    """Compute gradient of sum_k conv(a,a)_k^2. Uses FFT for O(n log n)."""
    n = len(a)
    b = fast_conv(a)
    a_rev = a[::-1]
    size = 1
    while size < len(b) + n - 1:
        size <<= 1
    B = np.fft.rfft(b, size)
    Ar = np.fft.rfft(a_rev, size)
    d = np.fft.irfft(B * Ar, size)
    # (grad f)_j = 4 * sum_k b[k] * a[k-j] = 4 * d[j + n - 1]
    grad = 4.0 * d[n - 1:n - 1 + n]
    return grad


def minimize_variance(a, deadline, max_iters=2000, verbose=True):
    """Minimize variance of conv(a,a) using gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    v = np.zeros(n)
    lr = 0.05
    momentum = 0.9

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad = variance_grad(a)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm

        v = momentum * v + (1 - momentum) * grad
        step = lr / (np.linalg.norm(v) + 1e-10)

        a_new = np.maximum(a - step * v, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        M_new = float(fast_conv(a_new).max())
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            if verbose and (it < 10 or it % 100 == 0):
                print(f"    Var it={it:4d}: J={2*n*M_new:.8f}, M={M_new:.10f}", flush=True)
        a = a_new
        lr = min(lr * 1.01, 0.5)

    return best_a


def smooth_max_descent(a, deadline, max_iters=3000, tau_start=2.0, tau_end=100.0, verbose=True):
    """Smooth max descent using exp soft-max approximation with Nesterov."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = 0.3 * (0.01 / 0.3) ** frac

        b = fast_conv(y)
        b_max = float(b.max())
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        grad = np.zeros(n)
        active = np.where(p > 1e-12)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = y[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * p[k] * seg

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new

        M_cur = float(fast_conv(a).max())
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):

            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:

        s = float(evaluate_sequence(a.tolist()))
