"""
FFT-based gradient descent + LP alternating for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth ||b||^2 gradient for fast progress, LP for precision, and peak-chasing for final polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except:
                continue
    except:
        pass
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except:
        pass
    return None

    Minimize ||conv(a,a)||^2 using FFT-based gradient descent.
    """

    a = np.maximum(a, 0).astype(np.float64)

    if s < 1e-10:
        return a

    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = 0.9 if use_momentum else 0.0
    v = np.zeros(n)
    
    # Precompute FFT size
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    
    a_flip = a[::-1].copy()
    
    stall = 0
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute b = conv(a, a)

        if s_new < 1e-10:

            v = np.zeros(n)

def chase_peak(a, time_budget=30.0):
    """Subgradient chase on the peak of the convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    eta = 0.1
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1

        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        k_star = int(np.argmax(b))
        
        # Subgradient of b[k_star] w.r.t. a[i]
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
        
        grad = grad - np.mean(grad)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.5)
        else:
            eta *= 0.95
            if eta < 1e-6:
                break
    
    return best_a

def random_init(n, rng):
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Edge-concentrated
        x = np.arange(n) / (n - 1)
        a = x**2 + (1-x)**2 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n//4, 3*n//4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c)**2) / (n//8)**2)
        a = a / np.sum(a)
    else:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    return a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def compute_obj_from_conv(a, b):
    """Compute objective given a (normalized) and its auto-convolution b."""
    n = len(a)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g_normalized or None.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.ECOS, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:

    except Exception:
        pass
    return None


def fft_gradient_descent(a, time_budget=120.0, mode='energy', verbose=False):

        a_flip = a[::-1]
        faf = rfft(a_flip, N)
        fb = rfft(b, N)
        c_full = irfft(faf * fb, N)[:3 * n - 2]
        c = c_full[n - 1:2 * n - 1]

        if mode == 'peak':
            # Softmax weighting: focus on peak

            w = np.exp(b_shifted)

                c = cw_full[n - 1:2 * n - 1]

        # Center gradient
        c = c - np.mean(c)

        # Momentum update
        v = momentum * v - eta * c
        a_new = a + v
        a_new = np.maximum(a_new, 0)

            v = np.zeros(n)

def chase_peak(a, time_budget=30.0):

def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 6)
    if mode == 0:

        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:

        centers = [n // 4, 3 * n // 4]

        x = np.arange(1, n + 1, dtype=np.float64)

    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            a_pc = chase_peak(best_seq, time_budget=min(remaining * 0.3, 20.0))
            obj_pc = compute_obj(a_pc)
            if obj_pc < best_score:
                best_score = obj_pc
                best_seq = a_pc.copy()
                print(f"  ** Final Peak: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
