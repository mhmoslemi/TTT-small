def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0: return np.array([])
    size=1
    while size < 2*n-1: size <<= 1
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa*fa, size)[:2*n-1]
    b = np.maximum(b, 0.0)
    return b

def norm_pos(a):
    a = np.asarray(a, dtype=np.float64)

    s = float(a.sum())

    return a / s

def resample_to_length(x, n):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.ones(n)
    if x.size == n:
        return x.copy()
    t_old = np.linspace(0.0, 1.0, x.size)
    t_new = np.linspace(0.0, 1.0, n)
    y = np.interp(t_new, t_old, x)
    return np.maximum(y, 0.0)

def lp_solve(p, beta, U, penalty, time_limit=10, active=None):
    p = np.asarray(p); n=len(p); m_full=2*n-1
    if U <= 0 or n*U < 1.0 + 1e-12: U = (1.0 + 1e-12)/n
    M_arr = fast_conv(p)

        k_vals = np.arange(m_full)
    else:

        if k_vals.size == 0: k_vals = np.arange(m_full)
    m = k_vals.size
    alpha = 1.0 - beta
    if beta <= 0 or beta >= 1: return None
    factor = 2.0*alpha*beta
    A = np.zeros((m, n+1), dtype=np.float64)
    A[:, -1] = -1.0
    for idx, k in enumerate(k_vals):
        k = int(k)
        js = k - (n - 1)
        if js < 0: js = 0
        je = k + 1
        if je > n: je = n
        if je > js:
            # p indices k - arange(js,je)
            A[idx, js:je] = factor * p[k - np.arange(js, je)]
    b_ub = -alpha*alpha * M_arr[k_vals]
    if penalty is not None:
        b_ub = b_ub - float(penalty)*float(penalty)
    A_eq = np.zeros((1, n+1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    c = np.zeros(n+1); c[-1] = 1.0
    bounds = []
    for _ in range(n): bounds.append((0.0, float(U)))
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=[1.0], bounds=bounds, method='highs', options={'time_limit': time_limit, 'presolve': True})

        return None

```python
"""
Minimize evaluate_sequence(a) = 2*n*max(conv(a,a))/(sum(a))^2 for nonnegative a.

This solver uses a multi-method local optimization stack fixed to length 768:
  1. Exact coordinate descent on the scale-invariant objective.
  2. Linearized minimax best-response steps via HiGHS LPs (optimistic,
     and a self-bounded safe variant for guaranteed descent when possible).
  3. Least-squares active-set balancing directions.
  4. Entropic multiplicative / mirror-descent updates using log-sum-exp and
     variance-flattening weights.

All helpers are top-level; no lambdas or nested closures are used.
"""

import numpy as np
import time
from scipy.optimize import linprog

try:
    from numpy.lib.stride_tricks import sliding_window_view
except Exception:
    sliding_window_view = None


def fast_conv(a):
    """FFT-based linear convolution a*a; clips tiny negative roundoff to zero."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    b = np.maximum(b, 0.0)
    return b


def norm_pos(a):
    """Project to nonnegative sequence with sum 1."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = len(a)
    if s < 1e-12:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def score_from_p(p):
    """Normalized objective value for p (p is rescaled to sum 1)."""
    p = norm_pos(p)
    b = fast_conv(p)
    if b.size == 0:
        return float("inf")
    return float(2.0 * len(p) * np.max(b))


def resample_to_length(x, n):
    """Linearly resample a 1D nonnegative sequence to length n."""
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    if x.size == 0:
        return np.ones(n, dtype=np.float64)
    if x.size == n:
        return x.copy()
    t_old = np.linspace(0.0, 1.0, x.size)
    t_new = np.linspace(0.0, 1.0, n)
    y = np.interp(t_new, t_old, x)
    return np.maximum(y, 0.0)


def softmax_shifted(x, tau=1.0):
    """Stable softmax of tau*x."""
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x
    xm = x - np.max(x)
    e = np.exp(tau * xm)
    s = float(e.sum())
    if s <= 1e-300:
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    return e / s


def gradient_from_weights(weights, p):
    """
    Compute dots d_i = sum_j p_j weights[i+j] for i=0..n-1.
    This is the derivative factor for convolution coefficients before multiplying
    by 2/4 as needed.
    """
    n = len(p)
    if sliding_window_view is not None:
        try:
            win = sliding_window_view(weights, n)
            g = win @ p
            if np.all(np.isfinite(g)):
                return np.asarray(g, dtype=np.float64)
        except Exception:
            pass
    g = np.zeros(n, dtype=np.float64)
    for i in range(n):
        g[i] = np.dot(weights[i:i + n], p)
    return g


def mirror_update(p, weights, factor, base_eta_scale=0.1):
    """
    One multiplicative/Frank-Wolfe-like update.

    If factor=2 and weights are LSE probabilities, this follows -grad log-sum-exp.
    If factor=4 and weights=b-mean(b), it follows -grad variance of b.
    """
    p = norm_pos(p)
    dots = gradient_from_weights(weights, p)
    g = factor * dots
    gt = g - g.mean()
    std = float(np.std(gt))
    if std < 1e-15:
        return None

    best_q = None
    best_score = score_from_p(p)
    base_eta = base_eta_scale / std
    shift = gt - np.min(gt)
    logp = np.log(np.maximum(p, 1e-300))

    mults = [0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 10.0, 30.0, 100.0]
    for mult in mults:
        eta = base_eta * mult
        if eta <= 0.0 or not np.isfinite(eta):
            continue
        exponent = -eta * shift
        exponent = np.clip(exponent, -700.0, 700.0)
        logq = logp + exponent
        q = np.exp(logq - np.max(logq))
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s <= 1e-300:
            continue
        q = q / s
        if not np.all(np.isfinite(q)):
            continue
        sc = score_from_p(q)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_q = q
    return best_q


def run_mirror_descent(p_init, deadline, tau_values=None):
    """Mirror descent with alternating LSE max-reduction and variance flattening."""
    if tau_values is None:
        tau_values = [50.0, 150.0, 400.0, 1000.0]

    p = norm_pos(p_init).copy()
    best = p.copy()
    best_score = score_from_p(p)
    it = 0
    no_improve = 0

    while time.monotonic() < deadline and no_improve < 25 and it < 10000:
        b = fast_conv(p)
        if it % 3 == 2:
            weights = b - np.mean(b)
            factor = 4.0
        else:
            tau = float(tau_values[it % len(tau_values)])
            weights = softmax_shifted(b, tau)
            factor = 2.0

        q = mirror_update(p, weights, factor)
        if q is None:
            no_improve += 1
        else:
            sc = score_from_p(q)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                p = q
                best = q.copy()
                best_score = sc
                no_improve = 0
            else:
                no_improve += 1
        it += 1

    return best, best_score


def build_deriv_matrix(p, k_vals):
    """
    Matrix D where D[idx, j] = p[k - j] if 0 <= k-j < n, else 0.
    The linear derivative of b_k wrt p_i is 2 * D[k, i].
    """
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    k_vals = np.asarray(k_vals, dtype=int)
    m = len(k_vals)
    D = np.zeros((m, n), dtype=np.float64)
    for idx in range(m):
        k = int(k_vals[idx])
        js = k - (n - 1)
        if js < 0:
            js = 0
        je = k + 1
        if je > n:
            je = n
        if je > js:
            D[idx, js:je] = p[k - np.arange(js, je)]
    return D


def lp_solve(p, beta, U, penalty, time_limit=10.0, active=None):
    """
    Solve a linearized minimax direction LP.

    Variables q_i in [0,U], sum q_i=1, plus T.
    Constraints:
      2 alpha beta L_k(q) - T <= -alpha^2 M_k - beta^2 penalty
    L_k(q)=conv(p,q)_k.

    If penalty is the same as U (or a valid upper bound for q_i), this gives a
    safe majorizer because conv(q,q)_k <= max_i q_i for nonnegative q sum 1.
    """
    p = norm_pos(p)
    n = len(p)
    m_full = 2 * n - 1
    beta = float(beta)
    U = float(U)
    if U <= 0.0 or n * U < 1.0 + 1e-12:
        U = (1.0 + 1e-12) / float(n)

    if active is None:
        k_vals = np.arange(m_full, dtype=int)
    else:
        k_vals = np.asarray(active, dtype=int).ravel()
        if k_vals.size == 0:
            k_vals = np.arange(m_full, dtype=int)

    m = len(k_vals)
    if m == 0:
        return None

    M_arr = fast_conv(p)
    alpha = 1.0 - beta
    if beta <= 1e-12 or beta >= 1.0:
        return None

    D = build_deriv_matrix(p, k_vals)
    factor = 2.0 * alpha * beta

    A = np.zeros((m, n + 1), dtype=np.float64)
    if m > 0:
        A[:, :n] = factor * D
    A[:, -1] = -1.0

    b_ub = -alpha * alpha * M_arr[k_vals]
    if penalty is not None:
        b_ub = b_ub - float(penalty) * float(penalty)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0

    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    bounds = []
    for _ in range(n):
        bounds.append((0.0, U))
    bounds.append((0.0, None))

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=[1.0],
            bounds=bounds,
            method="highs",
            options={"time_limit": max(1.0, float(time_limit)), "presolve": True},
        )
    except Exception:
        return None

    if (not res.success) or res.x is None:
        return None

    q = np.asarray(res.x[:n], dtype=np.float64)
    q = np.maximum(q, 0.0)
    q = np.minimum(q, U * 1.000001 + 1e-12)
    s = float(q.sum())
    if s <= 1e-12:
        return None
    q = q / s
    if not np.all(np.isfinite(q)):
        return None
    return q


def eval_tau(p, q, tau):
    """Score of (1-tau)p + tau q."""
    if tau <= 1e-9 or tau >= 1.0:
        return float("inf")
    r = (1.0 - tau) * p + tau * q
    return score_from_p(r)


def try_direction(p, q, current_score, tau_list):
    """Line-search a convex-combination direction q and return best accepted point."""
    p = norm_pos(p)
    q = norm_pos(q)
    if not np.all(np.isfinite(p)) or not np.all(np.isfinite(q)):
        return None, current_score

    base_taus = [0.001, 0.003, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
    if tau_list is not None:
        for t in tau_list:
            base_taus.append(float(t))

    unique = []
    for t in base_taus:
        if np.isfinite(t) and 1e-8 < t < 0.999:
            dup = False
            for u in unique:
                if abs(t - u) < 1e-12:
                    dup = True
                    break
            if not dup:
                unique.append(float(t))
    unique.sort()

    best_p = None
    best_sc = current_score
    best_tau = None

    for t in unique:
        sc = eval_tau(p, q, t)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            r = (1.0 - t) * p + t * q
            best_p = norm_pos(r)
            best_tau = t

    # Light local refinement around best grid point.
    if best_p is not None and best_tau is not None and len(unique) >= 2:
        idx = 0
        for ii, ut in enumerate(unique):
            if abs(ut - best_tau) < 1e-12:
                idx = ii
                break
        lo = unique[idx - 1] if idx > 0 else 0.0
        hi = unique[idx + 1] if idx + 1 < len(unique) else min(0.6, best_tau * 1.5 + 0.01)
        if hi - lo > 1e-5:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a = lo
            bb = hi
            c_x = bb - gr * (bb - a)
            d_x = a + gr * (bb - a)
            fc = eval_tau(p, q, c_x)
            fd = eval_tau(p, q, d_x)
            for _ in range(18):
                if fc < fd:
                    bb = d_x
                    d_x = c_x
                    fd = fc
                    c_x = bb - gr * (bb - a)
                    fc = eval_tau(p, q, c_x)
                else:
                    a = c_x
                    c_x = d_x
                    fc = fd
                    d_x = a + gr * (bb - a)
                    fd = eval_tau(p, q, d_x)
            t_opt = 0.5 * (a + bb)
            sc = eval_tau(p, q, t_opt)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                r = (1.0 - t_opt) * p + t_opt * q
                best_p = norm_pos(r)

    if best_p is None:
        return None, current_score
    return best_p, best_sc


def run_lp_mm(p_init, deadline):
    """Iterative linearized minimax best-response with active sets and safe fallback."""
    p = norm_pos(p_init).copy()
    best = p.copy()
    best_score = score_from_p(p)
    n = len(p)
    m_full = 2 * n - 1

    beta = 0.12
    no_improve = 0
    it = 0

    U_opts = [0.04, 0.03, 0.02, 0.01, max(1.2 / n, 0.004)]
    K_opts = [40, 80, 150, None, 200]

    while time.monotonic() < deadline and no_improve < 22:
        it += 1
        b = fast_conv(best)
        M = float(b.max())
        if M <= 0.0:
            break

        K = K_opts[it % len(K_opts)]
        U = U_opts[it % len(U_opts)]

        if K is None:
            active = np.arange(m_full, dtype=int)
        else:
            K = int(min(K, m_full))
            active = np.argsort(b)[::-1][:K]

        improved = False

        # Optimistic active LP.
        q = lp_solve(best, beta, U, None, 5.0, active)
        if q is not None:
            r, sc = try_direction(best, q, best_score, [0.3, 0.2, 0.15, beta, 0.07, 0.05, 0.03, 0.02, 0.01])
            if r is not None and sc < best_score - 1e-12:
                best = r
                best_score = sc
                improved = True
                if beta < 0.25:
                    beta = min(0.30, beta * 1.12)
                continue

        # Smaller beta optimistic retry.
        beta2 = max(0.025, min(0.05, beta * 0.5))
        q = lp_solve(best, beta2, U, None, 5.0, active)
        if q is not None:
            r, sc = try_direction(best, q, best_score, [0.05, 0.03, 0.02, 0.01, 0.005])
            if r is not None and sc < best_score - 1e-12:
                best = r
                best_score = sc
                improved = True
                beta = max(0.05, beta * 0.9)
                continue

        # Safe bounded self-conv surrogate with small active set.
        if not improved and time.monotonic() < deadline - 8.0:
            U_safe = max(1.05 / n, 0.002)
            active_safe = np.argsort(b)[::-1][:min(30, m_full)]
            q = lp_solve(best, 0.02, U_safe, U_safe, 7.0, active_safe)
            if q is not None:
                r, sc = try_direction(best, q, best_score, [0.02, 0.01, 0.005])
                if r is not None and sc < best_score - 1e-12:
                    best = r
                    best_score = sc
                    improved = True
                    beta = 0.05
                    continue

        # Full safe LP, rarely.
        if not improved and time.monotonic() < deadline - 15.0:
            U_safe = max(1.05 / n, 0.002)
            q = lp_solve(best, 0.01, U_safe, U_safe, 12.0, None)
            if q is not None:
                r, sc = try_direction(best, q, best_score, [0.01, 0.005])
                if r is not None and sc < best_score - 1e-12:
                    best = r
                    best_score = sc
                    improved = True
                    continue

        if not improved:
            no_improve += 1
            beta = max(0.025, beta * 0.85)

    return best, best_score


def coord_x_score(x, base, c_arr, s_arr, quad_c, S0, x0, n):
    """Exact one-coordinate normalized objective value."""
    s = S0 - x0 + x
    if s <= 1e-15:
        return float("inf")
    M = base
    if c_arr.size > 0:
        vals = c_arr + s_arr * x
        mv = float(np.max(vals))
        if mv > M:
            M = mv
    if quad_c is not None:
        qv = quad_c + x * x
        if qv > M:
            M = qv
    if M < 0.0:
        M = 0.0
    return (2.0 * n * M) / (s * s)


def run_coordinate_descent(p_init, deadline, seed=0):
    """
    Exact randomized coordinate descent on the scale-invariant unnormalized
    objective 2*n*max(conv(p,p))/(sum(p))^2.
    """
    p = norm_pos(p_init).copy()
    n = len(p)
    S = 1.0
    b = fast_conv(p)

    best = p.copy()
    best_score = 2.0 * n * float(np.max(b))

    rng = np.random.default_rng(seed)
    order = np.arange(n)
    no_improve = 0
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() < deadline:
        sweep += 1
        rng.shuffle(order)
        sweep_improved = False

        for i in order:
            if time.monotonic() > deadline:
                break
            if S <= 1e-12:
                break

            x0 = p[i]
            lo = i
            hi = i + n - 1

            base = 0.0
            if lo > 0:
                mv = float(np.max(b[:lo]))
                if mv > base:
                    base = mv
            if hi + 1 < b.size:
                mv = float(np.max(b[hi + 1:]))
                if mv > base:
                    base = mv

            c_list = []
            s_list = []
            quad_c = None

            for k in range(lo, hi + 1):
                j = k - i
                if j == i:
                    quad_c = b[k] - x0 * x0
                else:
                    c_list.append(b[k] - 2.0 * x0 * p[j])
                    s_list.append(2.0 * p[j])

            c_arr = np.asarray(c_list, dtype=np.float64) if c_list else np.empty(0, dtype=np.float64)
            s_arr = np.asarray(s_list, dtype=np.float64) if s_list else np.empty(0, dtype=np.float64)
            S0 = S

            max_b = float(np.max(b))
            f_cur = (2.0 * n * max_b) / (S * S)
            norm_M = max_b / (S * S)

            hi_x = max(
                2.0 * x0 + 1e-3,
                5.0 / n,
                2.0 * np.sqrt(max(norm_M, 0.0)) + 5.0 / n,
                0.02,
            )
            if x0 > 0.05:
                hi_x = max(hi_x, 3.0 * x0)
            hi_x = min(hi_x, max(0.25, 4.0 * x0 + 0.02))
            if hi_x <= 1e-8:
                hi_x = 10.0 / n

            lo_x = 0.0

            grid_x = np.linspace(0.0, hi_x, 7)
            if hi_x > 0.0 and not np.any(np.abs(grid_x - x0) < 1e-12):
                grid_x = np.concatenate((grid_x, np.array([x0], dtype=np.float64)))

            a = lo_x
            bb = hi_x
            c_x = bb - gr * (bb - a)
            d_x = a + gr * (bb - a)
            fc = coord_x_score(c_x, base, c_arr, s_arr, quad_c, S0, x0, n)
            fd = coord_x_score(d_x, base, c_arr, s_arr, quad_c, S0, x0, n)

            for _ in range(26):
                if fc < fd:
                    bb = d_x
                    d_x = c_x
                    fd = fc
                    c_x = bb - gr * (bb - a)
                    fc = coord_x_score(c_x, base, c_arr, s_arr, quad_c, S0, x0, n)
                else:
                    a = c_x
                    c_x = d_x
                    fc = fd
                    d_x = a + gr * (bb - a)
                    fd = coord_x_score(d_x, base, c_arr, s_arr, quad_c, S0, x0, n)

            if fc <= fd:
                best_x = c_x
                best_f = fc
            else:
                best_x = d_x
                best_f = fd

            cand_x = [0.0, hi_x, x0]
            for gx in grid_x:
                cand_x.append(float(gx))

            for cx in cand_x:
                if not np.isfinite(cx):
                    continue
                if cx < 0.0:
                    cx = 0.0
                if cx > hi_x:
                    cx = hi_x
                f = coord_x_score(cx, base, c_arr, s_arr, quad_c, S0, x0, n)
                if np.isfinite(f) and f < best_f - 1e-15:
                    best_f = f
                    best_x = cx

            tol = max(1e-12, 1e-10 * max(1.0, f_cur))
            if np.isfinite(best_f) and best_f < f_cur - tol:
                dx = best_x - x0
                if abs(dx) > 1e-14:
                    for k in range(lo, hi + 1):
                        j = k - i
                        if j == i:
                            b[k] = b[k] - x0 * x0 + best_x * best_x
                        else:
                            b[k] = b[k] + 2.0 * p[j] * dx
                    p[i] = best_x
                    S = S0 - x0 + best_x
                    sweep_improved = True

        if S <= 1e-12:
            break

        if S > 1.2 or S < 0.8333 or sweep % 5 == 0:
            c = 1.0 / S
            p = p * c
            b = b * (c * c)
            S = 1.0

        b = np.maximum(b, 0.0)
        cur_score = 2.0 * n * float(np.max(b))

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 5

        if no_improve > 20 and sweep > 5:
            break

    return best, best_score


def ls_direction(p, b, k_vals, target):
    """
    Least-squares direction d solving approximately:
      b_k + 2 * D_k d ~= target for selected lags k_vals,
    with zero-mean adjustment.
    """
    p = norm_pos(p)
    n = len(p)
    if len(k_vals) == 0:
        return None

    D = build_deriv_matrix(p, k_vals)
    A = 2.0 * D
    rhs = target - b[k_vals]

    if not np.all(np.isfinite(A)) or not np.all(np.isfinite(rhs)):
        return None

    try:
        d = np.linalg.lstsq(A, rhs, rcond=None)[0]
    except Exception:
        return None

    d = np.asarray(d, dtype=np.float64)
    if not np.all(np.isfinite(d)):
        return None
    d = d - np.mean(d)
    if np.linalg.norm(d) < 1e-15:
        return None
    return d


def run_ls_directions(p_init, deadline):
    """
    Fast local balancing via linearized least-squares on active lags.
    """
    p = norm_pos(p_init).copy()
    best = p.copy()
    best_score = score_from_p(p)
    n = len(p)
    m_full = 2 * n - 1

    K_cycle = [32, 64, 128, 200]
    gap_cycle = [0.002, 0.005, 0.01, 0.02]

    it = 0
    no_improve = 0

    while time.monotonic() < deadline and no_improve < 10:
        b = fast_conv(p)
        M = float(b.max())
        if M <= 0.0:
            break

        K = int(min(K_cycle[it % len(K_cycle)], m_full))
        active = np.argsort(b)[::-1][:K]
        target = M * (1.0 - gap_cycle[it % len(gap_cycle)])
        d = ls_direction(p, b, active, target)

        improved = False
        if d is not None:
            neg = d < -1e-15
            if np.any(neg):
                ratios = p[neg] / (-d[neg])
                tau_max = float(np.min(ratios)) * 0.95
            else:
                tau_max = 0.5
            tau_max = min(0.5, tau_max)
            if not np.isfinite(tau_max) or tau_max <= 0.0:
                tau_max = 0.0

            for mult in [1.0, 0.75, 0.5, 0.25, 0.125, 0.05, 0.01]:
                tau = tau_max * mult
                if tau <= 0.0:
                    continue
                r = p + tau * d
                r = norm_pos(r)
                sc = score_from_p(r)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best = r
                    best_score = sc
                    p = r
                    improved = True
                    break

        if not improved:
            no_improve += 1
        it += 1

    return best, best_score


def smooth_random(n, seed=1):
    """Smoothed random nonnegative sequence."""
    rng = np.random.default_rng(seed)
    x = rng.uniform(0.0, 1.0, n)
    k = max(5, n // 32)
    kernel = np.ones(k, dtype=np.float64) / float(k)
    x = np.convolve(x, kernel, mode="same")
    x = np.maximum(x, 1e-4)
    return x / x.sum()


def phase_deadline(hard_deadline, duration):
    """Return a deadline no later than hard_deadline, after at most duration seconds."""
    t = time.monotonic() + max(0.0, float(duration))
    if t > hard_deadline:
        t = hard_deadline
    return t


def propose_candidate(seed=42, budget_s=1500.0, **kwargs):
    """Search for a length-768 nonnegative sequence minimizing the target objective."""
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 10.0:
        budget = 1500.0

    n = 768
    hard_deadline = time.monotonic() + budget - 25.0
    if time.monotonic() > hard_deadline:
        hard_deadline = time.monotonic() + 10.0

    parent = None
    has_parent = False
    try:
        raw = height_sequence_1
        arr = np.asarray(raw, dtype=np.float64).ravel()
        if arr.size > 0:
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.maximum(arr, 0.0)
            if float(arr.sum()) > 1e-12:
                parent = arr
                has_parent = True
    except Exception:
        parent = None
        has_parent = False

    if parent is None or not has_parent:
        parent = np.ones(n, dtype=np.float64)
        has_parent = False

    parent = resample_to_length(parent, n)
    p = norm_pos(parent)

    best = p.copy()
    best_score = score_from_p(best)
    candidates = [best.copy()]

    def _report(label):
        print(f"{label}: normal_score={best_score:.10f} elapsed={time.monotonic():(time.monotonic() - (hard_deadline - budget + 25.0)):.1f}s", flush=True)

    print(f"start n={n} has_parent={has_parent} score={best_score:.10f}", flush=True)

    # Phase 1: cheap coordinate polish.
    if has_parent:
        dl = phase_deadline(hard_deadline, 45.0)
    else:
        dl = phase_deadline(hard_deadline, 120.0)
    p1, s1 = run_coordinate_descent(best, dl, seed=int(seed) + 1)
    if p1 is not None and np.isfinite(s1) and s1 < best_score - 1e-12:
        best = p1
        best_score = s1
        candidates.append(best.copy())
    _report("P1 coordinate")

    # Phase 2: least-squares active balancing.
    dl = phase_deadline(hard_deadline, 100.0 if has_parent else 160.0)
    p2, s2 = run_ls_directions(best, dl)
    if p2 is not None and np.isfinite(s2) and s2 < best_score - 1e-12:
        best = p2
        best_score = s2
        candidates.append(best.copy())
    _report("P2 LS")

    # Phase 3: mirror descent.
    dl = phase_deadline(hard_deadline, 70.0 if has_parent else 100.0)
    p3, s3 = run_mirror_descent(best, dl)
    if p3 is not None and np.isfinite(s3) and s3 < best_score - 1e-12:
        best = p3
        best_score = s3
        candidates.append(best.copy())
    _report("P3 mirror")

    # Phase 4: LP minimax best response, main engine.
    dl = phase_deadline(hard_deadline, 520.0 if has_parent else 600.0)
    p4, s4 = run_lp_mm(best, dl)
    if p4 is not None and np.isfinite(s4) and s4 < best_score - 1e-12:
        best = p4
        best_score = s4
        candidates.append(best.copy())
    _report("P4 LP")

    # Phase 5: coordinate polish.
    dl = phase_deadline(hard_deadline, 230.0 if has_parent else 180.0)
    p5, s5 = run_coordinate_descent(best, dl, seed=int(seed) + 123)
    if p5 is not None and np.isfinite(s5) and s5 < best_score - 1e-12:
        best = p5
        best_score = s5
        candidates.append(best.copy())
    _report("P5 coordinate")

    # Phase 6: another LS balancing pass.
    if time.monotonic() < hard_deadline - 90.0:
        dl = phase_deadline(hard_deadline, 120.0)
        p6, s6 = run_ls_directions(best, dl)
        if p6 is not None and np.isfinite(s6) and s6 < best_score - 1e-12:
            best = p6
            best_score = s6
            candidates.append(best.copy())
        _report("P6 LS")

    # Phase 7: final mirror and LP short passes.
    if time.monotonic() < hard_deadline - 110.0:
        dl = phase_deadline(hard_deadline, 70.0)
        p7, s7 = run_mirror_descent(best, dl)
        if p7 is not None and np.isfinite(s7) and s7 < best_score - 1e-12:
            best = p7
            best_score = s7
            candidates.append(best.copy())
        _report("P7 mirror")

    if time.monotonic() < hard_deadline - 60.0:
        dl = phase_deadline(hard_deadline, 130.0 if has_parent else 170.0)
        p8, s8 = run_lp_mm(best, dl)
        if p8 is not None and np.isfinite(s8) and s8 < best_score - 1e-12:
            best = p8
            best_score = s8
            candidates.append(best.copy())
        _report("P8 LP")

    # Optional escape from a local basin if still above target and time remains.
    if time.monotonic() < hard_deadline - 90.0 and best_score > 1.5027 + 0.0003:
        alt = norm_pos(0.75 * best + 0.25 * smooth_random(n, int(seed) + 999))
        dl = phase_deadline(hard_deadline, 80.0)
        pa, sa = run_lp_mm(alt, dl)
        if pa is not None and np.isfinite(sa) and sa < best_score - 1e-12:
            best = pa
            best_score = sa
            candidates.append(best.copy())
            _report("Alt LP")

        if time.monotonic() < hard_deadline - 40.0:
            dl2 = phase_deadline(hard_deadline, 50.0)
            pc, sc = run_coordinate_descent(best, dl2, seed=int(seed) + 3141)
            if pc is not None and np.isfinite(sc) and sc < best_score - 1e-12:
                best = pc
                best_score = sc
                candidates.append(best.copy())
                _report("Alt coordinate")

    # Final coordinate polish with remaining time.
    if time.monotonic() < hard_deadline - 5.0:
        dl = phase_deadline(hard_deadline, max(5.0, (hard_deadline - time.monotonic()) * 0.45))
        p9, s9 = run_coordinate_descent(best, dl, seed=int(seed) + 777)
        if p9 is not None and np.isfinite(s9) and s9 < best_score - 1e-12:
            best = p9
            best_score = s9
            candidates.append(best.copy())
        _report("P9 coordinate")

    # Select best by exact evaluator when possible.
    final_candidates = candidates[-12:]
    final_candidates.append(best)

    final_best = best.copy()
    final_score = float("inf")

    for cand in final_candidates:
        if time.monotonic() > hard_deadline - 3.0:
            break
        cand = np.clip(norm_pos(cand), 0.0, 1000.0)
        try:
            sc = float(evaluate_sequence([float(x) for x in cand]))
        except Exception:
            sc = score_from_p(cand)
        if np.isfinite(sc) and sc < final_score:
            final_score = sc
            final_best = cand

    if not np.isfinite(final_score):
        final_score = best_score
        final_best = norm_pos(best)

    final_best = np.clip(final_best, 0.0, 1000.0)
    final_best = norm_pos(final_best)

    try:
        exact_final = float(evaluate_sequence([float(x) for x in final_best]))
        if np.isfinite(exact_final):
            print(f"FINAL exact score={exact_final:.10f} n={len(final_best)}", flush=True)
        else:
            print(f"FINAL normal score={score_from_p(final_best):.10f}", flush=True)
    except Exception:
        print(f"FINAL normal score={score_from_p(final_best):.10f}", flush=True)

    return [float(x) for x in final_best]
```
