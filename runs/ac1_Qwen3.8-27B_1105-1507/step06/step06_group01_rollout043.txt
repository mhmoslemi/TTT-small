def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not np.any(cond):
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not np.any(~cond):
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def build_full_lp_matrix(a):
    """Build the (2n-1) x n constraint matrix for the full LP.
    A[k,j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    # The matrix is banded: for each k, the non-zero entries are at j where 0 <= k-j < n
    # i.e., max(0, k-n+1) <= j <= min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_lp_oracle(a, M, time_limit=5.0):
    """Solve the LP: max sum(g) s.t. A@g <= M, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    A = build_full_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

            best_a = a.copy()
            no_improve = 0

            g[i_min] = 1.0

    # Final check
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n <= 1:
        return np.maximum(v, 0.0)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv
    if not np.any(cond):
        theta = 0.0
    else:
        rho_val = int(np.argmax(~cond))
        if not np.any(~cond):
            rho_val = n - 1
        theta = (cv[rho_val] - 1.0) / (rho_val + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

    """Precompute row/col structure for the LP constraint matrix."""
    rows = []

def solve_lp_oracle(a, M, lp_rows, lp_cols, time_limit=3.0):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0.
    
    A[k,j] = a[k-j] (encoded via lp_rows, lp_cols).
    Returns normalized g or None.
    """
    n = len(a)
    vals = a[lp_rows - lp_cols]
    A = csr_matrix((vals, (lp_rows, lp_cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',

                             'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def run_fpw(a_init, deadline, lp_rows, lp_cols, max_iters=100000):
    """Full-constraint LP Frank-Wolfe with standard step size."""
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
        
        # Solve LP oracle
        g = solve_lp_oracle(a, M, lp_rows, lp_cols)
        if g is None:
            # Fallback: unit vector at min subgradient coordinate
            k_star = int(np.argmax(b))
            v = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    v[i] = 2.0 * a[j]
            i_min = int(np.argmin(v))
            g = np.zeros(n)
            g[i_min] = 1.0
        
        # Standard FW step size
        t = 2.0 / (it + 2)
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a

def cd_one_sweep(a, n, order=None):
    """One coordinate descent sweep. Returns (a, b) updated."""
    b = fast_conv(a)
    if order is None:
        order = np.arange(n)
    
    for i in order:
        x_cur = a[i]
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        nk = k_hi - k_lo + 1
        
        # Compute base and slope for each affected k
        # b[k] = base[k] + slope[k] * x + (x^2 if k == 2*i)
        base = np.zeros(nk)
        slope = np.zeros(nk)
        quad_idx = -1
        
        for idx in range(nk):
            k = k_lo + idx
            if k == 2 * i:
                base[idx] = b[k] - x_cur * x_cur
                slope[idx] = 0.0
                quad_idx = idx
            else:
                j = k - i
                if 0 <= j < n:
                    base[idx] = b[k] - 2.0 * x_cur * a[j]
                    slope[idx] = 2.0 * a[j]
                else:
                    base[idx] = b[k]
                    slope[idx] = 0.0
        
        # Unaffected max
        base_max = 0.0
        if k_lo > 0:
            base_max = max(base_max, b[:k_lo].max())
        if k_hi < 2 * n - 2:
            base_max = max(base_max, b[k_hi + 1:].max())
        
        # Golden section search for optimal x >= 0
        # f(x) = max(base_max, max(base + slope*x), quad_term)
        lo = 0.0
        hi = max(2.0 * x_cur + 0.01, 1.0 / n)
        
        # Ensure f(hi) > f(0) or we're at boundary
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        ga = hi - gr * (hi - lo)
        gb = lo + gr * (hi - lo)
        
        fa = _cd_obj(ga, base, slope, base_max, quad_idx)
        fb = _cd_obj(gb, base, slope, base_max, quad_idx)
        
        for _ in range(50):
            if fa < fb:
                hi = gb
                gb = ga
                fb = fa
                ga = hi - gr * (hi - lo)
                fa = _cd_obj(ga, base, slope, base_max, quad_idx)
            else:
                lo = ga
                ga = gb
                fa = fb
                gb = lo + gr * (hi - lo)
                fb = _cd_obj(gb, base, slope, base_max, quad_idx)
        
        x_opt = (lo + hi) / 2.0
        # Check boundary x=0
        f0 = _cd_obj(0.0, base, slope, base_max, quad_idx)
        fopt = _cd_obj(x_opt, base, slope, base_max, quad_idx)
        if f0 <= fopt:
            x_opt = 0.0
        
        # Update a[i] and b
        dx = x_opt - x_cur
        if abs(dx) > 1e-12:
            a[i] = x_opt
            for idx in range(nk):
                k = k_lo + idx
                if k == 2 * i:
                    b[k] = base[idx] + x_opt * x_opt
                else:
                    b[k] = base[idx] + slope[idx] * x_opt
    
    return a, b


def _cd_obj(x, base, slope, base_max, quad_idx):
    """Objective for coordinate descent at value x."""
    m = base_max
    if len(base) > 0:
        vals_m = (base + slope * x).max()
        if vals_m > m:
            m = vals_m
    if quad_idx >= 0:
        qv = base[quad_idx] + x * x
        if qv > m:
            m = qv
    return m

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)

    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    if len(base) > 0:
        vm = (base + slope * x).max()
        if vm > m:
            m = vm
    if quad_idx >= 0:
        qv = base[quad_idx] + x * x
        if qv > m:
            m = qv
    return m


def cd_one_sweep(a, b, n, order):
    """One coordinate descent sweep. Updates a and b in place."""
    for i in order:
        x_cur = a[i]
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        nk = k_hi - k_lo + 1

        base = np.zeros(nk)
        slope = np.zeros(nk)
        quad_idx = -1

        for idx in range(nk):
            k = k_lo + idx
            if k == 2 * i:
                base[idx] = b[k] - x_cur * x_cur
                slope[idx] = 0.0
                quad_idx = idx
            else:
                j = k - i
                if 0 <= j < n:
                    base[idx] = b[k] - 2.0 * x_cur * a[j]
                    slope[idx] = 2.0 * a[j]
                else:
                    base[idx] = b[k]
                    slope[idx] = 0.0

        base_max = 0.0
        if k_lo > 0:
            cm = b[:k_lo].max()
            if cm > base_max:
                base_max = cm
        if k_hi < 2 * n - 2:
            cm = b[k_hi + 1:].max()
            if cm > base_max:
                base_max = cm

        # Golden section search
        lo = 0.0
        hi = max(2.0 * x_cur + 0.01, 1.0 / n)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        ga = hi - gr * (hi - lo)
        gb = lo + gr * (hi - lo)
        fa = _cd_eval(ga, base, slope, base_max, quad_idx)
        fb = _cd_eval(gb, base, slope, base_max, quad_idx)

        for _ in range(45):
            if fa < fb:
                hi = gb
                gb = ga
                fb = fa
                ga = hi - gr * (hi - lo)
                fa = _cd_eval(ga, base, slope, base_max, quad_idx)
            else:
                lo = ga
                ga = gb
                fa = fb
                gb = lo + gr * (hi - lo)
                fb = _cd_eval(gb, base, slope, base_max, quad_idx)

        x_opt = (lo + hi) / 2.0
        f0 = _cd_eval(0.0, base, slope, base_max, quad_idx)
        fopt = _cd_eval(x_opt, base, slope, base_max, quad_idx)
        if f0 <= fopt:
            x_opt = 0.0

        dx = x_opt - x_cur
        if abs(dx) > 1e-13:
            a[i] = x_opt
            for idx in range(nk):
                k = k_lo + idx
                if k == 2 * i:
                    b[k] = base[idx] + x_opt * x_opt
                else:
                    b[k] = base[idx] + slope[idx] * x_opt

    return a, b


def run_cd_polish(a_init, deadline, max_sweeps=50, seed=0):
    """Run coordinate descent for multiple sweeps."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = np.arange(n)
        rng.shuffle(order)
        b = fast_conv(a)
        a, b = cd_one_sweep(a, b, n, order)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        M = b.max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()

    return best_a


def run_fpw_with_cd(a_init, deadline, lp_rows, lp_cols, cd_every=50):
    """LP Frank-Wolfe with periodic coordinate descent."""
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    rng = np.random.default_rng(12345)

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        # Periodic CD interlude
        if it > 0 and it % cd_every == 0:
            a_cd, _ = cd_one_sweep(a, fast_conv(a), n, np.arange(n))
            s = a_cd.sum()
            if s > 1e-15:
                a_cd = a_cd / s
                M_cd = fast_conv(a_cd).max()
                if M_cd < best_M - 1e-14:
                    best_M = M_cd
                    best_a = a_cd.copy()
                    a = a_cd
                    no_improve = 0

        b = fast_conv(a)
        M = b.max()

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        g = solve_lp_oracle(a, M, lp_rows, lp_cols)
        if g is None:
            k_star = int(np.argmax(b))
            v = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    v[i] = 2.0 * a[j]
            i_min = int(np.argmin(v))
            g = np.zeros(n)
            g[i_min] = 1.0

        # Adaptive step: start large, decrease
        t = min(0.8, 2.0 / (it + 2) * 5.0)

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gen_fourier_init(n, rng):
    """Generate initialization with controlled spectral content."""
    freqs = np.fft.rfftfreq(n)
    for width in [3.0, 7.0, 15.0]:
        amp = np.exp(-freqs * n / width)
        amp[0] = 1.0
        phase = rng.uniform(0, 2 * np.pi, len(amp))
        spec = amp * np.exp(1j * phase)
        a = np.real(np.fft.irfft(spec * n, n))
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            return a
    return np.ones(n) / n


def gen_random_smooth(n, rng):
    """Generate a random smooth non-negative sequence."""
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    a = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def gen_perturbed_parent(parent, n, rng, sigma=0.05):
    """Generate perturbation of parent resampled to length n."""
    p = resample_seq(parent, n)
    a = p + rng.normal(0, sigma, n)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best_score = score
            self.best = a.copy()
            elapsed = time.monotonic() - self.t_start
            print("  ** NEW BEST: J={:.8f} n={} t={:.0f}s".format(
                score, a.size, elapsed), flush=True)

    def report(self, label=""):
        if self.best is not None:
            print("  {}: J={:.8f} n={}".format(label, self.best_score, self.best.size),
                  flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Full-constraint LP Frank-Wolfe from parent + CD interludes
    Phase 2: CD polish
    Phase 3: Multi-length exploration
    Phase 4: Random restarts with diverse inits
    Phase 5: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            s = parent.sum()
            if s > 0.01:
                tracker.update(parent)
            print("Parent loaded: n={}, J={:.8f}".format(parent.size, tracker.best_score),
                  flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print("No parent, using uniform n=512", flush=True)

    n_parent = parent.size

    # Precompute LP structure for parent length
    lp_rows, lp_cols = precompute_lp_structure(n_parent)
    print("LP structure for n={}: {} nonzeros".format(n_parent, len(lp_rows)), flush=True)

    # ===== PHASE 1: LP-FW from parent =====
    print("\n" + "=" * 60, flush=True)
    print("PHASE 1: LP Frank-Wolfe (full constraints) + CD", flush=True)
    print("=" * 60, flush=True)

    t1_deadline = time.monotonic() + 700
    a0 = parent.copy()
    s0 = a0.sum()
    if s0 > 0.01:
        a0 = a0 / s0

    a_fpw = run_fpw_with_cd(a0, t1_deadline, lp_rows, lp_cols, cd_every=40)
    tracker.update(a_fpw)
    tracker.report("After Phase 1")

    # ===== PHASE 2: CD polish =====
    print("\n" + "=" * 60, flush=True)
    print("PHASE 2: Coordinate descent polish", flush=True)
    print("=" * 60, flush=True)

    t2_deadline = min(deadline - 400, time.monotonic() + 150)
    a_cd = run_cd_polish(tracker.best, t2_deadline, max_sweeps=100, seed=99)
    tracker.update(a_cd)
    tracker.report("After Phase 2")

    # Second CD pass from slightly perturbed
    if time.monotonic() < deadline - 350:
        a_pert = tracker.best + rng.normal(0, 0.005, tracker.best.size)
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()
        t2b_deadline = min(deadline - 350, time.monotonic() + 80)
        a_cd2 = run_cd_polish(a_pert, t2b_deadline, max_sweeps=50, seed=42)
        tracker.update(a_cd2)
        tracker.report("After Phase 2b")

    # ===== PHASE 3: Multi-length exploration =====
    print("\n" + "=" * 60, flush=True)
    print("PHASE 3: Multi-length exploration", flush=True)
    print("=" * 60, flush=True)

    lengths = []
    for delta in [-128, -64, 64, 128, 256, -256]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl not in lengths:
            lengths.append(nl)

    # Also try a few fixed lengths
    for fixed in [512, 640, 896, 1024]:
        if fixed not in lengths and 256 <= fixed <= 2048:
            lengths.append(fixed)

    lengths = lengths[:6]
    print("  Trying lengths: {}".format(lengths), flush=True)

    time_per_len = max(40, (deadline - time.monotonic() - 200) / max(1, len(lengths)))

    for nl in lengths:
        if time.monotonic() > deadline - 180:
            break
        sub_dl = min(deadline - 180, time.monotonic() + time_per_len)
        a_resample = resample_seq(tracker.best, nl)
        a_resample = np.maximum(a_resample, 1e-6)
        a_resample = a_resample / a_resample.sum()

        # Quick LP-FW + CD
        lp_r, lp_c = precompute_lp_structure(nl)
        a_opt = run_fpw_with_cd(a_resample, sub_dl, lp_r, lp_c, cd_every=30)
        tracker.update(a_opt)

        # Quick CD
        if time.monotonic() < deadline - 180:
            a_cd_try = run_cd_polish(a_opt, min(deadline - 180, time.monotonic() + 30),
                                     max_sweeps=30, seed=77)
            tracker.update(a_cd_try)

    tracker.report("After Phase 3")

    # ===== PHASE 4: Random restarts =====
    print("\n" + "=" * 60, flush=True)
    print("PHASE 4: Random restarts", flush=True)
    print("=" * 60, flush=True)

    t4_deadline = min(deadline - 60, time.monotonic() + 150)
    n_best = tracker.best.size

    for restart_i in range(10):
        if time.monotonic() > t4_deadline:
            break
        sub_dl = min(t4_deadline, time.monotonic() + 12)

        if restart_i % 3 == 0:
            a_init = gen_fourier_init(n_best, rng)
        elif restart_i % 3 == 1:
            a_init = gen_random_smooth(n_best, rng)
        else:
            a_init = gen_perturbed_parent(tracker.best, n_best, rng, sigma=0.1)

        # Quick LP-FW
        a_opt = run_fpw_with_cd(a_init, sub_dl, lp_rows, lp_cols, cd_every=20)
        tracker.update(a_opt)

    tracker.report("After Phase 4")

    # ===== PHASE 5: Final intensive polish =====
    print("\n" + "=" * 60, flush=True)
    print("PHASE 5: Final polish", flush=True)
    print("=" * 60, flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 20:
        # Long CD
        t5a = min(deadline - 20, time.monotonic() + 60)
        a_final = run_cd_polish(tracker.best, t5a, max_sweeps=200, seed=31337)
        tracker.update(a_final)

        # One more LP-FW round
        if time.monotonic() < deadline - 15:
            t5b = min(deadline - 15, time.monotonic() + 50)
            a_fpw2 = run_fpw_with_cd(tracker.best, t5b, lp_rows, lp_cols, cd_every=30)
            tracker.update(a_fpw2)

        # Final CD
        if time.monotonic() < deadline - 10:
            a_final2 = run_cd_polish(tracker.best, deadline - 5, max_sweeps=100, seed=7)
            tracker.update(a_final2)

    tracker.report("After Phase 5")

    # Ensure valid return
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print("\n" + "=" * 60, flush=True)
    print("FINAL: J={:.8f} n={} time={:.0f}s".format(
        tracker.best_score, tracker.best.size,
        time.monotonic() - tracker.t_start), flush=True)
    print("=" * 60, flush=True)

    return [float(x) for x in tracker.best]
