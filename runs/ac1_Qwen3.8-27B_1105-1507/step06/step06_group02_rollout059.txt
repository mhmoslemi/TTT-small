d(a*a)(k)/da_i = 2*a[k-i] (with a[j]=0 for j notin [0,n-1])

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def gs_init(n, rng, n_iters=200, beta=0.85):
    """Gerchberg-Saxton for flat autocorrelation initialization."""
    a = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    a /= a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_compressed = (mag ** 0.5) / ((mag ** 0.5).max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_compressed
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0.001)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, b, b_max


def nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=150.0,
                    lr_init=0.3, lr_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.85
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = np.maximum(y, 0.0) / sy
        a = a_new
    return best_a


def build_conv_matrix(a, n):
    """Build sparse matrix A where (A*g)[k] = (a*g)[k] for linear convolution."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M, n):
    """Solve: max sum(g) s.t. A(a)*g <= M, g >= 0. Return (g, dual_vars)."""
    A = build_conv_matrix(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True,
                             'dual_feasibility_tolerance': 1e-12,
                             'primal_feasibility_tolerance': 1e-12,
                             'time_limit': 10})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            dual = r.ineqlin.marginals if r.ineqlin is not None else np.zeros(2*n-1)
            dual = np.abs(dual)  # dual variables are non-negative in the max formulation
            return g, dual
    except Exception:
        pass
    return None, None


def newton_refine(a, active_set, max_iters=50, tol=1e-14):
    """
    Newton's method to solve:
      (a*a)(k) = M for k in active_set
      sum(a) = 1
      a >= 0
    Unknowns: (a_0, ..., a_{n-1}, M)
    """
    a = a.astype(np.float64).copy()
    n = len(a)
    a /= a.sum()
    active = sorted(active_set)
    m = len(active)
    
    # Build the system: F(x) = 0 where x = (a, M)
    # F_i = (a*a)(active[i]) - M for i in range(m)
    # F_m = sum(a) - 1
    # Total: m+1 equations, n+1 unknowns
    
    if m == 0:
        return a
    
    # If m > n-1, we have more equations than unknowns (minus M)
    # Use only the most important (largest gradient) constraints
    if m > n:
        # Rank by how close to the max they are
        b = fast_conv(a)
        b_max = b.max()
        scores = [b[k] for k in active]
        idx = np.argsort(-np.array(scores))[:n]
        active = [active[i] for i in idx]
        m = len(active)
    
    for it in range(max_iters):
        b = fast_conv(a)
        M = b[active].max() if m > 0 else b.max()
        
        # Residuals
        F = np.zeros(m + 1)
        for i, k in enumerate(active):
            F[i] = b[k] - M
        F[m] = a.sum() - 1.0
        
        if np.max(np.abs(F)) < tol:
            break
        
        # Jacobian: (m+1) x (n+1)
        J = np.zeros((m + 1, n + 1))
        for i, k in enumerate(active):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    J[i, j] = 2.0 * a[idx]
            J[i, n] = -1.0  # d/dM of ((a*a)(k) - M) = -1
        J[m, :n] = 1.0  # d/d(a_i) of sum(a) = 1
        J[m, n] = 0.0
        
        # Solve J * dx = -F
        try:
            # Use least squares if overdetermined
            if m + 1 > n + 1:
                dx, _, _, _ = np.linalg.lstsq(J, -F, rcond=None)
            else:
                dx = np.linalg.solve(J, -F)
        except np.linalg.LinAlgError:
            try:
                dx, _, _, _ = np.linalg.lstsq(J, -F, rcond=1e-12)
            except:
                break
        
        # Damping: ensure a stays non-negative
        a_new = a + dx[:n]
        M_new = a[-1] + dx[n] if False else M + dx[n]
        
        # Find max step that keeps a >= 0
        step = 1.0
        for _ in range(30):
            a_try = a + step * dx[:n]
            if np.min(a_try) >= -1e-15:
                break
            # Reduce step
            min_idx = np.argmin(a + step * dx[:n])
            if dx[:n][min_idx] < 0:
                step = 0.9 * step * a[min_idx] / (-dx[:n][min_idx])
            else:
                step *= 0.5
            if step < 1e-10:
                break
        
        a_try = np.maximum(a + step * dx[:n], 0.0)
        s = a_try.sum()
        if s < 1e-10:
            break
        a_try /= s
        
        # Verify improvement
        b_try = fast_conv(a_try)
        M_try = b_try[active].max() if m > 0 else b_try.max()
        F_try = max(abs(b_try[k] - M_try) for k in active) if m > 0 else 0
        
        if F_try < np.max(np.abs(F)) * 0.5 or step < 1e-6:
            a = a_try
        else:
            # Try smaller step
            a_try = np.maximum(a + 0.1 * step * dx[:n], 0.0)
            s = a_try.sum()
            if s > 1e-10:
                a_try /= s
                a = a_try
            else:
                break
    
    return a


def lp_mirror_descent(a, deadline, max_iters=200, t_init=0.3, t_decay=0.95):
    """LP mirror descent with active set tracking and Newton refinement."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t = t_init
    no_imp = 0
    active_set_stable = 0
    last_active = set()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        
        # Solve LP
        g, dual = solve_lp(a, M, n)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Identify active set from dual
        threshold = 1e-6 * (dual.max() + 1e-30) if dual.max() > 0 else 1e-8
        current_active = set(np.where(dual > threshold)[0].tolist())
        
        # Check active set stability
        if current_active == last_active:
            active_set_stable += 1
        else:
            active_set_stable = 0
        last_active = current_active
        
        # Update a
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        # If active set is stable, try Newton refinement
        if active_set_stable >= 3 and len(current_active) > 0:
            a_newton = newton_refine(a_new, current_active, max_iters=30, tol=1e-13)
            if a_newton is not None:
                b_n = fast_conv(a_newton)
                # Check all constraints are satisfied
                if b_n.max() <= M * 1.001:
                    M_n = b_n.max()
                    if M_n < M:
                        a_new = a_newton
                    else:
                        # Newton didn't improve; check if it's at the fixed point
                        if abs(M_n - M) < 1e-10 * M:
                            t *= t_decay
        
        # Backtracking: verify improvement on actual objective
        M_new = fast_conv(a_new).max()
        if M_new > M * 1.01:
            # Try smaller step
            for t_try in [t * 0.5, t * 0.25, t * 0.1]:
                a_try = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-10:
                    a_try /= s_try
                    M_try = fast_conv(a_try).max()
                    if M_try <= M * 1.001:
                        a_new = a_try
                        t = t_try
                        break
            else:
                a_new = a.copy()  # no improvement
                t *= 0.5
        else:
            if M_new < M:
                t = min(0.5, t * 1.1)  # increase step slightly
        
        a = a_new
        t = max(0.01, t * t_decay)
    
    return best_a

                    break
            else:

        # Update with line search
        improved = False

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent):
    """Generate diverse initializations emphasizing asymmetric structures."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.005))
        # Asymmetric perturbation
        asym = np.linspace(-0.1, 0.1, n)
        inits.append(np.maximum(p * (1 + asym), 0.005))
        inits.append(np.maximum(p * (1 + 2*asym), 0.005))
    
    # Asymmetric power law (key for J < 2)
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = (t + 0.1) ** alpha
        inits.append(a / a.sum())
    
    # Exponential decay (strongly asymmetric)
    for rate in [1, 2, 3, 5]:
        a = np.exp(-rate * t) + 0.01
        inits.append(a / a.sum())
    
    # Bump functions (asymmetric)
    for center in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for width in [0.1, 0.2, 0.3]:
            a = np.exp(-((t - center) / width) ** 2) + 0.01
            inits.append(a / a.sum())
    
    # Linear ramp
    inits.append(np.maximum(t + 0.05, 0.01))
    inits.append(np.maximum(1 - t + 0.05, 0.01))
    inits.append(np.maximum(2*t - 0.5 + 0.1, 0.01))
    
    # Quadratic (asymmetric)
    inits.append(np.maximum((t - 0.3)**2 + 0.05, 0.01))
    inits.append(np.maximum((t - 0.7)**2 + 0.05, 0.01))
    
    # GS from random
    inits.append(gs_init(n, rng, n_iters=80, beta=0.85))
    
    # Random smooth
    k = max(3, n // 30)
    kern = np.ones(k) / k
    a_rand = np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01)
    inits.append(a_rand / a_rand.sum())
    
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):

    parent = get_parent()

        best.update(parent)
        print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    else:

        best.update(parent)

    print(f"PHASE 1: Multi-length Smooth Initialization", flush=True)
    print(f"{'='*60}", flush=True)
    p1_start = time.monotonic()
    p1_time = 200.0
    
    lengths = [512, 768, 1024, 1280, 1536, 1792, 2048]
    # If parent exists, include its length
    if parent is not None and parent.size not in lengths:
        lengths.append(parent.size)
    lengths = sorted(set(l for l in lengths if 128 <= l <= 2560))
    
    best_per_length = {}
    time_per_len = p1_time / len(lengths)
    
    for n in lengths:
        t0 = time.monotonic()
        remaining = deadline - 600 - time.monotonic()
        if remaining < 20:
            break
        allot = min(time_per_len, remaining * 0.4)
        dl = time.monotonic() + allot
        if time.monotonic() >= dl:
            continue
        
        inits = make_diverse_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')
        
        # Time budget per init
        n_inits = min(len(inits), max(3, int(al)))
        time_per_init = allot / n_inits
        
        for idx in range(n_inits):
            if time.monotonic() > dl:
                break
            a0 = inits[idx]
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            
            # Quick Nesterov
            sub_dl = time.monotonic() + time_per_init
            a = nesterov_smooth(a0, sub_dl, max_iters=max(100, int(3000 * 256.0 / n)),
                               tau_start=1.0, tau_end=100.0,
                               lr_init=0.25, lr_end=0.005)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()
        
        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:5d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()
    
    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)
    
    # === PHASE 2: LP mirror descent + Newton on best lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent + Newton Refinement", flush=True)
    print(f"{'='*60}", flush=True)
    p2_start = time.monotonic()
    
    # Sort lengths by current J
    sorted_lens = sorted(best_per_length.keys(), key=lambda x: compute_J(best_per_length[x]))
    
    # Allocate time: more to the best length
    p2_budget = min(600.0, deadline - 300 - time.monotonic())
    if p2_budget < 30:
        p2_budget = 30
    
    for rank, n in enumerate(sorted_lens[:4]):
        t0 = time.monotonic()
        remaining = deadline - 150 - time.monotonic()
        if remaining < 15:
            break
        # Allocate more time to better lengths
        frac = [0.5, 0.25, 0.15, 0.10][min(rank, 3)]
        allot = p2_budget * frac
        dl = time.monotonic() + allot
        
        a_start = best_per_length[n].copy()
        
        # LP mirror descent
        a = lp_mirror_with_newton_polish(a_start, min(dl, deadline - 100), max_iters=150)
        M = fast_conv(a).max()
        J = compute_J(a)
        print(f"  n={n:5d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
        best.update(a)
        best_per_length[n] = a.copy()
    
    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)
    
    # === PHASE 3: Intensive Newton refinement on the best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Intensive Newton Refinement", flush=True)
    print(f"{'='*60}", flush=True)
    p3_start = time.monotonic()
    
    if best.seq is not None and time.monotonic() < deadline - 50:
        n_best = best.seq.size

        for round_num in range(10):

            b = fast_conv(a)
            M = b.max()
            # Active set: lags where b[k] >= M * 0.999
            active = set(np.where(b >= M * 0.999)[0].tolist())
            
            # Also include near-active
            active_near = set(np.where(b >= M * 0.99)[0].tolist())
            
            # Newton with full active set
            a_new = newton_refine(a, active, max_iters=50, tol=1e-15)
            if a_new is not None:
                M_new = fast_conv(a_new).max()
                if M_new < M * (1 - 1e-12):
                    a = a_new
                    print(f"    Newton round {round_num}: M={M:.12f} -> {M_new:.12f}", flush=True)
                    best.update(a)
                else:
                    # Try with near-active set
                    a_new2 = newton_refine(a, active_near, max_iters=30, tol=1e-15)
                    if a_new2 is not None:
                        M_new2 = fast_conv(a_new2).max()
                        if M_new2 < M * (1 - 1e-12):
                            a = a_new2
                            best.update(a)
                        else:
                            break
                    else:
                        break
            else:
                break
        
        # One more LP pass
        if time.monotonic() < deadline - 20:
            a_final = lp_mirror_descent(a, deadline - 10, max_iters=50, t_init=0.15, t_decay=0.93)
            best.update(a_final)
    
    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)
    
    # === PHASE 4: Length exploration around best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)
    p4_start = time.monotonic()
    
    if best.seq is not None and time.monotonic() < deadline - 30:
        n_best = best.seq.size
        deltas = [32, -32, 64, -64, 128, -128, 256, -256, 512, -512]
        
        for delta in deltas:
            if time.monotonic() > deadline - 15:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 3000:
                continue
            
            a_try = resample_seq(best.seq, n_try)
            a_try = np.maximum(a_try, 1e-6)
            a_try /= a_try.sum()
            
            # Quick smooth
            a_try = nesterov_smooth(a_try, min(time.monotonic() + 20, deadline - 20),
                                    max_iters=500, tau_start=3, tau_end=80,
                                    lr_init=0.15, lr_end=0.01)
            best.update(a_try)
            
            # LP polish
            if time.monotonic() < deadline - 15:
                a_lp = lp_mirror_with_newton_polish(a_try, deadline - 10, max_iters=30)
                best.update(a_lp)
    
    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)
    
    # === PHASE 5: Final aggressive Newton on the global best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)
    p5_start = time.monotonic()
    
    if best.seq is not None and time.monotonic() < deadline - 5:
        a = best.seq.copy()
        n = a.size
        
        # Repeated Newton with progressively tighter active sets
        for tightness in [0.999, 0.9995, 0.9999, 0.99999, 1.0]:
            if time.monotonic() > deadline - 3:
                break
            b = fast_conv(a)
            M = b.max()
            active = set(np.where(b >= M * tightness)[0].tolist())
            if len(active) == 0:
                active = {int(np.argmax(b))}
            a_new = newton_refine(a, active, max_iters=100, tol=1e-16)
            if a_new is not None:
                M_new = fast_conv(a_new).max()
                if M_new <= M * (1 + 1e-14):
                    a = a_new
                    best.update(a)
        
        # Final LP verification
        if time.monotonic() < deadline - 2:
            b = fast_conv(a)
            M = b.max()
            g, dual = solve_lp(a, M, n)
            if g is not None:
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    # Very small step
                    a_tiny = np.maximum(0.999 * a + 0.001 * g, 0.0)
                    a_tiny /= a_tiny.sum()
                    best.update(a_tiny)
    
    best.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)
    
    # Fallback
    if best.seq is None:
        best.seq = np.ones(256)
        best.score = compute_J(best.seq)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best.seq]
