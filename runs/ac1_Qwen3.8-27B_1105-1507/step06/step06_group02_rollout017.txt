        # Compress magnitude
        beta = beta_schedule[it] if beta_schedule else 0.85
        mag_norm = mag / (mag.max() + 1e-30)
        mag_target = (1 - beta) * mag + beta * mag.mean()
        F_new = mag_target * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        M = fast_conv(a_new).max()
        if M < best_M:
            best_M = M
            best_a = a_new.copy()
        a = a_new
    return best_a

def smooth_grad(x, tau, n):
    """Compute gradient of logsumexp(tau * conv(exp(x)/sum(exp(x)), ...)) w.r.t. x."""
    # Compute a from x
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    
    b = fast_conv(a)
    b_max = b.max()
    e = np.exp(tau * (b - b_max))
    p = e / e.sum()
    
    # Gradient in a-space: grad_a[j] = 2*tau * sum_k p[k] * a[k-j]
    # = 2*tau * conv(a, p[::-1])[2n-2-j]
    p_rev = p[::-1]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    conv_ap = np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(p_rev, size), size)[:n + len(p) - 1]
    grad_a = 2.0 * tau * conv_ap[n-1 : 2*n-1][::-1]
    
    # Transform to x-space
    grad_x = a * (grad_a - np.dot(grad_a, a))
    
    # Objective value
    val = b_max + np.log(e.sum()) / tau  # logsumexp(tau * b)
    # Actually: logsumexp(tau * b) = tau * b_max + log(sum(exp(tau*(b-b_max))))
    val = tau * b_max + np.log(e.sum())
    
    return val, grad_x

def run_smooth_lbfgs(a_init, tau, maxiter=200, deadline=None):
    """Run L-BFGS on the exp-parameterized smooth objective."""
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / a_init.sum()
    x0 = np.log(a_init)  # x such that exp(x)/sum(exp(x)) = a_init (approximately)
    
    def obj_and_grad(x):
        if deadline is not None and time.monotonic() > deadline:
            return compute_J(a_init), np.zeros(len(x))
        return smooth_grad(x, tau, len(a_init))
    
    try:
        res = minimize(obj_and_grad, x0, method='L-BFGS-B',
                      jac=True, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        x_opt = res.x
        a_opt = np.exp(x_opt - x_opt.max())
        a_opt = a_opt / a_opt.sum()
        a_opt = np.maximum(a_opt, 0)
        return a_opt
    except Exception:
        return a_init

def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=500.0):
    """Nesterov accelerated gradient descent with adaptive temperature."""
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    t_k = 1.0
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    size2 = 1
    while size2 < n + 2*n - 1:
        size2 <<= 1
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = 1.0 / (tau * (it + 1)**0.6)
        
        # Compute gradient at y
        b = fast_conv(y)
        b_max_val = b.max()
        e = np.exp(tau * (b - b_max_val))
        p = e / e.sum()
        
        p_rev = p[::-1]
        conv_yp = np.fft.irfft(np.fft.rfft(y, size2) * np.fft.rfft(p_rev, size2), size2)[:n + len(p) - 1]
        grad_y = 2.0 * tau * conv_yp[n-1 : 2*n-1][::-1]
        
        # Update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        
        a_new = y - lr * grad_y
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a_new
        else:
            y = y / sy
        
        a = a_new
        t_k = t_new
        
        # Track best
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
