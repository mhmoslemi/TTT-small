import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def subgrad_smooth_max(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[lo:hi + 1][::-1]
    return grad, b, b_max


def subgrad_exact_max(a):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    tol = 1e-10 * max(1.0, b_max)
    active = np.where(b >= b_max - tol)[0]
    p = np.zeros(2 * n - 1)
    p[active] = 1.0 / len(active)
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[lo:hi + 1][::-1]
    return grad, b, b_max


def gs_flat(a_init, n_iters=200, beta=0.85):
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = normalize(a)
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_power = mag ** 0.5
        mag_new = mag_power / (mag_power.max() + 1e-30) * mag.max()
        mag_mix = (1.0 - beta) * mag + beta * mag_new
        F_new = np.exp(1j * phase) * mag_mix
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        M_c = fast_conv(a_new).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_new.copy()
        a = a_new
    return best_a


def nesterov_optimize(a, deadline, max_iters=3000, tau_start=2.0, tau_end=300.0,
                      lr_init=0.5, lr_end=0.005):
    a = normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = subgrad_smooth_max(y, tau)
        if b_max < best_M - 1e-15:
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = normalize(y)
        a = a_new
    return best_a


def build_conv_matrix_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve_full(a, M):
    n = len(a)
    A = build_conv_matrix_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 60.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_solve_cutting(a, M, initial_constraints=30, max_rounds=30):
    n = len(a)
    b_ref = fast_conv(a)
    top_idx = np.argsort(b_ref)[-initial_constraints:]
    active_set = list(top_idx)
    g = None
    for rnd in range(max_rounds):
        rows = active_set
        if len(rows) < 2:
            rows = list(range(2 * n - 1))
        sub_A = build_conv_matrix_sparse(n, a)
        A_sub = sub_A[rows, :]
        c = -np.ones(n)
        b_ub = np.full(len(rows), M)
        bounds = [(0, None)] * n
        try:
            r = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 30.0})
            if not (r.success and r.x is not None and r.x.sum() > 1e-15):
                return g
            g = np.maximum(r.x, 0)
        except Exception:
            return g
        # Check all constraints
        # (a*g)(k) for all k
        ag = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)

def anderson_mix(history_x, history_Fx, m):
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 2.0)
    s_alpha = alpha.sum()
    if abs(s_alpha) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s_alpha
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)

        if gs < 1e-10:

        g /= gs
        # Backtracking line search
        best_t = t_base / (1 + 0.1 * max(0, it - 10))
        best_M_t = best_M
        candidates = [best_t, best_t * 0.5, best_t * 1.5, best_t * 0.25, best_t * 2.0, 0.05, 0.02]

        a_new = normalize((1.0 - best_t) * a + best_t * g)

        if len(hist_x) > anderson_window:

                if M_and <= M_new + 1e-10:

                else:

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    k = max(5, n // 10)
    kern = np.ones(k) / k
    inits.append(normalize(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01)))
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.2, 1.0, half_n), 0.01)
    if n % 2 == 0:
        full = np.concatenate([half, half[-2::-1]])[:n]
    else:
        full = np.concatenate([half, half[-1::-1]])[:n]
    inits.append(normalize(full))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None or len(a) == 0:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(512)
        best.update(parent)

    n_parent = parent.size
    print(f"Starting optimization, parent n={n_parent}", flush=True)

    # === PHASE 1: Initialize and optimize at parent length ===
    p1_deadline = time.monotonic() + 200.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Initialize + Nesterov at n={n_parent} ({200:.0f}s budget)", flush=True)
    print(f"{'='*60}", flush=True)

    n = n_parent
    # Resample parent to n if needed
    a_start = resample_seq(parent, n)
    a_start = normalize(a_start)

    # GS initialization
    a_gs = gs_flat(a_start, n_iters=300, beta=0.85)
    best.update(a_gs)
    print(f"  After GS: J={compute_J(a_gs):.6f}", flush=True)

    # Nesterov from GS
    a_nest = nesterov_optimize(a_gs, min(time.monotonic() + 60, p1_deadline),
                               max_iters=2000, tau_start=2.0, tau_end=200.0,
                               lr_init=0.5, lr_end=0.01)
    best.update(a_nest)
    print(f"  After Nesterov: J={compute_J(a_nest):.6f}", flush=True)

    # Try diverse inits
    inits = make_diverse_inits(n, rng, parent)
    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline - 10:
            break
        remaining = p1_deadline - time.monotonic()
        if remaining < 5:
            break
        allot = min(20.0, remaining * 0.7)
        a_opt = nesterov_optimize(a0, time.monotonic() + allot,
                                  max_iters=min(1000, int(allot * 100)),
                                  tau_start=5.0, tau_end=150.0,
                                  lr_init=0.3, lr_end=0.02)
        best.update(a_opt)

    best.report("Phase 1")
    p1_time = time.monotonic()
    print(f"Phase 1 done in {p1_time - (p1_deadline - 200):.1f}s", flush=True)

    # === PHASE 2: LP Mirror Descent with Anderson (main workhorse) ===
    p2_deadline = time.monotonic() + min(700.0, deadline - time.monotonic() - 200.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent + Anderson (budget {min(700.0, deadline - time.monotonic() - 200.0):.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        a_work = best.seq.copy()
        # First round: cutting-plane LP
        a_lp = lp_anderson_optimize(a_work, min(time.monotonic() + 300, p2_deadline),
                                    max_iters=200, anderson_window=5,
                                    t_base=0.4, use_cutting=True)
        best.update(a_lp)
        print(f"  After LP (cutting): J={compute_J(a_lp):.6f}", flush=True)

        # Second round: full LP with smaller step
        if time.monotonic() < p2_deadline - 30:
            a_lp2 = lp_anderson_optimize(a_lp, p2_deadline,
                                         max_iters=150, anderson_window=6,
                                         t_base=0.2, use_cutting=False)
            best.update(a_lp2)
            print(f"  After LP (full): J={compute_J(a_lp2):.6f}", flush=True)

    best.report("Phase 2")

    # === PHASE 3: Perturbation restarts ===
    p3_deadline = min(time.monotonic() + 300.0, deadline - 100.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Perturbation Restarts (budget {max(0, p3_deadline - time.monotonic()):.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        for restart in range(6):
            if time.monotonic() > p3_deadline - 20:
                break
            # Various perturbation strategies
            strategy = restart % 3
            if strategy == 0:
                # Gaussian noise
                a_pert = best.seq + rng.normal(0, 0.02, best.seq.size)
            elif strategy == 1:
                # Coordinate swap
                a_pert = best.seq.copy()
                i = rng.integers(0, len(a_pert))
                j = rng.integers(0, len(a_pert))
                delta = a_pert[i] - a_pert[j]
                a_pert[i] -= delta
                a_pert[j] += delta
            else:
                # Random rescaling of a segment
                a_pert = best.seq.copy()
                seg_len = max(5, len(a_pert) // 8)
                start = rng.integers(0, len(a_pert) - seg_len)
                a_pert[start:start + seg_len] *= rng.uniform(0.7, 1.3)
            a_pert = normalize(np.maximum(a_pert, 0))
            # Quick Nesterov
            allot = min(30.0, (p3_deadline - time.monotonic()) * 0.4)
            if allot < 3:
                break
            a_opt = nesterov_optimize(a_pert, time.monotonic() + allot,
                                      max_iters=500, tau_start=10, tau_end=200,
                                      lr_init=0.2, lr_end=0.02)
            best.update(a_opt)
            # Quick LP polish
            if time.monotonic() < p3_deadline - 5:
                allot2 = min(20.0, p3_deadline - time.monotonic() - 3)
                if allot2 > 3:
                    a_lp = lp_anderson_optimize(a_opt, time.monotonic() + allot2,
                                                max_iters=50, anderson_window=4,
                                                t_base=0.3, use_cutting=True)
                    best.update(a_lp)

    best.report("Phase 3")

    # === PHASE 4: Multi-length exploration ===
    p4_deadline = min(time.monotonic() + 200.0, deadline - 30.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration (budget {max(0, p4_deadline - time.monotonic()):.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_cur = best.seq.size
        # Try nearby lengths
        length_candidates = []
        for delta in [-64, -32, -16, 16, 32, 64, 128, 192, 256]:
            n_try = n_cur + delta
            if 256 <= n_try <= 2500 and n_try != n_cur:
                length_candidates.append(n_try)
        # Also try some round numbers
        for n_try in [768, 896, 1024, 1152, 1280, 1408, 1536, 1664, 1792, 1920, 2048]:
            if n_try != n_cur and 256 <= n_try <= 2500:
                length_candidates.append(n_try)
        length_candidates = sorted(set(length_candidates))

        for n_try in length_candidates:
            if time.monotonic() > p4_deadline - 15:
                break
            a_resamp = resample_seq(best.seq, n_try)
            a_resamp = normalize(a_resamp)
            # Quick Nesterov
            allot = min(25.0, (p4_deadline - time.monotonic()) * 0.5)
            if allot < 3:
                break
            a_opt = nesterov_optimize(a_resamp, time.monotonic() + allot,
                                      max_iters=400, tau_start=10, tau_end=150,
                                      lr_init=0.2, lr_end=0.02)
            best.update(a_opt)
            # Quick LP
            if time.monotonic() < p4_deadline - 5:
                allot2 = min(15.0, p4_deadline - time.monotonic() - 3)
                if allot2 > 2:
                    a_lp = lp_anderson_optimize(a_opt, time.monotonic() + allot2,
                                                max_iters=30, anderson_window=4,
                                                t_base=0.3, use_cutting=True)
                    best.update(a_lp)

    best.report("Phase 4")

    # === PHASE 5: Final intensive polishing ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 5:
        # Plain LP mirror descent for precise convergence
        a_final = lp_mirror_plain(best.seq, min(time.monotonic() + 80, deadline - 10),
                                  max_iters=80)
        best.update(a_final)

        # One more Anderson pass
        if time.monotonic() < deadline - 5:
            a_final2 = lp_anderson_optimize(a_final, deadline - 3,
                                            max_iters=30, anderson_window=5,
                                            t_base=0.15, use_cutting=False)
            best.update(a_final2)

        # Tiny perturbation + re-optimize
        if time.monotonic() < deadline - 3:
            a_tiny = best.seq + rng.normal(0, 0.003, best.seq.size)
            a_tiny = normalize(np.maximum(a_tiny, 0))
            a_tiny = nesterov_optimize(a_tiny, deadline - 1, max_iters=200,
                                       tau_start=50, tau_end=500, lr_init=0.1, lr_end=0.01)
            best.update(a_tiny)

    best.report("Phase 5 (Final)")
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    return [float(x) for x in best.seq]
