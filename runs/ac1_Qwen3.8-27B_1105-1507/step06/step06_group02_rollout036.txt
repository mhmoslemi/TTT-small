def fast_conv(a):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):

    s = a.sum()
    if s < 0.01:

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def gs_targeted(a, n_iters=200, beta=0.7):

    best_M = fast_conv(a).max()

        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)

        mag_mean = mag.mean()
        mag_std = mag.std()

            # Mild compression
            mag_new = mag ** 0.8
        elif it < 2 * n_iters // 3:
            # Medium compression
            mag_new = mag ** 0.6
        else:

            k = 3.0 - 2.0 * (it - 2 * n_iters // 3) / (n_iters // 3)
            mag_new = np.minimum(mag, mag_mean + k * mag_std)
        mag_new = mag_new / (mag_new.max() + 1e-30) * mag.max()

    cols = []

        for j in range(j_min, j_max + 1):

        #   d/d a_i of this sum:
        #     For j = i: d/d a_i [a_i * a_{k-i}] = a_{k-i} (treating a_{k-i} as constant if k-i != i)

        #     = a_{k-i} + a_{k-i}
        #     = 2 * a_{k-i}  (when 0 <= k-i <= n-1)

        #   Actually no: in the sum sum_j a_j * a_{k-j}, each j gives one term.
        #   If j = i and k-j = i (i.e., k = 2i), then the term a_i * a_i appears once (for j=i).
        #   d/d a_i of a_i^2 = 2*a_i.
        #   And our formula gives 2*a_{k-i} = 2*a_i. Correct!
        #   
        #   If j = i but k-i != i: term is a_i * a_{k-i}, d/d a_i = a_{k-i}
        #   If k-j = i (i.e., j = k-i) but j != i: term is a_{k-i} * a_i, d/d a_i = a_{k-i}
        #   Total: 2*a_{k-i} when both i and k-i are in [0,n-1]
        #   But if only one is in range (say i is in [0,n-1] but k-i is not), then only the j=i term
        #   exists, and d/d a_i = a_{k-i}... but k-i is out of range so a_{k-i} = 0.
        #   Hmm, this doesn't work.
        #   
        #   Let me just be explicit: c_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_j * a_{k-j}
        #   d(c_k)/da_i = 1[i in range] * a_{k-i} + 1[k-i in range] * a_{k-i}
        #   Wait, that's not right either. Let me index properly.
        #

        #   For a fixed i, the derivative is:
        #     - If i is in [L, U]: the term a_i * a_{k-i} contributes a_{k-i} to d/da_i
        #     - For each j in [L,U] where k-j = i (i.e., j = k-i): the term a_j * a_i contributes a_j = a_{k-i}

        #     k-i >= max(0, k-n+1):

        #   L = max(0, k-n+1), U = min(n-1, k)
        #   Condition 1: L <= i <= U => max(0, k-n+1) <= i <= min(n-1, k)
        #     Since 0 <= i <= n-1: this is k-n+1 <= i <= k (intersected with [0,n-1])

        #       if k >= n-1: k-i >= k-n+1 => i <= n-1 (always true)

        #   So condition 2: i >= k-(n-1) = k-n+1 and i <= (n-1 if k>=n-1 else k)
        #   
        #   Hmm, this is getting complicated. Let me just use the direct formula:
        #   d(c_k)/da_i = a_{k-i} if 0 <= k-i <= n-1 and (i + (k-i) = k is a valid convolution index)
        #   
        #   Actually, the simplest way: since c = conv(a,a) and conv is bilinear,
        #   d(c_k)/da_i = (conv(e_i, a))_k + (conv(a, e_i))_k where e_i is the unit vector

        #   (conv(a, e_i))_k = a_{k-i} (same thing! since conv is commutative for real sequences? No!)

        #   This is 2 * flip(a)[k-i] where flip(a) reverses... 
        #   Actually it's: grad_k[i] = 2 * a[k-i] for max(0, k-(n-1)) <= i <= min(n-1, k)
        
        # Simpler: grad_k = 2 * (a reversed and shifted by k, truncated to [0,n-1])
        # a[k-i] for i in [0, n-1] is: a[k], a[k-1], ..., a[k-(n-1)] but only valid where 0<=k-i<=n-1

        # Then a[k-i] = a_rev[n-1-(k-i)] = a_rev[n-1-k+i]
        # So grad_k[i] = 2 * a_rev[n-1-k+i] for valid i
        # The valid range of i: 0 <= k-i <= n-1 => k-(n-1) <= i <= k

        # Even simpler: just compute it directly
        grad_k = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            grad_k[i] = 2.0 * a[k - i]
        grad += p[k] * grad_k
    return grad, b, b_max

def run_lp_mirror(a, deadline, max_iters=500, use_line_search=True):
    """LP mirror descent with optional line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break
        # Solve LP
        g, t_lp = lp_min_peak(a)
        if g is None:
            break
        # Update
        if use_line_search:
            t_found, M_found = line_search(a, g, best_M)
            if t_found > 0:
                a = np.maximum((1 - t_found) * a + t_found * g, 0)
                s = a.sum()
                if s < 1e-10:
                    a = best_a.copy()
                else:
                    a = a / s
            else:
                # No improvement found, use small fixed step
                t = 0.1
                a_new = np.maximum((1 - t) * a + t * g, 0)
                s = a_new.sum()
                if s > 1e-10:
                    a = a_new / s
        else:
            t = max(0.05, 0.3 * (0.95 ** it))
            a_new = np.maximum((1 - t) * a + t * g, 0)
            s = a_new.sum()
            if s < 1e-10:
                a = best_a.copy()
            else:
                a = a_new / s
    return best_a

def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.3, lr_end=0.001):
    """Nesterov-accelerated smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M - 1e-12:
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.005))
        inits.append(gs_targeted(p, n_iters=100, beta=0.8))
    inits.append(np.ones(n) / n)
    for alpha in [0.3, 0.5, 1.0, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t**alpha + (1-t)**alpha), 0.01) / n)
    for f in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append((np.exp(-5*(t-0.5)**2) + 0.1) / n)
    # Random smooth
    k = max(3, n // 16)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01) / n)
    # Symmetric
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full / full.sum())
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_targeted(a_rand, n_iters=80, beta=0.85))
    return inits

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, bmat
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_targeted(a, n_iters=200):
    """Gerchberg-Saxton with progressive spectral compression."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        frac = it / max(1, n_iters)
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Progressive compression
        power = 1.0 - 0.7 * frac
        mag_new = mag ** power
        # Also apply soft clipping in later iterations
        if frac > 0.5:
            mag_mean = mag_new.mean()
            mag_std = mag_new.std() + 1e-15
            clip_threshold = mag_mean + (3.0 - 2.0 * (frac - 0.5) / 0.5) * mag_std
            mag_new = np.minimum(mag_new, clip_threshold)
        mag_new = mag_new / (mag_new.max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def build_conv_sparse(n, a):
    """Build sparse cross-correlation matrix: (A*g)_k = sum_j a[k-j]*g_j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_min_peak(a, time_limit=10):
    """LP: minimize t s.t. conv(a,g)_k <= t, sum(g)=1, g>=0, t>=0."""
    n = len(a)
    rows, cols, vals = build_conv_sparse(n, a)
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    I_sparse = csr_matrix((2 * n - 1, 2 * n - 1))
    A_ub_sparse = bmat([[A_sparse, -I_sparse]], format='csr')
    c = np.zeros(n + 1)
    c[-1] = 1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    try:
        r = linprog(c, A_ub=A_ub_sparse, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            t_val = float(r.x[n])
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs
                return g, t_val
    except Exception:
        pass
    return None, None


def line_search(a, g, best_M, n_samples=16):
    """Find best step size t in (0,1] minimizing M((1-t)*a + t*g)."""
    ts = np.concatenate([np.logspace(-3, 0, 10), np.linspace(0.1, 1.0, 6)])
    ts = np.sort(np.unique(np.clip(ts, 1e-4, 1.0)))
    best_t = 0.0
    best_val = best_M
    for t in ts:
        a_new = np.maximum((1 - t) * a + t * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        M = fast_conv(a_new).max()
        if M < best_val - 1e-14:
            best_val = M
            best_t = t
    return best_t, best_val


def smooth_max_grad(a, tau):
    """Compute smooth subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shift = b - b_max
    # Prevent overflow
    shift = np.clip(shift, -100.0 / tau, 0.0)
    exp_b = np.exp(tau * shift)
    p = exp_b / (exp_b.sum() + 1e-30)
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        pk = p[k]
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            grad[i] += pk * 2.0 * a[k - i]
    return grad, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=500.0,
                        lr_init=0.3, lr_end=0.001):
    """Nesterov-accelerated smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M - 1e-12:
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y = y / sy
        a = a_new
    return best_a


def run_lp_mirror(a, deadline, max_iters=200, lp_time_limit=8):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 15:
                break
        g, t_lp = lp_min_peak(a, time_limit=lp_time_limit)
        if g is None:
            break
        t_found, M_found = line_search(a, g, best_M)
        if t_found > 0:
            a = np.maximum((1 - t_found) * a + t_found * g, 0)
            s = a.sum()
            if s < 1e-10:
                a = best_a.copy()
            else:
                a = a / s
        else:
            t = max(0.05, 0.2 * (0.9 ** it))
            a_new = np.maximum((1 - t) * a + t * g, 0)
            s = a_new.sum()
            if s > 1e-10:
                a = a_new / s
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initializations emphasizing asymmetry."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    # Parent-based
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.2 * rng.normal(0, 1, n)), 0.005))
        inits.append(gs_targeted(p, n_iters=120))
    # Linear ramp (maximally asymmetric)
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    # Quadratic ramp
    ramp2 = (np.arange(1, n + 1, dtype=np.float64)) ** 2
    inits.append(ramp2 / ramp2.sum())
    # Reverse ramp
    inits.append((ramp[::-1] / ramp.sum()).copy())
    # Exponential decay
    exp_decay = np.exp(-3.0 * t)
    inits.append(exp_decay / exp_decay.sum())
    # Exponential growth
    exp_growth = np.exp(3.0 * t)
    inits.append(exp_growth / exp_growth.sum())
    # Symmetric shapes
    inits.append(np.ones(n) / n)
    for alpha in [0.3, 0.5, 1.0, 2.0, 4.0]:
        shp = t**alpha + (1 - t)**alpha
        inits.append(shp / shp.sum())
    # Multi-frequency
    for f in [1, 2, 3, 5]:
        shp = 0.5 + 0.3 * np.cos(2 * np.pi * f * t) + 0.2 * np.sin(2 * np.pi * f * t)
        shp = np.maximum(shp, 0.01)
        inits.append(shp / shp.sum())
    # Gaussian mixtures
    for mu in [0.3, 0.5, 0.7]:
        shp = np.exp(-5 * (t - mu) ** 2) + 0.05
        inits.append(shp / shp.sum())
    # Smooth random
    k = max(3, n // 12)
    kern = np.ones(k) / k
    shp = np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same'), 0.01)
    inits.append(shp / shp.sum())
    # Asymmetric random smooth
    shp = np.maximum(rng.uniform(0, 1, n) * (0.3 + 0.7 * t), 0.01)
    shp = np.convolve(shp, kern, 'same')
    shp = np.maximum(shp, 0.005)
    inits.append(shp / shp.sum())
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_targeted(a_rand, n_iters=80))
    # GS from ramp
    inits.append(gs_targeted(ramp / ramp.sum(), n_iters=80))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            s = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
