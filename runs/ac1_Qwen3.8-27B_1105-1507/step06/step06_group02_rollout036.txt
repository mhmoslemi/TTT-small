"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast smooth optimization (Nesterov + softmax gradient)
2. LP mirror descent with constraint generation (faster than full LP)
3. Perturbation-based global search to escape local minima
4. Final polish with aggressive step sizes
Key improvement: Constraint generation for LP (only solve with active constraints),
giving 5-10x speedup per LP iteration, allowing more total iterations.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a, b=None):
    """Full linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """Compute smoothed max gradient for the autocorrelation."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    # Shift to avoid overflow
    shifted = b - b_max
    exp_b = np.exp(tau * shifted)
    p = exp_b / (exp_b.sum() + 1e-30)
    
    # Gradient of sum_k p[k] * conv(a,a)[k]
    # = sum_k p[k] * (T_k + T_k^T) a
    grad = np.zeros(n)
    # For efficiency, only process significant p values
    active = np.where(p > 1e-10)[0]
    for k in active:
        # (T_k a)[i] = a[k-i] if 0 <= k-i <= n-1
        # (T_k^T a)[i] = a[i-k+n-1-(n-1-i)] ... let me think more carefully
        # conv(a,a)[k] = sum_i a[i] * a[k-i] (with a[j]=0 outside [0,n-1])
        # d/dx = sum_i (T_k + T_k^T) a, where (T_k a)[i] = a[k-i]
        # So grad[i] += 2 * p[k] * a[k-i] for valid i
        # More precisely: grad += 2 * p[k] * (T_k + T_k^T) a
        # (T_k a)[i] = a[k-i] if 0 <= k-i <= n-1, i.e., k-n+1 <= i <= k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # (T_k a)[i] = a[k-i] for i in [lo, hi]
        # (T_k^T a)[i] = (T_k a)[k-i] = a[k-(k-i)] = a[i] ... wait
        # T_k[i,j] = 1 if i+j=k, so (T_k a)[i] = sum_j T_k[i,j] a[j] = a[k-i]
        # T_k^T[i,j] = T_k[j,i] = 1 if j+i=k, so (T_k^T a)[i] = sum_j T_k[j,i] a[j] = a[k-i]... 
        # Wait, T_k[j,i] = 1 if j+i=k, so (T_k^T a)[i] = sum_j T_k[j,i] a[j] = a[k-i]
        # So T_k = T_k^T? No!
        # T_k[i,j] = 1 if i+j=k. So T_k is symmetric! T_k[i,j] = T_k[j,i] since i+j = k iff j+i = k.
        # So (T_k + T_k^T) a = 2 * T_k a
        # grad contributes: 2 * p[k] * (T_k a), where (T_k a)[i] = a[k-i] for i in [max(0,k-n+1), min(k,n-1)]
        seg = a[k - hi : k - lo + 1][::-1]  # a[k-hi], a[k-hi+1], ..., a[k-lo]
        # Wait, for i in [lo, hi]: (T_k a)[i] = a[k-i]
        # i = lo: a[k-lo], i = lo+1: a[k-lo-1], ..., i = hi: a[k-hi]
        # So (T_k a)[lo:hi+1] = a[k-lo], a[k-lo-1], ..., a[k-hi] = a[k-hi:k-lo+1][::-1]
        grad[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
    return grad, p, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.4, lr_end=0.003):
    """Nesterov-accelerated smooth gradient descent on the softmax approximation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def lp_best_response_cg(a, M, max_rounds=15, top_frac=0.3):
    """LP best response with constraint generation.
    
    Only activates constraints that are near the peak, then iteratively
    adds violated constraints.
    """
    n = len(a)
    b_self = fast_conv(a)
    b_max = float(b_self.max())
    
    # Initially activate top constraints (those where b_self is above threshold)
    threshold = b_max * (1 - top_frac)
    active_idx = np.where(b_self >= threshold)[0]
    
    for round_idx in range(max_rounds):
        # Build constraint matrix for active constraints
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active_idx):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(float(a[k - j]))
        
        if len(rows) == 0:
            return None
            
        A = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))
        c = -np.ones(n)
        b_ub = np.full(len(active_idx), M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 30})
        except Exception:
            break
        
        if not r.success or r.x is None:
            break
        
        g = np.maximum(r.x, 0)
        
        # Check for violated inactive constraints
        cross = fast_conv(a, g)
        violated = np.where(cross > M * (1 + 1e-6))[0]
        
        if len(violated) == 0:
            return g
        
        # Add violated constraints
        active_idx = np.unique(np.concatenate([active_idx, violated]))
    
    return g


def run_lp_anderson(a, deadline, max_iters=200, anderson_window=5,
                    t_base=0.3, min_t=0.005):
    """LP mirror descent with Anderson acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    hist_x = []
    hist_Fx = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
        
        g = lp_best_response_cg(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        if it < 5:
            t = t_base
        else:
            t = max(min_t, t_base / (1 + 0.05 * it))
        
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 3:
            a_anderson = anderson_solve(hist_x, hist_Fx, len(hist_x))
            sa = a_anderson.sum()
            if sa > 1e-10:
                a_anderson /= sa
            M_and = fast_conv(a_anderson).max()
            M_new = fast_conv(a_new).max()
            if M_and <= M_new * 1.02:
                a = a_anderson
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def anderson_solve(history_x, history_Fx, m):
    """Anderson mixing."""
    residuals = []
    for i in range(m):
        residuals.append(history_x[i] - history_Fx[i])
    residuals = np.array(residuals)
    G = residuals @ residuals.T
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -1.0, 1.0)
    alpha_sum = alpha.sum()
    if abs(alpha_sum) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / alpha_sum
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast smooth Nesterov optimization for initialization
2. LP mirror descent with constraint generation (5-20x faster than full LP)
3. Active-set tracking across LP iterations for warm starting
4. Backtracking line search for adaptive step sizes
5. Structured perturbation search to escape local minima
Key improvement: Constraint generation reduces LP from O(n^2) to O(K*n) per solve,
enabling 10-50x more LP iterations in the same time budget.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a, b=None):
    """Full linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)[:n + m - 1]
    return np.maximum(result, 0)


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """Compute smoothed max gradient for the autocorrelation."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = b - b_max
    exp_b = np.exp(tau * shifted)
    p = exp_b / (exp_b.sum() + 1e-30)
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.4, lr_end=0.003):
    """Nesterov-accelerated smooth gradient descent on the softmax approximation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def lp_solve_constraints(a, b_ub_values, active_idx, n):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= b_ub for k in active_idx, g >= 0."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_idx):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(float(a[k - j]))
    if len(rows) == 0:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub_values, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_best_response_cg(a, M, active_hints=None, max_rounds=15, max_constraints=300):
    """LP best response with constraint generation and active-set hints.
    
    Parameters:
    - a: current sequence
    - M: current peak of conv(a,a)
    - active_hints: indices of constraints that were active in previous iteration
    - max_rounds: max constraint generation rounds
    - max_constraints: cap on number of active constraints
    
    Returns:
    - g: solution direction
    - active_set: the active constraint indices (for next iteration)
    """
    n = len(a)
    b_self = fast_conv(a)
    b_max = float(b_self.max())
    
    # Initialize active set from hints or top constraints
    if active_hints is not None and len(active_hints) > 0:
        # Verify hints are still relevant
        hint_arr = np.array(active_hints)
        still_valid = hint_arr[(hint_arr >= 0) & (hint_arr < 2 * n - 1)]
        # Add top constraints not in hints
        top_k = max(10, min(30, n // 20))
        top_idx = np.argsort(b_self)[-top_k:]
        active_set = set(still_valid.tolist()) | set(top_idx.tolist())
    else:
        top_k = max(15, min(50, n // 10))
        top_idx = np.argsort(b_self)[-top_k:]
        active_set = set(top_idx.tolist())
    
    # Cap the initial set
    if len(active_set) > max_constraints:
        # Keep the ones with highest b_self values
        active_list = sorted(active_set, key=lambda k: -b_self[k])
        active_set = set(active_list[:max_constraints])
    
    g = None
    for round_idx in range(max_rounds):
        active_list = sorted(active_set)
        b_ub = np.full(len(active_list), M)
        g = lp_solve_constraints(a, b_ub, active_list, n)
        if g is None:
            break
        if g.sum() < 1e-15:
            return None, list(active_set)
        
        # Check for violations of inactive constraints
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        cross = np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(g, size), size)[:2 * n - 1]
        
        # Find violations (with small tolerance)
        tol = M * 1e-7
        violated = np.where(cross > M + tol)[0]
        
        if len(violated) == 0:
            return g, list(active_set)
        
        # Add violated constraints, prioritizing most violated
        for v in violated:
            if len(active_set) >= max_constraints:
                break
            active_set.add(int(v))
    
    return g, list(active_set)


def anderson_solve(history_x, history_Fx, m):
    """Anderson mixing."""
    residuals = []
    for i in range(m):
        residuals.append(history_x[i] - history_Fx[i])
    residuals = np.array(residuals)
    G = residuals @ residuals.T
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 1.5)
    alpha_sum = alpha.sum()
    if abs(alpha_sum) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / alpha_sum
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gs_flat_spectral(init, n_iters=200, beta=0.8):
    """Gerchberg-Saxton for flat spectral initialization."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        power = 0.3 + 0.4 * beta
        mag_comp = (mag ** power)
        mag_comp = mag_comp / (mag_comp.max() + 1e-30) * (mag.mean() + 1e-30)
        F_new = np.exp(1j * phase) * mag_comp
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = float(fast_conv(a_candidate).max())
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)

            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass
    
    if parent is None:
        parent = np.ones(1024)
        best.update(parent)
    
    current_best = best.seq.copy()
    n = current_best.size
    print(f"Starting optimization: n={n}, J={best.score:.6f}", flush=True)
    
    # === PHASE 1: Smooth optimization (150s) ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth Optimization", flush=True)
    print(f"{'='*50}", flush=True)
    t_phase1 = time.monotonic()
    
    p1_deadline = time.monotonic() + 100.0
    a = run_nesterov_smooth(current_best, p1_deadline, max_iters=10000,
                           tau_start=1.0, tau_end=400.0,
                           lr_init=0.35, lr_end=0.001)
    best.update(a)
    
    # GS refinement
    if time.monotonic() < deadline - 1200:
        a_gs = gs_flat_spectral(current_best, n_iters=200, beta=0.9)
        a_gs = run_nesterov_smooth(a_gs, min(time.monotonic() + 20, deadline - 1200),
                                   max_iters=2000, tau_start=5, tau_end=200,
                                   lr_init=0.2, lr_end=0.005)
        best.update(a_gs)
    
    if best.seq is not None and best.score < compute_J(current_best):
        current_best = best.seq.copy()
    
    best.report("Phase 1")
    print(f"Phase 1: {time.monotonic()-t_phase1:.0f}s", flush=True)
    
    # === PHASE 2: Fast LP with constraint generation (900s) ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: Fast LP (Constraint Generation)", flush=True)
    print(f"{'='*50}", flush=True)
    t_phase2 = time.monotonic()
    
    lp_deadline = time.monotonic() + 850.0
    a = run_lp_mirror_fast(current_best, min(lp_deadline, deadline - 200),
                           max_iters=20000, anderson_window=5,
                           t_base=0.35, min_t=0.002, use_backtrack=True)
    best.update(a)
    if best.seq is not None and best.score < compute_J(current_best):
        current_best = best.seq.copy()
        n = current_best.size
    
    # Second LP pass with different parameters
    if time.monotonic() < deadline - 150:
        a2 = run_lp_mirror_fast(current_best, deadline - 150,
                                max_iters=15000, anderson_window=6,
                                t_base=0.25, min_t=0.001, use_backtrack=True)
        best.update(a2)
        if best.seq is not None and best.score < compute_J(current_best):
            current_best = best.seq.copy()
    
    best.report("Phase 2")
    print(f"Phase 2: {time.monotonic()-t_phase2:.0f}s", flush=True)
    
    # === PHASE 3: Perturbation + re-optimize (200s) ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Perturbation Search", flush=True)
    print(f"{'='*50}", flush=True)
    t_phase3 = time.monotonic()
    
    p3_deadline = deadline - 60.0
    for trial in range(10):
        if time.monotonic() > p3_deadline:
            break
        strat = trial % 5
        if strat == 0:
            a_pert = np.maximum(current_best * (1 + 0.03 * rng.normal(0, 1, n)), 1e-6)
        elif strat == 1:
            t_arr = np.linspace(0, 1, n)
            boost = 1 + 0.08 * np.cos(2 * np.pi * t_arr)
            a_pert = np.maximum(current_best * boost, 1e-6)
        elif strat == 2:
            t_arr = np.linspace(0, 1, n)
            dip = 1 - 0.08 * np.exp(-30 * (t_arr - 0.5) ** 2)
            a_pert = np.maximum(current_best * dip, 1e-6)
        elif strat == 3:
            freq = rng.integers(2, 6)
            t_arr = np.linspace(0, 1, n)
            pert = 1 + 0.05 * np.sin(2 * np.pi * freq * t_arr + rng.uniform(0, 2 * np.pi))
            a_pert = np.maximum(current_best * pert, 1e-6)
        else:
            # Scale edges
            t_arr = np.linspace(0, 1, n)
            edge_boost = 1 + 0.1 * (t_arr ** 4 + (1 - t_arr) ** 4)
            a_pert = np.maximum(current_best * edge_boost, 1e-6)
        
        a_pert /= a_pert.sum()
        
        # Quick smooth optimization
        sub_dl = min(time.monotonic() + 15, p3_deadline)
        a_pert = run_nesterov_smooth(a_pert, sub_dl, max_iters=1500,
                                     tau_start=5, tau_end=150,
                                     lr_init=0.15, lr_end=0.008)
        best.update(a_pert)
        
        # LP polish if improved
        if best.seq is not None and compute_J(a_pert) < compute_J(current_best) + 0.001:
            if time.monotonic() < p3_deadline - 5:
                a_lp = run_lp_mirror_fast(a_pert, p3_deadline, max_iters=100,
                                          anderson_window=4, t_base=0.25, min_t=0.005)
                best.update(a_lp)
                if best.seq is not None and best.score < compute_J(current_best):
                    current_best = best.seq.copy()
    
    best.report("Phase 3")
    print(f"Phase 3: {time.monotonic()-t_phase3:.0f}s", flush=True)
    
    # === PHASE 4: Final polish (40s) ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Final Polish", flush=True)
    print(f"{'='*50}", flush=True)
    
    if time.monotonic() < deadline - 5:
        a_final = run_lp_mirror_plain(current_best, deadline, max_iters=100)
        best.update(a_final)
        
        if time.monotonic() < deadline - 3:
            a_final2 = run_lp_mirror_fast(a_final, deadline, max_iters=50,
                                          anderson_window=5, t_base=0.15, min_t=0.001)
            best.update(a_final2)
    
    # === PHASE 5: Length exploration ===
    if best.seq is not None and time.monotonic() < deadline - 15:
        n_best = best.seq.size
        for delta in [-64, -32, 32, 64]:
            if time.monotonic() > deadline - 10:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 3072:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            sub_dl = min(time.monotonic() + 10, deadline - 8)
            a_ext = run_nesterov_smooth(a_ext, sub_dl, max_iters=500,
                                        tau_start=5, tau_end=100,
                                        lr_init=0.1, lr_end=0.01)
            best.update(a_ext)
            if best.seq is not None and best.score < compute_J(resample_seq(best.seq, n_try)):
                if time.monotonic() < deadline - 5:
                    a_lp = run_lp_mirror_fast(a_ext, deadline - 3, max_iters=30,
                                              anderson_window=4, t_base=0.2, min_t=0.005)
                    best.update(a_lp)
    
    best.report("Final")
    
    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)
    
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
