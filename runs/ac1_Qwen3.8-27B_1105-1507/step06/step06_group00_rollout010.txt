def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_standard(a, A=None):
    """Standard LP: min M s.t. A@g <= M, sum(g)=1, g>=0."""
    n = len(a)
    if A is None:
        A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def chase_peak_refine(a, n_steps=200, time_budget=30.0):
    """Chase-the-peak with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.02
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Center the gradient
        grad_centered = grad - np.mean(grad)
        
        # Try multiple step sizes
        improved = False
        for ss in [step_size, step_size * 0.5, step_size * 0.1, step_size * 2]:
            a_new = a - ss * grad_centered
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    break
        
        if not improved:
            step_size *= 0.7
            if step_size < 1e-6:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            a = a / s
        else:
            a = np.ones(n) / n
    elif mode == 2:
        # Exponential decay from one end
        a = np.exp(-np.linspace(0, 3, n))
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        c1 = n // 3
        c2 = 2 * n // 3
        w = n // 8
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w**2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w**2))
        a = a / np.sum(a)
    elif mode == 4:
        # Linear ramp down
        a = np.linspace(1, 0.1, n)
        a = a + rng.normal(0, 0.05, n)
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    else:
        # Random smooth (low-pass filtered noise)
        a = rng.uniform(0, 1, n)
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    return a

    return result
