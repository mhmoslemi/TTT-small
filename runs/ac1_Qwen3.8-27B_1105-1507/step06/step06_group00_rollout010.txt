def compute_conv(a):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):

        return np.inf
    a_norm = a / s

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s
    conv = compute_conv(a_norm)
    
    max_c = np.max(conv)
    if max_c < 1e-15:
        return 0, np.zeros(n)
    
    # For numerical stability, work with log
    # objective = 2*n * (sum conv^p)^(1/p)
    # gradient: 2 * sum_k conv[k]^(p-1) * a[k-m] (then scale)
    
    # Use log-sum-exp for stability
    log_conv = np.log(np.maximum(conv, 1e-300))
    log_pconv = p * log_conv
    max_log = np.max(log_pconv)
    log_sum = max_log + np.log(np.sum(np.exp(log_pconv - max_log)))
    
    # (sum conv^p)^(1/p) = exp(log_sum / p)
    lp_val = np.exp(log_sum / p)

    scale = np.exp(log_sum * (1.0/p - 1.0) + max_lw)
    grad = grad * scale
    
    obj_val = 2.0 * n * lp_val
    return obj_val, grad

"""
Fast iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth Lp surrogate with gradient+momentum, subgradient project-on-simplex,
multi-n exploration, structured initializations, and LP polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def _compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def _compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = _compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0)


def _subgradient_step(a, p=1.0):
    """
    Compute gradient of (sum conv^p)^(1/p) w.r.t. a (on simplex, sum=1).
    Returns (objective_value, gradient_array).
    """
    n = len(a)
    conv = _compute_conv(a)
    
    max_c = np.max(conv)
    if max_c < 1e-15:
        return 0.0, np.zeros(n)
    
    if p <= 1.0:
        # Pure subgradient of max
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        obj_val = 2.0 * n * max_c
        return obj_val, grad
    
    # Smooth Lp: objective = 2*n * (sum conv^p)^(1/p)
    # Use log for stability
    safe_conv = np.maximum(conv, 1e-300)
    log_pconv = p * np.log(safe_conv)
    max_log = np.max(log_pconv)
    log_sum = max_log + np.log(np.sum(np.exp(log_pconv - max_log)))
    lp_val = np.exp(log_sum / p)
    obj_val = 2.0 * n * lp_val
    
    # weights[k] = conv[k]^(p-1)
    log_weights = (p - 1.0) * np.log(safe_conv)
    max_lw = np.max(log_weights)
    weights = np.exp(log_weights - max_lw)
    
    # grad[m] = 2 * scale * sum_{j=0}^{n-1} a[j] * weights[m+j]
    # = 2 * scale * cross_corr(a, weights)[m]
    size_ft = 1
    while size_ft < 3 * n - 2:
        size_ft <<= 1
    fa = rfft(a, size_ft)
    fw = rfft(weights, size_ft)
    cross = irfft(np.conj(fa) * fw, size_ft)[:n]
    
    # scale = (sum conv^p)^(1/p - 1) * exp(max_lw)
    scale = np.exp(log_sum * (1.0 / p - 1.0) + max_lw)
    grad = 2.0 * scale * cross
    
    return obj_val, grad


def _smooth_lp_optimize(a, time_budget=60.0, p_max=100.0, verbose=False):
    """
    Optimize using smooth Lp with increasing p, momentum, and adaptive step.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = _compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    mu = 0.85
    step_size = 1.0
    
    # Phases with increasing p
    p_schedule = [2, 4, 8, 16, 32, 64, 128, 256, p_max]
    p_schedule = [p for p in p_schedule if p <= p_max]
    
    total_iters = 0
    stall_count = 0
    
    for p in p_schedule:
        if time.monotonic() - start > time_budget:
            break
        
        # Adapt mu and step for this p
        mu_p = 0.8 + 0.1 * min(p / 32.0, 1.0)
        step_p = 0.8 / np.sqrt(p)
        
        inner_budget = time_budget * 0.7 / len(p_schedule)
        phase_start = time.monotonic()
        
        for it in range(300):
            if time.monotonic() - start > time_budget:
                break
            if time.monotonic() - phase_start > inner_budget:
                break
            
            obj_val, grad = _subgradient_step(a, float(p))
            
            # Center gradient (project onto tangent of simplex)
            grad_centered = grad - np.mean(grad)
            
            # Momentum update
            momentum = mu_p * momentum - step_p * grad_centered
            a_new = a + momentum
            a_new = _project_simplex(a_new)
            
            actual_obj = _compute_obj(a_new)
            total_iters += 1
            
            if actual_obj < best_obj:
                best_obj = actual_obj
                best_a = a_new.copy()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 30:
                    # Reduce step and reset momentum
                    step_p *= 0.6
                    momentum *= 0.5
                    stall_count = 0
            
            a = a_new
    
    return best_a


def _subgradient_momentum_optimize(a, time_budget=60.0, verbose=False):
    """
    Pure subgradient (p=infinity) with Nesterov momentum and backtracking.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()

    if s < 1e-10:
        return a

    best_obj = _compute_obj(a)

    momentum = np.zeros(n)
    mu = 0.9
    step_size = 2.0
    stall = 0
    
    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = _compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        grad_centered = grad - np.mean(grad)
        
        # Nesterov momentum
        momentum = mu * momentum - step_size * grad_centered
        a_new = a + momentum
        a_new = _project_simplex(a_new)
        
        actual_obj = _compute_obj(a_new)
        
        if actual_obj < best_obj:
            best_obj = actual_obj
            best_a = a_new.copy()
            stall = 0
            # Slowly reduce step
            if it % 50 == 0:
                step_size *= 0.98
        else:
            stall += 1
            if stall > 50:
                step_size *= 0.5
                momentum *= 0.3
                stall = 0
            elif stall > 200:

                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = _project_simplex(a)
                momentum = np.zeros(n)
                step_size = 2.0
                stall = 0
        
        a = a_new if actual_obj < _compute_obj(a) else a
    
    return best_a


def _lp_polish(a, time_budget=30.0, verbose=False):
    """
    A few iterations of LP alternating for final polish.
    Uses scipy linprog for speed.
    """
    from scipy.optimize import linprog
    
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = _compute_obj(a)
    start = time.monotonic()
    
    for it in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        # Build constraint matrix: A @ g = conv(a, g)
        # conv(a, g)[k] = sum_i a[i] * g[k-i] for valid indices
        # We want: conv(a, g)[k] <= M for all k
        # LP: min M s.t. A@g <= M, sum(g) = 1, g >= 0
        
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            row_start = j
            row_end = min(2 * n - 1, j + n)
            length = row_end - row_start
            A[row_start:row_end, j] = a[:length]
        
        # Variables: g[0..n-1], M
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize M
        
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0 or res.x is None:
                break
        except Exception:
            break
        
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs < 1e-10:
            break
        g_val = g_val / gs
        
        # Line search
        best_t_obj = np.inf
        best_t = 0.5
        for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g_val
            a_try = _project_simplex(a_try)
            obj_try = _compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:
            a = (1.0 - best_t) * a + best_t * g_val
            a = _project_simplex(a)
            best_obj = best_t_obj
            best_a = a.copy()
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            break
    
    return best_a


def _random_init(n, rng, mode=0):
    """Generate random non-negative sequence on simplex."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / np.sum(a)
    elif mode == 3:
        # Edge-enhanced
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 4:
        # Quadratic residue pattern (smoothed)
        p = n
        a = np.zeros(n)
        a[0] = 1
        for i in range(1, n):
            a[i] = 1.0 if (i * i) % n < n // 2 else 0.3

        a = a / np.sum(a)
    elif mode == 5:

        center1 = n // 4
        center2 = 3 * n // 4
        width = n // 8
        x = np.arange(n)
        a = np.exp(-((x - center1) ** 2) / (2 * width ** 2)) + \
            np.exp(-((x - center2) ** 2) / (2 * width ** 2))
        a = a / np.sum(a)
    elif mode == 6:
        # Tent (triangle)
        x = np.arange(n)
        center = n // 2
        a = 1.0 - np.abs(x - center) / center
        a = np.maximum(a, 0.1)
        a = a / np.sum(a)
    else:
        # Random smooth
        freqs = rng.integers(1, 5, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.arange(n)
        a = np.ones(n)
        for f, ph in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x / n + ph)
        a = np.maximum(a, 0.05)
        a = a / np.sum(a)
    return a


def _resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # === Phase 3: Random restarts with diverse initializations ===
    print(f"\n=== Phase 3: Random Restarts ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_restarts = min(10, int(remaining / 25))
        restart_budget = min(25.0, remaining * 0.6 / max(1, n_restarts))
        
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < restart_budget + 5:
                break
            
            mode = r % 7
            n_r = n_best
            # Occasionally try different n
            if r % 3 == 0 and n_best > 400:
                n_r = n_best + rng.integers(-100, 100)
                n_r = max(300, min(1500, n_r))
            
            if mode == 1:
                # Perturbed best
                a_r = best_seq[:n_best] if n_r == n_best else _resample(best_seq, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.1, len(a_r)))
                a_r = _project_simplex(a_r)
            else:
                a_r = _random_init(n_r, rng, mode=mode)
            
            # Optimize
            a_opt = _smooth_lp_optimize(a_r, time_budget=restart_budget * 0.7, p_max=150)
            obj_opt = _compute_obj(a_opt)
            
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                n_best = len(a_opt)
                print(f"  Restart {r} (mode={mode}, n={n_best}): obj={obj_opt:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}, n={n_r}): obj={obj_opt:.8f}")
    
    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()
    a_final = best_seq.copy()
    
    if remaining > 20:
        # Smooth Lp with high p
        alloc = min(remaining * 0.3, remaining - 10)
        a_f1 = _smooth_lp_optimize(a_final, time_budget=alloc, p_max=500)
        obj_f1 = _compute_obj(a_f1)
        print(f"  Final SmoothLP(p=500): obj={obj_f1:.8f}")
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            a_final = a_f1
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Subgradient
        alloc = min(remaining * 0.3, remaining - 10)
        a_f2 = _subgradient_momentum_optimize(a_final, time_budget=alloc)
        obj_f2 = _compute_obj(a_f2)
        print(f"  Final SubGrad: obj={obj_f2:.8f}")
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            a_final = a_f2
    
    remaining = deadline - time.monotonic()
    if remaining > 10:
        # LP polish
        alloc = min(remaining * 0.5, remaining - 5)
        a_f3 = _lp_polish(a_final, time_budget=alloc, verbose=True)
        obj_f3 = _compute_obj(a_f3)
        print(f"  Final LP: obj={obj_f3:.8f}")
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
    
    # One more round of smooth Lp on the very best
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_f4 = _smooth_lp_optimize(best_seq, time_budget=min(remaining * 0.5, 30), p_max=300)
        obj_f4 = _compute_obj(a_f4)
        print(f"  Final SmoothLP2: obj={obj_f4:.8f}")
        if obj_f4 < best_score:
            best_score = obj_f4
            best_seq = a_f4.copy()
    
    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = _compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
