    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=30):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def parabolic_line_search(f, x0, x1, x2, max_iter=20):
    """Parabolic interpolation line search. f should be a 1D function."""
    # Assume x1 is the best of the three
    if f(x1) > f(x0):
        x0, x1 = x1, x0
    if f(x1) > f(x2):
        x1, x2 = x2, x1
    
    for _ in range(max_iter):
        x0, x1, x2 = float(x0), float(x1), float(x2)
        if abs(x2 - x0) < 1e-6:
            break
        d1 = (x1 - x0) / (x2 - x0)
        d2 = (x1 - x2) / (x2 - x0)
        denom = d1 * d1 - d2 * d2
        if abs(denom) < 1e-15:
            break
        xm = x1 - (d1*d1*f(x1)*(x1-x0) - d2*d2*f(x1)*(x1-x2) + (x1-x0)*(x1-x2)*(f(x0)-f(x2))) / (2.0 * denom)
        # Actually let me use the standard parabolic formula
        # x_new = x1 - 0.5 * (f(x0)*(x1-x2)^2 - f(x2)*(x1-x0)^2) / (f(x0)*(x1-x2) - f(x2)*(x1-x0))
        # Hmm, this is getting complicated. Let me just use golden section.
        break
    
    # Fall back to golden section
    return golden_section_simple(f, min(x0, x1, x2), max(x0, x1, x2))


def golden_section_simple(f, lo, hi, max_iter=20):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-4:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def lp_alternate_multi(a, max_iters=40, time_budget=60.0):
    """LP alternating with multiple M values for diverse directions."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Try multiple M values
        best_g = None
        best_combo_obj = best_obj
        for m_factor in [1.0, 1.001, 0.999, 1.005]:
            if time.monotonic() - start > time_budget - 5:
                break
            M_try = M * m_factor
            g = solve_lp(a, M_try, time_limit=max(3, int(remaining * 0.4)))
            if g is None:
                continue
            
            # Find best blend
            def obj_at_t(t):
                a_try = normalize((1.0 - t) * a + t * g)
                return compute_obj(a_try)
            
            t_opt = golden_section_simple(obj_at_t, 0.0, 1.0, max_iter=15)
            a_new = normalize((1.0 - t_opt) * a + t_opt * g)
            obj_new = compute_obj(a_new)
            
            if obj_new < best_combo_obj:
                best_combo_obj = obj_new
                best_g = a_new
        
        if best_g is not None and best_combo_obj < best_obj - 1e-12:
            best_obj = best_combo_obj
            best_a = best_g.copy()
            a = best_g
        else:
            # Perturb to escape
            rng = np.random.default_rng((it * 7919 + 9999) % (2**31))
            sigma = 0.03 + 0.02 * (it % 7)
            a = normalize(best_a * np.exp(rng.normal(0, sigma, n)))
    
    return best_a


def logsumexp_gradient(a, max_steps=300, time_budget=60.0):
    """
    Gradient descent using log-sum-exp smooth approximation of max.
    Adaptive temperature with reheating.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 3 * n:
        size <<= 1
    
    tau = 0.5 / n  # Initial temperature
    step_size = 0.05
    no_improve_count = 0
    
    for step in range(max_steps):
        if time.monotonic() - start > time_budget - 2:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Adaptive temperature
        tau = max(0.1 / n, tau * 0.995)
        
        # Compute softmax weights over convolution
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (correlation)
        # More precisely: d/d a[i] of sum_k w[k] * conv[k] where conv[k] = sum_j a[j]*a[k-j]
        # = 2 * sum_k w[k] * a[k-i] for valid k-i
        # This is a correlation operation
        
        # Use FFT for efficiency
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n-1:2*n-1]
        
        # Center the gradient (project onto sum=const)
        grad_centered = grad - np.mean(grad)
        
        # Adaptive step size with backtracking
        for _ in range(10):
            a_new = normalize(a - step_size * grad_centered)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                no_improve_count = 0
                break
            else:
                step_size *= 0.7
                if step_size < 1e-8:
                    break
        
        if step_size < 1e-8:
            # Reheat
            step_size = 0.03
            tau = 0.5 / n
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            step_size = min(step_size * 1.05, 0.1)
    
    return best_a


def spectral_phase_iter(a, max_iters=80, time_budget=50.0):
    """
    Spectral method with phase correction for non-negativity.
    Alternates between:
    1. Flattening the power spectrum (reducing peak of |FFT|^2)
    2. Projecting to non-negative cone using a soft projection
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break
        
        # FFT
        A = np.fft.fft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        P = mag ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        
        # Flatten: P_new = P^alpha * p_max^(1-alpha) (geometric interpolation with flat)
        alpha = 0.6 + 0.3 * (it / max_iters)
        P_flat = np.ones(size) * p_max / size
        P_new = (P ** alpha) * (P_flat ** (1 - alpha))
        
        # Reconstruct
        mag_new = np.sqrt(P_new)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(A_new, size))[:n]
        
        # Soft non-negativity projection (Laplace approximation)
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            
            # Check if close to non-negative already (good solution)
            neg_frac = float(np.sum(a[~(a_new > 0)])) / s if s > 0 else 0
            a = a_new
        else:
            # Blend
            blend = 0.3 + 0.4 * (it / max_iters)
            a = normalize(blend * a_new + (1 - blend) * a)
    
    return best_a


def coordinate_descent_improved(a, time_budget=30.0):
    """
    Improved coordinate descent: targets the highest peak and moves mass
    from contributors to positions that reduce the peak most.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget - 2:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find top contributors to the peak
        contrib = []
        for j in range(min(n, k_star + 1)):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)
        
        if not contrib:
            break
        
        improved = False
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget - 3:
                break
            if j1 == j2:
                # Move mass from j1 to a random position
                for attempt in range(5):
                    j_target = int(rng.integers(0, n))
                    if j_target == j1:
                        continue
                    a_try = a.copy()
                    delta = a_try[j1] * rng.uniform(0.02, 0.08)
                    a_try[j1] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
            else:
                # Move mass from the pair (j1, j2) to random positions
                for attempt in range(3):
                    j_target = int(rng.integers(0, n))
                    if j_target == j1 or j_target == j2:
                        continue
                    a_try = a.copy()
                    delta1 = a_try[j1] * rng.uniform(0.01, 0.05)
                    delta2 = a_try[j2] * rng.uniform(0.01, 0.05)
                    a_try[j1] -= delta1
                    a_try[j2] -= delta2
                    a_try[j_target] += delta1 + delta2
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
        
        if not improved and rng.random() < 0.15:
            # Random perturbation
            a = normalize(a * np.exp(rng.normal(0, 0.04, n)))
    
    return best_a


def chirp_init(n):
    """Initialize with a chirp-like pattern (quadratic phase)."""
    x = np.arange(n) / n
    # Use a quadratic phase to create a spread-out autocorrelation
    phase = 2 * np.pi * (x ** 2) * (n // 4)
    a = 1.0 + 0.5 * np.cos(phase)
    return normalize(a)


def gaussian_init(n):
    """Initialize with a Gaussian-bump pattern."""
    x = np.arange(n) / n - 0.5
    sigma = 0.15
    a = np.exp(-x**2 / (2 * sigma**2))
    return normalize(a)


def multi_bump_init(n, n_bumps=3):
    """Initialize with multiple Gaussian bumps."""
    a = np.zeros(n)
    centers = np.linspace(0.1, 0.9, n_bumps)
    for c in centers:
        x = np.arange(n) / n - c
        a += np.exp(-x**2 / (2 * 0.05**2))
    return normalize(a)


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=30):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def blend_obj(a, g, t):
    a_try = normalize((1.0 - t) * a + t * g)
    return compute_obj(a_try)


def golden_section(f, lo, hi, max_iter=20):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-5:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def golden_blend(a, g, max_iter=18):
    """Golden section search for best blend t in [0,1] between a and g."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = blend_obj(a, g, c)
    fd = blend_obj(a, g, d)
    for _ in range(max_iter):
        if hi - lo < 1e-4:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = blend_obj(a, g, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = blend_obj(a, g, d)
    return (lo + hi) / 2.0

def logsumexp_gradient(a, max_steps=250, time_budget=60.0):
    """Log-sum-exp smoothed gradient descent with adaptive temperature and reheating."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 3 * n:
        size <<= 1

    tau = 0.5 / n
    step_size = 0.05
    no_improve = 0

    for step in range(max_steps):
        if time.monotonic() - start > time_budget - 2:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        tau = max(0.05 / n, tau * 0.994)

        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        grad_centered = grad - np.mean(grad)

        improved = False
        for _ in range(8):
            a_new = normalize(a - step_size * grad_centered)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
                improved = True
                break
            else:
                step_size *= 0.7
                if step_size < 1e-9:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
            if no_improve % 10 == 0:
                step_size = 0.03
                tau = 0.5 / n
                rng = np.random.default_rng(step * 31 + 7)
                a = normalize(best_a * np.exp(rng.normal(0, 0.02, n)))
        else:
            step_size = min(step_size * 1.03, 0.12)

    return best_a


def spectral_phase_iter(a, max_iters=70, time_budget=50.0):
    """Spectral method: flatten power spectrum and project to non-negative cone."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break

        A = np.fft.fft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        P = mag ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        alpha = 0.55 + 0.35 * (it / float(max_iters))
        P_flat = np.full(size, p_max / size)
        P_new = (P ** alpha) * (P_flat ** (1.0 - alpha))

        mag_new = np.sqrt(P_new)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(A_new, size))[:n]

        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.25 + 0.45 * (it / float(max_iters))
            a = normalize(blend * a_new + (1.0 - blend) * a)

    return best_a

def chirp_init(n):
    x = np.arange(n) / n
    phase = 2.0 * np.pi * (x ** 2) * (n // 4)
    a = 1.0 + 0.5 * np.cos(phase)
    return normalize(a)


def gaussian_init(n):
    x = np.arange(n) / n - 0.5
    sigma = 0.15
    a = np.exp(-x ** 2 / (2.0 * sigma ** 2))
    return normalize(a)


def multi_bump_init(n, n_bumps=3):
    a = np.zeros(n)
    centers = np.linspace(0.1, 0.9, n_bumps)
    for c in centers:
        x = np.arange(n) / n - c
        a += np.exp(-x ** 2 / (2.0 * 0.06 ** 2))
    return normalize(a)


def triangular_init(n):
    x = np.arange(n) / n
    a = np.minimum(x, 1.0 - x) * 4.0
    a = np.maximum(a, 0.01)
    return normalize(a)


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try2
            print(f"  ** n={n_try2}: {best_score:.8f} **")
        else:
            print(f"  n={n_try2}: {obj_try:.8f}")

    # Phase 6: Random restarts
    print(f"\n{'='*60}")
    print(f"Phase 6: Random restarts")
    print(f"{'='*60}")

    n_cur = len(best_seq)
    rng = np.random.default_rng(seed)

    for restart in range(12):
        if time.monotonic() > deadline - 70:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.09, 28.0)

        strategy = restart % 6
        if strategy == 0:
            a_try = best_seq * np.exp(rng.normal(0, 0.05, n_cur))
        elif strategy == 1:
            freq = rng.uniform(0.5, 10.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.18 * np.sin(2.0 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
            a_try = best_seq * mod
        elif strategy == 2:
            a_rand = rng.dirichlet(np.full(n_cur, 0.3))
            mix = rng.uniform(0.15, 0.35)
            a_try = (1.0 - mix) * normalize(best_seq) + mix * a_rand
        elif strategy == 3:
            a_try = best_seq.copy()
            block_len = int(rng.integers(n_cur // 5, max(n_cur // 5 + 1, n_cur // 3)))
            start_idx = int(rng.integers(0, max(1, n_cur - block_len)))
            a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]
        elif strategy == 4:
            a_try = best_seq.copy()
            k = int(rng.uniform(0.3, 0.6) * n_cur)
            idx = rng.choice(n_cur, size=k, replace=False)
            idx.sort()
            vals = a_try[idx].copy()
            rng.shuffle(vals)
            a_try[idx] = vals
        else:
            a_try = best_seq * np.exp(rng.normal(0, 0.03, n_cur))
            a_try[:n_cur // 4] *= 0.7
            a_try[-n_cur // 4:] *= 0.7

        a_try = normalize(a_try)

        if n_cur <= 900:
            a_try = lp_alternate_multi(a_try, max_iters=7, time_budget=alloc * 0.5)
        a_try = logsumexp_gradient(a_try, max_steps=45, time_budget=alloc * 0.4)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  Restart {restart} (s{strategy}): {best_score:.8f} **")
        else:
            print(f"  Restart {restart} (s{strategy}): {obj_try:.8f}")

    # Phase 7: Final polish
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")

    n_final = len(best_seq)
    a_final = normalize(best_seq.copy())

    for polish in range(5):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * 0.22, 35.0)

        if n_final <= 900 and alloc > 15:
            a_pol = lp_alternate_multi(a_final, max_iters=10, time_budget=alloc * 0.4)
        else:
            a_pol = a_final.copy()

        a_pol = logsumexp_gradient(a_pol, max_steps=70, time_budget=alloc * 0.35)

        if alloc > 12:
            a_pol = coordinate_descent_improved(a_pol, time_budget=min(8, alloc * 0.2))

        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            a_final = a_pol
            print(f"  Polish {polish}: {best_score:.8f} **")
        else:
            a_final = normalize(0.8 * a_final + 0.2 * a_pol)
            print(f"  Polish {polish}: {obj_pol:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s_total = sum(result)
    if s_total < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
