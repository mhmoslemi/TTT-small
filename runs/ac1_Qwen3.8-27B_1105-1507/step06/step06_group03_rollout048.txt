"""
Multi-method optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) projected subgradient with Polyak averaging, (2) enhanced coordinate
descent with continuous 1D optimization, (3) smooth gradient descent with Adam,
(4) multi-length exploration with structured initializations.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_method(a, max_iters, deadline, verbose=False):
    """
    Projected subgradient method with Polyak averaging.
    Minimizes max_k (conv(a,a)[k]) subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    
    # Initial objective for step size
    c0 = fast_conv(a)
    f0 = c0.max()
    
    a_avg = a.copy()
    best_a = a.copy()
    best_obj = compute_J(a)
    
    # Backtracking step size
    alpha = 0.05
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        k_star = np.argmax(c)
        
        # Compute subgradient
        # s_i = a[k_star - i] + a[i - k_star] (with out-of-range = 0)
        # Handle the case k_star = 2i separately (factor of 2)
        s = np.zeros(n)
        for i in range(n):
            val = 0.0
            j1 = k_star - i
            j2 = i - k_star
            if 0 <= j1 < n:
                val += a[j1]
            if 0 <= j2 < n:
                val += a[j2]
            s[i] = val
        
        # Projected gradient step
        a_new = project_simplex(a - alpha * s)
        
        # Evaluate
        obj_new = compute_J(a_new)
        
        # Polyak averaging
        a_avg = (it * a_avg + a_new) / (it + 1)
        obj_avg = compute_J(a_avg)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        elif obj_avg < best_obj:
            best_obj = obj_avg
            best_a = a_avg.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Decrease step size
            alpha = alpha * 0.95
        
        # Adapt step size
        if no_improve > 30:
            alpha = max(0.001, alpha * 0.8)
            no_improve = 0
        
        a = a_new
        
        if verbose and it % 100 == 0:
            print(f"    SubGrad it {it}: obj={best_obj:.8f} alpha={alpha:.5f}", flush=True)
    
    return best_a, best_obj
