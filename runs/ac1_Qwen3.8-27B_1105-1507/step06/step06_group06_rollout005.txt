"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Log-parameterized Adam optimizer for smooth non-negativity handling
2. Active-set LP (solves much faster than full LP by iterating on binding constraints)
3. Peak-equalization via weighted subgradient with adaptive temperature
4. Aggressive multi-length exploration (400-3000)
5. CMA-ES-inspired restart strategy for escaping local minima

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def fft_next_size(m):
    size = 1
    while size < m:
        size <<= 1
    return size


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def active_set_lp(a, max_rounds=8, top_k_init=15, time_limit=10.0):
    """
    Solve: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0
    Using active set method: start with top peaks, add violated constraints.
    Returns (g_normalized, t_opt) or (None, None).
    """
    n = len(a)
    if n > 2500:
        return None, None

    start = time.monotonic()
    conv_aa = compute_conv(a)
    num_constraints = 2 * n - 1

    # Build Toeplitz matrix A where (A@g)[k] = (a*g)[k]
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((num_constraints, n), dtype=np.float64)
    for j in range(n):
        k_max = min(j + n - 1, num_constraints - 1)
        A[j:k_max + 1, j] = a[:k_max - j + 1]

    # Initial active set: top peaks
    active = set(np.argsort(conv_aa)[-top_k_init:].tolist())

    for rnd in range(max_rounds):
        if time.monotonic() - start > time_limit:
            return None, None

        active_list = sorted(active)
        len_active = len(active_list)

        # LP: min t s.t. A[active]@g - t <= 0, sum(g)=1, g>=0, t>=0
        # Variables: [g_0,...,g_{n-1}, t]
        c = np.zeros(n + 1)
        c[-1] = 1.0

        A_ub = np.zeros((len_active, n + 1))
        A_ub[:, :n] = A[active_list, :]
        A_ub[:, -1] = -1.0
        b_ub = np.zeros(len_active)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0.0, None)] * n + [(0.0, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': min(5.0, time_limit - (time.monotonic() - start))})
        except Exception:
            return None, None

        if res.status != 0:
            return None, None

        g = np.maximum(res.x[:n], 0)
        s = np.sum(g)
        if s < 1e-10:
            return None, None
        g = g / s
        t_opt = float(res.x[n])

        # Check all constraints
        Ag = A @ g
        violated = np.where(Ag > t_opt + 1e-10)[0]

        if len(violated) == 0:
            return g, t_opt

        # Add most violated constraints
        violated_sorted = violated[np.argsort(Ag[violated])[::-1]]
        for idx in violated_sorted[:min(10, len(violated_sorted))]:
            active.add(int(idx))

    # Return best effort
    g = np.maximum(res.x[:n], 0)
    s = np.sum(g)
    if s > 1e-10:
        return g / s, float(res.x[n])
    return None, None


def log_param_adam(a, max_iters=200, time_budget=30.0, lr=0.02, beta1=0.9, beta2=0.999, tau=0.05):
    """
    Adam optimizer on log-parameterization x where a = softmax(x).
    Minimizes 2n * max(conv(a,a)) with sum(a)=1 enforced by softmax.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    # Initialize x such that softmax(x) ≈ a
    x = np.log(a + 1e-15)
    x -= np.max(x)  # for numerical stability

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = fft_next_size(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        t += 1

        # Compute a from x
        x_shifted = x - np.max(x)
        exp_x = np.exp(x_shifted)
        a = exp_x / np.sum(exp_x)

        # Compute convolution and gradient
        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights on convolution
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -50.0, 50.0)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            w = np.zeros(len(conv))
            w[idx] = 1.0
        else:
            w /= w_sum

        # Gradient of max(conv) w.r.t. a: grad_a[i] = sum_k w[k] * 2*a[k-i] * 1_{0<=k-i<n}
        grad_a = np.zeros(n)
        # Use FFT for efficiency: grad_a[i] = 2 * sum_k w[k] * a[k-i] 
        # This is a cross-correlation
        # sum_k w[k] * a[k-i] = (w * a_reverse)[n-1-i] ... let me just use conv
        # Actually: sum_k w[k]*a[k-i] where 0<=k-i<n means k is in [i, i+n-1]
        # = (w convolved with a) evaluated at position... 
        # Let's use the direct approach with FFT
        size_w = len(w)
        N_fft = fft_next_size(size_w + n - 1)
        fw = rfft(w, N_fft)
        fa_full = rfft(a, N_fft)
        # Cross-correlation: sum_k w[k]*a[k-i] = ifft(fft(w) * conj(fft(a))) shifted
        # Actually this gives sum_k w[k]*a[k+i] which is not quite right
        # Let me just compute grad_a directly using the convolution structure
        # grad_a[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
        # Let j = k-i, then k = i+j, j in [0, n-1]
        # grad_a[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
        # This is a cross-correlation of w and a
        # = ifft(fft(w) * conj(fft(a))) ... no
        # sum_j w[i+j]*a[j] = (w * a_flip)[?] 
        # Let's just use: grad_a = 2 * ifft(fft(w_padded) * conj(fft(a_padded))) ... 
        # Actually the simplest: use the fact that this is a sliding dot product
        # For efficiency with large n, use FFT-based cross-correlation
        # xcorr[w, a][i] = sum_j w[i+j] * a[j] for j=0..n-1
        # = sum_j w[n-1+j-i+n-1] ... this is getting confusing
        # Let me just do it with a loop over the significant w values
        # w is length 2n-1, a is length n
        # grad_a[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]  (for i+j < 2n-1)
        
        # Efficient: use FFT
        # Define w_rev[k] = w[2n-2-k]
        # Then sum_j w[i+j]*a[j] = sum_j w_rev[2n-2-i-j]*a[j]
        # = (w_rev * a)[2n-2-i]  (convolution)
        w_rev = w[::-1]
        N_c = fft_next_size(2 * n - 1)
        fwr = rfft(w_rev, N_c)
        fa2 = rfft(a, N_c)
        corr = irfft(fwr * fa2, N_c)
        # corr[m] = sum_j w_rev[j]*a[m-j] = sum_j w[2n-2-j]*a[m-j]
        # We want sum_j w[i+j]*a[j]. Let m = 2n-2-i, then:
        # corr[2n-2-i] = sum_j w_rev[j]*a[2n-2-i-j] = sum_j w[2n-2-j]*a[2n-2-i-j]
        # Let k = 2n-2-j, then j = 2n-2-k:
        # = sum_k w[k]*a[i+k-2n+2+2n-2] ... I'm getting confused.
        
        # Let me just do it the simple way for correctness (n up to ~3000 is fine)
        # grad_a[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
        # For efficiency, only iterate over significant w
        w_thresh = 1e-10
        significant = np.where(w > w_thresh)[0]
        if len(significant) < len(w) * 0.5:
            grad_a = np.zeros(n)
            for k_idx in significant:
                w_val = w[k_idx]
                j_max = min(n, 2 * n - 1 - k_idx)
                if j_max > 0:
                    i_range = range(max(0, k_idx - n + 1), k_idx + 1)
                    for i in i_range:
                        j = k_idx - i
                        if 0 <= j < n and 0 <= i < n:
                            grad_a[i] += w_val * a[j]
            grad_a *= 2.0
        else:
            # Full computation using the Toeplitz structure
            # grad_a[i] = 2 * sum_j w[i+j]*a[j]
            # Build at the end
            grad_a = np.zeros(n)
            for j in range(n):
                a_j = a[j]
                if a_j < 1e-15:
                    continue
                i_start = 0
                i_end = min(n, 2 * n - 1 - j)
                grad_a[i_start:i_end] += 2.0 * a_j * w[j:j + i_end - i_start]

        # Softmax gradient: df/dx[j] = a[j] * (df/da[j] - sum_i df/da[i]*a[i])
        mean_grad = np.dot(grad_a, a)
        grad_x = a * (grad_a - mean_grad)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr_t = lr * np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t)
        x -= lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)

        # Clamp x for numerical stability
        x = np.clip(x, -20.0, 20.0)

        # Compute objective
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def peak_equalize(a, max_iters=100, time_budget=20.0, tau_init=0.03, tau_end=0.01):
    """
    Weighted subgradient with adaptive temperature to equalize peaks.
    Uses Nesterov momentum for acceleration.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = 0.7
    v = np.zeros(n)
    step = 0.15

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        tau = tau_init + (tau_end - tau_init) * (it / max_iters)

        # Softmax weights
        shifted = np.clip((conv - cmax) / tau, -100.0, 100.0)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            w = np.zeros(len(conv))
            w[idx] = 1.0
        else:
            w /= w_sum

        # Gradient
        grad_a = np.zeros(n)
        w_thresh = 1e-8
        significant = np.where(w > w_thresh)[0]
        for k_idx in significant:
            w_val = w[k_idx]
            for i in range(n):
                j = k_idx - i
                if 0 <= j < n:
                    grad_a[i] += 2.0 * w_val * a[j]

        # Center gradient (for simplex projection)
        grad_centered = grad_a - np.mean(grad_a)

        # Nesterov momentum
        v = momentum * v - step * grad_centered
        a_new = a - v
        a_new = project_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Revert
            a_new = a.copy()

        step *= 0.995

    return best_a, best_obj


def lp_blend_step(a, max_iters=10, time_budget=30.0):
    """LP alternating: solve active-set LP, blend with current."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g, t_opt = active_set_lp(a, max_rounds=6, top_k_init=20,
                                  time_limit=min(5.0, time_budget - (time.monotonic() - start)))
        if g is None:
            break

        # Try several blend factors
        best_t = 0.5
        best_local_obj = best_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - alpha) * a + alpha * g
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_t = alpha
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_local_a.copy()
        else:
            break

    return best_a, best_obj


def chase_peak_fast(a, n_steps=30, step_size=0.03):
    """Fast chase-the-peak with vectorized operations."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: grad[i] = 2*a[k_star - i] if 0 <= k_star-i < n
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        j_start = k_star - i_end + 1
        j_end = k_star - i_start
        grad[i_start:i_end] = 2.0 * a[j_start:j_end]

        grad_centered = grad - np.mean(grad)

        improved = False
        for ss in [step_size, step_size * 0.5, step_size * 0.25]:
            a_try = a - ss * grad_centered
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Sine
    x = np.linspace(0, 1, n)
    sine = (np.sin(np.pi * x) ** 2 + 0.05)
    inits.append(sine / sine.sum())

    # Bimodal
    bimodal = np.exp(-((x - 0.3) ** 2) / 0.01) + np.exp(-((x - 0.7) ** 2) / 0.01) + 0.01
    inits.append(bimodal / bimodal.sum())

    # Random smooth
    rnd = rng.random(n)
    # Smooth with a kernel
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    rnd_smooth = np.convolve(rnd, kernel, mode='same')
    rnd_smooth = np.maximum(rnd_smooth, 0.01)
    inits.append(rnd_smooth / rnd_smooth.sum())

    # Tukey window
    r = 0.3
    window = np.ones(n)
    n_edge = int(n * r / 2)
    if n_edge > 0:
        edge = 0.5 * (1 + np.cos(np.pi * np.arange(1, n_edge + 1) / n_edge))
        window[:n_edge] = edge
        window[-n_edge:] = edge[::-1]
    inits.append(window / window.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.1
    inits.append(rc / rc.sum())

    if parent is not None:
        # Resampled parent
        p_resampled = resample_sequence(parent, n)
        inits.append(project_simplex(p_resampled))
        # Perturbed parent
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.15, n))
        inits.append(project_simplex(pert))

    return inits

                    print(f"  n={n_try}: NEW BEST {best_score:.8f}")

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates screened, best: {candidates[0][0] if candidates else 'N/A'}")

    # === Phase 2: Intensive optimization (55% time) ===
    print(f"\n=== Phase 2: Intensive Optimization (best={best_score:.8f}) ===")
    phase2_budget = min(800.0, (deadline - start_time) * 0.55)
    phase2_end = start_time + phase2_budget

    # Focus on top 3 candidates
    top_cands = [(best_score, best_seq.copy(), len(best_seq))]
    for obj_c, a_c, n_c in candidates[:2]:
        if obj_c < best_score + 0.01:
            top_cands.append((obj_c, a_c.copy(), n_c))

    for ci, (obj_init, a_init, n_init) in enumerate(top_cands):
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 20:
            break

        alloc = remaining / max(1, len(top_cands) - ci)
        a = project_simplex(a_init)
        local_obj = compute_obj(a)
        print(f"  Cand {ci} (n={n_init}): obj={local_obj:.8f}, budget={alloc:.0f}s")

        # Cycle through strategies
        strat_cycle = 0
        while time.monotonic() < phase2_end - 5:
            remaining = phase2_end - time.monotonic()
            if remaining < 8:
                break

            strat = strat_cycle % 4
            strat_cycle += 1

            if strat == 0:
                # Log-param Adam
                a_new, obj_new = log_param_adam(a, max_iters=50, time_budget=min(remaining * 0.3, 20.0),
                                                 lr=0.015, tau=0.04)
            elif strat == 1:
                # LP blending
                if n_init <= 1500:
                    a_new, obj_new = lp_blend_step(a, max_iters=8, time_budget=min(remaining * 0.3, 25.0))
                else:
                    a_new, obj_new = log_param_adam(a, max_iters=50, time_budget=min(remaining * 0.3, 20.0),
                                                     lr=0.01, tau=0.03)
            elif strat == 2:
                # Peak equalization with momentum
                a_new, obj_new = peak_equalize(a, max_iters=50, time_budget=min(remaining * 0.25, 15.0),
                                                tau_init=0.02, tau_end=0.005)
            else:
                # Chase peak + small Adam
                a_new, obj_new = chase_peak_fast(a, n_steps=20, step_size=0.02)
                a_new2, obj_new2 = log_param_adam(a_new, max_iters=30, time_budget=min(remaining * 0.2, 10.0),
                                                   lr=0.01, tau=0.03)
                if obj_new2 < obj_new:
                    a_new, obj_new = a_new2, obj_new2

            if obj_new < local_obj:
                local_obj = obj_new
                a = a_new.copy()
                if local_obj < best_score:
                    best_score = local_obj
                    best_seq = a.copy()
                    print(f"    ** Cand {ci} strat {strat}: obj={best_score:.8f} **")

    # === Phase 3: Diversification + re-convergence (20% time) ===
    print(f"\n=== Phase 3: Diversification (best={best_score:.8f}) ===")
    phase3_budget = min(300.0, (deadline - time.monotonic()) * 0.20)
    phase3_end = start_time + phase3_budget

    n_best = len(best_seq)

    # Try length adjustments
    n_adjusts = [n_best - 50, n_best + 50, n_best + 100, n_best - 100, n_best + 200]
    n_adjusts = [n for n in n_adjusts if 200 <= n <= 3000]
    n_adjusts = list(dict.fromkeys(n_adjusts))

    per_adj_time = max(5.0, phase3_budget / max(1, len(n_adjusts) + 1))

    # First: perturb current best
    a_pert = best_seq * np.exp(rng.normal(0, 0.05, n_best))
    a_pert = project_simplex(a_pert)
    a_pert, obj_pert = chase_peak_fast(a_pert, n_steps=20, step_size=0.02)
    a_pert, obj_pert = peak_equalize(a_pert, max_iters=30, time_budget=min(per_adj_time * 0.5, 10.0))
    if obj_pert < best_score:
        best_score = obj_pert
        best_seq = a_pert.copy()
        n_best = len(best_seq)
        print(f"  Perturbed: {best_score:.8f}")

    # Try different lengths
    for n_adj in n_adjusts:
        if time.monotonic() > phase3_end:
            break
        a_res = resample_sequence(best_seq, n_adj)
        a_res = project_simplex(a_res)
        a_res, obj_res = chase_peak_fast(a_res, n_steps=20, step_size=0.02)
        a_res, obj_res = peak_equalize(a_res, max_iters=30, time_budget=min(per_adj_time * 0.4, 10.0))
        if n_adj <= 1200:
            a_res, obj_res = lp_blend_step(a_res, max_iters=5, time_budget=min(per_adj_time * 0.3, 15.0))
        else:
            a_res, obj_res = log_param_adam(a_res, max_iters=30, time_budget=min(per_adj_time * 0.3, 10.0),
                                             lr=0.01, tau=0.03)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            n_best = len(best_seq)
            print(f"  n={n_adj}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_adj}: obj={obj_res:.8f}")

    # === Phase 4: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.8f}) ===")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 15 and best_seq is not None:
        a_final = project_simplex(best_seq.copy())
        n_final = len(a_final)
        polish_obj = compute_obj(a_final)

        print(f"  Polishing n={n_final} from {polish_obj:.8f}")

        polish_round = 0
        while time.monotonic() < deadline - 5:
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            polish_round += 1

            # Alternate: LP, Adam, PeakEq
            if n_final <= 1500 and polish_round % 3 != 2:
                a_p, obj_p = lp_blend_step(a_final, max_iters=5, time_budget=min(rem * 0.25, 20.0))
            else:
                a_p, obj_p = log_param_adam(a_final, max_iters=40, time_budget=min(rem * 0.25, 15.0),
                                             lr=0.008, tau=0.02)
            if obj_p < polish_obj:
                polish_obj = obj_p
                a_final = a_p.copy()

            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                a_p2, obj_p2 = peak_equalize(a_final, max_iters=30, time_budget=min(rem2 * 0.2, 10.0),
                                              tau_init=0.01, tau_end=0.003)
                if obj_p2 < polish_obj:
                    polish_obj = obj_p2
                    a_final = a_p2.copy()

            rem3 = deadline - time.monotonic()
            if rem3 > 3:
                a_p3, obj_p3 = chase_peak_fast(a_final, n_steps=15, step_size=0.01)
                if obj_p3 < polish_obj:
                    polish_obj = obj_p3
                    a_final = a_p3.copy()

            if polish_obj < best_score:
                best_score = polish_obj
                best_seq = a_final.copy()
                print(f"  Polish r{polish_round}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    # Convert to list of floats, clipped to [0, 1000]
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Validate
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * len(best_seq)
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
