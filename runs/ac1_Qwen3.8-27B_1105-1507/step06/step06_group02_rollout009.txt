Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_constraint_matrix(n, a):
    """Build sparse constraint matrix for LP: (a*g)(k) <= M for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    rows, cols, vals = build_constraint_matrix(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def anderson_mix(history_x, history_Fx, m):
    """Anderson mixing: find best convex combination of recent iterates."""
    if m < 2:
        return history_Fx[-1].copy()
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    # Allow mild extrapolation
    alpha = np.clip(alpha, -0.5, 2.0)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(history_Fx[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def gs_init(a0, n_iters=200, beta=0.85):
    """Gerchberg-Saxton spectral flattening."""
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-30)
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive constraint tightness (M scaling)
2. Backtracking line search for optimal step sizes
3. Anderson acceleration (window up to 10) for superlinear convergence
4. Multi-length exploration via interpolation from the parent sequence
5. Gerchberg-Saxton spectral initialization for diverse starting points
Key improvements: adaptive M (looser constraints give more LP flexibility),
backtracking instead of fixed step sizes, larger Anderson windows,

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_constraint_structure(n):
    """Precompute the (row, col) structure of the convolution constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def lp_best_response(a, M, structure=None):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0."""
    n = len(a)
    if structure is None:
        rows, cols = build_constraint_structure(n)
    else:
        rows, cols = structure
    vals = np.array([a[k - j] for k, j in zip(rows, cols)], dtype=np.float64)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9,
                             'time_limit': 5.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(hist_x, hist_Fx, m):
    """Anderson mixing: find best combination of recent (x, F(x)) pairs."""
    if m < 2:
        return hist_Fx[-1].copy()
    n_dim = len(hist_Fx[0])
    residuals = np.zeros((m, n_dim))
    for i in range(m):
        residuals[i] = hist_x[i] - hist_Fx[i]
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 2.0)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(n_dim)
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    return np.maximum(mixed, 0)


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def gs_init(a0, n_iters=150, beta=0.85):
    """Gerchberg-Saxton spectral flattening for initialization."""
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Soft compression
        mag_target = mag ** 0.5
        mag_new = (1.0 - beta) * mag + beta * (mag_target / (mag_target.max() + 1e-30)) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = float(fast_conv(a_candidate).max())
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a

    print(f"Phase 2: J={tracker.score:.8f} t={time.monotonic()-t0:.0f}s", flush=True)

    # === PHASE 3: Second LP pass with different parameters (200s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Second LP pass (different params)", flush=True)
    print(f"{'='*60}", flush=True)
    if time.monotonic() < deadline - 80:
        p3_end = min(time.monotonic() + 180, deadline - 60)
        a3, M3 = run_lp_adaptive(tracker.seq, p3_end, max_iters=500,
                                  anderson_window=12,
                                  m_scales=(1.0, 1.001, 1.004, 1.01, 1.02),
                                  structure=struct_parent)
        tracker.update(a3)
        print(f"Phase 3: J={tracker.score:.8f} t={time.monotonic()-t0:.0f}s", flush=True)

    # === PHASE 4: Length exploration (150s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    n_best = tracker.seq.size
    best_per_len = {}
    for d in [32, -32, 64, -64, 128, -128, 256, -256, 512, -512]:
        if time.monotonic() > deadline - 50:
            break
        n_try = n_best + d
        if n_try < 256 or n_try > 3000:
            continue
        a_res = resample_seq(tracker.seq, n_try)
        a_res = np.maximum(a_res, 1e-8)
        a_res /= a_res.sum()
        struct_try = build_constraint_structure(n_try)
        sub_dl = min(time.monotonic() + 25, deadline - 30)
        a_opt, _ = run_lp_adaptive(a_res, sub_dl, max_iters=60,
                                    anderson_window=6, m_scales=(1.0, 1.005),
                                    structure=struct_try)
        tracker.update(a_opt)
        J_try = compute_J(a_opt)
        best_per_len[n_try] = (a_opt.copy(), J_try)
        print(f"  n={n_try}: J={J_try:.8f}", flush=True)

    # === PHASE 5: Polish the best non-parent length if found ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Polish best alternative length", flush=True)
    print(f"{'='*60}", flush=True)
    if best_per_len and time.monotonic() < deadline - 30:
        # Find best alternative (not parent length)
        alts = [(n_try, a, J) for n_try, (a, J) in best_per_len.items() if n_try != n_best]
        if alts:
            alts.sort(key=lambda x: x[2])
            n_alt, a_alt, J_alt = alts[0]
            if J_alt < tracker.score:
                struct_alt = build_constraint_structure(n_alt)
                sub_dl = min(time.monotonic() + 40, deadline - 10)
                a_pol, _ = run_lp_with_anneal(a_alt, sub_dl, max_iters=300,
                                               anderson_window=8,
                                               m_scales=(1.0, 1.005),
                                               structure=struct_alt,
                                               t_start=0.3, t_end=0.05)
                tracker.update(a_pol)

    # === PHASE 6: Final polish of overall best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final polish", flush=True)
    print(f"{'='*60}", flush=True)
    if time.monotonic() < deadline - 5:
        n_final = tracker.seq.size
        struct_final = build_constraint_structure(n_final)
        a_fin, _ = run_lp_adaptive(tracker.seq, deadline, max_iters=100,
                                    anderson_window=10, m_scales=(1.0, 1.001),
                                    structure=struct_final)
        tracker.update(a_fin)

    # Fallback
    if tracker.seq is None:
        tracker.seq = np.ones(128)
        tracker.score = compute_J(tracker.seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.score:.8f} n={tracker.seq.size} time={time.monotonic()-t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.seq]
