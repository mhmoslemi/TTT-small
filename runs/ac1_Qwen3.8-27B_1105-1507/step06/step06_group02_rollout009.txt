"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth Nesterov optimization for initialization
2. LP best-response with Anderson acceleration and adaptive line search

4. Multi-scale refinement and length exploration
Key: More aggressive LP convergence with line search and bundle method.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    lo = v.min() - target_sum
    hi = v.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.maximum(v - mid, 0.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.maximum(v - theta, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.full(n, target_sum / n)
    return x * (target_sum / s)

def smooth_max_grad(a, tau):
    """Gradient of smooth approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(2 * n - 1)  # gradient wrt conv values
    # d(conv(a,a))/da is a 2*(2n-1) x n matrix
    # grad_a = 2 * sum_k p[k] * (d conv[k] / da)
    grad_a = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        # d conv[k] / da[i] = a[k-i] if 0 <= k-i < n
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            grad_a[i] += 2.0 * p[k] * a[k - i]
    return grad_a, p, b, b_max

def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.4, lr_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    mu = 1.0  # Nesterov momentum parameter
    beta_mom = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_mom * beta_mom)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta_mom * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a

def build_A_values(n, a):
    """Build sparse matrix for LP: (a*g)[k] = sum_i a[k-i]*g[i]"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return rows, cols, vals

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    rows, cols, vals = build_A_values(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def anderson_mix(history_x, history_Fx, m, gamma=1.0):
    """Anderson mixing: find best convex combination of Fx values."""
    if m < 2:
        return history_Fx[-1].copy()
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T + 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 1.5)
    alpha_sum = alpha.sum()
    if abs(alpha_sum) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / alpha_sum
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)

def run_lp_anderson_line_search(a, deadline, max_iters=300, anderson_window=5,

    if ls_factors is None:

    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    hist_x = []
    hist_Fx = []
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break
        
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Anderson direction
        direction = g
        hist_x.append(a.copy())
        hist_Fx.append(g.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        if len(hist_x) >= 3:
            mixed = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sm = mixed.sum()
            if sm > 1e-10:
                mixed /= sm
            # Use mixed as direction if it's better
            M_mixed = fast_conv(np.maximum((1-0.1)*a + 0.1*mixed, 0)).max()
            M_g = fast_conv(np.maximum((1-0.1)*a + 0.1*g, 0)).max()
            if M_mixed <= M_g * 1.01:
                direction = mixed
        
        # Line search for step size
        best_t = t_init
        best_M_t = fast_conv(np.maximum((1 - t_init) * a + t_init * direction, 0)).max()
        
        for f in ls_factors:
            t_try = best_t * f
            if t_try < t_min or t_try > 1.0:
                continue
            a_try = np.maximum((1.0 - t_try) * a + t_try * direction, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:

                best_M_t = M_try

        a_new = np.maximum((1.0 - best_t) * a + best_t * direction, 0.0)

def run_lp_plain_with_ls(a, deadline, max_iters=100):
    """Plain LP with line search for final polish."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    t = 0.3
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        improved = False
        for t_try in [t, t*0.5, t*2, t*0.25, t*0.1, t*4, t*0.05]:
            if t_try > 1.0 or t_try < 0.0001:
                continue
            a_t = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
            s_t = a_t.sum()
            if s_t < 1e-10:
                continue
            a_t /= s_t
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                t = t_try
                improved = True
                break
        
        if improved:
            no_imp = 0
            t = max(0.0001, t * 0.8)  # gradually decrease
        else:
            no_imp += 1
            t *= 0.7
            if no_imp > 20:
                break
    
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
