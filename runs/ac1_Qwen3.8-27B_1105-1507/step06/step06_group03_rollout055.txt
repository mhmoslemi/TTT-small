One more thought: the "Nelder-Mead on DCT subspace" idea is interesting because:
- DCT basis vectors are smooth (low-frequency components) and oscillatory (high-frequency)
- Low-frequency DCT components represent "global shape" changes
- High-frequency DCT components represent "local" changes
- By optimizing in a DCT basis, we can make both global and local changes efficiently

"""
Multi-strategy optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Combines: (1) improved coordinate descent with analytical 1D subproblem solving,
(2) Nelder-Mead optimization in DCT subspaces, (3) iterated local search with
structured perturbations, (4) multi-length exploration, (5) final polish.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def nelder_mead_dct(base, deadline, dim=15, max_iters=500, verbose=False):
    """
    Nelder-Mead optimization in a DCT subspace.
    Parametrizes a = project_simplex(base + V @ alpha) where V contains
    the first 'dim' DCT-II basis vectors.
    """
    n = len(base)
    base = np.maximum(base.astype(np.float64), 0)
    base = base / base.sum()

    # Build DCT-II basis vectors
    # DCT-II: v_k[i] = cos(pi * k * (2*i + 1) / (2*n))
    t = np.arange(n)
    V = np.zeros((n, dim))
    for k in range(dim):
        V[:, k] = np.cos(np.pi * k * (2 * t + 1) / (2 * n))
    # Normalize
    for k in range(dim):
        norm = np.linalg.norm(V[:, k])
        if norm > 1e-10:
            V[:, k] /= norm

    def objective(alpha):
        a_candidate = base + V @ alpha
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-10:
            return 1e10
        a_candidate /= s
        return compute_J(a_candidate)

    # Initial simplex
    alpha0 = np.zeros(dim)
    f0 = objective(alpha0)

    # Build initial Nelder-Mead simplex
    simplex = [alpha0.copy()]
    for k in range(dim):
        pert = np.zeros(dim)
        pert[k] = 0.05
        simplex.append(pert.copy())

    best_alpha = alpha0.copy()
    best_obj = f0

    # Run Nelder-Mead
    result = minimize(objective, alpha0, method='Nelder-Mead',
                      options={'maxiter': max_iters, 'xatol': 1e-8, 'fatol': 1e-10,
                               'adaptive': True, 'initial_simplex': simplex})

    obj_result = result.fun
    if obj_result < best_obj:
        best_obj = obj_result
        best_alpha = result.x.copy()

    a_final = base + V @ best_alpha
    a_final = np.maximum(a_final, 0)
    if a_final.sum() > 1e-10:
        a_final /= a_final.sum()

    return a_final, best_obj


def structured_perturb(base, rng, scale=0.05):
    """Apply a structured perturbation to the base sequence."""
    n = len(base)
    t = np.linspace(0, 1, n, endpoint=False)
    pert_type = rng.integers(0, 5)

    if pert_type == 0:
        # Sinusoidal perturbation
        freq = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        pert = scale * np.sin(2 * np.pi * freq * t + phase)
    elif pert_type == 1:
        # Gaussian bump add/remove
        center = rng.uniform(0.1, 0.9)
        width = rng.uniform(0.05, 0.3)
        amp = scale * rng.uniform(-1, 1)
        pert = amp * np.exp(-((t - center) / width) ** 2)
    elif pert_type == 2:
        # Smooth random (low-pass filtered noise)
        raw = rng.normal(0, scale, n)
        ksize = max(5, n // 8)
        kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) / (ksize / 3)) ** 2)
        kernel /= kernel.sum()
        pert = np.convolve(raw, kernel, mode='same')
    elif pert_type == 3:
        # Block perturbation
        n_blocks = rng.integers(3, 10)
        block_size = n // n_blocks
        pert = np.zeros(n)
        for b in range(n_blocks):
            start = b * block_size
            end = min(start + block_size, n)
            pert[start:end] = rng.normal(0, scale / 2)
    else:
        # Anti-symmetric perturbation (helps reduce center lag)
        freq = rng.uniform(0.5, 3)
        pert = scale * np.sin(np.pi * freq * (2 * t - 1))

    a_new = base + pert
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return base.copy()
    return a_new / s


def ilp_explore(base, deadline, verbose=False):
    """
    Iterated Local Search: perturb + CD, keep best.
    Uses structured perturbations of increasing then decreasing scale.
    """
    n = len(base)
    best_a = base.copy()
    best_obj = compute_J(base)
    rng = np.random.default_rng(12345)
    no_improve = 0

    # Schedule: start with larger perturbations, decrease over time
    n_iterations = 0
    start_time = time.monotonic()

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        total_budget = deadline - start_time
        frac = min(1.0, elapsed / max(1, total_budget))

        # Perturbation scale: starts large, decreases
        base_scale = 0.15 * (1.0 - frac) + 0.02
        scale = base_scale * rng.uniform(0.5, 1.5)

        # Generate perturbed candidate
        a_pert = structured_perturb(base, rng, scale)

        # Run CD on perturbed candidate with limited time
        cd_deadline = time.monotonic() + min(30, (deadline - time.monotonic()) * 0.3)
        if cd_deadline > deadline - 5:
            cd_deadline = deadline - 2

        a_opt = cd_exact(a_pert, max_sweeps=2000, deadline=cd_deadline,
                         random_order=True, seed=rng.integers(100000))

        obj_opt = compute_J(a_opt)
        n_iterations += 1

        if obj_opt < best_obj - 1e-10:
            best_a = a_opt.copy()
            best_obj = obj_opt
            no_improve = 0
            if verbose:
                print(f"    ILS iter {n_iterations}: NEW BEST J={obj_opt:.8f}", flush=True)
        else:
            no_improve += 1

        if verbose and n_iterations % 10 == 0:
            print(f"    ILS iter {n_iterations}: best={best_obj:.8f} scale={scale:.4f}", flush=True)

        if no_improve > 50:
            break

    return best_a, best_obj


def sca_lp_reduced(a, deadline, n_constraints=200, max_iters=50, verbose=False):
    """
    SCA-LP with reduced constraints (only top-k binding constraints).
    Much faster than full LP for large n.
    """
    n = len(a)
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Select top constraints (most binding)
        k_vals = np.argsort(c)[::-1][:n_constraints]

        rows = []
        cols = []
        vals = []
        b_ub = []

        for idx, k in enumerate(k_vals):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(idx)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(len(k_vals), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=np.array(b_ub), A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [0.5, 0.3, 0.2, 0.1, 0.05]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 5 == 0:
            print(f"    SCA-LP it {it}: obj={best_obj:.8f}", flush=True)

        if no_improve > 15:
            break

    return best_a, best_obj


def try_length(a_best, new_n, deadline, verbose=False):
    """Try optimizing at a different length."""
    a_resampled = resample_seq(a_best, new_n)
    a_resampled = np.maximum(a_resampled, 1e-8)
    a_resampled = a_resampled / a_resampled.sum()

    # CD at new length
    cd_deadline = time.monotonic() + min(60, (deadline - time.monotonic()) * 0.4)
    a_opt = cd_exact(a_resampled, max_sweeps=3000, deadline=cd_deadline,
                     random_order=True, seed=777)

    # Nelder-Mead refinement
    nm_deadline = time.monotonic() + min(30, (deadline - time.monotonic()) * 0.3)
    if time.monotonic() < nm_deadline:
        a_nm, obj_nm = nelder_mead_dct(a_opt, deadline=nm_deadline, dim=10, max_iters=200)
        if obj_nm < compute_J(a_opt):
            a_opt = a_nm

    return a_opt


def generate_structured_inits(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            p /= ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp (exploits asymmetry to reduce center lag)
    inits.append(project_simplex(np.linspace(0.1, 1.0, n)))
    inits.append(project_simplex(np.linspace(1.0, 0.1, n)))

    # Quadratic
    inits.append(project_simplex(t ** 2 + 0.1))
    inits.append(project_simplex((1 - t) ** 2 + 0.1))

    # Cosine window
    for alpha in [0.5, 1.0, 2.0]:
        inits.append(project_simplex(np.cos(np.pi * alpha * (t - 0.5)) ** 2 + 0.05))

    # Sine patterns
    for freq in [1, 2, 3]:
        inits.append(project_simplex(1.0 + 0.3 * np.sin(2 * np.pi * freq * t)))
        inits.append(project_simplex(1.0 + 0.5 * np.cos(2 * np.pi * freq * t)))

    # Exponential decay
    for rate in [1, 2, 4]:
        inits.append(project_simplex(np.exp(-rate * t) + 0.05))
        inits.append(project_simplex(np.exp(-rate * (1 - t)) + 0.05))

    # Two-sided exponential
    inits.append(project_simplex(np.exp(-3 * (t - 0.5) ** 2)))

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) / (ksize / 3)) ** 2)
    kernel /= kernel.sum()
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0.5, 1.5, n), kernel, mode='same')
        inits.append(project_simplex(smooth))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses: improved CD, Nelder-Mead in DCT subspace, ILS, SCA-LP, multi-length.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Initial optimization at parent length (200s) ===
    phase1_deadline = t_start + 200.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: CD + DCT-NM at n={n_parent} (budget {phase1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # CD from parent
    a1 = cd_exact(parent, max_sweeps=5000, deadline=t_start + 100.0,
                  random_order=True, seed=42)
    update_best(a1)
    print(f"  After CD: J={compute_J(a1):.8f}", flush=True)

    # Nelder-Mead in DCT subspace
    if time.monotonic() < phase1_deadline - 30:
        a2, obj2 = nelder_mead_dct(a1, deadline=phase1_deadline - 10, dim=20, max_iters=500)
        update_best(a2)
        print(f"  After DCT-NM: J={obj2:.8f}", flush=True)

    # === PHASE 2: ILS with structured perturbations (400s) ===
    phase2_deadline = t_start + 600.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Iterated Local Search (budget {phase2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        a3, obj3 = ilp_explore(best_seq, deadline=phase2_deadline - 20, verbose=True)
        update_best(a3)
        print(f"  After ILS: J={obj3:.8f}", flush=True)

    # === PHASE 3: SCA-LP reduced (200s) ===
    phase3_deadline = t_start + 800.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: SCA-LP (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        a4, obj4 = sca_lp_reduced(best_seq, deadline=phase3_deadline - 10,
                                  n_constraints=150, max_iters=40, verbose=True)
        update_best(a4)
        print(f"  After SCA-LP: J={obj4:.8f}", flush=True)

    # === PHASE 4: Second ILS round with tighter perturbations (200s) ===
    phase4_deadline = t_start + 1000.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: ILS round 2 (budget {phase4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        a5, obj5 = ilp_explore(best_seq, deadline=phase4_deadline - 20, verbose=True)
        update_best(a5)
        print(f"  After ILS2: J={obj5:.8f}", flush=True)

    # === PHASE 5: Multi-length exploration (250s) ===
    phase5_deadline = t_start + 1250.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Multi-length (budget {phase5_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        candidate_ns = [n_parent - 32, n_parent - 16, n_parent + 16, n_parent + 32,
                        n_parent - 64, n_parent + 64, 700, 800, 850]
        candidate_ns = sorted(set(n for n in candidate_ns if n >= 128 and n <= 2048))

        for new_n in candidate_ns[:6]:
            remaining = phase5_deadline - time.monotonic()
            if remaining < 40 or new_n == n_parent:
                continue
            alloc = min(40.0, remaining - 10)
            if alloc < 20:
                break
            print(f"  Trying n={new_n}...", flush=True)
            a_new = try_length(best_seq, new_n, deadline=time.monotonic() + alloc)
            update_best(a_new)
            print(f"    n={new_n}: J={compute_J(a_new):.8f}", flush=True)

    # === PHASE 6: Final intensive polish (remaining time) ===
    phase6_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 6: Final polish (budget {phase6_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase6_deadline - time.monotonic()
        a_curr = best_seq.copy()

        # Long CD
        if remaining > 60:
            t_cd = time.monotonic() + 0.35 * remaining
            a_curr = cd_exact(a_curr, max_sweeps=20000, deadline=t_cd,
                              random_order=True, seed=999)
            update_best(a_curr)
            remaining = phase6_deadline - time.monotonic()

        # DCT-NM
        if remaining > 40:
            t_nm = time.monotonic() + 0.30 * remaining
            a_nm, obj_nm = nelder_mead_dct(a_curr, deadline=t_nm, dim=25, max_iters=400)
            update_best(a_nm)
            remaining = phase6_deadline - time.monotonic()

        # Another SCA-LP
        if remaining > 40:
            t_lp = time.monotonic() + 0.20 * remaining
            a_lp, obj_lp = sca_lp_reduced(best_seq, deadline=t_lp,
                                          n_constraints=100, max_iters=20)
            update_best(a_lp)
            remaining = phase6_deadline - time.monotonic()

        # Final CD
        if remaining > 20:
            a_final = cd_exact(best_seq, max_sweeps=10000, deadline=phase6_deadline - 5,
                               random_order=True, seed=2024)
            update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
