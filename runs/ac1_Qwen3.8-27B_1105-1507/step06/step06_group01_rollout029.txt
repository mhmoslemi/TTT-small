And sum a[j]^2 >= (sum a[j])^2 / n = 1/n (by Cauchy-Schwarz again, with equality iff a is uniform).

So c[n-1] can be as low as... well, we need sum a[j]^2 to be small, which means a should be spread out. But if a is spread out (uniform), then c[n-1] = 1/n.

The issue is that for non-uniform a, some other c[k] might be larger.

        A = csr_matrix((vals, (rows, cols)), shape=(len(active_list), n))
        c = -np.ones(n)
        b_ub = np.full(len(active_list), M)
        bounds = [(0.0, None)] * n
        
        try:
            tl = min(3.0, max(0.5, deadline - time.monotonic() - 1.0))
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': tl, 'presolve': True})
            if not r.success or r.x is None:
                break
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g = g / g_sum
        except Exception:
            break
        
        # Check for violated constraints: compute (a*g)[k] for all k
        # (a*g)[k] = sum_j a[j] * g[k-j]
        # Use FFT for efficiency
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = np.fft.rfft(a, size)
        fg = np.fft.rfft(g, size)
        ag = np.fft.irfft(fa * fg, size)[:2 * n - 1]
        
        violations = ag - M
        max_viol = violations.max()
        
        if max_viol < 1e-10:
            # No violations: exact best response found
            return g
        
        # Add most violated constraint
        new_k = int(np.argmax(violations))
        if new_k not in active_set:
            active_set.add(new_k)
            # Cap active set size
            if len(active_set) > max_active:
                # Keep top max_active by violation
                viol_arr = ag.copy()
                for ak in active_set:
                    pass  # keep as is for now
                # Actually just keep all - it's bounded by 2n-1
                if len(active_set) > max_active:
                    # Remove least important (smallest ag value)
                    active_arr = np.array(list(active_set))
                    ag_vals = ag[active_arr]
                    remove_mask = np.argsort(ag_vals)[:len(active_set) - max_active]
                    for ri in remove_mask:
                        active_set.discard(int(active_arr[ri]))
    
    return g


def run_lp_mirror(a_init, deadline, n_active_start=10, n_active_end=80, max_iters=300):
    """LP mirror descent: alternate between LP best-response and mix update."""
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    current_active = n_active_start
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = lp_with_cuts(a, min(deadline, time.monotonic() + 5), max_active=current_active, max_iters=20)
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            current_active = min(int(current_active * 1.5), 2 * n - 1)
            continue
        
        # Line search: try different mix parameters
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
            # Slightly reduce active set (tighter LP)
            if current_active > n_active_start:
                current_active = max(n_active_start, int(current_active * 0.9))
        else:
            no_improve += 1
            current_active = min(int(current_active * 1.3), 2 * n - 1)
            if no_improve > 20:
                break
    
    return best_a


def coordinate_descent_focused(a_init, deadline, max_sweeps=50):
    """Coordinate descent focusing on coordinates that affect the max convolution."""
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        # Find coordinates that contribute to the top convolution values
        top_k = np.argsort(b)[::-1][:min(10, len(b))]
        
        # Determine which a[i] values affect these k's
        active_coords = set()
        for k in top_k:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                active_coords.add(j)
            # Also k-j
            for j in range(j_lo, j_hi + 1):
                active_coords.add(k - j)
        
        # Process active coordinates in random order
        coords = list(active_coords)
        np.random.shuffle(coords)
        
        sweep_improved = False
        for i in coords:
            if time.monotonic() > deadline:
                break
            
            x_cur = a[i]
            # Compute affected b values
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Base values (without x_cur contribution)
            c_vals = []
            slopes = []
            for k in range(k_lo, k_hi + 1):
                j = k - i
                if 0 <= j < n:
                    if j == i:
                        c_vals.append(b[k] - x_cur * x_cur)
                        slopes.append(0.0)  # quadratic: x^2
                    else:
                        c_vals.append(b[k] - 2.0 * x_cur * a[j])
                        slopes.append(2.0 * a[j])
            
            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            
            c_arr = np.array(c_vals) if c_vals else np.array([0.0])
            s_arr = np.array(slopes) if slopes else np.array([0.0])
            
            def obj_at(x_test):
                vals = c_arr + s_arr * x_test
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                # Check quadratic term (k=2i)
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    q_idx = 2 * i - k_lo
                    qv = c_arr[q_idx] + x_test * x_test
                    if qv > m:
                        m = qv
                return m
            
            # Golden section search
            lo = 0.0
            hi = max(3.0 * x_cur, 0.5 / n + 0.1, 0.05)
            
            # Expand hi if needed
            for _ in range(5):
                if obj_at(hi) > obj_at(hi * 0.5) and hi < 5.0:
                    hi *= 1.5
                else:
                    break
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_at(a_pt)
            fb = obj_at(b_pt)
            
            for _ in range(30):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_at(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)
            
            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt) + 1e-14:
                x_opt = 0.0
            
            if abs(x_opt - x_cur) > 1e-10 * max(1.0, x_cur):
                # Update b incrementally
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if j == i:
                            b[k] = b[k] - x_cur * x_cur + x_opt * x_opt
                        else:
                            b[k] = b[k] - 2.0 * x_cur * a[j] + 2.0 * x_opt * a[j]
                a[i] = max(0.0, x_opt)
                sweep_improved = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved or no_improve > 15:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p1 = np.maximum(p + rng.normal(0, 0.02, n), 1e-4)
        inits.append(p1 / p1.sum())
        p2 = np.maximum(p + rng.normal(0, 0.08, n), 1e-4)
        inits.append(p2 / p2.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    rc = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    rc = np.maximum(rc, 0.01)
    inits.append(rc / rc.sum())
    
    # Raised sine (alternative)
    rs = 0.5 + 0.5 * np.sin(np.pi * t)
    rs = np.maximum(rs, 0.01)
    inits.append(rs / rs.sum())
    
    # Double-hump
    dh = np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.05
    inits.append(dh / dh.sum())
    
    # Triangular
    tri = 1 - 2 * np.abs(t - 0.5)
    tri = np.maximum(tri, 0.05)
    inits.append(tri / tri.sum())
    
    # Pseudo-random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # "Anti-peak" init: try to equalize autocorrelation
    a_ap = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_ap)
        # Reduce a where it contributes most to the peak
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_ap[j]
        a_ap = a_ap - 0.05 * (corr - corr.mean())
        a_ap = np.maximum(a_ap, 0.001)
        a_ap = a_ap / a_ap.sum()
    inits.append(a_ap.copy())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_full_optimization(a0, n, deadline, seed):
    """Run the full optimization pipeline on a single initialization."""
    best = np.maximum(a0.copy(), 0.0)
    s = best.sum()
    if s < 1e-10:
        best = np.ones(n) / n
    else:
        best = best / s
    best_M = fast_conv(best).max()
    
    t_start = time.monotonic()
    phase_time = max(1.0, deadline - t_start)
    
    # Phase A: Dual-primal FISTA (40% of time)
    t_a = t_start + 0.4 * phase_time
    if time.monotonic() < t_a:
        tau_phases = [
            (20, 2000),
            (50, 2000),
            (100, 1500),
            (200, 1000),
            (400, 800),
        ]
        a_opt = run_dual_primal_fista(best, t_a, tau_phases, lr_init=0.6)
        M_opt = fast_conv(a_opt).max()
        if M_opt < best_M:
            best_M = M_opt
            best = a_opt.copy()
        print(f"    After FISTA: M={best_M:.10f}", flush=True)
    
    # Phase B: LP mirror descent (30% of time)
    t_b = t_start + 0.7 * phase_time
    if time.monotonic() < t_b:
        a_lp = run_lp_mirror(best, t_b, n_active_start=8, n_active_end=60, max_iters=200)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
        print(f"    After LP: M={best_M:.10f}", flush=True)
    
    # Phase C: Coordinate descent (20% of time)
    t_c = t_start + 0.9 * phase_time
    if time.monotonic() < t_c:
        max_sw = max(5, int((t_c - time.monotonic()) / max(0.05, n / 200.0)))
        a_cd = coordinate_descent_focused(best, t_c, max_sweeps=max_sw)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
        print(f"    After CD: M={best_M:.10f}", flush=True)
    
    # Phase D: Final LP polish (10%)
    if time.monotonic() < deadline:
        a_final = run_lp_mirror(best, deadline, n_active_start=15, n_active_end=100, max_iters=50)
        M_final = fast_conv(a_final).max()
        if M_final < best_M:
            best_M = M_final
            best = a_final.copy()
    
    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining dual-primal FISTA, LP best-response with cuts,
    and coordinate descent across multiple initializations and sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)
    
    n_parent = parent.size
    
    # Determine lengths to try
    lengths_to_try = [n_parent]
    for delta in [-64, -128, 64, 128, 256, -256]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl not in lengths_to_try:
            lengths_to_try.append(nl)
    if 768 not in lengths_to_try:
        lengths_to_try.append(768)
    
    # Sort: parent first, then by distance
    lengths_to_try.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    
    # Limit to 4 lengths max
    lengths_to_try = lengths_to_try[:4]
    
    print(f"\n{'='*60}", flush=True)
    print(f"Lengths to try: {lengths_to_try}", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Allocate time: 60% for multi-length search, 25% for intensive polish, 15% for final
    time_search = 0.6 * (deadline - time.monotonic())
    time_polish = 0.25 * (deadline - time.monotonic())
    time_final = 0.15 * (deadline - time.monotonic())
    
    # === Phase 1: Multi-length, multi-init search ===
    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - time_polish - time_final:
            break
        
        t0 = time.monotonic()
        remaining = deadline - time_polish - time_final - time.monotonic()
        if remaining < 30:
            break
        
        # More time for parent length
        time_alloc = remaining / max(1, len(lengths_to_try) - li)
        if n == n_parent:
            time_alloc *= 1.5
        time_alloc = min(time_alloc, remaining)
        
        print(f"\n  Length n={n}, time_alloc={time_alloc:.0f}s", flush=True)
        
        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        
        # Limit inits to fit time budget
        time_per_init = time_alloc / max(1, len(inits))
        max_inits = max(2, int(time_alloc / 15))  # at least 15s per init
        inits = inits[:max_inits]
        
        best_for_n = None
        best_M_for_n = float('inf')
        
        for ii, a0 in enumerate(inits):
            if time.monotonic() > deadline - time_polish - time_final:
                break
            
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            
            sub_deadline = min(t0 + time_alloc, deadline - time_polish - time_final)
            if time.monotonic() >= sub_deadline:
                break
            
            a_opt = run_full_optimization(a0, n, sub_deadline, seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()
            
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            tracker.update(best_for_n)
            print(f"  n={n}: best J={tracker.best_score:.6f} (elapsed={time.monotonic()-t0:.1f}s)", flush=True)
    
    tracker.report("After Phase 1")
    
    # === Phase 2: Intensive polish on global best ===
    if tracker.best is not None and time.monotonic() < deadline - time_final:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 2: Intensive polish", flush=True)
        print(f"{'='*60}", flush=True)
        
        n_best = tracker.best.size
        t_polish_deadline = min(deadline - time_final, time.monotonic() + time_polish)
        
        # LP with larger active set
        a_lp = run_lp_mirror(tracker.best, min(t_polish_deadline, time.monotonic() + time_polish * 0.5),
                             n_active_start=20, n_active_end=150, max_iters=100)
        tracker.update(a_lp)
        
        # FISTA with high tau
        if time.monotonic() < t_polish_deadline:
            tau_phases = [(200, 2000), (500, 1500), (1000, 1000)]
            a_f = run_dual_primal_fista(tracker.best, t_polish_deadline, tau_phases, lr_init=0.3)
            tracker.update(a_f)
        
        # Coordinate descent
        if time.monotonic() < t_polish_deadline:
            max_sw = max(10, int((t_polish_deadline - time.monotonic()) / max(0.1, n_best / 100.0)))
            a_cd = coordinate_descent_focused(tracker.best, t_polish_deadline, max_sweeps=max_sw)
            tracker.update(a_cd)
        
        print(f"  Polish elapsed: {time.monotonic() - (t_polish_deadline - time_polish):.1f}s", flush=True)
    
    tracker.report("After Phase 2")
    
    # === Phase 3: Final refinement ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 3: Final refinement", flush=True)
        print(f"{'='*60}", flush=True)
        
        t_final_deadline = deadline - 5
        
        # Quick LP
        a_lp = run_lp_mirror(tracker.best, min(t_final_deadline, time.monotonic() + time_final * 0.4),
                             n_active_start=30, n_active_end=200, max_iters=30)
        tracker.update(a_lp)
        
        # Quick CD
        if time.monotonic() < t_final_deadline:
            a_cd = coordinate_descent_focused(tracker.best, t_final_deadline, max_sweeps=20)
            tracker.update(a_cd)
    
    tracker.report("After Phase 3")
    
    # === Final answer ===
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
