import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def compute_M(a):
    """Compute max of conv(a,a) assuming sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return fast_conv(a / s).max()

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def solve_full_lp(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns normalized g or None."""
    n = len(a)
    # Build convolution matrix A: A[k,j] = a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.status == 0 and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception as e:
        pass
    return None

def solve_lp_active(a, M, active_set):

    cols = []
    vals = []

    bounds = [(0.0, None)] * n

            g = np.maximum(r.x, 0.0)

def bisection_line_search(a, g, M_current, n_evals_max=30):
    """Find optimal t in [0,1] minimizing max(conv((1-t)a+tg, ...)).
    Uses bisection on the derivative sign change."""
    n = len(a)
    
    def obj(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-15:
            return float('inf')
        return fast_conv(a_t / s).max()
    
    # Find the minimum of a unimodal (approximately) function on [0, 1]
    # Use golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = obj(a1)
    f2 = obj(a2)
    
    for _ in range(n_evals_max):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1 = obj(a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = obj(a2)
    
    t_opt = (lo + hi) / 2.0
    return t_opt, obj(t_opt)

def run_fista(a_init, deadline, tau_values=None):

        t_mom = 1.0
        
        for it in range(3000):

                break
            
            b = fast_conv(y)
            b_max = b.max()
            exp_b = np.exp(tau * (b - b_max))
            p = exp_b / exp_b.sum()
            
            # Gradient
            grad = np.zeros(n)
            for i in range(n):
                grad[i] = 2.0 * np.dot(p[i:i+n], y)
            
            grad_tan = grad - grad.mean()
            step = 0.3 / (1.0 + 0.01 * tau)
            
            a_new = proj_simplex(y - step * grad_tan)
            
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            alpha = (t_mom - 1.0) / t_new
            y_new = proj_simplex(a_new + alpha * (a_new - a_prev))
            
            a_prev = a
            a = a_new
            y = y_new
            t_mom = t_new
            
            if it % 10 == 0:
                M_cur = fast_conv(a).max()
                if M_cur < best_M:
                    best_M = M_cur
                    best_a = a.copy()
    
    return best_a, best_M

def coordinate_descent_step(a, i, b, n):
    """Optimize coordinate i given others fixed."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    # Unaffected max
    all_k = set(range(2 * n - 1))
    affected_k = set(range(k_lo, k_hi + 1))
    unaffected_k = all_k - affected_k
    if unaffected_k:
        base_max = max(b[k] for k in unaffected_k)
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(3.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.1)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

                        b[k] = (b[k] - (a[i]-delta) * (a[i]-delta)) + x_opt * x_opt

                sweep_improved = True

            best_a = a.copy()
            no_improve = 0

            no_improve += 1

def run_subgradient_averaging(a_init, deadline, n_iters=5000):
    """Subgradient method with iterate averaging."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # averaged iterate
    a_avg = a.copy()
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        # Pick the worst constraint (with some randomization)
        top_k = np.argsort(b)[::-1][:10]
        k_star = top_k[np.random.randint(len(top_k))]
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = int(k_star) - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        grad_tan = grad - grad.mean()
        gnorm = np.linalg.norm(grad_tan)
        if gnorm < 1e-15:
            break
        
        # Polyak step size (using gap estimate)
        # fCurrent - fLower ≈ M - (1/(2n-1)) (lower bound when sum=1)
        f_lower = 1.0 / (2 * n - 1)
        gap = max(M - f_lower, 1e-6)
        eta = gap / (gnorm * gnorm)
        eta = min(eta, 0.5)
        
        a_new = proj_simplex(a - eta * grad_tan)
        a = a_new
        
        # Exponential averaging
        alpha = 2.0 / (it + 2.0)
        a_avg = (1.0 - alpha) * a_avg + alpha * a
        
        # Track best
        if it % 50 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_M = M_avg
                best_a = a_avg.copy()
    
    # Final check
    M_cur = fast_conv(a).max()
    if M_cur < best_M:
        best_M = M_cur
        best_a = a.copy()
    M_avg = fast_conv(a_avg).max()
    if M_avg < best_M:
        best_M = M_avg
        best_a = a_avg.copy()
    
    return best_a, best_M

def resample_seq(src, dest_n):

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M_normalized(a):
    """Compute max of conv(a,a) with a normalized to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return float(fast_conv(a / s).max())


def compute_J_val(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            m = k - j
            rows.append(k)
            cols.append(j)
            vals.append(a[m])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_full_lp(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns normalized g or None."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.status == 0 and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def golden_section_mix(a, g, n_evals=25):
    """Find optimal t in [0,1] minimizing max(conv((1-t)a+tg, (1-t)a+tg)).
    Exploits convexity of f(t). Returns (t_opt, M_opt)."""
    n = len(a)

    def obj_at(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-15:
            return float('inf')
        return float(fast_conv(a_t / s).max())

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = obj_at(a1)
    f2 = obj_at(a2)

    for _ in range(n_evals):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1 = obj_at(a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = obj_at(a2)

    t_opt = (lo + hi) / 2.0
    return t_opt, obj_at(t_opt)


def run_lp_method(a_init, deadline, use_full=True, max_iters=500, n_evals_gs=22):
    """LP-based optimization with golden-section line search.
    Returns (best_a, best_M, converged)."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0
    converged = False

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M_cur = float(fast_conv(a).max())
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()

        if use_full:
            g = solve_full_lp(a, M_cur, time_limit=4)
        else:
            b = fast_conv(a)
            k_active = min(50, 2 * n - 1)
            active = np.argsort(b)[::-1][:k_active].tolist()
            g = solve_lp_subset(a, M_cur, active, time_limit=3)

        if g is None:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        # Check fixed point
        if float(np.max(np.abs(g - a))) < 5e-7:
            no_improve += 3
            if no_improve > 40:
                converged = True
                break
            continue

        # Golden section line search
        t_opt, M_opt = golden_section_mix(a, g, n_evals=n_evals_gs)

        if M_opt < best_M - 1e-13:
            a_new = (1.0 - t_opt) * a + t_opt * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a = a_new / s_new
                best_M = M_opt
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Try progressively smaller steps
            found = False
            for t_try in [0.15, 0.08, 0.04, 0.02, 0.01, 0.005]:
                a_trial = (1.0 - t_try) * a + t_try * g
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-10:
                    continue
                a_trial = a_trial / st
                Mt = float(fast_conv(a_trial).max())
                if Mt < best_M - 1e-13:
                    a = a_trial
                    best_M = Mt
                    best_a = a.copy()
                    no_improve = 0
                    found = True
                    break
            if not found:
                no_improve += 1
                if no_improve > 40:
                    converged = True
                    break

    return best_a, best_M, converged


def run_fista_smooth(a_init, deadline, tau_list=None):
    """FISTA with log-sum-exp smooth max approximation."""
    if tau_list is None:
        tau_list = [50, 100, 200, 500]

    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())

    for tau in tau_list:
        sub_dl = min(deadline, time.monotonic() + max(15, (deadline - time.monotonic()) / max(1, len(tau_list))))
        y = best_a.copy()
        a_prev = best_a.copy()
        t_mom = 1.0

        for it in range(5000):
            if time.monotonic() > sub_dl:
                break

            b = fast_conv(y)
            b_max = b.max()
            exp_b = np.exp(tau * (b - b_max))
            p = exp_b / exp_b.sum()

            grad = np.zeros(n)
            for i in range(n):
                grad[i] = 2.0 * np.dot(p[i:i + n], y)

            grad_tan = grad - grad.mean()
            step = 0.4 / (1.0 + 0.005 * tau)

            a_new = proj_simplex(y - step * grad_tan)

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            alpha = (t_mom - 1.0) / t_new
            y_new = proj_simplex(a_new + alpha * (a_new - a_prev))

            a_prev = a_new
            y = y_new
            t_mom = t_new

            if it % 20 == 0:
                Mc = float(fast_conv(y).max())
                if Mc < best_M:
                    best_M = Mc
                    best_a = y.copy()

    return best_a, best_M


def run_subgradient_avg(a_init, deadline, n_iters=20000):
    """Subgradient method with iterate averaging for escaping local optima."""
    rng = np.random.default_rng(777)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    a_avg = a.copy()

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        # Pick from top constraints with randomization
        top_k = np.argsort(b)[::-1][:15]
        k_star = int(top_k[rng.integers(len(top_k))])

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        grad_tan = grad - grad.mean()
        gnorm = np.linalg.norm(grad_tan)
        if gnorm < 1e-14:
            break

        gap = max(M - 1.0 / (2 * n - 1), 1e-8)
        eta = gap / (gnorm * gnorm)
        eta = min(eta, 0.3)

        a_new = proj_simplex(a - eta * grad_tan)
        a = a_new

        # Polyak averaging
        alpha = 2.0 / (it + 2.0)
        a_avg = (1.0 - alpha) * a_avg + alpha * a

        if it % 100 == 0:
            Mc = float(fast_conv(a).max())
            if Mc < best_M:
                best_M = Mc
                best_a = a.copy()
            Ma = float(fast_conv(a_avg).max())
            if Ma < best_M:
                best_M = Ma
                best_a = a_avg.copy()

    # Final checks
    Mc = float(fast_conv(a).max())
    if Mc < best_M:
        best_M = Mc
        best_a = a.copy()
    Ma = float(fast_conv(a_avg).max())
    if Ma < best_M:
        best_M = Ma
        best_a = a_avg.copy()

    return best_a, best_M


def coord_optimize_single(a, i, b, n):
    """Golden section to optimize coordinate i given others fixed."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    # Unaffected max
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_lo:k_hi + 1] = False
    if mask.any():
        base_max = float(b[mask].max())
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = float(vals.max())
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(3.0 * x_cur, float(np.sqrt(max(b.max(), 0.0))) + 0.3, 0.05)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coordinate_descent(a_init, deadline, max_sweeps=200, seed=42):
    """Coordinate descent with golden section per coordinate."""
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a).copy()
        order = np.arange(n)
        rng.shuffle(order)

        sweep_improved = False
        for i in order:
            x_opt = coord_optimize_single(a, i, b, n)
            delta = x_opt - a[i]
            if abs(delta) > 1e-13:
                old_val = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * delta
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = float(fast_conv(a).max())

        if M < best_M * (1.0 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved or no_improve > 40:
            break

    return best_a, best_M


def run_mirror_descent(a_init, deadline, n_iters=10000, step_init=0.15):
    """Mirror descent with KL geometry."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        grad_tan = grad - grad.mean()
        step = step_init / (1.0 + 0.01 * it)
        log_w = -step * grad_tan
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new

        if it % 20 == 0:
            Mc = float(fast_conv(a).max())
            if Mc < best_M - 1e-14:
                best_M = Mc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_M


def resample_to_length(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def update_best(state, candidate, t_start):
    """Update best state if candidate is better. state = [best_a, best_M]."""
    a_c = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
    s_c = a_c.sum()
    if s_c < 1e-10:
        return state
    a_c = a_c / s_c
    M_c = float(fast_conv(a_c).max())
    if M_c < state[1] - 1e-14:
        elapsed = time.monotonic() - t_start
        print(f"  ** NEW BEST: M={M_c:.12f} J={2*a_c.size*M_c:.8f} t={elapsed:.0f}s", flush=True)
        return [a_c.copy(), M_c]
    return state


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: FISTA -> Full LP -> CD -> Subgradient -> LP -> CD -> Alternate.
    Uses golden-section line search exploiting convexity of mixing parameter.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    t_start = time.monotonic()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768)

    parent = parent / parent.sum()
    best_a = parent.copy()
    best_M = float(fast_conv(parent).max())
    best_J = compute_J_val(best_a)

    print(f"Initial: n={parent.size}, M={best_M:.12f}, J={best_J:.8f}", flush=True)

    # === Phase 1: FISTA (180s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: FISTA (180s)", flush=True)
    print(f"{'='*60}", flush=True)
    t1 = min(t_start + 180, deadline - 600)
    a_f, M_f = run_fista_smooth(best_a, t1, tau_list=[50, 100, 200, 500])
    if M_f < best_M:
        best_a, best_M = a_f, M_f
        print(f"  FISTA: M={best_M:.12f}", flush=True)

    # === Phase 2: Full LP (350s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Full LP optimization (350s)", flush=True)
    print(f"{'='*60}", flush=True)
    t2 = min(t_start + 530, deadline - 400)
    a_lp, M_lp, conv = run_lp_method(best_a, t2, use_full=True, max_iters=2000, n_evals_gs=22)
    if M_lp < best_M:
        best_a, best_M = a_lp, M_lp
    print(f"  LP done: M={best_M:.12f} converged={conv}", flush=True)

    # === Phase 3: Coordinate descent (150s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Coordinate descent (150s)", flush=True)
    print(f"{'='*60}", flush=True)
    t3 = min(t_start + 680, deadline - 250)
    a_cd, M_cd = run_coordinate_descent(best_a, t3, max_sweeps=300, seed=111)
    if M_cd < best_M:
        best_a, best_M = a_cd, M_cd
    print(f"  CD done: M={best_M:.12f}", flush=True)

    # === Phase 4: Subgradient averaging (120s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Subgradient averaging (120s)", flush=True)
    print(f"{'='*60}", flush=True)
    t4 = min(t_start + 800, deadline - 150)
    a_sg, M_sg = run_subgradient_avg(best_a, t4, n_iters=30000)
    if M_sg < best_M:
        best_a, best_M = a_sg, M_sg
    print(f"  SG done: M={best_M:.12f}", flush=True)

    # === Phase 5: LP again (200s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: LP refinement (200s)", flush=True)
    print(f"{'='*60}", flush=True)
    t5 = min(t_start + 1000, deadline - 130)
    a_lp2, M_lp2, _ = run_lp_method(best_a, t5, use_full=True, max_iters=2000, n_evals_gs=20)
    if M_lp2 < best_M:
        best_a, best_M = a_lp2, M_lp2
    print(f"  LP2 done: M={best_M:.12f}", flush=True)

    # === Phase 6: CD (100s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: CD refinement (100s)", flush=True)
    print(f"{'='*60}", flush=True)
    t6 = min(t_start + 1100, deadline - 60)
    a_cd2, M_cd2 = run_coordinate_descent(best_a, t6, max_sweeps=400, seed=222)
    if M_cd2 < best_M:
        best_a, best_M = a_cd2, M_cd2
    print(f"  CD2 done: M={best_M:.12f}", flush=True)

    # === Phase 7: Mirror descent (80s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 7: Mirror descent (80s)", flush=True)
    print(f"{'='*60}", flush=True)
    t7 = min(t_start + 1180, deadline - 40)
    a_md, M_md = run_mirror_descent(best_a, t7, n_iters=15000, step_init=0.1)
    if M_md < best_M:
        best_a, best_M = a_md, M_md
    print(f"  MD done: M={best_M:.12f}", flush=True)

    # === Phase 8: Final LP + CD alternating ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 8: Final alternating LP/CD", flush=True)
    print(f"{'='*60}", flush=True)
    n_alt = 0
    while time.monotonic() < deadline - 10 and n_alt < 15:
        remaining = deadline - 10 - time.monotonic()
        if remaining < 20:
            break
        # Short LP
        sub_dl = min(deadline - 10, time.monotonic() + min(25, remaining * 0.6))
        a_a1, M_a1, _ = run_lp_method(best_a, sub_dl, use_full=True, max_iters=100, n_evals_gs=15)
        if M_a1 < best_M:
            best_a, best_M = a_a1, M_a1

        # Short CD
        sub_dl2 = min(deadline - 5, time.monotonic() + min(20, remaining * 0.4))
        a_a2, M_a2 = run_coordinate_descent(best_a, sub_dl2, max_sweeps=30, seed=5000 + n_alt)
        if M_a2 < best_M:
            best_a, best_M = a_a2, M_a2
        n_alt += 1

    # Final evaluation
    final_J = evaluate_sequence([float(x) for x in best_a])
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={final_J:.8f} M={best_M:.12f} n={best_a.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_a]
