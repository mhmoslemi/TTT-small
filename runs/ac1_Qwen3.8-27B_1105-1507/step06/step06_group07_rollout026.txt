import numpy as np

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def subgradient_at_peak(a, k_peak):
    """Compute subgradient of conv(a,a)[k_peak] w.r.t. a."""
    n = len(a)
    sg = np.zeros(n)
    for i in range(n):
        j1 = k_peak - i
        j2 = i - k_peak
        val = 0.0
        if 0 <= j1 < n:
            val += a[j1]
        if 0 <= j2 < n:
            val += a[j2]
        sg[i] = 2.0 * val
    return sg


def road_conv_cross(a, w):
    """Compute cross-correlation: (a*w_conv)[i] = sum_j w[j] * a[j+i] for i in [0, n-1].
    This is the gradient of sum_j w[j] * conv(a,a)[j] w.r.t. a[i]."""
    n = len(a)
    # conv(a,a)[k] = sum_i a[i] * a[k-i]
    # d/d a[i] sum_k w[k] * conv(a,a)[k] = sum_k w[k] * (a[k-i] + a[i-k])
    # = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
    # First term: let j = k-i, = sum_j w[j+i]*a[j] (i+j < 2n-1)
    # Second term: let j' = i-k, k = i-j', = sum_{j'} w[i-j']*a[j']
    # So gradient[i] = sum_j w[j+i]*a[j] + sum_j w[i-j]*a[j]
    # First sum is cross-correlation of w and a (reversed)
    # Let's compute via FFT

    size = 1
    while size < 4 * n:
        size <<= 1
    
    A = rfft(a, size)
    W = rfft(w, size)
    
    # sum_j w[j+i] * a[j] for j in [0,n-1], j+i < 2n-1
    # This is like: flip w, convolve with a
    w_flip = np.zeros(size)
    w_flip[:2*n-1] = w[::-1]
    WF = rfft(w_flip, size)
    first = irfft(WF * A, size)[:n]
    
    # sum_j w[i-j] * a[j] for j in [0,n-1], i-j in [0, 2n-2]
    # This is: convolve w with a, take positions [0, n-1]
    # Actually w[i-j] * a[j] summed over j is the (n-1-i)th element of conv(w, a) reversed...
    # Let me think again.
    # sum_j w[i-j] * a[j] = (w * a)[i] where * is cross-correlation
    # In FFT: ifft(conj(FFT(w)) * FFT(a))[i]
    second = irfft(np.conj(W) * A, size)[:n]
    
    return first + second

    # conv(a_norm, delta)[k] = sum_i a_norm[i] * delta[k-i]
    # We need this for k = 0, ..., 2n-2
    # Matrix A where A[k, i] = a_norm[k-i] if 0 <= k-i < n, else 0
    
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

        k_start = i

        A[k_start:k_end, i] = a_norm[:k_end - k_start]
    
    # LP: min t
    # s.t. 2 * A[k,:] @ delta + conv[k] <= t  for all k (or just peaks)
    #      sum(delta) = 0
    #      -a_norm[i] <= delta[i] <= a_norm[i] (trust region)

    # Sum constraint: sum(delta) = 0
    # In linprog, equality constraints are A_eq @ x = b_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def subgradient_descent(a, max_iters=5000, time_budget=60.0):
    """Subgradient descent on the exact (non-smooth) objective."""
    n = len(a)
    a = a.copy()

        return a
    a = a / s
    
    best_a = a.copy()

    # Polyak-like step size
    best_known = compute_obj(a)
    # Target: we know uniform gives 2.0, so target < 2.0
    target = 1.6  # reasonable target
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_peak = int(np.argmax(conv))
        M = conv[k_peak]
        
        # Subgradient
        sg = subgradient_at_peak(a, k_peak)
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-12:
            break
        
        # Polyak step size: (f(x) - f*) / ||g||^2
        step = (best_obj - target) / (sg_norm * sg_norm + 1e-8)
        step = min(step, 0.1)  # cap step size
        
        # Update (maintain non-negativity)
        a_new = a - step * sg
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            target = min(target, best_obj - 0.001)
        
        a = a_new

def coordinate_descent_v2(a, time_budget=30.0, seed=0):
    """More aggressive coordinate descent focusing on peak-reducing moves."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)
    
    for outer in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(best_a)
        # Find top peaks
        top_k = min(3, 2*n-1)
        peak_indices = np.argsort(conv)[::-1][:top_k]
        
        improved = False
        for p in peak_indices:
            # Elements contributing to conv[p]
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            
            # Try reducing the largest contributors to this peak
            contribs = []
            for i in range(i_min, i_max + 1):
                j = p - i
                contribs.append((best_a[i] * best_a[j], i, j))
            contribs.sort(reverse=True)
            
            # Try to reduce the top contributors
            for (val, i, j) in contribs[:5]:
                if time.monotonic() - start > time_budget:
                    break
                # Reduce a[i] or a[j] and redistribute
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * best_a[i]
                    # Move mass to a random position far from the peak
                    target = int(rng.integers(0, n))
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[target] += delta
                    if a_try[i] < 0:
                        continue
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a


def road_convolve(a, b):
    """Compute linear convolution of a and b."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    conv = irfft(fa * fb, size)
    return conv[:n + m - 1]

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP trust-region linearization (iterative), exact subgradient descent,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz_matrix(a):
    """Build matrix A where A[k, i] = a[k-i] if 0<=k-i<n else 0.
    So that A @ delta = conv(a, delta)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        A[k_start:k_end, i] = a[:k_end - k_start]
    return A

def subgradient_descent(a, max_iters=3000, time_budget=60.0):
    """Subgradient descent on the exact (non-smooth) objective with Polyak step."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    target = min(best_obj - 0.05, 1.6)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_peak = int(np.argmax(conv))
        
        # Subgradient of conv(a,a)[k_peak] w.r.t. a
        sg = np.zeros(n)
        for i in range(n):
            j1 = k_peak - i
            j2 = i - k_peak
            val = 0.0
            if 0 <= j1 < n:
                val += a[j1]
            if 0 <= j2 < n:
                val += a[j2]
            sg[i] = 2.0 * val
        
        sg_norm_sq = float(np.dot(sg, sg))
        if sg_norm_sq < 1e-16:
            break
        
        # Polyak step
        gap = best_obj - target
        if gap < 1e-6:
            target = best_obj - 0.001
            gap = 0.001
        step = gap / (sg_norm_sq + 1e-10)
        step = min(step, 0.05)
        
        a_new = a - step * sg
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if obj_new < target:
                target = obj_new - 0.001
        a = a_new
    
    return best_a


def adam_log_opt(a, max_iters=10000, time_budget=60.0, beta=50.0,
                 lr=0.03, lr_decay=0.99999):
    """Adam optimization in log-space with logsumexp smoothing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    
    start = time.monotonic()
    
    # Precompute FFT sizes
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    size2 = 1
    while size2 < 4 * n:
        size2 <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        x_centered = x - np.max(x)
        a_curr = np.exp(x_centered)
        a_curr = a_curr / np.sum(a_curr)
        
        # Compute convolution via FFT
        fa = rfft(a_curr, size)
        conv = irfft(fa * fa, size)[:2*n-1]
        
        sum_a = float(np.sum(a_curr))
        
        # Softmax weights
        cmax = float(np.max(conv))
        shifted = beta * (conv - cmax)
        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        e_sum = float(np.sum(e))
        if e_sum < 1e-300:
            e_sum = 1e-300
        w = e / e_sum
        ls = cmax + np.log(e_sum) / beta
        
        # Gradient via FFT cross-correlation
        W = rfft(w, size2)
        A_fft = rfft(a_curr, size2)
        
        w_flip = w[::-1]
        WF = rfft(w_flip, size2)
        term1 = irfft(WF * A_fft, size2)[:n]
        term2 = irfft(np.conj(W) * A_fft, size2)[:n]
        
        grad_ls = 2.0 * (term1 + term2)
        grad_a = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
        grad_x = grad_a * a_curr
        
        # Adam update
        t_count += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 100 == 0:
            x -= np.mean(x)
        
        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a

            break
        
        conv = compute_conv(best_a)
        peak_indices = np.argsort(conv)[::-1][:3]
        
        improved = False
        for p in peak_indices:
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            
            # Find top contributors to this peak
            contribs = []
            for i in range(i_min, i_max + 1):
                j = p - i
                contribs.append((best_a[i] * best_a[j], i, j))
            contribs.sort(reverse=True)
            
            for (val, i, j) in contribs[:4]:
                if time.monotonic() - start > time_budget:
                    break
                for frac in [0.02, 0.05, 0.01]:
                    delta = frac * max(best_a[i], best_a[j])
                    # Move to random position
                    target = int(rng.integers(0, n))
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[target] += delta
                    if a_try[i] < 0:
                        continue
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def neighborhood_search(a, time_budget=30.0, seed=0):
    """Random neighborhood search: try small random perturbations."""
    n = len(a)
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    best_a = a.copy().astype(np.float64)
    s = np.sum(best_a)
    if s > 1e-10:
        best_a = best_a / s
    best_obj = compute_obj(best_a)
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        # Random perturbation
        scale = 0.01 + 0.03 * (1.0 - (time.monotonic() - start) / time_budget)
        a_try = best_a * np.exp(rng.normal(0, scale, n))
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP trust-region, subgradient, Adam log-space,
    coordinate descent, neighborhood search, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            best_score = obj
            best_seq = parent.copy()
    except Exception:
        pass
    
    if best_seq is None:
        n = 700
        best_seq = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
        best_score = compute_obj(best_seq)
        print(f"Initial Hann (n={n}): obj={best_score:.8f}")
    
    n_curr = len(best_seq)
    print(f"Starting: n={n_curr}, best={best_score:.8f}, budget={budget_s}s")
    
    # === Phase 1: Iterative LP Trust Region (primary optimizer) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Iterative LP Trust Region")
    print(f"{'='*60}")
    
    if len(best_seq) <= 1200:
        # Run multiple rounds of LP with decreasing trust region
        for round_idx, trust_scale in enumerate([1.0, 0.7, 0.5, 0.3, 0.2, 0.1]):
            if time.monotonic() > deadline - 150:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            
            alloc = min(remaining * 0.2, 60.0)
            a_lp = lp_trust_region_iterative(best_seq, max_iters=30, time_budget=alloc)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  LP round {round_idx} (trust={trust_scale}): {best_score:.8f} ***")
            else:
                print(f"  LP round {round_idx} (trust={trust_scale}): {obj:.8f}")
    
    # === Phase 2: LP Direction (maximize sum approach) ===
    print(f"\n{'='*60}")
    print(f"Phase 2: LP Direction Max-Sum")
    print(f"{'='*60}")
    
    if len(best_seq) <= 1000:
        for lp_round in range(5):
            if time.monotonic() > deadline - 120:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            
            alloc = min(remaining * 0.2, 40.0)
            a_lp = lp_direction_max_sum(best_seq, time_budget=alloc)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  LP-dir {lp_round}: {best_score:.8f} ***")
            else:
                print(f"  LP-dir {lp_round}: {obj:.8f}")
                if obj >= best_score - 1e-9:
                    break
            
            # Follow up with Adam
            if time.monotonic() < deadline - 50:
                a_post = adam_log_opt(best_seq, max_iters=8000, time_budget=15.0,
                                      beta=200.0, lr=0.01, lr_decay=0.99999)
                obj_post = compute_obj(a_post)
                if obj_post < best_score:
                    best_score = obj_post
                    best_seq = a_post.copy()
                    print(f"  LP-dir+Adam {lp_round}: {best_score:.8f} ***")
    
    # === Phase 3: Adam Log-Space (multiple betas) ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Adam Log-Space Annealing")
    print(f"{'='*60}")
    
    adam_rounds = [
        (10.0, 0.04, 0.99998),
        (30.0, 0.03, 0.99998),
        (80.0, 0.02, 0.99999),
        (150.0, 0.015, 0.99999),
        (300.0, 0.01, 0.99999),
        (500.0, 0.008, 0.99999),
        (800.0, 0.005, 0.99999),
    ]
    
    for beta, lr, lr_dec in adam_rounds:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        
        alloc = min(remaining * 0.12, 50.0)
        a_opt = adam_log_opt(best_seq, max_iters=15000, time_budget=alloc,
                             beta=beta, lr=lr, lr_decay=lr_dec)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Adam beta={beta:.0f}: {best_score:.8f} ***")
        else:
            print(f"  Adam beta={beta:.0f}: {obj:.8f}")
    
    # === Phase 4: Subgradient Descent ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Subgradient Descent")
    print(f"{'='*60}")
    
    if time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.12, 40.0)
        a_sg = subgradient_descent(best_seq, max_iters=3000, time_budget=alloc)
        obj = compute_obj(a_sg)
        if obj < best_score:
            best_score = obj
            best_seq = a_sg.copy()
            print(f"  Subgrad: {best_score:.8f} ***")
        else:
            print(f"  Subgrad: {obj:.8f}")
    
    # === Phase 5: Multi-n Exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-n Exploration")
    print(f"{'='*60}")
    
    n_best = len(best_seq)
    n_options = sorted(set([
        max(300, n_best - 200),
        max(300, n_best - 100),
        max(300, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
        n_best + 700,
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500]
    
    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        
        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        s = np.sum(a_try)
        if s > 1e-10:
            a_try = a_try / s
        
        # Quick Adam on resampled
        alloc = min(remaining * 0.12, 40.0)
        a_opt = adam_log_opt(a_try, max_iters=10000, time_budget=alloc,
                             beta=50.0, lr=0.03, lr_decay=0.99998)
        obj = compute_obj(a_opt)
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  n={n_try}: {obj:.8f}")
    
    # === Phase 6: Coordinate Descent + Neighborhood Search ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Coordinate Descent + Neighborhood")
    print(f"{'='*60}")
    
    if time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 30.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, seed=seed)
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  CD: {best_score:.8f} ***")
        else:
            print(f"  CD: {obj:.8f}")
    
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 25.0)
        a_ns = neighborhood_search(best_seq, time_budget=alloc, seed=seed + 1)
        obj = compute_obj(a_ns)
        if obj < best_score:
            best_score = obj
            best_seq = a_ns.copy()
            print(f"  Neighborhood: {best_score:.8f} ***")
        else:
            print(f"  Neighborhood: {obj:.8f}")
    
    # === Phase 7: Final LP Trust Region ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final LP Trust Region")
    print(f"{'='*60}")
    
    if time.monotonic() < deadline - 30 and len(best_seq) <= 1200:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 50.0)
        a_final_lp = lp_trust_region_iterative(best_seq, max_iters=50, time_budget=alloc)
        obj = compute_obj(a_final_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_final_lp.copy()
            print(f"  Final LP: {best_score:.8f} ***")
        else:
            print(f"  Final LP: {obj:.8f}")
    
    # === Phase 8: Final High-Beta Adam ===
    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.5, 25.0)
        a_final = adam_log_opt(best_seq, max_iters=15000, time_budget=alloc,
                               beta=1000.0, lr=0.005, lr_decay=0.99999)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final Adam (beta=1000): {best_score:.8f} ***")
    
    # === Phase 9: Random Restarts ===
    if time.monotonic() < deadline - 20:
        rng = np.random.default_rng(seed + 200)
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        
        for restart in range(4):
            if time.monotonic() > deadline - 10:
                break
            if remaining < 10:
                break
            
            # Different perturbation strategies
            if restart == 0:
                a_pert = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            elif restart == 1:
                # Spike perturbation
                a_pert = best_seq.copy()
                for _ in range(max(1, n_best // 80)):
                    idx = int(rng.integers(0, n_best))
                    a_pert[idx] += rng.uniform(0.02, 0.08) * np.mean(best_seq)
            elif restart == 2:
                # Blend with flat
                alpha = 0.1
                flat = np.ones(n_best) / n_best
                b_norm = best_seq / np.sum(best_seq)
                a_pert = (1 - alpha) * b_norm + alpha * flat
            else:
                # Blend with Hann
                hann = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n_best) / max(n_best - 1, 1)))
                hann = hann / np.sum(hann)
                b_norm = best_seq / np.sum(best_seq)
                a_pert = 0.8 * b_norm + 0.2 * hann
            
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / np.sum(a_pert)
            
            alloc = min(remaining * 0.15, 12.0)
            a_opt = adam_log_opt(a_pert, max_iters=8000, time_budget=alloc,
                                 beta=100.0, lr=0.02, lr_decay=0.99999)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} ***")
            else:
                print(f"  Restart {restart}: {obj:.8f}")
    
    # Final validation
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")
    
    return result
