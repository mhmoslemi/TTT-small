import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def power_mean_obj_grad(a, p):
    """
    Compute (sum(conv^p)/len(conv))^(1/p) as smooth max approximation,
    and its gradient.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    m = len(conv)
    
    # Power mean: (sum(conv^p)/m)^(1/p)
    # Use log for stability
    conv_clip = np.maximum(conv, 1e-30)
    log_conv = np.log(conv_clip)
    # log_power_mean = (1/p) * (log(sum(exp(p*log_conv))) - log(m))
    p_log_conv = p * log_conv
    p_log_conv_shifted = p_log_conv - np.max(p_log_conv)
    log_sum_exp = np.max(p_log_conv) + np.log(np.sum(np.exp(p_log_conv_shifted)))
    log_pm = (log_sum_exp - np.log(m)) / p
    pm = np.exp(log_pm)
    
    # Weights: w_j = conv[j]^p / sum(conv^p)
    w = np.exp(p_log_conv_shifted)
    w_sum = np.sum(w)
    w = w / w_sum
    
    # Gradient of pm w.r.t. a[i]:
    # d(pm)/d(conv[j]) = (1/p) * pm * (p * w[j]) = pm * w[j]
    # d(conv[j])/d(a[i]) = 2 * cross_correlation(1)[j][i] = 2 * a[j-i] if valid
    # So d(pm)/d(a[i]) = pm * sum_j w[j] * d(conv[j])/d(a[i])
    # = pm * 2 * sum_j w[j] * a[|j-i|] (with bounds)
    # This is 2 * pm * (w * a)[i] (cross-correlation)
    
    size = 1
    while size < 2 * m:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_pm = 2.0 * pm * cc[:n]
    
    # Full objective: 2n * pm / sum_a^2
    obj = 2.0 * n * pm / (sum_a * sum_a)
    
    # d(obj)/d(a[i]) = (2n/sum_a^2) * (grad_pm[i] - 2*pm/sum_a)
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_pm - 2.0 * pm / sum_a)
    
    return obj, grad_obj

def adam_power_mean(a, max_iters=10000, time_budget=100.0, p_start=4.0, p_end=500.0,
                    lr_start=0.05, lr_decay=0.99995):
    """Adam optimization using power-mean smoothing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        p = p_start * (p_end / p_start) ** progress
        
        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)
        
        obj_sm, grad_a = power_mean_obj_grad(a, p)
        grad_x = grad_a * a
        
        t += 1
        lr_t = lr * (lr_decay ** it)
        m_adam = beta1 * m_adam + (1 - beta1) * grad_x
        v_adam = beta2 * v_adam + (1 - beta2) * (grad_x * grad_x)
        m_hat = m_adam / (1 - beta1 ** t)
        v_hat = v_adam / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 200 == 0:
            x -= np.mean(x)
        
        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a

def peak_suppress(a, n_peaks=5, steps=100, time_budget=30.0):
    """
    Directly suppress peaks by identifying contributing elements and reducing them.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for step in range(steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(best_a)
        # Get top peaks
        top_idx = np.argsort(conv)[::-1][:n_peaks]
        
        # For each top peak, find the elements contributing most
        for k in top_idx:
            # conv[k] = sum_i a[i]*a[k-i] for valid i
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            
            if i_min > i_max:
                continue
            
            # contributions[i] = a[i] * a[k-i]
            indices = np.arange(i_min, i_max + 1)
            partners = k - indices
            contribs = a[indices] * a[partners]
            
            # Find the elements with highest contribution and reduce them
            contrib_order = np.argsort(contribs)[::-1][:3]
            for ci in contrib_order:
                i = indices[ci]
                j = partners[ci]
                # Try reducing a[i] slightly
                for frac in [0.01, 0.005, 0.001]:
                    delta = frac * best_a[i]
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    if a_try[i] < 0:
                        continue
                    # Redistribute to elements NOT contributing to this peak
                    # Find elements that don't contribute to any top peak
                    a_try[j] += delta  # simple: add back to partner
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        break
    
    return best_a

def mirror_descent(a, max_iters=5000, time_budget=50.0, p=50.0):
    """Mirror descent in log-space with power-mean gradient."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Mirror descent: x_{t+1} = argmin <g, y> + D_KL(y || x_t)
    # In log-space: x_{t+1} = x_t - eta * g (where g is gradient w.r.t. log params)
    # With adaptive step size
    
    x = np.log(a)
    x -= np.mean(x)
    lr = 0.1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        x_c = x - np.max(x)
        a_cur = np.exp(x_c)
        a_cur = a_cur / np.sum(a_cur)
        
        obj_sm, grad_a = power_mean_obj_grad(a_cur, p)
        grad_x = grad_a * a_cur
        
        # Adaptive step: use norm of gradient
        g_norm = np.sqrt(np.sum(grad_x**2))
        step = lr / (g_norm + 1e-10)
        
        x = x - step * grad_x
        x -= np.mean(x)
        
        # Decay learning rate
        lr *= 0.999
        
        if it % 50 == 0:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()
    
    return best_a

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2:
1. Direct subgradient descent targeting peak positions
2. Power-mean and logsumexp smoothed Adam in log-space
3. Asymmetry injection to reduce center peak
4. Multi-length exploration with structured initializations
5. Peak-directed coordinate descent for final polish
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def logsumexp_smooth_grad(a, beta):
    """Compute logsumexp(beta*conv) and its gradient w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    m = len(conv)
    size = 1
    while size < 2 * m:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)

    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def power_mean_smooth_grad(a, p):
    """Compute power-mean smooth max and its gradient w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    m = len(conv)

    conv_safe = np.maximum(conv, 1e-30)
    log_conv = np.log(conv_safe)
    p_log_conv = p * log_conv
    p_shifted = p_log_conv - np.max(p_log_conv)
    log_sum_exp = np.max(p_log_conv) + np.log(np.sum(np.exp(p_shifted)))
    log_pm = (log_sum_exp - np.log(m)) / p
    pm = np.exp(log_pm)

    w = np.exp(p_shifted)
    w_sum = float(np.sum(w))
    w = w / w_sum

    obj = 2.0 * n * pm / (sum_a * sum_a)

    size = 1
    while size < 2 * m:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)

    grad_pm = 2.0 * pm * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_pm - 2.0 * pm / sum_a)

    return obj, grad_obj


def direct_subgradient(a, top_k=1):
    """
    Compute exact subgradient of 2n*max(conv)/sum^2 at current point.
    Returns gradient w.r.t. a (not log-space).
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    max_b = float(np.max(conv))
    obj = 2.0 * n * max_b / (sum_a * sum_a)

    # Find top_k peak positions (averaged subgradient)
    peak_idx = np.argsort(conv)[::-1][:top_k]
    # Weight by how close each peak is to the max
    peak_vals = conv[peak_idx]
    weights = (peak_vals / max_b) ** 10  # concentrate on the true max
    weights = weights / np.sum(weights)

    # Subgradient of max(conv) at peak position k: d/d(a[i]) = 2*a[k-i] if valid
    grad_max = np.zeros(n)
    for idx, pi in enumerate(peak_idx):
        k = pi
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i
            if 0 <= j < n:
                grad_max[i] += weights[idx] * 2.0 * a[j]

    # Full gradient of 2n*max_b/sum_a^2
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_max - max_b / sum_a)

    return obj, grad_obj


def adam_log_opt_v2(a, max_iters=20000, time_budget=100.0, method="logsumexp",

    """
    Adam optimization in log-space with configurable smoothing.

    """

    x = np.log(a)
    x -= np.mean(x)

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)

    beta2 = 0.999
    eps = 1e-8
    lr = lr_start

    best_true_obj = compute_obj(a)

    start = time.monotonic()

        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        param = beta_start * (beta_end / beta_start) ** progress

        x_centered = x - np.max(x)
        a_cur = np.exp(x_centered)
        a_cur = a_cur / np.sum(a_cur)

        if method == "logsumexp":
            _, grad_a = logsumexp_smooth_grad(a_cur, param)

            _, grad_a = power_mean_smooth_grad(a_cur, param)
        else:  # direct

        grad_x = grad_a * a_cur

        t += 1
        lr_t = lr * (lr_decay ** it)
        m_adam = beta1 * m_adam + (1 - beta1) * grad_x
        v_adam = beta2 * v_adam + (1 - beta2) * (grad_x * grad_x)
        m_hat = m_adam / (1 - beta1 ** t)
        v_hat = v_adam / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

                best_a = a_check.copy()

            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                step_size *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
            step_size *= 0.7

        if improved:
            patience = 0
            step_size = min(step_size * 1.1, 0.5)
        else:
            patience += 1
            step_size *= 0.8
            if patience > 200:
                break

    return best_a


def asymmetry_inject(a, rng):
    """
    Deliberately break symmetry to reduce center peak.
    Creates a version that is slightly asymmetric.
    """
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    # Strategy: multiply by a linearly varying factor
    # This breaks the symmetry while keeping the sequence smooth
    alpha = rng.uniform(0.05, 0.3)
    t = np.linspace(0, 1, n)
    factor = 1.0 + alpha * (2 * t - 1)  # varies from 1-alpha to 1+alpha
    factor = np.maximum(factor, 0.1)

    a_new = a * factor
    s_new = np.sum(a_new)
    if s_new > 1e-10:
        a_new = a_new / s_new
    else:
        a_new = a

    return a_new


def peak_coordinate_descent(a, time_budget=30.0, seed=0):
    """
    Coordinate descent focusing on peak reduction.
    Identifies elements contributing to the top peaks and adjusts them.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(best_a)
    top_peaks = np.argsort(conv)[::-1][:min(5, len(conv))]

    it = 0
    no_improve_count = 0

    while time.monotonic() - start < time_budget and it < 5000:
        it += 1
        improved = False

        for p in top_peaks:
            if time.monotonic() - start > time_budget:
                break

            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            if i_min > i_max:
                continue

            # Get contributions
            indices = np.arange(i_min, i_max + 1)
            partners = p - indices
            contribs = best_a[indices] * best_a[partners]

            # Sort by contribution (highest first)
            order = np.argsort(contribs)[::-1]

            for ci in order[:5]:
                i = indices[ci]
                # Try reducing a[i] and adding to a random non-contributing element
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    delta = frac * best_a[i]
                    if delta < 1e-15:
                        continue

                    # Find a good place to add: an element not contributing to any top peak
                    # Simple heuristic: add to an element with low value
                    a_try = best_a.copy()
                    a_try[i] = best_a[i] - delta

                    # Distribute to a random element
                    j = rng.integers(0, n)
                    a_try[j] += delta

                    if a_try[i] < 0:
                        continue
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        conv = compute_conv(best_a)
                        top_peaks = np.argsort(conv)[::-1][:min(5, len(conv))]
                        improved = True
                        no_improve_count = 0
                        break

        if not improved:
            no_improve_count += 1
            if no_improve_count > 200:
                break

    return best_a


def mirror_descent_log(a, max_iters=10000, time_budget=30.0, p=50.0):
    """Mirror descent in log-space with power-mean gradient."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.5

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        x_c = x - np.max(x)
        a_cur = np.exp(x_c)
        a_cur = a_cur / np.sum(a_cur)

        _, grad_a = power_mean_smooth_grad(a_cur, p)
        grad_x = grad_a * a_cur

        g_norm = np.sqrt(np.sum(grad_x**2)) + 1e-10
        step = lr / g_norm
        x = x - step * grad_x
        x -= np.mean(x)

        lr *= 0.9995

        if it % 30 == 0:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()

    return best_a


def try_load_parent():
    """Load parent sequence if available."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies:
    - Direct subgradient descent for peak targeting
    - Logsumexp and power-mean smoothed Adam in log-space
    - Asymmetry injection to reduce center peak concentration
    - Multi-length exploration
    - Peak-directed coordinate descent for final polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, budget={budget_s}s")

    # === Phase 1: Diverse initializations ===
    print(f"\n{'='*50}")
    print("Phase 1: Diverse initializations")
    print(f"{'='*50}")

    initializations = []

    if best_seq is not None:
        s = np.sum(best_seq)
        if s > 1e-10:
            initializations.append(("parent", best_seq.copy() / s))

    n_ref = n_parent
    x = np.arange(n_ref)

    # Various structured initializations
    fejer = np.maximum(1.0 - 2.0 * np.abs(x - (n_ref - 1) / 2.0) / (n_ref - 1), 0.01)
    initializations.append(("fejer", fejer / np.sum(fejer)))

    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n_ref))
    rc = np.maximum(rc, 0.01)
    initializations.append(("raised_cos", rc / np.sum(rc)))

    center = (n_ref - 1) / 2.0
    gauss = np.exp(-0.5 * (x - center) ** 2 / (n_ref / 4.0) ** 2)
    initializations.append(("gaussian", gauss / np.sum(gauss)))

    uniform = np.ones(n_ref) / n_ref
    initializations.append(("uniform", uniform))

    # Asymmetric: steeper on one side
    asym = np.exp(-x ** 2 / (n_ref * 0.3)) + 0.3 * np.exp(-(n_ref - x) ** 2 / (n_ref * 0.8))
    initializations.append(("asym_gauss", asym / np.sum(asym)))

    # Linear ramp
    ramp = (x + 1.0) / (n_ref + 1.0)
    initializations.append(("ramp", ramp / np.sum(ramp)))

    # Double peak
    dp = np.exp(-0.5 * (x - n_ref * 0.25) ** 2 / (n_ref * 0.1) ** 2) + \
         np.exp(-0.5 * (x - n_ref * 0.75) ** 2 / (n_ref * 0.1) ** 2)
    initializations.append(("double_peak", dp / np.sum(dp)))

    for name, a_init in initializations:
        if time.monotonic() > deadline - 200:
            break
        obj = compute_obj(a_init)
        print(f"  {name}: obj={obj:.6f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_init.copy()

    # === Phase 2: Subgradient descent (direct peak targeting) ===
    print(f"\n{'='*50}")
    print("Phase 2: Subgradient descent")
    print(f"{'='*50}")

    n_best = len(best_seq)
    if time.monotonic() < deadline - 150:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.1, 60.0)
        a_sub = subgradient_descent(best_seq, max_iters=5000, time_budget=alloc, top_k=1)
        obj_sub = compute_obj(a_sub)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  Subgradient (k=1): {best_score:.8f} ***")
        else:
            print(f"  Subgradient (k=1): {obj_sub:.8f}")

        if time.monotonic() < deadline - 100:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.08, 40.0)
            a_sub2 = subgradient_descent(best_seq, max_iters=5000, time_budget=alloc, top_k=3)
            obj_sub2 = compute_obj(a_sub2)
            if obj_sub2 < best_score:
                best_score = obj_sub2
                best_seq = a_sub2.copy()
                print(f"  Subgradient (k=3): {best_score:.8f} ***")
            else:
                print(f"  Subgradient (k=3): {obj_sub2:.8f}")

    # === Phase 3: Adam with logsumexp smoothing ===
    print(f"\n{'='*50}")
    print("Phase 3: Adam logsumexp optimization")
    print(f"{'='*50}")

    a = best_seq.copy()
    s = np.sum(a)
    if s > 1e-10:
        a = a / s

    rounds_ls = [
        (3.0, 20.0, 0.06, 0.99998),
        (5.0, 50.0, 0.05, 0.99998),
        (10.0, 100.0, 0.04, 0.99999),
        (20.0, 200.0, 0.03, 0.99999),
        (50.0, 400.0, 0.02, 0.99999),
        (100.0, 800.0, 0.01, 0.99999),
    ]

    for r_idx, (b_s, b_e, lr0, lr_dec) in enumerate(rounds_ls):
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        alloc = min(remaining * 0.1, 60.0)
        a_new = adam_log_opt_v2(a, max_iters=30000, time_budget=alloc,
                                method="logsumexp", beta_start=b_s, beta_end=b_e,
                                lr_start=lr0, lr_decay=lr_dec)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LSU round {r_idx} (beta {b_s:.0f}-{b_e:.0f}): {best_score:.8f} ***")
        else:
            print(f"  LSU round {r_idx} (beta {b_s:.0f}-{b_e:.0f}): {obj:.8f}")

    # === Phase 4: Adam with power-mean smoothing ===
    print(f"\n{'='*50}")
    print("Phase 4: Adam power-mean optimization")
    print(f"{'='*50}")

    if time.monotonic() < deadline - 80:
        rounds_pm = [
            (2.0, 10.0, 0.05, 0.99998),
            (5.0, 30.0, 0.04, 0.99998),
            (10.0, 80.0, 0.03, 0.99999),
            (30.0, 200.0, 0.02, 0.99999),
            (80.0, 500.0, 0.01, 0.99999),
        ]

        for r_idx, (p_s, p_e, lr0, lr_dec) in enumerate(rounds_pm):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            alloc = min(remaining * 0.1, 50.0)
            a_new = adam_log_opt_v2(a, max_iters=25000, time_budget=alloc,
                                    method="power_mean", beta_start=p_s, beta_end=p_e,
                                    lr_start=lr0, lr_decay=lr_dec)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  PM round {r_idx} (p {p_s:.0f}-{p_e:.0f}): {best_score:.8f} ***")
            else:
                print(f"  PM round {r_idx} (p {p_s:.0f}-{p_e:.0f}): {obj:.8f}")

    # === Phase 5: Asymmetry injection + refinement ===
    print(f"\n{'='*50}")
    print("Phase 5: Asymmetry injection")
    print(f"{'='*50}")

    rng = np.random.default_rng(seed)
    n_best = len(best_seq)

    for trial in range(8):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        a_asym = asymmetry_inject(best_seq, rng)
        obj_asym = compute_obj(a_asym)
        print(f"  Asym trial {trial}: raw obj={obj_asym:.8f}")

        alloc = min(remaining * 0.08, 30.0)
        a_opt = adam_log_opt_v2(a_asym, max_iters=20000, time_budget=alloc,
                                method="logsumexp", beta_start=10.0, beta_end=300.0,
                                lr_start=0.03, lr_decay=0.99998)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Asym trial {trial} optimized: {best_score:.8f} ***")
        else:
            print(f"  Asym trial {trial} optimized: {obj:.8f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n{'='*50}")
    print("Phase 6: Multi-length exploration")
    print(f"{'='*50}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 200), max(300, n_best - 100),
            n_best,
            n_best + 100, n_best + 200, n_best + 400, n_best + 600
        ]))
        n_options = [n for n in n_options if 200 <= n <= 2500]
        # Prioritize different lengths
        n_options = [n for n in n_options if n != n_best] + [n_best]

        for n_try in n_options:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if n_try == len(best_seq):
                a_try = best_seq.copy()
            else:
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, n_try)
                a_try = np.interp(x_new, x_old, best_seq)

            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s

            alloc = min(remaining * 0.1, 40.0)
            a_opt = adam_log_opt_v2(a_try, max_iters=20000, time_budget=alloc,
                                    method="logsumexp", beta_start=5.0, beta_end=200.0,
                                    lr_start=0.04, lr_decay=0.99998)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 7: Mirror descent ===
    print(f"\n{'='*50}")
    print("Phase 7: Mirror descent")
    print(f"{'='*50}")

    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.1, 30.0)
        a_md = mirror_descent_log(best_seq, max_iters=20000, time_budget=alloc, p=100.0)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Mirror descent: {best_score:.8f} ***")
        else:
            print(f"  Mirror descent: {obj_md:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n{'='*50}")
    print("Phase 8: Final polish")
    print(f"{'='*50}")

    if best_seq is not None:
        # High-temperature Adam
        if time.monotonic() < deadline - 25:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.15, 30.0)
            a_h = adam_log_opt_v2(best_seq, max_iters=25000, time_budget=alloc,
                                  method="logsumexp", beta_start=50.0, beta_end=1000.0,
                                  lr_start=0.01, lr_decay=0.99999)
            obj_h = compute_obj(a_h)
            if obj_h < best_score:
                best_score = obj_h
                best_seq = a_h.copy()
                print(f"  High-temp Adam: {best_score:.8f} ***")
            else:
                print(f"  High-temp Adam: {obj_h:.8f}")

        # Power-mean high
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.15, 25.0)
            a_h2 = adam_log_opt_v2(best_seq, max_iters=20000, time_budget=alloc,
                                   method="power_mean", beta_start=50.0, beta_end=1000.0,
                                   lr_start=0.01, lr_decay=0.99999)
            obj_h2 = compute_obj(a_h2)
            if obj_h2 < best_score:
                best_score = obj_h2
                best_seq = a_h2.copy()
                print(f"  High-p PM Adam: {best_score:.8f} ***")
            else:
                print(f"  High-p PM Adam: {obj_h2:.8f}")

        # Peak coordinate descent
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.2, 25.0)
            a_cd = peak_coordinate_descent(best_seq, time_budget=alloc, seed=seed)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Coord descent: {best_score:.8f} ***")
            else:
                print(f"  Coord descent: {obj_cd:.8f}")

        # Final subgradient
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.3, 20.0)
            a_sg = subgradient_descent(best_seq, max_iters=3000, time_budget=alloc, top_k=2)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Final subgradient: {best_score:.8f} ***")
            else:
                print(f"  Final subgradient: {obj_sg:.8f}")

    # === Phase 9: Random restarts ===
    print(f"\n{'='*50}")
    print("Phase 9: Random restarts")
    print(f"{'='*50}")

    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 1)

        for restart in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if restart % 3 == 0:
                scale = 0.02 + 0.015 * restart
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            elif restart % 3 == 1:
                a_pert = asymmetry_inject(best_seq, rng)
            else:
                # Blend with shifted version
                shift = rng.integers(1, max(2, n_best // 15))
                alpha = rng.uniform(0.05, 0.15)
                a_shift = np.roll(best_seq, shift)
                a_pert = (1 - alpha) * best_seq + alpha * a_shift

            a_pert = np.maximum(a_pert, 1e-10)
            s_p = np.sum(a_pert)
            if s_p < 1e-10:
                continue
            a_pert = a_pert / s_p

            alloc = min(remaining * 0.15, 12.0)
            a_opt = adam_log_opt_v2(a_pert, max_iters=15000, time_budget=alloc,
                                    method="logsumexp", beta_start=20.0, beta_end=500.0,
                                    lr_start=0.025, lr_decay=0.99998)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart}: {obj:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
