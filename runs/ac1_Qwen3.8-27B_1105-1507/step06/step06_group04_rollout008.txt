"""
Mirror descent (multiplicative weight) optimization with adaptive temperature,
FFT-accelerated gradients, LP refinement, and multi-start diverse exploration
to minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fft_convolve_auto(a):
    """Compute linear autocorrelation a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = fft_convolve_auto(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def fft_cross_corr(weights, a):
    """
    Compute cross-correlation: result[i] = sum_j a[j] * weights[i+j]
    for i = 0, ..., len(a)-1. Uses FFT for efficiency.
    weights should have length >= 2*len(a)-1.
    """
    n = len(a)
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    W = rfft(weights[:fft_size], fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    return corr[:n]


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    return a


def mirror_descent_optimize(a, time_budget=30.0, tau_init=0.01, eta_init=0.5,
                            tau_decay=0.997, eta_decay=0.9995,
                            eta_min=0.01, backtracking_factor=0.7):
    """
    Mirror descent (multiplicative weight) optimization on the simplex.
    
    Update: a[i] *= exp(-eta * g[i]); a /= sum(a)
    where g[i] = 2 * sum_p w[p] * a[p-i] (cross-correlation with softmax weights).
    """
    n = len(a)
    a = normalize_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_init
    eta = eta_init

    while time.monotonic() - start < time_budget:
        conv = fft_convolve_auto(a)
        cmax = float(np.max(conv))

        # Softmax weights over convolution values
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient via FFT cross-correlation
        grad = 2.0 * fft_cross_corr(weights, a)

        # Normalize gradient for stable updates
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean

        # Multiplicative update with backtracking
        found = False
        eta_try = eta
        for _bt in range(8):
            log_update = -eta_try * grad
            # Clip to avoid overflow/underflow
            log_update = np.clip(log_update, -50.0, 50.0)
            a_new = a * np.exp(log_update)
            a_new = normalize_simplex(a_new)
            obj_new = compute_obj(a_new)

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                found = True
                break
            elif obj_new <= best_obj * 1.001:
                # Accept if not too bad (for exploration)
                a = a_new
                found = True
                break
            else:
                eta_try *= backtracking_factor

        if not found:
            eta *= backtracking_factor
            if eta < eta_min:
                # Try perturbation instead
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = normalize_simplex(a)
                eta = eta_init * 0.5

        # Anneal parameters
        tau *= tau_decay
        eta = max(eta, eta_min)
        eta *= eta_decay

    return best_a, best_obj


def mirror_descent_adaptive(a, time_budget=30.0):

    Tries different (tau, eta) pairs and keeps the best.

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    configs = [
        (0.005, 0.8), (0.01, 0.5), (0.02, 0.3), (0.001, 1.0),
        (0.05, 0.2), (0.003, 0.6), (0.015, 0.4),
    ]

    # Pre-sort by expected quality (higher tau first for smooth start)
    configs.sort(key=lambda x: -x[0])

    per_config_time = min(time_budget / len(configs), 15.0)

    for tau, eta in configs:

            tau_init=tau, eta_init=eta

            best_a = a_cand.copy()

def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=20):
    """
    Solve: min t s.t. (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(fixed_seq)
    if n > 1000:
        return None, None
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(fixed_seq, M, time_limit=20):
    """
    Solve: max sum(g) s.t. (fixed_seq * g)[k] <= M for all k, g >= 0.
    Returns g normalized to sum=1 or None.
    """
    n = len(fixed_seq)
    if n > 1000:
        return None
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def generate_diverse_starts(n, num_starts=12, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    starts = []

    # Uniform
    starts.append(np.ones(n) / n)

    # Triangular
    tri = 1.0 - np.abs(i - center) / center
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    # Gaussian (narrow)
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 8.0) ** 2))
    starts.append(bell / np.sum(bell))

    # Gaussian (wide)
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 3.0) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # Semicircle (raised cosine)
    sc = np.sqrt(np.maximum(1.0 - ((i - center) / center) ** 2, 0.01))
    starts.append(sc / np.sum(sc))

    # Exponential decay
    exp_d = np.exp(-np.abs(i - center) / (n * 0.3))
    starts.append(exp_d / np.sum(exp_d))

    # Random parametric: bell * (1 + amp*cos)
    for _ in range(min(num_starts, 6)):
        sigma = n * (0.1 + 0.4 * rng.random())
        freq = int(1 + rng.integers(0, 10))
        amp = 0.1 + 0.6 * rng.random()
        phase = 2 * np.pi * rng.random()
        bell_p = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
        mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
        s = bell_p * mod
        s = np.maximum(s, 0)
        s_sum = np.sum(s)
        if s_sum > 1e-10:
            starts.append(s / s_sum)

    # Pure random
    rand_s = np.abs(rng.normal(0, 1, n)) + 0.01
    starts.append(rand_s / np.sum(rand_s))

    # Weighted random
    weights = rng.dirichlet(np.ones(n) * 0.5)
    starts.append(weights)

    return starts[:num_starts]


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

                rng = np.random.default_rng(int(time.monotonic() * 7777) % 1000000)
                scale = 0.05 + 0.05 * (stall % 4)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = normalize_simplex(a_pert)
                a_pert, obj_pert = mirror_descent_optimize(
                    a_pert, time_budget=min(5.0, remaining * 0.1),
                    tau_init=0.01, eta_init=0.4
                )
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                stall = 0

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        parent = parent[np.isfinite(parent)]
        if parent.size > 0:
            obj = compute_obj(parent)
            if np.isfinite(obj):
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start diverse initialization + mirror descent +
    LP refinement + stochastic polish + length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Phase 0: Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # Phase 1: Intensive optimization of parent (if available)
    print("\n=== Phase 1: Parent optimization ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.25, 180.0)
        a_pol, obj_pol = full_polish(best_seq, time_budget=alloc)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Parent polished: {best_score:.8f}")
        else:
            print(f"  Parent: no improvement ({obj_pol:.8f})")

    # Phase 2: Multi-start exploration at multiple lengths
    print(f"\n=== Phase 2: Multi-start (best: {best_score:.8f}) ===")

    n_values = sorted(set([
        n_parent,
        max(300, n_parent - 100),
        min(1200, n_parent + 100),
        max(200, n_parent - 200),
        min(1500, n_parent + 200),
    ]))

    # Filter to reasonable range
    n_values = [n for n in n_values if 200 <= n <= 1500]

    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Generate starts
        if n == n_parent and best_seq is not None:
            starts = [normalize_simplex(best_seq.copy())]
            starts.extend(generate_diverse_starts(n, num_starts=8, seed=seed + n))
        else:
            starts = generate_diverse_starts(n, num_starts=8, seed=seed + n * 7)

        # Quick screen each start
        alloc_per_start = min(remaining / len(starts), 20.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(alloc_per_start, 15.0)
            if t_alloc < 3:
                break

            a_opt, obj_opt = mirror_descent_optimize(
                s, time_budget=t_alloc,
                tau_init=np.random.default_rng(seed + si).uniform(0.005, 0.02),
                eta_init=np.random.default_rng(seed + si + 1).uniform(0.3, 0.7)
            )

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {best_score:.8f} ** NEW BEST **")

        # Intensive polish of best found at this length (if it improved)
        if best_seq is not None and len(best_seq) == n and time.monotonic() < deadline - 60:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.15, 40.0)
            a_pol, obj_pol = full_polish(best_seq, time_budget=alloc)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n} polished: {best_score:.8f}")

    # Phase 3: Length exploration around best
    print(f"\n=== Phase 3: Length exploration (best: {best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = sorted(set([
            n_best - 50, n_best + 50,
            n_best - 100, n_best + 100,
            n_best - 200, n_best + 200,
            n_best + 150, n_best - 150,
        ]))
        test_lengths = [x for x in test_lengths if 200 <= x <= 1500]

        for n_test in test_lengths:
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_resampled = normalize_simplex(resample_sequence(best_seq, n_test))
            alloc = min(remaining * 0.15, 30.0)
            a_pol, obj_pol = full_polish(a_resampled, time_budget=alloc)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: {best_score:.8f} ** UPDATE **")
            else:
                print(f"  n={n_test}: {obj_pol:.8f}")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")

    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.6, 120.0)
        a_final, obj_final = full_polish(best_seq, time_budget=alloc)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final: {best_score:.8f}")
        else:
            print(f"  Final: {obj_final:.8f} (no improvement)")

        # Last-ditch stochastic attempts
        remaining = deadline - time.monotonic()
        if remaining > 10 and best_seq is not None:
            n_final = len(best_seq)
            rng = np.random.default_rng(seed + 777777)
            for _ in range(min(5, int(remaining // 5))):
                if time.monotonic() > deadline - 5:
                    break
                scale = rng.uniform(0.02, 0.15)
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
                a_pert = normalize_simplex(a_pert)
                t_alloc = min(remaining * 0.1, 8.0)
                a_pert2, obj_pert2 = mirror_descent_optimize(
                    a_pert, time_budget=max(t_alloc, 2.0),
                    tau_init=rng.uniform(0.003, 0.015),
                    eta_init=rng.uniform(0.3, 0.6)
                )
                if obj_pert2 < best_score:
                    best_score = obj_pert2
                    best_seq = a_pert2.copy()
                    print(f"  Stochastic: {best_score:.8f} ** **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
