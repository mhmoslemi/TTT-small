def coord_descent_golden(a, max_sweeps, deadline, random_order=False, seed=0, golden_iters=5):
    """
    Fast coordinate descent with golden section search.
    Normalizes to sum=1 and minimizes 2n*max(conv(a,a)).
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order and sweep % 2 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved = False

        for i in order:
            old = a[i]

            # Determine search range for a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()

            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            def local_obj(x):
                delta = x - old
                if abs(delta) < 1e-15:
                    return best_obj
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                val = 2.0 * n * conv.max()
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
                return val

            # Golden section search in [0, upper_bound]
            # Upper bound: try a few multiples of old
            upper = max(old * 3.0, 0.01) if old > 0 else 0.05

            gr = (1 + np.sqrt(5)) / 2
            c = upper / gr
            d = upper - c

            fc = local_obj(c)
            fd = local_obj(d)

            for _ in range(golden_iters):
                if fc < fd:
                    upper = d
                    d = c
                    fd = fc
                    c = upper / gr
                    fc = local_obj(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = upper - c
                    fd = local_obj(d)

            # Also check 0 and the golden result
            best_x = old
            best_f = local_obj(old)
            candidates = [0.0, c, d, upper * 0.5]
            for x in candidates:
                x = max(0.0, min(x, upper))
                f = local_obj(x)
                if f < best_f:
                    best_f = f
                    best_x = x

            if best_x != old and abs(best_x - old) > 1e-13:
                delta = best_x - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_x
                s = s - old + best_x
                improved = True

        if s > 1e-10:
            current_obj = 2.0 * n * conv.max() / (s * s)
        else:
            current_obj = float('inf')

        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a

def lp_subgradient(a, max_iters, deadline, n_top=10, verbose=False):
    """
    LP-based subgradient method. At each step:
    1. Find the top-k positions where conv is close to max
    2. Solve LP to find best direction that reduces these positions
    3. Take a step in that direction
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Find top-k positions
        k = min(n_top, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]

        # Solve LP: minimize t
        # s.t. (a*g)[k_idx] <= M - t  for k in top_idx
        #      sum(g) = 1
        #      g >= 0
        # Where (a*g)[k] = sum_j a[k-j]*g[j]
        # Linearizing: we want to find g that minimizes the max of (a*g) at top positions
        # Approximation: minimize t s.t. (a*g)[k] <= M*(1-t/M) ... 
        # Actually let's use the standard approach:
        # minimize t
        # s.t. (a*g)[k] <= M  for k in top_idx (keep the bound)
        #      sum(g) = 1
        #      g >= 0
        # This gives the best g that doesn't increase the max at top positions.
        # Then we take a step towards g.
        #
        # Better: minimize t
        # s.t. (a*g)[k] - 2*(a*a)[k] + M <= t  for k in top_idx
        #      sum(g) = 1  
        #      g >= 0
        # This linearizes the quadratic around current a.
        # (a*g)[k] ≈ (a*a)[k] + (a*g)[k] - (a*a)[k] ... hmm this isn't right.
        #
        # The correct linearization:
        # We want to minimize max_k (a+t*(g-a)) * (a+t*(g-a)) at top positions
        # ≈ max_k c[k] + 2t * (a*(g-a))[k]
        # So we want to minimize max_k (c[k] + 2t*(a*(g-a))[k])
        # = c_max + 2t * max_k ((a*(g-a))[k]) for k in top
        # To reduce this, we want (a*(g-a))[k] < 0 for top positions
        # i.e., (a*g)[k] < (a*a)[k] = c[k] for top k
        #
        # LP: minimize t
        # s.t. (a*g)[k] - c[k] <= t  for k in top_idx
        #      sum(g) = 1
        #      g >= 0

        rows = []
        cols = []
        vals = []
        b_ub = []

        for k_val in top_idx:
            j_min = max(0, k_val - n + 1)
            j_max = min(n - 1, k_val)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub))
                cols.append(j)
                vals.append(a[k_val - j])
            rows.append(len(b_ub))
            cols.append(n)  # t variable
            vals.append(-1.0)
            b_ub.append(c[k_val])

        if not rows:
            no_improve += 1
            continue

        A = coo_matrix((vals, (rows, cols)), shape=(len(b_ub), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_eq = coo_matrix((np.ones(n), ([0] * n, list(range(n)))), shape=(1, n + 1)).tocsr()
        b_eq = [1.0]
        bounds = [(0, None)] * n + [(None, None)]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    # Line search
                    found = False
                    for frac in [0.5, 0.3, 0.2, 0.15, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns with various frequencies and amplitudes
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    # Product of cosines (beating patterns)
    for f1, f2 in [(1, 2), (1, 3), (2, 3), (3, 5), (1, 5)]:
        inits.append(project_simplex(1.0 + 0.5 * np.cos(2 * np.pi * f1 * t) * np.cos(2 * np.pi * f2 * t)))

    # Gaussian bumps
    for center in [0.1, 0.3, 0.5, 0.7, 0.9]:
        for width in [0.05, 0.1, 0.2, 0.3]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.05))

    # Power-law
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    # Smooth random
    ksize = max(3, n // 8)
    kernel = np.ones(ksize) / ksize
    for _ in range(8):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    # Sawtooth / triangular
    inits.append(project_simplex(np.abs(2 * t - 1) + 0.1))
    inits.append(project_simplex(1 - np.abs(2 * t - 1) + 0.1))

    # Periodic patterns
    for period in [2, 3, 4]:
        pattern = np.tile([1.0, 0.5][:period], n // period + 1)[:n]
        inits.append(project_simplex(pattern))

    return inits

"""
Multi-method optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) fast coordinate descent with adaptive ratios, (2) Nesterov-accelerated
smooth gradient descent on log-sum-exp, (3) LP subgradient with active set tracking,
(4) multi-length exploration with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - 1.0) / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = (cssv[k[-1]] - 1.0) / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent(a, max_sweeps, deadline, random_order=False, seed=0, verbose=False):

    Fast coordinate descent with adaptive candidate ratios.
    Normalizes to sum=1, minimizes 2n*max(conv(a,a)).
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)

        for i in order:
            old = a[i]

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()

            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            # Adaptive candidates based on current value
            if old < 1e-12:
                candidates = [0.0, 0.001 / n, 0.005 / n, 0.01 / n, 0.02 / n]
            else:
                base_candidates = np.array([0.0, 0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5])
                candidates = np.unique(np.concatenate([canvas if False else base_candidates * old]))

            best_val = old
            best_local = 2.0 * n * conv.max()

            for x in candidates:
                x = float(max(0.0, x))
                if abs(x - old) < 1e-14:
                    continue

                delta = x - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-13:

                conv[lo:hi + 1] += 2.0 * delta * a_slice

    return a


def coord_descent_binary(a, max_sweeps, deadline, random_order=False, seed=0,
                          binary_iters=6, verbose=False):

    Coordinate descent with binary search refinement for better per-coordinate optimization.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

        if time.monotonic() > deadline:
            break

            order = rng.permutation(n)
        else:

            a_slice = a[lo_idx - i:hi_idx - i + 1].copy()

            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            def eval_at(x):

                conv[lo_idx:hi_idx + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                new_s = s - old + x
                if new_s < 1e-10:
                    val = float('inf')
                else:
                    val = 2.0 * n * conv.max() / (new_s * new_s)
                conv[lo_idx:hi_idx + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
                return val

            if old < 1e-12:

            else:
                candidates = list(base_candidates * old)

            best_val = old
            best_f = eval_at(old)
            for x in candidates:
                x = max(0.0, float(x))
                f = eval_at(x)
                if f < best_f:

                        # Minimum is near mid
                        hi_b = (mid + hi_b) / 2.0
                        lo_b = (lo_b + mid) / 2.0

                for x in [lo_b, hi_b, (lo_b + hi_b) / 2.0]:
                    f = eval_at(x)
                    if f < best_f:
                        best_f = f
                        best_val = x

            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo_idx:hi_idx + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val

        if s > 1e-10:
            current_obj = 2.0 * n * conv.max() / (s * s)
        else:
            current_obj = float('inf')

        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 60:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def smooth_nesterov(a, max_iters, deadline, tau_init=40.0, tau_max=400.0,
                    lr_init=0.08, beta=0.9, verbose=False):
    """
    Nesterov-accelerated gradient descent on log-sum-exp smooth approximation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)
    no_improve = 0
    t_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_init + (tau_max - tau_init) * frac

        # Compute gradient at y
        c = fast_conv(y)
        c_shifted = c - c.max()
        exp_c = np.exp(np.minimum(tau * c_shifted, 50.0))
        w = exp_c / exp_c.sum()

        # Compute gradient via FFT
        a_rev = a[::-1]
        w_rev = w[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        Wrev_fft = np.fft.rfft(w_rev, size)

        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wrev_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad[i] += conv2[idx2]

        grad *= tau

        # Project to tangent space
        grad = grad - np.mean(grad)

        # Nesterov update
        t_count += 1
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * t_count * t_count)) / 2.0
        gamma_prev = (1.0 + np.sqrt(1.0 + 4.0 * (t_count - 1) * (t_count - 1))) / 2.0 if t_count > 1 else 1.0

        lr = lr_init / (1.0 + 0.05 * np.sqrt(t_count))
        v = beta * v + lr * grad
        y = a - v / gamma
        y = project_simplex(y)
        a_new = y - (gamma_prev / gamma) * (y - a)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                v = np.zeros(n)
                lr_init = max(0.01, lr_init * 0.5)
                a = best_a + np.random.default_rng(it).normal(0, 0.015, n)
                a = project_simplex(a)
                y = a.copy()
                no_improve = 0

        a = a_new

    return best_a, best_obj


def lp_active_set(a, max_iters, deadline, n_top=8, verbose=False):
    """
    LP-based method using active set of near-maximum convolution positions.
    Solves LP to find best direction reducing active constraints.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Active set: positions within 2% of max
        threshold = M * 0.98
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(c)[::-1][:n_top]
        else:
            # Limit to top n_top if too many
            if len(active) > n_top:
                c_active = c[active]
                top_in_active = active[np.argsort(c_active)[::-1][:n_top]]
                active = top_in_active

        # LP: minimize t
        # s.t. (a*g)[k] - c[k] <= t for k in active
        #      sum(g) = 1
        #      g >= 0
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k_val in active:
            j_min = max(0, k_val - n + 1)
            j_max = min(n - 1, k_val)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub))
                cols.append(j)
                vals.append(a[k_val - j])
            rows.append(len(b_ub))
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(float(c[k_val]))

        if not rows:
            no_improve += 1
            continue

        A = coo_matrix((vals, (rows, cols)), shape=(len(b_ub), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_eq = coo_matrix((np.ones(n), ([0] * n, list(range(n)))), shape=(1, n + 1)).tocsr()
        b_eq = [1.0]
        bounds = [(0, None)] * n + [(None, None)]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [0.5, 0.35, 0.25, 0.15, 0.1]:
                        a_try = (1.0 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 25:
            break

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.015, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.04, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.08, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.12, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns
    for freq in [1, 2, 3, 4, 5, 7, 11, 13]:
        for amp in [0.2, 0.4, 0.6, 0.8, 0.95]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    # Product of cosines
    for f1, f2 in [(1, 2), (1, 3), (2, 3), (1, 5), (3, 5), (2, 5)]:
        inits.append(project_simplex(1.0 + 0.5 * np.cos(2 * np.pi * f1 * t) * np.cos(2 * np.pi * f2 * t)))
        inits.append(project_simplex(1.0 + 0.7 * np.sin(2 * np.pi * f1 * t) * np.sin(2 * np.pi * f2 * t)))

    # Gaussian bumps
    for center in [0.1, 0.3, 0.5, 0.7, 0.9]:
        for width in [0.05, 0.1, 0.15, 0.25, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.03))

    # Double Gaussian
    for c1, c2 in [(0.3, 0.7), (0.2, 0.8), (0.4, 0.6)]:
        inits.append(project_simplex(
            np.exp(-((t - c1) / 0.15) ** 2) + np.exp(-((t - c2) / 0.15) ** 2) + 0.05))

    # Power-law
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1.0 - t) + 0.01) ** alpha))

    # Smooth random
    ksize = max(3, n // 10)
    kernel = np.ones(ksize) / ksize
    for _ in range(10):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.03))

    # Sawtooth
    inits.append(project_simplex(np.abs(2 * t - 1) + 0.1))
    inits.append(project_simplex(1.0 - np.abs(2 * t - 1) + 0.1))

    # Periodic
    for period in [2, 3, 4, 5]:
        for ratio in [0.3, 0.5, 0.7]:
            pattern = np.tile([1.0, ratio], n // 2 + 1)[:n]
            inits.append(project_simplex(pattern))

    # Chromatic / alternating smooth
    inits.append(project_simplex(1.0 + 0.4 * np.cos(4 * np.pi * t) * np.exp(-2 * t * (1 - t))))

    return inits


def try_update_best(best_seq, best_score, a_new):
    """Try updating best sequence and score. Returns (best_seq, best_score, improved)."""
    a_new = np.maximum(a_new, 0)
    if a_new.sum() < 1e-10:
        return best_seq, best_score, False
    a_new = a_new / a_new.sum()
    obj = compute_J(a_new)
    if np.isfinite(obj) and obj < best_score:
        return a_new.copy(), obj, True
    return best_seq, best_score, False


def solve_pipeline(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence.
    Returns (best_a, best_obj).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 1: Fast coordinate descent
    t1 = time.monotonic() + 0.30 * remaining
    a_cd = coord_descent(a, max_sweeps=8000, deadline=t1, random_order=True, seed=42)
    best_a, best_obj, _ = try_update_best(best_a, best_obj, a_cd)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 2: Nesterov smooth gradient
    t2 = time.monotonic() + 0.30 * remaining
    a_sg, obj_sg = smooth_nesterov(a, max_iters=3000, deadline=t2,
                                    tau_init=35.0, tau_max=250.0, lr_init=0.06)
    if obj_sg < best_obj:
        best_a = a_sg.copy()
        best_obj = obj_sg
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 3: LP active set
    t3 = time.monotonic() + 0.25 * remaining
    a_lp, obj_lp = lp_active_set(a, max_iters=80, deadline=t3, n_top=6)
    if obj_lp < best_obj:
        best_a = a_lp.copy()
        best_obj = obj_lp
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj

    # Stage 4: Binary search CD polish
    t4 = deadline - 3
    a_bin = coord_descent_binary(a, max_sweeps=3000, deadline=t4,
                                  random_order=True, seed=123, binary_iters=4)
    best_a, best_obj, _ = try_update_best(best_a, best_obj, a_bin)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses diverse initializations, coordinate descent, Nesterov smooth gradient,
    LP active set, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    # === PHASE 1: Optimize at parent length with diverse inits ===
    phase1_deadline = t_start + 380.0
    n_parent = len(parent)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-init optimization at n={n_parent}", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    # Score and sort initializations
    init_scores = []
    for idx, inp in enumerate(inits):
        inp = np.maximum(inp, 0)
        if inp.sum() < 1e-10:
            continue
        inp = inp / inp.sum()
        obj = compute_J(inp)
        init_scores.append((obj, idx, inp))
    init_scores.sort(key=lambda x: x[0])

    n_tries = min(12, len(init_scores))
    per_try_time = (phase1_deadline - time.monotonic()) / max(1, n_tries)

    for trial in range(n_tries):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 30:
            break
        alloc = min(per_try_time, remaining_check - 10)
        if alloc < 25:
            break

        obj_init, idx, a0 = init_scores[trial]

        a_opt, obj_opt = solve_pipeline(a0, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  Init {trial} (initidx={idx}, J0={obj_init:.6f}): J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Multi-length exploration ===
    phase2_deadline = t_start + 880.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidate_ns = [256, 320, 384, 448, 512, 576, 640, 768, 896, 1024, 1152, 1280]
    candidate_ns = [n for n in candidate_ns if n != n_parent]
    candidate_ns.sort(key=lambda n: abs(n - n_parent))

    for new_n in candidate_ns[:7]:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 60:
            break
        alloc = min(70.0, remaining_check - 20)
        if alloc < 40:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = solve_pipeline(a_ext, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive optimization on best ===
    phase3_deadline = deadline
    remaining = phase3_deadline - time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive optimization (budget {remaining:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and remaining > 30:
        a_curr = best_seq.copy()

        # Multiple CD passes with different seeds
        for cd_pass in range(4):
            remaining = phase3_deadline - time.monotonic()
            if remaining < 20:
                break
            t_pass = time.monotonic() + remaining * 0.18
            seed_val = 200 + cd_pass * 11
            a_curr = coord_descent(a_curr, max_sweeps=15000, deadline=t_pass,
                                    random_order=True, seed=seed_val)
            update_best(a_curr)
            a_curr = best_seq.copy()

        # Nesterov smooth gradient with different params
        remaining = phase3_deadline - time.monotonic()
        if remaining > 40:
            t_pass = time.monotonic() + remaining * 0.25
            a_sg, obj_sg = smooth_nesterov(a_curr, max_iters=5000, deadline=t_pass,
                                            tau_init=25.0, tau_max=200.0, lr_init=0.05)
            update_best(a_sg)
            a_curr = best_seq.copy()

        # LP active set
        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            t_pass = time.monotonic() + remaining * 0.25
            a_lp, obj_lp = lp_active_set(a_curr, max_iters=150, deadline=t_pass, n_top=10)
            update_best(a_lp)
            a_curr = best_seq.copy()

        # Binary search CD final polish
        remaining = phase3_deadline - time.monotonic()
        if remaining > 20:
            a_final = coord_descent_binary(a_curr, max_sweeps=5000,
                                            deadline=phase3_deadline - 5,
                                            random_order=True, seed=777, binary_iters=5)
            update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
