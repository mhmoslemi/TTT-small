"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses KKT-based power iteration, LP alternating with cubic line search,
simulated annealing with peak-targeting moves, spectral equalization,
and multi-n exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def cubic_line_search(f, x0, x1, max_iter=20):
    """Cubic interpolation line search."""
    if x1 <= x0:
        return x0
    # Find minimum in [x0, x1] using cubic interpolation
    # Start with golden section, then refine with cubic
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = x0
    b_pt = x1
    c_pt = x1 - gr * (x1 - x0)
    d_pt = x0 + gr * (x1 - x0)
    fc = f(c_pt)
    fd = f(d_pt)
    
    for _ in range(10):
        if b_pt - a_pt < 1e-6:
            break
        if fc < fd:
            b_pt = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b_pt - gr * (b_pt - a_pt)
            fc = f(c_pt)
        else:
            a_pt = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a_pt + gr * (b_pt - a_pt)
            fd = f(d_pt)
    
    # Refine with cubic interpolation in the best bracket
    lo = a_pt if f(a_pt) < f(b_pt) else b_pt
    hi = b_pt if f(a_pt) < f(b_pt) else a_pt
    mid = (lo + hi) / 2
    fl = f(lo)
    fh = f(hi)
    fm = f(mid)
    
    for _ in range(10):
        if hi - lo < 1e-7:
            break
        # Cubic interpolation
        h1 = lo - mid
        h2 = hi - mid
        if abs(h1) < 1e-12 or abs(h2) < 1e-12:
            break
        d1 = (fm - fl) / h1
        d2 = (fm - fh) / h2
        d = d1 - d2
        if abs(d) < 1e-12:
            break
        # Candidate point
        num = d1 * h1 * h1 * d2 - d2 * h2 * h2 * d1
        den = d1 * h1 * d2 - d2 * h2 * d1
        if den != 0:
            x_new = mid + num / den
        else:
            x_new = mid
        x_new = max(lo + 1e-8, min(hi - 1e-8, x_new))
        f_new = f(x_new)
        if f_new < fl:
            hi = mid
            fh = fm
            mid = x_new
            fm = f_new
        elif f_new < fh:
            lo = mid
            fl = fm
            mid = x_new
            fm = f_new
        else:
            if x_new > mid:
                hi = x_new
                fh = f_new
            else:
                lo = x_new
                fl = f_new
    
    return mid

def kkt_power_iteration(a, max_iters=100, time_budget=30.0, top_k=5):
    """
    KKT-based power iteration: at the optimum, the weighted sum of
    shifted a's should be constant on the support. We iterate toward this.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Find top K peaks
        peak_indices = np.argsort(conv)[-top_k:]
        
        # Compute "pressure" at each position: p[j] = sum of a[k-j] for k in peaks
        # This is the subgradient direction
        p = np.zeros(n)
        for k in peak_indices:
            # a[k-j] for j = 0..n-1 means: for each j, add a[k-j] if 0 <= k-j < n
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    p[j] += a[idx]
        
        # Water-filling update: a_new[j] ∝ 1/p[j] (inverse pressure)
        # This is motivated by the KKT condition that p[j] * a[j] ≈ const
        with np.errstate(divide='ignore', invalid='ignore'):
            a_new = np.where(p > 1e-12, 1.0 / p, n)  # Default to uniform if p=0
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Blend with current (to avoid oscillation)
        alpha = 0.3 + 0.4 * (it / max_iters)  # Increase blend over time
        a_try = (1 - alpha) * a + alpha * a_new
        a_try = normalize(a_try)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        else:
            # Try smaller step
            a_try = (1 - alpha * 0.3) * a + alpha * 0.3 * a_new
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
            # If still no improvement, try peak-targeting move
            else:
                # Reduce the contribution to the tallest peak
                k_star = int(np.argmax(conv))
                # Find the pair (i, k_star-i) with largest product
                best_i = -1
                best_prod = -1
                for i in range(n):
                    j = k_star - i
                    if 0 <= j < n:
                        prod = a[i] * a[j]
                        if prod > best_prod:
                            best_prod = prod
                            best_i = i
                if best_i >= 0:
                    j = k_star - best_i
                    if j != best_i and 0 <= j < n:
                        # Move some mass from the larger of a[best_i], a[j] to a random position
                        if a[best_i] > a[j]:
                            src = best_i
                        else:
                            src = j
                        tgt = int(np.random.randint(0, n))
                        if tgt != src:
                            delta = 0.02 * a[src]
                            a_try = a.copy()
                            a_try[src] -= delta
                            a_try[tgt] += delta
                            a_try = normalize(a_try)
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                a = a_try
    
    return best_a

def lp_alternate_v2(a, max_iters=40, time_budget=60.0):
    """LP alternating with cubic line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Try slightly different M values to get different LP solutions
        M_try = M * (1.0 + 0.01 * (it % 3 - 1))  # Vary by ±1%
        
        g = solve_lp(a, M_try, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break
        
        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g
            a_try = normalize(a_try)
            return compute_obj(a_try)
        
        t_opt = cubic_line_search(obj_at_t, 0.0, 1.0)
        
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Perturb
            rng = np.random.default_rng((it * 31337 + 7) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03 + 0.01 * (it % 7), n)))
    
    return best_a

def simulated_annealing(a, time_budget=60.0, t_start=0.1, t_end=0.001):
    """
    Simulated annealing with peak-targeting moves.
    Moves: move mass between two positions, biased toward reducing the tallest peak.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    n_steps = 0
    while time.monotonic() - start < time_budget:
        n_steps += 1
        # Temperature schedule (geometric cooling)
        progress = min(1.0, n_steps / (n_steps + 100))
        temp = t_start * (t_end / t_start) ** progress
        
        # Compute current peak
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Choose a move
        move_type = rng.integers(0, 4)
        a_try = a.copy()
        
        if move_type == 0:
            # Peak-targeting: reduce contribution to tallest peak
            # Find top contributor to peak at k_star
            contribs = []
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))
            contribs.sort(reverse=True)
            if contribs:
                _, i1, i2 = contribs[0]
                src = i1 if a[i1] > a[i2] else i2
                tgt = int(rng.integers(0, n))
                if tgt != src:
                    delta = rng.uniform(0.01, 0.05) * a[src]
                    a_try[src] -= delta
                    a_try[tgt] += delta
        elif move_type == 1:
            # Random pair swap
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2:
                delta = rng.uniform(0.01, 0.03) * min(a[i1], a[i2])
                if a[i1] >= delta:
                    a_try[i1] -= delta
                    a_try[i2] += delta
                else:
                    a_try[i2] -= delta
                    a_try[i1] += delta
        elif move_type == 2:
            # Block reversal
            block_len = int(rng.integers(n // 10, n // 3))
            start_idx = int(rng.integers(0, n - block_len))
            a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]
        else:
            # Gaussian perturbation
            a_try = a * np.exp(rng.normal(0, 0.02, n))
        
        a_try = normalize(a_try)
        obj_try = compute_obj(a_try)
        
        # Accept or reject
        if obj_try < current_obj or rng.random() < np.exp(-(obj_try - current_obj) / (temp * best_obj)):
            current_obj = obj_try
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
        
        # Periodically recompute to save time
        if n_steps % 50 == 0:
            pass  # Already computing conv above
    
    return best_a

def spectral_polish(a, max_iters=40, time_budget=30.0):
    """Spectral equalization with adaptive alpha."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        
        alpha = 0.3 + 0.5 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Blend
        blend = 0.5 + 0.3 * (it / max_iters)
        a_try = (1 - blend) * normalize(a) + blend * a_new
        a_try = normalize(a_try)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        else:
            a = 0.7 * normalize(a) + 0.3 * a_try
            a = normalize(a)
    
    return best_a

def coordinate_polish(a, time_budget=20.0):
    """
    Coordinate descent: for each position, try small adjustments to reduce the max peak.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Shuffle order for diversity
    order = rng.permutation(n)
    
    while time.monotonic() - start < time_budget:
        improved = False
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        for idx in order:
            if time.monotonic() - start > time_budget:
                    break
            j = int(idx)
            
            # Try increasing and decreasing a[j]
            for delta_frac in [0.02, 0.01, -0.02, -0.01]:
                a_try = a.copy()
                delta = delta_frac * a[j]
                a_try[j] += delta
                if a_try[j] < 0:
                    continue
                # Redistribute the change to maintain sum
                change = -delta
                # Add to a random other position
                tgt = int(rng.integers(0, n))
                if tgt == j:
                    tgt = (j + 1) % n
                a_try[tgt] += change
                a_try = normalize(a_try)
                
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved:
            # Shuffle and try again
            order = rng.permutation(n)
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

"""
Multi-strategy optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines KKT-based power iteration, LP alternating with cubic line search,
simulated annealing with peak-targeting moves, spectral equalization,
coordinate descent polishing, multi-n exploration, and diverse initializations.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix for conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def cubic_line_search(f, x0, x1, max_iter=15):
    """Line search combining golden section and cubic interpolation."""
    if x1 <= x0:
        return x0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = x0
    b_pt = x1
    c_pt = b_pt - gr * (b_pt - a_pt)
    d_pt = a_pt + gr * (b_pt - a_pt)
    fc = f(c_pt)
    fd = f(d_pt)

    for _ in range(8):
        if b_pt - a_pt < 1e-6:
            break
        if fc < fd:
            b_pt = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b_pt - gr * (b_pt - a_pt)
            fc = f(c_pt)
        else:
            a_pt = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a_pt + gr * (b_pt - a_pt)
            fd = f(d_pt)

    lo = a_pt if f(a_pt) < f(b_pt) else b_pt
    hi = b_pt if f(a_pt) < f(b_pt) else a_pt
    if hi - lo < 1e-7:
        return (lo + hi) / 2.0
    mid = (lo + hi) / 2.0
    fl = f(lo)
    fh = f(hi)
    fm = f(mid)

    for _ in range(8):
        if hi - lo < 1e-8:
            break
        h1 = mid - lo
        h2 = hi - mid
        if abs(h1) < 1e-12 or abs(h2) < 1e-12:

        d1 = (fm - fl) / h1
        d2 = (fm - fh) / h2
        d = d1 - d2
        if abs(d) < 1e-14:

        b_val = d1 * h1 - d2 * h2
        den = d1 * d2
        if abs(den) < 1e-14:

            den2 = d1 * h1 * d2 * h2 - d2 * h2 * d1 * h1

            fh = fm
            mid = x_new
            fm = f_new

            fl = fm
            mid = x_new
            fm = f_new
        else:

                fh = f_new
            else:

def compute_pressure(a, peak_indices):
    """Compute pressure vector p[j] = sum_{k in peaks} a[k-j]."""
    n = len(a)
    p = np.zeros(n)
    for k in peak_indices:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            seg = a[k - j_end:k - j_start + 1][::-1]
            p[j_start:j_end + 1] += seg
    return p


def kkt_power_iteration(a, max_iters=80, time_budget=40.0, top_k=5, blend_rate=0.4):
    """
    KKT-inspired power iteration. At the optimum, the weighted sum of
    shifted a's (over active peaks) should be constant on the support.
    We use inverse-pressure updates to approach this fixed point.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(7)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Find top K peaks (with slight randomization for diversity)
       threshold = cmax * 0.95
        peak_mask = conv >= threshold
        peak_indices = np.where(peak_mask)[0]
        if len(peak_indices) > top_k:
            # Keep only the top_k
            peak_indices = peak_indices[np.argsort(conv[peak_indices])][-top_k:]
        elif len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])

        # Compute pressure
        p = compute_pressure(a, peak_indices)
        p_mean = float(np.mean(p[p > 1e-10])) if np.any(p > 1e-10) else 1.0

        # Inverse-pressure update: a_new[j] ∝ a[j] * (p_mean / p[j])
        with np.errstate(divide='ignore', invalid='ignore'):
            ratio = np.where(p > 1e-12, p_mean / p, 1.0)
        ratio = np.clip(ratio, 0.1, 10.0)
        a_new = a * ratio
        a_new = normalize(a_new)

        # Adaptive blend (increasing over iterations)
        alpha = blend_rate * (0.5 + 0.5 * it / max_iters)
        a_try = (1.0 - alpha) * a + alpha * a_new
        a_try = normalize(a_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        else:
            # Try smaller step
            alpha_small = alpha * 0.3
            a_try = (1.0 - alpha_small) * a + alpha_small * a_new
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
            else:
                # Peak-targeting move: reduce the top contributor
                k_star = int(np.argmax(conv))
                # Find top contributor pair
                best_i = -1
                best_prod = -1.0
                for i in range(n):
                    j = k_star - i
                    if 0 <= j < n:
                        prod = a[i] * a[j]
                        if prod > best_prod:
                            best_prod = prod
                            best_i = i
                if best_i >= 0:
                    j = k_star - best_i
                    if 0 <= j < n and j != best_i:
                        src = best_i if a[best_i] > a[j] else j
                        tgt = int(rng.integers(0, n))
                        if tgt != src:
                            delta = 0.015 * a[src]
                            a_try = a.copy()
                            a_try[src] -= delta
                            a_try[tgt] += delta
                            a_try = normalize(a_try)
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                a = a_try

    return best_a

    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

        M = float(np.max(conv))

        # Perturb M slightly for diversity
        perturbation = rng.uniform(-0.005, 0.01)
        M_try = M * (1.0 + perturbation)
        M_try = max(M_try, M * 0.9)

        g = solve_lp(a, M_try, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break

        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g
            a_try = normalize(a_try)
            return compute_obj(a_try)

        t_opt = cubic_line_search(obj_at_t, 0.0, 1.0)

        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Perturb to escape
            noise_scale = 0.02 + 0.02 * (it % 5) / 5.0
            a = normalize(best_a * np.exp(rng.normal(0, noise_scale, n)))

    return best_a


def simulated_annealing(a, time_budget=50.0, t_start=0.08, t_end=0.001):
    """Simulated annealing with peak-targeting moves."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(555)
    step_count = 0

    while time.monotonic() - start < time_budget:
        step_count += 1
        progress = min(1.0, step_count / 2000.0)
        temp = t_start * (t_end / t_start) ** progress

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        cmax = float(conv[k_star])

        move_type = rng.integers(0, 5)
        a_try = a.copy()

        if move_type == 0:
            # Peak-targeting: reduce top contributor to tallest peak
            contribs = []
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))
            contribs.sort(reverse=True)
            if contribs:
                _, i1, i2 = contribs[0]
                src = i1 if a[i1] > a[i2] else i2
                tgt = int(rng.integers(0, n))
                if tgt != src:
                    delta = rng.uniform(0.005, 0.04) * a[src]
                    a_try[src] -= delta
                    a_try[tgt] += delta
        elif move_type == 1:
            # Directed pair: move mass from high to low pressure
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2:
                delta = rng.uniform(0.005, 0.02) * min(a[i1], a[i2])
                src = i1 if a[i1] > a[i2] else i2
                tgt = i1 if src == i2 else i2
                if a_try[src] >= delta:
                    a_try[src] -= delta
                    a_try[tgt] += delta
        elif move_type == 2:
            # Block reversal
            block_len = int(rng.integers(max(5, n // 15), n // 4))
            start_idx = int(rng.integers(0, n - block_len))
            a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]
        elif move_type == 3:
            # Mild Gaussian
            a_try = a * np.exp(rng.normal(0, 0.015, n))
        else:
            # Swap two values
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2:
                tmp = a_try[i1]
                a_try[i1] = a_try[i2]
                a_try[i2] = tmp

        a_try = normalize(a_try)
        obj_try = compute_obj(a_try)

        diff = obj_try - current_obj
        if diff < 0 or rng.random() < np.exp(-diff / (temp * max(best_obj, 1e-6))):
            current_obj = obj_try
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a


def spectral_polish(a, max_iters=40, time_budget=30.0):
    """Adaptive spectral equalization (POCS-style)."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Adaptive alpha: start gentle, get aggressive
        alpha = 0.2 + 0.6 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        blend = 0.4 + 0.4 * (it / max_iters)
        a_try = (1.0 - blend) * normalize(a) + blend * a_new
        a_try = normalize(a_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        else:
            a = 0.7 * normalize(a) + 0.3 * a_try
            a = normalize(a)

    return best_a


def coordinate_polish(a, time_budget=20.0):
    """Coordinate descent targeting the max convolution peak."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77)
    order = rng.permutation(n)

    while time.monotonic() - start < time_budget:
        improved = False
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        for idx in order:
            if time.monotonic() - start > time_budget:
                break
            j = int(idx)

            for delta_frac in [0.015, 0.008, -0.015, -0.008]:
                a_try = a.copy()
                delta = delta_frac * a[j]
                a_try[j] += delta
                if a_try[j] < 0:
                    continue
                # Redistribute to maintain sum
                change = -delta
                tgt = int(rng.integers(0, n))
                if tgt == j:
                    tgt = (j + 1) % n
                a_try[tgt] += change
                a_try = normalize(a_try)

                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            order = rng.permutation(n)

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def diverse_initialization(n, seed):
    """Generate diverse initial sequences for exploration."""
    rng = np.random.default_rng(seed)
    strategies = [
        # Triangle
        lambda i: np.concatenate([np.linspace(1, n/2, n//2), np.linspace(n/2, 1, n - n//2)]),
        # Gaussian
        lambda i: np.exp(-0.5 * ((np.arange(n) - n/2) / (n/4))**2),
        # Raised cosine
        lambda i: 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n)),
        # Random Dirichlet (concentrated)
        lambda i: rng.dirichlet(np.ones(n) * 0.3),
        # Random Dirichlet (spread)
        lambda i: rng.dirichlet(np.ones(n) * 2.0),
        # Sine-modulated
        lambda i: np.maximum(0, 1 + 0.5 * np.sin(2 * np.pi * 3 * np.arange(n) / n)),
    ]
    idx = seed % len(strategies)
    a = strategies[idx](np.arange(n))
    return normalize(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining KKT power iteration,
    LP alternating, simulated annealing, spectral methods, and multi-n search.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # =========================================================
    # Phase 0: Diverse initializations (quick, to find better start)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 0: Diverse initializations")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 1200:
        for init_seed in range(6):
            a_init = diverse_initialization(n_parent, seed + init_seed)
            # Quick KKT to optimize
            a_opt = kkt_power_iteration(a_init, max_iters=20, time_budget=8.0, top_k=5)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Init {init_seed}: {best_score:.8f} **")
            else:
                print(f"  Init {init_seed}: {obj_opt:.8f}")

    # =========================================================
    # Phase 1: KKT Power Iteration (main novel optimizer)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: KKT Power Iteration (n={len(best_seq)})")
    print(f"{'='*60}")

    a = normalize(best_seq.copy())
    for top_k in [3, 5, 7, 10]:
        if time.monotonic() > deadline - 700:
            break
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.1, 50.0)

        a_new = kkt_power_iteration(a, max_iters=100, time_budget=alloc, top_k=top_k)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  KKT top_k={top_k}: {best_score:.8f} **")
        else:
            print(f"  KKT top_k={top_k}: {obj_new:.8f}")
            a = 0.8 * normalize(a) + 0.2 * normalize(a_new)

    # =========================================================
    # Phase 2: Intensive LP Alternating
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP Alternating (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        n_cur = len(a)
        lp_cycles = 6 if n_cur <= 800 else 4

        for cycle in range(lp_cycles):
            if time.monotonic() > deadline - 500:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.12, 60.0)

            a_new = lp_alternate_v2(a, max_iters=35, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  LP cycle {cycle}: {best_score:.8f} **")
            else:
                a = a_new
                print(f"  LP cycle {cycle}: {obj_new:.8f}")

    # =========================================================
    # Phase 3: KKT + Spectral refinement
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: KKT + Spectral")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())

        if time.monotonic() < deadline - 350:
            a_spec = spectral_polish(a, max_iters=50, time_budget=30.0)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                a = a_spec
                print(f"  Spectral: {best_score:.8f} **")
            else:
                print(f"  Spectral: {obj_spec:.8f}")
                a = 0.7 * normalize(a) + 0.3 * normalize(a_spec)
                a = normalize(a)

        if time.monotonic() < deadline - 300:
            a_kkt = kkt_power_iteration(a, max_iters=80, time_budget=40.0, top_k=7)
            obj_kkt = compute_obj(a_kkt)
            if obj_kkt < best_score:
                best_score = obj_kkt
                best_seq = a_kkt.copy()
                print(f"  KKT refined: {best_score:.8f} **")
            else:
                print(f"  KKT refined: {obj_kkt:.8f}")

    # =========================================================
    # Phase 4: Simulated Annealing (multiple restarts)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Simulated Annealing")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed + 100)

        for sa_restart in range(3):
            if time.monotonic() > deadline - 200:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.15, 50.0)

            if sa_restart == 0:
                a_sa = simulated_annealing(best_seq, time_budget=alloc, t_start=0.06, t_end=0.0005)
            elif sa_restart == 1:
                # Perturbed start
                a_pert = best_seq * np.exp(rng.normal(0, 0.06, n_cur))
                a_pert = normalize(a_pert)
                a_sa = simulated_annealing(a_pert, time_budget=alloc, t_start=0.05, t_end=0.001)
            else:
                # Blend with random
                a_rand = rng.dirichlet(np.ones(n_cur) * 1.0)
                a_mix = 0.7 * normalize(best_seq) + 0.3 * a_rand
                a_mix = normalize(a_mix)
                a_sa = simulated_annealing(a_mix, time_budget=alloc, t_start=0.04, t_end=0.002)

            obj_sa = compute_obj(a_sa)
            if obj_sa < best_score:
                best_score = obj_sa
                best_seq = a_sa.copy()
                print(f"  SA restart {sa_restart}: {best_score:.8f} **")
            else:
                print(f"  SA restart {sa_restart}: {obj_sa:.8f}")

    # =========================================================
    # Phase 5: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700
    n_options = sorted(set([
        n_best - 200, n_best - 100, n_best - 50, n_best - 25,
        n_best, n_best + 25, n_best + 50, n_best + 100,
        n_best + 150, n_best + 200, n_best + 300
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        a_try = normalize(resample_sequence(best_seq, n_try))
        alloc = min(remaining * 0.08, 25.0)

        # Quick optimization
        a_try = kkt_power_iteration(a_try, max_iters=30, time_budget=alloc * 0.5, top_k=5)
        a_try = spectral_polish(a_try, max_iters=20, time_budget=alloc * 0.3)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 6: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        n_final = len(a_final)

        for round_idx in range(6):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            alloc = min(remaining * 0.18, 35.0)

            # KKT
            if alloc > 10:
                a_pol = kkt_power_iteration(a_final, max_iters=50, time_budget=alloc * 0.35, top_k=5)
            else:
                a_pol = a_final.copy()

            # LP (if n is manageable)
            if n_final <= 900 and alloc > 15:
                a_pol = lp_alternate_v2(a_pol, max_iters=20, time_budget=alloc * 0.35)

            # Coordinate descent
            if alloc > 10:
                a_pol = coordinate_polish(a_pol, time_budget=min(10, alloc * 0.2))

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {round_idx}: {best_score:.8f} **")
            else:
                a_final = a_pol
                print(f"  Polish {round_idx}: {obj_pol:.8f}")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
