"""
Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FISTA with adaptive log-sum-exp smoothing, LP alternation, spectral methods,
and systematic n exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

# --- Utility functions ---

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat - 1] - 1) / rho_hat
    return np.maximum(v - theta, 0)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

# --- LP alternation ---

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except:
        pass
    return None

def lp_direction(a, M, time_limit=20):
    """Find LP direction. Returns normalized g or None."""
    return solve_lp(a, M, time_limit)

def lp_alternate(a, max_iters=50, time_budget=100.0):
    """LP alternation with Armijo backtracking."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break

        # Armijo backtracking for step size
        obj_cur = compute_obj(a)
        # Gradient approximation: direction of improvement
        # Use the fact that moving toward g should decrease obj
        t = 1.0
        a_new = None
        for _ in range(10):
            a_try = normalize((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < obj_cur:
                a_new = a_try
                break
            t *= 0.5

        if a_new is not None and compute_obj(a_new) < best_obj:
            best_obj = compute_obj(a_new)
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step or break
            if it > 5:
                break
            a = normalize(0.95 * a + 0.05 * g)

    return best_a

# --- FISTA with log-sum-exp ---

def compute_smoothed_grad(a, mu, size):
    """Compute gradient of log-sum-exp smoothing of max(conv)."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = mu * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # Gradient = 2 * cross-correlation of weights and a
    a_flip = a[::-1]
    Wf = rfft(weights, size)
    Af = rfft(a_flip, size)
    corr = irfft(Wf * Af, size)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad

def fista_smooth(a, max_iter=500, time_budget=100.0, mu_start=None, mu_end=None):
    """FISTA with log-sum-exp smoothing."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    size = 1
    while size < 3 * n - 1:
        size <<= 1

    if mu_start is None:
        mu_start = n / 20.0
    if mu_end is None:
        mu_end = n / 2.0

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    x_prev = a.copy()
    t = 1.0
    L = 4.0  # Lipschitz constant estimate (conservative)

    for k in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive mu
        frac = (k + 1) / max_iter
        mu = mu_start + (mu_end - mu_start) * frac

        # Compute gradient at y
        grad = compute_smoothed_grad(y, mu, size)

        # Adaptive L: check if step is too aggressive
        x_new = project_simplex(y - grad / L)

        # Check if L needs to be increased (function value increased)
        # (simplified: just keep L fixed or slowly increase)
        
        x_prev = x
        x = x_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + (t - 1.0) / t_new * (x - x_prev)
        t = t_new

        # Periodically evaluate true objective
        if k % 20 == 0:
            obj = compute_obj(x)
            if obj < best_obj:
                best_obj = obj
                best_a = x.copy()

    # Final evaluation
    obj = compute_obj(x)
    if obj < best_obj:
        best_obj = obj
        best_a = x.copy()

    return best_a

def spectral_flatness_opt(a, max_iters=100, time_budget=60.0):
    """Iteratively flatten the power spectrum while maintaining a >= 0."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    size = 1
    while size < 2 * n:
        size <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compress dynamic range
        mag_min = np.min(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1e-10
        mag_max = np.max(mag)
        
        if mag_max < 1e-10:
            break

        # Soft compression: mag_new = mag^alpha where alpha < 1 compresses
        alpha = 0.7 + 0.2 * (it / max_iters)  # Start at 0.7, go to 0.9
        mag_norm = mag / mag_max
        mag_new = mag_norm ** alpha * mag_max

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            a = normalize(0.7 * a + 0.3 * a_new)

    return best_a

def coord_descent(a, time_budget=30.0):
    """Targeted coordinate descent on the convolution peak."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors
        contribs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)

        improved = False
        for val, j1, j2 in contribs[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            # Try reducing a[j1] and increasing a at a "good" location
            # "Good" = far from the peak
            for _ in range(5):
                targets = list(range(n))
                # Prefer targets far from j1 and j2
                dist = [abs(t - j1) + abs(t - j2) for t in targets]
                dist[j1] = 0
                dist[j2] = 0
                # Weighted random selection
                w = np.array(dist, dtype=float)
                w = w ** 2 + 1
                w /= w.sum()
                j_target = int(rng.choice(n, p=w))

                if j_target == j1 or j_target == j2:
                    continue

                a_try = a.copy()
                for frac in [0.02, 0.05, 0.1, 0.15]:
                    delta = frac * a_try[j1]
                    a_try2 = a_try.copy()
                    a_try2[j1] -= delta
                    a_try2[j_target] += delta
                    a_try2 = normalize(a_try2)
                    obj_try = compute_obj(a_try2)
                    if obj_try < best_obj * (1 - 1e-9):
                        best_obj = obj_try
                        best_a = a_try2.copy()
                        a = a_try2
                        improved = True
                        break

        if not improved:
            # Random perturbation to escape
            if rng.random() < 0.3:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FISTA with adaptive log-sum-exp smoothing, subgradient FISTA, LP alternation,
spectral flattening, systematic n exploration, and coordinate descent polishing.
Key improvements: Nesterov momentum, adaptive smoothing temperature,
reduced-constraint LP, and better time allocation.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = int(np.max(rho[cond]))
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))

    """Build Toeplitz matrix A where (A@g)[k] = sum_j a[k-j]*g[j]."""
    n = len(a)

    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        A[i, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status in (0, 3):
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                full_conv = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    j_min = max(0, k - n + 1)
                    j_max = min(n - 1, k)
                    full_conv[k] = np.dot(a[k - j_max:k - j_min + 1][::-1], g[j_min:j_max + 1])
                if np.max(full_conv) <= M * 1.005:
                    return g
    except Exception:
        pass
    return None


def lp_alternate(a, max_iters=40, time_budget=100.0):
    """LP alternation with backtracking step size."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_full(a, M, time_limit=max(3, int(remaining * 0.4)))
        if g is None:
            g = solve_lp_reduced(a, M, K=max(50, n // 3), time_limit=max(3, int(remaining * 0.3)))
        if g is None:
            break

        # Backtracking step size
        obj_cur = compute_obj(a)
        a_new = None
        for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.08]:
            a_try = normalize((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < obj_cur * (1.0 - 1e-10):
                a_new = a_try
                break

        if a_new is not None and compute_obj(a_new) < best_obj:
            best_obj = compute_obj(a_new)
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 6:
                # Perturb to escape
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                a = normalize(best_a * np.exp(rng.normal(0, 0.03 + 0.02 * stall_count, n)))
                stall_count = 0
            else:
                a = normalize(0.9 * a + 0.1 * g)

    return best_a


def fista_smoothed(a, max_iter=800, time_budget=80.0, mu_f_start=0.05, mu_f_end=0.5):
    """FISTA with log-sum-exp smoothing. mu_f is the fraction of n used for mu."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    size = 1
    while size < 3 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    x_prev = a.copy()
    t = 1.0
    L = 5.0

    for k in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        frac = (k + 1) / max_iter
        mu = n * (mu_f_start + (mu_f_end - mu_f_start) * frac)

        # Compute gradient of smoothed objective at y
        conv_y = compute_conv(y)
        cmax = float(np.max(conv_y))
        shifted = mu * (conv_y - cmax)
        shifted = np.clip(shifted, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv_y))
            weights = np.zeros(len(conv_y))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient via cross-correlation
        a_flip = y[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Adaptive L: increase if step is too large
        x_new = project_simplex(y - grad / L)
        # Simple check: if objective increased significantly, increase L
        if k % 50 == 0 and k > 0:
            obj_check = compute_obj(x_new)
            obj_y = compute_obj(y)
            if obj_check > obj_y * 1.05:
                L *= 1.5
                x_new = project_simplex(y - grad / L)

        x_prev = x
        x = x_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + (t - 1.0) / t_new * (x - x_prev)
        t = t_new

        if k % 40 == 0:
            obj = compute_obj(x)
            if obj < best_obj:
                best_obj = obj
                best_a = x.copy()

    obj = compute_obj(x)
    if obj < best_obj:
        best_obj = obj
        best_a = x.copy()

    return best_a


def fista_subgradient(a, max_iter=600, time_budget=60.0):
    """FISTA with exact subgradient of max(conv)."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    x_prev = a.copy()
    t = 1.0
    L = 4.0

    for k in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        # Subgradient at y
        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))
        grad = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        grad[j_min:j_max + 1] = 2.0 * y[k_star - j_max:k_star - j_min + 1][::-1]

        x_new = project_simplex(y - grad / L)
        x_prev = x
        x = x_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + (t - 1.0) / t_new * (x - x_prev)
        t = t_new

        if k % 50 == 0:
            obj = compute_obj(x)
            if obj < best_obj:
                best_obj = obj
                best_a = x.copy()

    obj = compute_obj(x)
    if obj < best_obj:
        best_obj = obj
        best_a = x.copy()

    return best_a


def spectral_flatness_opt(a, max_iters=150, time_budget=50.0):
    """Iteratively flatten power spectrum while maintaining a >= 0."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    size = 1
    while size < 2 * n:
        size <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)

        mag_max = float(np.max(mag))
        if mag_max < 1e-10:
            break

        # Adaptive compression: start aggressive, ease off
        alpha = 0.6 + 0.3 * (it / float(max_iters))
        mag_norm = mag / mag_max
        mag_new = mag_norm ** alpha * mag_max

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.5 + 0.3 * (it / float(max_iters))
            a = normalize(blend * a + (1.0 - blend) * a_new)

    return best_a


def coordinate_descent(a, time_budget=20.0):
    """Targeted mass-transfer coordinate descent."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        contribs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)

        improved = False
        for val, j1, j2 in contribs[:6]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            for _ in range(4):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue
                a_try = a.copy()
                delta = 0.04 * a_try[j1]
                a_try[j1] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj * (1.0 - 1e-10):
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and rng.random() < 0.2:
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search: LP + FISTA(smoothed + subgradient) + spectral + n exploration + polish.
    Optimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 35.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            best_score = compute_obj(parent)
            best_seq = parent.copy()
            print(f"Parent loaded: n={parent.size}, obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(700)
        best_score = compute_obj(best_seq)
        print(f"Uniform init: n=700, obj={best_score:.8f}")

    n_best = len(best_seq)

    # ============ Phase 1: LP + FISTA at parent n ============
    print(f"\n{'='*50}")
    print(f"Phase 1: Core optimization (n={n_best})")
    print(f"{'='*50}")

    a = normalize(best_seq.copy())

    # LP first round
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.18, 90.0)
        a = lp_alternate(a, max_iters=35, time_budget=alloc)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  LP round 1: {best_score:.8f}")
        else:
            print(f"  LP round 1: {obj:.8f}")

    # FISTA smoothed
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.18, 80.0)
        a_fista = fista_smoothed(a, max_iter=1000, time_budget=alloc,
                                  mu_f_start=0.05, mu_f_end=0.5)
        obj = compute_obj(a_fista)
        if obj < best_score:
            best_score = obj
            best_seq = a_fista.copy()
            a = a_fista
            print(f"  FISTA smoothed: {best_score:.8f}")
        else:
            print(f"  FISTA smoothed: {obj:.8f}")

    # LP second round (from FISTA result)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 70.0)
        a = lp_alternate(normalize(best_seq.copy()), max_iters=25, time_budget=alloc)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  LP round 2: {best_score:.8f}")
        else:
            print(f"  LP round 2: {obj:.8f}")

    # FISTA subgradient
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.12, 50.0)
        a_sg = fista_subgradient(normalize(best_seq.copy()), max_iter=500, time_budget=alloc)
        obj = compute_obj(a_sg)
        if obj < best_score:
            best_score = obj
            best_seq = a_sg.copy()
            print(f"  FISTA subgradient: {best_score:.8f}")
        else:
            print(f"  FISTA subgradient: {obj:.8f}")

    # Spectral
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 50.0)
        a_spec = spectral_flatness_opt(normalize(best_seq.copy()), max_iters=200, time_budget=alloc)
        obj = compute_obj(a_spec)
        if obj < best_score:
            best_score = obj
            best_seq = a_spec.copy()
            print(f"  Spectral: {best_score:.8f}")
        else:
            print(f"  Spectral: {obj:.8f}")

    # ============ Phase 2: n exploration ============
    print(f"\n{'='*50}")
    print(f"Phase 2: n exploration (current best n={len(best_seq)})")
    print(f"{'='*50}")

    n_cur = len(best_seq)
    n_candidates = sorted(set([
        n_cur - 150, n_cur - 100, n_cur - 75, n_cur - 50, n_cur - 25,
        n_cur,
        n_cur + 25, n_cur + 50, n_cur + 75, n_cur + 100, n_cur + 150, n_cur + 200
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 1200]

    # Prioritize n values closest to current best
    n_candidates.sort(key=lambda x: abs(x - n_cur))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        if n_try == len(best_seq):
            continue
        alloc = min(remaining * 0.08, 35.0)

        a_try = normalize(resample_sequence(best_seq, n_try))

        # Quick optimization pipeline
        if n_try <= 900:
            a_try = lp_alternate(a_try, max_iters=10, time_budget=alloc * 0.35)
        a_try = fista_smoothed(a_try, max_iter=300, time_budget=alloc * 0.35,
                                mu_f_start=0.05, mu_f_end=0.4)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # ============ Phase 3: Intensive polishing at best n ============
    print(f"\n{'='*50}")
    print(f"Phase 3: Intensive polish (n={len(best_seq)})")
    print(f"{'='*50}")

    n_final = len(best_seq)
    a = normalize(best_seq.copy())

    for polish in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.12, 50.0)

        # LP
        if n_final <= 950 and alloc > 20:
            a = lp_alternate(a, max_iters=20, time_budget=alloc * 0.35)
        # FISTA smoothed
        a = fista_smoothed(a, max_iter=300, time_budget=alloc * 0.3,
                           mu_f_start=0.03, mu_f_end=0.4)
        # FISTA subgradient
        a = fista_subgradient(a, max_iter=200, time_budget=alloc * 0.2)
        # Coordinate descent
        a = coordinate_descent(a, time_budget=min(8.0, alloc * 0.1))

        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Polish {polish}: {best_score:.8f} **")
        else:
            # Keep best, try small perturbation
            a = normalize(best_seq.copy())
            print(f"  Polish {polish}: {obj:.8f} (stuck)")

    # ============ Phase 4: Random restarts ============
    print(f"\n{'='*50}")
    print(f"Phase 4: Random restarts")
    print(f"{'='*50}")

    n_cur = len(best_seq)
    rng = np.random.default_rng(seed)

    for restart in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.08, 35.0)

        r = restart % 5
        if r == 0:
            a_try = best_seq * np.exp(rng.normal(0, 0.06, n_cur))
        elif r == 1:
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.12 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
            a_try = best_seq * mod
        elif r == 2:
            a_rand = rng.dirichlet(np.ones(n_cur) * 0.4)
            a_try = 0.75 * normalize(best_seq) + 0.25 * a_rand
        elif r == 3:
            # Block reversal
            a_try = best_seq.copy()
            blen = int(rng.integers(n_cur // 5, n_cur // 3))
            bstart = int(rng.integers(0, n_cur - blen))
            a_try[bstart:bstart + blen] = a_try[bstart:bstart + blen][::-1]
        else:
            # Dual-frequency modulation
            f1, f2 = sorted(rng.uniform(0.5, 6.0, 2))
            mod = 1.0 + 0.08 * np.sin(2 * np.pi * f1 * np.arange(n_cur) / n_cur) \
                          + 0.05 * np.cos(2 * np.pi * f2 * np.arange(n_cur) / n_cur)
            a_try = best_seq * mod

        a_try = normalize(a_try)

        if n_cur <= 900:
            a_try = lp_alternate(a_try, max_iters=10, time_budget=alloc * 0.3)
        a_try = fista_smoothed(a_try, max_iter=200, time_budget=alloc * 0.35,
                                mu_f_start=0.05, mu_f_end=0.35)

        obj = compute_obj(a_try)
        if obj < best_score:
            best_score = obj
            best_seq = a_try.copy()
            print(f"  Restart {restart}: {best_score:.8f} **")
        else:
            print(f"  Restart {restart}: {obj:.8f}")

    # ============ Final validation ============
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, eval={val:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
