def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def gs_initialize(n, n_iters=200, rng=None):
    """Gerchberg-Saxton initialization for flat autocorrelation."""
    if rng is None:
        rng = np.random.default_rng(42)
    a = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    a /= a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Target: constant magnitude (flat spectrum)
        mag_target = np.ones_like(mag) * (mag.mean() + 1e-15)
        # Blend: 50% target, 50% current (relaxed GS)
        if it < n_iters // 2:
            blend = 0.7
        else:
            blend = 0.5
        mag_new = (1 - blend) * mag + blend * mag_target
        F_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        M_c = fast_conv(a_new).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_new.copy()
        a = a_new
    return best_a


def build_lp_matrix(n, a):
    """Build sparse matrix for LP: (a*g)(k) = sum_j A[k,j]*g[j] <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_best_response(a, M, A_cache=None):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0."""
    n = len(a)
    if A_cache is None:
        A = build_lp_matrix(n, a)
    else:
        A = A_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-8,
                             'time_limit': 5})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(history, window=5):
    """Anderson acceleration: find best extrapolation from history."""
    m = min(window, len(history))
    if m < 2:
        return history[-1].copy()
    xs = [history[-i] for i in range(m, 0, -1)]
    # Residuals
    residuals = np.array([xs[i] - xs[i+1] if i < m-1 else xs[i] - xs[0] for i in range(m-1)])
    # Simpler: use standard Anderson
    points = [history[-i] for i in range(m, 0, -1)]
    diffs = [points[i] - points[i+1] for i in range(m-1)]
    G = np.array([np.dot(diffs[i], diffs[j]) for i in range(m-1) for j in range(m-1)]).reshape(m-1, m-1)
    G += 1e-12 * np.eye(m-1)
    ones = np.ones(m-1)
    try:
        alpha = np.linalg.solve(G, ones)
        s = alpha.sum()
        if abs(s) < 1e-15:
            alpha = ones / (m-1)
        else:
            alpha /= s
        alpha = np.clip(alpha, -0.5, 1.0)
        alpha /= alpha.sum()
        mixed = np.zeros_like(points[0])
        for i in range(m-1):
            mixed += alpha[i] * points[i]
        mixed += (1 - alpha.sum()) * points[m-1] if alpha.sum() != 1 else 0
        # Just use weighted average of last few
        w = np.ones(m)
        w *= np.linspace(0.5, 1.0, m)
        w /= w.sum()
        mixed = sum(w[i] * points[i] for i in range(m))
    except np.linalg.LinAlgError:
        mixed = points[0].copy()
    return np.maximum(mixed, 0)


def run_smooth_optimization(a, deadline, max_iters=5000, tau_start=3.0, tau_end=300.0,
                            lr_init=0.5, momentum_start=0.7, momentum_end=0.95):
    """Smooth max (log-sum-exp) gradient descent with Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        momentum = momentum_start + (momentum_end - momentum_start) * frac
        
        grad, p, b, b_max = smooth_max_gradient(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500 and it > 100:
                break
        
        # Backtracking line search
        lr = lr_init * (0.8 ** min(it // 50, 10))
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        direction = grad + momentum * (y - a)
        
        for ls in range(15):
            step = lr / (1.0 + ls * 0.5)
            a_cand = y - step * direction
            a_cand = np.maximum(a_cand, 0)
            s = a_cand.sum()
            if s < 1e-10:
                break
            a_cand /= s
            M_cand = fast_conv(a_cand).max()
            if M_cand < b_max + step * np.linalg.norm(direction) * tau * 1.1:
                break
        
        a_new = a_cand
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-10:
            y /= sy
        else:
            y = a_new.copy()
        a = a_new
    return best_a, best_M


def run_lp_refinement(a, deadline, A_cache=None, max_iters=200, t_start=0.3, t_min=0.01):
    """LP mirror descent with Anderson acceleration for final polish."""
    a = np.maximum(a.astype(np.float64), 1e-8)

        b = fast_conv(a)

            best_a = a.copy()
            no_imp = 0
        else:

            if no_imp > 40:

        g = solve_lp_best_response(a, M, A)
        if g is None:
            break

        if gs < 1e-10:
            break
        g /= gs
        
        # Adaptive step size
        if it < 10:

        if M_new > M * 1.01:
            # Bad step, reduce t

            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:

def resample_to_length(a, n_new):
    """Resample sequence to new length using interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    t_old = np.linspace(0, 1, n_old)
    t_new = np.linspace(0, 1, n_new)
    return np.maximum(np.interp(t_new, t_old, a), 0.0)


def make_diverse_inits(n, rng, parent=None):
    """Create diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_to_length(parent, n)
        p = np.maximum(p, 1e-6)
        p /= p.sum()
        inits.append(p.copy())
        # Perturbed versions
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.001))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.001))
        # Frequency perturbation
        F = np.fft.rfft(p, 2*n)
        F_new = F * (1 + 0.05 * rng.normal(0, 1, len(F)))
        p_f = np.fft.irfft(F_new, 2*n)[:n]
        inits.append(np.maximum(p_f, 0.001))
    
    # Parametric families
    inits.append(np.ones(n).astype(float))
    for alpha in [0.3, 0.5, 0.8, 1.2, 2.0]:
        inits.append(np.maximum(0.3 + 0.7 * (t**alpha + (1-t)**alpha) / 2, 0.01))
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.6 + 0.4 * np.cos(2*np.pi*f*t), 0.01))
    # Gaussian bumps
    for center, width in [(0.5, 0.3), (0.3, 0.2), (0.7, 0.2), (0.5, 0.15)]:
        inits.append(np.maximum(np.exp(-((t-center)/width)**2) + 0.05, 0.01))
    # Double bump
    inits.append(np.maximum(np.exp(-9*(t-0.3)**2) + np.exp(-9*(t-0.7)**2) + 0.03, 0.01))
    # Random smooth
    k = max(3, n // 16)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0.1, 1.0, n), kern, 'same') + 0.02, 0.01))
    # Symmetric random
    half = np.maximum(rng.uniform(0.5, 1.0, n // 2 + 1), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    
    # Normalize all
    inits = [np.maximum(x, 1e-6) / np.maximum(np.sum(np.maximum(x, 1e-6)), 1e-10) for x in inits]
    return inits

        else:
            print(f"  n={n:4d}: no result ({time.monotonic()-t0:.1f}s)", flush=True)
    
    best.report("Phase 1")
    print(f"Phase 1 total: {time.monotonic()-p1_start:.1f}s", flush=True)
    
    # === PHASE 2: Main optimization on best length ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Extended Smooth Optimization (budget: 500s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best.seq is not None:
        n_best = best.seq.size
        p2_budget = min(500.0, deadline - 300 - time.monotonic())
        if p2_budget > 60:
            dl2 = time.monotonic() + p2_budget
            
            # Multiple restarts from best
            n_restarts = 5
            for r in range(n_restarts):
                if time.monotonic() > dl2:
                    break
                if r == 0:
                    a_start = best.seq.copy()
                else:
                    noise_scale = 0.05 * (r + 1) / n_restarts
                    a_start = best.seq + rng.normal(0, noise_scale, n_best)
                    a_start = np.maximum(a_start, 1e-6)
                    a_start /= a_start.sum()
                
                sub_dl = time.monotonic() + (dl2 - time.monotonic()) / (n_restarts - r + 1) * 1.2
                a_opt, M_opt = run_smooth_optimization(a_start, sub_dl, max_iters=8000,
                                                        tau_start=10.0, tau_end=500.0,
                                                        lr_init=0.3, momentum_start=0.8,
                                                        momentum_end=0.97)
                best.update(a_opt)
            
            # Also try from phase 1 results for other lengths
            for n_try, a_try in sorted(best_per_length.items()):
                if time.monotonic() > dl2 - 30:
                    break
                if n_try == n_best:
                    continue
                a_up = resample_to_length(a_try, n_best)
                a_up = np.maximum(a_up, 1e-6)
                a_up /= a_up.sum()
                sub_dl = time.monotonic() + 40
                a_opt, M_opt = run_smooth_optimization(a_up, sub_dl, max_iters=2000,
                                                        tau_start=20.0, tau_end=200.0,
                                                        lr_init=0.2, momentum_start=0.85,
                                                        momentum_end=0.95)
                best.update(a_opt)
    
    best.report("Phase 2")
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)
    
    # === PHASE 3: LP Refinement ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Best-Response Refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best.seq is not None and time.monotonic() < deadline - 60:
        n = best.seq.size
        p3_budget = min(400.0, deadline - 60 - time.monotonic())
        
        if p3_budget > 30:
            # Build LP matrix once
            print(f"  Building LP matrix for n={n}...", flush=True)
            A = build_lp_matrix(n, best.seq)
            print(f"  LP matrix built. nnz={A.nnz}", flush=True)
            
            # Multiple LP passes with decreasing step sizes
            for pass_i in range(3):
                if time.monotonic() > deadline - 30:
                    break
                sub_dl = time.monotonic() + p3_budget / (3 - pass_i)
                max_it = max(30, int((sub_dl - time.monotonic())))
                print(f"  LP pass {pass_i+1}: budget={(sub_dl-time.monotonic()):.0f}s", flush=True)
                
                a_lp = run_lp_refinement(best.seq, sub_dl, A_cache=A,
                                         max_iters=max_it,
                                         t_start=0.35 / (pass_i + 1),
                                         t_min=0.005)
                best.update(a_lp)
        
        # Try LP on second-best length too
        if len(best_per_length) > 1 and time.monotonic() < deadline - 40:
            sorted_lens = sorted(best_per_length.keys(), key=lambda x: compute_J(best_per_length[x]))
            for n_try in sorted_lens[1:2]:
                if time.monotonic() > deadline - 30:
                    break
                a_cand = best_per_length[n_try]
                A_cand = build_lp_matrix(n_try, a_cand)
                a_lp = run_lp_refinement(a_cand, deadline - 25, A_cache=A_cand,
                                         max_iters=50, t_start=0.3, t_min=0.01)
                best.update(a_lp)
    
    best.report("Phase 3")
    print(f"Phase 3 total: {time.monotonic()-p3_start:.1f}s", flush=True)
    
    # === PHASE 4: Length exploration ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Exploration (budget: 150s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best.seq is not None and time.monotonic() < deadline - 30:
        n_best = best.seq.size
        # Try nearby lengths
        deltas = [-64, -32, -16, 16, 32, 64, 128, -128]
        p4_budget = min(150.0, deadline - 30 - time.monotonic())
        
        for delta in deltas:
            if time.monotonic() > deadline - 20:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 2048:
                continue
            allot = p4_budget / len(deltas) + 10
            sub_dl = time.monotonic() + min(allot, 35)
            
            a_ext = resample_to_length(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            
            a_opt, M_opt = run_smooth_optimization(a_ext, sub_dl, max_iters=2000,
                                                    tau_start=10.0, tau_end=150.0,
                                                    lr_init=0.25, momentum_start=0.8,
                                                    momentum_end=0.95)
            best.update(a_opt)
            
            # Quick LP if time permits
            if time.monotonic() < deadline - 15 and best.seq.size == n_try:
                A_try = build_lp_matrix(n_try, best.seq)
                a_lp = run_lp_refinement(best.seq, deadline - 15, A_cache=A_try,
                                         max_iters=20, t_start=0.25, t_min=0.01)
                best.update(a_lp)
    
    best.report("Phase 4")
    print(f"Phase 4 total: {time.monotonic()-p4_start:.1f}s", flush=True)
    
    # === PHASE 5: Final polish ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best.seq is not None and time.monotonic() < deadline - 5:
        # One more round of smooth optimization from best
        n_final = best.seq.size
        sub_dl = deadline - 3
        a_final, M_final = run_smooth_optimization(best.seq, sub_dl, max_iters=3000,
                                                    tau_start=50.0, tau_end=800.0,
                                                    lr_init=0.1, momentum_start=0.9,
                                                    momentum_end=0.98)
        best.update(a_final)
        
        # Final LP if time
        if time.monotonic() < deadline - 2:
            A_final = build_lp_matrix(n_final, best.seq)
            a_lp = run_lp_refinement(best.seq, deadline - 1, A_cache=A_final,
                                     max_iters=15, t_start=0.15, t_min=0.003)
            best.update(a_lp)
    
    best.report("Phase 5")
    print(f"Phase 5 total: {time.monotonic()-p5_start:.1f}s", flush=True)
    
    # Fallback
    if best.seq is None:
        best.seq = np.ones(256)
        best.score = compute_J(best.seq)
    
    total_time = time.monotonic() - best.t0
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={best.score:.8f} n={best.seq.size} time={total_time:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best.seq]
