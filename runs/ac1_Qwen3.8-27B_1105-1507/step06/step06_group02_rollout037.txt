    # grad_a_i = 2n * [sum_{k} p_k * a_{k-i} + sum_k p_k * a_{i-k}]
    # Note: sum_k p_k * a_{i-k} = sum_k p_k * a_{k-i} (by substitution j = i-k... no)
    # Let j = i-k, then k = i-j, sum_j p_{i-j} * a_j... 
    # Hmm, these are the same! sum_k p_k * a_{i-k} = sum_j p_{i-j} * a_j = (p * a reversed)... 
    
    # Actually: sum_k p_k * a_{k-i} and sum_k p_k * a_{i-k}
    # Let's substitute: in the second, let k' = i-k, then k = i-k', 
    # sum_{k'} p_{i-k'} * a_{k'} = sum_{k'} a_{k'} * p_{i-k'}
    # And the first: sum_k p_k * a_{k-i}, let j = k-i, k = j+i:
    # sum_j p_{j+i} * a_j
    
    # So grad_a_i = 2n * [sum_j a_j * p_{j+i} + sum_j a_j * p_{i-j}]
    # = 2n * [cross_corr(a, p)_i + cross_corr(a, reverse... )]
    
    # Let me just compute it directly for correctness:
    # For each i, grad_a[i] = 2n * sum_k p[k] * (a[k-i] if 0<=k-i<n else 0 + a[i-k] if 0<=i-k<n else 0)
    
    # Using FFT:
    # term1[i] = sum_k p[k] * a[k-i] (zero-padded)
    # This is the cross-correlation of p and a: (p ⋆ a)[i]
    # = ifft(fft(p, N) * conj(fft(a, N)))[i] ... let me verify
    # 
    # Ifft of FFT(p) * conj(FFT(a)):
    # = sum_k p[k] * a[n-k] (circular)... this is getting confusing.
    #
    # Let me just use the fact that:
    # np.correlate(p, a, mode='full') gives sum_k p[k+m] * a[k] for m from -(n-1) to n-1
    # 
    # I'll just use a simple approach: compute the gradient via the bilinear structure.
    
    # Actually, the cleanest way:
    # conv(a,a)_k = sum_j a[j] * a[k-j]
    # d/d a[i] of conv(a,a)_k = a[k-i] (if 0 <= k-i < n) + a[i-k] (if 0 <= i-k < n)
    # 
    # So grad_a[i] = 2n * sum_k p[k] * (a[k-i] * 1[0<=k-i<n] + a[i-k] * 1[0<=i-k<n])
    #
    # First sum: S1[i] = sum_k p[k] * a[k-i] * 1[0<=k-i<n]
    # = sum_{j=0}^{n-1} a[j] * p[j+i] * 1[0<=j+i<2n-1]
    # = cross_correlation of a and p, evaluated at position i
    #
    # Second sum: S2[i] = sum_k p[k] * a[i-k] * 1[0<=i-k<n]
    # = sum_{j=0}^{n-1} a[j] * p[i-j] * 1[0<=i-j<2n-1]
    # = convolution of a and p, evaluated at position i
    #
    # So grad_a = 2n * (cross_corr(a, p) + conv(a, p))[:n]
    # Wait, that's not quite right either. Let me use FFT directly.
    
    # For S1[i] = sum_j a[j] * p[j+i]:
    # This is the cross-correlation: np.correlate(p, a, 'full')[n-1+i]... 
    # Hmm, let me just use FFT:
    # S1 = ifft(fft(p, N) * conj(fft(a, N)))[:n]  -- this gives sum_k p[k]*a[(k-i) mod N]
    # With zero padding to N >= 2n-1, this gives the linear cross-correlation.
    
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    S1 = np.fft.irfft(fp * np.conj(fa), size)[:n]
    # Wait, this gives sum_k p[k] * a[(k-i) mod N], which with zero padding is sum_k p[k] * a[k-i] (valid range)
    # Hmm, actually IFFT(FFT(p) * conj(FFT(a))) gives the cross-correlation:
    # output[i] = sum_k p[k] * a[k-i]  (this is what we want for S1)
    
    # For S2[i] = sum_j a[j] * p[i-j]:
    # This is the convolution: (a * p)[i] = sum_j a[j] * p[i-j]
    # = IFFT(FFT(a) * FFT(p))
    S2 = np.fft.irfft(fa * fp, size)[:n]
    
    grad_a = 2.0 * n * (S1 + S2)
    
    # Now convert to gradient wrt x (softmax parameterization):
    # grad_x = a * (grad_a - dot(a, grad_a))
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x


def run_lbfgs(a_init, deadline, tau_start=10.0, tau_end=500.0, n_phases=5):
    """Run L-BFGS-B with continuation on tau."""
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / a_init.sum()
    n = len(a_init)
    
    # Start from x = log(a_init)
    x = np.log(a_init)
    
    best_a = a_init.copy()
    best_J = compute_J(a_init)
    
    # Continuation: increase tau over phases
    tau_schedule = np.linspace(tau_start, tau_end, n_phases)
    
    for phase_idx in range(n_phases):
        if time.monotonic() > deadline:
            break
        tau = tau_schedule[phase_idx]
        
        # Define objective and gradient for scipy
        x_current = x.copy()
        
        def func_grad(x_val):
            if time.monotonic() > deadline:
                return 0.0, np.zeros_like(x_val)
            return smooth_max_obj_and_grad(x_val, tau)
        
        try:
            result = minimize(func_grad, x_current, method='L-BFGS-B',
                             jac=True, options={'maxiter': 100, 'gtol': 1e-8,
                                                'maxls': 20})
            if result.success or result.fun < 0:
                x = result.x
        except Exception:
            pass
        
        # Update best
        x_shifted = x - x.max()
        a = np.exp(x_shifted) / np.exp(x_shifted).sum()
        a = np.maximum(a, 0)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
        
        print(f"    L-BFGS phase {phase_idx+1}/{n_phases} tau={tau:.0f}: J={best_J:.8f}", flush=True)
    
    return best_a


def run_adam_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=300.0,
                    lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on the softmax parameterization."""
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    n = len(a)
    x = np.log(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        
        obj, grad = smooth_max_obj_and_grad(x, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodically check J
        if (it + 1) % 50 == 0 or it == max_iters - 1:
            x_shifted = x - x.max()
            a_new = np.exp(x_shifted) / np.exp(x_shifted).sum()
            J = compute_J(a_new)
            if J < best_J:
                best_J = J
                best_a = a_new.copy()
    
    # Final check
    x_shifted = x - x.max()
    a_new = np.exp(x_shifted) / np.exp(x_shifted).sum()
    J = compute_J(a_new)
    if J < best_J:
        best_a = a_new.copy()
    
    return best_a


def build_A_values(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    n = len(a)
    rows, cols, vals = build_A_values(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_with_backtracking(a, deadline, max_iters=100, t_base=0.3):
    """LP best-response with backtracking line search."""
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break
        
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Backtracking line search
        improved = False
        t = t_base
        for _ in range(10):
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
            t *= 0.5
            if t < 1e-4:
                break
        
        if not improved:
            # Try direct replacement
            M_g = fast_conv(g).max()
            if M_g < best_M:
                a = g.copy()
                best_M = M_g
                best_a = a.copy()
            else:
                no_imp += 1
                if no_imp > 25:
                    break
    
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.005))
        # Asymmetric perturbation
        asym = np.linspace(-0.1, 0.1, n)
        inits.append(np.maximum(p + 0.1 * asym, 0.005))
    
    # Asymmetric inits
    inits.append(np.maximum(0.1 + 0.9 * t**2, 0.01))
    inits.append(np.maximum(0.1 + 0.9 * (1-t)**2, 0.01))
    inits.append(np.maximum(0.1 + 0.5 * t + 0.3 * t**3, 0.01))
    inits.append(np.exp(-2 * t) / np.exp(-2 * t).sum())
    inits.append(np.exp(-2 * (1-t)) / np.exp(-2 * (1-t)).sum())
    
    # Random smooth
    k = max(3, n // 10)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    
    # Bump functions
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        inits.append(np.exp(-20 * (t - center)**2) + 0.02)
    
    # Sine-modulated
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.4 * np.sin(2 * np.pi * f * t + 0.5), 0.01))
    
    # Power law with asymmetry
    for alpha in [0.3, 0.5, 1.0, 2.0, 3.0]:
        inits.append(np.maximum(t**alpha + (1-t)**alpha * 0.5 + 0.05, 0.01))
    
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with smooth max on softmax parameterization (O(n log n) gradients)
2. L-BFGS-B quasi-Newton method with continuation on tau for precise local optimization
3. LP best-response with backtracking line search
4. Random perturbation restarts to escape local minima
5. Length exploration around best found
Key improvements: L-BFGS-B for fast convergence, FFT-based gradient in softmax
parameterization (unconstrained optimization), continuation method on tau,
and asymmetric initialization that exploits the non-symmetric optimum.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_obj_and_grad(x, tau):
    """Compute smooth max objective and gradient in softmax parameterization.
    a = softmax(x), objective = 2n * smooth_max(conv(a,a)).
    Returns (objective, gradient_wrt_x).
    """
    # Compute a = softmax(x) with log-sum-exp trick
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    n = len(a)

    # Compute b = conv(a, a)
    b = fast_conv(a)

    # Smooth max: (1/tau) * log(sum(exp(tau * b)))
    tau_b = tau * b
    tau_b_max = tau_b.max()
    exp_tau_b = np.exp(tau_b - tau_b_max)
    log_sum_exp = tau_b_max + np.log(exp_tau_b.sum())
    obj = 2.0 * n * log_sum_exp / tau

    # Softmax weights p_k
    p = exp_tau_b / exp_tau_b.sum()

    # Compute gradient wrt a using FFT
    # grad_a_i = 2n * sum_k p[k] * (a[k-i] + a[i-k])
    # S1[i] = sum_k p[k] * a[k-i] = cross_correlation(p, a)[i]
    # S2[i] = sum_k p[k] * a[i-k] = convolution(p, a)[i]
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)

    # S1 = ifft(fft(p) * conj(fft(a)))[:n] = cross-correlation
    S1 = np.fft.irfft(fp * np.conj(fa), size)[:n]
    # S2 = ifft(fft(p) * fft(a))[:n] = convolution
    S2 = np.fft.irfft(fp * fa, size)[:n]

    grad_a = 2.0 * n * (S1 + S2)

    # Convert to gradient wrt x (softmax parameterization):
    # grad_x = a * (grad_a - dot(a, grad_a))
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)

    return obj, grad_x


def run_lbfgs(a_init, deadline, tau_start=20.0, tau_end=500.0, n_phases=5):
    """Run L-BFGS-B with continuation on tau for precise local optimization."""
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / a_init.sum()
    n = len(a_init)

    # Start from x = log(a_init)
    x = np.log(a_init)

    best_a = a_init.copy()
    best_J = compute_J(a_init)

    tau_schedule = np.linspace(tau_start, tau_end, n_phases)

    for phase_idx in range(n_phases):
        if time.monotonic() > deadline:
            break
        tau = tau_schedule[phase_idx]
        x_current = x.copy()
        n = len(x_current)

        def func_grad(x_val):
            if time.monotonic() > deadline:
                return 0.0, np.zeros_like(x_val)
            return smooth_max_obj_and_grad(x_val, tau)

        try:
            result = minimize(func_grad, x_current, method='L-BFGS-B',
                             jac=True, options={'maxiter': 200, 'gtol': 1e-10,
                                                'maxls': 30, 'ftol': 1e-12})
            if result.x is not None:
                x = result.x
        except Exception:
            pass

        # Update best
        x_shifted = x - x.max()
        a = np.exp(x_shifted) / np.exp(x_shifted).sum()
        a = np.maximum(a, 0)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()

    return best_a


def run_adam_smooth(a, deadline, max_iters=3000, tau_start=5.0, tau_end=300.0,
                    lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on the softmax parameterization with continuation on tau."""
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    n = len(a)
    x = np.log(a)

    best_a = a.copy()
    best_J = compute_J(a)

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)

        obj, grad = smooth_max_obj_and_grad(x, tau)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodically check J
        if (it + 1) % 100 == 0 or it == max_iters - 1:
            x_shifted = x - x.max()
            a_new = np.exp(x_shifted) / np.exp(x_shifted).sum()
            J = compute_J(a_new)
            if J < best_J:
                best_J = J
                best_a = a_new.copy()

    # Final check
    x_shifted = x - x.max()
    a_new = np.exp(x_shifted) / np.exp(x_shifted).sum()
    J = compute_J(a_new)
    if J < best_J:
        best_a = a_new.copy()

    return best_a


def build_A_values(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    n = len(a)
    rows, cols, vals = build_A_values(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_with_backtracking(a, deadline, max_iters=100, t_base=0.3):
    """LP best-response with backtracking line search."""
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 25:
                break

        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Backtracking line search
        improved = False
        t = t_base
        for _ in range(12):
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_t /= s
            M_t = float(fast_conv(a_t).max())
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
            t *= 0.5
            if t < 1e-5:
                break

        if not improved:
            # Try direct replacement
            M_g = float(fast_conv(g).max())
            if M_g < best_M:
                a = g.copy()
                best_M = M_g
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp > 30:
                    break

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.005))
        asym = np.linspace(-0.1, 0.1, n)
        inits.append(np.maximum(p + 0.1 * asym, 0.005))
        # Frequency perturbation
        F = np.fft.rfft(p, 2 * n)
        F += rng.normal(0, 0.005, F.shape)
        p_f = np.fft.irfft(F, 2 * n)[:n]
        inits.append(np.maximum(p_f, 0.005))

    # Asymmetric inits (optimum is non-symmetric)
    inits.append(np.maximum(0.1 + 0.9 * t**2, 0.01))
    inits.append(np.maximum(0.1 + 0.9 * (1.0 - t)**2, 0.01))
    inits.append(np.maximum(0.1 + 0.5 * t + 0.3 * t**3, 0.01))
    inits.append(np.exp(-2.0 * t) / np.exp(-2.0 * t).sum())
    inits.append(np.exp(-2.0 * (1.0 - t)) / np.exp(-2.0 * (1.0 - t)).sum())
    inits.append(np.maximum(0.2 + 0.6 * t**0.5 + 0.2 * (1.0 - t), 0.01))

    # Random smooth
    k = max(3, n // 10)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))

    # Bump functions
    for center in [0.2, 0.4, 0.6, 0.8]:
        inits.append(np.exp(-15.0 * (t - center)**2) + 0.03)

    # Sine-modulated asymmetric
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.4 + 0.3 * np.sin(2.0 * np.pi * f * t + 0.7) + 0.2 * t, 0.01))

    # Power law with asymmetry
    for alpha in [0.3, 0.7, 1.5, 2.5]:
        inits.append(np.maximum(t**alpha + 0.3 * (1.0 - t) + 0.05, 0.01))

    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    # === PHASE 1: Adam optimization from parent and diverse inits ===
    p1_budget = 250.0
    p1_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Adam Optimization ({p1_budget:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    n_parent = parent.size
    candidate_lengths = sorted(set([
        max(128, n_parent - 64),
        max(128, n_parent - 32),
        n_parent,
        n_parent + 32,
        n_parent + 64,
        n_parent + 128,
        n_parent + 256,
        n_parent + 384,
    ]))

    best_per_length = {}
    n_lengths = len(candidate_lengths)
    time_per_length = p1_budget / n_lengths

    for n in candidate_lengths:
        t0 = time.monotonic()
        remaining = p1_start + p1_budget - time.monotonic()
        if remaining < 10:
            break
        allot = min(time_per_length, remaining * 0.6)
        if allot < 5:
            continue
        dl = time.monotonic() + allot

        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()

            sub_dl = time.monotonic() + (dl - time.monotonic()) / (1.0 + idx * 0.4)
            a = run_adam_smooth(a0, sub_dl, max_iters=2000,
                               tau_start=5.0, tau_end=200.0, lr=0.05)
            M = float(fast_conv(a).max())
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: L-BFGS-B precise optimization ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: L-BFGS-B Optimization", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        # Run L-BFGS on the best sequence found
        dl2 = min(time.monotonic() + 350, deadline - 200)
        a_lbfgs = run_lbfgs(best.seq, dl2, tau_start=20.0, tau_end=500.0, n_phases=5)
        best.update(a_lbfgs)

        # Also try L-BFGS on top lengths from Phase 1
        sorted_lens = sorted(best_per_length.keys(),
                            key=lambda nn: compute_J(best_per_length[nn]))
        for n_try in sorted_lens[:3]:
            if time.monotonic() > dl2:
                break
            if n_try == best.seq.size:
                continue
            a_cand = best_per_length[n_try]
            sub_dl = min(time.monotonic() + 50, dl2)
            a_lbfgs2 = run_lbfgs(a_cand, sub_dl, tau_start=20.0, tau_end=300.0, n_phases=3)
            best.update(a_lbfgs2)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: LP best-response with backtracking ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: LP Best-Response", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 60:
        n = best.seq.size
        est_lp_ms = 3.0 * (n / 128.0) ** 2
        print(f"  n={n}, est LP time={est_lp_ms:.0f}ms", flush=True)
        dl3 = min(time.monotonic() + 250, deadline - 60)
        max_iters = max(30, int((dl3 - time.monotonic()) / max(0.01, est_lp_ms / 1000.0)))
        print(f"  max_iters={max_iters}", flush=True)

        a_lp = run_lp_with_backtracking(best.seq, dl3, max_iters=max_iters, t_base=0.3)
        best.update(a_lp)

        # L-BFGS polish after LP
        if time.monotonic() < deadline - 30:
            a_polish = run_lbfgs(a_lp, min(time.monotonic() + 60, deadline - 25),
                                tau_start=50.0, tau_end=400.0, n_phases=3)
            best.update(a_polish)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Random perturbation restarts ===
    p4_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Perturbation Restarts", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 40:
        n = best.seq.size
        n_restarts = 10
        time_per_restart = (deadline - 40 - time.monotonic()) / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break
            if r == 0:
                a_pert = np.maximum(best.seq + rng.normal(0, 0.01, n), 1e-6)
            elif r == 1:
                a_pert = np.maximum(best.seq + rng.normal(0, 0.04, n), 1e-6)
            elif r == 2:
                shift = np.linspace(-0.05, 0.05, n)
                a_pert = np.maximum(best.seq + shift, 1e-6)
            elif r == 3:
                F = np.fft.rfft(best.seq, 2 * n)
                F += rng.normal(0, 0.01, F.shape) + 1j * rng.normal(0, 0.01, F.shape)
                a_pert = np.fft.irfft(F, 2 * n)[:n]
                a_pert = np.maximum(a_pert, 1e-6)
            elif r == 4:
                mask = rng.random(n) > 0.08
                a_pert = best.seq * mask
                a_pert = np.maximum(a_pert, 1e-6)
            elif r == 5:
                half = best.seq[:n // 2]
                a_pert = np.concatenate([half, half[::-1]])[:n]
                a_pert = np.maximum(a_pert + rng.normal(0, 0.02, n), 1e-6)
            elif r == 6:
                a_norm = best.seq / (best.seq.max() + 1e-15)
                a_pert = a_norm ** rng.choice([0.5, 1.5, 2.0, 0.7, 3.0])
                a_pert = np.maximum(a_pert, 1e-6)
            elif r == 7:
                # Blend with a different-length resampled version
                n_alt = n + rng.choice([-64, -32, 32, 64])
                if n_alt > 100:
                    a_alt = resample_seq(best.seq, n_alt)
                    a_pert = resample_seq(a_alt, n)
                    a_pert = np.maximum(a_pert + rng.normal(0, 0.02, n), 1e-6)
                else:
                    a_pert = np.maximum(best.seq + rng.normal(0, 0.03, n), 1e-6)
            elif r == 8:
                # Random smooth bump added
                center = rng.random()
                a_pert = best.seq + 0.1 * np.exp(-50.0 * (np.linspace(0, 1, n) - center)**2)
                a_pert = np.maximum(a_pert, 1e-6)
            else:
                k = max(3, n // 20)
                kern = np.ones(k) / k
                a_pert = np.convolve(rng.uniform(0, 1, n), kern, 'same')
                a_pert = 0.5 * best.seq + 0.5 * a_pert
                a_pert = np.maximum(a_pert, 1e-6)

            a_pert /= a_pert.sum()

            # Quick Adam optimize
            sub_dl = time.monotonic() + min(time_per_restart * 0.5, 40)
            a_opt = run_adam_smooth(a_pert, sub_dl, max_iters=800,
                                   tau_start=10.0, tau_end=200.0, lr=0.03)

            # L-BFGS polish
            if time.monotonic() < deadline - 15:
                sub_dl2 = min(time.monotonic() + min(time_per_restart * 0.5, 30), deadline - 10)
                a_opt = run_lbfgs(a_opt, sub_dl2, tau_start=50.0, tau_end=300.0, n_phases=2)

            best.update(a_opt)
            print(f"  Restart {r+1}/{n_restarts}: J={best.score:.8f}", flush=True)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Final length exploration and polish ===
    p5_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Final Length Exploration", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 20:
        n_best = best.seq.size
        for delta in [-48, -32, -16, 16, 32, 48, 64, 96, 128, 192]:
            if time.monotonic() > deadline - 12:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 3000:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()

            sub_dl = min(time.monotonic() + 15, deadline - 8)
            a_opt = run_adam_smooth(a_ext, sub_dl, max_iters=400,
                                   tau_start=10.0, tau_end=150.0, lr=0.03)
            best.update(a_opt)

            if time.monotonic() < deadline - 6:
                a_opt = run_lbfgs(a_opt, min(time.monotonic() + 12, deadline - 4),
                                 tau_start=50.0, tau_end=250.0, n_phases=2)
                best.update(a_opt)

    # Final L-BFGS polish on the best
    if best.seq is not None and time.monotonic() < deadline - 3:
        a_final = run_lbfgs(best.seq, deadline, tau_start=100.0, tau_end=400.0, n_phases=2)
        best.update(a_final)

    best.report("Phase 5")

    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
