Step function peak minimization using FFT-accelerated smoothed subgradient
with Nesterov momentum, coordinate descent polishing, and multi-scale
diverse search. Targets improvement over 1.506547 toward 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def compute_gradient_fft(a, weights, fft_size):
    """Compute subgradient using FFT cross-correlation.
    grad[i] = 2 * sum_k weights[k] * a[k-i] for valid k-i
    = 2 * (a * weights reversed) correlation
    """
    # We want: grad[i] = 2 * sum_k w[k] * a[k-i]
    # Using FFT: corr = ifft(fft(a, N) * conj(fft(w_padded, N)))
    # corr[m] = sum_j a[j] * w[m+j] (for m+j < len(w), else 0)
    # So corr[i] = sum_j a[j] * w[i+j] = sum_k w[k] * a[k-i] (substituting k=i+j)
    # Yes! This is exactly what we want.
    w_padded = np.zeros(fft_size)
    w_padded[:len(weights)] = weights
    A = rfft(a, fft_size)
    W = rfft(w_padded, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    grad = 2.0 * corr[:len(a)]
    return grad

def smoothed_subgradient(a, max_iters=5000, time_budget=30.0, tau_init=0.01, tau_decay=0.999, 
                          step_init=0.1, momentum=0.9):
    """FFT-accelerated smoothed subgradient with Nesterov momentum."""
    n = len(a)
    a = normalize_simplex(a.copy()).astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    fft_size = 1
    while fft_size < 2*n - 1:
        fft_size <<= 1
    
    v = np.zeros(n)
    tau = tau_init
    step = step_init
    prev_x = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute convolution and softmax weights
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -100.0, 100.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= wsum
        
        # Compute gradient via FFT
        grad = compute_gradient_fft(a, weights, fft_size)
        # Center the gradient (project onto sum=const hyperplane)
        grad_centered = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-15:
            break
        
        # Nesterov lookahead
        x_lookup = a - momentum * (a - prev_x)
        prev_x = a.copy()
        
        # Normalize gradient
        grad_normed = grad_centered / gnorm
        
        # Take step
        a_new = x_lookup - step * grad_normed
        a_new = normalize_simplex(a_new)
        a = a_new
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        # Adapt step size and temperature
        if obj > best_obj * 1.001:
            step *= 0.9
        else:
            step = min(step * 1.01, 0.3)
        tau *= tau_decay
    
    return best_a, best_obj

def build_toeplitz(a):
    """Build Toeplitz matrix for conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2*n-1)
        A[lo:hi, j] = a[:hi-lo]
    return A

def minpeak_lp(fixed_seq, time_limit=10):
    """Solve min t s.t. conv(fixed_seq, g) <= t, sum(g)=1, g>=0."""
    from scipy.optimize import linprog
    n = len(fixed_seq)
    A = build_toeplitz(fixed_seq)
    c = np.zeros(n+1)
    c[n] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n+1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g/s
    except:
        pass
    return None

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def generate_starts(n, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n-1)/2.0
    
    # Uniform
    starts.append(np.ones(n)/n)
    
    # Triangular
    tri = 1.0 - np.abs(i - center)/center
    tri = np.maximum(tri, 0.01)
    starts.append(tri/np.sum(tri))
    
    # Gaussian (narrow)
    bell = np.exp(-((i-center)**2)/(2*(n/6)**2))
    starts.append(bell/np.sum(bell))
    
    # Gaussian (wide)
    bell_w = np.exp(-((i-center)**2)/(2*(n/3)**2))
    starts.append(bell_w/np.sum(bell_w))
    
    # Two-bump
    for gap in [0.1, 0.2, 0.3]:
        bump1 = np.exp(-((i - center*(1-gap))**2)/(2*(n/10)**2))
        bump2 = np.exp(-((i - center*(1+gap))**2)/(2*(n/10)**2))
        two = bump1 + bump2
        starts.append(two/np.sum(two))
    
    # Random smooth
    for _ in range(5):
        freqs = rng.integers(1, 6, size=3)
        amps = rng.random(3)
        phases = rng.random(3) * 2 * np.pi
        s = 1.0
        for f, a_, p in zip(freqs, amps, phases):
            s += a_ * np.cos(2*np.pi*f*i/n + p)
        s = np.maximum(s, 0.01)
        starts.append(s/np.sum(s))
    
    # Asymmetric
    for skew in [0.3, 0.5, 0.7]:
        asym = np.exp(-((i - center*(1-skew))**2)/(2*(n/4)**2))
        starts.append(asym/np.sum(asym))
    
    return starts

"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def fft_gradient(a, weights, fft_size):
    """Compute subgradient via FFT cross-correlation.
    grad[i] = 2 * sum_k weights[k] * a[k-i] for valid (k-i).
    """
    w_pad = np.zeros(fft_size)
    w_pad[:len(weights)] = weights
    A = rfft(a, fft_size)
    W = rfft(w_pad, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    return 2.0 * corr[:len(a)]


def smoothed_sg(a, time_budget=30.0, tau_init=0.008, tau_end=0.0008,
                step_init=0.08, momentum=0.85):
    """FFT-accelerated smoothed subgradient with Nesterov momentum and adaptive step."""
    n = len(a)
    a = normalize_simplex(a.copy()).astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    v = np.zeros(n)
    tau = tau_init
    step = step_init
    prev_a = a.copy()
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -200.0, 200.0)
        w = np.exp(shifted)
        wsum = np.sum(w)
        if wsum < 1e-300:
            idx = int(np.argmax(conv))
            w = np.zeros(len(conv))
            w[idx] = 1.0
        else:
            w /= wsum

        grad = fft_gradient(a, w, fft_size)
        grad_centered = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_centered))
        if gnorm < 1e-15:
            break

        # Nesterov lookahead
        lookup = a - momentum * (a - prev_a)
        prev_a = a.copy()

        a_new = normalize_simplex(lookup - step * grad_centered / gnorm)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            step = min(step * 1.02, 0.25)
        else:
            a = a_new
            stall += 1
            if stall > 5:
                step *= 0.85
                v *= 0.5
                if stall > 15:
                    # Perturb to escape
                    rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
                    a = normalize_simplex(best_a * np.exp(rng.normal(0, 0.03, n)))
                    prev_a = a.copy()
                    stall = 0
                    step = max(step, 0.03)

        tau = max(tau * 0.9992, tau_end)

    return best_a, best_obj


def peak_directed_refine(a, time_budget=20.0):
    """Targeted refinement: reduce mass at coordinates that contribute most to the peak."""
    n = len(a)
    a = normalize_simplex(a.copy()).astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        peak_val = float(conv[peak_idx])

        # Contribution of a[i] to conv[peak_idx] is proportional to a[peak_idx - i]
        contrib = np.zeros(n)
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        for i in range(lo, hi + 1):
            contrib[i] = a[peak_idx - i]

        # Find top contributor to decrease
        dec_idx = int(np.argmax(contrib))
        if contrib[dec_idx] < 1e-10:
            break

        # Find a safe index to increase (low contribution to peak)
        safe_scores = contrib.copy()
        safe_scores[dec_idx] = -1
        inc_idx = int(np.argmin(safe_scores))

        # Try transferring a small amount
        delta = a[dec_idx] * 0.05
        a[dec_idx] -= delta
        a[inc_idx] += delta
        a = normalize_simplex(a)
        obj_new = compute_obj(a)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a.copy()
        else:
            a = best_a.copy()

        # Also try the reverse direction sometimes
        if it % 3 == 2:
            delta2 = a[inc_idx] * 0.03
            a[dec_idx] += delta2
            a[inc_idx] -= delta2
            a = normalize_simplex(a)
            obj2 = compute_obj(a)
            if obj2 < best_obj - 1e-13:
                best_obj = obj2
                best_a = a.copy()
            else:
                a = best_a.copy()

        it += 1

    return best_a, best_obj


def coordinate_polish(a, time_budget=20.0):
    """Randomized coordinate descent: try pairwise mass transfers."""
    n = len(a)
    a = normalize_simplex(a.copy()).astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))

        # Structured move: transfer from high-contribution to low-contribution
        contrib = np.zeros(n)
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        for i in range(lo, hi + 1):
            contrib[i] = a[peak_idx - i]

        # Top 3 contributors
        top_contrib = np.argsort(contrib)[-3:]
        # Bottom 3 (safe to increase)
        bottom = np.argsort(contrib)[:3]

        moved = False
        for d in top_contrib:
            for c in bottom:
                if d == c:
                    continue
                delta = a[d] * (0.02 + 0.03 * rng.random())
                a[d] -= delta
                a[c] += delta
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s < 1e-15:
                    a = best_a.copy()
                    break
                a /= s
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a.copy()
                    moved = True
                    break
                else:
                    a = best_a.copy()
            if moved:
                break

        if not moved:
            # Random perturbation
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2:
                delta = max(a[i1], a[i2]) * 0.05 * rng.random()
                if rng.random() < 0.5:
                    a[i1] -= delta
                    a[i2] += delta
                else:
                    a[i1] += delta
                    a[i2] -= delta
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-15:
                    a /= s
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a.copy()
                    else:
                        a = best_a.copy()

        it += 1

    return best_a, best_obj


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def minpeak_lp(fixed_seq, time_limit=8):
    from scipy.optimize import linprog
    n = len(fixed_seq)
    A = build_toeplitz(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate(a, time_budget=40.0):

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def generate_starts(n, seed=42):
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    starts.append(np.ones(n) / n)

    tri = np.maximum(1.0 - np.abs(i - center) / center, 0.01)
    starts.append(tri / np.sum(tri))

    bell = np.exp(-((i - center) ** 2) / (2 * (n / 5) ** 2))
    starts.append(bell / np.sum(bell))

    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 3) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # Two-bump structures
    for gap in [0.15, 0.25, 0.35, 0.5]:
        bump1 = np.exp(-((i - center * (1 - gap)) ** 2) / (2 * (n / 8) ** 2))
        bump2 = np.exp(-((i - center * (1 + gap)) ** 2) / (2 * (n / 8) ** 2))
        two = bump1 + bump2
        starts.append(two / np.sum(two))

    # Asymmetric
    for skew in [0.2, 0.4, 0.6]:
        asym = np.exp(-((i - center * (1 - skew)) ** 2) / (2 * (n / 4) ** 2))
        starts.append(asym / np.sum(asym))

    # Sinusoidal modulations
    for freq in [2, 3, 5]:
        mod = 1.0 + 0.4 * np.cos(2 * np.pi * freq * i / n)
        mod = np.maximum(mod, 0.05)
        starts.append(mod / np.sum(mod))

    # Random smooth
    for _ in range(4):
        f1 = int(rng.integers(1, 5))
        f2 = int(rng.integers(2, 8))
        p1 = rng.random() * 2 * np.pi
        p2 = rng.random() * 2 * np.pi
        s = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * i / n + p1) + 0.2 * np.cos(2 * np.pi * f2 * i / n + p2)
        s = np.maximum(s, 0.01)
        starts.append(s / np.sum(s))

    return starts

    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print(f"\n=== Phase 5: Final polish (budget={remaining:.0f}s) ===")
            a_fin, obj_fin = polish_pipeline(best_seq, time_budget=remaining * 0.7)
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"  Final: {best_score:.8f}")

            # One more coordinate polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 8:
                a_cp, obj_cp = coordinate_polish(best_seq, time_budget=remaining2 * 0.6)
                if obj_cp < best_score:
                    best_score = obj_cp
                    best_seq = a_cp.copy()
                    print(f"  Coord: {best_score:.8f}")

    # Phase 6: Batch of small perturbations on final best
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            n_fin = len(best_seq)
            rng = np.random.default_rng(seed + 77777)
            n_pert = min(8, int(remaining / 3))
            for pi in range(n_pert):
                if time.monotonic() > deadline - 3:
                    break
                scale = 0.01 + 0.008 * pi
                a_p = best_seq * np.exp(rng.normal(0, scale, n_fin))
                a_p = normalize_simplex(a_p)
                t_alloc = min(remaining * 0.12, 8.0)
                if t_alloc < 2:
                    break
                a_p, obj_p = smoothed_sg(a_p, time_budget=t_alloc)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Perturb {pi} (s={scale:.3f}): ** {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = min(best_score, val)
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
