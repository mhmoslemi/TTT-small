"""
Multi-method optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: efficient coordinate descent with adaptive candidates, SCA-LP with warm-starting,
active-set tracking, and perturbation-based restarts to escape local minima.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent(a, deadline, seed=42, verbose=False):
    """
    Efficient coordinate descent with incremental convolution updates.
    Uses adaptive candidate generation based on active constraint analysis.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    cur_max = conv.max()
    best_obj = 2.0 * n * cur_max / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 200
    sweep_count = 0

    while time.monotonic() < deadline:
        sweep_count += 1
        improved = False

        # Random order for diversity
        order = rng.permutation(n)

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            kq = 2 * i
            has_quad = (0 <= kq <= 2 * n - 2)

            # Get the slice of a that interacts with a[i]
            # conv[k] involves a[i]*a[k-i] for k in [i, i+n-1]
            idx = np.arange(lo, hi + 1)
            other_idx = idx - i  # these are the indices j such that a[j] = a[k-i]
            other_vals = a[other_idx]

            # Try candidate values
            candidates = [0.0]
            if old > 1e-14:
                candidates.extend([old * r for r in [0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0]])

            # Add "scale" candidates based on 1/n
            scale = 1.0 / n
            candidates.extend([scale * r for r in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0]])

            # Add candidate from active constraint: if conv[k*] is max, find x that makes it tight
            k_star = np.argmax(conv)
            if lo <= k_star <= hi and k_star != kq:
                j_star = k_star - i
                if 0 <= j_star < n and a[j_star] > 1e-14:
                    # 2*x*a[j_star] + (conv[k_star] - 2*old*a[j_star]) = cur_max_target
                    # We want x such that this term equals some fraction of current max
                    base_k = conv[k_star] - 2.0 * old * a[j_star]
                    for frac in [0.9, 0.95, 1.0, 1.05]:
                        target = frac * cur_max
                        x_cand = (target - base_k) / (2.0 * a[j_star])
                        if x_cand > 1e-10:
                            candidates.append(x_cand)

            # Remove duplicates and filter
            candidates = np.array(candidates)
            candidates = candidates[candidates >= 0]
            candidates = np.unique(np.round(candidates, 12))

            best_x = old
            best_local = best_obj

            for x in candidates:
                x = float(x)
                delta = x - old
                if abs(delta) < 1e-15:
                    continue

                # Update conv incrementally
                conv[lo:hi + 1] += 2.0 * delta * other_vals
                if has_quad:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_max = conv.max()
                    new_obj = 2.0 * n * new_max / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_x = x
                    if new_s >= 1e-10:
                        cur_max = new_max

                # Revert
                conv[lo:hi + 1] -= 2.0 * delta * other_vals
                if has_quad:
                    conv[kq] -= delta * delta

            # Apply best
            if best_x != old:
                delta = best_x - old
                conv[lo:hi + 1] += 2.0 * delta * other_vals
                if has_quad:
                    conv[kq] += delta * delta
                a[i] = best_x
                s -= old
                s += best_x
                cur_max = conv.max()
                improved = True

        current_obj = 2.0 * n * cur_max / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and sweep_count % 50 == 0:
            print(f"    CD sweep {sweep_count}: obj={best_obj:.8f}", flush=True)

        if no_improve > max_no_improve:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def sca_lp_optimize(a, deadline, max_iters=500, verbose=False):
    """
    SCA-LP optimization with warm-starting.
    Linearizes the quadratic constraint and solves LP subproblems.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Build LP: minimize t
        # s.t. 2*(a*g)[k] - c[k] <= t  for k=0,...,2n-2
        #      sum(g) = 1, g >= 0
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=[1.0],
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    # Try line search
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-14:
                            a = a_try.copy()
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 20 == 0:
            print(f"    SCA it {it}: obj={best_obj:.8f}", flush=True)

        if no_improve > 40:
            break

    return best_a, best_obj


def adam_smooth(a, deadline, max_iters=5000, tau_init=15.0, tau_max=150.0,
                lr_init=0.04, verbose=False):
    """
    Adam optimization with log-sum-exp smoothing of max(conv).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_count = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_init + (tau_max - tau_init) * frac

        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()

        # Gradient computation via FFT
        a_rev = a[::-1]
        w_rev = w[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        Wrev_fft = np.fft.rfft(w_rev, size)

        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wrev_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad[i] += conv2[idx2]

        grad = grad - np.mean(grad)

        t_count += 1
        lr = lr_init / (1.0 + 0.05 * t_count ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                rng = np.random.default_rng(it + 12345)
                a = best_a + rng.normal(0, 0.03, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

        if verbose and it % 200 == 0:
            print(f"    Adam it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine (Hann window)
    inits.append(project_simplex(0.5 * (1 + np.cos(2 * np.pi * t))))

    # Gaussian centered
    for width in [0.15, 0.25, 0.35, 0.5]:
        inits.append(project_simplex(np.exp(-((t - 0.5) / width) ** 2) + 0.01))

    # Two-sided Gaussian
    for width in [0.1, 0.15, 0.2]:
        g = np.exp(-((t - 0.2) / width) ** 2) + np.exp(-((t - 0.8) / width) ** 2) + 0.01
        inits.append(project_simplex(g))

    # Sine patterns
    for freq in [1, 2, 3, 4, 5]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))

    # Smooth random
    ksize = max(5, n // 8)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    return inits

        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 30:
            break
        alloc = min(per_init_time, remaining_check - 5)
        if alloc < 15:
            break

        a0 = inits[idx].copy()
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        # Quick CD on this init
        a_opt = coord_descent(a0, deadline=time.monotonic() + alloc * 0.7, seed=100 + idx)
        update_best(a_opt)

        # If it improved, do SCA-LP polish
        if best_score < float('inf') and remaining_check - (time.monotonic() - t_start - 0) > 10:
            if time.monotonic() < phase2_deadline - 15:
                a_sca, _ = sca_lp_optimize(best_seq, deadline=phase2_deadline - 10, max_iters=50)
                update_best(a_sca)

        print(f"  Init {idx}: best J={best_score:.8f}", flush=True)

    # === PHASE 3: Try different lengths (15% of time) ===
    phase3_deadline = t_start + 0.85 * (budget_s - 15)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length exploration (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    candidate_ns = []
    for n_try in [384, 512, 640, 768, 896, 1024, 1152, 1280]:
        if n_try != n_parent and n_try <= 1536:
            candidate_ns.append(n_try)
    candidate_ns.sort(key=lambda x: abs(x - n_parent))

    for new_n in candidate_ns[:4]:
        remaining_check = phase3_deadline - time.monotonic()
        if remaining_check < 60:
            break
        alloc = min(80.0, remaining_check - 20)
        if alloc < 40:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext /= a_ext.sum()

        a_opt = coord_descent(a_ext, deadline=time.monotonic() + alloc * 0.6, seed=200)
        update_best(a_opt)

        if time.monotonic() < phase3_deadline - 20:
            a_sca, _ = sca_lp_optimize(best_seq, deadline=phase3_deadline - 15, max_iters=30)
            update_best(a_sca)

        print(f"  n={new_n}: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Intensive final polish (15% of time) ===
    phase4_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final polish (budget {phase4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        remaining = phase4_deadline - time.monotonic()

        # Pass A: Long CD
        if remaining > 30:
            t_a = time.monotonic() + 0.30 * remaining
            a = coord_descent(best_seq, deadline=t_a, seed=777, verbose=False)
            update_best(a)

        remaining = phase4_deadline - time.monotonic()
        # Pass B: SCA-LP
        if remaining > 30:
            t_b = time.monotonic() + 0.25 * remaining
            a_sca, _ = sca_lp_optimize(best_seq, deadline=t_b, max_iters=100, verbose=False)
            update_best(a_sca)

        remaining = phase4_deadline - time.monotonic()
        # Pass C: Adam with different params
        if remaining > 30:
            t_c = time.monotonic() + 0.20 * remaining
            a_adam, _ = adam_smooth(best_seq, deadline=t_c, max_iters=5000,
                                     tau_init=8.0, tau_max=80.0, lr_init=0.02)
            update_best(a_adam)

        remaining = phase4_deadline - time.monotonic()
        # Pass D: Final CD
        if remaining > 20:
            t_d = deadline - 5
            a = coord_descent(best_seq, deadline=t_d, seed=999, verbose=False)
            update_best(a)

    # Fallback
    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    # Final evaluation
    final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
