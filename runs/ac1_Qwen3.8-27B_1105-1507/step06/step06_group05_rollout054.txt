def solve_peak_reduction_lp(a, k_star, epsilon, time_limit=30):
    """
    Solve LP to find direction that reduces peak at k_star.
    min 2*sum_j a[k*-j]*delta[j]
    s.t. 2*sum_j a[k-j]*delta[j] <= epsilon for k != k_star
         sum(delta) = 0
         -a[j] <= delta[j]
    """
    n = len(a)
    m = 2*n - 1
    # Build coefficient matrix for constraints
    # For each k (0..2n-2), the coefficient of delta[j] in conv(a+delta,a+delta)[k] is 2*a[k-j]
    A = np.zeros((m, n))
    for j in range(n):
        for k in range(max(0, j), min(2*n-1, j+n)):
            A[k, j] = 2*a[k-j]
    
    # Objective: minimize A[k*,:] @ delta
    c = A[k_star, :]
    
    # Constraints: A[k,:] @ delta <= epsilon for k != k_star
    A_ub = np.delete(A, k_star, axis=0)
    b_ub = np.full(m-1, epsilon)
    
    # Equality: sum(delta) = 0
    A_eq = np.ones(1).reshape(1, n)  # actually np.ones((1,n))
    b_eq = np.zeros(1)
    
    # Bounds: -a[j] <= delta[j] <= some_upper
    bounds = [(-a[j], None) for j in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs', options={'time_limit': time_limit})
    if res.status == 0:
        return res.x
    return None

Multi-strategy optimization with peak-reduction LP, multi-peak constraints,
frequency-domain polishing, and structured perturbations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_max_sum(a, M, time_limit=30):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def solve_peak_reduction_lp(a, k_star, epsilon, time_limit=30):
    """LP that directly targets reducing the peak at position k_star."""
    n = len(a)
    m = 2 * n - 1
    A = build_toeplitz(a)  # A[k,j] = a[k-j] for valid k,j
    
    # Objective: minimize A[k_star, :] @ delta (reduces peak)
    c = A[k_star, :].copy()
    
    # Constraints: A[k,:] @ delta <= epsilon for k != k_star
    # But we only need to worry about peaks that are close to the max
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    # Only constrain peaks that are > 0.8 * cmax
    peak_threshold = 0.8 * cmax
    constraint_k = [k for k in range(m) if k != k_star and conv[k] > peak_threshold]
    
    if constraint_k:
        A_ub = A[constraint_k, :]
        b_ub = np.full(len(constraint_k), epsilon)
    else:
        A_ub = None
        b_ub = None
    
    # Equality: sum(delta) = 0
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    
    # Bounds: delta[j] >= -a[j] (to keep a + delta >= 0)
    # Upper bound: delta[j] <= a[j] (to keep delta from being too large)
    bounds = [(-a[j], a[j]) for j in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:

    return None

def solve_multi_peak_lp(a, top_k_idx, M, time_limit=30):

def golden_section_search(f, lo, hi, tol=0.01, max_iter=20):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0

def line_search_obj(a, direction, max_t=1.0):
    """Find best t in [0, max_t] for a + t*(direction - a), normalized."""
    def obj_at_t(t):
        a_try = (1.0 - t) * a + t * direction
        a_try = normalize(a_try)
        return compute_obj(a_try)
    
    t_opt = golden_section_search(obj_at_t, 0.0, max_t, tol=0.01, max_iter=15)
    return t_opt

def spectral_equalization_v2(a, max_iters=40, time_budget=30.0):
    """Improved frequency-domain flattening with adaptive compression."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        
        # Adaptive compression: start with high alpha (aggressive flattening)
        alpha = 0.3 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            blend = 0.3 + 0.4 * (it / float(max_iters))
            a = (1 - blend) * a + blend * a_new
            a = normalize(a)
    
    return best_a

def soft_max_gradient(a, tau_init=0.001, step_size=0.04, max_steps=100, time_budget=30.0):
    """Soft-max gradient descent with adaptive temperature."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        tau = tau_init + 0.5 * cmax / (n * (step + 1))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        w_ext = np.zeros(2 * n - 1)
        w_ext[:] = weights
        a_flip = a[::-1]
        
        Wf = rfft(w_ext, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        
        grad_centered = grad - np.mean(grad)
        a_new = normalize(a - step_size * grad_centered)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.7
            if step_size < 1e-8:
                break
        
        tau_init *= 0.97
    
    return best_a

def coordinate_descent(a, time_budget=20.0):
    """Targeted coordinate descent on peak contributors."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find top contributors to peak
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)
        
        if not contrib:
            break
        
        improved = False
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            
            # Try moving mass from j1 or j2 to a low-convolution position
            for _ in range(5):
                # Find position with minimum convolution contribution
                min_pos = int(rng.integers(0, n))
                a_try = a.copy()
                delta = 0.03 * (a_try[j1] + a_try[j2])
                # Move from the larger contributor
                if a_try[j1] >= a_try[j2]:
                    a_try[j1] -= delta
                    src = j1
                else:
                    a_try[j2] -= delta
                    src = j2
                a_try[min_pos] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved:
            # Try swapping two positions
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                a_try = a.copy()
                a_try[i], a_try[j] = a_try[j], a_try[i]
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
        
        if not improved and rng.random() < 0.05:
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

"""
Multi-strategy optimization combining LP alternating (standard, peak-reduction, multi-peak),
frequency-domain spectral equalization, soft-max gradient descent, and coordinate descent.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix A where (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=25):
    """LP: max sum(g) s.t. A@g <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_topk(a, M, top_k_indices, time_limit=25):
    """LP: max sum(g) s.t. conv(a,g) <= M only at top-k positions, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(len(top_k_indices), M)
    A_ub = A[top_k_indices, :]
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_peak_reduction_lp(a, k_star, epsilon, time_limit=25):
    """
    LP: minimize A[k_star,:] @ delta (reduce peak at k_star)
    s.t. A[k,:] @ delta <= epsilon for other high peaks
         sum(delta) = 0
         -a[j] <= delta[j] <= a[j]
    Returns delta or None.
    """
    n = len(a)
    m = 2 * n - 1
    A = build_toeplitz(a)
    c = A[k_star, :].copy()
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    threshold = 0.85 * cmax
    constraint_k = [k for k in range(m) if k != k_star and conv[k] > threshold]
    A_ub = None
    b_ub = None
    if constraint_k:
        A_ub = A[constraint_k, :]
        b_ub = np.full(len(constraint_k), epsilon)
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bounds = [(-a[j], a[j]) for j in range(n)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def golden_section_search(f, lo, hi, tol=0.005, max_iter=20):
    """Golden section search for minimum on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def multi_step_line_search(a, direction, max_t=0.8):
    """Combined grid + golden section line search."""
    def obj_at_t(t):
        a_try = (1.0 - t) * a + t * direction
        return compute_obj(normalize(a_try))
    # Coarse grid
    grid = np.linspace(0, max_t, 9)
    best_grid_obj = np.inf
    best_grid_t = 0.0
    for t in grid:
        o = obj_at_t(t)
        if o < best_grid_obj:
            best_grid_obj = o
            best_grid_t = t
    # Refine around best grid point
    lo = max(0, best_grid_t - max_t / 8)
    hi = min(max_t, best_grid_t + max_t / 8)
    t_opt = golden_section_search(obj_at_t, lo, hi, tol=0.005, max_iter=15)
    return t_opt

def spectral_equalization_v2(a, max_iters=50, time_budget=30.0):
    """Frequency-domain flattening with adaptive compression and blending."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        # Adaptive: start aggressive (low alpha = more flattening), ease off
        alpha = 0.2 + 0.5 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = np.maximum(P_norm ** alpha, 1e-12) * p_max
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend to maintain progress
            blend = 0.3 + 0.3 * (it / float(max_iters))
            a = normalize((1 - blend) * a + blend * a_new)

    return best_a


def soft_max_gradient(a, tau_init=0.001, step_size=0.03, max_steps=100, time_budget=30.0):
    """Soft-max (log-sum-exp) weighted gradient descent."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    size = 1
    while size < 3 * n - 1:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        tau = tau_init + 0.5 * cmax / (n * (step + 1))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        # Gradient via correlation
        w_ext = np.zeros(2 * n - 1)
        w_ext[:] = weights
        a_flip = a[::-1]
        Wf = rfft(w_ext, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        grad_centered = grad - np.mean(grad)
        a_new = normalize(a - step_size * grad_centered)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.7
            if step_size < 1e-8:
                break
        tau_init *= 0.98

    return best_a


def coordinate_descent(a, time_budget=20.0):
    """Targeted coordinate descent on top peak contributors with L1 moves."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))

        contrib = []
        for j in range(n):

            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)
        if not contrib:
            break
        improved = False
        for peak_val, j1, j2 in contrib[:10]:
            if time.monotonic() - start > time_budget:

            if j1 == j2:
                continue
            for _ in range(4):
                # Move mass to a position that minimizes impact
                j_target = int(rng.integers(0, n))
                if j_target in (j1, j2):
                    continue
                a_try = a.copy()
                # Try both directions of mass transfer

                        a_try[j1] -= delta
                        a_try[j_target] += delta
                    else:
                        delta = frac * a_try[j2]
                        a_try[j2] -= delta
                        a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
        if not improved:
            # Random swap or small perturbation
            if rng.random() < 0.3:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    a_try = a.copy()
                    a_try[i], a_try[j] = a_try[j], a_try[i]
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
            else:

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

    print(f"Phase 1: LP optimization (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        for cycle in range(10):
            if time.monotonic() > deadline - 250:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.11, 80.0)
            a_new = lp_alternate_combined(a, max_iters=35, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  LP cycle {cycle}: {best_score:.8f} **")
            else:
                a = a_new
                print(f"  LP cycle {cycle}: {obj_new:.8f}")

    # Phase 2: Spectral + gradient + LP follow-up
    print(f"\n{'='*60}")
    print(f"Phase 2: Spectral + gradient")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())

        if time.monotonic() < deadline - 200:
            a_spec = spectral_equalization_v2(a, max_iters=50, time_budget=35.0)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                a = a_spec
                print(f"  Spectral: {best_score:.8f} **")
            else:
                a = normalize(0.7 * normalize(a) + 0.3 * normalize(a_spec))
                print(f"  Spectral: {obj_spec:.8f}")

        if time.monotonic() < deadline - 200:
            a_soft = soft_max_gradient(a, tau_init=0.0008, step_size=0.03,
                                       max_steps=120, time_budget=35.0)
            obj_soft = compute_obj(a_soft)
            if obj_soft < best_score:
                best_score = obj_soft
                best_seq = a_soft.copy()
                a = a_soft
                print(f"  SoftMax: {best_score:.8f} **")
            else:
                print(f"  SoftMax: {obj_soft:.8f}")

        if time.monotonic() < deadline - 160:
            a_new = lp_alternate_combined(a, max_iters=30, time_budget=55.0)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  Post-Phase2 LP: {best_score:.8f} **")
            else:
                print(f"  Post-Phase2 LP: {obj_new:.8f}")

    # Phase 3: Multi-n exploration
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700
    n_options = sorted(set([
        n_best - 250, n_best - 200, n_best - 150, n_best - 100,
        n_best - 50, n_best - 25, n_best,
        n_best + 25, n_best + 50, n_best + 100, n_best + 150,
        n_best + 200, n_best + 300, n_best + 400
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1300]

    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        if n_try == n_best:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample_sequence(best_seq, n_try))
        alloc = min(remaining * 0.09, 28.0)
        if n_try <= 950:
            a_try = lp_alternate_combined(a_try, max_iters=12, time_budget=alloc * 0.5)
        a_try = soft_max_gradient(a_try, tau_init=0.001, step_size=0.025,
                                  max_steps=60, time_budget=alloc * 0.4)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # Phase 4: Structured restarts
    print(f"\n{'='*60}")
    print(f"Phase 4: Structured restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed)
        restart_strategies = [
            'circular_shift', 'freq_mod', 'gauss_mult',
            'block_reverse', 'end_shift', 'noise'
        ]
        for restart in range(10):
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.09, 28.0)
            strategy = restart_strategies[restart % len(restart_strategies)]

            if strategy == 'circular_shift':
                shift = int(rng.integers(1, max(2, n_cur // 15)))
                a_try = np.concatenate([best_seq[-shift:], best_seq[:-shift]])
            elif strategy == 'freq_mod':
                freq = rng.uniform(0.5, 4.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.12 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
                a_try = best_seq * mod
            elif strategy == 'gauss_mult':
                a_try = best_seq * np.exp(rng.normal(0, 0.07, n_cur))
            elif strategy == 'block_reverse':
                a_try = best_seq.copy()
                blen = int(rng.integers(n_cur // 5, n_cur // 2))
                s_idx = int(rng.integers(0, n_cur - blen))
                a_try[s_idx:s_idx + blen] = a_try[s_idx:s_idx + blen][::-1]
            elif strategy == 'end_shift':
                # Move a chunk from one end to the other
                chunk = int(n_cur // 6)
                s_idx = int(rng.integers(0, chunk + 1))
                a_try = np.concatenate([best_seq[s_idx:], best_seq[:s_idx]])
            else:
                a_try = best_seq + rng.normal(0, 0.02 * np.max(best_seq), n_cur)

            a_try = normalize(a_try)
            if n_cur <= 950:
                a_try = lp_alternate_combined(a_try, max_iters=10, time_budget=alloc * 0.5)
            a_try = soft_max_gradient(a_try, tau_init=0.001, step_size=0.02,
                                      max_steps=50, time_budget=alloc * 0.4)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart} ({strategy}): {best_score:.8f} **")
            else:
                print(f"  Restart {restart} ({strategy}): {obj_try:.8f}")

    # Phase 5: Final intensive polish
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        for polish in range(6):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            alloc = min(remaining * 0.22, 30.0)
            # Combine all methods in sequence
            if alloc > 12:
                a_pol = lp_alternate_combined(a_final, max_iters=15, time_budget=alloc * 0.4)
            else:
                a_pol = a_final.copy()
            a_pol = soft_max_gradient(a_pol, tau_init=0.0005, step_size=0.015,
                                      max_steps=60, time_budget=alloc * 0.3)
            if alloc > 6:
                a_pol = coordinate_descent(a_pol, time_budget=min(8, alloc * 0.2))
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish}: {best_score:.8f} **")
            else:
                a_final = a_pol
                print(f"  Polish {polish}: {obj_pol:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
