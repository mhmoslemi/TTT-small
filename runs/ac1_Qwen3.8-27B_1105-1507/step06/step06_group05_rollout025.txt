    while size < 2 * n - 1:

def build_toeplitz(a):

    m = 2 * n - 1

    for j in range(n):
        k_start = j
        k_end = min(j + n, m)

    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, tighten_top=0, tighten_frac=0.95, time_limit=15):
    n = len(a)
    m = 2 * n - 1
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(m, M)
    if tighten_top > 0:
        conv = compute_conv(a)
        peak_indices = np.argsort(conv)[::-1][:tighten_top]
        for idx in peak_indices:
            b_ub[idx] = M * tighten_frac
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section(f, lo, hi, tol=0.01, max_iter=18):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def lp_optimize(a, max_iters=30, time_budget=60.0):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Adaptive tightening: start loose, get tighter over iterations
        tighten_top = min(5 + it, 15)
        tighten_frac = max(0.85, 1.0 - 0.01 * it)

        g = solve_lp(a, M, tighten_top=tighten_top, tighten_frac=tighten_frac,
                     time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            # Fall back to no tightening
            g = solve_lp(a, M, tighten_top=0, time_limit=max(3, int(remaining * 0.4)))
            if g is None:
                break

        def obj_at_t(t):
            a_try = normalize((1.0 - t) * a + t * g)
            return compute_obj(a_try)

        t_opt = golden_section(obj_at_t, 0.0, 1.0, tol=0.02, max_iter=12)

        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            rng = np.random.default_rng((it * 7919 + 12345) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03 + 0.01 * (it % 5), n)))

    return best_a


def accelerated_gradient(a, max_steps=400, time_budget=50.0):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    v = np.zeros(n)
    beta = 0.9
    step_size = 0.06
    tau_base = 0.0

    size = 1
    while size < 3 * n - 1:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(y)
        cmax = float(np.max(conv))
        cmean = float(np.mean(conv))

        # Adaptive temperature based on peak-to-mean ratio
        tau = max((cmax - cmean) * 0.2, cmax * 0.05, 1e-8)
        shifted = np.clip((conv - cmax) / tau, -500.0, 0.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_ext = np.zeros(2 * n - 1)
        w_ext[:] = weights
        y_flip = y[::-1]

        Wf = rfft(w_ext, size)
        Yf = rfft(y_flip, size)
        corr = irfft(Wf * Yf, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        grad_centered = grad - np.mean(grad)

        # Heavy-ball momentum
        v = beta * v + (1.0 - beta) * grad_centered
        y_new = normalize(y - step_size * v)
        obj_new = compute_obj(y_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = y_new.copy()
            a = y_new
        else:
            step_size *= 0.6
            if step_size < 1e-8:
                break
            if step % 15 == 0:
                v = np.zeros(n)
                beta = 0.9

    return best_a


def spectral_optimize(a, max_iters=100, time_budget=40.0):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Progressive compression: start aggressive, ease off
        alpha = 0.2 + 0.5 * (it / max_iters)
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.3 + 0.3 * (it / max_iters)
            a = normalize((1.0 - blend) * a + blend * a_new)

    return best_a


def symmetrize_and_optimize(a, time_budget=30.0):
    n = len(a)
    # Symmetrize: a_sym[i] = (a[i] + a[n-1-i]) / 2
    a_sym = normalize((a + a[::-1]) / 2.0)
    obj_sym = compute_obj(a_sym)

    if obj_sym < compute_obj(a):
        # Symmetrization helped, optimize from symmetric point
        a_sym = lp_optimize(a_sym, max_iters=10, time_budget=time_budget * 0.5)
        a_sym = accelerated_gradient(a_sym, max_steps=100, time_budget=time_budget * 0.4)
        return a_sym

    return a


def peak_shaving(a, time_budget=30.0):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n and j != k2:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        improved = False
        for peak_prod, j1, j2 in contrib[:6]:
            if time.monotonic() - start > time_budget:
                break

            src = j1 if a[j1] >= a[j2] else j2

            # Find trough positions (low convolution values)
            trough_candidates = []
            for k in range(n):
                if k != src and k != j1 and k != j2:
                    trough_candidates.append((conv[k], k))
            trough_candidates.sort()

            for _, k_dst in trough_candidates[:4]:
                for frac in [0.08, 0.15, 0.25]:
                    delta = frac * a[src]
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[k_dst] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            a = normalize(a * np.exp(rng.normal(0, 0.04, n)))
            obj_check = compute_obj(a)
            if obj_check < best_obj:
                best_obj = obj_check
                best_a = a.copy()

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def blend_sequences(a1, a2, t):
    n = len(a1)
    if len(a2) != n:
        a2 = resample_sequence(a2, n)
    return normalize((1.0 - t) * normalize(a1) + t * normalize(a2))
