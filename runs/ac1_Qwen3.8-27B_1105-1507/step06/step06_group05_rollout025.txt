```python
    tau_min = c_max / (n * 200)
    
    # Nesterov momentum parameters
    beta = 0.9  # momentum coefficient
    v = np.zeros(n)
    alpha = 0.005  # initial step size
    
    start = time.monotonic()
    step = 0
    stall = 0
    best_stall_obj = best_obj
    
    while time.monotonic() - start < budget - 5:
        # Nesterov lookahead
        a_lookahead = a - beta * v
        a_lookahead = np.maximum(a_lookahead, 0)
        s_la = float(np.sum(a_lookahead))
        if s_la > 1e-15:
            a_lookahead /= s_la
        else:
            a_lookahead = a
        
        # Compute gradient at lookahead point
        conv_la = compute_conv(a_lookahead)
        cmax_la = float(np.max(conv_la))
        
        shifted = np.clip((conv_la - cmax_la) / tau, -500.0, 0.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(len(conv_la))
            weights[int(np.argmax(conv_la))] = 1.0
        else:
            weights /= w_sum
        
        grad = gradient_with_weights(a_lookahead, weights, size)
        grad_centered = grad - np.mean(grad)
        
        # Update momentum
        v = beta * v + alpha * grad_centered
        
        # Compute new position
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-15:

            v = np.zeros(n)
```

The key is to handle the projection carefully. When we clip and renormalize, the effective step is different from what we intended. This can cause oscillation.

One approach: instead of projecting after the update, use a "mirror map" (exponential family) which naturally handles non-negativity:
```
log(a_new[i]) = log(a[i]) - alpha * grad[i]
a_new = exp(log(a_new)) / sum(exp(log(a_new)))
```

This is the **exponential weights update** or **multiplicative weights** update. It's equivalent to multiplicative update:
```
a_new[i] = a[i] * exp(-alpha * grad[i]) / Z
```

Where Z is the normalizing constant. This is always positive and automatically normalized!

This might be better than the additive update with projection. Let me try this.

The exponential weights update is:
```
a_new[i] = a[i] * exp(-alpha * (grad[i] - mean(grad))) / Z
```

where Z = sum_i a[i] * exp(-alpha * (grad[i] - mean(grad)))

This is equivalent to a multiplicative step in the log-domain, which is a mirror descent step with the negative entropy as the Bregman divergence.

The advantage is:
- Always stays in the positive orthant (no clipping needed)
- Automatically normalized (no renormalization needed, it's built in)
- The step size alpha controls the "loudness" of the update

Actually, one more important insight: the multiplicative update `a_new = a * exp(-alpha * grad_centered) / Z` is equivalent to a step in the "exponential family". The step size alpha controls how "aggressive" the update is. 

For a well-conditioned problem, a larger alpha converges faster. For an ill-conditioned problem, a smaller alpha is more stable. The backtracking line search handles this adaptively.

However, there's a subtlety: the objective `2n * max(conv(a,a)) / (sum(a))^2` is not convex in `a` (it's the max of linear functions divided by a quadratic). So the standard convergence guarantees don't apply. But in practice, the gradient method works well for this type of problem.

```python
"""
Annealed multiplicative gradient descent with adaptive step size, restless
temperature scheduling, basin hopping, and multi-n exploration for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-based gradient computation with
log-sum-exp smoothing of the max operator. Includes optional LP polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_gradient(a, weights, size):
    """Compute gradient of weighted sum of convolution values.
    grad[i] = 2 * sum_k weights[k] * a[k-i] * I(0<=k-i<n)
    """
    n = len(a)
    w_ext = np.zeros(2 * n - 1)
    w_ext[:] = weights
    a_flip = a[::-1]
    Wf = rfft(w_ext, size)
    Af = rfft(a_flip, size)
    corr = irfft(Wf * Af, size)
    return 2.0 * corr[n - 1:2 * n - 1]


def gradient_optimize(a, budget, seed=0):
    """Core multiplicative gradient descent with adaptive step size and annealing."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    size = 1
    while size < 3 * n - 2:
        size <<= 1

    conv_init = compute_conv(a)
    c_max = float(np.max(conv_init))
    tau = c_max / (n * 8.0)
    tau_min = c_max / (n * 300.0)

    alpha = 1.0
    start = time.monotonic()
    step = 0
    stall = 0
    best_at_step = best_obj

    rng = np.random.default_rng(seed)

    while time.monotonic() - start < budget - 3.0:
        # Compute convolution and gradient
        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights with temperature
        shifted = np.clip((conv - cmax) / tau, -500.0, 0.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum

        # Gradient via FFT
        grad = compute_gradient(a, weights, size)
        grad_centered = grad - np.mean(grad)

        # Normalize gradient for stable step size
        grad_rms = float(np.sqrt(np.mean(grad_centered ** 2)))
        if grad_rms < 1e-30:
            alpha *= 0.5
            continue
        grad_norm = grad_centered / grad_rms

        # Multiplicative update
        log_update = -alpha * grad_norm
        log_update -= np.max(log_update)
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        s = float(np.sum(a_new))
        if s < 1e-15:
            alpha *= 0.5
            continue
        a_new /= s

        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            alpha = min(alpha * 1.03, 3.0)
        else:
            stall += 1
            alpha *= 0.75
            a = best_a.copy()

        # Anneal temperature
        if step % 25 == 0:
            tau = max(tau * 0.985, tau_min)

        # Restless: occasionally increase temperature
        if step % 800 == 799:
            tau = min(tau * 2.5, c_max / (n * 4.0))

        # Aggressive restart if stuck
        if stall > 35:
            noise_scale = 0.04 + 0.02 * min(stall // 35, 5)
            a = normalize(best_a * np.exp(rng.normal(0, noise_scale, n)))
            alpha = 1.0
            tau = c_max / (n * 8.0)
            stall = 0

        step += 1

    return best_a


def spectral_perturb(a, rng):
    """Perturb via random smooth spectral filter."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = fft(a, size)
    # Create smooth random phase perturbation
    freqs = np.fft.fftfreq(size)
    smooth_freq = 1.0 + rng.uniform(2, 8)
    phase = rng.uniform(0, 2 * np.pi, size)
    # Low-pass smooth the phase
    mask = np.exp(-np.abs(freqs) * smooth_freq)
    phase_smooth = np.real(ifft(fft(phase) * mask, size))
    A_pert = A * np.exp(1j * phase_smooth * 0.1)
    a_new = np.real(ifft(A_pert, size))[:n]
    return normalize(a_new)


def block_perturb(a, rng):
    """Perturb by reversing a random block."""
    n = len(a)
    a_try = a.copy()
    if n < 10:
        return a_try
    block_len = int(rng.integers(n // 5, max(n // 2, n // 5 + 1)))
    start_idx = int(rng.integers(0, n - block_len))
    a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]
    return normalize(a_try)


def resample_to_n(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def solve_lp_polish(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    m = 2 * n - 1
    # Build constraint matrix: A_ub @ g <= b_ub
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None
```
