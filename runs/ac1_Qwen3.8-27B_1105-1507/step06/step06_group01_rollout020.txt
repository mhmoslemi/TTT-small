"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:

2. FISTA on log-sum-exp smooth approximation
3. Coordinate descent with affected-convolution tracking
4. Multi-start with symmetry and perturbation
Key innovation: adaptive constraint generation for LP (start with 1 constraint,
grow as needed), combined with periodic perturbation to escape local minima.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        # Find first index where cond is False
        falses = np.where(~cond)[0]
        if len(falses) == 0:
            rho = n - 1
        else:
            rho = int(falses[0])
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def softmax_weights(b, tau):
    """Compute softmax weights for log-sum-exp gradient."""
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    return exp_b / exp_b.sum()


def compute_smooth_grad(a, p):
    """Compute gradient of log-sum-exp smooth max at a, given softmax weights p.
    
    grad[i] = 2 * sum_k p[k] * a[k-i] (where a[m]=0 outside [0,n-1])
    This is 2 * (p circ a)[i] where circ denotes circular correlation,
    but we want linear correlation. We can compute it with FFT.
    """
    n = len(a)
    conv_len = 2 * n - 1
    # Pad p and a for linear cross-correlation
    size = 1
    while size < conv_len:
        size <<= 1
    # We need: grad[i] = 2 * sum_k p[k] * a[k-i] for k-i in [0,n-1]
    # This is the cross-correlation of p and a
    # corr[i] = sum_k p[k] * a[k-i] = (p * flip(a))[i] in convolution terms
    # Actually: sum_k p[k]*a[k-i] = sum_j p[j+i]*a[j] (substituting j=k-i)
    # = (reverse(p) conv a)[i + n - 1] ... hmm let me just do it directly with FFT
    
    # Alternative: use the fact that if we pad p to length 2n-1 and a to length 2n-1,
    # the cross-correlation can be computed as IFFT(FFT(p) * conj(FFT(a)))
    # But we need the linear version.
    
    # Simplest: for each i, grad[i] = 2 * dot(p[i:i+n], a)
    # This is O(n^2). For n=768, that's 590K ops which is fine.
    
    # FFT approach: pad a to length size, compute FFT, then for each i:
    # grad[i] = 2 * IFFT(FFT(p_padded) * conj(FFT(a_padded)))[i + offset]
    # Let me just use the direct method for clarity and correctness
    
    grad = np.zeros(n)
    for i in range(n):
        # sum_{j=0}^{n-1} p[i+j] * a[j]
        # p has length 2n-1, so p[i:i+n] is valid for i=0,...,n-1
        grad[i] = 2.0 * np.dot(p[i:i + n], a)
    return grad


def run_fista(a_init, deadline, tau_schedule=None, max_iters=10000, seed=0):
    """FISTA on log-sum-exp smooth approximation with projected gradient on simplex."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if tau_schedule is None:
        tau_schedule = [50, 100, 200, 500, 1000]
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break
        
        y = a.copy()
        a_prev = a.copy()
        t = 1.0
        lr = 0.8 / (1.0 + 0.1 * tau)
        
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            
            b = fast_conv(y)
            p = softmax_weights(b, tau)
            grad_y = compute_smooth_grad(y, p)
            
            # Project gradient to simplex tangent
            grad_tan = grad_y - grad_y.mean()
            
            # Step
            step = lr / (1.0 + 0.01 * it)
            a_new = y - step * grad_tan
            a_new = proj_simplex(a_new)
            
            # Nesterov momentum
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            alpha = (t - 1.0) / t_new
            y_new = a_new + alpha * (a_new - a_prev)
            y_new = np.maximum(y_new, 0.0)
            y_sum = y_new.sum()
            if y_sum > 0:
                y_new = y_new / y_sum
            else:
                y_new = a_new
            
            a_prev = a
            a = a_new
            y = y_new
            t = t_new
            
            if it % 20 == 0:
                M_cur = fast_conv(a).max()
                if M_cur < best_M:
                    best_M = M_cur
                    best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def lp_constraint_generation(a, M, max_constraints=100, deadline_lp=None):
    """LP best response with constraint generation (cutting plane).
    
    Start with the single worst constraint, add violated constraints iteratively.
    """
    n = len(a)
    b = fast_conv(a)
    conv_len = 2 * n - 1
    
    # Active set: start with argmax
    active = [int(np.argmax(b))]
    
    for cg_iter in range(max_constraints):
        if deadline_lp is not None and time.monotonic() > deadline_lp:
            break
        
        # Build constraint matrix for active set
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active):
            # (a * g)[k] = sum_j a[k-j] * g[j] for j where 0 <= k-j < n
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                m = k - j
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
        
        if len(rows) == 0:
            return np.ones(n) / n
            
        A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
        c_obj = -np.ones(n)  # maximize sum(g) = minimize -sum(g)
        b_ub = np.full(len(active), M)
        bounds = [(0.0, None)] * n
        
        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if not r.success or r.x is None:
                # Fallback: return uniform on "good" coordinates
                b_check = fast_conv(a)
                good_coords = np.where(b_check < M * 0.8)[0]
                if len(good_coords) > 0:
                    g = np.zeros(n)
                    g[good_coords] = 1.0 / len(good_coords)
                    return g
                return np.ones(n) / n
        except Exception:
            return np.ones(n) / n
        
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            return np.ones(n) / n
        g = g / g_sum
        
        # Check for violated constraints
        # Compute conv(a, g) and find violations
        conv_ag = fast_conv(a * g)  # Wait, this isn't right!
        # Actually, we need (a * g)[k] = sum_j a[j] * g[k-j]
        # This is the cross-convolution, not a*g pointwise!
        
        # Let me compute it properly:
        # (a * g)[k] = sum_j a[j] * g[k-j]
        # This is the standard linear convolution of a and g
        ag = np.convolve(a, g)  # This gives (a*g)[k] = sum_j a[j]*g[k-j]
        
        violated = np.where(ag > M + 1e-10)[0]
        if len(violated) == 0:
            break
        
        # Add the most violated constraint
        worst = int(violated[np.argmax(ag[violated])])
        if worst not in active:
            active.append(worst)
    
    return g


def run_lp_optimization(a_init, deadline, max_rounds=500):
    """LP-based optimization with constraint generation and line search."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:

        if time.monotonic() > deadline:
            break
        
        # Get best response direction
        g = lp_constraint_generation(a, best_M, max_constraints=50,
                                     deadline_lp=min(deadline, time.monotonic() + 15))
        
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Line search: find best t in (0, 1]
        improved = False
        for t in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-12):
                a = a_trial.copy()
                best_M = M_trial

    return best_a

            #       (where a[k-i] is the "other" factor)
            
            # Unaffected max
            if k_lo > 0 or k_hi < conv_len - 1:
                mask = np.ones(conv_len, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            
            # Build the set of linear/quadratic terms
            # For each affected k, the coefficient of x is 2*a[k-i] (or 1 if k==2i, it's x^2)
            slopes = []
            intercepts = []
            has_quad = False
            quad_idx = -1
            
            for k in range(k_lo, k_hi + 1):
                other = k - i
                if other == i:  # k = 2*i, quadratic term
                    # b_new[2i] = b[2i] - x_cur^2 + x^2
                    intercepts.append(b[k] - x_cur * x_cur)
                    slopes.append(0.0)
                    has_quad = True
                    quad_idx = len(slopes) - 1
                elif 0 <= other < n:
                    # b_new[k] = b[k] - 2*x_cur*a[other] + 2*x*a[other]
                    intercepts.append(b[k] - 2.0 * x_cur * a[other])
                    slopes.append(2.0 * a[other])
                else:
                    # This shouldn't happen for valid k
                    intercepts.append(b[k])
                    slopes.append(0.0)
            
            c_arr = np.array(intercepts)
            s_arr = np.array(slopes)
            
            def obj_fn(x_test):
                vals = c_arr + s_arr * x_test
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if has_quad:
                    qv = c_arr[quad_idx] + x_test * x_test
                    if qv > m:
                        m = qv
                return m
            
            # Golden section search for minimum
            lo = 0.0
            hi = max(3.0 * x_cur, 1.0)
            
            # Ensure hi is large enough
            for _ in range(10):
                if obj_fn(hi) > obj_fn(hi * 0.5) and hi < 100.0:
                    hi *= 2.0
                else:
                    break
            
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_fn(a_pt)
            fb = obj_fn(b_pt)
            
            for _ in range(30):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_fn(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_fn(b_pt)
            
            x_opt = (lo + hi) / 2.0
            if obj_fn(0.0) <= obj_fn(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - x_cur) > 1e-12:
                # Update b
                for k in range(k_lo, k_hi + 1):
                    other = k - i
                    if other == i:
                        b[k] = b[k] - x_cur * x_cur + x_opt * x_opt
                    elif 0 <= other < n:
                        b[k] = b[k] - 2.0 * x_cur * a[other] + 2.0 * x_opt * a[other]
                a[i] = x_opt
                sweep_improved = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-10:
            break
        # Don't renormalize during CD (it changes the problem), just track
        # Actually, J is scale-invariant, so we can normalize
        a = a / s
        # Rebuild b after normalization
        b = fast_conv(a)
        
        M_cur = b.max()
        if M_cur < best_M * (1 - 1e-14):
            best_M = M_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 5
        if no_improve > 30:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=True)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed parent
        for sigma in [0.02, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window (Hann-like)
    hann = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    hann = np.maximum(hann, 0.01)
    inits.append(hann / hann.sum())
    
    # Gaussian bump
    gauss = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Two bumps
    two_bump = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2) + 0.02
    inits.append(two_bump / two_bump.sum())
    
    # Power law
    power = t**0.5 + (1-t)**0.5 + 0.1
    inits.append(power / power.sum())
    
    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # "Ruler" pattern: high at powers of 2 positions
    ruler = np.ones(n) * 0.3
    pos = 1
    while pos < n:
        ruler[pos] = 1.0
        pos *= 2
    inits.append(ruler / ruler.sum())
    
    return inits


def run_mirror_descent(a_init, deadline, max_iters=10000, step_init=0.3):
    """Multiplicative weights / mirror descent."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Subgradient
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        for j in range(j_lo, j_hi + 1):
            grad[j] = 2.0 * a[k_star - j]
        
        grad_tan = grad - grad.mean()
        
        step = step_init / (1.0 + 0.01 * it)
        log_w = -step * grad_tan
        log_w -= log_w.max()
        a = a * np.exp(log_w)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        M_cur = fast_conv(a).max()
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()
    
    return best_a


def run_perturbation_restart(a, deadline, rng, max_restarts=5):
    """Try perturbed versions and keep the best."""
    best_a = a.copy()
    best_M = fast_conv(a).max()
    n = len(a)
    
    for r in range(max_restarts):
        if time.monotonic() > deadline:
            break
        
        # Perturb
        sigma = 0.05 * (1.0 + r * 0.5)
        a_pert = np.maximum(a + rng.normal(0, sigma, n), 1e-8)
        a_pert = a_pert / a_pert.sum()
        
        # Quick FISTA
        a_opt = run_fista(a_pert, min(deadline, time.monotonic() + 30),
                          tau_schedule=[200, 500], max_iters=2000, seed=1000 + r)
        M_opt = fast_conv(a_opt).max()
        
        if M_opt < best_M:
            best_M = M_opt
            best_a = a_opt.copy()
            print(f"    Perturbation restart {r}: M={M_opt:.10f}", flush=True)
    
    return best_a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a, label=""):
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)
