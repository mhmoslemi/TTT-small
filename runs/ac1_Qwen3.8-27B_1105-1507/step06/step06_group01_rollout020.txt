from scipy.sparse import csr_matrix
import time

# --- Top-level helper functions ---

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

def compute_M(a):

def compute_J_local(a):

    if s < 0.01:
        return float('inf')
    n = len(a)
    return 2.0 * n * compute_M(a) / (s * s)

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)

    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g_normalized, opt_value) or (None, None) on failure.
    """
    n = len(a)

            opt_val = -r.fun  # since we negate the objective
            if opt_val > 1e-10:
                return g / g.sum(), opt_val
            else:
                return g / max(g.sum(), 1e-15), opt_val
    except Exception as e:
        pass
    return None, None

def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and gradient wrt a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    # grad[i] = 2 * sum_j p[i+j] * a[j]
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    P = p[idx]
    grad = 2.0 * (P.dot(a))
    return smooth_val, grad

def run_adam_smooth(a, deadline, tau_start=50, tau_end=500, lr=0.01):
    """Adam optimization on smooth max approximation."""
    a = a.copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    total_iters = 5000
    
    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / total_iters)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad = smooth_max_grad(a, tau)
        grad_tan = grad - grad.mean()
        
        m = beta1 * m + (1 - beta1) * grad_tan
        v = beta2 * v + (1 - beta2) * grad_tan ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)
        
        if it % 10 == 0:
            M = compute_M(a)
            if M < best_M:
                best_M = M
                best_a = a.copy()
    
    M = compute_M(a)
    if M < best_M:
        best_a = a.copy()
    
    return best_a

def run_lp_optimize(a, deadline, max_iters=500):
    """LP-based optimization with mixing."""
    a = a.copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g, opt_val = solve_lp_best_response(a, best_M)
        if g is None:
            break
        
        # Gap
        gap = opt_val  # opt_val = max sum(g) with (a*g) <= M
        if gap <= 1.0 + 1e-6:
            # At optimum (or very close)
            no_improve += 1
            if no_improve > 5:
                break
            continue
        
        # Mix
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 0.01:
                continue
            a_trial = a_trial / s
            M_trial = compute_M(a_trial)
            if M_trial < best_M:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
    
    return best_a

def run_coord_descent(a, deadline, max_sweeps=100):
    """Coordinate descent with exact 1D optimization."""
    a = a.copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        order = np.random.permutation(n)
        sweep_changed = False
        
        for i in order:
            # Optimize a[i] given others
            # (a*a)[k] for affected k: k in [i, i+n-1] ∩ [0, 2n-2]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # For each affected k, (a*a)[k] = base_k + s_k * x + q_k * x^2
            # where x = a[i], base_k is the contribution from j ≠ i
            c_arr = []
            s_arr = []
            q_arr = []
            
            for k in range(k_lo, k_hi + 1):
                j = k - i
                if 0 <= j < n:
                    if j == i:
                        # a[i] * a[i] term
                        c_arr.append(b[k] - a[i] * a[i])
                        s_arr.append(0.0)
                        q_arr.append(1.0)
                    else:
                        # a[i] * a[j] + a[j] * a[i] = 2 * a[j] * a[i]
                        # But we need to be careful: (a*a)[k] = sum_j a_j * a_{k-j}
                        # The term involving a[i] is: a[i] * a[k-i] + a[k-i] * a[i] (if k-i ≠ i)
                        # = 2 * a[k-i] * a[i]
                        # If k-i = i (i.e., k = 2i): a[i] * a[i] = a[i]²
                        # Already handled above
                        c_arr.append(b[k] - 2.0 * a[j] * a[i])
                        s_arr.append(2.0 * a[j])
                        q_arr.append(0.0)
                else:
                    c_arr.append(b[k])
                    s_arr.append(0.0)
                    q_arr.append(0.0)
            
            c_arr = np.array(c_arr)
            s_arr = np.array(s_arr)
            q_arr = np.array(q_arr)
            
            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            
            # Golden section search for x that minimizes max(base_max, max(c + s*x + q*x²))
            def obj(x_val):
                vals = c_arr + s_arr * x_val + q_arr * x_val * x_val
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                return m
            
            # Search range
            x_lo = 0.0
            x_hi = max(2.0 * a[i], 0.1)
            for _ in range(5):
                if obj(x_hi) > obj(x_hi * 0.5):
                    x_hi *= 0.5
                else:
                    break
            x_hi = max(x_hi, 0.01)
            
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = x_hi - gr * (x_hi - x_lo)
            b_pt = x_lo + gr * (x_hi - x_lo)
            fa = obj(a_pt)
            fb = obj(b_pt)
            
            for _ in range(30):
                if fa < fb:
                    x_hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = x_hi - gr * (x_hi - x_lo)
                    fa = obj(a_pt)
                else:
                    x_lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = x_lo + gr * (x_hi - x_lo)
                    fb = obj(b_pt)
            
            x_opt = (x_lo + x_hi) / 2.0
            if obj(0.0) <= obj(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - a[i]) > 1e-10 * max(1.0, a[i]):
                a[i] = x_opt
                sweep_changed = True
        
        if sweep_changed:
            a = a / a.sum()
            M = compute_M(a)
            if M < best_M * (1 - 1e-12):
                best_M = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 3
        
        if no_improve > 15:
            break
    
    return best_a

def resample(a, new_n):
    """Resample sequence to new length."""
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 0.0)

def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample(parent, n)
        s = p.sum()
        if s > 0:
            inits.append(p / s)
        for noise_level in [0.02, 0.05, 0.1, 0.2]:
            p_noisy = np.maximum(p + rng.normal(0, noise_level, n), 0)
            s = p_noisy.sum()
            if s > 0:
                inits.append(p_noisy / s)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    cosw = 0.5 + 0.5 * np.cos(np.pi * t)
    cosw = np.maximum(cosw, 0.01)
    inits.append(cosw / cosw.sum())
    
    # Random smooth
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Power law
    pw = (t ** 0.5) + 0.1
    inits.append(pw / pw.sum())
    
    # Two peaks
    peak = np.exp(-10*(t-0.3)**2) + np.exp(-10*(t-0.7)**2) + 0.05
    inits.append(peak / peak.sum())
    
    return inits
