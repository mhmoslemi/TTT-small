"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho = np.max(np.where(w * np.arange(1, n + 1) > (cumsum - 1))[0])
    theta = (cumsum[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_toeplitz_conv(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]. Shape: (2n-1, n)."""
    n = len(a)

    for j in range(n):

        # conv[k] gets a[k-j] * g[j]
        a_slice = a[:n]
        A[j:j+n, j] = a_slice
    return A

        conv = compute_conv(a)

        # Find optimal step size by trying several values

def fourier_flatten(a, max_iters=40, time_budget=30.0):
    """Flatten the power spectrum to reduce time-domain peaks."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_mean = float(np.mean(P))
        p_max = float(np.max(P))
        
        if p_max < 1e-30:
            break

        # Compress the power spectrum toward uniform
        # Use different compression strategies
        if it % 3 == 0:
            # Power compression
            alpha = 0.5 + 0.3 * (it / max_iters)
            P_new = (P / p_max) ** alpha * p_max
        elif it % 3 == 1:
            # Log compression
            P_log = np.log1p(P)
            P_log_max = float(np.max(P_log))
            P_new = (P_log / P_log_max) * p_max
        else:
            # Soft clipping
            threshold = p_mean * 3.0
            P_new = np.where(P > threshold, threshold + (P - threshold) * 0.3, P)

        # Reconstruct with modified magnitude, preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        # Project onto simplex
        a_new = project_simplex(a_new)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            a = normalize(0.6 * a + 0.4 * a_new)

    return best_a


def subgradient_momentum(a, max_steps=200, time_budget=30.0):
    """Projected subgradient with Nesterov momentum and adaptive step size."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Momentum
    v = np.zeros(n)
    mu = 0.9  # momentum coefficient
    step = 0.02
    tau = 0.05  # temperature for softmax weighting

    size = 1
    while size < 3 * n:
        size <<= 1

    prev_grad = None
    prev_step = None
    prev_a = None

    for step_i in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights over peaks
        shifted = np.clip((conv - cmax) / (tau * cmax + 1e-10), -50, 50)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient: grad[j] = 2 * sum_k weights[k] * a[k-j] (with bounds)
        # Using FFT: grad = 2 * ifft(rfft(weights) * rfft(a_flipped))
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Barzilai-Borwein step size
        if prev_grad is not None and prev_a is not None:
            y = grad - prev_grad
            s_vec = a - prev_a
            yy = float(np.dot(y, y))
            sy = float(np.dot(s_vec, y))
            if yy > 1e-15:
                step_bb = float(np.dot(s_vec, s_vec)) / sy if abs(sy) > 1e-15 else step
                step = 0.5 * step + 0.5 * np.clip(step_bb, 0.001, 0.1)

        # Nesterov momentum update
        v = mu * v + (1 - mu) * grad
        a_trial = a - step * v

        # Project onto simplex
        a_new = project_simplex(a_trial)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        prev_grad = grad.copy()
        prev_a = a.copy()
        a = a_new

        # Anneal temperature
        tau *= 0.98
        step = max(step * 0.995, 0.001)

    return best_a


def anneal_search(a, time_budget=30.0):
    """Simulated annealing with diverse perturbation strategies."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    T = 0.05  # initial temperature
    T_min = 0.001
    cooling = 0.97

    while time.monotonic() - start < time_budget:
        # Perturbation
        strategy = rng.integers(0, 4)
        if strategy == 0:
            # Gaussian
            a_try = a * np.exp(rng.normal(0, 0.1, n))
        elif strategy == 1:
            # Random swap of mass
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                a_try = a.copy()
                delta = 0.1 * max(a[i], a[j])
                a_try[i] = max(0, a[i] - delta)
                a_try[j] = a[j] + delta
            else:
                a_try = a.copy()
        elif strategy == 2:
            # Sinusoidal perturbation
            freq = rng.uniform(1, 10)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            a_try = a * mod
        else:
            # Block reversal
            a_try = a.copy()
            block_len = int(rng.integers(max(2, n // 10), n // 3))
            start_idx = int(rng.integers(0, n - block_len))
            a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]

        a_try = normalize(a_try)
        obj_try = compute_obj(a_try)

        # Acceptance criterion
        delta = obj_try - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-10)):
            a = a_try
            cur_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

        T = max(T * cooling, T_min)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def binary_sequence_optimize(n, time_budget=30.0):
    """Generate and optimize binary-like sequences (related to difference sets)."""
    start = time.monotonic()

    for attempt in range(5):

            break
        if best_a is not None and best_obj < 1.51:
            break

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    cond = w * np.arange(1, n + 1) > (cumsum - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cumsum[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_toeplitz_conv(a):
    """Build matrix A where (A @ g)[k] = convolve(a, g)[k]. Shape: (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_direction(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz_conv(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

        if remaining < 5:

def peak_shave(a, time_budget=20.0):
    """Directly reduce peak contributors and redistribute mass to safe positions."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(555)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find top contributors to the peak
        contrib = []
        for j in range(n):
            i = k_star - j
            if 0 <= i < n:
                contrib.append((a[j] * a[i], j, i))
        contrib.sort(reverse=True)

        if not contrib:
            break

        improved = False
        for peak_contrib, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            # Try moving mass from the larger contributor to a "safe" position
            # Find positions that don't contribute to top peaks
            top_k = set()
            for k in range(2 * n - 1):
                if conv[k] > 0.85 * peak_val:
                    for j in range(n):
                        i = k - j
                        if 0 <= i < n:
                            top_k.add(j)
                            top_k.add(i)

            # Find safe target positions
            safe_positions = [p for p in range(n) if p not in top_k and a[p] < np.mean(a)]
            if not safe_positions:
                safe_positions = [p for p in range(n) if p != j1 and p != j2]
            if not safe_positions:
                continue

            j_target = safe_positions[int(rng.integers(0, len(safe_positions)))]

            for frac in [0.02, 0.05, 0.1, 0.15]:
                delta = frac * a[j1]
                a_try = a.copy()
                a_try[j1] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            # Occasional global perturbation
            if rng.random() < 0.15:
                a = normalize(a * np.exp(rng.normal(0, 0.04, n)))
            else:
                break

    return best_a


def subgradient_momentum(a, max_steps=300, time_budget=40.0):
    """Projected subgradient with Nesterov momentum, BB step, and softmax weighting."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.85
    step = 0.015
    tau = 0.03

    size = 1
    while size < 3 * n:
        size <<= 1

    prev_grad = None
    prev_a = None
    stall_count = 0

    for step_i in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Adaptive softmax weights
        shifted = np.clip((conv - cmax) / (tau * cmax + 1e-10), -50, 50)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient via correlation
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Barzilai-Borwein step size
        if prev_grad is not None and prev_a is not None:
            y = grad - prev_grad
            s_vec = a - prev_a
            yy = float(np.dot(y, y))
            sy = float(np.dot(s_vec, y))
            if yy > 1e-15 and abs(sy) > 1e-15:
                step_bb = float(np.dot(s_vec, s_vec)) / sy
                step = 0.6 * step + 0.4 * np.clip(step_bb, 0.002, 0.08)

        # Nesterov momentum
        v = mu * v + (1.0 - mu) * grad
        a_trial = a - step * v
        a_new = project_simplex(a_trial)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 30:
                # Restart momentum and reduce step
                v = np.zeros(n)
                step *= 0.7
                tau *= 0.8
                stall_count = 0
                a = best_a.copy()

        prev_grad = grad.copy()
        prev_a = a.copy()
        a = a_new
        tau *= 0.995

    return best_a


def fourier_flatten(a, max_iters=60, time_budget=35.0):
    """Flatten power spectrum using multiple compression strategies."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        p_mean = float(np.mean(P[1:n])) if n > 1 else p_max

        # Cycle through compression strategies
        strategy = it % 4
        if strategy == 0:
            alpha = 0.4 + 0.3 * (it / max_iters)
            P_new = (P / p_max) ** alpha * p_max
        elif strategy == 1:
            # Soft clip at 3x mean
            thresh = max(p_mean * 2.5, p_max * 0.3)
            P_new = np.minimum(P, thresh + (P - thresh) * 0.2)
        elif strategy == 2:
            # Log compression
            P_log = np.log1p(P / p_max)
            P_max_log = float(np.max(P_log))
            P_new = (P_log / (P_max_log + 1e-10)) * p_max
        else:
            # Inverse: boost low-power frequencies
            P_norm = P / p_max
            P_new = (0.3 + 0.7 * P_norm) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = project_simplex(a_new)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.7 + 0.1 * (it / max_iters)
            a = normalize(blend * a + (1 - blend) * a_new)

    return best_a


def anneal_search(a, time_budget=35.0, seed=999):
    """Simulated annealing with diverse perturbation strategies."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    T = 0.04
    T_min = 0.0005
    cooling = 0.975

    while time.monotonic() - start < time_budget:
        strategy = rng.integers(0, 5)
        if strategy == 0:
            a_try = a * np.exp(rng.normal(0, 0.08, n))
        elif strategy == 1:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                a_try = a.copy()
                delta = 0.05 * max(a[i], a[j], 1.0 / n)
                a_try[i] = max(0, a[i] - delta)
                a_try[j] += delta
            else:
                a_try = a * np.exp(rng.normal(0, 0.05, n))
        elif strategy == 2:
            freq = rng.uniform(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.15 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            a_try = a * mod
        elif strategy == 3:
            a_try = a.copy()
            blen = int(rng.integers(max(2, n // 8), n // 3))
            s_idx = int(rng.integers(0, n - blen))
            a_try[s_idx:s_idx + blen] = a_try[s_idx:s_idx + blen][::-1]
        else:
            # Blend with a smooth random function
            t = np.linspace(0, 2 * np.pi, n)
            rand_func = 1.0 + 0.2 * np.sin(rng.uniform(1, 5) * t + rng.uniform(0, 2 * np.pi))
            a_try = a * rand_func

        a_try = normalize(a_try)
        obj_try = compute_obj(a_try)

        delta = obj_try - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-10)):
            a = a_try
            cur_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

        T = max(T * cooling, T_min)

    return best_a


def resample_sequence(a, new_n):
    """Resample via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

            if time.monotonic() > deadline - 100:
                break
            a_ann = anneal_search(a, time_budget=25.0, seed=ann_seed)
            obj_ann = compute_obj(a_ann)
            if obj_ann < best_score:
                best_score = obj_ann
                best_seq = a_ann.copy()
                a = a_ann
                print(f"  Anneal({ann_seed}): {best_score:.8f} *")
            else:
                print(f"  Anneal({ann_seed}): {obj_ann:.8f}")

    # =========================================================
    # Phase 6: Multi-n exploration
    # =========================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 6: Multi-n exploration")
    print(f"{'=' * 60}")

    n_best = len(best_seq) if best_seq is not None else 700
    n_options = sorted(set([
        n_best - 100, n_best - 75, n_best - 50, n_best - 25,
        n_best, n_best + 25, n_best + 50, n_best + 75,
        n_best + 100, n_best + 150, n_best + 200, n_best + 300,
        # Structural lengths
        200, 300, 400, 500, 600, 700, 800, 900, 1000, 1100, 1300
    ]))
    n_options = [n for n in n_options if 120 <= n <= 1400]

    for n_try in n_options:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break

        if best_seq is not None:
            a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try

        alloc = min(remaining * 0.08, 25.0)

        if n_try <= 800 and alloc > 10:
            a_try = lp_optimize(a_try, max_iters=10, time_budget=alloc * 0.4)
        a_try = subgradient_momentum(a_try, max_steps=60, time_budget=alloc * 0.5)
        a_try = peak_shave(a_try, time_budget=alloc * 0.2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 7: Final intensive polish
    # =========================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 7: Final intensive polish")
    print(f"{'=' * 60}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())

        for round_i in range(6):
            if time.monotonic() > deadline - 8:
                break
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            alloc = min(remaining * 0.18, 35.0)

            # Cycle through methods
            if round_i % 3 == 0:
                if len(a_final) <= 900 and alloc > 15:
                    a_pol = lp_optimize(a_final, max_iters=15, time_budget=alloc * 0.4)
                else:
                    a_pol = a_final.copy()
                a_pol = peak_shave(a_pol, time_budget=alloc * 0.3)
                a_pol = subgradient_momentum(a_pol, max_steps=80, time_budget=alloc * 0.3)
            elif round_i % 3 == 1:
                a_pol = fourier_flatten(a_final, max_iters=40, time_budget=alloc * 0.4)
                a_pol = subgradient_momentum(a_pol, max_steps=100, time_budget=alloc * 0.4)
                if len(a_pol) <= 900:
                    a_pol = lp_optimize(a_pol, max_iters=10, time_budget=alloc * 0.2)
            else:
                a_pol = subgradient_momentum(a_final, max_steps=150, time_budget=alloc * 0.3)
                a_pol = peak_shave(a_pol, time_budget=alloc * 0.25)
                if len(a_pol) <= 900 and alloc > 10:
                    a_pol = lp_optimize(a_pol, max_iters=8, time_budget=alloc * 0.25)

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score - 1e-14:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {round_i}: {best_score:.8f} *")
            else:
                a_final = a_pol
                print(f"  Polish {round_i}: {obj_pol:.8f}")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'=' * 60}")

    return result
