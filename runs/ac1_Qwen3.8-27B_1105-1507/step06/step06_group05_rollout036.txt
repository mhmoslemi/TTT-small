        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def pocs_flatten(a, max_iters=200, time_budget=60.0):
    """
    POCS-style algorithm to flatten the self-convolution.
    Alternates between:
    1. Frequency domain: compress |A[k]| to be more uniform
    2. Time domain: enforce non-negativity and normalization
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Forward FFT
        A = fft(a, size)
        mag = np.abs(A)
        
        # Target uniform magnitude
        target_mag = np.sum(mag) / size  # average magnitude
        
        # Compress magnitudes toward uniform
        # Use a mixing parameter that decreases over iterations
        mix = 0.3 + 0.5 * (1 - it / max_iters)  # start aggressive, ease off
        
        # New magnitude: blend current with target
        # Use square-root compression to reduce dynamic range
        mag_new = mag * (1 - mix) + target_mag * mix
        
        # Alternatively: power compression
        # mag_new = mag * (mag / max(mag))**(-0.1)  # mild compression
        
        # Reconstruct
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        
        # IFFT
        a_new = np.real(ifft(A_new, size))[:n]
        
        # Project: non-negativity + normalization
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def logsumexp_gradient_descent(a, max_steps=300, time_budget=60.0):
    """
    Gradient descent on log-sum-exp of convolution (smooth approximation of max).
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    lr = 0.1
    beta = 50.0  # temperature parameter for log-sum-exp
    
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv_fft(a)
        cmax = float(np.max(conv))
        
        # Log-sum-exp with adaptive temperature
        beta_local = cmax * beta / n  # scale with problem
        shifted = (conv - cmax) / (cmax / n + 1e-10)
        shifted = np.clip(shifted, -100, 0)
        weights = np.exp(beta_local * shifted / cmax)
        # Actually, let me use a simpler formulation
        # LSE = cmax + (1/beta) * log(sum(exp(beta*(conv - cmax))))
        # Gradient of LSE w.r.t. a[i] = sum_k w_k * (conv contribution)
        
        # Simpler: use softmax weights
        tau = cmax / (5 + step * 0.1)  # cooling temperature
        shifted_conv = (conv - cmax) / tau
        shifted_conv = np.clip(shifted_conv, -50, 0)
        w = np.exp(shifted_conv)
        w = w / np.sum(w)
        
        # Gradient: d/d(a[i]) of sum_k w_k * conv[k]
        # conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (when valid)
        # So gradient[i] = sum_k w_k * (a[k-i] + a[i-k]) for valid indices
        # This is a weighted cross-correlation
        
        # Compute via FFT for efficiency
        a_padded = np.zeros(size)
        a_padded[:n] = a
        a_flip = np.zeros(size)
        a_flip[:n] = a[::-1]
        
        w_padded = np.zeros(size)
        w_padded[:len(w)] = w
        
        # corr[i] = sum_k w[k] * a[k-i] (circular correlation)
        Wf = fft(w_padded)
        Af = fft(a_padded)
        # We want: grad[i] = sum_k w[k] * (a_flip contribution)
        # Actually, let's think of it as:
        # grad[i] = sum_k w[k] * d(conv[k])/d(a[i])
        # d(conv[k])/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # = a[k-i] + a[-(k-i)] (with the convention that a[j]=0 for j outside [0,n-1])
        # Let d = k-i, then sum_k w[k] * (a[k-i] + a[i-k])
        # = sum_d w[d+i] * a[d] + sum_d w[d+i] * a[-d]
        # = (w shifted by i) * a + (w shifted by i) * a_flipped
        # This is a cross-correlation computation
        
        # Simpler approach: grad = conv(w, a) + conv(w, a[::-1]) evaluated appropriately
        # Let me use the correlation formula
        # grad[i] = sum_k w[k] * (a[k-i] if 0<=k-i<n else 0) + sum_k w[k] * (a[i-k] if 0<=i-k<n else 0)
        # First term: correlation of w and a, evaluated at i
        # Second term: correlation of w and a (reversed), evaluated at i
        
        # Using FFT: correlation(w, a)[i] = ifft(fft(w) * conj(fft(a)))... 
        # Actually for the convolution formulation:
        # (w * a)[i] = sum_k w[k] * a[i-k]
        # That's not quite what we want either.
        
        # Let me just do it directly for clarity (n is at most ~1500)
        # Actually for n up to 1500, direct is O(n^2) which might be too slow
        # Let me use the FFT approach properly.
        
        # d(conv[k])/d(a[i]) where conv[k] = sum_j a[j]*a[k-j]
        # = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
        # = a[k-i] + a[k-i] ... no
        # = a[k-i] (from first term, when j=i) + a[k-i] (from second term, when j=k-i)
        # Wait: d/d(a[i]) sum_j a[j]*a[k-j]
        # = sum_j [delta(j-i)*a[k-j] + a[j]*delta(k-j-i)]
        # = a[k-i] + a[k-i]
        # Hmm, that gives 2*a[k-i]? No:
        # = a[k-i] (when j=i in first sum) + a[k-i] (when k-j=i, i.e., j=k-i, in second sum)
        # But these are the same term! So it's a[k-i] + a[k-i] only if i != k-i, 
        # i.e., when i != k/2.
        
        # Actually, let me be more careful:
        # conv[k] = sum_{j=0}^{n-1} a[j] * a[k-j] (where a[m]=0 if m<0 or m>=n)
        # d(conv[k])/d(a[i]) = a[k-i] + a[k-i] if we differentiate both factors
        # Wait no. conv[k] = sum_j a[j] * a[k-j]
        # d/d(a[i]): only terms where j=i or k-j=i contribute
        # j=i: a[i] * a[k-i] -> derivative is a[k-i]
        # k-j=i, i.e., j=k-i: a[k-i] * a[i] -> derivative is a[i]
        # But if i = k-i (i.e., k=2i), then these are the same term, and the derivative is 2*a[i]
        # Otherwise, the derivative is a[k-i] + a[k-i]...

        # So in all cases: d(conv[k])/d(a[i]) = 2*a[k-i] (with a[m]=0 for invalid m)

        # conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
        # d(conv[k])/d(a[i]) = sum of contributions:
        #   - If j=i is valid (0<=i<n): term is a[i]*a[k-i], derivative = a[k-i] (if k-i valid)
        #   - If j=k-i is valid (0<=k-i<n): term is a[k-i]*a[i], derivative = a[i]... 
        #     wait no, derivative of a[k-i]*a[i] w.r.t. a[i] is a[k-i]
        # So: d(conv[k])/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n)
        # = 2*a[k-i] (if 0<=k-i<n), else 0 or a[k-i] (if only one is valid)
        
        # Hmm, but when i = k-i, i.e., k = 2i, the term a[i]*a[i] appears only once in the sum (j=i=k-i), so its derivative is 2*a[i].
        # When i != k-i, the terms a[i]*a[k-i] (j=i) and a[k-i]*a[i] (j=k-i) are separate, each contributing a[k-i] to the derivative.
        # So total is 2*a[k-i].
        # In both cases: d(conv[k])/d(a[i]) = 2*a[k-i] (if 0<=k-i<n), else 0.
        
        # Wait, that's only if both i and k-i are in [0, n-1]. If only one is in range:
        # - If i in [0,n-1] but k-i not in [0,n-1]: only the j=i term exists, derivative = 0 (since a[k-i]=0)
        #   Actually no: the j=i term is a[i]*a[k-i], but if k-i is not valid, this term doesn't exist in the sum.
        #   So derivative = 0.
        # - If k-i in [0,n-1] but i not in [0,n-1]: impossible since i is always in [0,n-1] (we're differentiating w.r.t. a[i] for valid i).
        
        # So: d(conv[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n, else 0.
        
        # Wait, I think I'm overcomplicating this. Let me just verify with a small example.
        # a = [x, y, z], conv[1] = x*y + y*x = 2xy
        # d(conv[1])/d(x) = 2y = 2*a[1] where k-i = 1-0 = 1. Yes!
        # conv[2] = x*z + y*y + z*x = xz + y^2 + zx
        # d(conv[2])/d(y) = 2y = 2*a[0] where k-i = 2-1 = 1... wait that's a[1] = y. Yes, 2*y.
        # d(conv[2])/d(x) = z + 0 + z = 2z = 2*a[2] where k-i = 2-0 = 2. Yes!
        
        # Great, so grad[i] = sum_k w[k] * 2*a[k-i] (where a[m]=0 for m outside [0,n-1])
        # = 2 * (w * a_flipped)[i] where a_flipped[m] = a[n-1-m]... 
        # Actually: sum_k w[k] * a[k-i] is the cross-correlation of w and a at lag i.
        # In convolution terms: (w * a_rev)[i + offset] where a_rev is reversed...
        
        # Let me use a different formulation. Define b[m] = a[m] for 0<=m<n, 0 otherwise.
        # Then grad[i]/2 = sum_k w[k] * b[k-i]
        # Let d = k-i, so k = d+i:
        # = sum_d w[d+i] * b[d]
        # This is the cross-correlation of w and b at lag i.
        # In FFT terms: ifft(fft(w) * conj(fft(b)))... no, that's for circular correlation.

        # Let me just use the direct approach since n <= 1500 and the conv has 2n-1 <= 2999 elements.

        # = 2 * sum_{m: 0<=m<n} w[i+m] * a[m] (where m = k-i)
        
        # This is O(n * (2n-1)) = O(n^2) per gradient step, which for n=700 is about 1M operations.
        # With 300 steps, that's 300M operations. Might be too slow.

        # r = ifft(fft(w_padded, N) * conj(fft(a_padded, N)))[:n]
        # where N >= n + (2n-1) - 1 = 3n-2

        # w has length 2n-1, a has length n.
        # r[i] = sum_m w[i+m] * a[m] for m=0,...,n-1
        # This is the cross-correlation of w and a.
        # In terms of convolution: r[i] = (w_rev * a)[i + (2n-2) - (n-1)]... this is getting complicated.
        
        # Let me just use the direct computation but vectorize it.
        # For each i, grad[i] = 2 * dot(w[i:i+n], a)
        # We can compute all at once using a Toeplitz-like matrix, or just loop.
        
        # For n up to 1500, a loop of 1500 iterations each doing a dot product of length up to 1500
        # is 1500*1500 = 2.25M operations per gradient step. With 300 steps, that's 675M.
        # In numpy, each dot product is fast, so this should be OK (maybe 1-2 seconds total).
        
        # Actually, let me use a smarter FFT approach.
        # r[i] = sum_m a[m] * w[i+m]
        # Let's define w_ext of length 3n (padded), a_ext of length 3n (padded).
        # Then the circular correlation (with appropriate padding) gives us what we want.
        
        # r[i] = sum_{m=0}^{n-1} a[m] * w[i+m]
        # If we reverse w: w_rev[j] = w[(2n-2)-j], then:
        # r[i] = sum_m a[m] * w_rev[(2n-2)-(i+m)] = sum_m a[m] * w_rev[2n-2-i-m]
        # = (a * w_rev)[2n-2-i] (linear convolution)
        
        # So r = reverse of the middle n elements of conv(a, w_rev)
        # conv(a, w_rev) has length n + (2n-1) - 1 = 3n-1
        # We want elements at indices 2n-2-i for i=0,...,n-1
        # = elements at indices n-1, n, ..., 2n-2
        # = conv(a, w_rev)[n-1:2n-1] reversed
        
        # Let me just implement this directly.
        
        # Actually, the simplest vectorized approach:
        # Stack w[i:i+n] for all i into a matrix W (n x n), then grad = 2 * W @ a
        # But W is a Hankel matrix, which can be applied via FFT.
        
        # For now, let me just do it in a loop for correctness, and optimize if needed.
        grad = np.zeros(n)
        for i in range(n):
            # grad[i] = 2 * sum_m a[m] * w[i+m] where 0<=i+m<2n-1
            m_max = min(n, 2 * n - 1 - i)
            if m_max > 0:
                grad[i] = 2.0 * np.dot(a[:m_max], w[i:i + m_max])
        
        # Center the gradient (remove mean to preserve sum)
        grad -= np.mean(grad)
        
        # Step
        a_new = a - lr * grad / (cmax + 1e-10)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.8
            if lr < 1e-10:
                break
        
        # Anneal temperature
        beta = max(5.0, beta * 0.98)
    
    return best_a

def build_toeplitz_conv_matrix(a):
    """Build matrix T such that (T @ g) = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Using for loop (n up to 1500, so m x n = 3000 x 1500 = 4.5M entries)
    T = np.zeros((m, n))
    for j in range(n):
        for k in range(j, j + n):
            if k < m:
                T[k, j] = a[k - j]
    return T

def build_toeplitz_fast(a):
    """Build Toeplitz matrix more efficiently using scipy."""
    n = len(a)
    m = 2 * n - 1
    # conv(a, g)[k] = sum_j a[j] * g[k-j]
    # = sum_j T[k,j] * g[j] where T[k,j] = a[j] if... 
    # Actually: conv(a,g)[k] = sum_j a[k-j] * g[j] (standard definition)
    # Wait: np.convolve(a, g)[k] = sum_j a[j] * g[k-j]
    # So T[k,j] where the j-th component of g: T[k,j] = a[k-j] if 0<=k-j<n
    # Equivalently, for fixed j, T[k,j] = a[k-j] for k=j,...,j+n-1 (and k < m)
    
    # Let me use scipy's toa
    from scipy.linalg import toeplitz
    # Actually, the matrix isn't exactly Toeplitz in the standard sense.
    # Let me just build it with a loop.
    T = np.zeros((m, n))
    for j in range(n):
        start = j
        end = min(j + n, m)
        length = end - start
        T[start:end, j] = a[:length]
    return T

def lp_refine(a, time_budget=40.0, max_iters=20):
    """LP-based refinement: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv_fft(a)
        M = float(np.max(conv))
        
        # Build constraint matrix
        T = build_toeplitz_fast(a)
        c = -np.ones(n)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=T, b_ub=np.full(2 * n - 1, M),
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10,
                                  'mip_rel_gap': 0.0})
            if res.status == 0:
                g = np.maximum(res.x, 0.0)
                s = float(np.sum(g))
                if s > 1e-10:
                    g = g / s
                    # Line search for blending
                    best_t = 0
                    best_blend_obj = best_obj
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                        blend = normalize((1 - t) * a + t * g)
                        obj_blend = compute_obj(blend)
                        if obj_blend < best_blend_obj:
                            best_blend_obj = obj_blend
                            best_t = t
                    if best_t > 0:
                        a = normalize((1 - best_t) * a + best_t * g)
                        if best_blend_obj < best_obj:
                            best_obj = best_blend_obj
                            best_a = a.copy()
        except Exception:
            pass
    
    return best_a

def multi_scale_pocs(a, time_budget=60.0):
    """
    Multi-scale POCS: start with coarse (downsampled) sequence, flatten, then upsample and refine.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Start with a downsampled version
    coarse_n = max(50, n // 8)
    # Downsample by averaging blocks
    block_size = n // coarse_n
    if block_size < 1:
        coarse_n = n
        block_size = 1
    
    coarse_a = np.zeros(coarse_n)
    for i in range(coarse_n):
        s = i * block_size
        e = min((i + 1) * block_size, n)
        coarse_a[i] = np.mean(a[s:e])
    coarse_a = normalize(coarse_a)
    
    # Flatten at coarse scale
    coarse_a = pocs_flatten(coarse_a, max_iters=100, time_budget=time_budget * 0.3)
    
    # Upsample
    x_coarse = np.linspace(0, 1, coarse_n)
    x_fine = np.linspace(0, 1, n)
    upsampled = np.interp(x_fine, x_coarse, coarse_a)
    upsampled = normalize(upsampled)
    
    # Blend with original
    blend = 0.5 * normalize(a) + 0.5 * upsampled
    blend = normalize(blend)
    
    # Refine at full scale
    refined = pocs_flatten(blend, max_iters=80, time_budget=time_budget * 0.4)
    obj_refined = compute_obj(refined)
    
    if obj_refined < best_obj:
        best_obj = obj_refined
        best_a = refined
    
    return best_a

def diverse_initializations(n, seed=42):
    """Generate diverse initial sequences of length n."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangle (concentrated in middle)
    x = np.arange(n)
    tri = 1.0 - np.abs(x - n / 2) / (n / 2)
    inits.append(tri / np.sum(tri))
    
    # Exponential decay from left
    exp_decay = np.exp(-np.arange(n) / (n / 4))
    inits.append(exp_decay / np.sum(exp_decay))
    
    # Exponential decay from right
    exp_decay_r = np.exp(-np.arange(n - 1, -1, -1) / (n / 4))
    inits.append(exp_decay_r / np.sum(exp_decay_r))
    
    # Sine wave
    sine = np.abs(np.sin(np.pi * np.arange(n) / n))
    inits.append(sine / np.sum(sine))
    
    # Two bumps
    bump1 = np.exp(-((np.arange(n) - n * 0.25) ** 2) / (2 * (n * 0.1) ** 2))
    bump2 = np.exp(-((np.arange(n) - n * 0.75) ** 2) / (2 * (n * 0.1) ** 2))
    two_bump = bump1 + bump2
    inits.append(two_bump / np.sum(two_bump))
    
    # Random smooth
    smooth = np.convolve(rng.random(n), np.ones(n // 10) / (n // 10), mode='same')
    smooth = np.maximum(smooth, 0)
    inits.append(smooth / np.sum(smooth))
    
    # Linear ramp up
    ramp = np.arange(1, n + 1, dtype=float)
    inits.append(ramp / np.sum(ramp))
    
    # Linear ramp down
    ramp_d = np.arange(n, 0, -1, dtype=float)
    inits.append(ramp_d / np.sum(ramp_d))
    
    # Wigner-like (quadratic)
    quad = 1.0 - (2 * np.arange(n) / n - 1) ** 2
    quad = np.maximum(quad, 0)
    inits.append(quad / np.sum(quad))
    
    return inits

def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Strategy:
    1. Try multiple n values with diverse initializations
    2. Use POCS flattening + gradient descent + LP refinement
    3. Multi-scale approach for global optimization
    4. Final intensive polishing
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent

        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s
