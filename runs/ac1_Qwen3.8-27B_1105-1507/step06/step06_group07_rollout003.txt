import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

# --- Core computation functions ---

def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_obj_grad(a, beta):
    """Smoothed objective and gradient using logsumexp smoothing."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    # Gradient via cross-correlation
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj

def adam_log_opt(a, max_iters=10000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995,
                 seed=0):
    """Adam in log-space with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_start
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        # Use a smoother beta schedule (cosine annealing)
        beta = beta_start + (beta_end - beta_start) * (1 - np.cos(np.pi * progress)) / 2
        
        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a = a / np.sum(a)
        
        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a
        
        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 200 == 0:
            x -= np.mean(x)
        
        if it % 100 == 0 or it == max_iters - 1:
            xc = x - np.max(x)
            a_check = np.exp(xc)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()
    
    return best_a

def peak_shave(a, max_iters=200, time_budget=30.0, top_k=5):
    """
    Direct peak shaving: at each step, compute the subgradient from the top-k
    peaks and take a projected gradient step.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    lr = 0.01
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Find top-k peak indices
        peak_indices = np.argsort(conv)[::-1][:top_k]
        
        # Compute subgradient: average of d(conv[k])/d(a) for k in peak_indices
        grad = np.zeros(n)
        for k in peak_indices:
            # d(conv[k])/d(a[i]) = a[i+k] (if i+k < n) + a[i-k] (if i-k >= 0)
            # Vectorized:
            if k == 0:
                grad += 2 * a
            else:
                # a[i+k] for i = 0..n-1-k
                grad[:n-k] += a[k:]
                # a[i-k] for i = k..n-1
                grad[k:] += a[:n-k]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Subgradient of max(conv)/(sum(a))^2:
        # d/d(a) [max(conv) / S^2] = (grad * S^2 - max_conv * 2*S * 1) / S^4
        # = (grad / S^2) - 2*max_conv / S^3
        # = (1/S^2) * (grad - 2*max_conv/S * ones)
        
        S = np.sum(a)
        grad_obj = (grad / (S * S)) - 2 * max_conv / (S * S * S) * np.ones(n)
        # Simplify: grad_obj = (grad - 2*max_conv/S * ones) / S^2
        # The direction of descent is -grad_obj
        
        # Take a step
        a_new = a - lr * grad_obj
        
        # Project to a >= 0 and normalize
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            # Keep learning rate
        else:
            # Reduce learning rate
            lr *= 0.95
            if lr < 1e-8:
                break
    
    return best_a

def pair_swap_refine(a, time_budget=30.0, seed=0):

        i_min = max(0, k_star - (n-1))

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2*n-1)
        A[k_start:k_end, j] = a[:k_end-k_start]
    return A

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def structured_init(n, kind='raised_cosine', seed=0):
    """Generate structured initial sequences."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    
    if kind == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2*np.pi*x/(n-1+1e-10)))
    elif kind == 'gaussian':
        center = (n-1)/2
        sigma = n/4
        a = np.exp(-0.5 * ((x-center)/sigma)**2)
    elif kind == 'triangular':
        center = (n-1)/2
        a = np.maximum(0, 1 - np.abs(x - center)/(n/2))
    elif kind == 'random':
        a = rng.random(n) + 0.1
    elif kind == 'fourier':
        # Sum of a few Fourier modes with random phases
        a = np.ones(n)
        for freq in [2, 3, 5, 7]:
            phase = rng.uniform(0, 2*np.pi)
            a += 0.3 * np.cos(2*np.pi*freq*x/n + phase)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-10)
    s = float(np.sum(a))
    return a / s
