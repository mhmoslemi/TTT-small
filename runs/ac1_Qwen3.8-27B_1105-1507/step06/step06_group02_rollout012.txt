    n = len(a)

            cols.append(j)

def build_lp_matrix(n):
    """Build the sparse structure for the LP. Returns index arrays for reconstruction."""
    # (a*g)(k) = sum_j a[k-j] * g[j] for valid j
    # For each k in [0, 2n-2], the valid j range is [max(0, k-n+1), min(n-1, k)]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

def make_lp_matrix(a, row_idx, col_idx):

    n = len(a)
    vals = a[row_idx - col_idx]  # a[k-j] where k=row_idx, j=col_idx
    return csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))

        pass
    return None

    invphi = 1 / phi

        return bt.max()

    t_opt = (lo + hi) / 2
    # Check against current best
    v_opt = obj(t_opt)

    return best_t

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def conv_ab(a, b):
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def compute_J(a):

def build_lp_indices(n):

def minifind_line_search(b_a, b_ag, b_g, max_evals=50):
    """Find t in [0,1] minimizing max_k [(1-t)^2 b_a(k) + 2t(1-t) b_ag(k) + t^2 b_g(k)]"""
    def obj(t):
        bt = (1.0 - t) ** 2 * b_a + 2.0 * t * (1.0 - t) * b_ag + t ** 2 * b_g
        return float(bt.max())

    # Coarse grid
    grid = np.linspace(0.0, 1.0, 21)
    vals = [obj(t) for t in grid]
    best_t = grid[np.argmin(vals)]
    best_val = min(vals)

    # Golden section refinement
    invphi = (np.sqrt(5) - 1) / 2
    lo = max(0.0, best_t - 0.05)
    hi = min(1.0, best_t + 0.05)
    if hi - lo < 1e-10:
        return best_t, best_val

    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = obj(c)
    fd = obj(d)
    evals = 2

    for _ in range(max_evals):

            d = c
            fd = fc

            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = obj(d)
        evals += 1

    t_opt = (lo + hi) / 2.0
    v_opt = obj(t_opt)
    if v_opt < best_val:
        return t_opt, v_opt
    return best_t, best_val

def coordinate_descent_polish(a_init, deadline, max_rounds=200):
    """Try adjusting individual elements to reduce the peak autocorrelation."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_peak = int(np.argmax(b))
        improved = False

        # Try perturbing elements that contribute most to the peak
        # (a*a)(k_peak) = sum_j a[j]*a[k_peak-j]
        j_min = max(0, k_peak - n + 1)
        j_max = min(n - 1, k_peak)

        # Get the elements contributing to the peak
        contributors = []
        for j in range(j_min, j_max + 1):
            pair_val = a[j] * a[k_peak - j]
            contributors.append((pair_val, j, k_peak - j))
        contributors.sort(reverse=True)

        # Try adjusting top contributors
        tried = set()
        for pair_val, j1, j2 in contributors[:min(20, len(contributors))]:
            if j1 in tried or j2 in tried:
                continue
            tried.add(j1)
            tried.add(j2)

            # Try reducing the larger of the two
            if a[j1] >= a[j2]:
                j_try = j1
            else:
                j_try = j2

            # Binary search for best reduction
            old_val = a[j_try]
            lo_frac = 0.0
            hi_frac = 0.5  # reduce by at most 50%
            a[j_try] = old_val * (1.0 - hi_frac)
            b_test = fast_conv(a)
            m_hi = float(b_test.max())

            a[j_try] = old_val
            for _ in range(8):
                mid_frac = (lo_frac + hi_frac) / 2.0
                a[j_try] = old_val * (1.0 - mid_frac)
                b_test = fast_conv(a)
                m_mid = float(b_test.max())
                if m_mid < m_hi:
                    lo_frac = mid_frac
                else:
                    hi_frac = mid_frac

            a[j_try] = old_val * (1.0 - lo_frac)
            b_new = fast_conv(a)
            m_new = float(b_new.max())

            if m_new < best_M - 1e-14:
                best_M = m_new
                best_a = a.copy()
                improved = True
            else:
                a[j_try] = old_val

            if not improved:
                # Also try increasing a non-contributor to spread mass
                pass

        if not improved:
            break

        a = np.maximum(a, 1e-10)
        a = a / a.sum()

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.005))
        # Anti-symmetric perturbation (helps reduce middle autocorrelation)
        pert = rng.normal(0, 0.02, n)
        pert = pert - pert.mean()
        inits.append(np.maximum(p + pert, 0.005))

    # Uniform
    inits.append(np.ones(n) / n)

    # Various shaped initializations
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        shape = (t ** alpha + (1 - t) ** alpha) / 2.0
        inits.append(0.1 + 0.9 * shape / shape.max())

    # Oscillatory
    for f in [1, 2, 3, 4, 7, 11]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t + rng.uniform(0, 2 * np.pi)), 0.01))

    # Gaussian bumps
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        inits.append(np.exp(-5 * (t - center) ** 2) + 0.05)

    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.2, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)

    # Smooth random
    k = max(3, n // 16)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same'), 0.01))

    # Anti-correlated pattern (high-low alternating with smoothing)
    pattern = np.ones(n)
    pattern[::2] = 1.0
    pattern[1::2] = 0.3
    k2 = max(3, n // 32)
    kern2 = np.ones(k2) / k2
    inits.append(np.convolve(pattern, kern2, 'same'))

    # Triangular wave
    inits.append(np.abs(2 * (t - 0.5) - 1) + 0.1)

            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    n_parent = parent.size
    print(f"Starting optimization. Parent n={n_parent}", flush=True)

    # Determine candidate lengths (focus around parent length)
    candidate_lengths = sorted(set([
        max(128, n_parent - 64),
        n_parent,
        min(2048, n_parent + 64),
        min(2048, n_parent + 128),
    ]))
    # Also add some fixed good lengths
    for nl in [512, 768, 1024, 1280, 1536]:
        if 128 <= nl <= 2048:
            candidate_lengths.append(nl)
    candidate_lengths = sorted(set(candidate_lengths))
    print(f"Lengths: {candidate_lengths}", flush=True)

    # === PHASE 1: FW with miniFind on primary length ===
    p1_start = time.monotonic()
    p1_budget = min(600.0, (deadline - p1_start) * 0.4)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: FW miniFind (budget={p1_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_primary = n_parent if n_parent in candidate_lengths else min(candidate_lengths, key=lambda x: abs(x - n_parent))
    time_per_init = p1_budget / 12  # try up to 12 initializations

    inits = make_initializations(n_primary, rng, best.seq if best.seq is not None else parent)

    best_n_primary = None
    best_M_primary = float('inf')

    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_start + p1_budget:
            break
        a0 = np.maximum(a0.astype(np.float64), 1e-6)
        a0 = a0 / a0.sum()

        # Allocate decreasing time to later inits
        remaining = (p1_start + p1_budget) - time.monotonic()
        allot = remaining / (len(inits) - idx)
        sub_deadline = time.monotonic() + max(30.0, min(allot, 120.0))

        print(f"  Init {idx+1}/{len(inits)}: M0={fast_conv(a0).max():.6f}", flush=True)
        a = run_fw_minifind(a0, sub_deadline, max_iters=5000, use_accel=False,
                            verbose=(idx == 0))
        M = float(fast_conv(a).max())
        if M < best_M_primary:
            best_M_primary = M
            best_n_primary = a.copy()
            J = compute_J(a)
            print(f"    -> M={M:.8f} J={J:.6f}", flush=True)
            best.update(a)

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Accelerated FW + more iterations on best ===
    p2_start = time.monotonic()
    p2_budget = min(400.0, (deadline - p2_start) * 0.3)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Accelerated FW + continued vanilla (budget={p2_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        # Run accelerated FW
        a_acc = run_fw_minifind(best.seq, min(time.monotonic() + p2_budget * 0.4, deadline - 100),
                                max_iters=5000, use_accel=True, verbose=True)
        best.update(a_acc)

        # Continue vanilla FW from the accelerated result
        if time.monotonic() < deadline - 80:
            a_van = run_fw_minifind(a_acc, min(time.monotonic() + p2_budget * 0.4, deadline - 80),
                                    max_iters=10000, use_accel=False, verbose=True)
            best.update(a_van)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Coordinate descent polish ===
    p3_start = time.monotonic()
    p3_budget = min(150.0, (deadline - p3_start) * 0.15)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Coordinate Descent Polish (budget={p3_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        a_cd = coordinate_descent_polish(best.seq, time.monotonic() + p3_budget, max_rounds=100)
        best.update(a_cd)

        # Another FW round after CD
        if time.monotonic() < deadline - 50:
            a_fw2 = run_fw_minifind(a_cd, min(time.monotonic() + 60, deadline - 50),
                                    max_iters=2000, use_accel=False, verbose=True)
            best.update(a_fw2)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Try other lengths ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        for n_try in candidate_lengths:
            if time.monotonic() > deadline - 60:
                break
            if n_try == best.seq.size:
                continue
            # Skip if this length is too expensive
            if n_try > 1536 and time.monotonic() > deadline - 200:
                continue

            a_res = resample_seq(best.seq, n_try)
            a_res = np.maximum(a_res, 1e-6)
            a_res = a_res / a_res.sum()

            allot = min(60.0, (deadline - 60) - time.monotonic())
            if allot < 20:
                break

            a_opt = run_fw_minifind(a_res, time.monotonic() + allot,
                                     max_iters=3000, use_accel=False, verbose=False)
            J_test = compute_J(a_opt)
            print(f"  n={n_try}: J={J_test:.6f}", flush=True)
            best.update(a_opt)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Final intensive FW on best ===
    p5_start = time.monotonic()
    remaining = deadline - p5_start - 5
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final intensive FW (remaining={remaining:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and remaining > 30:
        # Vanilla FW with miniFind
        a_final = run_fw_minifind(best.seq, deadline - 5, max_iters=20000,
                                   use_accel=False, verbose=True)
        best.update(a_final)

        # Quick CD polish
        if time.monotonic() < deadline - 10:
            a_cd2 = coordinate_descent_polish(a_final, deadline - 3, max_rounds=50)
            best.update(a_cd2)

    best.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best.seq]
