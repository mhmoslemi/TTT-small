import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_val_and_grad(a, tau):
    """Compute smooth max value and gradient using softmax approximation."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    val = b_max + np.log(exp_b.sum()) / tau  # log-sum-exp / tau
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return val, grad


def obj_and_grad(x, tau):

    a = np.maximum(x, 0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    a_norm = a / s
    val, grad = smooth_max_val_and_grad(a_norm, tau)
    # Chain rule: d(val)/dx_i = (grad_i * s - val * a_i) / s^2 ... 
    # Actually, let's work with the normalized version directly
    # J = 2n * max(conv(a/s, a/s)) = 2n * max(conv(a,a)) / s^2
    # This is more complex. Let's just use the normalized version.
    # We'll optimize over a directly with normalization constraint.
    return val, grad / s  # approximate gradient for unnormalized


def run_lbfgs_smooth(a, deadline, tau=50.0, max_iter=200):
    """Use L-BFGS-B to minimize smooth max of convolution."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    def objective(x):
        if time.monotonic() > deadline:
            return float('inf')
        a_x = np.maximum(x, 1e-12)
        s = a_x.sum()
        if s < 1e-10:
            return 1e10
        a_n = a_x / s
        M = fast_conv(a_n).max()
        return M
    
    def gradient(x):
        a_x = np.maximum(x, 1e-12)
        s = a_x.sum()
        if s < 1e-10:
            return np.zeros_like(x)
        a_n = a_x / s
        _, grad = smooth_max_val_and_grad(a_n, tau)
        return grad / s
    
    try:
        res = minimize(objective, a, method='L-BFGS-B', jac=gradient,
                      options={'maxiter': max_iter, 'gtol': 1e-8,
                               'iprint': -1, 'maxfun': 5000})
        if res.x is not None:
            a_res = np.maximum(res.x, 0)
            s = a_res.sum()
            if s > 1e-10:
                a_res /= s
                M = fast_conv(a_res).max()
                if M < best_M:
                    best_M = M
                    best_a = a_res.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix for (a*g)(k) = sum_j a[j]*g[k-j]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, prev_x=None):
    """Solve LP: max sum(g) s.t. Ag <= M, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    x0 = prev_x if prev_x is not None and len(prev_x) == n else None
    try:
        if x0 is not None:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                       options={'presolve': True, 'x0': x0,
                                'dual_feasibility_tolerance': 1e-9})
        else:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                       options={'presolve': True,
                                'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(history_x, history_Fx, m, clip_bound=0.5):
    """Anderson mixing: find alpha minimizing ||sum alpha_i (x_i - Fx_i)||^2."""
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -clip_bound, clip_bound)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha /= s
    mixed = sum(alpha[i] * history_Fx[i] for i in range(m))
    return np.maximum(mixed, 0)


def run_lp_anderson_v2(a, deadline, max_iters=300, anderson_window=6,
                       t_base=0.3, min_t=0.005, backtrack=True):
    """LP mirror descent with Anderson acceleration and backtracking on J."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_imp = 0
    hist_x = []
    hist_Fx = []
    prev_g = None
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        J_cur = 2 * n * M  # since sum(a)=1
        
        g = lp_best_response(a, M, prev_g)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g_norm = g / gs
        prev_g = g.copy()
        
        # Determine step size with backtracking
        if backtrack:
            t_options = [t_base / (1 + 0.1 * it)]
            for factor in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                t_options.append(max(min_t, t_base * factor / (1 + 0.05 * it)))
            t_options = list(dict.fromkeys(t_options))  # deduplicate, preserve order
            t_options.sort(reverse=True)
            
            best_t = min_t
            best_a_new = None
            best_M_new = float('inf')
            for t in t_options:
                a_new = (1 - t) * a + t * g_norm
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new /= s
                M_new = fast_conv(a_new).max()
                if M_new < best_M_new:
                    best_M_new = M_new
                    best_a_new = a_new.copy()
                    best_t = t
            if best_a_new is None:
                break
            a_new = best_a_new
        else:
            t = max(min_t, t_base / (1 + 0.05 * it))
            a_new = (1 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new /= s
            best_M_new = fast_conv(a_new).max()
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
                M_and = fast_conv(a_and).max()
                if M_and <= best_M_new * 1.02:
                    a_new = a_and
                    best_M_new = M_and
        
        if best_M_new < best_J / (2 * n) * (1 - 1e-10):
            best_J = 2 * n * best_M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        
        a = a_new
    
    return best_a


def run_nesterov_v2(a, deadline, max_iters=3000, tau_start=5.0, tau_end=500.0,
                    lr_init=0.3, lr_end=0.001):
    """Improved Nesterov with adaptive tau and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    momentum = 0.9
    no_imp = 0
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        val, grad = smooth_max_val_and_grad(y, tau)
        b_max = fast_conv(y).max()
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200:
                break
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        a_new = np.maximum(y - lr * (grad + momentum * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        for sigma in [0.01, 0.02, 0.05, 0.1]:
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.005))
    inits.append(np.ones(n))
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t**alpha + (1-t)**alpha) / 2, 0.01))
    for f in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2*np.pi*f*t), 0.01))
        inits.append(np.maximum(0.5 + 0.4 * np.sin(2*np.pi*f*t), 0.01))
    inits.append(np.exp(-3*(t-0.5)**2) + 0.05)
    inits.append(np.exp(-6*(t-0.3)**2) + np.exp(-6*(t-0.7)**2) + 0.03)
    # Smooth random
    k = max(5, n // 15)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # Symmetric
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    # GS-like
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    size = 1
    while size < 2*n-1:
        size <<= 1
    for _ in range(50):
        F = np.fft.rfft(a_rand, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag = 0.5 * mag + 0.5 * mag.mean()
        a_rand = np.maximum(np.fft.irfft(np.exp(1j*phase)*mag, size)[:n], 0.01)
        a_rand /= a_rand.sum()
    inits.append(a_rand)
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_val_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    val = b_max + np.log(exp_b.sum() + 1e-30) / tau
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return val, grad


def run_lbfgs_smooth(a, deadline, tau=80.0, max_iter=500):

    a = a / a.sum()
    n = len(a)

        if time.monotonic() > deadline:
            return 1e10
        a_x = np.maximum(x, 1e-12)
        s = a_x.sum()
        if s < 1e-10:
            return 1e10
        a_n = a_x / s
        return float(fast_conv(a_n).max())

        a_x = np.maximum(x, 1e-12)
        s = a_x.sum()
        if s < 1e-10:
            return np.zeros_like(x)
        a_n = a_x / s
        _, grad = smooth_max_val_grad(a_n, tau)
        return grad / s

    try:
        res = minimize(objective, a.copy(), method='L-BFGS-B', jac=gradient,

def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=5.0, tau_end=500.0,

        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        val, grad = smooth_max_val_grad(y, tau)
        b_max = float(fast_conv(y).max())
        if b_max < best_M * (1 - 1e-10):

            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

            a_new = a.copy()

            a_new /= s

        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:

        a = a_new
    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, prev_x=None):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        opts = {'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                'primal_feasibility_tolerance': 1e-9}
        if prev_x is not None:
            opts['x0'] = prev_x
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=opts)
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(history_x, history_Fx, m, clip_bound=0.8):
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -clip_bound, clip_bound)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha /= s
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)

        if time.monotonic() > deadline:

        b = fast_conv(a)

        J_cur = 2.0 * n * M_cur

        gs = g.sum()
        if gs < 1e-10:

        g_norm = g / gs

        best_t = min_t
        best_a_new = None
        best_M_new = float('inf')
        best_J_new = float('inf')

                best_M_new = M_new
                best_a_new = a_new.copy()
                best_t = t
        if best_a_new is None:
            break
        a_new = best_a_new

        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)

            if sa > 1e-10:

        if best_J_new < best_J * (1 - 1e-12):
            best_J = best_J_new
            best_a = a_new.copy()
            no_imp = 0
            stuck_count = 0
        else:
            no_imp += 1
            stuck_count += 1
            if no_imp > 50:

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_gs_init(n, rng, beta=0.85, n_iters=80):
    a = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    a /= a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    for _ in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-15)
        power = 0.5 + 0.3 * beta
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        a_new = np.fft.irfft(np.exp(1j * phase) * mag_final, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        a = a_new
    return a

        inits.append(np.maximum(0.1 + 0.9 * (t**alpha + (1 - t)**alpha) / 2, 0.01))
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.05)
    inits.append(np.exp(-8 * (t - 0.3)**2) + np.exp(-8 * (t - 0.7)**2) + 0.03)
    k = max(5, n // 15)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    inits.append(make_gs_init(n, rng))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(1024)
        best.update(parent)

    # === PHASE 1: Smooth optimization (L-BFGS + Nesterov) ===
    p1_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth Optimization", flush=True)
    print(f"{'='*50}", flush=True)

    p1_budget = 250.0
    dl1 = min(time.monotonic() + p1_budget, deadline - 200)

    # L-BFGS on parent
    if time.monotonic() < dl1 - 50:
        a_lbfgs = run_lbfgs_smooth(parent, min(time.monotonic() + 120, dl1 - 50),
                                   tau=100.0, max_iter=400)
        best.update(a_lbfgs)

    # Nesterov on best
    if time.monotonic() < dl1 - 20:
        start_a = best.seq if best.seq is not None else parent
        a_nest = run_nesterov_smooth(start_a, dl1,
                                     max_iters=5000, tau_start=10.0, tau_end=400.0,
                                     lr_init=0.25, lr_end=0.001)
        best.update(a_nest)

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: LP Mirror Descent (primary optimization) ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Mirror Descent with Anderson", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        # Estimate LP time: for n=1024, each LP takes ~50-200ms with HiGHS
        est_lp_ms = 2.0 * (n / 256.0)**2 + 10
        print(f"  n={n}, est LP={est_lp_ms:.0f}ms/iter", flush=True)

        lp_budget = min(700.0, max(150.0, (deadline - 100) - time.monotonic()))
        max_iters = max(40, int(lp_budget / max(0.02, est_lp_ms / 1000.0)))
        print(f"  LP budget={lp_budget:.0f}s, max_iters={max_iters}", flush=True)

        # First LP pass: larger step sizes
        a_lp1 = run_lp_anderson_v2(best.seq,
                                   min(time.monotonic() + lp_budget * 0.45, deadline - 80),
                                   max_iters=max_iters, anderson_window=8,
                                   t_base=0.35, min_t=0.003, adaptive_M=True)
        best.update(a_lp1)

        # Second LP pass: smaller step sizes for convergence
        if time.monotonic() < deadline - 60:
            a_lp2 = run_lp_anderson_v2(a_lp1,
                                       min(time.monotonic() + lp_budget * 0.35, deadline - 40),
                                       max_iters=int(max_iters * 0.7), anderson_window=10,
                                       t_base=0.15, min_t=0.001, adaptive_M=True)
            best.update(a_lp2)

        # Third LP pass: very small steps
        if time.monotonic() < deadline - 30:
            a_lp3 = run_lp_anderson_v2(a_lp2, deadline - 20,
                                       max_iters=int(max_iters * 0.4), anderson_window=10,
                                       t_base=0.08, min_t=0.001, adaptive_M=False)
            best.update(a_lp3)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Length exploration ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Length Exploration", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 40:
        n_best = best.seq.size
        deltas = [-64, -32, 32, 64, 128, -128]
        for delta in deltas:
            if time.monotonic() > deadline - 25:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 3000:
                continue
            a_try = resample_seq(best.seq, n_try)
            a_try = np.maximum(a_try, 1e-6)
            a_try /= a_try.sum()

            # Nesterov
            sub_dl = min(time.monotonic() + 25, deadline - 20)
            a_nest = run_nesterov_smooth(a_try, sub_dl, max_iters=800,
                                         tau_start=10, tau_end=200,
                                         lr_init=0.2, lr_end=0.005)
            best.update(a_nest)

            # LP if improved
            if best.seq.size == n_try and time.monotonic() < deadline - 15:
                a_lp = run_lp_anderson_v2(best.seq, deadline - 10,
                                          max_iters=30, anderson_window=6,
                                          t_base=0.25, min_t=0.005, adaptive_M=True)
                best.update(a_lp)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Final polish with plain LP ===
    p4_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Final Plain LP Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 5:
        a_final = run_lp_plain(best.seq, min(time.monotonic() + 80, deadline - 2),
                               max_iters=60)
        best.update(a_final)

        # One more Anderson pass
        if time.monotonic() < deadline - 3:
            a_final2 = run_lp_anderson_v2(a_final, deadline - 1,
                                          max_iters=25, anderson_window=8,
                                          t_base=0.1, min_t=0.001, adaptive_M=False)
            best.update(a_final2)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Diverse restart if time remains ===
    if best.seq is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*50}", flush=True)
        print(f"PHASE 5: Diverse Restarts", flush=True)
        print(f"{'='*50}", flush=True)

        n_best = best.seq.size
        for trial in range(5):
            if time.monotonic() > deadline - 10:
                break
            a0 = np.maximum(best.seq + rng.normal(0, 0.03, n_best), 0.005)
            a0 /= a0.sum()
            a0 = run_nesterov_smooth(a0, min(time.monotonic() + 20, deadline - 8),
                                     max_iters=500, tau_start=20, tau_end=300,
                                     lr_init=0.15, lr_end=0.005)
            best.update(a0)
            if time.monotonic() < deadline - 8:
                a0_lp = run_lp_anderson_v2(a0, deadline - 5, max_iters=15,
                                           anderson_window=5, t_base=0.2,
                                           min_t=0.005, adaptive_M=True)
                best.update(a0_lp)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
