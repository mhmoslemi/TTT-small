from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    """One subgradient step targeting peak at k_star."""
    n = len(a)

        if 0 <= idx < n:

    g = g - np.mean(g)
    a_new = a * np.exp(-lr * g)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / np.sum(a_new)
    return a_new

This is just a reversed and shifted version of a! Specifically:
g[j] = 2*a[k*-j] = 2*flip(a)[j - (k*-n+1)]... hmm, let me think.

**Algorithm: Multi-strategy optimization**

Phase 1: Subgradient descent with adaptive target
- Use top-k peaks with softmax weighting for a smoother gradient
- Adaptive step size with backtracking
- Multiplicative updates for stability

Phase 2: Spectral shaping
- In frequency domain, try to flatten the spectrum
- This corresponds to reducing peak autocorrelation

Phase 3: LP refinement
- Use the LP approach on the best sequences found

Phase 4: Coordinate descent / local search
- Fine-tune individual coordinates

Let me now write the full implementation. I'll focus on making it efficient and effective.

Actually, let me think about what makes the LP approach work well and how to complement it:

The LP approach finds the "best direction" to move (the g that minimizes the peak of conv(a,g) given sum(g)=1). This is a very strong per-iteration move, but each LP solve is expensive (O(n^2) or more).

The subgradient approach is fast (O(n log n) per iteration) but takes many small steps.

A good hybrid:
1. Use subgradient for many fast iterations to get close to a good solution
2. Then use LP for a few iterations to make big improvements
3. Alternate between them

Let me also consider the **dual problem**. The LP in the previous approach is:
min M s.t. A@g <= M, sum(g) = 1, g >= 0

The dual is:
max sum(y) s.t. A^T*y >= 0, sum(y) = 1, y >= 0

where y is a distribution over the lags. The optimal dual y tells us which lags are "important" - the ones that are binding.

This gives us a nice interpretation: the optimal subgradient is a mixture of the subgradients at the important lags, weighted by the dual y.

So instead of just using the argmax, we can use a weighted combination of top lags, where the weights come from the dual solution.

But computing the dual is as expensive as solving the LP. So as an approximation, we can use a softmax over the top convolution values.

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines subgradient descent, spectral shaping, and LP-based refinement.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def get_subgradient(a, k_star):
    """Compute subgradient of max(conv(a,a)) at specific lag k_star."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(k_star + 1, n)
    if lo < hi:
        g[lo:hi] = 2.0 * a[k_star - hi + 1 : k_star - lo + 1 : -1]
    return g

def get_smoothed_subgradient(a, conv, top_k=5, temperature=0.05):
    """Compute smoothed subgradient using top-k peaks with softmax weighting."""
    n = len(a)
    # Get top-k lags
    idx = np.argsort(conv)[::-1][:top_k]
    # Softmax weights
    w = np.exp(conv[idx] / temperature - np.max(conv[idx] / temperature))
    w = w / np.sum(w)
    # Weighted subgradient
    g = np.zeros(n)
    for i, k in enumerate(idx):
        g += w[i] * get_subgradient(a, int(k))
    return g

def subgradient_descent(a, time_budget=100.0, verbose=False):
    """Subgradient descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1  # learning rate
    top_k = 3
    temperature = 1.0 / (n * 0.1)
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        k_star = int(np.argmax(conv))
        
        # Compute subgradient
        g = get_smoothed_subgradient(a, conv, top_k=top_k, temperature=temperature)
        
        # Project onto sum=const hyperplane
        g = g - np.mean(g)
        
        # Try a few step sizes (backtracking)
        improved = False
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            a_new = a - lr_try * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                lr = lr_try  # keep successful step size
                break
            elif obj_new < best_obj:
                # Accept but don't update best tracking
                a = a_new
                improved = True
                break
        if not improved:
            lr *= 0.5
        
        if verbose and it % 100 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, lr={lr:.6f}")
    
    return best_a

def multiplicative_descent(a, time_budget=100.0, verbose=False):
    """Multiplicative (mirror) descent."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    a[a == 0] = 1e-10
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1
    top_k = 3
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        g = get_smoothed_subgradient(a, conv, top_k=top_k, temperature=1.0/(n*0.1))
        g = g - np.mean(g)
        
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 1e-10)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # still accept for exploration
            # Decay learning rate
            if obj_new > best_obj * 1.01:
                lr *= 0.9
    
    return best_a

def spectral_flatten(a, time_budget=100.0, verbose=False):
    """Spectral flattening approach."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Size for FFT
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        
        # Forward FFT
        fa = rfft(a, size)
        power = np.abs(fa)**2
        
        # Flatten: reduce peaks in frequency domain
        # Target: make power as uniform as possible
        target = np.mean(power)
        # Soft threshold: reduce values above target
        new_power = np.where(power > target, target + 0.5*(power - target), power)
        
        # Also boost values below target
        new_power = np.where(power < target, target - 0.3*(target - power), power)
        
        new_power = np.maximum(new_power, 1e-20)
        
        # Preserve phase, update magnitude
        phase = fa / np.abs(fa)
        phase[np.abs(fa) < 1e-10] = 1.0
        new_fa = phase * np.sqrt(new_power)
        
        # Ensure Hermitian symmetry (real signal)
        a_new = irfft(new_fa, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
    
    return best_a

def lp_refine(a, time_budget=60.0, verbose=False):
    """LP-based refinement (min-peak LP)."""
    try:
        import cvxpy as cp
    except ImportError:
        return a
    
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        # Build convolution matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a[:r]
        
        # Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        g_val = g_val / gs
                        solved = True
                        break
            except:
                continue
        
        if not solved:
            break
        
        # Blend
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1-t)*a + t*g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break
    
    return best_a

def coordinate_descent(a, time_budget=30.0):
    """Coordinate descent: adjust pairs of coordinates."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    
    for outer in range(20):
        if time.monotonic() - start > time_budget:
            break
        
        k_star = int(np.argmax(conv))
        improved = False
        
        # Try moving mass from high-contributing to low-contributing positions
        for i in range(0, n, max(1, n // 100)):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i:
                for frac in [0.01, 0.05, 0.1, 0.2]:
                    delta = frac * min(a[i], a[j])
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv)) / (float(np.sum(a))**2)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break
        
        if not improved:
            break
    
    return best_a

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

Let me think about another approach: **Strategic coordinate manipulation**.

The key observation is that the peak of the autocorrelation is determined by a few "important" pairs of indices. If we can identify which pairs contribute most to the peak and reduce their contribution, we can lower the peak.

For lag k, the contribution is: (a*a)[k] = sum_i a[i]*a[k-i].
The main contributors are the terms where both a[i] and a[k-i] are large.

So a strategy would be:
1. Find the peak lag k*
2. Identify the top contributors (i, k*-i) pairs
3. Reduce the product a[i]*a[k*-i] for these pairs (by reducing one of the two)
4. Compensate by increasing other coordinates (to maintain sum)

This is similar to coordinate descent but more targeted.

Actually, let me think about an even simpler but potentially effective approach: **iterative water-filling on the autocorrelation**.

At each step:
1. Compute conv = a*a
2. Identify the peak lag k*
3. The peak is sum_i a[i]*a[k*-i]. To reduce it, we want to reduce the "concentration" of a.
4. Specifically, for the pairs (i, k*-i) that contribute most, transfer mass from the larger to other positions.

Let me try a different angle. What if I use ** simulated annealing** with a specific move proposal?

Moves:
1. Pick a random pair (i, j) and swap their values
2. Pick a random pair (i, j) and transfer a small amount of mass
3. Pick the peak-contributing pair and reduce it
4. Randomly perturb a few coordinates

Acceptance: Metropolis criterion (accept if better, or accept with probability exp(-Delta/T) if worse)

def topk_lp(a, k_constraints=10, time_budget=30.0):
    """LP with only top-k convolution constraints."""
    import cvxpy as cp
    
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    
    # Get top-k lags
    top_idx = np.argsort(conv)[::-1][:k_constraints]
    M_target = conv[top_idx[0]]  # current max
    
    # Build constraint matrix for only these lags
    A_k = np.zeros((len(top_idx), n))
    for row, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        for j in range(lo, hi):
            A_k[row, j] = a[k - j]
    
    # LP: max sum(g) s.t. A_k @ g <= M_target, g >= 0
    g = cp.Variable(n, nonneg=True)
    constraints = [A_k @ g <= M_target]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except:
            continue
    return None

    return 2.0 * n * float(np.max(conv))


def get_subgradient_at_lag(a, k):
    """Subgradient of (a*a)[k] w.r.t. a: equals 2*a[k-j] for valid j."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k + 1, n)
    if lo < hi:
        g[lo:hi] = 2.0 * a[k - hi + 1:k - lo + 1:-1]
    return g


def get_weighted_subgradient(a, conv, top_k=5, temperature=None):
    """Weighted subgradient using top-k peaks with softmax weights."""
    n = len(a)
    if temperature is None:
        temperature = 0.05 * float(np.max(conv))
    idx = np.argsort(conv)[::-1][:top_k]
    w = np.exp(conv[idx] / temperature - conv[idx[0]] / temperature)
    w = w / w.sum()
    g = np.zeros(n)
    for i, k in enumerate(idx):
        g += w[i] * get_subgradient_at_lag(a, int(k))
    return g
