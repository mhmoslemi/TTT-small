import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def obj_val(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.where(u * np.arange(1, n+1) > (cssv - 1))[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_gradient(a, tau):
    """Compute smooth approximation of max(conv) and its gradient."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    smooth_val = max_c + np.log(sum_exp) / tau
    w = exp_shifted / sum_exp
    # Gradient: grad[i] = 2 * sum_j a[j] * w[i+j]
    # Using FFT: IFFT(FFT(w,N) * conj(FFT(a,N)))[:n]
    N = 1
    while N < 3 * n:
        N <<= 1
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)
    grad = 2.0 * corr[:n]
    return smooth_val, grad

def gradient_descent(a, time_budget=100.0, verbose=False):
    """Smoothed gradient descent with Adam and simplex projection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    lr = 0.05
    
    # Temperature schedule
    tau_start = 5.0
    tau_end = 2000.0
    total_iters_estimate = 50000
    
    stall = 0
    stall_threshold = 500
    
    while time.monotonic() - start < time_budget:
        # Batch of iterations
        batch = 100
        for _ in range(batch):
            if time.monotonic() - start > time_budget:
                break
            
            t += 1
            # Temperature annealing
            progress = min(1.0, t / total_iters_estimate)
            tau = tau_start * (tau_end / tau_start) ** progress
            
            # Learning rate decay
            lr_t = lr / (1.0 + 0.01 * t)
            
            smooth_val, grad = smooth_gradient(a, tau)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            a_new = a - step
            a_new = simplex_project(a_new)
            
            if np.sum(a_new) > 1e-10:
                a = a_new

        if obj < best_obj - 1e-10:

            best_a = a.copy()
            stall = 0

                best_obj = obj_val(a)
                stall = 0
                # Reset Adam
                m = np.zeros(n)
                v = np.zeros(n)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)

    # Use scipy linprog (most reliable)
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

        s = float(np.sum(g_val))

"""
Hybrid smoothed gradient descent + LP alternating optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses log-sum-exp smoothing for the max operator,
Adam optimizer with simplex projection, and LP-based jumps to escape local minima.
Multi-n exploration and temperature annealing for global search.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_val(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Euclidean projection onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, tau):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient.
    Uses log-sum-exp: (1/tau)*log(sum_k exp(tau*conv[k]))
    Returns (smooth_value, gradient_array).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    if sum_exp < 1e-300:
        return max_c, np.zeros(n)
    smooth_val = max_c + np.log(sum_exp) / tau
    w = exp_shifted / sum_exp
    # Gradient: grad[i] = 2 * sum_j a[j] * w[i+j]
    # Computed via FFT cross-correlation: IFFT(FFT(w,N)*conj(FFT(a,N)))[:n]
    N = 1
    while N < 3 * n:
        N <<= 1
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)
    grad = 2.0 * corr[:n]
    return smooth_val, grad

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2 * n - 1), A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_val = float(np.sum(g_val))
            if s_val > 1e-10:
                return g_val / s_val, float(res.x[n])
    except Exception:
        pass
    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.3
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        c1, c2 = n * 0.3, n * 0.7
        w1, w2 = n * 0.1, n * 0.1
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w1 ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w2 ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Triangular
        a = 1.0 - np.abs(np.arange(n) - n // 2) / (n // 2)
        a = a / np.sum(a)
    else:
        # Random smooth
        freqs = rng.integers(1, 5, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * np.arange(n) / n + p)
        a = np.maximum(a, 0)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    return a

    """

    Multi-phase: intensive optimization, multi-n exploration, final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s > 0.01:
            best_seq = parent / s
            best_score = obj_val(best_seq)
            print(f"Parent (n={len(best_seq)}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800
        best_score = obj_val(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    # Phase 1: Intensive mixed optimization on parent
    print(f"\n=== Phase 1: Mixed Optimization (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.45, 600.0)
    a1 = mixed_optimize(best_seq.copy(), time_budget=p1_budget, verbose=True)
    obj1 = obj_val(a1)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        n_best = len(best_seq)
        print(f"  Phase 1 result: obj={best_score:.8f}")
    else:
        print(f"  Phase 1 result: obj={obj1:.8f} (no improvement)")

    # Phase 2: Pure gradient descent with different hyperparameters
    print(f"\n=== Phase 2: GD with varied params ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(remaining * 0.2, 200.0)
        # Higher temperature for more exploration
        a2 = gradient_descent(best_seq.copy(), time_budget=p2_budget,
                              verbose=True, tau_start=2.0, tau_end=5000.0)
        obj2 = obj_val(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            n_best = len(best_seq)
            print(f"  Phase 2 result: obj={best_score:.8f}")
        else:
            print(f"  Phase 2 result: obj={obj2:.8f}")

    # Phase 3: LP alternating polish
    print(f"\n=== Phase 3: LP Alternating Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30 and n_best <= 1500:
        p3_budget = min(remaining * 0.2, 150.0)
        a3 = alternating_optimize(best_seq.copy(), time_budget=p3_budget, verbose=True)
        obj3 = obj_val(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3 result: obj={best_score:.8f}")
        else:
            print(f"  Phase 3 result: obj={obj3:.8f}")

    # Phase 4: Multi-n exploration
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100), n_best,
        n_best + 100, n_best + 200, n_best + 400
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.15, 50.0)
        a_opt = mixed_optimize(a_try, time_budget=alloc)
        obj_try = obj_val(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 5: Random restarts from diverse inits
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        n_r = n_best + int(rng.integers(-100, 200))
        n_r = max(300, min(1500, n_r))
        a_r = diverse_init(n_r, rng)
        alloc = min(remaining * 0.12, 35.0)
        a_opt = mixed_optimize(a_r, time_budget=alloc)
        obj_r = obj_val(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (n={n_r}): obj={obj_r:.8f}")

    # Phase 6: Final intensive polish
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20 and n_best <= 1200:
        a_f = best_seq.copy()
        alloc_lp = min(remaining * 0.4, 80.0)
        a_lp = alternating_optimize(a_f, time_budget=alloc_lp, verbose=True)
        obj_lp = obj_val(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_gd = gradient_descent(best_seq.copy(), time_budget=min(remaining * 0.4, 40.0),
                                    verbose=True, tau_start=50.0, tau_end=10000.0)
            obj_gd = obj_val(a_gd)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  Final GD: obj={best_score:.8f}")

    # Final validation and output
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = obj_val(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
