from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_for_direction(a, M, time_limit=10):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def line_search_blend(a, g, obj_a, tol=0.005, max_iter=20):
    """Find best t in [0,1] for a*(1-t) + g*t, return (best_a, best_obj, best_t)."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = compute_obj(normalize((1 - c) * a + c * g))
    fd = compute_obj(normalize((1 - d) * a + d * g))
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = compute_obj(normalize((1 - c) * a + c * g))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = compute_obj(normalize((1 - d) * a + d * g))
    t = (lo + hi) / 2.0
    a_best = normalize((1 - t) * a + t * g)
    obj_best = compute_obj(a_best)
    return a_best, obj_best, t


def adam_smooth_gradient(a, time_budget=60.0, lr_init=0.05, beta1=0.9, beta2=0.999,
                          tau_init=0.002, tau_decay=0.97, max_steps=500):
    """Adam optimizer with log-sum-exp smoothing of the max."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    tau = tau_init
    lr = lr_init
    
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Temperature annealing
        tau = max(tau * tau_decay, 1e-6)
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: grad_i = 2 * sum_k weights[k] * a[k-i] (with bounds)
        # This is a cross-correlation
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        
        # Center gradient (remove mean for sum-invariance)
        grad_centered = grad - np.mean(grad)
        
        # Adam update
        step_num = step + 1
        m_state = beta1 * m_state + (1 - beta1) * grad_centered
        v_state = beta2 * v_state + (1 - beta2) * grad_centered ** 2
        m_hat = m_state / (1 - beta1 ** step_num)
        v_hat = v_state / (1 - beta2 ** step_num)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce learning rate
            lr *= 0.8
            if lr < 1e-8:
                break
        
        # Occasional learning rate increase
        if step % 50 == 49 and lr < lr_init:
            lr = min(lr * 1.5, lr_init)
    
    return best_a


def peak_shaving(a, time_budget=30.0):
    """Targeted coordinate descent: reduce the highest peak by adjusting its contributors."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find contributors to the peak
        contribs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n and j != k2:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)
        
        if not contribs:
            break
        
        improved = False
        for peak_val, j1, j2 in contribs[:8]:
            if time.monotonic() - start > time_budget:
                break
            
            # Try reducing j1 and increasing a random other position
            for attempt in range(5):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue
                
                # Try different delta values
                for frac in [0.02, 0.05, 0.1, 0.15]:
                    delta = frac * a[j1]
                    if a[j1] - delta < 0:
                        continue
                    a_try = a.copy()
                    a_try[j1] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            # Try a small random perturbation
            a = normalize(best_a * np.exp(rng.normal(0, 0.02, n)))
            obj_try = compute_obj(a)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()
    
    return best_a


def resample_fft(a, new_n):
    """Resample using FFT (better than linear interp for smooth signals)."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    
    # Zero-pad to max(n, new_n) power of 2
    size = 1
    while size < max(n, new_n) * 2:
        size <<= 1
    
    A = fft(a, size)
    
    # Keep only the lowest new_n/2 frequencies (low-pass filter)
    keep = new_n // 2
    A_high = np.zeros(size, dtype=complex)
    A_high[:keep] = A[:keep]
    A_high[-keep:] = A[-keep:]
    # Handle Nyquist
    if new_n % 2 == 0:
        A_high[size // 2] = A[size // 2]
    
    a_new = np.real(ifft(A_high, size))[:new_n]
    a_new = np.maximum(a_new, 0.0)
    return normalize(a_new)


def adaptive_lp_alternate(a, time_budget=100.0, max_iters=50):
    """
    LP alternating with adaptive M.
    Tries multiple M values per iteration and picks the best direction.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        conv = compute_conv(a)
        M_max = float(np.max(conv))
        
        # Try multiple M values
        m_factors = [1.0, 0.99, 0.97, 0.95]
        best_dir = None
        best_dir_obj = best_obj
        best_dir_a = None
        
        for mf in m_factors:
            if time.monotonic() - start > time_budget - 3:
                break
            M = M_max * mf
            tp = max(3, int(remaining * 0.2 / len(m_factors)))
            g = solve_lp_for_direction(a, M, time_limit=tp)
            if g is None:
                continue
            
            a_try, obj_try, t = line_search_blend(a, g, best_obj, tol=0.01, max_iter=12)
            if obj_try < best_dir_obj:
                best_dir_obj = obj_try
                best_dir_a = a_try
                best_dir = g
        
        if best_dir_a is not None and best_dir_obj < best_obj:
            best_obj = best_dir_obj
            best_a = best_dir_a.copy()
            a = best_dir_a
        else:
            # No improvement from LP; try a small perturbation
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03 + 0.01 * (it % 3), n)))
            obj_a = compute_obj(a)
            if obj_a < best_obj:
                best_obj = obj_a
                best_a = a.copy()
        # Continue
    
    return best_a

The problem is: minimize `max_k (a^T Q_k a)` subject to `sum(a) = 1, a >= 0`, where `Q_k` is the matrix with `(Q_k)_{ij} = 1 if i+j=k`.

This is a non-convex problem (Q_k is indefinite for k not at the extremes).

Key approaches:
1. **LP alternating** (fix a, optimize g): Gives a descent direction. Provable convergence to a local minimum.
2. **Gradient methods** (with smoothing): Can be faster in practice but may converge to different local minima.
3. **Second-order methods**: Use the Hessian of the smoothed objective. More expensive per step but potentially faster convergence.
4. **Evolutionary / global methods**: Can escape local minima but expensive.

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive-M LP alternating, Adam-based smooth gradient descent,
targeted peak shaving, multi-n exploration with diverse starts,
and iterative final polish. Key improvements over prior: multi-M LP
direction selection, Adam optimizer with adaptive LR, extended line
search, and diverse initialization in multi-n phase.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_direction(a, M, time_limit=8):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'mip_rel_gap': 0.0})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def line_search(a, g, max_iter=18, tol=0.008):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    a_c = normalize((1 - c) * a + c * g)
    a_d = normalize((1 - d) * a + d * g)
    fc = compute_obj(a_c)
    fd = compute_obj(a_d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            a_c = normalize((1 - c) * a + c * g)
            fc = compute_obj(a_c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            a_d = normalize((1 - d) * a + d * g)
            fd = compute_obj(a_d)
    t = (lo + hi) / 2.0
    a_best = normalize((1 - t) * a + t * g)
    obj_best = compute_obj(a_best)
    return a_best, obj_best


def adaptive_lp_alternate(a, time_budget=120.0, max_iters=60):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 4:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        conv = compute_conv(a)
        M_max = float(np.max(conv))
        if M_max < 1e-15:
            break

        m_factors = [1.0, 0.985, 0.97, 0.94]
        best_dir_obj = best_obj
        best_dir_a = None
        tp_each = max(2, int(remaining * 0.25 / len(m_factors)))

        for mf in m_factors:
            if time.monotonic() - start > time_budget - 2:
                break
            M = M_max * mf
            g = solve_lp_direction(a, M, time_limit=tp_each)
            if g is None:
                continue
            a_try, obj_try = line_search(a, g, max_iter=14, tol=0.012)
            if obj_try < best_dir_obj:
                best_dir_obj = obj_try
                best_dir_a = a_try

        if best_dir_a is not None and best_dir_obj < best_obj - 1e-14:
            best_obj = best_dir_obj
            best_a = best_dir_a.copy()
            a = best_dir_a
        else:
            rng = np.random.default_rng((it * 6271 + 31337) % (2**31))
            sigma = 0.025 + 0.015 * (it % 4)
            a = normalize(best_a * np.exp(rng.normal(0, sigma, n)))
            obj_a = compute_obj(a)
            if obj_a < best_obj:
                best_obj = obj_a
                best_a = a.copy()

    return best_a


def adam_gradient(a, time_budget=60.0, lr_init=0.06, beta1=0.9, beta2=0.999,
                  tau_init=0.002, tau_decay=0.965, max_steps=400):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m_state = np.zeros(n)
    v_state = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall_count = 0

    size = 1
    while size < 3 * n - 1:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        tau = max(tau * tau_decay, 1e-7)
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(weights.sum())
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        grad_centered = grad - np.mean(grad)

        step_num = step + 1
        m_state = beta1 * m_state + (1 - beta1) * grad_centered
        v_state = beta2 * v_state + (1 - beta2) * grad_centered ** 2
        m_hat = m_state / (1 - beta1 ** step_num)
        v_hat = v_state / (1 - beta2 ** step_num)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = normalize(a_new)

        if not np.all(np.isfinite(a_new)):
            a_new = normalize(a)
            lr *= 0.5
            continue

        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            lr *= 0.85
            if lr < 1e-9:
                break

        if stall_count > 30 and step % 20 == 0:
            lr = min(lr * 2.0, lr_init)
            stall_count = 0

        if step % 40 == 39 and lr < lr_init * 0.5:
            lr = lr * 1.4

    return best_a


def peak_shaving(a, time_budget=40.0):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        contribs = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n and j != k2:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)

        if not contribs:
            break

        improved = False
        for peak_val, j1, j2 in contribs[:10]:
            if time.monotonic() - start > time_budget:
                break
            for attempt in range(6):
                j_t = int(rng.integers(0, n))
                if j_t == j1 or j_t == j2:
                    continue
                for frac in [0.015, 0.04, 0.08, 0.12, 0.18]:
                    delta = frac * a[j1]
                    if a[j1] - delta < -1e-12:
                        continue
                    a_try = a.copy()
                    a_try[j1] -= delta
                    a_try[j_t] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            a = normalize(best_a * np.exp(rng.normal(0, 0.015, n)))
            if compute_obj(a) < best_obj:
                best_obj = compute_obj(a)
                best_a = a.copy()

    return best_a


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return normalize(np.interp(x_new, x_old, a))
