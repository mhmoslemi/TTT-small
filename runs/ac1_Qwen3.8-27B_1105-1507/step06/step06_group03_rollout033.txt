def fourier_smooth(a, alpha=0.5, beta=2.0):
    """Apply Fourier-domain spectral flattening."""
    n = len(a)
    size = 1
    while size < 4 * n:
        size <<= 1
    A = np.fft.rfft(a, size)
    P = np.abs(A)**2
    P_mean = P.mean()
    if P_mean < 1e-15:
        return a.copy()
    # Soft compression of peaks
    ratio = P / P_mean
    # Compress: values above mean get compressed, values below get slightly boosted
    P_new = P_mean * np.sign(ratio) * np.abs(ratio) ** (1.0 - alpha * np.log2(np.maximum(ratio, 1e-10)))
    # Hmm, this is getting complicated. Let me use a simpler approach:

    # Reconstruct with new magnitude, old phase

    a_new = np.fft.irfft(A_new, size)[:n]

        else:

                lr *= 0.5
                no_improve = 0
                # Reset momentum

                t = 1.0
                # Add small perturbation
                a = project_simplex(best_a + np.random.default_rng(it).normal(0, 0.01, n))

def optimize_single_coord(a, i, conv, s, n):
    """Optimize a[i] by 1D search. Returns optimal value."""
    old = a[i]
    
    # Determine which conv entries are affected
    # conv[k] is affected by a[i] if 0 <= k-i < n
    # i.e., i <= k <= i + n - 1
    lo = i
    hi = min(2*n - 2, i + n - 1)
    
    # For each affected k, determine the linear and quadratic coefficients
    # conv[k](x) = conv[k](old) - contrib_old + contrib_new
    # contrib of a[i] to conv[k]:
    #   if 0 <= k-i < n and k-i != i: 2*x*a[k-i]
    #   if k == 2i: x^2
    
    # Precompute base values (conv[k] with a[i] contribution removed)
    base = np.zeros(hi - lo + 1)
    lin = np.zeros(hi - lo + 1)
    quad = np.zeros(hi - lo + 1)
    
    for idx, k in enumerate(range(lo, hi + 1)):
        ki = k - i
        base[idx] = conv[k]
        if 0 <= ki < n:
            if ki == i:
                # conv[k] includes a[i]^2 = old^2
                base[idx] -= old * old
                quad[idx] = 1.0
            else:
                # conv[k] includes 2*old*a[ki]
                base[idx] -= 2.0 * old * a[ki]
                lin[idx] = 2.0 * a[ki]
    
    # Unaffected conv entries
    # The max of unaffected entries is:
    if lo > 0:
        max_unaffected_low = conv[:lo].max()
    else:
        max_unaffected_low = 0.0  # no entries
    if hi < 2*n - 2:
        max_unaffected_high = conv[hi+1:].max()
    else:
        max_unaffected_high = 0.0
    max_unaffected = max(max_unaffected_low, max_unaffected_high)
    
    # J(x) = 2n * max(max_unaffected, max_k(base[k] + lin[k]*x + quad[k]*x^2)) / (s - old + x)^2
    # We want to minimize this over x >= 0
    
    s_new_base = s - old
    
    def J_of_x(x):
        affected = base + lin * x + quad * x * x
        M = max(max_unaffected, affected.max())
        s_new = s_new_base + x
        if s_new < 1e-10:
            return float('inf')
        return 2.0 * n * M / (s_new * s_new)
    
    # Golden section search on [0, s_new_base + old] = [0, s]
    # (x > s makes no sense since sum would exceed original)
    # Actually x can be up to s (making all mass on one coordinate)
    a_lo = 0.0
    a_hi = s  # upper bound
    
    # First, check if the minimum is at x=0
    J_0 = J_of_x(0.0)
    J_hi = J_of_x(a_hi)
    
    if J_0 <= J_hi:
        # Minimum might be near 0, do golden section on [0, s]
        pass
    else:
        # Minimum is somewhere in (0, s)
        pass
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = J_of_x(c)
    fd = J_of_x(d)
    
    for _ in range(50):  # 50 iterations gives ratio (1-gr)^50 ~ 1e-11

            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = J_of_x(c)

            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = J_of_x(d)
    
    x_opt = (a_lo + a_hi) / 2.0

"""
Convex optimizer for J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses golden-section coordinate descent on the simplex parameterization,
FISTA with log-sum-exp smoothing, and multi-start strategy.
The problem is convex (min of max of convex quadratics over a polytope),
so all local minima are global.
"""

import numpy as
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.astype(np.fft.irfft(fa * fa, size)[:2 * n - 1], np.float64)


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
