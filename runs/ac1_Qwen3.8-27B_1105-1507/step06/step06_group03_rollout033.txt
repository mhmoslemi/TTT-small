def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lp_best_response(a, M, time_limit=15.0):
    """
    Solve LP: maximize sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (normalized to sum=1) and the optimal sum value.
    """
    n = len(a)
    
    # Build constraint matrix: (a*g)[k] = sum_j a[k-j] * g[j] for valid j
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs, gs
            return g, 0.0
    except Exception:
        pass
    return None, 0.0

def smooth_mirror_descent(a, deadline, max_iters=20000, tau=50.0, lr=0.02):

            break

        # Smooth max via log-sum-exp
        c_shifted = c / tau
        logsumexp = c_max/tau + np.log(np.sum(np.exp(c_shifted - c_max/tau)))
        
        # Gradient of smooth max wrt a: 
        # d(logsumexp)/da[i] = sum_k w[k] * (a[k-i] + a[i-k] terms)
        w = np.exp(c_shifted - c_max/tau) / np.sum(np.exp(c_shifted - c_max/tau))
        w = w / (2.0 * n)  # normalize

        a_rev = a[::-1]
        size = 1

        Wrev_fft = np.fft.rfft(w[::-1], size)
        
        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wrev_fft * A_fft, size)
        
        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad[i] += conv2[idx2]
        
        grad = grad - np.mean(grad)
        
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size == n:
        p = np.maximum(parent, 0) / max(parent.sum(), 1e-10)
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    inits.append(np.ones(n) / n)
    
    # Cosine patterns
    for freq in [1, 2, 3, 4]:
        for amp in [0.2, 0.5, 0.8]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
    
    # Bimodal
    inits.append(project_simplex(np.exp(-((t-0.25)/0.1)**2) + np.exp(-((t-0.75)/0.1)**2)))
    inits.append(project_simplex(0.1 + np.exp(-((t-0.3)/0.15)**2) + np.exp(-((t-0.7)/0.15)**2)))
    
    # Smooth random
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    # Power law
    for alpha in [0.5, 1.5, 3.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))

    return inits

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lp_best_response(a, M, time_limit=12.0):
    """
    Solve LP: maximize sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g_normalized, optimal_sum_value).
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs, gs
    except Exception:
        pass
    return None, 0.0

def lp_sca_step(a, time_limit=12.0):
    """
    SCA-LP: minimize t s.t. 2*(a*g)[k] - t <= c[k], sum(g)=1, g>=0.
    This gives a tighter linearization.
    """
    n = len(a)
    c = fast_conv(a)
    rows = []
    cols = []
    vals = []
    b_ub = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * a[k - j])
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
        b_ub.append(c[k])
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    rows_eq = list(range(n))
    A_eq = coo_matrix((np.ones(n), (rows_eq, rows_eq)), shape=(1, n + 1)).tocsr()
    b_eq = [1.0]
    bounds = [(0, None)] * n + [(None, None)]
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            gs = g.sum()

        pass

        else:
            order = range(n)

        if ps > 1e-10:
            parent_n = parent_n / ps
        inits.append(parent_n.copy())
        inits.append(project_simplex(parent_n + rng.normal(0, 0.03, n)))
        inits.append(project_simplex(parent_n + rng.normal(0, 0.08, n)))
    inits.append(np.ones(n) / n)
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.6]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
    inits.append(project_simplex(0.1 + np.exp(-((t-0.3)/0.12)**2) + np.exp(-((t-0.7)/0.12)**2)))
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer: LP best-response with binary search steps,
    targeted CD, SCA-LP, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)
    print(f"\n{'='*60}", flush=True)
    print(f"Starting optimization: n={n_parent}, budget={budget_s}s", flush=True)
    print(f"{'='*60}", flush=True)

    # === Phase 1: Quick CD baseline (80s) ===
    print(f"\nPHASE 1: Quick CD baseline", flush=True)
    t1 = time.monotonic() + 80.0
    a_curr = parent.copy()
    a_curr = np.maximum(a_curr, 0)
    a_curr = a_curr / a_curr.sum()
    a_curr = opt_cd(a_curr, deadline=t1, max_sweeps=20000, seed=42)
    update_best(a_curr)

    remaining = deadline - time.monotonic()
    if remaining < 50:
        return [float(x) for x in best_seq]

    # === Phase 2: LP optimization (35% of remaining) ===
    print(f"\nPHASE 2: LP best-response optimization", flush=True)
    t2 = time.monotonic() + 0.35 * remaining
    a_lp, obj_lp = lp_optimize(a_curr, deadline=t2, max_iters=200, use_tightening=False, verbose=False)
    update_best(a_lp)
    # Quick CD polish
    t2c = min(time.monotonic() + 40.0, t2)
    a_p = opt_cd(a_lp, deadline=t2c, max_sweeps=5000, seed=77)
    update_best(a_p)

    remaining = deadline - time.monotonic()
    if remaining < 50:
        return [float(x) for x in best_seq]

    # === Phase 3: SCA-LP (20% of remaining) ===
    print(f"\nPHASE 3: SCA-LP optimization", flush=True)
    t3 = time.monotonic() + 0.20 * remaining
    a_sca, obj_sca = lp_sca_optimize(best_seq, deadline=t3, max_iters=80)
    update_best(a_sca)
    # CD polish
    t3c = min(time.monotonic() + 30.0, t3)
    a_p2 = targeted_cd(a_sca, deadline=t3c, max_sweeps=5000, seed=123)
    update_best(a_p2)

    remaining = deadline - time.monotonic()
    if remaining < 50:
        return [float(x) for x in best_seq]

    # === Phase 4: Multi-n exploration (15% of remaining) ===
    print(f"\nPHASE 4: Multi-n exploration", flush=True)
    t4 = time.monotonic() + 0.15 * remaining
    candidate_ns = [n for n in [384, 512, 640, 768, 896, 1024, 1152, 1280] if n != n_parent and n <= 1536]
    candidate_ns.sort(key=lambda x: abs(x - n_parent))
    for new_n in candidate_ns[:4]:
        if time.monotonic() > t4 - 40:
            break
        alloc = min(80.0, t4 - time.monotonic() - 15)
        if alloc < 40:
            break
        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()
        a_opt = opt_cd(a_ext, deadline=time.monotonic() + 0.3 * alloc, max_sweeps=3000, seed=new_n)
        a_opt, _ = lp_optimize(a_opt, deadline=time.monotonic() + alloc * 0.6, max_iters=30)
        a_opt = opt_cd(a_opt, deadline=time.monotonic() + 0.1 * alloc, max_sweeps=2000, seed=new_n+1)
        update_best(a_opt)

    remaining = deadline - time.monotonic()
    if remaining < 50:
        return [float(x) for x in best_seq]

    # === Phase 5: Intensive final optimization ===
    print(f"\nPHASE 5: Intensive final optimization ({remaining:.0f}s)", flush=True)
    a_curr = best_seq.copy()

    # 5a: Mixed LP+CD (30%)
    t5a = time.monotonic() + 0.30 * remaining
    print(f"  Mixed LP+CD ({t5a - time.monotonic():.0f}s)...", flush=True)
    a_mixed, obj_mixed = mixed_descent(a_curr, deadline=t5a, max_iters=100)
    update_best(a_mixed)
    a_curr = best_seq.copy()

    remaining = deadline - time.monotonic()
    if remaining < 30:
        return [float(x) for x in best_seq]

    # 5b: Targeted CD (25%)
    t5b = time.monotonic() + 0.25 * remaining
    print(f"  Targeted CD ({t5b - time.monotonic():.0f}s)...", flush=True)
    a_curr = targeted_cd(a_curr, deadline=t5b, max_sweeps=50000, seed=456)
    update_best(a_curr)

    remaining = deadline - time.monotonic()
    if remaining < 30:
        return [float(x) for x in best_seq]

    # 5c: LP with tightening (20%)
    t5c = time.monotonic() + 0.20 * remaining
    print(f"  LP tightened ({t5c - time.monotonic():.0f}s)...", flush=True)
    a_tight, obj_tight = lp_optimize(a_curr, deadline=t5c, max_iters=100, use_tightening=True)
    update_best(a_tight)
    a_curr = best_seq.copy()

    remaining = deadline - time.monotonic()
    if remaining < 30:
        return [float(x) for x in best_seq]

    # 5d: Final CD (25%)
    t5d = deadline - 5
    print(f"  Final CD ({t5d - time.monotonic():.0f}s)...", flush=True)
    a_curr = opt_cd(a_curr, deadline=t5d, max_sweeps=50000, seed=789)
    update_best(a_curr)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
