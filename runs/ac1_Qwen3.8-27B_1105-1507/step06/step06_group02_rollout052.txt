```python
def optimize_fast(a, deadline):
    a = np.maximum(a, 0)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = conv(a).max()
    
    # Adam optimizer
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    exp_lr = lr  # for bias correction
    
    for it in range(max_iters):
        if time > deadline: break
        
        b = conv(a, a)
        b_max = b.max()
        
        if b_max < best_M:
            best_M = b_max
            best_a = a.copy()
        
        # Compute gradient using top-k lags
        # Use adaptive k based on how peaked the autocorrelation is
        k_active = min(10, n)  # or adaptive
        top_k = np.argsort(b)[-k_active:]
        b_top = b[top_k]
        b_top_max = b_top.max()
        weights = np.exp(10 * (b_top - b_top_max))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for k in top_k:
            w = weights[np.where(top_k == k)[0][0]]
            # grad contribution from lag k
            for j in range(n):
                # d/d a[j] of (a*a)[k] = a[k-j] + a[j-k] (if valid)
                kj = k - j
                jk = j - k
                val = 0
                if 0 <= kj < n:
                    val += a[kj]
                if 0 <= jk < n:
                    val += a[jk]
                grad[j] += w * val
        
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * grad**2
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        
        a_new = a - exp_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        
        a = a_new
        # Decay learning rate
        exp_lr = lr * (0.999 ** it)
    
    return best_a
```

```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def grad_active(a, b, tau=10.0, k_top=50):
    """Compute gradient of smooth max of autocorrelation."""
    n = len(a)
    b_max = b.max()
    # Use top-k for efficiency
    if k_top < len(b):
        top_idx = np.argpartition(b, -k_top)[-k_top:]
        b_top = b[top_idx]
        # Weight by exp(tau * (b - b_max))
        w = np.exp(tau * (b_top - b_max))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w /= w_sum
    else:
        top_idx = np.arange(len(b))
        w = np.exp(tau * (b - b_max))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(b)) / len(b)
        else:
            w /= w_sum
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wi = w[i]
        if wi < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # grad[j] += 2 * wi * a[k-j]
        # a[k-j] for j in [j_start, j_end] = a[k-j_end : k-j_start+1][::-1]
        i_start = k - j_end
        i_end = k - j_start
        grad[j_start:j_end+1] += 2.0 * wi * a[i_start:i_end+1][::-1]
    return grad

def adam_optimize(a, deadline, lr=0.05, tau=10.0, k_top=50, 
                  beta1=0.9, beta2=0.999, eps=1e-8, decay=0.9995):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr_t = lr
    
    while time.monotonic() < deadline:
        t += 1
        b = fast_conv(a)
        b_max = float(b.max())
        
        if b_max < best_M - 1e-12:
            best_M = b_max
            best_a = a.copy()
        
        grad = grad_active(a, b, tau=tau, k_top=k_top)
        
        #Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        lr_t *= decay
    
    return best_a
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Adam optimization with active-set (top-k) subgradient
2. Multi-length, multi-start exploration
3. LP best-response polish with Anderson acceleration
4. Gerchberg-Saxton spectral initialization
Key innovation: O(n*k) gradient computation (vs O(n^2) for full smooth max),
enabling 10-100x more iterations than previous approach.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def grad_active(a, b, tau, k_top):
    n = len(a)
    b_max = float(b.max())
    if k_top < len(b):
        top_idx = np.argpartition(b, -k_top)[-k_top:]
        b_top = b[top_idx]
    else:
        top_idx = np.arange(len(b))
        b_top = b
    
    w = np.exp(tau * (b_top - b_max))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        wi = w[i]
        if wi < 1e-15:
            continue
        k = int(top_idx[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        i_start = k - j_end
        i_end = k - j_start
        grad[j_start:j_end+1] += 2.0 * wi * a[i_start:i_end+1][::-1]
    return grad

def compute_J(a):
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2 * n * b.max() / (s * s))

def adam_optimize(a, deadline, lr=0.05, tau=15.0, k_top=100,
                  beta1=0.9, beta2=0.999, eps=1e-8, decay=0.9998):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr_t = lr
    no_imp = 0
    
    while time.monotonic() < deadline:
        t += 1
        b = fast_conv(a)
        b_max = float(b.max())
        
        if b_max < best_M - 1e-13:
            best_M = b_max
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50000:
                break
        
        grad = grad_active(a, b, tau=tau, k_top=k_top)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        lr_t *= decay
    
    return best_a

def gs_init(n, rng, n_iters=200, beta=0.85):
    a = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    a = a / a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for _ in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-30)
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        M = fast_conv(a_new).max()
        if M < best_M:
            best_M = M
            best_a = a_new.copy()
        a = a_new
    return best_a

def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        # Resample parent
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            t_old = np.linspace(0, 1, len(parent))
            p = np.maximum(np.interp(t, t_old, parent), 0.0)
            inits.append(p)
        # Perturbed parent
        inits.append(np.maximum(parent_resampled(parent, n) + rng.normal(0, 0.03, n), 0.001))
        inits.append(np.maximum(parent_resampled(parent, n) + rng.normal(0, 0.08, n), 0.001))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # U-shape (high at edges)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        u = (2 * t - 1) ** 2
        a = u ** alpha + 0.1
        inits.append(a / a.sum())
    
    # Inverted U (high in middle)
    for alpha in [0.5, 1.0, 2.0]:
        a = np.maximum(1.0 - (2 * t - 1) ** 2, 0.01) ** alpha
        inits.append(a / a.sum())
    
    # Cosine
    for f in [1, 2, 3]:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * f * t)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    
    # Random smooth
    k = max(3, n // 30)
    kern = np.ones(k) / k
    a = np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same'), 0.01)
    inits.append(a / a.sum())
    
    # GS random
    inits.append(gs_init(n, rng, n_iters=100))
    
    # GS from parent
    if parent is not None:
        p = parent_resampled(parent, n)
        inits.append(gs_init_from(p, n, rng, n_iters=150))
    
    return inits

def parent_resampled(parent, n):
    if len(parent) == n:
        return parent.copy()
    t_old = np.linspace(0, 1, len(parent))
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, parent), 0.0)

def gs_init_from(p, n, rng, n_iters=150):
    a = np.maximum(p, 0.01).astype(np.float64)
    if len(a) != n:
        a = parent_resampled(a, n)
    a = a / a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for _ in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_new = 0.15 * mag + 0.85 * (mag.mean() + 1e-30)

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals

def lp_best_response(a, M):
    n = len(a)
    rows, cols, vals = build_A_sparse(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def lp_polish(a, deadline, max_iters=50, t_base=0.3):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break
        
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        t = t_base / (1 + 0.1 * it)
        a_new = np.maximum((1 - t) * a + t * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s
    
    return best_a
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Adam optimization with active-set (top-k) subgradient - O(n*k) per iter
2. Multi-phase learning rate schedule (exploration -> refinement -> fine-tuning)
3. Multi-length, multi-start exploration with GS spectral initialization
4. LP best-response polish with adaptive step size for final convergence
Key: 10-50x faster per-iteration than previous smooth-max approach, enabling
far more iterations and better convergence to the non-negativity constrained optimum.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2 * n * b.max() / (s * s))


def grad_active(a, b, tau, k_top):
    n = len(a)
    b_max = float(b.max())
    nb = len(b)
    if k_top < nb:
        top_idx = np.argpartition(b, -k_top)[-k_top:]
        b_top = b[top_idx]
    else:
        top_idx = np.arange(nb)
        b_top = b.copy()

    w = np.exp(tau * (b_top - b_max))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for i in range(len(top_idx)):
        wi = float(w[i])
        if wi < 1e-14:
            continue
        k = int(top_idx[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        i_start = k - j_end
        i_end = k - j_start
        grad[j_start:j_end + 1] += 2.0 * wi * a[i_start:i_end + 1][::-1]
    return grad


def grad_topk(a, b, k_top, linear_decay=True):
    """Gradient using top-k lags with optional linear decay weighting."""
    n = len(a)
    nb = len(b)
    k = min(k_top, nb)
    top_idx = np.argpartition(b, -k)[-k:]
    b_top = b[top_idx]
    sorted_order = np.argsort(b_top)[::-1]
    top_idx = top_idx[sorted_order]
    b_sorted = b_top[sorted_order]

    if linear_decay:
        ranks = np.arange(k, 0, -1, dtype=np.float64)
        w = ranks / ranks.sum()
    else:
        w = np.ones(k) / k

    grad = np.zeros(n)
    for i in range(k):
        wi = float(w[i])
        if wi < 1e-15:
            continue
        idx = int(top_idx[i])
        j_start = max(0, idx - n + 1)
        j_end = min(n - 1, idx)
        i_start = idx - j_end
        i_end = idx - j_start
        grad[j_start:j_end + 1] += 2.0 * wi * a[i_start:i_end + 1][::-1]
    return grad


def adam_optimize(a, deadline, lr=0.05, tau=15.0, k_top=100,
                  beta1=0.9, beta2=0.999, eps=1e-8, decay=0.9998):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr_t = lr
    no_imp = 0

    while time.monotonic() < deadline:
        t += 1
        b = fast_conv(a)
        b_max = float(b.max())

        if b_max < best_M - 1e-13:
            best_M = b_max
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 100000:
                break

        grad = grad_active(a, b, tau=tau, k_top=k_top)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        lr_t = max(lr_t * decay, lr * 1e-4)

    return best_a


def adam_multirate(a, deadline,
                   lr_A=0.08, tau_A=5.0, k_A=200, decay_A=0.9995,
                   lr_B=0.03, tau_B=15.0, k_B=100, decay_B=0.9998,
                   lr_C=0.01, tau_C=30.0, k_C=50, decay_C=0.99995,
                   frac_A=0.3, frac_B=0.4):
    """Three-phase Adam: broad exploration, refinement, fine-tuning."""
    total_time = deadline - time.monotonic()
    if total_time <= 0:
        return a
    dl_A = time.monotonic() + total_time * frac_A
    dl_B = time.monotonic() + total_time * (frac_A + frac_B)

    a = adam_optimize(a, dl_A, lr=lr_A, tau=tau_A, k_top=k_A, decay=decay_A)
    a = adam_optimize(a, dl_B, lr=lr_B, tau=tau_B, k_top=k_B, decay=decay_B)
    a = adam_optimize(a, deadline, lr=lr_C, tau=tau_C, k_top=k_C, decay=decay_C)
    return a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gs_spectral(a, n_iters=200, beta=0.85):
    a = np.maximum(a.astype(np.float64), 0.01)
    a = a / a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    for _ in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-30)
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        M = float(fast_conv(a_new).max())
        if M < best_M:
            best_M = M
            best_a = a_new.copy()
        a = a_new
    return best_a


def make_inits(n, rng, best_seq, parent):

    t = np.linspace(0, 1, n, endpoint=False)

        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.001))
        inits.append(np.maximum(p * np.exp(rng.normal(0, 0.15, n)), 0.001))

    if parent is not None and parent is not best_seq:
        p2 = resample_seq(parent, n)
        inits.append(p2.copy())

    inits.append(np.ones(n) / n)

    for alpha in [0.5, 1.0, 2.0, 3.0]:
        u = (2.0 * t - 1.0) ** 2
        a = u ** alpha + 0.05
        inits.append(a / a.sum())

    for alpha in [0.5, 1.0, 2.0]:
        a = np.maximum(1.0 - (2.0 * t - 1.0) ** 2, 0.01) ** alpha
        inits.append(a / a.sum())

    for f in [1, 2, 3, 5]:
        a = 0.5 + 0.4 * np.cos(2.0 * np.pi * f * t)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())

    k = max(3, n // 25)
    kern = np.ones(k) / k
    a = np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same'), 0.01)
    inits.append(a / a.sum())

    inits.append(gs_spectral(np.maximum(rng.normal(0.5, 0.3, n), 0.01), n_iters=100))

    if best_seq is not None:
        p = resample_seq(best_seq, n)
        inits.append(gs_spectral(p, n_iters=150, beta=0.9))

    return inits


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    n = len(a)
    rows, cols, vals = build_A_sparse(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_polish(a, deadline, max_iters=50, t_base=0.25):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 25:
                break

        g = lp_best_response(a, M)
        if g is None:
            break
        gs = float(g.sum())
        if gs < 1e-10:
            break
        g = g / gs

        t = t_base / (1.0 + 0.08 * it)
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = float(a_new.sum())
        if s < 1e-10:
            break
        a = a_new / s

    return best_a


def subgradient_with_momentum(a, deadline, lr=0.02, momentum=0.9, k_top=5):
    """Simple subgradient with heavy-ball momentum on top-k lags."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    v = np.zeros(n)
    no_imp = 0

    while time.monotonic() < deadline:
        b = fast_conv(a)
        b_max = float(b.max())
        if b_max < best_M - 1e-13:
            best_M = b_max
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50000:
                break

        grad = grad_topk(a, b, k_top=k_top, linear_decay=True)
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        lr *= 0.9999

    return best_a

                best_seq = a_lp2.copy()
                print(f"  LP polish 2: J={J_lp2:.8f}", flush=True)

    # Also try LP on second-best length if different
    if len(best_per_length) > 1:
        sorted_lens = sorted(best_per_length.keys(), key=lambda x: compute_J(best_per_length[x]))
        for n_alt in sorted_lens[1:3]:
            if time.monotonic() > deadline - 60:
                break
            if n_alt == best_seq.size:
                continue
            a_alt = best_per_length[n_alt]
            est_iter_ms = 2.5 * (n_alt / 128.0) ** 2
            max_iters_alt = min(30, max(5, int((deadline - time.monotonic() - 40) * 1000.0 / max(50.0, est_iter_ms))))
            if max_iters_alt > 3:
                a_alt_lp = lp_polish(a_alt, min(time.monotonic() + 40, deadline - 40),
                                     max_iters=max_iters_alt, t_base=0.25)
                J_alt = compute_J(a_alt_lp)
                if J_alt < best_score:
                    best_score = J_alt
                    best_seq = a_alt_lp.copy()
                    print(f"  Alt length {n_alt} LP: J={J_alt:.8f}", flush=True)

    print(f"\nPhase 2 complete. Best: J={best_score:.8f} n={best_seq.size}", flush=True)

    # Phase 3: Final fine-tuning with Adam (very small lr, high tau)
    p3_deadline = deadline - 15
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Final Adam Fine-Tuning", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and time.monotonic() < p3_deadline - 5:
        a_fin = adam_optimize(best_seq, p3_deadline, lr=0.005, tau=40.0, k_top=40,
                              beta1=0.9, beta2=0.999, eps=1e-8, decay=0.99998)
        J_fin = compute_J(a_fin)
        if J_fin < best_score:
            best_score = J_fin
            best_seq = a_fin.copy()
            print(f"  Fine-tune: J={J_fin:.8f}", flush=True)

    # Phase 4: Try subgradient with momentum on best (different optimizer might escape)
    if time.monotonic() < deadline - 10 and best_seq is not None:
        a_sg = subgradient_with_momentum(best_seq, min(time.monotonic() + 30, deadline - 8),
                                         lr=0.01, momentum=0.85, k_top=3)
        J_sg = compute_J(a_sg)
        if J_sg < best_score:
            best_score = J_sg
            best_seq = a_sg.copy()
            print(f"  Subgradient: J={J_sg:.8f}", flush=True)

    # Phase 5: Quick length exploration
    if time.monotonic() < deadline - 5 and best_seq is not None:
        n_best = best_seq.size
        for delta in [64, -64, 128, 256, -128]:
            if time.monotonic() > deadline - 5:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 4096:
                continue
            a_try = resample_seq(best_seq, n_try)
            a_try = np.maximum(a_try, 0.001)
            a_try = a_try / a_try.sum()
            a_try = adam_optimize(a_try, min(time.monotonic() + 15, deadline - 3),
                                  lr=0.03, tau=20, k_top=80, decay=0.9998)
            J_try = compute_J(a_try)
            if J_try < best_score:
                best_score = J_try
                best_seq = a_try.copy()
                print(f"  Length {n_try}: J={J_try:.8f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Verify
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Verified score: {final_score:.8f}", flush=True)

    if not np.isfinite(final_score) or final_score > best_score * 1.01:
        if parent is not None:
            print("Falling back to parent", flush=True)
            return [float(x) for x in parent]

    return [float(x) for x in best_seq]
```
