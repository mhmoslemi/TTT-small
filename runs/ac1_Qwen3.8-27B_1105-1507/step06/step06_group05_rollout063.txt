```python
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s*s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s
```

```python
def two_coord_optimize(a, i, j):
    n = len(a)
    vi = a[i]
    vj = a[j]
    c = vi + vj
    
    if c < 1e-15:
        return None
    
    conv = compute_conv(a)
    
    # Determine affected lags
    affected = []
    for k in range(max(0, i), min(2*n-1, i+n)):
        affected.append(k)
    for k in range(max(0, j), min(2*n-1, j+n)):
        if k not in set(affected):  # This is O(n) lookup, slow
            affected.append(k)
    
    # Better: use a boolean array
    affected_mask = np.zeros(2*n-1, dtype=bool)
    for k in range(max(0, i), min(2*n-1, i+n)):
        affected_mask[k] = True
    for k in range(max(0, j), min(2*n-1, j+n)):
        affected_mask[k] = True
    affected = np.where(affected_mask)[0]
    
    # Precompute max of unaffected lags
    if not affected_mask.all():
        max_unaffected = float(np.max(conv[~affected_mask]))
    else:
        max_unaffected = 0.0
    
    # Precompute the "base" values: conv[k] - old contributions for affected lags
    # Actually, let's just compute delta on the fly
    
    def obj_at_x(x):
        new_xj = c - x
        max_val = max_unaffected
        for k in affected:
            # Compute delta using the 4-candidate method
            candidates = set()
            if 0 <= k - i < n:
                candidates.add(i)
            if 0 <= k - j < n:
                candidates.add(j)
            ki = k - i
            if 0 <= ki < n and ki != i:
                candidates.add(ki)
            kj = k - j
            if 0 <= kj < n and kj != j:
                candidates.add(kj)
            
            delta = 0.0
            for ip in candidates:
                jp = k - ip
                if 0 <= jp < n:
                    old_val = a[ip] * a[jp]
                    new_ip = x if ip == i else (new_xj if ip == j else a[ip])
                    new_jp = x if jp == i else (new_xj if jp == j else a[jp])
                    new_val = new_ip * new_jp
                    delta += new_val - old_val
            
            new_conv_k = conv[k] + delta
            if new_conv_k > max_val:
                max_val = new_conv_k
        
        return max_val
    
    # Ternary search
    lo, hi = 0.0, c
    for _ in range(35):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_x(m1)
        f2 = obj_at_x(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    x_opt = (lo + hi) / 2.0
    
    # Check if there's actually an improvement
    obj_current = float(np.max(conv))
    obj_new = obj_at_x(x_opt)
    if obj_new >= obj_current * (1 - 1e-10):
        return None
    
    a_new = a.copy()
    a_new[i] = x_opt
    a_new[j] = c - x_opt
    return a_new
```

**Idea: Spectral method with concave envelope**
The objective `max_k (a*a)[k]` can be bounded above by the integral of the power spectrum. Specifically, by Parseval's theorem:
`sum_k (a*a)[k]^2 = sum_k |A(k)|^2` where A is the DFT of a.

But we want to minimize the MAX, not the sum of squares. However, if we can make the power spectrum very flat, the time-domain convolution should be more uniform.

**Idea: Use the relationship between the peak and the rest**
`sum_k (a*a)[k] = (sum a)^2 = 1` (when normalized)
So `max_k (a*a)[k] >= 1/(2n-1)` (pigeonhole)

To get close to this bound, the convolution should be as flat as possible. The ideal convolution would be a constant 1/(2n-1) at all lags, which is impossible for a non-negative sequence (since the convolution at lag 0 is a[0]^2 >= 0, and at lag 2n-2 is a[n-1]^2 >= 0, while the total is 1).

**Idea: Start from a "Barker-like" sequence**
Barker sequences have the property that their aperiodic autocorrelation has low sidelobes. While they're typically defined for ±1 sequences, we can use their structure as inspiration for non-negative sequences.

For a non-negative sequence, the analog would be a sequence that is close to constant (to minimize the peak of the autocorrelation). But we need to break the symmetry slightly to reduce the peak.

**Idea: Use the fact that the problem is equivalent to minimizing the peak of the "probability distribution" of the sum of two independent random variables**

"""
Multi-strategy optimization combining LP alternating, two-coordinate block
coordinate descent (exact 1D search), soft-max gradient descent, and multi-n
exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key new idea: two-coordinate BCD performs exact 1D optimization over pairs of
coordinates, exploiting the structure that the objective is a max of at-most-quadratic
functions in 1D, enabling precise local improvements.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def cubic_line_search(f, x0, x1, f0, f1, tol=0.01, max_iter=20):
    """Cubic interpolation line search for minimum of f on [x0, x1]."""
    if abs(x1 - x0) < 1e-8:
        return (x0 + x1) / 2.0
    for _ in range(max_iter):
        if abs(x1 - x0) < tol:
            break
        # Cubic interpolation
        dx = x1 - x0
        df = f1 - f0
        if abs(df) < 1e-15:
            t = 0.5
        else:
            t = df / (2.0 * df / dx) if dx != 0 else 0.5
        t = max(0.1, min(0.9, t))
        x_new = x0 + t * dx
        f_new = f(x_new)
        if f_new < f0 and f_new < f1:
            if f_new < f0:
                x1, f1 = x0, f0
            x0, f0 = x_new, f_new
        else:
            if f0 < f1:
                x1, f1 = x_new, f_new
            else:
                x0, f0 = x_new, f_new
    return (x0 + x1) / 2.0
