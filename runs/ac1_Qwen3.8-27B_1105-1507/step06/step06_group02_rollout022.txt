2. Finding the a_i that minimizes max_k(C_k + a_i * a_{k-i}) (with the quadratic term for k=2i)

This is a 1D minimax problem that can be solved by bisection or by finding the "crossing point".

The function g(a_i) = max_k(C_k + a_i * B_k + (k==2i)*a_i^2) is a convex piecewise-linear (or piecewise-quadratic) function of a_i. Its minimum can be found efficiently.

2. Compute C_k and B_k for all k
3. Find the optimal a_i that minimizes max_k(C_k + a_i * B_k + quadratic)
4. Update a_i

This is O(n) per coordinate update (to compute C_k and B_k) and O(n log n) or O(n) to find the optimal a_i. So one full pass over all coordinates is O(n^2).

This is comparable to the LP in cost, but:
- No solver overhead
- Can be warm-started naturally
- Can be combined with the smooth optimization

    # So: grad[i]/2 = conv(a, q)[2n-2-i] where q = p[::-1]
    # Or equivalently: grad/2 = conv(a, p[::-1])[2n-2-i] for i=0..n-1
    # = conv(a, p[::-1])[n-1 : 2n-2][::-1]  (slicing and reversing)
    
    # Let me verify: conv(a, q) has length n + (2n-1) - 1 = 3n-2
    # q has length 2n-1
    # conv(a, q)[k] = sum_{j=max(0,k-(2n-2))}^{min(n-1,k)} a[j]*q[k-j]
    # 
    # For k = 2n-2-i (where i is in [0, n-1], so k is in [n-1, 2n-2]):
    # conv(a,q)[2n-2-i] = sum_{j} a[j]*q[2n-2-i-j]
    # q[2n-2-i-j] = p[i+j] (since q[m] = p[2n-2-m], so q[2n-2-i-j] = p[2n-2-(2n-2-i-j)] = p[i+j])
    # Valid range: 0 <= i+j <= 2n-2, and 0 <= j <= n-1
    # Since i >= 0 and j >= 0, i+j >= 0 always.
    # i+j <= 2n-2: since i <= n-1 and j <= n-1, i+j <= 2n-2. Always valid!
    # So: conv(a,q)[2n-2-i] = sum_{j=0}^{n-1} a[j]*p[i+j] = grad[i]/2 ✓
    
    # Great! So the algorithm is:
    # 1. Compute p = softmax(tau*(b - b_max))
    # 2. Compute q = p[::-1] (reverse p)
    # 3. Compute conv_aq = convolve(a, q) via FFT
    # 4. grad = 2 * conv_aq[n-1:2n-1][::-1]  (take the right slice and reverse)
    
    # Wait, let me double-check the indices.
    # conv_aq has length n + (2n-1) - 1 = 3n - 2
    # We want conv_aq[2n-2-i] for i = 0, 1, ..., n-1
    # That's indices: 2n-2, 2n-3, ..., 2n-2-(n-1) = n-1
    # So: grad[i] = 2 * conv_aq[2n-2-i]
    # In array notation: grad = 2 * conv_aq[n-1:2n-1][::-1]
    # conv_aq[n-1:2n-1] gives elements at indices n-1, n, ..., 2n-2
    # Reversed: 2n-2, 2n-3, ..., n-1
    # So grad[0] = 2*conv_aq[2n-2], grad[1] = 2*conv_aq[2n-3], ..., grad[n-1] = 2*conv_aq[n-1]
    # ✓
    
    # Let me implement this properly.
    
    # Actually, I realize I should just compute it directly for clarity and correctness:
    
    # Method: use the fact that grad[i] = 2 * sum_{m=0}^{n-1} a[m] * p[m+i]
    # This is a valid cross-correlation. I'll compute it using FFT.
    
    size2 = 1
    while size2 < 3 * n - 2:
        size2 <<= 1
    fa2 = np.fft.rfft(a, size2)
    fq = np.fft.rfft(p[::-1], size2)  # q = p reversed, padded to size2
    conv_aq = np.fft.irfft(fa2 * fq, size2)[:3*n-2]
    grad = 2.0 * conv_aq[n-1:2*n-1][::-1]
    
    return b_max, grad, b, p


def nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=300.0,
                     lr_init=0.3, lr_end=0.005):
    """Nesterov-accelerated gradient descent on smooth max of autocorrelation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        b_max, grad, b, p = smooth_max_and_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def coord_update(a, i, c):

def refined_smooth_descend(a, deadline, max_iters=5000):
    """
    Refined smooth descent with:
    - Adaptive tau based on effective support size
    - Line search for step size
    - Restarts when progress stalls
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    
    # Adaptive parameters
    tau = 5.0
    lr = 0.3
    beta = 0.9
    y = a.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b_max, grad, b, p = smooth_max_and_grad(y, tau)
        
        # Adaptive tau: increase tau when effective support is too large
        eff_support = 1.0 / (p * p).sum()
        target_support = max(n * 0.05, 2.0)
        if eff_support > target_support * 1.5:
            tau = min(tau * 1.2, 5000.0)
        elif eff_support < target_support * 0.5:
            tau = max(tau / 1.1, 1.0)
        
        # Line search
        best_lr = lr
        best_obj = b_max
        for lr_try in [lr * 2, lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
            a_try = np.maximum(y - lr_try * (grad + beta * (y - a)), 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try /= s_try
            M_try = fast_conv(a_try).max()
            if M_try < best_obj:
                best_obj = M_try
                best_lr = lr_try
        
        lr = best_lr * 1.0  # maintain or slightly adjust
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - best_lr * (grad + beta * (y - a)), 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s_new
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
        
        M_cur = fast_conv(a).max()
        if M_cur < best_M * (1 - 1e-10):
            best_M = M_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Restart with higher tau
                tau = min(tau * 2, 5000.0)
                if tau >= 5000:
                    break
                no_improve = 0
    
    return best_a


def gs_flat_spectral(init, n_iters=200, beta=0.8):
    """Gerchberg-Saxton: flatten power spectrum."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        # Compress dynamic range
        power = 0.3 + 0.5 * beta
        mag_new = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
