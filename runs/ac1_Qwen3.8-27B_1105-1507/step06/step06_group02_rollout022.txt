def gs_flat_spectral(init, n_iters=200, beta=0.7):
    """Gerchberg-Saxton: iteratively flatten the power spectrum."""
    a = np.maximum(init.astype(np.float64), 1e-10)
    s = a.sum()
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress dynamic range
        mag_min = mag.min()
        mag_range = mag.max() - mag_min
        if mag_range < 1e-15:
            break
        mag_norm = (mag - mag_min) / mag_range
        mag_new = mag_min + mag_range * (1 - beta) + mag_range * beta * (1.0 / (1.0 + mag_norm * 10) / (1.0/11))
        # Simpler: power compression
        power = 0.3 + 0.4 * (1 - beta)
        mag_alt = (mag ** power)
        mag_final = mag_alt / mag_alt.max() * mag.max()
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:

        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate

    p = exp_b / exp_b.sum()

    """Find t in [0,1] minimizing max_k [(1-t)^2*ba[k] + 2t(1-t)*bag[k] + t^2*bg[k]]"""
    # Precompute coefficients for each k:
    # f_k(t) = ba[k]*(1-t)^2 + 2*bag[k]*t*(1-t) + bg[k]*t^2
    #        = ba[k] + (2*bag[k] - 2*ba[k])*t + (bg[k] - 2*bag[k] + ba[k])*t^2
    A = ba  # constant term
    B = 2*(bag - ba)  # linear coefficient
    C = bg - 2*bag + ba  # quadratic coefficient
    
    def max_quad(t):
        return np.maximum(A + B*t + C*t*t)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

            hi = d; d = c; fd = fc
            c = hi - gr * (hi - lo); fc = max_quad(c)
        else:
            lo = c; c = d; fc = fd
            d = lo + gr * (hi - lo); fd = max_quad(d)
    t_opt = (lo + hi) / 2
    return t_opt

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton spectral initialization
2. LP best-response with exact golden-section line search on J
3. Multi-start with diverse initializations across lengths
4. Adaptive M-tightening and perturbation restarts
Key improvement: exact line search along LP direction guarantees
monotone J decrease and faster convergence to the fixed point.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv2(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_spectral(init, n_iters=300, beta=0.7):
    a = np.maximum(init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_min = mag.min()
        mag_range = mag.max() - mag_min
        if mag_range < 1e-15:
            break
        power = 0.3 + 0.5 * (1.0 - beta)
        mag_c = (mag ** power)
        mag_c = mag_c / mag_c.max() * mag.max()
        F_new = np.exp(1j * phase) * mag_c
        a_new = np.fft.irfft(F_new, size)[:n]
        a_cand = np.maximum(a_new, 0)
        sc = a_cand.sum()
        if sc < 1e-15:
            break
        a_cand = a_cand / sc
        M_c = fast_conv(a_cand).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_cand.copy()
        a = a_cand
    return best_a


def smooth_grad_J(a, tau=100.0):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = a.sum()
    # grad_b[i] = 2 * sum_j p[i+j] * a[j] for valid indices
    grad_b = np.zeros(n)
    for j in range(n):
        hi = min(n, 2 * n - 1 - j)
        if hi > 0:
            grad_b[:hi] += 2.0 * a[j] * p[j:j + hi]
    # J = 2n * b_max / s^2
    # dJ/d a_i = 2n * (s * grad_b_i - 2 * b_max) / s^3
    J = 2.0 * n * b_max / (s * s)
    grad_J = 2.0 * n * (s * grad_b - 2.0 * b_max) / (s * s * s)
    return J, grad_J


def nesterov_smooth(a, deadline, max_iters=3000, tau_start=5.0, tau_end=300.0,
                     lr_init=0.3, lr_end=0.005):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta_n = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        J_y, grad_y = smooth_grad_J(y, tau)
        b_y = fast_conv(y)
        if b_y.max() < best_M * (1 - 1e-10):
            best_M = b_y.max()
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_n * beta_n)) / 2.0
        a_new = np.maximum(y - lr * (grad_y + beta_n * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new
    return best_a


def build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, prev_solution=None):
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def line_search_golden(ba, bag, bg, n_evals=50):
    """Find t in [0,1] minimizing max_k [(1-t)^2*ba[k] + 2t(1-t)*bag[k] + t^2*bg[k]]."""
    A = ba
    B = 2.0 * (bag - ba)
    C = bg - 2.0 * bag + ba

    def max_quad(t):
        vals = A + B * t + C * t * t
        return float(np.max(vals))

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = max_quad(c)
    fd = max_quad(d)
    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = max_quad(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = max_quad(d)
    t_opt = (lo + hi) / 2.0
    return t_opt


def run_lp_linesearch(a, deadline, max_iters=500, m_shrink=1.0):
    """LP best response with exact golden-section line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = 2.0 * n * fast_conv(a).max()

    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        ba = fast_conv(a)
        M = float(ba.max()) * m_shrink
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        bag = fast_conv2(a, g)
        bg = fast_conv(g)
        t_opt = line_search_golden(ba, bag, bg, n_evals=45)
        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
            break
        a_new = a_new / s
        M_new = fast_conv(a_new).max()
        if M_new < best_J * (1 - 1e-12):
            best_J = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        a = a_new
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(gs_flat_spectral(p, n_iters=100, beta=0.75))
    inits.append(np.ones(n))
    for alpha in [0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t ** alpha + (1 - t) ** alpha) / 2.0, 0.01))
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-5 * (t - 0.5) ** 2) + 0.05)
    inits.append(0.5 + 0.3 * np.sin(np.pi * t) + 0.1 * np.cos(2 * np.pi * t))
    # Asymmetric
    inits.append(np.maximum(0.2 + 0.8 * t, 0.01))
    inits.append(np.maximum(0.2 + 0.8 * (1 - t), 0.01))
    # Random smooth
    k = max(5, n // 16)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_flat_spectral(a_rand, n_iters=150, beta=0.8))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:

            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
