Optimization using mirror descent (multiplicative weight updates) with

import numpy as np
from scipy.fft import rfft, irfft

# --- Core computation functions ---

def fft_conv_square(a):
    """Compute a*a (linear convolution) using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def obj_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = fft_conv_square(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def gradient_smoothed(a, conv, tau):
    """
    Compute smoothed subgradient of max(conv) using softmax weights.
    Returns gradient vector of length n.
    Uses FFT for efficiency.
    """
    n = len(a)
    # Softmax weights over convolution values
    shifted = (conv - np.max(conv)) / tau
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w /= np.sum(w)
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # = 2 * (cross-correlation of w and a) at lag i
    # Using FFT: corr[i] = sum_j a[j] * w[i+j]
    # ifft(fft(a) * conj(fft(w_padded)))[i] = sum_j a[j] * w[i+j]
    
    fft_size = 1
    while fft_size < 2*n - 1:
        fft_size <<= 1
    
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    
    A = rfft(a, fft_size)
    W = rfft(w_padded, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    
    return 2.0 * corr[:n]

def mirror_descent_step(a, grad, eta):
    """Multiplicative weight update: a *= exp(-eta*grad), normalize."""
    log_a = np.log(np.maximum(a, 1e-300))
    log_a -= eta * grad
    # Normalize (softmax)
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    s = np.sum(a_new)
    if s < 1e-300:
        return a.copy()
    return a_new / s

def targeted_peak_reduction(a, num_peaks=3, time_budget=30, seed=42):
    """
    Directly target the top K peaks in the convolution.
    For each peak, compute its specific gradient and take a coordinated step.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = obj_value(a)
    start = time.monotonic()
    eta = 0.1
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        conv = fft_conv_square(a)
        
        # Find top K peak positions
        top_k = np.argsort(conv)[-num_peaks:]
        
        # Weight: inverse of (conv value - second max + epsilon) to focus on the highest
        cvals = conv[top_k]
        weights = 1.0 / (cvals - conv.min() + 1e-10)
        weights /= np.sum(weights)
        
        # Combined gradient: weighted sum of individual peak gradients
        grad = np.zeros(n)
        for wk, k in zip(weights, top_k):
            # grad from peak at k: grad[i] = 2*a[k-i] if 0<=k-i<n
            valid = np.arange(max(0, k-n+1), min(k+1, n))
            for i in valid:
                grad[i] += 2.0 * wk * a[k-i]
        
        # Mirror descent step
        grad_centered = grad - np.mean(grad)
        a_new = mirror_descent_step(a, grad_centered, eta)
        obj_new = obj_value(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new

                eta = 0.1
