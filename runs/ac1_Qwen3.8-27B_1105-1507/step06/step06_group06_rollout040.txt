df/da[i] = 2n/S^2 * dM/da[i] - 4n*M/S^3 * 1

Since we're working with normalized a (S=1):
df/da[i] = 2n * dM/da[i] - 4n * M

dM/da[i] = 2 * sum_j w[i+j] * a[j]

So: df/da[i] = 4n * (sum_j w[i+j] * a[j]) - 4n * M = 4n * (sum_j w[i+j] * a[j] - M)

This is clean! The gradient direction at each point i is:
- If the weighted average of a's "partners" at position i is above M, decrease a[i]
- If below M, increase a[i]

This makes intuitive sense: we want to reduce the peaks of the convolution.

For the softplus parameterization: a[i] = softplus(x[i]) = log(1 + exp(x[i]))
da[i]/dx[i] = sigmoid(x[i])
dx[i] gradient = (df/da[i]) * sigmoid(x[i])

But actually, since we're normalizing a (dividing by sum), the parameterization complicates things. Let me just work with a directly and project to {a >= 0, sum(a) = 1} after each step.

**Algorithm:**
1. Initialize a (normalized, non-negative)
2. For each iteration:
   a. Compute c = conv(a, a)
   b. Compute soft-max weights w
   c. Compute gradient g[i] = 4n * (sum_j w[i+j] * a[j] - M)
   d. Update: x = x - lr * g (in Adam)
   e. Project: a = max(0, x) / sum(max(0, x))
3. Track best objective

- a_new = IFFT(A).real
- Project to non-negative and normalize

"""
Adam-based optimization with soft-max gradient and spectral projection
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key innovations: (1) proper Adam optimizer with analytically-computed soft-max
gradient via FFT, (2) Gerchberg-Saxton style spectral projection for escaping
local minima, (3) adaptive temperature annealing, (4) multi-length exploration
with resampling.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def compute_gradient_softmax(a, tau):
    """
    Compute gradient of soft-max objective.
    Returns (gradient array of length n, soft_max_value, weights of length 2n-1).
    
    The gradient of f = 2n*M/S^2 (with S=1) w.r.t. a[i] is:
    4n * (sum_j w[i+j]*a[j] - M)
    where w[k] = exp(c[k]/tau) / sum exp(c/tau), M = tau*log(sum exp(c/tau))
    """
    n = len(a)
    c = compute_conv(a)
    c_len = len(c)  # 2n-1
    
    # Soft-max computation (numerically stable)
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / tau)
    sum_exp = np.sum(exp_c)
    M = c_max + tau * np.log(sum_exp)
    w = exp_c / sum_exp
    
    # Gradient: g[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1
    # This is a cross-correlation-like operation
    # g[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j] where w[i+j]=0 if i+j >= 2n-1
    
    # Use FFT for efficiency
    # g[i] = sum_j a[j] * w[i+j]
    # This is (a cross-correlated with w) truncated to n elements
    # = ifft(fft(a) * conj(fft(w))) ... let me think
    # Actually: sum_j a[j] * w[i+j] = (a * w_reversed)[i + n - 1] ... 
    # Let me use a different approach: 
    # g[i] = sum_j a[j] * w[i+j]
    # Let's define v = w (length 2n-1)
    # g[i] = sum_{j=0}^{n-1} a[j] * v[i+j]
    # This is a "sliding window dot product" = cross-correlation
    # cross_corr[i] = sum_j a[j] * v[i+j]
    # = sum_j a[j] * conj(v[i+j]) (since real)
    # = ifft(fft(a_padded) * conj(fft(v_padded))) -- not quite
    # 
    # Actually, cross-correlation of a and v:
    # xcorr[i] = sum_j a[j] * v[i+j]
    # = (a ⋆ v)[i] where ⋆ denotes cross-correlation
    # 
    # Via FFT: xcorr = ifft(fft(a,N) * conj(fft(v,N)))
    # But we need to be careful about padding and alignment.
    
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1
    
    fa_full = fft(a, N_fft)
    fv_full = fft(w, N_fft)
    
    # cross-correlation: sum_j a[j] * v[i+j]
    # = ifft(fft(a) * conj(fft(v))) gives: x[i] = sum_j a[j] * v[(j-i) mod N]
    # That's not quite what we want. We want sum_j a[j] * v[i+j].
    # 
    # Let me define: h = flip(v) (so h[k] = v[c_len-1-k])
    # Then sum_j a[j] * v[i+j] = sum_j a[j] * h[c_len-1-i-j] = (a * h)[c_len-1-i]
    # where * is linear convolution.
    # 
    # So: g[i] = (a * h)[c_len-1-i] = conv(a, h)[c_len-1-i]
    # conv(a, h) has length n + c_len - 1 = n + 2n-2 = 3n-2
    # We need indices c_len-1-i = 2n-2-i for i=0..n-1, so indices n-1..2n-2
    
    h = w[::-1]  # flip
    # conv(a, h) via FFT
    conv_ah = irfft(fa_full * fft(h, N_fft), N_fft)
    # conv_ah[k] = sum_j a[j] * h[k-j] for valid j
    # We need g[i] = 2 * conv_ah[2n-2-i]
    # Actually: conv_ah[m] = sum_j a[j] * h[m-j] = sum_j a[j] * w[c_len-1-(m-j)]
    # = sum_j a[j] * w[2n-2-m+j]
    # We want sum_j a[j] * w[i+j], so set 2n-2-m+j = i+j => m = 2n-2-i
    # So g[i] = 2 * conv_ah[2n-2-i]
    
    indices = 2 * n - 2 - np.arange(n)  # [2n-2, 2n-3, ..., n-1]
    g = 2.0 * conv_ah[indices]
    
    # Full gradient of objective (with S=1):
    # df/da[i] = 4n * (g[i]/2 - M) = 4n * (sum_j w[i+j]*a[j] - M)
    # Wait: dM/da[i] = sum_k w[k] * dc[k]/da[i] = 2 * sum_j w[i+j]*a[j] = g[i] (as computed)
    # df/da[i] = 2n * dM/da[i] - 4n*M = 2n*g[i] - 4n*M
    # But since S=1: df/da[i] = 2n * (2*sum_j w[i+j]*a[j]) - 4n*M
    # = 4n * (sum_j w[i+j]*a[j] - M)
    
    grad = 4.0 * n * (g / 2.0 - M)
    
    return grad, M, w, c


def spectral_projection(a, beta=0.5, width=5.0, iterations=50):
    """
    Gerchberg-Saxton style projection:
    - In freq domain: concentrate energy at DC
    - In time domain: enforce non-negativity
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    a_cur = a.copy()
    
    for _ in range(iterations):
        # Frequency domain: push energy toward DC
        A = fft(a_cur, N)
        freqs = np.arange(N)
        # DC concentration filter
        dc_filter = np.exp(-(freqs.astype(float) ** 2) / (width ** 2))
        # Preserve DC, reduce others
        A_mod = A.copy()
        A_mod[0] = A[0]  # keep DC exact
        # For non-DC, blend toward the DC-filtered version
        A_mod[1:] = (1 - beta) * A[1:] + beta * A[1:] * dc_filter[1:] / max(1e-10, dc_filter[1:][0] if N > 1 else 1.0)
        # Simpler: just scale non-DC by a factor
        A_mod[1:] = (1 - beta) * A[1:]
        
        a_new = ifft(A_mod, N).real[:n]
        a_cur = project_nonneg_sum1(a_new)
    
    return a_cur


def spectral_projection_v2(a, alpha=0.3, iterations=30):
    """
    Alternative spectral projection: modify the magnitude spectrum to be
    more uniform (flat power spectrum = flat autocorrelation in a sense).
    Actually we want the OPPOSITE: we want |A|^2 to be peaked at DC.
    So we reduce non-DC magnitudes.
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    a_cur = a.copy()
    
    for _ in range(iterations):
        A = fft(a_cur, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Reduce non-DC components
        mag_new = mag.copy()
        mag_new[0] = mag[0]  # keep DC
        # Apply power compression to non-DC
        max_non_dc = np.max(mag_new[1:]) if N > 1 else 1.0
        if max_non_dc > 1e-10:
            mag_new[1:] = max_non_dc * (mag_new[1:] / max_non_dc) ** (1 + alpha)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = ifft(A_new, N).real[:n]
        a_cur = project_nonneg_sum1(a_new)
    
    return a_cur


def adam_optimize(a, n_iters, tau_init, tau_min, lr_init, time_budget,
                  spectral_interval=200, spectral_iter=20):
    """
    Adam optimizer with soft-max gradient for the self-convolution peak minimization.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam state
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Precompute FFT size
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature
        tau = tau_init * (tau_min / tau_init) ** (it / max(1, n_iters))
        tau = max(tau, 1e-6)
        
        # Compute gradient
        grad, M, w, c = compute_gradient_softmax(a, tau)
        
        # Adam update
        lr = lr_init / (1.0 + it * 0.001)  # simple decay
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_try = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_nonneg_sum1(a_try)
        
        # Periodic spectral projection
        if (it + 1) % spectral_interval == 0:
            a_spec = spectral_projection_v2(a, alpha=0.5, iterations=spectral_iter)
            obj_spec = compute_obj(a_spec)
            obj_cur = compute_obj(a)
            if obj_spec < obj_cur:
                a = a_spec
        
        # Track best
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

def coordinate_descent_refine(a, n_rounds=5, time_budget=30.0):
    """
    Simple coordinate descent: for each coordinate, try increasing/decreasing
    to reduce the max convolution peak.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for rnd in range(n_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        c = compute_conv(a)
        k_peak = int(np.argmax(c))
        
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            # Try adjusting a[i]
            # The partners of a[i] in the peak are a[k_peak - i] (if valid)
            j = k_peak - i
            if 0 <= j < n:
                partner = a[j]
                # Decreasing a[i] reduces c[k_peak] by 2*a[i]*partner
                # Try decreasing
                for frac in [0.1, 0.3, 0.5]:
                    a_try = a.copy()
                    a_try[i] *= (1 - frac)
                    a_try = project_nonneg_sum1(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break
    
    return best_a, best_obj


def peak_chasing(a, n_steps=100, time_budget=30.0):
    """
    Iteratively chase the peak: find the peak, reduce contributions to it.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        c = compute_conv(a)
        k_peak = int(np.argmax(c))
        
        # Find all pairs (i, j) with i+j = k_peak that contribute to c[k_peak]
        # c[k_peak] = sum_i a[i]*a[k_peak-i]
        # Gradient direction: reduce a[i] where a[k_peak-i] is large
        
        contributions = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                contributions[i] = a[j]
            j2 = i + k_peak  # for the other side (if k_peak >= n)
            if 0 <= i + (2*n-2-k_peak) < n:
                contributions[i] += a[2*n-2-k_peak - i] if 0 <= 2*n-2-k_peak-i < n else 0
        
        # Reduce a in proportion to contributions
        total_contrib = np.sum(contributions)
        if total_contrib < 1e-10:
            break
        
        # Direction: decrease a where contribution is high
        direction = contributions - np.mean(contributions)
        
        for alpha in [0.05, 0.02, 0.01, 0.005]:
            a_try = a - alpha * direction
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break
    
    return best_a, best_obj

                                                time_budget=min(remaining2, 30.0))
        if obj2b < best_score:
            best_score = obj2b
            best_seq = a2b.copy()
            print(f"  Coord descent improved: {best_score:.8f}")
    
    # === Phase 3: Multi-length exploration (15%) ===
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    print(f"Best so far: {best_score:.8f}")
    phase3_budget = min(200.0, (deadline - start_time) * 0.15)
    phase3_start = time.monotonic()
    
    n_cur = len(best_seq)
    n_tries = [n_cur - 50, n_cur - 100, n_cur + 50, n_cur + 100,
               n_cur + 200, n_cur - 200, 800, 900, 1000, 1200]
    n_tries = sorted(set(n for n in n_tries if 100 <= n <= 2500))
    
    per_n_time = max(5.0, phase3_budget / len(n_tries))
    
    for n_try in n_tries:
        if time.monotonic() > phase3_start + phase3_budget:
            break
        if n_try == n_cur:
            continue
        
        a_res = resample_sequence(best_seq, n_try)
        a_res = project_nonneg_sum1(a_res)
        
        # Quick Adam on resampled
        tau_init = max(compute_conv(a_res).max() * 0.2, 0.01)
        tau_min = max(compute_conv(a_res).max() * 0.01, 0.001)
        a_opt, obj_opt = adam_optimize_v2(
            a_res, n_iters=100000,
            tau_init=tau_init, tau_min=tau_min,
            lr_init=0.03, time_budget=min(per_n_time, 20.0)
        )
        
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = n_try
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # === Phase 4: Final intensive polish (remaining) ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining4 = deadline - time.monotonic()
    
    if remaining4 > 20 and best_seq is not None:
        n_fin = len(best_seq)
        
        # Multiple Adam runs with different hyperparameters
        configs = [
            {'lr': 0.02, 'tau_mult': 0.2, 'tau_min_mult': 0.005},
            {'lr': 0.01, 'tau_mult': 0.5, 'tau_min_mult': 0.01},
            {'lr': 0.05, 'tau_mult': 0.1, 'tau_min_mult': 0.003},
            {'lr': 0.03, 'tau_mult': 0.3, 'tau_min_mult': 0.008},
        ]
        
        per_config = max(5.0, remaining4 / (len(configs) + 2))
        
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            
            c_peak = compute_conv(best_seq).max()
            a_p, obj_p = adam_optimize_v2(
                best_seq,
                n_iters=200000,
                tau_init=max(c_peak * cfg['tau_mult'], 0.005),
                tau_min=max(c_peak * cfg['tau_min_mult'], 0.0005),
                lr_init=cfg['lr'],
                time_budget=min(per_config, rem - 5)
            )
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Config {ci}: {best_score:.8f}")
        
        # Final spectral projection
        rem = deadline - time.monotonic()
        if rem > 10:
            a_spec = spectral_projection_v2(best_seq, alpha=1.0, iterations=50)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                print(f"  Final spectral: {best_score:.8f}")
            
            # Quick peak chase
            a_pc, obj_pc = peak_chasing(best_seq, n_steps=100, time_budget=min(10, rem - 5))
            if obj_pc < best_score:
                best_score = obj_pc
                best_seq = a_pc.copy()
                print(f"  Final peak chase: {best_score:.8f}")
    
    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200) / 200
    
    # Ensure valid output
    best_seq = np.clip(best_seq, 0, 1000)
    if np.sum(best_seq) < 0.01:
        best_seq = np.ones(100)
    
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    print(f"\n=== FINAL: n={len(result)}, score={val:.8f} ===")
    
    if val < np.inf and val < best_score:
        best_score = val
    
    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s")
    
    return result
