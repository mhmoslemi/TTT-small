"""
Optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) FFT-based subgradient with momentum and adaptive step size,
(2) LP-based alternating optimization, (3) frequency-domain power spectrum
smoothing, (4) multi-length population-based search.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_gradient(a, conv=None):
    """
    Compute the (softmax-smoothed) subgradient of max_k conv(a,a)[k].
    Returns gradient vector of length n.
    """
    n = len(a)
    if conv is None:
        conv = compute_conv(a)
    L = len(conv)  # 2n-1
    
    # Use argmax subgradient (exact, not smoothed)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    # grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    if i_max >= i_min:
        indices = np.arange(i_min, i_max + 1)
        a_indices = k_star - indices
        grad[indices] = 2.0 * a[a_indices]
    
    return grad


def compute_smooth_gradient(a, conv, tau=0.1):
    """
    Compute the softmax-smoothed subgradient.
    grad[i] = 2 * sum_k weights[k] * a[k-i] for valid k-i
    Computed via FFT in O(n log n).
    """
    n = len(a)
    L = len(conv)  # 2n-1
    
    # Softmax weights
    cmax = float(np.max(conv))
    if cmax < 1e-300:
        return np.zeros(n)
    shifted = (conv - cmax) / tau
    weights = np.exp(np.clip(shifted, -50, 50))
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros(L)
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= wsum
    
    # grad[i] = 2 * sum_k weights[k] * a[k-i] for 0 <= k-i < n
    # = 2 * corr[weights, a][L-1-i] where corr is the cross-correlation
    # Computed as: corr[m] = sum_k weights[k] * a[m - L + 1 + k]
    # We want m = L-1-i, so corr[L-1-i] = sum_k weights[k] * a[k-i] ✓
    
    # Use FFT: corr = ifft(fft(weights_rev) * fft(a))
    # where weights_rev = weights[::-1]
    size = 1
    while size < L + n - 1:
        size <<= 1
    w_rev = weights[::-1]
    fw = rfft(w_rev, size)
    fa = rfft(a, size)
    corr = irfft(fw * fa, size)
    
    # grad[i] = 2 * corr[L-1-i] for i=0,...,n-1
    # So grad = 2 * corr[L-1 : L-n : -1] = 2 * corr[L-n : L][::-1]
    grad = 2.0 * corr[L - n:L][::-1]
    
    return grad


def subgradient_optimize(a, max_steps=10000, time_budget=60.0, tau_init=0.05, 
                         step_init=None, momentum=0.9, verbose=False):
    """
    Subgradient optimization with momentum and adaptive step size.
    a is assumed to be on the simplex (sum=1, non-negative).
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    
    if step_init is None:
        step = 0.5 / np.sqrt(n)
    
    buf = np.zeros(n)
    start = time.monotonic()
    stall_count = 0
    
    for it in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        if it % 100 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    SG it{it}: obj={cur_obj:.8f}, step={step:.6f}, elapsed={elapsed:.1f}s")
        
        # Compute gradient
        conv = compute_conv(a)
        tau = tau_init * (1 + it / 200)  # slowly increase tau
        grad = compute_smooth_gradient(a, conv, tau)
        
        # Center gradient (tangent to simplex)
        grad_centered = grad - np.mean(grad)
        
        # Momentum step
        buf = momentum * buf - step * grad_centered
        a_new = a + buf
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj_new = compute_obj(a_new)

            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
            stall_count = 0
            # Slightly increase step
            step *= 1.01
        else:
            stall_count += 1
            if stall_count > 20:
                step *= 0.7
                if step < 1e-6:
                    break
            # Try a smaller step
            buf = momentum * buf * 0.5
    
    return best_a, best_obj

    best_obj = compute_obj(a)
    cur_obj = best_obj
    
    step = 0.3 / np.sqrt(n)
    buf = np.zeros(n)
    momentum = 0.9
    start = time.monotonic()
    stall_count = 0
    
    for it in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        
        # Alternate between exact and smoothed gradient
        if it % 3 == 0:
            grad = compute_gradient(a, conv)
        else:
            tau = 0.05 + 0.1 * (it / max_steps)
            grad = compute_smooth_gradient(a, conv, tau)
        
        grad_centered = grad - np.mean(grad)
        
        buf = momentum * buf - step * grad_centered
        a_new = a + buf
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < cur_obj - 1e-12:
            a = a_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
            stall_count = 0
            step = min(step * 1.02, 2.0 / np.sqrt(n))
        else:
            stall_count += 1
            step *= 0.8
            if step < 1e-7:
                step = 0.3 / np.sqrt(n)
                buf = np.zeros(n)
            if stall_count > 50:
                break
    
    return best_a, best_obj


def build_toeplitz_conv(a):
    """Build matrix A where A@test = conv(a, test) (linear convolution)."""
    n = len(a)
    # conv(a, test)[k] = sum_i a[i] * test[k-i]
    # A has shape (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # test[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # conv[j+i] += a[i] * test[j] for i = 0, ..., n-1
        for i in range(n):
            k = i + j
            if k < 2 * n - 1:
                A[k, j] += a[i]
    return A


def solve_lp_max_sum(a, M, time_limit=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 1500:
        return None
    A = build_toeplitz_conv(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate(a, max_iters=30, time_budget=40.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp_max_sum(a, M, time_limit=min(8, time_budget * 0.3))
        if g is None:
            break
        
        # Line search
        best_t_obj = best_obj
        best_t = 0.5
        best_a_local = best_a.copy()
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t}")
        else:
            break
    
    return best_a, best_obj


def freq_smooth(a, max_iters=100, time_budget=20.0):
    """
    Frequency-domain smoothing: modify the power spectrum to reduce time-domain peaks.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = np.fft.fft(a, L)
        power = np.abs(fa) ** 2
        
        # Smooth the power spectrum: reduce peaks
        p_max = np.max(power)
        if p_max < 1e-30:
            break
        
        # Apply logarithmic compression
        alpha = 0.85 + 0.1 * (it / max_iters)
        power_smooth = (power / p_max) ** alpha * p_max
        
        # Reconstruct
        phase = np.angle(fa)
        mag = np.sqrt(power_smooth)
        fa_new = mag * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
    
    return best_a, best_obj


def chase_peak_fast(a, n_steps=200, time_budget=20.0):
    """
    Fast chase-the-peak using the exact subgradient (no smoothing).
    Very fast since each step is O(n) after the initial O(n log n) conv computation.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    
    step = 0.1 / np.sqrt(n)
    start = time.monotonic()
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Exact subgradient
        grad = np.zeros(n)
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        if i_max >= i_min:
            indices = np.arange(i_min, i_max + 1)
            grad[indices] = 2.0 * a[k_star - indices]
        
        grad_centered = grad - np.mean(grad)
        
        # Try the step
        a_new = a - step * grad_centered
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < cur_obj - 1e-14:
            a = a_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.7
            if step < 1e-8:
                break
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new
