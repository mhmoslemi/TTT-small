import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k] for k=0..2n-2"""
    n = len(a)
    # A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For column j, A[k, j] = a[k-j] if 0 <= k-j < n
        # k ranges from j to j+n-1 (clipped to 2n-2)
        k_min = j
        k_max = min(j + n, 2 * n - 1)
        A[k_min:k_max, j] = a[:k_max - k_min]
    return A

def lp_alternate_max_sum(a, M, n_iters=500, time_budget=60.0):
    """
    LP alternating: repeatedly solve max sum(g) s.t. (a*g) <= M, g >= 0
    and update a.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    A = build_toeplitz(a)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    t = 0.5
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Rebuild A (since a changed)
        A = build_toeplitz(a)
        b_ub = np.full(2 * n - 1, M)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, 
                         method='highs', options={'presolve': True})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s < 1e-10:
                break
            g = g / s
        except Exception:
            break
        
        # Line search for best t
        best_t = t
        best_t_obj = np.inf
        for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t_try) * a + t_try * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t_try
        
        a_new = (1 - best_t) * a + best_t * g
        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            t = best_t
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
    
    return best_a, best_obj

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_max = min(j + n, 2 * n - 1)
        A[j:k_max, j] = a[:k_max - j]
    return A


def smoothed_max_grad(a, tau):
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    shifted = np.clip((conv - cmax) / tau, -400, 400)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= wsum
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.fft(weights, N)
    fa = np.fft.fft(a, N)
    corr = np.fft.ifft(fw * np.conj(fa)).real[:n]
    grad = (2.0 / tau) * corr
    val = tau * (np.log(wsum) if wsum > 0 else 0) + cmax
    return val, grad


def gradient_descent_smoothed(a, max_iters=300, time_budget=30.0):
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    momentum = 0.8
    v = np.zeros(n)
    tau = 0.02

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        cmax = np.max(conv)
        med = np.median(conv)
        tau = max(0.003, 0.15 * (cmax - med) * (1.0 - 0.5 * it / max_iters) + 0.003)
        _, grad = smoothed_max_grad(a, tau)
        grad = grad - np.mean(grad)
        v = momentum * v - step * grad
        a_new = a - v
        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.95
            v *= 0.5
    return best_a


def lp_alternate(a, n_iters=5000, time_budget=120.0, adaptive_M=False, verbose=False):
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    t = 0.4
    stall = 0
    last_report = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        M = float(np.max(conv))
        if adaptive_M and it > 100:
            M_eff = M * (1.0 - 0.002 * ((it - 100) % 200) / 200.0)
        else:
            M_eff = M
        A = build_toeplitz(a)
        b_ub = np.full(2 * n - 1, M_eff)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s < 1e-10:
                break
            g = g / s
        except Exception:
            break

        best_t = t
        best_t_obj = np.inf
        for t_try in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t_try

        a_new = (1.0 - best_t) * a + best_t * g
        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            t = best_t
            stall = 0
            if verbose and it - last_report > 200:
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, {elapsed:.1f}s")
                last_report = it
        else:
            stall += 1
            if stall > 40:
                break

    return best_a, best_obj


def freq_compress(a, n_iters=150, time_budget=20.0):
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        cmax = np.max(conv)
        if cmax < 1e-15:
            break
        c_mod = conv.copy()
        med = np.median(c_mod[c_mod > 0]) if np.any(c_mod > 0) else cmax * 0.5
        alpha = 0.5 + 0.35 * (it / n_iters)
        c_mod = np.maximum(c_mod, 0)
        c_mod = (c_mod / cmax) ** alpha * cmax
        cf = np.fft.fft(c_mod, N)
        fa_new = np.sqrt(cf + 0j)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
    return best_a, best_obj


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def simulated_annealing(a, rng, n_iters=300, T_init=0.15, time_budget=60.0):
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    cur_a = a.copy()
    cur_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = cur_obj
    start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        T = T_init * 0.995 ** it
        a_trial = cur_a.copy()
        mt = rng.integers(0, 3)
        if mt == 0:
            ws = max(5, n // 20)
            ws0 = rng.integers(0, n - ws)
            a_trial[ws0:ws0 + ws] += rng.normal(0, 0.15, ws) * (cur_a[ws0:ws0 + ws].mean() + 0.01)
        elif mt == 1:
            ws = max(10, n // 10)
            ws0 = rng.integers(0, n - ws)
            mass = a_trial[ws0:ws0 + ws].sum()
            a_trial[ws0:ws0 + ws] = 0
            tgt = rng.choice(n, size=ws, replace=False)
            a_trial[tgt] += mass / ws
        else:
            ws = max(5, n // 25)
            w1 = rng.integers(0, n - ws)
            w2 = rng.integers(0, n - ws)
            if abs(w1 - w2) >= ws:
                a_trial[w1:w1 + ws], a_trial[w2:w2 + ws] = \
                    a_trial[w2:w2 + ws].copy(), a_trial[w1:w1 + ws].copy()
        a_trial = project_nonneg_sum1(a_trial)
        trial_obj = compute_obj(a_trial)
        delta = trial_obj - cur_obj
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-min(delta / T, 50))):
            cur_a = a_trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()
    return best_a, best_obj


def lp_minpeak_blend(a, time_budget=20.0):
    n = len(a)
    if n > 2000:
        return a, compute_obj(a)
    conv = compute_conv(a)
    M_cur = float(np.max(conv))
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    best_a = a.copy()
    best_obj = compute_obj(a)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': min(time_budget * 0.6, 15)})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - t) * a + t * g
                    a_try = project_nonneg_sum1(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
    except Exception:
        pass
    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-strategy: aggressive LP alternating (primary), smoothed-max gradient,
    LP min-peak blend, frequency compression, SA, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    n_parent = len(best_seq)
    print(f"Start: n={n_parent}, best={best_score:.8f}, budget={budget_s}s")

    # === Phase 1: Aggressive LP Alternating (45% budget) ===
    print("\n--- Phase 1: LP Alternating ---")
    p1_budget = min(650.0, (deadline - start_time) * 0.45)
    p1_start = time.monotonic()

    # First round: standard LP alt
    a1, obj1 = lp_alternate(best_seq, n_iters=10000, time_budget=p1_budget * 0.4, verbose=True)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        print(f"  LP round 1: {best_score:.8f}")

    # Second round: adaptive M
    rem1 = p1_start + p1_budget - time.monotonic()
    if rem1 > 20:
        a2, obj2 = lp_alternate(best_seq, n_iters=10000, time_budget=rem1,
                                adaptive_M=True, verbose=True)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  LP round 2 (adaptive M): {best_score:.8f}")

    # === Phase 2: LP Min-Peak + Gradient (20% budget) ===
    print(f"\n--- Phase 2: LP MinPeak + Gradient (best={best_score:.8f}) ---")
    p2_budget = min(300.0, (deadline - time.monotonic()) * 0.20)
    p2_start = time.monotonic()

    # LP min-peak
    n_cur = len(best_seq)
    if n_cur <= 2000:
        a3, obj3 = lp_minpeak_blend(best_seq, time_budget=min(p2_budget * 0.3, 25.0))
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  LP minpeak: {best_score:.8f}")

    # Gradient descent on smoothed max
    rem2 = p2_start + p2_budget - time.monotonic()
    if rem2 > 10:
        a4 = gradient_descent_smoothed(best_seq, max_iters=500, time_budget=rem2 * 0.5)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Gradient: {best_score:.8f}")

    # Freq compression
    rem2 = p2_start + p2_budget - time.monotonic()
    if rem2 > 5:
        a5, obj5 = freq_compress(best_seq, n_iters=200, time_budget=rem2)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Freq compress: {best_score:.8f}")

    # === Phase 3: SA + Perturbations (15% budget) ===
    print(f"\n--- Phase 3: SA + Perturb (best={best_score:.8f}) ---")
    p3_budget = min(220.0, (deadline - time.monotonic()) * 0.15)
    p3_start = time.monotonic()

    a6, obj6 = simulated_annealing(best_seq, rng, n_iters=500, T_init=0.12,
                                    time_budget=p3_budget * 0.4)
    if obj6 < best_score:
        best_score = obj6
        best_seq = a6.copy()
        print(f"  SA: {best_score:.8f}")

    rem3 = p3_start + p3_budget - time.monotonic()
    n_perturb = min(8, max(1, int(rem3 / 15)))
    for p in range(n_perturb):
        if time.monotonic() > p3_start + p3_budget:
            break
        scale = 0.02 + 0.02 * p
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = project_nonneg_sum1(a_pert)
        a_pert, obj_pert = lp_alternate(a_pert, n_iters=1000, time_budget=12.0)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Perturb {p} (scale={scale:.3f}): {best_score:.8f}")

    # === Phase 4: Multi-length exploration (12% budget) ===
    print(f"\n--- Phase 4: Multi-length (best={best_score:.8f}) ---")
    rem4 = deadline - time.monotonic()
    if rem4 > 40:
        p4_budget = min(180.0, rem4 * 0.45)
        p4_start = time.monotonic()
        n_cur = len(best_seq)
        n_tries = sorted(set([
            max(100, n_cur - 100), max(100, n_cur - 50),
            n_cur + 50, n_cur + 100, n_cur + 200,
            max(100, n_cur - 200), n_cur + 300
        ]))
        n_tries = [n for n in n_tries if 100 <= n <= 3000]
        per_budget = max(8.0, p4_budget / max(1, len(n_tries)))

        for n_try in n_tries:
            if time.monotonic() > p4_start + p4_budget:
                break
            a_res = resample_sequence(best_seq, n_try)
            a_res = project_nonneg_sum1(a_res)
            a_res, obj_res = lp_alternate(a_res, n_iters=3000, time_budget=per_budget)
            if obj_res < best_score:
                best_score = obj_res
                best_seq = a_res.copy()
                print(f"  n={n_try}: {best_score:.8f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_res:.8f}")

    # === Phase 5: Final intensive polish (remaining) ===
    print(f"\n--- Phase 5: Final Polish (best={best_score:.8f}) ---")
    rem5 = deadline - time.monotonic()
    if rem5 > 15 and best_seq is not None:
        n_f = len(best_seq)
        # Alternating LP + freq + gradient rounds
        for r in range(min(8, int(rem5 / 15))):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            # LP
            a_pol, obj_pol = lp_alternate(best_seq, n_iters=5000, time_budget=min(rem * 0.5, 25.0))
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            # Freq
            a_fc, obj_fc = freq_compress(best_seq, n_iters=80, time_budget=min(rem * 0.3, 12.0))
            if obj_fc < best_score:
                best_score = obj_fc
                best_seq = a_fc.copy()
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            # Gradient
            a_gd = gradient_descent_smoothed(best_seq, max_iter=200, time_budget=min(rem * 0.4, 15.0))
            obj_gd = compute_obj(a_gd)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
            rem = deadline - time.monotonic()
            if rem < 10 or n_f > 2000:
                continue
            # LP minpeak
            a_mp, obj_mp = lp_minpeak_blend(best_seq, time_budget=min(rem * 0.4, 15.0))
            if obj_mp < best_score:
                best_score = obj_mp
                best_seq = a_mp.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
