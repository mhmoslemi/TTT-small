    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Continuation-based projected subgradient with FFT-accelerated gradients,
LP alternating refinement, multi-n exploration, and random restarts for
minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.ones(n) / n
    return w


def compute_smoothed_grad_fft(a, weights, n):
    """
    Compute smoothed gradient of max(conv) using FFT cross-correlation.
    grad_i = 2 * sum_j weights[i+j] * a[j]
    Computed as: 2 * conv(weights_reversed, a)[2n-2-i]
    """
    L = 2 * n - 1
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    w_rev = weights[::-1]
    W = rfft(w_rev, size)
    A = rfft(a, size)
    c = irfft(W * A, size)
    
    # grad_i = 2 * c[2n-2-i] for i=0,...,n-1
    # So grad = 2 * c[n-1 : 2n-1] reversed
    grad = 2.0 * c[n-1:2*n-1][::-1]
    return grad


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.

    M = cp.Variable()
