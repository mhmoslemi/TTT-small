    ```python
    """
    Fast LP-alternating + Nesterov subgradient optimization for minimizing
    2*n*max(conv(a,a))/(sum(a))^2. Uses scipy HiGHS for fast LP solves.
    """

    import numpy as np
    from scipy.fft import rfft, irfft
    from scipy.optimize import linprog
    import time


    def compute_conv(a):
        """Full linear convolution a*a via FFT."""
        n = len(a)
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = rfft(a, size)
        return irfft(fa * fa, size)[:2 * n - 1]


    def compute_obj(a):
        """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
        n = len(a)
        s = float(np.sum(a))
        if s < 1e-10:
            return np.inf
        a_n = a / s
        c = compute_conv(a_n)
        return 2.0 * n * float(np.max(c))


    def build_conv_matrix(a):
        """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
        n = len(a)
        A = np.zeros((2 * n - 1, n))
        for m in range(n):
            length = min(n, 2 * n - 1 - m)
            A[m:m + length, m] = a[:length]
        return A


    def solve_minpeak_lp(a):
        """
        Solve LP: min M s.t. conv(a,g)_k <= M for all k, sum(g)=1, g>=0.
        Returns (g_normalized, M_opt) or (None, None).
        """
        n = len(a)
        a_norm = a / np.sum(a)
        A = build_conv_matrix(a_norm)
        
        # LP: min M s.t. A@g - M <= 0, sum(g)=1, g>=0, M>=0
        # Variables: [g_0,...,g_{n-1}, M]
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize M
        
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * (n + 1)
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-12:
                return g_val / s, float(res.x[n])
        return None, None


    def simplex_project(v, target_sum=1.0):
        """Project v onto {x >= 0, sum(x) = target_sum}."""
        # Standard simplex projection
        n = len(v)
        if n == 0:
            return v
        # Sort in descending order
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
        theta = (cssv[rho] - target_sum) / (rho + 1.0)
        return np.maximum(v - theta, 0)


    def nesterov_subgradient(a, time_budget=60.0, verbose=False):
        """
        Nesterov-accelerated subgradient method for minimizing max(conv(a,a))
        subject to sum(a)=1, a>=0.
        """
        n = len(a)
        a = np.maximum(a, 0).astype(np.float64)
        s = float(np.sum(a))
        if s < 1e-10:
            return a
        a = a / s
        
        best_a = a.copy()
        best_obj = compute_obj(a)
        start = time.monotonic()
        
        # Nesterov parameters
        beta = 1.0
        alpha0 = 0.5
        a_prev = a.copy()
        
        max_iters = 5000
        for it in range(max_iters):
            if time.monotonic() - start > time_budget:
                break
            
            # Nesterov extrapolation
            if it > 0:
                y = a + ((beta - 1.0) / beta) * (a - a_prev)
                y = np.maximum(y, 0)
                sy = float(np.sum(y))
                if sy > 1e-10:
                    y = y / sy
            else:
                y = a.copy()
            
            # Compute subgradient at y
            conv_y = compute_conv(y)
            peak_val = float(np.max(conv_y))
            # Find all peak positions (within tolerance)
            peak_mask = conv_y >= peak_val - 1e-10
            peak_positions = np.where(peak_mask)[0]
            # Pick a random peak position for subgradient
            k_star = int(peak_positions[np.random.default_rng(it).integers(len(peak_positions))])
            
            # Subgradient of conv(y,y)_{k_star} w.r.t. y_i = 2*y[k_star - i] (with bounds)
            grad = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] = 2.0 * y[j]
            
            # Condition on sum=1: subtract mean
            grad_centered = grad - np.mean(grad)
            
            # Step size
            alpha = alpha0 / (1.0 + 0.01 * it)
            a_new = a - alpha * grad_centered
            a_new = simplex_project(a_new, target_sum=1.0)
            
            # Check improvement
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a_prev = a
            a = a_new
            beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        
        return best_a

    def resample_sequence(a, new_n):
        """Resample sequence to new length."""
        n = len(a)
        if n == new_n:
            return a.copy()
        if n == 1:
            return np.full(new_n, a[0])
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
        return np.maximum(a_new, 0)


    def coordinate_refine(a, time_budget=20.0):

        """

            return a

            if time.monotonic() - start > time_budget:
                break
            
            conv = compute_conv(a)
            k_star = int(np.argmax(conv))
            improved = False
            
            # Find the pair (i, j) where moving mass from i to j reduces the peak
            best_delta_obj = 0
            best_i, best_j, best_frac = -1, -1, 0
    ```
