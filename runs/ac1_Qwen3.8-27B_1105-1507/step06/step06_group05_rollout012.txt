    """
    Lp-smooth descent for minimizing (sum_k Q_k^p)^{1/p} on the simplex.
    As p increases, this approaches the max.
    """
    n = len(a)
    a = proj_simplex(a, 1.0).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive p: increase over time
        t = it / max_iters
        p = p_init + (p_final - p_init) * t
        
        conv = compute_conv_fft(a)
        cmax = float(np.max(conv))
        cmin = float(np.min(conv))
        
        if cmax < 1e-15:
            break
        
        # Normalized weights: w[k] = Q_k^{p-1} / sum Q_k^{p-1}
        # But Q_k can be zero, so use (Q_k + eps)^{p-1}
        eps = 1e-10 * cmax
        qp1 = np.power(conv + eps, p - 1)
        w = qp1 / np.sum(qp1)

        b = a[::-1].copy()
        size = 1

        Wf = rfft(w, size)
        Bf = rfft(b, size)
        crosscorr_full = irfft(Wf * Bf, size)[:2*n-1+n-1]
        crosscorr = crosscorr_full[n-1:2*n-1]  # valid part, length n
        
        grad = 2.0 * crosscorr
        
        # Backtracking line search
        improved = False
        for _ in range(20):
            a_try = proj_simplex(a - eta * grad, 1.0)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                eta = min(eta * 1.2, 5.0)

                break
            eta *= 0.5
