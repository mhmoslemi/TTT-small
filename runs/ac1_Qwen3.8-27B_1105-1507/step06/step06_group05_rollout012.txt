    n = len(a)

    A = np.zeros((m, n), dtype=np.float64)

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Core: LP alternating with adaptive M tightening, spectral equalization,
subgradient descent with simplex projection, multi-n exploration,
and coordinate descent polishing.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(y):
    """Euclidean projection onto probability simplex {x: x>=0, sum(x)=1}."""
    n = len(y)
    if n == 0:
        return y.copy()
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(y - theta, 0.0)


def build_toeplitz(a):
    """Build matrix A where A @ g = conv(a, g). Shape (2n-1, n)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        A[k, js:je] = a[js:je]
    return A


def solve_lp_direction(a, M, time_limit=45):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Return normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate_refined(a, max_iters=100, time_budget=120.0, M_init_scale=1.02):

    conv = compute_conv(a)

def spectral_equalization_v2(a, max_iters=100, time_budget=60.0):
    """
    Improved spectral equalization with adaptive compression exponent
    and proper simplex projection.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    alpha = 0.5  # power compression: starts gentle
    blend_weight = 0.6  # weight of old a in blend

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Compress: raise flat parts, suppress peaks
        P_norm = P / p_max
        # Smooth the compression to avoid artifacts
        alpha_eff = alpha + 0.1 * np.sin(0.1 * it)  # slight oscillation
        P_new = P_norm ** alpha_eff * p_max

        # Reconstruct preserving phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_freq = np.real(ifft(A_new, size))[:n]

        # Project to simplex (Euclidean)
        a_proj = project_simplex(a_freq)

        # Blend
        a_candidate = project_simplex(blend_weight * a + (1.0 - blend_weight) * a_proj)

            best_a = a_candidate.copy()
            a = a_candidate
            alpha = min(1.8, alpha * 1.015)
            blend_weight = max(0.2, blend_weight * 0.99)
        else:
            alpha = max(0.3, alpha * 0.985)
            blend_weight = min(0.8, blend_weight * 1.01)

    return best_a


def subgradient_projected(a, max_iters=400, time_budget=60.0):
    """
    Projected subgradient descent with FOMC-style acceleration.
    Minimizes max conv(a,a) over the probability simplex.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # FOMC parameters
    y = a.copy()  # extrapolation point
    v = np.zeros(n)  # mean of steps
    t_k = 1.0  # step size counter
    theta = 0.5  # extrapolation parameter

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute subgradient at y
        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))

        # Subgradient: sg[i] = 2*a_ref[k_star - i] where a_ref is the current iterate
        # For the max, use y for gradient computation
        sg = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sg[i] = 2.0 * y[j]

        # Step size
        eta = 1.0 / (np.sqrt(t_k) * np.sqrt(float(n)))

        # Gradient step
        a_new = y - eta * sg

        # Project to simplex
        a_new = project_simplex(a_new)

        # Update FOMC state
        t_k_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        tau = (t_k - 1.0) / t_k_new if t_k_new > 0 else 0.0
        y_new = (1.0 + theta * tau) * a_new - theta * tau * v

        # Evaluate
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        # Update
        v = (v * t_k + a_new) / t_k_new
        y = y_new
        t_k = t_k_new

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    vals = np.interp(x_new, x_old, a)
    return np.maximum(vals, 0.0)


def coordinate_polish(a, time_budget=20.0, seed=123):
    """
    Coordinate descent: transfer mass between coordinates contributing
    to the peak convolution, trying to reduce the peak.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall_count = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        contribs = []
        for j in range(min(n, 100)):  # limit to first 100 for speed
            j2 = k_star - j
            if 0 <= j2 < n:
                contribs.append((a[j] * a[j2], j, j2))
        contribs.sort(key=lambda x: x[0], reverse=True)

        improved = False
        for peak_val, j1, j2 in contribs[:6]:
            if time.monotonic() - start > time_budget - 0.5:
                break
            if j1 == j2:
                continue

            # Try moving mass from the larger contributor to a random far position
            src = j1 if a[j1] >= a[j2] else j2
            for _ in range(4):
                dst = int(rng.integers(0, n))
                if dst == src:
                    continue

                for frac in [0.03, 0.08, 0.15]:
                    delta = frac * a[src]
                    if delta < 1e-10:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[dst] += delta
                    a_try = project_simplex(a_try)
                    obj = compute_obj(a_try)
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break

        if improved:
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 10:
                # Perturb
                a = project_simplex(a * np.exp(rng.normal(0, 0.02, n)))
                stall_count = 0

    return best_a

                bo = obj
                ba = a.copy()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = ba.copy()
            elif rng.random() < np.exp(-(obj - bo) / (2.0 * bd * bd * max(bo, 0.001))):
                pass  # accept (simulated annealing)
            else:
                a = ba.copy()
                bd *= 0.98

        iran = max(0.001, iran * 0.95)

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function with phased strategy:
    Phase 1: LP alternating with adaptive M (primary)
    Phase 2: Spectral equalization + subgradient (diversification)
    Phase 3: Multi-n exploration
    Phase 4: Random restarts with diverse perturbations
    Phase 5: Intensive final polish (LP + coordinate + monkey)
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            best_score = obj
            best_seq = parent.copy()
            print(f"Parent loaded: n={parent.size}, obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        print(f"Uniform start: n={len(best_seq)}, obj={best_score:.8f}")

    n_cur = len(best_seq)

    # ================================================================
    # Phase 1: LP Alternating (primary optimizer)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: LP Alternating (n={n_cur})")
    print(f"{'='*60}")

    a = normalize(best_seq.copy())

    # First pass: standard M_scale
    if time.monotonic() < deadline - 300:
        alloc = min(120.0, (deadline - time.monotonic()) * 0.15)
        a1 = lp_alternate_refined(a, max_iters=80, time_budget=alloc, M_init_scale=1.02)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            a = a1
            print(f"  LP pass 1: {best_score:.8f}")
        else:
            print(f"  LP pass 1: {obj1:.8f} (no gain)")
            a = a1

    # Second pass: tighter M
    if time.monotonic() < deadline - 250:
        alloc = min(100.0, (deadline - time.monotonic()) * 0.12)
        a2 = lp_alternate_refined(a, max_iters=70, time_budget=alloc, M_init_scale=1.005)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            a = a2
            print(f"  LP pass 2 (tight): {best_score:.8f}")
        else:
            print(f"  LP pass 2 (tight): {obj2:.8f} (no gain)")

    # Third pass: even tighter
    if time.monotonic() < deadline - 200:
        alloc = min(80.0, (deadline - time.monotonic()) * 0.1)
        a3 = lp_alternate_refined(a, max_iters=60, time_budget=alloc, M_init_scale=1.0)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            a = a3
            print(f"  LP pass 3 (tighter): {best_score:.8f}")
        else:
            print(f"  LP pass 3 (tighter): {obj3:.8f} (no gain)")

    # ================================================================
    # Phase 2: Spectral + Subgradient (diversification)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Spectral + Subgradient")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 200:
        a = normalize(best_seq.copy())
        a_spec = spectral_equalization_v2(a, max_iters=100, time_budget=50.0)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: {best_score:.8f}")
        else:
            print(f"  Spectral: {obj_spec:.8f} (no gain)")
            # Blend with best
            a = project_simplex(0.7 * normalize(best_seq) + 0.3 * a_spec)

        # Subgradient
        if time.monotonic() < deadline - 150:
            a_sg = subgradient_projected(a, max_iters=300, time_budget=45.0)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Subgradient: {best_score:.8f}")
            else:
                print(f"  Subgradient: {obj_sg:.8f} (no gain)")

        # Follow-up LP on best so far
        if time.monotonic() < deadline - 100:
            alloc = min(60.0, (deadline - time.monotonic()) * 0.1)
            a_lp = lp_alternate_refined(normalize(best_seq.copy()), max_iters=40,
                                        time_budget=alloc, M_init_scale=1.01)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Post Phase2 LP: {best_score:.8f}")

    # ================================================================
    # Phase 3: Multi-n exploration
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-n Exploration")
    print(f"{'='*60}")

    n_best_cur = len(best_seq)
    n_options = sorted(set([
        max(200, n_best_cur - 200),
        max(200, n_best_cur - 100),
        max(200, n_best_cur - 50),
        n_best_cur + 50,
        n_best_cur + 100,
        n_best_cur + 200,
    ]))

    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_best_cur:
            continue

        a_try = normalize(resample_sequence(best_seq, n_try))
        alloc = min(30.0, max(10.0, (deadline - time.monotonic()) * 0.08))

        # Optimize at this n
        if n_try <= 900:
            a_try = lp_alternate_refined(a_try, max_iters=25, time_budget=alloc * 0.5,
                                         M_init_scale=1.01)
        a_try = spectral_equalization_v2(a_try, max_iters=40, time_budget=alloc * 0.3)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best_cur = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # ================================================================
    # Phase 4: Random restarts with diverse strategies
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Random Restarts")
    print(f"{'='*60}")

    n_cur = len(best_seq)
    rng = np.random.default_rng(seed)

    for restart in range(6):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(35.0, remaining * 0.12)

        # Diverse perturbation
        if restart % 3 == 0:
            # Gaussian multiplier
            a_try = best_seq * np.exp(rng.normal(0, 0.06, n_cur))
        elif restart % 3 == 1:
            # Sinusoidal modulation
            freq = rng.uniform(0.5, 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.12 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
            a_try = best_seq * mod
        else:
            # Blend with Dirichlet
            a_rand = rng.dirichlet(np.ones(n_cur) * 0.3)
            mix = rng.uniform(0.15, 0.4)
            a_try = (1 - mix) * normalize(best_seq) + mix * a_rand

        a_try = project_simplex(a_try)

        # Optimize
        if n_cur <= 900 and alloc > 15:
            a_try = lp_alternate_refined(a_try, max_iters=20, time_budget=alloc * 0.5,
                                         M_init_scale=1.01)
        a_try = spectral_equalization_v2(a_try, max_iters=30, time_budget=alloc * 0.3)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  Restart {restart}: {best_score:.8f} **")
        else:
            print(f"  Restart {restart}: {obj_try:.8f}")

    # ================================================================
    # Phase 5: Final intensive polish
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Polish")
    print(f"{'='*60}")

    for polish_round in range(4):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break
        alloc = min(remaining * 0.25, 45.0)

        a = normalize(best_seq.copy())

        # LP pass
        if len(a) <= 900 and alloc > 15:
            a = lp_alternate_refined(a, max_iters=25, time_budget=alloc * 0.35,
                                     M_init_scale=1.005)

        # Subgradient
        if alloc > 10:
            a = subgradient_projected(a, max_iters=100, time_budget=alloc * 0.2)

        # Coordinate polish
        if alloc > 8:
            a = coordinate_polish(a, time_budget=min(12.0, alloc * 0.2))

        # Monkey tune
        if alloc > 5:
            a = monkey_tune(a, time_budget=min(8.0, alloc * 0.15))

        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Polish {polish_round}: {best_score:.8f}")
        else:
            print(f"  Polish {polish_round}: {obj:.8f} (no gain)")

    # ================================================================
    # Final validation
    # ================================================================
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Verify with actual evaluation function
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
