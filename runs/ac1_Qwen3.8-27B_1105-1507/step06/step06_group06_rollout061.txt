import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

LP-alternating optimization with frequency-domain refinement and
adaptive multi-length search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: (1) tighter LP with decreasing M,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=25):
    """Solve: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 4000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

    A_eq = np.zeros((1, n + 1))

"""
LP-alternating optimization with min-peak LP, frequency-domain water-filling,
dual-guided perturbation, and aggressive multi-length search to minimize
2*n*max(conv(a,a))/(sum(a))^2. Key novel elements: (1) alternating between

frequency domain preserving conjugate symmetry, (3) dual-variable-guided

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=25):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""

    try:

                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:

    except Exception:

    return None

def freq_water_filling(a, max_iters=200, time_budget=30.0, beta=0.3):
    """
    Multiplicative water-filling in the frequency domain.
    Pulls convolution values toward their geometric mean.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = fft(a, N)
        fA2 = fa * fa
        conv = ifft(fA2).real[:2 * n - 1]

        conv_pos = np.maximum(conv, 1e-15)
        gm = np.exp(np.mean(np.log(conv_pos)))

        # Multiplicative water-filling: pull toward geometric mean
        # conv_new = conv * (gm/conv)^beta = conv^(1-beta) * gm^beta
        conv_mod = conv_pos ** (1.0 - beta) * gm ** beta
        conv_mod = np.minimum(conv_mod, np.max(conv))  # don't increase peak

        # Permute back: DFT -> square root -> IDFT
        cf = fft(conv_mod, N)
        fa_new = np.sqrt(cf + 0j)

        # Ensure conjugate symmetry (should be automatic for real input)
        a_new = ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def freq_power_compress(a, max_iters=200, time_budget=25.0, power=0.8):
    """
    Power compression of the convolution in time domain, then sqrt to recover a.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = fft(a, N)
        conv = ifft(fa * fa).real[:2 * n - 1]

        # Power compression
        c_max = np.max(conv)
        if c_max < 1e-15:
            break
        conv_norm = np.maximum(conv, 0) / c_max
        conv_comp = conv_norm ** power * c_max

        # Recover a
        cf = fft(conv_comp, N)
        fa_new = np.sqrt(cf + 0j)
        a_new = ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj

            if 0 <= j < n:
                direction[i] = 2.0 * a[j]
        
        direction = direction - np.mean(direction)
        
        # Try small steps in this direction (negative = decrease peak)
        for step_size in [0.01, 0.005, 0.002]:
            a_try = a - step_size * direction
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating (min-peak + max-sum), frequency water-filling,
    dual-guided perturbation, multi-length sweep, and final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent = project_nonneg_sum1(parent)
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()
    else:
        n_init = 700
        x = np.linspace(-3, 3, n_init)
        best_seq = np.exp(-x**2 / 3)
        best_seq = project_nonneg_sum1(best_seq)
        best_score = compute_obj(best_seq)
        print(f"Generated initial (n=700): obj={best_score:.8f}")

    n_start = len(best_seq)
    print(f"Starting: n={n_start}, best={best_score:.8f}")

    # === Phase 1: Intensive LP Alternating (40%) ===
    print("\n=== Phase 1: Intensive LP Alternating ===")
    phase1_budget = min(580.0, (deadline - start_time) * 0.40)
    phase1_start = time.monotonic()

    a_work = best_seq.copy()
    a_work, obj_work = lp_alternate(a_work, max_iters=200, time_budget=phase1_budget * 0.55, verbose=True)
    if obj_work < best_score:
        best_score = obj_work
        best_seq = a_work.copy()
        print(f"  LP round 1: {best_score:.8f}")

    # Frequency water-filling
    remain = phase1_start + phase1_budget - time.monotonic()
    if remain > 15:
        a_wf, obj_wf = freq_water_filling(best_seq, max_iters=200, time_budget=min(remain*0.3, 30.0), beta=0.2)
        if obj_wf < best_score:
            best_score = obj_wf
            best_seq = a_wf.copy()
            print(f"  Water-filling: {best_score:.8f}")

    # More LP
    remain = phase1_start + phase1_budget - time.monotonic()
    if remain > 15:
        a_lp2, obj_lp2 = lp_alternate(best_seq, max_iters=150, time_budget=remain * 0.5, verbose=True)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP round 2: {best_score:.8f}")

    # Subgradient polishing
    remain = phase1_start + phase1_budget - time.monotonic()
    if remain > 15:
        a_sub, obj_sub = subgradient_smoothed(best_seq, max_iters=500, time_budget=remain * 0.5)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  Subgradient: {best_score:.8f}")

    # Power compression
    remain = phase1_start + phase1_budget - time.monotonic()
    if remain > 10:
        a_pc, obj_pc = freq_power_compress(best_seq, max_iters=100, time_budget=remain * 0.5, power=0.85)
        if obj_pc < best_score:
            best_score = obj_pc
            best_seq = a_pc.copy()
            print(f"  Power compress: {best_score:.8f}")

    # Final LP after all refinements
    remain = phase1_start + phase1_budget - time.monotonic()
    if remain > 10:
        a_f, obj_f = lp_alternate(best_seq, max_iters=50, time_budget=remain * 0.7)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  LP final P1: {best_score:.8f}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Dual-guided perturbation (15%) ===
    print(f"\n=== Phase 2: Dual-guided Perturbation ===")
    phase2_budget = min(220.0, (deadline - time.monotonic()) * 0.15)
    phase2_start = time.monotonic()

    for attempt in range(15):
        if time.monotonic() > phase2_start + phase2_budget:
            break
        remain = phase2_start + phase2_budget - time.monotonic()
        if remain < 10:
            break

        a_p, obj_p = dual_guided_perturb(best_seq, rng, n_perturbations=5, time_budget_per=8.0)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Dual perturb {attempt}: {best_score:.8f}")

        # Also try small random perturbation + LP
        remain = phase2_start + phase2_budget - time.monotonic()
        if remain < 8:
            break
        sigma = 0.003 + 0.01 * (attempt / 15.0)
        a_r = best_seq * np.exp(rng.normal(0, sigma, len(best_seq)))
        a_r = project_nonneg_sum1(a_r)
        a_r, obj_r = lp_alternate(a_r, max_iters=20, time_budget=min(remain * 0.4, 12.0))
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Random perturb {attempt}: {best_score:.8f}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Multi-length exploration (25%) ===
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    phase3_budget = min(370.0, (deadline - time.monotonic()) * 0.25)
    phase3_start = time.monotonic()

    n_current = len(best_seq)
    # Dense sweep around current best + some outliers
    n_tries = list(range(max(100, n_current - 60), min(3000, n_current + 61), 10))
    n_tries.extend([400, 500, 800, 1000, 1200, 1500, 2000])
    n_tries = sorted(set(n for n in n_tries if 100 <= n <= 3000))

    per_n_budget = max(4.0, phase3_budget / len(n_tries) * 0.8)
    print(f"  Trying {len(n_tries)} lengths, ~{per_n_budget:.1f}s each")

    # Sort by proximity to current best (try nearby first)
    n_tries.sort(key=lambda x: abs(x - n_current))

    for n_try in n_tries:
        if time.monotonic() > phase3_start + phase3_budget:
            break
        remain = phase3_start + phase3_budget - time.monotonic()
        if remain < 5:
            break

        a_n = resample_sequence(best_seq, n_try)
        a_n = project_nonneg_sum1(a_n)

        a_opt, obj_opt = lp_alternate(a_n, max_iters=15, time_budget=min(per_n_budget, remain * 0.7))

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: ** NEW BEST ** {best_score:.8f}")
        elif abs(n_try - n_current) <= 20:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Final intensive polish (20%) ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        phase4_start = time.monotonic()
        phase4_budget = min(300.0, remaining)

        a_final = best_seq.copy()

        for r in range(30):
            if time.monotonic() > phase4_start + phase4_budget:
                break
            rem = phase4_start + phase4_budget - time.monotonic()
            if rem < 8:
                break

            # LP
            a_p, obj_p = lp_alternate(a_final, max_iters=30, time_budget=min(rem * 0.3, 20.0))
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  Polish LP r{r}: {best_score:.8f}")

            rem = phase4_start + phase4_budget - time.monotonic()
            if rem < 5:
                break

            # Freq water-filling (small beta for fine-tuning)
            beta_small = 0.05 + 0.1 * (r / 30.0)
            a_wf, obj_wf = freq_water_filling(a_final, max_iters=50, time_budget=min(rem * 0.2, 10.0), beta=beta_small)
            if obj_wf < best_score:
                best_score = obj_wf
                best_seq = a_wf.copy()
                a_final = a_wf.copy()
                print(f"  Polish WF r{r}: {best_score:.8f}")

            rem = phase4_start + phase4_budget - time.monotonic()
            if rem < 5:
                break

            # Tiny perturbation + LP
            a_tp = a_final * np.exp(rng.normal(0, 0.002, len(a_final)))
            a_tp = project_nonneg_sum1(a_tp)
            a_tp, obj_tp = lp_alternate(a_tp, max_iters=15, time_budget=min(rem * 0.3, 10.0))
            if obj_tp < best_score:
                best_score = obj_tp
                best_seq = a_tp.copy()
                a_final = a_tp.copy()
                print(f"  Polish TP r{r}: {best_score:.8f}")

            rem = phase4_start + phase4_budget - time.monotonic()
            if rem < 5:
                break

            # Subgradient (short)
            a_sg, obj_sg = subgradient_smoothed(a_final, max_iters=200, time_budget=min(rem * 0.2, 15.0))
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                a_final = a_sg.copy()
                print(f"  Polish SG r{r}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(700)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 700
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
