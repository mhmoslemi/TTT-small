def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """FFT-based linear cross-convolution of a and b."""
    na, nb = len(a), len(b)
    if na == 0 or nb == 0:
        return np.array([])
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_and_grad(a, tau):
    """Compute smooth max (log-sum-exp) of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(p[i:i + n], a)
    return smooth_val, grad


def run_fista_smooth(a_init, tau_start, tau_end, max_iters, deadline, lr=None):
    """FISTA on smooth max approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    if lr is None:
        lr = 0.5
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        _, grad_y = smooth_max_and_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        step = lr / (1.0 + 0.05 * tau * it)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        it += 1
        if it % 5 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    M_a = fast_conv(a).max()
    if M_a < best_M:
        best_M = M_a
        best_a = a.copy()
    return best_a


def run_mirror_descent(a_init, max_iters, deadline, step_init=0.3):
    """Mirror descent with multi-constraint subgradients."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        # Use top-3 active constraints for better direction
        top_k = np.argsort(b)[::-1][:3]
        weights = np.exp(5.0 * (b[top_k] - b[top_k].max()))
        weights = weights / weights.sum()
        grad = np.zeros(n)
        for w_k, k in zip(weights, top_k):
            for i in range(n):
                j = int(k) - i
                if 0 <= j < n:
                    grad[i] += 2.0 * w_k * a[j]
        grad_tan = grad - grad.mean()
        step = step_init / (1.0 + 0.02 * it)
        log_weights = -step * grad_tan
        log_weights -= log_weights.max()
        a_new = a * np.exp(log_weights)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
        M_cur = b.max()  # quick check using old b as proxy
        if M < best_M:
            best_M = M
            best_a = a.copy()
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    return best_a


def build_lr_matrix(n, a, active_indices):
    """Build sparse matrix for LP: (a*g)[k] for k in active_indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        k = int(k)
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response(a, active_indices, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    n = len(a)
    A = build_lr_matrix(n, a, active_indices)
    if A.nnz == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def golden_blend_search(a, g, A, C, G, t_min=0.001, t_max=0.8):
    """Find optimal blend parameter t via golden section search.
    
    f(t) = max_k [(1-t)^2 * A[k] + 2t(1-t) * C[k] + t^2 * G[k]]
    """
    def f(t):
        vals = (1.0 - t) ** 2 * A + 2.0 * t * (1.0 - t) * C + t ** 2 * G
        return vals.max()
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = t_min
    hi = t_max
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = f(a_pt)
    fb = f(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = f(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = f(b_pt)
        if hi - lo < 1e-10:
            break
    
    t_opt = (lo + hi) / 2.0
    # Also check boundaries
    t_candidates = [t_min, t_opt, t_max, 0.01, 0.05, 0.1, 0.2, 0.3, 0.5]
    best_t = t_opt
    best_f = f(t_opt)
    for tc in t_candidates:
        if t_min <= tc <= t_max:
            fc = f(tc)
            if fc < best_f:
                best_f = fc
                best_t = tc
    return best_t, best_f


def run_active_set_lp(a_init, max_iters, deadline, init_active=10, max_active=None):
    """Violation-based active set LP optimization with proper line search."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1
    if max_active is None:
        max_active = min(conv_len, n)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    current_active = init_active
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Get active set: top current_active by value
        top_indices = np.argsort(b)[::-1][:min(current_active, conv_len)]
        active = top_indices.tolist()
        
        # Solve LP
        g = lp_best_response(a, active, M)
        if g is None:
            current_active = min(current_active + 5, max_active)
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Precompute convolutions for line search
        A = b  # conv(a, a)
        C = cross_conv(a, g)  # conv(a, g)
        G = fast_conv(g)  # conv(g, g)
        
        # Pad to same length
        max_len = max(len(A), len(C), len(G))
        if len(A) < max_len:
            A = np.pad(A, (0, max_len - len(A)))
        if len(C) < max_len:
            C = np.pad(C, (0, max_len - len(C)))
        if len(G) < max_len:
            G = np.pad(G, (0, max_len - len(G)))
        
        # Golden section search for optimal t
        t_opt, f_opt = golden_blend_search(a, g, A, C, G)
        
        # Apply blend
        a_trial = (1.0 - t_opt) * a + t_opt * g
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 0.01:
            no_improve += 1
            continue
        a_trial = a_trial / s_t
        
        # Verify actual improvement
        M_trial = fast_conv(a_trial).max()
        
        if M_trial < best_M * (1.0 - 1e-12):
            a = a_trial.copy()
            best_M = M_trial
            best_a = a.copy()
            no_improve = 0
            # Can try smaller active set
            if current_active > 5:
                current_active = max(5, int(current_active * 0.85))
        else:
            no_improve += 1
            # Expand active set
            current_active = min(int(current_active * 1.3) + 2, max_active)
            if no_improve > 20:
                break
    
    return best_a


def coordinate_descent_1d(a, i, b, n):
    """Find optimal value for a[i] given all others via golden section."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals) if c_vals else np.array([0.0])
    s_arr = np.array(slopes) if slopes else np.array([0.0])
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 1.0, 0.5)
    
    for _ in range(10):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 100.0:
            hi *= 2.0
        else:
            break
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Run coordinate descent sweeps."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-10:
                old_val = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - old_val * old_val)
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old_val)
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_improve:
            no_improve += 3
        if no_improve > 20:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.02, n), 1e-4)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.08, n), 1e-4)
        inits.append(p_n2 / p_n2.sum())
        # Symmetrized version
        p_sym = 0.5 * (p + p[::-1])
        p_sym = np.maximum(p_sym, 1e-4)
        inits.append(p_sym / p_sym.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Symmetric cosine
    cos_sym = 0.5 + 0.5 * np.cos(np.pi * (2 * t - 1))
    cos_sym = np.maximum(cos_sym, 0.01)
    inits.append(cos_sym / cos_sym.sum())
    
    # Raised cosine (Hann window)
    hann = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    hann = np.maximum(hann, 0.01)
    inits.append(hann / hann.sum())
    
    # Bartlett (triangle)
    bart = 1.0 - 2.0 * np.abs(t - 0.5)
    bart = np.maximum(bart, 0.01)
    inits.append(bart / bart.sum())
    
    # Flat-top-ish
    flat = 0.3 + 0.7 * np.cos(2 * np.pi * t / (2 * np.pi)) ** 4
    flat = np.maximum(flat, 0.01)
    inits.append(flat / flat.sum())
    
    # Frequency-domain inspired: try to make spectrum flat
    a_spec = np.ones(n) / n
    for _ in range(50):
        bs = fast_conv(a_spec)
        M_s = bs.max()
        # Reduce where convolution peaks
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.05 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    
    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Symmetric random
    half_n = n // 2
    half = rng.uniform(0.1, 1.0, half_n)
    sym = np.concatenate([half, half[::-1]])[:n]
    sym = np.maximum(sym, 0.01)
    inits.append(sym / sym.sum())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_combined_optimize(a0, n, deadline, seed=0):
    """Run combined optimization: FISTA -> LP -> Mirror -> CD."""
    best = np.maximum(a0.astype(np.float64), 1e-8)
    best = best / best.sum()
    best_M = fast_conv(best).max()
    t_start = time.monotonic()
    phase_duration = max(1, deadline - t_start)
    
    # Stage 1: FISTA (30% of time)
    t1 = t_start + 0.3 * phase_duration
    if time.monotonic() < t1:
        for tau in [50, 200, 500]:
            if time.monotonic() > t1:
                break
            a_f = run_fista_smooth(best, tau, tau * 3, 2000,
                                   min(t1, time.monotonic() + max(5, (t1 - time.monotonic()) // 3)),
                                   lr=0.3)
            M_f = fast_conv(a_f).max()
            if M_f < best_M:
                best_M = M_f
                best = a_f.copy()
    
    # Stage 2: LP with violation-based active set (40% of time)
    t2 = t_start + 0.7 * phase_duration
    if time.monotonic() < t2:
        est_per_iter = 0.3
        max_iters = max(10, int((t2 - time.monotonic()) / est_per_iter))
        a_lp = run_active_set_lp(best, max_iters, t2, init_active=8, max_active=min(2 * n - 1, n))
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
    
    # Stage 3: Mirror descent (15% of time)
    t3 = t_start + 0.85 * phase_duration
    if time.monotonic() < t3:
        max_md = max(100, int((t3 - time.monotonic()) / 0.01))
        a_md = run_mirror_descent(best, max_md, t3, step_init=0.15)
        M_md = fast_conv(a_md).max()
        if M_md < best_M:
            best_M = M_md
            best = a_md.copy()
    
    # Stage 4: Coordinate descent (15% of time)
    if time.monotonic() < deadline:
        est_sweep = n / 200.0
        max_sw = max(3, int((deadline - time.monotonic()) / max(0.1, est_sweep)))
        a_cd = run_coordinate_descent(best, max_sw, deadline, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
    
    return best

        if remaining < 30:
            break
        time_alloc = min(time_per_length, remaining / (n_lengths - li))
        print(f"\n  Length n={n}, time_alloc={time_alloc:.1f}s", flush=True)
        
        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        
        # Limit number of inits based on time
        n_inits = min(len(inits), max(2, int(time_alloc / 15)))
        
        best_for_n = None
        best_M_for_n = float('inf')
        
        for ii in range(n_inits):
            if time.monotonic() > deadline - reserve:
                break
            sub_deadline = min(time.monotonic() + time_alloc / n_inits + 5, deadline - reserve)
            if time.monotonic() >= sub_deadline - 5:
                break
            
            a0 = np.maximum(inits[ii], 1e-6)
            a0 = a0 / a0.sum()
            
            a_opt = run_combined_optimize(a0, n, sub_deadline, seed=seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()
            
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - (deadline - reserve - time_for_phase1)
            print(f"  n={n}: best M={best_M_for_n:.10f} J={J_val:.6f}", flush=True)
            tracker.update(best_for_n)
    
    tracker.report("After Phase 1")
    
    # === Phase 2: Intensive LP polish on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive LP polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 100:
        n_best = tracker.best.size
        t_lp_deadline = min(deadline - 80, time.monotonic() + 180)
        
        # First pass: moderate active set
        a_lp = run_active_set_lp(tracker.best, 100, t_lp_deadline, init_active=10, max_active=min(2 * n_best - 1, n_best // 2))
        tracker.update(a_lp)
        
        # Second pass: larger active set from improved point
        if time.monotonic() < deadline - 60:
            a_lp2 = run_active_set_lp(a_lp, 80, min(deadline - 40, time.monotonic() + 100), init_active=20, max_active=min(2 * n_best - 1, n_best))
            tracker.update(a_lp2)
    
    tracker.report("After Phase 2")
    
    # === Phase 3: FISTA refinement with high tau ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: High-tau FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t3_deadline = min(deadline - 30, time.monotonic() + 80)
        current = tracker.best.copy()
        for tau in [300, 800, 1500]:
            if time.monotonic() > t3_deadline:
                break
            a_f = run_fista_smooth(current, tau, tau * 2, 3000,
                                   min(t3_deadline, time.monotonic() + 25), lr=0.1)
            M_f = fast_conv(a_f).max()
            if M_f < fast_conv(current).max():
                current = a_f.copy()
            tracker.update(current)
    
    tracker.report("After Phase 3")
    
    # === Phase 4: Coordinate descent polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_cd = tracker.best.size
        est_sweep = n_cd / 150.0
        max_sweeps = max(5, int((deadline - 20 - time.monotonic()) / max(0.2, est_sweep)))
        a_cd = run_coordinate_descent(tracker.best, max_sweeps, deadline - 10, seed=99999)
        tracker.update(a_cd)
        
        # Final LP pass
        if time.monotonic() < deadline - 5:
            a_final = run_active_set_lp(a_cd, 20, deadline, init_active=15, max_active=n_cd)
            tracker.update(a_final)
    
    tracker.report("After Phase 4")
    
    # === Phase 5: Try length variations from best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 20:
        n_cur = tracker.best.size
        for mult in [1.2, 0.8, 1.4]:
            if time.monotonic() > deadline - 15:
                break
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 2048)
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 5, time.monotonic() + 40)
            a_opt = run_combined_optimize(a_ext, new_n, sub_dl, seed=seed + 777)
            tracker.update(a_opt)
    
    tracker.report("After Phase 5")
    
    # === Final LP + CD alternating on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL POLISH: Alternating LP/CD", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 5:
        current = tracker.best.copy()
        n_f = current.size
        # Alternate LP and CD a few times
        for round_i in range(4):
            if time.monotonic() > deadline - 5:
                break
            # LP step
            t_lp = min(deadline - 5, time.monotonic() + 15)
            a_new = run_active_set_lp(current, 15, t_lp, init_active=10, max_active=n_f // 2)
            M_new = fast_conv(a_new).max()
            if M_new < fast_conv(current).max():
                current = a_new.copy()
            # CD step
            t_cd = min(deadline - 5, time.monotonic() + 15)
            a_new2 = run_coordinate_descent(current, 5, t_cd, seed=seed + 500 + round_i)
            M_new2 = fast_conv(a_new2).max()
            if M_new2 < fast_conv(current).max():
                current = a_new2.copy()
            tracker.update(current)
    
    tracker.report("FINAL")
    
    # Ensure we return something valid
    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL ANSWER: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
