Wait, there's an issue with the L-BFGS-B approach. The objective function uses the smoothed max, but the gradient I compute should be the gradient of the smoothed max with respect to x. Let me verify:

F(x) = (1/β) log Σ exp(β * conv_k(a(x)))
where a(x) = [x_0, ..., x_{n-2}, 1 - Σx_i]

dF/dx_j = (1/β) * Σ_k β * (d conv_k / d a(x)) * (da(x)/dx_j) * exp(β*conv_k) / Σ_l exp(β*conv_l)
= Σ_k W_k * (d conv_k / d a(x)) * (da(x)/dx_j)

where W_k = exp(β*conv_k) / Σ_l exp(β*conv_l)

And d conv_k / d a(x) = grad_k (the gradient of conv_k with respect to a), so:
dF/dx_j = Σ_k W_k * grad_k[j] - Σ_k W_k * grad_k[n-1]
= grad_smoothed[j] - grad_smoothed[n-1]

where grad_smoothed = Σ_k W_k * grad_k is the full gradient we computed.

So the gradient with respect to x is indeed grad[:n-1] - grad[n-1]. ✓

But wait, there's a subtlety: when I compute a = [x, 1-sum(x)] and then clip with a = max(a, 0) and renormalize, I'm changing the mapping. For L-BFGS to work correctly, the mapping from x to a should be smooth (or at least differentiable almost everywhere).

The issue is: if 1 - sum(x) < 0, the mapping is infeasible. If some x_j < 0, the clipping makes it non-smooth.

A better approach: use the simplex projection in the objective and gradient, but accept that the optimization might be slightly perturbed near the boundary. In practice, this works fine.

Actually, the cleanest approach for this problem is:

1. Use the **exponential parameterization**: a_j = exp(z_j) / Σ exp(z_i)
2. Optimize over z ∈ R^n using unconstrained L-BFGS-B (no bounds needed)
3. The gradient with respect to z is:
   ∂F/∂z_j = a_j * (∂F/∂a_j - Σ_i a_i * ∂F/∂a_i)

This handles the non-negativity constraint automatically (exp is always positive) and the normalization is built in.

The downside is that when some a_j are very small, the corresponding z_j is very negative, and the optimizer might struggle. But with a good initialization, this should be fine.

def _compute_conv(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def _compute_obj(a):
    """Objective: 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = _compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using the standard algorithm."""
    v = np.array(v, dtype=np.float64)
    n = len(n) if False else len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smooth_max_grad(a, beta):
    """
    Compute (smooth_max_of_conv, gradient) where smooth_max uses log-sum-exp.
    Gradient is with respect to a (free variables, not on simplex).
    """
    n = len(a)
    conv = _compute_conv(a)

    # Numerically stable softmax
    cmax = np.max(conv)
    shifted = beta * (conv - cmax)
    shifted -= np.max(shifted)
    W = np.exp(shifted)
    W_sum = np.sum(W)
    if W_sum < 1e-300:
        idx = int(np.argmax(conv))
        W = np.zeros_like(conv)
        W[idx] = 1.0
    else:
        W /= W_sum

    sm = cmax + np.log(np.sum(np.exp(shifted))) / beta

    # Gradient via FFT
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    A = rfft(a, fft_size)
    Wp = np.zeros(fft_size)
    Wp[:2 * n - 1] = W
    Wf = rfft(Wp, fft_size)

    # term1[m] = sum_k W[k] * a[k-m] (cross-correlation)
    term1 = irfft(Wf * np.conj(A), fft_size)[:n]
    # term2[m] = sum_k W[k] * a[m-k] (convolution-like)
    term2 = irfft(Wf * A, fft_size)[:n]

    grad = term1 + term2
    return sm, grad

    a = np.exp(z_shifted)
    a /= np.sum(a)

def _build_toeplitz_conv_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A

            if it > 5 and it % 8 == 0:
                rng = np.random.default_rng(it * 7919 + 1)
                a = _project_simplex(best_a * np.exp(rng.normal(0, 0.03, n)))
            elif it > 20:
                break

    return best_a, best_max


def _generate_diverse_starts(n, num_starts=10, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    starts.append(np.ones(n) / n)

    # Gaussian (narrow)
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 8.0) ** 2))
    starts.append(bell / np.sum(bell))

    # Gaussian (wide)
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 4.0) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # Triangular
    tri = 1.0 - np.abs(i - center) / max(center, 1)
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    # Cosine taper (Hann-like)
    hann = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
    starts.append(hann / np.sum(hann))

    # Random parametric
    for _ in range(max(0, num_starts - 5)):
        sigma = n * (0.1 + 0.4 * rng.random())
        freq = int(1 + rng.integers(0, 6))
        amp = 0.1 + 0.6 * rng.random()
        phase = 2 * np.pi * rng.random()
        bell_p = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
        a = np.maximum(bell_p * mod, 0)
        s = np.sum(a)
        if s > 1e-15:
            starts.append(a / s)
        else:
            starts.append(np.ones(n) / n)

    return starts


def _resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def _try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = _compute_obj(parent)
            print(f"  Parent loaded: n={parent.size}, obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Continuation L-BFGS + LP polishing + FISTA + multi-start.
    Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_obj = np.inf

    # Phase 0: Load parent
    print("=== Phase 0: Loading parent ===")
    parent, parent_obj = _try_load_parent()
    if parent is not None:
        best_seq = parent.copy()
        best_obj = parent_obj

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"  Initial best: {best_obj:.8f} (n={n_parent})")

    # Phase 1: Quick L-BFGS on parent
    print("\n=== Phase 1: Continuation L-BFGS on parent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 120.0)
        a_opt, max_opt = _continuation_lbfgs(best_seq, time_budget=alloc)
        obj_opt = 2.0 * len(a_opt) * max_opt / 1.0  # sum=1 after normalization
        # Actually compute properly
        obj_opt = _compute_obj(a_opt)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_seq = a_opt.copy()
            print(f"  L-BFGS improved: {best_obj:.8f}")
        else:
            print(f"  L-BFGS: {obj_opt:.8f} (no improvement over {best_obj:.8f})")

    # Phase 2: LP polishing
    print("\n=== Phase 2: LP polishing ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_cur = len(best_seq)
        if n_cur <= 1000 and remaining > 60:
            alloc = min(remaining * 0.2, 80.0)
            a_lp, max_lp = _lp_alternate(best_seq, method='minpeak',
                                         max_iters=30, time_budget=alloc * 0.6)
            obj_lp = _compute_obj(a_lp)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_seq = a_lp.copy()
                print(f"  MinPeak LP: {best_obj:.8f} ** IMPROVED **")
            else:
                print(f"  MinPeak LP: {obj_lp:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_lp2, _ = _lp_alternate(best_seq, method='maxsum',
                                         max_iters=20, time_budget=min(remaining * 0.3, 40.0))
                obj_lp2 = _compute_obj(a_lp2)
                if obj_lp2 < best_obj:
                    best_obj = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  MaxSum LP: {best_obj:.8f} ** IMPROVED **")
                else:
                    print(f"  MaxSum LP: {obj_lp2:.8f}")

    # Phase 3: FISTA refinement
    print("\n=== Phase 3: FISTA subgradient refinement ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.2, 60.0)
            a_fista, max_fista = _fista_subgradient(best_seq, time_budget=alloc)
            obj_fista = _compute_obj(a_fista)
            if obj_fista < best_obj:
                best_obj = obj_fista
                best_seq = a_fista.copy()
                print(f"  FISTA: {best_obj:.8f} ** IMPROVED **")
            else:
                print(f"  FISTA: {obj_fista:.8f}")

    # Phase 4: Multi-start exploration at different lengths
    print("\n=== Phase 4: Multi-start exploration ===")
    n_values = sorted(set([
        n_parent - 50, n_parent, n_parent + 50,
        n_parent - 100, n_parent + 100
    ]))
    n_values = [nv for nv in n_values if 400 <= nv <= 1200]

    for n_trial in n_values:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_trial == n_parent and best_seq is not None:
            starts = [_project_simplex(best_seq.copy())]
        else:
            if best_seq is not None:
                starts = [_project_simplex(_resample(best_seq, n_trial))]
            else:
                starts = [np.ones(n_trial) / n_trial]
            starts.extend(_generate_diverse_starts(n_trial, num_starts=5, seed=seed + n_trial))

        alloc_per = min(remaining / (len(starts) + 1), 25.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 80:
                break
            t_alloc = min(alloc_per, 20.0)
            if t_alloc < 5:
                break

            # Quick L-BFGS
            a_opt, max_opt = _continuation_lbfgs(s, time_budget=t_alloc * 0.7)
            obj_opt = _compute_obj(a_opt)

            if obj_opt < best_obj:
                best_obj = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_trial}, start {si}: {best_obj:.8f} ** NEW BEST **")
            else:
                print(f"  n={n_trial}, start {si}: {obj_opt:.8f}")

    # Phase 5: Intensive polishing on the best
    print(f"\n=== Phase 5: Intensive polishing (best: {best_obj:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Final L-BFGS with more iterations
            a_final, max_final = _continuation_lbfgs(best_seq, time_budget=min(remaining * 0.3, 60.0))
            obj_final = _compute_obj(a_final)
            if obj_final < best_obj:
                best_obj = obj_final
                best_seq = a_final.copy()
                print(f"  Final L-BFGS: {best_obj:.8f} ** IMPROVED **")

            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_lp, _ = _lp_alternate(best_seq, method='minpeak',
                                        max_iters=30, time_budget=min(remaining * 0.5, 50.0))
                obj_lp = _compute_obj(a_lp)
                if obj_lp < best_obj:
                    best_obj = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  Final LP: {best_obj:.8f} ** IMPROVED **")

    # Phase 6: Perturbation-based local search
    print("\n=== Phase 6: Perturbation search ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 12345)

        for pidx in range(20):
            if time.monotonic() > deadline - 10:
                break
            if remaining < 5:
                break

            scale = 0.01 + 0.015 * (pidx % 5)
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_pert = _project_simplex(a_pert)

            t_alloc = min(remaining * 0.15, 15.0)
            if t_alloc < 4:
                break

            a_opt, max_opt = _continuation_lbfgs(a_pert, time_budget=t_alloc)
            obj_opt = _compute_obj(a_opt)

            if obj_opt < best_obj:
                best_obj = obj_opt
                best_seq = a_opt.copy()
                print(f"  Perturb {pidx} (s={scale:.3f}): {best_obj:.8f} ** IMPROVED **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(500) / 500
        best_obj = _compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if np.isfinite(final_val) and final_val < best_obj:
        best_obj = final_val
    elif not np.isfinite(final_val):
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
