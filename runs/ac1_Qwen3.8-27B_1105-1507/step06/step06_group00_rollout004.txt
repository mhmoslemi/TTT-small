import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_grad_conv(a, w):
    """
    Compute gradient of weighted sum of convolution values.
    grad[i] = 2 * sum_j a[j] * w[i+j]
    where w has length 2n-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1 + n - 1:
        size <<= 1
    # flip_a * w gives us what we need
    flip_a = a[::-1]
    fa = rfft(flip_a, size)
    fw = rfft(w, size)
    conv = irfft(fa * fw, size)
    return 2.0 * conv[n-1:n-1+n]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Binary search for threshold
    lo = np.min(v) - 1
    hi = np.max(v) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > 1:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.maximum(v - tau, 0)

def smooth_objective(a, beta):
    """Compute (1/beta) * log(sum_k exp(beta * conv_k)) as smooth max approximation."""
    conv = compute_conv(a)
    c = beta * conv
    c_max = np.max(c)
    return (c_max + np.log(np.sum(np.exp(c - c_max)))) / beta

def smooth_grad(a, beta):
    """Gradient of smooth objective w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    c = beta * conv
    c_max = np.max(c)
    w = np.exp(c - c_max)
    w /= np.sum(w)
    # w are the softmax weights over convolution positions
    return compute_grad_conv(a, w)

def gradient_descent_smooth(a, beta_schedule, iters_per_beta=100, lr_init=0.1, verbose=False):
    """
    Gradient descent on smooth approximation with beta annealing.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = np.inf
    start = time.monotonic()
    
    for step, beta in enumerate(beta_schedule):
        lr = lr_init / (1 + 0.01 * step)
        for it in range(iters_per_beta):
            if time.monotonic() - start > 1200:
                break
            grad = smooth_grad(a, beta)
            # Project gradient onto tangent space of simplex (sum=0)
            grad_centered = grad - np.mean(grad)
            a_new = a - lr * grad_centered
            a_new = project_simplex(a_new)
            
            obj = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
            a = a_new
        
        if verbose and step % 5 == 0:
            obj = 2.0 * n * float(np.max(compute_conv(a)))
            print(f"    beta={beta:.1f}, obj={obj:.6f}, time={time.monotonic()-start:.1f}s")
    
    return best_a
