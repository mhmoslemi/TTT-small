"""
Hybrid optimizer: Fast Multiplicative Weight Updates (MWU) + LP Alternating +

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def mwu_gradient(a, k_indices):
    """Compute gradient of average b_k over k in k_indices w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    K = len(k_indices)
    for k in k_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        d_start = k + 1 - j_end
        d_end = k - j_start + 1
        grad[j_start:j_end] += a[d_start:d_end][::-1]
    grad /= K
    return grad


def mwu_step(a, eta, k_indices):
    """One MWU step targeting specific peak positions."""
    n = len(a)
    grad = mwu_gradient(a, k_indices)
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new /= s
    return a_new

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]
    if solver:
        solvers = [solver] + solvers
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None

                    best_a = a.copy()

def subgradient_momentum(a, time_budget=60.0, top_k=5, verbose=False):
    """
    Nesterov-accelerated subgradient method with momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.9
    step_size = 0.02
    x = a.copy()  # Nesterov's lookahead point

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        # Compute gradient at lookahead point
        conv = compute_conv(x)
        k_indices = np.argsort(conv)[::-1][:top_k]
        grad = mwu_gradient(x, k_indices)
        
        # Center the gradient (project to sum=0 hyperplane)
        grad_centered = grad - np.mean(grad)

        # Momentum update
        momentum = beta * momentum - step_size * grad_centered
        new_a = a + momentum

        # Project to simplex (a >= 0, sum(a) = 1)
        new_a = np.maximum(new_a, 0)
        s_new = float(np.sum(new_a))
        if s_new < 1e-10:
            new_a = a.copy()
            s_new = 1.0
        new_a = new_a / s_new

        # Nesterov: update lookahead
        t_k = 2.0 / (it + 2.0)
        x = new_a + (1 - t_k) * (new_a - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = new_a
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive step size
        if it % 100 == 99:
            step_size = max(step_size * 0.95, 0.001)

    return best_a


def coordinate_descent_polish(a, time_budget=30.0):
    """
    Coordinate descent: adjust pairs of coordinates to reduce peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        k_star = int(np.argmax(conv))

        # Try moving mass from high contributors to low contributors
        for i in range(0, n, 2):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i:
                # Also check other peaks
                for frac in [0.01, 0.03, 0.05]:
                    delta = frac * a[i]
                    if delta > a[i]:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        if not improved:
            # Try small random perturbations
            rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
            for _ in range(10):
                idx = rng.integers(0, n)
                jdx = (idx + 1 + rng.integers(0, n - 1)) % n
                delta = 0.01 * a[idx]
                a[idx] -= delta
                a[jdx] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a /= s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 50:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 4: Multi-n exploration (250s) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        max(200, n_parent - 100), max(200, n_parent - 50),
        n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick MWU + LP
        alloc_mwu = min(30.0, remaining * 0.2)
        alloc_lp = min(40.0, remaining * 0.3)
        
        a_opt = mwu_optimize(a_try, time_budget=alloc_mwu, top_k=10)
        a_opt = lp_alternating(a_opt, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final polish (remaining) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    a_final = best_seq.copy()

    if remaining > 20:
        # LP polish
        alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_alternating(a_final, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    if remaining > 15:
        # Coordinate descent
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 40.0)
        a_cd = coordinate_descent_polish(a_final, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a_final = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_cd:.8f}")

    if remaining > 15:
        # One more LP round
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.4, 30.0)
        a_last = lp_alternating(a_final, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")
        else:
            print(f"  Final LP2: obj={obj_last:.8f}")

    # === Phase 6: Random restarts with best structure ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(10):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # Perturb best solution
            scale = 0.03 + 0.02 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

            # Quick optimization
            alloc = min(remaining * 0.3, 25.0)
            a_opt = mwu_optimize(a_r, time_budget=alloc * 0.4, top_k=10)
            a_opt = lp_alternating(a_opt, time_budget=alloc * 0.6)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
