"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best response (min t s.t. conv(a,g)<=t, sum(g)=1, g>=0)
2. Line search over convex combination step sizes
3. Gerchberg-Saxton spectral initialization
4. Multi-length search (1024, 1536, 2048, 3072) with time allocation based on expected LP cost
5. Anderson acceleration on the LP mirror descent iterates

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv_cross(a, b):
    """FFT-based linear convolution of a with b."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def precompute_conv_indices(n):
    """Precompute (row, col, a_idx) triples for convolution matrix."""
    triples = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            triples.append((k, i, k - i))
    return triples


def build_lp_matrices(n, a, triples):
    """Build sparse matrices for the LP: min t s.t. A@g - t <= 0, sum(g)=1, g>=0."""
    # A is (2n-1) x n, plus a -1 column for t, giving (2n-1) x (n+1)
    rows = []
    cols = []
    vals = []
    for (k, i, a_idx) in triples:
        rows.append(k)
        cols.append(i)
        vals.append(a[a_idx])
    # Add the -1 column for t
    for k in range(2 * n - 1):
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    return A_ub, b_ub, A_eq, b_eq


def solve_best_response(n, a, triples):
    """Solve LP: min t s.t. conv(a,g) <= t, sum(g)=1, g>=0. Returns (g*, t*)."""
    A_ub, b_ub, A_eq, b_eq = build_lp_matrices(n, a, triples)
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * n + [(None, None)]
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30.0})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            gs = g.sum()
            if gs > 1e-10:
                g /= gs
            t_val = float(r.x[n])
            return g, t_val
    except Exception:
        pass
    return None, float('inf')


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: iteratively flatten the power spectrum while maintaining non-negativity."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress dynamic range to flatten autocorrelation
        mag_min = mag.min()
        mag_max = mag.max()
        if mag_max - mag_min < 1e-15:
            break
        # Power-law compression
        power = 0.3 + 0.4 * beta
        mag_comp = (mag - mag_min) / (mag_max - mag_min + 1e-30)
        mag_comp = mag_comp ** power
        mag_new = mag_min + mag_comp * (mag_max - mag_min)
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def resample_seq(src, n):
    """Resample a sequence to length n using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            M_try = fast_conv(a_try).max()
            if M_try < best_M_trial:
                best_M_trial = M_try
                best_t = t
        
        if best_M_trial < best_M - 1e-14:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            s = a.sum()
            a = a / s
            M_new = fast_conv(a).max()
            if M_new < best_M:
                best_M = M_new
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        
        if no_imp > 80:
            break
        
        if (it + 1) % 200 == 0:
            J_cur = 2.0 * n * best_M
            print(f"    iter {it+1}: M={best_M:.8f} J={J_cur:.6f} step={step_base:.4f}", flush=True)
    
    return best_a, best_M


def propose_candidate(seed=42, budget_s=1500, **kw):
    """Main search function: multi-length LP mirror descent with line search."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    
    best_seq = None
    best_score = float('inf')
    
    def report_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            s = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a.copy()
            el = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={s:.8f} n={a.size} t={el:.0f}s {label}", flush=True)
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            report_update(parent, "(parent)")
            print(f"Parent: n={parent.size}, score={best_score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass
    
    if parent is None:
        parent = np.ones(512)
        report_update(parent, "(default)")
    
    # Determine lengths to try
    # Longer sequences have better theoretical bounds but slower LP
    lengths_to_try = [1024, 1536, 2048]
    # Add parent length if reasonable
    if parent.size in [768, 1280, 2560]:
        if parent.size not in lengths_to_try:
            lengths_to_try.append(parent.size)
    lengths_to_try = sorted(set(lengths_to_try))
    
    print(f"Lengths to try: {lengths_to_try}", flush=True)
    print(f"Budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s", flush=True)
    
    # Pre-compute indices for each length
    triples_cache = {}
    for n in lengths_to_try:
        triples_cache[n] = precompute_conv_indices(n)
        print(f"  n={n}: {len(triples_cache[n])} non-zero entries in conv matrix", flush=True)
    
    # === Main optimization loop over lengths ===
    # Allocate time proportionally to expected value (longer = better bound)
    # but inversely to LP cost
    total_time = deadline - time.monotonic() - 60  # reserve 60s for polish
    weights = []
    for n in lengths_to_try:
        # Expected LP cost scales roughly as n^1.5
        cost = (n / 1024.0) ** 1.5
        # Value scales as 1/n (longer sequences have better bounds)
        value = 1.0 / np.sqrt(n)
        weights.append(value / cost)
    total_w = sum(weights)
    weights = [w / total_w for w in weights]
    
    print(f"\nTime allocation: " + ", ".join(f"n={lengths_to_try[i]}: {weights[i]*total_time:.0f}s" for i in range(len(lengths_to_try))), flush=True)
    
    results_per_length = {}
    
    for idx, n in enumerate(lengths_to_try):
        time_for_n = weights[idx] * total_time
        phase_deadline = time.monotonic() + time_for_n
        remaining = phase_deadline - time.monotonic()
        if remaining < 30:
            continue
        
        print(f"\n{'='*60}", flush=True)
        print(f"Optimizing n={n} (budget {time_for_n:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        
        triples = triples_cache[n]
        
        # Generate initializations
        inits = make_initializations(n, rng, parent)
        
        # For each initialization, run a short GS + LP
        best_a_n = None
        best_M_n = float('inf')
        
        # Allocate per-init time
        n_inits = len(inits)
        # Give more time to first few inits (parent-based)
        init_weights = []
        for i in range(n_inits):
            if i < 3 and parent is not None:
                init_weights.append(2.0)
            else:
                init_weights.append(1.0)
        tw = sum(init_weights)
        init_weights = [w / tw for w in init_weights]
        
        for i_init, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline - 10:
                break
            init_budget = init_weights[i_init] * (phase_deadline - time.monotonic())
            if init_budget < 5:
                continue
            
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()
            
            # Quick GS polish
            a_gs = gs_flat_spectral(a0, n_iters=50, beta=0.85)
            M_gs = fast_conv(a_gs).max()
            
            if M_gs < best_M_n:
                # Run LP mirror descent on this init
                sub_deadline = time.monotonic() + init_budget * 0.9
                a_opt, M_opt = lp_mirror_descent_v2(n, a_gs, sub_deadline, triples, max_iters=5000)
                if M_opt < best_M_n:
                    best_M_n = M_opt
                    best_a_n = a_opt.copy()
                    J_cur = 2.0 * n * M_opt
                    report_update(a_opt, f"(n={n}, init={i_init})")
            else:
                # Still try LP, might escape local min
                sub_deadline = time.monotonic() + init_budget * 0.5
                a_opt, M_opt = lp_mirror_descent_v2(n, a_gs, sub_deadline, triples, max_iters=3000)
                if M_opt < best_M_n:
                    best_M_n = M_opt
                    best_a_n = a_opt.copy()
                    J_cur = 2.0 * n * M_opt
                    report_update(a_opt, f"(n={n}, init={i_init})")
        
        if best_a_n is not None:
            results_per_length[n] = best_a_n.copy()
            J_n = 2.0 * n * best_M_n
            print(f"  n={n} best: M={best_M_n:.8f} J={J_n:.6f}", flush=True)
    
    # === Phase: Polish the best result ===
    print(f"\n{'='*60}", flush=True)
    print(f"POLISH PHASE", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        n_best = best_seq.size
        # Make sure we have triples for this length
        if n_best not in triples_cache:
            triples_cache[n_best] = precompute_conv_indices(n_best)
        triples_best = triples_cache[n_best]
        
        polish_deadline = deadline
        remaining = polish_deadline - time.monotonic()
        if remaining > 30:
            # Continue LP on the best sequence
            a_polish, M_polish = lp_mirror_descent_v2(
                n_best, best_seq, polish_deadline, triples_best, max_iters=10000)
            report_update(a_polish, "(polish)")
            
            # Try v1 on the result too (different dynamics)
            if time.monotonic() < deadline - 30:
                a_polish2, M_polish2 = lp_mirror_descent(
                    n_best, a_polish, deadline, triples_best, max_iters=5000)
                report_update(a_polish2, "(polish2)")
    
    # === Phase: Try perturbing the best and re-optimizing ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = best_seq.size
        if n_best not in triples_cache:
            triples_cache[n_best] = precompute_conv_indices(n_best)
        triples_best = triples_cache[n_best]
        
        for noise_level in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline - 10:
                break
            a_pert = best_seq + rng.normal(0, noise_level, n_best)
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / a_pert.sum()
            # Short LP polish
            a_res, _ = lp_mirror_descent_v2(n_best, a_pert, min(time.monotonic() + 30, deadline - 5),
                                             triples_best, max_iters=2000)
            report_update(a_res, f"(perturb {noise_level})")
    
    # === Fallback ===
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = compute_J(best_seq)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_seq]
