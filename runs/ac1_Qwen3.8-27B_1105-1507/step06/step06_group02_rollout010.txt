import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse convolution matrix A where (A*g)_k = sum_j a[k-j]*g_j"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, prev_x=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0"""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        # Warm start: if we have a previous solution, use it
        args = {}
        if prev_x is not None:
            args['x0'] = np.maximum(prev_x, 0)
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception as e:
        pass
    return None

def subgradient_step(a, active_indices, step_size):
    """Take a subgradient step using exact active set."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    # Gradient at active indices
    grad = np.zeros(n)
    for k in active_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= len(active_indices)  # average over active set
    # Projected gradient step
    a_new = np.maximum(a - step_size * grad, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new

def run_lp_mirror_adaptive(a, deadline, max_iters=500, initial_t=0.5):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    prev_g = a.copy()
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M, prev_x=prev_g)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        prev_g = g.copy()
        
        # Backtracking line search
        improved = False
        t = initial_t
        for _ in range(8):
            a_t = (1 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-12:
                a = a_t.copy()
                best_M = M_t
                best_a = a.copy()
                improved = True
                no_imp = 0
                break
            t *= 0.5
            if t < 1e-4:
                break
        
        if not improved:
            no_imp += 1
            if no_imp > 20:
                # Try a random perturbation

                a_pert = np.maximum(a + pert, 1e-8)
                a_pert /= a_pert.sum()
                M_pert = fast_conv(a_pert).max()
                if M_pert < best_M:
                    a = a_pert
                    best_M = M_pert
                    best_a = a.copy()
                    no_imp = 0
                else:
                    break
    
    return best_a

def run_subgradient_momentum(a, deadline, max_iters=2000, step_init=0.1, step_end=0.001, momentum=0.9):
    """Subgradient with Nesterov momentum, using exact active set."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        step = step_init * (step_end / step_init) ** frac
        
        b = fast_conv(a)
        M = b.max()
        # Find active set (within 5% of max)
        threshold = M * (1 - 0.05)
        active = np.where(b >= threshold)[0].tolist()
        if not active:
            active = [int(np.argmax(b))]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * a[k - j]
        grad /= (len(active) + 1e-10)
        
        # Momentum step
        v = momentum * v + step * grad
        a_new = np.maximum(a - v, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton spectral flattening."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress dynamic range
        mag_max = mag.max() + 1e-30
        mag_mean = mag.mean() + 1e-30
        mag_new = (1 - beta) * mag + beta * mag_mean
        # Power compression
        power = 0.7
        mag_p = (mag / mag_max) ** power
        mag_p = mag_p / (mag_p.max() + 1e-30) * mag_max
        mag_final = 0.5 * mag_new + 0.5 * mag_p
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 1e-8)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a

def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=5.0, tau_end=500.0,
                        lr_init=0.3, lr_end=0.002):
    """Nesterov smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta_mom = 0.9
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / (exp_b.sum() + 1e-30)
        
        grad = np.zeros(n)
        active = np.where(p > 1e-10)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * p[k] * y[k - j]
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_mom * beta_mom)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta_mom * (y - a)), 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
        
        M_new = fast_conv(a).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a.copy()

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def fft_subgradient(a, active_indices):
    """Compute subgradient of max(conv) at active set using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    v = np.zeros(size)
    for k in active_indices:
        if 0 <= k < size:
            v[k] = 1.0
    # Cross-correlation: grad_j = sum_k v_k * a_{k-j}
    # This is the convolution of v (reversed) with a
    v_rev = np.zeros(size)
    for k in active_indices:
        if 0 <= k < size:
            v_rev[(size - k) % size] = 1.0
    # Actually, simpler: grad_j = sum_k a_{k-j} = (a convolved with v_indicator) at position j
    # In FFT: FFT(grad) = FFT(a) * FFT(v)  (with appropriate handling)
    fa = np.fft.rfft(a, size)
    fv = np.fft.rfft(v, size)
    grad_full = np.fft.irfft(fa * fv, size)
    return grad_full[:n]

def resample_seq(src, n):

    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kw):
    global rng_global

    deadline = time.monotonic() + budget_s - 25.0

        pass

        best.update(parent)

    print(f"Starting optimization, base_n={base_n}", flush=True)

    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Spectral + Nesterov (n={base_n})", flush=True)
    print(f"{'='*50}", flush=True)
    t_phase_start = time.monotonic()
    p1_budget = 150.0
    dl1 = min(time.monotonic() + p1_budget, deadline - 800)

    a_gs = gs_flat_spectral(parent, n_iters=150, beta=0.9)
    best.update(a_gs)

    a_smooth = run_nesterov_fft(a_gs, dl1, max_iters=8000,
                                 tau_start=5.0, tau_end=400.0,
                                 lr_init=0.25, lr_end=0.001, active_frac=0.15)
    best.update(a_smooth)
    print(f"  Phase 1: {time.monotonic()-t_phase_start:.1f}s", flush=True)

    # === PHASE 2: LP Mirror with extrapolation ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Mirror + Extrapolation (n={base_n})", flush=True)
    print(f"{'='*50}", flush=True)
    t_phase_start = time.monotonic()
    remaining = deadline - time.monotonic() - 200
    p2_budget = min(500.0, max(100.0, remaining * 0.5))
    dl2 = time.monotonic() + p2_budget

    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Subgradient Momentum (n={base_n})", flush=True)
    print(f"{'='*50}", flush=True)
    t_phase_start = time.monotonic()
    remaining = deadline - time.monotonic() - 150
    p3_budget = min(150.0, max(30.0, remaining * 0.2))
    dl3 = time.monotonic() + p3_budget

    a_sub = run_subgradient_momentum_fft(a_lp, dl3, max_iters=5000,
                                          step_init=0.06, step_end=0.0003, momentum=0.94)
    best.update(a_sub)

    # LP polish from subgradient result
    if time.monotonic() < deadline - 60:
        a_lp2 = run_lp_mirror_extrap(a_sub, deadline - 60, max_iters=150, initial_t=0.25)
        best.update(a_lp2)

    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*50}", flush=True)
    t_phase_start = time.monotonic()

    for delta in [-128, -64, -32, 32, 64, 128, 256, 512]:

        if n_abs != base_n and 256 <= n_abs <= 4000:
            lengths_to_try.append(n_abs)
    lengths_to_try = sorted(set(lengths_to_try))

    remaining = deadline - time.monotonic() - 100

        lengths_to_try = lengths_to_try[:2]
    per_length = max(15.0, remaining / max(1, len(lengths_to_try)))

        a_res = resample_seq(best.seq, n_try)
        a_res = np.maximum(a_res, 1e-6)

        a_opt = run_nesterov_fft(a_res, dl_sub, max_iters=3000,
                                  tau_start=10, tau_end=300,
                                  lr_init=0.2, lr_end=0.003, active_frac=0.1)
        J_opt = compute_J(a_opt)
        best.update(a_opt)
        results_per_length[n_try] = (a_opt, J_opt)

        # LP polish if time allows
        if time.monotonic() < deadline - 40:
            n_est = n_try
            est_lp_time = 0.005 * (n_est / 256.0) ** 2
            dl_lp = time.monotonic() + min(per_length * 0.5, max(10, 60 / max(1, est_lp_time / 0.1)))
            a_lp_opt = run_lp_mirror_extrap(a_opt, dl_lp, max_iters=30, initial_t=0.3)
            J_lp = compute_J(a_lp_opt)
            best.update(a_lp_opt)
            if J_lp < J_opt:
                results_per_length[n_try] = (a_lp_opt, J_lp)

        print(f"  n={n_try}: best J so far for this length", flush=True)

    print(f"  Phase 4: {time.monotonic()-t_phase_start:.1f}s", flush=True)

    # === PHASE 5: Final intensive optimization on best ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*50}", flush=True)
    t_phase_start = time.monotonic()

    if best.seq is not None and time.monotonic() < deadline - 15:

        remaining_final = deadline - time.monotonic()

        if remaining_final > 100:

            a_f1 = run_lp_mirror_extrap(best.seq, time.monotonic() + remaining_final * 0.4,
                                         max_iters=200, initial_t=0.3)
            best.update(a_f1)
            if time.monotonic() < deadline - 20:
                a_f2 = run_subgradient_momentum_fft(a_f1, deadline - 20, max_iters=3000,
                                                     step_init=0.03, step_end=0.0002, momentum=0.95)

        elif remaining_final > 30:
            a_f1 = run_lp_mirror_extrap(best.seq, deadline - 5, max_iters=50, initial_t=0.2)
            best.update(a_f1)
        else:

                                     tau_start=20, tau_end=200,
                                     lr_init=0.1, lr_end=0.001, active_frac=0.05)
            best.update(a_f1)

    best.report("Phase 5")
    print(f"  Phase 5: {time.monotonic()-t_phase_start:.1f}s", flush=True)

    # Final report
    best.report("FINAL")
    print(f"\n{'='*50}", flush=True)
    print(f"RESULT: J={best.score:.8f} n={best.seq.size} total_time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is None:
        best.seq = np.ones(128)

    return [float(x) for x in best.seq]
