    while size < 2 * n - 1:

    fa = rfft(a, size)
    conv = irfft(fa * fa, size)

def normalize(a):

def fast_subgradient(a, max_iters=2000, time_budget=60.0, top_k=10, seed=42):
    """Fast FFT-based subgradient with momentum and adaptive top-k peak tracking."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)
    lr = 0.02
    momentum = 0.85
    start = time.monotonic()
    
    size = 1
    while size < 3 * n:
        size <<= 1
    
    iter_count = 0
    while iter_count < max_iters:
        if time.monotonic() - start > time_budget:
            break
        iter_count += 1
        
        a_look = np.maximum(a + momentum * v, 0)
        if float(np.sum(a_look)) < 1e-15:
            a_look = a.copy()
        
        conv = compute_conv(a_look)
        cmax = float(np.max(conv))
        
        # Top-k with soft weights
        k_eff = min(top_k, len(conv))
        if k_eff < len(conv):
            peak_indices = np.argsort(conv)[-k_eff:]
        else:
            peak_indices = np.arange(len(conv))
        
        peak_vals = conv[peak_indices]
        # Adaptive temperature
        tau = max(0.02 * cmax / n, 1e-12) * (1.0 + 0.5 * iter_count / max_iters)
        weights_vec = np.exp((peak_vals - peak_vals.max()) / tau)
        wsum = float(np.sum(weights_vec))
        if wsum < 1e-300:
            weights_vec = np.zeros(len(peak_indices))
            weights_vec[0] = 1.0
        else:
            weights_vec /= wsum
        
        w = np.zeros(2 * n - 1)
        for idx, k in enumerate(peak_indices):
            w[k] = weights_vec[idx]
        
        # Gradient via FFT cross-correlation
        a_flip = a_look[::-1]
        Wf = rfft(w, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = corr[n-1:2*n-1]
        
        s = float(np.sum(a_look))
        full_grad = 2.0 * n * (grad / (s*s) - 2.0 * cmax / (s*s*s))
        full_grad -= np.mean(full_grad)
        
        # Adaptive per-coordinate scaling
        grad_mag = np.sqrt(np.mean(full_grad ** 2) + 1e-30)
        
        v = momentum * v - (lr / grad_mag) * full_grad
        a_new = np.maximum(a + v, 0)
        a_new = normalize(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Occasionally accept worse solutions (simulated annealing flavor)
            if np.random.default_rng(iter_count + seed).random() < 0.01 * (1 - iter_count/max_iters):
                a = a_new
            else:
                lr *= 0.95
                if lr < 1e-6:
                    break
    
    return best_a

def spectral_flatten(a, max_iters=200, time_budget=40.0):
    """Spectral domain optimization: iteratively flatten power spectrum."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size >= 2 * n:
        size <<= 1
    
    alpha = 0.3  # compression strength
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        P_max = float(np.max(P))
        P_min = float(np.min(P[P > 0])) if np.any(P > 0) else 0
        
        if P_max < 1e-30:
            break
        
        # Compress: P_new = P^alpha (normalized to preserve total power)
        P_new = P ** alpha
        P_new *= P.sum() / P_new.sum()
        
        # Reconstruct with new amplitude, keep phase
        phase = np.angle(A)
        mag = np.sqrt(P_new)
        A_new = mag * np.exp(1j * phase)
        
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            a = 0.7 * normalize(a) + 0.3 * normalize(a_new)
        
        # Schedule
        alpha = 0.2 + 0.4 * (it / max_iters)
    
    return best_a

            a = a_try

mirror descent (multiplicative) updates, spectral equalization, and structured

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    Af = rfft(a_flip, size)

        else:
            peak_indices = np.arange(len(conv))

        peak_vals = conv[peak_indices]
        # Temperature: starts narrow (focus on main peak), widens over time
        tau = max(0.01 * cmax / n, 1e-12) * (1.0 + 2.0 * min(it / max_iters, 1.0))
        weights_vec = np.exp((peak_vals - peak_vals.max()) / tau)
        wsum = float(np.sum(weights_vec))
        if wsum < 1e-300:
            weights_vec = np.zeros(len(peak_indices))
            weights_vec[np.argmax(peak_vals)] = 1.0
        else:
            weights_vec /= wsum

        # Build weight vector
        w = np.zeros(2 * n - 1)
        for idx, k in enumerate(peak_indices):
            w[k] = weights_vec[idx]

        # Gradient via FFT
        grad = compute_gradient_fft(a, w, size)

        # Full subgradient of 2n * max_b (with sum=1 constraint, so we just need grad of max_b)
        # d(max_b)/da[i] ≈ sum_k w[k] * a[max(0,k-n+1)..min(k,n-1)] (the valid range)
        # Already computed as 'grad'
        # Project: remove component along (1,1,...,1) for sum constraint
        grad_centered = grad - np.mean(grad)

        # Momentum
        momentum_buf = momentum_alpha * momentum_buf + (1 - momentum_alpha) * grad_centered

        # Mirror descent (multiplicative) update
        scaled = eta * momentum_buf / (np.std(momentum_buf) + 1e-15)
        scaled = np.clip(scaled, -50, 50)
        a_new = a * np.exp(-scaled)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            eta = min(eta * 1.02, 0.3)
        else:
            patience += 1
            if patience > max_patience:
                # Reheat: perturb to escape local minimum
                a = normalize(a * np.exp(rng.normal(0, 0.05, n)))
                eta = 0.05
                patience = 0
                momentum_buf = np.zeros(n)
            else:
                eta *= 0.97

    return best_a


def subgradient_euclidean(a, max_iters=5000, time_budget=60.0, top_k=8, seed=42):
    """
    Euclidean (additive) subgradient with Nesterov momentum.
    Standard projected gradient with adaptive step size.
    """
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed + 7)

    size = 1
    while size < 3 * n:
        size <<= 1

    lr = 0.03
    momentum = 0.85
    v = np.zeros(n)
    patience = 0
    max_patience = 400

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov lookahead
        a_look = np.maximum(a + momentum * v, 0)
        s_look = float(np.sum(a_look))
        if s_look < 1e-15:
            a_look = a.copy()
        else:
            a_look = a_look / s_look

        conv = compute_conv(a_look)
        cmax = float(np.max(conv))
        if cmax < 1e-15:
            break

        k_eff = min(top_k + it // 80, len(conv))
        if k_eff < len(conv):
            peak_indices = np.argsort(conv)[-k_eff:]
        else:
            peak_indices = np.arange(len(conv))

        peak_vals = conv[peak_indices]
        tau = max(0.02 * cmax / n, 1e-12) * (1.0 + 1.5 * min(it / max(float(max_iters), 1), 1.0))
        weights_vec = np.exp((peak_vals - peak_vals.max()) / tau)
        wsum = float(np.sum(weights_vec))
        if wsum < 1e-300:
            weights_vec = np.zeros(len(peak_indices))
            weights_vec[np.argmax(peak_vals)] = 1.0
        else:
            weights_vec /= wsum

        w = np.zeros(2 * n - 1)
        for idx, k in enumerate(peak_indices):
            w[k] = weights_vec[idx]

        grad = compute_gradient_fft(a_look, w, size)
        grad_centered = grad - np.mean(grad)

        # Normalize gradient for stable updates
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm > 1e-15:
            grad_centered = grad_centered / grad_norm

        v = momentum * v - lr * grad_centered
        a_new = np.maximum(a + v, 0)
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            lr = min(lr * 1.01, 0.2)
        else:
            patience += 1
            if patience > max_patience:
                a = normalize(a * np.exp(rng.normal(0, 0.04, n)))
                v = np.zeros(n)
                lr = 0.03
                patience = 0
            else:
                lr *= 0.98

    return best_a


def spectral_optimize(a, max_iters=300, time_budget=50.0):
    """
    Iterative spectral flattening with adaptive compression.
    Alternates between frequency-domain amplitude compression and
    time-domain non-negativity projection.
    """
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        P_max = float(np.max(P))
        if P_max < 1e-30:
            break

        # Adaptive compression: stronger early, weaker later
        progress = it / float(max_iters)
        alpha = 0.15 + 0.5 * progress

        # Compress power spectrum
        P_safe = np.maximum(P, 1e-30)
        P_new = P_safe ** alpha
        # Preserve total power
        P_new *= P.sum() / P_new.sum()

        # Reconstruct
        phase = np.angle(A)
        mag = np.sqrt(P_new)
        A_new = mag * np.exp(1j * phase)

        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Gradual blend
            blend = 0.3 + 0.3 * progress
            a = normalize((1 - blend) * a + blend * a_new)

    return best_a


def structured_sa(a, time_budget=50.0, seed=123):
    """
    Simulated annealing with structured moves designed for this problem:
    - Peak-directed mass transfer
    - Symmetry breaking (reversal of sub-blocks)
    - Frequency-domain phase shuffling
    """
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    T_init = 0.05
    T_final = 0.001

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(elapsed / time_budget, 1.0)
        T = T_init * (T_final / T_init) ** progress

        a_try = a.copy()
        move = rng.integers(0, 5)

        if move == 0:
            # Peak-directed: move mass from high-autocorrelation-contributing positions
            conv = compute_conv(a)
            k_peak = int(np.argmax(conv))
            # Find i that contributes most to b[k_peak]: a[i]*a[k_peak-i]
            contribs = []
            for i in range(n):
                j = k_peak - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))
            contribs.sort(reverse=True)
            if contribs:
                _, i_src, j_src = contribs[0]
                i_dst = int(rng.integers(0, n))
                if i_dst != i_src and i_dst != j_src:
                    delta = rng.uniform(0.02, 0.08) * max(a[i_src], a[j_src])
                    a_try[i_src] = max(0, a[i_src] - delta)
                    a_try[i_dst] += delta
        elif move == 1:
            # Block reversal (symmetry breaking)
            blen = int(rng.integers(max(3, n // 15), max(4, n // 4)))
            sidx = int(rng.integers(0, n - blen))
            a_try[sidx:sidx + blen] = a_try[sidx:sidx + blen][::-1]
        elif move == 2:
            # Frequency modulation
            freq = rng.uniform(0.3, 10.0)
            amp = rng.uniform(0.03, 0.2)
            ph = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
            a_try = a * mod
        elif move == 3:
            # Gaussian perturbation
            sigma = 0.03 + 0.05 * T / T_init
            a_try = a * np.exp(rng.normal(0, sigma, n))
        else:
            # Swap two blocks
            blen = int(rng.integers(max(2, n // 20), max(3, n // 6)))
            i1 = int(rng.integers(0, n - blen))
            i2 = int(rng.integers(0, n - blen))
            if abs(i1 - i2) > blen:
                tmp = a_try[i1:i1 + blen].copy()
                a_try[i1:i1 + blen] = a_try[i2:i2 + blen]
                a_try[i2:i2 + blen] = tmp

        a_try = normalize(a_try)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        elif rng.random() < np.exp(-(obj_try - best_obj) / (T * best_obj + 1e-10)):
            a = a_try

    return best_a


def resample_sequence(a, new_n):
    """Resample via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def polish_sequence(a, time_budget=30.0, seed=77):
    """Final intensive polish: rapid short alternating subgradient runs."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        half = remaining * 0.45

        # Alternate between mirror and Euclidean
        a1 = subgradient_mirror(a, max_iters=2000, time_budget=half, top_k=5, seed=seed)
        seed += 1
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            a = a1

        if time.monotonic() - start < time_budget - 3:
            half2 = time_budget - (time.monotonic() - start)
            if half2 > 3:
                a2 = subgradient_euclidean(a, max_iters=2000, time_budget=half2 * 0.8, top_k=5, seed=seed)
                seed += 1
                obj2 = compute_obj(a2)
                if obj2 < best_obj:
                    best_obj = obj2
                    best_a = a2.copy()
                    a = a2

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase strategy combining fast subgradient
    (both mirror and Euclidean), spectral optimization, structured SA,
    and length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 35.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            best_score = compute_obj(parent)
            best_seq = parent.copy()
            print(f"Parent loaded (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        print(f"Default uniform start: obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting optimization with n={n_parent}")

    # ============================================================
    # Phase 1: Primary subgradient optimization (mirror descent)
    # This is the main workhorse - fast, many iterations
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Mirror descent subgradient (n={n_parent})")
    print(f"{'='*60}")

    for cycle in range(8):
        if time.monotonic() > deadline - 450:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.12, 70.0)

        # Vary top_k and seed for diversity
        tk = 6 + (cycle % 4) * 2  # 6, 8, 10, 6, 8, 10, 6, 8
        a = normalize(best_seq.copy())
        a_new = subgradient_mirror(a, max_iters=5000, time_budget=alloc,
                                   top_k=tk, seed=seed + cycle * 100)
        obj_new = compute_obj(a_new)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Mirror cycle {cycle} (top_k={tk}): {best_score:.8f} "
                  f"[t={time.monotonic()-start_time:.0f}s]")
        else:
            print(f"  Mirror cycle {cycle} (top_k={tk}): {obj_new:.8f} (no improvement)")

    # ============================================================
    # Phase 2: Euclidean subgradient (different geometry)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Euclidean subgradient")
    print(f"{'='*60}")

    for cycle in range(4):
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.12, 55.0)

        a = normalize(best_seq.copy())
        # Add small perturbation for diversity
        rng = np.random.default_rng(seed + 500 + cycle)
        a = normalize(a * np.exp(rng.normal(0, 0.02, len(a))))
        a_new = subgradient_euclidean(a, max_iters=5000, time_budget=alloc,
                                      top_k=8, seed=seed + 500 + cycle)
        obj_new = compute_obj(a_new)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Euclidean cycle {cycle}: {best_score:.8f} "
                  f"[t={time.monotonic()-start_time:.0f}s]")
        else:
            print(f"  Euclidean cycle {cycle}: {obj_new:.8f} (no improvement)")

    # ============================================================
    # Phase 3: Spectral optimization
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Spectral optimization")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 200:
        a = normalize(best_seq.copy())
        a_new = spectral_optimize(a, max_iters=400, time_budget=50.0)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Spectral: {best_score:.8f}")
        else:
            print(f"  Spectral: {obj_new:.8f} (no improvement)")
            # Blend for potential future improvement
            a_blend = normalize(0.6 * best_seq + 0.4 * a_new)
            obj_blend = compute_obj(a_blend)
            if obj_blend < best_score:
                best_score = obj_blend
                best_seq = a_blend.copy()

    # ============================================================
    # Phase 4: Structured simulated annealing
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Structured SA")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 150:
        a = normalize(best_seq.copy())
        a_new = structured_sa(a, time_budget=55.0, seed=seed + 900)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  SA: {best_score:.8f}")
        else:
            print(f"  SA: {obj_new:.8f} (no improvement)")

        # Second SA run with different seed
        if time.monotonic() < deadline - 100:
            a_new2 = structured_sa(normalize(best_seq), time_budget=40.0, seed=seed + 999)
            obj_new2 = compute_obj(a_new2)
            if obj_new2 < best_score:
                best_score = obj_new2
                best_seq = a_new2.copy()
                print(f"  SA2: {best_score:.8f}")
            else:
                print(f"  SA2: {obj_new2:.8f}")

    # ============================================================
    # Phase 5: Length exploration
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Length exploration")
    print(f"{'='*60}")

    n_best = len(best_seq)
    n_candidates = sorted(set([
        n_best - 100, n_best - 50, n_best - 25, n_best,
        n_best + 25, n_best + 50, n_best + 100, n_best + 200
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 1600]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == n_best:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample_sequence(best_seq, n_try))

        alloc = min(remaining * 0.1, 35.0)
        # Quick optimization at this length
        a_try = subgradient_mirror(a_try, max_iters=2000, time_budget=alloc,
                                   top_k=8, seed=seed + n_try)
        obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # ============================================================
    # Phase 6: Final intensive polish
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        a = normalize(best_seq.copy())
        a_new = polish_sequence(a, time_budget=min(remaining * 0.7, 80.0), seed=seed + 2000)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Polish: {best_score:.8f}")
        else:
            print(f"  Polish: {obj_new:.8f}")

    # Final attempt with very short mirror runs
    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a = normalize(best_seq.copy())
            rng_final = np.random.default_rng(seed + 3000)
            for _ in range(3):
                a = normalize(a * np.exp(rng_final.normal(0, 0.01, len(a))))
                a = subgradient_mirror(a, max_iters=1000, time_budget=min(remaining * 0.2, 15.0),
                                       top_k=5, seed=seed + 3000 + int(rng_final.integers(0, 100)))
                obj = compute_obj(a)
                if obj < best_score:
                    best_score = obj
                    best_seq = a.copy()
                    print(f"  Final attempt: {best_score:.8f}")

    # ============================================================
    # Final validation
    # ============================================================
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
