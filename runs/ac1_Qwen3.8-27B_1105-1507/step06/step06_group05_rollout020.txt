"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft

    import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_cvxpy(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0 using CVXPY."""
    n = len(a)
    A = build_toeplitz(a)
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M]
    prob = cp.Problem(objective, constraints)
    try:
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                if prob.status == 'optimal' and g.value is not None:
                    gv = np.maximum(g.value, 0.0)
                    s = float(np.sum(gv))
                    if s > 1e-10:
                        return gv / s
            except Exception:
                continue
    except Exception:
        pass
    return None


def solve_lp_scipy(a, M, time_limit=30):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def lp_alternate(a, max_iters=50, time_budget=80.0):
    """LP alternating with golden section line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break

        # Line search for optimal blend
        def obj_at_t(t):
            a_try = normalize((1.0 - t) * a + t * g)
            return compute_obj(a_try)

        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc, fd = obj_at_t(c), obj_at_t(d)
        for _ in range(12):
            if hi - lo < 0.01:
                break
            if fc < fd:
                hi, d, fd = d, c, fc
                c = hi - gr * (hi - lo)
                fc = obj_at_t(c)
            else:
                lo, c, fc = c, d, fd
                d = lo + gr * (hi - lo)
                fd = obj_at_t(d)
        t_opt = (lo + hi) / 2.0

        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            rng = np.random.default_rng((it * 7919 + 12345) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.05, n)))

    return best_a


def soft_max_grad(a, conv, tau):
    """Compute soft-max gradient of max(conv)."""
    n = len(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # grad[i] = 2 * sum_k weights[k] * a[k-i] (where 0 <= k-i < n)
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    a_flip = a[::-1]
    Wf = rfft(weights, size)
    Af = rfft(a_flip, size)
    corr = irfft(Wf * Af, size)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, weights


def kl_mirror_descent(a, max_iters=300, time_budget=50.0, tau_init=0.001):
    """KL mirror descent with soft-max gradient."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1
    tau = tau_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        grad, weights = soft_max_grad(a, conv, tau)

        # KL mirror descent: a_new = a * exp(-eta * grad) / Z
        # This is equivalent to: log(a_new) = log(a) - eta*grad - log(Z)
        log_a = np.log(np.maximum(a, 1e-30))
        log_new = log_a - eta * grad
        # Shift for numerical stability
        log_new -= np.max(log_new)
        a_new = np.exp(log_new)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.05, 1.0)  # slightly increase step
        else:
            eta *= 0.7
            if eta < 1e-8:
                break

        tau *= 0.985  # cool down

    return best_a


def nesterov_soft_max_grad(a, max_iters=200, time_budget=50.0):
    """Nesterov-accelerated soft-max gradient descent."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    tau = 0.001
    step = 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        t = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        beta = (it + 1) / t  # Nesterov momentum

        y = normalize(a + beta * (a - y) if it > 0 else a)

        conv = compute_conv(y)
        grad, _ = soft_max_grad(y, conv, tau)
        grad_centered = grad - np.mean(grad)

        a_new = normalize(a - step * grad_centered)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.7
            a = a + beta * (a - y)  # rollback
            if step < 1e-8:
                break

        tau *= 0.99

    return best_a


def frank_wolfe(a, max_iters=500, time_budget=30.0):
    """Frank-Wolfe with away steps."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # FW step: move toward argmin grad
        i_min = int(np.argmin(grad))
        
        # Away step: check if moving away from a vertex in support helps
        support = np.where(a > 1e-6)[0]
        gamma = 0.0
        if len(support) > 1:
            for i_s in support:
                # Direction: a - e_{i_s}
                direction = -grad[i_s]  # <grad, -e_i> = -grad[i]
                if direction > gamma:
                    gamma = direction
                    i_away = int(i_s)

        # Line search
        step = 2.0 / (it + 2.0)
        
        if gamma > 0 and it > 20:
            # Away step
            a_try = a.copy()
            a_try[i_away] -= step * gamma
            a_try[i_min] += step * (1.0 - gamma)
            a_try = np.maximum(a_try, 0.0)
            s = np.sum(a_try)
            if s > 1e-10:
                a_try /= s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
            else:
                # Regular FW step
                a_try = (1.0 - step) * a + step * np.zeros(n)
                a_try[i_min] = step
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    a = (1.0 - step) * a
                    a[i_min] += step
                    a = normalize(a)
        else:
            # Regular FW step
            a = (1.0 - step) * a
            a[i_min] += step
            a = normalize(a)
            obj_now = compute_obj(a)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()

    return best_a


def coordinate_descent(a, time_budget=20.0):
    """Paired coordinate descent targeting convolution peak."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        if not contrib:
            break

        improved = False
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            for _ in range(2):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue

                a_try = a.copy()
                delta = 0.03 * a_try[j1]
                a_try[j1] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)

                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and rng.random() < 0.15:
            a = normalize(a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, np.maximum(a, 0))

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating (CVXPY/scipy), KL mirror descent with soft-max gradient,
Frank-Wolfe with away steps, progressive multi-n refinement, and coordinate descent.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def build_toeplitz(a):
    """Toeplitz matrix for conv(a, .) constraints."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, time_limit=20):
    """LP: max sum(g) s.t. conv(a,g)<=M, g>=0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)

    # Try CVXPY first
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [A @ g_var <= M])
            for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDDLP]:
                try:
                    prob.solve(solver=solver, **{'time_limit': time_limit})
                    if prob.status in ('optimal', 'optimal_inaccurate') and g_var.value is not None:
                        gv = np.maximum(g_var.value, 0.0)
                        s = float(np.sum(gv))
                        if s > 1e-10:
                            return gv / s
                except Exception:
                    continue
        except Exception:
            pass

    # Fallback to scipy
    try:
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0.0, None)]*n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate(a, max_iters=40, time_budget=60.0):
    """LP alternating with golden-section line search."""
    n = len(a)
    a = normalize(a)
    best_a, best_obj = a.copy(), compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):

        else:

def soft_max_gradient(a, conv, tau):
    """Smoothed subgradient using log-sum-exp weights."""
    n = len(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    w = np.exp(shifted)
    ws = float(np.sum(w))
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    # grad[i] = 2*sum_k w[k]*a[k-i]
    size = 1
    while size < 3*n - 1:
        size <<= 1
    corr = irfft(rfft(w, size) * rfft(a[::-1], size), size)
    return 2.0 * corr[n-1:2*n-1]


def kl_mirror_descent(a, max_iters=300, time_budget=50.0, tau_init=0.001):
    """KL-divergence mirror descent with soft-max gradient."""
    n = len(a)
    a = normalize(a)
    best_a, best_obj = a.copy(), compute_obj(a)
    start = time.monotonic()
    eta, tau = 0.1, tau_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        grad = soft_max_gradient(a, compute_conv(a), tau)
        # KL update: a * exp(-eta*grad) / Z
        log_new = np.log(np.maximum(a, 1e-30)) - eta * grad
        log_new -= np.max(log_new)
        a_new = np.exp(log_new)
        a_new /= np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj, best_a, a = obj_new, a_new.copy(), a_new
            eta = min(eta * 1.03, 0.5)
        else:
            eta *= 0.7
            if eta < 1e-8:
                break
        tau *= 0.99

    return best_a


def nesterov_gradient(a, max_iters=200, time_budget=45.0):
    """Nesterov-accelerated gradient with soft-max smoothing."""
    n = len(a)
    a = normalize(a)
    y = a.copy()
    best_a, best_obj = a.copy(), compute_obj(a)
    start = time.monotonic()
    tau, step = 0.001, 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        t = (1.0 + np.sqrt(1.0 + 4.0*(it+1)**2)) / 2.0
        beta = (it+1) / t if it > 0 else 0.0
        y = normalize(a + beta*(a - y)) if it > 0 else a

        grad = soft_max_gradient(y, compute_conv(y), tau)
        grad -= np.mean(grad)
        a_new = normalize(a - step * grad)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj, best_a, a = obj_new, a_new.copy(), a_new
        else:
            step *= 0.7
            if step < 1e-8:
                break
        tau *= 0.995

    return best_a


def frank_wolfe(a, max_iters=3000, time_budget=25.0):
    """Frank-Wolfe with away steps for simplex-constrained problem."""
    n = len(a)
    a = normalize(a)
    best_a, best_obj = a.copy(), compute_obj(a)
    start = time.monotonic()
    support = set(np.where(a > 1e-8)[0].tolist())

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        ks = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j = ks - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        step = 2.0 / (it + 2.0)
        i_min = int(np.argmin(grad))

        # Check for away step
        best_away = 0.0
        i_away = -1
        for i_s in support:
            d = -grad[i_s]
            if d > best_away:
                best_away = d
                i_away = int(i_s)

        if best_away > 0 and it > 50 and i_away >= 0:
            a_try = a.copy()
            a_try[i_away] -= step * best_away
            a_try[i_min] += step * (1.0 - best_away)
            a_try = np.maximum(a_try, 0.0)
            s = float(np.sum(a_try))
            if s > 1e-10:
                a_try /= s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj, best_a, a = obj_try, a_try.copy(), a_try
                support = set(np.where(a > 1e-8)[0].tolist())
                continue

        # Regular FW step
        a = (1.0 - step) * a
        a[i_min] += step
        a = normalize(a)
        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj, best_a = obj_now, a.copy()
        support = set(np.where(a > 1e-8)[0].tolist())

    return best_a


def coord_descent(a, time_budget=20.0):
    """Targeted coordinate descent on convolution peak."""
    n = len(a)
    a = normalize(a)
    best_a, best_obj = a.copy(), compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        ks = int(np.argmax(conv))
        contrib = sorted(
            ((a[j]*a[ks-j], j, ks-j) for j in range(n) if 0 <= ks-j < n),
            reverse=True
        )
        improved = False
        for _, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue
            for _ in range(3):
                jt = int(rng.integers(0, n))
                if jt == j1 or jt == j2:
                    continue
                a_try = a.copy()
                d = 0.04 * a_try[j1]
                a_try[j1] -= d
                a_try[jt] += d
                a_try = normalize(a_try)
                if compute_obj(a_try) < best_obj - 1e-14:
                    best_obj = compute_obj(a_try)
                    best_a, a = a_try.copy(), a_try
                    improved = True
                    break
        if not improved and rng.random() < 0.12:
            a = normalize(a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    return np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, n), np.maximum(a, 0))
