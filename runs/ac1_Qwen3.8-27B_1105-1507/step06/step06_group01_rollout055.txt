import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def build_full_lp_matrix(a, n):
    """Build the full sparse constraint matrix for (a*g)[k] <= M for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_lp_best_response(a, M, n, time_limit=5.0):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    A = build_full_lp_matrix(a, n)
    if A.nnz == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def line_search_mirror(a, g, best_M):
    """Find best t for a_new = normalize((1-t)*a + t*g)."""
    best_t = 0.0
    best_M_trial = best_M
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9]:
        a_trial = (1.0 - t) * a + t * g
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        M_trial = fast_conv(a_trial).max()
        if M_trial < best_M_trial:
            best_M_trial = M_trial
            best_t = t
    return best_t, best_M_trial

def coordinate_1d(a, i, n):
    """Exact 1D optimization for coordinate i."""
    b = fast_conv(a)
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_arr = []
    s_arr = []
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c_arr.append(b[k] - x_cur * x_cur)
            s_arr.append(0.0)
            quad_pos = len(c_arr) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c_arr.append(b[k] - 2.0 * x_cur * a[j])
                s_arr.append(2.0 * a[j])
            else:
                c_arr.append(b[k])
                s_arr.append(0.0)
    
    # Unaffected max
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_lo:k_hi + 1] = False
    base_max = b[mask].max() if mask.any() else 0.0
    
    c_arr = np.array(c_arr)
    s_arr = np.array(s_arr)
    
    def obj_at(x):
        vals = c_arr + s_arr * x
        m = base_max
        if vals.size > 0:
            m = max(m, vals.max())
        if quad_pos >= 0:
            m = max(m, c_arr[quad_pos] + x * x)
        return m
    
    lo = 0.0
    hi = max(3.0 * x_cur, 2.0 * np.sqrt(max(b.max(), 0.0)), 1.0)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)

def subgradient_descent(a_init, deadline, eta0=0.01):
    """Subgradient descent with running average."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    a_avg = a.copy()
    count = 0
    
    it = 0
    while time.monotonic() < deadline:
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Subgradient: grad[i] = 2*a[k_star - i] if 0 <= k_star-i < n else 0
        grad = np.zeros(n)
        j_lo = max(0, k_star - (n - 1))
        j_hi = min(k_star, n - 1)
        for i in range(j_lo, j_hi + 1):
            grad[i] = 2.0 * a[k_star - i]
        
        # Projected subgradient step
        eta = eta0 / (1.0 + 0.01 * it)
        a_new = a - eta * (grad - grad.mean())
        a_new = proj_simplex(a_new)
        
        a_avg = (count * a_avg + a_new) / (count + 1)
        count += 1
        a = a_new
        it += 1
        
        if it % 100 == 0:
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_M = M_avg
                best_a = a_avg.copy()
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    # Final check
    M_avg = fast_conv(a_avg).max()
    if M_avg < best_M:
        best_a = a_avg.copy()
    
    return best_a

def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = np.interp(t, np.linspace(0, 1, parent.size), parent)
        p = np.maximum(p, 1e-4)
        inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.02, n), 1e-4)
        inits.append(p_n / p_n.sum())
    
    inits.append(np.ones(n) / n)
    
    # Concave (parabola)
    poly = (t * (1 - t)) + 0.1
    inits.append(poly / poly.sum())
    
    # Cosine
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())
    
    # Gaussian
    gauss = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Raised cosine (Hann window)
    hann = 0.5 * (1 + np.cos(2 * np.pi * t))
    hann = np.maximum(hann, 0.01)
    inits.append(hann / hann.sum())
    
    return inits

    k_hi = min(2 * n - 2, i + n - 1)
    
    x_cur = a[i]
    delta = x_test - x_cur

    ks = np.arange(k_lo, k_hi + 1)
    js = ks - i  # j = k - i for each affected k

    # b[k] + 2*delta*a[j] (for k != 2i)
    # b[k] - x_cur^2 + x_test^2 (for k == 2i)

    valid_mask = (js >= 0) & (js < n)

    new_vals[valid_mask] += 2.0 * delta * a[j_clipped[valid_mask]]
    # Fix the k=2i case: we added 2*delta*a[i] but should have added x_test^2 - x_cur^2

    # What we added: 2*delta*a[i] = 2*delta*x_cur
    # What we should add: delta*(2*x_cur + delta)

    idx_2i = 2 * i - k_lo

        new_vals[idx_2i] += delta * delta

    max_unaff = 0.0
    if k_lo > 0:
        max_unaff = max(max_unaff, float(b[:k_lo].max()))

        max_unaff = max(max_unaff, float(b[k_hi+1:].max()))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-constraint LP mirror descent (all 2n-1 constraints for exact best response)
2. Vectorized coordinate descent with exact 1D golden-section solves
3. Subgradient descent with running average
4. Iterative LP+CD polish loop
Key improvement over previous: uses ALL LP constraints (not just top-k),
precomputed sparse structure for fast LP rebuilds, and vectorized CD.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def precompute_lp_structure(n):
    rows = []
    cols = []
    val_idx = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            val_idx.append(k - j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), np.array(val_idx, dtype=np.int32)


class LPContext:
    def __init__(self, n):
        self.n = n
        self.row_idx, self.col_idx, self.val_idx = precompute_lp_structure(n)
        self.last_a = None
        self.last_A = None

    def get_matrix(self, a):
        if self.last_a is not None and np.array_equal(a, self.last_a):
            return self.last_A
        vals = a[self.val_idx]
        A = csr_matrix((vals, (self.row_idx, self.col_idx)), shape=(2 * self.n - 1, self.n))
        self.last_a = a.copy()
        self.last_A = A
        return A


def solve_lp_best_response(lp_ctx, a, M, time_limit=5.0):
    n = lp_ctx.n
    A = lp_ctx.get_matrix(a)
    if A.nnz == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def line_search_mirror(a, g, best_M):
    n = len(a)
    best_t = 0.0
    best_M_trial = best_M
    for t in [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        a_trial = (1.0 - t) * a + t * g
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        M_trial = fast_conv(a_trial).max()
        if M_trial < best_M_trial:
            best_M_trial = M_trial
            best_t = t
    return best_t, best_M_trial


def run_lp_mirror_descent(a_init, deadline, lp_ctx, label=""):
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        b = fast_conv(a)
        M_cur = b.max()
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()

        lp_ctx.last_a = None
        g = solve_lp_best_response(lp_ctx, a, M_cur, time_limit=5.0)
        if g is None:
            break

        t, M_trial = line_search_mirror(a, g, best_M)
        if t == 0.0 or M_trial >= best_M * (1.0 - 1e-13):
            no_improve += 1
            if no_improve > 50:
                break
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            a = a_new
            no_improve = 0

        it += 1
        if it % 100 == 0:
            cur_M = fast_conv(a).max()
            print(f"    {label} LP-MD iter {it}: best_M={best_M:.10f} cur_M={cur_M:.10f}", flush=True)

    return best_a


def coord_obj_vec(b, a, i, n, x_test):
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    x_cur = a[i]
    delta = x_test - x_cur

    new_vals = b[k_lo:k_hi + 1].astype(np.float64).copy()
    ks = np.arange(k_lo, k_hi + 1)
    js = ks - i
    valid = (js >= 0) & (js < n)
    j_clip = np.clip(js, 0, n - 1)
    new_vals[valid] += 2.0 * delta * a[j_clip[valid]]

    idx_2i = 2 * i - k_lo
    if 0 <= idx_2i < len(new_vals):
        new_vals[idx_2i] += delta * delta

    max_unaff = 0.0
    if k_lo > 0:
        max_unaff = max(max_unaff, float(b[:k_lo].max()))
    if k_hi + 1 < 2 * n - 1:
        max_unaff = max(max_unaff, float(b[k_hi + 1:].max()))

    return max(max_unaff, float(new_vals.max()) if len(new_vals) > 0 else 0.0)


def coordinate_1d_solve(b, a, i, n):
    x_cur = a[i]
    lo = 0.0
    hi = max(3.0 * x_cur, 2.0 * np.sqrt(max(b.max(), 0.0)) + 0.1, 0.5)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = coord_obj_vec(b, a, i, n, a_pt)
    fb = coord_obj_vec(b, a, i, n, b_pt)

    for _ in range(45):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = coord_obj_vec(b, a, i, n, a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = coord_obj_vec(b, a, i, n, b_pt)

    x_opt = (lo + hi) / 2.0
    if coord_obj_vec(b, a, i, n, 0.0) <= coord_obj_vec(b, a, i, n, x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, deadline, seed=0):
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    rng = np.random.default_rng(seed)
    order = np.arange(n)
    sweep = 0

    while time.monotonic() < deadline:
        rng.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False

        for i in order:
            if time.monotonic() > deadline:
                break
            x_opt = coordinate_1d_solve(b, a, i, n)
            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 5
        if no_improve > 30:
            break
        sweep += 1

    return best_a


def subgradient_descent(a_init, deadline, eta0=0.015):
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    a_avg = a.copy()
    count = 0
    it = 0

    while time.monotonic() < deadline:
        b = fast_conv(a)
        k_star = int(np.argmax(b))

        grad = np.zeros(n)
        j_lo = max(0, k_star - (n - 1))
        j_hi = min(k_star, n - 1)
        for i in range(j_lo, j_hi + 1):
            grad[i] = 2.0 * a[k_star - i]

        eta = eta0 / (1.0 + 0.005 * it)
        a_new = a - eta * (grad - grad.mean())
        a_new = proj_simplex(a_new)

        a_avg = (count * a_avg + a_new) / (count + 1)
        count += 1
        a = a_new
        it += 1

        if it % 200 == 0:
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_M = M_avg
                best_a = a_avg.copy()
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()

    M_avg = fast_conv(a_avg).max()
    if M_avg < best_M:
        best_a = a_avg.copy()

    return best_a


def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = np.interp(t, np.linspace(0, 1, parent.size), parent)
        p = np.maximum(p, 1e-4)
        inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.02, n), 1e-4)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.05, n), 1e-4)
        inits.append(p_n2 / p_n2.sum())

    inits.append(np.ones(n) / n)

    poly = t * (1.0 - t) + 0.1
    inits.append(poly / poly.sum())

    cos_w = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())

    gauss = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    hann = np.maximum(hann, 0.01)
    inits.append(hann / hann.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best_score = score
            self.best = a.copy()
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768) / 768
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)

    n = parent.size
    lp_ctx = LPContext(n)

    # Phase 1: Full LP mirror descent from parent
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Full LP mirror descent (n={n})", flush=True)
    print(f"{'='*60}", flush=True)
    t1_deadline = min(deadline - 500, time.monotonic() + 400)
    a_lp = run_lp_mirror_descent(parent, t1_deadline, lp_ctx, label="P1")
    tracker.update(a_lp)

    # Phase 2: Coordinate descent
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Coordinate descent", flush=True)
    print(f"{'='*60}", flush=True)
    t2_deadline = min(deadline - 350, time.monotonic() + 200)
    a_cd = run_coordinate_descent(a_lp, t2_deadline, seed=seed)
    tracker.update(a_cd)

    # Phase 3: LP from CD result
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP from CD result", flush=True)
    print(f"{'='*60}", flush=True)
    t3_deadline = min(deadline - 200, time.monotonic() + 150)
    a_lp2 = run_lp_mirror_descent(a_cd, t3_deadline, lp_ctx, label="P3")
    tracker.update(a_lp2)

    # Phase 4: Subgradient with averaging
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Subgradient descent", flush=True)
    print(f"{'='*60}", flush=True)
    t4_deadline = min(deadline - 100, time.monotonic() + 100)
    Best = tracker.best if tracker.best is not None else a_lp2
    a_sg = subgradient_descent(Best, t4_deadline, eta0=0.02)
    tracker.update(a_sg)

    # Phase 5: CD on best
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: CD on best", flush=True)
    print(f"{'='*60}", flush=True)
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t5_deadline = min(deadline - 30, time.monotonic() + 80)
        a_cd2 = run_coordinate_descent(tracker.best, t5_deadline, seed=seed + 100)
        tracker.update(a_cd2)

    # Phase 6: Iterative LP+CD polish loop
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: LP+CD polish loop", flush=True)
    print(f"{'='*60}", flush=True)
    if tracker.best is not None:
        cur = tracker.best.copy()
        cur_M = fast_conv(cur).max()
        polish_iter = 0
        while time.monotonic() < deadline - 10:
            sub_dl = min(deadline - 10, time.monotonic() + 50)
            a_new = run_lp_mirror_descent(cur, sub_dl, lp_ctx, label=f"P6-{polish_iter}")
            M_new = fast_conv(a_new).max()
            if M_new < cur_M * (1.0 - 1e-12):
                cur = a_new
                cur_M = M_new
                tracker.update(cur)
                polish_iter += 1
            else:
                # Try CD
                sub_dl2 = min(deadline - 10, time.monotonic() + 30)
                a_new2 = run_coordinate_descent(cur, sub_dl2, seed=int(time.monotonic() * 1000))
                M_new2 = fast_conv(a_new2).max()
                tracker.update(a_new2)
                if M_new2 < cur_M * (1.0 - 1e-12):
                    cur = a_new2
                    cur_M = M_new2
                    polish_iter += 1
                else:
                    break

    # Phase 7: Try a different length if time remains
    if time.monotonic() < deadline - 60 and tracker.best is not None:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 7: Try n=1024", flush=True)
        print(f"{'='*60}", flush=True)
        n2 = 1024
        lp_ctx2 = LPContext(n2)
        t_arr = np.linspace(0, 1, n2, endpoint=False)
        a2 = np.interp(t_arr, np.linspace(0, 1, tracker.best.size), tracker.best)
        a2 = np.maximum(a2, 1e-4)
        a2 = a2 / a2.sum()
        t7_dl = min(deadline - 30, time.monotonic() + 60)
        a2_lp = run_lp_mirror_descent(a2, t7_dl, lp_ctx2, label="P7")
        tracker.update(a2_lp)

    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
