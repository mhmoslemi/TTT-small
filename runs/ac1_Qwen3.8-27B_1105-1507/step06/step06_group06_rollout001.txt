import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(j+n, 2*n-1)
        A[k_start:k_end, j] = a[:k_end-k_start]
    return A

def solve_lp(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0"""
    n = len(a)
    if n > 3000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit, 'ips': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except:
        pass
    return None

"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 3000:
        return None
    try:
        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns (g, M_opt)."""
    n = len(a)
    if n > 2500:
        return None, None
    try:
        A = build_toeplitz(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0.0, None)] * n + [(0.0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def lp_alternate_advanced(a, max_iters=300, time_budget=120.0, cool_rate=0.002):

    Uses min-peak LP to find the theoretical minimum, then targets M values

def logsumexp_gradient_descent(a, max_iters=200, time_budget=60.0):
    """
    Gradient descent on the smooth log-sum-exp approximation of the peak.
    Uses Barzilai-Borwein step sizes for faster convergence.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = 0.01  # Temperature for log-sum-exp
    s_prev = None
    g_prev = None
    step = 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Smooth approximation of max
        shifted = conv / tau
        shifted = np.clip(shifted, -300, 300)
        max_shifted = np.max(shifted)
        log_sum_exp = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))

        # Gradient of log-sum-exp w.r.t. a_i
        weights = np.exp(shifted - log_sum_exp)  # Softmax weights
        # grad[i] = (1/tau) * sum_k weights[k] * (d conv_k / da_i)
        # d conv_k / da_i = a[k-i] (if 0<=k-i<n) + a[i-(2n-2-k)] (if valid for non-symmetric)
        # For linear conv: d/d a_i of sum_j a_j a_{k-j} = a[k-i] (if 0<=k-i<n) + a[k-i] (if j=k-i=i, i.e., k=2i)
        # Actually: d/d a_i [sum_j a_j * a_{k-j}] = a[k-i] (contributing when j gives a_j's derivative)
        #   + a[k-i] (contributing when k-j gives a_{k-j}'s derivative, i.e., j = k-i)
        # Wait, these are the same term! Let me be more careful.
        # conv_k = sum_j a_j * a_{k-j}
        # d/d a_i conv_k = sum_j [delta_{j,i} * a_{k-j} + a_j * delta_{k-j,i}]
        #   = a_{k-i} (if 0 <= k-i < n) + a_{k-i} (if 0 <= k-i < n)
        # Hmm no: delta_{k-j,i} means k-j=i, so j=k-i, giving a_j = a_{k-i}
        # And delta_{j,i} gives j=i, giving a_{k-j} = a_{k-i}
        # So d/d a_i conv_k = 2 * a_{k-i} if 0 <= k-i < n, else 0
        # Wait that's only if i != k-i (i.e., 2i != k). If 2i = k, then both terms give the same a_i, so it's still 2*a_i.
        # Actually no: if j = i and k-j = i (i.e., k = 2i), then the term a_i * a_i appears once in the sum,
        # and d/d a_i (a_i * a_i) = 2 * a_i. So yes, it's always 2 * a_{k-i} when valid.

        grad = np.zeros(n)
        for k_idx in range(len(weights)):
            w_k = weights[k_idx]
            if w_k < 1e-15:
                continue
            # d conv[k_idx] / da_i = 2 * a[k_idx - i] for valid i
            # i ranges such that 0 <= k_idx - i < n, i.e., k_idx - n + 1 <= i <= k_idx
            i_start = max(0, k_idx - n + 1)
            i_end = min(n - 1, k_idx)
            if i_end >= i_start:
                indices = np.arange(i_start, i_end + 1)
                a_indices = k_idx - indices
                grad[indices] += w_k * 2.0 * a[a_indices]

        grad *= (1.0 / tau)

        # Center the gradient (since sum is fixed)
        grad_centered = grad - np.mean(grad)

        # Barzilai-Borwein step
        if s_prev is not None and g_prev is not None:
            gamma = np.dot(s_prev, s_prev) / np.abs(np.dot(s_prev, g_prev))
            step_bb = min(0.5, max(0.01, gamma))
            # Blend with current step
            step = 0.5 * step + 0.5 * step_bb

        s_curr = step * grad_centered
        a_new = a - s_curr
        a_new = project_nonneg_sum1(a_new)

        s_prev = a_new - a
        g_prev = grad_centered
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Adapt tau: decrease over time for sharper max approximation
        tau = max(0.005, tau * 0.995)
        step = max(0.01, step * 0.998)

    return best_a, best_obj


def freq_domain_flatten(a, max_iters=100, time_budget=30.0):
    """
    Frequency domain approach: flatten the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power_spec = np.abs(fa) ** 2
        ps_max = np.max(power_spec)
        if ps_max < 1e-15:
            break

        # Compress the power spectrum toward flatness
        ps_norm = power_spec / ps_max
        # Apply power compression (reduces dynamic range)
        alpha = 0.5 + 0.3 * (it / max_iters)
        ps_compressed = np.maximum(ps_norm, 1e-10) ** alpha

        # Reconstruct
        phase = np.angle(fa)
        magnitude = np.sqrt(ps_compressed * ps_max)
        fa_new = magnitude * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def chase_peak_fast(a, n_steps=50, step_size=0.03):

        i_end = min(n - 1, k_star)

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Sinusoidal
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    # Gaussian
    xg = np.linspace(-3, 3, n)
    gauss = np.exp(-xg ** 2 / 3.0)
    inits.append(gauss / gauss.sum())

    # Chirp-like (modulated sinusoid)
    freq = np.linspace(0, 5, n)
    chirp = np.abs(np.sin(np.pi * x + 2 * np.pi * freq * x)) + 0.01
    inits.append(chirp / chirp.sum())

    # Bimodal
    bimodal = np.exp(-((x - 0.3) ** 2) / 0.01) + np.exp(-((x - 0.7) ** 2) / 0.01) + 0.01
    inits.append(bimodal / bimodal.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Comb (alternating pattern)
    comb = np.ones(n)
    comb[::3] = 0.3
    inits.append(comb / comb.sum())

    # Triangle
    tri = np.minimum(x, 1 - x) + 0.05
    inits.append(tri / tri.sum())

    # Multi-frequency
    mf = 1 + 0.5 * np.sin(4 * np.pi * x) + 0.3 * np.sin(8 * np.pi * x)
    mf = np.maximum(mf, 0.01)
    inits.append(mf / mf.sum())

    if parent is not None:
        inits.append(resample_sequence(parent, n))
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.1, n))
        inits.append(project_nonneg_sum1(pert))

    return inits

            if time.monotonic() - p3_start > min(p3_budget, remaining3 * 0.35):
                break
            if time.monotonic() > deadline - 50:
                break

            a_res = resample_sequence(best_seq, n_try)
            a_res = project_nonneg_sum1(a_res)

            # Optimize
            a_res, obj_res = chase_peak_fast(a_res, n_steps=20, step_size=0.02)
            if n_try <= 1200:
                a_res2, obj_res2 = lp_alternate_advanced(a_res, max_iters=30,
                                                         time_budget=min(per_n_budget * 0.5, 40.0))
                if obj_res2 < obj_res:
                    a_res, obj_res = a_res2, obj_res2

            a_res3, obj_res3 = logsumexp_gradient_descent(a_res, max_iters=30,
                                                          time_budget=min(per_n_budget * 0.3, 20.0))
            if obj_res3 < obj_res:
                a_res, obj_res = a_res3, obj_res3

            if obj_res < best_score:
                best_score = obj_res
                best_seq = a_res.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_res:.8f}")

    # === Phase 4: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 20 and best_seq is not None:
        n_final = len(best_seq)
        a_final = project_nonneg_sum1(best_seq.copy())

        # Multiple intensive LP rounds
        polish_rounds = min(8, int(remaining4 / 30))
        for pr in range(polish_rounds):
            rem = deadline - time.monotonic()
            if rem < 15:
                break

            # LP with adaptive M
            if n_final <= 1500:
                a_p, obj_p = lp_alternate_advanced(a_final, max_iters=30,
                                                   time_budget=min(rem * 0.4, 50.0))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Polish LP r{pr}: {best_score:.8f}")

            # LSE gradient
            rem = deadline - time.monotonic()
            if rem > 10:
                a_p2, obj_p2 = logsumexp_gradient_descent(a_final, max_iters=30,
                                                          time_budget=min(rem * 0.3, 25.0))
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_final = a_p2.copy()
                    print(f"  Polish LSE r{pr}: {best_score:.8f}")

            # Chase peak
            rem = deadline - time.monotonic()
            if rem > 5:
                a_p3, obj_p3 = chase_peak_fast(a_final, n_steps=20, step_size=0.01)
                if obj_p3 < best_score:
                    best_score = obj_p3
                    best_seq = a_p3.copy()
                    a_final = a_p3.copy()

            # Small perturbation + re-optimize
            rem = deadline - time.monotonic()
            if rem > 20:
                a_pert = a_final * np.exp(rng.normal(0, 0.02, n_final))
                a_pert = project_nonneg_sum1(a_pert)
                a_pert, obj_pert = chase_peak_fast(a_pert, n_steps=20, step_size=0.02)
                if n_final <= 1200:
                    a_pert2, obj_pert2 = lp_alternate_advanced(a_pert, max_iters=15,
                                                               time_budget=min(rem * 0.3, 25.0))
                    if obj_pert2 < obj_pert:
                        a_pert, obj_pert = a_pert2, obj_pert2
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_final = a_pert.copy()
                    print(f"  Polish pert r{pr}: {best_score:.8f}")

        # Final min-peak LP attempt
        rem = deadline - time.monotonic()
        if rem > 15 and n_final <= 1200:
            g_final, M_final = solve_lp_minpeak(a_final, time_limit=min(rem - 3, 20))
            if g_final is not None:
                for t in [0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - t) * a_final + t * g_final
                    a_try = project_nonneg_sum1(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_score:
                        best_score = obj_try
                        best_seq = a_try.copy()
                        print(f"  Final LP blend t={t}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
