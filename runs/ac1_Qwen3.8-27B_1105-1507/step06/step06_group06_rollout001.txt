Key improvements over prior: (1) min-peak LP formulation for stronger descent direction,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]. Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None)."""
    n = len(a)
    if n > 2500:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(-np.inf, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = g.sum()
            if s > 1e-10:

        # Wait: d/da[i] sum_j a[j]*a[k-j] = a[k-i] (from j=i term) + a[i] (from k-j=i i.e. j=k-i term)
        # Only if both i and k-i are in [0,n). But when j=i, we get a[k-i]. When j=k-i, we get a[i].
        # If i == k-i (i.e., k=2i), we get 2*a[i].
        # So grad[i] = sum_k weights[k] * (a[k-i] if 0<=k-i<n else 0) + sum_k weights[k] * (a[i] if 0<=k-i<n else 0)
        #          = sum_k weights[k] * a[k-i] * (1 if 0<=k-i<n else 0) + a[i] * sum_k weights[k] * (1 if 0<=k-i<n else 0)
        # Simplifies to: grad[i] = sum_{k: 0<=k-i<n} weights[k] * (a[k-i] + a[i])
        # But this double-counts when i = k-i. Let me just compute it directly.
        
        # Efficient computation via the Toeplitz structure
        # grad[i] = sum_k w[k] * a[k-i] for k in [i, i+n-1] (where a[k-i] is valid)
        #         + sum_k w[k] * a[i] for k in [i, i+n-1]
        #         = (correlation of w and a)[i] + a[i] * (sum of w over valid range)
        
        # Use FFT for the correlation
        size_w = len(weights)
        N_fft = 1
        while N_fft < size_w + n - 1:
            N_fft <<= 1
        fw = rfft(weights, N_fft)
        fa = rfft(a, N_fft)
        # Cross-correlation: corr[m] = sum_k w[k+m] * a[k] for appropriate shifts
        # Actually we want: for each i, sum_k w[k] * a[k-i] where 0<=k-i<n
        # This is (w * a_rev) type operation... let me just do it via the Toeplitz approach
        # grad[i] = A^T @ weights where A is the Toeplitz built from... 
        # Actually: (A@g)[k] = sum_j A[k,j]*g[j] = (conv(a,g))[k]
        # So grad = A^T @ (weights * 2) ... no
        # Let me think again.
        # conv[k] = sum_i a[i] * a[k-i] (for valid k-i)
        # d(conv[k])/da[i] = a[k-i] (if 0<=k-i<n)
        # So grad[i] = sum_k weights[k] * a[k-i] * (1 if 0<=k-i<n else 0)
        # This is: grad = A^T @ weights where A[k,i] = a[k-i] if 0<=k-i<n else 0
        # But A is (2n-1) x n, and we built it as build_toeplitz(a)
        # So grad[i] = sum_k A[k,i] * weights[k] = (A.T @ weights)[i]
        
        if n <= 2000:
            A = build_toeplitz(a)
            grad = A.T @ weights
        else:
            # Approximate via FFT
            # grad[i] = sum_k w[k] * a[k-i] for max(0,k-n+1) <= i <= min(n-1,k)
            # This is a cross-correlation
            grad = np.zeros(n)
            # Use the fact that for each k, it contributes to i in [max(0,k-n+1), min(n-1,k)]
            # with value a[k-i]
            # Equivalent to: for offset d = k-i, grad[i] += w[k]*a[d]
            # Let's use FFT
            fw_full = np.fft.fft(weights, N_fft)
            fa_full = np.fft.fft(a, N_fft)
            # sum_k w[k]*a[k-i] = ifft(fft(w) * conj(fft(a))) shifted... 
            # Actually: ifft(F_w * conj(F_a))[m] = sum_k w[k] * a[(k+m) mod N]
            # We want sum_k w[k]*a[k-i] = sum_k w[k]*a[k-i]
            # = ifft(F_w * F_a_neg)[i] where F_a_neg[k] = F_a[-k mod N]... 
            # This is getting complicated. For n>2000, just use the direct method sparsely.
            top_k = np.argsort(weights)[-200:]
            for k_idx in top_k:
                if weights[k_idx] < 1e-10:
                    continue
                i_start = max(0, int(k_idx) - n + 1)
                i_end = min(n, int(k_idx) + 1)
                if i_end > i_start:
                    indices = np.arange(i_start, i_end)
                    a_vals = a[int(k_idx) - indices]
                    grad[indices] += weights[k_idx] * a_vals

        # Center the gradient (remove mean to preserve sum)
        grad_centered = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - step * grad_centered
        a_new = a - v
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert and reduce step
            step *= 0.9
            v = 0.5 * v

        step = max(step * 0.999, 0.001)

    return best_a, best_obj


def freq_domain_smooth(a, max_iters=150, time_budget=20.0, alpha=0.85):
    """Frequency-domain: smooth the power spectrum to reduce time-domain peaks."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power_spec = np.abs(fa) ** 2
        ps_max = np.max(power_spec)
        if ps_max < 1e-20:
            break

        # Compress the power spectrum
        ps_norm = power_spec / ps_max
        # Use a time-varying alpha (start aggressive, end gentle)
        current_alpha = alpha + (1.0 - alpha) * (it / max_iters)
        ps_compressed = np.maximum(ps_norm, 1e-12) ** current_alpha

        # Reconstruct
        phase = np.angle(fa)
        magnitude = np.sqrt(ps_compressed) * np.sqrt(ps_max)
        fa_new = magnitude * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def chase_peak_subgradient(a, n_steps=100, step_size=0.03):
    """Chase the single maximum peak with subgradient descent."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient at the peak: grad[i] = a[k_star - i] for valid i
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += a[j]
            # Also the "mirror" contribution
            j2 = 2 * n - 2 - k_star + i
            if 0 <= j2 < n:
                grad[i] += a[j2]

        grad_centered = grad - np.mean(grad)

        for ss in [step_size, step_size * 0.5, step_size * 2.0, step_size * 0.25]:
            a_try = a - ss * grad_centered
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a, best_obj


def perturb_and_reoptimize(a, rng, n_perturb=5, time_budget=30.0):
    """Perturb the sequence and re-optimize to escape local minima."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for p in range(n_perturb):
        if time.monotonic() - start > time_budget:
            break
        # Random multiplicative perturbation
        noise_scale = 0.05 + 0.05 * p
        a_pert = a * np.exp(rng.normal(0, noise_scale, n))
        a_pert = project_nonneg_sum1(a_pert)

        # Quick local optimization
        a_opt, obj_opt = chase_peak_subgradient(a_pert, n_steps=30, step_size=0.02)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Resampled parent
    if parent is not None and len(parent) > 0:
        inits.append(resample_sequence(parent, n))
        # Perturbed parent
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_nonneg_sum1(pert))

    # Uniform
    inits.append(np.ones(n) / n)

    # Sinusoidal
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    # Gaussian
    x = np.linspace(-3, 3, n)
    gauss = np.exp(-x ** 2 / 3)
    inits.append(gauss / gauss.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Two-peak
    two = np.zeros(n)
    two[n // 4 - 5:n // 4 + 5] = 1.0
    two[3 * n // 4 - 5:3 * n // 4 + 5] = 1.0
    inits.append(two / two.sum())

    return inits


def optimize_sequence(a, rng, time_budget=120.0):
    """Run multi-strategy optimization on a single sequence."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Choose strategy based on remaining time and current state
        frac_done = (time.monotonic() - start) / time_budget

        if frac_done < 0.3:
            # Early: LP alternating (most powerful)
            if n <= 1500:
                a_new, obj_new = lp_alternate_minpeak(a, max_iters=30,
                                                       time_budget=min(remaining * 0.4, 30.0))
            else:
                a_new, obj_new = freq_domain_smooth(a, max_iters=50,
                                                     time_budget=min(remaining * 0.4, 20.0))
        elif frac_done < 0.6:
            # Middle: subgradient + LP
            a_new = None
            obj_new = np.inf
            # Subgradient
            a_sg, obj_sg = subgradient_multi_peak(a, max_iters=100,
                                                   time_budget=min(remaining * 0.3, 20.0))
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                a = a_sg.copy()
            # LP on improved
            remain2 = time_budget - (time.monotonic() - start)
            if remain2 > 15 and n <= 1200:
                a_new, obj_new = lp_alternate_maxsum(a, max_iters=20,
                                                      time_budget=min(remain2 * 0.4, 20.0))
        elif frac_done < 0.8:
            # Later: chase peak + perturb
            a_new, obj_new = chase_peak_subgradient(a, n_steps=50, step_size=0.02)
            remain3 = time_budget - (time.monotonic() - start)
            if remain3 > 10:
                a_p, obj_p = perturb_and_reoptimize(a_new, rng, n_perturb=3,
                                                     time_budget=min(remain3 * 0.5, 15.0))
                if obj_p < (obj_new if obj_new < np.inf else best_obj):
                    a_new, obj_new = a_p, obj_p
        else:
            # Final: fine-tuning
            a_new, obj_new = chase_peak_subgradient(a, n_steps=30, step_size=0.01)

        if a_new is not None and obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        elif a_new is None:
            # Subgradient already updated best
            a = best_a.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-length LP-alternating optimization with peak targeting.
    Explores lengths around the parent, uses min-peak LP as primary driver,
    subgradient and frequency-domain methods as complements.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None and len(parent) > 0:
        parent = np.clip(parent, 0, 1000.0)
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        best_seq = parent.copy()
        best_score = parent_obj
    else:
        n_init = 700
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"No parent, starting uniform n={n_init}: obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # Phase 1: Quick screening of lengths and initializations (10% budget)
    print("\n=== Phase 1: Length & Initialization Screening ===")
    phase1_budget = min(120.0, (deadline - start_time) * 0.08)
    phase1_end = start_time + phase1_budget

    n_candidates = [n_parent]
    # Add nearby lengths
    for delta in [-50, -25, 25, 50, 75, 100, 150, -75]:
        n_try = n_parent + delta
        if 200 <= n_try <= 2500:
            n_candidates.append(n_try)
    n_candidates = list(dict.fromkeys(n_candidates))

    candidates = []  # (obj, seq)
    per_n_time = max(5.0, phase1_budget / len(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > phase1_end:
            break
        inits = generate_initializations(n_try, rng, parent)
        per_init_time = max(2.0, per_n_time / max(len(inits), 1))

        for init in inits:
            if time.monotonic() > phase1_end:
                break
            a_init = project_nonneg_sum1(init)
            # Quick chase peak to get a rough objective
            a_quick, obj_quick = chase_peak_subgradient(a_init, n_steps=30, step_size=0.03)
            if obj_quick < np.inf:
                candidates.append((obj_quick, a_quick))

    candidates.sort(key=lambda x: x[0])
    print(f"  Screened {len(candidates)} candidates, best: {candidates[0][0] if candidates else 'N/A'}")

    for obj_c, seq_c in candidates[:5]:
        if obj_c < best_score:
            best_score = obj_c
            best_seq = seq_c.copy()
            print(f"  New best from screening: n={len(seq_c)}, obj={best_score:.8f}")

    # Phase 2: Intensive optimization on best candidates (55% budget)
    print(f"\n=== Phase 2: Intensive Optimization ===")
    print(f"Best before: {best_score:.8f}")
    phase2_budget = min(700.0, (deadline - start_time) * 0.50)
    phase2_end = start_time + phase2_budget

    # Build list of sequences to optimize (top from screening + current best)
    opt_list = [(best_score, best_seq.copy())]
    for obj_c, seq_c in candidates[:4]:
        if len(seq_c) != len(best_seq) or obj_c > best_score + 0.01:
            opt_list.append((obj_c, seq_c.copy()))

    per_opt_budget = max(60.0, phase2_budget / max(len(opt_list), 1))

    for oi, (obj_init, seq_init) in enumerate(opt_list):
        if time.monotonic() > phase2_end:
            break
        alloc = min(per_opt_budget, phase2_end - time.monotonic())
        if alloc < 20:
            break

        print(f"  Opt {oi} (n={len(seq_init)}): start={compute_obj(seq_init):.8f}, budget={alloc:.0f}s")
        a_opt, obj_opt = optimize_sequence(seq_init, rng, time_budget=alloc)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"    ** NEW BEST: n={len(a_opt)}, obj={best_score:.8f} **")
        print(f"  Opt {oi} final: {obj_opt:.8f}")

    # Phase 3: Further refinement on the best (25% budget)
    print(f"\n=== Phase 3: Refinement ===")
    print(f"Best before: {best_score:.8f}")
    phase3_budget = min(350.0, (deadline - start_time) * 0.25)
    phase3_end = start_time + phase3_budget
    n_best = len(best_seq)

    # Multiple LP rounds
    a_cur = project_nonneg_sum1(best_seq.copy())
    a_cur_obj = compute_obj(a_cur)

    for round_idx in range(10):
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 15:
            break

        # Alternate between min-peak LP and subgradient
        if round_idx % 3 == 0 and n_best <= 1500:
            a_new, obj_new = lp_alternate_minpeak(a_cur, max_iters=20,
                                                   time_budget=min(remaining * 0.3, 25.0))
        elif round_idx % 3 == 1:
            a_sg, obj_new = subgradient_multi_peak(a_cur, max_iters=80,
                                                    time_budget=min(remaining * 0.3, 20.0))
            a_new = a_sg
        else:
            a_new, obj_new = lp_alternate_maxsum(a_cur, max_iters=15,
                                                  time_budget=min(remaining * 0.3, 20.0))

        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  Round {round_idx}: NEW BEST {best_score:.8f}")
        else:
            a_cur = a_cur  # keep current

        # Occasional perturbation
        if round_idx % 4 == 3:
            a_pert = a_cur * np.exp(rng.normal(0, 0.03, n_best))
            a_pert = project_nonneg_sum1(a_pert)
            a_pert, obj_pert = chase_peak_subgradient(a_pert, n_steps=20, step_size=0.015)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                a_cur = a_pert.copy()
                print(f"  Round {round_idx} perturb: {best_score:.8f}")

    # Phase 4: Length exploration (12% budget)
    print(f"\n=== Phase 4: Length Exploration ===")
    print(f"Best before: {best_score:.8f}")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 40:
        phase4_budget = min(200.0, remaining4 * 0.50)
        phase4_end = start_time + (budget_s - 30.0) - (remaining4 - phase4_budget)
        # Actually just use time directly
        phase4_end = time.monotonic() + phase4_budget

        n_current = len(best_seq)
        n_tries = []
        for delta in [-30, -60, -100, 30, 60, 100, 150, 200, -150]:
            n_try = n_current + delta
            if 300 <= n_try <= 2500 and n_try != n_current:
                n_tries.append(n_try)
        n_tries = list(dict.fromkeys(n_tries))

        per_n_budget = max(10.0, phase4_budget / len(n_tries))

        for n_try in n_tries:
            if time.monotonic() > phase4_end:
                break
            a_res = resample_sequence(best_seq, n_try)
            a_res = project_nonneg_sum1(a_res)

            # Quick optimization
            a_opt, obj_opt = chase_peak_subgradient(a_res, n_steps=40, step_size=0.02)
            remain_n = phase4_end - time.monotonic()
            if remain_n > 15 and n_try <= 1200:
                a_opt2, obj_opt2 = lp_alternate_minpeak(a_opt, max_iters=10,
                                                         time_budget=min(remain_n * 0.5, 15.0))
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 5: Final polish (remaining time)
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 15 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        n_final = len(a_final)

        for pr in range(min(8, int(remaining5 / 8))):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            if n_final <= 1200:
                a_p, obj_p = lp_alternate_minpeak(a_final, max_iters=15,
                                                   time_budget=min(rem * 0.3, 15.0))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Polish LP {pr}: {best_score:.8f}")

            remain = deadline - time.monotonic()
            if remain > 5:
                a_p2, obj_p2 = chase_peak_subgradient(a_final, n_steps=30, step_size=0.01)
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_final = a_p2.copy()
                    print(f"  Polish CP {pr}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
        print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f} ===")
    else:
        print(f"\nEvaluation failed, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s")

    return result
