def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Optimized minimization of J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Combines fast coordinate descent with golden section, LP mirror descent,
and smooth log-sum-exp gradient with Adam optimizer.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent_golden(a, deadline, max_sweeps=5000, seed=42, verbose=False):
    """
    Coordinate descent with golden section search for 1D sub-optimization.
    Uses incremental convolution updates for efficiency.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 200

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Random order every few sweeps
        if sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        sweep_improved = False

        for i in order:
            old = a[i]
            
            # Determine search direction from subgradient
            max_val = conv.max()
            # Active set: entries at (or very near) the max
            tol = 1e-8 * max_val
            active_mask = conv >= max_val - tol
            
            # Subgradient direction
            # dM/da[i] = max over active k of (contribution of a[i] to conv[k])
            # For k in active set, contribution is 2*a[k-i] if 0 <= k-i < n
            lo_k = max(0, i)
            hi_k = min(2*n-2, i+n-1)
            
            # Get active entries in range [lo_k, hi_k]
            active_in_range = active_mask[lo_k:hi_k+1]
            if active_in_range.any():
                j_indices = np.arange(lo_k, hi_k+1) - i
                contribs = 2.0 * a[j_indices]
                # Special case: k = 2i has extra delta^2 term, but at delta=0 it's just 2*old
                max_contrib = contribs[active_in_range].max() if active_in_range.any() else 0.0
            else:
                max_contrib = 0.0
            
            direction = max_contrib * s - 2.0 * max_val
            
            if abs(direction) < 1e-12 * max_val:
                continue  # Near optimal
            
            if direction > 0:
                # J increases with a[i], search in [0, old]
                lo, hi = 0.0, old
            else:
                # J decreases with a[i], search in [old, upper_bound]
                # Upper bound: want new_s to stay reasonable
                lo, hi = old, min(old + 0.5, s * 0.5)
            
            if hi - lo < 1e-12:
                continue
            
            # Precompute outside max
            outside_lo = 0
            outside_hi = lo_k - 1
            outside_lo2 = hi_k + 1
            outside_hi2 = 2*n-2
            
            outside_parts = []
            if outside_hi >= outside_lo:
                outside_parts.append(conv[outside_lo:outside_hi+1].max())
            if outside_hi2 >= outside_lo2:
                outside_parts.append(conv[outside_lo2:outside_hi2+1].max())
            conv_outside_max = max(outside_parts) if outside_parts else 0.0
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_i = a  # shorthand
            
            def eval_x(x):
                delta = x - old
                if abs(delta) < 1e-14:
                    return best_obj
                new_s = s + delta
                if new_s < 1e-10:
                    return float('inf')
                j_range = np.arange(lo_k, hi_k+1) - i
                changes = 2.0 * delta * a_i[j_range]
                k_2i = 2 * i
                if lo_k <= k_2i <= hi_k:
                    idx = k_2i - lo_k
                    changes[idx] = 2.0 * delta * old + delta * delta
                affected = conv[lo_k:hi_k+1] + changes
                total_max = max(conv_outside_max, float(affected.max()))
                return 2.0 * n * total_max / (new_s * new_s)
            
            # Golden section
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = eval_x(x1)
            f2 = eval_x(x2)
            
            for _ in range(15):  # 15 iterations for ~30 bits of precision
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - gr * (hi - lo)
                    f1 = eval_x(x1)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + gr * (hi - lo)
                    f2 = eval_x(x2)
                if hi - lo < 1e-10:
                    break
            
            x_opt = (lo + hi) / 2.0
            f_opt = min(f1, f2)
            
            if x_opt != old and f_opt < best_obj - 1e-12:
                # Apply the change
                delta = x_opt - old
                j_range = np.arange(lo_k, hi_k+1) - i
                changes = 2.0 * delta * a[j_range]
                k_2i = 2 * i
                if lo_k <= k_2i <= hi_k:
                    idx = k_2i - lo_k
                    changes[idx] = 2.0 * delta * old + delta * delta
                conv[lo_k:hi_k+1] += changes
                a[i] = x_opt
                s += delta
                best_obj = f_opt
                sweep_improved = True
        
        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        if verbose and sweep % 100 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a, compute_J(a)


def lp_mirror_descent(a, deadline, max_iters=200, verbose=False):
    """
    LP-based mirror descent: at each step, solve LP for best response direction.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Build LP: maximize sum(g) s.t. (a*g)[k] <= M, g >= 0
        # Constraint matrix: A[k,j] = a[k-j] if 0 <= k-j < n
        rows = []
        cols = []
        vals = []
        
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 30, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    # Try different step sizes
                    found = False
                    for frac in [0.8, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_sum = a_try.sum()
                        if a_sum < 1e-10:
                            continue
                        a_try /= a_sum
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-10:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 10 == 0:
            print(f"    LP it {it}: obj={best_obj:.8f}", flush=True)

        if no_improve > 30:
            break

    return best_a, best_obj
