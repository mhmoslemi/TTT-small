from scipy.optimize import linprog

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent_golden(a, deadline, max_sweeps=500, seed=42, verbose=False):
    """
    Coordinate descent with golden section 1D search.
    Works in un-normalized space (J is scale-invariant).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S
        S = 1.0

    conv = fast_conv(a)
    max_b = conv.max()
    best_obj = 2.0 * n * max_b / (S * S)
    best_a = a.copy()
    rng = np.random.default_rng(seed)

    # Golden section constants
    gr = (np.sqrt(5) - 1) / 2  # ~0.618

    no_improve = 0
    max_no_improve = 200

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Randomize order every few sweeps
        if sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        sweep_improved = False

        for i in order:
            old = a[i]
            
            # Determine search range
            S = a.sum()
            x_max = max(2.0 * old if old > 0 else 0, 4.0 / n)
            # Also ensure x_max isn't too large
            x_max = min(x_max, 4.0 / n)
            
            if x_max < 1e-15:
                continue

            # Golden section search for x in [0, x_max]
            # f(x) = 2n * max_b_new(x) / (S - old + x)^2
            
            # Precompute the affected indices
            # Affected k: those in [i, i+n-1] (clamped to [0, 2n-2])
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Save original conv values for affected range
            conv_saved = conv[k_lo:k_hi+1].copy()
            
            # The "unaffected max" is the max of conv outside [k_lo, k_hi]
            if k_lo > 0:
                unaffected_max = max(0, conv[0:k_lo].max() if k_lo > 0 else 0)
            else:
                unaffected_max = 0.0
            if k_hi < 2*n-2:
                tmp = conv[k_hi+1:].max() if k_hi < 2*n-2 else 0
                unaffected_max = max(unaffected_max, tmp)
            
            S_old = a.sum()
            
            def f_1d(x):
                """Evaluate J at a[i] = x (others unchanged)."""
                # Update affected conv values
                # For k in [k_lo, k_hi]:
                #   delta = x - old
                #   if k == 2*i: conv[k] += delta*x + delta*old = delta*(x+old)
                #     Actually: conv[2i] was old^2 + other, now x^2 + other, delta = x^2 - old^2
                #   else: conv[k] += 2*delta*a[k-i]
                # Let me recompute: conv_new[k] = conv_old[k] + change
                # change at k=2i: x^2 - old^2
                # change at k≠2i (in range): 2*(x-old)*a[k-i]
                
                delta = x - old
                if abs(delta) < 1e-15:
                    mb = conv.max()
                else:
                    # Update conv in place temporarily
                    for k in range(k_lo, k_hi + 1):
                        if k == 2 * i:
                            conv[k] += x * x - old * old
                        else:
                            conv[k] += 2.0 * delta * a[k - i]
                    mb = conv.max()
                    for k in range(k_lo, k_hi + 1):
                        if k == 2 * i:
                            conv[k] -= x * x - old * old
                        else:
                            conv[k] -= 2.0 * delta * a[k - i]
                
                S_new = S_old - old + x
                if S_new < 1e-10:
                    return float('inf')
                return 2.0 * n * mb / (S_new * S_new)

            # Golden section search
            a_lo = 0.0
            a_hi = x_max
            c = a_hi - gr * (a_hi - a_lo)
            d = a_lo + gr * (a_hi - a_lo)
            f_c = f_1d(c)
            f_d = f_1d(d)
            
            for _ in range(15):  # 15 iterations gives ~0.618^15 ≈ 0.0005 precision
                if f_c < f_d:
                    a_hi = d
                    d = c
                    f_d = f_c
                    c = a_hi - gr * (a_hi - a_lo)
                    f_c = f_1d(c)
                else:
                    a_lo = c
                    c = d
                    f_c = f_d
                    d = a_lo + gr * (a_hi - a_lo)
                    f_d = f_1d(d)
            
            x_opt = (a_lo + a_hi) / 2
            
            # Compare with current value
            f_current = 2.0 * n * conv.max() / (S_old * S_old)
            f_opt = f_1d(x_opt)
            
            if f_opt < f_current - 1e-12:
                # Update conv
                delta = x_opt - old
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        conv[k] += x_opt * x_opt - old * old
                    else:
                        conv[k] += 2.0 * delta * a[k - i]
                a[i] = x_opt
                S = S_old - old + x_opt
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (S * S)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 50 == 0:
            print(f"    CD-Golden sweep {sweep}: obj={best_obj:.8f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a = a / a.sum()
    return a
