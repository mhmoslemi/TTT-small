    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, b, b_max


def run_nesterov(a, deadline, max_iters=3000, tau_start=2.0, tau_end=300.0,
                 lr_init=0.4, lr_end=0.005):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta_ns = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_ns * beta_ns)) / 2.0
        momentum = beta_ns * (y - a)
        a_new = np.maximum(y - lr * (grad + momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def run_lp_adaptive(a, deadline, max_iters=2000, t_init=0.4, min_t=0.002):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    indices = precompute_conv_indices(n)
    best_a = a.copy()
    best_J = compute_J(a)
    no_imp = 0
    current_t = t_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        A = build_conv_matrix_fast(n, a, indices)
        g = lp_best_response(a, M, A)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        t = current_t
        for trial in range(10):
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                t /= 2
                continue
            a_t /= s
            J_t = compute_J(a_t)
            if J_t < best_J - 1e-13:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                current_t = min(t * 1.5, t_init)
                improved = True
                break
            t /= 2

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            current_t = max(min_t, t * 0.5)
            if no_imp > 30:
                break

    return best_a, best_J


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        p /= p.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.005))

    inits.append(np.ones(n) / n)

    for alpha in [0.3, 0.7, 1.0, 1.5, 3.0]:
        shape = np.maximum(0.3 + 0.7 * (t**alpha + (1 - t)**alpha) / 2, 0.01)
        inits.append(shape / shape.sum())

    for f in [1, 2, 3, 5, 7]:
        shape = np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01)
        inits.append(shape / shape.sum())

    inits.append((np.exp(-4 * (t - 0.5)**2) + 0.05) / n)
    inits.append((np.exp(-8 * (t - 0.3)**2) + np.exp(-8 * (t - 0.7)**2) + 0.03) / n)

    k = max(3, n // 16)
    kern = np.ones(k) / k
    smooth_rand = np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01)
    inits.append(smooth_rand / smooth_rand.sum())

    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full / full.sum())

    inits.append(spectral_init(n, rng, n_iters=200, beta=0.8))
    inits.append(spectral_init(n, rng, n_iters=200, beta=0.9))

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        p /= p.sum()
        inits.append(spectral_init(p * p.astype(float) if False else np.maximum(rng.normal(0, 0.5, n), 0.1), rng, n_iters=150, beta=0.85))

    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    base_n = parent.size if parent.size > 0 else 1024
    print(f"Base length: {base_n}", flush=True)

    # === PHASE 1: Multi-length smooth optimization ===
    p1_budget = 250.0
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Nesterov Optimization ({p1_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Focus on lengths around the parent and some larger ones
    lengths_to_try = sorted(set([
        base_n,
        max(128, base_n - 128),
        max(128, base_n - 64),
        base_n + 64,
        base_n + 128,
        base_n + 256,
        768, 896, 1024, 1152, 1280, 1408, 1536
    ]))
    lengths_to_try = [n for n in lengths_to_try if 128 <= n <= 2048]
    print(f"Lengths: {lengths_to_try}", flush=True)

    best_per_length = {}
    n_lengths = len(lengths_to_try)
    time_per_length = p1_budget / n_lengths

    for n in lengths_to_try:
        t0 = time.monotonic()
        remaining = deadline - 300 - time.monotonic()
        if remaining < 30:
            break
        allot = min(time_per_length, remaining * 0.4)
        dl = time.monotonic() + allot
        if time.monotonic() >= dl:
            continue

        inits = make_initializations(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')

        n_iters = max(300, int(5000 * (256.0 / n)))
        for idx, a0 in enumerate(inits):
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            frac = 1.0 / (1.0 + idx * 0.4)
            sub_dl = time.monotonic() + (dl - time.monotonic()) * frac
            a = run_nesterov(a0, sub_dl, max_iters=n_iters,
                            tau_start=2.0, tau_end=400.0,
                            lr_init=0.35, lr_end=0.003)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: M={best_M:.7f} J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: LP mirror descent on best sequences ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (Adaptive Step)", flush=True)
    print(f"{'='*60}", flush=True)

    # Sort lengths by their J value from Phase 1
    if best_per_length:
        sorted_lengths = sorted(best_per_length.keys(),
                                key=lambda x: compute_J(best_per_length[x]))
    else:
        sorted_lengths = [base_n]

    # Run LP on top 3 lengths
    for rank, n_try in enumerate(sorted_lengths[:3]):
        if time.monotonic() > deadline - 100:
            break

        a_start = best_per_length.get(n_try)
        if a_start is None:
            if best.seq is not None:
                a_start = resample_seq(best.seq, n_try)
            else:
                continue
        a_start = np.maximum(a_start, 1e-8)
        a_start /= a_start.sum()

        # Allocate more time to the best length
        if rank == 0:
            lp_time = min(500.0, max(60.0, (deadline - 120) - time.monotonic()))
        elif rank == 1:
            lp_time = min(300.0, max(30.0, (deadline - 100) - time.monotonic()))
        else:
            lp_time = min(150.0, max(20.0, (deadline - 80) - time.monotonic()))

        dl_lp = time.monotonic() + lp_time
        print(f"  LP on n={n_try}, budget={lp_time:.0f}s", flush=True)

        a_opt, J_opt = run_lp_adaptive(a_start, dl_lp, max_iters=5000,
                                        t_init=0.4, min_t=0.002)
        print(f"    Result: J={J_opt:.8f} ({time.monotonic()-p2_start:.1f}s elapsed)", flush=True)
        best.update(a_opt)

        if a_opt is not None and J_opt < best.score + 1e-8:
            best_per_length[n_try] = a_opt.copy()

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Randomized restarts to escape local minima ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Randomized Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 60:
        n_best = best.seq.size
        n_restarts = 5
        restart_budget = min(200.0, (deadline - 60) - time.monotonic())
        time_per_restart = restart_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break
            dl_r = time.monotonic() + time_per_restart

            # Create perturbed version of best
            a_pert = best.seq.copy()
            noise_scale = 0.05 + 0.05 * r
            a_pert = np.maximum(a_pert + rng.normal(0, noise_scale, n_best), 0.001)
            a_pert /= a_pert.sum()

            # Quick smooth optimization
            a_smooth = run_nesterov(a_pert, min(time.monotonic() + time_per_restart * 0.4, dl_r),
                                    max_iters=500, tau_start=5.0, tau_end=100.0,
                                    lr_init=0.2, lr_end=0.01)
            J_s = compute_J(a_smooth)
            print(f"  Restart {r+1}: smooth J={J_s:.6f}", flush=True)
            best.update(a_smooth)

            # Quick LP polish
            if time.monotonic() < dl_r:
                a_lp, J_lp = run_lp_adaptive(a_smooth, dl_r, max_iters=200,
                                              t_init=0.3, min_t=0.005)
                print(f"  Restart {r+1}: LP J={J_lp:.6f}", flush=True)
                best.update(a_lp)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Final fine-tuning LP ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 5:
        # Very careful LP with small steps
        a_final, J_final = run_lp_adaptive(best.seq, deadline, max_iters=3000,
                                            t_init=0.2, min_t=0.001)
        print(f"  Final LP: J={J_final:.8f}", flush=True)
        best.update(a_final)

        # One more pass with even smaller steps
        if time.monotonic() < deadline - 3:
            a_final2, J_final2 = run_lp_adaptive(a_final, deadline, max_iters=1000,
                                                  t_init=0.1, min_t=0.0005)
            print(f"  Second polish: J={J_final2:.8f}", flush=True)
            best.update(a_final2)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Length neighborhood exploration ===
    if best.seq is not None and time.monotonic() < deadline - 10:
        n_best = best.seq.size
        for delta in [32, -32, 64, -64, 128, -128]:
            if time.monotonic() > deadline - 5:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2500:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = run_nesterov(a_ext, min(time.monotonic() + 10, deadline - 3),
                                 max_iters=200, tau_start=5, tau_end=80,
                                 lr_init=0.1, lr_end=0.01)
            best.update(a_ext)
            if time.monotonic() < deadline - 3:
                a_lp, _ = run_lp_adaptive(a_ext, deadline - 2, max_iters=100,
                                           t_init=0.3, min_t=0.01)
                best.update(a_lp)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best.seq]
