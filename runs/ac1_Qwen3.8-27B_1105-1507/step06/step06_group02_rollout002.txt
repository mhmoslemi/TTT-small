def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_M(a):
    """Return max autocorrelation value (assuming sum=1)."""
    return float(fast_conv(a).max())

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2 * len(a) * compute_M(a) / (s * s))

def fft_cross_correlate(p, a):
    """Compute cross[i] = sum_j p[i+j]*a[j] for i=0..n-1 using FFT."""
    n = len(a)
    N = 1
    while N < 2 * n + len(p) - 1:
        N <<= 1
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    # cross-correlation: ifft(conj(P) * A)
    # But we want sum_j p[i+j]*a[j], which is a specific slice
    # Actually: (p_flipped * a)[i] where p_flipped[k] = p[-k]
    # Let me just use: cross[i] = sum_j p[i+j]*a[j]
    # = correlation of p and a at lag i (with p indexed forward)
    # Using FFT: ifft(conj(FFT(p_zero)) * FFT(a_zero)) gives (p ⋆ a)
    # where (p ⋆ a)[i] = sum_j p[j+i]*conj(a[j])
    # For real: (p ⋆ a)[i] = sum_j p[j+i]*a[j]
    # So we want the first n values of (p ⋆ a)
    Pc = np.conj(P)
    cross = np.fft.irfft(Pc * A, N)
    return cross[:n]

def smooth_max_grad_fft(a, tau):
    """Compute gradient of smooth max using FFT-based cross-correlation."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    # Compute probabilities
    shifted = (b - b_max) * tau  # all <= 0
    # For numerical stability
    if shifted.min() < -700:
        shifted = np.maximum(shifted, -700)
    exp_b = np.exp(shifted)
    total = exp_b.sum()
    if total < 1e-300:
        p = np.zeros(2*n-1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_b / total
    # Gradient: grad[i] = 2 * sum_j p[i+j] * a[j] = 2 * (p ⋆ a)[i]
    grad = 2.0 * fft_cross_correlate(p, a)
    return grad, p, b, b_max

def gs_flat_spectral(init, n_iters=200, beta=0.85):
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_mean = mag.mean()
        # Compress toward mean
        decay = beta * (1.0 - 0.5 * it / n_iters)
        mag_new = mag_mean + decay * (mag - mag_mean)
        mag_new = np.maximum(mag_new, 1e-10)
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a

def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=2.0, tau_end=500.0,
                        lr_init=0.5, lr_end=0.005):
    """Nesterov-accelerated smooth max gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta_mom = 0.9
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        # Adaptive temperature: start smooth, end sharp
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_max_grad_fft(y, tau)
        if b_max < best_M - 1e-12:
            best_M = b_max
            best_a = a.copy()
        
        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_mom**2)) / 2.0
        y = np.maximum(y - lr * (grad + beta_mom * (y - a)), 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y = y / sy
        
        a_new = y + ((t_k - 1.0) / (t_k + 1.0)) * (y - a)
        a_new = np.maximum(a_new, 0.0)
        sa = a_new.sum()
        if sa < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / sa
        a = a_new
        
        # Periodically check
        if it % 100 == 0:
            M_check = fast_conv(a).max()
            if M_check < best_M:
                best_M = M_check
                best_a = a.copy()
    
    return best_a

def build_conv_matrix(n, a):
    """Build sparse matrix for conv(a, g) = A @ g."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            g = np.maximum(r.x, 0)
            # Extract dual variables for KKT analysis
            y_dual = None
            if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                if hasattr(r.ineqlin, 'marginals'):
                    y_dual = r.ineqlin.marginals
            return g, y_dual
    except Exception:
        pass
    return None, None

def anderson_accelerate(history_x, history_Fx, m, extrap_range=(0.5, 1.5)):
    """Anderson acceleration: find extrapolated point with minimum residual."""
    # Residuals
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T + 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        alpha = np.linalg.solve(G, ones)
        denom = alpha.sum()
        if abs(denom) < 1e-15:
            alpha = np.ones(m) / m
        else:
            alpha = alpha / denom
    except np.linalg.LinAlgError:
        alpha = np.ones(m) / m
    
    # Clamp for stability
    alpha = np.clip(alpha, -2.0, 2.0)
    alpha = alpha / alpha.sum()
    
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)

        s = a_new.sum()
        if s < 1e-10:
