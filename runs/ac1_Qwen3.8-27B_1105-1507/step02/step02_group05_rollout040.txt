def fast_conv(a):

    n = len(a)

def compute_J(a):

def hankel_matvec(a, w):
    """Compute (w ⊛ a)[i] = sum_j a[j] * w[i+j] for i=0..n-1 using FFT."""

                fd = float(np.max(conv_base + d_gs * c + d_gs**2 * d_coeff))
                
                for _ in range(40):
                    if fc < fd:
                        b_gs = d_gs
                        d_gs = c_gs
                        fd = fc
                        c_gs = b_gs - gr * (b_gs - a_gs)
                        fc = float(np.max(conv_base + c_gs * c + c_gs**2 * d_coeff))
                    else:
                        a_gs = c_gs
                        c_gs = d_gs
                        fc = fd
                        d_gs = a_gs + gr * (b_gs - a_gs)
                        fd = float(np.max(conv_base + d_gs * c + d_gs**2 * d_coeff))
                
                t_best = (a_gs + b_gs) / 2
            
            # Apply update
            a[i] = max(t_best, 0.0)
            new_s = a.sum()
            if new_s > 1e-10:
                pass  # will normalize at end
            if abs(t_best - save_ai) > 1e-10:
                improved = True
        
        a = a / a.sum()
        
        if not improved:
            break
    
    return a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # A[k,j] = a[k-j] for j in [jl, jh]
            # a[k-j] for j = jl, ..., jh gives a[k-jl], a[k-jl-1], ..., a[k-jh]
            # = a[k-jh : k-jl+1] reversed
            segment = a[k-jh:k-jl+1][::-1]
            vals.extend(segment.tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker_callback=None):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker_callback:
        tracker_callback(a)
    
    n = len(a)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Try multiple step sizes
        base_t = max(0.03, 0.4 * (1.0 - it / float(2 * max_iters)))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5)]
        
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            t_small = 0.01
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J
        
        if tracker_callback and it % 20 == 0:
            tracker_callback(a)
    
    if tracker_callback:
        tracker_callback(a)
    return a


class ScoreTracker:
    """Tracks the best sequence found so far."""
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.3, 1.0, n))
    
    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 4, 5, 6, 7, 8]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr))
    
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * t_arr ** 2 * 3))
    inits.append(1.0 + 0.5 * (2.0 * t_arr - 1.0) ** 2)
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.1)
    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * t_arr + rng.uniform(0, 6.28)))
    
    # S-curve
    inits.append(1.0 / (1.0 + np.exp(-8.0 * (t_arr - 0.5))))
    
    # Triangular wave
    inits.append(1.0 + 0.3 * (2.0 * np.abs(2.0 * t_arr - 1.0) - 1.0))
    
    # Sparse structure
    sparse = np.zeros(n)
    n_spikes = max(4, n // 20)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel = np.ones(5) / 5.0
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.03)
    
    # Power-law
    inits.append((t_arr + 0.1) ** rng.uniform(-1, 1))
    inits.append((1.0 - t_arr + 0.1) ** rng.uniform(-1, 1))
    
    # Two-hump
    inits.append(np.exp(-10*(t_arr-0.25)**2) + np.exp(-10*(t_arr-0.75)**2) + 0.05)
    
    # Symmetric random
    half = n // 2
    h = 0.5 + 0.5 * rng.normal(0, 0.3, half + 1)
    h = np.maximum(h, 0.01)
    full = np.concatenate([h, h[:0:-1]])
    inits.append(full)
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    1. Soft-max L-BFGS-B with temperature annealing (log-parameterized)
    2. Coordinate descent polish
    3. LP mirror descent
    4. Multi-length exploration
    5. Final combined polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None
        print("No parent available, starting fresh.", flush=True)
    
    # === Phase 1: Soft-max L-BFGS-B with annealing ===
    # Try multiple lengths
    if parent is not None:
        base_n = parent.size
    else:
        base_n = 128
    
    lengths = sorted(set([base_n, base_n * 2, base_n // 2 if base_n > 64 else 64,
                            128, 256, 512]))
    lengths = [n for n in lengths if 32 <= n <= 2048]
    
    print(f"\n{'='*50} Phase 1: Soft-max L-BFGS-B {'='*50}", flush=True)
    
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- n={n} ---", flush=True)
        
        inits = generate_inits(n, rng, parent)
        
        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()
            
            # Soft-max refinement
            a_ref, obj_ref = softmax_refine(a0, deadline=deadline, max_iter=150)
            tracker.update(a_ref)
            
            # Quick coordinate descent
            a_cd = coord_descent(a_ref, max_sweeps=10, deadline=deadline)
            tracker.update(a_cd)
            
            if (idx + 1) % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    init {idx+1}/{len(inits)} done, best so far: {tracker.best_score:.8f}, t={elapsed:.1f}s", flush=True)
        
        elapsed = time.monotonic() - t0
        print(f"  n={n} done in {elapsed:.1f}s, best: {tracker.best_score:.8f}", flush=True)
    
    # === Phase 2: LP Mirror Descent on best sequence ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 2: LP Mirror Descent n={n} {'='*50}", flush=True)
        
        a = lp_mirror_descent(tracker.best, max_iters=200, deadline=deadline,
                              tracker_callback=tracker.update)
        
        # Short coordinate descent
        a = coord_descent(a, max_sweeps=20, deadline=deadline)
        tracker.update(a)
        
        print(f"  LP done in {time.monotonic()-t0:.1f}s, best: {tracker.best_score:.8f}", flush=True)
    
    # === Phase 3: SA-style restarts with combined refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 3: SA Restarts n={n} {'='*50}", flush=True)
        
        temperature = 0.1
        n_restarts = 40
        
        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, temperature, n)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            
            # Combined: soft-max + coord descent
            a_ref, _ = softmax_refine(a, deadline=deadline, max_iter=50)
            a_ref = coord_descent(a_ref, max_sweeps=5, deadline=deadline)
            tracker.update(a_ref)
            
            if r % 10 == 0:
                print(f"    restart {r+1}/{n_restarts}, T={temperature:.4f}, best={tracker.best_score:.8f}", flush=True)
            
            temperature *= 0.95
        
        print(f"  SA done in {time.monotonic()-t0:.1f}s, best: {tracker.best_score:.8f}", flush=True)
    
    # === Phase 4: Final polish - alternating LP + CD + L-BFGS-B ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 4: Final Polish n={n} {'='*50}", flush=True)
        
        a = tracker.best.copy()
        
        for round_num in range(10):
            if time.monotonic() > deadline:
                break
            
            # LP step
            a = lp_mirror_descent(a, max_iters=30, deadline=deadline)
            # CD step
            a = coord_descent(a, max_sweeps=10, deadline=deadline)
            tracker.update(a)
            
            # L-BFGS-B with soft-max
            a_ref, _ = softmax_refine(a, deadline=deadline, max_iter=100)
            a_ref = coord_descent(a_ref, max_sweeps=10, deadline=deadline)
            tracker.update(a_ref)
            
            if time.monotonic() > deadline:
                break
        
        print(f"  Final polish done in {time.monotonic()-t0:.1f}s, best: {tracker.best_score:.8f}", flush=True)
    
    # === Phase 5: Try different lengths for the best pattern ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 5: Multi-length exploration {'='*50}", flush=True)
        
        base_n = tracker.best.size
        candidate_ns = []
        for mult in [0.75, 1.0, 1.25, 1.5, 2.0]:
            cn = int(base_n * mult)
            cn = max(32, min(4096, cn))
            if cn not in candidate_ns:
                candidate_ns.append(cn)
        
        for cn in candidate_ns:
            if time.monotonic() > deadline:
                break
            if cn == base_n:
                continue
            
            a_resampled = resample_seq(tracker.best, cn)
            a_resampled = np.maximum(a_resampled, 1e-6)
            a_resampled = a_resampled / a_resampled.sum()
            
            a_ref, _ = softmax_refine(a_resampled, deadline=deadline, max_iter=100)
            a_ref = coord_descent(a_ref, max_sweeps=15, deadline=deadline)
            tracker.update(a_ref)
            
            if time.monotonic() > deadline:
                break
        
        print(f"  Multi-length done in {time.monotonic()-t0:.1f}s, best: {tracker.best_score:.8f}", flush=True)
    
    # === Final: one more CD + LP on the best ===
    if tracker.best is not None and time.monotonic() < deadline:
        a = coord_descent(tracker.best, max_sweeps=30, deadline=deadline)
        tracker.update(a)
        a = lp_mirror_descent(a, max_iters=50, deadline=deadline, tracker_callback=tracker.update)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    
    return [float(x) for x in tracker.best]
