    """One step of projected gradient descent."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    # Log-sum-exp weights
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum
    
    # Compute gradient: grad[i] = 2 * sum_j a[j] * weights[i+j]
    # Using FFT-based convolution
    size = 1
    while size < 3 * n:

    rev_a = a[::-1].copy()
    fa = np.fft.rfft(rev_a, size)
    fw = np.fft.rfft(weights, size)
    crosscorr = np.fft.irfft(fa * fw, size)
    grad = 2.0 * crosscorr[n-1 : 2*n-1]
    
    # Gradient step

"""

smoothing, FFT-accelerated gradients, multiple initializations (including ZC-inspired),

import numpy as np
import time
import math


def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v, s=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    alpha = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - alpha, 0)


def compute_gradient(conv, a, tau):
    """Compute smoothed gradient of max(conv) using log-sum-exp weights.
    Returns gradient of length n using FFT-based Hankel matrix-vector product."""
    n = len(a)
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum

    # grad[i] = 2 * sum_j a[j] * weights[i+j]
    # = 2 * convolve(a[::-1], weights)[n-1+i]
    size = 1
    while size < 3 * n:
        size <<= 1
    rev_a = a[::-1]
    fa = np.fft.rfft(rev_a, size)
    fw = np.fft.rfft(weights, size)
    crosscorr = np.fft.irfft(fa * fw, size)
    grad = 2.0 * crosscorr[n - 1:2 * n - 1]
    return grad


def projected_gradient_optimize(a, max_iter=10000, tau_start=10.0, tau_end=80.0,
                                eta_init=0.05, verbose=False, rng=None):
    """Run projected gradient descent to minimize max(conv(a,a)) with sum(a)=1."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        return a, np.max(compute_conv(a))
    a = a / s

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    eta = eta_init
    stall_count = 0
    patience = 500

    for it in range(max_iter):
        frac = it / max(1, max_iter)
        # Animate tau: start smooth, end sharp
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)

        conv = compute_conv(a)
        current_max = float(np.max(conv))

        if current_max < best_max - 1e-12:
            best_max = current_max
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                # Restart with perturbation (simulated annealing style)
                noise_scale = 0.1 * max(1.0, 1.0 - frac)
                a_pert = a * np.exp(rng.normal(0, noise_scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p < 1e-10:
                    break
                a_pert = a_pert / s_p
                obj_p = np.max(compute_conv(a_pert))
                if obj_p < best_max:
                    a = a_pert
                    best_max = obj_p
                    best_a = a.copy()
                else:
                    # Accept with probability based on "temperature"
                    temp = 0.01 * (1.0 - frac)
                    if rng.random() < math.exp(-(obj_p - best_max) / max(temp, 1e-10)):
                        a = a_pert
                    else:
                        a = best_a.copy()
                stall_count = 0
                continue

        grad = compute_gradient(conv, a, tau)

        # Adaptive step size: scale by the magnitude of a to keep things stable
        eta_adapt = eta / (1.0 + 0.01 * current_max * n)
        a_new = a - eta_adapt * grad
        a_new = project_simplex(a_new, 1.0)
        a = a_new

        # Decrease step slowly
        if it > 100:
            eta *= 0.9997

    return best_a, best_max

def make_init_cosine(n, rng):
    t = np.arange(n) / n
    phase = rng.uniform(0, 2 * np.pi)
    a = 0.5 * (1 + np.cos(2 * np.pi * t + phase))
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:

    return a / s

    freq = rng.uniform(0.5, 3.0)

    a = 1.0 + np.cos(np.pi * freq * i * i / n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return make_init_uniform(n)
    return a / s

    center = n / 2.0
    a = np.exp(-rate * (np.arange(n) - center) ** 2 / (2 * center ** 2))
    s = np.sum(a)
    return a / s


def make_init_random(n, rng):

    a = a * window
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return make_init_uniform(n)
    return a / s


def make_init_piecewise(n, rng):
    """Piecewise constant with random breakpoints."""
    n_segs = rng.integers(3, 10)
    breakpoints = np.sort(rng.choice(np.arange(1, n), n_segs - 1, replace=False))
    values = rng.uniform(0.1, 1.0, n_segs)
    a = np.zeros(n)
    starts = [0] + list(breakpoints)
    ends = list(breakpoints) + [n]
    for j in range(n_segs):
        a[starts[j]:ends[j]] = values[j]
    # Apply smooth window
    t = np.arange(n) / n
    a = a * (0.2 + 0.8 * np.sin(np.pi * t))
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return make_init_uniform(n)
    return a / s


def make_init_shifted_uniform(n, rng):
    """Uniform but with one end tapered."""
    a = np.ones(n)
    taper_len = int(n * rng.uniform(0.2, 0.5))

    else:
        a[-taper_len:] *= np.linspace(1, 0, taper_len)
    s = np.sum(a)
    return a / s

def resize_sequence(a, new_n):
    """Resize sequence to new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

        best_score = parent_obj
        best_seq = parent.copy()

    # Sequence lengths to explore (ordered by expected value)
    n_values = [
        100, 120, 150, 200, 250, 300, 400, 500, 600,
        800, 1000, 1200, 1500, 2000
    ]

    # If we have a parent, include its length and neighbors
    if parent is not None:
        n_parent = len(parent)
        extra = [n_parent, max(50, n_parent - 50), min(2500, n_parent + 50),
                 max(50, n_parent - 100), min(2500, n_parent + 100)]
        n_values = sorted(set(n_values + [int(x) for x in extra]))

    print(f"Exploring n values: {n_values}")
    print(f"Time budget: {budget_s:.0f}s, deadline in {deadline - start_time:.0f}s")

    # Phase 1: Explore different lengths
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        # Time allocation: give more time to promising lengths
        if n <= 200:
            alloc = min(30.0, remaining * 0.06)
        elif n <= 400:
            alloc = min(45.0, remaining * 0.08)
        elif n <= 800:
            alloc = min(60.0, remaining * 0.1)
        else:
            alloc = min(80.0, remaining * 0.12)
        alloc = max(8.0, alloc)

        if remaining < alloc:
            alloc = max(5.0, remaining * 0.5)

        print(f"N={n}: {remaining:.0f}s remaining, allocating {alloc:.0f}s")

        a_opt, obj = optimize_single(n, seed=seed + n * 7919,
                                     time_budget=alloc, verbose=(n in [150, 250, 500]))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  UPDATE: n={n}, obj={best_score:.6f}")

    # Phase 2: Focus on the best length and nearby lengths
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        nearby = sorted(set([
            max(50, n_best - 20), max(50, n_best - 50),
            n_best,
            n_best + 20, n_best + 50, n_best + 100
        ]))
        print(f"\nPhase 2: Focusing on n near {n_best}")

        for n in nearby:
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(60.0, remaining * 0.15)
            alloc = max(10.0, alloc)

            print(f"  N={n}: allocating {alloc:.0f}s")
            a_opt, obj = optimize_single(n, seed=seed + n * 31337,
                                         time_budget=alloc, verbose=False)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, obj={best_score:.6f}")

    # Phase 3: Intensive polishing of the best sequence
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 3: Polishing best sequence (n={len(best_seq)}), {remaining:.0f}s remaining")

        # Split remaining time among polish rounds
        a_polished, obj_polished = polish_sequence(
            best_seq, time_budget=remaining * 0.7, rng=rng, verbose=True
        )
        if obj_polished < best_score:
            best_score = obj_polished
            best_seq = a_polished.copy()

    # Phase 4: Final micro-polish if time remains
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            est_time = 0.0005 * (len(best_seq) / 250) ** 1.5
            max_iter = int(min(20000, max(500, remaining / max(est_time, 0.0001))))
            a_final, _ = projected_gradient_optimize(
                best_seq, max_iter=max_iter,
                tau_start=50.0, tau_end=200.0,
                eta_init=0.01, rng=rng
            )
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final micro-polish: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation != best_score and validation < np.inf:
        print(f"  Validation: computed={best_score:.6f}, validated={validation:.6f}")

    return result
