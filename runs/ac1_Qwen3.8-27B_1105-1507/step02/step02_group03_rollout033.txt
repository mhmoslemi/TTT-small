When k = j+l for l = 0, ..., n-1: f[k-j] = f[l]
So `g[j] = sum_{l=0}^{n-1} w[j+l] * f[l]` (with appropriate clipping)

In my code: `k_min = j`, `k_max = j+n`, and `f[0:k_max-k_min] = f[0:n]`.
`w[k_min:k_max] = w[j:j+n]` (clamped to valid range).

Yes, this looks correct.

Now let me think about the step size (eta) more carefully. The update is:
`f_new[j] = f[j] * exp(-eta * (g[j] - dot(f, g)))`

For a typical f with sum=1 and n=512:
- f[j] ≈ 1/512 ≈ 0.002
- g[j] ≈ sum of w[k]*f[k-j] ≈ (average w) * (some f value) ≈ (1/1023) * 0.002 ≈ 2e-6
- dot(f, g) ≈ sum f[j]*g[j] ≈ (1/n) * (1/n) * n ≈ 1/n ≈ 0.002... 

Hmm, let me compute this more carefully. 
`dot(f, g) = sum_j f[j] * sum_k w[k]*f[k-j] = sum_k w[k] * sum_j f[j]*f[k-j] = sum_k w[k] * conv(f,f)[k]`

So `dot(f, g) = inner product of w and conv(f,f)`. Since w is a probability distribution concentrated near the max of conv, this is approximately `M * w[argmax] + ...` ≈ `M * (something close to 1 if T is small)`.

If T is small, `dot(f,g) ≈ M ≈ 0.0014` (for score 1.5, n=512).
And `g[j] ≈ f[k*-j]` where k* is the argmax. So `g[j] ≈ 1/n ≈ 0.002`.

So `g[j] - dot(f,g) ≈ 0.002 - 0.0014 = 0.0006`.
The relative change in f[j] is `exp(-eta * 0.0006) - 1 ≈ -eta * 0.0006`.
For a 10% change: `eta * 0.0006 ≈ 0.1`, so `eta ≈ 167`.

So eta0 should be around 100-500. Let me use eta0=200.

Actually, I realize I should be more careful. The "centering" `g[j] - dot(f, g)` ensures that the weighted sum of the log-changes is zero:
`sum_j f[j] * (g[j] - dot(f,g)) = dot(f,g) - dot(f,g) * sum_j f[j] = dot(f,g) - dot(f,g) = 0`

Wait, that's not right. `sum_j f[j] * (g[j] - dot(f,g)) = sum_j f[j]*g[j] - dot(f,g) * sum_j f[j] = dot(f,g) - dot(f,g) * 1 = 0`. Yes! So the first-order change in the sum is zero, which means the normalization step is a second-order correction. Good.

"""
Multi-strategy optimizer combining mirror descent with soft-max gradient,
LP-based projection refinement, coordinate descent polishing, and diverse
initializations to minimize 2n*max(conv(f,f))/sum(f)^2.
"""

import numpy as np
from scipy.optimize import linprog
import time


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def compute_score(f):
    """Compute the score: 2*n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))


def compute_pressure(f, conv, T):
    """Compute soft-pressured gradient: g[j] = sum_k w[k]*f[k-j]."""
    n = len(f)
    max_c = conv.max()
    exp_c = np.exp((conv - max_c) / T)
    w = exp_c / exp_c.sum()

    g = np.zeros(n)
    for j in range(n):
        k_min = max(0, j)
        k_max = min(len(conv), j + n)
        if k_min < k_max:
            g[j] = np.dot(w[k_min:k_max], f[0:k_max - k_min])
    return g


def soft_gradient_descent(f, n_iter, T0, T_end, eta0, eta_decay, rng=None):
    """Exponentiated gradient (mirror descent) with annealed temperature."""
    if rng is None:
        rng = np.random.default_rng(42)

    n = len(f)
    f = f.copy().astype(np.float64)
    f = np.clip(f, 1e-15, None)
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        progress = it / max(1, n_iter - 1)
        T = T0 * (T_end / T0) ** progress
        eta = eta0 * (eta_decay ** it)

        conv = np.convolve(f, f)
        g = compute_pressure(f, conv, T)

        # Center to preserve sum (first order)
        fg = np.dot(f, g)
        g_centered = g - fg

        # Mirror descent update
        f_new = f * np.exp(-eta * g_centered)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 300 and it > n_iter * 0.5:
                # Perturb to escape
                f = f * (1.0 + 0.15 * rng.normal(0, 1, n))
                f = np.clip(f, 1e-15, None)
                f = normalize(f)
                stall = 0
                if stall > 500:
                    break
    return best_f, best_score


def build_conv_matrix(f):
    """Build matrix A such that A @ x = np.convolve(f, x, mode='full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    idx = np.arange(n)
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[k, lo:hi] = f[k - (lo:hi)]
    return A


def lp_refine(f, time_budget, t0=0.5, rng=None, lp_time=2):
    """LP-based refinement: find best direction g keeping conv(f,g) <= M."""
    if rng is None:
        rng = np.random.default_rng(42)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                   'time_limit': lp_time})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = np.clip(res.x, 0, None)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        improved = False
        for step in [min(1.0, t * 2.0), t, t * 0.5, t * 0.25]:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.3)
                improved = True
                break

        if not improved:
            t *= 0.65
            if t < 0.01:
                f_pert = f + 0.05 * rng.normal(0, 1, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.3
                else:
                    break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, rng=None):
    """Coordinate descent polishing with geometric factors."""
    if rng is None:
        rng = np.random.default_rng(42)

    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    factors = [0.3, 0.5, 0.7, 0.85, 1.0 / 0.85, 1.0 / 0.7, 1.0 / 0.5, 1.0 / 0.3]

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                continue
            best_val = original
            best_local = best_score

            for factor in factors:
                f[i] = max(0.0, original * factor)
                conv = np.convolve(f, f)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def diverse_init(n, rng):
    """Generate a diverse initial sequence."""
    r = rng.random()
    if r < 0.25:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.5:
        # U-shaped (high at ends, low in middle) - reduces central autocorr peak
        f = np.cos(np.pi * np.arange(n) / n) ** 2 + 0.1
    elif r < 0.7:
        # Sine product
        f = np.abs(np.sin(np.pi * np.arange(n) / n * rng.uniform(1.5, 3.0))) + 0.05
    elif r < 0.85:
        # Multi-bump
        f = np.ones(n) * 0.3
        for _ in range(4):
            c = int(rng.integers(0, n))
            w = int(rng.integers(5, n // 4))
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
            f += rng.uniform(0.5, 1.5) * kernel
    else:
        # Smooth random
        f = np.ones(n)
        for _ in range(6):
            c = int(rng.integers(0, n))
            w = int(rng.integers(4, n // 3))
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
            f *= (1.0 + 0.4 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05

    f = np.clip(f, 0.01, None)
    f = f * (1.0 + 0.08 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def spectral_init(n, rng):
    """Initialize from flat-spectrum idea: generate random phases, IRFFT, clip."""
    n_fft = 1 << (2 * n - 1).bit_length()
    # Random phases, flat magnitude
    phases = rng.uniform(0, 2 * np.pi, n_fft // 2 + 1)
    mag = np.ones(n_fft // 2 + 1)
    mag[0] = np.sqrt(n)  # DC
    Z = mag * np.exp(1j * phases)
    # Make it conjugate symmetric for real output
    Z_full = np.zeros(n_fft, dtype=complex)
    Z_full[:n_fft // 2 + 1] = Z
    for k in range(1, n_fft // 2):
        Z_full[n_fft - k] = np.conj(Z[k])
    f = np.fft.irfft(Z_full, n_fft)[:n]
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-10:
        return diverse_init(n, rng)
    f = f / s
    f = f * (1.0 + 0.05 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f
