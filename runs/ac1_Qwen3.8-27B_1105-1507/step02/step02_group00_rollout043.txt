import time

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A_matrix(a):
    """Build the constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp_given_a(a, M):
    """Solve: max 1^T g s.t. A a-related constraints, g >= 0."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternation(a_init, max_iter=500, relax=1.0, blend_rate=0.5):
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    best = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iter):
        conv_a = np.convolve(a, a)
        M = conv_a.max() * relax
        g = solve_lp_given_a(a, M)
        if g is None:

        if g_sum < 1e-15:

        g /= g_sum

            best = g.copy()

"""
Optimized step function search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternation, smoothed gradient descent (log-sum-exp), and coordinate refinement.
Exploits non-palindromic structure for better-than-uniform solutions.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a, M):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternate(a_init, max_iter=500, relax=1.0, blend=0.5):
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(len(a), dtype=np.float64) / len(a)
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    prev_g = None
    
    for it in range(max_iter):
        conv_a = np.convolve(a, a)
        M = float(conv_a.max()) * relax
        g = solve_lp(a, M)
        if g is None:
            break
        gs = np.sum(g)
        if gs < 1e-15:
            break
        g = g / gs
        
        obj_g = compute_obj(g)
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()
        
        # Check convergence
        if prev_g is not None:
            diff = np.max(np.abs(g - prev_g))
            if diff < 1e-10:
                break
        prev_g = g.copy()
        
        # Adaptive blend
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s2 = np.sum(a_new)
        if s2 < 1e-15:
            a = g.copy()
        else:
            a = a_new / s2
        
        obj_a = compute_obj(a)
        if obj_a < best_obj:
            best_obj = obj_a
            best_a = a.copy()
    
    return best_a, best_obj


def smoothed_gradient_descent(a_init, n_steps=10000, tau_init=0.1, tau_min=0.001, eta=None):
    """Smoothed GD using log-sum-exp approximation of max."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for step in range(n_steps):
        conv = np.convolve(a, a)
        tau = max(tau_min, tau_init * (tau_min / tau_init) ** (step / n_steps))
        
        # Log-sum-exp for numerical stability
        conv_shifted = conv / tau
        conv_max = np.max(conv_shifted)
        log_sum_exp = conv_max + np.log(np.sum(np.exp(conv_shifted - conv_max)))
        
        # Weights: softmax of conv/tau
        weights = np.exp(conv_shifted - conv_max)
        weights = weights / np.sum(weights)
        
        # Gradient: weighted average of 2*a[k-j]
        b = a[::-1]
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-15:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx_s = n - 1 - k + j_start
            idx_e = n - 1 - k + j_end + 1
            grad[j_start:j_end + 1] += weights[k] * 2.0 * b[idx_s:idx_e]
        
        # Step size
        if eta is None:
            eta = 1.0 / (2.0 * max(1.0, np.max(a)) * n)
        else:
            eta = eta / (1.0 + 0.01 * step)
        
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s2 = np.sum(a_new)
        if s2 < 1e-15:
            break
        a_new = a_new / s2
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_obj
