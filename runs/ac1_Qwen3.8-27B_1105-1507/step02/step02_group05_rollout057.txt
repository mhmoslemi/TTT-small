"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based mirror descent with adaptive step sizes
2. Coordinate descent polish with geometric grid search
3. Simulated annealing with LP restarts
4. Multi-length search with resampling

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # a[k-j] for j in range(jl, jh+1)
            # When j=jl, k-j = k-jl; when j=jh, k-j = k-jh
            vals.extend(a[k-jh:k-jl+1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

- The change in b[k] due to a[i] going from x to t is:

  - Else: 2*(t-x)*a[k-i]

- When k = 2i: the term is a[i]*a[i] = a[i]^2, which changes from x^2 to t^2. Change = t^2 - x^2.
- When k != 2i but 0 <= k-i < n: the term a[i]*a[k-i] appears twice (once with j=i and once with j=k-i). Change = 2*(t-x)*a[k-i].

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based mirror descent with adaptive geometric step sizes
2. Vectorized coordinate descent with exact 1D subproblem solving
3. Simulated annealing with LP restarts  
4. Multi-length search with resampling
5. Final slow LP polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k-jh:k-jl+1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def lp_mirror_descent(a, max_iters, deadline, tracker, step_decay=0.995):
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    t = 0.3  # initial step size
    n = len(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Try multiple step sizes
        best_a = None
        best_J = current_J
        
        for t_candidate in [t, t * 0.5, t * 1.5, t * 0.25]:
            t_candidate = min(t_candidate, 1.0)
            a_trial = (1.0 - t_candidate) * a + t_candidate * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # No improvement, reduce step
            t *= 0.8
            if t < 0.01:
                break
            continue
        
        t *= step_decay
        
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def coord_desc_polish(a, max_sweeps=50, deadline=None):
    """Vectorized coordinate descent with exact 1D subproblem."""
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a
    b = fast_conv(a)
    total_len = 2 * n - 1
    
    J_prev = 2.0 * n * b.max() / (S * S)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        for i in range(n):
            x = a[i]
            C = S - x
            if C < 1e-12:
                continue
            
            k_lo = i
            k_hi = min(total_len - 1, i + n - 1)
            
            # Non-affected max
            if k_lo > 0 or k_hi < total_len - 1:
                mask = np.ones(total_len, dtype=bool)
                mask[k_lo:k_hi+1] = False
                non_affected_max = b[mask].max() if mask.any() else 0.0
            else:
                non_affected_max = 0.0
            
            # Generate candidates from top active k values
            b_max = b.max()
            if b_max < 1e-15:
                continue
            
            # Top active indices
            num_active = min(50, total_len)
            active_ks = np.argsort(b)[::-1][:num_active]
            
            candidates = {0.0, x, 3.0 * a.max()}
            
            for k in active_ks:
                if k == 2 * i:
                    B_val = b[k] - x * x
                    if C > 1e-14 and B_val > 0:
                        candidates.add(B_val / C)
                else:
                    ki = k - i
                    if 0 <= ki < n and a[ki] > 1e-14:
                        t_star = S + x - b[k] / a[ki]
                        if t_star > 0:
                            candidates.add(t_star)
            
            candidates = sorted(t for t in candidates if t >= 0)
            
            # Evaluate candidates vectorized
            best_J = float('inf')
            best_t = x
            
            for t in candidates:
                S_new = S + t - x
                if S_new < 1e-10:
                    continue
                
                dx = t - x
                # Compute affected b values
                delta = 2.0 * dx * a[i:i+n]  # a[k-i] for k in [i, i+n-1]
                b_affected = b[k_lo:k_hi+1] + delta
                
                # Fix k=2i
                two_i = 2 * i
                if k_lo <= two_i <= k_hi:
                    idx = two_i - k_lo
                    b_affected[idx] = b[two_i] + t * t - x * x
                
                max_b = max(b_affected.max(), non_affected_max)
                J_val = 2.0 * n * max_b / (S_new * S_new)
                
                if J_val < best_J:
                    best_J = J_val
                    best_t = t
            
            if best_t != x and abs(best_t - x) > 1e-15 * max(1.0, x):
                dx = best_t - x
                a[i] = best_t
                S += dx
                
                # Update b
                delta = 2.0 * dx * a[i:i+n]
                b[k_lo:k_hi+1] += delta
                two_i = 2 * i
                if k_lo <= two_i <= k_hi:
                    idx = two_i - k_lo
                    # The delta already added 2*dx*a[i], but we need t^2 - x^2
                    # So adjust: add (t^2 - x^2) - 2*dx*a[i] = (t-x)(t+x) - 2(t-x)x = (t-x)(t-x) = dx^2
                    b[two_i] += dx * dx
        
        J_curr = 2.0 * n * b.max() / (S * S)
        if abs(J_prev - J_curr) < 1e-13 * J_prev:
            break
        J_prev = J_curr
    
    return a

def generate_initializations(n, rng, parent):
    inits = []
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    # Various tapers and shapes
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        inits.append(1.0 - alpha * (2.0 * t_arr - 1.0) ** 2)
    
    # Cosine variations
    for freq in [1, 2, 3, 4]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(1.0 + amp * np.cos(2.0 * np.pi * freq * t_arr))
    
    # Edge-enhanced
    inits.append(1.0 + 0.5 * np.cos(np.pi * t_arr))
    inits.append(np.abs(np.cos(np.pi * t_arr)) + 0.1)
    inits.append((1.0 - t_arr) ** 2 + 0.05)
    inits.append(t_arr ** 2 + 0.05)
    
    # Random
    inits.append(rng.uniform(0.2, 1.0, n))
    
    # Sparse-ish
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)
    
    # Ensure non-negative
    inits = [np.maximum(x, 0.01) for x in inits]
    
    return inits
