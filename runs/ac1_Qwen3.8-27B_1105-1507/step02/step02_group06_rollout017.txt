"""
Optimized search for low-peak-autocorrelation non-negative sequences.
Uses smoothed (log-sum-exp) gradient descent with Nesterov acceleration,
projected onto the probability simplex, combined with periodic LP-based
direction finding and multi-restart strategy.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(a):
    """Fast computation of 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    S = float(np.sum(a))
    if S < 1e-10:
        return float('inf')
    return 2.0 * n * M / (S * S)


def simplex_project(x):
    """Project x onto {a >= 0, sum(a) = 1}."""
    n = len(x)
    if np.sum(x) <= 0:
        return np.ones(n) / n
    # Simple approach: clip and renormalize (approximate but fast)
    x_clipped = np.clip(x, 0.0, None)
    s = x_clipped.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return x_clipped / s


def simplex_project_exact(x):
    """Exact projection onto {a >= 0, sum(a) = 1} via Duchi et al."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n + 1) - (cssv - 1.0) > 0)
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)


def smooth_max_and_grad(a, t):
    """Compute smooth max approximation of conv(a,a) and its gradient.
    Returns (smooth_value, gradient) where gradient is w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    # Numerically stable log-sum-exp
    conv_max = conv.max()
    conv_shifted = (conv - conv_max) / t
    exp_vals = np.exp(conv_shifted)
    log_sum = np.log(exp_vals.sum()) + conv_max / t
    smooth_val = t * log_sum
    # Weights
    w = exp_vals / exp_vals.sum()
    # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1
    # This is np.correlate(w, a, mode='valid')
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return smooth_val, grad, conv


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k] for k=0..2n-2."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)[k] <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def smoothed_gradient_descent(a_init, deadline, t_init=None, t_min=None,
                              nesterov=True, verbose=False, label=""):
    """Nesterov accelerated gradient descent on the simplex with log-sum-exp smoothing.
    Minimizes 2n*max(conv(a,a)) subject to a>=0, sum(a)=1."""
    a = simplex_project_exact(a_init.copy())
    n = len(a)
    S = 1.0  # We work with sum=1
    score = compute_score(a)
    best_a = a.copy()
    best_score = score

    conv = np.convolve(a, a)
    M = float(np.max(conv))
    if t_init is None:
        t_init = M / 5.0
    if t_min is None:
        t_min = M / 5000.0

    t = t_init
    lr = 0.01
    v = np.zeros(n)  # Nesterov momentum

    no_improve = 0
    it = 0
    patience = 2000

    while time.monotonic() < deadline:
        # Nesterov lookahead
        if nesterov:
            z = simplex_project_exact(a + v)
        else:
            z = a.copy()

        smooth_val, grad, conv = smooth_max_and_grad(z, t)
        M = float(np.max(conv))

        # Gradient step
        new_a = z - lr * grad
        new_a = simplex_project_exact(new_a)

        # Evaluate
        new_score = compute_score(new_a)
        it += 1

        if np.isfinite(new_score) and new_score < best_score:
            if nesterov:
                v = (a - new_a) / lr  # implicit momentum
                v = 0.9 * v  # dampen
            a = new_a
            best_score = new_score
            best_a = a.copy()
            no_improve = 0
            # Adapt learning rate
            if it % 500 == 0 and verbose:
                print(f"    {label} iter {it}: score={best_score:.6f} t={t:.4f} M={M:.6f}")
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.8
            if no_improve > patience:
                break

        # Anneal temperature
        if it % 50 == 0:
            t = max(t * 0.97, t_min)
            # Occasionally boost lr
            if it % 500 == 0:
                lr = max(lr * 0.95, 0.0001)

        # Check deadline more frequently for large n
        if it % 200 == 0 and time.monotonic() > deadline - 2:
            break

    return best_a, best_score

def subgradient_step(a, top_k=5):
    """One subgradient step: reduce the top-k largest convolution values."""
    n = len(a)
    conv = np.convolve(a, a)
    top_idx = np.argsort(conv)[-top_k:]
    # Weights proportional to excess
    mean_c = conv.mean()
    weights = np.maximum(conv[top_idx] - mean_c, 0)
    if weights.sum() < 1e-15:
        weights = np.ones(top_k)
    weights = weights / weights.sum()

    grad = np.zeros(n)
    for k, w in zip(top_idx, weights):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * w * a[k - j]

    return grad


def fast_subgradient_optimize(a_init, deadline, lr=0.005, momentum_decay=0.9,
                              top_k=5, label=""):
    """Fast subgradient descent with momentum."""
    a = simplex_project_exact(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    momentum = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < 50000:
        grad = subgradient_step(a, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        momentum = momentum_decay * momentum - lr * grad
        new_a = simplex_project_exact(a + momentum)
        new_score = compute_score(new_a)
        it += 1

        if np.isfinite(new_score) and new_score < best_score:
            a = new_a
            best_a = a.copy()
            best_score = new_score
            no_improve = 0
            if it % 1000 == 0 and verbose_flag:
                print(f"    {label} subgrad iter {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            lr *= 0.95
            momentum *= 0.5
            if lr < 1e-7 or no_improve > 300:
                break

    return best_a, best_score


def make_initializations(n, rng, parent=None):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Cosine taper
    cos_t = (1.0 + np.cos(np.pi * idx / max(n - 1, 1))) / 2.0
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(('cosine', cos_t / cos_t.sum()))

    # Raised cosine (smooth edges)
    rc = 0.5 * (1 + np.cos(2 * np.pi * idx / (n - 1)))
    rc = np.maximum(rc, 0.001)
    inits.append(('raised_cos', rc / rc.sum()))

    # Exponential decay from center
    center = (n - 1) / 2.0
    bell = np.exp(-((idx - center) ** 2) / (2 * (n / 3.0) ** 2))
    inits.append(('bell', bell / bell.sum()))

    # One-sided: heavy at start
    ramp = np.linspace(1.0, 0.05, n)
    inits.append(('ramp', ramp / ramp.sum()))

    # One-sided: heavy at end
    inits.append(('ramp_rev', np.flip(ramp) / ramp.sum()))

    # Bimodal
    bim = (np.exp(-((idx - 0.25 * n) ** 2) / (0.1 * n) ** 2) +
           np.exp(-((idx - 0.75 * n) ** 2) / (0.1 * n) ** 2))
    inits.append(('bimodal', bim / bim.sum()))

    # Power law
    pw = (idx + 1) ** (-0.5)
    inits.append(('power', pw / pw.sum()))

    # Random
    r = rng.uniform(0.1, 2.0, n)
    inits.append(('random', r / r.sum()))

    # Sine
    sine = np.abs(np.sin(np.pi * idx / (n - 1))) + 0.05
    inits.append(('sine', sine / sine.sum()))

    # Triangle (peak at center)
    tri = 1.0 - np.abs(idx - center) / center
    tri = np.maximum(tri, 0.01)
    inits.append(('triangle', tri / tri.sum()))

    # If parent available, interpolate
    if parent is not None and len(parent) != n:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        p_interp = np.interp(x_new, x_old, parent)
        p_interp = np.maximum(p_interp, 0)
        if p_interp.sum() > 0:
            inits.append(('parent_interp', p_interp / p_interp.sum()))
    elif parent is not None and len(parent) == n:
        p = np.maximum(parent, 0)
        if p.sum() > 0:
            inits.append(('parent', p / p.sum()))

    return inits

    # Phase B: Smoothed gradient descent (40% of time)
    smooth_dl = time.monotonic() + (deadline - time.monotonic()) * 0.40
    a, s = smoothed_gradient_descent(a, smooth_dl, label=label + "sgd")
    if s < best_score:
        best_score = s
        best_a = a.copy()

    # Phase C: LP refinement (25% of time)
    lp_dl = time.monotonic() + (deadline - time.monotonic()) * 0.25
    a, s = lp_refine(a, lp_dl, max_iters=15, step_init=0.08, label=label + "lp1")
    if s < best_score:
        best_score = s
        best_a = a.copy()

    # Phase D: Another round of smoothed GD with lower temperature
    a, s = smoothed_gradient_descent(a, time.monotonic() + (deadline - time.monotonic()) * 0.4,
                                      t_init=None, label=label + "sgd2")
    if s < best_score:
        best_score = s
        best_a = a.copy()

    # Phase E: Final LP + subgradient polish
    rem = deadline - time.monotonic()
    a, s = lp_refine(a, time.monotonic() + rem * 0.5, max_iters=10, step_init=0.04,
                     label=label + "lp2")
    if s < best_score:
        best_score = s
        best_a = a.copy()
    a, s = fast_subgradient_optimize(a, deadline, lr=0.002, top_k=7, label=label + "sg2")
    if s < best_score:
        best_score = s
        best_a = a.copy()

    return best_a, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry point: search for the best non-negative sequence minimizing the score."""
    global verbose_flag
    verbose_flag = True

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(42)

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals():
            parent = np.asarray(globals()['height_sequence_1'], dtype=np.float64).copy()
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.ndim != 1 or parent.size == 0 or not np.all(np.isfinite(parent)):
                parent = None
    except Exception as e:
        print(f"Could not load parent: {e}")

    best_score = float('inf')
    best_seq = None

    if parent is not None:
        ps = compute_score(parent)
        if np.isfinite(ps):
            best_score = ps
            best_seq = parent.copy()
            print(f"Parent (n={parent.size}): score={ps:.6f}")

    # Phase 1: Quick exploration
    print("\n=== Phase 1: Exploration ===")
    expl_dl = t0 + min(250.0, (deadline - t0) * 0.2)

    lengths = [512, 1024, 2048]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.insert(0, pn)
        lengths = sorted(set(lengths))

    candidates = []  # (score, seq, n, label)

    for n in lengths:
        if time.monotonic() > expl_dl - 20:
            break
        print(f"\n--- Exploring n={n} ---")
        inits = make_initializations(n, rng, parent)

        for label, f_init in inits:
            if time.monotonic() > expl_dl - 5:
                break
            f_init = simplex_project_exact(f_init)
            score = compute_score(f_init)

            if score < best_score:
                best_score = score
                best_seq = f_init.copy()
                print(f"  New best: {label} n={n} -> {score:.6f}")

            # Quick optimization: subgradient for a few seconds
            q_dl = min(expl_dl, time.monotonic() + 8.0)
            f_opt, s_opt = fast_subgradient_optimize(f_init, q_dl, lr=0.005, top_k=5,
                                                      label=f"  {label}[{n}]")
            if s_opt < score:
                score = s_opt
                f_init = f_opt
            if score < best_score:
                best_score = score
                best_seq = f_init.copy()

            candidates.append((score, f_init.copy(), n, label))

        print(f"  Best for n={n}: min({min(sc for sc, _, _, _ in candidates if _ == n):.6f})"
              if any(_ == n for _, _, _, _ in candidates) else "")

    # Sort and select top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates after exploration:")
    for sc, seq, n, lbl in candidates[:8]:
        print(f"  n={n:4d} {lbl:15s}: {sc:.6f}")
    print(f"Best overall: {best_score:.6f}")

    # Phase 2: Focused optimization
    print("\n=== Phase 2: Focused Optimization ===")
    focus_dl = deadline - 200.0

    # Select top candidates (diverse in length)
    seen_n = set()
    top_cands = []
    for sc, seq, n, lbl in candidates:
        if n not in seen_n and sc < best_score + 0.05:
            top_cands.append((sc, seq, n, lbl))
            seen_n.add(n)
        if len(top_cands) >= 3:
            break

    # Always include the current best
    if best_seq is not None:
        bn = len(best_seq)
        if bn not in seen_n:
            top_cands.append((best_score, best_seq, bn, 'best'))

    # Also always try n=1024 from parent if available
    if parent is not None and len(parent) == 1024:
        if 1024 not in seen_n:
            parent_norm = simplex_project_exact(parent)
            top_cands.append((compute_score(parent_norm), parent_norm, 1024, 'parent_direct'))

    # Allocate time proportionally
    n_focus = len(top_cands)
    time_per_cand = (focus_dl - time.monotonic()) / max(n_focus, 1)

    for i, (sc, f_init, n, lbl) in enumerate(top_cands):
        if time.monotonic() > focus_dl - 30:
            break
        # Last candidate gets all remaining time up to focus_dl
        cand_dl = min(focus_dl - 10, time.monotonic() + time_per_cand * 1.2)
        print(f"\n--- Focused: n={n} {lbl} (start={sc:.6f}, time={cand_dl - time.monotonic():.0f}s) ---")

        f_opt, s_opt = mixed_optimize(f_init, cand_dl, label=f"[{n}:{lbl}]")
        if s_opt < best_score:
            best_score = s_opt
            best_seq = f_opt.copy()
            print(f"  Improved to: {best_score:.6f}")
        else:
            print(f"  No improvement (best remains {best_score:.6f})")

    # Phase 3: Final polish
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_best = len(best_seq)
        print(f"Polishing n={n_best}...")

        # Multiple polish passes with different methods
        passes = [
            ("polish_sgd", lambda a, dl: smoothed_gradient_descent(a, dl, t_init=None, label="poly_sgd")),
            ("polish_lp", lambda a, dl: lp_refine(a, dl, max_iters=10, step_init=0.03, label="poly_lp")),
            ("polish_sg", lambda a, dl: fast_subgradient_optimize(a, dl, lr=0.001, top_k=7, label="poly_sg")),
        ]

        # Try to also improve at nearby lengths
        alt_length = n_best
        if n_best > 512:
            alt_length = n_best // 2
        elif n_best < 2048:
            alt_length = n_best * 2

        remaining = deadline - time.monotonic() - 20
        time_per_pass = remaining / (len(passes) + 1)

        for pname, pfunc in passes:
            if time.monotonic() > deadline - 20:
                break
            p_dl = time.monotonic() + time_per_pass
            try:
                f_pol, s_pol = pfunc(best_seq.copy(), p_dl)
                if s_pol < best_score:
                    best_score = s_pol
                    best_seq = f_pol.copy()
                    print(f"  {pname}: improved to {best_score:.6f}")
            except Exception as e:
                print(f"  {pname} failed: {e}")

        # Try alt length
        if time.monotonic() < deadline - 60:
            alt_dl = deadline - 30
            print(f"\n  Trying alt length n={alt_length}...")
            alt_inits = make_initializations(alt_length, rng, best_seq)
            for albl, af in alt_inits[:4]:
                if time.monotonic() > deadline - 30:
                    break
                af = simplex_project_exact(af)
                a_opt, s_opt = smoothed_gradient_descent(af, deadline - 10, label=f"alt[{alt_length}:{albl}]")
                if s_opt < best_score:
                    best_score = s_opt
                    best_seq = a_opt.copy()
                    print(f"  Alt n={alt_length} {albl}: improved to {best_score:.6f}")

    # Final verification and scaling
    if best_seq is None:
        best_seq = np.ones(512) / 512.0

    # The score is scale-invariant, so any positive scaling works.
    # Use sum=1 for cleanliness (avoids the 1000 cap issue).
    final_seq = best_seq.copy()
    final_sum = final_seq.sum()
    if final_sum > 0:
        # Scale so values are in a safe range [0, 1000]
        # sum=1 gives max element <= 1, well within [0, 1000]
        final_seq = final_seq / final_sum

    # Verify
    final_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in final_seq])
    if not np.isfinite(final_score):
        print("WARNING: Verification failed, using fallback")
        final_seq = np.ones(512) / 512.0
        final_score = evaluate_sequence(final_seq.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(final_seq)}, score={final_score:.6f}, time={elapsed:.1f}s")

    return [float(max(0.0, min(1000.0, x))) for x in final_seq]
