import time

def fft_convolve(f):
    """FFT-based linear convolution."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    return np.fft.irfft(F * F, fft_size)[:size]

def soft_max_and_grad(conv, beta):
    """Compute soft-max value and probability vector for entropic max."""
    cmax = conv.max()
    shifted = beta * (conv - cmax)
    exp_c = np.exp(shifted)
    total = exp_c.sum()
    p = exp_c / total
    value = cmax + np.log(total) / beta
    return p, value

def grad_of_soft_max(f, p):
    """Gradient of soft-max of conv(f,f) w.r.t. f."""
    return 2.0 * np.correlate(p, f, mode='valid')

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    idx = np.arange(1, n + 1)

    rho = np.max(np.where(positive)[0]) if np.any(positive) else n - 1
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
