Let me analyze the problem carefully:

We want to minimize $J(a) = \frac{2n \cdot \max_k (a*a)_k}{(\sum_j a_j)^2}$ where $a$ is a non-negative sequence of length $n$.

This is essentially minimizing the ratio of the peak of the autocorrelation to the square of the sum. If we normalize so that $\sum a_j = 1$, we want to minimize $2n \cdot \max_k (a*a)_k$.

The key insight is that for a "flat" autocorrelation, we'd want $(a*a)_k$ to be as uniform as possible. The minimum possible value of $\max_k (a*a)_k$ given $\sum a_j = 1$ is related to the "ideal" flat autocorrelation.

For a sequence with $\sum a_j = S$, the sum of all autocorrelation values is $(\sum a_j)^2 = S^2$. There are $2n-1$ autocorrelation values. So the average autocorrelation value is $S^2/(2n-1)$. The minimum possible maximum is at least $S^2/(2n-1)$, achieved when the autocorrelation is perfectly flat.

So the lower bound on $J$ is approximately $2n \cdot S^2/(2n-1) / S^2 = 2n/(2n-1) \approx 1$ for large $n$.

But achieving a perfectly flat autocorrelation is essentially the " Barker sequence" or "perfect autocorrelation" problem, which is known to be very hard and generally impossible for most lengths.

"""

import numpy as np
from scipy.optimize import linprog
import time
from numpy.lib.stride_tricks import sliding_window_view


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A where A @ x = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jl:k - jh + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def compute_score(a):
    """Compute J(a) directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_smoothed_obj_and_grad(a, beta):
    """
    Compute LSE-smoothed objective and gradient for normalized a.
    Returns (objective_value, gradient).
    Objective: (1/beta) * log(sum_k exp(beta * b_k)) where b = a*a
    This approximates max(b) as beta -> inf.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    log_sum = b_max + np.log(exp_b.sum()) / beta
    p = exp_b / exp_b.sum()
    
    # Gradient: g_i = 2 * sum_k p_k * a_{k-i}
    # Using sliding window: P[i,j] = p[i+j], g = 2 * (P @ a)
    P = sliding_window_view(p, n)
    grad = 2.0 * (P @ a)
    
    return log_sum, grad

def lp_iteration(a, max_iters, deadline, base_t=0.5):
    """LP best-response with adaptive step size."""
    n = len(a)
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size
        t = base_t * (1.0 - 0.5 * it / max_iters)
        t = max(0.02, min(0.8, t))

        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            a = a_new
        else:
            t2 = t * 0.3
            a_new2 = (1.0 - t2) * a + t2 * g
            a_new2 = np.clip(a_new2, 0.0, None)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                score2 = compute_score(a_new2)
                if score2 < best_score:
                    best_score = score2
                    best_local = a_new2.copy()
                a = a_new2
            if t2 < 0.01:
                break

    """Generate diverse initializations."""
    inits = []
    t_arr = np.linspace(0, 1, n)

    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t_arr, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())

    inits.append(np.ones(n) / n)
    c = 1.0 + 0.5 * np.cos(2 * np.pi * t_arr)
    inits.append(np.maximum(c, 0.01) / np.maximum(c, 0.01).sum())
    c = 1.0 + 0.3 * np.cos(4 * np.pi * t_arr)
    inits.append(np.maximum(c, 0.01) / np.maximum(c, 0.01).sum())
    c = 1.0 + 0.2 * np.cos(6 * np.pi * t_arr)
