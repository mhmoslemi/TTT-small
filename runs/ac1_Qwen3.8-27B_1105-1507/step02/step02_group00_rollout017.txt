Phase 1: Fast mirror descent warm start
Phase 2: LP polishing (exact inner problem)

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(n-1, k)
        A[k, j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
    return A

def solve_lp_g(a, M_relax=1.0):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv)) * M_relax
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_polish(a, max_iter=200, verbose=False):
    """LP-based polishing. Start from a good initial point."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a/s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iter):
        g = solve_lp_g(a)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g/g_sum
        
        # Check g and blend
        obj_g = compute_obj(g)
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()
        
        # Blend update
        t = 0.5
        a_new = (1-t)*a + t*g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new/s
        
        # Check blended
        obj_blend = compute_obj(a)
        if obj_blend < best_obj:
            best_obj = obj_blend
            best_a = a.copy()
        
        if verbose and it % 50 == 0:
            print(f"    LP it={it}: obj={best_obj:.6f}")
    
    return best_a, best_obj

def md_fast(a_init, n_steps=2000, verbose=False):
    """Fast mirror descent with log-sum-exp."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)/n
    else:
        a = a/s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    tau_min = 30.0
    tau_max = 300.0
    
    for step in range(n_steps):
        frac = step / max(1, n_steps-1)
        tau = tau_min * (tau_max/tau_min)**frac
        
        h = np.convolve(a, a)
        h_max = np.max(h)
        exp_h = np.exp(tau*(h - h_max))
        sum_exp = np.sum(exp_h)
        w = exp_h / sum_exp
        
        # Gradient
        grad = np.zeros(n)
        for m in range(n):
            end = min(n, 2*n-1-m)
            if end > 0:
                grad[m] = 2.0 * np.dot(w[m:m+end], a[:end])
        
        eta = 0.4 / (1.0 + 0.5*frac)
        
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / np.sum(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def coord_refine(a, n_rounds=3):
    """Coordinate descent refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for rnd in range(n_rounds):
        for j in range(n):
            old = a[j]
            best_x = old
            best_local = best_obj
            for mult in [0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
                a[j] = old * mult
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = a[j]
            a[j] = best_x
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

    choice = int(rng.integers(0, 5))

    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a/s

The mirror descent update is:
a_new[i] ∝ a[i] * exp(-eta * grad[i])

The gradient grad[m] = 2 * sum_k w[k] * a[k-m]

The magnitude of grad is roughly 2 * sum(a) = 2 (since sum(a)=1 and sum_k w[k]=1, so sum_m grad[m] = 2 * sum_k w[k] * sum_m a[k-m]... hmm, that's not quite right).

Actually, sum_m grad[m] = 2 * sum_m sum_k w[k] * a[k-m] = 2 * sum_k w[k] * sum_m a[k-m] = 2 * sum_k w[k] * sum_j a[j] = 2 * 1 * 1 = 2.

So the average gradient is 2/n. The typical magnitude of grad[m] is O(1/n) (since it's a weighted average of a's values, and a has sum 1 over n elements, so typical a[i] ~ 1/n).

For the multiplicative update to be meaningful, we need eta * grad[i] to be O(1) for the coordinates that should change the most. So eta should be O(n) if grad is O(1/n). But that seems too large.

Let me think differently. The log of the update is:
log(a_new[i]/a[i]) = -eta * grad[i] - log(Z)

where Z = sum_j a[j] * exp(-eta * grad[j]).

For the update to be "one step" (not too aggressive), we want |log(a_new[i]/a[i])| < 1 for most i. This means eta * |grad[i]| < 1 roughly.

Since grad[i] ~ 2 * (weighted average of a) ~ 2/n (for uniform a), we need eta ~ n/2.

But for a peaked a, grad can be much larger at some coordinates. So we need eta to be adaptive.

A good rule: eta = 1 / (max(grad) - min(grad) + eps) = 1 / (range(grad) + eps).

Or even simpler: eta = c / max(grad) for some constant c < 1.

Let me use: eta = 0.5 / max(grad)

This ensures that the largest coordinate gets multiplied by exp(-0.5) ≈ 0.6, which is a reasonable update.
