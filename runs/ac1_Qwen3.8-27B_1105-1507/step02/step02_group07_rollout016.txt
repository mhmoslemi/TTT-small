def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)[k] <= M for all k."""
    n = len(f)
    # Build constraint matrix A where (A@g)[k] = conv(f,g)[k]
    # A is (2n-1) x n, A[k,j] = f[k-j] if valid
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k, n-1)+1):
            A[k, j] = f[k-j]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 10})
        if result.success:
            return result.x
    except:
        pass
    return None

def lp_step(f, deadline, max_iters=100):
    """LP-based iteration."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    best_f = f.copy()
    best_score = compute_score_direct(f)
    step_size = 0.1
    no_improve = 0
    
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        
        g = lp_direction(f, M)
        it += 1
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            continue
        
        g = g / S  # same sum as f
        
        # Line search
        best_t = 0
        best_s = best_score
        for t in [step_size, step_size*1.5, step_size*0.5, step_size*2, step_size*0.75]:
            new_f = (1-t)*f + t*g
            new_f = np.maximum(new_f, 0)
            ns = np.sum(new_f)
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = compute_score_direct(new_f)
            if sc < best_s:
                best_s = sc
                best_t = t
        
        if best_t > 0 and best_s < best_score:
            f = (1-best_t)*f + best_t*g
            f = np.maximum(f, 0)
            f = f / np.sum(f)
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.01 or no_improve > 30:
                break
    
    return best_f, best_score

def asymmetric_window(n, skew=0.3):
    """Generate an asymmetric window (wider on one side)."""
    idx = np.arange(n, dtype=np.float64)
    # Asymmetric Gaussian
    center = n/2 + skew * n/4
    sigma = n/4
    w = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    return w

def raised_cosine_asymmetric(n, alpha=0.3, skew=0.2):
    """Asymmetric raised cosine window."""
    idx = np.arange(n, dtype=np.float64)
    center = n/2 + skew * n/4
    half_width = n/2 * (1 + skew)
    if center - half_width < 0:
        # Shift
        w = np.exp(-((idx - center) / half_width) ** 2)
    else:
        w = np.ones(n)
        left = max(0, int(center - half_width))
        right = min(n, int(center + half_width))
        w[:left] = 0
        if left > 0:
            taper = (right - left) // 4
            for i in range(taper):
                w[left + i] = 0.5 * (1 + np.cos(np.pi * (1 - i/taper)))
                w[right - 1 - i] = 0.5 * (1 + np.cos(np.pi * (1 - i/taper)))
    w = np.maximum(w, 0.01)
    return w

def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2*i/(2*m) - 1)))
            w[n-1-i] = w[i]
    return w

"""
Multi-phase optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses softmax-reparameterized L-BFGS-B with multi-stage beta,
LP-based refinement, asymmetric initializations, and perturbation restarts.
Key insight: asymmetric sequences can achieve flatter autocorrelation.

import numpy as np
import time
from scipy.optimize import minimize, linprog


def compute_score(f):
    """Compute score = 2n * max(conv(f,f)) / sum(f)^2."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    conv_f = np.convolve(f, f)
    max_conv = float(np.max(conv_f))
    sum_f = float(np.sum(f))
    if sum_f < 0.01:
        return np.inf
    return 2.0 * n * max_conv / (sum_f ** 2)


def softmax_parameterize(x):
    """Map x in R^n to f on the probability simplex via softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def softmax_objective(f, beta):
    """Smooth approximation of max(conv(f,f))."""
    conv_f = np.convolve(f, f)
    m = np.max(conv_f)
    shifted = beta * (conv_f - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_conv = np.exp(shifted)
    total = np.sum(exp_conv)
    if total < 1e-300:
        return m
    return m + np.log(total) / beta


def softmax_gradient_f(f, beta):
    """Gradient of soft_max(conv(f,f)) w.r.t. f."""
    conv_f = np.convolve(f, f)
    m = np.max(conv_f)
    shifted = beta * (conv_f - m)
    shifted = np.clip(shifted, -700, 700)
    exp_conv = np.exp(shifted)
    total = np.sum(exp_conv)
    if total < 1e-300:
        return np.zeros_like(f)
    p = exp_conv / total
    n = len(f)
    # d(soft_max)/d(f[j]) = 2 * sum_l f[l] * p[l+j]
    grad = 2.0 * np.correlate(p, f, mode='valid')
    return grad

    """Objective in reparameterized space."""
    f = softmax_parameterize(x)
    return softmax_objective(f, beta)


def gradient_reparam(x, beta):
    """Gradient in reparameterized space (chain rule through softmax)."""
    f = softmax_parameterize(x)
    g_f = softmax_gradient_f(f, beta)
    # dL/dx_i = g_f[i]*f[i] - f[i]*sum_j(g_f[j]*f[j])
    dot = np.dot(g_f, f)
    grad_x = f * (g_f - dot)
    return grad_x


def lbfgsb_optimize(f_init, deadline, beta_schedule=None):
    """Optimize using L-BFGS-B with softmax reparameterization."""
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, compute_score(f)
    f = f / S

    best_f = f.copy()
    best_score = compute_score(f)

    if beta_schedule is None:
        beta_schedule = [5, 15, 40, 100, 300, 800]

    # Initialize x such that softmax(x) = f
    # x = log(f) works (up to a constant)
    f_safe = np.maximum(f, 1e-30)
    x = np.log(f_safe) - np.mean(np.log(f_safe))

    bounds = [(-50, 50)] * n

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break

        # Run L-BFGS-B
        try:
            def obj(x_curr):
                return objective_reparam(x_curr, beta)

            def grad(x_curr):
                return gradient_reparam(x_curr, beta)

            result = minimize(
                obj, x, jac=grad, method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 300, 'gtol': 1e-10, 'maxls': 100,
                         'ftol': 1e-12}
            )

            if result.fun < objective_reparam(x, beta) + 1e-10:
                f_new = softmax_parameterize(result.x)
                sc = compute_score(f_new)
                if sc < best_score:
                    best_score = sc
                    best_f = f_new.copy()
                x = result.x
                f = f_new
        except Exception as e:
            print(f"    L-BFGS-B error at beta={beta}: {e}")

    return best_f, best_score


def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)[k] <= M for all k."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = f[k - j]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': 5})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_step(f, deadline, max_iters=80, step_init=0.1):
    """LP-based descent iterations."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    best_f = f.copy()
    best_score = compute_score(f)
    step_size = step_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M < 1e-15:
            break

        g = lp_direction(f, M)
        it += 1

        if g is None:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            continue

        g = g / S

        best_t = 0.0
        best_s = best_score
        for t in [step_size, step_size * 0.7, step_size * 1.3, step_size * 0.4, step_size * 1.8]:
            new_f = (1.0 - t) * f + t * g
            new_f = np.maximum(new_f, 0.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = compute_score(new_f)
            if sc < best_s:
                best_s = sc
                best_t = t

        if best_t > 0 and best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0.0)
            f = f / float(np.sum(f))
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 30 == 0:
                print(f"    LP iter {it}: score={best_score:.6f} (t={best_t:.3f})")
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.01 or no_improve > 40:
                break

    return best_f, best_score


def tukey_window(n, r=0.3):
    """Generate a Tukey window."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def asymmetric_gaussian(n, skew=0.3):
    """Asymmetric Gaussian window."""
    idx = np.arange(n, dtype=np.float64)
    center = n / 2.0 + skew * n / 4.0
    sigma = n / 4.0
    w = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    return np.maximum(w, 1e-6)


def asymmetric_power(n, p=1.5, skew=0.2):
    """Asymmetric power-law window."""
    idx = np.arange(n, dtype=np.float64)
    center = n * 0.5 + skew * n * 0.2
    half = n * 0.5
    w = np.maximum(1.0 - np.abs(idx - center) / half, 0.0) ** p
    return np.maximum(w, 0.01)


def two_hump(n, separation=0.3, width=0.25):
    """Two-hump Gaussian window (asymmetric)."""
    idx = np.arange(n, dtype=np.float64)
    c1 = n * (0.5 - separation / 2)
    c2 = n * (0.5 + separation / 2)
    w1 = 0.6 * np.exp(-0.5 * ((idx - c1) / (n * width)) ** 2)
    w2 = 0.4 * np.exp(-0.5 * ((idx - c2) / (n * width)) ** 2)
    w = w1 + w2
    return np.maximum(w, 0.01)


def full_optimize(f_init, deadline):
    """Run full multi-phase optimization."""
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, compute_score(f)
    f = f / S

    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    remaining = deadline - t0
    if remaining < 10:
        return best_f, best_score

    # Phase 1: L-BFGS-B coarse (lower beta)
    p1_deadline = t0 + remaining * 0.35
    if time.monotonic() < p1_deadline:
        f1, s1 = lbfgsb_optimize(f, p1_deadline, beta_schedule=[3, 10, 30, 80])
        if s1 < best_score:
            best_score = s1
            best_f = f1

    # Phase 2: LP refinement
    p2_deadline = t0 + remaining * 0.6
    if time.monotonic() < p2_deadline:
        f2, s2 = lp_step(best_f, p2_deadline, max_iters=60)
        if s2 < best_score:
            best_score = s2
            best_f = f2

    # Phase 3: L-BFGS-B fine (high beta)
    p3_deadline = t0 + remaining * 0.85
    if time.monotonic() < p3_deadline:
        f3, s3 = lbfgsb_optimize(best_f, p3_deadline, beta_schedule=[150, 400, 1000])
        if s3 < best_score:
            best_score = s3
            best_f = f3

    # Phase 4: Final LP polish
    if time.monotonic() < deadline - 3:
        f4, s4 = lp_step(best_f, deadline - 3, max_iters=30, step_init=0.05)
        if s4 < best_score:
            best_score = s4
            best_f = f4

    return best_f, best_score


def perturb(f, rng, strength=0.08):
    """Add multiplicative noise and renormalize."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.0, 3.0)
    new_f = np.maximum(new_f, 0.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(777)

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Focus on lengths: 512 (proven), 1024 (more DoF), 768 (intermediate)
    lengths = [512, 1024, 768]
    time_fracs = {512: 0.35, 1024: 0.30, 768: 0.15}
    total_budget = deadline - t0

    for n in lengths:
        if time.monotonic() > deadline - 80:
            break

        n_budget = total_budget * time_fracs.get(n, 0.1)
        n_deadline = time.monotonic() + n_budget
        print(f"\n=== n={n} (budget {n_budget:.0f}s) ===")

        # Generate diverse initializations
        inits = []

        # Symmetric windows
        inits.append(('tukey_0.25', tukey_window(n, 0.25)))
        inits.append(('tukey_0.4', tukey_window(n, 0.4)))
        inits.append(('tukey_0.6', tukey_window(n, 0.6)))

        # Asymmetric windows (key for breaking symmetry!)
        for skew in [-0.3, -0.2, 0.2, 0.3]:
            inits.append((f'asym_gauss_{skew}', asymmetric_gaussian(n, skew)))

        # Asymmetric power-law
        for skew in [-0.2, 0.2]:
            for p in [1.0, 2.0]:
                inits.append((f'asym_pow_{p}_{skew}', asymmetric_power(n, p, skew)))

        # Two-hump
        inits.append(('two_hump', two_hump(n, 0.3, 0.25)))
        inits.append(('two_hump_wide', two_hump(n, 0.4, 0.35)))

        # Smooth random
        for seed in [1, 2]:
            r_rng = np.random.default_rng(seed * 1000 + n)
            x = r_rng.standard_normal(n)
            kernel = np.exp(-0.5 * (np.arange(-15, 16) / 4.0) ** 2)
            smooth_x = np.convolve(x, kernel, mode='same')
            smooth_x = np.abs(smooth_x) + 0.05
            inits.append((f'rand_smooth_{seed}', smooth_x))

        # Parent (interpolated if needed)
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.insert(0, ('parent', best_sequence.copy()))
            else:
                parent_interp = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, len(best_sequence)),
                    best_sequence)
                parent_interp = np.maximum(parent_interp, 0)
                inits.insert(0, ('parent_interp', parent_interp))

        # Score and sort initializations
        scored_inits = []
        for label, f_init in inits:
            f_init = np.maximum(f_init.astype(np.float64), 0.0)
            S_init = float(np.sum(f_init))
            if S_init < 0.01:
                continue
            f_init = f_init / S_init
            sc = compute_score(f_init)
            scored_inits.append((sc, label, f_init))

        scored_inits.sort(key=lambda x: x[0])
        top_labels = [(l, f'{s:.4f}') for s, l, _ in scored_inits[:6]]
        print(f"  Top inits: {top_labels}")

        # Update best from inits
        for sc, label, f_init in scored_inits:
            if sc < best_score:
                best_score = sc
                best_sequence = f_init.copy()

        # Optimize top initializations
        n_to_opt = min(4, len(scored_inits))
        per_init_time = n_budget / max(n_to_opt, 1) * 0.75

        for i in range(n_to_opt):
            if time.monotonic() > n_deadline - 15:
                break
            sc, label, f_init = scored_inits[i]
            init_deadline = min(n_deadline - 5, time.monotonic() + per_init_time)
            if time.monotonic() < init_deadline - 10:
                try:
                    bf, bs = full_optimize(f_init, init_deadline)
                    if bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        print(f"  [n={n} {label}] -> {bs:.6f}")
                except Exception as e:
                    print(f"  [n={n} {label}] error: {e}")

    # Perturbation restarts on best found
    if best_sequence is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Perturbation restarts (best: {best_score:.6f}) ===")
        n_best = len(best_sequence)
        restart_deadline = deadline - 15
        n_restarts = 4
        per_restart = (restart_deadline - time.monotonic()) / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > restart_deadline - 5:
                break
            strength = 0.05 + 0.04 * r
            f_pert = perturb(best_sequence, rng, strength)
            r_deadline = time.monotonic() + per_restart * 0.8
            try:
                bf, bs = full_optimize(f_pert, r_deadline)
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strength:.3f}): {bs:.6f}")
                else:
                    print(f"  Restart {r}: no improvement ({bs:.6f})")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # Final intensive polish on the absolute best
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final intensive polish (n={len(best_sequence)}, score={best_score:.6f}) ===")
        polish_deadline = deadline - 5

        # High-beta L-BFGS-B
        if time.monotonic() < polish_deadline - 20:
            try:
                bf, bs = lbfgsb_optimize(best_sequence, polish_deadline - 20,
                                         beta_schedule=[200, 500, 1200, 3000])
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf
                    print(f"  After high-beta L-BFGS-B: {best_score:.6f}")
            except Exception as e:
                print(f"  Polish L-BFGS-B error: {e}")

        # LP polish
        if time.monotonic() < polish_deadline - 5:
            try:
                bf, bs = lp_step(best_sequence, polish_deadline - 5,
                                 max_iters=40, step_init=0.03)
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf
                    print(f"  After LP polish: {best_score:.6f}")
            except Exception as e:
                print(f"  Polish LP error: {e}")

        # One more L-BFGS-B after LP
        if time.monotonic() < polish_deadline - 3:
            try:
                bf, bs = lbfgsb_optimize(best_sequence, polish_deadline - 3,
                                         beta_schedule=[2000, 5000])
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf
                    print(f"  Final L-BFGS-B: {best_score:.6f}")
            except Exception as e:
                print(f"  Final L-BFGS-B error: {e}")

    if best_sequence is None:
        best_sequence = tukey_window(512, 0.3)
        best_score = compute_score(best_sequence)

    # Validation
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = compute_score(best_sequence)

    # Final validation with evaluate_sequence
    final_score = evaluate_sequence(best_sequence.tolist())
    if not np.isfinite(final_score):
        best_sequence = tukey_window(512, 0.3)
        final_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
