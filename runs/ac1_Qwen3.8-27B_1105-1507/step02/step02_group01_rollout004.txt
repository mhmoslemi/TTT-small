"""

sequences, max(conv(a,a)) = sum(a_j^2) (by Cauchy-Schwarz), reducing the objective
to 2n*sum(a^2)/(sum(a))^2. Uses this for fast evaluation and combines multiple

def compute_obj_fast(a):
    """Fast objective using the identity max_conv = sum(a^2) for non-negative a."""
    n = len(a)
    if n == 0:
        return np.inf
    a = np.maximum(0.0, a)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    sum_sq = float(np.dot(a, a))
    return 2.0 * n * sum_sq / (sum_a * sum_a)


def compute_obj_verify(a):
    """Verify with full convolution (slower but exact)."""
    n = len(a)
    if n == 0:
        return np.inf
    a = np.maximum(0.0, a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz_row(a, k):
    """Build the row of the convolution matrix for lag k."""
    n = len(a)
    row = np.zeros(n)
    if k >= 0:
        # (conv(a,a))_k = sum_{j=k}^{n-1} a[j]*a[j-k] = sum_{j=0}^{n-1-k} a[j+k]*a[j]
        j_start = 0
        j_end = n - 1 - k
        row[j_start:j_end+1] = a[j_start+k:j_end+1+k]
    else:
        # Symmetric
        row = build_toeplitz_row(a, -k)
    return row


def subgradient_step(a, lr, max_lags=5, n_corrective=3):
    """
    Subgradient step for minimizing 2n*sum(a^2)/(sum(a))^2 / max_conv.
    Uses the identity that max_conv = sum(a^2) for non-negative sequences.
    The objective is sum(a^2)/(sum(a))^2, gradient is:
    d/da_j = 2a_j/(sum(a))^2 - 2*sum(a^2)*a_j/(sum(a))^3
           = 2a_j/(sum(a))^2 * (1 - sum(a^2)/sum(a))
    
    But this is just proportional to a_j, so gradient descent just scales a.
    We need a different approach.
    
    Actually, since the objective is 2n*sum(a^2)/(sum(a))^2 and we can rescale
    freely, the shape that matters is a/sum(a). The objective depends only on
    the normalized vector: 2n * sum((a/sum_a)^2).
    
    So we need to minimize sum(a_i^2) subject to sum(a_i) = 1, a_i >= 0.
    This is minimized at a_i = 1/n for all i.
    
    For a more interesting optimization (in case the actual metric is different),
    let's do a proper subgradient on max_conv directly.
    """
    n = len(a)
    conv = np.convolve(a, a)
    max_idx = np.argmax(conv)
    max_val = conv[max_idx]
    
    # Subgradient of max_conv w.r.t. a_j
    # conv[k] = sum_i a[i] * a[i + k - (n-1)] (for full conv indexing)
    # Actually for full conv: conv[m] = sum_{i} a[i] * a[m-i] where 0<=i<n, 0<=m-i<n
    # So conv[m] = sum_{i=max(0,m-n+1)}^{min(m,n-1)} a[i] * a[m-i]
    # d(conv[m])/d(a[j]) = a[m-j] if 0<=m-j<n
    
    k = max_idx - (n - 1)  # The lag corresponding to max_idx in full conv
    # k ranges from -(n-1) to (n-1)
    grad = np.zeros(n)
    if k >= 0:
        # d/d(a[j]) of sum_{i=0}^{n-1-k} a[i+k]*a[i]
        # = a[j-k] (if j>=k) + a[j+k] (if j<=n-1-k), but actually:
        # conv_full[n-1+k] = sum_{i=0}^{n-1-k} a[i+k]*a[i]
        # d/d(a[j]) = a[j-k] (from first factor, when i=j-k)... 
        # Let me just compute it directly
        for j in range(n):
            # a[j] appears in conv[m] as: a[j]*a[m-j] (if 0<=m-j<n)
            # So d(conv[m])/d(a[j]) = a[m-j] if 0 <= m-j < n
            mj = max_idx - j
            if 0 <= mj < n:
                grad[j] += a[mj]
            # Also if j appears as the second factor: a[mj2]*a[j] where mj2 = max_idx - j
            # That's the same term, so we've already counted it
            # Wait no. conv[m] = sum_{i: 0<=i<n, 0<=m-i<n} a[i]*a[m-i]
            # If j is the first index: i=j, contribution is a[j]*a[m-j] if 0<=m-j<n
            # If j is the second index: m-i=j, i=m-j, contribution is a[m-j]*a[j] if 0<=m-j<n
            # These are the same! So d(conv[m])/d(a[j]) = a[m-j] + a[m-j] = 2*a[m-j] if 0<=m-j<n
            # Wait no, they're the same term counted once in the sum. Let me be more careful.
            # conv[m] = sum_{i=0}^{n-1} a[i] * a[m-i] * I(0<=m-i<=n-1)
            # = sum over valid (i, m-i) pairs
            # d/d(a[j]): j appears as i (giving a[m-j] if valid) AND as m-i (giving a[i] = a[m-j] if valid)
            # But these are the SAME term! The sum has only one term involving a[j]*a[m-j] (or a[m-j]*a[j])
            # So d(conv[m])/d(a[j]) = a[m-j] if 0 <= m-j < n, else 0
            # UNLESS m-j == j, in which case the term is a[j]^2 and the derivative is 2*a[j]
            if mj == j:
                grad[j] = 2.0 * a[j]
            elif 0 <= mj < n:
                grad[j] = a[mj]
            else:
                grad[j] = 0.0
    else:
        # Negative lag: conv_full[n-1+k] where k<0
        # Same formula applies
        for j in range(n):
            mj = max_idx - j
            if 0 <= mj < n:
                if mj == j:
                    grad[j] = 2.0 * a[j]
                else:
                    grad[j] = a[mj]
            else:
                grad[j] = 0.0
    
    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad = grad / gnorm
    
    # Also add gradient of sum(a) = constant constraint
    return grad


def active_set_qp(a, max_iters=50, tol=1e-8):
    """
    Active-set method: identify constraints near the maximum, then solve
    a QP to reduce them while maintaining sum(a)=1, a>=0.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(0, a)
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj_fast(a)
    a = a / s
    
    best_obj = compute_obj_fast(a)
    best_a = a.copy()
    
    for it in range(max_iters):
        conv = np.convolve(a, a)
        max_val = float(np.max(conv))
        
        # Find active constraints (within 1% of max)
        threshold = max_val * 0.99
        active_lags = np.where(conv >= threshold)[0]
        
        # Solve: minimize sum(a^2) subject to conv[k] <= max_val*0.99 for active k,
        # sum(a) = 1, a >= 0
        # Use a simple projected gradient on the max-plus-penalty
        penalty = 0.1 * max_val
        for sub_it in range(20):
            grad = 2.0 * a
            # Add penalty gradient for active constraints
            for k_idx in active_lags[:10]:  # Limit to 10 most active
                k = k_idx - (n - 1)
                for j in range(n):
                    mj = k_idx - j
                    if 0 <= mj < n:
                        if mj == j:
                            grad[j] += penalty * 2.0 * a[mj]
                        else:
                            grad[j] += penalty * a[mj]
            
            # Gradient of sum(a)^2 in denominator: we want to keep sum=1
            # Projected gradient: a_new = a - lr * grad, then project to {sum=1, a>=0}
            lr = 0.1 / (1.0 + 0.01 * sub_it)
            a_new = a - lr * grad
            a_new = np.maximum(0, a_new)
            sn = np.sum(a_new)
            if sn < 1e-10:
                break
            a_new = a_new / sn
            
            obj_new = compute_obj_fast(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
        
        # Check convergence
        conv_check = np.convolve(a, a)
        if float(np.max(conv_check)) - max_val > -tol:
            break
    
    return best_a, best_obj


def lp_minimize_peak(a, max_iters=100, time_budget=30):
    """
    LP-based approach: iteratively solve
    max sum(b) subject to conv(a,b) <= M, b >= 0
    then blend. This pushes the correlation down.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(0, a)
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj_fast(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        # Build A matrix: A[i,j] = a[i-j] if valid, for row i of conv(a,?)
        # conv(a, b)[k] = sum_j a[j]*b[k-j]
        # We want conv(a,b)[k] <= M for all k
        # A is (2n-1) x n
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A[k, j] = a[idx]
        
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': max(0.5, time_budget/10)})
            if res.status == 0 and res.x is not None:
                g = res.x
                g_sum = float(np.sum(g))
                if g_sum > 1.0 + 1e-6:
                    g = g / g_sum
                    # Try different blend factors
                    for t in [0.3, 0.15, 0.05, 0.5]:
                        a_new = (1-t)*a + t*g
                        a_new = np.maximum(0, a_new)
                        sn = np.sum(a_new)
                        if sn < 1e-10:
                            continue
                        a_new = a_new / sn
                        obj = compute_obj_fast(a_new)
                        if obj < best_obj:
                            best_obj = obj
                            best_a = a_new.copy()
                            a = a_new
                            break
        except Exception:
            break
    
    return best_a, best_obj


def coordinate_descent(a, max_time=20):
    """
    Coordinate descent on the normalized vector. Since the objective is
    2n*sum(a^2)/(sum(a))^2, and we can normalize, we minimize sum((a/sum_a)^2)

    For a more general approach (in case the metric is different), do
    coordinate descent on max_conv.
    """

    a = a.copy().astype(np.float64)
    a = np.maximum(0, a)
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj_fast(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        improved = False
        for j in range(n):
            if time.monotonic() - start > max_time:
                break
            
            old_val = a[j]
            # Try different values for a[j]
            for factor in [0.5, 0.75, 1.25, 1.5, 2.0, 0.25, 3.0]:
                a[j] = old_val * factor
                sn = np.sum(a)

                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
                elif obj < compute_obj_fast(best_a):
                    pass
            if not improved and factor == 1.0:
                a[j] = old_val
        
        if not improved:
            break
    
    return best_a, best_obj


def diverse_init(n, rng, method='random'):
    """Generate diverse initial sequences."""
    if method == 'flat':
        return np.ones(n) / n
    elif method == 'edge':
        a = np.zeros(n)
        a[0] = a[-1] = 2.0 / np.sqrt(2*n)
        a[1] = a[-2] = 1.5 / np.sqrt(2*n)
        interior = np.ones(n-4) * 0.5
        a[2:n-2] = interior
        s = np.sum(a)
        return a / s
    elif method == 'gaussian':
        center = n / 2.0
        width = n * rng.uniform(0.3, 0.7)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width)**2)
        a += 0.05
        return a / np.sum(a)
    elif method == 'periodic':
        period = max(2, int(n * rng.uniform(0.05, 0.3)))
        freq = 1.0 / period
        phase = rng.uniform(0, 2*np.pi)
        a = np.maximum(0.01, np.cos(2*np.pi*freq*np.arange(n) + phase))
        return a / np.sum(a)
    elif method == 'bipolar':
        a = np.ones(n)
        a[::2] = rng.uniform(0.3, 1.0)
        a[1::2] = rng.uniform(0.3, 1.0)
        return a / np.sum(a)
    else:  # random
        a = np.maximum(0.001, rng.power(2.0, n))
        return a / np.sum(a)

                break
            method = methods[(r + n) % len(methods)]
            a_init = diverse_init(n, rng, method)
            t_budget = min(time_per_restart, deadline - time.monotonic() - 5)
            if t_budget < 10:
                break
            
            a_opt, obj_opt = full_optimize(a_init, t_budget)
            obj_verify = compute_obj_verify(a_opt)
            
            status = " ** NEW BEST **" if obj_verify < best_score else ""
            print(f"  n={n} r={r} ({method}): verify={obj_verify:.6f}{status}")
            
            if obj_verify < best_score:
                best_score = obj_verify
                best_seq = a_opt.copy()
    
    # Step 5: Final refinement
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s remaining")
        
        # Try optimizing the best found
        a_final, obj_final = full_optimize(best_seq.copy(), min(remaining * 0.6, 60.0))
        obj_final_v = compute_obj_verify(a_final)
        if obj_final_v < best_score:
            best_score = obj_final_v
            best_seq = a_final.copy()
            print(f"  After final opt: {best_score:.6f}")
        
        # Random perturbation search
        if time.monotonic() < deadline - 5:
            n_best = len(best_seq)
            rng2 = np.random.default_rng(999)
            bp_time = deadline - time.monotonic() - 3
            for _ in range(int(bp_time * 100)):
                if time.monotonic() > deadline - 3:
                    break
                a_p = best_seq.copy()
                # Block perturbation
                w = max(1, int(n_best * rng2.uniform(0.02, 0.15)))
                pos = int(rng2.integers(0, n_best - w + 1))
                a_p[pos:pos+w] *= np.exp(rng2.normal(0, 0.05, w))
                a_p = np.maximum(0, a_p)
                sp = np.sum(a_p)
                if sp < 1e-10:
                    continue
                obj_p = compute_obj_fast(a_p)
                if obj_p < best_score - 1e-10:
                    obj_p_v = compute_obj_verify(a_p)
                    if obj_p_v < best_score:
                        best_score = obj_p_v
                        best_seq = a_p.copy()
    
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj_verify(best_seq)
    
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
