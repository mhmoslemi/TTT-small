"""
Adam-based optimization with log-sum-exp smoothing and coordinate descent polish
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
import time

def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def logsumexp_max_grad(conv, tau):
    """Compute smoothed max and its gradient w.r.t. conv values."""
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    return weights

def smoothed_objective_and_grad(a, tau):
    """Compute smoothed objective and gradient w.r.t. a (unnormalized)."""
    n = len(a)
    conv = compute_conv(a)
    weights = logsumexp_max_grad(conv, tau)
    # Gradient of sum_k w_k * (a*a)[k] w.r.t. a[j]
    # (a*a)[k] = sum_i a[i]*a[k-i]
    # d/d(a[j]) (a*a)[k] = a[k-j] + a[2j+k-(n-1) ...] 
    # Actually, let's use the correlation trick:
    # d/d(a[j]) sum_k w_k (a*a)[k] = 2 * sum_k w_k * a[k-j] (with valid index)
    # = 2 * (correlate(weights, a))[j]  (full correlation, take first n)
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    smoothed_max = float(np.sum(weights * conv))
    return smoothed_max, grad

                  lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer for the smoothed subgradient problem."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s  # normalize to sum = 1
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    stall_count = 0
    patience = 5000
    
    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac
        
        _, grad = smoothed_objective_and_grad(a, tau)
        
        # Center gradient for sum constraint
        grad = grad - np.mean(grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a
            s_new = s
        else:
            a_new = a_new / s_new
            s = s_new
        
        a = a_new

        if it % 100 == 0:
            obj = compute_obj(a)
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 100
                if stall_count > patience:
                    break
    
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj

def coordinate_descent(a, max_rounds=100, n_trials=5):
    """Coordinate descent: for each coordinate, try optimizing it in 1D."""
    n = len(a)
    a = a.copy()
    best_obj = compute_obj(a)
    
    for rd in range(max_rounds):
        improved = False
        for j in range(n):
            # Try different values for a[j]
            current_val = a[j]
            best_val = current_val
            best_local_obj = best_obj
            
            # Generate candidate values
            candidates = [
                0.0,
                current_val * 0.5,
                current_val * 0.8,
                current_val * 1.2,
                current_val * 1.5,
                current_val + 0.1 / n,
                current_val - 0.1 / n,
                1.0 / n,  # uniform value
            ]
            candidates = [max(0, c) for c in candidates]
            
            for cand in candidates:
                a[j] = cand
                s = np.sum(a)
                if s < 1e-10:
                    a[j] = current_val
                    continue
                a_norm = a / s
                obj = compute_obj(a_norm)
                if obj < best_local_obj - 1e-10:
                    best_local_obj = obj
                    best_val = cand
                a[j] = current_val
            
            if best_val != current_val:
                a[j] = best_val
                s = np.sum(a)
                a = a / s
                best_obj = best_local_obj
                improved = True
        
        if not improved:
            break
    
    return a, best_obj

Uses L-BFGS-B with log-sum-exp smoothing, Adam with adaptive tau,

import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, tau):
    """Compute smoothed objective and gradient using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    w_sum = float(np.sum(exp_shifted))
    w = exp_shifted / w_sum
    
    smoothed_max = max_conv + np.log(w_sum) / tau
    
    # Gradient of smoothed_max w.r.t. a
    c = np.correlate(w, a, mode='full')
    S_grad = 2.0 * c[:n]
    
    obj = 2.0 * n * smoothed_max / (sum_a * sum_a)
    grad = (2.0 * n / (sum_a * sum_a)) * (S_grad - 2.0 * smoothed_max / sum_a)
    
    return obj, grad

    n = len(a)

        return smooth_obj_grad(a, tau)[0]

        return smooth_obj_grad(a, tau)[1]

        if (it + 1) % check_interval == 0:
            obj = compute_obj(a)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                stall = 0
            else:
                stall += check_interval
                if stall >= patience:
                    break
    
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj


def coordinate_descent(a, max_rounds=50, time_limit=None):
    """Coordinate descent with multiple candidate values per coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng_local = np.random.default_rng(12345)
    
    for rd in range(max_rounds):
        if time_limit is not None and time.monotonic() - start > time_limit:
            break
        improved = False
        order = rng_local.permutation(n)
        for j in order:
            current_val = a[j]
            sum_a = float(np.sum(a))
            
            candidates = np.array([
                0.0,
                current_val * 0.3,
                current_val * 0.5,
                current_val * 0.7,
                current_val * 1.3,
                current_val * 1.6,
                1.0 / n,
                0.5 / n,
                2.0 / n,
                0.3 * sum_a / n,
            ])
            candidates = np.unique(candidates)
            candidates = candidates[(candidates >= 0.0) & (candidates < sum_a - 1e-10)]
            
            best_val = current_val
            best_local = best_obj
            
            for x in candidates:
                if abs(x - current_val) < 1e-12:
                    continue
                a[j] = x
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[j] = current_val
                    continue
                obj = compute_obj(a)
                if obj < best_local - 1e-13:
                    best_local = obj
                    best_val = x
                a[j] = current_val
            
            if best_val != current_val:
                a[j] = best_val
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                    best_obj = best_local
                    improved = True
        
        if not improved:
            break
    
    return a, best_obj


def generate_init(n, rng, init_type=0, parent=None):
    """Generate an initialization of length n."""
    if init_type == 0:
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(0.5, 3.0)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase))
    elif init_type == 1:
        a = 1.0 + rng.normal(0, 0.3, n)
    elif init_type == 2:
        a = np.linspace(rng.uniform(0.5, 1.5), rng.uniform(0.5, 1.5), n)
    elif init_type == 3:
        a = np.where(rng.random(n) < 0.5, rng.uniform(0.5, 1.5), rng.uniform(0.1, 0.5))
    elif init_type == 4:
        if parent is not None and len(parent) == n:
            a = parent + rng.normal(0, 0.05, n)
        else:
            a = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
    elif init_type == 5:
        # Sawtooth-like
        a = np.abs(np.sin(2.0 * np.pi * rng.uniform(1, 4) * np.arange(n) / n + rng.uniform(0, 3)))
    elif init_type == 6:
        # Gaussian bump
        center = rng.uniform(0.2 * n, 0.8 * n)
        width = rng.uniform(0.1 * n, 0.4 * n)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    else:
        a = rng.random(n)
    
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def optimize_single(a, time_budget, seed=0):
    """Run full optimization pipeline on a single initialization."""
    rng_local = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + time_budget
    n = len(a)
    
    # Phase 1: L-BFGS-B with increasing tau
    for tau in [5, 20, 60, 150]:
        if time.monotonic() > deadline - 10:
            break
        a = lbfgs_optimize(a, tau=tau, max_iter=150)
    
    # Phase 2: Adam
    remaining = deadline - time.monotonic()
    if remaining > 10:
        max_iter = int(min(50000, remaining * 6))
        a, _ = adam_optimize(a, max_iter=max_iter, tau_start=10.0,
                             tau_end=120.0, lr=0.03, patience=3000)
    
    # Phase 3: Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 5:
        a, _ = coordinate_descent(a, max_rounds=30, time_limit=remaining - 5)
    
    # Phase 4: Try reversal
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_rev = a[::-1].copy()
        a_rev = lbfgs_optimize(a_rev, tau=60, max_iter=100)
        if time.monotonic() < deadline - 5:
            a_rev, _ = adam_optimize(a_rev, max_iter=int(min(10000, (deadline - time.monotonic() - 3) * 4)),
                                     tau_start=30.0, tau_end=150.0, lr=0.02, patience=2000)
        if time.monotonic() < deadline - 3:
            a_rev, _ = coordinate_descent(a_rev, max_rounds=10, time_limit=3.0)
        if compute_obj(a_rev) < compute_obj(a):
            a = a_rev
    
    return a


def optimize_for_n(n, seed, time_budget=60.0, parent=None, verbose=False):
    """Optimize a sequence of length n within the time budget."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + time_budget
    
    best_a = None
    best_obj = np.inf
    
    n_inits = max(2, int(time_budget / 18))
    
    for init_idx in range(n_inits):
        if time.monotonic() > deadline - 12:
            break
        
        init_type = (init_idx + n % 7) % 7
        if parent is not None and init_idx == 0:
            init_type = 4
        
        a = generate_init(n, rng, init_type, parent)
        
        if verbose:
            print(f"    n={n} init{init_idx} type{init_type}: obj={compute_obj(a):.6f}")
        
        remaining = deadline - time.monotonic()
        alloc = max(10.0, remaining / (n_inits - init_idx) - 2)
        
        a_opt = optimize_single(a, time_budget=alloc, seed=seed + init_idx * 31)
        obj = compute_obj(a_opt)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    n={n} init{init_idx}: obj={best_obj:.6f}")
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)
    
    return best_a, best_obj


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-stage optimization: L-BFGS-B + Adam + coordinate descent + reversal.
    Focuses on parent length and neighbors, with final intensive polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    
    best_seq = None
    best_score = np.inf
    
    # Load and optimize parent
    parent, parent_obj = try_parent_sequence()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        n_par = len(parent)
        
        print(f"Optimizing parent (n={n_par}) for 150s...")
        a_par = parent.copy()
        a_par = optimize_single(a_par, time_budget=150.0, seed=seed)
        obj_par = compute_obj(a_par)
        if obj_par < best_score:
            best_score = obj_par
            best_seq = a_par.copy()
            print(f"  Parent improved to: {best_score:.6f}")
    
    # Explore other n values
    n_values = [200, 220, 250, 280, 300, 350, 400]
    if parent is not None:
        n_par = len(parent)
        n_values = sorted(set(n_values + [n_par, max(150, n_par - 30), min(500, n_par + 30)]))
    
    print(f"\nExploring n={n_values}")
    
    for n in n_values:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        if parent is not None and n == len(parent):
            alloc = min(100.0, remaining * 0.18)
        elif n <= 250:
            alloc = min(70.0, remaining * 0.13)
        else:
            alloc = min(55.0, remaining * 0.10)
        
        alloc = max(25.0, alloc)
        print(f"\nN={n}: {alloc:.0f}s allocated, {remaining:.0f}s remaining")
        
        par = (parent if (parent is not None and len(parent) == n) else None)
        a_opt, obj = optimize_for_n(n, seed=seed + n * 7919,
                                     time_budget=alloc, parent=par,
                                     verbose=False)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  UPDATE: n={n}, obj={best_score:.6f}")
    
    # === Final intensive polishing ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Final polishing (n={n_best}), {remaining:.0f}s remaining ===")
        
        a_cur = best_seq.copy()
        round_num = 0
        no_improve_rounds = 0
        
        while time.monotonic() < deadline - 10:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            old_score = best_score
            
            # High-tau Adam
            adam_budget = min(remaining * 0.35, 80.0)
            if adam_budget > 8:
                max_iter = int(min(80000, adam_budget * 7))
                a_cur, _ = adam_optimize(a_cur, max_iter=max_iter,
                                         tau_start=40.0, tau_end=200.0,
                                         lr=0.015, patience=6000,
                                         check_interval=20)
                obj = compute_obj(a_cur)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_cur.copy()
                    print(f"  R{round_num} Adam: {best_score:.7f}")
            
            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 10:
                cd_budget = min(remaining * 0.25, 40.0)
                a_cur, obj_cd = coordinate_descent(a_cur,
                                                    max_rounds=int(cd_budget * 2),
                                                    time_limit=cd_budget)
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cur.copy()
                    print(f"  R{round_num} CD: {best_score:.7f}")
            
            # L-BFGS-B with very high tau
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_cur = lbfgs_optimize(a_cur, tau=300, max_iter=200)
                obj = compute_obj(a_cur)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_cur.copy()
                    a_cur = a_cur.copy()
                    print(f"  R{round_num} L-BFGS: {best_score:.7f}")
            
            # Reversal trick
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_rev = a_cur[::-1].copy()
                a_rev = lbfgs_optimize(a_rev, tau=100, max_iter=100)
                if time.monotonic() < deadline - 5:
                    a_rev, _ = adam_optimize(a_rev, max_iter=int(min(15000, (deadline - time.monotonic() - 3) * 5)),
                                             tau_start=50.0, tau_end=200.0, lr=0.015, patience=3000)
                if time.monotonic() < deadline - 3:
                    a_rev, _ = coordinate_descent(a_rev, max_rounds=10, time_limit=3.0)
                obj_rev = compute_obj(a_rev)
                if obj_rev < best_score:
                    best_score = obj_rev
                    best_seq = a_rev.copy()
                    a_cur = a_rev.copy()
                    print(f"  R{round_num} Rev: {best_score:.7f}")
            
            # Perturbation escape
            remaining = deadline - time.monotonic()
            if remaining > 15 and round_num > 2:
                scales = [0.05, 0.1, 0.15]
                for sc in scales:
                    if time.monotonic() > deadline - 5:
                        break
                    a_pert = a_cur * np.exp(rng.normal(0, sc, n_best))
                    a_pert = np.maximum(a_pert, 0.0)
                    s_p = float(np.sum(a_pert))
                    if s_p > 1e-10:
                        a_pert = a_pert / s_p
                        # Quick local optimization from perturbation
                        a_pert = lbfgs_optimize(a_pert, tau=30, max_iter=50)
                        if time.monotonic() < deadline - 5:
                            a_pert, _ = adam_optimize(a_pert, max_iter=3000,
                                                       tau_start=30.0, tau_end=100.0,
                                                       lr=0.03, patience=2000)
                        if time.monotonic() < deadline - 3:
                            a_pert, _ = coordinate_descent(a_pert, max_rounds=5, time_limit=3.0)
                        obj_p = compute_obj(a_pert)
                        if obj_p < best_score:
                            best_score = obj_p
                            best_seq = a_pert.copy()
                            a_cur = a_pert.copy()
                            print(f"  R{round_num} Pert(sc={sc}): {best_score:.7f}")
            
            # Track stagnation
            if best_score < old_score - 1e-12:
                no_improve_rounds = 0
            else:
                no_improve_rounds += 1
                if no_improve_rounds > 5:
                    break
        
        # Absolute final coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_final, obj_final = coordinate_descent(best_seq, max_rounds=80,
                                                     time_limit=remaining - 5)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final CD: {best_score:.7f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(250) / 250
        best_score = compute_obj(best_seq)
    
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.7f}, time={elapsed:.1f}s")
    
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf and abs(validation - best_score) > 1e-6:
        print(f"  Validation: computed={best_score:.7f}, validated={validation:.7f}")
        best_score = validation
    
    return result
