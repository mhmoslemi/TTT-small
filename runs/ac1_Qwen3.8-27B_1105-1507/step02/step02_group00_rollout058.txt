"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed projected gradient with adaptive tau, LP-based alternating
optimization with HiGHS, structured initializations, and diverse sequence lengths.
Key improvements: FFT-accelerated smooth gradient computation, Nesterov-like
momentum, adaptive tau annealing, and LP warm-starting from smoothed solutions.

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(x):
    """Project x onto {a >= 0, sum(a) = 1}."""
    n = len(x)
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    ts = (cumsum - 1.0) / np.arange(1, n + 1)
    valid = x_sorted > ts
    if not valid.any():
        k = 1
    else:
        k = int(np.max(np.where(valid)[0])) + 1
    t = (cumsum[k - 1] - 1.0) / k
    return np.maximum(x - t, 0.0)


def smooth_gradient(a, tau):
    """
    Compute gradient of tau*logsumexp(conv(a,a)/tau) w.r.t. a.
    Returns (gradient, softmax weights p).
    """
    n = len(a)
    h = np.convolve(a, a)  # length 2n-1
    h_max = np.max(h)
    shifted = (h - h_max) / tau
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(h)
        p[np.argmax(h)] = 1.0
    else:
        p /= p_sum

    # grad[j] = 2 * sum_k p[k] * a[k-j] (with a[m]=0 for invalid m)
    # Use FFT: ifft(FFT(p) * conj(FFT(a))) gives cross-correlation
    size = 1
    needed = 3 * n - 2
    while size < needed:
        size *= 2
    P = np.fft.rfft(p, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(P * np.conj(A), n=size)
    grad = 2.0 * corr[:n]
    return grad, p, h_max


def smoothed_gradient_descent(a_init, max_iter=5000, verbose=False):
    """
    Smoothed projected gradient with momentum for minimizing max(conv(a,a))
    subject to sum(a)=1, a>=0.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    # Initial tau based on spread of convolution
    h0 = np.convolve(a, a)
    tau0 = float(np.max(h0)) / 5.0
    if tau0 < 1e-6:
        tau0 = 0.01

    # Momentum
    v = np.zeros(n)
    momentum = 0.9

    for it in range(max_iter):
        # Adaptive tau: decrease over time
        tau = tau0 / (1.0 + it / 200.0)
        if tau < tau0 * 0.01:
            tau = tau0 * 0.01

        grad, p, h_max = smooth_gradient(a, tau)

        # Adaptive step size
        max_a = np.max(a)
        step = min(1.0, 0.5 / (max_a + 1e-6)) / (1.0 + 0.005 * it)

        # Momentum update
        v = momentum * v - step * grad
        a_new = a + v

        # Project onto simplex
        a_new = project_simplex(a_new)

        # Check objective
        if it % 10 == 0 or it == max_iter - 1:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

        a = a_new

        if verbose and it % 200 == 0:
            obj = compute_obj(a)
            print(f"      SGD it={it}: obj={obj:.6f}, tau={tau:.6f}")

    # Final check
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def build_A(a):
    """Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_g(a, M=None):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    if M is None:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternate(a_init, max_iter=200, verbose=False):
    """LP alternating optimization with adaptive blending."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(max_iter):
        conv_a = np.convolve(a, a)
        M = float(np.max(conv_a))

        # Try slightly tighter M to force improvement
        M_tight = M * 0.95
        g = solve_lp_g(a, M=M_tight)
        if g is None:
            g = solve_lp_g(a, M=M)
        if g is None:
            break

        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Check if g alone improves
        obj_g = compute_obj(g)
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()

        # Adaptive blend
        if it < 10:
            t = 0.8
        elif it < 50:
            t = 0.6
        else:
            t = 0.4

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a = g
        else:
            a = a_new / s_new

        # Also check blend
        obj_blend = compute_obj(a)
        if obj_blend < best_obj:
            best_obj = obj_blend
            best_a = a.copy()

        if verbose and it % 50 == 0:
            print(f"    LP it={it}: best_obj={best_obj:.6f}")

    return best_a, best_obj


def subgradient_polish(a, n_steps=3000, verbose=False):
    """Subgradient method with adaptive active set."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for step in range(n_steps):
        h = np.convolve(a, a)
        max_h = np.max(h)
        if max_h < 1e-15:
            break

        # Active set with relative tolerance
        tol = max_h * (0.005 + 0.05 * (step / n_steps))
        active = h >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = h >= max_h - 1e-10
            n_active = 1

        # Compute subgradient
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            vals = a[k - j_end:k - j_start + 1][::-1]
            grad[j_start:j_end + 1] += 2.0 * vals
            if k % 2 == 0:
                mid = k // 2
                if 0 <= mid < n:
                    grad[mid] -= a[mid]
        grad /= n_active

        # Step size
        step_size = 0.3 / (1.0 + 0.005 * step)

        a_new = a - step_size * grad
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj

        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + rng.uniform(0.3, 0.8) * np.cos(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.01)
    elif mode == 1:
        # U-shape (high at edges)
        alpha = rng.uniform(0.3, 1.0)
        a = 1.0 + alpha * (2 * x - 1) ** 2
    elif mode == 2:
        # Inverted U (high in middle)
        alpha = rng.uniform(0.3, 0.8)
        a = 1.0 + alpha * np.cos(np.pi * (x - 0.5))
        a = np.maximum(a, 0.01)
    elif mode == 3:
        # Periodic binary-like
        period = int(rng.integers(max(2, n // 50), max(3, n // 5)))
        pattern = np.where(rng.random(period) > 0.5, rng.uniform(0.5, 1.5), rng.uniform(0.1, 0.5))
        a = np.tile(pattern, n // period + 1)[:n]
    elif mode == 4:
        # Gaussian bump
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = 0.3 + 0.7 * a
    elif mode == 5:
        # Sum of two cosines
        f1 = rng.uniform(1, 3)
        f2 = rng.uniform(3, 8)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.3 * np.cos(2 * np.pi * f2 * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
    elif mode == 6:
        # Linear ramp
        slope = rng.uniform(-1, 1)
        a = 1.0 + slope * (x - 0.5)
        a = np.maximum(a, 0.01)
    else:
        # Random smooth
        a = rng.random(n)
        # Smooth with moving average
        kernel = np.ones(max(1, n // 20)) / max(1, n // 20)
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)

    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n) + 0.1
        s = np.sum(a)
    return a / s


def coord_refine(a, n_rounds=2):
    """Quick coordinate descent refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            old_val = a[j]
            best_local = best_obj
            best_x = old_val
            for mult in [0.0, 0.3, 0.7, 1.0, 1.4, 2.0]:
                x = old_val * mult
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x
            a[j] = best_x
            if best_x != old_val:
                improved = True
        if compute_obj(a) < best_obj:
            best_obj = compute_obj(a)
            best_a = a.copy()
        if not improved:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining smoothed gradient, LP alternating,
    subgradient polishing, and coordinate refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent_arr = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_arr = parent
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # N values to try (prioritize around parent length and moderately larger)
    n_parent = len(parent_arr) if parent_arr is not None else 300
    n_values = []
    # Include parent n and nearby values
    for delta in [0, -50, -100, 50, 100, 200, 300]:
        nn = n_parent + delta
        if 50 <= nn <= 800:
            n_values.append(nn)
    # Also include some fixed values
    for nn in [100, 150, 200, 250, 300, 400, 500]:
        if nn not in n_values:
            n_values.append(nn)
    n_values = sorted(set(n_values))

    # Phase 1: For each n, try smoothed gradient + LP
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        print(f"N={n}: {remaining:.0f}s remaining, current best={best_score:.6f}")

        n_restarts = 4 if remaining > 600 else (3 if remaining > 300 else 2)

        for restart in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break

            # Choose initialization
            if restart == 0 and parent_arr is not None and len(parent_arr) == n:
                a_init = parent_arr.copy()
            elif restart == 0 and best_seq is not None:
                # Resize best_seq to n
                from numpy import interp
                if len(best_seq) != n:
                    old_n = len(best_seq)
                    new_x = np.linspace(0, 1, n)
                    old_x = np.linspace(0, 1, len(best_seq))
                    a_init = np.interp(new_x, old_x, best_seq)
                    a_init = np.maximum(a_init, 0)
                    a_init = a_init / np.sum(a_init)
                else:
                    a_init = best_seq.copy() / np.sum(best_seq)
            else:
                a_init = structured_init(n, rng)

            # Time budget for this restart
            t_remaining = deadline - time.monotonic()
            t_per_phase = max(5.0, t_remaining / (3 * (n_restarts - restart)))

            # Phase 1a: Smoothed gradient descent
            max_iter_sgd = int(min(3000, t_per_phase / 0.005))
            if max_iter_sgd > 100:
                a1, obj1 = smoothed_gradient_descent(a_init, max_iter=max_iter_sgd)
                if obj1 < best_score:
                    best_score = obj1
                    best_seq = a1.copy()
                    print(f"  SGD n={n} r={restart}: obj={obj1:.6f}")

            if time.monotonic() > deadline - 20:
                break

            # Phase 1b: LP alternating (start from best of a1 and a_init)
            a_start_lp = a1 if obj1 < compute_obj(a_init) else a_init
            max_iter_lp = int(min(150, t_per_phase / 0.1))
            if max_iter_lp > 10:
                a2, obj2 = lp_alternate(a_start_lp, max_iter=max_iter_lp)
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a2.copy()
                    print(f"  LP n={n} r={restart}: obj={obj2:.6f}")

            if time.monotonic() > deadline - 20:
                break

            # Phase 1c: Subgradient polish
            a_start_sg = a2 if 'a2' in dir() and obj2 < best_score else a1
            remaining_sg = deadline - time.monotonic() - 10
            ns = int(min(2000, max(200, remaining_sg / 0.01)))
            if ns > 100:
                a3, obj3 = subgradient_polish(a_start_sg, n_steps=ns)
                if obj3 < best_score:
                    best_score = obj3
                    best_seq = a3.copy()
                    print(f"  SG n={n} r={restart}: obj={obj3:.6f}")

    # Phase 2: Intensive refinement of the best solution
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPhase 2: refining n={n_best}, obj={best_score:.6f}, {remaining:.0f}s left")

        # Extended smoothed gradient
        max_iter = int(min(5000, remaining / 0.008))
        if max_iter > 500:
            a_ref, obj_ref = smoothed_gradient_descent(best_seq, max_iter=max_iter)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  Refine SGD: obj={obj_ref:.6f}")

        # LP polish
        if time.monotonic() < deadline - 20 and n_best <= 600:
            remaining = deadline - time.monotonic()
            max_iter = int(min(300, remaining / 0.1))
            if max_iter > 20:
                a_lp, obj_lp = lp_alternate(best_seq, max_iter=max_iter)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  Refine LP: obj={obj_lp:.6f}")

        # Subgradient with tight tolerance
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            ns = int(min(5000, remaining / 0.008))
            if ns > 500:
                a_sg, obj_sg = subgradient_polish(best_seq, n_steps=ns)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    print(f"  Refine SG: obj={obj_sg:.6f}")

        # Coordinate refinement (only for small n)
        if time.monotonic() < deadline - 10 and n_best <= 200:
            a_coord, obj_coord = coord_refine(best_seq, n_rounds=2)
            if obj_coord < best_score:
                best_score = obj_coord
                best_seq = a_coord.copy()
                print(f"  Refine Coord: obj={obj_coord:.6f}")

    # Phase 3: Try a few more random restarts if time permits
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 3: random restarts, {remaining:.0f}s left")
        n_try = len(best_seq) if best_seq is not None else 300
        n_attempts = int(remaining / 30)
        for att in range(min(n_attempts, 5)):
            if time.monotonic() > deadline - 10:
                break
            a_init = structured_init(n_try, rng)
            # Perturb from best
            if best_seq is not None and len(best_seq) == n_try:
                noise = rng.random(n_try) * 0.1
                a_init = 0.7 * best_seq / np.sum(best_seq) + 0.3 * a_init
                a_init = np.maximum(a_init, 0)
                a_init = a_init / np.sum(a_init)

            max_iter = int(min(2000, (deadline - time.monotonic() - 10) / 0.005))
            if max_iter > 200:
                a_new, obj_new = smoothed_gradient_descent(a_init, max_iter=max_iter)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"  Restart {att}: obj={obj_new:.6f}")

            # Quick LP
            if time.monotonic() < deadline - 15:
                max_iter_lp = int(min(100, (deadline - time.monotonic() - 10) / 0.1))
                if max_iter_lp > 20:
                    a_lp2, obj_lp2 = lp_alternate(a_new, max_iter=max_iter_lp)
                    if obj_lp2 < best_score:
                        best_score = obj_lp2
                        best_seq = a_lp2.copy()
                        print(f"  Restart {att} LP: obj={obj_lp2:.6f}")

    # Final fallback
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
