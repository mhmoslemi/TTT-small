import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J_normalized(a):
    """Compute J for a normalized (sum=1) vector."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a = a / s
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def compute_objective(a):
    """Compute the actual J value (scale invariant)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_dense(a):
    """Build dense constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        # A[k, jl:jh+1] = a[k-jh : k-jl+1][::-1]
        A[k, jl:jh+1] = a[k-jh:k-jl+1][::-1]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_dense(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smoothed_objective(a, tau):
    """Log-sum-exp smoothed max of convolution."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    a = a / s
    b = fast_conv(a)
    b_shift = b - b.max()
    return float(b.max() + tau * np.log(np.exp(b_shift / tau).sum()))


def smoothed_gradient(a, tau):
    """Gradient of the smoothed objective."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros_like(a)
    a = a / s
    b = fast_conv(a)
    b_shift = (b - b.max()) / tau
    w = np.exp(b_shift)
    w /= w.sum()
    
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        # d(b_k)/d(a_j) = 2 * a_{k-j} for valid j
        # a_{k-j} for j in [jl, jh] is a[k-jh:k-jl+1][::-1]
        grad[jl:jh+1] += (2.0 * w[k] / tau) * a[k-jh:k-jl+1][::-1]
    return grad
