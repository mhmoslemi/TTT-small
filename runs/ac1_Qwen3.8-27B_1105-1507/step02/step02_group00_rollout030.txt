def compute_obj(a):

    return 2.0 * n * max_b / (sum_a * sum_a)

    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        A[k, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp_g(a, M):

    A = build_A(a)
    c = -np.ones(n)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_optimize(a_init, max_iter, deadline):
    """LP alternation with adaptive M."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a_init, compute_obj(a_init)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))

def build_grad_matrix(a):
    n = len(a)
    # Method: use scipy's toeplitz or manual construction
    # G[k, i] = a[k-i] + a[i-k]
    
    # First part: H[k, i] = a[k-i] (valid when 0 <= k-i < n)
    # This is a lower-bidiagonal-like matrix
    # H is (2n-1) x n
    # Row k, column i: a[k-i] if 0 <= k-i < n else 0
    
    # We can construct this using shifting
    H = np.zeros((2*n-1, n))
    for shift in range(n):
        # a[shift] appears at H[k, i] where k-i = shift => k = i+shift
        # So H[i+shift, i] = a[shift] for valid i
        i_max = min(n-1, 2*n-2-shift)
        H[shift:shift+i_max+1, :i_max+1] += np.diag(a[shift] * np.ones(i_max+1))
    
    # This is O(n^2) with numpy but involves n iterations...
    # Better approach:
    
    # Actually, the simplest vectorized approach:
    idx_k = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    idx_i = np.arange(n)[None, :]      # (1, n)
    diff = idx_k - idx_i                # (2n-1, n): diff[k,i] = k-i
    # H[k,i] = a[k-i] if 0 <= k-i < n else 0
    H = np.where((diff >= 0) & (diff < n), a[diff.clip(0, n-1)], 0.0)
    # G[k,i] = a[k-i] + a[i-k] if 0 <= i-k < n else 0
    # a[i-k] where i-k = -diff, so a[i-k] = a[-diff] if 0 <= -diff < n
    G = H + np.where((diff <= 0) & (diff > -n), a[(-diff).clip(0, n-1)], 0.0)
    
    return G

                ns_final = int(min(2000, (phase3_deadline - time.monotonic()) / 0.01))
                if ns_final > 100:
                    a_final, obj_final = subgradient_polish(best_seq, ns_final, phase3_deadline)
                    if obj_final < best_score:
                        best_score = obj_final
                        best_seq = a_final.copy()
                        print(f"  Final SG: obj={obj_final:.6f} *")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"DONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Ensure valid output
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    return [float(x) for x in best_seq]
