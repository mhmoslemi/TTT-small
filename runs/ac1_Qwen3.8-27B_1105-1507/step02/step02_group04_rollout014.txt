Nesterov-accelerated smoothed gradient descent with softmax parameterization,
backtracking line search, and LP polishing for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Explores multiple sequence lengths with diverse initializations.

import numpy as np
from scipy.optimize import linprog

def fft_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]


def softmax(x):
    """Numerically stable softmax."""
    xshift = x - np.max(x)
    e = np.exp(xshift)
    return e / np.sum(e)


def compute_obj_normalized(a):
    """Compute objective assuming sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fft_convolve(a)
    return 2.0 * n * np.max(conv)


def smoothed_objective(a, tau):
    """Compute smoothed log-sum-exp of convolution, scaled."""
    conv = fft_convolve(a)
    # Numerically stable log-sum-exp
    m = np.max(conv)
    return m + np.log(np.mean(np.exp(tau * (conv - m)))) / tau


def compute_grad_smoothed(a, tau):
    """Compute gradient of smoothed objective w.r.t. a (unconstrained)."""
    n = len(a)
    conv = fft_convolve(a)
    m = np.max(conv)
    # p_k = exp(tau*(conv_k - m)) / sum_j exp(tau*(conv_j - m))
    shifted = tau * (conv - m)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    p = e / np.sum(e)
    # grad[i] = sum_k p[k] * a[k-i] (for valid k-i)
    # = cross-correlation of p and a at lag i
    # np.correlate(p, a, 'full')[i] = sum_j p[j+i] * a[j]
    grad = np.correlate(p, a, mode='full')[:n]
    return grad

    a = softmax(b)

    dot = np.dot(grad_a, a)
    grad_b = a * (grad_a - dot)
    return grad_b


def line_search(b, grad_b, tau, alpha_init=1.0, max_iter=30):

    f_init = smoothed_objective(a, tau)

        alpha *= 0.5
    return b - alpha * grad_b, alpha

def build_A_matrix(a):
    """Build Toeplitz matrix for LP constraints."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax+1] = a[k - jmax:k - jmin + 1][::-1]
    return A

"""
Momentum-accelerated smoothed gradient descent in softmax b-space, with LP polishing,
adaptive tau scheduling, and multi-length exploration for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated convolution and backtracking line search for efficient convergence.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a):
    """Compute full linear autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def safe_softmax(x):
    """Numerically stable softmax."""
    xshift = x - np.max(x)
    e = np.exp(xshift)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fft_convolve(a)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_val(a, tau):
    """Value of smoothed log-sum-exp objective."""
    conv = fft_convolve(a)
    m = float(np.max(conv))
    shifted = tau * (conv - m)
    shifted = np.clip(shifted, -500, 500)
    return m + float(np.log(np.mean(np.exp(shifted)))) / tau


def grad_smoothed_a(a, tau):
    """Gradient of smoothed max(conv(a,a)) w.r.t. a (unconstrained)."""
    n = len(a)
    conv = fft_convolve(a)
    m = float(np.max(conv))
    shifted = np.clip(tau * (conv - m), -500, 500)
    e = np.exp(shifted)
    p = e / np.sum(e)
    # grad[i] = sum_j p[i+j] * a[j] = cross-correlation of p and a
    grad = np.correlate(p, a, mode='full')[:n]
    return grad


def grad_b_space(b, tau):
    """Gradient of smoothed objective in b-space (softmax parameterization)."""
    a = safe_softmax(b)
    ga = grad_smoothed_a(a, tau)
    dot = float(np.dot(ga, a))
    return a * (ga - dot)


def momentum_optimize(a_init, max_iter=50000, tau_start=10.0, tau_end=300.0,

    """Momentum gradient descent in softmax b-space with adaptive step."""
    n = len(a_init)

    a_init = a_init / np.sum(a_init)

        else:

def build_toeplitz(a):
    """Build Toeplitz matrix A such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_optimize(a, max_iter=30, t_mix=0.3, time_budget=30.0):
    """Alternating LP: maximize sum(g) s.t. conv(a,g) <= M, g>=0."""
    n = len(a)
    a = a.copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    stall = 0
    start = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        conv = fft_convolve(a)
        M = float(np.max(conv))

        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            remaining_lp = max(1.0, time_budget - (time.monotonic() - start))
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': remaining_lp})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(-res.fun)
        except Exception:
            break

        if sum_g <= 1.0 + 1e-5:
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0

        g = g / sum_g
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def coordinate_descent(a, max_sweeps=50):
    """Coordinate descent: optimize each element of a one at a time."""
    n = len(a)
    a = a.copy()
    a = a / np.sum(a)
    best_obj = compute_obj(a)

    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            # Compute current convolution contributions
            # conv[k] = C[k] + 2*a[i]*S[k] + (a[i]^2 if k == 2i else 0)
            conv_before = fft_convolve(a)
            
            # Find optimal a[i] by ternary search (convex in a[i] for the max)
            a_old = a[i]
            a[i] = 0
            base_conv = fft_convolve(a)
            # Now conv[k] as function of x = a[i]:
            # conv[k] = base_conv[k] + 2*x * a[k-i] (if 0<=k-i<n) + x^2 (if k==2i)
            # Actually base_conv already has a[i]=0, so:
            # The terms involving a[i]=x:
            # For element j of a (j != i): a[j]*a[k-j] is in base_conv
            # Additional terms: a[i]*a[k-i] = x*a[k-i] and a[k-i]*a[i] = a[k-i]*x
            # So conv[k] = base_conv[k] + x * (a[k-i] if 0<=k-i<n else 0) + (x*a[i-k] if 0<=i-k<n and i-k != i else 0)
            # Wait, since a[i]=0 in base, the full conv is:
            # conv_full[k] = base_conv[k] + x*a[k-i] (if valid) + a[i-k]*x (if valid and i-k != i)
            # = base_conv[k] + x * (a[k-i] + a[i-k] if both valid, or just one)
            # + x^2 if k-i==i, i.e., k==2i
            
            # Let's compute the linear coefficient for each k
            lin = np.zeros(2 * n - 1)
            quad = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                coeff = 0.0
                if 0 <= k - i < n:
                    coeff += a[k - i]
                if 0 <= i - k < n and (i - k) != i:
                    coeff += a[i - k]
                lin[k] = coeff
                if k == 2 * i and 0 <= i < n:
                    quad[k] = 1.0

            # Minimize max_k (base_conv[k] + x*lin[k] + x^2*quad[k]) over x >= 0
            # Use bisection on the subgradient
            lo = 0.0
            hi = a_old * 3.0 + 0.01
            if hi < 1e-6:
                hi = 0.01

            for _ in range(40):
                mid = (lo + hi) / 2
                # Subgradient of max at x=mid: check which k achieves max
                vals = base_conv + mid * lin + mid * mid * quad
                kmax = np.argmax(vals)
                deriv = lin[kmax] + 2 * mid * quad[kmax]
                if deriv > 0:
                    hi = mid
                elif deriv < 0:
                    lo = mid
                else:
                    break

            x_opt = (lo + hi) / 2
            a[i] = x_opt

            # Renormalize
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

            obj = compute_obj(a)
            if obj < best_obj - 1e-8:
                best_obj = obj
                improved = True

        if not improved and sweep > 5:
            break

    return a, best_obj

            continue

        # Time per n
        if parent is not None and abs(n - len(parent)) <= 25:
            alloc = min(40.0, remaining * 0.15)
        elif n <= 250:
            alloc = min(25.0, remaining * 0.1)
        else:
            alloc = min(20.0, remaining * 0.08)
        alloc = max(10.0, alloc)

        num_inits = max(2, int(alloc / 8))
        inits = generate_initializations(n, rng, num_inits)

        # Include parent if same length
        if parent is not None and len(parent) == n:
            inits.insert(0, parent.copy())

        for init_idx, a_init in enumerate(inits):
            if time.monotonic() > phase1_end - 5:
                break
            init_time = min(alloc / max(1, num_inits), phase1_end - time.monotonic() - 5)
            if init_time < 5:
                break

            a_opt, obj = optimize_single(a_init, init_time)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n} init={init_idx}: NEW BEST obj={best_score:.6f}")

    print(f"\nAfter Phase 1: best={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}")

    # Phase 2: Focused optimization on best neighborhood (35% of time)
    phase2_end = start_time + (budget_s - 25.0) * 0.75
    if best_seq is not None and time.monotonic() < phase2_end:
        n_best = len(best_seq)
        print(f"\nPhase 2: Focused optimization on n={n_best}")

        # Try nearby lengths
        nearby_ns = [n_best]
        for delta in [25, 50, 75]:
            nearby_ns.append(max(20, n_best - delta))
            nearby_ns.append(min(600, n_best + delta))
        nearby_ns = sorted(set(nearby_ns))

        for n in nearby_ns:
            if time.monotonic() > phase2_end - 10:
                break
            remaining = phase2_end - time.monotonic()
            if remaining < 20:
                continue

            num_inits = max(3, int(remaining / 25))
            inits = generate_initializations(n, rng, num_inits)

            if n == n_best and best_seq is not None:
                # Start from best and add perturbations
                inits.insert(0, best_seq.copy())
                for pert_scale in [0.05, 0.1, 0.15]:
                    a_pert = best_seq * np.exp(rng.normal(0, pert_scale, n_best))
                    a_pert = np.maximum(a_pert, 1e-4)
                    a_pert = a_pert / np.sum(a_pert)
                    inits.append(a_pert)

            for init_idx, a_init in enumerate(inits):
                if time.monotonic() > phase2_end - 5:
                    break
                init_time = min(remaining / max(1, num_inits), phase2_end - time.monotonic() - 5)
                if init_time < 5:
                    break
                a_opt, obj = optimize_single(a_init, init_time)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  n={n} init={init_idx}: NEW BEST obj={best_score:.6f}")

    print(f"\nAfter Phase 2: best={best_score:.6f}")

    # Phase 3: Intensive final polishing (remaining time)
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPhase 3: Final polishing (n={n_best}, {remaining:.0f}s)")

        a_cur = best_seq.copy()
        round_num = 0

        while time.monotonic() < deadline - 5:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            # High-tau gradient
            max_iter = int(min(20000, remaining * 3000))
            a_cur, obj = momentum_optimize(a_cur, max_iter=max_iter,
                                            tau_start=80.0, tau_end=1000.0,
                                            momentum=0.85, alpha_init=0.05)
            if obj < best_score:
                best_score = obj
                best_seq = a_cur.copy()
                print(f"  Round {round_num} grad: obj={best_score:.6f}")

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 5:
                a_cur, obj = coordinate_descent(a_cur, max_sweeps=min(20, int(remaining)))
                if obj < best_score:
                    best_score = obj
                    best_seq = a_cur.copy()
                    print(f"  Round {round_num} coord: obj={best_score:.6f}")

            # LP step
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 450:
                a_lp, obj_lp = lp_optimize(a_cur, max_iter=min(25, int(remaining // 3)),
                                            t_mix=0.2, time_budget=min(remaining * 0.4, 30.0))
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    print(f"  Round {round_num} LP: obj={best_score:.6f}")

            # Stochastic perturbation + re-optimize
            remaining = deadline - time.monotonic()
            if remaining > 20:
                for scale in [0.03, 0.07, 0.12]:
                    a_pert = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_pert = np.maximum(a_pert, 1e-4)
                    a_pert = a_pert / np.sum(a_pert)
                    obj_p = compute_obj(a_pert)
                    if obj_p < best_score * 1.005:
                        a_pert, obj_p = momentum_optimize(a_pert, max_iter=5000,
                                                           tau_start=30.0, tau_end=300.0,
                                                           momentum=0.9, alpha_init=0.2)
                        if obj_p < best_score:
                            best_score = obj_p
                            best_seq = a_pert.copy()
                            a_cur = a_pert.copy()
                            print(f"  Round {round_num} perturb({scale}): obj={best_score:.6f}")
                            break

    # Fallback
    if best_seq is None:
        best_seq = np.ones(50) / 50
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if np.isfinite(validation):
        if validation < best_score:
            print(f"  Validation lower: {validation:.6f} vs computed {best_score:.6f}")
            best_score = validation
        elif abs(validation - best_score) > 1e-6:
            print(f"  Note: validation={validation:.6f}, computed={best_score:.6f}")

    return result
