Convex optimization approach for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated subgradient method on the simplex, combined with
scipy trust-constr refinement for final polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def compute_conv(a):
    """Compute full convolution."""
    return np.convolve(a, a)


def subgradient_of_max_conv(a, active_indices):
    """
    Compute the average subgradient of max_k conv(a,a)[k] at point a.
    grad_k[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in active_indices:
        # grad_k[m] = 2*a[k-m] for 0 <= k-m < n
        # i.e., m ranges from max(0, k-n+1) to min(k, n-1)
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            # a[k-m] for m in [m_start, m_end]
            # = a[k-m_end : k-m_start+1][::-1]
            grad[m_start:m_end+1] += 2.0 * a[k-m_end:k-m_start+1][::-1]
    grad /= len(active_indices)
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    a = project_simplex(a_init.copy().astype(np.float64))
    
    # Nesterov momentum
    y = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_f = np.max(np.convolve(a, a))

        conv_y = np.convolve(y, y)
        max_val = np.max(conv_y)
        
        # Active set: indices where conv is close to max
        # Use adaptive tolerance
        tol = max_val * 0.01 + 1e-10
        active = np.where(conv_y >= max_val - tol)[0]

        g = subgradient_of_max_conv(y, active)

        x_new = project_simplex(x_new)

def averaged_subgradient(a_init, n_iter=20000, verbose=False):
    """
    Averaged subgradient method (Polyak's method).
    Minimize f(a) = max_k conv(a,a)[k] on the simplex.
    Uses Polyak step size with running average.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy().astype(np.float64))
    
    best_a = a.copy()
    best_f = np.max(np.convolve(a, a))
    
    # We need a lower bound on the optimal value for Polyak step size
    # Lower bound: sum(a)^2 / (2n-1) = 1/(2n-1) (since sum(a)=1)
    f_lb = 1.0 / (2 * n - 1)
    
    a_avg = a.copy()
    
    for t in range(1, n_iter + 1):
        conv_a = np.convolve(a, a)
        max_val = np.max(conv_a)
        
        # Active set
        tol = max_val * 0.05 + 1e-10
        active = np.where(conv_a >= max_val - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv_a)])
        
        g = subgradient_of_max_conv(a, active)
        
        # Polyak step size
        gamma = (max_val - f_lb) / (np.dot(g, g) + 1e-12)
        
        a_new = a - gamma * g
        a_new = project_simplex(a_new)
        
        # Running average
        a_avg = a_avg + (a_new - a_avg) / t
        
        # Track best
        f_new = np.max(np.convolve(a_new, a_new))
        if f_new < best_f:
            best_f = f_new
            best_a = a_new.copy()
        
        # Also check the average
        f_avg = np.max(np.convolve(a_avg, a_avg))
        if f_avg < best_f:
            best_f = f_avg
            best_a = a_avg.copy()
        
        a = a_new
        
        if verbose and t % 2000 == 0:
            obj_direct = compute_obj(a_new)
            obj_avg = compute_obj(a_avg)
            print(f"  AvgSG t={t}: f_new={f_new:.8f}, f_avg={f_avg:.8f}, best={best_f:.8f}")
    
    return best_a, best_f


def smooth_max_subgradient(a_init, n_iter=10000, nu=50.0, verbose=False):
    """
    Use smooth approximation of max: 
    f_nu(a) = (1/nu) * log(sum_k exp(nu * conv(a,a)[k]))
    This is smooth, and we can use gradient descent with momentum.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy().astype(np.float64))
    
    best_a = a.copy()
    best_f = np.max(np.convolve(a, a))
    
    # Adam optimizer parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    for t in range(1, n_iter + 1):
        conv_a = np.convolve(a, a)
        # For numerical stability, subtract max
        conv_shifted = conv_a - np.max(conv_a)
        weights = np.exp(nu * conv_shifted)
        w_sum = np.sum(weights)
        if w_sum == 0:
            weights = np.zeros(len(conv_a))
            weights[np.argmax(conv_a)] = 1.0
            w_sum = 1.0
        else:
            weights /= w_sum
        
        # Gradient of smooth max is weighted average of individual gradients
        # grad = sum_k weights[k] * grad_k
        # grad_k[m] = 2*a[k-m] if valid
        grad = np.zeros(n)
        for k_idx in range(len(conv_a)):
            if weights[k_idx] < 1e-10:
                continue
            k = k_idx
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start <= m_end:
                grad[m_start:m_end+1] += weights[k_idx] * 2.0 * a[k-m_end:k-m_start+1][::-1]
        
        # Adam update
        lr = 0.01 / (1 + 0.001 * t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s  # Projection onto simplex (approximate)
        
        # True projection
        a_new = project_simplex(a_new)
        
        f_new = np.max(np.convolve(a_new, a_new))
        if f_new < best_f:
            best_f = f_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and t % 1000 == 0:
            print(f"  SmoothMax t={t}: f={f_new:.8f}, best={best_f:.8f}, nu={nu}")
    
    return best_a, best_f

def lp_alternate(a_init, max_iter=200, verbose=False):
    """LP-based alternating optimization (improved from previous)."""
    from scipy.optimize import linprog
    n = len(a_init)
    a = project_simplex(a_init.copy().astype(np.float64))
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iter):
        conv_a = np.convolve(a, a)
        M = np.max(conv_a)
        
        # LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0
        # conv(a,g)[k] = sum_j a[j]*g[k-j]
        # This is: A @ g <= M * ones, where A[k,j] = a[j] if 0<=k-j<n
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            A[k, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
        
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Blend
        t_blend = 0.5
        a_new = (1 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = g
        else:
            a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 50 == 0:
            print(f"    LP it={it}: obj={best_obj:.6f}")
    
    return best_a, best_obj

"""
Convex optimization for step function coefficients.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated subgradient,
smooth-max gradient descent with Adam, LP alternation, and coordinate descent.
The core insight is that fixing sum(a)=1, the problem is convex (max of PSD quadratic forms),
enabling efficient subgradient methods with provable convergence.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_simplex(v):
    """Project v onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, active_indices):
    """
    Compute average subgradient of max_k conv(a,a)[k].
    grad_k[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in active_indices:
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            idx = np.arange(m_start, m_end + 1)
            src = k - idx
            grad[m_start:m_end + 1] += 2.0 * a[src]
    grad /= float(len(active_indices))
    return grad


def nesterov_subgrad(a_init, n_iter=5000, verbose=False):
    """
    Nesterov-accelerated subgradient on simplex.
    Minimize f(a) = max_k conv(a,a)[k] s.t. sum(a)=1, a>=0.
    """
    n = len(a_init)
    a = project_simplex(a_init.astype(np.float64))
    y = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_f = float(np.max(np.convolve(a, a)))

    L = 2.5
    for t in range(1, n_iter + 1):
        conv_y = np.convolve(y, y)
        max_val = float(np.max(conv_y))

        # Adaptive active set tolerance
        frac = 0.02 + 0.03 * (1.0 - t / float(n_iter))
        tol = max_val * frac + 1e-12
        active = np.where(conv_y >= max_val - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv_y))])

        g = compute_subgradient(y, active)

        alpha = 1.0 / (L * t)
        x_new = y - alpha * g
        x_new = project_simplex(x_new)

        f_new = float(np.max(np.convolve(x_new, x_new)))
        if f_new < best_f:
            best_f = f_new
            best_a = x_new.copy()

        beta = (t - 1.0) / (t + 1.0)
        y = x_new + beta * (x_new - x_prev)
        y = project_simplex(y)
        x_prev = x_new

        if verbose and t % 2000 == 0:
            print(f"    Nesterov t={t}: f={best_f:.8f}, obj={2.0*n*best_f:.6f}")

    return best_a, best_f


def smoothmax_adam(a_init, n_iter=8000, nu=50.0, lr0=0.05, verbose=False):
    """
    Gradient descent with Adam on smooth approximation of max.
    f_nu(a) = (1/nu) * log(sum_k exp(nu * conv_k(a)))
    Gradient is the softmax-weighted average of individual gradients.
    """
    n = len(a_init)
    a = project_simplex(a_init.astype(np.float64))
    best_a = a.copy()
    best_f = float(np.max(np.convolve(a, a)))

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for t in range(1, n_iter + 1):
        conv_a = np.convolve(a, a)
        conv_shifted = conv_a - np.max(conv_a)
        exp_vals = np.exp(nu * conv_shifted)
        w_sum = np.sum(exp_vals)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv_a))
            weights[int(np.argmax(conv_a))] = 1.0
        else:
            weights = exp_vals / w_sum

        grad = np.zeros(n)
        for ki in range(len(conv_a)):
            if weights[ki] < 1e-12:
                continue
            k = ki
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start <= m_end:
                idx = np.arange(m_start, m_end + 1)
                src = k - idx
                grad[m_start:m_end + 1] += weights[ki] * 2.0 * a[src]

        lr = lr0 / (1.0 + 0.0005 * t)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        f_new = float(np.max(np.convolve(a_new, a_new)))
        if f_new < best_f:
            best_f = f_new
            best_a = a_new.copy()

        a = a_new

        if verbose and t % 2000 == 0:
            print(f"    SmoothMax t={t}: f={best_f:.8f}, obj={2.0*n*best_f:.6f}")

    return best_a, best_f


def multiplicative_weights(a_init, n_iter=10000, eta0=0.1, verbose=False):
    """
    Multiplicative weights (mirror descent with entropy mirror) on simplex.
    a_j <- a_j * exp(-eta * g_j) / sum(...)
    """
    n = len(a_init)
    a = project_simplex(a_init.astype(np.float64))
    best_a = a.copy()
    best_f = float(np.max(np.convolve(a, a)))

    for t in range(1, n_iter + 1):
        conv_a = np.convolve(a, a)
        max_val = float(np.max(conv_a))

        tol = max_val * 0.03 + 1e-12
        active = np.where(conv_a >= max_val - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv_a))])

        g = compute_subgradient(a, active)

        eta = eta0 / (1.0 + 0.001 * t)
        a_new = a * np.exp(-eta * g)
        s = np.sum(a_new)
        if s < 1e-300:
            a_new = a
        else:
            a_new = a_new / s
        a_new = project_simplex(a_new)

        f_new = float(np.max(np.convolve(a_new, a_new)))
        if f_new < best_f:
            best_f = f_new
            best_a = a_new.copy()

        a = a_new

        if verbose and t % 2000 == 0:
            print(f"    MW t={t}: f={best_f:.8f}, obj={2.0*n*best_f:.6f}")

    return best_a, best_f


def lp_alternate(a_init, max_iter=100, verbose=False):
    """LP-based alternating optimization."""
    n = len(a_init)
    a = project_simplex(a_init.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(max_iter):
        conv_a = np.convolve(a, a)
        M = float(np.max(conv_a))

        # Build constraint matrix for LP: max sum(g) s.t. (a*g)_conv[k] <= M, g>=0
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status != 0:
            break

        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Try different blend factors
        best_local = best_obj
        best_local_a = best_a
        for t_blend in [0.3, 0.5, 0.7]:
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_local:
                best_local = obj
                best_local_a = a_new.copy()

        if best_local < best_obj:
            best_obj = best_local
            best_a = best_local_a.copy()
            a = best_local_a.copy()
        else:
            a_new = 0.5 * a + 0.5 * g
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s > 1e-10:
                a = a_new / s

        if verbose and it % 25 == 0:
            print(f"    LP it={it}: obj={best_obj:.6f}")

    return best_a, best_obj


def coord_descent(a_init, n_rounds=10, n_samples=40, verbose=False):
    """
    Coordinate descent: for each j, sweep a[j] over sample values,
    keeping others fixed, then renormalize.
    """
    n = len(a_init)
    a = project_simplex(a_init.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for rnd in range(n_rounds):
        any_improved = False
        for j in range(n):
            old_val = a[j]
            if old_val < 1e-8:
                # Try small positive values
                candidates = np.concatenate([
                    np.linspace(0, 0.05 / n, 10),
                    np.array([old_val])
                ])
            else:
                candidates = np.concatenate([
                    np.linspace(0, 2.5 * old_val, n_samples),
                    np.array([old_val])
                ])

            best_x = old_val
            best_local = np.inf

            for x in candidates:
                x = max(0.0, x)
                a[j] = x
                conv = np.convolve(a, a)
                local_max = float(np.max(conv))
                if local_max < best_local:
                    best_local = local_max
                    best_x = x

            a[j] = best_x

        # Renormalize
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            any_improved = True

        if verbose and rnd % 3 == 0:
            print(f"    CoordDescent rnd={rnd}: obj={obj:.6f}")

        if not any_improved and rnd > 3:
            break

    return best_a, best_obj


def diverse_init(n, rng):
    """Generate diverse initializations."""
    choice = int(rng.integers(0, 8))
    x = np.arange(n, dtype=np.float64)
    if choice == 0:
        freq = rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * x / n + phase))
    elif choice == 1:
        a = np.exp(-3.0 * (x / n - 0.5) ** 2)
    elif choice == 2:
        period = int(rng.integers(max(2, n // 20), max(3, n // 3)))
        pattern = np.abs(np.sin(np.arange(period) * np.pi / period)) + 0.1
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        mid = n // 2
        a = np.ones(n)
        a[:mid] = rng.uniform(0.3, 1.0)
        a[mid:] = rng.uniform(0.3, 1.0)
    elif choice == 4:
        a = np.abs(np.sin(x * rng.uniform(0.5, 2.0))) + 0.05
    elif choice == 5:
        a = 1.0 / (1.0 + np.exp(-(x - n * rng.uniform(0.3, 0.7)) / (n * 0.1)))
    elif choice == 6:
        a = (np.sin(x * np.pi / n) ** 2 + 0.1)
    else:
        a = rng.random(n) + 0.01

    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

        print(f"  [{time.monotonic()-t0:.1f}s]")

    # Phase 5: Coordinate descent
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        if n <= 400:
            print(f"\n[Phase 5] Coordinate descent, n={n}")
            t0 = time.monotonic()
            rounds = int(min(12, (deadline - time.monotonic()) / 3.0))
            if rounds > 2:
                a_out, _ = coord_descent(best_seq, n_rounds=rounds, n_samples=35, verbose=True)
                obj = compute_obj(a_out)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_out.copy()
                    print(f"  -> obj={obj:.6f}")
                else:
                    print(f"  -> obj={obj:.6f} (no improvement)")
            print(f"  [{time.monotonic()-t0:.1f}s]")

    # Phase 6: Try other sequence lengths
    if time.monotonic() < deadline - 120:
        print(f"\n[Phase 6] Explore other n values")
        n_tries = [200, 250, 350, 400, 500, 600]
        for n_try in n_tries:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 45:
                break

            print(f"  n={n_try} ({remaining:.0f}s available)")
            best_local = best_score
            best_local_seq = best_seq

            # Multiple restarts
            n_restarts = 2 if n_try <= 400 else 1
            for rst in range(n_restarts):
                if time.monotonic() > deadline - 30:
                    break
                a_init = diverse_init(n_try, rng)

                # Nesterov
                n_iter = int(min(6000, (deadline - time.monotonic()) / 0.4))
                if n_iter > 500:
                    a_out, _ = nesterov_subgrad(a_init, n_iter=n_iter)
                    obj = compute_obj(a_out)
                    if obj < best_local:
                        best_local = obj
                        best_local_seq = a_out.copy()
                        print(f"    Nesterov r{rst}: obj={obj:.6f}")

                # SmoothMax
                if time.monotonic() < deadline - 30:
                    n_iter2 = int(min(4000, (deadline - time.monotonic()) / 0.4))
                    if n_iter2 > 300:
                        a_out2, _ = smoothmax_adam(best_local_seq if best_local_seq is not None and len(best_local_seq) == n_try else a_init,
                                                    n_iter=n_iter2, nu=50.0, lr0=0.03)
                        obj2 = compute_obj(a_out2)
                        if obj2 < best_local:
                            best_local = obj2
                            best_local_seq = a_out2.copy()
                            print(f"    SmoothMax r{rst}: obj={obj2:.6f}")

                # LP
                if time.monotonic() < deadline - 20:
                    a_out3, _ = lp_alternate(best_local_seq if best_local_seq is not None and len(best_local_seq) == n_try else a_init, max_iter=40)
                    obj3 = compute_obj(a_out3)
                    if obj3 < best_local:
                        best_local = obj3
                        best_local_seq = a_out3.copy()
                        print(f"    LP r{rst}: obj={obj3:.6f}")

            if best_local < best_score:
                best_score = best_local
                best_seq = best_local_seq.copy()
                print(f"  New best: obj={best_score:.6f} at n={n_try}")

    # Phase 7: Final polishing
    if best_seq is not None and time.monotonic() < deadline - 5:
        print(f"\n[Phase 7] Final polishing, n={len(best_seq)}, obj={best_score:.6f}")
        remaining = deadline - time.monotonic()

        # One more Nesterov pass
        if remaining > 20:
            n_iter = int(min(5000, remaining / 0.15))
            if n_iter > 200:
                a_out, _ = nesterov_subgrad(best_seq, n_iter=n_iter)
                obj = compute_obj(a_out)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_out.copy()
                    print(f"  Final Nesterov: obj={obj:.6f}")

        # Coordinate descent
        n = len(best_seq)
        if n <= 350 and time.monotonic() < deadline - 5:
            rounds = int(min(8, max(1, (deadline - time.monotonic()) / 3.0)))
            if rounds > 0:
                a_out, _ = coord_descent(best_seq, n_rounds=rounds, n_samples=30)
                obj = compute_obj(a_out)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_out.copy()
                    print(f"  Final CoordDescent: obj={obj:.6f}")

        # SmoothMax one more
        if time.monotonic() < deadline - 5:
            n_iter = int(min(3000, max(100, (deadline - time.monotonic()) / 0.2)))
            if n_iter > 100:
                a_out, _ = smoothmax_adam(best_seq, n_iter=n_iter, nu=80.0, lr0=0.01)
                obj = compute_obj(a_out)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_out.copy()
                    print(f"  Final SmoothMax: obj={obj:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)
        print(f"Fallback used: n=100, obj={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
