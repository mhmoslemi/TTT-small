    for k in range(2 * n - 1):
        # Determine valid range of j for this k
        j_low = max(0, k - (n - 1))
        j_high = min(k, n - 1)
        for j in range(j_low + 1):
            i = int(k - j)
            # i is the index in f to use for weight in the inner product contribution
            row, col = k, j
            weight = f[i]
            rows.append(row)
            cols.append(col)
            vals.append(weight)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return rows, cols, vals

def evaluate_sequence(seq):
    """Evaluate a sequence using the given criteria."""
    if not isinstance(seq, list):
        return np.inf
    if not seq:
        return np.inf
    if not all(isinstance(x, float) for x in seq):
        return np.inf  # Only check for floats in the real function
    # Convert to numpy and ensure non-negative
    numpy_seq = np.array(seq, dtype=float)
    numpy_seq = np.clip(numpy_seq, 0, 1000)  # Clipping internally, but we'll do separately to preserve sum
    sum_a = np.sum(numpy_seq)
    if sum_a < 0.01:
        # Fallback by scaling to meet the sum requirement
        numpy_seq = np.clip(numpy_seq, 0, 1000)  # Ensure numbers are not too small and within bounds
        kernel = np.exp(-0.5 * (np.arange(-50) + 50)**2)
        smooth_seq = np.convolve(numpy_seq, kernel, mode='same')
        smooth_seq = np.abs(smooth_seq)
        smooth_seq = np.clip(smooth_seq, 0, 1000)
        numpy_seq = smooth_seq.copy() / 0.99  # Force sum to be 0.99
        # Adjust the fallback policy to require explicit computation
        sum_a = 0.99
    # Compute full convolution (mode='full')
    conv_result = np.convolve(numpy_seq, numpy_seq, mode='full')
    max_b = np.max(conv_result)
    n_val = float(len(numpy_seq))
    result = float(2 * n_val * max_b / (sum_a**2))
    return result

def objective_function(x):

def optimization_step(x):
    """Optimize the sequence using gradient descent."""
    x = np.array(x, dtype=float)
    best_score = float('inf')
    # Try both maximizing and minimizing directions based on the objective
    try:
        # Always ensure the sequence sum is at least 0.01
        sum_x = np.sum(x)
        if sum_x < 0.01:
            # If too small, scale to a fixed sum
            scale = 0.1 / sum_x if sum_x < 0.01 else 1.0
            x = x * scale
        
        # Evaluate current score
        current_score = evaluate_sequence(x.tolist())
        best_score = current_score
        best_x = x.copy()
        # Perturb the sequence and update the best
        # Define a step size based on the exponent of the sum
        step_size = 0.05 / (np.sqrt(np.abs(np.sum(x))) if np.any(x) else 1.0)
        step_size = min(0.05, step_size)  # Ensure it's a small perturbation
        
        # Try small perturbations in multiple directions
        v = np.random.randn(len(x))
        # We can either: a) Randomly select indices to perturb, or b) use a full optimization
        # To avoid infinite loops, we perturb randomly
        pertIdx = np.random.choice(len(x), int(0.1 * len(x)), replace=False)
        v[pertIdx] = 1.0
        v = min(step_size, 0.5) * v
        new_seq = np.clip(x + v, 0.0001, 1000.0)
        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < best_score:
            best_score = new_score
            best_x = new_seq
    
    except Exception as e:
        print(f"Error in optimization_step: {e}")
    
    return list(best_x), best_score
