"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using multiplicative
weight descent (mirror descent on simplex) with adaptive peak focusing, 
combined with subgradient methods and periodic restarts.
"""

import numpy as np
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def compute_grad_smoothing(a, weights):
    """Compute grad[j] = 2 * sum_k weights[k] * a[k-j] using FFT."""
    n = len(a)
    # We need: grad[j] = 2 * sum_k w[k] * a[k-j]
    # Let a_rev[i] = a[n-1-i]
    # Then sum_k w[k] * a[k-j] = (w * a_rev)[n-1-j] ... let me verify
    # (w * a_rev)[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-(m-k)]
    # = sum_k w[k] * a[n-1-m+k]
    # We want this to equal sum_k w[k] * a[k-j], so n-1-m+k = k-j => m = n-1+j
    # So grad[j] = 2 * (w * a_rev)[n-1+j]
    
    a_rev = a[::-1]
    size = 1
    while size < n + len(weights):
        size <<= 1
    fw = np.fft.rfft(weights, size)
    far = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * far, size)
    grad = 2.0 * conv[n-1 : 2*n-1]
    return grad

def multiplicative_optimize(a, max_iter=50000, eta_init=0.5, eta_decay=0.99995,
                            tau_init=0.5, verbose=False):
    """Mirror descent with multiplicative updates on the simplex."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a /= s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    eta = eta_init
    tau = tau_init
    stall = 0
    patience = 2000
    
    for it in range(max_iter):
        conv = compute_conv(a)
        current_max = float(np.max(conv))
        current_obj = 2.0 * n * current_max  # sum_a = 1
        
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                break
        
        # Smoothed weights
        max_conv = current_max
        shifted = (conv - max_conv) / tau
        shifted = np.clip(shifted, -500, 0)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
        
        # Gradient
        grad = compute_grad_smoothing(a, weights)
        
        # Center the gradient (for the simplex constraint)
        grad -= grad.mean()
        
        # Multiplicative update
        update = np.exp(-eta * grad)
        a = a * update
        a = np.maximum(a, 1e-30)
        a /= a.sum()
        
        # Decay
        eta *= eta_decay
        tau *= 0.9999
        tau = max(tau, 0.01)
        
        if verbose and it % 10000 == 0:
            print(f"    MultOpt it={it}, obj={current_obj:.6f}, best={best_obj:.6f}")
    
    return best_a, best_obj
