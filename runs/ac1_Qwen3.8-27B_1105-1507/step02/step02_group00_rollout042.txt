def lse_objective_and_grad(a, beta):
    n = len(a)
    conv = np.convolve(a, a)  # length 2n-1
    max_conv = np.max(conv)
    exp_conv = np.exp(beta * (conv - max_conv))
    p = exp_conv / np.sum(exp_conv)  # length 2n-1
    
    obj = (max_conv + np.log(np.sum(exp_conv))) / beta  # LSE value
    
    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    grad = np.zeros(n)
    for i in range(n):
        # p[i:i+n] * a -> dot product
        grad[i] = 2.0 * np.dot(p[i:i+n], a)
    
    return obj, grad

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def lse_grad(a, beta):
    """Gradient of (1/beta)*log(sum exp(beta*conv(a,a)))."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    exp_conv = np.exp(beta * (conv - max_conv))
    p = exp_conv / np.sum(exp_conv)
    
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(p[i:i+n], a)
    return grad

def lse_value(a, beta):
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    return (max_conv + np.log(np.sum(np.exp(beta * (conv - max_conv))))) / beta

def lse_gradient_descent(a, max_iter=2000, beta_start=5.0, beta_end=500.0, verbose=False):
    """LSE-smoothed gradient descent on the simplex."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        beta = beta_start * (beta_end / beta_start) ** (it / max(1, max_iter - 1))
        
        grad = lse_grad(a, beta)
        
        # Adaptive step size
        # Use the ratio of current LSE value to gradient norm
        lse_val = lse_value(a, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lse_val / (grad_norm * grad_norm) * 0.5
        step = min(step, 1.0)

        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 200 == 0:
            print(f"    LSE it={it}, beta={beta:.1f}, obj={best_obj:.6f}")
    
    return best_a, best_obj

"""
Optimizes step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LSE-smoothed gradient descent with beta-annealing, combined with
LP-based refinement, subgradient polishing, and diverse multi-scale search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    s = np.sum(v)
    if s < 1e-15:
        return np.ones(n) / n
    # Standard simplex projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def lse_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*conv_k))."""
    n = len(a)
    conv = np.convolve(a, a)  # length 2n-1
    max_conv = np.max(conv)
    exp_conv = np.exp(beta * (conv - max_conv))
    p = exp_conv / np.sum(exp_conv)
    
    # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    idx = np.arange(n)
    P = p[idx[:, None] + idx[None, :]]  # n x n Hankel matrix
    grad = 2.0 * P.dot(a)
    return grad, p, conv


def lse_anneal(a_init, n_stages=10, iters_per_stage=200,
               beta_start=5.0, beta_end=2000.0, verbose=False):
    """LSE-smoothed gradient descent with beta annealing."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    momenta = np.zeros(n)
    momentum = 0.85
    
    for stage in range(n_stages):
        frac = stage / max(1, n_stages - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        for it in range(iters_per_stage):
            grad, p, conv = lse_grad(a, beta)
            
            # Momentum
            momenta = momentum * momenta + (1.0 - momentum) * grad
            gnorm = np.linalg.norm(momenta)
            if gnorm < 1e-15:
                break
            
            # Adaptive step size
            # Scale step by the "flatness" of the distribution p
            max_p = np.max(p)
            step = 0.3 / (gnorm * np.sqrt(max_p + 1e-10))
            step = min(step, 2.0)
            
            a_new = a - step * momenta
            a_new = project_simplex(a_new)
            
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
            
            a = a_new
            
            if verbose and (stage * iters_per_stage + it) % 500 == 0:
                print(f"      stage={stage}, it={it}, beta={beta:.1f}, obj={best_obj:.6f}")
    
    return best_a, best_obj


def build_A(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_refine(a, max_iter=100, verbose=False):
    """LP-based alternating refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    A = build_A(a)  # Will be rebuilt when a changes
    
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        # Solve LP: max sum(g) s.t. A @ g <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Blend
        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new / s
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Rebuild A
        A = build_A(a)
        
        if verbose and it % 20 == 0:
            print(f"    LP it={it}, obj={best_obj:.6f}")
    
    return best_a, best_obj


def subgradient_polish(a, n_steps=2000, verbose=False):
    """Subgradient polishing with adaptive tolerance."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for step in range(n_steps):
        conv = np.convolve(a, a)
        max_conv = np.max(conv)
        if max_conv < 1e-15:
            break
        
        # Active set with adaptive tolerance
        tol = max_conv * (0.0005 + 0.01 * (step / n_steps))
        active = conv >= max_conv - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = conv >= max_conv - 1e-10
            n_active = int(np.sum(active))
        
        # Compute subgradient: average of active gradients
        # grad[i] = (2/n_active) * sum_{k active} a[k-i] * 1(0 <= k-i < n)
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            grad[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
        grad /= n_active
        
        # Step size
        step_size = 0.3 / (1.0 + 0.05 * step)
        
        a_new = a - step_size * grad
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def make_init(n, rng, variant=0):
    """Generate diverse initializations."""
    if variant == 0:
        # Cosine
        freq = rng.uniform(1.0, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif variant == 1:
        # Two-lobe (U-shaped)
        mid = n / 2.0
        sigma = rng.uniform(0.1, 0.3)
        a = np.exp(-((np.arange(n) - mid) / (sigma * n)) ** 2)
        # Add edge peaks
        a[0] += rng.uniform(0.5, 2.0)
        a[-1] += rng.uniform(0.5, 2.0)
    elif variant == 2:
        # Periodic
        period = int(rng.integers(max(3, n // 20), max(4, n // 3)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif variant == 3:
        # Gaussian window
        center = rng.uniform(0.3, 0.7)
        sigma = rng.uniform(0.05, 0.3)
        a = np.exp(-((np.arange(n) / n - center) / sigma) ** 2)
    elif variant == 4:
        # Inverse cosine (high at edges)
        a = 1.0 / (1.0 + np.exp(-rng.uniform(2, 8) * (2 * np.arange(n) / n - 1)))
    elif variant == 5:
        # Random
        a = rng.random(n)
    elif variant == 6:
        # Power law
        alpha = rng.uniform(0.5, 3.0)
        x = np.arange(1, n + 1) / n
        a = x ** alpha + (1 - x) ** alpha
    else:
        a = rng.random(n)
    
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LSE-smoothed gradient descent with beta-annealing, LP-based refinement,
subgradient polishing, and diverse multi-scale search across sequence lengths.

import numpy as np
from scipy.optimize import linprog

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    s = np.sum(v)
    if s < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def lse_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*conv_k)) and the weights."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    exp_conv = np.exp(beta * (conv - max_conv))
    p = exp_conv / np.sum(exp_conv)
    idx = np.arange(n)
    P = p[idx[:, None] + idx[None, :]]
    grad = 2.0 * P.dot(a)
    return grad, p, conv


def lse_anneal(a_init, n_stages=10, iters_per_stage=200,
               beta_start=5.0, beta_end=2000.0, verbose=False):
    """LSE-smoothed gradient descent with beta annealing and momentum."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    momentum_vec = np.zeros(n)
    mu = 0.85

    total_iters = n_stages * iters_per_stage
    global_it = 0

    for stage in range(n_stages):
        frac = stage / max(1, n_stages - 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        for it in range(iters_per_stage):
            grad, p, conv = lse_grad(a, beta)

            momentum_vec = mu * momentum_vec + (1.0 - mu) * grad
            gnorm = np.linalg.norm(momentum_vec)
            if gnorm < 1e-15:
                break

            # Adaptive step: scale by inverse gradient norm
            max_p = np.max(p)
            step = 0.15 / (gnorm * np.sqrt(max_p + 0.1))
            step = min(step, 1.0)
            # Decay with global iteration
            step *= 1.0 / (1.0 + 0.002 * global_it)

            a_new = a - step * momentum_vec
            a_new = project_simplex(a_new)

            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

            a = a_new
            global_it += 1

            if verbose and global_it % 500 == 0:
                print(f"      LSE stage={stage} it={it} beta={beta:.1f} obj={best_obj:.6f}")

    return best_a, best_obj


def build_A(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_refine(a, max_iter=100, verbose=False):
    """LP-based alternating refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)

def subgradient_polish(a, n_steps=2000, verbose=False):

    n = len(a)
