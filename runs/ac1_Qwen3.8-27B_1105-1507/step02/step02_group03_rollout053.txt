"""
Improved multi-phase optimization for step function coefficient minimization.
Key improvements: adaptive LP constraint, FFT-based subgradient, multi-restart,
and aggressive line search.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_score_normalized(f):
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def build_conv_matrix(f):
    n = len(f)
    # For position k in [0, 2n-2], (conv(f,g))[k] = sum_i f[i]*g[k-i]
    # = sum_i A[k,i] * g[i] where A[k,i] = f[k-i] if 0 <= k-i < n
    n_rows = 2 * n - 1
    A = np.zeros((n_rows, n))
    for i in range(n):
        # g[i] contributes to conv[k] for k in [i, i+n-1] with coefficient f[k-i]
        for k in range(i, i + n):
            A[k, i] = f[k - i]
    return A

def spectral_flat(f, n_iter=200, rng=None):
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    stall = 0
    
    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target flat magnitude
        target_mag = np.sqrt(n_fft / (len(mag) - 1))  # Keep DC separate
        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0
        
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        
        score = compute_score_normalized(f_new)
        if score < best_score - 1e-10:
            best_score = score
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        f = f_new
    
    return best_f, best_score

    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    stall = 0
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Active set: positions within 10% of max
        threshold = 0.9 * M
        active = np.where(conv >= threshold)[0]
        
        # Gradient: for each i, grad[i] = sum_{k in active} f[k-i] / |active|
        # This is convolve(f, indicator_active) but we need to be careful with indices
        indicator = np.zeros(n)
        for k in active:
            # f[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            indicator[i_start:i_end] += 1
        
        # Actually, the gradient of (f*f)[k] w.r.t. f[i] is f[k-i] (if valid)
        # So the subgradient of max is the average gradient over active set
        # grad[i] = (1/|active|) * sum_{k in active} f[k-i] * 1{0<=k-i<n}
        
        # This is: for each active k, add f[k-i] to grad[i] for valid i
        # Equivalently: grad = some convolution
        # Let's just compute it directly for correctness (n=512 is small enough)
        
        grad = np.zeros(n)
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            # f[k-i] for i in [i_start, i_end)
            # Let j = k-i, so j ranges from k-(i_end-1) to k-i_start
            # f[k-i] = f[j] where j = k-i
            # For i = i_start: j = k - i_start
            # For i = i_end-1: j = k - (i_end-1)
            js = k - np.arange(i_start, i_end)
            grad[i_start:i_end] += f[js]
        
        grad /= len(active)

        eta_eff = eta * (1.0 / (1.0 + 0.005 * it))
        f_new = f * np.exp(-eta_eff * grad / (grad.max() + 1e-10))
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)
        
        score = compute_score_normalized(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:

                f_pert = f * (1 + 0.1 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score_normalized(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score_normalized(f)
                    stall = 0
                else:
                    break
    
    return best_f, best_score

def lp_refine_adaptive(f, time_budget, rng, delta=0.02, lp_time_limit=2):
    """LP refinement with adaptive (tighter) M constraint."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    start = time.monotonic()
    t = 0.3
    A = None  # Cache the matrix
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Use tighter constraint
        M_tight = M * (1 - delta)
        
        if A is None:
            A = build_conv_matrix(f)
        else:
            # Update A since f changed
            A = build_conv_matrix(f)
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_tight)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time_limit})
        except Exception:
            break
        
        if not res.success or res.x is None:
            # Try with original M
            b_ub = np.full(2 * n - 1, M)
            try:
                res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presolve': True, 'time_limit': lp_time_limit})
            except Exception:
                break
            if not res.success or res.x is None:
                break
        
        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum
        
        # Line search
        improved = False
        for step in [t, t * 0.5, t * 0.25, t * 0.1]:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score_normalized(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.2)
                improved = True
                break
        
        if not improved:
            t *= 0.5
            if t < 0.01:
                # Try perturbation
                f_pert = f + rng.normal(0, 0.02, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                sc = compute_score_normalized(f_pert)
                if sc < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = sc
                    t = 0.2
                else:
                    # Try perturbation in direction of g
                    f_pert = f + 0.05 * (g - f) + rng.normal(0, 0.01, n)
                    f_pert = np.clip(f_pert, 0, None)
                    f_pert = normalize(f_pert)
                    sc = compute_score_normalized(f_pert)
                    if sc < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = sc
                        t = 0.1
                    else:
                        break
    
    return best_f, best_score

def coordinate_descent(f, n_sweeps=10, rng=None):
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score_normalized(f)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-10:
                continue
            best_val = original
            best_local = best_score
            
            for factor in [0.2, 0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0]:
                f[i] = max(0.0, original * factor)
                conv = np.convolve(f, f)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        
        f = np.clip(f, 0, None)
        f = normalize(f)
    
    return f, best_score

def simulated_annealing(f, rng, n_iter=5000, T0=0.03, T_end=0.0001):
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    cur_score = best_score
    no_improve = 0
    
    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()
        r = rng.random()
        
        if r < 0.25:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.3))
        elif r < 0.45:
            # Single element
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.04))
        elif r < 0.6:
            # Pair swap
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.8:
            # Gaussian bump
            center = int(rng.integers(0, n))
            width = max(2, int(n // 40))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.normal(0, 0.08) * kernel
        else:
            # Global reweight
            f_new *= np.exp(rng.normal(0, 0.2, n))
        
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score_normalized(f_new)
        
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 1500 and it > n_iter // 2:
            break
    
    return best_f, best_score

def diverse_init(n, rng):

Improved multi-phase optimization for step function coefficient minimization.
Uses adaptive LP constraint (tighter M), FFT-based pressure updates,
multi-restart strategy, and aggressive line search to beat previous bound.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score_normalized(f):
    """Compute score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        for k in range(i, i + n):
            A[k, i] = f[k - i]
    return A


def spectral_flat(f, n_iter=200, rng=None):
    """Iterative spectral flattening via zero-padded FFT."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        target_mag = np.sqrt(n_fft / max(1, len(mag) - 1))
        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0

                    best_score = compute_score_normalized(f)
                    stall = 0
                else:
                    break

    return best_f, best_score


def lp_refine_adaptive(f, time_budget, rng, delta=0.02, lp_time_limit=2, t0=0.3):
    """LP refinement with adaptive (tighter) M constraint and line search."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    start = time.monotonic()
    t = t0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        M_tight = M * (1.0 - delta)

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_tight)
        bounds = [(0, None)] * n

        res = None
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time_limit})
        except Exception:
            pass

        if res is None or not res.success or res.x is None:
            # Fall back to original M
            b_ub = np.full(2 * n - 1, M)
            try:
                res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presolve': True, 'time_limit': lp_time_limit})
            except Exception:
                break
            if res is None or not res.success or res.x is None:
                break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum

        improved = False
        for step in [t, t * 0.6, t * 0.3, t * 0.15, t * 0.05]:
            f_new = (1.0 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score_normalized(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.15)
                improved = True
                break

        if not improved:
            t *= 0.5
            if t < 0.008:
                # Perturb to escape
                f_pert = f + rng.normal(0, 0.015, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                sc = compute_score_normalized(f_pert)
                if sc < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = sc
                    t = 0.2
                else:
                    # Try direction toward g with small noise
                    f_pert = f + 0.1 * (g - f) + rng.normal(0, 0.01, n)
                    f_pert = np.clip(f_pert, 0, None)
                    f_pert = normalize(f_pert)
                    sc = compute_score_normalized(f_pert)
                    if sc < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = sc
                        t = 0.1
                    else:
                        break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, rng=None):
    """Coordinate descent with geometric value search."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score_normalized(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-10:
                continue
            best_val = original
            best_local = best_score

            for factor in [0.25, 0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0, 3.0]:
                f[i] = max(0.0, original * factor)
                conv = np.convolve(f, f)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def simulated_annealing(f, rng, n_iter=8000, T0=0.02, T_end=0.00005):
    """Simulated annealing with diverse structured moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / max(1, n_iter))
        f_new = f.copy()
        r = rng.random()

        if r < 0.25:
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.3))
        elif r < 0.45:
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.04))
        elif r < 0.6:
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.8:
            center = int(rng.integers(0, n))
            width = max(2, int(n // 40))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.normal(0, 0.06) * kernel
        else:
            f_new *= np.exp(rng.normal(0, 0.15, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score_normalized(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 2000 and it > n_iter // 2:
            break

    return best_f, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequence."""
    r = rng.random()
    if r < 0.2:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.4:
        f = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
    elif r < 0.6:
        f = np.abs(np.sin(np.pi * np.arange(n) / n * rng.uniform(1, 3))) + 0.1
    elif r < 0.8:
        f = np.ones(n)
        for _ in range(4):
            center = int(rng.integers(0, n))
            width = int(rng.integers(3, max(4, n // 4)))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / max(1.0, width ** 2))
            f *= (1.0 + 0.5 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05
    else:
        f = np.ones(n) * (1.0 + 0.1 * rng.normal(0, 1, n))

    f = np.clip(f, 0.01, None)
    f = normalize(f)
    f = f * (1.0 + 0.03 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def multi_scale_init(n_target, n_start=32, rng=None):
    """Optimize small sequence, upsample to target length."""
    if rng is None:
        rng = np.random.default_rng(0)

    f_small = np.ones(n_start) * (1 + 0.2 * rng.normal(0, 1, n_start))
    f_small = np.clip(f_small, 0.05, None)
    f_small = normalize(f_small)
    f_small, _ = pressure_update_fft(f_small, n_iter=100, eta=0.2, rng=rng)
    f_small, _ = spectral_flat(f_small, n_iter=80, rng=rng)

    x_small = np.linspace(0, 1, n_start)
    x_large = np.linspace(0, 1, n_target)
    f_large = np.interp(x_large, x_small, f_small)
    f_large = np.clip(f_large, 0.01, None)
    f_large = normalize(f_large)
    return f_large


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase pipeline starting from parent, with adaptive LP,
    pressure updates, coordinate descent, and simulated annealing.
    Also explores different sequence lengths if time permits.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0

    best_overall = None
    best_score_overall = np.inf

    # Load parent
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            parent = normalize(parent)
            parent_score = compute_score_normalized(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_overall = parent.copy()
            best_score_overall = parent_score

    print(f"Target: 1.5027, Current best: {best_score_overall:.6f}")
    print(f"Budget: {budget_s}s, Deadline in {deadline - time.monotonic():.0f}s")

    # === Main pipeline on parent ===
    if parent is not None:
        n = len(parent)
        f = parent.copy()

        # Phase 1: Spectral flattening
        remaining = deadline - time.monotonic()
        if remaining > 25:
            t_spec = min(15, remaining * 0.08)
            n_iter_spec = int(t_spec * 60)
            f, s = spectral_flat(f, n_iter=n_iter_spec, rng=rng)
            print(f"Phase 1 (spectral): {s:.6f}")
            if s < best_score_overall:
                best_score_overall = s
                best_overall = f.copy()

        # Phase 2: Pressure update
        remaining = deadline - time.monotonic()
        if remaining > 35:
            t_press = min(30, remaining * 0.12)
            n_iter_press = int(t_press * 60)
            f, s = pressure_update_fft(f, n_iter=n_iter_press, eta=0.15, rng=rng)
            print(f"Phase 2 (pressure): {s:.6f}")
            if s < best_score_overall:
                best_score_overall = s
                best_overall = f.copy()

        # Phase 3: LP refinement (adaptive, main workhorse)
        remaining = deadline - time.monotonic()
        if remaining > 80:
            t_lp = min(remaining * 0.35, 300)
            f, s = lp_refine_adaptive(f, t_lp, rng, delta=0.02, lp_time_limit=2, t0=0.3)
            print(f"Phase 3 (LP delta=0.02): {s:.6f}")
            if s < best_score_overall:
                best_score_overall = s
                best_overall = f.copy()

        # Phase 4: Second LP with different delta
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_lp2 = min(remaining * 0.2, 120)
            f, s = lp_refine_adaptive(best_overall, t_lp2, rng, delta=0.05, lp_time_limit=2, t0=0.2)
            print(f"Phase 4 (LP delta=0.05): {s:.6f}")
            if s < best_score_overall:
                best_score_overall = s
                best_overall = f.copy()

        # Phase 5: Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 25:
            n_sweeps = max(5, int(min(15, remaining * 0.25)))
            f_cd, s_cd = coordinate_descent(best_overall, n_sweeps=n_sweeps, rng=rng)
            print(f"Phase 5 (CD): {s_cd:.6f}")
            if s_cd < best_score_overall:
                best_score_overall = s_cd
                best_overall = f_cd.copy()

        # Phase 6: Simulated annealing
        remaining = deadline - time.monotonic()
        if remaining > 40:
            n_iter_sa = min(12000, int(remaining * 180))
            f_sa, s_sa = simulated_annealing(best_overall, rng, n_iter=n_iter_sa, T0=0.015, T_end=0.00003)
            print(f"Phase 6 (SA): {s_sa:.6f}")
            if s_sa < best_score_overall:
                best_score_overall = s_sa
                best_overall = f_sa.copy()

        # Phase 7: Final LP polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_lp3 = min(remaining * 0.4, 100)
            f_lp3, s_lp3 = lp_refine_adaptive(best_overall, t_lp3, rng, delta=0.01, lp_time_limit=2, t0=0.15)
            print(f"Phase 7 (LP final): {s_lp3:.6f}")
            if s_lp3 < best_score_overall:
                best_score_overall = s_lp3
                best_overall = f_lp3.copy()

        # Phase 8: Final CD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            f_cd2, s_cd2 = coordinate_descent(best_overall, n_sweeps=8, rng=rng)
            print(f"Phase 8 (CD final): {s_cd2:.6f}")
            if s_cd2 < best_score_overall:
                best_score_overall = s_cd2
                best_overall = f_cd2.copy()

    # === Try different lengths ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        for n_try in [768, 1024]:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            print(f"\n--- Exploring n={n_try} (remaining: {remaining:.0f}s) ---")

            f0 = diverse_init(n_try, rng)
            f1, s1 = spectral_flat(f0, n_iter=80, rng=rng)
            f2, s2 = pressure_update_fft(f1, n_iter=150, eta=0.15, rng=rng)
            print(f"  n={n_try} after pressure: {s2:.6f}")

            if s2 < best_score_overall:
                best_score_overall = s2
                best_overall = f2.copy()
                print(f"  New best from n={n_try}: {s2:.6f}")

            remaining = deadline - time.monotonic()
            if remaining > 30:
                t_lp = min(remaining * 0.4, 80)
                f3, s3 = lp_refine_adaptive(f2, t_lp, rng, delta=0.02, lp_time_limit=2, t0=0.3)
                print(f"  n={n_try} after LP: {s3:.6f}")
                if s3 < best_score_overall:
                    best_score_overall = s3
                    best_overall = f3.copy()

    # === Try multi-scale init on parent length ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and parent is not None:
        n = len(parent)
        print(f"\n--- Multi-scale restart on n={n} (remaining: {remaining:.0f}s) ---")
        f_ms = multi_scale_init(n, n_start=max(16, n // 16), rng=rng)
        f_ms1, s_ms1 = spectral_flat(f_ms, n_iter=60, rng=rng)
        f_ms2, s_ms2 = pressure_update_fft(f_ms1, n_iter=100, eta=0.15, rng=rng)
        print(f"  Multi-scale after pressure: {s_ms2:.6f}")

        if s_ms2 < best_score_overall:
            best_score_overall = s_ms2
            best_overall = f_ms2.copy()

        remaining = deadline - time.monotonic()
        if remaining > 20 and s_ms2 < best_score_overall + 0.01:
            t_lp_ms = min(remaining * 0.5, 60)
            f_ms3, s_ms3 = lp_refine_adaptive(f_ms2, t_lp_ms, rng, delta=0.02, lp_time_limit=2, t0=0.3)
            print(f"  Multi-scale after LP: {s_ms3:.6f}")
            if s_ms3 < best_score_overall:
                best_score_overall = s_ms3
                best_overall = f_ms3.copy()

    # === Final verification ===
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal: score={final_score:.6f}, n={len(final_list)}")
        return final_list

    print("Fallback")
    return [1.0] * 256
