"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Projected subgradient descent on max autocorrelation with momentum
2. LP best-response polishing at periodic intervals
3. Adaptive step sizes with backtracking
4. Multi-length exploration with resampling
5. Simulated annealing restarts from best-so-far

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J_normalized(a):
    """Compute max_b for normalized a (sum=1). J = 2*n*max_b."""
    b = fast_conv(a)
    return float(b.max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def subgradient_max_b(a):
    """
    Compute subgradient of max_k b_k w.r.t. a, where b = conv(a,a).
    Returns (gradient, max_b, active_set_size)
    """
    n = len(a)
    b = fast_conv(a)
    max_b = float(b.max())
    active = np.where(b >= max_b * (1 - 1e-12))[0]
    
    # g_j = (1/|S|) * sum_{k in S} 2*a_{k-j} for 0 <= k-j < n
    # Equivalently: g = (1/|S|) * 2 * sum_{k in S} circconv_indicator(a, k)
    # where circconv_indicator gives a_{k-j} for valid k-j
    
    g = np.zeros(n)
    for k in active:
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            idx = k - j
            if 0 <= idx < n:
                g[j] += 2.0 * a[idx]
    g /= len(active)
    
    return g, max_b, len(active)


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_descent(a, max_iters, deadline, tracker, use_lp_every=50, noise_level=0.0):
    """
    Main optimization loop: subgradient descent with momentum and periodic LP polish.
    """
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    
    # Momentum
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.99
    
    # Adaptive step size
    eta = 1.0 / np.sqrt(2 * n)
    best_max_b = compute_J_normalized(a)
    patience = 0
    max_patience = 30
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient
        g, max_b, n_active = subgradient_max_b(a)
        
        if max_b < best_max_b - 1e-14:
            best_max_b = max_b
            patience = 0
            # Occasionally update tracker
            if it % 20 == 0 or max_b < best_max_b + 1e-10:
                tracker.update(a)
        else:
            patience += 1
        
        # Momentum update
        v = beta1 * v + beta2 * g
        v_hat = v / (1 - beta1 ** (it + 1))
        
        # Step
        a_new = a - eta * v_hat
        
        # Add noise if specified
        if noise_level > 0:
            a_new += noise_level * np.random.standard_normal(n)
        
        # Project: non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new /= s
        
        a = a_new
        
        # Adaptive step size: decrease if no progress
        if patience > max_patience:
            eta *= 0.8
            patience = 0
        
        # Periodic LP polish
        if use_lp_every > 0 and it % use_lp_every == use_lp_every - 1:
            M = compute_J_normalized(a)
            g_lp = lp_best_response(a, M)
            if g_lp is not None:
                g_lp = np.maximum(g_lp, 0.0)
                g_lp_sum = g_lp.sum()
                if g_lp_sum > 1e-10:
                    g_lp /= g_lp_sum
                    # Blend
                    t_blend = 0.3
                    a = (1 - t_blend) * a + t_blend * g_lp
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    # Reset momentum after LP
                    v = np.zeros(n)
    
    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Random uniform
    inits.append(rng.uniform(0.2, 1.0, n))
    
    # Cosine waves
    for freq in [1, 2, 3, 4, 5, 7, 8]:
        inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t_arr))
    
    # Gaussian
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.1)
    
    # Triangle
    inits.append(1.0 - 2.0 * np.abs(t_arr - 0.5))
    
    # Two peaks
    inits.append(np.exp(-20.0 * (t_arr - 0.3) ** 2) + np.exp(-20.0 * (t_arr - 0.7) ** 2) + 0.05)
    
    # Sine squared
    inits.append(1.0 + 0.5 * np.sin(np.pi * t_arr) ** 2)
    
    # Random smooth (low-pass filtered noise)
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 10)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size / 3)) ** 2)
    kernel /= kernel.sum()
    inits.append(np.convolve(noise, kernel, mode='same') + 1.0)
    inits[-1] = np.maximum(inits[-1], 0.01)
    
    # Turyn-like: alternating blocks
    block = n // 8
    pattern = np.ones(block)
    for i in range(7):
        if i % 2 == 0:
            pattern = np.concatenate([pattern, -np.ones(block)])
        else:
            pattern = np.concatenate([pattern, np.ones(block)])
    pattern = pattern[:n]
    inits.append(1.0 + 0.4 * pattern)
    
    # Sparse spikes with smoothing
    n_spikes = max(3, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    k = np.ones(max(3, n // 32)) / max(3, n // 32)
    inits.append(np.convolve(sparse, k, mode='same') + 0.05)
    
    return inits

Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated projected subgradient descent on max autocorrelation
2. Periodic LP best-response polishing (sparse matrix)

5. Perturbation-based local search with annealing
6. Final targeted coordinate refinement on peak indices
Key advantages: O(n log n) per subgradient step (vs O(n^2) for LP),

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_max_b(a):
    """Compute max of self-convolution for normalized a."""
    b = fast_conv(a)
    return float(b.max()), b


def subgradient_max_b(a):
    """
    Compute subgradient of max_k b_k w.r.t. a where b = conv(a,a).
    db_k/da_j = 2*a_{k-j} if 0<=k-j<n, else 0.
    Subgradient = average over active (max) indices.
    Returns (gradient, max_b, active_indices)
    """
    n = len(a)
    b = fast_conv(a)
    max_b = float(b.max())
    # Use tolerance to include near-max indices for stability
    tol = max_b * 1e-10 + 1e-15
    active = np.where(b >= max_b - tol)[0]
    
    g = np.zeros(n)
    for k in active:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            # g[j] += 2 * a[k-j] for j in [jl, jh]
            idx = np.arange(jl, jh + 1)
            g[idx] += 2.0 * a[k - idx]
    g /= len(active)
    
    return g, max_b, active


def build_A_sparse(a):
    """Build sparse constraint matrix A[k,j] = a[k-j] for valid indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            idx = np.arange(jl, jh + 1)
            vals.extend(a[k - idx].tolist())
    A = lil_matrix((2 * n - 1, n))
    if rows:
        A[np.array(rows), np.array(cols)] = vals
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def nesterov_subgradient(a, max_iters, deadline, tracker, use_lp_every=200, noise_temp=0.0, rng=None):

    improvement_count = 0

        if time.monotonic() > deadline:
            break
        
        # Compute subgradient at y (Nesterov lookahead)
        g, max_b_y, active = subgradient_max_b(y)
        
        # Nesterov update
        t_k = 2.0 / (it + 2.0 + 6.0)  # Nesterov parameter
        new_a = a - eta * (g + t_k * (y - a))
        y = new_a  # Update lookahead
        
        # Project: non-negative
        new_a = np.maximum(new_a, 0.0)
        
        # Add noise
        if noise_temp > 1e-8:
            new_a += noise_temp * rng.standard_normal(n)
            new_a = np.maximum(new_a, 0.0)
        
        # Normalize
        s = new_a.sum()
        if s < 1e-10:
            new_a = np.ones(n) / n
        else:
            new_a /= s
        
        a = new_a
        y = a + (t_k / (t_k + 1.0)) * (a - new_a)  # simplified: just set y = a for stability
        y = a.copy()

        max_b_now = compute_max_b(a)[0]
        if max_b_now < best_max_b - 1e-13:
            best_max_b = max_b_now

                tracker.update(a)
        else:
            no_progress += 1

            eta *= 0.85
            no_progress = 0

            eta *= 1.02

                    a /= a.sum()
                    y = a.copy()
                    v = np.zeros(n)
                    # Reset progress counter
                    no_progress = 0
        
    tracker.update(a)
    return a

def generate_initializations(n, rng, parent):

    # Uniform
    inits.append(np.ones(n))
    
    # Random

    # Cosine waves (various frequencies)

        inits.append(1.0 + 0.6 * np.cos(2.0 * np.pi * freq * t_arr))
    
    # Sine squared
    inits.append(1.0 + 0.5 * np.sin(np.pi * t_arr) ** 2)
    
    # Gaussian bump
    inits.append(np.exp(-5.0 * (t_arr - 0.5) ** 2) + 0.15)
    
    # Two bumps
    inits.append(np.exp(-30.0 * (t_arr - 0.25) ** 2) + np.exp(-30.0 * (t_arr - 0.75) ** 2) + 0.1)
    
    # Triangle
    inits.append(1.0 - 1.5 * np.abs(t_arr - 0.5))
    
    # Random smooth
    noise = rng.standard_normal(n)
    ks = max(3, n // 8)
    kernel = np.exp(-0.5 * (np.arange(-ks, ks + 1) / max(1, ks // 3)) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(noise, kernel, mode='same')
    inits.append(np.maximum(smooth + 1.5, 0.01))
    
    # Block pattern (Turyn-like)
    block = max(1, n // 6)
    pattern = np.zeros(n)
    for i in range(n):
        pattern[i] = 1.0 if (i // block) % 2 == 0 else -1.0
    inits.append(1.0 + 0.5 * pattern)
    
    # Sparse with smoothing
    n_spikes = max(3, n // 20)
    positions = rng.choice(n, size=n_spikes, replace=False)

    sparse[positions] = 1.0
    k2 = max(3, n // 25)
    ker2 = np.ones(k2) / k2
    inits.append(np.convolve(sparse, ker2, mode='same') + 0.08)
    
    # Quadratic
    inits.append(1.0 + 0.8 * (2.0 * t_arr - 1.0) ** 2)
    
    # Log-cosine (Barker-like for real)
    inits.append(np.abs(np.cos(np.pi * t_arr * 3)) + 0.3)
    
    return inits

                break
            a0 = resample_seq(tracker.best, nl)
            a0 = np.maximum(a0, 0.001)
            a0 /= a0.sum()
            
            a = nesterov_subgradient(a0, 1000, deadline, tracker,
                                     use_lp_every=100, noise_temp=0.0, rng=rng)
        
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    
    return [float(x) for x in tracker.best]
