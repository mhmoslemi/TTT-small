"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Extended LP mirror descent with momentum and running-average tracking
2. Multi-scale log-sum-exp gradient polishing via L-BFGS-B
3. Multi-length exploration with adaptive focus
4. Simulated annealing restarts on LP best-response

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_array, csr_array
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_sparse_conv_matrix(n):
    """Build the structure (row, col indices) for the convolution constraint matrix.
    Values depend on a, but structure is fixed for given n."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def make_A(a, row_idx, col_idx):
    """Build sparse matrix with given structure and values from a."""
    n = len(a)
    # A[k, j] = a[k-j] where k-j is in [0, n-1]
    # From our construction, for row k and col j, k-j is the value index
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)])
    from scipy.sparse import coo_array
    A = coo_array((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A.tocsr()


def lp_best_response(a, M, row_idx, col_idx):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = make_A(a, row_idx, col_idx)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_smooth_objective(a, T):
    """Log-sum-exp smoothing of max(conv(a,a))."""
    b = fast_conv(a)
    b_max = b.max()
    shifted = b - b_max
    max_shifted = shifted.max()
    if max_shifted > 700:  # numerical stability
        shifted = shifted - max_shifted
        b_max += max_shifted
    logits = shifted / T
    logits = logits - logits.max()
    log_sum = b_max + T * (np.log(np.sum(np.exp(logits))) + logits.max())
    p = np.exp(logits) / np.sum(np.exp(logits))
    return log_sum, p


def lse_gradient(a, p):
    """Gradient of log-sum-exp(conv(a,a)/T) w.r.t. a."""
    # grad_i = 2 * sum_j a[j] * p[j+i]
    grad = 2.0 * np.correlate(a, p, mode='valid')
    return grad


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_lp_with_momentum(a, max_iters, deadline, row_idx, col_idx, tracker_update, momentum=0.3):
    """LP mirror descent with momentum and running average."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    
    v = np.zeros(n)  # momentum
    running_avg = a.copy()
    count = 0
    
    best_local = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, row_idx, col_idx)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive step size: 2/(it+2) is optimal for strongly convex
        # But our problem isn't strongly convex, so use 1/sqrt(it+1)
        t_step = 2.0 / (np.sqrt(it + 1) + 1)
        t_step = min(t_step, 0.5)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * (g - a)
        a_new = a + t_step * v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Update running average
        count += 1
        running_avg = running_avg + (a_new - running_avg) / count
        
        a = a_new
        
        # Track best
        if it % 50 == 0:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_local = a.copy()
            
            J_avg = compute_J(running_avg)
            if J_avg < best_J:
                best_J = J_avg
                best_local = running_avg.copy()
            
            tracker_update(a)
            tracker_update(running_avg)
    
    # Final checks
    J_final = compute_J(a)
    if J_final < best_J:
        best_local = a.copy()
    J_avg_final = compute_J(running_avg)
    if J_avg_final < best_J:
        best_local = running_avg.copy()
    
    tracker_update(best_local)

def lse_polish(a, deadline, tracker_update, n_T_stages=5):

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    T_start = n * 0.5  # start with large T (smooth)
    T_min = 0.01  # end with small T (sharp)
    
    T_values = np.geomspace(T_start, T_min, n_T_stages)
    
    for T in T_values:
        if time.monotonic() > deadline:
            break
        
        def objective(x):
            x = np.maximum(x, 0.0)
            s_x = x.sum()
            if s_x < 1e-10:
                return 1e10, np.zeros_like(x)
            x = x / s_x
            val, p = lse_smooth_objective(x, T)
            grad = lse_gradient(x, p)
            return val, grad

        def obj_wrapper(x):
            val, _ = objective(x)
            return val
        
        def grad_wrapper(x):
            _, grad = objective(x)
            return grad
        
        bounds = [(0.0, None)] * n
        res = minimize(obj_wrapper, a, jac=grad_wrapper, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-14, 'gtol': 1e-12})

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Extended LP mirror descent with momentum and running-average tracking
2. Multi-scale log-sum-exp gradient polishing via L-BFGS-B
3. Multi-length exploration with adaptive focus
4. Simulated annealing restarts on LP best-response
Key improvements over previous: longer LP runs, momentum, running averages,

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import coo_array
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_sparse_structure(n):
    """Build row/col index arrays for the convolution constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def make_A(a, row_idx, col_idx):
    """Build sparse matrix A[k,j] = a[k-j] using precomputed structure."""
    n = len(a)
    vals = a[row_idx - col_idx]
    A = coo_array((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A.tocsr()


def lp_best_response(a, M, row_idx, col_idx):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = make_A(a, row_idx, col_idx)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_smooth_value(a, T):
    """Compute log-sum-exp of conv(a,a)/T for numerical stability."""
    b = fast_conv(a)
    b_max = b.max()
    shifted = (b - b_max) / T
    shifted = shifted - shifted.max()
    log_sum = (b_max + T * shifted.max()) + T * np.log(np.sum(np.exp(shifted)))
    p = np.exp(shifted) / np.sum(np.exp(shifted))
    return log_sum, p


def lse_grad(a, p):
    """Gradient of LSE(conv(a,a)/T) with respect to a.
    grad_i = 2 * sum_j a_j * p_{j+i}"""
    return 2.0 * np.correlate(a, p, mode='valid')


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def run_lp_momentum(a, max_iters, deadline, row_idx, col_idx, tracker, momentum=0.25):
    """LP mirror descent with momentum and running average."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()

    v = np.zeros(n)
    running_avg = a.copy()
    count = 0
    best_local = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, row_idx, col_idx)
        if g is None:

        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        t_step = 1.0 / (np.sqrt(it + 1) * 0.5 + 1.0)
        t_step = min(t_step, 0.4)

        # Momentum
        v = momentum * v + (1.0 - momentum) * (g - a)
        a_new = a + t_step * v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        count += 1
        running_avg = running_avg + (a_new - running_avg) / count
        a = a_new

        if it % 100 == 0:
            J_cur = compute_J(a)
            J_avg = compute_J(running_avg)
            if J_cur < best_J:

            if J_avg < best_J:
                best_J = J_avg

def lse_polish(a, deadline, tracker, n_stages=6):
    """Polish using L-BFGS-B on LSE smoothing with geometrically decreasing T."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    T_start = float(n) * 0.3
    T_min = 0.005
    T_values = np.geomspace(T_start, T_min, n_stages)

    for T in T_values:
        if time.monotonic() > deadline:
            break

        val, p = lse_smooth_value(a, T)
        grad0 = lse_grad(a, p)

        bounds = [(0.0, None)] * n

        def _obj(x):
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-10:
                return 1e10
            x = x / sx
            v, _ = lse_smooth_value(x, T)
            return v

        def _grad(x):
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-10:
                return np.zeros_like(x)
            x = x / sx
            _, p2 = lse_smooth_value(x, T)
            return lse_grad(x, p2)

        res = minimize(_obj, a, jac=_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': 300, 'max_fun': 100000,
                                'ftol': 1e-15, 'gtol': 1e-13})

    inits.append(np.exp(-4.0 * (t - 0.3) ** 2) + np.exp(-4.0 * (t - 0.7) ** 2) + 0.05)
    inits.append(rng.uniform(0.2, 1.0, n))
    inits.append(1.0 - 1.5 * np.abs(t - 0.5) + 0.1)

    return [np.maximum(x, 0.001) for x in inits]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: LP mirror descent + LSE polishing + multi-length + SA."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        print("No parent, using uniform n=128", flush=True)

    n_parent = parent.size

    # === Phase 1: LP Mirror Descent (main workhorse) ===
    # Try parent length first, then a few others
    lengths = [n_parent, 128, 100, 150, 200, 192, 256]
    # Remove duplicates, sort by closeness to parent
    seen = set()
    unique_lengths = []
    for ln in lengths:
        if ln not in seen and ln >= 50 and ln <= 512:
            seen.add(ln)
            unique_lengths.append(ln)
    unique_lengths.sort(key=lambda x: abs(x - n_parent))

    for n in unique_lengths[:5]:  # At most 5 lengths
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 LP n={n} {'='*50}", flush=True)

        row_idx, col_idx = build_sparse_structure(n)
        inits = generate_inits(n, rng, parent)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 = a0 / s0

            # Parent gets the most iterations
            if idx == 0:
                max_iters = 8000
            elif idx <= 3:
                max_iters = 4000
            else:
                max_iters = 2000

            a = run_lp_momentum(a0, max_iters, deadline, row_idx, col_idx, tracker, momentum=0.25)

        print(f"  Phase1 n={n} done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 2: LSE Polishing ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 LSE Polish n={n} {'='*50}", flush=True)
        a_pol = lse_polish(tracker.best, deadline, tracker, n_stages=7)
        tracker.update(a_pol)
        print(f"  Phase2 done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: LP Refinement on polished ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 LP Refine n={n} {'='*50}", flush=True)
        row_idx, col_idx = build_sparse_structure(n)
        a = run_lp_momentum(tracker.best, 5000, deadline, row_idx, col_idx, tracker, momentum=0.1)
        tracker.update(a)
        print(f"  Phase3 done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Final LSE + LP ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 Final Polish {'='*50}", flush=True)
        a_fin = lse_polish(tracker.best, deadline, tracker, n_stages=5)
        tracker.update(a_fin)

        n = tracker.best.size
        if time.monotonic() < deadline:
            row_idx, col_idx = build_sparse_structure(n)
            a = run_lp_momentum(a_fin, 3000, deadline, row_idx, col_idx, tracker, momentum=0.05)
            tracker.update(a)
        print(f"  Phase4 done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: SA Restarts ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase5 SA Restarts n={n} {'='*50}", flush=True)
        row_idx, col_idx = build_sparse_structure(n)
        temperature = 0.08
        for r in range(40):
            if time.monotonic() > deadline:
                break
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, temperature, n)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            a = run_lp_momentum(a, 1000, deadline, row_idx, col_idx, tracker, momentum=0.15)
            temperature *= 0.96

        print(f"  Phase5 done in {time.monotonic()-t0:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
