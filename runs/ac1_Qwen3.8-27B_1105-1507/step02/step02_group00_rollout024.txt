def smooth_max_obj(a, tau):
    """Compute smooth approximation to max(conv(a,a)) and its gradient."""
    c = np.convolve(a, a)
    # For numerical stability, subtract max
    c_shifted = c - np.max(c)
    exp_c = np.exp(tau * c_shifted)
    log_sum = np.log(np.sum(exp_c)) / tau + np.max(c)
    p = exp_c / np.sum(exp_c)
    # Gradient: 2 * correlate(p, a, 'full')[:n]
    grad = 2.0 * np.correlate(p, a, mode='full')[:len(a)]
    return log_sum, p, grad

Optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-based alternating optimization for fast initial convergence
- Log-sum-exp smooth max with mirror descent for fine polishing

from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A_mat(a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    return A


def smooth_max_and_grad(a, tau):
    """Compute smooth max of conv(a,a) and gradient w.r.t. a."""
    c = np.convolve(a, a)
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_c = np.exp(tau * c_shifted)
    sum_exp = np.sum(exp_c)
    p = exp_c / sum_exp
    # grad[i] = 2 * sum_k p[k] * a[k-i] (valid indices)
    # = 2 * correlate(p, a, 'full')[:n]
    grad = 2.0 * np.correlate(p, a, mode='full')[:len(a)]
    return c_max + np.log(sum_exp) / tau, p, grad


def mirror_descent(a, max_steps=5000, tau_start=5.0, tau_end=500.0, lr_start=0.1, lr_end=0.01, verbose=False):
    """Mirror descent with smooth max, annealing tau and learning rate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for step in range(max_steps):
        frac = step / max(max_steps - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        c_max, p, grad = smooth_max_and_grad(a, tau)
        
        # Mirror descent: a[i] *= exp(-lr * grad[i]), then normalize
        # Subtract max grad for numerical stability
        grad_shifted = grad - np.max(grad)
        a_new = a * np.exp(-lr * grad_shifted)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Track best
        if step % 10 == 0:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose and step % 100 == 0:
                    print(f"    MD step={step}: obj={best_obj:.6f}, tau={tau:.1f}")
        
        a = a_new
    
    # Final check
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj


def lp_step(a, M_multiplier=1.0):
    """One LP step: find best g given current a."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv)) * M_multiplier
    A = build_A_mat(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternate(a, max_iter=100, blend=0.5, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        g = lp_step(a)
        if g is None:
            break
        g_s = np.sum(g)
        if g_s < 1e-10:
            break
        g = g / g_s
        
        # Try different blends

            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_a = a_try.copy()
        
        a = (1 - blend) * a + blend * g
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    
    return best_a, best_obj


def coord_descent(a, n_rounds=2, verbose=False):

I'll introduce a **log-sum-exp smooth max with mirror descent (multiplicative weight updates)** as the core polishing method, combined with LP alternation and coordinate descent. The smooth max provides a differentiable surrogate that anneals from a global view (small τ) to a sharp peak-focusing view (large τ), allowing escape from local minima that pure LP gets stuck in.

Key improvements:
1. **Smooth max gradient** via `2 * correlate(p, a, 'valid')` where `p ∝ exp(τ·conv)`
2. **Mirror descent** (multiplicative weights) naturally maintains positivity
3. **τ-annealing** from smooth to sharp for progressive refinement
4. **Upsampling initialization** from smaller-n solutions to larger-n
5. **More diverse initializations** and n values up to 500

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-based alternating optimization for initial convergence
- Log-sum-exp smooth max with mirror descent (multiplicative weight updates) and tau-annealing
- Coordinate descent for final refinement
- Multiple sequence lengths, diverse initializations, and upsampled starts
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A_mat(a):
    """Build (2n-1) x n constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def smooth_max_and_grad(a, tau):
    """
    Compute smooth max (log-sum-exp) of conv(a,a) and gradient w.r.t. a.
    Returns (smooth_max_value, probability_vector_p, gradient).
    """
    c = np.convolve(a, a)
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_c = np.exp(tau * c_shifted)
    sum_exp = np.sum(exp_c)
    p = exp_c / sum_exp
    # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j] = 2 * correlate(p, a, 'valid')[i]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    smooth_val = c_max + np.log(sum_exp) / tau
    return smooth_val, p, grad


def mirror_descent_polish(a, max_steps, tau_start, tau_end, lr_start, lr_end,
                          check_interval=20, verbose=False, verbose_interval=500):
    """
    Mirror descent (multiplicative weight updates) with log-sum-exp smooth max.
    Anneals tau (smooth->sharp) and learning rate (large->small).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    tau_ratio = tau_end / max(tau_start, 1e-6)
    lr_ratio = lr_end / max(lr_start, 1e-10)

    for step in range(max_steps):
        frac = step / max(max_steps - 1, 1)
        tau = tau_start * (tau_ratio ** frac)
        lr = lr_start * (lr_ratio ** frac)

        smooth_val, p, grad = smooth_max_and_grad(a, tau)

        # Mirror descent update with stability shift
        grad_max = np.max(grad)
        a_new = a * np.exp(-lr * (grad - grad_max))
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        a = a_new

        # Periodically check actual objective
        if (step + 1) % check_interval == 0 or step == max_steps - 1:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                if verbose and (step + 1) % verbose_interval == 0:
                    print(f"      MD step={step+1}/{max_steps}: obj={best_obj:.6f}, tau={tau:.1f}, lr={lr:.5f}")

    # Final evaluation
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def lp_one_step(a):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    conv_aa = np.convolve(a, a)
    M = float(np.max(conv_aa))
    if M < 1e-15:
        return None
    A = build_A_mat(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternate(a, max_iter, blend=0.5, verbose=False):
    """LP alternating optimization with multiple blend factors."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(max_iter):
        g = lp_one_step(a)
        if g is None:
            break
        g_s = np.sum(g)
        if g_s < 1e-12:
            break
        g = g / g_s

        # Try multiple blend factors
        for t in [blend, 0.3, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            ss = np.sum(a_try)
            if ss < 1e-12:
                continue
            a_try = a_try / ss
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_a = a_try.copy()

        a = (1.0 - blend) * a + blend * g
        a = np.maximum(a, 0.0)
        ss = np.sum(a)
        if ss < 1e-12:
            a = g
        else:
            a = a / ss

    return best_a, best_obj


def coord_descent(a, n_rounds=2):
    """Coordinate descent with multiple trial multipliers per coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    multipliers = [0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0]

    for rnd in range(n_rounds):
        for j in range(n):
            old = a[j]
            if old < 1e-15:
                # Try setting to small positive values
                for x in [0.0, old + 0.01 / n, 0.05 / n]:
                    a[j] = x
                    obj = compute_obj(a)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                a[j] = old
                continue
            best_x = old
            best_local = compute_obj(a)
            for mult in multipliers:
                x = old * mult
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x
            a[j] = best_x

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 10))
    x = np.arange(n, dtype=np.float64)
    if choice == 0:
        freq = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0.01, 1.0 + 0.8 * np.cos(2 * np.pi * freq * x / n + phase))
    elif choice == 1:
        # Two-lobe
        a = np.ones(n)
        mid = n // 2
        a[:mid] *= rng.uniform(0.6, 1.4)
        a[mid:] *= rng.uniform(0.6, 1.4)
    elif choice == 2:
        # Periodic
        period = int(rng.integers(max(2, n // 50), max(3, n // 4)))
        pattern = 0.5 + 0.5 * rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        # Gaussian bell
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.15, 0.45) * n
        a = np.exp(-0.5 * ((x - center) / width) ** 2) + 0.01
    elif choice == 4:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(int(rng.integers(2, 5))):
            f = rng.uniform(0.5, 6.0)
            ph = rng.uniform(0, 2 * np.pi)
            a = a * (1.0 + 0.25 * np.cos(2 * np.pi * f * x / n + ph))
            a = np.maximum(a, 0.01)
    elif choice == 5:
        # Linear ramp
        a = np.linspace(rng.uniform(0.4, 1.6), rng.uniform(0.4, 1.6), n)
    elif choice == 6:
        # Random smooth
        a = 0.5 + 0.5 * rng.random(n)
        # Smooth with moving average
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
    elif choice == 7:
        # Concentrated center
        a = np.full(n, 0.3)
        center = n // 2
        spread = int(n * rng.uniform(0.15, 0.35))
        a[max(0, center - spread):min(n, center + spread)] = rng.uniform(1.0, 2.5)
    elif choice == 8:
        # Quadratic (U-shape or inverted)
        t = x / max(n - 1, 1) - 0.5
        a = 1.0 + rng.uniform(0.5, 2.0) * (t ** 2) * rng.choice([1, -1])
        a = np.maximum(a, 0.01)
    else:
        # Pure random
        a = 0.3 + 0.7 * rng.random(n)

    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n) + 0.1
        s = np.sum(a)
    return a / s


def upsample(a, new_n):
    """Upsample a sequence to new_n by linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return np.interp(new_x, old_x, a)
