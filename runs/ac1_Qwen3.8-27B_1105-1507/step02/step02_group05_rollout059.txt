```python
def subgradient_descent(a, n_steps, deadline):
    a = a.copy()
    s = a.sum()
    a = a / s  # normalize to sum=1
    
    best = a.copy()
    best_val = fast_conv(a).max()
    
    for t in range(1, n_steps+1):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_star = int(b.argmax())
        
        # Subgradient: sg_i = 2*a[k*-i] for valid i
        sg = np.zeros_like(a)
        for i in range(len(a)):
            j = k_star - i
            if 0 <= j < len(a):
                sg[i] = 2.0 * a[j]
        
        # Step size (Polyak or decreasing)
        lr = 1.0 / np.sqrt(t)
        # Scale lr to be reasonable
        lr *= best_val  # scale by current objective value
        
        a_new = a - lr * sg
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        val = fast_conv(a_new).max()
        if val < best_val:
            best_val = val
            best = a_new.copy()
        
        a = a_new
    
    return best, best_val
```

At `k = k*` (where `c_k(a)` is max):
`= (1-t)^2 M + 2t(1-t)(a*g)_{k*} + t^2 c_{k*}(g)`

The LP ensures `(a*g)_{k*} <= M`. And `c_{k*}(g) <= max_k c_k(g)`.

So `c_{k*}(new) <= (1-t)^2 M + 2t(1-t)M + t^2 * max c_k(g) = M - 2t(1-t)M + t^2 * max c_k(g)`

For this to be less than M, we need `t^2 * max c_k(g) < 2t(1-t)M`, i.e., `t * max c_k(g) < 2(1-t)M`.

The gradient of `logsumexp(beta * c_k) / beta` with respect to `a_i` is:
`(1/beta) * sum_k (beta * exp(beta * c_k) / sum_j exp(beta * c_j)) * dc_k/da_i`
`= sum_k softmax_k * dc_k/da_i`

Where `softmax_k = exp(beta * c_k) / sum_j exp(beta * c_j)`.

This is a "soft" subgradient that weights all constraints by their proximity to the max. For large beta, it concentrates on the argmax.

This could be more effective than the hard subgradient because it considers "near-max" constraints too.

OK this is getting too complicated in my head. Let me just use a practical approach.

For the soft gradient, I can compute it as follows:
- `soft_sg[i] = 2 * sum_k w[k] * a[k-i]` (valid indices)
- This is equivalent to the cross-correlation of `w` and `a` at offset `i`
- I can compute this using FFT: `cross_corr = ifft(fft(w, N) * conj(fft(a, N)))` where `N >= 4n-2`
- The result at index `i + (2n-2)` gives the cross-correlation at offset `i`... 

Actually, let me just use `np.correlate` or compute it via the convolution trick:

`sum_k w[k] * a[k-i]` for `i = 0, ..., n-1`:
- Let `a_rev = a[::-1]`
- Then `sum_k w[k] * a[k-i] = sum_k w[k] * a_rev[n-1-(k-i)] = sum_k w[k] * a_rev[n-1-k+i]`
- `= (w * a_rev)[n-1+i]` (where `*` is convolution)

So `soft_sg[i] = 2 * (convolve(w, a_rev))[n-1+i]`

This is O(n log n) via FFT.

Let me now write the complete algorithm:

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Analytic initialization with parameter sweeps
2. Fast soft-subgradient descent with log-sum-exp smoothing
3. LP-based mirror descent for polishing
4. L-BFGS-B gradient refinement
5. Multi-length exploration
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s*s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k-jh:k-jl+1][::-1].tolist())
    A = lil_matrix((2*n-1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def soft_subgradient_descent(a, max_iters, deadline, beta_init=10, beta_final=100):
    """Fast subgradient method with log-sum-exp smoothing."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    
    best = a.copy()
    best_val = fast_conv(a).max()
    
    a_rev = a[::-1].copy()  # will update each iteration
    
    for t in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        
        # Anneal beta
        beta = beta_init + (beta_final - beta_init) * (t / max_iters)
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Soft weights (log-sum-exp)
        log_w = beta * (b - b_max)
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2*n-1)
            w[int(b.argmax())] = 1.0
        else:
            w /= w_sum
        
        # Soft gradient: sg[i] = 2 * sum_k w[k] * a[k-i]
        # = 2 * (convolve(w, a_rev))[n-1+i]
        N = 1
        while N < 4*n:
            N <<= 1
        W = np.fft.rfft(w, N)
        A = np.fft.rfft(a_rev, N)
        conv = np.fft.irfft(W * A, N)
        sg = 2.0 * conv[n-1 : n-1+n]
        
        # Step size
        # Use Polyak-like step: lr = (f(a) - f*) / ||sg||^2
        # We don't know f*, so use heuristic
        sg_norm = np.sqrt(np.sum(sg**2))
        if sg_norm < 1e-15:
            break
        lr = 1.0 / (np.sqrt(t) * sg_norm)
        # Scale by current objective to make it dimensionless
        lr *= b_max * 0.5
        
        # Update
        a_new = a - lr * sg
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        # Check improvement
        b_new = fast_conv(a_new)
        val = b_new.max()
        if val < best_val:
            best_val = val
            best = a_new.copy()
        
        a = a_new
        a_rev = a[::-1].copy()
    
    return best, best_val

def lp_mirror_descent(a, max_iters, deadline):
    """LP-based mirror descent."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    
    best = a.copy()
    best_val = fast_conv(a).max()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g = g / g.sum()
        
        # Adaptive step size
        base_t = max(0.05, 0.4 * (1.0 - t / (2.0 * max_iters)))
        
        best_a = None
        best_J = best_val
        for t_step in [base_t, base_t * 0.5, base_t * 1.5]:
            a_trial = (1.0 - t_step) * a + t_step * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = fast_conv(a_trial).max()
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            break
        
        a = best_a
        if best_J < best_val:
            best_val = best_J
            best = a.copy()
    
    return best, best_val

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    
    # Cosine sweeps
    for freq in [1, 2, 3, 4, 5, 8, 12]:
        for amp in [0.2, 0.5, 0.8]:
            inits.append(1.0 + amp * np.cos(2*np.pi*freq*t))
    
    # Gaussian
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            inits.append(np.exp(-((t-center)/width)**2) + 0.05)
    
    # Power law
    for p in [-0.5, -0.2, 0.2, 0.5]:
        inits.append((t + 0.01)**p)
    
    # Triangle / tent
    inits.append(1.0 - np.abs(2*t - 1))
    inits.append(np.abs(2*t - 1))
    
    # Two peaks
    inits.append(np.exp(-((t-0.25)/0.1)**2) + np.exp(-((t-0.75)/0.1)**2) + 0.05)
    
    # Alternating
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.3)
    inits.append(alt)
    
    # Sparse spikes
    for n_spikes in [4, 8, 16]:
        if n_spikes <= n:
            sparse = np.zeros(n)
            positions = rng.choice(n, size=n_spikes, replace=False)
            sparse[positions] = 1.0
            inits.append(sparse + 0.02)
    
    return inits
```

```python
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_precomputed(n):
    """Precompute sparsity structure for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def build_A_values(a, rows, cols):
    """Fill in values for the precomputed structure."""
    n = len(a)
    vals = np.zeros(len(rows), dtype=np.float64)
    for idx in range(len(rows)):
        k = rows[idx]
        j = cols[idx]
        if 0 <= k - j < n:
            vals[idx] = a[k - j]
    return vals


def lp_best_response(a, M, rows, cols):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    vals = build_A_values(a, rows, cols)
    from scipy.sparse import coo_matrix
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def soft_subgradient_descent(a, max_iters, deadline, beta_init=5.0, beta_end=80.0):
    """Fast subgradient method with log-sum-exp smoothing and momentum."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)

    best = a.copy()
    best_val = fast_conv(a).max()

    # Momentum
    v = np.zeros(n)
    momentum = 0.9

    for t in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        # Anneal beta
        frac = min(1.0, t / max_iters)
        beta = beta_init + (beta_end - beta_init) * frac * frac

        b = fast_conv(a)
        b_max = b.max()

        # Soft weights via log-sum-exp
        shifted = beta * (b - b_max)
        # Clip for numerical safety
        shifted = np.clip(shifted, -50, 0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(b.argmax())] = 1.0
        else:
            w /= w_sum

        # Soft gradient: sg[i] = 2 * sum_k w[k] * a[k-i] (valid)
        # = 2 * (convolve(w, a_rev))[n-1+i]
        a_rev = a[::-1]
        N_fft = 1
        while N_fft < (2 * n - 1) + n:
            N_fft <<= 1
        W = np.fft.rfft(w, N_fft)
        Ar = np.fft.rfft(a_rev, N_fft)
        conv_res = np.fft.irfft(W * Ar, N_fft)
        sg = 2.0 * conv_res[n - 1:2 * n - 1]

        # Ensure non-negative gradient direction (we want to decrease peak)
        sg_norm = np.sqrt(np.sum(sg ** 2))
        if sg_norm < 1e-15:
            break

        # Step size: proportional to objective / gradient norm
        lr = 0.5 * b_max / (sg_norm + 1e-10)
        # Damp early iterations
        lr *= min(1.0, t / 50.0)

        # Momentum update
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        # Evaluate
        b_new = fast_conv(a_new)
        val = b_new.max()
        if val < best_val:
            best_val = val
            best = a_new.copy()

        a = a_new

    return best, best_val


def lp_mirror_descent(a, max_iters, deadline, rows, cols):
    """LP-based mirror descent with adaptive step size and restarts."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)

    best = a.copy()
    best_val = fast_conv(a).max()
    stall_count = 0
    max_stall = 10

    for t in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        b_max_val = M

        g = lp_best_response(a, M, rows, cols)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size: try multiple values
        base_t = max(0.08, 0.5 * (1.0 - t / float(2.0 * max_iters)))
        step_candidates = [base_t, base_t * 0.5, base_t * 1.5, base_t * 0.25]

        best_a = None
        best_trial_val = best_val

        for t_step in step_candidates:
            a_trial = (1.0 - t_step) * a + t_step * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            trial_val = fast_conv(a_trial).max()
            if trial_val < best_trial_val:
                best_trial_val = trial_val
                best_a = a_trial.copy()

        if best_a is None:
            stall_count += 1
            if stall_count >= max_stall:
                # Perturb and restart
                a = a + 0.05 * np.random.default_rng(t).normal(0, 1, n)
                a = np.maximum(a, 0.001)
                a = a / a.sum()
                stall_count = 0
                continue
            break
        else:
            a = best_a
            stall_count = 0
            if best_trial_val < best_val:
                best_val = best_trial_val
                best = a.copy()

    return best, best_val


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    # Uniform
    inits.append(np.ones(n))

    # Random smooth
    rough = rng.uniform(0.3, 1.0, n)
    kernel = np.ones(5) / 5.0
    inits.append(np.convolve(rough, kernel, mode='same'))

    # Cosine sweeps
    for freq in [1, 2, 3, 4, 5, 6, 8, 12]:
        for amp in [0.3, 0.6, 0.9]:
            inits.append(1.0 + amp * np.cos(2 * np.pi * freq * t))

    # Gaussian bumps
    for center in [0.25, 0.5, 0.75]:
        for width in [0.08, 0.15, 0.25]:
            inits.append(np.exp(-((t - center) / width) ** 2) + 0.03)

    # Two peaks
    inits.append(np.exp(-((t - 0.2) / 0.1) ** 2) + np.exp(-((t - 0.8) / 0.1) ** 2) + 0.03)
    inits.append(np.exp(-((t - 0.3) / 0.15) ** 2) + np.exp(-((t - 0.7) / 0.15) ** 2) + 0.05)

    # Power law
    for p in [-0.5, -0.3, 0.3, 0.5]:
        inits.append((t + 0.01) ** p)

    # Triangle
    inits.append(1.0 - np.abs(2 * t - 1))
    inits.append(np.abs(2 * t - 1) + 0.1)

    # Alternating (smoothed)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.4)
    kernel3 = np.ones(3) / 3.0
    inits.append(np.convolve(alt, kernel3, mode='same'))

    # Chirp
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * (3 * t + t ** 2)))

    # Sparse
    for n_spikes in [max(4, n // 32), max(8, n // 16)]:
        if n_spikes <= n:
            sparse = np.zeros(n)
            positions = rng.choice(n, size=min(n_spikes, n), replace=False)
            sparse[positions] = 1.0
            kernel5 = np.ones(5) / 5.0
            inits.append(np.convolve(sparse, kernel5, mode='same') + 0.02)

    return inits


class ScoreTracker:

            shifted = beta_ls * (b - b_max)
            shifted = np.clip(shifted, -50, 0)
            return float(b_max + np.log(np.sum(np.exp(shifted))) / beta_ls)

        def lbb_gradient(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros_like(x)
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            beta_ls = 30.0
            shifted = beta_ls * (b - b_max)
            shifted = np.clip(shifted, -50, 0)
            w = np.exp(shifted)
            w /= w.sum()
            # Gradient of logsumexp + b_max approx: sum_k w_k * dc_k/da_i
            # dc_k/da_i = 2*a[k-i] (valid)
            # So grad_i = 2 * sum_k w[k] * a[k-i] = 2*(conv(w, a_rev))[n-1+i]
            a_rev = ax[::-1]
            N_fft = 1
            while N_fft < (2 * n - 1) + n:
                N_fft <<= 1
            W = np.fft.rfft(w, N_fft)
            Ar = np.fft.rfft(a_rev, N_fft)
            conv_res = np.fft.irfft(W * Ar, N_fft)
            grad = 2.0 * conv_res[n - 1:2 * n - 1]
            return grad

        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(lbb_objective, tracker.best.copy(), jac=lbb_gradient,
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 400, 'max_fun': 60000, 'ftol': 1e-18})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        print(f"  L-BFGS-B done ({time.monotonic() - t0:.1f}s)", flush=True)

    # === Phase 5: Multi-length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        print("\n" + "=" * 50 + " Phase 5: Multi-length " + "=" * 50, flush=True)

        target_lengths = [128, 192, 256, 384, 512, 768, 1024, 1536, 2048, 3072, 4096]
        for target_n in target_lengths:
            if time.monotonic() > deadline:
                break
            if target_n == tracker.best.size:
                continue

            t0 = time.monotonic()
            a_res = resample_seq(tracker.best, target_n)
            a_res = np.maximum(a_res, 0.001)
            a_res = a_res / a_res.sum()

            # Quick subgradient
            max_sg = max(200, min(2000, 300000 // target_n))
            a_sg, _ = soft_subgradient_descent(a_res, max_sg, deadline,
                                               beta_init=5.0, beta_end=40.0)
            tracker.update(a_sg)

            # Short LP
            if time.monotonic() < deadline:
                rows, cols = build_A_precomputed(target_n)
                max_lp = max(10, min(80, 40000 // target_n))
                a_lp, _ = lp_mirror_descent(a_sg, max_lp, deadline, rows, cols)
                tracker.update(a_lp)

            print(f"  n={target_n}: best J={tracker.best_score:.8f} ({time.monotonic() - t0:.1f}s)",
                  flush=True)

    # === Phase 6: Final deep LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        if n <= 4096:
            print("\n" + "=" * 50 + " Phase 6: Final Polish " + "=" * 50, flush=True)
            rows, cols = build_A_precomputed(n)
            remaining = max(0, int(deadline - time.monotonic()))
            print(f"  {remaining}s remaining, n={n}", flush=True)
            max_lp = max(50, min(500, 200000 // n))
            t0 = time.monotonic()
            a_final, _ = lp_mirror_descent(tracker.best, max_lp, deadline, rows, cols)
            tracker.update(a_final)
            print(f"  Final: J={tracker.best_score:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 50} FINAL {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
```
