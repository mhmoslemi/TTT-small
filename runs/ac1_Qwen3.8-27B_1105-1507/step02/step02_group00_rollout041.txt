"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth gradient (log-sum-exp), LP direction finding, and fast coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]

def smooth_gradient_step(a, t, eta):
    """One step of smooth gradient method with log-sum-exp."""
    n = len(a)
    h = np.convolve(a, a)
    h_max = np.max(h)
    h_shifted = h - h_max
    p = np.exp(h_shifted / t)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(h)] = 1.0
    else:
        p /= p_sum
    
    # Compute gradient: grad[i] = sum_k p[k] * 2 * a[k-i] (for valid k-i)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        grad[i_lo:i_hi + 1] += 2.0 * pk * a[k - i_hi:k - i_lo + 1][::-1]
    
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-15:
        return a, h, p
    a_new = a_new / s
    return a_new, h, p

def coord_refine(a_init, n_rounds=5, deadline=None, verbose=False):
    """Fast coordinate descent with golden section search."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        sum_a = float(np.sum(a))
        
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            
            old_val = a[j]
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            
            # Golden section search range
            hi_bound = max(3.0 * old_val + 0.01, 3.0 * sum_a / n)
            
            def eval_x(x):
                # Update affected conv entries
                delta = x - old_val
                new_max = 0.0
                for k in range(k_lo, k_hi + 1):
                    if 0 <= k - j < n:
                        if k == 2 * j:
                            new_conv_k = conv[k] - old_val * old_val + x * x
                        else:
                            new_conv_k = conv[k] + 2.0 * delta * a[k - j]
                    else:
                        new_conv_k = conv[k]
                    if new_conv_k > new_max:
                        new_max = new_conv_k
                # Also need to check unaffected entries
                # For efficiency, precompute max of unaffected entries
                # (This is O(n) which is fine)
                unaffected_max = 0.0
                for k in range(2 * n - 1):
                    if k < k_lo or k > k_hi:
                        if conv[k] > unaffected_max:
                            unaffected_max = conv[k]
                    else:
                        if 0 <= k - j < n:
                            if k == 2 * j:
                                val = conv[k] - old_val * old_val + x * x
                            else:
                                val = conv[k] + 2.0 * delta * a[k - j]
                        else:
                            val = conv[k]
                        if val > new_max:
                            new_max = val
                
                new_sum = sum_a - old_val + x
                if new_sum < 0.01:
                    return 1e10
                return 2.0 * n * new_max / (new_sum * new_sum)
            
            # Golden section search
            phi = 0.618033988749895
            lo = 0.0
            hi = hi_bound
            a1 = hi - phi * (hi - lo)
            a2 = lo + phi * (hi - lo)
            f1 = eval_x(a1)
            f2 = eval_x(a2)
            
            for _ in range(40):
                if f1 < f2:
                    hi = a2
                    a2 = a1
                    f2 = f1
                    a1 = hi - phi * (hi - lo)
                    f1 = eval_x(a1)
                else:
                    lo = a1
                    a1 = a2
                    f1 = f2
                    a2 = lo + phi * (hi - lo)
                    f2 = eval_x(a2)
            
            opt_x = (lo + hi) / 2
            f_opt = eval_x(opt_x)
            
            if f_opt < best_obj - 1e-12:
                a[j] = opt_x
                conv = np.convolve(a, a)
                sum_a = float(np.sum(a))
                best_obj = f_opt
                best_a = a.copy()
    
    return best_a, best_obj


def structured_init(n, rng):
    """Generate structured initialization."""
    choice = int(rng.integers(0, 5))
    if choice == 0:
        a = np.ones(n)
    elif choice == 1:
        freq = rng.uniform(0.5, 2.0)
        a = np.maximum(0.1, np.cos(2 * np.pi * freq * np.arange(n) / n))
    elif choice == 2:
        a = np.exp(-2.0 * (np.arange(n) / n - 0.5) ** 2)
    elif choice == 3:
        period = max(2, n // 20)
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    else:
        a = rng.random(n)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

def compute_gradient_direct(a, p):
    n = len(a)
    grad = np.zeros(n)
    # For each i, grad[i] = sum_k p[k] * 2 * a[k-i] (for 0 <= k-i < n)
    # = 2 * sum_k p[k] * a[k-i]

    # Valid i for given k: i in [max(0, k-n+1), min(n-1, k)]
    # a[k-i] for i in that range: a[k - min(n-1,k) : k - max(0,k-n+1) + 1][::-1]

        if p[k] < 1e-18:

One more idea: **use the subgradient from the LP dual**. After solving the LP, the dual variables give us a subgradient of the value function. We can use this to take a step in a more informed direction.

The LP dual is:
```
min M * sum(y)
s.t. sum_k A[k,j] * y[k] >= 1 for all j
     y >= 0
```

The dual variable y[k] tells us the "shadow price" of constraint k. Normalizing y gives us weights that indicate which constraints are most binding.

We can then use these weights to compute a subgradient:
`subgrad[j] = sum_k y[k] * A[k,j]` (up to normalization)

Wait, that's just the gradient of the constraint function `g(a) = max_k (a*a)[k]` evaluated at the current point, weighted by the dual variables.

Actually, the subgradient of `max_k h_k(a)` at a point where the active set is S is:
`conv({grad h_k(a) : k in S})`

The dual variables tell us which active constraints to emphasize. If we normalize y to sum to 1 and restrict to active constraints, we get a specific subgradient.

But in the LP, the dual variables are for the relaxed problem (with RHS = M), so they might not directly correspond to the subgradient of the max function. Still, they provide useful information.

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smooth gradient, LP direction finding, and coordinate descent.
Focused strategy: fewer n values with deeper optimization, perturbation restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a, M_scale=1.0):
    """Solve LP: max sum(g) s.t. a*g <= M*M_scale, g >= 0."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv)) * M_scale
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def compute_gradient_smooth(a, p):
    """Compute gradient: grad[i] = sum_k p[k] * 2 * a[k-i] for valid indices."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-18:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        grad[i_lo:i_hi + 1] += 2.0 * pk * a[k - i_hi:k - i_lo + 1][::-1]
    return grad


def smooth_optimize(a_init, n_iters=5000, deadline=None, verbose=False):
    """Log-sum-exp smooth gradient with adaptive temperature and step size."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    h = np.convolve(a, a)
    t_start = float(np.max(h)) * 0.05
    t_end = t_start * 0.0005

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    patience = 0
    PATIENCE_LIMIT = 800

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        frac = (it + 1) / n_iters
        # Temperature: exponential decay
        t = t_start * (t_end / t_start) ** frac
        # Step size: starts moderate, decreases
        eta = 0.08 * (1.0 - 0.5 * frac)

        # Compute softmax weights
        h = np.convolve(a, a)
        h_max = float(np.max(h))
        h_shifted = h - h_max
        p = np.exp(h_shifted / t)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(h))] = 1.0
        else:
            p /= p_sum

        grad = compute_gradient_smooth(a, p)

        # Gradient step with projection
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new

        if verbose and it % 500 == 0:
            print(f"    Smooth it={it}: t={t:.2e}, eta={eta:.4f}, best={best_obj:.6f}")

        # Early stop if stuck
        if patience > PATIENCE_LIMIT and frac > 0.3:
            break

    return best_a, best_obj


def lp_polish(a_init, n_iters=100, deadline=None, verbose=False):
    """LP-based directional optimization."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g = solve_lp(a, M_scale=1.0)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Try multiple blend factors
        best_local = best_obj
        best_local_a = a.copy()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7]:
            a_try = (1 - alpha) * a + alpha * g
            a_try = np.maximum(a_try, 0.0)
            s_try = np.sum(a_try)
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local:
                best_local = obj_try
                best_local_a = a_try.copy()

        if best_local < best_obj:
            best_obj = best_local
            best_a = best_local_a.copy()
            a = best_local_a.copy()
        else:
            # Move towards g slightly anyway for diversity
            a = 0.9 * a + 0.1 * g
            a = np.maximum(a, 0.0)
            s = np.sum(a)
            if s < 1e-15:
                break
            a = a / s

        if verbose and it % 20 == 0:
            print(f"    LP it={it}: obj={best_obj:.6f}")

    return best_a, best_obj


def coord_refine(a_init, n_rounds=5, deadline=None, verbose=False):
    """Coordinate descent with multi-point evaluation per coordinate."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break

        improved = False
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break

            old_val = a[j]
            # Try multiple values
            candidates = [0.0, 0.25 * old_val, 0.5 * old_val, 0.75 * old_val,
                          1.0 * old_val, 1.25 * old_val, 1.5 * old_val, 2.0 * old_val]
            # Also try values based on the mean
            mean_val = np.mean(a)
            candidates.append(mean_val)
            candidates.append(0.5 * mean_val)
            candidates.append(2.0 * mean_val)

            best_x = old_val
            best_local = best_obj

            for x in candidates:
                if x < 0:
                    continue
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x

            a[j] = best_x
            if best_x != old_val:
                improved = True

        if improved:
            obj_now = compute_obj(a)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()
        else:
            if rnd > 0:
                break

        if verbose:
            print(f"    Coord round {rnd}: obj={best_obj:.6f}")

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 7))
    if choice == 0:
        a = np.ones(n)
    elif choice == 1:
        freq = rng.uniform(0.3, 2.0)
        a = np.maximum(0.05, np.cos(2 * np.pi * freq * np.arange(n) / n))
    elif choice == 2:
        sigma = rng.uniform(0.1, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) / n - 0.5) / sigma) ** 2)
    elif choice == 3:
        period = max(2, int(n / rng.integers(5, 30)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 4:
        a = np.abs(np.sin(np.arange(n) * rng.uniform(0.05, 0.5)))
    elif choice == 5:
        # Triangular
        mid = n // 2
        a = 1.0 - np.abs(np.arange(n) - mid) / (mid + 1)
    else:
        a = rng.random(n)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s
