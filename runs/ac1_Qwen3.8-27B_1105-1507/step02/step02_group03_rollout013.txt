Optimized step function search using:
1. Nesterov-accelerated subgradient descent on the simplex
2. LP-based refinement with adaptive constraint tightness
3. Coordinate descent polishing
4. Multi-start with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_score_normalized(f):
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def simplex_project(v):
    """Project onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.zeros(n)
    r = np.sort(v)[::-1]
    cv = np.cumsum(r)
    rho = np.nonzero(r * np.arange(1, n+1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    mu = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - mu, 0)

def nesterov_subgradient(f, n_iter=5000, eta0=0.01, rng=None):
    """Nesterov-accelerated subgradient descent on the simplex."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    
    # Nesterov momentum
    y = f.copy()
    t = 1.0
    no_improve = 0
    
    for it in range(n_iter):
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        alpha = (t - 1) / t_new
        
        # Compute subgradient at y
        conv = np.convolve(y, y)
        M = conv.max()
        # Find all positions achieving (near) max
        threshold = M * 0.999
        active = np.where(conv >= threshold)[0]
        
        # Subgradient: for each i, sum over active k of y[k-i] (if valid)
        sg = np.zeros(n)
        for k in active:
            # y[k-i] for valid i: i in [max(0, k-n+1), min(k, n-1)]
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                sg[i] += y[k - i]
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            sg = sg / sg_norm
        
        # Step size
        eta = eta0 / (np.sqrt(it + 1))
        
        # Update
        y = simplex_project(y - eta * sg)
        f = (1 - alpha) * y + alpha * f
        
        # Check improvement
        score = compute_score_normalized(f)
        if score < best_score - 1e-10:
            best_score = score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 1000 and it > n_iter // 2:
            break
    
    return best_f, best_score

def compute_score_normalized(f):
    """Compute score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def simplex_project(v):
    """Project vector onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0)])
    r = np.sort(v)[::-1]
    cv = np.cumsum(r)
    rho_candidates = np.nonzero(r * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    mu = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - mu, 0.0)


def subgradient_at(f, conv=None):
    """Compute subgradient of max(conv(f,f)) w.r.t. f. Returns (sg, max_val, active_set)."""
    if conv is None:
        conv = np.convolve(f, f)
    M = conv.max()
    # Use a "near-active" set for more stable gradients
    threshold = M * 0.9995
    active = np.where(conv >= threshold)[0]
    
    n = len(f)
    sg = np.zeros(n)
    for k in active:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            sg[i_start:i_end + 1] += f[k - i_start:k - i_start + (i_end - i_start + 1)]
    return sg, M, active


def nesterov_subgradient(f, n_iter=5000, eta0=0.05, momentum_decay=1.0, rng=None):

        t = t_new

def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A

def coordinate_descent_golden(f, n_sweeps=5, rng=None):
    """Coordinate descent with golden-section search for optimal value of each coordinate."""
    if rng is None:

                # Try small positive values
                for trial in [1e-4, 1e-3, 0.01, 0.05]:
                    f[i] = trial
                    s = f.sum()
                    if s < 1e-10:
                        f[i] = original
                        continue
                    score = 2.0 * n * np.convolve(f, f).max() / (s * s)
                    if score < best_score:
                        best_score = score
                        original = trial
                        break
                f[i] = original
                continue
            
            # Golden section search in log-space
            # Search range: [original * 0.1, original * 10]
            log_lo = np.log(original) - 2.3  # ~10x smaller
            log_hi = np.log(original) + 2.3  # ~10x larger
            
            f[i] = original  # reset
            best_val = original
            best_local = best_score
            
            # Sample several points
            for log_val in np.linspace(log_lo, log_hi, 7):
                f[i] = np.exp(log_val)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = 2.0 * n * np.convolve(f, f).max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            
            # Refine around best
            if best_val != original:
                log_center = np.log(best_val)
                log_range = abs(log_center - log_lo) * 0.5
                for log_val in np.linspace(log_center - log_range, log_center + log_range, 5):
                    f[i] = np.exp(log_val)
                    s = f.sum()
                    if s < 1e-10:
                        f[i] = best_val
                        continue
                    score = 2.0 * n * np.convolve(f, f).max() / (s * s)
                    if score < best_local:
                        best_local = score
                        best_val = f[i]
            
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        
        f = normalize(f)
    
    return f, best_score


def multiplicative_weights(f, n_iter=200, eta=0.1, rng=None):

def diverse_init(n, rng, strategy=0):
    """Generate diverse initial sequences."""
    if strategy == 0:
        # Raised cosine
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        f = np.maximum(f, 0.01)
    elif strategy == 1:
        # Asymmetric ramp
        f = np.linspace(0.1, 1.0, n)
    elif strategy == 2:
        # Two-bump
        f = np.exp(-0.5 * (np.arange(n) - n * 0.3) ** 2 / (n * 0.05)) + \
            np.exp(-0.5 * (np.arange(n) - n * 0.7) ** 2 / (n * 0.05))
        f = np.maximum(f, 0.01)
    elif strategy == 3:
        # Geometric
        f = 0.5 ** (np.arange(n) / n * 3)
        f = np.maximum(f, 0.01)
    elif strategy == 4:
        # Random smooth
        f = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 5) * np.arange(n) / n)
        f = np.maximum(f, 0.01)
    else:
        # Triangular
        center = n // 2
        f = 1.0 - np.abs(np.arange(n) - center) / center
        f = np.maximum(f, 0.01)
    
    f = f * (1.0 + 0.05 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f

        print(f"  Alt nesterov: {s_alt:.8f}")
        
        remaining2 = deadline - time.monotonic()
        if remaining2 > 40:
            f_alt, s_alt = lp_refine_adaptive(f_alt, min(remaining2 * 0.5, 120),
                                              M_factor_range=(0.98, 1.02, 1.0),
                                              rng=rng, lp_time_limit=5)
            print(f"  Alt LP: {s_alt:.8f}")
        
        if s_alt < best_score_overall:
            best_score_overall = s_alt
            best_overall = f_alt.copy()
            print(f"  NEW BEST (alt length): {best_score_overall:.8f}")
        ensemble.append(f_alt.copy())
    
    # Phase 7: Final polishing
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_overall is not None:
        print(f"\n--- Phase 7: Final polish (remaining: {remaining:.0f}s) ---")
        
        # Quick Nesterov from best
        t_polish = min(60, remaining * 0.3)
        if t_polish > 10:
            f_p, s_p = nesterov_subgradient(best_overall, n_iter=int(t_polish * 40), eta0=0.03, rng=rng)
            if s_p < best_score_overall:
                best_score_overall = s_p
                best_overall = f_p.copy()
                print(f"  Polish Nesterov: {best_score_overall:.8f}")
        
        # Final coordinate descent
        remaining3 = deadline - time.monotonic()
        if remaining3 > 15:
            f_cd_final, s_cd_final = coordinate_descent_golden(best_overall, n_sweeps=3, rng=rng)
            if s_cd_final < best_score_overall:
                best_score_overall = s_cd_final
                best_overall = f_cd_final.copy()
                print(f"  Final CD: {best_score_overall:.8f}")
        
        # One more LP if time
        remaining4 = deadline - time.monotonic()
        if remaining4 > 30:
            f_lp_final, s_lp_final, _ = lp_refine_adaptive(best_overall, remaining4 - 10,
                                                            M_factor_range=(0.995, 1.005, 1.0),
                                                            rng=rng, lp_time_limit=10)
            if s_lp_final < best_score_overall:
                best_score_overall = s_lp_final
                best_overall = f_lp_final.copy()
                print(f"  Final LP: {best_score_overall:.8f}")
    
    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.8f}, n={len(final_list)} ===")
        print(f"    Target: 1.5027, Previous best: 1.512006")
        if final_score < 1.5027:
            print("    SUCCESS: Below target!")
        elif final_score < 1.512006:
            print(f"    IMPROVEMENT: {1.512006 - final_score:.6f} better than previous")
        return final_list
    
    # Fallback
    print("Fallback to uniform sequence")
    return [1.0 / 256.0] * 256
