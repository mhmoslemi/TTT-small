"""
Optimized step function search using FFT-accelerated gradients, mirror descent,
spectral initialization, and adaptive tau schedules. Explores n up to 2000.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def fft_cross_correlate(w, a):
    """Compute cross-correlation (w * a)[j] = sum_k w[k] * a[k-j] using FFT.
    Returns array of length n."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w[:L], L)
    # Cross-correlation via FFT: IFFT(FFT(w) * conj(FFT(a))) gives sum_k w[k]*a[k-j]
    corr = np.fft.irfft(fw * np.conj(fa), L)
    return corr[:n]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = fft_convolve(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smoothed_weights(conv, tau):
    """Compute log-sum-exp smoothed weights over convolution values."""
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    return weights


def compute_gradient_fft(weights, a):
    """Compute gradient of sum_k weights[k]*conv(a,a)[k] w.r.t. a using FFT.
    Returns gradient of length n."""
    n = len(a)
    corr = fft_cross_correlate(weights, a)
    return 2.0 * corr


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update: a_new = a * exp(-eta*grad) / Z."""
    log_a = np.log(np.maximum(a, 1e-300)) - eta * grad
    # Subtract max for numerical stability
    log_a = log_a - np.max(log_a)
    a_new = np.exp(log_a)
    s = np.sum(a_new)
    if s < 1e-300:
        return a.copy()
    return a_new / s


def backtracking_step(a, grad, tau, max_linesearch=20):
    """Additive gradient step with backtracking line search.
    Minimizes smoothed objective with sum constraint handled by centering."""
    n = len(a)
    grad_centered = grad - np.mean(grad)
    
    # Try decreasing steps
    for ls in range(max_linesearch):
        step = 0.1 * (0.5 ** ls)
        a_try = a - step * grad_centered
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-10:
            continue
        a_try = a_try / s
        # Check improvement in smoothed objective
        conv_try = fft_convolve(a_try)
        curr_val = float(np.sum(smoothed_weights(conv_try, tau) * conv_try))
        conv_cur = fft_convolve(a)
        cur_val = float(np.sum(smoothed_weights(conv_cur, tau) * conv_cur))
        if curr_val < cur_val + 1e-15:
            return a_try
    return a

                stall = 0
                continue
        
        weights = smoothed_weights(conv, tau)
        grad = compute_gradient_fft(weights, a)
        
        # Alternate between mirror and additive
        mode_phase = (it // 100) % 2
        if mode_phase == 0:
            eta = 0.3 * (1.0 - frac) + 0.02
            a = mirror_descent_step(a, grad, eta)
        else:
            step = step_start * (1.0 - 0.5 * frac)
            grad_c = grad - np.mean(grad)
            a_new = a - step * grad_c
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s > 1e-10:
                a = a_new / s
    
    return best_a


def build_A_matrix(a):
    """Build the Toeplitz constraint matrix for the LP (convolution with a)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp_for_g(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_optimize(a, max_iter=50, t=0.3, time_per_lp=15):
    """Alternating LP optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    stall = 0
    for it in range(max_iter):
        conv = fft_convolve(a)
        M = float(np.max(conv))
        
        g, sum_g = solve_lp_for_g(a, M, time_limit=time_per_lp)
        if g is None:
            break
        
        if sum_g <= 1.0 + 1e-4:
            stall += 1
            if stall > 15:
                break
            # Try slightly larger M
            M2 = M * 1.01
            g, sum_g = solve_lp_for_g(a, M2, time_limit=time_per_lp)
            if g is None or sum_g <= 1.0 + 1e-4:
                continue
            g = g / sum_g
            # Renormalize M
            g = g * M / (M2)
        else:
            stall = 0
            g = g / sum_g
        
        g = np.maximum(g, 0)
        s_g = np.sum(g)
        if s_g < 1e-10:
            break
        g = g / s_g
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_obj


def spectral_init(n, rng):
    """Initialize using flat-magnitude spectrum with random phase."""
    L = 1
    while L < 2 * n:
        L <<= 1
    
    # Create flat magnitude spectrum with some variation
    magnitude = np.ones(L // 2 + 1)
    # Add some random variation to avoid perfectly symmetric solutions
    variation = 1.0 + rng.normal(0, 0.1, L // 2 + 1)
    variation = np.maximum(variation, 0.5)
    magnitude *= variation
    
    # Random phase
    phase = rng.uniform(-np.pi, np.pi, L // 2 + 1)
    # Force DC component to be real and positive
    phase[0] = 0.0
    # Force Nyquist to be real (if L is even)
    if L % 2 == 0:
        phase[-1] = 0.0 if rng.random() > 0.5 else np.pi
    
    spectrum = magnitude * np.exp(1j * phase)
    # Symmetrize for real output
    full_spectrum = np.zeros(L, dtype=complex)
    full_spectrum[:L//2+1] = spectrum
    for k in range(1, L//2):
        full_spectrum[L-k] = np.conj(spectrum[k])
    if L % 2 == 0:
        full_spectrum[L//2] = np.real(spectrum[L//2])
    
    a_full = np.fft.ifft(full_spectrum).real
    a = a_full[:n]
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def parametric_init(n, rng):
    """Generate a parametric initialization."""
    choice = rng.integers(0, 6)
    x = np.arange(n) / n
    
    if choice == 0:
        # Cosine window
        freq = rng.uniform(0.5, 3.0)
        a = np.abs(np.cos(np.pi * freq * x + rng.uniform(0, np.pi)))
    elif choice == 1:
        # Gaussian
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif choice == 2:
        # Power law
        p = rng.uniform(0.5, 3.0)
        a = (1 - x) ** p
    elif choice == 3:
        # Bimodal
        c1, c2 = 0.25, 0.75
        w = rng.uniform(0.05, 0.15)
        a = np.exp(-((x - c1)**2)/(2*w**2)) + np.exp(-((x - c2)**2)/(2*w**2))
    elif choice == 4:
        # Sinc-like
        freq = rng.uniform(1, 5)
        a = np.abs(np.sinc(freq * x - freq/2))
    else:
        # Random smooth
        a = rng.random(n)
        # Smooth with a small kernel
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    
    # Add small random perturbation
    a = a + rng.normal(0, 0.01, n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def run_optimization_for_n(n, seed, time_budget, parent_a=None, verbose=False):
    """Run full optimization pipeline for a given n."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    
    best_a = None
    best_obj = np.inf
    
    # Generate initializations
    inits = []
    
    if parent_a is not None and len(parent_a) == n:
        p = parent_a.copy()
        s = np.sum(p)
        if s > 1e-10:
            inits.append(p / s)
    
    inits.append(spectral_init(n, rng))
    inits.append(parametric_init(n, rng))
    inits.append(parametric_init(n, rng))
    
    if n <= 200:
        inits.append(spectral_init(n, rng))
        inits.append(np.ones(n) / n)
    
    else:
        # Uniform with perturbation
        a_u = np.ones(n) / n
        a_u = a_u * np.exp(rng.normal(0, 0.3, n))
        a_u = np.maximum(a_u, 0)
        a_u = a_u / np.sum(a_u)
        inits.append(a_u)
    
    # Allocate time per initialization
    time_per_init = time_budget / len(inits)
    
    for idx, a_init in enumerate(inits):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:
            break
        
        remaining = time_budget - elapsed
        t_alloc = min(time_per_init, remaining - 2)
        if t_alloc < 3:
            break
        
        a = a_init.copy()
        
        # Phase 1: Mirror descent (exploration)
        if verbose:
            print(f"      n={n} init{idx} phase1 (mirror)", end=" ")
        md_iters = int(min(30000, max(1000, t_alloc * 4)))
        a = optimize_mirror_descent(a, max_iter=md_iters,
                                     tau_start=5.0, tau_end=100.0,
                                     eta_start=0.5, eta_end=0.02)
        obj = compute_obj(a)
        if verbose:
            print(f"obj={obj:.6f}", end=" ")
        
        # Phase 2: Combined optimization (exploitation)
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining > 5:
            t2 = min(remaining * 0.4, t_alloc * 0.5)
            lsb_iters = int(min(30000, max(1000, t2 * 4)))
            a = optimize_lsb(a, max_iter=lsb_iters,
                             tau_start=20.0, tau_end=200.0)
            obj2 = compute_obj(a)
            if obj2 < obj:
                obj = obj2
            if verbose:
                print(f"phase2={obj:.6f}", end=" ")
        
        # Phase 3: Additive gradient (fine-tuning)
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining > 5:
            t3 = min(remaining * 0.3, t_alloc * 0.3)
            add_iters = int(min(20000, max(500, t3 * 3)))
            a = optimize_additive(a, max_iter=add_iters,
                                   tau_start=50.0, tau_end=300.0)
            obj3 = compute_obj(a)
            if obj3 < obj:
                obj = obj3
            if verbose:
                print(f"phase3={obj:.6f}", end=" ")
        
        # Phase 4: LP refinement (only for small n)
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining > 10 and n <= 350:
            a_lp, obj_lp = lp_optimize(a, max_iter=30, t=0.25,
                                       time_per_lp=min(10, remaining // 2))
            if obj_lp < obj:
                obj = obj_lp
                a = a_lp
            if verbose:
                print(f"LP={obj:.6f}", end=" ")
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose:
            print(f" -> best={best_obj:.6f}")
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)
    
    return best_a, best_obj


def final_polish(a, time_budget, verbose=True):
    """Intensive final polishing of the best sequence found."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    deadline = start + time_budget
    
    print(f"  Polish start: n={n}, obj={best_obj:.6f}")
    
    round_idx = 0
    while time.monotonic() < deadline - 5:
        round_idx += 1
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        a_cur = best_a.copy()
        improved = False
        
        # Strategy 1: Mirror descent with high tau
        if remaining > 10:
            t_alloc = min(remaining * 0.3, 60.0)
            iters = int(min(50000, max(2000, t_alloc * 5)))
            a_try = optimize_mirror_descent(a_cur, max_iter=iters,
                                             tau_start=50.0, tau_end=500.0,
                                             eta_start=0.2, eta_end=0.005)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-8:
                best_obj = obj_try
                best_a = a_try.copy()
                a_cur = a_try.copy()
                improved = True
                if verbose:
                    print(f"  Round {round_idx} mirror: obj={best_obj:.6f}")
        
        # Strategy 2: Additive with very high tau
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_alloc = min(remaining * 0.25, 40.0)
            iters = int(min(50000, max(2000, t_alloc * 5)))
            a_try = optimize_additive(a_cur, max_iter=iters,
                                       tau_start=100.0, tau_end=800.0)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-8:
                best_obj = obj_try
                best_a = a_try.copy()
                a_cur = a_try.copy()
                improved = True
                if verbose:
                    print(f"  Round {round_idx} additive: obj={best_obj:.6f}")
        
        # Strategy 3: LP step
        remaining = deadline - time.monotonic()
        if remaining > 15 and n <= 400:
            a_try, obj_try = lp_optimize(a_cur, max_iter=30, t=0.2,
                                         time_per_lp=min(10, remaining // 3))
            if obj_try < best_obj - 1e-8:
                best_obj = obj_try
                best_a = a_try.copy()
                a_cur = a_try.copy()
                improved = True
                if verbose:
                    print(f"  Round {round_idx} LP: obj={best_obj:.6f}")
        
        # Strategy 4: Perturbation + re-optimization
        remaining = deadline - time.monotonic()
        if remaining > 15:
            rng = np.random.default_rng(round_idx * 997 + 13)
            a_perturb = a_cur * np.exp(rng.normal(0, 0.05, n))
            a_perturb = np.maximum(a_perturb, 0)
            s = np.sum(a_perturb)
            if s > 1e-10:
                a_perturb = a_perturb / s
                # Quick optimization from perturbed state
                iters = int(min(10000, max(1000, remaining * 2)))
                a_try = optimize_mirror_descent(a_perturb, max_iter=iters,
                                                 tau_start=30.0, tau_end=200.0,
                                                 eta_start=0.3, eta_end=0.01)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-8:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    if verbose:
                        print(f"  Round {round_idx} perturb: obj={best_obj:.6f}")
        
        if not improved and round_idx > 5:
            # Try a bigger perturbation
            rng = np.random.default_rng(round_idx * 7919 + 42)
            a_perturb = a_cur * np.exp(rng.normal(0, 0.3, n))
            a_perturb = np.maximum(a_perturb, 0)
            s = np.sum(a_perturb)
            if s > 1e-10:
                a_perturb = a_perturb / s
                iters = int(min(20000, max(2000, remaining * 3)))
                a_try = optimize_lsb(a_perturb, max_iter=iters,
                                     tau_start=20.0, tau_end=300.0)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-8:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    if verbose:
                        print(f"  Round {round_idx} big_perturb: obj={best_obj:.6f}")
                
                if time.monotonic() > deadline - 10:
                    break
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for optimal step function sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FFT-accelerated gradients, mirror descent, spectral initialization,
    and multiple optimization phases. Explores n from 30 to 2000.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Parent: n={parent.size}, obj={best_score:.6f}")
    
    # Sequence lengths to explore
    n_values = [30, 50, 80, 100, 120, 150, 200, 250, 300, 400, 500, 750, 1000, 1500, 2000]
    
    # If parent exists, prioritize its neighborhood
    if parent is not None:
        n_p = len(parent)
        extra_ns = [n_p, max(10, n_p - 50), n_p + 50, max(10, n_p - 100), n_p + 100]
        n_values = sorted(set(n_values + [int(x) for x in extra_ns if 10 <= x <= 3000]))
    
    print(f"Exploring n values: {n_values}")
    
    # Time allocation strategy: spend more time on promising n values
    # Give initial quick pass to all, then intensive pass on best
    total_n = len(n_values)
    
    for i, n in enumerate(n_values):
        elapsed = time.monotonic() - start_time
        remaining = deadline - elapsed
        if remaining < 20:
            break
        
        # Time allocation: larger n gets more time, but cap total
        if n <= 100:
            alloc = min(20.0, remaining * 0.06)
        elif n <= 250:
            alloc = min(50.0, remaining * 0.10)
        elif n <= 500:
            alloc = min(80.0, remaining * 0.12)
        elif n <= 1000:
            alloc = min(120.0, remaining * 0.15)
        else:
            alloc = min(150.0, remaining * 0.18)
        
        alloc = max(8.0, alloc)
        
        if remaining < alloc + 10:
            alloc = remaining - 10
            if alloc < 8:
                break
        
        verbose = n in [100, 250, 500, 1000]
        parent_for_n = None
        if parent is not None and len(parent) == n:
            parent_for_n = parent
        
        print(f"N={n}: {remaining:.0f}s left, alloc={alloc:.0f}s")
        a_opt, obj = run_optimization_for_n(n, seed=seed + n * 7919,
                                             time_budget=alloc,
                                             parent_a=parent_for_n,
                                             verbose=verbose)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  *** UPDATE: n={n}, obj={best_score:.6f}")
    
    # Final intensive polishing
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"\nFinal polish: n={len(best_seq)}, {remaining:.0f}s remaining")
            a_pol, obj_pol = final_polish(best_seq, time_budget=remaining)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
    
    # One more round: try to improve by trying n values near the best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_best = len(best_seq)
            nearby_ns = [n_best - 10, n_best - 20, n_best + 10, n_best + 20,
                        n_best + 50, n_best + 100]
            nearby_ns = sorted(set([x for x in nearby_ns if 10 <= x <= 3000
                                     and x != n_best]))
            for n_try in nearby_ns[:3]:
                remaining = deadline - time.monotonic()
                if remaining < 15:
                    break
                alloc = min(remaining * 0.3, 60.0)
                a_t, obj_t = run_optimization_for_n(n_try, seed=seed + n_try * 31,
                                                     time_budget=alloc)
                if obj_t < best_score:
                    best_score = obj_t
                    best_seq = a_t.copy()
                    print(f"  Nearby update: n={n_try}, obj={best_score:.6f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)
    
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    
    # Validate with evaluate_sequence
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < best_score:
        print(f"  Validation better: {validation:.6f}")
        best_score = validation
    
    return result
