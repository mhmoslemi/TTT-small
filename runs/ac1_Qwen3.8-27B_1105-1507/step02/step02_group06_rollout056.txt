"""
Optimized search for minimizing 2n*max(conv(a,a))/sum(a)^2 using:
- Nesterov-accelerated subgradient with adaptive step size
- LP-based direction finding for polishing
- Multiple structured initializations (ramp, S-curve, etc.)
- Adaptive peak-tracking (increasing K of top lags as we converge)

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Compute the objective value directly."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    nnz = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        nnz += j_max - j_min + 1
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    vals = np.empty(nnz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_subgrad(a, top_k=5, use_soft=False, p=10):
    """Compute subgradient of max_k c[k] with respect to a."""
    n = len(a)
    conv = np.convolve(a, a)
    if use_soft:
        weights = conv.astype(np.float64)
        weights = np.maximum(weights, 0.0)
        weights = weights ** p
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(len(conv)) / len(conv)
        else:
            weights = weights / w_sum
        # Sum over top lags weighted
        # Focus on top 50% for efficiency
        threshold = np.median(weights)
        active = np.where(weights > threshold)[0]
        if len(active) == 0:
            active = np.argsort(weights)[-5:]
    else:
        active = np.argsort(conv)[-top_k:]
        weights = np.ones(len(active)) / len(active)

    n = len(a)
    grad = np.zeros(n)
    max_weight = 0.0
    for k, w in zip(active, weights):
        if use_soft:
            w = weights[k]
        else:
            w = 1.0 / len(active)
        if w < 1e-10:
            continue
        max_weight = max(max_weight, w)
        j_vals = np.arange(n)
        idx1 = k - j_vals
        valid1 = (idx1 >= 0) & (idx1 < n)
        grad[valid1] += w * a[idx1[valid1]]
        idx2 = j_vals - k
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad[valid2] += w * a[idx2[valid2]]
    return grad, float(np.max(conv)), conv


def nesterov_subgrad_optimize(f, deadline, lr_init=0.01, top_k_init=3,
                               top_k_max=15, momentum_init=0.9,
                               max_iter=500000, verbose_prefix=""):
    """Nesterov accelerated subgradient descent with adaptive step size."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = 0.0

    y = f.copy()  # Nesterov lookahead point
    no_improve = 0
    it = 0
    top_k = top_k_init

    while time.monotonic() < deadline and it < max_iter:

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Nesterov update
        momentum = momentum_decay * momentum - lr * grad
        y_new = f + (momentum_decay + 1) * momentum / (momentum_decay + 1)
        # Actually standard Nesterov: y = x + beta * (x - x_prev)
        # Then x_new = x - lr * grad(y)

        # v = beta * v - lr * grad
        # x = x + v

        y_new = np.clip(y_new, 0.0, None)

        if new_sum < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 200:
                break
            it += 1
            continue
        y_new = y_new * (S / new_sum)

        new_score = compute_score(y_new)
        it += 1

            f = y_new
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            # Gradually increase top_k
            if it % 500 == 0:

        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.9
            if no_improve > 500:
                # Reset momentum and try with larger top_k
                momentum = np.zeros(n)
                momentum = 0.0
                momentum = np.zeros(n)
                top_k = min(top_k_max, top_k + 3)
                no_improve = 0
                if lr < 1e-8:

                # Restart from best

        # Line search

        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            step *= 0.99
            step = max(step, 0.01)
            if it % 5 == 0:
                print(f"  {verbose_prefix}LP iter {it}: score={best_score:.6f} "
                      f"(step={step:.4f}, lp_t={lp_time:.2f}s)")
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.005 or no_improve > 40:
                break

    return best_f, best_score


def ramp_init(n, start=1.0, end=None):
    """Linear ramp initialization."""
    if end is None:
        end = 1.0 + 0.5 * (n - 1) / n
    return np.linspace(start, end, n)


def scurve_init(n):
    """S-curve (sigmoid) initialization."""
    x = np.linspace(0, 1, n)
    return 1.0 / (1.0 + np.exp(-6.0 * (x - 0.3)))


def quadratic_ramp_init(n):
    """Quadratic ramp."""
    x = np.linspace(0, 1, n)
    return 1.0 + x ** 2


def exp_ramp_init(n):
    """Exponential ramp."""
    x = np.linspace(0, 1, n)
    return np.exp(1.5 * x)


def offcenter_bump_init(n):
    """Gaussian bump off-center."""
    x = np.linspace(0, 1, n)
    return np.exp(-5.0 * (x - 0.3) ** 2) + 0.1


def sawtooth_init(n):
    """Sawtooth-like pattern."""
    x = np.linspace(0, 1, n)
    return (x * 3.0) % 1.0 + 0.1


def make_initializations(n, rng):
    """Create diverse initializations for length n."""
    inits = []
    inits.append(('ramp', ramp_init(n)))
    inits.append(('ramp2', ramp_init(n, start=0.1, end=2.0)))
    inits.append(('scurve', scurve_init(n)))
    inits.append(('quad_ramp', quadratic_ramp_init(n)))
    inits.append(('exp_ramp', exp_ramp_init(n)))
    inits.append(('offcenter', offcenter_bump_init(n)))
    inits.append(('sawtooth', sawtooth_init(n)))
    # Random with structure
    base = ramp_init(n, 0.5, 1.5)
    noise = rng.normal(0, 0.2, size=n)
    inits.append(('ramp_noise', np.clip(base + noise, 0.01, None)))
    return inits


def combined_optimize(f, deadline, verbose_prefix=""):
    """Combined optimization: subgradient for fast progress, LP for polishing."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)

    # Phase A: Fast subgradient (40% of time)
    sub_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.4
    f_sub, s_sub = nesterov_optimize_v2(f, sub_deadline, lr_init=0.02,
                                         top_k_init=3, top_k_max=15,
                                         verbose_prefix=verbose_prefix + "sub ")
    if s_sub < best_score:
        best_score = s_sub
        best_f = f_sub.copy()
        f = f_sub.copy()

    # Phase B: LP refinement (40% of time)
    remaining = deadline - time.monotonic()
    lp_deadline = time.monotonic() + remaining * 0.5
    f_lp, s_lp = lp_optimize(f, lp_deadline, step_init=0.05,
                              verbose_prefix=verbose_prefix + "LP ")
    if s_lp < best_score:
        best_score = s_lp
        best_f = f_lp.copy()
        f = f_lp.copy()

    # Phase C: Final subgradient polish (remaining time)
    f_pol, s_pol = nesterov_optimize_v2(f, deadline, lr_init=0.005,
                                         top_k_init=10, top_k_max=25,
                                         verbose_prefix=verbose_prefix + "polish ")
    if s_pol < best_score:
        best_score = s_pol
        best_f = f_pol.copy()

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence minimizing 2n*max(conv)/sum^2."""
    t0 = time.monotonic()
    deadline = t0 + max(50.0, float(budget_s) - 5.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Try to load parent sequence
    parent_n = 1024
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                parent_n = len(parent)
                print(f"Parent n={parent_n}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Determine which lengths to try
    lengths = [1024]
    if parent_n != 1024:
        lengths.insert(0, parent_n)
    # Also try 2048 if we have time
    if budget_s > 1000:
        lengths.append(2048)

    # Phase 1: Quick exploration (first 25% of time)
    t_explore_end = t0 + (budget_s - 5.0) * 0.25
    print(f"\n=== Phase 1: Exploration (until t={t_explore_end - t0:.0f}s) ===")

    candidates = []  # (score, sequence, n, label)

    for n in lengths:
        if time.monotonic() > t_explore_end - 60:
            break
        print(f"\n--- n={n} ---")

        inits = make_initializations(n, rng)

        # Add parent if applicable
        if best_sequence is not None and len(best_sequence) == n:
            inits.insert(0, ('parent', best_sequence.copy()))
        elif best_sequence is not None:
            x_old = np.linspace(0, 1, len(best_sequence))
            x_new = np.linspace(0, 1, n)
            parent_interp = np.interp(x_new, x_old, best_sequence)
            inits.insert(0, ('parent_interp', parent_interp))

        # For large n, limit number of initializations
        max_inits = 6 if n >= 2048 else 8

        for i, (label, f_init) in enumerate(inits[:max_inits]):
            if time.monotonic() > t_explore_end - 10:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            if np.sum(f_init) < 1e-10:
                continue
            f_init = f_init / np.sum(f_init) * np.sqrt(2 * n)

            score_init = compute_score(f_init)
            if not np.isfinite(score_init):
                continue

            # Quick optimization
            init_time = min(30.0 if n < 2048 else 15.0,
                            max(5.0, (t_explore_end - 10) - time.monotonic()))
            quick_deadline = time.monotonic() + init_time

            f_opt, s_opt = nesterov_optimize_v2(f_init, quick_deadline,
                                                 lr_init=0.02, top_k_init=3,
                                                 top_k_max=10,
                                                 verbose_prefix=f"[{label}] ")
            if s_opt < score_init:
                score_init = s_opt
                f_init = f_opt

            if score_init < best_score:
                best_score = score_init
                best_sequence = f_init.copy()
                print(f"  NEW BEST: n={n} {label}: {best_score:.6f}")

            candidates.append((score_init, f_init.copy(), n, label))
            print(f"  n={n} {label}: {score_init:.6f}")

    # Sort and keep top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates after exploration:")
    for sc, seq, n, lbl in candidates[:5]:
        print(f"  n={n} {lbl}: {sc:.6f}")
    print(f"Best so far: n={len(best_sequence) if best_sequence is not None else '?'}: {best_score:.6f}")

    # Phase 2: Focused optimization (next 50% of time)
    t_focus_end = t0 + (budget_s - 5.0) * 0.75
    print(f"\n=== Phase 2: Focused Optimization (until t={t_focus_end - t0:.0f}s) ===")

    # Pick top candidates with different structures
    seen = set()
    top_candidates = []
    for sc, seq, n, lbl in candidates:
        key = (n, lbl)
        if key not in seen:
            top_candidates.append((sc, seq, n, lbl))
            seen.add(key)
        if len(top_candidates) >= 3:
            break

    # Always include the best
    if best_sequence is not None:
        best_n = len(best_sequence)
        already_included = any(n == best_n for _, _, n, _ in top_candidates)
        if not already_included:
            top_candidates.insert(0, (best_score, best_sequence.copy(), best_n, 'best'))

    for sc, f_init, n, lbl in top_candidates:
        if time.monotonic() > t_focus_end - 30:
            break
        print(f"\n--- Focused: n={n} {lbl} (start={sc:.6f}) ---")

        # Use combined optimization with more time
        f_opt, s_opt = combined_optimize(f_init, t_focus_end,
                                          verbose_prefix=f"[n={n} {lbl}] ")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  => NEW BEST: {best_score:.6f}")

    # Phase 3: Final polish (last 25% of time, minus safety margin)
    t_polish_end = deadline
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")

    if best_sequence is not None and time.monotonic() < t_polish_end - 10:
        n_best = len(best_sequence)
        print(f"Polishing n={n_best}...")

        # LP polish
        if n_best <= 1500:  # LP is too slow for very large n
            lp_deadline = min(t_polish_end, time.monotonic() + (t_polish_end - time.monotonic()) * 0.6)
            f_pol, s_pol = lp_optimize(best_sequence, lp_deadline, step_init=0.03,
                                        verbose_prefix="[polish-LP] ")
            if s_pol < best_score:
                best_score = s_pol
                best_sequence = f_pol.copy()

        # Final subgradient
        if time.monotonic() < t_polish_end - 5:
            f_pol, s_pol = nesterov_optimize_v2(best_sequence, t_polish_end,
                                                 lr_init=0.003, top_k_init=15,
                                                 top_k_max=30,
                                                 verbose_prefix="[polish-sub] ")
            if s_pol < best_score:
                best_score = s_pol
                best_sequence = f_pol.copy()

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(1024)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Final verification
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed evaluation! Using fallback.")
        best_sequence = np.ones(512)
        final_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
