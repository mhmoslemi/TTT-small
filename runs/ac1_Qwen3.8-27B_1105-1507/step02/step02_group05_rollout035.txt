    """Build the matrix A such that A @ g = convolve(a, g, 'full')"""
    n = len(a)
    m = 2*n - 1
    rows = []
    cols = []
    vals = []
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    return A

    c = -np.ones(n)

from scipy.optimize import linprog, minimize

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):

def resample_seq(src, dest_n):
    """Resample 1D array to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def cross_correlation_fft(p, a):
    """Compute c[j] = sum_k p[k] * a[k-j] for j=0..len(a)-1 using FFT."""
    np_len = len(p)
    na_len = len(a)
    N = np_len + na_len - 1
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1
    P = np.fft.rfft(p, N_fft)
    A = np.fft.rfft(a, N_fft)
    result = np.fft.irfft(P * np.conj(A), N_fft)[:na_len]
    return result


def logsumexp_grad(a, T):
    """
    Compute gradient of f(a) = (1/T)*log(sum_k exp(T*b_k)) where b = conv(a,a).
    Returns (f_value, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    p = exp_Tb / exp_Tb.sum()
    f_val = float(Tb.max() / T + np.log(exp_Tb.sum()) / T)

    # grad_j = 2 * sum_k p_k * a[k-j] - p[2j]*a[j] (if 2j valid)
    cross = cross_correlation_fft(p, a)
    grad = 2.0 * cross
    # Correction for k = 2j
    idx_2j = 2 * np.arange(n)
    valid = idx_2j < len(p)
    if np.any(valid):
        grad[valid] -= p[idx_2j[valid]] * a[valid]
    return f_val, grad

    # Since homogeneous, minimizing smooth_max(b) with sum=1 is equivalent

    scale = (1.0 / s - np.maximum(x, 0.0) / (s * s))
    grad_x = grad_a * mask * np.maximum(scale, 0)
    # Actually, for L-BFGS-B let's just use the normalized a directly

def lse_lbfgs(a0, T, deadline, tracker, max_iter=200):
    """Run L-BFGS-B on log-sum-exp smoothed objective."""
    a = np.maximum(a0.copy(), 1e-6)
    a = a / a.sum()

    def obj_and_grad(x):
        if time.monotonic() > deadline:
            return 1e10, np.zeros_like(x)
        ax = np.maximum(x, 1e-8)
        ax = ax / ax.sum()
        try:
            f, g = logsumexp_grad(ax, T)
            # Project gradient back: we're optimizing on the simplex implicitly
            # by normalizing inside; for L-BFGS-B just return grad of f w.r.t. x
            # Since f depends on ax = x/sum(x), the chain rule gives:
            # df/dx_j = sum_k (df/dax_k) * dax_k/dx_j
            # dax_k/dx_j = (delta_kj * sum(x) - ax_k) / sum(x)^2
            # This is complex; simpler to just return g directly as approximation
            return f, g / ax.sum()
        except Exception:
            return 1e10, np.zeros_like(x)

    bounds = [(0.0, None)] * len(a0)
    try:
        res = minimize(obj_and_grad, a, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'gtol': 1e-10, 'ftol': 1e-14})
        if res.fun < 1e9:
            a_res = np.maximum(res.x, 0.0)
            if a_res.sum() > 1e-8:
                a_res = a_res / a_res.sum()
                tracker.update(a_res)
                return a_res
    except Exception as e:
        print(f"  LSE-LBFGS error: {e}", flush=True)
    return a

    tracker.update(a)
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # Line search: try several step sizes
        best_t = 0.0
        best_J = current_J
        best_a = None

        # Generate candidate step sizes
        base = max(0.05, 0.4 * (1.0 - it / float(max(1, max_iters))))
        candidates = [base, base * 0.5, base * 1.5, base * 0.25, 1.0]
        candidates = [min(1.0, max(0.01, t)) for t in candidates]

        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-12:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = t
                best_a = a_trial

        if best_a is None:
            # No improvement found, try very small step
            t_small = 0.01
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-12:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - tracker.t_start
            print(f"    LP it={it} J={current_J:.8f} t={elapsed:.0f}s", flush=True)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n))

    # Random
    inits.append(rng.uniform(0.2, 1.0, n))

    # Cosine families
    for freq in [1, 2, 3, 4, 5, 7]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))

    # Quadratic bump
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.05)

    # W-shaped
    inits.append(1.0 + 0.5 * np.cos(4.0 * np.pi * t))

    # Smooth random (low-pass filtered noise)
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(noise, kernel, mode='same') + 1.0)
    inits[-1] = np.maximum(inits[-1], 0.1)

    # Lacunary structure (sparse spikes smoothed)
    n_spikes = max(4, n // 32)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    kw = max(3, n // 128)
    kern = np.ones(kw) / kw
    inits.append(np.convolve(sparse, kern, mode='same') + 0.05)

    # Power law
    inits.append((t + 0.05) ** rng.uniform(-0.5, 1.0))

    # Two-peaked
    inits.append(np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2) + 0.05)

    # Chirp
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * (5 * t + 3 * t ** 2)))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Minimize J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using LP mirror descent,
    log-sum-exp gradient polishing, and multi-scale exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
            print(f"Parent loaded: n={parent.size} J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-scale LP exploration ===
    t0_phase1 = time.monotonic()
    phase1_budget = 0.30 * (deadline - time.monotonic())
    phase1_deadline = time.monotonic() + phase1_budget

    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-scale LP exploration ({phase1_budget:.0f}s budget)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [128, 256, 512, 1024]
    if parent is not None and parent.size in [64, 256, 512, 2048]:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()

        # Determine iterations based on remaining time and n
        remaining = phase1_deadline - time.monotonic()
        n_inits = min(6, max(3, int(remaining / max(5, n // 100))))
        iters_per_init = max(20, min(200, int(remaining / (n_inits * max(0.1, n / 2000)))))

        print(f"\n  n={n}: {n_inits} inits, ~{iters_per_init} LP iters each", flush=True)

        inits = generate_initializations(n, rng, parent)
        inits = inits[:n_inits]

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()
            a = run_lp_iterations(a0, iters_per_init, phase1_deadline, tracker)

        elapsed = time.monotonic() - t0
        tracker.report(f"Phase1 n={n} ({elapsed:.1f}s)")

    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    # === Phase 2: Deep LP optimization ===
    t0_phase2 = time.monotonic()
    phase2_budget = 0.35 * (deadline - time.monotonic())
    phase2_deadline = time.monotonic() + phase2_budget

    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Deep LP optimization ({phase2_budget:.0f}s budget)", flush=True)
    print(f"{'='*60}", flush=True)

    n_best = tracker.best.size
    a = tracker.best.copy()

    # Run LP iterations with increasing patience
    total_iters = 0
    while time.monotonic() < phase2_deadline:
        # Run a batch of LP iterations
        batch_deadline = min(phase2_deadline, time.monotonic() + 30.0)
        a = run_lp_iterations(a, 100, batch_deadline, tracker)
        total_iters += 100

        # Periodic restart: add small noise and continue
        if total_iters % 500 == 0 and time.monotonic() < phase2_deadline - 10:
            noise_scale = 0.02 * a.mean()
            a_noisy = a + rng.normal(0, noise_scale, n_best)
            a_noisy = np.maximum(a_noisy, 1e-6)
            a_noisy = a_noisy / a_noisy.sum()
            J_noisy = compute_J(a_noisy)
            if J_noisy < tracker.best_score + 0.01:
                a = a_noisy
    tracker.report("Phase2")

    # === Phase 3: Log-sum-exp gradient polishing ===
    t0_phase3 = time.monotonic()
    phase3_budget = 0.20 * (deadline - time.monotonic())
    phase3_deadline = time.monotonic() + phase3_budget

    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: Log-sum-exp gradient polish ({phase3_budget:.0f}s budget)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_polish = tracker.best.size
        if n_polish <= 2048:
            T_values = [5, 10, 20, 50, 100]
            a_curr = tracker.best.copy()
            for T in T_values:
                if time.monotonic() > phase3_deadline:
                    break
                max_iter = max(50, int((phase3_deadline - time.monotonic()) / (T_values.index(T) + 1) / max(1, n_polish / 500)))
                a_curr = lse_lbfgs(a_curr, T, phase3_deadline, tracker, max_iter=min(max_iter, 300))
            tracker.report("Phase3")

    # === Phase 4: Final LP polish ===
    t0_phase4 = time.monotonic()
    phase4_budget = max(30, deadline - time.monotonic())
    phase4_deadline = time.monotonic() + phase4_budget

    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Final LP polish ({phase4_budget:.0f}s budget)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a_final = tracker.best.copy()
        n_final = a_final.size

        # Multiple rounds of LP with decreasing step sizes
        for round_idx in range(5):
            if time.monotonic() > phase4_deadline:
                break
            a_final = run_lp_iterations(a_final, 200, phase4_deadline, tracker)

            # Try a restart with structured perturbation
            if time.monotonic() < phase4_deadline - 20:
                # Smooth perturbation
                smooth_noise = rng.standard_normal(n_final)
                k_size = max(3, n_final // 64)
                kern = np.ones(k_size) / k_size
                smooth_noise = np.convolve(smooth_noise, kern, mode='same')
                a_perturbed = a_final + 0.03 * smooth_noise
                a_perturbed = np.maximum(a_perturbed, 1e-6)
                a_perturbed = a_perturbed / a_perturbed.sum()
                if compute_J(a_perturbed) < tracker.best_score + 0.005:
                    a_final = a_perturbed

        tracker.update(a_final)
        tracker.report("Phase4")

    # === Final verification ===
    if tracker.best is not None:
        final_score = float(evaluate_sequence(tracker.best.tolist()))
        if np.isfinite(final_score):
            tracker.best_score = final_score

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic() - tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
