- The feasible set is {f >= 0, sum(f) = 1} (the probability simplex)
- The vertices are the unit vectors e_i
- So the FW step is: find the i that minimizes the directional derivative, then move toward e_i

This is very fast (no LP needed) but converges slowly. However, it could be useful as a polishing step.

def pair_adjust(f, n_trials=5000, rng=None):

        if i == j:
            continue
        if f[i] < 1e-10 and f[j] < 1e-10:
            continue
        
        # Try moving mass from i to j and vice versa
        best_local = best_score
        best_i, best_j = f[i], f[j]
        
        for delta in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5] * max(0.01, f[i], f[j]):
            for sign in [1, -1]:

                fj_new = max(0, f[j] - sign * delta)
                if fi_new < 0 or fj_new < 0:

                f[i], f[j] = fi_new, fj_new
                score = compute_score(f)

                    best_i, best_j = fi_new, fj_new

                # Reset
                f[i], f[j] = fi_new, fj_new
        
        # Restore
        f[i], f[j] = best_i, best_j
        if best_local < best_score - 1e-15:

    return f, best_score

"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_score(f):
    """Compute 2*n*max(conv(f,f))/(sum(f))^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)


def compute_score_fft(f, n_fft=None):
    """FFT-based score computation (faster for large n)."""
    n = len(f)
    if n_fft is None:
        n_fft = 1 << (2 * n - 1).bit_length()
    s = f.sum()
    if s < 1e-15:
        return np.inf
    F = np.fft.rfft(f, n_fft)
    conv = np.fft.irfft(F * F, n_fft)[:2 * n - 1]
    return 2.0 * n * conv.max() / (s * s)


def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A[k, i] = f[k - i]
    return A


def build_reduced_conv_matrix(f, active_indices):
    """Build conv matrix for only specified convolution positions."""
    n = len(f)
    A = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A[idx, i] = f[k - i]
    return A


def lp_direction(f, M, active_indices=None, time_limit=3):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M for active k's, g >= 0."""
    n = len(f)
    if active_indices is not None:
        A = build_reduced_conv_matrix(f, active_indices)
        b = np.full(len(active_indices), M)
    else:
        A = build_conv_matrix(f)
        b = np.full(2 * n - 1, M)

    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': time_limit})
    if not res.success or res.x is None:
        return None
    return np.clip(res.x, 0, None)


def lp_refine(f, time_budget, rng, t0=0.3, n_reduce=None, full_every=15, lp_time=2):
    """LP-based refinement with reduced constraints and adaptive step size."""
    n = len(f)
    if n_reduce is None:
        n_reduce = max(20, n // 4)

    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    stuck = 0
    start = time.monotonic()
    iteration = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        # Choose constraint set
        use_full = (iteration % full_every == 0) or (stuck > 3)
        if use_full:
            active = None
        else:
            active = np.argsort(conv)[::-1][:n_reduce]

        g = lp_direction(f, M, active, time_limit=lp_time)
        if g is None or g.sum() < 1e-12:
            stuck += 5
            if stuck > 10:
                break
            f = f * (1 + 0.05 * rng.normal(0, 1, n))
            f = np.clip(f, 0, None)
            f = normalize(f)
            continue

        g = g / g.sum()

        # Adaptive step size: try several, pick best
        improved = False
        steps = [min(1.0, t * 1.5), t, t * 0.5, t * 0.25, t * 0.1]
        for step in steps:
            f_new = (1.0 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score(f_new)
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_f = f_new.copy()
                f = f_new
                improved = True
                # Adjust step size based on improvement magnitude
                if step <= t:
                    t = min(1.0, t * 1.03)
                else:
                    t = max(0.01, t * 0.95)
                break

        if not improved:
            t = max(0.001, t * 0.6)
            stuck += 1
            if stuck > 8:
                # Perturbation to escape
                f_pert = f * (1 + 0.1 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.1
                    stuck = 0
                elif stuck > 20:
                    break

        iteration += 1

    return best_f, best_score


def fft_pressure(f, n_iter=1000, eta=0.05, rng=None, hot_frac=0.5):
    """FFT-accelerated pressure update: reduce elements contributing to conv peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score(f)
    stuck = 0

    for it in range(n_iter):
        # FFT convolution
        F = np.fft.rfft(f, n_fft)
        conv = np.fft.irfft(F * F, n_fft)[:2 * n - 1]
        M = conv.max()

        # Identify hot positions (top fraction of peaks)
        threshold = M * (1.0 - hot_frac)
        hot_mask = conv >= threshold
        n_hot = hot_mask.sum()

        if n_hot == 0:
            break

        # Compute pressure vectorized:
        # pressure[i] = sum_{j: hot} f[j-i] where 0 <= j-i < n
        # This is cross-correlation of f with hot indicator
        hot_vec = np.zeros(2 * n - 1)
        hot_vec[hot_mask] = 1.0

        # pressure[i] = sum_k f[k] * hot_vec[i+k] for valid k, i+k
        # = (f * reverse(hot_vec))[i + n - 1] ... let me use direct conv
        # Actually: pressure[i] = sum_k f[k] * hot_vec[i+k] = cross_correlate(f, hot_vec, lag=i)
        # np.correlate(hot_vec, f, 'valid') gives sum_k hot_vec[k] * f[k] for offset...
        # Let me just use: for each i, sum_{k=0}^{n-1} f[k] * hot_vec[i+k] if i+k < 2n-1
        # This is the "valid" part of the convolution of f with reversed hot_vec
        hot_rev = hot_vec[::-1]
        # np.convolve(f, hot_rev, 'valid') has length 2n-1-n+1 = n
        pressure = np.convolve(f, hot_rev, mode='valid')

        if pressure.max() > 1e-15:
            pressure = pressure / pressure.max()

        eta_eff = eta * (0.99 ** (it // 50))
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        new_score = compute_score(f_new)
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 80:
                f_pert = f * (1 + 0.15 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stuck = 0
                else:
                    break

    return best_f, best_score


def softmax_gradient_descent(f, rng, time_budget=60, T=10.0, lr=0.01):
    """Gradient descent using soft-max relaxation of max(conv)."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()

    def soft_objective(x):
        """Soft-max based objective (differentiable)."""
        x = np.abs(x)  # enforce non-negativity
        s = x.sum()
        if s < 1e-10:
            return 1e6
        conv = np.convolve(x, x)
        # Soft max approximation
        max_c = conv.max()
        soft_max = max_c + np.log(np.sum(np.exp((conv - max_c) / T)) / len(conv)) * T
        return 2.0 * n * soft_max / (s * s) + 0.01 * np.sum(np.clip(-x, 0, None) ** 2)

    x = f.copy()
    current_val = soft_objective(x)
    prev_val = current_val

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        # Numerical gradient
        grad = np.zeros(n)
        eps = 0.001
        for i in range(n):
            x[i] += eps
            val_plus = soft_objective(x)
            x[i] -= 2 * eps
            val_minus = soft_objective(x)
            x[i] += eps
            grad[i] = (val_plus - val_minus) / (2 * eps)

        # Project gradient step
        x_new = x - lr * grad
        x_new = np.clip(x_new, 0, None)
        s = x_new.sum()
        if s < 1e-10:
            break
        x_new = x_new / s

        new_val = soft_objective(x_new)
        if new_val < current_val:
            x = x_new
            current_val = new_val
            actual_score = compute_score(x)
            if actual_score < best_score:
                best_score = actual_score
                best_f = x.copy()
        else:
            lr *= 0.9

    return best_f, best_score


def coordinate_descent(f, n_sweeps=5, rng=None):
    """Coordinate descent with geometric search."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_score = compute_score(f)

    factors = [0.2, 0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0, 5.0]

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            best_val = orig
            best_local = best_score

            if orig < 1e-10:
                candidates = [0.001, 0.005, 0.01, 0.02, 0.05]
            else:
                candidates = [orig * fac for fac in factors]

            for val in candidates:
                val = max(0.0, val)
                f[i] = val
                score = compute_score(f)
                if score < best_local:
                    best_local = score
                    best_val = val

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def pair_swap_search(f, n_trials=2000, rng=None):
    """Try swapping/adjusting pairs of elements."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_score = compute_score(f)
    improvements = 0

    for _ in range(n_trials):
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        fi, fj = f[i], f[j]
        if fi < 1e-12 and fj < 1e-12:
            continue

        best_local = best_score
        best_i, best_j = fi, fj

        # Try splitting mass between i and j
        total = fi + fj
        for frac in [0.0, 0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9, 1.0]:
            f[i] = total * frac
            f[j] = total * (1.0 - frac)
            score = compute_score(f)
            if score < best_local:
                best_local = score
                best_i, best_j = f[i], f[j]

        f[i], f[j] = best_i, best_j
        if best_local < best_score - 1e-15:
            best_score = best_local
            improvements += 1
        elif improvements > 50 and rng.random() > 0.1:
            continue

    return f, best_score
