"""
Optimized step function coefficient search using LP-based alternating,
soft-max L-BFGS polish, and multi-scale initialization.
Targets objective below 1.5027 by exploiting large sequence lengths
and efficient gradient computation via FFT-based convolution.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_obj(a):
    """Compute objective: 2*n*max(convolve(a,a))/(sum(a))^2"""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_lp_matrix(a):
    """Build matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_min_m(a):
    """
    Solve: min M s.t. (a*b)[k] <= M for all k, sum(b)=1, b>=0, M>=0.
    Returns (b_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    A = build_lp_matrix(a)
    # Variables: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    # Inequality constraints: A @ b - M <= 0 for each row
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], float(res.x[-1])
    return None, None


def soft_max_value_and_weights(conv, T):
    """Compute soft-max value and probability weights."""
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    sm_val = max_c + T * np.log(np.sum(exp_vals))
    return sm_val, p


def compute_soft_max_grad(a, T):
    """
    Compute gradient of 2n * soft_max(conv(a,a), T) with respect to a.
    Uses FFT-based convolution for efficiency.
    """
    n = len(a)
    conv = np.convolve(a, a)
    sm_val, p = soft_max_value_and_weights(conv, T)

    # grad[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j in [0, n-1])
    # This equals 2 * convolve(p, a[::-1], 'full')[n-1 : 2*n-1]
    a_flip = a[::-1]
    conv_grad = np.convolve(p, a_flip, mode='full')
    grad = 2.0 * conv_grad[n - 1:2 * n - 1]

    return sm_val, grad


def lp_alternating(a, max_iters=50, time_limit=30.0):
    """
    LP-based alternating method:
    Iteratively solve min M s.t. a*b <= M, sum(b)=1, b>=0.
    Set a = b* and repeat.
    """
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        b_opt, M_opt = solve_lp_min_m(a)
        if b_opt is None:
            break

        # Normalize
        s = np.sum(b_opt)
        if s < 1e-10:
            break
        b_opt = b_opt / s
        b_opt = np.maximum(b_opt, 0.0)

        # Check improvement
        obj = compute_obj(b_opt)
        if obj < best_obj:
            best_a = b_opt.copy()
            best_obj = obj

        # Damped update to avoid cycling
        a_new = 0.7 * a + 0.3 * b_opt
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = b_opt
        else:
            a = a_new / s

        # Check convergence
        if obj >= best_obj * (1 - 1e-10):
            # Try harder damp
            a_new = 0.5 * a + 0.5 * b_opt
            a_new = np.maximum(a_new, 0.0)
            s = np.sum(a_new)
            if s > 1e-10:
                a = a_new / s
            else:
                a = b_opt
        else:
            a = b_opt.copy()

    return best_a, best_obj


def lbfgs_polish(a, T_init=0.01, max_time=30.0, max_iter=500):
    """
    Polish using L-BFGS-B on softmax-parameterized objective.
    Minimizes 2n * soft_max(conv(a,a), T) where a = softmax(x).
    """
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Initialize x from a (softmax inverse)
    x = np.log(a + 1e-30) - np.log(a + 1e-30).mean()
    T = T_init

    def objective_and_grad(x):
        nonlocal T
        # Softmax
        x_shift = x - x.max()
        exp_x = np.exp(x_shift)
        a_sm = exp_x / np.sum(exp_x)

        conv = np.convolve(a_sm, a_sm)
        sm_val, p = soft_max_value_and_weights(conv, T)
        obj = 2.0 * n * sm_val

        # Gradient w.r.t. a
        a_flip = a_sm[::-1]
        conv_grad = np.convolve(p, a_flip, mode='full')
        grad_a = 2.0 * conv_grad[n - 1:2 * n - 1]

        # Chain rule: dx -> da via softmax Jacobian
        # da_i/dx_j = a_i * (delta_ij - a_j)
        # df/dx_j = a_j * grad_a_j - a_j * sum_i(grad_a_i * a_i)
        dot = np.dot(grad_a, a_sm)
        grad_x = a_sm * (grad_a - dot)

        return obj, grad_x

    # Use L-BFGS-B
    try:
        result = minimize(
            lambda x: objective_and_grad(x)[0],
            x,
            jac=lambda x: objective_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': 5000, 'gtol': 1e-8}
        )
        if result.x is not None:
            x_opt = result.x
            x_shift = x_opt - x_opt.max()
            a_opt = np.exp(x_shift) / np.sum(np.exp(x_shift))
            a_opt = np.maximum(a_opt, 0.0)
            s = np.sum(a_opt)
            if s > 1e-10:
                a_opt = a_opt / s
                obj = compute_obj(a_opt)
                if obj < best_obj:
                    best_a = a_opt.copy()
                    best_obj = obj
    except Exception as e:
        pass

    # Adaptive T: try smaller T for sharper max approximation
    for T_try in [T_init * 0.5, T_init * 0.1, T_init * 0.01]:
        if time.monotonic() - start > max_time:
            break
        T = T_try
        x = np.log(best_a + 1e-30) - np.log(best_a + 1e-30).mean()
        try:
            result2 = minimize(
                lambda x: objective_and_grad(x)[0],
                x,
                jac=lambda x: objective_and_grad(x)[1],
                method='L-BFGS-B',
                options={'maxiter': 200, 'maxfun': 2000, 'gtol': 1e-9}
            )
            if result2.x is not None:
                x_opt = result2.x
                x_shift = x_opt - x_opt.max()
                a_opt = np.exp(x_shift) / np.sum(np.exp(x_shift))
                a_opt = np.maximum(a_opt, 0.0)
                s = np.sum(a_opt)
                if s > 1e-10:
                    a_opt = a_opt / s
                    obj = compute_obj(a_opt)
                    if obj < best_obj:
                        best_a = a_opt.copy()
                        best_obj = obj
        except Exception:
            pass
        if time.monotonic() - start > max_time:
            break

    return best_a, best_obj


def structured_init(n, rng, init_type=None):
    """Generate structured initializations."""
    if init_type is None:
        init_type = rng.integers(0, 8)

    if init_type == 0:
        # Uniform
        a = np.ones(n)
    elif init_type == 1:
        # Gaussian
        center = n / 2.0
        width = n * rng.uniform(0.15, 0.4)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    elif init_type == 2:
        # Cosine window
        a = np.cos(np.pi * np.arange(n) / (n - 1 + 1e-10)) ** 2 + 0.1
    elif init_type == 3:
        # Sine modulation
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(a, 0.01)
    elif init_type == 4:
        # Edge-heavy
        a = np.ones(n)
        edge_w = rng.uniform(1.5, 4.0)
        a[:n // 10] = edge_w
        a[-(n // 10):] = edge_w
    elif init_type == 5:
        # Periodic pattern
        period = max(2, int(rng.integers(3, max(4, n // 50))))
        pattern = np.maximum(0.1, rng.random(period))
        a = np.tile(pattern, n // period + 1)[:n]
    elif init_type == 6:
        # Random with smoothing
        a = np.maximum(0.1, rng.random(n))
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    else:
        # Logarithmic spacing
        a = 1.0 / (1.0 + 0.5 * np.abs(np.arange(n) - n / 2.0) / (n / 2.0))
        a = np.maximum(a, 0.05)

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def block_perturb_refine(a, rng, max_time=10.0):
    """Perturb random blocks to escape local minima."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        width = max(1, int(n * rng.uniform(0.005, 0.08)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        sigma = 0.02 + 0.05 * max(0.0, 1.0 - it / 50000)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.maximum(candidate, 0.0)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate.copy()
            best_obj = obj

    return best_a, best_obj
