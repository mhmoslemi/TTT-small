1. F = FFT(f, L)
2. |F_target[k]| = sqrt(max(0, G_ideal[k])) (or some scaled version)

4. F_new = |F_target| * exp(i * Phase_F_target)
5. f_new = Re(IFFT(F_new))
6. f_new = max(f_new, 0)
7. f_new = f_new / sum(f_new)

This should work! Let me also consider a "soft" version where we interpolate between the current |F| and the target:

Wait, the direct computation of pressure in the soft version is O(n^2) per iteration, which for n=1024 is about 1M operations per iteration. That's fine for a few thousand iterations.

But actually, I can use FFT to make it O(n log n). Let me think...

pressure[j] = sum_{i=0}^{n-1} f[i] * w[j+i]

This is the cross-correlation of f and w, evaluated at specific lags. Specifically:
correlation(f, w)[lag] = sum_i f[i] * w[i+lag]

So pressure[j] = correlation(f, w)[j] for j = 0, ..., n-1.

In terms of FFT:
correlation(f, w) = IFFT(FFT(f) * conj(FFT(w))) (with appropriate zero-padding)

Actually, the standard formula is:
(x ⊛ y)[k] = sum_n x[n+k] * conj(y[n]) = IFFT(FFT(x) * conj(FFT(y)))

So: pressure = IFFT(FFT(w, L) * conj(FFT(f, L)))[:n] (taking the first n elements)

Wait, I need to be more careful. Let me define:
(f ⊛ w)[k] = sum_{n=0}^{n-1} f[n] * w[n+k] (cross-correlation, lag k)

This equals: IFFT(conj(FFT(f, L)) * FFT(w, L))[k] for k = 0, 1, ... (circular correlation)

With zero-padding to L >= 2n-1+max_lag, this gives the linear cross-correlation.

So:
F = rfft(f, L)
W = rfft(w, L)
corr = irfft(conj(F) * W, L)  # This gives sum f[n]*w[n+k] for k=0,...,L-1 (circular)
pressure = corr[:n]  # First n lags

Wait, I need to double-check the convention. In numpy:
np.fft.irfft(np.conj(np.fft.rfft(f, L)) * np.fft.rfft(w, L), L)

This computes the circular cross-correlation: sum_n f[n] * w[(n+k) mod L]

With L >= 2n-1 (actually L >= (n-1) + (2n-2) + 1 = 3n-2 to be safe for the lag range we need), this becomes the linear cross-correlation for lags 0 to L-1-(n-1).

Since we need lags 0 to n-1, and w has length 2n-1, we need (n-1) + (2n-2) = 3n-3 < L. So L >= 3n-2 should work.

"""
Multi-method optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses multiplicative updates, IFTD (Iterative Fourier Transform for flat autocorrelation),
softmax gradient descent, and LP-based refinement with diverse initializations.

import numpy as np

def multiplicative_update(f, deadline, lr=0.05, max_iters=20000, verbose_prefix=""):
    """
    Multiplicative update: f[j] *= exp(-lr * f[m-j]) where m = argmax(f*f).
    Plus a soft variant using softmax weighting.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    it = 0
    no_improve = 0
    
    # Precompute FFT sizes
    L_corr = 1
    while L_corr < 3 * n:
        L_corr *= 2
    
    while time.monotonic() < deadline and it < max_iters:
        conv = np.convolve(f, f)
        m = int(np.argmax(conv))
        current_max = conv[m]
        
        if it % 1000 == 0:
            score = evaluate_sequence(f.tolist())
            if score < best_score - 1e-10:
                best_score = score
                best_f = f.copy()
                no_improve = 0
                if it % 5000 == 0:
                    print(f"    {verbose_prefix}MU iter {it}: score={best_score:.6f}, max_conv={current_max:.6e}")
            else:
                no_improve += 1000
                if no_improve > 5000:
                    lr *= 0.5
                    no_improve = 0
                    if lr < 1e-10:
                        break
        
        # Hard multiplicative update
        # pressure[j] = f[m-j] if 0 <= m-j < n, else 0
        j_min = max(0, m - (n - 1))
        j_max = min(m, n - 1)
        # f[m-j] for j in [j_min, j_max] is f[m-j_max] to f[m-j_min]
        # m-j ranges from m-j_max to m-j_min
        idx_start = m - j_max
        idx_end = m - j_min + 1
        pressure_segment = f[idx_start:idx_end]
        
        # Apply update
        f[j_min:j_max+1] *= np.exp(-lr * pressure_segment)
        
        # Also apply a softer global pressure using the full convolution
        # Use FFT-based cross-correlation
        F = np.fft.rfft(f, L_corr)
        # For the soft part, use a decaying weight
        # Actually, let's just do the hard update for now and add soft later
        
        f = np.clip(f, 1e-15, 1000.0)
        S = float(np.sum(f))
        if S < 1e-10:
            break
        f = f / S
        
        it += 1
    
    score = evaluate_sequence(f.tolist())
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def iftd_flat_autocorr(f, deadline, max_iters=5000, t = 0.3, verbose_prefix=""):
    """
    IFTD: Iterative Fourier Transform to make f*f flat.
    Target spectrum: |F_target[k]|^2 proportional to max(0, Dirichlet_kernel[k])
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    
    M = 2 * n - 1  # length of convolution
    # FFT size >= M
    L = 1
    while L < M:
        L *= 2
    # Actually we need L >= M for the convolution to be captured
    # But for the IFTD to work well, larger L gives more "room"
    L = max(L, 4 * n)  # Use 4n for better frequency resolution
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    it = 0
    no_improve = 0
    
    # Precompute target spectrum
    # If g[k] = c for k=0,...,M-1 (flat), then G[k] = c * sum_{m=0}^{M-1} exp(-2*pi*i*k*m/L)
    # |G[k]| = c * |sin(pi*k*M/L) / sin(pi*k/L)|
    # We want |F[k]|^2 = G[k], so |F_target[k]| = sqrt(c * |D_k|) where D_k is the Dirichlet kernel
    # But G[k] can be negative! So we use max(0, G[k]).
    # Actually, let's use a different target: we want |F[k]| to be small for k != 0
    # but not zero (to allow the convolution to have the right shape).
    
    # Alternative target: make |F[k]|^2 proportional to a raised cosine or similar
    # that's peaked at DC.
    
    # Let's try: |F_target[k]| = (1 + cos(pi*k/L)) for k = 0,...,L/2
    # This is a Hann-like spectrum that's peaked at DC.
    
    k_vals = np.arange(L // 2 + 1)
    # Target: peaked at DC, decaying
    target_spec = 1.0 + np.cos(np.pi * k_vals / (L // 2))
    target_spec = np.sqrt(np.maximum(target_spec, 0))  # |F_target|
    
    # Scale target to match energy
    target_energy = np.sum(target_spec ** 2)
    target_spec = target_spec * np.sqrt(L / target_energy)  # Parseval: sum|f|^2 = (1/L)sum|F|^2
    
    while time.monotonic() < deadline and it < max_iters:
        # Forward FFT
        F = np.fft.rfft(f, L)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Update magnitude toward target
        mag_new = (1 - t) * mag + t * target_spec * (mag / (mag + 1e-15))
        # Actually, let's just set the magnitude to target
        # mag_new = target_spec
        # But this might be too aggressive. Use interpolation.
        mag_new = (1 - t) * mag + t * target_spec
        
        # Reconstruct
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, L)[:n]  # Take first n elements
        
        # Enforce non-negativity
        f_new = np.maximum(f_new, 0)
        
        # Normalize
        S_new = float(np.sum(f_new))
        if S_new < 1e-15:
            break
        f_new = f_new / S_new
        
        f = f_new
        it += 1
        
        if it % 500 == 0:
            score = evaluate_sequence(f.tolist())
            if score < best_score - 1e-10:
                best_score = score
                best_f = f.copy()
                no_improve = 0
                if it % 2000 == 0:
                    print(f"    {verbose_prefix}IFTD iter {it}: score={best_score:.6f}")
            else:
                no_improve += 500
                if no_improve > 5000:
                    t *= 0.5
                    no_improve = 0
                    if t < 0.01:
                        break
    
    score = evaluate_sequence(f.tolist())
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def softmax_gradient_descent(f, deadline, beta_start=20.0, beta_end=500.0,
                              lr_start=0.02, max_iters=10000, verbose_prefix=""):
    """
    Softmax gradient descent with Adam optimizer.
    Minimizes soft_max(conv(f,f)) with sum(f)=1 constraint (via projection).
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline and it < max_iters:
        # Adaptive beta
        it_frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** it_frac
        
        # Compute softmax gradient
        conv = np.convolve(f, f)
        max_conv = conv.max()
        exp_conv = np.exp(beta * (conv - max_conv))
        total = exp_conv.sum()
        if total < 1e-300:
            break
        p = exp_conv / total
        
        # Gradient: grad[j] = 2 * sum_i f[i] * p[j+i]
        # This is 2 * cross_correlation(f, p) at lag j
        # Use FFT
        L = 1
        while L < 3 * n:
            L *= 2
        F = np.fft.rfft(f, L)
        P = np.fft.rfft(p, L)
        corr = np.fft.irfft(np.conj(F) * P, L)[:n]
        grad = 2.0 * corr
        
        # Adam update
        it += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * (grad ** 2)
        m_hat = m_adam / (1 - beta1 ** it)
        v_hat = v_adam / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = f - lr * update
        f_new = np.clip(f_new, 0, 1000)
        S_new = float(np.sum(f_new))
        if S_new < 1e-10:
            f_new = f
            S_new = S
        f_new = f_new / S_new
        
        if it % 200 == 0:
            sc = evaluate_sequence(f_new.tolist())
            if np.isfinite(sc) and sc < best_score - 1e-10:
                f = f_new
                best_f = f.copy()
                best_score = sc
                no_improve = 0
                if it % 2000 == 0:
                    print(f"    {verbose_prefix}SM iter {it}: score={best_score:.6f} (beta={beta:.0f})")
            else:
                f = f_new  # Still accept the step (for exploration)
                no_improve += 200
                if no_improve > 3000:
                    lr = max(lr * 0.5, 1e-7)
                    no_improve = 0
                    if lr < 1e-7:
                        break
        else:
            f = f_new
    
    score = evaluate_sequence(f.tolist())
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def lp_based_iteration(f, deadline, max_iters=100, step_size=0.1, verbose_prefix=""):
    """
    LP-based direction finding: max sum(g) s.t. g>=0, conv(f,g) <= M.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    it = 0
    no_improve = 0
    
    # Build the constraint matrix (sparse would be better, but for n<=2048, dense is OK)
    # A @ g = conv(f, g), shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = f[k - j]
    
    while time.monotonic() < deadline and it < max_iters:
        conv = np.convolve(f, f)
        M = float(conv.max())
        
        # Solve LP: max 1^T g s.t. A g <= M, g >= 0
        c = -np.ones(n)
        try:
            result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                           bounds=[(0, None)]*n, method='highs')
            if not result.success:
                no_improve += 1
                if no_improve > 10:
                    break
                it += 1
                continue
        except Exception:
            no_improve += 1
            if no_improve > 10:
                break
            it += 1
            continue
        
        g = result.x
        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            it += 1
            continue
        
        g = g / sum_g  # Normalize to sum=1
        
        # Line search
        best_t = 0
        best_s = best_score
        for t in [step_size, step_size*0.5, step_size*1.5, step_size*0.25]:
            new_f = (1-t)*f + t*g
            new_f = np.clip(new_f, 0, 1000)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = t
        
        if best_s < best_score:
            f = (1-best_t)*f + best_t*g
            f = np.clip(f, 0, 1000)
            f = f / float(np.sum(f))
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 20 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.005 or no_improve > 20:
                break
        
        # Update A for next iteration
        for k in range(2 * n - 1):
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                A[k, j] = f[k - j]
        
        it += 1
    
    return best_f, best_score


def generate_initializations(n, rng):
    """Generate diverse initializations for length n."""
    inits = []
    
    # Uniform
    inits.append(('uniform', np.ones(n)))
    
    # Tukey windows
    for r in [0.2, 0.4, 0.6]:
        w = np.ones(n)
        m = int(np.floor(r * n / 4))
        if m > 0:
            for i in range(m):
                w[i] = 0.5 * (1 + np.cos(np.pi * (2*i/(2*m) - 1)))
                w[n-1-i] = w[i]
        inits.append((f'tukey_r{r}', w))
    
    # Real chirp: 1 + a*cos(2*pi*k^2/N)
    for a in [0.1, 0.2, 0.3, 0.5]:
        k = np.arange(n, dtype=np.float64)
        w = 1.0 + a * np.cos(2 * np.pi * k**2 / n)
        w = np.maximum(w, 0)
        if w.sum() > 0.01:
            inits.append((f'chirp_a{a}', w))
    
    # Linear ramp
    inits.append(('ramp', np.arange(1, n+1, dtype=np.float64)))
    
    # Two-lobe: 1 + b*cos(2*pi*k/n)
    for b in [0.3, 0.5, 0.7]:
        k = np.arange(n, dtype=np.float64)
        w = 1.0 + b * np.cos(2 * np.pi * k / n)
        w = np.maximum(w, 0)
        if w.sum() > 0.01:
            inits.append((f'twoloop_b{b}', w))
    
    # Super-Gaussian
    for alpha in [1.5, 2.0, 4.0]:
        k = np.arange(n, dtype=np.float64)
        center = (n-1) / 2.0
        w = np.exp(-((k - center) / (n*0.4)) ** (2*alpha))
        inits.append((f'supauss_a{alpha}', w))
    
    # Asymmetric: exponential decay
    inits.append(('exp_decay', np.exp(-np.arange(n, dtype=np.float64) / (n*0.3))))
    
    # Random smooth
    x = rng.standard_normal(n)
    kernel_len = max(3, n // 32)
    kernel = np.exp(-0.5 * (np.arange(-kernel_len, kernel_len+1) / (kernel_len/3)) ** 2)
    smooth = np.convolve(x, kernel, mode='same')
    smooth = np.abs(smooth) + 0.1
    inits.append(('rand_smooth', smooth))
    
    return inits


def perturbation_restart(f, rng, strength):
    """Add multiplicative noise."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 3.0)
    new_f = np.clip(new_f, 0, 1000)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def propose_candidate(budget_s=1500, **kwargs):
    """
    Multi-method optimization combining multiplicative updates, IFTD,
    softmax gradient descent, and LP-based refinement.
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)
    
    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)
    
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")
    
    # Choose lengths to try
    # Focus on 512 (parent length) and nearby
    lengths_to_try = [512, 768, 1024]
    
    # Allocate time
    total_budget = deadline - t0
    time_fractions = [0.45, 0.25, 0.15]
    # Remaining 15% for perturbation restarts
    
    for idx, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - 30:
            break
        
        n_budget = total_budget * time_fractions[idx]
        n_deadline = time.monotonic() + n_budget
        print(f"\n{'='*50}")
        print(f"n={n}, budget={n_budget:.0f}s")
        print(f"{'='*50}")
        
        # Generate initializations
        inits = generate_initializations(n, rng)
        
        # If parent is available, add it (interpolated if needed)
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.insert(0, ('parent', best_sequence.copy()))
            else:
                from numpy import interp
                parent_interp = interp(np.linspace(0, 1, n),
                                       np.linspace(0, 1, len(best_sequence)),
                                       best_sequence)
                parent_interp = np.maximum(parent_interp, 0)
                inits.insert(0, ('parent_interp', parent_interp))
        
        # For each initialization, run a short multi-method optimization
        n_inits = len(inits)
        per_init_time = n_budget / max(n_inits, 1)
        
        for label, f_init in inits:
            if time.monotonic() > n_deadline - 5:
                break
            
            f_init = np.clip(f_init.astype(np.float64), 0, 1000)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue
            f_init = f_init / S
            
            init_score = evaluate_sequence(f_init.tolist())
            if not np.isfinite(init_score):
                continue
            if init_score < best_score:
                best_score = init_score
                best_sequence = f_init.copy()
                print(f"  {label}: init score={init_score:.6f}")
            
            # Run optimization methods in sequence
            init_deadline = min(n_deadline, time.monotonic() + per_init_time)
            f_curr = f_init.copy()
            
            # Method 1: IFTD (new, fast)
            td = min(init_deadline, time.monotonic() + per_init_time * 0.3)
            if time.monotonic() < td - 1:
                f_curr, s = iftd_flat_autocorr(f_curr, td, max_iters=5000, t=0.3)
                if s < best_score:
                    best_score = s
                    best_sequence = f_curr.copy()
                    print(f"  {label} IFTD: {s:.6f}")
            
            # Method 2: Multiplicative update (new)
            td = min(init_deadline, time.monotonic() + per_init_time * 0.3)
            if time.monotonic() < td - 1:
                f_curr, s = multiplicative_update(f_curr, td, lr=0.05, max_iters=20000)
                if s < best_score:
                    best_score = s
                    best_sequence = f_curr.copy()
                    print(f"  {label} MU: {s:.6f}")
            
            # Method 3: Softmax gradient (proven effective)
            td = min(init_deadline, time.monotonic() + per_init_time * 0.3)
            if time.monotonic() < td - 1:

                                                     lr_start=0.02, max_iters=5000)

                    best_score = s
                    best_sequence = f_curr.copy()
                    print(f"  {label} SM: {s:.6f}")

        if best_sequence is not None and time.monotonic() < n_deadline - 10:

            if len(best_sequence) != n:
                from numpy import interp
                f_refine = interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(best_sequence)),
                                  best_sequence)
                f_refine = np.maximum(f_refine, 0)
                f_refine = f_refine / float(np.sum(f_refine))
            else:
                f_refine = best_sequence.copy()
            
            # IFTD refinement
            td = min(n_deadline, time.monotonic() + (n_deadline - time.monotonic()) * 0.3)
            if time.monotonic() < td - 2:
                f_r, s = iftd_flat_autocorr(f_refine, td, max_iters=5000, t=0.2)
                if s < best_score:
                    best_score = s
                    best_sequence = f_r.copy()
                    print(f"  Refine IFTD: {s:.6f}")
            
            # LP refinement
            td = min(n_deadline, time.monotonic() + (n_deadline - time.monotonic()) * 0.4)
            if time.monotonic() < td - 2 and n <= 1024:

        print(f"Perturbation restarts (best: {best_score:.6f})")
        print(f"{'='*50}")
        
        restart_budget = deadline - time.monotonic() - 10
        n_restarts = min(8, max(3, int(restart_budget / 40)))
        per_restart = restart_budget / n_restarts
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            
            strength = 0.02 + 0.02 * r
            f_pert = perturbation_restart(best_sequence, rng, strength)
            f_pert_deadline = time.monotonic() + per_restart * 0.8
            
            # Quick IFTD + SM
            td1 = min(f_pert_deadline, time.monotonic() + per_restart * 0.4)
            f_r, s = iftd_flat_autocorr(f_pert, td1, max_iters=3000, t=0.25)
            
            td2 = min(f_pert_deadline, time.monotonic() + per_restart * 0.4)
            f_r, s = softmax_gradient_descent(f_r, td2, beta_start=50, beta_end=400,
                                              lr_start=0
