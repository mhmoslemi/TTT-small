    linear_mask = valid & (k_vals != 2*i)
    quad_mask = valid & (k_vals == 2*i)
    
    def F(x):
        if x < 0:
            x = 0
        if x > 1:
            x = 1
        denom = (S + x) ** 2
        if denom < 1e-30:
            return np.inf
        full = base_conv.copy()
        full[linear_mask] += 2 * x * a[mi[linear_mask]]
        if quad_mask.any():
            full[quad_mask] += x * x
        return np.max(full) / denom
    
    # Golden section search on [0, 1]
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    lo, hi = 0.0, 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    fc = F(c)
    fd = F(d)
    
    for _ in range(60):  # ~60 iterations gives precision ~ (0.618)^60 ≈ 10^-13
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = F(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            fd = F(d)
    
    x_opt = (lo + hi) / 2
    a[i] = x_opt
    # Renormalize
    a = a / np.sum(a)
    return a
