def build_conv_matrix_sparse(f):
    n = len(f)
    # The convolution matrix is a "Kirkwood" (convolution) matrix
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # For row k: non-zero entries are at columns j = max(0, k-n+1) to min(k, n-1)
    
    # Build in COO format
    n_rows = 2 * n - 1
    rows_list = []
    cols_list = []
    vals_list = []
    
    for k in range(n_rows):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            rows_list.extend([k] * len(js))
            cols_list.extend(js.tolist())
            vals_list.extend(f[k - js].tolist())

def spectral_shaping(f, n_iter=10):
    """Try to flatten the power spectrum of f."""
    f = f.copy()
    n = len(f)
    F = np.fft.rfft(f)
    power = np.abs(F)**2
    # Target: flatten the power spectrum
    # Keep DC component fixed
    for _ in range(n_iter):
        # Compute target power: geometric mean of non-DC powers
        non_dc_power = power[1:]
        target = np.geometric_mean(non_dc_power)
        # Blend power spectrum towards flat
        new_power = power.copy()
        new_power[1:] = 0.5 * power[1:] + 0.5 * target
        # Take square root to get amplitude
        new_amp = np.sqrt(new_power)
        # Keep the phase
        phase = np.angle(F)
        new_F = new_amp * np.exp(1j * phase)
        new_F[0] = F[0]  # Keep DC
        # Inverse FFT
        new_f = np.fft.irfft(new_F, n=n)
        # Clip to non-negative
        new_f = np.clip(new_f, 0, None)
        if np.sum(new_f) < 1e-10:
            break
        f = new_f
    return f

"""
LP-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP direction finding with adaptive step size, spectral initialization,
and multi-length exploration with restarts.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Compute the objective value directly (fast, no type checks)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    j = np.repeat(np.arange(n), n)
    k_offset = np.tile(np.arange(n), n)
    k = j + k_offset
    valid = k < 2 * n - 1
    j = j[valid]
    k = k[valid]
    return csr_matrix((f[k - j], (k, j)), shape=(2 * n - 1, n))


def lp_step(f, M, step=0.3):
    """Perform one LP-based step. Returns (new_f, new_score, improved)."""
    n = len(f)
    S = float(np.sum(f))
    
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
    except Exception:
        return f, compute_score(f), False
    
    if not result.success or result.x is None:
        return f, compute_score(f), False
    
    g = result.x
    sum_g = float(np.sum(g))
    
    if sum_g < 1e-10:
        return f, compute_score(f), False
    
    # Rescale g to have same sum as f
    g = g * (S / sum_g)
    
    # Blend
    new_f = (1.0 - step) * f + step * g
    new_f = np.clip(new_f, 0.0, None)
    new_sum = float(np.sum(new_f))
    if new_sum < 1e-10:
        return f, compute_score(f), False
    new_f = new_f * (S / new_sum)
    
    new_score = compute_score(new_f)
    old_score = compute_score(f)
    
    if new_score < old_score:
        return new_f, new_score, True
    else:
        return f, old_score, False


def subgradient_step(f, lr=0.01, top_k=5):
    """Perform one subgradient step to reduce max convolution."""
    n = len(f)
    S = float(np.sum(f))
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    
    # Get top-k lags
    top_indices = np.argsort(conv)[-top_k:]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in top_indices:
        # ∂c_k/∂a_i = a[k-i] if 0 <= k-i < n
        k_minus_i = k - np.arange(n)
        valid = (k_minus_i >= 0) & (k_minus_i < n)
        grad[valid] += f[k_minus_i[valid]]
    
    # Normalize
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return f
    
    grad = grad / grad_norm
    
    # Step
    new_f = f - lr * grad
    new_f = np.clip(new_f, 0.0, None)
    new_sum = float(np.sum(new_f))
    if new_sum < 1e-10:
        return f
    new_f = new_f * (S / new_sum)
    
    return new_f


def spectral_init(n, rng, n_iter=20):
    """Create an initialization with approximately flat power spectrum."""
    # Start with random
    f = rng.uniform(0.5, 1.5, size=n)
    
    for _ in range(n_iter):
        F = np.fft.rfft(f)
        power = np.abs(F) ** 2
        
        # Target: make non-DC power more uniform
        non_dc = power[1:]
        target = np.sqrt(np.sum(non_dc) / len(non_dc))  # RMS
        
        # Blend towards flat
        new_power = power.copy()
        new_power[1:] = 0.7 * power[1:] + 0.3 * target
        new_power = np.maximum(new_power, 1e-10)
        
        # Reconstruct with same phase
        phase = np.angle(F)
        new_F = np.sqrt(new_power) * np.exp(1j * phase)
        new_f = np.fft.irfft(new_F, n=n)
        new_f = np.clip(new_f, 0.0, None)
        
        if np.sum(new_f) < 1e-10:
            break
        f = new_f
    
    return f
