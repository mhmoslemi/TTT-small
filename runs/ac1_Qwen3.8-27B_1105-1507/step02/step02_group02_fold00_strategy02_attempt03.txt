def evaluate_sequence(seq):
    # Defensive clipping
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)) or np.isnan(x) or np.isinf(x):
            return np.inf
    seq = [float(x) for x in seq]

    for x in seq:
        if x < 0:

    # Ensure no single very large number even if scaled differently
    # This check ensures we don't have situations where the fitness is driven by one giant number
    if max(seq) > 1000.0:

    sum_a = np.sum(seq)

        # But our clip already ensures nothing is beyond bounds
        return np.inf
    # Avoid nonsense numbers due to floating point errors

        global_score = np.inf
        print("Initial sequence invalid or evaluation time out", flush=True)
    best_sequence = global_heights
    best_score = global_score

    # Define the time budget
    total_budget = 1500
    start_time = time.time()
    remaining = total_budget

    # Using uniform starting point (if parent evaluation was rushed, we might need to start elsewhere but not provided here)
    # We'll do block-coordinate descent: optimize parts with adaptive DE

    # Define parameterization of DE based on sequence length
    def modify_bounds(original_length):
        """Generates adaptive bounds for DE step-function optimization."""
        # Lower bound: after clipping, must be non-negative
        lower_bound = 0.0
        # Upper bound: normalized to not exceed 1000 even if extended beyond typical values
        upper_bound = 1000.0
        return [(lower_bound, upper_bound)] * original_length

    # We'll define a transformation to standard length (power-of-two convention) for FFT behavior if needed
    # Since the full convolution depends on the length, we'll use the parent length to determine the scale
    parent_length = 512
    if len(best_sequence) != parent_length:
        # Normalize to parent_length if necessary
        parent_proportion = np.ones(parent_length) / parent_length
        # Then scale to reconcile the sum
        factor = len(best_sequence) / parent_length
        best_sequence = [x * factor for x in parent_proportion]

    # Block-coordinate descent: break the sequence into segments (not meaningful, but set to 1 segment) for DE
    def fitness_segment(segment_of_coefficients, current_center):
        """Fitness function that slightly perturbs the segment and evaluates."""
        # Evaluate the segment fitness only through the overall function
        segment_score = evaluate_sequence(segment_of_coefficients)
        # Also evaluate neighbors to ensure we aren't at a local optimum
        if current_center is not None:
            perturbation_noise = 0.01 * (np.random.normal(0, 1, len(segment_of_coefficients)) if len(segment_of_coefficients) > 1 else random.uniform(-0.01, 0.01))
            perturbed_center = [min(1000, max(0, x + noise)) for x, noise in zip(current_center, perturbation_noise.tolist()[:len(segment_of_coefficients)])]
            try:
                candidate_score = evaluate_sequence(perturbed_center)
            except:
                candidate_score = evaluate_sequence(current_center)
            # Check for relative improvement from the fixed point
            if candidate_score < segment_score:
                return candidate_score, perturbed_center
            else:
                return segment_score, perturbed_center
        else:
            # Either way, we return the segment's score
            return segment_score, segment_of_coefficients

    # Algorithm selection: iterative block DE with knowledge segmentation
    def block_evolution(desired_length=512):
        """Performs differential evolution on a fixed-length segment arrangement from the uniform population or parent."""
        nonlocal best_sequence, best_score
        try:
            # Choose length: when the product of coefficients is large, we want segmentation to align with structure
            # Clamp the segment lengths to powers of two for optimal FFT convolution if desired, but evaluate_sequence doesn't use FFT here
            # We'll use adaptive bounds depending on the current best data
            bounds = modify_bounds(desired_length)
            # Build initial population: keep best as a focus point
            # DE parameters: crossover=0.9, mutation=0.1
            result = differential_evolution(
                lambda x: evaluate_sequence(x),
                bounds=bounds,
                maxiter=50,
                popsize=200,
                mutation=(0.5, 1.0),
                crossover=0.7,
                seed=42
            )
            # Get the best from DE
            candidate = result.x
            candidate_score = result.fun
        except Exception as e:
            # In case of optimization failure or too many evaluations, use the current best
            candidate = best_sequence
            candidate_score = best_score
            print(f" DE Optimization failure: {e} Resorting to best candidate", flush=True)
        # Candidate normalization check: make sure sum >=0.01 and clipping wasn't extreme
        if candidate_score >= np.inf or candidate_score <= 0.0:
            return False
    
    # We'll go through several phases of DE evolution with diminishing budgets
    # Phase 1: optimize using the parent length
    # Total phase 1 time: 1000 seconds
    budget_phase1 = 800
    # Initial DE call for parent length (512)
    if time.time() - start_time < budget_phase1 and len(best_sequence) == 512:
        print(f"===Phase1: DE on fixed length {512}===", flush=True)
        optimal_candidate, exists_improvement = optimize_segment(512)
        if exists_improvement:
            best_sequence = optimal_candidate
            best_score = evaluate_sequence(optimal_candidate)
        remaining = max(0, budget_phase1 - (time.time() - start_time))
        # If remaining time allows, we can try higher dimensions with segmentation by multiple factors
        # But note: the evaluations are more expensive for longer arrays
    
    # Phase 2: try slightly larger or smaller segments
    # Let's consider length factors: 0.8, 0.9, 1.1, 1.2, 1.5 (powers that could be computationally manageable)
    length_factors = [0.8, 0.9, 1.1, 1.2, 1.5]
    for factor in length_factors:
        new_length = int(parent_length * factor)
        # But new_length must be positive and at worst only segment further
        if new_length < 1:
            new_length = 1
        budget = ((time.time() - start_time) if factor is longer, but let's decouple) but we have a total time constraint
        new_length_time = remaining / 20   # إحتمال
        remaining = remaining - new_length_time * factor  # Not exactly, better with fixed subtraction
        budget = budget_phase2 / (len(length_factors))
        if (time.time() - start_time) > total_budget - budget and new_length < 1000000:
            print(f"Not enough time for length factor {factor} optimization, {new_length} left.")
            continue
        else:
            print(f"===Phase2 Factor={factor}: DE on length {new_length}===", flush=True)
            if new_length == parent_length:
                continue   # Skip duplicate optimization if we are here
            optimal_candidate, exists_improvement = block_evolution(new_length)
            if exists_improvement:
                best_sequence = optimal_candidate
                best_score = evaluate_sequence(optimal_candidate)
            else:
                print("No improvement for this length, using previous best.")
    
    # Phase 3: final return unless invalid
    n_final = len(best_sequence)
    final_score = evaluate_sequence(best_sequence)
    if final_score == np.inf:
        # Fallback: use uniform
        best_sequence = np.ones(n_final) / n_final
        final_score = evaluate_sequence(best_sequence)
    elif final_score > best_score:
        print("===Final phase failed and returned worse score! Resorting to best known candidate from phase1/2 before failure.")  # But actually, phase2 might have been more promising
        best_sequence = [x for x in best_sequence]

    print(f"PARETO OPTIMAL: J={final_score:.6f} for sequence of length {n_final}")
    # Verify and format the result
    return [float(x) if x >= 0 else float(0) for x in best_sequence[:n_final]]
