    invphi2 = 1.0 - invphi

    d = lo + invphi * (hi - lo)

            c = d
            fc = fd

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if T > 0.0005:
                T = max(0.0005, T * 0.997)
        else:
            if it % 50 == 0:
                lr *= 0.95
                T = max(0.0005, T * 0.9)
            if lr < 1e-10:
                break

    return best_a, best_obj


def lag_targeted_perturb(a, rng, time_limit=20.0):
    """Perturb a by reducing contributions to the hottest lags."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(100000):
        if time.monotonic() - start > time_limit:
            break

        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        # Find top lags
        top_k = max(1, int(0.05 * (2 * n - 1)))
        hot_lags = np.argsort(conv)[-top_k:]

        # For each hot lag, reduce the a[j] that contribute most
        j = int(rng.integers(0, n))
        k = int(hot_lags[int(rng.integers(0, len(hot_lags)))])

        # Reduce a[j] and redistribute
        a_new = a.copy()
        reduction = a[j] * rng.uniform(0.05, 0.3)
        a_new[j] -= reduction
        if a_new[j] < 0:
            a_new[j] = 0
            reduction = a[j]
        # Distribute to random other positions
        num_targets = int(rng.integers(1, 4))
        targets = rng.integers(0, n, size=num_targets)
        for t_idx in targets:
            a_new[t_idx] += reduction / num_targets

        a_new = np.clip(a_new, 0.0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def block_refine(a, rng, time_limit=20.0):
    """Random block multiplicative perturbations."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200000):
        if time.monotonic() - start > time_limit:
            break

        width = max(1, int(n * rng.uniform(0.02, 0.15)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = a.copy()
        sigma = 0.02 + 0.06 * max(0.0, 1.0 - it / 200000.0)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s_new = np.sum(candidate)
        if s_new < 1e-10:
            continue
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = candidate.copy()
            a = candidate

    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        # Uniform with multiplicative noise
        a = np.ones(n) * np.exp(rng.normal(0, 0.3, n))
    elif mode == 1:
        # Edge-enhanced
        c = 1.0 / np.sqrt(2.0 * n - 1.0)
        a = np.ones(n) * 0.5
        a[0] = c * 2
        a[1] = c * 1.5
        a[-1] = c * 2
        a[-2] = c * 1.5
        a *= np.exp(rng.normal(0, 0.2, n))
    elif mode == 2:
        # Periodic
        period = max(2, int(n / rng.uniform(5, 30)))
        pattern = np.maximum(0, rng.random(period) + 0.1)
        a = np.tile(pattern, n // period + 1)[:n]
    elif mode == 3:
        # Bimodal (mass at two edges)
        center1 = n // 4
        center2 = 3 * n // 4
        w = n * 0.1
        a = np.exp(-0.5 * ((np.arange(n) - center1) / w) ** 2)
        a += np.exp(-0.5 * ((np.arange(n) - center2) / w) ** 2)
        a *= np.exp(rng.normal(0, 0.15, n))
    elif mode == 4:
        # Power law decay from center
        center = n / 2.0
        a = 1.0 / (1.0 + 0.1 * np.abs(np.arange(n) - center))
        a *= np.exp(rng.normal(0, 0.2, n))
    elif mode == 5:
        # Sinc-like
        x = np.linspace(-1, 1, n)
        a = np.sinc(2 * x) + 0.3
        a = np.maximum(0, a)
        a *= np.exp(rng.normal(0, 0.1, n))
    elif mode == 6:
        # Random with exponential distribution
        a = rng.exponential(1.0, n)
    else:
        # Random symmetric
        half = n // 2
        a_half = np.maximum(0, rng.normal(0.5, 0.3, half + 1))
        a = np.concatenate([a_half, a_half[-2::-1]])[:n]

    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def full_pipeline(a_init, time_budget):
    """Run the full optimization pipeline on a given initial sequence."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LP alternating
    lp_time = min(time_budget * 0.5, 40.0)
    max_iters = max(20, int(lp_time / 0.1))
    a, obj = lp_alternate_improved(a, max_iters=max_iters, time_limit=lp_time)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    # Phase 2: Softmax GD
    if time.monotonic() - start < time_budget * 0.8:
        gd_time = min(time_budget * 0.3, 20.0)
        a2, obj2 = softmax_gd(a, time_limit=gd_time)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

    # Phase 3: Second LP
    if time.monotonic() - start < time_budget * 0.9:
        lp_time2 = min(time_budget * 0.15, 15.0)
        a3, obj3 = lp_alternate_improved(a2, max_iters=30, time_limit=lp_time2)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: diverse init -> LP alternating with line search ->
    softmax GD -> lag-targeted perturbation -> block refinement.
    Sweeps n values and tries multiple initializations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 8.0

    best_seq = None
    best_score = np.inf

    # Try parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Optimize parent first (it's our best starting point)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            print("Optimizing parent...")
            p_opt, p_obj = full_pipeline(parent.copy(), 120.0)
            if p_obj < best_score:
                best_score = p_obj
                best_seq = p_opt.copy()
                print(f"  Parent optimized: obj={p_obj:.6f}")

            # Additional refinement on parent
            if time.monotonic() < deadline - 60:
                a_ref, obj_ref = softmax_gd(p_opt, time_limit=30.0, T_init=0.002)
                if obj_ref < best_score:
                    best_score = obj_ref
                    best_seq = a_ref.copy()
                    print(f"  Parent after GD: obj={obj_ref:.6f}")

                a_ref2, obj_ref2 = block_refine(best_seq, rng, time_limit=20.0)
                if obj_ref2 < best_score:
                    best_score = obj_ref2
                    best_seq = a_ref2.copy()
                    print(f"  Parent after block: obj={obj_ref2:.6f}")
    except Exception as e:
        print(f"Parent optimization error: {e}")

    # Main search over different n values
    n_values = [100, 150, 200, 250, 350, 400, 500]
    # Remove n=300 since we already optimized parent
    n_values = [n for n in n_values if abs(n - 300) > 20]

    for n in n_values:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        num_remaining = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining)
        time_per_n = min(time_per_n, 120.0)

        num_restarts = max(2, min(4, int(time_per_n / 25)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts, {time_per_restart:.1f}s each")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 50:
                break
            remaining_r = deadline - time.monotonic() - 50
            if remaining_r < 10:
                break

            mode = (r * 3 + n) % 7
            a_init = diverse_init(n, rng, mode=mode)
            t_budget = min(time_per_restart, remaining_r * 0.9)
            a_opt, obj_opt = full_pipeline(a_init, t_budget)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r} (mode={mode}): obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r} (mode={mode}): obj={obj_opt:.6f}")

    # Final refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s remaining")

        # Softmax GD with small T
        gd_time = min(remaining * 0.3, 30.0)
        a_ref, obj_ref = softmax_gd(best_seq, time_limit=gd_time, T_init=0.001)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After final GD: obj={best_score:.6f}")

        # Block refinement
        if time.monotonic() < deadline - 5:
            bp_time = min(deadline - time.monotonic() - 3.0, 25.0)
            a_bp, obj_bp = block_refine(best_seq, rng, time_limit=bp_time)
            if obj_bp < best_score:
                best_score = obj_bp
                best_seq = a_bp.copy()
                print(f"  After block refine: obj={best_score:.6f}")

        # Lag-targeted perturbation
        if time.monotonic() < deadline - 5:
            lt_time = min(deadline - time.monotonic() - 3.0, 15.0)
            a_lt, obj_lt = lag_targeted_perturb(best_seq, rng, time_limit=lt_time)
            if obj_lt < best_score:
                best_score = obj_lt
                best_seq = a_lt.copy()
                print(f"  After lag perturb: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
