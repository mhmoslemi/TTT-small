import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def safe_compute_obj(a):
    """Safe version that clips to valid range."""
    a = np.clip(a, 0.0, 1000.0)
    return compute_obj(a)


def soft_max_value_and_grad(a, T):
    """Compute soft-max approximation of max(conv) and its gradient wrt a."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    sm = max_c + T * np.log(sum_exp)

    # grad_a[i] = sum_j p[i+j] * a[j]
    # This is the cross-correlation of p and a, taking the first n elements
    grad_conv = np.correlate(p, a, mode='full')[:n]

    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return 1e10, np.zeros(n)

    # Full gradient: d(2*n*sm/S^2)/d(a[i]) = 2*n*(grad_conv[i]*S - 2*sm)/S^3
    S = sum_a
    grad = 2.0 * n * (grad_conv * S - 2.0 * sm) / (S ** 3)
    obj_val = 2.0 * n * sm / (S * S)
    return obj_val, grad


def soft_max_objective(x, T):
    """Objective function for scipy minimize (x is the parameter vector)."""
    a = np.maximum(0.0, x)
    S = float(np.sum(a))
    if S < 0.01:
        return 1e10
    a = a / S  # normalize to sum=1 for numerical stability
    val, _ = soft_max_value_and_grad(a, T)
    return val


def soft_max_gradient(x, T):
    """Gradient for scipy minimize."""
    a = np.maximum(0.0, x)
    S = float(np.sum(a))
    if S < 0.01:
        return np.zeros_like(x)
    a = a / S
    _, grad_a = soft_max_value_and_grad(a, T)
    # Chain rule for normalization: a = x / S
    # d(a[i])/d(x[j]) = delta_ij/S - x[j]*x[i]/S^2 (after positive clipping)
    # For simplicity, approximate gradient as grad_a / S (ignoring the -x*x/S^2 term
    # which is a correction)
    return grad_a / S

    """Uniform in the middle, tapered at edges."""
    a = np.ones(n)
    edge_len = int(n * taper / 2)
    if edge_len > 0:
        for i in range(edge_len):
            frac = (i + 1) / (edge_len + 1)
            a[i] = frac
            a[n - 1 - i] = frac
    a = np.maximum(0.0, a)
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n


def init_random_smooth(n, rng, smoothness=3):
    """Generate a smooth random sequence."""
    a = rng.random(n)
    # Apply moving average for smoothness
    kernel = np.ones(smoothness) / smoothness
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(0.0, a)
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n


def init_parent_based(n, rng, parent=None):
    """Initialize based on parent sequence, possibly modified."""
    if parent is not None and len(parent) == n:
        a = parent.copy()
        # Add small perturbation
        a *= np.exp(rng.normal(0, 0.1, n))
        a = np.maximum(0.0, a)
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n
    elif parent is not None:
        # Resample parent to length n
        from scipy.interpolate import interp1d
        old_n = len(parent)
        x_old = np.linspace(0, 1, old_n)
        x_new = np.linspace(0, 1, n)
        f = interp1d(x_old, parent, kind='cubic', fill_value='extrapolate')
        a = np.maximum(0, f(x_new))
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n
    else:
        return init_random_smooth(n, rng)


def lbfgs_optimize(a_init, T_init, max_time, T_min=0.0001):
    """Run L-BFGS-B optimization with adaptive temperature."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = T_init

    # Adaptive L-BFGS-B with decreasing temperature
    while time.monotonic() - start < max_time:
        result = minimize(
            fun=soft_max_objective,
            x0=a.copy(),
            jac=soft_max_gradient,
            args=(T,),
            method='L-BFGS-B',
            bounds=[(0.0, None)] * n,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': 50000}
        )

        if result.success or result.fun < best_obj:
            a_new = np.clip(result.x, 0.0, None)
            S_new = np.sum(a_new)
            if S_new > 0.01:
                a_new = a_new / S_new
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                    a = a_new

        # Check time and reduce temperature
        if time.monotonic() - start >= max_time:
            break

        # Reduce temperature
        T = max(T_min, T * 0.5)
        if T == T_min:
            # One more final pass
            result = minimize(
                fun=soft_max_objective,
                x0=a.copy(),
                jac=soft_max_gradient,
                args=(T,),
                method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': 100000}
            )
            if result.fun < best_obj:
                a_new = np.clip(result.x, 0.0, None)
                S_new = np.sum(a_new)
                if S_new > 0.01:
                    a_new = a_new / S_new
                    obj = compute_obj(a_new)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a_new.copy()
            break

    return best_a, best_obj


def local_refine(a, max_time, rng):
    """Local refinement using random coordinate and block perturbations."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    max_stall = 500

    it = 0
    while time.monotonic() - start < max_time and stall < max_stall:
        it += 1
        mode = rng.integers(0, 3)

        if mode == 0:
            # Single coordinate
            i = int(rng.integers(0, n))
            candidate = best_a.copy()
            factor = np.exp(rng.normal(0, 0.1))
            candidate[i] = max(0.0, candidate[i] * factor)
        elif mode == 1:
            # Block
            width = max(1, int(n * rng.uniform(0.02, 0.15)))
            pos = int(rng.integers(0, n - width + 1))
            candidate = best_a.copy()
            sig = 0.05 + 0.1 * max(0.0, 1.0 - it / 10000)
            candidate[pos:pos + width] *= np.exp(rng.normal(0, sig, width))
        else:
            # Two-block (swap-like)
            w1 = max(1, int(n * rng.uniform(0.05, 0.2)))
            w2 = max(1, int(n * rng.uniform(0.05, 0.2)))
            p1 = int(rng.integers(0, n - w1 + 1))
            p2 = int(rng.integers(0, n - w2 + 1))
            if abs(p1 - p2) < max(w1, w2):
                continue
            candidate = best_a.copy()
            s1 = candidate[p1:p1+w1].sum()
            s2 = candidate[p2:p2+w2].sum()
            total = s1 + s2
            if total < 1e-10:
                continue
            ratio = rng.uniform(0.3, 0.7)
            candidate[p1:p1+w1] *= (ratio * total / max(s1, 1e-10))
            candidate[p2:p2+w2] *= ((1-ratio) * total / max(s2, 1e-10))
            candidate = np.maximum(0.0, candidate)

        candidate = np.maximum(0.0, candidate)
        S_c = np.sum(candidate)
        if S_c < 0.01:
            stall += 1
            continue
        obj = compute_obj(candidate)
        if obj < best_obj - 1e-12:
            best_a = candidate
            best_obj = obj
            stall = 0
        else:
            stall += 1

    return best_a, best_obj


def lp_alternate(a, max_time, rng):
    """LP-based alternating method from the state of the art."""
    from scipy.optimize import linprog
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    iteration = 0
    while time.monotonic() - start < max_time and stall < 30:
        iteration += 1
        conv = np.convolve(a, a)
        M = float(np.max(conv))

        # Build constraint matrix A where A @ b = a * b (convolution)
        # A has shape (2n-1, n)
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A[k, j] = a[k - j]

        # Maximize sum(b) subject to A @ b <= M, b >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, 10.0)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 10})
        except Exception:
            break

        if res.status != 0:
            break

        b_opt = res.x
        b_sum = float(np.sum(b_opt))
        if b_sum < 1.0 + 1e-6:
            stall += 1
            continue
        stall = 0

        # Normalize and blend
        b_opt = b_opt / b_sum
        t = max(0.1, 0.7 * (1.0 - iteration / 100.0))

        for tt in [t, t * 0.5, t * 0.25]:
            a_new = (1.0 - tt) * a + tt * b_opt
            a_new = np.maximum(0.0, a_new)
            S_new = np.sum(a_new)
            if S_new < 0.01:
                continue
            a_new = a_new / S_new
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                break
        else:
            a = (1.0 - t) * a + t * b_opt
            a = np.maximum(0.0, a)
            S_new = np.sum(a)
            if S_new > 0.01:
                a = a / S_new

    return best_a, best_obj


def de_global_search(n, max_time, rng, parent=None):
    """Differential evolution for global search."""
    bounds = [(0.0, 1.0)] * n

    def de_objective(x):
        a = np.array(x)
        S = np.sum(a)
        if S < 0.01:
            return 1e6
        a = a / S
        return compute_obj(a)

    # Population size adapted to time budget
    pop_size = min(30, max(10, n // 10))
    start = time.monotonic()

    try:
        result = differential_evolution(
            de_objective, bounds,
            population_size=pop_size,
            maxiter=50,
            tol=1e-6,
            seed=rng.integers(0, 2**31),
            mutation=(0.3, 0.9),
            recombination=0.8,
            polish=True,
            init='sobol' if n <= 50 else 'latinhypercube',
            updating='immediate',
            callback=None
        )
        if result.fun < 1e5:
            a = result.x
            S = np.sum(a)
            if S > 0.01:
                a = a / S
                return a, compute_obj(a)
    except Exception as e:
        print(f"  DE error: {e}")

    return None, None


def full_optimize(a_init, time_budget, rng):
    """Full optimization pipeline for a single initialization."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(0.0, a)
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: L-BFGS-B with soft-max (main workhorse)
    T_init = 0.1 * float(np.max(np.convolve(a, a)))
    lbfgs_time = min(time_budget * 0.5, 60.0)
    if lbfgs_time > 5:
        a1, obj1 = lbfgs_optimize(a, T_init, lbfgs_time)
        if obj1 < best_obj:
            best_a, best_obj = a1, obj1
            print(f"    After L-BFGS: {best_obj:.6f}")

    # Phase 2: LP alternating
    remaining = time_budget - (time.monotonic() - start)
    lp_time = min(remaining * 0.3, 30.0)
    if lp_time > 5 and n <= 200:
        a2, obj2 = lp_alternate(best_a, lp_time, rng)
        if obj2 < best_obj:
            best_a, best_obj = a2, obj2
            print(f"    After LP: {best_obj:.6f}")

    # Phase 3: Another L-BFGS pass with lower temperature
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        T2 = 0.01 * float(np.max(np.convolve(best_a, best_a)))
        a3, obj3 = lbfgs_optimize(best_a, T2, min(remaining * 0.4, 30.0))
        if obj3 < best_obj:
            best_a, best_obj = a3, obj3
            print(f"    After L-BFGS2: {best_obj:.6f}")

    # Phase 4: Local refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a4, obj4 = local_refine(best_a, remaining * 0.8, rng)
        if obj4 < best_obj:
            best_a, best_obj = a4, obj4
            print(f"    After local: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-strategy optimization: DE global search + L-BFGS-B soft-max +
    LP alternating + local refinement. Tries multiple initializations
    and sequence lengths, focusing on n=300.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 8.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.sum() > 0.01:
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        parent = None

    # If parent exists, optimize it with a quick pipeline
    if parent is not None and parent.size <= 400:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\nOptimizing parent (n={parent.size}), budget={remaining:.0f}s")
            t_parent = min(remaining * 0.15, 120.0)
            a_opt, obj_opt = full_optimize(parent.copy(), t_parent, rng)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Parent optimized: {best_score:.6f}")

    # Define initialization strategies for given n
    def get_inits(n, rng):
        inits = []
        inits.append(init_uniform(n))
        inits.append(init_raised_cosine(n, beta=0.3))
        inits.append(init_anti_palindrome(n, rng))
        inits.append(init_tapered(n, taper=0.4))
        inits.append(init_random_smooth(n, rng, smoothness=5))
        inits.append(init_bump(n, 0.3, 0.3, rng))
        inits.append(init_windowed_sine(n, 3, 0.0, floor=0.1))
        inits.append(init_windowed_sine(n, 7, np.pi/3, floor=0.2))
        if parent is not None:
            inits.append(init_parent_based(n, rng, parent))
        return inits

    # Main search over n values
    n_values = [300]
    # Add a few other sizes to explore
    if parent is not None:
        n_parent = parent.size
        if n_parent != 300:
            n_values.insert(0, n_parent)
    n_values.extend([200, 150, 100])

    # Budget allocation
    total_search_time = deadline - time.monotonic()
    print(f"\nTotal search time available: {total_search_time:.0f}s")

    for n in n_values:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Time allocation per n
        num_remaining = len([x for x in n_values if x >= n])
        if n == n_values[0]:
            # Give more time to primary n
            time_for_n = min(remaining * 0.5, 400.0)
        else:
            time_for_n = min(remaining * 0.25, 100.0)

        time_for_n = min(time_for_n, remaining - 20)
        if time_for_n < 15:
            continue

        print(f"\n=== N={n}, time budget={time_for_n:.0f}s ===")

        # How many restarts
        num_restarts = max(3, min(6, int(time_for_n / 30)))
        time_per_restart = time_for_n / num_restarts

        inits = get_inits(n, rng)

        for r in range(num_restarts):
            if time.monotonic() > deadline - 20:
                break

            # Pick initialization (shuffle for diversity)
            init_idx = r % len(inits)
            a_init = inits[init_idx].copy()

            t_remaining = deadline - time.monotonic() - 20
            t_alloc = min(time_per_restart, t_remaining * 0.3)
            if t_alloc < 10:
                break

            try:
                a_opt, obj_opt = full_optimize(a_init, t_alloc, rng)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  n={n} r={r} init={init_idx}: {obj_opt:.6f} ** NEW BEST **")
                else:
                    print(f"  n={n} r={r} init={init_idx}: {obj_opt:.6f}")
            except Exception as e:
                print(f"  n={n} r={r} error: {e}")

    # Final refinement on the best
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 10
        print(f"\n=== Final refinement: {remaining:.0f}s remaining ===")

        # Multiple refinement passes
        for pass_num in range(3):
            if time.monotonic() > deadline - 10:
                break
            t_pass = min(remaining / (3 - pass_num), 60.0)
            if t_pass < 5:
                break
            try:
                a_ref, obj_ref = full_optimize(best_seq.copy(), t_pass, rng)
                if obj_ref < best_score:
                    best_score = obj_ref
                    best_seq = a_ref.copy()
                    print(f"  Refinement pass {pass_num+1}: {best_score:.6f}")
                else:
                    print(f"  Refinement pass {pass_num+1}: no improvement ({obj_ref:.6f})")
                    break
            except Exception:
                break

        # Final local polish
        if time.monotonic() < deadline - 5:
            t_local = deadline - time.monotonic() - 4
            if t_local > 2:
                a_pol, obj_pol = local_refine(best_seq, t_local, rng)
                if obj_pol < best_score:
                    best_score = obj_pol
                    best_seq = a_pol.copy()
                    print(f"  After polish: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        n_fb = 300
        best_seq = init_raised_cosine(n_fb, beta=0.2)
        best_score = compute_obj(best_seq)
        print(f"Fallback used: n={n_fb}, score={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
