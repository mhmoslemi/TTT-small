from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    """
    Given fixed sequence a, solve LP:
    minimize T
    subject to: (a*b)[k] <= T for k=0,...,2n-2
                b_i >= 0
                sum(b_i) = 1
    Returns (b, T).
    """
    n = len(a)
    # Build constraint matrix
    # For each k in 0..2n-2, the constraint is: sum_i A[k,i] * b[i] - T <= 0
    # where A[k,i] = a[i] if k-i is in [0, n-1], else 0

    # So (a*b)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[k-j]*b[j]
    
    # Variables: [b_0, ..., b_{n-1}, T]
    # Objective: minimize T -> c = [0, ..., 0, 1]
    
    # Inequality constraints: (a*b)[k] - T <= 0 for each k

"""
Optimized step function coefficient search using alternating LP projection,
subgradient descent, and coordinate descent refinement.
The key innovation is using an alternating LP method where each step solves
a linear program to find the "best response" sequence, providing a natural
descent direction for the non-smooth max-autocorrelation objective.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a)) / sum(a)^2"""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s

def solve_cross_lp(fixed_seq):
    """
    Given fixed_seq (non-negative, sum=1), solve:
    minimize T
    subject to: (fixed_seq * x)[k] <= T for all k
                x >= 0, sum(x) = 1
    Returns (x, T) or (None, None).
    """
    n = len(fixed_seq)
    # Build constraint matrix efficiently
    # Row k of A_ub: for j in valid range, A_ub[k,j] = fixed_seq[k-j], A_ub[k,n] = -1
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        js = np.arange(j_start, j_end+1)
        A_ub[k, js] = fixed_seq[k - js]
        A_ub[k, n] = -1.0
    
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n+1)
    c[n] = 1.0
    
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n], float(res.x[n])
    return None, None

def alternating_lp(a_init, max_iters=200, time_limit=60.0):
    """
    Alternating LP: alternate between finding best response to a and to b.
    Uses blending to ensure convergence.
    """
    start_time = time.monotonic()
    a = normalize(a_init.copy())
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_limit:
            break
        
        # Step 1: Given a, find b that minimizes max(a*b)
        b, T_b = solve_cross_lp(a)
        if b is None:
            break
        
        # Step 2: Given b, find a_new that minimizes max(b*a_new) = max(a_new*b)
        a_new, T_a = solve_cross_lp(b)
        if a_new is None:
            break
        
        # Evaluate candidates
        candidates = []
        
        # Candidate 1: a_new directly
        candidates.append(a_new)
        
        # Candidate 2: b directly
        candidates.append(b)
        
        # Candidate 3: Blend of a and a_new
        for t in [0.3, 0.5, 0.7, 1.0]:
            cand = normalize((1-t) * a + t * a_new)
            candidates.append(cand)
        
        # Candidate 4: Blend of a and b
        for t in [0.3, 0.5, 0.7]:
            cand = normalize((1-t) * a + t * b)
            candidates.append(cand)
        
        # Pick the best candidate
        best_cand = a
        best_cand_obj = compute_obj(a)
        for cand in candidates:
            obj = compute_obj(cand)
            if obj < best_cand_obj:
                best_cand_obj = obj
                best_cand = cand
        
        if best_cand_obj < best_obj:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
        
        a = best_cand
    
    return best_a, best_obj

def subgradient_descent(a_init, time_limit=30.0, eta_init=0.1):
    """
    Subgradient descent with mirror step (multiplicative weights) on max(a*a).
    """
    start_time = time.monotonic()
    a = normalize(a_init.copy())
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    eta = eta_init
    
    for it in range(100000):
        if time.monotonic() - start_time > time_limit:
            break
        
        # Compute (a*a)
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]
        
        # Compute subgradient at a
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n-1, k_star)
        for j in range(j_start, j_end+1):
            # a[j] appears in conv[k_star] as a[j]*a[k_star-j]
            grad[j] += a[k_star - j]
            # a[k_star-j] also gets contribution from a[j]*a[k_star-j]
            if k_star - j != j:
                grad[k_star - j] += a[j]
        
        # Mirror step: a_new[j] = a[j] * exp(-eta * grad[j])
        log_update = -eta * grad
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = max(eta * 0.999, 0.001)
        else:
            eta *= 0.95
            if eta < 1e-6:
                break
    
    return best_a, best_obj

def coordinate_descent(a_init, time_limit=30.0):
    """
    Coordinate descent: optimize one coordinate at a time.
    For each coordinate j, find the optimal value of a[j] given all others fixed.
    """
    start_time = time.monotonic()
    a = normalize(a_init.copy())
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Precompute which convolution positions each coordinate affects
    # a[j] affects conv[k] for k in [j, j+n-1] (intersected with [0, 2n-2])
    
    for outer_iter in range(50):
        if time.monotonic() - start_time > time_limit:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() - start_time > time_limit:
                break
            
            # Current value
            a_j = a[j]
            
            # The convolution values affected by a[j]:
            # conv[k] for k where a[j] appears
            # a[j] appears in conv[k] as:
            #   - a[j] * a[k-j] (if 0 <= k-j < n, i.e., j <= k <= j+n-1)
            #   - a[k-j] * a[j] (same condition, but this is the same set of k)
            # So a[j] affects conv[k] for k in [j, min(j+n-1, 2n-2)]
            
            k_min = j
            k_max = min(j + n - 1, 2*n - 2)
            
            # For each affected k, conv[k] = (other terms) + a[j]*a[k-j] + a[k-j]*a[j]
            # = (other terms) + 2*a[j]*a[k-j] (if k != 2j)
            # = (other terms) + a[j]^2 (if k == 2j)
            
            # Actually let me just compute the current conv and figure out the linear part
            conv = np.convolve(a, a)
            
            # For the affected positions, conv[k] = c0[k] + a[j] * coeff[k]
            # where c0[k] is the part not involving a[j], and coeff[k] depends on a
            # This is getting complex. Let me just do a 1D search.
            
            # Try a few values of a[j] around the current value
            # The optimal a[j] should be at a point where the max of affected conv[k] is minimized
            # This is a 1D minimax problem.
            
            # Simple approach: try a range of values
            if a_j < 1e-10:
                # Try setting it to small positive values
                candidates = [0.0, 1e-4, 1e-3, a_j * 2 if a_j > 0 else 1e-3,

                             a_j * 0.1, a_j * 0.3, a_j * 0.7,

            obj_now = compute_obj(a)

                improved = True

def init_uniform(n):
    return np.ones(n) / n

def init_triangular(n):
    a = np.minimum(np.arange(1, n+1), np.arange(n, 0, -1)).astype(float)
    return a / np.sum(a)

def init_gaussian(n, sigma_frac=0.3):
    center = (n-1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * (np.arange(n) - center)**2 / sigma**2)
    return a / np.sum(a)

def init_two_bump(n, pos_frac=0.25):
    a = np.zeros(n)
    pos = int(pos_frac * n)
    width = max(1, n // 10)
    a[max(0, pos-width):pos+width+1] = 1.0
    a[n-1-pos-max(0, pos-width):n-1-pos+width+1] = 1.0
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_offset_bump(n, offset=0.1):

    center = (n-1) / 2.0 + offset * n
    sigma = 0.2 * n
    a = np.exp(-0.5 * (np.arange(n) - center)**2 / sigma**2)
    a = np.clip(a, 0, None)

def init_periodic(n, period=5):
    """Periodic pattern."""
    pattern = np.array([1.0, 0.5, 0.0, 0.5, 1.0])[:period]
    a = np.tile(pattern, n // period + 1)[:n]

def full_optimize(a_init, time_budget=60.0):

    Phase 1: Alternating LP
    Phase 2: Subgradient descent
    Phase 3: Coordinate descent
    """
    start = time.monotonic()

    best_a = a.copy()

    n = len(a)

The objective is `2n * max_k c[k] / S^2` where `c = a * a` (full convolution) and `S = sum(a)`.

By scaling invariance, we can set `S = 1` and minimize `2n * max_k c[k]`.

The minimum possible value of `max_k c[k]` for non-negative `a` with sum 1:
- `c[n-1] = sum_i a[i] * a[n-1-i]`
- If `a` is symmetric: `c[n-1] = sum(a_i^2) >= 1/n`
- If `a` is not symmetric: `c[n-1]` can be less than `1/n`

But other `c[k]` values also matter. For example, `c[0] = a[0]^2`, `c[1] = 2*a[0]*a[1]`, etc.
