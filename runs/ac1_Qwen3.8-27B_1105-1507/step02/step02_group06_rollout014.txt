    fb = np.fft.rfft(b, size)

def water_filling(x, target_sum=1.0):
    """Euclidean projection of x onto {a >= 0, sum(a) = target_sum}."""
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.maximum(x - mid, 0.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    result = np.maximum(x - t, 0.0)
    # Ensure exact sum (floating point correction)
    diff = target_sum - float(np.sum(result))
    if abs(diff) > 1e-15:
        # Distribute the difference among positive entries
        pos_mask = result > 0
        n_pos = int(np.sum(pos_mask))
        if n_pos > 0:
            result[pos_mask] += diff / n_pos
            result = np.maximum(result, 0.0)
    return result

                         momentum=0.9, verbose_prefix=""):

    a = np.array(a_init, dtype=np.float64).copy()

    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

        fft_size *= 2

    it = 0

        fa = np.fft.rfft(a, fft_size)
        conv = np.fft.irfft(fa * fa, fft_size)[:2*n-1]

            #   the rest are 0)

            #   g_k[m] = a[k-m] if 0 <= k-m <= n-1, else 0

                # which is a[k-j_max : k-j_min+1][::-1]

                # = a[k-j_max : k-j_min+1][::-1]

                # So the values are a[indices] = a[k-j_min : k-j_max-1 : -1] ... this is getting confusing

        # g = 2 * cross_correlation(w_lags, a) where w_lags is sparse
        # Since w_lags has only top_k nonzero entries, I can just loop:
        g = np.zeros(n)
        for ki, wi in zip(top_indices, weights):
            k = int(ki)
            # g[m] += 2 * wi * a[k-m] for valid m
            # Valid: 0 <= m < n and 0 <= k-m < n => max(0, k-n+1) <= m <= min(k, n-1)
            m_start = max(0, k - n + 1)
            m_end = min(k, n - 1)
            if m_end >= m_start:
                # a[k-m] for m = m_start, ..., m_end
                # = a[k-m_start], a[k-m_start-1], ..., a[k-m_end]
                a_vals = a[k-m_end:k-m_start+1][::-1]
                g[m_start:m_end+1] += 2.0 * wi * a_vals
        
        # Momentum update
        v = momentum * v - lr * g
        a_new = water_filling(a + v, 1.0)
        
        a = a_new
        it += 1

            score = compute_score(a)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if it % 500 == 0:
                print(f"  {verbose_prefix}iter {it}: score={best_score:.6f} lr={lr:.6f}")
            
            if no_improve > 50:
                lr *= 0.85
                v *= 0.5
                no_improve = 0
            if lr < 1e-8:
                break
    
    return best_a, best_score

Let me also think about the LP approach. For n=1024, the LP has 1024 variables and 2047 constraints. With a sparse structure (each constraint involves at most n variables), this might be solvable in a few seconds with a good solver.

The key insight for the LP is: we don't need ALL constraints. We only need the constraints that are "active" (tight) at the optimum. So we can use a cutting plane approach:
1. Start with a small set of constraints (e.g., top 20 lags)
2. Solve the LP
3. Check if the solution violates any other constraints
4. Add violated constraints
5. Repeat

This is much faster than solving the full LP.

For the LP formulation:
max sum(g) subject to:
- conv(a, g)_k <= M for k in active_set
- g >= 0
- (optionally: sum(g) = 1, to normalize the direction)

where M = max(conv(a,a)) is the current peak.

Actually, we want to find a direction that reduces the peak. The LP gives us a point g that is "as large as possible" while keeping conv(a, g) bounded. Then we move from a towards g.

def compute_score(f):
    """Compute objective value directly."""
    n = len(f)
