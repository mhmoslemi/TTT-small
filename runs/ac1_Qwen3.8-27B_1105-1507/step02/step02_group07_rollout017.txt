Multi-phase optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses LP direction finding with principled step sizes, Nesterov-accelerated
soft-max gradient descent, asymmetric initializations, coordinate descent
polish, and diverse restarts.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix(f):
    """Build dense matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = f[k - j]
    return A


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_sparse=None):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""

def lp_iterate(f, deadline, step_size_init=0.1, max_iters=300, verbose_prefix=""):
    """Run LP-based iterations with principled step size."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    it = 0
    no_improve = 0
    step_size = step_size_init

    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        A_sp = build_conv_matrix_sparse(f)
        g = solve_lp_direction(f, M, A_sp)
        it += 1

        if g is None:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue

        g = g / S  # normalize g to have same sum as f

        # Compute principled step size from KKT
        t_kkt = compute_step_size_from_kkt(f, g, M)
        t_kkt = min(t_kkt, 1.0)
        
        # Try several step sizes
        candidates = [step_size, t_kkt * 0.5, t_kkt * 0.8, step_size * 1.5, step_size * 0.5]
        candidates = [max(0.001, min(1.0, t)) for t in candidates]
        candidates = list(set([round(t, 4) for t in candidates]))
        
        best_t = step_size
        best_s = best_score
        
        for trial_t in sorted(candidates, reverse=True):
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t

        if best_s < best_score - 1e-10:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, 1000.0)
            S = float(np.sum(f))
            f = f / S
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            step_size = best_t
            if it % 20 == 1:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f} (t={best_t:.4f})")
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.003 or no_improve > 50:
                break

    return best_f, best_score

    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_sparse=None):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    if A_sparse is None:
        A_sparse = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_step_size_kkt(f, g, M):
    """Compute max step from KKT: t <= -2*(f*g)[k]/(g*g)[k] at active k."""
    conv_fg = np.convolve(f, g)
    conv_gg = np.convolve(g, g)
    conv_ff = np.convolve(f, f)
    active = conv_ff >= M * 0.999
    t_max = 1.0
    active_idx = np.where(active)[0]
    for k in active_idx:
        den = conv_gg[k]
        if den > 1e-15:
            candidate = -2.0 * conv_fg[k] / den
            if candidate < t_max:
                t_max = candidate
    return max(t_max, 0.0)


def lp_iterate(f, deadline, step_size_init=0.1, max_iters=200, verbose_prefix=""):
    """LP-based iterations with KKT step sizes and adaptive line search."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    it = 0
    no_improve = 0
    step_size = step_size_init

    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 1e-15:
            break

        A_sp = build_conv_matrix_sparse(f)
        g = solve_lp_direction(f, M, A_sp)
        it += 1

        if g is None:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue

        g = g / S
        t_kkt = compute_step_size_kkt(f, g, M)
        t_kkt = min(t_kkt, 1.0)

        candidates = sorted(set([
            round(step_size, 5),
            round(min(1.0, t_kkt * 0.6), 5),
            round(min(1.0, t_kkt * 0.9), 5),
            round(min(1.0, step_size * 1.5), 5),
            round(max(0.001, step_size * 0.5), 5),
        ]), reverse=True)

        best_t = step_size
        best_s = best_score

        for trial_t in candidates:
            if trial_t <= 0:
                continue
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t

        if best_s < best_score - 1e-12:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, 1000.0)
            S = float(np.sum(f))
            f = f / S
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            step_size = max(0.005, best_t * 1.1)
            if it % 15 == 1:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f} (t={best_t:.4f})")
        else:
            no_improve += 1
            step_size *= 0.65
            if step_size < 0.002 or no_improve > 40:
                break

    return best_f, best_score


def compute_softmax_gradient(f, beta):
    """Gradient of soft-max of conv(f,f) using FFT-based convolution."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    shifted = beta * (C - max_C)
    shifted -= shifted.max()
    exp_C = np.exp(shifted)
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), max_C + np.log(total) / beta
    p = exp_C / total

    p_rev = p[::-1]
    conv_fp = np.convolve(f, p_rev, mode='full')
    grad = 2.0 * conv_fp[n - 1:2 * n - 1][::-1]
    soft_max_val = max_C + np.log(total) / beta + shifted.max() / beta
    soft_max_val = max_C + (shifted.max() + np.log(total)) / beta
    return grad, soft_max_val


def nesterov_softmax_optimize(f, deadline, beta_start=10.0, beta_end=500.0,
                               lr_start=0.02, max_iters=5000, verbose_prefix=""):
    """Nesterov-accelerated soft-max gradient descent with Adam."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    no_improve = 0
    it = 0

    # Nesterov momentum
    f_nested = f.copy()
    mu = 0.9

    while time.monotonic() < deadline and it < max_iters:
        it_frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** it_frac

        grad, _ = compute_softmax_gradient(f_nested, beta)

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)

        new_f = f - lr * update
        new_f = np.clip(new_f, 0.0, 1000.0)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            new_f = f
            ns = S
        new_f = new_f / ns

        sc = evaluate_sequence(new_f.tolist())
        if np.isfinite(sc) and sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            f_nested = f + mu * (f - f + 0)  # reset nested
            f_nested = f.copy()
            if it % 500 == 0:
                print(f"    {verbose_prefix}NM iter {it}: score={best_score:.6f} (beta={beta:.1f}, lr={lr:.5f})")
        else:
            no_improve += 1
            if no_improve > 150:
                lr *= 0.6
                no_improve = 0
                if lr < 1e-7:
                    break

    return best_f, best_score


def coordinate_descent_polish(f, deadline, max_sweeps=50, verbose_prefix=""):
    """Coordinate descent: optimize each coordinate by 1D line search."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    n = len(f)
    best_score = evaluate_sequence(f.tolist())
    best_f = f.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Try several values for f[j]
            old_val = f[j]
            if old_val < 1e-12:
                candidates = [old_val * 2 + 1e-6, old_val + 0.001, S / n * 0.5, S / n]
            else:
                candidates = [old_val * 0.5, old_val * 1.5, old_val * 0.8,
                             old_val * 1.2, 0.0, old_val + S/n*0.1, old_val - S/n*0.1]
            candidates = [max(0.0, c) for c in candidates]

            best_c = old_val
            best_s = best_score
            for c in candidates:
                f[j] = c
                ns = float(np.sum(f))
                if ns < 1e-10:
                    continue
                f_norm = f / ns
                sc = evaluate_sequence(f_norm.tolist())
                if np.isfinite(sc) and sc < best_s:
                    best_s = sc
                    best_c = c
            f[j] = best_c
            if best_s < best_score - 1e-10:
                f = f / float(np.sum(f))
                best_score = best_s
                best_f = f.copy()
                improved = True

        if not improved:
            break

    return best_f, best_score


def tukey_window(n, r=0.3):
    """Tukey (tapered cosine) window."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def asymmetric_ramp(n, peak_frac=0.4, taper=0.3):
    """Asymmetric window: rises gradually, peaks off-center, falls faster."""
    x = np.linspace(0, 1, n)
    peak = peak_frac
    width = 0.3
    w = np.exp(-((x - peak) / width) ** 2)
    # Add a linear taper on the right
    right_start = int(n * (1 - taper))
    if right_start < n:
        taper_x = np.linspace(1, 0, n - right_start)
        w[right_start:] *= taper_x
    return w


def shifted_gaussian(n, center_frac=0.35, width_frac=0.4):
    """Gaussian with peak off-center."""
    x = np.linspace(0, 1, n)
    center = center_frac
    width = width_frac
    w = np.exp(-((x - center) / width) ** 2)
    return w


def random_smooth(n, seed, kernel_width=5):
    """Random smooth non-negative sequence."""
    rng = np.random.default_rng(seed)
    x = rng.standard_normal(n)
    kernel = np.exp(-0.5 * (np.arange(-kernel_width, kernel_width + 1) / 2.0) ** 2)
    kernel /= kernel.sum()
    smooth_x = np.convolve(x, kernel, mode='same')
    smooth_x = np.abs(smooth_x) + 0.01
    return smooth_x


def perturb(f, rng, strength=0.05):
    """Multiplicative noise perturbation."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.05, 3.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def full_optimize(f_init, deadline, verbose_prefix=""):
    """Multi-phase optimization: LP -> Nesterov gradient -> LP -> CD polish."""
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    f_init = f_init / S

    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())
    t0 = time.monotonic()
    total_budget = max(10.0, deadline - t0)

    # Phase 1: LP iterations (40% of budget)
    p1_deadline = t0 + total_budget * 0.35
    if time.monotonic() < p1_deadline - 5:
        f1, s1 = lp_iterate(f_init, p1_deadline, step_size_init=0.12,
                            max_iters=150, verbose_prefix=verbose_prefix + "P1")
        if s1 < best_score:
            best_f, best_score = f1, s1

    # Phase 2: Nesterov soft-max gradient (35% of budget)
    p2_deadline = t0 + total_budget * 0.70
    if time.monotonic() < p2_deadline - 5:
        f2, s2 = nesterov_softmax_optimize(best_f, p2_deadline,
                                            beta_start=20.0, beta_end=600.0,
                                            lr_start=0.015,
                                            max_iters=8000,
                                            verbose_prefix=verbose_prefix + "P2")
        if s2 < best_score:
            best_f, best_score = f2, s2

    # Phase 3: LP polish (15% of budget)
    p3_deadline = t0 + total_budget * 0.85
    if time.monotonic() < p3_deadline - 5:
        f3, s3 = lp_iterate(best_f, p3_deadline, step_size_init=0.05,
                            max_iters=60, verbose_prefix=verbose_prefix + "P3")
        if s3 < best_score:
            best_f, best_score = f3, s3

    # Phase 4: Coordinate descent polish (10% of budget)
    if time.monotonic() < deadline - 3:
        f4, s4 = coordinate_descent_polish(best_f, deadline - 3,
                                            max_sweeps=30,
                                            verbose_prefix=verbose_prefix + "P4")
        if s4 < best_score:
            best_f, best_score = f4, s2 if False else f4, s4

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Load parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Generate initializations for multiple lengths
    lengths = [512, 1024, 768, 1536]
    time_fracs = {512: 0.35, 1024: 0.25, 768: 0.15, 1536: 0.12}
    total_budget = deadline - t0

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break

        n_budget = total_budget * time_fracs.get(n, 0.1)
        n_deadline = time.monotonic() + n_budget
        print(f"\n=== n={n} (budget {n_budget:.0f}s) ===")

        inits = []

        # Asymmetric initializations (key insight: optimum is NOT symmetric)
        inits.append(('asym_ramp', asymmetric_ramp(n, 0.35, 0.35)))
        inits.append(('asym_ramp2', asymmetric_ramp(n, 0.55, 0.25)))
        inits.append(('shift_gauss_l', shifted_gaussian(n, 0.3, 0.35)))
        inits.append(('shift_gauss_r', shifted_gaussian(n, 0.7, 0.35)))

        # Standard windows
        inits.append(('tukey_03', tukey_window(n, 0.3)))
        inits.append(('tukey_05', tukey_window(n, 0.5)))

        # Random smooth
        inits.append(('rand_s1', random_smooth(n, seed=7 + n))))
        inits.append(('rand_s2', random_smooth(n, seed=13 + n * 3)))

        # Power-law shaped
        x = np.linspace(0.01, 1, n)
        inits.append(('power_law', x ** 0.3))
        inits.append(('power_law2', (1 - x + 0.01) ** 0.3))

        # Use parent if same length or interpolate
        if parent is not None:
            if len(parent) == n:
                inits.insert(0, ('parent', parent.copy()))
            else:
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n)
                parent_interp = np.interp(x_new, x_old, parent)
                parent_interp = np.clip(parent_interp, 0, None)
                inits.insert(0, ('parent_interp', parent_interp))

        # Evaluate all initializations
        scored_inits = []
        for label, f_init in inits:
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue
            sc = evaluate_sequence(f_init.tolist())
            if np.isfinite(sc):
                scored_inits.append((sc, label, f_init))
                if sc < best_score:
                    best_score = sc
                    best_sequence = f_init.copy()
                    print(f"  [n={n} {label}] init score={sc:.6f} (NEW BEST)")

        scored_inits.sort(key=lambda x: x[0])

        # Optimize top initializations
        n_top = min(4, len(scored_inits))
        per_init_time = n_budget / max(n_top + 1, 1) * 0.85

        for i in range(n_top):
            if time.monotonic() > n_deadline - 10:
                break
            sc, label, f_init = scored_inits[i]
            init_deadline = min(n_deadline - 5, time.monotonic() + per_init_time)
            if time.monotonic() >= init_deadline - 10:
                continue
            print(f"  [n={n} {label}] optimizing (init score={sc:.6f})...")
            try:
                bf, bs = full_optimize(f_init, init_deadline,
                                       verbose_prefix=f"[n={n} {label}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  [n={n} {label}] NEW BEST: {bs:.6f}")
                else:
                    print(f"  [n={n} {label}] best: {bs:.6f}")
            except Exception as e:
                print(f"  [n={n} {label}] error: {e}")

    # Perturbation restarts
    if best_sequence is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Perturbation restarts (best: {best_score:.6f}, n={len(best_sequence)}) ===")
        restart_budget = deadline - time.monotonic() - 20
        n_restarts = min(8, max(3, int(restart_budget / 40)))
        per_restart = restart_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            strength = 0.03 + 0.04 * r
            f_pert = perturb(best_sequence, rng, strength=strength)
            r_deadline = time.monotonic() + per_restart * 0.75

            try:
                bf, bs = full_optimize(f_pert, r_deadline,
                                       verbose_prefix=f"[rst{r}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strength:.3f}): NEW BEST {bs:.6f}")
                else:
                    print(f"  Restart {r} (str={strength:.3f}): {bs:.6f}")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # Final polish on the absolute best
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final polish (n={len(best_sequence)}, score={best_score:.6f}) ===")
        try:
            bf, bs = lp_iterate(best_sequence, time.monotonic() + 60,
                                step_size_init=0.03, max_iters=40,
                                verbose_prefix="[final_LP] ")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()

            remaining = deadline - time.monotonic() - 5
            if remaining > 20:
                bf, bs = nesterov_softmax_optimize(best_sequence, time.monotonic() + remaining * 0.5,
                                                    beta_start=200.0, beta_end=1000.0,
                                                    lr_start=0.005, max_iters=3000,
                                                    verbose_prefix="[final_NM] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()

            remaining = deadline - time.monotonic() - 3
            if remaining > 10:
                bf, bs = coordinate_descent_polish(best_sequence, time.monotonic() + remaining * 0.8,
                                                    max_sweeps=20, verbose_prefix="[final_CD] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()

            print(f"  Final score: {best_score:.6f}")
        except Exception as e:
            print(f"  Final polish error: {e}")

    # Fallback
    if best_sequence is None:
        best_sequence = tukey_window(512, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())
        print(f"  FALLBACK: score={best_score:.6f}")

    # Final validation
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
