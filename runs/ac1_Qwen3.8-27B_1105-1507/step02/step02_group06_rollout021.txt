But we can use a surrogate: replace (g*g)[k] ≤ M with (f+g)*(f+g)[k] ≤ M for some fixed f. Or use a first-order approximation.

Actually, a better approach: **Use the fact that we want to minimize the peak of the autocorrelation, and formulate a sequence of LPs that progressively tighten the bound.**

Algorithm:
1. Start with current bound M.
2. Solve LP: max sum(g) s.t. (f*g)[k] ≤ M, g ≥ 0.
3. Compute max(g*g). Call it M_g.
4. If M_g < M: great, mix and continue with new M = (1-t)^2 M + 2t(1-t)M + t^2 M_g < M.
5. If M_g ≥ M: the LP didn't help. Try:
   a. Reduce M in the LP: solve max sum(g) s.t. (f*g)[k] ≤ M*(1-eps), g ≥ 0.
   b. Or use a different approach (subgradient, etc.)

**Another key idea: Use the dual of the LP to identify "hot spots".**

The dual of the LP:
Primal: max 1^T g s.t. A g ≤ M*1, g ≥ 0
Dual: min M * 1^T y s.t. A^T y ≥ 1, y ≥ 0

where A is the matrix with A[k,j] = f[k-j] (the convolution matrix).

The dual variables y[k] tell us how "active" each constraint is. Lags with large y[k] are the ones most limiting. We can use this to:
- Focus subgradient updates on those lags
- Adjust the LP by adding extra weight to those constraints

"""
Optimization of step function heights to minimize 2*n*max(autocorr)/sum^2.
Uses improved LP-based iterative refinement with reduced-bound forcing,
dual variable exploitation, and spectral domain polishing.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix


def compute_score_fast(f):
    """Compute 2*n*max(conv(f,f))/sum(f)^2 without type checks."""
    n = len(f)
    conv = np.convolve(f, f)
    M = np.max(conv)
    S = np.sum(f)
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A @ g)[k] = (convolve(f,g))[k].
    A has shape (2n-1, n)."""
    n = len(f)
    # For each k in 0..2n-2, (f*g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[k-j] * g[j]
    # = sum_j A[k,j] * g[j] where A[k,j] = f[k-j] if valid, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M_bound, verbose=False):
    """Solve: max sum(g) s.t. (f*g)[k] <= M_bound for all k, g >= 0.
    Returns (g, dual_vars) or (None, None)."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M_bound)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            dual_vars = result.ineqlin.marginals if hasattr(result.ineqlin, 'marginals') else None
            return result.x, dual_vars
    except Exception:
        pass
    return None, None


def subgradient_step(f, top_k=5):
    """Compute subgradient of max_k(f*f)[k] w.r.t. f.
    Returns gradient vector (unnormalized)."""
    n = len(f)
    conv = np.convolve(f, f)
    top_indices = np.argsort(conv)[-top_k:]
    
    grad = np.zeros(n)
    for k in top_indices:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += f[k - j]
    
    return grad


def subgradient_optimize(f, deadline, lr=0.01, top_k=5, momentum_decay=0.9, verbose_prefix=""):
    """Subgradient descent with momentum."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score_fast(f)
    n = len(f)
    S = np.sum(f)
    mom = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = np.max(conv)
        if M < 1e-15:
            break
        
        grad = subgradient_step(f, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        mom = momentum_decay * mom - lr * grad
        new_f = np.clip(f + mom, 0.0, None)
        new_S = np.sum(new_f)
        if new_S < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 100:
                break
            it += 1
            continue
        new_f *= S / new_S
        
        new_score = compute_score_fast(new_f)
        it += 1
        
        if new_score < best_score:
            best_score = new_score
            best_f = new_f.copy()
            f = new_f
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.97
            mom *= 0.5
            if lr < 1e-8 or no_improve > 500:
                break
    
    return best_f, best_score

def spectral_refine(f, deadline):
    """Try to flatten the power spectrum in the frequency domain."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()

    n = len(f)
    S = np.sum(f)
    rng = np.random.default_rng(42)
    t = 0.01
    
    while time.monotonic() < deadline:
        # Compute power spectrum
        F = np.fft.rfft(f, n=2*n-1)
        power = np.abs(F)**2
        
        # Try to equalize the power spectrum (reduce peaks)
        mean_power = np.mean(power)
        # Perturb: reduce components with high power, increase low power
        target = mean_power + 0.1 * (power - mean_power)  # Compress
        
        # Convert back and project
        F_new = F * np.sqrt(np.maximum(target / np.maximum(power, 1e-15), 0.3))
        F_new = F * (1 - t) + F_new * t
        
        # Inverse FFT and take real part, truncate to length n
        f_new = np.fft.irfft(F_new, n=2*n-1)[:n]
        f_new = np.clip(f_new, 0, None)
        fS = np.sum(f_new)
        if fS < 1e-15:
            t *= 0.5
            continue
        f_new *= S / fS
        
        new_score = compute_score_fast(f_new)
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            t = min(0.05, t * 1.1)
        else:
            t *= 0.7
            if t < 1e-5:
                break
    
    return best_f, best_score

    # Sine (single period)
    inits.append(('sine', 0.5 + 0.5 * np.sin(2 * np.pi * idx / n)))

    inits.append(('two_hump', two_hump + 0.01))

    center = (n - 1) / 2.0
    gauss = np.exp(-0.5 * ((idx - center) / (n * 0.4))**2)
    inits.append(('gauss_wide', gauss + 0.01))
    
    # Gaussian (narrow)
    gauss_n = np.exp(-0.5 * ((idx - center) / (n * 0.15))**2)
    inits.append(('gauss_narrow', gauss_n + 0.01))
    
    if parent is not None:
        if len(parent) == n:
            inits.append(('parent', parent.copy()))
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            p_interp = np.interp(x_new, x_old, parent)
            inits.append(('parent_interp', p_interp))
    
    return inits

import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score_fast(f):
    """Compute 2*n*max(conv(f,f))/sum(f)^2 without type checks."""
    n = len(f)
    if n == 0:
        return float('inf')
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def build_conv_matrix_sparse(f, row_col_cache=None):
    """Build sparse matrix A where (A @ g)[k] = (convolve(f,g))[k].
    A has shape (2n-1, n) with A[k,j] = f[k-j] when valid."""
    n = len(f)
    if row_col_cache is not None and row_col_cache[0] == n:
        rows, cols = row_col_cache[1], row_col_cache[2]
        vals = np.empty(len(rows), dtype=np.float64)
        for idx in range(len(rows)):
            k = rows[idx]
            j = cols[idx]
            vals[idx] = f[k - j]
        mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        return mat, (n, rows, cols)
    else:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
        rows = np.array(rows, dtype=np.int32)
        cols = np.array(cols, dtype=np.int32)
        vals = np.empty(len(rows), dtype=np.float64)
        for idx in range(len(rows)):
            k = rows[idx]
            j = cols[idx]
            vals[idx] = f[k - j]
        mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        return mat, (n, rows, cols)


def solve_lp_direction(f, M_bound, cache=None):
    """Solve: max sum(g) s.t. (f*g)[k] <= M_bound for all k, g >= 0.
    Returns (g, updated_cache) or (None, cache)."""
    n = len(f)
    A, cache = build_conv_matrix_sparse(f, cache)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_bound)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            return result.x, cache
    except Exception:
        pass
    return None, cache


def subgradient_optimize(f, deadline, lr_init=0.02, top_k=5, momentum_decay=0.9, verbose_prefix=""):
    """Subgradient descent with momentum for max autocorrelation minimization."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score_fast(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    mom = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M < 1e-15:
            break

        top_indices = np.argsort(conv)[-top_k:]
        grad = np.zeros(n)
        for k in top_indices:
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += f[k - j]

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        grad /= gnorm

        mom = momentum_decay * mom - lr * grad
        new_f = np.clip(f + mom, 0.0, None)
        new_S = float(np.sum(new_f))
        if new_S < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 200:
                break
            it += 1
            continue
        new_f *= S / new_S

        new_score = compute_score_fast(new_f)
        it += 1

        if new_score < best_score:
            best_score = new_score
            best_f = new_f.copy()
            f = new_f
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.97
            mom *= 0.5
            if lr < 1e-8 or no_improve > 500:
                break

    return best_f, best_score


def lp_optimize(f, deadline, delta_init=0.015, step_init=0.5, verbose_prefix=""):
    """LP-based optimization with reduced-bound forcing and line search."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score_fast(f)
    n = len(f)
    S = float(np.sum(f))
    delta = delta_init
    step = step_init
    no_improve = 0
    it = 0
    cache = None

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M < 1e-15:
            break

        M_bound = M * (1.0 - delta)
        g, cache = solve_lp_direction(f, M_bound, cache)

        if g is None:
            delta *= 0.5
            if delta < 0.0005:
                no_improve += 1
                if no_improve > 80:
                    break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue

        g *= S / sum_g

        best_step_score = best_score
        best_step_f = None
        for alpha in [step, step * 0.5, step * 0.75, step * 0.25]:
            alpha = min(alpha, 1.0)
            if alpha < 0.01:
                continue
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0.0, None)
            cS = float(np.sum(candidate))
            if cS < 1e-15:
                continue
            candidate *= S / cS
            cs = compute_score_fast(candidate)
            if cs < best_step_score:
                best_step_score = cs
                best_step_f = candidate

        it += 1

        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            step = max(0.05, step * 0.995)
            delta = min(0.04, delta * 1.005)
        else:
            no_improve += 1
            delta = max(0.0005, delta * 0.6)
            step *= 0.85
            if step < 0.02 or no_improve > 120:
                break

        if it % 100 == 0:
            print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f} "
                  f"delta={delta:.5f} step={step:.3f}")

    return best_f, best_score


def spectral_refine(f, deadline, rng_seed=42):
    """Flatten the power spectrum in the frequency domain to reduce autocorrelation peak."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score_fast(f)
    n = len(f)
    S = float(np.sum(f))
    rng = np.random.default_rng(rng_seed)
    t = 0.05
    stall = 0

    while time.monotonic() < deadline:
        F = np.fft.rfft(f, n=2 * n - 1)
        power = np.abs(F) ** 2
        mean_power = float(np.mean(power))

        compression = 0.3
        target_ratio = 1.0 - compression * (power - mean_power) / (mean_power + 1e-15)
        target_ratio = np.clip(target_ratio, 0.2, 2.0)

        F_new = F * (1.0 - t) + F * np.sqrt(target_ratio) * t
        f_new = np.fft.irfft(F_new, n=2 * n - 1)[:n]
        f_new = np.clip(f_new, 0.0, None)
        fS = float(np.sum(f_new))
        if fS < 1e-15:
            t *= 0.5
            stall += 1
            if stall > 20:
                break
            continue
        f_new *= S / fS

        new_score = compute_score_fast(f_new)
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            stall = 0
            t = min(0.1, t * 1.05)
        else:
            stall += 1
            t *= 0.7
            if t < 1e-5 or stall > 30:
                break

    return best_f, best_score


def random_perturb(f, sigma, rng):
    """Add small random perturbation and reproject to non-negative with same sum."""
    n = len(f)
    S = float(np.sum(f))
    f_pert = f + sigma * rng.standard_normal(n)
    f_pert = np.clip(f_pert, 0.0, None)
    pS = float(np.sum(f_pert))
    if pS < 1e-15:
        return f.copy()
    f_pert *= S / pS
    return f_pert


def make_inits(n, rng, parent=None):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    inits.append(('uniform', np.ones(n)))

    cos_w = 0.5 * (1.0 + np.cos(np.pi * idx / (n - 1)))
    inits.append(('cosine', cos_w + 0.1))

    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * idx / n))
    inists_edge = 1.0 + 0.3 * np.cos(np.pi * idx / (n - 1))
    inits.append(('raised_cos', rc + 0.2))
    inits.append(('edge_enh', inists_edge))

    center = (n - 1) / 2.0
    gauss_wide = np.exp(-0.5 * ((idx - center) / (n * 0.45)) ** 2)
    inits.append(('gauss_wide', gauss_wide + 0.05))

    gauss_med = np.exp(-0.5 * ((idx - center) / (n * 0.25)) ** 2)
    inits.append(('gauss_med', gauss_med + 0.05))

    two_hump = (np.exp(-((idx - n * 0.2) ** 2) / (n * 0.08)) +
                np.exp(-((idx - n * 0.8) ** 2) / (n * 0.08)))
    inits.append(('two_hump', two_hump + 0.05))

    inits.append(('random', rng.uniform(0.3, 2.0, n)))

    sin_shape = 0.5 + 0.5 * np.sin(np.pi * idx / (n - 1))
    inits.append(('sin_half', sin_shape))

    quad = 1.0 - 0.7 * (2.0 * idx / (n - 1) - 1.0) ** 2
    inits.append(('quad', np.clip(quad, 0.1, None)))

    if parent is not None:
        if len(parent) == n:
            inits.append(('parent', parent.copy()))
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            p_interp = np.interp(x_new, x_old, parent)
            inits.append(('parent_interp', p_interp))

    return inits


def deep_optimize(f, deadline, verbose_prefix=""):
    """Deep optimization: alternate LP, spectral, and subgradient with restarts."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score_fast(f)
    n = len(f)
    rng = np.random.default_rng(77)
    phase = 0
    last_best = best_score
    stall_count = 0

    while time.monotonic() < deadline:
        phase_deadline = min(deadline, time.monotonic() + 120.0)
        phase += 1

        if phase % 3 == 1:
            f_cur, s_cur = lp_optimize(f, phase_deadline, delta_init=0.015,
                                        step_init=0.5, verbose_prefix=verbose_prefix + "LP")
        elif phase % 3 == 2:
            f_cur, s_cur = spectral_refine(f, phase_deadline, rng_seed=phase)
        else:
            f_cur, s_cur = subgradient_optimize(f, phase_deadline, lr_init=0.01,
                                                 top_k=7, verbose_prefix=verbose_prefix + "SG")

        if s_cur < best_score:
            best_score = s_cur
            best_f = f_cur.copy()
            f = f_cur.copy()
            stall_count = 0
            print(f"    {verbose_prefix}Phase {phase}: improved to {best_score:.6f}")
        else:
            stall_count += 1
            if stall_count >= 2:
                f = random_perturb(f, 0.02 * np.max(f), rng)
                f = np.clip(f, 0.0, None)

        if time.monotonic() > deadline:
            break

    return best_f, best_score


def propose_candidate(budget_s=1500):
    """Main entry point: search for the best sequence minimizing the objective."""
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 15.0

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(2024)

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.all(np.isfinite(parent)):
            s = evaluate_sequence(parent.tolist())
            if np.isfinite(s):
                best_score = s
                best_sequence = parent.copy()
                print(f"Parent loaded: n={parent.size}, score={s:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Phase 1: Quick exploration (first 25% of budget)
    print("\n" + "=" * 60)
    print("PHASE 1: Quick Exploration")
    print("=" * 60)

    expl_deadline = t0 + float(budget_s) * 0.25
    lengths_to_try = [1024, 2048, 512, 1536]

    exploration_results = []

    for n in lengths_to_try:
        if time.monotonic() > expl_deadline - 20:
            break
        print(f"\n--- Exploring n={n} ---")
        inits = make_inits(n, rng, parent)
        per_init = max(5.0, (expl_deadline - time.monotonic()) / max(len(inits) * 2, 1))

        for label, f_init in inits:
            if time.monotonic() > expl_deadline - 5:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            if float(np.sum(f_init)) < 1e-10:
                continue

            # Quick subgradient (short)
            sg_dl = min(expl_deadline, time.monotonic() + per_init * 0.4)
            f_sg, s_sg = subgradient_optimize(f_init, sg_dl, lr_init=0.03, top_k=5)

            # Quick LP (short)
            lp_dl = min(expl_deadline, time.monotonic() + per_init * 0.5)
            f_lp, s_lp = lp_optimize(f_sg, lp_dl, delta_init=0.02, step_init=0.6)

            exploration_results.append((s_lp, f_lp, n, label))
            if s_lp < best_score:
                best_score = s_lp
                best_sequence = f_lp.copy()
                print(f"  *** NEW BEST: n={n} {label}: {s_lp:.6f}")
            else:
                print(f"  n={n} {label}: {s_lp:.6f}")

    # Sort exploration results
    exploration_results.sort(key=lambda x: x[0])
    print(f"\nTop 5 after exploration:")
    for sc, seq, nn, lbl in exploration_results[:5]:
        print(f"  n={nn} {lbl}: {sc:.6f}")
    print(f"Best so far: {best_score:.6f}")

    # Phase 2: Deep optimization of top candidates (next 55%)
    print("\n" + "=" * 60)
    print("PHASE 2: Deep Optimization")
    print("=" * 60)

    deep_deadline = t0 + float(budget_s) * 0.80

    # Select top unique-length candidates
    seen_lengths = set()
    top_cands = []
    for sc, seq, nn, lbl in exploration_results:
        if nn not in seen_lengths:
            top_cands.append((sc, seq, nn, lbl))
            seen_lengths.add(nn)
        if len(top_cands) >= 3:
            break

    # Also ensure parent length is included
    if parent is not None and len(parent) not in seen_lengths:
        top_cands.append((best_score, parent.copy(), len(parent), 'parent'))

    for i, (sc, f_init, nn, lbl) in enumerate(top_cands):
        if time.monotonic() > deep_deadline - 60:
            break
        time_alloc = (deep_deadline - time.monotonic()) / (len(top_cands) - i)
        cand_deadline = time.monotonic() + time_alloc
        print(f"\n--- Deep opt: n={nn} {lbl} (start={sc:.6f}, time={time_alloc:.0f}s) ---")

        f_best, s_best = deep_optimize(f_init, cand_deadline, verbose_prefix=f"[{nn}/{lbl}]")
        if s_best < best_score:
            best_score = s_best
            best_sequence = f_best.copy()
            print(f"  *** NEW BEST: {s_best:.6f}")

    # Phase 3: Final polish (last 20%)
    print("\n" + "=" * 60)
    print(f"PHASE 3: Final Polish (best={best_score:.6f})")
    print("=" * 60)

    if best_sequence is not None and time.monotonic() < deadline - 20:
        n_best = len(best_sequence)
        print(f"Polishing n={n_best}...")
        f_pol, s_pol = deep_optimize(best_sequence, deadline, verbose_prefix="[polish]")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()
            print(f"  *** FINAL BEST: {s_pol:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(1024)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Final validation
    final_list = [float(x) for x in best_sequence]
    final_score = evaluate_sequence(final_list)
    if not np.isfinite(final_score):
        print("WARNING: Final sequence invalid! Using fallback.")
        best_sequence = np.ones(1024)
        final_list = [float(x) for x in best_sequence]
        final_score = evaluate_sequence(final_list)

    elapsed = time.monotonic() - t0
    print(f"\n{'=' * 60}")
    print(f"FINAL RESULT: n={len(final_list)}, score={final_score:.6f}, time={elapsed:.1f}s")
    print(f"{'=' * 60}")

    return final_list
