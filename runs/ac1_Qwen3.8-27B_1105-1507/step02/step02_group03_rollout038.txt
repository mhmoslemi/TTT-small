"""
Optimized sequence search using:

4. Multi-start with diverse initializations
5. Simulated annealing for global exploration

import numpy as np
import time
from scipy.optimize import linprog


def normalize(x):
    s = x.sum()
    if s < 1e-15:
        return x.copy()
    return x / s


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for the given sequence."""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def compute_peak_normalized(f):
    """Compute peak of convolution for normalized (sum=1) sequence. Score = 2*n*peak."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def spectral_shape(f, n_iter=200, alpha=0.3, rng=None):
    """Adaptive spectral shaping to flatten |F|^2 while maintaining non-negativity."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 2 * n  # Use at least 2n for linear convolution via circular
    
    best_f = f.copy()
    best_score = compute_peak_normalized(f)
    stall = 0
    
    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag2 = np.abs(F) ** 2
        phase = np.angle(F)
        
        # Target: flat magnitude spectrum (DC fixed)
        # Mean of non-DC bins
        mean_mag2 = np.mean(mag2[1:])
        target_mag2 = np.full_like(mag2, mean_mag2)
        target_mag2[0] = mag2[0]  # DC fixed
        
        # Soft update
        beta = alpha * max(0.1, 1.0 - it / n_iter)
        new_mag2 = mag2 + beta * (target_mag2 - mag2)
        new_mag2 = np.maximum(new_mag2, 0)
        
        new_mag = np.sqrt(new_mag2)
        F_new = new_mag * np.exp(1j * phase)
        
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        
        score = compute_peak_normalized(f_new)
        if score < best_score - 1e-10:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 100:

                f = f * (1 + 0.1 * rng.normal(0, 1, n))
                f = np.clip(f, 0, None)
                f = normalize(f)
                stall = 0
                if it > n_iter - 20:
                    break
    
    return best_f, best_score

        for i in order:
            orig_val = f[i]
            if orig_val < 1e-10:
                # Try small positive values
                candidates = [1e-4, 1e-3, 2e-3, 5e-3, orig_val]
            else:
                # Ternary search range
                lo = 0.0
                hi = max(3.0 * orig_val, 4.0 / n)
                
                # Quick check: is the current value already good?
                best_local = compute_peak_normalized(f)
                
                # Ternary search (peak is convex in f[i])
                for _ in range(30):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f[i] = m1
                    p1 = np.convolve(f, f).max()
                    f[i] = m2
                    p2 = np.convolve(f, f).max()
                    if p1 < p2:
                        hi = m2
                    else:
                        lo = m1
                    if hi - lo < 1e-14:
                        break
                
                f[i] = (lo + hi) / 2.0
                new_peak = np.convolve(f, f).max()
                
                if new_peak >= best_local:
                    f[i] = orig_val
                    continue
                
                improved_this_sweep += 1
            
            # Handle the case where orig was near zero
            if orig_val < 1e-10:
                best_val = orig_val
                best_p = np.convolve(f, f).max()
                for cv in candidates:
                    f[i] = cv
                    p = np.convolve(f, f).max()
                    if p < best_p:
                        best_p = p
                        best_val = cv
                f[i] = best_val
                if best_val > orig_val:
                    improved_this_sweep += 1
        
        f = normalize(f)
        new_score = compute_peak_normalized(f)
        if new_score < best_score:
            best_score = new_score
    
    return f, best_score


def lp_direction(f, M_factor=1.0, lp_time=5):
    """Find LP direction to reduce peak."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max() * M_factor
    
    # Build constraint matrix: A @ g = f * g (convolution)
    # A has shape (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, j] = f[k - j]
    
    c = -np.ones(n)  # maximize sum(g) = minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': lp_time})
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    g = np.clip(res.x, 0, None)
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s


def lp_refine(f, time_budget, rng=None):
    """LP-based refinement loop."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_peak_normalized(f)
    start = time.monotonic()
    t = 0.5
    
    while time.monotonic() - start < time_budget:
        g = lp_direction(f, M_factor=1.0, lp_time=min(3, time_budget - (time.monotonic() - start)))
        if g is None:
            break
        
        improved = False
        for alpha in [min(1.0, t), t * 0.5, t * 0.25, 0.1]:
            f_new = (1 - alpha) * f + alpha * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            score = compute_peak_normalized(f_new)
            if score < best_score - 1e-12:
                best_score = score
                best_f = f_new.copy()
                f = f_new
                t = min(1.0, t * 1.2)
                improved = True
                break
        
        if not improved:
            t *= 0.7
            if t < 0.01:
                # Perturb
                f = f * (1 + 0.2 * rng.normal(0, 1, n))
                f = np.clip(f, 0, None)
                f = normalize(f)
                if compute_peak_normalized(f) < best_score:
                    best_f = f.copy()
                    best_score = compute_peak_normalized(f)
                    t = 0.3
                else:
                    break
    
    return best_f, best_score


def pressure_update(f, n_iter=300, eta=0.2, rng=None):
    """Pressure-based multiplicative updates."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_peak_normalized(f)
    stall = 0
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Pressure: for each i, sum of f[j] over j where conv at (i+j) is near peak
        # Simplified: use the gradient of the peak
        # d(conv[k])/d(f[i]) = f[k-i] (for valid k-i)
        # For the peak at k*, pressure on f[i] is f[k*-i]
        
        # Use top-K peaks for robustness
        top_k = max(3, n // 16)
        top_indices = np.argsort(conv)[::-1][:top_k]
        weights = np.zeros(2 * n - 1)
        for rank, k in enumerate(top_indices):
            weights[k] = 1.0 / (rank + 1)
        
        pressure = np.zeros(n)
        for i in range(n):
            for k in top_indices:
                j = k - i
                if 0 <= j < n:
                    pressure[i] += weights[k] * f[j]
        
        pressure = np.maximum(pressure, 1e-10)
        pressure = pressure / pressure.max()
        
        eta_eff = eta * max(0.1, 1.0 - it / n_iter)
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = f_new / f_new.sum()
        
        score = compute_peak_normalized(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                f = f * (1 + 0.15 * rng.normal(0, 1, n))
                f = np.clip(f, 0, None)
                f = normalize(f)
                if compute_peak_normalized(f) < best_score:
                    best_f = f.copy()
                    best_score = compute_peak_normalized(f)
                stall = 0
                if it > n_iter - 30:
                    break
    
    return best_f, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    r = rng.random()
    if r < 0.2:
        # Gaussian
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
    elif r < 0.4:
        # Cosine
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.6:
        # Tapered
        f = np.abs(np.sin(np.pi * np.arange(n) / n))
        f = f ** rng.uniform(0.5, 2.0)
    elif r < 0.8:
        # Product of sines
        f = np.ones(n)
        for _ in range(3):
            freq = rng.uniform(1, 4)
            f *= 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n)
    else:
        # Random smooth
        f = np.ones(n)
        for _ in range(5):
            c = int(rng.integers(0, n))
            w = max(2, int(n * rng.uniform(0.05, 0.3)))
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
            f *= (1 + rng.normal(0, 0.5) * kernel)
        f = np.abs(f)
    
    f = np.clip(f, 0.01, None)
    f = normalize(f)
    f = f * (1 + 0.05 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def simulated_annealing(f, rng, n_iter=10000, T0=0.02, T_end=1e-5):
    """Simulated annealing with diverse moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_peak_normalized(f)
    cur_score = best_score
    no_improve = 0
    
    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()
        r = rng.random()
        
        if r < 0.25:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.5))
        elif r < 0.45:
            # Single element
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.1 / n))
        elif r < 0.6:
            # Pair swap
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.8:
            # Smooth perturbation
            c = int(rng.integers(0, n))
            w = max(2, n // 20)
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
            f_new += rng.normal(0, 0.05) * kernel
        else:
            # Large random perturbation
            f_new *= np.exp(rng.normal(0, 0.4, n))
        
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_peak_normalized(f_new)
        
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 3000 and it > n_iter // 2:
            break
    
    return best_f, best_score

"""
Optimized sequence search combining spectral shaping, coordinate descent with exact
1D minimization, LP-based direction finding, pressure updates, and simulated annealing.
Target: score < 1.5027 for the evaluation function 2*n*max(conv)/sum^2.

import numpy as np
import time
from scipy.optimize import linprog


def normalize(x):
    """Normalize to sum=1."""
    s = x.sum()
    if s < 1e-15:
        return x.copy()
    return x / s


def compute_peak_score(f):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def spectral_shape(f, n_iter=200, alpha=0.35, rng=None):
    """Adaptive spectral shaping: flatten |F|^2 while maintaining non-negativity."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 2 * n

    best_f = f.copy()
    best_score = compute_peak_score(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag2 = np.abs(F) ** 2
        phase = np.angle(F)

        # Target: flatten non-DC bins toward their mean
        mean_mag2 = np.mean(mag2[1:])
        target_mag2 = np.full_like(mag2, mean_mag2)
        target_mag2[0] = mag2[0]

        # Soft update with decaying alpha
        beta = alpha * max(0.05, 1.0 - 0.5 * it / n_iter)
        new_mag2 = mag2 + beta * (target_mag2 - mag2)
        new_mag2 = np.maximum(new_mag2, 0)

        new_mag = np.sqrt(new_mag2)
        F_new = new_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s

        score = compute_peak_score(f_new)
        if score < best_score - 1e-11:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                f = f * (1 + 0.12 * rng.normal(0, 1, n))
                f = np.clip(f, 0, None)
                f = normalize(f)
                stall = 0
                if it > n_iter - 30:
                    break

    return best_f, best_score


def coordinate_descent_exact(f, n_sweeps=30, rng=None):
    """Coordinate descent with ternary search for exact 1D minimization per coordinate."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_peak_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        improved = 0

        for i in order:
            orig_val = f[i]
            best_local_score = compute_peak_score(f)

            if orig_val < 1e-8:
                # Try a few small positive values
                best_val = orig_val
                for cv in [1e-4, 5e-4, 1e-3, 2e-3, 5e-3]:
                    f[i] = cv
                    sc = compute_peak_score(f)
                    if sc < best_local_score:
                        best_local_score = sc
                        best_val = cv
                f[i] = best_val
                if best_val > orig_val:
                    improved += 1
            else:
                # Ternary search: peak is convex in f[i]
                lo = 0.0
                hi = max(3.0 * orig_val, 5.0 / n)

                for _ in range(35):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f[i] = m1
                    p1 = np.convolve(f, f).max()
                    f[i] = m2
                    p2 = np.convolve(f, f).max()
                    if p1 < p2:
                        hi = m2
                    else:
                        lo = m1
                    if hi - lo < 1e-14 * max(1.0, orig_val):
                        break

                f[i] = (lo + hi) / 2.0
                new_score = compute_peak_score(f)
                if new_score < best_local_score - 1e-12:
                    improved += 1
                else:
                    f[i] = orig_val

        f = normalize(f)
        new_score = compute_peak_score(f)
        if new_score < best_score:
            best_score = new_score

        if improved == 0 and sweep > 5:
            # Stuck - add small perturbation
            f = f * (1 + 0.05 * rng.normal(0, 1, n))
            f = np.clip(f, 0, None)
            f = normalize(f)
            if compute_peak_score(f) < best_score:
                best_score = compute_peak_score(f)

    return f, best_score


def build_conv_matrix(f):
    """Build matrix A where A @ g = np.convolve(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        for j in range(j_lo, j_hi):
            A[k, j] = f[k - j]
    return A


def lp_direction(f, M_factor=1.0, lp_time=4):
    """Find LP direction to reduce convolution peak."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max() * M_factor

    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': lp_time, 'presolve': True})
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = np.clip(res.x, 0, None)
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s


def lp_refine(f, time_budget, rng=None):
    """LP-based refinement: iteratively find LP direction and blend."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_peak_score(f)
    start = time.monotonic()
    t = 0.4

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        t_lp = min(4, max(1, int(remaining * 0.3)))
        g = lp_direction(f, M_factor=1.0, lp_time=t_lp)
        if g is None:
            break

        improved = False
        for alpha in [min(1.0, t * 1.3), t, t * 0.5, t * 0.25, 0.05]:
            f_new = (1 - alpha) * f + alpha * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            score = compute_peak_score(f_new)
            if score < best_score - 1e-12:
                best_score = score
                best_f = f_new.copy()
                f = f_new
                t = min(1.0, t * 1.15)
                improved = True
                break

        if not improved:
            t *= 0.7
            if t < 0.02:
                f = f * (1 + 0.2 * rng.normal(0, 1, n))
                f = np.clip(f, 0, None)
                f = normalize(f)
                sc = compute_peak_score(f)
                if sc < best_score:
                    best_f = f.copy()
                    best_score = sc
                    t = 0.3
                else:
                    break

    return best_f, best_score


def pressure_update(f, n_iter=300, eta=0.18, rng=None):
    """Pressure-based multiplicative updates targeting peak lags."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_peak_score(f)
    stall = 0
    top_k = max(5, n // 12)

    for it in range(n_iter):
        conv = np.convolve(f, f)
        top_indices = np.argsort(conv)[::-1][:top_k]

        pressure = np.zeros(n)
        for rank, k in enumerate(top_indices):
            w = 1.0 / (rank + 1)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += w * f[j]

        if pressure.max() < 1e-15:
            break
        pressure = pressure / pressure.max()

        eta_eff = eta * max(0.1, 1.0 - it / (n_iter + 1))
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = f_new / f_new.sum()

        score = compute_peak_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 40:
                f = f * (1 + 0.15 * rng.normal(0, 1, n))
                f = np.clip(f, 0, None)
                f = normalize(f)
                sc = compute_peak_score(f)
                if sc < best_score:
                    best_f = f.copy()
                    best_score = sc
                    stall = 0
                if it > n_iter - 40:
                    break

    return best_f, best_score


def diverse_init(n, rng):

    r = rng.random()

    f = np.clip(f, 0.005, None)
    f = normalize(f)
    f = f * (1 + 0.05 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)

def simulated_annealing(f, rng, n_iter=8000, T0=0.015, T_end=1e-5):
    """Simulated annealing with diverse perturbation moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_peak_score(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()
        r = rng.random()

        if r < 0.25:
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.5))
        elif r < 0.45:
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.15 / np.sqrt(n)))
        elif r < 0.6:
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.8:
            c = int(rng.integers(0, n))
            w = max(2, n // 16)
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
            f_new += rng.normal(0, 0.08) * kernel
        else:
            f_new *= np.exp(rng.normal(0, 0.35, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_peak_score(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 2500 and it > n_iter // 2:
            break

    return best_f, best_score
