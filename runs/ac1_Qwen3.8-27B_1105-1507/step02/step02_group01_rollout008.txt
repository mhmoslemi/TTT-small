    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_max = min(n - 1, k)
        j_min = max(0, k - n + 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp_for_a(a):
    """Solve LP: max sum(b) s.t. A(a) @ b <= M, b >= 0."""
    n = len(a)
    conv = convolve_fast(a)
    M = float(np.max(conv))
    A = build_A(a)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

def lp_phase(a, max_iters=500, time_budget=60.0):
    """Run LP alternating until convergence or time budget exceeded."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        b, b_sum, M = solve_lp_for_a(a)
        if b is None:
            break
        ratio = b_sum / 1.0  # since sum(a) = 1
        if ratio < 1.0 + 1e-8:
            break  # Converged
        
        b_norm = b / b_sum  # Now sum(b_norm) = 1
        
        # Line search
        improved = False
        for t in [1.0, 0.9, 0.7, 0.5, 0.3, 0.1]:
            a_new = (1.0 - t) * a + t * b_norm
            a_new = np.clip(a_new, 0.0, None)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break
        
        if not improved:
            # Accept even without improvement if we're stuck
            a = b_norm.copy()
    
    return best_a, best_obj

def gd_phase(a, time_budget=30.0, T_init=None):
    """Gradient descent on soft-max autocorrelation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv0 = convolve_fast(a)
    if T_init is None:
        T_init = 0.05 * float(np.max(conv0))
    T = T_init
    lr = 0.003
    patience = 0
    max_patience = 200
    
    while time.monotonic() - start < time_budget:
        conv = convolve_fast(a)
        max_c = float(np.max(conv))
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        p = exp_vals / sum_exp
        
        # Gradient computation
        P = np.array([p[j:j+n] for j in range(n)])
        grad = (2.0 / T) * (P @ a)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break
        
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            lr *= 0.5
            patience += 1
            if patience > max_patience:
                break
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
            patience = 0
            if T > 0.0005 * max_c:
                T *= 0.99
        else:
            lr *= 0.92
            patience += 1
            if patience >= max_patience:
                T = max(0.0005 * max_c, T * 0.7)
                lr = 0.001
                patience = 0
            if lr < 1e-12:
                break
    
    return best_a, best_obj

def perturb_and_optimize(a, rng, lp_time=15.0, gd_time=10.0):
    """Perturb a and re-optimize."""
    n = len(a)
    # Random multiplicative perturbation
    sigma = rng.uniform(0.05, 0.3)
    a_pert = a * np.exp(rng.normal(0, sigma, n))
    a_pert = np.clip(a_pert, 0.0, None)
    s = np.sum(a_pert)
    if s < 1e-10:
        a_pert = a.copy()
    else:
        a_pert = a_pert / s
    
    # LP
    a_opt, obj_opt = lp_phase(a_pert, max_iters=200, time_budget=lp_time)
    
    # GD refinement
    if gd_time > 0:
        a_ref, obj_ref = gd_phase(a_opt, time_budget=gd_time)
        if obj_ref < obj_opt:
            a_opt = a_ref
            obj_opt = obj_ref
    
    return a_opt, obj_opt
