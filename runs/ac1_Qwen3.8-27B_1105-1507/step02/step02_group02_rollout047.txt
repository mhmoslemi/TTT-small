1. a = current point (on simplex)
2. b = conv(a, a)  [length 2n-1]
3. p = softmax(beta * b)  [length 2n-1]
4. grad[i] = 2 * beta * sum_j p[j+i] * a[j]  for i in [0, n-1]
5. Update: a_new = project_simplex(a - step_size * grad)
6. Adjust step size (barzilai-borwein or backtracking)

The projection onto the simplex {a >= 0, sum(a) = 1} can be done efficiently.

This is essentially the **FISTA / Nesterov accelerated proximal gradient** method applied to our problem!

```python
    n = len(v)

    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    shifted = np.clip(shifted, -50, 50)
    p = np.exp(shifted)
    p /= p.sum()

    # This is the cross-correlation of p and a at lags 0, 1, ..., n-1
    grad = np.zeros(n)
    for i in range(n):

        # p has length 2n-1, so p[i:i+n] is valid for i in [0, n-1]
        grad[i] = 2.0 * beta * np.dot(p[i:i+n], a)

    prev_a = a.copy()

        beta_mom = (t_new - 1) / t_new

        y = project_simplex(y)
        
        grad_y = smooth_gradient(y, beta)
        
        # Barzilai-Borwein step size update
        if it > 0:

            pass

        # Check improvement
        obj = compute_score(a_new)
        if obj < best_obj:

            best_a = a_new.copy()
        
        prev_a = a
        prev_grad = grad_y
        a = a_new
        t = t_new
```

```python
def cross_corr_fft(p, a):
    """Compute cross[i] = sum_j p[j+i] * a[j] for i in [0, n-1] using FFT."""
    n = len(a)
    m = 2 * n - 1  # length of p
    size = 1
    while size < m + n - 1:
        size <<= 1
    
    p_rev = p[::-1]
    P = np.fft.rfft(p_rev, size)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(P * A, size)[:m + n - 1]
    
    # cross[i] = conv[2n-2-i] for i in [0, n-1]
    # So cross = conv[2n-2 : n-1 : -1] = conv[n-1 : 2n-1][::-1]
    # Wait: 2n-2-i for i=0 gives 2n-2, for i=n-1 gives n-1
    # So cross[i] = conv[2n-2-i], i.e., cross = conv[n-1:2n-1][::-1]
    cross = conv[n-1:2*n-1][::-1]
    return cross
```

This is very efficient to compute with FFT!

And for the p-norm with general p:
f_p(a) = 2n * (Σ_k b_k^p)^(1/p)
∂f_p/∂a_i = (2n/p) * (Σ b^p)^(1/p - 1) * Σ_k p * b_k^(p-1) * 2*a_{k-i}
= (4n) * (Σ b^p)^(1/p - 1) * Σ_j b[j+i]^(p-1) * a_j

For p=2: (4n) * (Σ b^2)^(-1/2) * Σ_j b[j+1]... wait, b[j+i]^(2-1) = b[j+i]
So ∂f_2/∂a_i = 4n * (Σ b^2)^(-1/2) * Σ_j b[j+i] * a_j ✓

For p=4: ∂f_4/∂a_i = 4n * (Σ b^4)^(-3/4) * Σ_j b[j+i]^3 * a_j

As p → ∞, this approaches the gradient of 2n * max(b), which is what we want.

So the strategy is:
1. Start with p=2 (or p=4) for fast convergence
2. Gradually increase p to get closer to the true objective
3. Use FISTA for the optimization

This is a "homotopy" approach! Start with an easy problem and gradually make it harder.

Let me implement this:

```python
def pnorm_gradient(a, p, n):
    """Gradient of 2n * ||conv(a,a)||_p w.r.t. a."""
    b = fast_conv(a)  # length 2n-1
    
    if p == 2:
        norm_sq = np.dot(b, b)
        if norm_sq < 1e-30:
            return np.zeros(n)
        # grad[i] = 4n / sqrt(norm_sq) * sum_j b[j+i] * a[j]
        cross = cross_corr_fft(b, a, n)
        return 4.0 * n / np.sqrt(norm_sq) * cross
    else:
        # General p
        b_pos = np.abs(b)
        norm_p = np.sum(b_pos ** p) ** (1.0 / p)
        if norm_p < 1e-30:
            return np.zeros(n)
        w = b_pos ** (p - 1) / norm_p ** (p - 1)  # weights for the gradient
        # Actually: grad[i] = 4n * (sum b^p)^(1/p - 1) * sum_j b[j+i]^(p-1) * a[j]
        # = 4n / norm_p^(p-1) * sum_j b[j+i]^(p-1) * a[j]
        # = 4n / norm_p^(p-1) * cross(b^(p-1), a)[i]
        b_powered = b ** (p - 1)  # b >= 0 since a >= 0
        cross = cross_corr_fft(b_powered, a, n)
        return 4.0 * n / norm_p ** (p - 1) * cross
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    m = 2 * n - 1
    s = 1
    while s < m:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:m]

def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def cross_corr_fft(p, a, n):
    """Compute cross[i] = sum_j p[j+i] * a[j] for i in [0, n-1]."""
    m = 2 * n - 1
    size = 1
    while size < m + n - 1:
        size <<= 1
    p_rev = p[::-1]
    P = np.fft.rfft(p_rev, size)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(P * A, size)[:m + n - 1]
    return conv[n-1:2*n-1][::-1]

def pnorm_grad(a, p, n):
    b = fast_conv(a)
    if p == 2:
        norm_sq = np.dot(b, b)
        if norm_sq < 1e-30:
            return np.zeros(n)
        cross = cross_corr_fft(b, a, n)
        return (4.0 * n / np.sqrt(norm_sq)) * cross
    else:
        b_pow = b ** (p - 1)
        norm_p = np.sum(b ** p) ** (1.0 / p)
        if norm_p < 1e-30:
            return np.zeros(n)
        cross = cross_corr_fft(b_pow, a, n)
        return (4.0 * n / norm_p ** (p - 1)) * cross

def fista_pnorm(a0, deadline, p, max_iters=1000, verbose=False):
    """FISTA for the p-norm objective."""
    n = len(a0)
    x = project_simplex(a0.copy())
    y = x.copy()
    t = 1.0
    
    # Initial step size
    g0 = pnorm_grad(y, p, n)
    L = max(np.linalg.norm(g0) / max(np.linalg.norm(y), 1e-10), 1.0)
    step = 1.0 / (L * 1.1)  # slightly conservative
    
    best_x = x.copy()
    best_score = compute_score(x)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        gy = pnorm_grad(y, p, n)
        x_new = project_simplex(y - step * gy)
        
        if it % 50 == 0:
            score = compute_score(x_new)
            if score < best_score:
                best_score = score
                best_x = x_new.copy()
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t_new - 1.0) / t_new
        y = x_new + momentum * (x_new - x)
        y = project_simplex(y)
        x = x_new
        t = t_new
    
    score = compute_score(x)
    if score < best_score:
        best_score = score
        best_x = x.copy()
    
    if verbose:
        print(f"  FISTA p={p}: score={best_score:.6f}", flush=True)
    
    return best_x, best_score

def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

def lp_best_response(a, M, n):
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None

def lp_iterate(a, deadline, max_iters=500, verbose=False):
    """LP best-response iterations."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, n)
        if g is None:
            break
        g = project_simplex(g)
        
        # Adaptive step size
        t = 0.3 * (1.0 - 0.5 * it / max_iters)
        t = max(0.05, t)
        
        a_new = (1.0 - t) * a + t * g
        a_new = project_simplex(a_new)
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            t2 = t * 0.3
            a_new2 = (1.0 - t2) * a + t2 * g
            a_new2 = project_simplex(a_new2)
            score2 = compute_score(a_new2)
            if score2 < best_score:
                best_score = score2
                best_a = a_new2.copy()
            a = a_new2
    
    if verbose:
        print(f"  LP: score={best_score:.6f}", flush=True)
    
    return best_a, best_score
```

One more idea: **mirror descent with the negative entropy regularizer**.

Instead of Euclidean projection onto the simplex, use the Bregman projection with the negative entropy:
D(x, y) = sum x_i * log(x_i / y_i) - sum x_i + sum y_i

The Bregman projection of v onto the simplex (with negative entropy) is just:
x_i = exp(v_i) / sum(exp(v_i)) = softmax(v_i)

This is the **multiplicative weight update**! And it's known to work well for problems on the simplex.

The mirror descent step:
y = softmax(log(a) - step * grad) = a * exp(-step * grad) / sum(a * exp(-step * grad))

This naturally keeps a in the interior of the simplex (all positive) and might converge faster for our problem.

```python
def mirror_descent(a0, deadline, p, max_iters=500):
    """Mirror descent with negative entropy (multiplicative weights)."""
    n = len(a0)
    a = np.maximum(a0, 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    step = 1.0
    for it in range(max_iters):
        if time.monotonic() > deadline:

                best_score = score
                best_a = a_new.copy()
        
        a = a_new
        step *= 0.99  # slowly decrease step size
    
    return best_a, best_score
```
