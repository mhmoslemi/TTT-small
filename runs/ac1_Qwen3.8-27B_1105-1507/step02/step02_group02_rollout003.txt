Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with backtracking line search (primary method)
2. Smooth log-sum-exp gradient refinement with analytical FFT-based gradients
3. Multi-start with diverse initializations and adaptive length selection
4. Stagnation detection with perturbation restarts

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Full linear cross-convolution a * b via FFT."""
    na, nb = len(a), len(b)
    s = 1
    while s < na + nb - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:na + nb - 1]


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def smooth_objective_and_grad(x, beta, n):
    """
    Log-sum-exp smoothed objective and gradient for x on probability simplex.
    f(x) = 2n * (1/beta) * log(sum_k exp(beta * (x*x)_k))
    Gradient computed via FFT cross-correlation.
    """
    x = np.maximum(x, 1e-15)
    s = x.sum()
    if s < 0.01:
        return 1e10, np.zeros(n)
    x_norm = x / s
    b = fast_conv(x_norm)
    max_b = b.max()
    shifted = beta * (b - max_b)
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    log_sum_exp = max_b + np.log(exp_shifted.sum()) / beta
    value = 2.0 * n * log_sum_exp
    # Gradient: df/dx_j = 4n * sum_k p[k] * x_norm[k-j] for valid k-j
    # Computed as cross-correlation via FFT
    p_rev = p[::-1]
    cross = fast_cross_conv(p_rev, x_norm)
    grad_core = cross[2 * n - 2: 3 * n - 2]
    grad = 4.0 * n * grad_core
    return value, grad

    resampled = np.maximum(resampled, 0)
    if resampled.sum() < 0.01:

                a = np.maximum(a, 1e-6)
                a = a / a.sum()
                n_iters = int((half_deadline - time.monotonic()) / 0.05)
                a_result, score = lp_iteration_with_linesearch(a, n_iters, half_deadline)
            else:
                half_deadline2 = time.monotonic() + (deadline - time.monotonic()) / 2
                a_result, score = smooth_gradient_descent(best, half_deadline2)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  Round {round_idx}: J={best_score:.6f}", flush=True)

    # Phase 6: Perturbation search with remaining time
    if time.monotonic() < deadline:
        print(f"\n=== Phase 6: Perturbation search ===", flush=True)
        n = best.size
        remaining = deadline - time.monotonic()
        n_tries = max(3, int(remaining / 20))
        for trial in range(n_tries):
            if time.monotonic() >= deadline:
                break
            scale = 0.05 + 0.1 * (trial % 4)
            noise = rng.normal(0, scale, n)
            a_pert = best + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            sub_deadline = min(time.monotonic() + 15, deadline)
            n_iters = int((sub_deadline - time.monotonic()) / 0.05)
            a_result, score = lp_iteration_with_linesearch(a_pert, n_iters, sub_deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  Pert trial {trial}: J={best_score:.6f}", flush=True)

    # Final validation
    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
