import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = (a*b)[k] for full convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_find_b(a, M=None):
    """Solve LP: maximize sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    if M is None:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        return res.x / max(res.x.sum(), 1e-15), float(-res.fun), M
    return None, None, M

def init_ramp(n, rng):
    """Linearly increasing initialization."""
    a = np.arange(1, n + 1, dtype=np.float64)
    a += rng.normal(0, 0.1, n)
    a = np.maximum(0, a)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_v_shape(n, rng):
    """V-shaped (high at edges, low in middle)."""
    center = (n - 1) / 2.0
    a = np.abs(np.arange(n) - center) + 1.0
    a += rng.normal(0, 0.1, n)
    a = np.maximum(0, a)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_exp_decay(n, rng):
    """Exponential decay from one end."""
    decay = rng.uniform(0.5, 2.0)
    a = np.exp(-decay * np.arange(n) / n)
    # Also try from the other end
    if rng.random() > 0.5:
        a = a[::-1]
    a += 0.05 * rng.random(n)
    a = np.maximum(0, a)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_gaussian(n, rng):
    """Gaussian centered."""
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.2, 0.5)

def gd_phase(a, max_time, T_init=None):
    """Soft-max gradient descent with adaptive learning rate."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)

    if T_init is None:

        T_init = 0.05 * float(np.max(conv))

    lr = 0.05
    lr_min = 1e-6
    last_improve = 0
    momentum = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    for it in range(50000):
        if time.monotonic() - start > max_time:
            break
        sm_val, grad = soft_max_grad(a, T)
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-10:
            lr *= 0.5
            if lr < lr_min:
                break
            continue
        a_new /= s
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            last_improve = it
            if T > 0.001 * float(np.max(np.convolve(a, a))):
                T *= 0.98
        else:
            lr *= 0.95
            if it - last_improve > 100:
                T = max(0.001, T * 0.5)
                lr = max(lr_min, lr * 2)
                # Random restart around best
                if it - last_improve > 300:
                    a = best_a * (1 + 0.02 * rng_np.random(n))
                    a = np.maximum(0, a)
                    s = a.sum()
                    if s > 1e-10:
                        a /= s
                    last_improve = it
                    lr = 0.02
            if lr < lr_min:
                break

    return best_a, best_obj


def coordinate_descent(a, max_time):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    current = a.copy()

    passes = 0
    while time.monotonic() - start < max_time:
        improved = False
        for i in range(n):
            if time.monotonic() - start > max_time:
                break
            # Try several values for a[i]
            base = current.copy()
            best_val = current[i]
            best_local_obj = compute_obj(base)
            # Candidate values: 0, current, current*0.5, current*1.5, current*random
            candidates = [0.0, current[i] * 0.3, current[i] * 0.7, current[i] * 1.3, current[i] * 2.0]
            candidates.append(current[i] * (1 + 0.3 * rng_np.standard_normal()))
            candidates = [max(0.0, c) for c in candidates if c >= 0]
            
            for cv in candidates:
                base[i] = cv
                s = base.sum()
                if s < 1e-10:
                    continue
                obj = compute_obj(base)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_val = cv
            
            if best_val != current[i]:
                current[i] = best_val
                s = current.sum()
                if s > 1e-10:
                    current /= s
                if best_local_obj < best_obj:
                    best_obj = best_local_obj
                    best_a = current.copy()
                    improved = True

        passes += 1
        if not improved and passes > 2:
            break

    return best_a, best_obj


def block_refine(a, max_time):
    """Block perturbation refinement."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        width = max(1, int(n * rng_np.uniform(0.02, 0.15)))
        pos = int(rng_np.integers(0, n - width + 1))
        candidate = best_a.copy()
        sigma = 0.05 + 0.1 * max(0.0, 1.0 - (time.monotonic() - start) / max_time)
        candidate[pos:pos + width] *= np.exp(rng_np.normal(0, sigma, width))
        candidate = np.maximum(0, candidate)
        s = candidate.sum()
        if s < 1e-10:
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj

    return best_a, best_obj


def full_pipeline(a_init, time_budget):

    best_obj = compute_obj(a)

        best_obj = obj

    # Phase 2: GD
    if time.monotonic() - start < time_budget * 0.7:
        gd_time = min(time_budget * 0.3, 20.0)
        a, obj = gd_phase(a, gd_time)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    # Phase 3: Second LP round
    if time.monotonic() - start < time_budget * 0.85:
        lp_time2 = min(time_budget * 0.15, 15.0)
        a2, obj2 = lp_alternate(a, max(5, int(lp_time2 / 0.1)), lp_time2)
        if obj2 < best_obj:
            best_a = a2.copy()
            best_obj = obj2
            a = a2

    # Phase 4: Fine GD
    if time.monotonic() - start < time_budget * 0.93:
        gd_time2 = min(time_budget * 0.07, 10.0)
        a, obj = gd_phase(a, gd_time2, T_init=0.005)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: structured init -> LP alternating -> gradient
    descent on soft-max -> coordinate descent -> block perturbation.
    Tries multiple sequence lengths and initialization shapes with random restarts.
    """
    global rng_np
    rng_np = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 8.0

    best_seq = None
    best_score = np.inf

    # Try parent if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Optimize the parent if it exists and is reasonable size
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.size <= 400:
            p_opt, p_obj = full_pipeline(parent.copy(), min(50.0, float(budget_s) * 0.08))
            if p_obj < best_score:
                best_score = p_obj
                best_seq = p_opt.copy()
                print(f"Parent optimized: obj={p_obj:.6f}")
    except Exception:
        pass

    # Main search over different n values
    n_values = [30, 40, 50, 60, 80, 100, 120, 150, 180, 200, 250, 300, 350, 400]

    for n in n_values:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        num_remaining_n = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining_n)
        time_per_n = min(time_per_n, 120.0)

        # Number of restarts based on available time
        num_restarts = max(2, min(5, int(time_per_n / 20)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts, {time_per_restart:.1f}s each")

        # Try different initialization types
        inits = all_inits(n, rng_np)
        rng_np.shuffle(inits)
        inits = inits[:num_restarts]

        for r in range(num_restarts):
            if time.monotonic() > deadline - 50:
                break
            a_init = inits[r % len(inits)]
            a_opt, obj_opt = full_pipeline(a_init, time_per_restart)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r}: obj={obj_opt:.6f}")

    # Final refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s remaining")

        # GP phase
        a_ref, obj_ref = full_pipeline(best_seq.copy(), min(remaining * 0.4, 50.0))
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After GP: obj={best_score:.6f}")

        # Coordinate descent
        if time.monotonic() < deadline - 10:
            cd_time = min(remaining * 0.25, 30.0)
            a_cd, obj_cd = coordinate_descent(best_seq, cd_time)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  After CD: obj={best_score:.6f}")

        # Block perturbation
        if time.monotonic() < deadline - 5:
            bp_time = deadline - time.monotonic() - 4.0
            a_bp, obj_bp = block_refine(best_seq, bp_time)
            if obj_bp < best_score:
                best_score = obj_bp
                best_seq = a_bp.copy()
                print(f"  After BP: obj={best_score:.6f}")

        # One more CD pass
        if time.monotonic() < deadline - 3:
            cd_time2 = deadline - time.monotonic() - 3.0
            a_cd2, obj_cd2 = coordinate_descent(best_seq, min(cd_time2, 20.0))
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  After CD2: obj={best_score:.6f}")

    if best_seq is None:
        # Fallback: use a ramp which should give ~2
        best_seq = np.arange(1, 101, dtype=np.float64) / np.sum(np.arange(1, 101, dtype=np.float64))
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
