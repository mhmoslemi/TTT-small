```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2 quickly."""
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    return 2.0 * n * M / (S * S)

def build_sparse_conv(f):
    """Build sparse matrix C such that (C @ g)[k] = convolve(f, g)[k] for k=0..2n-2."""
    n = len(f)
    # conv(k) = sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[k-j] * g[j]
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_step(f, M, n):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0. Returns g."""
    C = build_sparse_conv(f)
    c = -np.ones(n)
    res = linprog(c, A_ub=C, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

def lp_optimize(f, deadline, verbose=False):
    """Iterative LP optimization."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S0 = np.sum(f)
    best_f = f.copy()
    best_score = compute_obj(f)
    M = np.max(np.convolve(f, f))
    no_improve = 0
    
    it = 0
    while time.monotonic() < deadline:
        g = lp_step(f, M, n)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        sum_g = np.sum(g)
        if sum_g < 1e-15:
            no_improve += 1
            continue
        
        # Rescale g to have same sum as f
        g = g * (S0 / sum_g)
        
        # Try different blending factors
        best_alpha = None
        best_new_score = best_score
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            candidate = (1 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0, None)
            cand_sum = np.sum(candidate)
            if cand_sum < 1e-15:
                continue
            candidate = candidate * (S0 / cand_sum)
            score = compute_obj(candidate)
            if score < best_new_score:
                best_new_score = score
                best_alpha = alpha
                best_candidate = candidate
        
        it += 1
        if best_alpha is not None:
            f = best_candidate
            new_M = np.max(np.convolve(f, f))
            M = new_M
            if best_new_score < best_score:
                best_score = best_new_score
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"  LP iter {it}: score={best_score:.6f}, M={M:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            break
    
    return best_f, best_score
```

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    return 2.0 * n * M / (S * S)


def build_conv_matrix(f):
    """Sparse matrix C where (C @ g)[k] = sum_j f[k-j]*g[j] = conv(f,g)[k]."""
    n = len(f)
    nnz = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        nnz += j_max - j_min + 1
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    vals = np.empty(nnz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M, top_k=None):
    """Solve: max sum(g) s.t. conv(f,g)_k <= M (for top_k lags or all), g >= 0."""
    n = len(f)
    C = build_conv_matrix(f)
    b = np.full(2 * n - 1, M)
    
    if top_k is not None and top_k < 2 * n - 1:
        # Only enforce top_k constraints
        conv_vals = C @ f  # This gives conv(f,f)
        # Get indices of top_k
        top_idx = np.argsort(conv_vals)[::-1][:top_k]
        C_sub = C[top_idx, :]

"""
LP-based iterative optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find descent directions with multiple initializations,
sequence lengths, and constraint relaxation strategies. Includes Soilenka-style
multiplicative refinement for fast local optimization.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    return 2.0 * n * M / (S * S)


def build_conv_structure(n):
    """Precompute row/col indices for the convolution matrix (structure only)."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def make_conv_matrix(f, row_idx, col_idx):
    """Build sparse conv matrix using precomputed structure."""
    n = len(f)
    2n_minus_1 = 2 * n - 1
    vals = np.empty(len(row_idx), dtype=np.float64)
    # For entry (k, j): value is f[k-j]
    k_arr = row_idx.astype(np.int64)
    j_arr = col_idx.astype(np.int64)
    vals = f[k_arr - j_arr]
    return csr_matrix((vals, (row_idx, col_idx)), shape=(2n_minus_1, n))


def solve_lp_full(f, M, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0."""
    n = len(f)
    C = make_conv_matrix(f, row_idx, col_idx)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None


def solve_lp_topk(f, M, top_k, row_idx, col_idx):
    """Solve LP with only top_k constraints enforced."""
    n = len(f)
    conv_ff = np.convolve(f, f)
    top_indices = np.argsort(conv_ff)[::-1][:top_k]
    # Filter rows
    mask = np.isin(row_idx, top_indices, assume_unique=False)
    # More efficient: build set
    top_set = set(top_indices)
    keep = np.array([k in top_set for k in row_idx])
    r_sub = row_idx[keep]
    c_sub = col_idx[keep]
    C = make_conv_matrix(f, r_sub, c_sub)
    c = -np.ones(n)
    b = np.full(len(top_indices), M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def lp_optimize(f, deadline, row_idx, col_idx, use_topk=None, verbose=False, vp=""):
    """Main LP optimization loop."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S0 = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_obj(f)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M <= 0:
            break

        if use_topk is not None and use_topk < 2 * n - 1:
            g = solve_lp_topk(f, M, use_topk, row_idx, col_idx)
        else:
            g = solve_lp_full(f, M, row_idx, col_idx)

        if g is None:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue

        g = g * (S0 / sum_g)

        # Line search
        best_cand_score = best_score
        best_cand = None
        for alpha in [0.15, 0.3, 0.5, 0.75, 1.0]:
            cand = (1.0 - alpha) * f + alpha * g
            cand = np.clip(cand, 0.0, None)
            cs = float(np.sum(cand))
            if cs < 1e-15:
                continue
            cand = cand * (S0 / cs)
            sc = compute_obj(cand)
            if sc < best_cand_score:
                best_cand_score = sc
                best_cand = cand

        it += 1

        if best_cand is not None:
            f = best_cand
            if best_cand_score < best_score:
                best_score = best_cand_score
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"    {vp}iter {it}: {best_score:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 100:
            break

    return best_f, best_score


def soilenka_optimize(f, deadline, verbose=False, vp=""):
    """Soilenka-style multiplicative update optimization."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S0 = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_obj(f)
    no_improve = 0
    it = 0
    eta = 0.1

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M <= 0:
            break

        # Get top peaks
        n_peaks = min(5, 2 * n - 1)
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        # Build correction
        correction = np.zeros(n)
        for k in peak_indices:
            excess = conv[k] - M * 0.8  # threshold
            if excess <= 0:
                continue
            for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
                correction[j] += f[k - j]

        # Multiplicative update
        f_new = f * (1.0 - eta * correction / max(M, 1e-15))
        f_new = np.clip(f_new, 0.0, None)
        s = float(np.sum(f_new))
        if s < 1e-15:
            eta *= 0.5
            no_improve += 1
            if no_improve > 100:
                break
            it += 1
            continue
        f_new = f_new * (S0 / s)

        sc = compute_obj(f_new)
        it += 1

        if sc < best_score:
            f = f_new
            best_score = sc
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    {vp}soilenka iter {it}: {best_score:.6f}")
        else:
            no_improve += 1
            eta *= 0.9
            if eta < 1e-4 or no_improve > 200:
                break

    return best_f, best_score


def alt_lp_optimize(f, deadline, row_idx, col_idx, verbose=False, vp=""):
    """Alternating LP: alternate between solving for g given f, and for f given g."""
    n = len(f)
    f = f.astype(np.float64).copy()
    g = f.copy()
    S0 = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_obj(f)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        # Solve for g given f
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        g_new = solve_lp_full(f, M, row_idx, col_idx)
        if g_new is not None:
            g_new = g_new * (S0 / max(np.sum(g_new), 1e-15))
            g = g_new

        # Try blending
        best_cand_score = best_score
        best_cand = None
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = f + alpha * (g - f)
            cand = np.clip(cand, 0.0, None)
            cs = float(np.sum(cand))
            if cs < 1e-15:
                continue
            cand = cand * (S0 / cs)
            sc = compute_obj(cand)
            if sc < best_cand_score:
                best_cand_score = sc
                best_cand = cand

        it += 1

        if best_cand is not None:
            f = best_cand
            if best_cand_score < best_score:
                best_score = best_cand_score
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    {vp}alt_lp iter {it}: {best_score:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            break

    return best_f, best_score


def make_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Triangular
    tri = 1.0 - np.abs(idx - center) / center
    inits.append(('tri', np.clip(tri, 0.01, None)))

    # Gaussians with various widths
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((idx - center) / (n * sf)) ** 2)
        inits.append((f'gauss_{sf}', g))

    # Hann
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * idx / max(n - 1, 1)))
    inits.append(('hann', np.clip(hann, 0.001, None)))

    # Quartic
    q = np.clip(1.0 - ((idx - center) / center) ** 4, 0.001, None)
    inits.append(('quartic', q))

    # Bimodal
    bim = (np.exp(-0.5 * ((idx - 0.3 * n) / (0.12 * n)) ** 2) +
           np.exp(-0.5 * ((idx - 0.7 * n) / (0.12 * n)) ** 2))
    inits.append(('bimodal', bim))

    # Flat-top
    edge = max(1, int(0.1 * n))
    ft = np.ones(n)
    ramp = 0.5 * (1.0 + np.cos(np.pi * np.arange(edge) / edge))
    ft[:edge] = ramp
    ft[-edge:] = ramp[::-1]
    inits.append(('flattop', ft))

    # Random
    inits.append(('rand1', rng.uniform(0.1, 1.0, n)))
    inits.append(('rand2', rng.uniform(0.1, 1.0, n) * np.sin(np.pi * idx / n) + 0.1))

    # Parent interpolated
    if parent is not None:
        if len(parent) == n:
            inits.append(('parent', parent.copy()))
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            pi = np.interp(x_new, x_old, parent)
            inits.append(('parent_interp', pi))

    return inits


def propose_candidate(budget_s=1500):
    t0 = time.monotonic()
    total_budget = float(budget_s)
    deadline = t0 + total_budget - 15.0

    best_score = float('inf')
    best_seq = None
    rng = np.random.default_rng(2024)

    # Load parent
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 1:
            sc = evaluate_sequence(parent.tolist())
            if np.isfinite(sc):
                best_score = sc
                best_seq = parent.copy()
                print(f"Parent loaded: n={parent.size}, score={sc:.6f}")
            else:
                parent = None
    except Exception as e:
        print(f"No parent: {e}")

    # Define n values and time allocation
    # Allocate: 45% to n=1024, 30% to n=2048, 15% to n=512, 10% to n=4096
    n_configs = [
        (1024, 0.40),
        (2048, 0.30),
        (512, 0.15),
        (4096, 0.08),
    ]

    # Phase 1: Explore different n and initializations
    print("\n=== Phase 1: Exploration ===")
    phase1_deadline = t0 + total_budget * 0.55

    for n, frac in n_configs:
        n_deadline = min(phase1_deadline, t0 + frac * (total_budget * 0.55))
        if time.monotonic() > n_deadline:
            continue
        remaining_for_n = n_deadline - time.monotonic()
        if remaining_for_n < 30:
            continue

        print(f"\n--- n={n} ({remaining_for_n:.0f}s allocated) ---")

        # Precompute conv matrix structure
        row_idx, col_idx = build_conv_structure(n)

        inits = make_inits(n, rng, parent)
        n_inits = len(inits)
        time_per_init = remaining_for_n * 0.7 / max(n_inits, 1)

        for label, init in inits:
            if time.monotonic() > n_deadline - 10:
                break

            init = np.clip(init.astype(np.float64), 0.0, 1000.0)
            if np.sum(init) < 1e-10:
                continue

            init_score = compute_obj(init)
            if init_score < best_score:
                best_score = init_score
                best_seq = init.copy()

            # Run optimization with time limit
            init_dl = min(n_deadline, time.monotonic() + time_per_init)
            if time.monotonic() > init_dl:
                continue

            # Strategy 1: Full LP
            f1, s1 = lp_optimize(init, init_dl, row_idx, col_idx,
                                 use_topk=None, verbose=False, vp=f"[{n}/{label}]")
            if s1 < best_score:
                best_score = s1
                best_seq = f1.copy()

            # Strategy 2: Top-k LP on the best result
            remaining2 = n_deadline - time.monotonic()
            if remaining2 > 10 and s1 < compute_obj(init):
                f2, s2 = lp_optimize(f1, time.monotonic() + remaining2 * 0.5,
                                     row_idx, col_idx, use_topk=20,
                                     verbose=False, vp=f"[{n}/{label}/topk]")
                if s2 < best_score:
                    best_score = s2
                    best_seq = f2.copy()

            # Strategy 3: Soilenka polish
            remaining3 = n_deadline - time.monotonic()
            if remaining3 > 10:
                f3, s3 = soilenka_optimize(best_seq, time.monotonic() + remaining3 * 0.5,
                                           verbose=False, vp=f"[{n}/{label}/sol]")
                if s3 < best_score:
                    best_score = s3
                    best_seq = f3.copy()

            print(f"  {label}: init={init_score:.6f} best={best_score:.6f}")

    print(f"\nAfter Phase 1: best={best_score:.6f}")

    # Phase 2: Focused optimization on best sequence
    print("\n=== Phase 2: Focused Optimization ===")
    phase2_deadline = deadline - 60.0

    if best_seq is not None:
        n_best = len(best_seq)
        row_idx, col_idx = build_conv_structure(n_best)
        print(f"Optimizing n={n_best} from score={best_score:.6f}")

        # Extended LP optimization
        f_ext, s_ext = lp_optimize(best_seq, phase2_deadline - 60,
                                   row_idx, col_idx, use_topk=None,
                                   verbose=True, vp="[ext]")
        if s_ext < best_score:
            best_score = s_ext
            best_seq = f_ext.copy()
            print(f"  Extended LP: {best_score:.6f}")

        # Top-k LP refinement
        remaining = phase2_deadline - time.monotonic()
        if remaining > 20:
            f_tk, s_tk = lp_optimize(best_seq, time.monotonic() + remaining * 0.6,
                                     row_idx, col_idx, use_topk=15,
                                     verbose=False, vp="[tk]")
            if s_tk < best_score:
                best_score = s_tk
                best_seq = f_tk.copy()
                print(f"  Top-k LP: {best_score:.6f}")

        # Soilenka polish
        remaining = phase2_deadline - time.monotonic()
        if remaining > 10:
            f_sol, s_sol = soilenka_optimize(best_seq, time.monotonic() + remaining * 0.7,
                                             verbose=False, vp="[sol2]")
            if s_sol < best_score:
                best_score = s_sol
                best_seq = f_sol.copy()
                print(f"  Soilenka: {best_score:.6f}")

    # Phase 3: Random restarts to escape local minima
    print(f"\n=== Phase 3: Random Restarts (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_r = len(best_seq)
        row_idx_r, col_idx_r = build_conv_structure(n_r)

        for restart in range(10):
            if time.monotonic() > deadline - 20:
                break
            # Perturb
            noise_scale = 0.05 + 0.05 * restart
            noise = rng.normal(0, noise_scale, n_r)
            perturbed = best_seq * (1.0 + noise)
            perturbed = np.clip(perturbed, 0.0, None)
            if np.sum(perturbed) < 1e-10:
                continue
            perturbed = perturbed / np.sum(perturbed) * np.sum(best_seq)

            restart_dl = min(deadline - 15, time.monotonic() + 30.0)
            f_r, s_r = lp_optimize(perturbed, restart_dl, row_idx_r, col_idx_r,
                                   use_topk=25, verbose=False, vp="[restart]")
            if s_r < best_score:
                best_score = s_r
                best_seq = f_r.copy()
                print(f"  Restart {restart}: {best_score:.6f} *BETTER*")
            else:
                print(f"  Restart {restart}: {s_r:.6f}")

    # Phase 4: Try interpolating best to different n values
    print(f"\n=== Phase 4: N-Interpolation Polish (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_base = len(best_seq)
        for n_try in [n_base * 2, n_base // 2]:
            if n_try < 64 or n_try > 8192:
                continue
            if time.monotonic() > deadline - 15:
                break
            x_old = np.linspace(0, 1, n_base)
            x_new = np.linspace(0, 1, n_try)
            interp_seq = np.interp(x_new, x_old, best_seq)
            row_i, col_i = build_conv_structure(n_try)
            dl = min(deadline - 10, time.monotonic() + 40.0)
            f_i, s_i = lp_optimize(interp_seq, dl, row_i, col_i,
                                   use_topk=None, verbose=False, vp=f"[n={n_try}]")
            if s_i < best_score:
                best_score = s_i
                best_seq = f_i.copy()
                print(f"  n={n_try}: {best_score:.6f} *BETTER*")
            else:
                print(f"  n={n_try}: {s_i:.6f}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512)

    # Ensure valid
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if np.sum(best_seq) < 0.01:
        best_seq = np.ones(256)

    final_score = evaluate_sequence([float(x) for x in best_seq])
    if not np.isfinite(final_score):
        print("WARNING: Invalid final sequence, using fallback!")
        best_seq = np.ones(256)
        final_score = evaluate_sequence([float(x) for x in best_seq])

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_seq)}, score={final_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_seq]
