def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:

    b = fast_conv(a)

    return A

def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def smooth_objective_and_grad(x, beta, n):
    """Compute log-sum-exp smoothed objective and its gradient.
    Returns (value, gradient)."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10, np.zeros_like(x)
    b = fast_conv(x_pos)
    b_max = b.max()
    shifted = np.clip(beta * (b - b_max), -200.0, 200.0)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        return 1e10, np.zeros_like(x)
    p = exp_shifted / sum_exp
    LSE = b_max + np.log(sum_exp) / beta

    # Gradient of LSE w.r.t. x_i = 2 * sum_k p_k * x_{k-i}
    # Compute via FFT: c[i] = (p * x_rev)[n-1+i]
    x_rev = x_pos[::-1]
    conv_len = (2 * n - 1) + n - 1
    s_fft = 1
    while s_fft < conv_len + 1:
        s_fft <<= 1
    fp = np.fft.rfft(p, s_fft)
    fxr = np.fft.rfft(x_rev, s_fft)
    p_conv_xr = np.fft.irfft(fp * fxr, s_fft)
    c = p_conv_xr[n - 1:2 * n - 1]

    dLSE_dx = 2.0 * c
    LSE_val = float(LSE)
    S = float(s)
    J = 2.0 * n * LSE_val / (S * S)
    grad = 2.0 * n * (S * dLSE_dx - 2.0 * LSE_val) / (S ** 3)
    # Zero out gradient where x is at boundary
    grad = np.where(x_pos > 0, grad, 0.0)
    return float(J), grad


def smooth_refine(a, deadline, betas=None, maxiter=200):
    """Refine using L-BFGS-B with log-sum-exp, beta annealing."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)

    if betas is None:
        betas = [10, 50, 200, 1000, 5000]

    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = np.clip(best, 1e-8, None)
        x0 = x0 / x0.sum()

        try:
            def obj(jac_x):
                val, grad = smooth_objective_and_grad(jac_x, beta, n)
                if jac_x is None:
                    return val
                return val, grad

            def obj_wrapper(x):
                return smooth_objective_and_grad(x, beta, n)[0]

            def grad_wrapper(x):
                return smooth_objective_and_grad(x, beta, n)[1]

            res = minimize(obj_wrapper, x0, jac=grad_wrapper,
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,

            if res.fun < 1e10:

                    if score < best_score:

def build_conv_matrix(a):
    """Build matrix A where A @ g = full convolve(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M, timeout=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': timeout})
        if res.success and res.x is not None and res.x.sum() > 1e-10:
            return res.x
    except Exception:
        pass
    return None


def lp_iteration(a, max_iters, deadline):
    """LP best-response iteration with adaptive step size."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = np.clip(a, 1e-6, None)
    a = a / a.sum()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, timeout=15)
        if g is None:
            break
        g = g / g.sum()

        improved = False
        for t in [0.5, 0.3, 0.1, 0.8, 0.2]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new / a_new.sum()
            score = compute_score(a_new)
            if score < best_score - 1e-12:
                best_score = score
                best_local = a_new.copy()
                a = a_new
                improved = True
                break

        if not improved:
            break

    return best_local, best_score


def generate_starting_points(n, rng, parent=None):

    a = np.maximum(a, 1e-10)
    points.append((a / a.sum(), "geo"))

    # Sine concentrated at end

    a = np.maximum(a, 1e-10)
    points.append((a / a.sum(), "sine_end"))

    # Triangle peaking at right
    a = np.minimum(t + 1, n - t + 1) * 0.1 + t / n
    a = np.maximum(a, 1e-10)
    points.append((a / a.sum(), "tri_right"))

    # Uniform (baseline)
    points.append((np.ones(n) / n, "uniform"))

    # Resampled parent
    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        a = np.maximum(np.interp(t_new, t_old, parent), 0)
        if a.sum() > 0.01:
            points.append((a / a.sum(), "parent"))

    # Random
    a = rng.uniform(0.1, 1.0, n)
    points.append((a / a.sum(), "random"))

    # Two-segment: low then high
    a = np.concatenate([np.full(n // 2, 0.1), np.full(n - n // 2, 1.0)])
    points.append((a / a.sum(), "step"))

    return points

            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fxr = np.fft.rfft(a_rev, s_fft)
        p_conv_xr = np.fft.irfft(fp * fxr, s_fft)
        c = p_conv_xr[n - 1:2 * n - 1]

        S = a.sum()
        grad = 2.0 * c - 2.0 * b_max / S  # simplified subgradient

        lr = lr_init / (1.0 + 0.01 * step)
        a = a - lr * grad
        a = np.clip(a, 0, None)
        if a.sum() < 0.01:
            a = best.copy()
            continue
        a = a / a.sum()

        if step % 100 == 0:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best = a.copy()

    score = compute_score(a)
    if score < best_score:
        best = a.copy()
        best_score = score
    return best, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: asymmetric starts + L-BFGS-B (log-sum-exp) + LP polish + perturbation.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0

    best = None
    best_score = float('inf')

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            sc = compute_score(p)
            if sc < best_score:
                best = p.copy()
                best_score = sc
                print(f"Parent: J={sc:.6f} (n={p.size})", flush=True)
    except (NameError, AttributeError):
        pass

    # Lengths to try
    lengths = [256, 384, 512, 768]
    if best is not None:
        pl = best.size
        lengths = [l for l in lengths if l == pl] + [l for l in lengths if l != pl]

    # === PHASE 1: Smooth optimization from diverse asymmetric starts ===
    print("\n=== Phase 1: Smooth Optimization (asymmetric starts) ===", flush=True)
    phase1_deadline = time.monotonic() + total_budget * 0.45

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break

        remaining = phase1_deadline - time.monotonic()
        if remaining < 30:
            break

        # Estimate time per L-BFGS-B call
        est_time = max(2.0, 0.003 * n)
        max_starts = max(2, int(remaining / (est_time * 3)))

        inits = generate_starting_points(n, rng, best)

        # Prioritize asymmetric starts
        priority_order = ["linear", "quad", "cubic", "exp_1.0", "exp_2.0",
                         "sqrt", "quart", "sine_end", "tri_right", "step",
                         "parent", "uniform", "random", "geo", "exp_0.5", "exp_3.0"]
        inits_sorted = sorted(
            [(name, a) for a, name in inits],
            key=lambda x: priority_order.index(x[0]) if x[0] in priority_order else 99
        )

        print(f"\n  n={n} ({len(inits_sorted)} starts, ~{est_time:.1f}s each):", flush=True)

        for i, (name, a0) in enumerate(inits_sorted[:max_starts]):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.clip(a0, 1e-8, None)
            a0 = a0 / a0.sum()

            t0 = time.monotonic()
            a_result, score = smooth_refine(a0, phase1_deadline,
                                            betas=[10, 50, 200, 1000],
                                            maxiter=150)
            dt = time.monotonic() - t0

            marker = " *** NEW BEST ***" if score < best_score else ""
            print(f"    {name:12s}: J={score:.6f} (t={dt:.1f}s){marker}", flush=True)

            if score < best_score:
                best = a_result.copy()
                best_score = score

        # Also try subgradient for the best init at this length
        if time.monotonic() < phase1_deadline - 10:
            a0 = np.clip(inits_sorted[0][1], 1e-8, None)
            a0 = a0 / a0.sum()
            a_result, score = subgradient_step(a0, phase1_deadline, n_steps=3000)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"    subgrad {n:4d}: J={score:.6f} *** NEW BEST ***", flush=True)

    # === PHASE 2: LP polishing ===
    print(f"\n=== Phase 2: LP Polishing === (best: {best_score:.6f})", flush=True)
    phase2_deadline = min(time.monotonic() + total_budget * 0.20, deadline)

    if best is not None and time.monotonic() < phase2_deadline:
        n = best.size
        remaining = phase2_deadline - time.monotonic()
        est_iter_time = max(0.03, 0.00005 * n)
        max_iters = max(50, int(remaining * 0.85 / est_iter_time))
        max_iters = min(max_iters, 5000)

        a_result, score = lp_iteration(best, max_iters, phase2_deadline)
        marker = " *** NEW BEST ***" if score < best_score else ""
        print(f"  LP (n={n}): J={score:.6f}{marker}", flush=True)
        if score < best_score:
            best = a_result.copy()
            best_score = score

    # === PHASE 3: Deep smooth refinement with high beta ===
    print(f"\n=== Phase 3: Deep Smooth Refinement === (best: {best_score:.6f})", flush=True)
    phase3_deadline = min(time.monotonic() + total_budget * 0.15, deadline)

    if best is not None and time.monotonic() < phase3_deadline:
        a_result, score = smooth_refine(best, phase3_deadline,
                                        betas=[500, 2000, 10000],
                                        maxiter=100)
        marker = " *** NEW BEST ***" if score < best_score else ""
        print(f"  Deep smooth: J={score:.6f}{marker}", flush=True)
        if score < best_score:
            best = a_result.copy()
            best_score = score

        # LP again after deep smooth
        if time.monotonic() < phase3_deadline - 10:
            a_result, score = lp_iteration(best, 500, phase3_deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  LP2: J={score:.6f} *** NEW BEST ***", flush=True)

    # === PHASE 4: Perturbation + quick optimize ===
    print(f"\n=== Phase 4: Perturbation Search === (best: {best_score:.6f})", flush=True)

    if best is not None and time.monotonic() < deadline:
        n = best.size
        remaining = deadline - time.monotonic()
        n_tries = max(3, int(remaining / 15))

        for trial in range(n_tries):
            if time.monotonic() > deadline:
                break

            scales = [0.03, 0.06, 0.01, 0.1, 0.02, 0.05, 0.08]
            scale = scales[trial % len(scales)]
            noise = rng.normal(0, scale, n)
            a_pert = np.clip(best + noise, 0, None)
            if a_pert.sum() < 0.01:
                noise = rng.uniform(0, scale * 2, n)
                a_pert = np.clip(best * (1 + (noise - scale)), 0, None)
            if a_pert.sum() < 0.01:
                continue
            a_pert = a_pert / a_pert.sum()

            sub_deadline = min(deadline, time.monotonic() + remaining / max(n_tries - trial, 1) - 2)
            a_result, score = smooth_refine(a_pert, sub_deadline,
                                            betas=[100, 500], maxiter=80)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  pert {trial}: J={score:.6f} *** NEW BEST ***", flush=True)

    # === PHASE 5: Try other lengths with best shape ===
    if best is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Phase 5: Cross-length exploration === (best: {best_score:.6f})", flush=True)
        n_best = best.size
        other_lengths = [l for l in [192, 320, 448, 640, 896] if l != n_best]
        for n2 in other_lengths[:3]:
            if time.monotonic() > deadline - 30:
                break
            t_old = np.linspace(0, 1, n_best)
            t_new = np.linspace(0, 1, n2)
            a2 = np.maximum(np.interp(t_new, t_old, best), 0)
            if a2.sum() < 0.01:
                continue
            a2 = a2 / a2.sum()
            a_result, score = smooth_refine(a2, deadline - 30,
                                            betas=[50, 500], maxiter=100)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  cross n={n2}: J={score:.6f} *** NEW BEST ***", flush=True)

    # Final: validate and return
    if best is None:
        best = np.ones(256) / 256
        best_score = compute_score(best)

    final_seq = [float(x) for x in best]
    try:
        final_score = float(evaluate_sequence(final_seq))
        if np.isfinite(final_score):
            best_score = min(best_score, final_score)
    except Exception:
        pass

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return final_seq
