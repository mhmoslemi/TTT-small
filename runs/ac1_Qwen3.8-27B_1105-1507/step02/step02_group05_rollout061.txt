```python
def minimize_1d(f, lo, hi, iters=50):
    """Minimize f on [lo, hi] using golden section search."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in iters:
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2
```

```python
def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * len(a) * b.max() / (S * S))


def coord_descent(a, n_sweeps, rng, refine=True):
    """
    Coordinate descent minimizing max(conv(a,a)) subject to a>=0, sum(a)=1.
    Uses incremental convolution updates for O(n) per coordinate.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    n = len(a)
    b = fast_conv(a)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for m in order:
            x0 = a[m]
            if x0 < 1e-15:
                continue

            # B[k] = 2*a[k-m] for k in [m, min(2n-2, m+n-1)]
            k_lo = m
            k_hi = min(2 * n - 2, m + n - 1)
            B_slice = 2.0 * a[0:k_hi - m + 1]

            # C: 1 at position k=2m if in range
            k_2m = 2 * m
            has_C = (0 <= k_2m <= 2 * n - 2)
            c_pos = k_2m - k_lo if has_C else -1

            # Unchanged max (precompute once)
            if k_lo > 0:
                ul = b[:k_lo].max()
            else:
                ul = 0.0
            if k_hi < 2 * n - 2:
                ur = b[k_hi + 1:].max()
            else:
                ur = 0.0
            unchanged_max = max(ul, ur)

            # Try candidates
            candidates = [0.0, x0, 0.5 * x0, 2.0 * x0]
            # Add a candidate based on the "balanced" value
            # If the dominant k has large B, try reducing x to counteract
            b_at_range = b[k_lo:k_hi + 1]
            if b_at_range.max() > unchanged_max:
                # The max is in the changed range; try reducing
                candidates.append(0.3 * x0)
                candidates.append(0.7 * x0)

            best_x = x0
            best_val = float('inf')

            for x_trial in candidates:
                if x_trial < 0:
                    continue
                u = x_trial - x0
                S_new = S + u
                if S_new < 1e-10:
                    continue

                # b_new in changed range
                bc = b[k_lo:k_hi + 1].copy() + B_slice * u
                if has_C and c_pos >= 0 and c_pos < len(bc):
                    bc[c_pos] += u * u
                changed_max = bc.max()
                total_max = max(changed_max, unchanged_max)
                val = total_max / (S_new * S_new)

                if val < best_val:
                    best_val = val
                    best_x = x_trial

            # Optional: refine with golden section around best candidate
            if refine and best_x > 0:
                lo = max(0.0, best_x * 0.3)
                hi = best_x * 3.0
                # Quick refinement with 10 iterations
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c = hi - gr * (hi - lo)
                d = lo + gr * (hi - lo)

                def eval_f(xv):
                    uu = xv - x0
                    Sn = S + uu
                    if Sn < 1e-10:
                        return 1e10
                    bc = b[k_lo:k_hi + 1] + B_slice * uu
                    if has_C and c_pos >= 0 and c_pos < len(bc):
                        bc = bc.copy()
                        bc[c_pos] += uu * uu
                    cm = bc.max()
                    return max(cm, unchanged_max) / (Sn * Sn)

                fc = eval_f(c)
                fd = eval_f(d)
                for _ in range(12):
                    if fc < fd:
                        hi = d
                        d = c
                        fd = fc
                        c = hi - gr * (hi - lo)
                        fc = eval_f(c)
                    else:
                        lo = c
                        c = d
                        fc = fd
                        d = lo + gr * (hi - lo)
                        fd = eval_f(d)
                x_mid = (lo + hi) / 2.0
                f_mid = eval_f(x_mid)
                if f_mid < best_val:
                    best_x = x_mid
                    best_val = f_mid

            # Apply update
            a[m] = best_x
            u = best_x - x0
            S += u
            if abs(u) > 1e-15:
                b[k_lo:k_hi + 1] += B_slice * u
                if has_C and (0 <= k_2m <= 2 * n - 2):
                    b[k_2m] += u * u

        # Renormalize periodically
        S = a.sum()
        if S > 0 and abs(S - 1.0) > 1e-6:
            a = a / S
            S = 1.0
            b = fast_conv(a)  # Recompute to fix drift

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        t_old = np.linspace(0, 1, len(parent))
        t_new = np.linspace(0, 1, n)
        inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.001))

    # Uniform
    inits.append(np.ones(n))

    # Random
    inits.append(rng.uniform(0.2, 1.0, n))

    # Cosine patterns
    for freq in [1, 2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))

    # Anti-symmetric trick: high on first half, low on second
    half = n // 2
    anti = np.concatenate([1.0 + 0.3 * rng.standard_normal(half),
                           0.3 + 0.1 * rng.standard_normal(n - half)])
    inits.append(np.maximum(anti, 0.01))

    # Gaussian bell
    inits.append(np.exp(-5.0 * (t - 0.3) ** 2) + 0.05)
    inits.append(np.exp(-5.0 * (t - 0.7) ** 2) + 0.05)

    # Sine |sin| pattern
    inits.append(np.abs(np.sin(2 * np.pi * t * 3)) + 0.1)

    # Power law
    inits.append((t + 0.05) ** rng.uniform(-2, 2))

    # Sparse spikes (smoothed)
    sparse = np.zeros(n)
    n_sp = max(4, n // 16)
    pos = rng.choice(n, size=n_sp, replace=False)
    sparse[pos] = 1.0
    kernel_size = 3
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Alternating pattern with noise
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.3)
    inits.append(alt + 0.1 * rng.standard_normal(n))

    # Two-scale pattern
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.cos(2 * np.pi * 7 * t))

    # Random smooth (low-pass filtered random)
    raw = rng.standard_normal(n)
    freqs = np.fft.rfftfreq(n)
    cutoff = n // 4
    mask = np.ones_like(freqs)
    mask[cutoff:] = 0
    filtered = np.fft.irfft(np.fft.rfft(raw) * mask)
    inits.append(1.0 + 0.5 * filtered / (np.abs(filtered).max() + 1e-10))

    for i in range(len(inits)):
        inits[i] = np.maximum(inits[i].astype(np.float64), 0.001)
        inits[i] /= inits[i].sum()

    return inits

            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Scale sweeps by length
        sweeps = max(50, min(500, int(200000 // n)))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a = coord_descent(a0, sweeps, rng, refine=False)
            update_tracker(a)

        # Second round with refinement
        if time.monotonic() < deadline and best_seq is not None:
            # Resample best to this length and refine
            t_old = np.linspace(0, 1, best_seq.size)
            t_new = np.linspace(0, 1, n)
            a0 = np.maximum(np.interp(t_new, t_old, best_seq), 0.001)
            a0 /= a0.sum()
            sweeps2 = max(50, min(300, int(100000 // n)))
            a = coord_descent(a0, sweeps2, rng, refine=True)
            update_tracker(a)

        elapsed = time.monotonic() - t0
        print(f"  Phase1 n={n}: {elapsed:.1f}s, best J={best_score:.8f}", flush=True)

    # === Phase 2: Perturbation restarts on best ===
    if best_seq is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = best_seq.size
        print(f"\n{'='*50} Phase2: Perturbation n={n} {'='*50}", flush=True)

        n_restarts = 40
        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            temp = 0.1 * (1.0 - r / float(n_restarts)) + 0.02
            a = best_seq.copy() + rng.normal(0, temp, n)
            a = np.maximum(a, 0.001)
            a /= a.sum()
            sweeps = max(30, min(100, 200 - r * 3))
            a = coord_descent(a, sweeps, rng, refine=(r < 10))
            update_tracker(a)

        elapsed = time.monotonic() - t0
        print(f"  Phase2: {elapsed:.1f}s, best J={best_score:.8f}", flush=True)

    # === Phase 3: Try different lengths by resampling ===
    if best_seq is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase3: Length adaptation {'='*50}", flush=True)

        for n_try in [512, 1024, 1536, 2048, 3072]:
            if time.monotonic() > deadline:
                break
            if n_try == best_seq.size:
                continue
            t_old = np.linspace(0, 1, best_seq.size)
            t_new = np.linspace(0, 1, n_try)
            a0 = np.maximum(np.interp(t_new, t_old, best_seq), 0.001)
            a0 /= a0.sum()
            sweeps = max(30, min(200, int(80000 // n_try)))
            a = coord_descent(a0, sweeps, rng, refine=True)
            update_tracker(a)

        elapsed = time.monotonic() - t0
        print(f"  Phase3: {elapsed:.1f}s, best J={best_score:.8f}", flush=True)

    # === Phase 4: L-BFGS-B polish ===
    if best_seq is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = best_seq.size
        print(f"\n{'='*50} Phase4: L-BFGS-B n={n} {'='*50}", flush=True)

        if n <= 2048:
            def obj_and_grad(x):
                ax = np.maximum(x, 0.0)
                S = ax.sum()
                if S < 1e-8:
                    return 1e10, np.zeros_like(x)
                ax = ax / S
                b = fast_conv(ax)
                kmax = int(np.argmax(b))
                val = b[kmax]
                # Gradient of b[kmax] w.r.t. ax
                grad_b = np.zeros(n)
                for i in range(max(0, kmax - n + 1), min(n, kmax + 1)):
                    j = kmax - i
                    if 0 <= j < n:
                        grad_b[i] += ax[j]
                        if j != i:
                            grad_b[j] += ax[i]
                # Chain rule: d(b_kmax)/d(x_i) / S - b_kmax * x_i / S^2
                grad = grad_b / S - val * ax / (S * S)
                return val, grad

            bounds_list = [(0.0, None)] * n
            try:
                res = minimize(obj_and_grad, best_seq.copy(), method='L-BFGS-B',
                               bounds=bounds_list, jac=True,
                               options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-14})
                if res.fun < 1e10:
                    ax = np.maximum(res.x, 0.0)
                    if ax.sum() > 0.01:
                        update_tracker(ax)
            except Exception as e:
                print(f"  L-BFGS-B error: {e}", flush=True)

        elapsed = time.monotonic() - t0
        print(f"  Phase4: {elapsed:.1f}s, best J={best_score:.8f}", flush=True)

    # === Phase 5: Final intensive coordinate descent ===
    if best_seq is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = best_seq.size
        print(f"\n{'='*50} Phase5: Final CD n={n} {'='*50}", flush=True)

        remaining = deadline - time.monotonic()
        # Estimate sweeps we can afford: ~20ms per sweep for n=1024
        est_sweep_time = 0.02 * (n / 1024.0)
        n_sweeps = int(remaining / (est_sweep_time + 0.001)) - 5
        n_sweeps = max(20, min(2000, n_sweeps))
        print(f"  Running {n_sweeps} sweeps", flush=True)

        # Run in chunks to check time
        chunk = max(1, n_sweeps // 4)
        a = best_seq.copy()
        for i in range(4):
            if time.monotonic() > deadline:
                break
            a = coord_descent(a, chunk, rng, refine=True)
            update_tracker(a)

        elapsed = time.monotonic() - t0
        print(f"  Phase5: {elapsed:.1f}s, best J={best_score:.8f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = compute_J(best_seq)
        print(f"Fallback: J={best_score:.8f}", flush=True)

    total_time = time.monotonic() - t_start
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={best_score:.8f} n={best_seq.size} time={total_time:.0f}s", flush=True)

    return [float(x) for x in best_seq]
```
