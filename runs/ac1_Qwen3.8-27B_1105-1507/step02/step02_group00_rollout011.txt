def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf

    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp_g(a, M):
    """Solve LP: max sum(g) s.t. A@G <= M*1, g >= 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:

    return None

def compute_obj(a):

            #                = base_conv[2j] + x'^2 for k = 2j

            # Bisection on M: find smallest M such that exists x' >= 0 with f_k(x') <= M for all k.

            # For k != 2j (and 0 <= k-j < n): x' <= (M - base_conv[k]) / (2*a[k-j]) if a[k-j] > 0
            # For k = 2j: x' <= sqrt(max(0, M - base_conv[2j]))
            # Also x' >= 0.

            # Upper bound on x': let's try a range [0, x_max]
            # x_max: try up to 3*current value or up to 1 (whichever is reasonable)
            x_max = max(x * 3, 0.1, 1.0 / n * 10)

            # Bisection on M in [current max, upper bound]
            current_max = float(np.max(base_conv)) + 2.0 * x * (float(np.max(a)) if n > 0 else 0)
            # Actually let's just compute the actual current max
            a[j] = x
            actual_max = float(np.max(np.convolve(a, a)))
            a[j] = 0.0

            lo = max(float(np.max(base_conv)), 0.0)
            hi = actual_max  # current value as upper bound

            # We want to find the minimum achievable max
            # Bisection: for a given M, check feasibility
            best_x = x
            best_M = actual_max

            for _ in range(40):
                mid = (lo + hi) / 2.0
                # Check if there exists x' in [0, x_max] with all f_k(x') <= mid
                # Constraints on x':
                upper = x_max
                # From k = 2j: x'^2 <= mid - base_conv[2j]
                if 2 * j < 2 * n - 1:
                    bound_2j = mid - base_conv[2 * j]
                    if bound_2j < 0:
                        upper = -1  # infeasible
                    else:
                        upper = min(upper, np.sqrt(bound_2j))

                # From other k: x' <= (mid - base_conv[k]) / (2*a[k-j])
                # We only need to check k where a[k-j] > 0
                for k in range(2 * n - 1):
                    if k == 2 * j:
                        continue
                    kj = k - j
                    if 0 <= kj < n and a[kj] > 1e-15:
                        bound_k = (mid - base_conv[k]) / (2.0 * a[kj])
                        upper = min(upper, bound_k)

                # Also x' >= 0
                if upper >= 0.0:
                    # Feasible with x' = 0 (worst case) or any x' in [0, upper]
                    # The tightest constraint determines the max achievable M
                    hi = mid
                    best_x = 0.0  # x'=0 is always feasible when upper >= 0
                    # Actually we want to find the optimal x', not just feasibility
                    # For now, just track the minimum M
                else:
                    lo = mid

            # After bisection, hi is approximately the minimum achievable max
            # Now find the optimal x' that achieves this
            # Try x' = 0 and x' = hi/2 * some factor
            # Actually, let's just try several values

            a[j] = 0.0
            candidates = [0.0, x, x * 0.5, x * 1.5, hi * 0.1, min(x_max, hi * 0.5)]
            for xc in candidates:
                if xc < 0:
                    continue
                a[j] = xc
                obj_c = compute_obj(a)
                if obj_c < best_obj:
                    best_obj = obj_c
                    best_x = xc

            a[j] = best_x
            a = a / np.sum(a)

        # After full round, update best
        obj_round = compute_obj(a)
        if obj_round < best_obj:
            best_obj = obj_round
            best_a = a.copy()

        if verbose:
            print(f"    Coord rnd={rnd}: obj={best_obj:.6f}")

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 7))
    if choice == 0:
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif choice == 1:
        mid = n // 2
        a = np.ones(n)
        a[:mid] = rng.uniform(0.3, 1.5)
        a[mid:] = rng.uniform(0.3, 1.5)
    elif choice == 2:
        period = int(rng.integers(max(2, n // 30), max(3, n // 5)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        a = np.exp(-rng.uniform(0, 3) * (np.arange(n) / n - 0.5) ** 2)
    elif choice == 4:
        a = np.abs(np.sin(np.arange(n) * rng.uniform(0.1, 1.5))) + 0.1
    elif choice == 5:
        # Smooth random with low-pass
        a = rng.random(n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, 'same')
    else:
        a = rng.random(n)

    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search with corrected subgradient, LP alternation,
    momentum polishing, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            best_score = obj
            best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Primary n values
    n_values = [200, 250, 300, 350, 400, 500]

    # Phase 1: LP optimization + subgradient polish for multiple n
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        print(f"N={n}: {remaining:.0f}s remaining, best={best_score:.6f}")

        n_restarts = 4 if n <= 250 else 2

        for restart in range(n_restarts):
            if time.monotonic() > deadline:
                break

            # Initialize
            if restart == 0 and best_seq is not None and len(best_seq) == n:
                a_init = best_seq.copy()
            elif restart == 1 and best_seq is not None:
                # Interpolate best to this n
                if len(best_seq) != n:
                    from scipy.interpolate import interp1d
                    old_x = np.linspace(0, 1, len(best_seq))
                    new_x = np.linspace(0, 1, n)
                    f = interp1d(old_x, best_seq, kind='cubic', fill_value='extrapolate')
                    a_init = np.maximum(f(new_x), 0)
                    a_init = a_init / np.sum(a_init)
                else:
                    a_init = best_seq.copy()
            else:
                a_init = structured_init(n, rng)

            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # LP alternation
            max_lp_iter = int(min(300, max(50, remaining / 3.0)))
            a1, obj1 = lp_alternate(a_init, max_iter=max_lp_iter,
                                    verbose=(restart == 0 and n == 300))
            if obj1 < best_score:
                best_score = obj1
                best_seq = a1.copy()
                print(f"  LP n={n} r={restart}: obj={obj1:.6f}")

            # Subgradient polish with momentum (FIXED subgradient)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                ns = int(min(8000, remaining / 0.008))
                if ns > 100:
                    a2, obj2 = subgrad_polish(a1, n_steps=ns,
                                              momentum_beta=0.9,
                                              verbose=(restart == 0 and n == 300))
                    if obj2 < best_score:
                        best_score = obj2
                        best_seq = a2.copy()
                        print(f"  SG n={n} r={restart}: obj={obj2:.6f}")

                    # Second LP round on polished solution
                    remaining = deadline - time.monotonic()
                    if remaining > 15:
                        mx2 = int(min(150, remaining / 0.05))
                        if mx2 > 10:
                            a3, obj3 = lp_alternate(a2, max_iter=mx2)
                            if obj3 < best_score:
                                best_score = obj3
                                best_seq = a3.copy()
                                print(f"  LP2 n={n} r={restart}: obj={obj3:.6f}")

    # Phase 2: Final refinement on best solution
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"Phase 2: n={n_best}, obj={best_score:.6f}, {remaining:.0f}s left")

        # Extended subgradient
        if n_best <= 500:
            ns = int(min(10000, remaining / 0.01))
            if ns > 500:
                a4, obj4 = subgrad_polish(best_seq, n_steps=ns, momentum_beta=0.95)
                if obj4 < best_score:
                    best_score = obj4
                    best_seq = a4.copy()
                    print(f"  Final SG: obj={obj4:.6f}")

        # Coordinate descent (exact 1D minimax per coordinate)
        remaining = deadline - time.monotonic()
        if remaining > 10 and n_best <= 200:
            rounds = int(min(3, remaining / (n_best * 0.05)))
            if rounds > 0:
                a5, obj5 = coord_descent(best_seq, n_rounds=rounds, verbose=True)
                if obj5 < best_score:
                    best_score = obj5
                    best_seq = a5.copy()
                    print(f"  Final Coord: obj={obj5:.6f}")

        # One more LP + SG cycle
        remaining = deadline - time.monotonic()
        if remaining > 20 and n_best <= 400:
            mx = int(min(100, remaining / 0.1))
            if mx > 10:
                a6, obj6 = lp_alternate(best_seq, max_iter=mx)
                if obj6 < best_score:
                    best_score = obj6
                    best_seq = a6.copy()
                    print(f"  Final LP: obj={obj6:.6f}")

                remaining = deadline - time.monotonic()
                if remaining > 10:
                    ns2 = int(min(5000, remaining / 0.01))
                    if ns2 > 100:
                        a7, obj7 = subgrad_polish(a6, n_steps=ns2, momentum_beta=0.95)
                        if obj7 < best_score:
                            best_score = obj7
                            best_seq = a7.copy()
                            print(f"  Final SG2: obj={obj7:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    # Final validation
    final_obj = compute_obj(best_seq)
    if final_obj != best_score:
        best_score = final_obj

    elapsed = time.monotonic() - start_time
    print(f"DONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
