```python
"""
Nesterov-accelerated projected subgradient optimization with exact subgradient,
active set tracking, and multiplicative updates for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiple initialization strategies and adaptive step sizes.
"""

import numpy as np
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def subgradient_from_active(a, conv, k_active):
    """Compute exact subgradient using active set of lags.

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the standard algorithm
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Nesterov-accelerated subgradient optimization with exact subgradient, Frank-Wolfe steps,
and multiplicative updates for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines multiple optimization strategies with adaptive active-set tracking and
diverse initialization to escape local minima.
"""

import numpy as np
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = v.copy()
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_topk(a, conv, top_k):
    """Compute exact subgradient using top-k active lags by softmax weights."""
    n = len(a)
    conv_len = len(conv)
    M = np.max(conv)
    # Softmax weights with temperature
    T = max(0.01 * (M - np.min(conv)), 1e-10)
    shifted = (conv - M) / T
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros(conv_len)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum
    # Now compute g[j] = 2 * sum_k weights[k] * a[k-j] for valid k-j
    # Using correlate: g[j] = 2 * np.correlate(weights, a, 'full')[j]
    if n <= 2000:
        corr = np.correlate(weights, a, mode='full')
        g = 2.0 * corr[:n]
    else:
        # FFT-based
        size = 1
        while size < conv_len + n - 1:
            size <<= 1
        fw = np.fft.rfft(weights, size)
        fa = np.fft.rfft(a, size)
        corr = np.fft.irfft(fw * np.conj(fa), size)
        g = 2.0 * corr[:n]
    return g


def subgradient_soft_weighted(a, conv, tau):
    """Compute smoothed subgradient using exponential weights (log-sum-exp smoothing)."""
    n = len(a)
    conv_len = len(conv)
    M = np.max(conv)
    shifted = tau * (conv - M)
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros(conv_len)
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= wsum
    if n <= 2000:
        corr = np.correlate(weights, a, mode='full')
        g = 2.0 * corr[:n]
    else:
        size = 1
        while size < conv_len + n - 1:
            size <<= 1
        fw = np.fft.rfft(weights, size)
        fa = np.fft.rfft(a, size)
        corr = np.fft.irfft(fw * np.conj(fa), size)
        g = 2.0 * corr[:n]
    return g


def nesterov_optimize(a, max_iters, tau_start=2.0, tau_end=50.0,
                      lr_init=0.1, momentum=0.9, verbose=False, time_limit=None):
    """Nesterov-accelerated projected subgradient with annealed smoothing."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    y = a.copy()
    v = np.zeros(n)
    beta = momentum

    best_a = a.copy()
    best_obj = compute_obj(a)
    lr = lr_init
    stall = 0
    patience = 300

    start_t = time.monotonic() if time_limit else None

    for it in range(max_iters):
        if time_limit and (time.monotonic() - start_t > time_limit):
            break

        # Nesterov look-ahead
        y = a + beta * v
        y = project_simplex(y)

        # Compute objective
        conv = compute_conv(y)
        S = float(np.sum(y))
        M = float(np.max(conv))
        obj = 2.0 * n * M / (S * S) if S > 1e-10 else np.inf

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                beta = max(0.5, beta * 0.8)
                lr *= 0.5
                v *= 0.5
                if lr < 1e-12:
                    break
                stall = 0
                patience = max(100, patience // 2)

        # Anneal tau
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * frac

        # Compute smoothed gradient
        g = subgradient_soft_weighted(y, conv, tau)
        # Center for sum constraint
        g = g - np.mean(g)

        # Nesterov update
        v = beta * v - lr * g
        a_new = project_simplex(y + v)
        a = a_new

        # Adaptive learning rate
        if it < 50:
            lr *= 0.99
        elif it < 500:
            lr *= 0.998
        else:
            lr *= 0.9995

    return best_a, best_obj


def multiplicative_optimize(a, max_iters, tau=10.0, lr_init=0.1,
                            verbose=False, time_limit=None):
    """Multiplicative update optimization (naturally maintains non-negativity)."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())

    best_a = a.copy()
    best_obj = compute_obj(a)
    lr = lr_init
    stall = 0
    patience = 200

    start_t = time.monotonic() if time_limit else None

    for it in range(max_iters):
        if time_limit and (time.monotonic() - start_t > time_limit):
            break

        conv = compute_conv(a)
        S = float(np.sum(a))
        M = float(np.max(conv))
        obj = 2.0 * n * M / (S * S) if S > 1e-10 else np.inf

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                lr *= 0.6
                if lr < 1e-12:
                    break
                # Restart with noise
                noise = np.random.default_rng(it + 12345).normal(0, 0.02, n)
                a = a * np.exp(noise)
                a = project_simplex(a)
                stall = 0
                patience = max(50, patience // 2)

        g = subgradient_soft_weighted(a, conv, tau)
        g = g - np.mean(g)

        # Multiplicative update
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s
        a = a_new

        # Decay
        if it > 100:
            lr *= 0.999
        else:
            lr *= 0.995

    return best_a, best_obj


def frank_wolfe_optimize(a, max_iters, tau=5.0, lr_init=None,
                         verbose=False, time_limit=None):
    """Frank-Wolfe (conditional gradient) method on the simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())

    best_a = a.copy()
    best_obj = compute_obj(a)
    stall = 0
    patience = 500

    if lr_init is None:
        lr_init = 2.0 / 3.0  # Default FW learning rate

    start_t = time.monotonic() if time_limit else None

    for it in range(max_iters):
        if time_limit and (time.monotonic() - start_t > time_limit):
            break

        conv = compute_conv(a)
        S = float(np.sum(a))
        M = float(np.max(conv))
        obj = 2.0 * n * M / (S * S) if S > 1e-10 else np.inf

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Perturb and reset
                rng = np.random.default_rng(it + 99999)
                a = a + rng.normal(0, 0.01, n)
                a = project_simplex(a)
                stall = 0
                patience = max(200, patience // 2)

        g = subgradient_soft_weighted(a, conv, tau)
        # Center gradient
        g = g - np.mean(g)

        # FW step: find s in simplex minimizing <g, s>
        # Optimal s is e_{j*} where j* = argmin_j g[j]
        j_star = int(np.argmin(g))
        s_vec = np.zeros(n)
        s_vec[j_star] = 1.0

        # FW learning rate
        gamma = 2.0 / (it + 2.0)  # Standard FW rate

        a_new = (1.0 - gamma) * a + gamma * s_vec
        a = a_new

    return best_a, best_obj


def hybrid_optimize(a, time_budget, verbose=False):
    """Hybrid optimization combining Nesterov, multiplicative, and FW methods."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    deadline = start + time_budget

    # Phase 1: Nesterov subgradient (50% of time)
    t1 = (start + time_budget * 0.45 - start)
    a, obj = nesterov_optimize(a, max_iters=200000, tau_start=3.0, tau_end=60.0,
                               lr_init=0.08, momentum=0.9, time_limit=t1)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
        if verbose:
            print(f"    Phase 1 (Nesterov): obj={obj:.6f}")

    # Phase 2: Multiplicative updates (25% of time)
    remaining = deadline - time.monotonic()
    if remaining > 5:
        t2 = remaining * 0.3
        a, obj = multiplicative_optimize(a, max_iters=100000, tau=15.0,
                                         lr_init=0.05, time_limit=t2)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    Phase 2 (Multiplicative): obj={obj:.6f}")

    # Phase 3: Nesterov with higher tau (focus on peak) (15% of time)
    remaining = deadline - time.monotonic()
    if remaining > 5:
        t3 = remaining * 0.3
        a, obj = nesterov_optimize(a, max_iters=200000, tau_start=20.0, tau_end=80.0,
                                   lr_init=0.03, momentum=0.95, time_limit=t3)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    Phase 3 (Nesterov hi-tau): obj={obj:.6f}")

    # Phase 4: Frank-Wolfe (10% of time)
    remaining = deadline - time.monotonic()
    if remaining > 10 and n <= 800:
        t4 = remaining * 0.5
        a, obj = frank_wolfe_optimize(a, max_iters=50000, tau=20.0, time_limit=t4)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    Phase 4 (FW): obj={obj:.6f}")

    # Phase 5: Final Nesterov polish
    remaining = deadline - time.monotonic()
    if remaining > 5:
        a, obj = nesterov_optimize(a, max_iters=200000, tau_start=40.0, tau_end=120.0,
                                   lr_init=0.02, momentum=0.95, time_limit=remaining)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def generate_initialization(n, init_type, rng):
    """Generate an initial sequence of length n."""
    if init_type == 0:
        # Cosine
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n + phase))
    elif init_type == 1:
        # Gaussian centered
        center = n / 2.0
        rate = rng.uniform(1.0, 4.0)
        a = np.exp(-rate * (np.arange(n) - center) ** 2 / (n * n))
    elif init_type == 2:
        # Random
        a = rng.random(n)
    elif init_type == 3:
        # Multi-peak cosine
        freq = rng.uniform(1.0, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.abs(np.cos(np.pi * freq * np.arange(n) / n + phase))
    elif init_type == 4:
        # Binary-like
        a = (rng.random(n) > 0.3).astype(float)
        a += rng.random(n) * 0.3
    elif init_type == 5:
        # Linear ramp with noise
        a = np.linspace(0.1, 1.0, n)
        a += rng.normal(0, 0.1, n)
    elif init_type == 6:
        # Two-peak
        c1, c2 = sorted(rng.choice(range(1, n-1), 2, replace=False))
        a = np.exp(-2.0 * (np.arange(n) - c1) / n) + np.exp(-2.0 * (np.arange(n) - c2) / n)
    else:
        a = np.ones(n)
    return np.maximum(a, 0)


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Hybrid Nesterov + multiplicative + Frank-Wolfe optimization.
    Explores sequence lengths 50-800 with diverse initializations.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Define candidate (n, init_type) pairs to explore
    candidates = []

    # If parent exists, explore around its length
    if parent is not None:
        n_parent = len(parent)
        n_values = sorted(set([
            max(30, n_parent - 50), max(30, n_parent - 20), n_parent,
            n_parent + 20, n_parent + 50, n_parent + 100
        ]))
    else:
        n_values = [50, 100, 150, 200, 250, 300, 400, 500]

    # Add diverse lengths
    for n in [50, 100, 150, 200, 250, 300, 400, 500, 700]:
        if n not in n_values:
            n_values.append(n)
    n_values = sorted(set(n_values))

    print(f"Exploring n values: {n_values}")

    # Quick screen phase: evaluate multiple initializations quickly
    screen_candidates = []
    screen_time = min(120.0, (deadline - start_time) * 0.15)
    screen_deadline = start_time + screen_time

    for n in n_values:
        if time.monotonic() > screen_deadline:
            break
        for init_type in range(7):
            if time.monotonic() > screen_deadline:
                break
            a = generate_initialization(n, init_type, rng)
            a = project_simplex(a)
            obj = compute_obj(a)
            screen_candidates.append((obj, n, init_type, a.copy()))

    # Sort by objective and keep top ones
    screen_candidates.sort(key=lambda x: x[0])
    print(f"Screened {len(screen_candidates)} candidates, best screen obj: {screen_candidates[0][0]:.6f}")

    # Also add parent to the list
    if parent is not None:
        screen_candidates.append((parent_obj, len(parent), -1, parent.copy()))
        screen_candidates.sort(key=lambda x: x[0])

    # Deep optimization phase: work on top candidates
    top_k_deep = min(5, len(screen_candidates))
    deep_budget = (deadline - time.monotonic())
    per_candidate = deep_budget / (top_k_deep + 2)  # +2 for final polish and exploration

    for rank in range(top_k_deep):
        if time.monotonic() > deadline - 60:
            break
        obj_init, n, init_type, a_init = screen_candidates[rank]
        remaining = deadline - time.monotonic() - 80  # Reserve 80s for polish
        alloc = min(per_candidate, remaining)
        if alloc < 15:
            break

        verb = rank < 2
        print(f"\nDeep optimize candidate {rank}: n={n}, init={init_type}, "
              f"obj={obj_init:.6f}, budget={alloc:.0f}s")

        a_opt, obj_opt = hybrid_optimize(a_init, time_budget=alloc, verbose=verb)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  *** UPDATE: n={n}, obj={best_score:.6f}")
        else:
            print(f"  No improvement (obj={obj_opt:.6f} >= best={best_score:.6f})")

    # Final intensive polish on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Final Polish: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        polish_round = 0

        while time.monotonic() < deadline - 10:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # Subgradient polish with very high tau (extreme peak focus)
            half_time = remaining / 2.5
            a_cur, obj = nesterov_optimize(a_cur, max_iters=500000,
                                           tau_start=30.0, tau_end=150.0,
                                           lr_init=0.02, momentum=0.95,
                                           time_limit=half_time)
            if obj < best_score:
                best_score = obj
                best_seq = a_cur.copy()
                print(f"  Polish {polish_round} Nesterov: obj={best_score:.6f}")

            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # Multiplicative polish
            half_time = remaining / 2.5
            a_mp, obj_mp = multiplicative_optimize(a_best_round := best_seq.copy(),
                                                    max_iters=200000, tau=30.0,
                                                    lr_init=0.03, time_limit=half_time)
            if obj_mp < best_score:
                best_score = obj_mp
                best_seq = a_mp.copy()
                a_cur = a_mp.copy()
                print(f"  Polish {polish_round} Multi: obj={best_score:.6f}")

            # Random perturbation and quick re-optimization
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            noise_scale = 0.03
            a_pert = best_seq * np.exp(rng.normal(0, noise_scale, n_best))
            a_pert = project_simplex(a_pert)
            obj_p = compute_obj(a_pert)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_pert.copy()
                print(f"  Polish {polish_round} Perturb: obj={best_score:.6f}")

            # Quick Nesterov from perturbed
            half_time = min(remaining / 3, 30.0)
            if half_time > 5:
                a_p2, obj_p2 = nesterov_optimize(a_pert, max_iters=100000,
                                                  tau_start=10.0, tau_end=50.0,
                                                  lr_init=0.05, momentum=0.9,
                                                  time_limit=half_time)
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    print(f"  Polish {polish_round} Perturb+Nesterov: obj={best_score:.6f}")

    # Try a few different lengths from the best shape (stretching/compressing)
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Length exploration: {remaining:.0f}s remaining ===")

        for scale in [0.8, 1.2, 1.5]:
            if time.monotonic() > deadline - 15:
                break
            n_new = int(round(n_best * scale))
            n_new = max(30, min(1000, n_new))
            if n_new == n_best:
                continue

            # Interpolate the best sequence to new length
            old_t = np.linspace(0, 1, n_best)
            new_t = np.linspace(0, 1, n_new)
            a_interp = np.interp(new_t, old_t, best_seq)
            a_interp = project_simplex(a_interp)

            alloc = min(remaining / 3, 60.0)
            if alloc < 10:
                continue

            a_opt, obj_opt = hybrid_optimize(a_interp, time_budget=alloc, verbose=False)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Length {n_new}: obj={best_score:.6f}")
            else:
                print(f"  Length {n_new}: obj={obj_opt:.6f} (no improvement)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation != np.inf and abs(validation - best_score) > 1e-6:
        print(f"  Validation: computed={best_score:.6f}, validated={validation:.6f}")
        if validation < best_score:
            best_score = validation

    return result
```
