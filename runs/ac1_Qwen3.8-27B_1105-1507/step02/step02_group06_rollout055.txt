```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import math

def compute_score(f):
    """Compute the objective value directly."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))

def solve_lp(f, S):
    """Solve: min t s.t. conv(f,g)[k] <= t, sum(g) = S, g >= 0."""
    n = len(f)
    # Build constraint matrix: A[k,j] = f[k-j] if 0 <= k-j < n
    # Shape: (2n-1, n)
    # Plus variable t: total n+1 variables
    
    # For efficiency, use the Toeplitz structure
    # A @ g = convolve(f, g)
    
    # Build A as dense (2n-1 x n) - for n=1024, this is 2047x1024 ≈ 2M entries
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k, j] = f[k-j] for k = j, j+1, ..., j+n-1
        A[j:j+n, j] = f  # f[0] to f[n-1] placed at rows j to j+n-1
    
    # Full constraint matrix: [A, -ones(2n-1,1)]
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                        bounds=bounds, method='highs', options={'presolve': True})
        if result.success:
            return result.x[:n], result.x[-1]
    except Exception:
        pass
    return None, None

def optimize_lp(f_init, deadline, verbose_prefix=""):
    """LP-based optimization with line search."""
    f = f_init.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Solve LP
        g, t_opt = solve_lp(f, S)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g = np.clip(g, 0, 1000)
        
        # Line search
        best_alpha_score = best_score
        best_alpha_f = None
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
            candidate = (1 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0, 1000)
            cand_sum = float(np.sum(candidate))
            if cand_sum < 1e-10:
                continue
            # Normalize to maintain sum
            candidate = candidate * (S / cand_sum)
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_alpha_score:
                best_alpha_score = cand_score
                best_alpha_f = candidate
        
        it += 1
        
        if best_alpha_f is not None:
            f = best_alpha_f
            best_f = f.copy()
            best_score = best_alpha_score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    
    return best_f, best_score
```
