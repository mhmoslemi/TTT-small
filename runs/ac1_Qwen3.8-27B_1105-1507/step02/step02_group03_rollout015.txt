import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute the objective: 2*n*max(conv)/(sum(f))^2"""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def fast_convolve(f):
    """Compute full convolution using FFT for efficiency."""
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, n_fft)
    conv = np.fft.irfft(F * F, n_fft)[:2 * n - 1]

with active set reduction, and adaptive multi-scale optimization.
Key improvements: top-k constraint LP for speed, better spectral target shapes,
symmetric sequence exploitation, and adaptive time allocation.

import numpy as np

def compute_score(f):
    """Compute score: 2*n*max(conv)/(sum(f))^2"""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def normalize(f):
    """Normalize to sum=1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def convolve_fft(f):
    """Fast full convolution using FFT."""
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, n_fft)
    return np.fft.irfft(F * F, n_fft)[:2 * n - 1]


def build_conv_matrix_topk(f, k):
    """Build convexolution matrix for top-k constraints only."""
    n = len(f)
    conv = convolve_fft(f)
    # Get top-k indices
    topk = np.argsort(conv)[-k:]
    # Also always include the center
    if (n - 1) not in topk:
        topk = np.concatenate([topk[:-1], [n - 1]])
    A = np.zeros((len(topk), n))
    for row, kk in enumerate(topk):
        j_start = max(0, kk - n + 1)
        j_end = min(n, kk + 1)
        for j in range(j_start, j_end):
            A[row, j] = f[kk - j]
    return A, topk


def build_conv_matrix_full(f):
    """Build full convolution matrix."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for kk in range(2 * n - 1):
        j_start = max(0, kk - n + 1)
        j_end = min(n, kk + 1)
        for j in range(j_start, j_end):
            A[kk, j] = f[kk - j]
    return A


def spectral_refine(f, n_iter=300, rng=None):
    """Spectral flattening with adaptive target."""
    if rng is None:
        rng = np.random.default_rng(42)
    f = np.array(f, dtype=np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        # Target: flatten with slight DC emphasis
        n_bins = len(mag)
        target_mag = np.ones(n_bins)
        # Slightly boost low frequencies to help with boundary effects
        freq_ratio = np.arange(n_bins) / n_bins
        target_mag = 1.0 + 0.3 * np.exp(-freq_ratio * 5.0)

        # Scale to preserve sum
        F_new = target_mag * np.exp(1j * phase)
        # Scale so DC = 1 (sum = 1)
        F_new = F_new / F_new[0] * 1.0
        f_new = np.fft.irfft(F_new, n_fft)[:n]

        # Soft clipping (better than hard zero)
        f_new = np.maximum(f_new, 0.0)
        # Add small floor to avoid zero elements
        f_new = f_new + 0.001
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:

def pressure_refine(f, n_iter=500, eta=0.08, rng=None):
    """Pressure-based refinement with FFT convolution and top-k focus."""

def coordinate_descent_fast(f, n_sweeps=20, rng=None):

def simulated_annealing(f, rng, n_iter=3000, T0=0.03, T_end=0.00001):
    """Simulated annealing with diverse moves."""
    f = np.array(f, dtype=np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0
    n_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()
        r = rng.random()

        if r < 0.25:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.3))
        elif r < 0.45:
            # Smooth Gaussian perturbation
            center = int(rng.integers(0, n))
            width = max(2, int(n // 32))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.normal(0, 0.08) * kernel
        elif r < 0.65:
            # Two-point swap with scaling
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j] * 0.8, f_new[i] * 0.8
        elif r < 0.85:
            # Global smooth perturbation
            f_new *= np.exp(rng.normal(0, 0.15, n))
        else:
            # Large restructure
            f_new *= np.exp(rng.normal(0, 0.5, n))

        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        F = np.fft.rfft(f_new, n_fft)
        conv = np.fft.irfft(F * F, n_fft)[:2 * n - 1]
        new_score = 2.0 * n * conv.max() / 1.0  # sum=1 so s^2=1

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 1500 and it > n_iter // 2:
            break

    return best_f, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""

    if r < 0.2:

            f += 0.3 * np.sin(2 * np.pi * fr * np.arange(n) / n + phase)
        f = np.maximum(f, 0.05)
    elif r < 0.4:
        # Raised cosine

        f = np.maximum(f, 0.01)
    elif r < 0.6:

        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
        f = np.maximum(f, 0.01)
    elif r < 0.8:
        # Product of cosines
        f = np.ones(n)

            phase = rng.uniform(0, np.pi)

    else:
        # Random smooth (interpolation of few random points)

    f = f * (1 + 0.05 * rng.normal(0, 1, n))
    f = np.maximum(f, 0.005)
    f = f / f.sum()
    return f


def symmetric_init(n, rng):
    """Generate a symmetric initial sequence."""
    half = n // 2 + 1
    f_half = rng.uniform(0.1, 1.0, half)

    f[:half] = f_half
    f[half:] = f_half[:n - half][::-1]
    f = f / f.sum()
    return f

            n_iter_spec = max(50, int(t_spec * 30))
            f1, s1 = spectral_refine(f0, n_iter=n_iter_spec, rng=rng)
            print(f"  Init {init_i} after spectral: {s1:.6f}")

            # Phase 2: Pressure refinement
            t_press = min(30, t_alloc * 0.2)
            n_iter_press = max(100, int(t_press * 20))
            f2, s2 = pressure_refine(f1, n_iter=n_iter_press, eta=0.08, rng=rng)
            print(f"  Init {init_i} after pressure: {s2:.6f}")

            candidates.append((f2, s2))

            if s2 < best_score_overall:
                best_score_overall = s2
                best_overall = f2.copy()
                print(f"  New best: {s2:.6f}")

        # Phase 3: LP refinement on best candidates
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_alloc * 0.45)
        if t_lp > 10 and candidates:
            # Sort candidates by score
            candidates.sort(key=lambda x: x[1])

            # LP refine the best 1-2 candidates
            n_lp_candidates = min(2, len(candidates))
            t_per_lp = t_lp / n_lp_candidates

            for ci in range(n_lp_candidates):
                rem = deadline - time.monotonic()
                if rem < 10:
                    break
                fc, sc = candidates[ci]
                t_lp_eff = min(t_per_lp, rem - 5)
                if t_lp_eff < 5:
                    break

                print(f"  LP refine candidate {ci} (score={sc:.6f}, time={t_lp_eff:.0f}s), topk=16")
                f3, s3 = lp_refine_efficient(fc, t_lp_eff, t0=0.3, rng=rng, topk=16)
                print(f"  After LP (topk=16): {s3:.6f}")

                if s3 < best_score_overall:
                    best_score_overall = s3
                    best_overall = f3.copy()
                    print(f"  New best: {s3:.6f}")

                # Try with more constraints if time allows
                rem = deadline - time.monotonic()
                if rem > 10 and s3 < best_score_overall + 0.01:
                    t_lp2 = min(rem * 0.5, 30)
                    if t_lp2 > 5:
                        print(f"  LP refine with full constraints (time={t_lp2:.0f}s)")
                        f4, s4 = lp_refine_full(f3, t_lp2, t0=0.2, rng=rng)
                        print(f"  After LP (full): {s4:.6f}")
                        if s4 < best_score_overall:
                            best_score_overall = s4
                            best_overall = f4.copy()
                            print(f"  New best: {s4:.6f}")

    # Final polishing phase
    if best_overall is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            print(f"\n--- Final Polishing (remaining: {rem:.0f}s) ---")
            n_best = len(best_overall)

            # Coordinate descent
            if rem > 10:
                n_sweeps = max(5, int(min(30, rem * 0.3)))
                f_cd, s_cd = coordinate_descent_fast(best_overall, n_sweeps=n_sweeps, rng=rng)
                if s_cd < best_score_overall:
                    best_score_overall = s_cd
                    best_overall = f_cd.copy()
                    print(f"  After CD: {best_score_overall:.6f}")

            # Simulated annealing
            rem = deadline - time.monotonic()
            if rem > 10:
                n_iter_sa = min(5000, int(rem * 200))
                f_sa, s_sa = simulated_annealing(best_overall, rng, n_iter=n_iter_sa, T0=0.02, T_end=0.00001)
                if s_sa < best_score_overall:
                    best_score_overall = s_sa
                    best_overall = f_sa.copy()
                    print(f"  After SA: {best_score_overall:.6f}")

            # Final CD
            rem = deadline - time.monotonic()
            if rem > 8:
                f_cd2, s_cd2 = coordinate_descent_fast(best_overall, n_sweeps=10, rng=rng)
                if s_cd2 < best_score_overall:
                    best_score_overall = s_cd2
                    best_overall = f_cd2.copy()
                    print(f"  Final CD: {best_score_overall:.6f}")

            # One more LP if time allows
            rem = deadline - time.monotonic()
            if rem > 15:
                print(f"  Final LP polish (time={rem:.0f}s)")
                f_lp, s_lp = lp_refine_efficient(best_overall, min(rem - 5, 60), t0=0.1, rng=rng, topk=20)
                if s_lp < best_score_overall:
                    best_score_overall = s_lp
                    best_overall = f_lp.copy()
                    print(f"  After final LP: {best_score_overall:.6f}")

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    print("Using fallback")
    return [1.0] * 256
