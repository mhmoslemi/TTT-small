import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_linearized_lp(f, deadline_check):
    """Solve LP: min M_new s.t. linearized conv(f+δ, f+δ) <= M_new, sum(δ)=0, f+δ>=0"""
    n = len(f)
    conv_ff = np.convolve(f, f)
    
    # Build the 2*C matrix (sparse)
    A_conv = build_conv_matrix_sparse(f)
    A_full = 2.0 * A_conv  # 2 * conv matrix
    
    # Variables: δ (n vars) + M_new (1 var) = n+1 vars
    # Objective: minimize M_new → c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_full @ δ - M_new <= -conv_ff
    # i.e., A_full @ δ + (-1) * M_new <= -conv_ff
    A_ineq = np.hstack([A_full.toarray(), -np.ones((2*n-1, 1))])
    b_ineq = -conv_ff
    
    # Equality: sum(δ) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: δ_i >= -f_i, M_new free (or >= 0)
    bounds = [(-f[i], None) for i in range(n)] + [(0, None)]
    
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,

Optimization of step function heights to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses alternating LP (fixing one copy, optimizing the other) with line search,
combined with softmax subgradient polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix


def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_conv_matrix(f):
    """Dense matrix C where (C @ x)[k] = convolve(f, x)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            C[k, j] = f[k - j]
    return C


def solve_fixed_lp(f, S):
    """Solve: min M s.t. conv(f, x) <= M, sum(x) = S, x >= 0.
    Returns optimal x and M."""
    n = len(f)
    C = build_conv_matrix(f)
    
    # Variables: x (n) + M (1)
    # Objective: min M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: C @ x - M <= 0 → [C, -1] @ [x; M] <= 0
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(x) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: x >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[-1]
    return None, None


def line_search_blend(f, x, S):
    """Find best t in [0,1] to minimize score of (1-t)*f + t*x (renormalized to sum S)."""
    n = len(f)
    
    def obj(t):
        candidate = (1 - t) * f + t * x
        candidate = np.maximum(candidate, 0)
        csum = np.sum(candidate)
        if csum < 1e-10:
            return 1e10
        candidate = candidate * (S / csum)
        return compute_score(candidate)
    
    result = minimize_scalar(obj, bounds=(0, 1), method='bounded',
                             options={'xatol': 1e-6})
    return result.x, result.fun


def lp_optimize(f, deadline, verbose=True):

    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        x, M_lp = solve_fixed_lp(f, S)
        if x is None:

        # Line search
        t_opt, score_opt = line_search_blend(f, x, S)
        
        if t_opt > 0.001 and score_opt < best_score:
            f_new = (1 - t_opt) * f + t_opt * x
            f_new = np.maximum(f_new, 0)
            csum = np.sum(f_new)
            f_new = f_new * (S / csum)
            f = f_new
            best_f = f.copy()
            best_score = score_opt
            if verbose and it % 5 == 0:
                print(f"    LP iter {it}: score={best_score:.6f}, t={t_opt:.4f}, M_lp={M_lp:.6f}")
        else:

            improved = False
            for t_try in [0.5 * t_opt, 0.1, 0.05, 0.01]:
                if t_try <= 0:

                candidate = (1 - t_try) * f + t_try * x
                candidate = np.maximum(candidate, 0)
                csum = np.sum(candidate)
                if csum < 1e-10:
                    continue
                candidate = candidate * (S / csum)
                sc = compute_score(candidate)
                if sc < best_score:
                    f = candidate
                    best_f = f.copy()
                    best_score = sc
                    improved = True
                    break
            if not improved:
                # Try the other direction: optimize with x fixed, find y
                y, M_lp2 = solve_fixed_lp(x, S)
                if y is not None:
                    t2, s2 = line_search_blend(x, y, S)
                    if t2 > 0.001 and s2 < best_score:

Is H_k PSD? H_k is a "permutation-like" matrix (it picks out pairs (j, k-j)). For k < n, H_k has nonzeros only on the "anti-diagonal" at position k. This is a symmetric matrix that's a scaled permutation matrix (or zero). It's not PSD in general.

For example, with n=3, k=1: H_1 = [[0,1,0],[1,0,0],[0,0,0]]. The eigenvalues are 1, -1, 0. Not PSD!

So the objective is NOT convex. This means we're dealing with a non-convex optimization problem, and local minima are possible.

This confirms that we need good initializations and possibly multiple restarts.

Given the non-convexity, the LP-based approach (which is a linearization at the current point) is essentially a local method. It will converge to a local minimum. The quality of the local minimum depends on the starting point.

"""
Optimization for minimizing 2n*max(convolve(a,a))/sum(a)^2 using:
1. LP-based direction finding (fix one copy, optimize other via LP)
2. Line search along blend direction
3. Softmax-weighted subgradient for polishing
4. Multiple initializations with the parent sequence as primary start
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar


def compute_score(f):
    """Compute objective: 2n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def fast_convolve(f):
    """FFT-based convolution for speed."""
    n = len(f)
    n_fft = 1
    while n_fft < 2 * n - 1:
        n_fft *= 2
    F = np.fft.rfft(f, n_fft)
    result = np.fft.irfft(F * F, n_fft)
    return result[:2 * n - 1]


def build_conv_matrix_dense(f):
    """Dense matrix C where (C@x)[k] = convolve(f, x)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            C[k, j] = f[k - j]
    return C


def solve_fixed_lp(f, S, time_limit=30.0):
    """Solve: min M s.t. conv(f,x) <= M, sum(x)=S, x>=0. Returns (x, M)."""
    n = len(f)
    C = build_conv_matrix_dense(f)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': time_limit})
        if result.success:
            return result.x[:n].copy(), float(result.x[-1])
    except Exception:
        pass
    return None, None


def line_search(f, direction, S, t_max=1.0):
    """Line search: find best t in [0, t_max] for f + t*(direction - f)."""
    n = len(f)
    
    def obj(t):
        candidate = (1.0 - t) * f + t * direction
        candidate = np.maximum(candidate, 0.0)
        csum = float(np.sum(candidate))
        if csum < 1e-10:
            return 1e10
        candidate = candidate * (S / csum)
        return compute_score(candidate)
    
    try:
        res = minimize_scalar(obj, bounds=(0.0, t_max), method='bounded',
                              options={'xatol': 1e-5})
        return float(res.x), float(res.fun)
    except Exception:
        # Fallback: try a few values
        best_t, best_s = 0.0, obj(0.0)
        for t in [0.1, 0.25, 0.5, 0.75, 0.9]:
            s = obj(t)
            if s < best_s:
                best_s = s
                best_t = t
        return best_t, best_s


def softmax_subgrad_step(f, T, lr):
    """One softmax-weighted subgradient step."""
    n = len(f)
    conv = np.convolve(f, f)
    S = float(np.sum(f))
    
    shifted = T * (conv - np.max(conv))
    exp_v = np.exp(shifted)
    weights = exp_v / np.sum(exp_v)
    
    # grad[j] = 2 * sum_k weights[k] * f[k-j] for valid k-j
    grad = np.zeros(n)
    for j in range(n):
        k_min = max(0, j)
        k_max = min(j + n - 1, 2 * n - 2)
        if k_max >= k_min:
            grad[j] = 2.0 * np.dot(weights[k_min:k_max + 1], f[0:k_max - j + 1])
    
    f_new = f - lr * grad
    f_new = np.maximum(f_new, 0.0)
    csum = float(np.sum(f_new))
    if csum < 1e-10:
        return f, compute_score(f)
    f_new = f_new * (S / csum)
    return f_new, compute_score(f_new)


def subgradient_polish(f, deadline, T_init=5.0, T_final=100.0,
                       lr_init=0.005, verbose_prefix=""):
    """Subgradient descent with annealed temperature."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    no_improve = 0
    it = 0
    
    total_iter_est = int((deadline - time.monotonic()) / 0.05)
    total_iter_est = max(total_iter_est, 10)
    
    while time.monotonic() < deadline:
        progress = min(1.0, it / total_iter_est)
        T = T_init * (T_final / T_init) ** progress
        
        f_new, score_new = softmax_subgrad_step(f, T, lr)
        it += 1
        
        if score_new < best_score:
            f = f_new
            best_f = f.copy()
            best_score = score_new
            no_improve = 0
            if it % 200 == 0:
                print(f"    {verbose_prefix}subgrad iter {it}: score={best_score:.6f} T={T:.1f} lr={lr:.5f}")
        else:
            no_improve += 1
            lr *= 0.97
            if no_improve > 200 or lr < 1e-7:
                break
    
    return best_f, best_score

def make_initializations(n, rng):
    """Create diverse initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    
    inits.append(('uniform', np.ones(n)))
    
    cos_t = 0.5 * (1.0 + np.cos(2.0 * np.pi * idx / (n - 1)))
    inits.append(('hann', np.maximum(cos_t, 0.01)))
    
    center = (n - 1) / 2.0
    gauss = np.exp(-0.5 * ((idx - center) / (n / 4.0)) ** 2)
    inits.append(('gaussian', gauss))
    
    two_peak = (np.exp(-0.5 * ((idx - n * 0.25) / (n / 8.0)) ** 2) +
                np.exp(-0.5 * ((idx - n * 0.75) / (n / 8.0)) ** 2))
    inits.append(('bimodal', two_peak))
    
    raised = np.sin(np.pi * idx / (n - 1))
    inits.append(('raised_cos', np.maximum(raised, 0.01)))
    
    # Flat top with tapered edges
    flat = np.ones(n)
    taper = n // 8
    for i in range(taper):
        flat[i] = 0.5 * (1 + np.cos(np.pi * i / taper))
        flat[n - 1 - i] = 0.5 * (1 + np.cos(np.pi * i / taper))
    inits.append(('flat_top', flat))
    
    r = rng.uniform(0.1, 2.0, size=n)
    inits.append(('random', r))
    
    # Power-law taper
    pw = (1.0 + idx / (n - 1)) ** (-0.5)
    inits.append(('power', pw))
    
    # Cosine squared
    cos_sq = (0.5 * (1.0 + np.cos(2.0 * np.pi * idx / (n - 1)))) ** 2
    inits.append(('cos_sq', np.maximum(cos_sq, 0.001)))
    
    return inits
