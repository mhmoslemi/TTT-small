def optimize_coordinate(conv, a, i, num_trials=20):
    """Find optimal new value for a[i] to minimize max(conv)."""
    n = len(a)
    v = a[i]
    best_d = 0
    best_max = np.max(conv)
    
    # Try several delta values
    # Scale of delta: related to a[i] and the conv values
    scale = max(v, 0.01)
    
    for d in np.linspace(-0.5*scale, 0.5*scale, num_trials):
        new_v = v + d
        if new_v < 0:
            continue
        # Compute new max of conv (approximately)
        # Only conv[k] for k in [i, i+n-1] change
        k_start = max(0, i)  # k where k-i >= 0
        k_end = min(2*n-2, i+n-1)  # k where k-i < n
        
        # For k in [k_start, k_end], k != 2i:
        new_vals = conv[k_start:k_end+1].copy()
        for offset in range(k_end - k_start + 1):
            k = k_start + offset
            if k == 2*i:
                new_vals[offset] = conv[k] + 2*d*v + d*d
            else:
                new_vals[offset] = conv[k] + 2*d*a[k-i]
        
        new_max = np.max(new_vals)
        if new_max < best_max:
            best_max = new_max
            best_d = d
    
    return best_d, best_max
```

    idx_2i = 2*i - k_start  # index of k=2i in the ks array
    if 0 <= idx_2i < num_k:
        # At k=2i, the formula is conv[2i] + 2*d*v + d^2
        # This is quadratic, not linear
        # For the linear approximation, coeff = 2*v, and we add d^2 separately
        coeffs[idx_2i] = 2.0 * v  # linear part
    
    # For each delta, new_conv = conv_base + deltas[:, None] * coeffs[None, :]
    # Shape: (num_deltas, num_k)
    new_convs = conv_base[None, :] + deltas[:, None] * coeffs[None, :]
    
    # Add quadratic correction for k=2i
    if 0 <= idx_2i < num_k:
        new_convs[:, idx_2i] += deltas**2
    
    # But wait, we also need to consider conv values OUTSIDE [k_start, k_end]
    # Those don't change, so the overall max is max(new_convs, max_outside)
    if k_start > 0:
        max_outside_left = np.max(conv[:k_start])
    else:
        max_outside_left = -np.inf
    if k_end < 2*n-2:
        max_outside_right = np.max(conv[k_end+1:])
    else:
        max_outside_right = -np.inf
    max_outside = max(max_outside_left, max_outside_right)
    
    overall_max = np.maximum(np.max(new_convs, axis=1), max_outside)
    
    best_idx = np.argmin(overall_max)
    return deltas[best_idx], overall_max[best_idx]
```

"""
Optimized step function search using parametric initialization + efficient coordinate descent
with incremental convolution updates + simulated annealing. Minimizes 2*n*max(conv(a,a))/sum(a)^2
by exploiting scale invariance to reduce to minimizing max(conv) with sum(a)=1.
"""

import numpy as np
import time


def compute_conv_fast(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fast(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize so sum(a) = 1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def make_cosine_seq(n, A, f, phase):
    """Generate a[j] = 1 + A*cos(2*pi*f*j/n + phase), normalized."""
    j = np.arange(n)
    a = 1.0 + A * np.cos(2.0 * np.pi * f * j / n + phase)
    return normalize(a)


def make_power_u_seq(n, p, C):
    """Generate a[j] = |2*j/n - 1|^p + C, normalized."""
    j = np.arange(n)
    x = np.abs(2.0 * j / (n - 1) - 1.0) if n > 1 else np.array([0.0])
    a = x ** p + C
    return normalize(a)


def make_gaussian_u_seq(n, B):
    """Generate a[j] = exp(B*((j-(n-1)/2)/(n/2))^2), normalized. B>0 for U-shape."""
    j = np.arange(n).astype(np.float64)
    center = (n - 1) / 2.0
    width = max(1.0, n / 2.0)
    a = np.exp(B * ((j - center) / width) ** 2)
    return normalize(a)


def make_rational_u_seq(n, B):
    """Generate a[j] = 1/(1 + B*((j-(n-1)/2)/(n/2))^2), normalized. B>0 for U-shape."""
    j = np.arange(n).astype(np.float64)
    center = (n - 1) / 2.0
    width = max(1.0, n / 2.0)
    a = 1.0 / (1.0 + B * ((j - center) / width) ** 2)
    return normalize(a)


def make_exp_cos_seq(n, D, f):

    j = np.arange(n)
    a = np.exp(D * np.cos(2.0 * np.pi * f * j / n))

            i2 = rng.integers(0, n)

            frac = rng.uniform(0.1, 0.3)
            count = max(1, int(n * frac))
            idx = rng.choice(n, count, replace=False)
            a_pert[idx] = rng.uniform(0, 2.0/n, count)

        a_pert = np.maximum(a_pert, 0)
        s = np.sum(a_pert)
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        # Run coordinate descent from perturbed point
        a_cd, _ = coord_descent_fast(a_pert, max_passes=max_cd_passes, seed=seed + r)
        obj_new = compute_obj(a_cd)

        # Accept/reject
        if obj_new < best_obj:
            a = a_cd
            best_a = a.copy()
            best_obj = obj_new
            if verbose:
                print(f"  Anneal {r}: improved to {best_obj:.6f}")
        else:
            # Metropolis acceptance
            delta = obj_new - best_obj
            if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-8)):
                a = a_cd

        t *= cool_rate

    return best_a, best_obj


def parametric_search(n, max_time=30.0, verbose=False):
    """Search over parametric families for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    # Cosine family
    for A in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0, 1.2]:
        for f in [1, 2, 3]:
            for phase in [0, np.pi / 2, np.pi]:
                if time.monotonic() - start > max_time:
                    return best_a, best_obj
                a = make_cosine_seq(n, A, f, phase)
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

    # Power U family
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for C in [0.01, 0.05, 0.1, 0.3, 0.5]:
            if time.monotonic() - start > max_time:
                return best_a, best_obj
            a = make_power_u_seq(n, p, C)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    # Gaussian U family
    for B in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0]:
        if time.monotonic() - start > max_time:
            return best_a, best_obj
        a = make_gaussian_u_seq(n, B)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Rational U family
    for B in [1.0, 2.0, 5.0, 10.0, 20.0, 50.0]:
        if time.monotonic() - start > max_time:
            return best_a, best_obj
        a = make_rational_u_seq(n, B)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Exp-cos family
    for D in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]:
        for f in [1, 2]:
            if time.monotonic() - start > max_time:
                return best_a, best_obj
            a = make_exp_cos_seq(n, D, f)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    if verbose and best_a is not None:
        print(f"  Parametric n={n}: best_obj={best_obj:.6f}")

    return best_a, best_obj


def refine_with_cd(a, max_time=60.0, seed=0, verbose=False):
    """Run coordinate descent with multiple restarts within time budget."""
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    n = len(a)

    # Phase 1: Standard CD
    remaining = max_time - (time.monotonic() - start)
    passes = int(min(5000, remaining * 50))
    a_cd, _ = coord_descent_fast(a, max_passes=max(100, passes), seed=seed)
    obj = compute_obj(a_cd)
    if obj < best_obj:
        best_obj = obj
        best_a = a_cd.copy()
        if verbose:
            print(f"    CD: {best_obj:.6f}")

    # Phase 2: Try smaller perturbations + CD
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        rng = np.random.default_rng(seed + 1000)
        for attempt in range(5):
            if time.monotonic() - start > max_time - 3:
                break
            scale = 0.02 / (attempt + 1)
            a_p = best_a * np.exp(rng.normal(0, scale, n))
            a_p = np.maximum(a_p, 0)
            a_p = a_p / np.sum(a_p)
            passes2 = int(min(2000, remaining / 2))
            a_cd2, _ = coord_descent_fast(a_p, max_passes=max(50, passes2), seed=seed + 2000 + attempt)
            obj2 = compute_obj(a_cd2)
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a_cd2.copy()
                if verbose:
                    print(f"    CD perturb {attempt}: {best_obj:.6f}")

    return best_a, best_obj


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for best step function sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses parametric initialization + efficient coordinate descent + annealing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Sequence lengths to explore (biased toward medium-length)
    n_values = [30, 40, 50, 60, 80, 100, 120, 150, 200, 250, 300, 400, 500, 600, 800, 1000, 1200, 1500]

    # If parent exists, include its length
    if parent is not None:
        np_ = len(parent)
        n_values = sorted(set(n_values + [np_, max(10, np_ - 30), min(2000, np_ + 30)]))

    print(f"Exploring n values: {n_values}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Phase 1: Parametric search + quick CD for each n
    for n in n_values:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Allocate time based on n
        if n <= 80:
            alloc = min(15.0, remaining * 0.05)
        elif n <= 200:
            alloc = min(30.0, remaining * 0.08)
        elif n <= 400:
            alloc = min(45.0, remaining * 0.1)
        elif n <= 800:
            alloc = min(60.0, remaining * 0.12)
        else:
            alloc = min(80.0, remaining * 0.15)

        alloc = max(8.0, alloc)

        # Parametric search
        a_par, obj_par = parametric_search(n, max_time=alloc * 0.4, verbose=False)

        # Also try random initializations
        rng = np.random.default_rng(seed + n * 31)
        a_rand = rng.uniform(0, 1, n)
        a_rand = normalize(a_rand)
        obj_rand = compute_obj(a_rand)

        # Pick better starting point
        if a_par is not None and obj_par < obj_rand:
            a_start = a_par
            obj_start = obj_par
        else:
            a_start = a_rand
            obj_start = obj_rand

        # Coordinate descent refinement
        remaining_cd = alloc - (time.monotonic() - start_time) + (deadline - start_time)
        remaining_cd = min(alloc * 0.6, deadline - time.monotonic() - 5)
        if remaining_cd > 3:
            a_cd, obj_cd = refine_with_cd(a_start, max_time=remaining_cd, seed=seed + n, verbose=False)
            if obj_cd < obj_start:
                a_start = a_cd
                obj_start = obj_cd

        if obj_start < best_score:
            best_score = obj_start
            best_seq = a_start.copy()
            print(f"  UPDATE: n={n}, obj={best_score:.6f}")

    print(f"\nAfter Phase 1 (parametric+CD): best={best_score:.6f} (n={len(best_seq) if best_seq is not None else 'N/A'})")

    # Phase 2: Intensive optimization on best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPhase 2: Intensive optimization, n={n_best}, {remaining:.0f}s remaining")

        a_cur = best_seq.copy()

        # Multiple rounds of CD + perturbation
        round_num = 0
        while time.monotonic() < deadline - 20:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # CD with many passes
            cd_time = min(remaining * 0.4, 120.0)
            cd_passes = int(cd_time * 30)
            a_cd, _ = coord_descent_fast(a_cur, max_passes=cd_passes, seed=seed + 10000 + round_num)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                a_cur = a_cd.copy()
                print(f"  Round {round_num} CD: {best_score:.6f}")

            # Perturbation + CD (annealing step)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                rng = np.random.default_rng(seed + 20000 + round_num)
                n_try = min(5, int(remaining / 15))
                for t in range(n_try):
                    if time.monotonic() > deadline - 20:
                        break
                    # Various perturbation strategies
                    strat = rng.integers(0, 4)
                    a_p = a_cur.copy()
                    if strat == 0:
                        noise_scale = 0.03 / (1 + round_num * 0.1)
                        a_p = a_cur * np.exp(rng.normal(0, noise_scale, n_best))
                    elif strat == 1:
                        # Block perturbation
                        blen = max(1, n_best // 5)
                        start = rng.integers(0, n_best - blen)
                        a_p[start:start + blen] = rng.uniform(0, 3.0/n_best, blen)
                    elif strat == 2:
                        # Random subset reset
                        frac = rng.uniform(0.05, 0.2)
                        count = max(1, int(n_best * frac))
                        idx = rng.choice(n_best, count, replace=False)
                        a_p[idx] = rng.uniform(0, 2.0/n_best, count)
                    else:
                        # Mirror perturbation (exploit symmetry)
                        a_p[:n_best//2] = a_cur[n_best//2:] * rng.uniform(0.8, 1.2, n_best//2)

                    a_p = np.maximum(a_p, 0)
                    s_p = np.sum(a_p)
                    if s_p < 1e-10:
                        continue
                    a_p = a_p / s_p

                    # Quick CD from perturbed
                    a_cd2, _ = coord_descent_fast(a_p, max_passes=500, seed=seed + 30000 + t)
                    obj_p = compute_obj(a_cd2)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_cd2.copy()
                        a_cur = a_cd2.copy()
                        print(f"  Round {round_num} perturb {strat}: {best_score:.6f}")

    # Phase 3: Try longer sequences if time remains
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 3: Trying longer sequences, {remaining:.0f}s remaining")

        long_ns = [2000, 3000]
        if len(best_seq) < 1000:
            long_ns = [1500, 2000, 3000]

        for n_long in long_ns:
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # Quick parametric init for long sequence
            a_long = make_cosine_seq(n_long, 0.7, 1, 0)
            obj_long = compute_obj(a_long)

            # Quick CD
            a_cd, _ = coord_descent_fast(a_long, max_passes=min(2000, int(remaining * 20)), seed=seed + 50000)
            obj_cd = compute_obj(a_cd)
            if obj_cd < obj_long:
                a_long = a_cd
                obj_long = obj_cd

            if obj_long < best_score:
                best_score = obj_long
                best_seq = a_long.copy()
                print(f"  Long n={n_long}: {best_score:.6f}")

    # Phase 4: Final ultra-fine polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 4: Final polish, {remaining:.0f}s remaining")
        n_final = len(best_seq)

        a_final = best_seq.copy()
        # Many short CD passes with different random orders
        for polish_round in range(max(1, int(remaining / 10))):
            if time.monotonic() > deadline - 5:
                break
            a_final, _ = coord_descent_fast(a_final, max_passes=2000, seed=seed + 60000 + polish_round)
            obj_f = compute_obj(a_final)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_final.copy()
                if polish_round % 5 == 0:
                    print(f"  Polish {polish_round}: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Validate with evaluate_sequence
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-6:
            print(f"  Validation: computed={best_score:.6f}, validated={validation:.6f}")
            if validation < best_score:
                best_score = validation

    return result
