**Key innovation: "Peak shaving" with constraint activation**

Instead of solving an LP at each step (which is expensive for large n), I'll use a simpler approach:

1. Compute b = conv(f, f)
2. Find the top-K peaks in b
3. For each peak at position k*, the contribution is b[k*] = sum_j f[j] f[k*-j]
4. To reduce b[k*], we can reduce f[j] for j that contribute significantly to b[k*]
5. But reducing f[j] also affects other b[k] values
6. Use a weighted update: reduce f[j] proportionally to its contribution to the top peaks, while being careful not to increase other peaks

This is essentially a subgradient method for the max function, but with careful weighting.

**Even better: Use the "active set" approach**

At the optimum, some of the constraints b[k] <= t are active (tight). The active set determines the structure. We can:
1. Start with all constraints active
2. At each iteration, solve the active set problem
3. Check if any inactive constraint is violated
4. Add violated constraints to the active set
5. Repeat

For the active set problem (with m active constraints):
min t s.t. b[k_i] = t for i in active set, b[k] <= t for all k, sum(f) = 1, f >= 0

This is still non-convex, but with the linearization trick it becomes an LP.

"""
Optimized step function search using LP-based peak shaving,
adaptive length selection, FFT-based convolution, and multi-start strategy.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix
from scipy.fft import fft, ifft


def fast_convolve(f, g=None):
    """Fast convolution using FFT."""
    if g is None:
        g = f
    n = len(f)
    m = len(g)
    size = n + m - 1
    N = 1
    while N < size:
        N *= 2
    F = fft(f, N)
    G = fft(g, N)
    result = np.real(ifft(F * G))
    return result[:size]


def build_conv_matrix(f):
    """Build matrix A where (A @ g)[k] = convolve(f, g)[k]."""
    n = len(f)
    # A is (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_peak_shave(f, deadline, max_iters=500, verbose=False):
    """
    LP-based peak shaving: iteratively find direction g that maximizes sum(g)
    subject to conv(f,g) <= M (current peak), then take a step in direction g.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    f = f / S  # normalize to sum=1
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    # For speed, let's use the fact that A is determined by f.
    
    step_size = 0.3
    no_improve = 0
    it = 0

        it += 1
        b = fast_convolve(f)
        M = np.max(b)

            break
        
        # Build A matrix
        A = build_conv_matrix(f)
        
        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                         bounds=[(0, None)]*n, method='highs',
                         options={'presolve': True, 'time_limit': 5})
            if not res.success:
                no_improve += 5
                if no_improve > 30:
                    break
                continue
            g = res.x
        except Exception:
            no_improve += 5
            if no_improve > 30:
                break
            continue
        
        sum_g = np.sum(g)
        if sum_g < 1e-15:
            no_improve += 1
            if no_improve > 20:

        g = g / S  # same sum as f

        for t in [step_size, step_size*0.5, step_size*1.5, step_size*0.25, step_size*2]:

def soft_max_objective(f, beta):
    """Compute soft-max of conv(f,f) and its gradient."""
    b = fast_convolve(f)
    max_b = np.max(b)
    exp_b = np.exp(beta * (b - max_b))
    Z = np.sum(exp_b)
    p = exp_b / Z  # probability distribution over lag positions
    soft_val = max_b + np.log(Z) / beta
    
    # Gradient: d/dx_j [sum_k p[k] * b[k]] where b[k] = sum_i f[i] f[k-i]
    # = 2 * sum_k p[k] * sum_i f[i] * delta[k - i - j]
    # = 2 * sum_i f[i] * p[i+j]  (where i+j is valid)
    # This is 2 * correlate(p, f, 'valid')
    grad = 2.0 * np.convolve(p, f, mode='valid')[:len(f)]
    
    return soft_val, grad


def soft_max_optimize(f, deadline, beta_start=10, beta_end=500, lr=0.02, max_iters=5000, verbose=False):
    """Optimize using soft-max with Adam."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    f = f / S
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        _, grad = soft_max_objective(f, beta)
        
        # Adam update
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad**2
        m_hat = m / (1 - b1**it)
        v_hat = v / (1 - b2**it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        new_f = f - lr * update
        new_f = np.maximum(new_f, 0)
        ns = np.sum(new_f)
        if ns < 1e-10:
            new_f = f.copy()
            ns = S
        new_f = new_f / ns
        
        sc = evaluate_sequence(new_f.tolist())
        if sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    SM iter {it}: score={best_score:.6f}, beta={beta:.0f}")
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-7:
                    break
    
    return best_f, best_score


def coordinate_descent(f, deadline, max_iters=5000, verbose=False):
    """Coordinate descent: optimize one element at a time."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    f = f / S
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline and it < max_iters:
        improved = False
        for j in range(n):
            if time.monotonic() > deadline - 5:
                break
            
            old_val = f[j]
            # Try a few values
            for trial in [old_val * 0.9, old_val * 0.95, old_val * 1.05, old_val * 1.1]:
                if trial < 0:
                    continue
                f[j] = trial
                sc = evaluate_sequence(f.tolist())
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                    improved = True
                else:
                    f[j] = old_val
            
            if not improved and old_val == f[j]:
                f[j] = old_val
        
        it += 1
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0
    
    return best_f, best_score

"""
Multi-strategy optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses flat-autocorrelation initialization, LP peak shaving, soft-max gradient descent,
and fast vectorized coordinate descent with multiple lengths and restarts.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft


def fast_convolve(f, g=None):
    """Fast linear convolution using FFT."""
    if g is None:
        g = f
    n = len(f)
    m = len(g)
    size = n + m - 1
    N = 1
    while N < size:
        N *= 2
    F = fft(f, N)
    G = fft(g, N)
    result = np.real(ifft(F * G))
    return result[:size]


def compute_score(f):
    """Compute the score directly (faster than calling evaluate_sequence)."""
    f = np.maximum(f, 0)
    S = np.sum(f)
    if S < 0.01:
        return np.inf
    b = fast_convolve(f)
    return 2.0 * len(f) * np.max(b) / (S * S)


def build_conv_mat(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f, g)[k]."""
    n = len(f)
    rows = np.zeros(2 * n - 1 + n, dtype=np.int32)
    cols = np.zeros(2 * n - 1 + n, dtype=np.int32)
    vals = np.zeros(2 * n - 1 + n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def flat_autocorr_init(n):
    """
    Generate a sequence whose autocorrelation is flat (=1) for the first n lags,
    using the recursive formula. This gives an excellent initialization.
    """
    f = np.zeros(n)
    f[0] = 1.0
    for m in range(1, n):
        # b[m] = 2*f[0]*f[m] + 2*sum_{i=1}^{floor((m-1)/2)} f[i]*f[m-i]
        #       + (if m even) f[m/2]^2 = 1
        s = 0.0
        # Terms: 2*sum_{i=1}^{m-1} f[i]*f[m-i] but only count each pair once
        # b[m] = 2*f[0]*f[m] + 2*sum_{i=1}^{floor(m/2)} f[i]*f[m-i] - (if m even) f[m/2]^2
        # Wait, let me be careful:
        # b[m] = sum_{i=0}^{m} f[i]*f[m-i]  (all indices 0 to m-1 are valid since m < n)
        # = f[0]*f[m] + f[1]*f[m-1] + ... + f[m]*f[0]
        # = 2*f[0]*f[m] + 2*sum_{i=1}^{floor((m-1)/2)} f[i]*f[m-i] + (if m even) f[m//2]^2
        known_sum = 0.0
        for i in range(1, m):
            if i < m - i:
                known_sum += 2.0 * f[i] * f[m - i]
            elif i == m - i:
                known_sum += f[i] * f[m - i]
        # b[m] = 2*f[0]*f[m] + known_sum = 1
        # f[m] = (1 - known_sum) / (2*f[0])
        f[m] = (1.0 - known_sum) / (2.0 * f[0])
    # Ensure non-negativity
    f = np.maximum(f, 0.0)
    # Apply a mild taper at the end to help with the tail of the convolution
    taper_len = max(1, n // 10)
    for i in range(taper_len):
        t = (i + 1) / (taper_len + 1)
        taper = 0.5 * (1 + np.cos(np.pi * (1 - t)))
        f[n - 1 - i] *= taper
    f = np.maximum(f, 0.0)
    S = np.sum(f)
    if S > 0:
        f = f / S
    return f


def tukey_window(n, r=0.5):
    """Tukey window."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def gauss_window(n, sigma_frac=0.4):
    """Gaussian window."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n - 1) / 2.0
    w = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    return w


def lp_optimize(f, deadline, max_iters=300, verbose_prefix=""):
    """
    LP-based peak shaving. Rebuilds the constraint matrix each iteration.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    if S < 1e-10:
        return f, np.inf
    f = f / S

    best_f = f.copy()
    best_score = compute_score(f)

    step_size = 0.3
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        b = fast_convolve(f)
        M = float(np.max(b))
        if M <= 0:
            break

        A = build_conv_mat(f)
        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 8})
            if not res.success:
                no_improve += 5
                if no_improve > 40:
                    break
                continue
            g = res.x
        except Exception:
            no_improve += 5
            if no_improve > 40:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        # g has sum = sum_g. We want to blend: f_new = (1-t)*f + t*(g/sum_g)
        # so that sum stays 1.
        g_norm = g / sum_g

        # Line search
        best_t = 0.0
        best_s = best_score
        for t in [step_size, step_size * 0.5, step_size * 1.5, step_size * 0.25, 0.8, 0.95]:
            if t <= 0 or t > 1:
                continue
            new_f = (1.0 - t) * f + t * g_norm
            new_f = np.maximum(new_f, 0.0)
            ns = np.sum(new_f)
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = compute_score(new_f)
            if sc < best_s:
                best_s = sc
                best_t = t

        if best_t > 0 and best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g_norm
            f = np.maximum(f, 0.0)
            f = f / np.sum(f)
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 50 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f}, t={best_t:.3f}")
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.005 or no_improve > 50:
                break

    return best_f, best_score


def softmax_grad_optimize(f, deadline, beta_start=5.0, beta_end=300.0,
                          lr_start=0.03, max_iters=5000, verbose_prefix=""):
    """Soft-max gradient descent with Adam."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    if S < 1e-10:
        return f, np.inf
    f = f / S

    best_f = f.copy()
    best_score = compute_score(f)

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr_start
    it = 0
    no_improve = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        frac = min(1.0, it / (max_iters * 0.85))
        beta = beta_start * (beta_end / beta_start) ** frac

        # Compute soft-max gradient
        b = fast_convolve(f)
        max_b = np.max(b)
        exp_b = np.exp(beta * (b - max_b))
        Z = np.sum(exp_b)
        if Z < 1e-300:
            break
        p = exp_b / Z
        # grad[j] = 2 * sum_k p[k] * f[k-j] = 2 * correlate(p, f, 'valid')
        grad = 2.0 * np.correlate(p[:2*n-1], f, mode='valid')[:n]

        # Adam
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1 - b1 ** it)
        v_hat = v / (1 - b2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)

        new_f = f - lr * update
        new_f = np.maximum(new_f, 0.0)
        ns = np.sum(new_f)
        if ns < 1e-10:
            new_f = f.copy()
            ns = 1.0
        new_f = new_f / ns

        sc = compute_score(new_f)
        if sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 1000 == 0:
                print(f"    {verbose_prefix}SM iter {it}: score={best_score:.6f}, beta={beta:.0f}")
        else:
            no_improve += 1
            if no_improve % 150 == 0:
                lr *= 0.6
                no_improve = 0
                if lr < 1e-7:
                    break

    return best_f, best_score


def coord_descent_fast(f, deadline, max_sweeps=200, verbose_prefix=""):
    """
    Fast coordinate descent with vectorized convolution updates.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    if S < 1e-10:
        return f, np.inf
    f = f / S

    b = fast_convolve(f).copy()
    best_score = 2.0 * n * np.max(b)
    best_f = f.copy()

    no_improve_sweeps = 0
    sweep = 0

    while time.monotonic() < deadline and sweep < max_sweeps:
        sweep += 1
        improved = False

        # Random order for diversity
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() > deadline - 2:
                break

            old_val = f[j]
            k_min = j
            k_max = min(j + n, 2 * n - 1)  # exclusive

            if old_val < 1e-12:
                trials = [1e-4, 1e-3, 5e-3]
            else:
                trials = [old_val * 0.7, old_val * 0.85, old_val * 1.15, old_val * 1.3]

            for trial in trials:
                if trial <= 0:
                    continue
                delta = trial - old_val
                if abs(delta) < 1e-15:
                    continue

                # Vectorized update of b
                idx_f = np.arange(k_min, k_max) - j
                # b[k] changes by 2*delta*f[k-j] for k != 2j
                # b[2j] changes by 2*old_val*delta + delta^2
                b_delta = 2.0 * delta * f[idx_f]

                # Handle the diagonal case k = 2j
                if k_min <= 2 * j < k_max:
                    diag_pos = 2 * j - k_min
                    b_delta[diag_pos] = 2.0 * old_val * delta + delta * delta

                b[k_min:k_max] += b_delta

                new_max = np.max(b)
                new_score = 2.0 * n * new_max  # sum(f) is approximately 1

                if new_score < best_score:
                    best_score = new_score
                    best_f = f.copy()
                    f[j] = trial
                    improved = True
                else:
                    # Revert
                    b[k_min:k_max] -= b_delta
                    f[j] = old_val

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 5:
                break
        else:
            no_improve_sweeps = 0

        if sweep % 20 == 0:
            print(f"    {verbose_prefix}CD sweep {sweep}: score={best_score:.6f}")

    # Renormalize
    S_final = np.sum(best_f)
    if S_final > 0:
        best_f = best_f / S_final
    return best_f, compute_score(best_f)


def subgrad_optimize(f, deadline, lr_start=0.05, max_iters=3000, verbose_prefix=""):
    """
    Subgradient method using the active peak.
    For the top-K peaks, compute the subgradient and step.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    if S < 1e-10:
        return f, np.inf
    f = f / S

    best_f = f.copy()
    best_score = compute_score(f)
    lr = lr_start
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        b = fast_convolve(f)

        # Find top-K peaks
        K = min(10, 2 * n - 1)
        top_k_indices = np.argsort(b)[-K:]
        top_k_values = b[top_k_indices]
        # Weight by how much above the mean
        mean_b = np.mean(b)
        weights = np.maximum(top_k_values - mean_b, 0)
        if np.sum(weights) < 1e-15:
            weights = np.ones(K) / K
        else:
            weights = weights / np.sum(weights)

        # Subgradient: for each peak k*, grad[j] += w_k * 2 * f[k*-j]
        grad = np.zeros(n)
        for k_star, w in zip(top_k_indices, weights):
            # f[k_star - j] for valid j
            j_min = max(0, k_star - (n - 1))
            j_max = min(k_star, n - 1)
            if j_min <= j_max:
                grad[j_min:j_max + 1] += w * 2.0 * f[k_star - (np.arange(j_min, j_max + 1))]

        # Update
        new_f = f - lr * grad
        new_f = np.maximum(new_f, 0.0)
        ns = np.sum(new_f)
        if ns < 1e-10:
            new_f = f.copy()
            ns = 1.0
        new_f = new_f / ns

        sc = compute_score(new_f)
        if sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-7:
                    break

        if it % 500 == 0:
            print(f"    {verbose_prefix}SG iter {it}: score={best_score:.6f}, lr={lr:.5f}")

    return best_f, best_score


def optimize_pipeline(f, deadline, do_lp=True, do_sm=True, do_cd=True, verbose_prefix=""):
    """Run the full optimization pipeline on a starting point."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    if S < 1e-10:
        return f, np.inf
    f = f / S

    best_f = f.copy()
    best_score = compute_score(f)
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_f, best_score

    # Phase 1: LP (40% of time)
    if do_lp and remaining > 60:
        lp_deadline = time.monotonic() + remaining * 0.4
        f1, s1 = lp_optimize(f, lp_deadline, max_iters=200, verbose_prefix=verbose_prefix + "LP: ")
        if s1 < best_score:
            best_f, best_score = f1, s1

    # Phase 2: Soft-max gradient (35% of time)
    remaining = deadline - time.monotonic()
    if do_sm and remaining > 30:
        sm_deadline = time.monotonic() + remaining * 0.5
        f2, s2 = softmax_grad_optimize(best_f, sm_deadline, beta_start=5.0, beta_end=400.0,
                                        lr_start=0.02, verbose_prefix=verbose_prefix + "SM: ")
        if s2 < best_score:
            best_f, best_score = f2, s2

    # Phase 3: Subgradient (15% of time)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sg_deadline = time.monotonic() + remaining * 0.3
        f3, s3 = subgrad_optimize(best_f, sg_deadline, lr_start=0.03,
                                   verbose_prefix=verbose_prefix + "SG: ")
        if s3 < best_score:
            best_f, best_score = f3, s3

    # Phase 4: LP again
    remaining = deadline - time.monotonic()
    if do_lp and remaining > 30:
        lp2_deadline = time.monotonic() + remaining * 0.3
        f4, s4 = lp_optimize(best_f, lp2_deadline, max_iters=100, verbose_prefix=verbose_prefix + "LP2: ")
        if s4 < best_score:
            best_f, best_score = f4, s4

    # Phase 5: Coordinate descent (remaining time)
    remaining = deadline - time.monotonic()
    if do_cd and remaining > 15:
        cd_deadline = time.monotonic() + remaining * 0.9
        f5, s5 = coord_descent_fast(best_f, cd_deadline, max_sweeps=100,
                                     verbose_prefix=verbose_prefix + "CD: ")
        if s5 < best_score:
            best_f, best_score = f5, s5

    return best_f, best_score


def perturb_restart(f, rng, strength=0.05):
    """Perturb f with multiplicative noise and renormalize."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 3.0)
    new_f = np.maximum(new_f, 0.0)
    S = np.sum(new_f)
    if S < 1e-10:
        return f.copy()
    return new_f / S


def propose_candidate(budget_s=1500, **kwargs):
    """
    Main search function. Tries multiple initializations and lengths,
    using LP + soft-max + subgradient + coordinate descent pipeline.
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Load parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, UnboundLocalError, Exception) as e:
        print(f"No parent available: {e}")

    # Define lengths to try and time budgets
    # Total budget: deadline - t0
    total_budget = deadline - t0
    print(f"Total budget: {total_budget:.0f}s")

    # Strategy: focus on n=512 (refine parent) and n=1024, n=2048 (explore longer)
    lengths = [512, 1024, 2048]
    time_fracs = [0.35, 0.35, 0.20]
    # Remaining 0.10 for restarts and final polish

    for idx_n, n in enumerate(lengths):
        if time.monotonic() > deadline - 60:
            print("Time running out, stopping length exploration")
            break

        n_budget = total_budget * time_fracs[idx_n]
        n_deadline = time.monotonic() + n_budget
        print(f"\n{'='*60}")
        print(f"=== n={n} (budget {n_budget:.0f}s) ===")
        print(f"{'='*60}")

        # Generate initializations
        inits = []

        # 1. Flat autocorrelation initialization (novel!)
        f_flat = flat_autocorr_init(n)
        inits.append(('flat_ac', f_flat))

        # 2. Tukey window
        inits.append(('tukey', tukey_window(n, 0.5)))

        # 3. Gaussian window
        inits.append(('gauss', gauss_window(n, 0.35)))

        # 4. Parent (interpolated)
        if parent is not None and len(parent) > 1:
            x_parent = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            f_parent_interp = np.interp(x_new, x_parent, parent)
            f_parent_interp = np.maximum(f_parent_interp, 0)
            inits.insert(0, ('parent_interp', f_parent_interp))

        # 5. Symmetric flat_ac (mirror)
        f_flat_sym = f_flat.copy()
        f_flat_sym = np.maximum(f_flat_sym, f_flat_sym[::-1])
        inits.append(('flat_sym', f_flat_sym))

        # 6. Random smooth
        r_rng = np.random.default_rng(n + 7)
        x = r_rng.standard_normal(n)
        kernel = np.exp(-0.5 * (np.arange(-15, 16) / 4.0) ** 2)
        smooth_x = np.convolve(x, kernel, mode='same')
        smooth_x = np.abs(smooth_x) + 0.1
        inits.append(('rand_smooth', smooth_x))

        # Evaluate and sort initializations
        scored_inits = []
        for label, f_init in inits:
            f_init = np.maximum(f_init.astype(np.float64), 0.0)
            S = np.sum(f_init)
            if S < 0.01:
                continue
            f_init = f_init / S
            sc = compute_score(f_init)
            scored_inits.append((sc, label, f_init))
            print(f"  Init '{label}': score={sc:.6f}")

        scored_inits.sort(key=lambda x: x[0])

        # Optimize top initializations (top 3)
        per_init_time = n_budget / 3.0

        for rank, (init_sc, label, f_init) in enumerate(scored_inits[:3]):
            if time.monotonic() > n_deadline - 10:
                break

            if init_sc < best_score:
                best_score = init_sc
                best_sequence = f_init.copy()

            init_deadline = min(n_deadline, time.monotonic() + per_init_time)
            if time.monotonic() > init_deadline - 5:
                continue

            print(f"  Optimizing '{label}' (init score={init_sc:.6f})...")
            try:
                bf, bs = optimize_pipeline(f_init, init_deadline, verbose_prefix=f"[n={n} {label}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"    -> improved to {bs:.6f}")
                else:
                    print(f"    -> no improvement (best from init: {bs:.6f})")
            except Exception as e:
                print(f"    Error: {e}")

        # After optimizing all inits for this length, do one more round on the best
        remaining_for_n = n_deadline - time.monotonic()
        if remaining_for_n > 30 and best_sequence is not None and len(best_sequence) == n:
            print(f"  Extra refinement for n={n} ({remaining_for_n:.0f}s remaining)...")
            try:
                bf, bs = optimize_pipeline(best_sequence, time.monotonic() + remaining_for_n * 0.9,
                                           do_lp=False, do_sm=True, do_cd=True,
                                           verbose_prefix=f"[n={n} refine] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"    -> improved to {bs:.6f}")
            except Exception as e:
                print(f"    Error: {e}")

    # Perturbation restarts on the global best
    if best_sequence is not None and time.monotonic() < deadline - 40:
        print(f"\n{'='*60}")
        print(f"=== Perturbation restarts (best: {best_score:.6f}, n={len(best_sequence)}) ===")
        print(f"{'='*60}")
        restart_budget = deadline - time.monotonic() - 20
        n_restarts = min(6, max(2, int(restart_budget / 40)))
        per_restart = restart_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break

            strength = 0.02 + 0.02 * r
            f_pert = perturb_restart(best_sequence, rng, strength=strength)
            f_deadline = time.monotonic() + per_restart * 0.75

            try:
                bf, bs = optimize_pipeline(f_pert, f_deadline, do_lp=True, do_sm=True, do_cd=True,
                                           verbose_prefix=f"[restart {r}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strength:.3f}): improved to {bs:.6f}")
                else:
                    print(f"  Restart {r} (str={strength:.3f}): no improvement ({bs:.6f})")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # Final polish
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Final polish (n={len(best_sequence)}, score={best_score:.6f}) ===")
        final_deadline = deadline
        try:
            # Coordinate descent for final squeeze
            bf, bs = coord_descent_fast(best_sequence, final_deadline - 3, max_sweeps=200,
                                        verbose_prefix="[final CD] ")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()

            # Quick LP
            if time.monotonic() < final_deadline - 10:
                bf, bs = lp_optimize(best_sequence, final_deadline - 3, max_iters=50,
                                     verbose_prefix="[final LP] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()

            # More CD
            if time.monotonic() < final_deadline - 5:
                bf, bs = coord_descent_fast(best_sequence, final_deadline - 2, max_sweeps=100,
                                            verbose_prefix="[final CD2] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()

            print(f"  Final score: {best_score:.6f}")
        except Exception as e:
            print(f"  Final polish error: {e}")

    # Verify with evaluate_sequence
    if best_sequence is not None:
        best_sequence = np.clip(best_sequence, 0.0, 1000.0)
        S = np.sum(best_sequence)
        if S < 0.01:
            best_sequence = flat_autocorr_init(512)
        final_score = evaluate_sequence(best_sequence.tolist())
        print(f"\n{'='*60}")
        print(f"FINAL: n={len(best_sequence)}, evaluate_score={final_score:.6f}, "
              f"internal_score={best_score:.6f}, time={time.monotonic()-t0:.1f}s")
        print(f"{'='*60}")

    if best_sequence is None:
        best_sequence = flat_autocorr_init(512)
        evaluate_sequence(best_sequence.tolist())

    return [float(x) for x in best_sequence]
