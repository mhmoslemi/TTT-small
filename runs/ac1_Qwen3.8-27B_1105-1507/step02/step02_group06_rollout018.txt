   - Move toward the solution

"""
Convex optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding with adaptive step sizes,
smoothed max (log-sum-exp) with accelerated gradient,
and multi-length search with parent initialization.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = sum_j f[|k-j|... no, convolve(f,g)[k]."""
    n = len(f)
    # convolve(f, g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # For a fixed f, this is linear in g
    # (A @ g)[k] = sum_j A[k,j] * g[j] where A[k,j] = f[k-j] if 0<=k-j<n
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def compute_score(f):
    """Compute the objective value directly."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def solve_lp_direction(f, M, tolerance=1e-8):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + tolerance)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, 
                         method='highs', options={'presolve': True, 'time_limit': 60})
        if result.success or result.status == 1:  # 1 = local optimal for nonlinear, but for LP it should be 0
            if result.x is not None:
                return result.x
        elif result.status == 3:  # Infeasible
            return None
    except Exception:
        pass
    return None


def smoothed_max_gradient(f, tau=1.0):
    """Compute gradient of (1/tau)*log(sum exp(tau*(f*f))) as a smooth approximation of max(f*f)."""
    conv = np.convolve(f, f)
    # For numerical stability, subtract max
    M = np.max(conv)
    shifted = tau * (conv - M)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient of max_k (f*f)[k] is in the convex hull of gradients of active terms
    # For smoothed max, gradient = sum_k weights[k] * grad_k
    n = len(f)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += weights[k] * 2.0 * f[k - j]
    return grad, M + np.log(np.sum(exp_vals)) / tau

def subgradient_optimize(f, deadline, lr_init=0.005, top_k=5, verbose_prefix=""):
    """Subgradient descent with momentum for the max convolution."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        
        # Top-k subgradient
        top_indices = np.argsort(conv)[-top_k:]
        weights = np.maximum(conv[top_indices] - np.mean(conv), 0)
        w_sum = np.sum(weights)
        if w_sum < 1e-15:
            weights = np.ones(top_k)
            w_sum = top_k
        else:
            weights = weights / w_sum
        
        grad = np.zeros(n)
        for k, w in zip(top_indices, weights):
            if w <= 0:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += w * 2.0 * f[k - j]

        if norm_grad < 1e-15:
            break

        new_sum = float(np.sum(new_f))
        if new_sum < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 100:
                break
            it += 1
            continue

        it += 1
        
        if np.isfinite(new_score) and new_score < best_score:

def smoothed_optimize(f, deadline, tau_init=50.0, lr_init=0.001, verbose_prefix=""):
    """Optimize using smoothed max (log-sum-exp) with adaptive tau."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    tau = tau_init
    lr = lr_init
    no_improve = 0
    it = 0
    
    # Momentum for acceleration
    y = f.copy()
    t_k = 1.0
    
    while time.monotonic() < deadline:
        conv_y = np.convolve(y, y)
        M_y = float(np.max(conv_y))
        
        # Compute smoothed gradient
        shifted = tau * (conv_y - M_y)
        exp_vals = np.exp(shifted)
        exp_sum = np.sum(exp_vals)
        weights = exp_vals / exp_sum
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-10:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += weights[k] * 2.0 * f[k - j]
        
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break
        grad = grad / norm_grad
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = f - lr * grad + ((t_k - 1.0) / t_new) * (f - y)
        y = np.clip(y_new, 0.0, None)
        y_sum = float(np.sum(y))
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        
        f_new = np.clip(f - lr * grad, 0.0, None)
        f_sum = float(np.sum(f_new))
        if f_sum > 1e-15:
            f_new = f_new * (S / f_sum)
            f = f_new
        
        t_k = t_new
        new_score = compute_score(f)
        it += 1
        
        # Decay tau (make the smoothing sharper over time)
        tau *= 0.999
        tau = max(tau, 5.0)
        
        if np.isfinite(new_score) and new_score < best_score:
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            if it % 500 == 0:
                print(f"    {verbose_prefix}smoothed iter {it}: score={best_score:.6f} tau={tau:.1f}")
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-6:
                    break

    return best_f, best_score

def build_conv_matrix_sparse(f, row_indices=None):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k] for k in row_indices."""
    n = len(f)
    if row_indices is None:
        row_indices = range(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for k in row_indices:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    if not rows:
        return csr_matrix((2 * n - 1, n))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

            candidate = np.clip(candidate, 0.0, None)
            cand_sum = float(np.sum(candidate))
            if cand_sum < 1e-15:
                alpha *= 0.5
                continue
            candidate = candidate * (S / cand_sum)
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate
                best_alpha = alpha
                break
            alpha *= 0.5
            if alpha < 0.001:
                break
        
        it += 1

        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            # Adapt step size
            if best_alpha == step:
                step = min(step * 1.05, 0.6)
            elif best_alpha > step * 0.5:
                step = min(step * 1.02, 0.5)
            else:
                step = max(step * 0.9, 0.01)
            if it % 5 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f} "
                      f"(step={step:.3f}, lp_time={lp_time:.2f}s, sum_g/S={sum_g/S:.4f})")
        else:
            no_improve += 1
            step *= 0.8
            if step < 0.003 or no_improve > 40:
                break

    return best_f, best_score


def subgradient_optimize(f, deadline, lr_init=0.003, top_k=3, verbose_prefix=""):
    """Subgradient descent with momentum."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        
        # Top-k subgradient
        top_indices = np.argsort(conv)[-top_k:]
        weights = np.maximum(conv[top_indices] - np.mean(conv), 0)
        w_sum = np.sum(weights)
        if w_sum < 1e-15:
            weights = np.ones(top_k) / top_k
        else:
            weights = weights / w_sum
        
        grad = np.zeros(n)
        for k, w in zip(top_indices, weights):
            if w <= 0:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += w * 2.0 * f[k - j]
        
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break
        grad = grad / norm_grad
        
        momentum = 0.8 * momentum - lr * grad
        new_f = f + momentum
        new_f = np.clip(new_f, 0.0, None)
        new_sum = float(np.sum(new_f))
        if new_sum < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 100:
                break
            it += 1
            continue
        new_f = new_f * (S / new_sum)
        
        new_score = compute_score(new_f)
        it += 1
        
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.98
            momentum *= 0.5
            if lr < 1e-9 or no_improve > 500:
                break

    return best_f, best_score


def smoothed_optimize(f, deadline, tau_init=40.0, lr_init=0.001, verbose_prefix=""):
    """Optimize using smoothed max with FISTA acceleration."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    tau = tau_init
    lr = lr_init
    no_improve = 0
    it = 0
    y = f.copy()
    t_k = 1.0
    
    while time.monotonic() < deadline:
        conv_y = np.convolve(y, y)
        M_y = float(np.max(conv_y))
        
        # Compute smoothed gradient
        shifted = tau * (conv_y - M_y)
        # Clip for numerical stability
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        exp_sum = np.sum(exp_vals)
        if exp_sum < 1e-300:
            break
        weights = exp_vals / exp_sum
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-8:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += weights[k] * 2.0 * f[k - j]
        
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break
        grad = grad / norm_grad
        
        # FISTA
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = f - lr * grad + ((t_k - 1.0) / t_new) * (f - y)
        y = np.clip(y_new, 0.0, None)
        y_sum = float(np.sum(y))
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        
        f_new = np.clip(f - lr * grad, 0.0, None)
        f_sum = float(np.sum(f_new))
        if f_sum > 1e-15:
            f_new = f_new * (S / f_sum)
        f = f_new
        t_k = t_new
        
        # Decay tau gradually
        tau = max(tau * 0.9995, 3.0)
        
        new_score = compute_score(f)
        it += 1
        
        if np.isfinite(new_score) and new_score < best_score:
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                lr *= 0.8
                no_improve = 0
                t_k = 1.0
                y = f.copy()
            if lr < 1e-7:
                break

    return best_f, best_score


def coordinate_descent(f, deadline, verbose_prefix=""):
    """Paired coordinate descent: update pairs (i, j) to reduce max conv."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    it = 0
    no_improve = 0
    
    rng = np.random.default_rng(77)
    
    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        
        # Pick the worst lag
        k_worst = int(np.argmax(conv))
        
        # Find a pair (i, j) with i != j that affects lag k_worst
        # f[i] and f[k_worst - i] both contribute to conv[k_worst]
        candidates = []
        for i in range(min(k_worst + 1, n)):
            j = k_worst - i
            if 0 <= j < n and j != i:
                candidates.append((i, j))
        
        if not candidates:
            break
        
        # Try a few random pairs
        best_pair_score = best_score
        best_pair_f = None
        best_pair = None
        
        num_tries = min(10, len(candidates))
        tried = set()
        for _ in range(num_tries):
            idx = rng.integers(0, len(candidates))
            i, j = candidates[idx]
            if (i, j) in tried or (j, i) in tried:
                continue
            tried.add((i, j))
            tried.add((j, i))
            
            # Current contributions
            fi = f[i]
            fj = f[j]
            other = conv[k_worst] - 2.0 * fi * fj  # approximate other contribution
            
            # Try transferring mass from higher to lower
            delta = 0.01 * S / n
            for sign in [1, -1]:
                fi_new = fi + sign * delta
                fj_new = fj - sign * delta
                if fi_new < 0 or fj_new < 0:
                    continue
                # Estimate new max conv
                new_conv_k = other + 2.0 * fi_new * fj_new
                # Check if this helps
                if new_conv_k < conv[k_worst] - 1e-10:
                    f_test = f.copy()
                    f_test[i] = fi_new
                    f_test[j] = fj_new
                    f_test = f_test * (S / float(np.sum(f_test)))
                    s_test = compute_score(f_test)
                    if s_test < best_pair_score:
                        best_pair_score = s_test
                        best_pair_f = f_test.copy()
                        best_pair = (i, j, sign, delta)
        
        it += 1
        if best_pair_f is not None:
            f = best_pair_f
            best_f = f.copy()
            best_score = best_pair_score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry: multi-method optimization for the step function problem."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = compute_score(parent)
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")
    
    # Determine lengths to try
    if parent is not None:
        n_parent = len(parent)
        # Try parent length and one bigger
        candidates_n = [n_parent]
        if n_parent * 2 <= 4096:
            candidates_n.append(n_parent * 2)
        if n_parent >= 2048:
            candidates_n.append(n_parent // 2)
    else:
        candidates_n = [1024, 2048]
    
    candidates_n = sorted(set(candidates_n))
    print(f"Lengths to try: {candidates_n}")
    print(f"Initial best: {best_score:.6f}")
    
    # Time allocation
    final_polish_time = 150.0
    main_deadline = deadline - final_polish_time
    time_per_n = (main_deadline - t0) / len(candidates_n)
    
    results = {}  # n -> (score, sequence)
    
    for n in candidates_n:
        if time.monotonic() > main_deadline - 30:
            break
        
        print(f"\n{'='*60}")
        print(f"Optimizing n={n}")
        print(f"{'='*60}")
        
        # Initialize
        if parent is not None and len(parent) == n:
            f_init = parent.copy()
        elif parent is not None:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            f_init = np.interp(x_new, x_old, parent)
            f_init = np.clip(f_init, 0.0, None)
        else:
            f_init = np.ones(n)
        
        S_init = float(np.sum(f_init))
        if S_init < 1e-10:
            f_init = np.ones(n)
            S_init = float(n)
        f_init = f_init * (S_init / S_init)  # already normalized
        
        init_score = compute_score(f_init)
        print(f"  Initial score: {init_score:.6f}")
        
        # Phase A: LP with active set (main workhorse)
        n_budget = time_per_n
        alloc_A = n_budget * 0.5
        d_A = time.monotonic() + alloc_A
        print(f"  Phase A: LP active set ({alloc_A:.0f}s)")
        f_A, s_A = lp_optimize(f_init, d_A, step_init=0.12,
                                verbose_prefix=f"[n={n} A] ", use_active_set=True)
        if s_A < best_score:
            best_score = s_A
            best_sequence = f_A.copy()
            print(f"  Phase A result: {s_A:.6f}")
        
        # Phase B: Continue LP from Phase A result with full constraints
        alloc_B = n_budget * 0.25
        d_B = time.monotonic() + alloc_B
        if time.monotonic() < main_deadline - 30:
            print(f"  Phase B: LP full ({alloc_B:.0f}s)")
            f_B, s_B = lp_optimize(f_A, d_B, step_init=0.06,
                                    verbose_prefix=f"[n={n} B] ", use_active_set=False)
            if s_B < s_A:
                f_A = f_B
                s_A = s_B
            if s_A < best_score:
                best_score = s_A
                best_sequence = f_A.copy()
                print(f"  Phase B result: {s_A:.6f}")
        
        # Phase C: Smoothed max / FISTA
        alloc_C = n_budget * 0.15
        d_C = time.monotonic() + alloc_C
        if time.monotonic() < main_deadline - 30:
            print(f"  Phase C: Smoothed FISTA ({alloc_C:.0f}s)")
            f_C, s_C = smoothed_optimize(f_A, d_C, tau_init=30.0,
                                          lr_init=0.001, verbose_prefix=f"[n={n} C] ")
            if s_C < s_A:
                f_A = f_C
                s_A = s_C
            if s_A < best_score:
                best_score = s_A
                best_sequence = f_A.copy()
                print(f"  Phase C result: {s_A:.6f}")
        
        # Phase D: Subgradient polish
        alloc_D = n_budget * 0.10
        d_D = time.monotonic() + alloc_D
        if time.monotonic() < main_deadline - 30:
            print(f"  Phase D: Subgradient ({alloc_D:.0f}s)")
            f_D, s_D = subgradient_optimize(f_A, d_D, lr_init=0.002, top_k=5,
                                             verbose_prefix=f"[n={n} D] ")
            if s_D < s_A:
                f_A = f_D
                s_A = s_D
            if s_A < best_score:
                best_score = s_A
                best_sequence = f_A.copy()
                print(f"  Phase D result: {s_A:.6f}")
        
        # Phase E: Restart with perturbation + LP
        alloc_E = n_budget * 0.10
        d_E = time.monotonic() + alloc_E
        if time.monotonic() < main_deadline - 30:
            print(f"  Phase E: Restart + LP ({alloc_E:.0f}s)")
            rng = np.random.default_rng(n + 1)
            f_restart = f_A + rng.normal(0, 0.005 * S_init / n, size=n)
            f_restart = np.clip(f_restart, 0.0, None)
            f_restart = f_restart * (S_init / float(np.sum(f_restart)))
            f_E, s_E = lp_optimize(f_restart, d_E, step_init=0.08,
                                    verbose_prefix=f"[n={n} E] ", use_active_set=True)
            if s_E < s_A:
                f_A = f_E
                s_A = s_E
            if s_A < best_score:
                best_score = s_A
                best_sequence = f_A.copy()
                print(f"  Phase E result: {s_A:.6f}")
        
        results[n] = (s_A, f_A.copy())
        print(f"  Best for n={n}: {s_A:.6f}")
    
    # Final polish on the overall best
    print(f"\n{'='*60}")
    print(f"Final Polish: best_score={best_score:.6f}")
    print(f"{'='*60}")
    
    if best_sequence is not None and time.monotonic() < deadline - 5:
        # LP polish with very small steps
        f_pol, s_pol = lp_optimize(best_sequence, deadline, step_init=0.02,
                                    verbose_prefix="[final-LP] ", use_active_set=True)
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()
            print(f"  After final LP: {best_score:.6f}")
        
        # Quick subgradient
        if time.monotonic() < deadline - 5:
            d_sub = min(deadline, time.monotonic() + 20)
            f_pol, s_pol = subgradient_optimize(best_sequence, d_sub, lr_init=0.0005,
                                                 top_k=7, verbose_prefix="[final-sub] ")
            if s_pol < best_score:
                best_score = s_pol
                best_sequence = f_pol.copy()
                print(f"  After final sub: {best_score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(1024)
        best_score = compute_score(best_sequence)
    
    # Final verification with official evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_sequence]
    final_score = evaluate_sequence(final_list)
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed evaluation! Using fallback.")
        best_sequence = np.ones(512)
        final_list = [1.0] * 512
        final_score = evaluate_sequence(final_list)
    
    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, time={elapsed:.1f}s")
    
    return final_list
