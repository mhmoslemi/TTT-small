```
def correlate_w_a(w, a):
    n = len(a)
    if n == 1:
        return np.array([w[0]*a[0]])
    # full conv of a and w[::-1]
    size = 1
    target = n + (2*n -1) -1 = 3n-2
    while size < target: size <<=1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w[::-1], size)
    conv = np.fft.irfft(fa*fw, size)
    start = 2*n-2
    stop = n-2
    return conv[start:stop:-1]
```

```
def resample_seq(src, dest_n):
    src=np.asarray(src,float)
    if src.size==dest_n: return src.copy()
    if src.size==1: return np.full(dest_n, src[0])
    x_old=np.linspace(0,1,src.size)
    x_new=np.linspace(0,1,dest_n)
    y=np.interp(x_new, x_old, src)
    return np.maximum(y,0.0)
```

```python
"""
Minimize J(a) = 2*n*max(conv(a,a))/(sum(a))^2 for nonnegative sequences.

Strategy:
- Work on the simplex via softmax parameters z, a = softmax(z).
- Use a smooth p-norm of the convolution as a differentiable surrogate for
  the maximum good-convolution coefficient:
      F_p(a) = (sum_k c_k^p)^(1/p),  c = conv(a,a), sum(a)=1.
  The scaled objective is J = 2n * F_p(a), so we minimize log(F_p(a)).
- Optimize z with L-BFGS-B using analytic gradients of log(F_p).
- Use many diverse initializations, p-annealing, random perturbations,
  exact active-set / soft-active-set multiplicative subgradient polishing,
  and an LP best-response refinement.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

CURRENT_P = 4.0


def set_current_p(p):
    global CURRENT_P
    CURRENT_P = float(p)


def pair_first(item):
    return item[0]


def logsumexp(v):
    v = np.asarray(v, dtype=float)
    if v.size == 0:
        return -np.inf
    m = float(np.max(v))
    if not np.isfinite(m):
        return m
    if m == -np.inf:
        return -np.inf
    sh = v - m
    s = float(np.sum(np.exp(sh)))
    if s <= 0.0 or not np.isfinite(s):
        return -np.inf
    return m + float(np.log(s))


def fast_conv(a):
    a = np.asarray(a, dtype=float).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=float)
    L = 2 * n - 1
    size = 1
    while size < L:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:L]
    c = np.maximum(c, 0.0)
    return c


def correlate_w_a(w, a):
    """
    Compute inner_i = sum_{l=0}^{n-1} a_l * w_{i+l}
    for i=0..n-1, using FFT.
    """
    a = np.asarray(a, dtype=float).ravel()
    w = np.asarray(w, dtype=float).ravel()
    n = a.size
    if n == 1:
        return np.array([w[0] * a[0]], dtype=float)

    L = 3 * n - 2
    size = 1
    while size < L:
        size <<= 1

    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w[::-1], size)
    conv = np.fft.irfft(fa * fw, size)
    conv = np.maximum(conv, 0.0)

    start = 2 * n - 2
    stop = n - 2
    out = conv[start:stop:-1]
    out = np.ascontiguousarray(out)
    return out


def fast_J(a):
    a = np.asarray(a, dtype=float).ravel()
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        return float("inf")
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    c = fast_conv(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * np.max(c) / (s * s))


def softmax(z):
    z = np.asarray(z, dtype=float).ravel()
    if z.size == 0:
        return np.array([], dtype=float)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(z.size, 1.0 / z.size, dtype=float)
    return e / s


def a_to_z(a):
    a = np.asarray(a, dtype=float).ravel()
    a = np.maximum(a, 1e-12)
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return np.zeros(a.size, dtype=float)
    a = a / s
    z = np.log(a)
    z = z - float(np.mean(z))
    return z


def pnorm_obj(x):
    z = np.asarray(x, dtype=float).ravel()
    if z.size == 0 or not np.all(np.isfinite(z)):
        return 1e100
    z = z - float(np.mean(z))
    a = softmax(z)
    c = fast_conv(a)
    if c.size == 0:
        return 1e100

    p = float(CURRENT_P)
    pos = c > 1e-300
    if not np.any(pos):
        return 1e100

    logc = np.full(c.shape, -np.inf, dtype=float)
    logc[pos] = np.log(c[pos])
    logF = (1.0 / p) * logsumexp(p * logc)
    if not np.isfinite(logF):
        return 1e100
    return float(logF)


def pnorm_grad(x):
    z = np.asarray(x, dtype=float).ravel()
    n = z.size
    if n == 0 or not np.all(np.isfinite(z)):
        return np.zeros(n, dtype=float)

    z = z - float(np.mean(z))
    a = softmax(z)
    c = fast_conv(a)
    if c.size == 0:
        return np.zeros(n, dtype=float)

    p = float(CURRENT_P)
    pos = c > 1e-300
    if not np.any(pos):
        return np.zeros(n, dtype=float)

    logc = np.full(c.shape, -np.inf, dtype=float)
    logc[pos] = np.log(c[pos])
    logF = (1.0 / p) * logsumexp(p * logc)
    if not np.isfinite(logF):
        return np.zeros(n, dtype=float)

    # W_k = 2 * c_k^(p-1) / F^p
    logW = np.log(2.0) + (p - 1.0) * logc - p * logF
    W = np.zeros(c.shape, dtype=float)
    valid = logW > -745.0
    W[valid] = np.exp(np.minimum(logW[valid], 700.0))

    inner = correlate_w_a(W, a)
    if not np.all(np.isfinite(inner)):
        return np.zeros(n, dtype=float)

    eta = float(np.dot(a, inner))
    grad = a * (inner - eta)
    grad = grad - float(np.mean(grad))

    if not np.all(np.isfinite(grad)):
        return np.zeros(n, dtype=float)
    return grad


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float("inf")
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.asarray(a, dtype=float).ravel()
        if a.size == 0:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(np.sum(a))
        if not np.isfinite(s) or s < 0.01:
            return

        if self.best is not None:
            fj = fast_J(a)
            if not np.isfinite(fj) or fj + 1e-8 >= self.best_score:
                return

        try:
            score = evaluate_sequence(a.tolist())
        except Exception:
            return

        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = float(score)
            print(
                f"  * new best J={self.best_score:.10f} n={a.size} "
                f"t={time.monotonic() - self.t0:.1f}s",
                flush=True,
            )

    def report(self, label="best"):
        if self.best is None:
            print(f"  {label}: none", flush=True)
        else:
            print(
                f"  {label}: J={self.best_score:.10f} n={self.best.size} "
                f"t={time.monotonic() - self.t0:.1f}s",
                flush=True,
            )


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=float).ravel()
    if src.size == 0:
        return np.ones(dest_n, dtype=float)
    if src.size == dest_n:
        return src.copy()
    if src.size == 1:
        return np.full(dest_n, float(src[0]), dtype=float)
    x_old = np.linspace(0.0, 1.0, src.size)
    x_new = np.linspace(0.0, 1.0, int(dest_n))
    y = np.interp(x_new, x_old, src)
    return np.maximum(y, 0.0)


def normalize_nonnegative(a, floor=1e-12):
    a = np.asarray(a, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, floor)
    s = float(np.sum(a))
    if s <= 0.0:
        n = a.size
        return np.full(n, 1.0 / n if n else 0.0, dtype=float)
    return a / s


def generate_inits(n, rng, parent, max_count=20):
    n = int(n)
    if n <= 0:
        return []

    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0.0, 1.0, n)

    inits.append(np.ones(n))
    inits.append(0.05 + 0.95 * (t * t + (1.0 - t) * (1.0 - t)))
    inits.append(0.05 + 0.95 * np.abs(2.0 * t - 1.0))
    inits.append(0.05 + 0.95 * np.exp(-3.0 * (t - 0.5) ** 2))
    inits.append(0.05 + 0.95 * np.exp(-4.0 * (t - 0.35) ** 2))
    inits.append(0.05 + 0.95 * np.exp(-4.0 * (t - 0.65) ** 2))
    inits.append(0.05 + 0.55 * np.exp(-8.0 * (t - 0.25) ** 2) +
                  0.75 * np.exp(-8.0 * (t - 0.80) ** 2))
    inits.append(np.linspace(0.1, 1.0, n))
    inits.append(np.linspace(1.0, 0.1, n))

    for freq, phase, mult in (
        (1.0, 0.0, 0.5),
        (2.0, 0.5 * np.pi, 0.5),
        (3.0, 0.0, 0.4),
        (4.0, np.pi, 0.35),
        (5.0, 1.0, 0.3),
        (7.0, 2.0, 0.25),
    ):
        inits.append(
            np.maximum(1.0 + mult * np.cos(2.0 * np.pi * freq * t + phase), 0.01)
        )

    for _ in range(4):
        freq = float(rng.integers(1, 13))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        mult = float(rng.uniform(0.25, 0.75))
        inits.append(
            np.maximum(1.0 + mult * np.cos(2.0 * np.pi * freq * t + phase), 0.01)
        )

    beta = float(rng.uniform(-6.0, 6.0))
    inits.append(np.exp(beta * (t - 0.5)))

    k = max(3, n // 32)
    kernel = np.ones(k) / float(k)

    x = rng.uniform(0.0, 1.0, n)
    inits.append(np.convolve(x, kernel, mode="same") + 0.05)

    m = max(1, min(n, n // 16))
    sparse = np.zeros(n)
    if 1 <= m <= n:
        pos = rng.choice(n, size=m, replace=False)
        sparse[pos] = 1.0
    inits.append(np.convolve(sparse, kernel, mode="same") + 0.02)

    seg = int(rng.integers(3, 6))
    seg_len = max(1, n // seg)
    heights = rng.uniform(0.05, 1.0, seg)
    block = np.repeat(heights, seg_len)
    if block.size < n:
        block = np.concatenate([block, np.full(n - block.size, heights[-1])])
    else:
        block = block[:n]
    inits.append(block + 0.05)

    out = []
    for arr in inits:
        arr = np.asarray(arr, dtype=float).ravel()
        if arr.size != n:
            continue
        arr = np.maximum(arr, 1e-6)
        s = float(np.sum(arr))
        if not np.isfinite(s) or s <= 0.0:
            continue
        arr = arr / s
        out.append(arr)
        if len(out) >= int(max_count):
            break

    if not out:
        out.append(np.ones(n) / float(n))
    return out


def estimate_eval_time(n, p):
    set_current_p(p)
    z = np.zeros(int(n), dtype=float)
    if z.size == 0:
        return 1e-3
    t0 = time.monotonic()
    pnorm_obj(z)
    pnorm_grad(z)
    pnorm_obj(z)
    pnorm_grad(z)
    dt = (time.monotonic() - t0) / 2.0
    return max(float(dt), 1e-5)


def run_pnorm_schedule(z0, p_list, deadline, tracker, maxiter, maxfun):
    z = np.asarray(z0, dtype=float).ravel().copy()
    if z.size == 0:
        return None
    z = z - float(np.mean(z))
    a = softmax(z)
    tracker.update(a)

    for p in p_list:
        if time.monotonic() >= deadline:
            break

        set_current_p(float(p))
        base = pnorm_obj(z)
        options = {
            "maxiter": int(maxiter),
            "maxfun": int(maxfun),
            "ftol": 1e-18,
            "gtol": 1e-10,
        }
        try:
            res = minimize(
                pnorm_obj,
                z,
                jac=pnorm_grad,
                method="L-BFGS-B",
                options=options,
            )
        except Exception:
            break

        if res is not None and hasattr(res, "x") and res.x is not None:
            cand = np.asarray(res.x, dtype=float).ravel()
            cand_ok = np.all(np.isfinite(cand))
            fun_ok = np.isfinite(res.fun)
            if cand_ok and fun_ok and res.fun < base + 1e-7:
                z = cand
                z = z - float(np.mean(z))

        a = softmax(z)
        tracker.update(a)

    return z


def soft_max_weights(c, alpha_mult):
    c = np.asarray(c, dtype=float).ravel()
    if c.size == 0:
        return np.array([], dtype=float)
    m = float(np.max(c))
    if m <= 0.0:
        return np.ones(c.size) / float(c.size)

    if alpha_mult <= 1e-12:
        w = np.zeros_like(c)
        active = c >= (m - 1e-10 * m)
        if not np.any(active):
            active[np.argmax(c)] = True
        w[active] = 1.0
        s = float(np.sum(w))
        if s > 0.0:
            return w / s
        return np.ones(c.size) / float(c.size)

    alpha = float(alpha_mult) * m
    vals = (c - m) / alpha
    vals = np.clip(vals, -50.0, 0.0)
    w = np.exp(vals)
    s = float(np.sum(w))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return w / s


def polish_subgradient(a, deadline, tracker, max_steps=800):
    a = normalize_nonnegative(a)
    if a.size == 0:
        return a

    current = fast_J(a)
    tracker.update(a)

    etas = (1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.2)
    alpha_mults = (1e-4, 1e-3, 1e-2, 5e-2)

    improved_total = 0
    for _ in range(int(max_steps)):
        if time.monotonic() >= deadline:
            break

        c = fast_conv(a)
        if c.size == 0:
            break
        m = float(np.max(c))
        if m <= 0.0:
            break

        best_a = None
        best_J = current

        for am in alpha_mults:
            w = soft_max_weights(c, am)
            G = 2.0 * correlate_w_a(w, a)
            if not np.all(np.isfinite(G)):
                continue

            for eta in etas:
                ax = a * np.exp(-np.clip(eta * G, -700.0, 700.0))
                s = float(np.sum(ax))
                if s <= 0.0 or not np.isfinite(s):
                    continue
                ax = ax / s
                J = fast_J(ax)
                if np.isfinite(J) and J < best_J - 1e-13:
                    best_J = J
                    best_a = ax

        if best_a is None:
            break

        a = best_a
        current = best_J
        improved_total += 1
        if improved_total % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def lp_best_response(a):
    a = np.asarray(a, dtype=float).ravel()
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        return None
    s = float(np.sum(a))
    if s <= 0.0:
        return None
    a = a / s

    m = 2 * n - 1
    rows = []
    cols = []
    vals = []

    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        cnt = jh - jl + 1
        if cnt > 0:
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)

    A = csr_matrix((vals, (rows, cols)), shape=(m, n + 1))

    cobj = np.zeros(n + 1)
    cobj[-1] = 1.0

    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    beq = np.ones(1)

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(
            cobj,
            A_ub=A,
            b_ub=np.zeros(m),
            A_eq=Aeq,
            b_eq=beq,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        return None

    if res is None or not res.success or res.x is None:
        return None

    y = np.maximum(res.x[:n], 0.0)
    y = np.where(np.isfinite(y), y, 0.0)
    s = float(np.sum(y))
    if s <= 0.0:
        return None
    return y / s


def lp_refine(a, max_iters, deadline, tracker):
    a = normalize_nonnegative(a)
    if a.size == 0:
        return a

    current = fast_J(a)
    tracker.update(a)

    for _ in range(int(max_iters)):
        if time.monotonic() >= deadline:
            break

        y = lp_best_response(a)
        if y is None or y.size != a.size:
            break

        best_a = None
        best_J = current

        for lam in (0.1, 0.25, 0.5, 0.75):
            ax = (1.0 - lam) * a + lam * y
            ax = normalize_nonnegative(ax)
            J = fast_J(ax)
            if np.isfinite(J) and J < best_J - 1e-12:
                best_J = J
                best_a = ax

        if best_a is None:
            break

        a = best_a
        current = best_J
        tracker.update(a)

    return a


def optimize_length(n, rng, parent, phase_deadline, tracker):
    n = int(n)
    if n < 2:
        return

    inits = generate_inits(
        n,
        rng,
        parent,
        max_count=14 if n <= 128 else 10 if n <= 256 else 8,
    )

    scored = []
    for a in inits:
        fj = fast_J(a)
        if np.isfinite(fj):
            scored.append((fj, a))

    if not scored:
        return

    scored.sort(key=pair_first)

    try:
        est = estimate_eval_time(n, 8.0)
    except Exception:
        est = 1e-4

    p_list = (4.0, 8.0, 16.0, 32.0)

    for idx, (_, a) in enumerate(scored):
        if time.monotonic() >= phase_deadline:
            break

        remaining = max(1, len(scored) - idx)
        time_left = max(1e-3, float(phase_deadline) - time.monotonic())
        time_per = max(8.0, time_left / float(remaining) * 0.95)

        if idx >= min(4, len(scored)):
            p_use = p_list[1:]
        else:
            p_use = p_list

        total_fun = int(time_per / max(est, 1e-5) / max(1, len(p_use)))
        if n <= 128:
            total_fun = min(total_fun, 16000)
        elif n <= 256:
            total_fun = min(total_fun, 10000)
        else:
            total_fun = min(total_fun, 5000)

        per_fun = max(200, total_fun // max(1, len(p_use)))
        maxiter = max(120, per_fun // 3)

        z = a_to_z(a)
        run_pnorm_schedule(z, p_use, phase_deadline, tracker, maxiter, per_fun)


def refine_best(tracker, deadline, rng, parent):
    if tracker.best is None:
        return

    a = tracker.best.copy()
    n = int(a.size)
    if n <= 0:
        return

    a = normalize_nonnegative(a)

    # High-precision p-annealing from current best.
    r1_deadline = min(float(deadline), time.monotonic() + 450.0)
    r1_deadline = min(r1_deadline, float(deadline) - 320.0)
    if r1_deadline > time.monotonic() + 10.0:
        try:
            est = estimate_eval_time(n, 32.0)
        except Exception:
            est = 1e-4
        time_left = max(1.0, r1_deadline - time.monotonic())
        total_fun = int(time_left / max(est, 1e-5) / 4.0)
        if n <= 128:
            total_fun = min(total_fun, 24000)
        elif n <= 256:
            total_fun = min(total_fun, 14000)
        else:
            total_fun = min(total_fun, 7000)
        per_fun = max(300, total_fun // 4)
        maxiter = max(200, per_fun // 3)

        z = a_to_z(a)
        run_pnorm_schedule(z, (8.0, 16.0, 32.0, 64.0), r1_deadline, tracker, maxiter, per_fun)

        if tracker.best is not None:
            a = tracker.best.copy()

    # Random perturbation restarts in z-space.
    r2_deadline = min(float(deadline), float(deadline) - 220.0)
    r2_deadline = min(r2_deadline, r2_deadline if r2_deadline > time.monotonic() + 20 else
                      float(deadline) - 150.0)

    if r2_deadline > time.monotonic() + 20.0:
        for r in range(30):
            if time.monotonic() >= r2_deadline:
                break

            scale = 0.25 + 0.05 * float(r)
            scale = min(scale, 1.2)

            z = a_to_z(a)
            z = z + rng.normal(0.0, scale, n)
            z = run_pnorm_schedule(z, (32.0, 64.0), r2_deadline, tracker, 350, 2500)

            if z is not None:
                cand = softmax(z - float(np.mean(z)))
                tracker.update(cand)

            if tracker.best is not None:
                a = tracker.best.copy()

    # Subgradient polish.
    r3_deadline = float(deadline) - 90.0
    if r3_deadline > time.monotonic() + 5.0:
        a = polish_subgradient(a, r3_deadline, tracker, max_steps=700)
        if tracker.best is not None:
            a = tracker.best.copy()

    # Optional LP best-response polish for moderate sizes.
    if a.size <= 256:
        r4_deadline = float(deadline) - 40.0
        if r4_deadline > time.monotonic() + 5.0:
            a = lp_refine(a, 80, r4_deadline, tracker)
            if tracker.best is not None:
                a = tracker.best.copy()

    # Final very sharp L-BFGS pass.
    r5_deadline = float(deadline) - 5.0
    if r5_deadline > time.monotonic() + 5.0:
        z = a_to_z(a)
        run_pnorm_schedule(z, (64.0,), r5_deadline, tracker, 500, 3000)

    # Try thresholding tiny values to zero.
    if tracker.best is not None:
        b = tracker.best.copy()
        b = np.where(b < 1e-10, 0.0, b)
        s = float(np.sum(b))
        if s > 0.0:
            b = b / s
            tracker.update(b)


def propose_candidate(seed=42, budget_s=1500):
    start = time.monotonic()
    deadline = start + max(120.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed))
    tracker = ScoreTracker()

    parent = None
    try:
        hs = height_sequence_1
        arr = np.asarray(hs, dtype=float).ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            parent = np.clip(arr, 0.0, 1000.0)
            if float(np.sum(parent)) > 0.0:
                tracker.update(parent)
                print(
                    f"Parent loaded: n={parent.size}, J={tracker.best_score:.10f}",
                    flush=True,
                )
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        parent = np.ones(128, dtype=float)

    lengths = []
    for n in (64, int(parent.size), 128, 256, 384, 512):
        if 16 <= int(n) <= 512 and int(n) not in lengths:
            lengths.append(int(n))
    lengths.sort()

    print(f"Searching lengths: {lengths}", flush=True)

    for idx, n in enumerate(lengths):
        if time.monotonic() >= deadline:
            break

        remaining_lengths = max(1, len(lengths) - idx)
        phase_time = max(90.0, (deadline - time.monotonic()) / float(remaining_lengths) * 0.93)
        phase_deadline = min(float(deadline), time.monotonic() + phase_time)

        print(
            f"Phase length n={n}, phase_deadline in "
            f"{phase_deadline - time.monotonic():.1f}s",
            flush=True,
        )
        optimize_length(n, rng, parent, phase_deadline, tracker)
        tracker.report(f"After length {n}")

    if tracker.best is not None:
        refine_best(tracker, deadline, rng, parent)
    else:
        tracker.update(np.ones(128, dtype=float))

    tracker.report("Final")

    if tracker.best is None:
        out = np.ones(128, dtype=float)
    else:
        out = tracker.best.copy()

    return [float(x) for x in out]
```
