def project_simplex(v):
    """Project v onto probability simplex {x>=0, sum(x)=1} via water-filling."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    return weights[idx]

def coordinate_descent(f0, time_budget, rng=None):
    """Coordinate descent with geometric trial values."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f0.copy().astype(np.float64)
    f = np.clip(f, 0, None)
    f = normalize(f)

    start_time = time.monotonic()
    
    factors = [0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.5, 2.0, 3.0]
    
    while time.monotonic() - start_time < time_budget:
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start_time >= time_budget:
                break
            original = f[i]
            if original < 1e-12:
                continue
            best_val = original
            best_local = best_score
            
            prev_conv = np.convolve(f, f)
            prev_max = prev_conv.max()
            
            for factor in factors:
                f[i] = max(0.0, original * factor)
                # Quick evaluation: only need to check affected convolution entries
                # For efficiency, just compute full conv (fast for n=512)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                conv_max = np.convolve(f, f).max()
                score = 2.0 * n * conv_max / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        
        f = np.clip(f, 0, None)
        f = normalize(f)
    
    return f, best_score

            # Single element
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] * np.exp(rng.normal(0, 0.3)))
        elif r < 0.65:
            # Smooth perturbation
            center = int(rng.integers(0, n))
            width = max(2, int(n // 32))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.normal(0, 0.1) * kernel
        elif r < 0.8:
            # Two-element swap-like
            i, j = rng.integers(0, n, 2)
            if i != j:
                swap = f_new[i]
                f_new[i] = f_new[j]
                f_new[j] = swap
        else:
            # Global reweight
            f_new *= np.exp(rng.normal(0, 0.3, n))
        
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)
        
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        it += 1
        if no_improve > 3000 and it > total_iters // 3:
            break
    
    return best_f, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    r = rng.random()
    if r < 0.2:
        # Uniform with noise
        f = np.ones(n) * (1 + 0.3 * rng.normal(0, 1, n))
    elif r < 0.4:
        # Gaussian bump off-center
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.6:
        # Cosine window
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        f = f ** rng.uniform(0.5, 3.0)
    elif r < 0.8:
        # Two-sided (mass at edges)
        f = np.ones(n)
        decay = np.arange(n)
        center = n / 2
        f = np.exp(-0.3 * np.abs(np.arange(n) - center) / center)
        # Add edge boost
        f[:n//4] += 0.5
        f[3*n//4:] += 0.5
    else:
        # Product structure
        f = np.ones(n)
        for _ in range(3):
            center = int(rng.integers(0, n))
            width = int(rng.integers(5, n // 3))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f *= (0.5 + kernel)
    
    f = np.clip(f, 0.01, None)
    f = f * (1 + 0.1 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def multi_scale_init(n_target, rng):
    """Create initial sequence by optimizing small then upsample."""
    n_small = max(16, n_target // 32)
    f_small = np.ones(n_small) * (1 + 0.5 * rng.random(n_small))
    f_small = normalize(f_small)
    
    # Quick pressure-based optimization on small
    for _ in range(100):
        conv = np.convolve(f_small, f_small)
        j_star = np.argmax(conv)
        pressure = np.zeros(n_small)
        for i in range(n_small):
            idx = j_star - i
            if 0 <= idx < n_small:
                pressure[i] = f_small[idx]
        f_small = f_small * np.exp(-0.3 * pressure / (pressure.max() + 1e-10))
        f_small = np.clip(f_small, 1e-6, None)
        f_small = normalize(f_small)
    
    # Upsample
    x_small = np.linspace(0, 1, n_small)
    x_large = np.linspace(0, 1, n_target)
    f_large = np.interp(x_large, x_small, f_small)
    f_large = np.clip(f_large, 0.01, None)
    f_large = normalize(f_large)
    return f_large


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-strategy search: mirror descent + LP refinement + coordinate descent
    + simulated annealing across multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0
    
    best_overall = None
    best_score_overall = np.inf
    
    # Try parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None
    
    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            parent = normalize(parent)
            parent_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_overall = parent.copy()
            best_score_overall = parent_score
    
    # Lengths to try (priority order)
    lengths = [512, 256, 1024, 768, 2048]
    if parent is not None and len(parent) not in lengths:
        lengths.insert(0, len(parent))
    
    print(f"Lengths to try: {lengths}")
    print(f"Total budget: {budget_s}s, target: 1.5027")
    
    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        print(f"\n{'='*50}")
        print(f"n={n}, remaining={remaining:.0f}s")
        
        # Determine time allocation
        t_md = min(80, remaining * 0.3)
        t_lp = min(60, remaining * 0.25)
        t_cd = min(30, remaining * 0.1)
        t_sa = min(30, remaining * 0.1)
        # Scale down if needed
        total_alloc = t_md + t_lp + t_cd + t_sa
        if total_alloc > remaining * 0.8:
            scale = remaining * 0.8 / total_alloc
            t_md *= scale
            t_lp *= scale
            t_cd *= scale
            t_sa *= scale
        
        # Try up to 2 different initializations per length
        n_inits = 2 if remaining > 200 else 1

        for init_idx in range(n_inits):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            
            # Choose initialization
            if init_idx == 0 and parent is not None and len(parent) == n:
                f0 = parent * (1 + 0.1 * rng.normal(0, 1, n))

                    best_score_overall = s_md

            # Phase 2: LP refinement
