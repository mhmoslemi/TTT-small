"""
Coordinate descent + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact 1D subproblem solving via golden section search for each coordinate,
combined with LP-based global improvements and multiple diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time

def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def coord_step(a, i, S, C):
    """Optimize coordinate i using precomputed sum S and convolution C.
    Returns new value of a[i] and new objective."""
    n = len(a)
    t_old = a[i]
    
    # C_base[k] = C[k] minus a[i]'s contribution
    # a[i] contributes to C[k] as:
    #   2*t_old*a[k-i] if 0<=k-i<n and k-i != i
    #   t_old^2 if k-i == i (k=2i)
    
    # Build the "linear coefficients" for the 1D subproblem
    # C[k](t) = C_base[k] + 2*t*a[k-i] (for k-i valid and k!=2i)
    #          = C_base[k] + t^2 (for k=2i)
    
    # Precompute arrays for fast evaluation
    # For each k, store: (base, linear_coeff, is_quadratic)
    # C[k](t) = base + linear_coeff * t + (t^2 if is_quadratic)
    
    base = C.copy()  # Start with current C
    lin_coeff = np.zeros(2*n-1)
    quad = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        j = k - i
        if 0 <= j < n:
            if j == i:
                base[k] -= t_old * t_old
                quad[k] = 1.0
            else:
                base[k] -= 2 * t_old * a[j]
                lin_coeff[k] = 2.0 * a[j]
    
    def obj_1d(t):
        S_new = S - t_old + t
        if S_new < 0.01:
            return np.inf
        # Compute max of C[k](t) = base[k] + lin_coeff[k]*t + quad[k]*t^2
        C_t = base + lin_coeff * t + quad * t * t
        max_c = float(np.max(C_t))
        return 2.0 * n * max_c / (S_new * S_new)
    
    # 1D minimization
    # Search range: [0, max(t_old * 3, S)]
    hi = max(3.0 * t_old, S * 0.5) if t_old > 0 else S * 0.5
    if hi < 1e-10:
        hi = S
    
    result = minimize_scalar(obj_1d, bounds=(0.0, hi), method='bounded',
                             options={'maxiter': 100, 'xatol': 1e-10})
    
    t_new = max(0.0, result.x)

"""
Exact coordinate descent + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: Instead of approximate subgradient methods, we solve the exact 1D subproblem
for each coordinate using golden section search. This leads to higher quality local minima.
Combined with LP-based global improvement and diverse initializations.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def golden_section_min(f, lo, hi, n_iter=60):
    """Golden section search for 1D minimization. Returns minimizer."""
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(n_iter):
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2


def coord_step(a, i):
    """Optimize coordinate i exactly via 1D golden section search.
    Uses current values of all other coordinates."""
    n = len(a)
    t_old = a[i]
    S_other = float(np.sum(a)) - t_old

    # Compute base convolution (a with a[i]=0)
    a_zero = a.copy()
    a_zero[i] = 0.0
    C_base = compute_conv(a_zero)

    # Precompute linear and quadratic coefficients for a[i]=t
    lin_coeffs = np.zeros(2 * n - 1)
    quad_coeffs = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        j = k - i
        if 0 <= j < n:
            if j == i:
                quad_coeffs[k] = 1.0
            else:
                lin_coeffs[k] = 2.0 * a[j]

    # Precompute partial values for speed
    C_base_local = C_base
    lin_local = lin_coeffs
    quad_local = quad_coeffs
    n_local = n
    S_other_local = S_other

    def obj_1d(t):
        S_new = S_other_local + t
        if S_new < 0.01:
            return 1e10
        Ct = C_base_local + lin_local * t + quad_local * t * t
        return (2.0 * n_local * np.max(Ct)) / (S_new * S_new)

    # Determine search bounds
    hi = max(3.0 * t_old + 0.1, S_other / max(1, n - 1) * 3.0)
    hi = min(hi, S_other * 1.5 + 1.0)
    if hi < 1e-8:
        hi = 1.0
    lo = 0.0

    t_new = golden_section_min(obj_1d, lo, hi, n_iter=55)
    t_new = max(0.0, min(t_new, hi))
    a[i] = t_new
    return a


def coordinate_descent(a, max_sweeps=500, time_limit=30.0, start_time=None, verbose=False, rng=None):
    """Run coordinate descent with exact 1D subproblem solving."""

    best_a = a.copy()
    best_obj = compute_obj(a)

def build_A_matrix(a):
    """Build the matrix A where (A*g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        for h in range(n):
            j = k - h
            if 0 <= j < n:
                A[k, h] = a[j]
    return A


def build_A_matrix_fast(a):
    """Build A matrix more efficiently using toeplitz-like structure."""
    n = len(a)
    # A[k, h] = a[k-h] if 0 <= k-h < n, else 0
    # This is a Toeplitz matrix with first column = [a[0], 0, 0, ...] (shifted)
    # Actually A[k, h] = a[k-h] for max(0, k-n+1) <= h <= min(n-1, k)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for h in range(n):
        # A[k, h] = a[k-h] for k = h, h+1, ..., h+n-1
        A[h:h + n, h] = a
    return A


def solve_lp_for_g(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_matrix_fast(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_optimize(a, max_iter=50, t_blend=0.3, time_limit=30.0, verbose=False):
    """LP-based optimization: find g that maximizes sum subject to conv(a,g)<=M,
    then blend a with g."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    stall = 0

    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_for_g(a, M, time_limit=5)
        if g is None:
            break

        if sum_g <= 1.0 + 1e-3:
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0

        # Normalize g to have same sum as a
        S_a = np.sum(a)
        g = g * (S_a / sum_g)

        # Blend
        a_new = (1.0 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 0.01:
            break
        a_new = a_new / S_new * S_a  # preserve sum

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"      LP it={it}: obj={best_obj:.8f}")
        a = a_new

    return best_a, best_obj

    return np.maximum(a, 0)


def init_oscillating(n, rng):
    """Initialize with oscillating structure."""
    freq = rng.uniform(1.0, 5.0)
    phase = rng.uniform(0, 2 * np.pi)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    a += rng.normal(0, 0.03, n)
    return np.maximum(a, 0)


def init_bimodal(n, rng):
    """Initialize with bimodal (two peaks) structure."""
    center1 = n * rng.uniform(0.1, 0.35)
    center2 = n * rng.uniform(0.65, 0.9)
    width = n * rng.uniform(0.05, 0.15)
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center1) / width) ** 2) + np.exp(-0.5 * ((x - center2) / width) ** 2)
    a += rng.normal(0, 0.02, n)
    return np.maximum(a, 0)


def init_sine_half(n, rng):
    """Half-sine shape (zero at ends)."""
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a += rng.normal(0, 0.03, n)
    return np.maximum(a, 0)


def init_asymmetric_exp(n, rng):
    """Asymmetric exponential: rises fast, decays slow."""
    x = np.arange(n)
    rise = n * rng.uniform(0.1, 0.3)
    decay = n * rng.uniform(0.3, 0.8)
    peak = n * rng.uniform(0.3, 0.6)
    a = np.exp(-np.abs(x - peak) / (np.where(x < peak, rise, decay)))
    a += rng.normal(0, 0.02, n)
    return np.maximum(a, 0)


def init_random_smooth(n, rng):
    """Random smooth sequence via random Fourier coefficients."""
    num_freqs = max(2, n // 5)
    coeffs_r = rng.normal(0, 1, num_freqs)
    coeffs_i = rng.normal(0, 1, num_freqs)
    x = np.arange(n)
    a = np.ones(n)
    for k in range(1, num_freqs):
        a += (coeffs_r[k - 1] * np.cos(2 * np.pi * k * x / n) +
              coeffs_i[k - 1] * np.sin(2 * np.pi * k * x / n)) * 0.3
    return np.maximum(a, 0)


def normalize_sequence(a):
    """Normalize sequence to have sum = 1, non-negative."""
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / S


def run_pipeline_for_n(n, seed, time_budget, rng):
    """Run the full optimization pipeline for a given sequence length."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    # Try multiple initializations
    inits = [
        init_front_heavy,
        init_back_heavy,
        init_oscillating,
        init_bimodal,
        init_sine_half,
        init_asymmetric_exp,
        init_random_smooth,
    ]

    n_inits = min(len(inits), max(2, int(time_budget / 15)))
    time_per_init = time_budget / n_inits

    for idx in range(n_inits):
        if time.monotonic() - start > time_budget:
            break

        init_fn = inits[idx % len(inits)]
        a = init_fn(n, rng)
        a = normalize_sequence(a)
        obj = compute_obj(a)

        # Phase 1: Coordinate descent
        t1 = time.monotonic()
        cd_time = min(time_per_init * 0.5, time_budget - (t1 - start))
        if cd_time > 3:
            a, obj_cd = coordinate_descent(a, max_sweeps=200, time_limit=cd_time,
                                           start_time=start, rng=rng)
            obj = obj_cd

        # Phase 2: LP (if time allows and n is reasonable)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 500:
            lp_time = min(remaining * 0.3, 30.0)
            a_lp, obj_lp = lp_optimize(a, max_iter=30, t_blend=0.3, time_limit=lp_time)
            if obj_lp < obj:
                a, obj = a_lp, obj_lp

        # Phase 3: Final coordinate descent polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a_pol, obj_pol = coordinate_descent(a, max_sweeps=200, time_limit=remaining * 0.6,
                                                start_time=start, rng=rng)
            if obj_pol < obj:
                a, obj = a_pol, obj_pol

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = 2.0

    return best_a, best_obj


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def intensive_polish(a, time_limit, rng):
    """Intensive polishing with alternating CD and LP."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    round_num = 0
    while time.monotonic() - start < time_limit:
        round_num += 1
        remaining = time_limit - (time.monotonic() - start)
        if remaining < 3:
            break

        # Coordinate descent
        cd_time = remaining * 0.5
        a_cd, obj_cd = coordinate_descent(a, max_sweeps=500, time_limit=cd_time,
                                          start_time=start, rng=rng)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()

        # LP step
        remaining = time_limit - (time.monotonic() - start)
        if remaining > 10 and n <= 400:
            a_lp, obj_lp = lp_optimize(a_cd, max_iter=30, t_blend=0.25,
                                       time_limit=remaining * 0.3)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                a = a_lp.copy()
            else:
                a = a_cd.copy()
        else:
            a = a_cd.copy()

        # Random perturbation to escape local minima
        remaining = time_limit - (time.monotonic() - start)
        if remaining > 10 and round_num % 3 == 0:
            scale = rng.uniform(0.02, 0.08)
            a_pert = a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            S = np.sum(a_pert)
            if S > 0.01:
                obj_p = compute_obj(a_pert)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_pert.copy()
                    a = a_pert.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for the best sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses exact coordinate descent (1D subproblem solving) combined with LP-based
    global improvement and diverse initializations across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0  # leave buffer for final validation

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Sequence lengths to explore
    n_values = [30, 40, 50, 60, 80, 100, 120, 150, 200, 250, 300, 400, 500]

    if parent is not None:
        n_parent = len(parent)
        n_values = sorted(set(n_values + [n_parent, max(20, n_parent - 30), min(600, n_parent + 30)]))

    print(f"Exploring n values: {n_values}")
    print(f"Total time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # First pass: quick exploration
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        # Time allocation based on remaining time and position
        idx = n_values.index(n)
        frac = (idx + 1) / len(n_values)
        alloc = min(remaining * 0.15, 40.0 * (n / 100))
        alloc = max(8.0, alloc)

        print(f"\n  N={n}: {remaining:.0f}s remaining, allocating {alloc:.0f}s")

        a_opt, obj = run_pipeline_for_n(n, seed + n * 7919, time_budget=alloc, rng=rng)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  UPDATE: n={n}, obj={best_score:.8f}")

    # Second pass: intensive polishing on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n\nIntensive polishing (n={n_best}), {remaining:.0f}s remaining")

        a_pol, obj_pol = intensive_polish(best_seq, time_limit=remaining * 0.7, rng=rng)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After intensive polish: obj={best_score:.8f}")

    # Third pass: try a few more diverse lengths if time allows
    if time.monotonic() < deadline - 60 and best_seq is not None:
        remaining = deadline - time.monotonic()
        # Try lengths near the best
        n_best = len(best_seq)
        extra_ns = sorted(set([max(20, n_best - 50), max(20, n_best - 25),
                               n_best + 25, n_best + 50, n_best + 100]))
        extra_ns = [x for x in extra_ns if x <= 600 and x not in n_values]

        for n in extra_ns[:3]:
            if time.monotonic() > deadline - 30:
                break
            alloc = min(remaining * 0.2, 30.0)
            if alloc < 8:
                break
            print(f"\n  Extra N={n}: allocating {alloc:.0f}s")
            a_opt, obj = run_pipeline_for_n(n, seed + n * 104729, time_budget=alloc, rng=rng)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, obj={best_score:.8f}")

    # Final validation and cleanup
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    S = np.sum(best_seq)
    if S < 0.01:
        best_seq = np.ones(50)
        S = np.sum(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s")

    # Validate using evaluate_sequence
    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if np.isfinite(validation) and validation < best_score:
            print(f"  Validation gave better score: {validation:.8f}")
            best_score = validation
        elif np.isfinite(validation):
            print(f"  Validation: computed={best_score:.8f}, validated={validation:.8f}")
            best_score = validation
    except Exception as e:
        print(f"  Validation error: {e}")

    return result
