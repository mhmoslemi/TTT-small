    A = build_conv_matrix(f)

    # In scipy's linprog, the dual variables for A_ub x <= b_ub are in result.ineqlin.marginals
    # But we need negative of that (or check the sign convention)
    # Actually, for a minimization problem with A_ub @ x <= b_ub,
    # the dual variables are the shadow prices, which should be >= 0
    y = None

    current_score = compute_score(f)
    t = t_init
    for _ in range(max_backtracks):
        f_new = (1-t)*f + t*g
        f_new = np.clip(f_new, 0, None)

        if s < 1e-10:

            continue
        f_new = f_new / s
        new_score = compute_score(f_new)
        if new_score < current_score:
            return f_new, new_score, y
        t *= 0.5
    return None, None, y

                    f = np.clip(f, 0, None)
                    f = normalize(f)
                    if compute_score(f) < best_score:
                        best_f = f.copy()
                        best_score = compute_score(f)
                    stall = 0
                    if stall > 100:
                        break

    return best_f, best_score


def dual_pressure_update(f, dual_y, n_iter=50, eta=0.2, rng=None):
    """Pressure update using LP dual variables."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)

    if dual_y is None or dual_y.sum() < 1e-15:
        return best_f, best_score

    for it in range(n_iter):
        # pressure_i = sum_k A[k,i] * y_k = sum_k f[k-i] * y_k
        pressure = np.zeros(n)
        for i in range(n):
            k_min = i
            k_max = min(len(dual_y) - 1, i + n - 1)
            idx = k_max - k_min + 1
            if idx > 0:
                pressure[i] = np.dot(f[:idx], dual_y[k_min:k_min + idx])

        if pressure.max() > 1e-15:
            pressure = pressure / pressure.max()
            eta_eff = eta * (1.0 / (1.0 + 0.01 * it))
            f_new = f * np.exp(-eta_eff * pressure)
            f_new = np.clip(f_new, 1e-15, None)
            f_new = normalize(f_new)
            score = compute_score(f_new)
            if score < best_score - 1e-12:
                best_score = score
                best_f = f_new.copy()
                f = f_new

    return best_f, best_score


def spectral_flat(f, n_iter=200, rng=None):
    """FFT-based spectral flattening with non-negative projection."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        n_bins = len(mag)
        current_energy = np.sum(f ** 2)
        target_mag = np.sqrt(current_energy * n_fft / n_bins)

        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0

        if it % 40 == 39:
            phase = phase + rng.normal(0, 0.08, len(phase))

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score - 1e-10:
            best_score = score
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 150:
                break
        f = f_new

    return best_f, best_score


def lp_refine(f, budget_s, t_init=0.7, lp_time_limit=8, rng=None):
    """LP-based refinement with backtracking line search and dual-informed updates."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t = t_init
    start = time.monotonic()
    stall_count = 0
    iter_count = 0

    while time.monotonic() - start < budget_s:
        iter_count += 1
        remaining = budget_s - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        tlim = min(lp_time_limit, max(2, remaining * 0.2))
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': tlim})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b_raw = np.clip(res.x, 0, None)
        b_sum = b_raw.sum()
        if b_sum < 1e-15:
            break

        g = b_raw / b_sum

        # Extract dual variables
        dual_y = None
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            dual_y = np.abs(res.ineqlin.marginals)

        # Backtracking line search
        improved = False
        t_trial = min(1.0, t * 1.2)
        for bt in range(12):
            t_use = t_trial * (0.5 ** bt)
            f_new = (1 - t_use) * f + t_use * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score(f_new)
            if new_score < best_score - 1e-13:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = t_use
                improved = True
                break

        if improved:
            stall_count = 0
            if stall_count >= 0 and iter_count % 10 == 0:
                # Occasionally use dual for pressure update
                if dual_y is not None and dual_y.sum() > 1e-10:
                    f, s_dual = dual_pressure_update(f, dual_y, n_iter=30, eta=0.15, rng=rng)
                    if s_dual < best_score:
                        best_f = f.copy()
                        best_score = s_dual
        else:
            stall_count += 1
            t *= 0.7

            if stall_count % 5 == 0:
                # Try dual-informed pressure
                if dual_y is not None and dual_y.sum() > 1e-10:
                    f_new2, s2 = dual_pressure_update(f, dual_y, n_iter=40, eta=0.2, rng=rng)
                    if s2 < best_score:
                        best_f = f_new2.copy()
                        best_score = s2
                        f = f_new2
                        stall_count = 0
                        t = 0.5
                else:
                    f_new2, s2 = weighted_pressure_update(f, n_iter=40, eta=0.15, rng=rng)
                    if s2 < best_score:
                        best_f = f_new2.copy()
                        best_score = s2
                        f = f_new2
                        stall_count = 0
                        t = 0.5

            if stall_count > 20:
                # Aggressive perturbation
                f = f * (1 + 0.1 * rng.normal(0, 1, n))
                f = np.clip(f, 0, None)
                f = normalize(f)
                if compute_score(f) < best_score:
                    best_f = f.copy()
                    best_score = compute_score(f)
                t = 0.4
                stall_count = 0

            if stall_count > 50:
                break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, rng=None):
    """Coordinate descent polishing with geometric trial values."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                continue
            best_val = original
            best_local = best_score

            for factor in [0.3, 0.5, 0.7, 0.85, 1.0 / 0.85, 1.0 / 0.7, 1.0 / 0.5, 1.0 / 0.3]:
                f[i] = max(0.0, original * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def simulated_annealing(f, rng, n_iter=5000, T0=0.04, T_end=0.00005):
    """Simulated annealing with diverse move types."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()
        r = rng.random()

        if r < 0.25:
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start_idx = int(rng.integers(0, n - width))
            f_new[start_idx:start_idx + width] *= np.exp(rng.normal(0, 0.4))
        elif r < 0.45:
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.05))
        elif r < 0.6:
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.8:
            center = int(rng.integers(0, n))
            width = max(2, int(n // 32))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.normal(0, 0.08) * kernel
        else:
            f_new *= np.exp(rng.normal(0, 0.25, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 1500 and it > n_iter // 2:
            break

    return best_f, best_score


def multi_start_optimize(n, budget_s, rng, parent=None):
    """Run multi-start optimization for a given sequence length."""
    best_overall = None
    best_score_overall = np.inf

    # Determine number of starts based on budget
    n_starts = max(2, min(5, int(budget_s / 80)))
    time_per_start = budget_s / n_starts

    for start_idx in range(n_starts):
        remaining = budget_s - start_idx * time_per_start
        if remaining < 30:
            break

        if start_idx == 0 and parent is not None:
            f = parent.copy()
        else:
            f = diverse_init(n, rng)

        current_score = compute_score(f)
        if current_score < best_score_overall:
            best_score_overall = current_score
            best_overall = f.copy()
            print(f"    Start {start_idx}: initial score = {current_score:.6f}")

        # Phase 1: Spectral flattening (quick)
        f, s1 = spectral_flat(f, n_iter=min(150, int(time_per_start * 20)), rng=rng)
        if s1 < best_score_overall:
            best_score_overall = s1
            best_overall = f.copy()

        # Phase 2: Weighted pressure
        f, s2 = weighted_pressure_update(f, n_iter=min(80, int(time_per_start * 10)),
                                         eta=0.12, rng=rng)
        if s2 < best_score_overall:
            best_score_overall = s2
            best_overall = f.copy()

        # Phase 3: LP refinement (main workhorse)
        elapsed = 0
        lp_time = max(20, time_per_start * 0.5 - elapsed)
        f, s3 = lp_refine(f, lp_time, t_init=0.6, lp_time_limit=10, rng=rng)
        if s3 < best_score_overall:
            best_score_overall = s3
            best_overall = f.copy()
            print(f"    Start {start_idx}: after LP = {s3:.6f}")

        # Phase 4: Another round of pressure using the LP result
        f, s4 = weighted_pressure_update(f, n_iter=50, eta=0.1, rng=rng)
        if s4 < best_score_overall:
            best_score_overall = s4
            best_overall = f.copy()

        # Phase 5: Short LP polish
        remaining2 = budget_s - (start_idx + 1) * time_per_start
        if remaining2 > 15:
            f, s5 = lp_refine(f, min(20, remaining2 * 0.5), t_init=0.4, lp_time_limit=8, rng=rng)
            if s5 < best_score_overall:
                best_score_overall = s5
                best_overall = f.copy()

    return best_overall, best_score_overall


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multi-start LP refinement, pressure updates,
    spectral flattening, coordinate descent, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0
    start_time = time.monotonic()

    # Try parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = np.inf

    # Primary optimization on parent length
    if parent is not None:
        parent = np.clip(parent, 0, None)
        parent = normalize(parent)
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
        best_overall = parent.copy()
        best_score_overall = parent_score

        # Optimize on parent length
        n_parent = len(parent)
        alloc_parent = min(budget_s * 0.55, 800)
        print(f"\n--- Optimizing n={n_parent} (budget: {alloc_parent:.0f}s) ---")
        f_opt, s_opt = multi_start_optimize(n_parent, alloc_parent, rng, parent=parent)
        if s_opt < best_score_overall:
            best_score_overall = s_opt
            best_overall = f_opt.copy()
            print(f"  Best after parent-length optimization: {s_opt:.6f}")
    else:
        n_parent = 512
        alloc_parent = budget_s * 0.5
        print(f"\n--- Optimizing n={n_parent} from scratch (budget: {alloc_parent:.0f}s) ---")
        f_opt, s_opt = multi_start_optimize(n_parent, alloc_parent, rng)
        best_overall = f_opt
        best_score_overall = s_opt

    # Try a different length (longer)
    remaining = deadline - time.monotonic()
    n_long = 1024
    if remaining > 150:
        alloc_long = min(remaining * 0.5, 300)
        print(f"\n--- Optimizing n={n_long} (budget: {alloc_long:.0f}s) ---")
        f_opt2, s_opt2 = multi_start_optimize(n_long, alloc_long, rng)
        if s_opt2 < best_score_overall:
            best_score_overall = s_opt2
            best_overall = f_opt2.copy()
            print(f"  New best from n={n_long}: {s_opt2:.6f}")

    # Final polishing phase
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 20:
        print(f"\n--- Final polishing (remaining: {remaining:.0f}s) ---")
        n_best = len(best_overall)

        # Coordinate descent
        cd_time = min(remaining * 0.3, 120)
        n_sweeps = max(3, int(cd_time * 0.3))
        f_cd, s_cd = coordinate_descent(best_overall, n_sweeps=n_sweeps, rng=rng)
        if s_cd < best_score_overall:
            best_score_overall = s_cd
            best_overall = f_cd.copy()
            print(f"  After CD: {s_cd:.6f}")

        # Simulated annealing
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            n_iter_sa = min(8000, int(remaining2 * 150))
            f_sa, s_sa = simulated_annealing(best_overall, rng, n_iter=n_iter_sa,
                                             T0=0.03, T_end=0.00002)
            if s_sa < best_score_overall:
                best_score_overall = s_sa
                best_overall = f_sa.copy()
                print(f"  After SA: {s_sa:.6f}")

        # One more short LP refinement on the best
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            f_lp, s_lp = lp_refine(best_overall, min(remaining3 * 0.7, 120),
                                   t_init=0.3, lp_time_limit=10, rng=rng)
            if s_lp < best_score_overall:
                best_score_overall = s_lp
                best_overall = f_lp.copy()
                print(f"  Final LP: {s_lp:.6f}")

        # Final CD
        remaining4 = deadline - time.monotonic()
        if remaining4 > 10:
            f_cd2, s_cd2 = coordinate_descent(best_overall, n_sweeps=5, rng=rng)
            if s_cd2 < best_score_overall:
                best_score_overall = s_cd2
                best_overall = f_cd2.copy()
                print(f"  Final CD: {s_cd2:.6f}")

    # Verify and return
    elapsed = time.monotonic() - start_time
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal: score={final_score:.6f}, n={len(final_list)}, time={elapsed:.1f}s")
        return final_list
    else:
        print("Fallback: uniform sequence")
        return [1.0] * 512
