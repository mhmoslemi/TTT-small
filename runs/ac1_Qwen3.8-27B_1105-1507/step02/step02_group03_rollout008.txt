import numpy as np

import time

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()

def build_conv_matrix(f):
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A

def spectral_equalize(f, n_iter=300, rng=None, strength=1.0):
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0
    
    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: geometric mean of magnitudes (excluding DC)
        # Use a power to control strength
        non_dc = mag[1:]
        if non_dc.min() < 1e-15:
            non_dc_target = np.sqrt(np.mean(non_dc**2))
        else:
            log_mean = np.mean(np.log(np.clip(non_dc, 1e-15, None)))
            non_dc_target = np.exp(log_mean)
        
        # Blend: target = current * (target/current)^strength
        # This controls how aggressively we flatten
        ratio = non_dc_target / np.clip(mag[1:], 1e-15, None)
        mag_new = mag.copy()
        mag_new[1:] = mag[1:] * ratio ** strength
        
        # Ensure DC is preserved
        mag_new[0] = 1.0
        
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        
        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            f = f_new  # Still update to explore
            stall += 1
            if stall > 100:
                break
    
    return best_f, best_score

def pressure_update_v2(f, n_iter=300, eta=0.15, rng=None):
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_score = compute_score(f)
    stall = 0

        conv = np.convolve(f, f)
        M = conv.max()
        
        # Compute pressure using top k peaks
        top_k = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]
        peak_values = conv[peak_indices]
        
        pressure = np.zeros(n)
        for pk, pv in zip(peak_indices, peak_values):
            weight = (pv / M) ** 2
            for i in range(n):

                if 0 <= idx < n:
                    pressure[i] += weight * f[idx]

            f = f_new

def multiplicative_update(f, n_iter=200, rng=None):
    """
    Multiplicative weight update based on convolution imbalance.
    """
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        avg = 1.0 / (2 * n - 1)  # average convolution value
        
        # For each i, compute the "excess" it contributes to peaks
        # vs the "deficit" it helps fill in valleys
        pressure = np.zeros(n)
        for i in range(n):
            # Contribution of f[i] to convolution at various positions
            # Position k gets f[i] * f[k-i]
            # If k is a peak, this is "bad"; if k is a valley, this is "good"
            for j in range(n):
                k = i + j
                if k < 2 * n - 1:
                    if conv[k] > M * 0.9:
                        pressure[i] += f[j]
                    elif conv[k] < avg * 1.1:
                        pressure[i] -= 0.5 * f[j]
        
        # Normalize
        pmax = np.abs(pressure).max()
        if pmax > 0:
            pressure = pressure / pmax
        
        eta = 0.1 / (1.0 + 0.02 * it)
        f_new = f * np.exp(-eta * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)
        
        score = compute_score(f_new)
        if score < best_score - 1e-13:
            best_score = score
            best_f = f_new.copy()
            f = f_new
    
    return best_f, best_score

def diverse_init(n, rng):
    r = rng.random()

        width = n * rng.uniform(0.08, 0.3)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.4:
        # Cosine window
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.6:
        # Product of sines
        f = 1.0
        for _ in range(int(rng.integers(1, 4))):
            freq = rng.uniform(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            f *= 0.5 + 0.5 * np.sin(np.pi * freq * np.arange(n) / n + phase)
        f = np.abs(f) + 0.05
    elif r < 0.8:
        # Random smooth (low-pass filtered noise)
        f = np.ones(n)
        for _ in range(int(rng.integers(3, 8))):
            center = int(rng.integers(0, n))
            width = int(rng.integers(3, max(4, n // 4)))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / max(1, width ** 2))
            f *= (1.0 + rng.standard_normal() * 0.3 * kernel)
        f = np.abs(f) + 0.05
    else:
        # Block structure
        n_blocks = int(rng.integers(2, 8))
        f = np.zeros(n)
        for b in range(n_blocks):
            start = int(b * n / n_blocks)
            end = int((b + 1) * n / n_blocks)
            f[start:end] = rng.uniform(0.3, 1.0)
        # Smooth
        f = np.convolve(f, np.ones(5) / 5, mode='same')
    
    f = np.clip(f, 0.01, None)
    f = normalize(f)
    f = f * (1 + 0.03 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f

"""
Improved multi-strategy optimization: aggressive spectral equalization,
adaptive LP refinement with M-bisection, pressure updates, and diverse
initializations. Key improvements over previous: tighter LP loop with
adaptive constraint tightening, more aggressive spectral flattening,
n=1024 exploration, and better step-size search.

import numpy as np
from scipy.optimize import linprog
import time


def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def spectral_equalize(f, n_iter=300, rng=None, strength=1.0):
    """Iterative spectral flattening with adaptive strength."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F).copy()
        phase = np.angle(F)

        # Target: geometric mean of non-DC magnitudes
        non_dc = mag[1:]
        clip_val = np.clip(non_dc, 1e-15, None)
        log_mean = np.mean(np.log(clip_val))
        target_mag = np.exp(log_mean)

        # Blend current toward target
        ratio = target_mag / clip_val
        new_mag = np.empty_like(mag)
        new_mag[0] = 1.0
        new_mag[1:] = mag[1:] * ratio ** strength

        F_new = new_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
        f = f_new

    return best_f, best_score


def pressure_update(f, n_iter=200, eta=0.15, rng=None):
    """Pressure-based multiplicative update targeting convolution peaks."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        # Use top peaks with weighted contributions
        top_k = min(8, len(conv))
        peak_idx = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[peak_idx]

        pressure = np.zeros(n)
        for pk, pv in zip(peak_idx, peak_vals):
            w = (pv / M) ** 2
            # f[i] contributes f[i]*f[pk-i] to conv[pk]
            valid_i = range(max(0, pk - n + 1), min(n, pk + 1))
            for i in valid_i:
                pressure[i] += w * f[pk - i]

        pmax = pressure.max()
        if pmax > 1e-15:
            pressure /= pmax

        eta_eff = eta / (1.0 + 0.03 * it)
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-13:
            best_score = score
            best_f = f_new.copy()
            stall = 0
        else:
            # Accept with SA probability
            T = 0.005 / (1.0 + 0.05 * it)
            if rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                f = f_new
            stall += 1
            if stall > 200:
                f_pert = f * (1.0 + 0.08 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                ps = compute_score(f_pert)
                if ps < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = ps
                    stall = 0
                elif stall > 400:
                    break
        f = f_new if score < best_score else f

    return best_f, best_score


def lp_refine(f, time_budget, rng=None, initial_t=0.4):
    """
    Aggressive LP refinement with adaptive M and binary search step sizes.
    Tries progressively tighter convolution bounds and uses the LP solution
    as a direction for updating f.
    """
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score(f)
    t = initial_t
    start = time.monotonic()
    lp_calls = 0
    no_improve_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n

        g = None
        # Try progressively tighter M values
        for frac in [0.92, 0.95, 0.98, 1.0]:
            M_try = M * frac
            if M_try <= 0:
                continue
            b_ub = np.full(2 * n - 1, M_try)
            tl = max(0.5, min(2.5, remaining * 0.04))

            try:
                res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presolve': True,
                                                       'time_limit': tl})
                lp_calls += 1
            except Exception:
                continue

            if res.success and res.x is not None:
                g = np.clip(res.x, 0, None)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    break

        if g is None:
            # Fallback: pressure update
            f, best_score = pressure_update(f, n_iter=30, eta=0.1, rng=rng)
            t = initial_t
            no_improve_count = 0
            continue

        # Binary search for best step size
        lo, hi = 0.0, 1.0
        best_step = 0.0
        best_trial_score = best_score

        for _ in range(8):
            mid = (lo + hi) / 2.0
            if mid < 1e-6:
                break
            f_trial = (1 - mid) * f + mid * g
            f_trial = np.clip(f_trial, 0, None)
            ts = f_trial.sum()
            if ts < 1e-10:
                hi = mid
                continue
            f_trial = f_trial / ts
            trial_score = compute_score(f_trial)
            if trial_score < best_trial_score:
                best_trial_score = trial_score
                best_step = mid
                lo = mid
            else:
                hi = mid

        if best_step > 1e-5 and best_trial_score < best_score - 1e-13:
            f_new = (1 - best_step) * f + best_step * g
            f_new = np.clip(f_new, 0, None)
            f_new = f_new / f_new.sum()
            best_f = f_new.copy()
            best_score = best_trial_score
            f = f_new
            t = min(1.0, t * 1.1)
            no_improve_count = 0
        else:
            no_improve_count += 1
            t *= 0.6
            if no_improve_count > 15:
                # Perturb and restart
                f_pert = f * (1.0 + 0.05 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                ps = compute_score(f_pert)
                if ps < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = ps
                    t = initial_t * 0.5
                    no_improve_count = 0
                else:
                    # Try a few random perturbations
                    found = False
                    for _ in range(5):
                        f_try = f * (1.0 + 0.1 * rng.normal(0, 1, n))
                        f_try = np.clip(f_try, 0, None)
                        f_try = normalize(f_try)
                        if compute_score(f_try) < best_score:
                            f = f_try
                            best_f = f.copy()
                            best_score = compute_score(f)
                            t = initial_t * 0.3
                            no_improve_count = 0
                            found = True
                            break
                    if not found:
                        break

    return best_f, best_score, lp_calls


def diverse_init(n, rng):

        f = np.abs(f) + 0.03
    elif r < 0.75:
        f = np.ones(n)
        for _ in range(int(rng.integers(3, 8))):

            f *= (1.0 + rng.standard_normal() * 0.3 * ker)
        f = np.abs(f) + 0.03
    elif r < 0.9:
        n_blocks = int(rng.integers(3, 9))
        f = np.zeros(n)
        for b in range(n_blocks):

            f[s:e] = rng.uniform(0.2, 1.0)
        f = np.convolve(f, np.ones(max(3, n // 64)) / max(3, n // 64), mode='same')
    else:
        # Low-frequency random

        amps = rng.uniform(0.2, 1.0, 3)

    f = np.clip(f, 0.001, None)
