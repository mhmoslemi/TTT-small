def compute_soft_max_gradient(f, beta):
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    p = exp_C / total
    corr = np.correlate(p, f, mode='valid')
    grad = 2.0 * corr
    return grad

"""
Multi-phase optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses FFT-based variance/peak gradients, LP direction finding,
spectral pre-conditioning, and adaptive multi-start search.
Key improvements: O(n log n) gradient computation via FFT, variance
minimization phase for better conditioning, and more diverse initializations.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_size(n):
    """Smallest power of 2 >= n."""
    s = 1
    while s < n:
        s *= 2
    return s


def _conv_fft(f):
    """Linear convolution f*f via FFT."""
    n = len(f)
    N = _fft_size(2 * n - 1)
    fp = np.zeros(N)
    fp[:n] = f
    F = np.fft.fft(fp)
    return np.real(np.fft.ifft(F * F))[:2 * n - 1]


def _variance_gradient(f):
    """Gradient of sum(conv(f,f))^2 computed via FFT. O(n log n)."""
    n = len(f)
    N = _fft_size(2 * n - 1)
    fp = np.zeros(N)
    fp[:n] = f
    F = np.fft.fft(fp)
    grad_full = 4.0 * np.real(np.fft.ifft(np.abs(F) ** 2 * F))
    return grad_full[:n]


def _softmax_gradient_fft(f, beta):
    """Soft-max gradient via FFT cross-correlation. Returns (grad, soft_max_val)."""
    n = len(f)
    C = _conv_fft(f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0
    p = exp_C / total
    N = _fft_size(3 * n - 2)
    p_pad = np.zeros(N)
    p_pad[:2 * n - 1] = p
    f_pad = np.zeros(N)
    f_pad[:n] = f
    P = np.fft.fft(p_pad)
    Ff = np.fft.fft(f_pad)
    corr = np.real(np.fft.ifft(P * np.conj(Ff)))
    grad = 2.0 * corr[:n]
    soft_max_val = max_C + np.log(total) / beta
    return grad, soft_max_val


def _build_conv_matrix_sparse(f):
    """Sparse matrix A where (A @ g)[k] = convolve(f, g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (np.array(rows, dtype=np.int32),
                              np.array(cols, dtype=np.int32))),
                      shape=(2 * n - 1, n))


def _solve_lp(f, M, A_cache=None):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = _build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _adam_step(grad, m, v, t, lr, b1=0.8, b2=0.999, eps=1e-8):
    """One Adam update step. Returns (update, m_new, v_new)."""
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * (grad ** 2)
    m_hat = m / (1 - b1 ** t)
    v_hat = v / (1 - b2 ** t)
    update = m_hat / (np.sqrt(v_hat) + eps)
    return lr * update, m, v


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (css - 1))[0][-1]
    theta = (css[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def _lp_optimize(f, deadline, step_init=0.08, max_iters=100):
    """Phase 3: LP-based direction finding with line search."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f
    f = f / S
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step = step_init
    no_improve = 0
    A_cache = None

    while time.monotonic() < deadline:
        C = _conv_fft(f)
        M = float(C.max())
        if M <= 0:
            break

            A_cache = _build_conv_matrix_sparse(f)
        g = _solve_lp(f, M, A_cache)

            no_improve += 1

                break
            continue

        if sum_g < 1e-15:
            no_improve += 1
            continue

        best_s = best_score

            new_f = (1.0 - trial_t) * f + trial_t * g

            best_score = best_s
            no_improve = 0
            A_cache = None  # invalidate cache since f changed

            step *= 0.6

def _tukey_window(n, r=0.3):
    w = np.ones(n)
    m = max(1, int(np.floor(r * n / 2)))
    for i in range(m):
        w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
        w[n - 1 - i] = w[i]
    return w


def _hann_window(n):
    if n < 2:
        return np.ones(n)
    return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))


def _gauss_tukey(n, r=0.2, gamma=1.0):
    """Gaussian center with Tukey taper."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    g = np.exp(-gamma * ((idx - center) / (n / 2.0)) ** 2)
    m = max(1, int(np.floor(r * n / 2)))
    for i in range(m):
        taper = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
        g[i] *= taper
        g[n - 1 - i] *= taper
    return g


def _raised_cosine_super(n, power=2.0):
    """Raised cosine window raised to a power."""
    if n < 2:
        return np.ones(n)
    x = np.linspace(-1, 1, n)
    w = np.cos(np.pi * x / 2.0) ** 2
    return np.power(np.clip(w, 1e-10, 1.0), power)


def _spectral_init(n):
    """Initialize from a target flat spectrum, non-negativized."""
    N = _fft_size(2 * n - 1)
    # Target: flat |F|^2 with DC boost
    target = np.ones(N)
    target[0] = 2.0  # boost DC
    F_target = np.sqrt(target).astype(np.complex128)
    F_target.imag = 0
    f_raw = np.real(np.fft.ifft(F_target))[:n]
    f_raw = np.clip(f_raw, 0.0, None)
    S = float(np.sum(f_raw))
    if S < 1e-10:
        f_raw = _tukey_window(n, 0.3)
        S = float(np.sum(f_raw))
    return f_raw / S

                best_f, best_score = f3, s2

    # Phase 4: Fine-tune with high-beta gradient (20% of budget)
    if time.monotonic() < deadline - 1:
        f4 = _peak_optimize(best_f, deadline, beta_start=200.0, beta_end=800.0, lr_start=0.008)
        s4 = evaluate_sequence(f4.tolist())
        if np.isfinite(s4) and s4 < best_score:
            best_f, best_score = f4, s4

    return best_f


def propose_candidate(budget_s=1500):
    """Main search function. Returns best sequence found within budget."""
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s) - 5.0)
    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                parent = parent / S
                sc = evaluate_sequence(parent.tolist())
                if np.isfinite(sc):
                    best_score = sc
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={sc:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Lengths to try with time fractions
    # Focus on 512 (proven) and explore 640, 768, 1024
    plans = [
        (512, 0.28),
        (768, 0.18),
        (1024, 0.17),
        (640, 0.12),
        (384, 0.08),
    ]

    total_budget = deadline - t0
    elapsed_before = 0.0

    for n, frac in plans:
        if time.monotonic() > deadline - 60:
            print("Time running out, stopping exploration")
            break
        n_budget = total_budget * frac
        n_deadline = time.monotonic() + n_budget
        print(f"\n=== n={n} (budget {n_budget:.0f}s) ===")

        # Generate initializations
        inits = []
        # Tukey with various r
        for r in [0.2, 0.3, 0.4]:
            inits.append(_tukey_window(n, r))
        # Gauss-Tukey
        for gamma in [0.5, 1.0, 1.5]:
            inits.append(_gauss_tukey(n, r=0.25, gamma=gamma))
        # Raised cosine power
        for pw in [1.5, 2.0, 3.0]:
            inits.append(_raised_cosine_super(n, power=pw))
        # Spectral init
        inits.append(_spectral_init(n))
        # Hann
        inits.append(_hann_window(n))
        # Parent interpolation if available
        if best_sequence is not None and len(best_sequence) != n:
            x_old = np.linspace(0, 1, len(best_sequence))
            x_new = np.linspace(0, 1, n)
            parent_interp = np.interp(x_new, x_old, best_sequence)
            parent_interp = np.clip(parent_interp, 0, None)
            S = float(np.sum(parent_interp))
            if S > 0.01:
                inits.insert(0, parent_interp / S)
        elif best_sequence is not None and len(best_sequence) == n:
            inits.insert(0, best_sequence.copy())

        # Evaluate and optimize promising inits
        n_inits = len(inits)
        # Sort by initial score (best first)
        scored_inits = []
        for fi in inits:
            fi = np.clip(fi.astype(np.float64), 0.0, None)
            S = float(np.sum(fi))
            if S < 0.01:
                continue
            fi = fi / S
            sc = evaluate_sequence(fi.tolist())
            if np.isfinite(sc):
                scored_inits.append((sc, fi))
        scored_inits.sort(key=lambda x: x[0])

        # Optimize top initializations
        n_to_optimize = min(5, len(scored_inits))
        per_init_time = n_budget * 0.85 / max(n_to_optimize, 1)

        for idx in range(n_to_optimize):
            if time.monotonic() > n_deadline - 5:
                break
            sc_init, fi = scored_inits[idx]
            if sc_init < best_score:
                best_score = sc_init
                best_sequence = fi.copy()
            # Only optimize if close to best
            if sc_init > best_score * 1.03 and idx > 0:
                continue
            init_deadline = min(n_deadline, time.monotonic() + per_init_time)
            if time.monotonic() < init_deadline - 3:
                try:
                    bf = _full_pipeline(fi, init_deadline)
                    bs = evaluate_sequence(bf.tolist())
                    if np.isfinite(bs) and bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        print(f"  n={n} init {idx} (score {sc_init:.4f}): improved to {bs:.6f}")
                except Exception as e:
                    print(f"  n={n} init {idx} error: {e}")

        if best_sequence is not None and len(best_sequence) == n:
            print(f"  n={n} best after this block: {best_score:.6f}")

    # Perturbation restarts
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Perturbation restarts (best: {best_score:.6f}) ===")
        n_best = len(best_sequence)
        restart_time = deadline - time.monotonic() - 15
        n_restarts = min(6, max(2, int(restart_time / 25)))
        per_restart = restart_time / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            strength = 0.03 + 0.02 * r
            f_pert = _perturb(best_sequence, rng, strength=strength)
            if time.monotonic() < deadline - 15:
                try:
                    bf = _full_pipeline(f_pert, min(deadline - 10, time.monotonic() + per_restart * 0.8),
                                        do_variance=(r % 2 == 0))
                    bs = evaluate_sequence(bf.tolist())
                    if np.isfinite(bs) and bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        print(f"  Restart {r} (str={strength:.3f}): {bs:.6f}")
                    else:
                        print(f"  Restart {r} (str={strength:.3f}): {bs:.6f} (no improve)")
                except Exception as e:
                    print(f"  Restart {r} error: {e}")

    # Final LP + gradient polish
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Final polish (n={len(best_sequence)}, score={best_score:.6f}) ===")
        try:
            bf = _lp_optimize(best_sequence, min(deadline - 3, time.monotonic() + 15), step_init=0.05)
            bs = evaluate_sequence(bf.tolist())
            if np.isfinite(bs) and bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
                print(f"  After LP: {best_score:.6f}")
        except Exception:
            pass
        if time.monotonic() < deadline - 3:
            try:
                bf = _peak_optimize(best_sequence, deadline, beta_start=300.0, beta_end=1000.0, lr_start=0.005)
                bs = evaluate_sequence(bf.tolist())
                if np.isfinite(bs) and bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  After fine gradient: {best_score:.6f}")
            except Exception:
                pass

    if best_sequence is None:
        best_sequence = _tukey_window(512, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Final validation
    best_sequence = np.clip(best_sequence.astype(np.float64), 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
