"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. L-BFGS-B with log-sum-exp smoothed objective and beta annealing
2. Multiple lengths and diverse initializations
3. LP-based best-response for final polish
4. Subgradient FISTA for the exact objective
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            segment = a[jl:jh + 1]
            # A[k, j] = a[k - j] for j in jl..jh
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):

def smooth_conv_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient w.r.t. a.
    Uses log-sum-exp for numerical stability.
    Returns (smooth_max_value, gradient_array).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)

    Z = float(exp_shifted.sum())

def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_multiplicative=False):
    """
    LP-based mirror descent (Frank-Wolfe style) iterations.
    At each step: solve LP for best response g, then update a.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size: try several and pick the best
        base_t = max(0.03, 0.4 * (1.0 - it / float(max(2 * max_iters, 1))))
        t_candidates = [base_t, min(1.0, base_t * 2.0), max(0.01, base_t * 0.5), 0.1]

        best_a = None
        best_J = current_J

        for t in t_candidates:
            if use_multiplicative:
                # Geometric mean update: a_new = a^(1-t) * g^t
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # Try very small step
            t_small = 0.02
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                # Stall: try a slightly different approach
                # Perturb and retry
                break

        if tracker is not None and it % 20 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):

        # Also a slightly perturbed version

        p = p + rng.normal(0, 0.05, n) * p

    # Uniform
    inits.append(np.ones(n))

    # Cosine families
    for freq in [2, 3, 4, 5, 7, 8]:
        inits.append(1.0 + 0.35 * np.cos(2.0 * np.pi * freq * t_arr))

    # Smooth bump

    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2)

    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    # Phase-shifted cosine

    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * t_arr + phase))

    # Two-peak structure
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr))

    # Sparse + smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 20)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Power law
    inits.append((t_arr + 0.1) ** rng.uniform(-0.8, 0.8))

    # Alternating high/low
    alt = np.ones(n)
    alt[::2] = 0.6
    inits.append(alt)

    # Exponential decay
    inits.append(np.exp(-1.5 * t_arr) + 0.2)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: L-BFGS-B with smooth objective (fast, finds good basin)
    Phase 2: LP mirror descent (exact, refines)
    Phase 3: Multi-restart with perturbation (escape local minima)
    Phase 4: Final LP polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: L-BFGS-B with smooth objective ===
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B {'=' * 50}", flush=True)

    lengths_p1 = [128, 256, 512]
    beta_schedule_p1 = [(1.0, 60), (5.0, 60), (20.0, 80), (50.0, 100)]

    for n in lengths_p1:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Limit inits based on time
        max_inits = max(3, min(len(inits), int(120.0 / max(1, n / 128))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_smooth_optimize(a0, beta_schedule_p1, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP mirror descent on best solution ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: LP MD n={n} {'=' * 50}", flush=True)

        # Run LP iterations from the best
        a = lp_mirror_descent(tracker.best, max_iters=500, deadline=deadline, tracker=tracker)

        # Also try from resampled versions
        for n_try in [128, 256, 512]:
            if time.monotonic() > deadline:
                break
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            a_res = lp_mirror_descent(a_res, max_iters=300, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Multi-restart with perturbation ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: Restarts n={n} {'=' * 50}", flush=True)

        # Try different perturbation strategies
        strategies = [
            ('gaussian', 0.1),
            ('gaussian', 0.05),
            ('uniform', 0.15),
            ('spike', 5),
            ('gaussian', 0.03),
            ('smooth_noise', 0.1),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(8):
                if time.monotonic() > deadline:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n)
                elif pert_type == 'spike':
                    k = min(param, n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'smooth_noise':
                    noise = rng.normal(0, param, n)
                    # Smooth the noise
                    kernel = np.ones(5) / 5.0
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Run a short LP refinement
                max_iters = max(30, min(150, int(60.0 * (deadline - time.monotonic()) / max(1, budget_s * 0.4))))
                a = lp_mirror_descent(a, max_iters, deadline, tracker)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 4: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Final Polish n={n} {'=' * 50}", flush=True)

        # Long run of LP iterations
        remaining = deadline - time.monotonic()
        max_iters = int(min(1000, max(50, remaining * 50)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker,
                              use_multiplicative=False)

        # Also try multiplicative update
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            max_iters = int(min(500, max(20, remaining * 30)))
            a2 = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker,
                                   use_multiplicative=True)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try adjacent lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Length Search {'=' * 50}", flush=True)

        for n_try in [n - 32, n - 16, n + 16, n + 32, n + 64]:
            if n_try < 32 or n_try > 2048 or time.monotonic() > deadline:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            # Quick L-BFGS then LP
            beta_sched = [(10.0, 40), (30.0, 40)]
            a_opt = lbfgs_smooth_optimize(a_res, beta_sched, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                a_lp = lp_mirror_descent(a_opt, 200, deadline, tracker=tracker)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
