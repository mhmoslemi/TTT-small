"""
QP-based optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses proximal QP direction finding with adaptive constraint selection,
combined with subgradient polishing. Key innovation: formulates each
step as a Euclidean projection onto a polytope (intersection of
half-spaces from top active convolution lags and non-negativity),
solved via CVXPY/OSQP with warm-starting.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix, eye as speye
from scipy.optimize import linprog


def compute_score(f):
    """Compute 2*n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 1e-15:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_qp_direction(f, M, active_lags, mu):
    """
    Solve: min mu/2 * ||g - f||^2 - sum(g)
    s.t. conv(f,g)_k <= M for k in active_lags
         g >= 0
    Returns g or None if infeasible.
    """
    n = len(f)
    A = build_conv_matrix(f)
    # Select only active constraints
    A_sub = A[active_lags, :]
    
    g = cp.Variable(n)
    # Objective: mu/2 * ||g - f||^2 - sum(g)
    # = mu/2 * (g-f)^T (g-f) - 1^T g
    cost = 0.5 * mu * cp.sum_squares(g - f) - cp.sum(g)
    
    constraints = [A_sub @ g <= M]
    if g >= 0 not in constraints:
        constraints.append(g >= 0)
    
    prob = cp.Problem(cp.Minimize(cost), constraints)
    
    try:
        prob.solve(solver=cp.OSQP, max_iter=5000, eps_abs=1e-6, eps_rel=1e-6,
                   adaptive=True, polish=True)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-6, reltol=1e-6)
        except Exception:
            return None
    
    if g.value is None:
        return None
    
    g_val = np.asarray(g.value).flatten()
    if not np.all(np.isfinite(g_val)):
        return None
    
    return g_val


def solve_qp_all_constraints(f, M, mu, max_constraints=None):
    """
    Solve QP with all constraints (or top max_constraints if specified).
    Uses iterative constraint addition for efficiency.
    """
    n = len(f)
    conv_f = np.convolve(f, f)
    
    # Start with top-K active lags
    if max_constraints is not None:
        K = min(max_constraints, 2 * n - 1)
        active_lags = list(np.argsort(conv_f)[-K:])
    else:
        active_lags = list(range(2 * n - 1))
    
    max_round = 10
    for _ in range(max_round):
        g = solve_qp_direction(f, M, active_lags, mu)
        if g is None:
            return None
        
        # Check for violations of non-active constraints
        conv_g = np.convolve(f, g)
        violations = np.where(conv_g > M + 1e-10)[0]
        # Exclude already-active lags
        active_set = set(active_lags)
        new_violations = [v for v in violations if v not in active_set]
        
        if not new_violations:
            return g
        
        # Add most violated constraints
        new_lags = sorted(new_violations, key=lambda k: -conv_g[k])[:5]
        active_lags.extend(new_lags)
    
    return g


def subgradient_step(f, lr, top_k=5):
    """Compute a subgradient step to reduce max convolution."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    
    # Get top-k lags
    top_indices = np.argsort(conv)[-top_k:]
    
    # Weight by how much above the mean
    mean_conv = (S * S) / (2 * n - 1)
    weights = np.array([max(conv[k] - mean_conv, 0.0) for k in top_indices])
    if np.sum(weights) < 1e-15:
        weights = np.ones(len(top_indices))
    
    # Compute gradient
    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        if w <= 0:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * w * f[k - j]
    
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return f.copy()
    
    grad = grad / norm
    new_f = f - lr * grad
    new_f = np.clip(new_f, 0.0, None)
    new_sum = np.sum(new_f)
    if new_sum < 1e-15:
        return f.copy()
    new_f = new_f * (S / new_sum)
    return new_f


def fast_subgradient_optimize(f, deadline, lr_init=0.005, momentum_decay=0.9,
                              top_k=5, max_iter=50000):
    """Fast subgradient descent with momentum."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M <= 0:
            break

        n = len(f)
        top_indices = np.argsort(conv)[-top_k:]
        mean_conv = (S * S) / (2 * n - 1)
        weights = np.array([max(conv[k] - mean_conv, 0.0) for k in top_indices])
        if np.sum(weights) < 1e-15:
            weights = np.ones(len(top_indices))

        grad = np.zeros(n)
        for k, w in zip(top_indices, weights):
            if w <= 0:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += 2.0 * w * f[k - j]

        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break
        grad = grad / norm_grad

        momentum = momentum_decay * momentum - lr * grad
        new_f = f + momentum
        new_f = np.clip(new_f, 0.0, None)
        new_sum = float(np.sum(new_f))
        if new_sum < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 100:
                break
            it += 1
            continue
        new_f = new_f * (S / new_sum)

        new_score = compute_score(new_f)
        it += 1

        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.98
            momentum *= 0.7
            if lr < 1e-8 or no_improve > 200:
                break

    return best_f, best_score


def qp_optimize(f, deadline, mu_init=0.01, k_init=10, verbose=False):

    """
    f = f.copy().astype(np.float64)

    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    
    mu = mu_init
    K = k_init
    no_improve = 0
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline and it < 5000:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M <= 0:
            break
        
        # Solve QP direction
        t_start = time.monotonic()
        g = solve_qp_all_constraints(f, M, mu, max_constraints=K)
        qp_time = time.monotonic() - t_start
        
        if g is None:
            no_improve += 1
            mu *= 2.0
            if no_improve > 20:
                break
            it += 1
            continue
        
        g = np.clip(g, 0.0, None)
        g_sum = float(np.sum(g))
        if g_sum < 1e-15:
            no_improve += 1
            mu *= 2.0
            it += 1
            continue
        
        # Rescale to maintain sum
        g = g * (S / g_sum)
        
        # Try the update
        new_score = compute_score(g)
        
        # Also try a blend for safety
        best_candidate = None
        best_cand_score = new_score

            candidate = alpha * g + (1.0 - alpha) * f

def make_structured_initializations(n, rng):
    """Create diverse structured initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    inits.append(('uniform', np.ones(n)))
    
    # Various tapers
    cos_t = (1.0 + np.cos(np.pi * idx / max(n - 1, 1))) / 2.0
    inits.append(('cosine', cos_t))
    
    # Triangular
    tri = 1.0 - np.abs(idx - center) / (center + 1e-10)
    inits.append(('triangular', np.clip(tri, 0.01, None)))
    
    # Sine
    sin_s = np.sin(np.pi * (idx + 1) / (n + 1))
    inits.append(('sine', sin_s))
    
    # Gaussian (wide)
    gauss_wide = np.exp(-0.5 * ((idx - center) / (n / 3.0)) ** 2)
    inits.append(('gauss_wide', gauss_wide))
    
    # Random smooth
    r = rng.uniform(0.5, 1.5, size=n)
    # Smooth with a kernel
    kernel = np.ones(21) / 21.0
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(('random_smooth', r_smooth))
    
    # Bimodal
    two_peak = (np.exp(-0.5 * ((idx - n * 0.3) / (n / 10.0)) ** 2) +
                np.exp(-0.5 * ((idx - n * 0.7) / (n / 10.0)) ** 2))
    inits.append(('bimodal', two_peak))
    
    # Flat top with smooth edges
    edge = n // 10
    flat = np.ones(n)
    fade = np.linspace(0.3, 1.0, edge)
    flat[:edge] = fade
    flat[-edge:] = fade[::-1]
    inits.append(('flat_top', flat))

    return inits

        print(f"Interpolated to n={target_n}: score={best_score:.6f}")
    
    # Phase 1: Quick exploration of different initializations (100s)
    print("\n=== Phase 1: Quick Exploration ===")
    explore_deadline = t0 + 100.0
    
    # Also try a different length
    alt_n = 2048
    
    candidates = []
    
    # Try target_n with various inits
    for n_try in [target_n]:
        if best_sequence is not None and len(best_sequence) == n_try:
            inits = [('parent', best_sequence.copy())]
        else:
            inits = make_structured_initializations(n_try, rng)
        
        for label, f_init in inits:
            if time.monotonic() > explore_deadline - 5:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            if np.sum(f_init) < 1e-10:
                continue
            s = compute_score(f_init)
            if not np.isfinite(s):
                continue
            if s < best_score:
                best_score = s
                best_sequence = f_init.copy()
            candidates.append((s, f_init.copy(), n_try, label))
            print(f"  n={n_try} {label}: {s:.6f}")
    
    # Try alt_n with parent interpolation
    if best_sequence is not None:
        x_old = np.linspace(0, 1, len(best_sequence))
        x_new = np.linspace(0, 1, alt_n)
        alt_init = interp(x_new, x_old, best_sequence)
        s_alt = compute_score(alt_init)
        if np.isfinite(s_alt):
            candidates.append((s_alt, alt_init.copy(), alt_n, 'parent_interp_alt'))
            print(f"  n={alt_n} parent_interp: {s_alt:.6f}")
            if s_alt < best_score:
                best_score = s_alt
                best_sequence = alt_init.copy()
    
    # Quick subgradient on top candidates
    candidates.sort(key=lambda x: x[0])
    for sc, seq, n_c, lbl in candidates[:3]:
        if time.monotonic() > explore_deadline:
            break
        quick_dl = min(explore_deadline, time.monotonic() + 20.0)
        f_opt, s_opt = fast_subgradient_optimize(seq, quick_dl, lr_init=0.01, top_k=5)
        if s_opt < sc:
            print(f"  {lbl} quick subgrad: {sc:.6f} -> {s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_sequence = f_opt.copy()
    
    print(f"\nAfter exploration: best={best_score:.6f}")
    
    # Phase 2: QP-based optimization on best candidate (900s)
    print("\n=== Phase 2: QP Optimization ===")
    qp_deadline = t0 + 1010.0
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        print(f"QP optimizing n={n_best}, start={best_score:.6f}")
        
        # Try multiple mu values
        for mu_try in [0.005, 0.01, 0.05]:
            if time.monotonic() > qp_deadline - 30:
                break
            mu_deadline = min(qp_deadline, time.monotonic() + (qp_deadline - time.monotonic()) * 0.5)
            f_qp, s_qp = qp_optimize(best_sequence, mu_deadline, mu_init=mu_try, k_init=15)
            if s_qp < best_score:
                best_score = s_qp
                best_sequence = f_qp.copy()
                print(f"  QP mu={mu_try}: improved to {best_score:.6f}")
        
        # Continue QP if time
        if time.monotonic() < qp_deadline - 30:
            f_qp, s_qp = qp_optimize(best_sequence, qp_deadline, mu_init=0.02, k_init=20)
            if s_qp < best_score:
                best_score = s_qp
                best_sequence = f_qp.copy()
                print(f"  QP continued: improved to {best_score:.6f}")
    
    print(f"After QP: best={best_score:.6f}")
    
    # Phase 3: Subgradient polish (300s)
    print("\n=== Phase 3: Subgradient Polish ===")
    sub_deadline = t0 + 1310.0
    
    if best_sequence is not None:
        f_sub, s_sub = fast_subgradient_optimize(
            best_sequence, sub_deadline, lr_init=0.003, top_k=7, momentum_decay=0.92)
        if s_sub < best_score:
            best_score = s_sub
            best_sequence = f_sub.copy()
            print(f"  Subgrad polish: improved to {best_score:.6f}")
    
    # Phase 4: LP refinement (150s)
    print("\n=== Phase 4: LP Refinement ===")
    lp_deadline = t0 + 1460.0
    
    if best_sequence is not None:
        f_lp, s_lp = lp_optimize(best_sequence, lp_deadline, step_init=0.05)
        if s_lp < best_score:
            best_score = s_lp
            best_sequence = f_lp.copy()
            print(f"  LP refinement: improved to {best_score:.6f}")
    
    # Phase 5: Final subgradient micro-polish
    print("\n=== Phase 5: Final Polish ===")
    final_deadline = deadline
    
    if best_sequence is not None and time.monotonic() < final_deadline - 5:
        f_final, s_final = fast_subgradient_optimize(
            best_sequence, final_deadline, lr_init=0.001, top_k=10, momentum_decay=0.95)
        if s_final < best_score:
            best_score = s_final
            best_sequence = f_final.copy()
            print(f"  Final polish: improved to {best_score:.6f}")
    
    # Verify
    if best_sequence is None:
        best_sequence = np.ones(1024)
    
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    
    if not np.isfinite(final_score):
        print("WARNING: Failed evaluation, using fallback")
        best_sequence = np.ones(1024)
        final_score = evaluate_sequence([float(x) for x in best_sequence])
    
    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, time={elapsed:.1f}s")
    
    return [float(x) for x in best_sequence]
