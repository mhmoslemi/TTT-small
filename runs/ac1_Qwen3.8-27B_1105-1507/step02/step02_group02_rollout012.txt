"""
Minimizes J(a) = 2*n*max(conv(a,a))/sum(a)^2 using:
1. Mirror descent (multiplicative weight updates) with softmax-subgradient of the max-convolution
2. Beta annealing to progressively focus on the true maximum
3. Multiple initializations (parent, random, structured) at several lengths
4. LP best-response polish for final refinement
5. Simulated-annealing restarts to escape local minima
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = fft(a, s)
    return np.real(ifft(fa * fa, s)[:2 * n - 1])


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None

    if w_sum < 1e-300:

    w_rev = w[::-1]

    s_fft = 1
    while s_fft < n + (2 * n - 1) - 1:
        s_fft <<= 1

    W = fft(w_rev, s_fft)
    conv_result = np.real(ifft(A * W, s_fft))
    # g[m] = 2 * conv_result[2n-2-m] for m = 0, ..., n-1
    g = 2.0 * conv_result[2 * n - 2 : n - 1 : -1]
    
    # Multiplicative weight (mirror descent) update on simplex
    # a_new[i] = a[i] * exp(-eta * g[i]) / Z
    log_shift = -eta * g
    log_shift -= log_shift.max()  # numerical stability
    unnormalized = a * np.exp(log_shift)
    z = unnormalized.sum()
    if z < 1e-300:
        a_new = np.ones(n) / n
    else:
        a_new = unnormalized / z
    
    return a_new, c_max


def optimize_mirror_descent(a0, deadline, max_iters=100000, beta_start=1.0, beta_end=200.0,
                            eta_start=None, min_eta=0.001):
    """
    Run mirror descent with beta annealing and adaptive eta.
    """
    n = len(a0)
    a = np.clip(a0, 1e-10, None).astype(np.float64)
    a /= a.sum()
    
    best = a.copy()
    best_score = compute_score(a)
    
    if eta_start is None:
        eta_start = 1.0 / np.sqrt(n)
    eta = eta_start
    
    no_improve = 0
    stuck_count = 0
    iter_count = 0
    
    while time.monotonic() < deadline and iter_count < max_iters:
        # Adaptive beta: gradually increase to focus on true max
        progress = min(1.0, iter_count / (0.9 * max_iters))
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = min(beta, beta_end)
        
        a_new, c_max = mirror_descent_step(a, beta, eta)
        
        # Ensure minimum values don't underflow to zero
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        score = compute_score(a_new)
        iter_count += 1
        
        if score < best_score:
            best = a_new.copy()
            best_score = score
            no_improve = 0
            stuck_count = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new  # still move forward
            
            if no_improve > 500:
                # Restart from best with fresh eta
                a = best.copy()
                eta = eta_start
                no_improve = 0
                stuck_count += 1
                if stuck_count > 5:
                    break
            elif no_improve > 200:
                eta *= 0.7
    
    return best, best_score


def lp_polish(a, deadline, max_iters=200):
    """Apply LP best-response iterations for final polish."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 1e-6, None)
    a /= a.sum()
    best = a.copy()
    best_score = compute_score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Try several step sizes
        improved = False
        for t in [0.3, 0.1, 0.05, 0.02]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new /= s
            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best = a_new.copy()
                a = a_new
                improved = True
                break
        
        if not improved:
            break
    
    return best, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    t = np.linspace(0, 1, n)
    
    # Resampled parent
    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t, t_old, parent), 1e-6)
        inits.append(resampled / resampled.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Exponential random
    r = rng.exponential(1.0, n)
    inits.append(r / r.sum())
    
    # Cosine (1st harmonic)
    c = 1.0 + 0.5 * np.cos(2 * np.pi * t)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # Cosine (2nd harmonic)
    c = 1.0 + 0.5 * np.cos(4 * np.pi * t)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # Bimodal
    c = np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # Random uniform
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())
    
    # "Sidelobe-suppressed" pattern
    c = 1.0 / (1.0 + 0.3 * np.abs(np.sin(np.pi * t * 4)))
    inits.append(c / c.sum())
    
    return inits


def propose_candidate():

    Main search function. Uses mirror descent with softmax-subgradient as the primary
    optimizer, combined with LP polish, multiple initializations, and simulated annealing restarts.
    """

        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            sc = compute_score(p)
            if np.isfinite(sc):
                best = p.copy()
                best_score = sc
                print(f"Parent: J={sc:.6f} (n={p.size})", flush=True)
    except (NameError, AttributeError):
        pass
    
    if best is None:
        best = np.ones(512) / 512
        best_score = compute_score(best)
        print(f"Default uniform: J={best_score:.6f} (n=512)", flush=True)
    
    # === PHASE 1: Mirror descent from parent (50% of time) ===
    md1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.45
    print(f"\n=== Phase 1: Mirror descent from parent (n={best.size}) ===", flush=True)
    t0 = time.monotonic()
    
    a = best.copy()
    a = np.clip(a, 1e-6, None)
    a /= a.sum()

        beta_start=2.0, beta_end=300.0,
        eta_start=2.0 / np.sqrt(best.size)
    )
    if score < best_score:
        best = result.copy()
        best_score = score
        print(f"  Phase 1 result: J={score:.6f}", flush=True)
    
    # Continue with higher beta for fine-tuning
    md1b_deadline = min(md1_deadline + (deadline - time.monotonic()) * 0.1, deadline)
    result, score = optimize_mirror_descent(
        best, md1b_deadline,
        max_iters=100000,
        beta_start=100.0, beta_end=1000.0,
        eta_start=0.5 / np.sqrt(best.size)
    )
    if score < best_score:
        best = result.copy()
        best_score = score
        print(f"  Phase 1b result: J={score:.6f}", flush=True)
    else:
        print(f"  Phase 1b: no improvement (best={best_score:.6f})", flush=True)
    
    print(f"  Phase 1 total time: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)
    
    # === PHASE 2: Mirror descent from diverse initializations (25% of time) ===
    phase2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.4
    print(f"\n=== Phase 2: Diverse initializations ===", flush=True)
    t0 = time.monotonic()
    
    # Try at the best length
    n_best = best.size
    lengths_to_try = [n_best]
    # Also try some other lengths
    for n_alt in [n_best // 2, n_best * 2, 384, 768]:
        if n_alt != n_best and 128 <= n_alt <= 2048:
            lengths_to_try.append(n_alt)
    
    remaining_for_phase2 = phase2_deadline - time.monotonic()
    per_length = max(30, remaining_for_phase2 / len(lengths_to_try))
    
    for n in lengths_to_try:
        if time.monotonic() > phase2_deadline:
            break
        
        if n != n_best:
            # Resample best to this length
            t_old = np.linspace(0, 1, n_best)
            t_new = np.linspace(0, 1, n)
            base = np.maximum(np.interp(t_new, t_old, best), 1e-6)
            base /= base.sum()
        else:
            base = best.copy()
        
        inits = generate_initializations(n, rng, base)
        
        per_init_time = max(10, per_length / max(len(inits), 1))
        
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase2_deadline:
                break
            a0 = np.clip(a0, 1e-6, None)
            a0 /= a0.sum()
            
            init_deadline = time.monotonic() + per_init_time
            
            # Quick mirror descent
            result, score = optimize_mirror_descent(
                a0, init_deadline,
                max_iters=50000,
                beta_start=5.0, beta_end=200.0,
                eta_start=3.0 / np.sqrt(n)
            )
            if score < best_score:
                best = result.copy()
                best_score = score

    phase3_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.5
    print(f"\n=== Phase 3: LP Polish ===", flush=True)
    t0 = time.monotonic()
    
    n = best.size
    # Only do LP if n is small enough
    if n <= 768:
        result, score = lp_polish(best, phase3_deadline, max_iters=100)
        if score < best_score:
            best = result.copy()
            best_score = score
            print(f"  LP polish: J={score:.6f}", flush=True)
    else:
        print(f"  Skipping LP (n={n} too large)", flush=True)
    
    # Mirror descent polish after LP
    if time.monotonic() < phase3_deadline:
        result, score = optimize_mirror_descent(
            best, phase3_deadline,
            max_iters=50000,
            beta_start=200.0, beta_end=500.0,
            eta_start=0.3 / np.sqrt(n)
        )
        if score < best_score:
            best = result.copy()
            best_score = score
            print(f"  MD polish after LP: J={score:.6f}", flush=True)
    
    print(f"  Phase 3 total time: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)
    
    # === PHASE 4: Simulated annealing restarts (remaining time) ===
    if time.monotonic() < deadline:
        print(f"\n=== Phase 4: Simulated Annealing Restarts ===", flush=True)
        t0 = time.monotonic()
        n = best.size
        
        remaining = deadline - time.monotonic()
        n_restarts = max(5, int(remaining / 20))
        n_restarts = min(n_restarts, 50)
        
        noise_scales = [0.02, 0.05, 0.1, 0.03, 0.08, 0.01, 0.15]
        
        for trial in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            
            scale = noise_scales[trial % len(noise_scales)]
            
            # Different perturbation strategies
            strategy = trial % 3
            if strategy == 0:
                # Gaussian noise
                pert = best + rng.normal(0, scale, n)
            elif strategy == 1:
                # Multiplicative noise
                pert = best * np.exp(rng.normal(0, scale, n))
            else:
                # Coordinate swap perturbation
                pert = best.copy()
                n_swaps = max(5, n // 20)
                for _ in range(n_swaps):
                    i, j = rng.choice(n, 2, replace=False)
                    delta = (pert[i] - pert[j]) * scale
                    pert[i] -= delta
                    pert[j] += delta
            
            pert = np.clip(pert, 1e-8, None)
            s = pert.sum()
            if s < 0.01:
                continue
            pert /= s
            
            # Quick optimization from perturbation
            restart_deadline = time.monotonic() + max(5, remaining / (n_restarts - trial))
            result, score = optimize_mirror_descent(
                pert, restart_deadline,
                max_iters=30000,
                beta_start=50.0, beta_end=500.0,
                eta_start=1.0 / np.sqrt(n)
            )
            if score < best_score:
                best = result.copy()
                best_score = score
                print(f"  NEW BEST (restart {trial}): J={score:.6f}", flush=True)
        
        print(f"  Phase 4 total time: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)
    
    # === Final: one more MD pass at very high beta ===
    if time.monotonic() < deadline - 5:
        n = best.size
        result, score = optimize_mirror_descent(
            best, deadline,
            max_iters=100000,
            beta_start=500.0, beta_end=2000.0,
            eta_start=0.1 / np.sqrt(n)
        )
        if score < best_score:
            best = result.copy()
            best_score = score
    
    # Final validation
    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    
    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
