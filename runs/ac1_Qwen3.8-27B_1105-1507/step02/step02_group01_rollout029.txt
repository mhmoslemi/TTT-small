def golden_section_1d(f, lo, hi, tol=1e-8, max_iter=50):
    """Minimize f on [lo, hi] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:

            d = c
            fd = fc
            c = hi - gr * (hi - lo)

            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
