    weights = weights / w_sum
    
    # Gradient of LSE w.r.t. a: grad_a = 2 * (weights correlated with a reversed)
    # d(conv_k)/d(a_i) = a[k-i] + a[i-k] (for valid indices)
    # Actually, d(conv_k)/d(a_i) = sum_j delta(k - i - j) a[j] = a[k-i] (if valid)
    # But since conv_k = sum_i a[i]*a[k-i], d(conv_k)/d(a_i) = a[k-i] + (if k-i != i) a[i-k]
    # Wait, for i != k-i: d(conv_k)/d(a_i) = a[k-i] + a[i-k]... no.
    # conv_k = sum_j a[j]*a[k-j]. d/d(a_i) = a[k-i] + (if i != k-i, which is always true unless k is even and i=k/2) a[i-k+...] 
    # Actually: d(conv_k)/d(a_i) = a[k-i] if 0 <= k-i <= n-1, plus a[k-i] again if... 
    # No, simpler: conv_k = sum_j a[j] * a[k-j]. 
    # d(conv_k)/d(a_i) = a[k-i] (from the first factor being a_i) + a[k-i] (from the second factor being a_i, i.e., k-j=i means j=k-i)
    # Wait: the second term: a[k-j] = a_i when j = k-i. So d(conv_k)/d(a_i) = a[k-i] + a[k-i] = 2*a[k-i] (if k-i is in [0,n-1])
    # But if k-i = i (i.e., k = 2i), then we're differentiating a_i * a_k/2, which gives 2*a_i.
    # Actually, it's always 2*a[k-i] when 0 <= k-i <= n-1, because:
    # conv_k = sum_j a[j]*a[k-j]
    # The term involving a_i in the first slot: j=i, gives a_i * a[k-i]
    # The term involving a_i in the second slot: k-j=i, j=k-i, gives a[k-i] * a_i
    # These are the same term! So d(conv_k)/d(a_i) = a[k-i] + a[k-i] = 2*a[k-i]
    # Unless k-i = i (k=2i), in which case it's the same term counted twice, giving 2*a_i.
    # Either way, it's 2*a[k-i] (or 2*a_i when k=2i).
    
    # So grad_a[i] = sum_k weights[k] * 2 * a[k-i] (where k-i is in [0,n-1])
    # This is 2 * (weights * a) autocorrelation at lag... 
    # grad_a = 2 * correlate(weights, a, 'full')[n-1:n-1+n]
    # Wait, let me be more careful.
    # grad_a[i] = 2 * sum_k weights[k] * a[k-i] where 0 <= k-i <= n-1, i.e., i <= k <= i+n-1
    # = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    # This is the cross-correlation of weights and a, evaluated at lag i.
    
    c = np.correlate(weights, a, mode='full')
    # c[m] = sum_j weights[j]*a[j+m-n+1]... let me just use the direct formula
    # Actually, np.correlate(weights, a, 'full')[i+n-1] = sum_j weights[j]*a[j+i] for valid j
    # Hmm, let me just compute it directly with FFT for efficiency
    
    # For large n, use FFT
    if n > 100:
        size = 1
        while size < 2 * n:
            size <<= 1
        fw = np.fft.rfft(weights, size)
        fa_ = np.fft.rfft(a, size)
        # We want grad_a[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
        # This is the cross-correlation of (weights reversed) and a, or...
        # Let me think: sum_j weights[i+j]*a[j] = sum_j a[j]*weights[i+j]
        # This is (a * weights_flipped)[i] where * is correlation
        # In FFT: if A = FFT(a), W = FFT(weights), then correlation = IFFT(conj(A) * W)
        # But I need to be careful about the indexing.
        
        # Alternative: use the fact that grad_a[i] = 2 * sum_k weights[k] * a[k-i] (k-i in [0,n-1])
        # Let b[k] = weights[k] for k in [0, 2n-2], 0 otherwise
        # Then grad_a[i] = 2 * sum_{j=0}^{n-1} b[i+j] * a[j] = 2 * (b conv a)[i] (cross-correlation)
        # In FFT: cross-correlation = IFFT(conj(FFT(b)) * FFT(a))
        
        fb = np.fft.rfft(weights, size)
        grad_a = 2.0 * np.fft.irfft(np.conj(fb) * fa_, size)[:n]
    else:
        # Direct computation
        grad_a = np.zeros(n)
        for i in range(n):
            k_min = i
            k_max = min(2*n-2, i+n-1)
            if k_min <= k_max:
                grad_a[i] = 2.0 * np.sum(weights[k_min:k_max+1] * a[:k_max-k_min+1])
    
    # Now, grad w.r.t. x (where a = softmax(x)):
    # a_i = exp(x_i) / sum(exp(x_j))
    # d(a_i)/d(x_l) = a_i * (delta_il - a_l)
    # grad_x[l] = sum_i grad_a[i] * d(a_i)/d(x_l) = sum_i grad_a[i] * a_i * (delta_il - a_l)
    # = grad_a[l] * a_l - a_l * sum_i grad_a[i] * a_i
    # = a_l * (grad_a[l] - sum_i grad_a[i] * a_i)
    
    inner = np.sum(grad_a * a)
    grad_x = a * (grad_a - inner)
    
    # Objective value (LSE)
    obj_val = np.log(w_sum) / tau + max_conv  # LSE = max_conv + log(w_sum)/tau
    
    return obj_val, grad_x

def lse_optimize(x_init, tau, max_iter=5000, tol=1e-10):
    """Optimize using L-BFGS-B on the LSE approximation."""
    def fun(x):
        val, grad = lse_objective_and_grad(x, tau)
        return val, grad
    
    res = minimize(fun, x_init, method='L-BFGS-B', jac=True,
                   options={'maxiter': max_iter, 'ftol': tol, 'gtol': 1e-8,
                            'maxls': 50})
    return res.x, res.fun

def build_A_matrix(a):
    """Build the Toeplitz constraint matrix for the LP: A[k, i] such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = a[j]
    return A

def solve_lp_for_g(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0."""
    n = len(a)
    # For large n, building the full matrix is expensive
    if n > 500:
        return None, None
    
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        # A[k, i] = a[k - i] for i in [j_min, j_max]
        # When i = j_min: a[k - j_min] = a[k - max(0, k-n+1)]
        # When i = j_max: a[k - j_max] = a[k - min(n-1, k)]
        if j_min <= j_max:
            A[k, j_min:j_max+1] = a[k - j_max : k - j_min + 1][::-1]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception as e:
        pass
    return None, None

def lp_step(a, M, t=0.5):
    """One LP-based update step."""
    g, sum_g = solve_lp_for_g(a, M)
    if g is None or sum_g <= 0:
        return None
    g = g / sum_g
    a_new = (1 - t) * a + t * g
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return None
    a_new = a_new / s
    return a_new

def smoothed_subgrad_step(a, tau, step):
    """One smoothed subgradient step."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    
    # Compute gradient
    size = 1
    while size < 2 * n:
        size <<= 1
    fw = np.fft.rfft(weights, size)
    fa = np.fft.rfft(a, size)
    grad_a = 2.0 * np.fft.irfft(np.conj(fw) * fa, size)[:n]
    
    # Center for sum constraint
    grad_centered = grad_a - np.mean(grad_a)
    
    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a
    a_new = a_new / s
    return a_new

def optimize_sequence(a, max_time=60.0, use_lp=True, verbose=False):
    """Main optimization loop for a given initial sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Phase 1: LSE optimization with increasing tau (annealing)
    taus = [3.0, 5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0]
    
    for tau in taus:
        if time.monotonic() - start > max_time * 0.5:
            break
        remaining = max_time * 0.5 - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Convert to x (log-softmax parameterization)
        a_safe = np.maximum(a, 1e-15)
        x = np.log(a_safe / np.sum(a_safe))
        
        # L-BFGS-B optimization
        try:
            x_opt, _ = lse_optimize(x, tau, max_iter=2000)
            a_safe = np.maximum(a, 1e-15)
            x = np.log(a_safe / np.sum(a_safe))
            
            exp_x = np.exp(x_opt - np.max(x_opt))
            a_new = exp_x / np.sum(exp_x)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    LSE tau={tau}: obj={best_obj:.6f}")
            a = a_new
        except Exception:
            pass
    
    # Phase 2: Alternating subgradient + LP
    step = 0.05
    stall = 0
    patience = 100
    
    while time.monotonic() - start < max_time * 0.8:
        remaining = max_time * 0.8 - (time.monotonic() - start)
        if remaining < 2:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        obj = 2.0 * n * M / (np.sum(a) ** 2)
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > patience:
            # Try perturbation
            noise = np.random.default_rng(hash((n, int(stall))) % 2**31).normal(0, 0.03, n)
            a_pert = a * np.exp(noise)
            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s > 1e-10:
                a_pert = a_pert / s
                obj_p = compute_obj(a_pert)
                if obj_p < best_obj:
                    a = a_pert
                    best_obj = obj_p
                    best_a = a.copy()
                    stall = 0
                else:
                    step *= 0.5
                    if step < 1e-10:
                        break
            else:
                break
        
        # Subgradient step
        tau = 20.0 + 50.0 * (stall / patience)
        a = smoothed_subgrad_step(a, tau, step)
        
        # LP step (if n is small enough)
        if use_lp and n <= 400 and remaining > 5:
            conv = compute_conv(a)
            M = float(np.max(conv))
            a_lp = lp_step(a, M, t=0.4)
            if a_lp is not None:
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_obj:
                    best_obj = obj_lp
                    best_a = a_lp.copy()
                    a = a_lp
                    stall = 0
                    if verbose:
                        print(f"    LP: obj={best_obj:.6f}")
        
        # Slowly decrease step
        step *= 0.999
    
    # Phase 3: Final polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        for _ in range(int(remaining // 5)):
            if time.monotonic() - start > max_time:
                break
            # Small LSE step
            a_safe = np.maximum(a, 1e-15)
            x = np.log(a_safe / np.sum(a_safe))
            try:
                x_opt, _ = lse_optimize(x, tau=200.0, max_iter=500)
                exp_x = np.exp(x_opt - np.max(x_opt))
                a_new = exp_x / np.sum(exp_x)
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                    a = a_new
            except Exception:
                pass
        
        # Final LP polish
        if n <= 350:
            for _ in range(20):
                if time.monotonic() - start > max_time:
                    break
                conv = compute_conv(a)
                M = float(np.max(conv))
                a_lp = lp_step(a, M, t=0.3)
                if a_lp is not None:
                    obj_lp = compute_obj(a_lp)
                    if obj_lp < best_obj:
                        best_obj = obj_lp
                        best_a = a_lp.copy()
                        a = a_lp
                    else:
                        break
    
    return best_a, best_obj

def generate_initialization(n, rng, method=None):
    """Generate a random initialization."""
    if method is None:
        method = rng.integers(0, 8)
    
    if method == 0:
        # Uniform
        a = np.ones(n)
    elif method == 1:
        # Cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n + rng.uniform(0, 2 * np.pi)))
    elif method == 2:
        # Two bumps
        c1 = n // 3
        c2 = 2 * n // 3
        w = n // 10
        a = np.zeros(n)
        a = np.exp(-0.5 * ((np.arange(n) - c1) / w) ** 2)
        a += np.exp(-0.5 * ((np.arange(n) - c2) / w) ** 2)
    elif method == 3:
        # Random
        a = rng.random(n)
    elif method == 4:
        # Ramp
        a = np.linspace(0.1, 1.0, n)
    elif method == 5:
        # Asymmetric bump
        center = rng.uniform(0.2 * n, 0.8 * n)
        width = rng.uniform(0.1 * n, 0.4 * n)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    elif method == 6:
        # Sinusoidal
        freq = rng.uniform(1.0, 5.0)
        a = np.abs(np.sin(np.pi * freq * np.arange(n) / n)) + 0.1
    else:
        # Combination
        a = 0.5 + 0.3 * np.cos(2 * np.pi * rng.uniform(1, 3) * np.arange(n) / n)
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf

def propose_candidate(seed=42, budget_s=1500):
    """
    Improved optimization combining LSE (L-BFGS-B), LP, and subgradient methods.
    Explores multiple sequence lengths with multi-start.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    # Determine n values to explore
    n_values = [50, 80, 100, 120, 150, 180, 200, 250, 300, 350, 400, 500]
    if parent is not None:
        n_parent = len(parent)
        n_values = sorted(set(n_values + [n_parent, max(20, n_parent - 50), min(600, n_parent + 50)]))
    
    print(f"Exploring n values: {n_values}")
    print(f"Budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")
    
    # First pass: quick optimization of each n
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        # Allocate time based on n
        if n == (parent.size if parent is not None else 250):
            alloc = min(remaining * 0.2, 120.0)
        elif n <= 100:
            alloc = min(remaining * 0.05, 30.0)
        elif n <= 200:
            alloc = min(remaining * 0.08, 60.0)
        elif n <= 350:
            alloc = min(remaining * 0.1, 80.0)
        else:
            alloc = min(remaining * 0.08, 60.0)
        
        alloc = max(10.0, alloc)
        
        if verbose := (n in [100, 150, 200, 250, 300]):
            print(f"\nN={n}: {remaining:.0f}s remaining, allocating {alloc:.0f}s")
        
        # Multiple initializations
        n_inits = max(2, int(alloc / 15))
        for init_idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            
            a_init = generate_initialization(n, rng)
            
            # If we have a parent of the same length, also try starting from it
            if parent is not None and len(parent) == n and init_idx == 0:
                a_init = parent.copy()
            
            use_lp = n <= 400
            a_opt, obj = optimize_sequence(a_init, max_time=alloc / n_inits,
                                           use_lp=use_lp, verbose=verbose and init_idx == 0)
            
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                if verbose:
                    print(f"  UPDATE: n={n}, init={init_idx}, obj={best_score:.6f}")
    
    # Second pass: intensive optimization of the best n
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        print(f"\nIntensive optimization: n={n_best}, {remaining:.0f}s remaining, current obj={best_score:.6f}")
        
        # Try multiple restarts from the best solution with perturbations
        for restart in range(10):
            if time.monotonic() > deadline - 10:
                break
            
            a_cur = best_seq.copy()
            # Perturb
            noise_scale = 0.05 + 0.02 * restart
            noise = rng.normal(0, noise_scale, n_best)
            a_pert = a_cur * np.exp(noise)
            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s > 1e-10:
                a_pert = a_pert / s
            
            alloc = min(30.0, (deadline - time.monotonic() - 10) / (10 - restart))
            alloc = max(5.0, alloc)
            
            use_lp = n_best <= 400
            a_opt, obj = optimize_sequence(a_pert, max_time=alloc,
                                           use_lp=use_lp, verbose=False)
            
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: obj={best_score:.6f}")
    
    # Final intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        print(f"\nFinal polish: {remaining:.0f}s remaining")
        
        a_cur = best_seq.copy()
        
        # Intensive LSE polishing
        while time.monotonic() < deadline - 5:
            a_safe = np.maximum(a_cur, 1e-15)
            x = np.log(a_safe / np.sum(a_safe))
            
            # Use very high tau for sharp focus on the peak
            for tau in [100, 200, 500]:
                if time.monotonic() > deadline - 5:
                    break
                try:
                    x_opt, _ = lse_optimize(x, tau, max_iter=1000)
                    exp_x = np.exp(x_opt - np.max(x_opt))
                    a_new = exp_x / np.sum(exp_x)
                    obj = compute_obj(a_new)
                    if obj < best_score:
                        best_score = obj
                        best_seq = a_new.copy()
                        a_cur = a_new
                        x = np.log(np.maximum(a_new, 1e-15) / np.sum(a_new))
                        print(f"  Final polish tau={tau}: obj={best_score:.6f}")
                    else:
                        break
                except Exception:
                    break
            
            # LP polish
            if n_best <= 350 and time.monotonic() < deadline - 10:
                for _ in range(10):
                    if time.monotonic() > deadline - 5:
                        break
                    conv = compute_conv(a_cur)
                    M = float(np.max(conv))
                    a_lp = lp_step(a_cur, M, t=0.2)
                    if a_lp is not None:
                        obj_lp = compute_obj(a_lp)
                        if obj_lp < best_score:
                            best_score = obj_lp
                            best_seq = a_lp.copy()
                            a_cur = a_lp
                            print(f"  Final polish LP: obj={best_score:.6f}")
                        else:
                            break
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)
    
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    
    # Validate
    result = [float(x) for x in best_seq]
