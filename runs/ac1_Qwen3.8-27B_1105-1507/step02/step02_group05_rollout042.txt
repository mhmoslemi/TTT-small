import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def spectral_reweighting(a, alpha=0.3):
    """Flatten the periodogram slightly and inverse DFT (clip to non-negative)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    power = np.abs(fa) ** 2
    # Blend with flat power spectrum
    log_power = np.log1p(power)
    log_power = (1 - alpha) * log_power + alpha * log_power.max()
    new_power = np.expm1(log_power)
    # Preserve phase
    phase = np.angle(fa)
    new_fa = np.sqrt(new_power) * np.exp(1j * phase)
    new_a = np.fft.irfft(new_fa, s)[:n]
    new_a = np.maximum(new_a, 0.0)
    if new_a.sum() < 1e-10:
        return a.copy()
    return new_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = compute_J(a)
        if np.isfinite(score) and score < self.best_score - 1e-12:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Aggressive line search: try many step sizes
        base_t = max(0.03, 0.4 * (1.0 + np.cos(np.pi * it / max_iters)) / 2)
        t_candidates = np.unique(np.concatenate([
            np.array([base_t]),
            np.linspace(0.01, 1.0, 8),
            np.array([base_t * 0.5, base_t * 1.5, base_t * 2.0, base_t * 0.25])
        ]))
        t_candidates = t_candidates[(t_candidates > 0) & (t_candidates <= 1.0)]

        best_a = None
        best_J = current_J
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # No improvement from LP step; try coordinate descent
            if it % cd_interval == 0 and a.size <= 512:
                a_cd, _ = coord_descent_step(a)
                a_cd = np.maximum(a_cd, 0.0)
                s = a_cd.sum()
                if s > 1e-10:
                    a_cd /= s
                    J_cd = compute_J(a_cd)
                    if J_cd < current_J - 1e-12:
                        a = a_cd
                        current_J = J_cd
                        tracker.update(a)
                        continue
            break
        else:
            a = best_a
            current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    # Final coordinate descent pass
    if a.size <= 256 and time.monotonic() < deadline:
        for _ in range(3):
            if time.monotonic() > deadline:
                break
            a_cd, _ = coord_descent_step(a)
            a_cd = np.maximum(a_cd, 0.0)
            s = a_cd.sum()
            if s > 1e-10:
                a_cd /= s
                J_cd = compute_J(a_cd)
                if J_cd < current_J - 1e-12:
                    a = a_cd
                    current_J = J_cd
                else:
                    break
    tracker.update(a)
    return a


def make_inits_1sqrt(n, rng):
    """Generate initializations based on 1/sqrt envelope."""
    t = np.arange(n, dtype=np.float64)
    inits = []

    # Pure 1/sqrt
    a = 1.0 / np.sqrt(t + 0.5)
    inits.append(a)

    # 1/sqrt with raised cosine modulation
    for freq in [1, 2, 3]:
        mod = 1.0 + 0.15 * np.cos(2.0 * np.pi * freq * t / n)
        inits.append(a * mod)

    # 1/sqrt with linear tilt
    inits.append(a * (1.0 + 0.1 * (t / n - 0.5)))

    # 1/sqrt with exponential taper at edges
    taper = np.exp(-2.0 * ((t / n - 0.5) ** 2))
    inits.append(a * (1.0 + 0.1 * (1.0 - taper)))

    # Symmetric V-shape: 1/sqrt(min(j, n-1-j) + 0.5)
    d = np.minimum(t, n - 1 - t).astype(np.float64)
    inits.append(1.0 / np.sqrt(d + 0.5))

    # 1/sqrt with small random perturbation
    inits.append(a * (1.0 + 0.05 * rng.standard_normal(n)))

    # Different power: 1/(j+0.5)^0.45
    inits.append(1.0 / (t + 0.5) ** 0.45)

    # 1/(j+0.5)^0.55
    inits.append(1.0 / (t + 0.5) ** 0.55)

    # Uniform (baseline)
    inits.append(np.ones(n))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: 1/sqrt init -> LP mirror descent -> coord descent
    -> spectral reweighting -> SA restarts -> L-BFGS-B -> final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: LP from 1/sqrt inits at n=128 ===
    n = 128
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 1: LP n={n} {'='*50}", flush=True)

    inits = make_inits_1sqrt(n, rng)
    if parent is not None and parent.size != n:
        inits.insert(0, resample_seq(parent, n))

    for idx, a0 in enumerate(inits):
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0.astype(np.float64), 1e-6)
        a0 /= a0.sum()
        J0 = compute_J(a0)
        print(f"  Init {idx}: J={J0:.6f}", flush=True)
        iters = max(80, min(400, int(120000 // n)))
        a = run_lp_with_line_search(a0, iters, deadline, tracker, cd_interval=8)

    tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP at n=256 ===
    if time.monotonic() < deadline:
        n2 = 256
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: LP n={n2} {'='*50}", flush=True)

        inits2 = make_inits_1sqrt(n2, rng)
        if tracker.best is not None:
            inits2.insert(0, resample_seq(tracker.best, n2))

        for idx, a0 in enumerate(inits2[:6]):  # Limit number of inits
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            J0 = compute_J(a0)
            print(f"  Init {idx}: J={J0:.6f}", flush=True)
            iters = max(50, min(250, int(80000 // n2)))
            a = run_lp_with_line_search(a0, iters, deadline, tracker, cd_interval=10)

        tracker.report(f"Phase2 n={n2}: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Spectral reweighting iterations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 3: Spectral n={n} {'='*50}", flush=True)

        for alpha in [0.1, 0.2, 0.3, 0.2, 0.15]:
            if time.monotonic() > deadline:
                break
            a_spec = spectral_reweighting(tracker.best, alpha)
            a_spec /= a_spec.sum()
            J_spec = compute_J(a_spec)
            print(f"  Spectral alpha={alpha}: J={J_spec:.6f}", flush=True)
            if J_spec < tracker.best_score:
                # Polish with LP
                a = run_lp_with_line_search(a_spec, 100, deadline, tracker, cd_interval=5)

        # Also try: take current best, blend with its spectral version
        a_blend = 0.7 * tracker.best + 0.3 * spectral_reweighting(tracker.best, 0.5)
        a_blend = np.maximum(a_blend, 0.0)
        a_blend /= a_blend.sum()
        tracker.update(a_blend)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Simulated Annealing with LP ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 4: SA n={n} {'='*50}", flush=True)

        temperature = 0.1
        n_restarts = max(20, int((deadline - time.monotonic()) / 15))
        n_restarts = min(n_restarts, 80)

        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            #多元化 perturbations
            noise_scale = temperature * (1.0 + 0.5 * np.sin(r * 0.3))
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, noise_scale, n)
            a = np.maximum(a, 1e-5)
            a /= a.sum()

            iters = max(15, min(60, int(120 - r)))
            a = run_lp_with_line_search(a, iters, deadline, tracker, cd_interval=5)
            temperature *= 0.96

        tracker.report(f"SA: {time.monotonic()-t0:.1f}s")

    # === Phase 5: L-BFGS-B refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 5: L-BFGS-B n={n} {'='*50}", flush=True)

        def lbb_objective(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            return float(2 * n * b.max())

        bounds_list = [(0.0, None)] * n
        x0 = tracker.best.astype(np.float64)
        x0 = x0 / x0.sum()

        res = minimize(lbb_objective, x0, method='L-BFGS-B',
                       bounds=bounds_list,
                       options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18, 'gtol': 1e-12})
        if res.fun < 1e10:
            a_res = np.maximum(res.x, 0.0)
            if a_res.sum() > 1e-8:
                a_res /= a_res.sum()
                J_res = compute_J(a_res)
                print(f"  L-BFGS-B: J={J_res:.8f}", flush=True)
                tracker.update(a_res)

        tracker.report(f"L-BFGS-B: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final slow LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        if n <= 256:
            t0 = time.monotonic()
            print(f"\n{'='*50} Phase 6: Final LP n={n} {'='*50}", flush=True)
            # Use very small steps for careful convergence
            a = tracker.best.copy().astype(np.float64)
            a = np.maximum(a, 1e-6)
            a /= a.sum()

            for it in range(500):
                if time.monotonic() > deadline:
                    break
                b = fast_conv(a)
                M = float(b.max())
                g = lp_best_response(a, M)
                if g is None:
                    break
                g = np.maximum(g, 0.0)
                g_sum = g.sum()
                if g_sum < 1e-10:
                    break
                g /= g_sum

                # Very small step
                t = 0.01
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    break
                a_trial /= s
                J_trial = compute_J(a_trial)
                J_current = compute_J(a)
                if J_trial < J_current - 1e-14:
                    a = a_trial
                else:
                    # Try even smaller step
                    t = 0.003
                    a_trial = (1.0 - t) * a + t * g
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        break
                    a_trial /= s
                    J_trial = compute_J(a_trial)
                    if J_trial < J_current - 1e-14:
                        a = a_trial
                    else:
                        break

                if it % 50 == 0:
                    tracker.update(a)

            # Coordinate descent polish
            if a.size <= 256:
                for _ in range(5):
                    if time.monotonic() > deadline:
                        break
                    a_cd, _ = coord_descent_step(a)
                    a_cd = np.maximum(a_cd, 0.0)
                    s = a_cd.sum()
                    if s > 1e-10:
                        a_cd /= s
                        J_cd = compute_J(a_cd)
                        if J_cd < compute_J(a) - 1e-12:
                            a = a_cd
                        else:
                            break
            tracker.update(a)
            tracker.report(f"Final LP: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        a = 1.0 / np.sqrt(np.arange(128, dtype=np.float64) + 0.5)
        a /= a.sum()
        tracker.best = a
        tracker.best_score = compute_J(a)

    # Final validation
    final_score = compute_J(tracker.best)
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={final_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
