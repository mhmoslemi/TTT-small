import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s*s))

def simplex_project(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    # This is the Euclidean projection onto the probability simplex scaled by target_sum
    # Use the water-filling algorithm
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # We want to find τ such that sum(max(v - τ, 0)) = target_sum
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - target_sum) / rho > 0
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho_max = np.max(np.where(cond)[0]) + 1
    tau = (cssv[rho_max-1] - target_sum) / rho_max
    return np.maximum(v - tau, 0)

    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n) / (resample_seq(parent, n).sum() + 1e-15))
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    inits[-1] /= inits[-1].sum()
    
    # Cosine-based
    for freq in [1, 2, 3, 4, 5, 7, 8]:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    
    # Raised cosine
    a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(a / a.sum())
    
    # Triangular
    a = 1.0 - 2.0 * np.abs(t - 0.5)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    
    # Two-lobe
    a = 1.0 + 0.5 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    
    # Random Fourier features
    for _ in range(3):
        coeffs = rng.normal(0, 1, min(n, 20))
        a = np.ones(n)
        for j, c in enumerate(coeffs):
            a += c * np.cos(2 * np.pi * (j + 1) * t + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_lp_mirror_descent(a, max_iters, deadline, tracker, k_frac=0.25):
    """LP mirror descent with top-k constraints and adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    k = max(16, int(k_frac * (2 * n - 1)))
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response_topk(a, M, k)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Adaptive step: try several step sizes
        base_t = max(0.03, 0.4 * (1.0 - it / float(max(2*max_iters, 1))))
        candidates = [base_t, base_t * 2.0, base_t * 0.5, 0.1, 0.2]
        candidates = [t for t in candidates if 0 < t <= 1.0]
        
        best_J = compute_J(a)
        best_a = a
        
        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is a:
            # No improvement, try very small step
            t = 0.01
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial /= s
            if compute_J(a_trial) < best_J:
                a = a_trial
            else:
                break
        else:
            a = best_a
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def run_multiplicative_descent(a, max_iters, deadline, tracker):
    """Multiplicative LSE gradient descent."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    n = len(a)
    
    tau = 5.0
    eta = 0.5
    best_local = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        a = multiplicative_step(a, tau, eta)
        
        if it % 10 == 0:
            J = compute_J(a)
            if J < best_local:
                best_local = J
                tracker.update(a)
        
        # Anneal
        if it % 50 == 0:
            tau = min(tau * 1.5, 50.0)
            eta = max(eta * 0.9, 0.05)
    
    tracker.update(a)
    return a
