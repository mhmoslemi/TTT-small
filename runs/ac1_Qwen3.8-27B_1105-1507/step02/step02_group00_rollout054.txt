def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    w /= np.sum(w)

    # This is 2 * (w * flipped_a) convolved appropriately
    # More precisely: grad[j] = 2 * sum_{k=j}^{j+n-1} w[k] * a[k-j]
    #                   = 2 * (w * a)[j] in the cross-correlation sense
    
    # Let's compute it as: grad = 2 * np.convolve(w, a)[:n] ... no
    # Actually: grad[j] = 2 * sum_i w[j+i] * a[i] for i=0..n-1
    # This is 2 * (reversed_w_cross_correlation_with_a)
    # = 2 * np.correlate(w, a, mode='valid') ... hmm

def lse_descent(a_init, n_steps=5000, beta_start=10, beta_end=100, 
                momentum=0.9, verbose=False, deadline=None):
    """
    Log-sum-exp gradient descent with adaptive beta and Nesterov momentum.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=constant.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s0 = np.sum(a)
    a = a / s0  # normalize to sum=1
    
    v = np.zeros(n)  # velocity for Nesterov
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Adaptive beta: starts small, grows over iterations
        frac = step / max(n_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Nesterov lookahead
        a_lookahead = a + momentum * v
        
        # Compute gradient at lookahead point
        c = np.convolve(a_lookahead, a_lookahead)
        c_max = np.max(c)
        w = np.exp(beta * (c - c_max))
        w_sum = np.sum(w)
        w = w / w_sum
        
        # Gradient
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = 2.0 * np.dot(w[j:j+n], a_lookahead)
        
        # Update velocity and position
        v = momentum * v - grad
        a_new = a + v
        
        # Project: non-negativity and normalization
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        # Check objective
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and step % 500 == 0:
            print(f"    LSE step={step}, beta={beta:.1f}, obj={best_obj:.6f}")
    
    return best_a, best_obj

def lp_alternate(a_init, max_iter=200, deadline=None):
    """Improved LP alternating method."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Solve LP: max sum(g) s.t. conv(a,g) <= max(conv(a,a)), g >= 0
        conv_aa = np.convolve(a, a)
        M = np.max(conv_aa)
        
        # Build constraint matrix
        # conv(a,g)[k] = sum_i a[i]*g[k-i] <= M for k=0..2n-2
        # For variable g[j], it appears in conv(a,g)[k] with coefficient a[k-j]
        # So constraint k: sum_j a[k-j]*g[j] <= M (for valid k-j)
        
        # Use scipy linprog
        c = -np.ones(n)  # minimize -sum(g)
        A_ub = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A_ub[k, j] = a[idx]
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Adaptive blending: use more aggressive blending when improvement is large
        obj_g = compute_obj(g)
        improvement = best_obj - obj_g
        if improvement > 0.01:
            t = 0.7
        elif improvement > 0:
            t = 0.5
        else:
            t = 0.3
        
        a_new = (1-t)*a + t*g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-15:
            a = g
        else:
            a = a_new / s
        
        # Track best
        obj_a = compute_obj(a)
        if obj_a < best_obj:
            best_obj = obj_a
            best_a = a.copy()
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()
    
    return best_a, best_obj

def fourier_init(n, rng):
    """
    Initialize using the magnitude of sqrt(1+x+...+x^{2n-2}).
    The coefficients of this polynomial give a "flat convolution" target.
    """
    # Compute coefficients of sqrt(1+x+x^2+...+x^{2n-2})
    # Using polynomial arithmetic
    target = np.ones(2*n-1)
    # We want p such that p*p (convolution) = target
    # p has n coefficients
    # This is essentially computing the "square root" of a polynomial
    # We can do this iteratively: p_{k+1} = p_k + (target - p_k*p_k) / (2*p_k)
    
    p = np.ones(n) / np.sqrt(n)  # initial guess
    for _ in range(50):
        p2 = np.convolve(p, p)  # length 2n-1
        # Newton step: p_new = p + (target - p2) / (2*p) (in polynomial sense)
        # (target - p2) / (2*p) is a polynomial division
        # The result should have at most n terms (the rest should be ~0)
        diff = target - p2  # length 2n-1
        # Divide diff by 2*p (polynomial long division)
        # We only need the first n coefficients of the quotient
        quotient = np.zeros(n)
        for i in range(n):
            if abs(2*p[0]) > 1e-15:
                quotient[i] = diff[i] / (2*p[0])
                diff[i:i+n] -= quotient[i] * (2*p)
        
        p_new = p + 0.5 * quotient  # damping factor
        if np.max(np.abs(p_new - p)) < 1e-12:
            p = p_new
            break
        p = p_new
    
    # Clip to non-negative
    p = np.maximum(p, 0)
    s = np.sum(p)
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    
    # Add small random perturbation
    p = p + rng.random(n) * 0.01
    p = np.maximum(p, 0)
    p = p / np.sum(p)
    
    return p

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        a = np.ones(n)
    elif choice == 1:

        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2*np.pi)

    elif choice == 2:
        # Periodic
        period = int(rng.integers(max(2, n//20), max(3, n//3)))
        pattern = rng.random(period)
        a = np.tile(pattern, n//period + 1)[:n]
    elif choice == 3:
        # Gaussian bump
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.1, 0.4) * n
        a = np.exp(-((np.arange(n) - center)**2) / (2*width**2))
    elif choice == 4:
        # Two bumps
        c1 = rng.uniform(0.1, 0.4) * n
        c2 = rng.uniform(0.6, 0.9) * n
        w = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((np.arange(n)-c1)**2)/(2*w**2)) + np.exp(-((np.arange(n)-c2)**2)/(2*w**2))
    elif choice == 5:
        # Linear ramp
        a = np.linspace(rng.uniform(0.1, 1), rng.uniform(0.1, 1), n)
    elif choice == 6:
        # Sine with random frequency
        freq = rng.uniform(1, 10)
        a = np.maximum(0, np.sin(2*np.pi*freq*np.arange(n)/n)) + 0.1
    else:
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def coord_refine(a, n_rounds=2, deadline=None):
    """Coordinate descent refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    
    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        for j in range(n):
            old = a[j]
            best_x = old
            best_local = best_obj
            # Try several values
            candidates = [0.0, old*0.5, old*0.7, old*1.3, old*1.5, old*2.0, 
                         np.max(a)*0.5, np.max(a)]
            for x in candidates:
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x
            a[j] = best_x
        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj = obj_now
    
    return a, best_obj

"""
Optimization of non-negative step functions to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp gradient descent with adaptive beta, Fourier-based initialization,
LP alternating optimization, and coordinate refinement across multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * float(np.max(conv))


def fourier_init(n, rng):
    """
    Initialize using polynomial square root of (1+x+...+x^{2n-2}).
    Newton iteration: p_{k+1} = p_k + (target - p_k*p_k)/(2*p_k)
    """
    target = np.ones(2 * n - 1)
    p = np.ones(n) / np.sqrt(n)

    for _ in range(100):
        p2 = np.convolve(p, p)
        diff = target - p2
        quotient = np.zeros(n)
        for i in range(n):
            denom = 2.0 * p[0]
            if abs(denom) > 1e-15:
                quotient[i] = diff[i] / denom
                diff[i:i + n] -= quotient[i] * (2.0 * p)
        p_new = p + 0.5 * quotient
        if np.max(np.abs(p_new - p)) < 1e-14:
            p = p_new
            break
        p = p_new

    p = np.maximum(p, 0.0)
    s = np.sum(p)
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    # Add perturbation for diversity
    perturbation = rng.random(n) * 0.05
    p = p + perturbation
    p = np.maximum(p, 0.0)
    p = p / np.sum(p)
    return p

def lse_gradient(a, beta):
    """Compute gradient of log-sum-exp approximation of max(conv(a,a))."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = np.max(c)
    w = np.exp(beta * (c - c_max))
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w = w / w_sum

    # grad[j] = 2 * sum_{i=0}^{n-1} w[j+i] * a[i]
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(w[j:j + n], a)
    return grad, w, c


def lse_descent(a_init, n_steps=5000, beta_start=5.0, beta_end=60.0,
                momentum=0.85, step_scale=0.3, deadline=None):
    """
    Log-sum-exp gradient descent with adaptive beta and Nesterov momentum.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj_normalized(best_a)

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        frac = step / max(n_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        # Nesterov lookahead
        a_la = a + momentum * v

        # Compute gradient
        grad, w, c = lse_gradient(a_la, beta)

        # Update
        v = momentum * v - step_scale * grad
        a_new = a + v

        # Project onto simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj = compute_obj_normalized(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def lse_descent_adaptive(a_init, n_steps=5000, beta_start=5.0, beta_end=80.0,
                         momentum=0.85, deadline=None):
    """LSE descent with adaptive step size (reduces if objective worsens)."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj_normalized(best_a)
    step_scale = 0.3
    obj_prev = best_obj

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        frac = step / max(n_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        a_la = a + momentum * v
        grad, w, c = lse_gradient(a_la, beta)

        v = momentum * v - step_scale * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            a_new = a.copy()
            v = np.zeros(n)
            step_scale *= 0.5
        else:
            a_new = a_new / s_new

        obj = compute_obj_normalized(a_new)

        # Adaptive step size
        if obj > obj_prev * 1.01:
            step_scale *= 0.7
            a_new = a  # revert
            v = np.zeros(n)
        elif obj < best_obj:
            step_scale = min(step_scale * 1.05, 1.0)
            best_obj = obj
            best_a = a_new.copy()

        obj_prev = obj
        a = a_new

    return best_a, best_obj


def build_constraint_matrix(a):
    """Build A_ub for LP: A_ub[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def lp_alternate(a_init, max_iter=100, deadline=None, blend_start=0.5, blend_end=0.8):
    """
    LP alternating: find best g given a, blend back.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(best_a)
    A_ub = build_constraint_matrix(a)  # Will be rebuilt each iter if a changes

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv_aa = np.convolve(a, a)
        M = float(np.max(conv_aa)) * 1.001  # tiny slack for feasibility

        A_ub = build_constraint_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 2.0})

        if res.status != 0:
            break

        g = np.maximum(res.x, 0.0)
        g_sum = np.sum(g)
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Adaptive blending
        frac = it / max(max_iter - 1, 1)
        t = blend_start + (blend_end - blend_start) * frac

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            a = g
        else:
            a = a_new / s_new

        obj_a = compute_obj_normalized(a)
        obj_g = compute_obj_normalized(g)

        if obj_a < best_obj:
            best_obj = obj_a
            best_a = a.copy()
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()

    return best_a, best_obj


def subgradient_polish(a_init, n_steps=3000, deadline=None, tol_frac=0.001):
    """
    Subgradient method on the true max with active set tracking.
    Uses Polyak-type step size.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(best_a)
    obj_lb = 2.0 * n / (2 * n - 1)  # theoretical lower bound
    gap = best_obj - obj_lb

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        c_max = np.max(c)

        # Active set: lags within relative tolerance of max
        tol = c_max * tol_frac * (1.0 + step / n_steps)
        active = c >= c_max - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = np.zeros(len(c), dtype=bool)
            active[np.argmax(c)] = True
            n_active = 1

        # Subgradient: average of gradients at active lags
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            vals = a[k - j_max:k - j_min + 1][::-1]
            grad[j_min:j_max + 1] += 2.0 * vals
        grad /= n_active

        # Polyak step size
        gap = max(best_obj - obj_lb, 1e-10)
        step_size = gap / (np.dot(grad, grad) + 1e-15)
        step_size = min(step_size, 1.0)

        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj = compute_obj_normalized(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            gap = max(best_obj - obj_lb, 1e-10)

        a = a_new

    return best_a, best_obj


def coord_refine(a_init, n_rounds=2, deadline=None):
    """Coordinate descent: try adjusting each element."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_obj = compute_obj_normalized(a)

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        for j in range(n):
            old = a[j]
            best_x = old
            best_local = best_obj
            # Try several multipliers and absolute values
            max_a = np.max(a)
            candidates = [0.0, old * 0.3, old * 0.6, old * 0.9,
                          old * 1.1, old * 1.4, old * 2.0,
                          max_a * 0.3, max_a * 0.8]
            # Also try bisection-like: geometric mean with neighbors
            if j > 0:
                candidates.append(np.sqrt(old * a[j - 1]))
            if j < n - 1:
                candidates.append(np.sqrt(old * a[j + 1]))

            for x in candidates:
                if x < 0:
                    continue
                a[j] = x
                s_curr = np.sum(a)
                if s_curr < 1e-15:
                    a[j] = old
                    continue
                a_norm = a / s_curr
                obj = compute_obj_normalized(a_norm)
                if obj < best_local:
                    best_local = obj
                    best_x = x
            a[j] = best_x
            # Renormalize
            s_curr = np.sum(a)
            if s_curr > 1e-15:
                a = a / s_curr

        obj_now = compute_obj_normalized(a)
        if obj_now < best_obj:
            best_obj = obj_now

    return a, best_obj

        for restart in range(n_restarts):
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # Choose initialization
            if restart == 0 and best_seq is not None and len(best_seq) == n:
                a_init = best_seq / np.sum(best_seq)
            elif restart == 1:
                a_init = fourier_init(n, rng)
            elif restart == 2:
                a_init = structured_init(n, rng)
            else:
                a_init = fourier_init(n, rng)

            # LSE descent (medium budget)
            t_budget = min(remaining * 0.35, 25.0)
            t_dl = time.monotonic() + t_budget
            n_steps = max(1000, int(t_budget / 0.003))

            a1, obj1 = lse_descent(a_init, n_steps=n_steps,
                                    beta_start=5.0, beta_end=50.0,
                                    momentum=0.85, step_scale=0.3,
                                    deadline=t_dl)
            # Convert to unnormalized objective
            obj1_real = compute_obj(a1 * np.sqrt(2 * n))

            if obj1_real < best_score:
                best_score = obj1_real
                best_seq = a1.copy()
                print(f"  LSE n={n} r={restart}: obj={obj1_real:.6f}")

            # LP alternating on result
            if time.monotonic() < deadline - 15:
                t_budget2 = min((deadline - time.monotonic()) * 0.25, 15.0)
                lp_dl = time.monotonic() + t_budget2
                n_iter = max(20, int(t_budget2 / 0.05))

                a2, obj2_norm = lp_alternate(a1, max_iter=n_iter, deadline=lp_dl)
                obj2_real = compute_obj(a2 * np.sqrt(2 * n))

                if obj2_real < best_score:
                    best_score = obj2_real
                    best_seq = a2.copy()
                    print(f"  LP n={n} r={restart}: obj={obj2_real:.6f}")

                # Second LSE pass with higher beta
                if time.monotonic() < deadline - 10:
                    t_budget3 = min((deadline - time.monotonic()) * 0.15, 10.0)
                    lse_dl = time.monotonic() + t_budget3
                    n_steps3 = max(500, int(t_budget3 / 0.003))

                    a3, obj3_norm = lse_descent(a2, n_steps=n_steps3,
                                                 beta_start=20.0, beta_end=80.0,
                                                 momentum=0.9, step_scale=0.2,
                                                 deadline=lse_dl)
                    obj3_real = compute_obj(a3 * np.sqrt(2 * n))
                    if obj3_real < best_score:
                        best_score = obj3_real
                        best_seq = a3.copy()
                        print(f"  LSE2 n={n} r={restart}: obj={obj3_real:.6f}")

    # Phase 2: Intensive refinement of the best solution
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        a_best = best_seq / np.sum(best_seq)
        print(f"\nPhase 2: n={n_best}, obj={best_score:.6f}, {remaining:.0f}s")

        # Subgradient polish
        t_b = min(remaining * 0.3, 20.0)
        sg_dl = time.monotonic() + t_b
        ns = max(1000, int(t_b / 0.002))
        a_sg, obj_sg_norm = subgradient_polish(a_best, n_steps=ns, deadline=sg_dl)
        obj_sg_real = compute_obj(a_sg * np.sqrt(2 * n_best))
        if obj_sg_real < best_score:
            best_score = obj_sg_real
            best_seq = a_sg.copy()
            print(f"  SG: obj={obj_sg_real:.6f}")
            a_best = a_sg / np.sum(a_sg)

        # LSE with high beta
        if time.monotonic() < deadline - 10:
            t_b = min((deadline - time.monotonic()) * 0.25, 15.0)
            lse_dl = time.monotonic() + t_b
            ns = max(1000, int(t_b / 0.003))
            a_l, obj_l_norm = lse_descent_adaptive(a_best, n_steps=ns,
                                                    beta_start=30.0, beta_end=100.0,
                                                    momentum=0.9, deadline=lse_dl)
            obj_l_real = compute_obj(a_l * np.sqrt(2 * n_best))
            if obj_l_real < best_score:
                best_score = obj_l_real
                best_seq = a_l.copy()
                print(f"  LSE-adaptive: obj={obj_l_real:.6f}")
                a_best = a_l / np.sum(a_l)

        # LP alternating
        if time.monotonic() < deadline - 10:
            t_b = min((deadline - time.monotonic()) * 0.2, 15.0)
            lp_dl = time.monotonic() + t_b
            n_iter = max(20, int(t_b / 0.05))
            a_lp, obj_lp_norm = lp_alternate(a_best, max_iter=n_iter,
                                              blend_start=0.6, blend_end=0.9,
                                              deadline=lp_dl)
            obj_lp_real = compute_obj(a_lp * np.sqrt(2 * n_best))
            if obj_lp_real < best_score:
                best_score = obj_lp_real
                best_seq = a_lp.copy()
                print(f"  LP: obj={obj_lp_real:.6f}")
                a_best = a_lp / np.sum(a_lp)

        # Coordinate refinement
        if time.monotonic() < deadline - 5 and n_best <= 500:
            a_c, obj_c_norm = coord_refine(a_best, n_rounds=2,
                                            deadline=min(deadline - 2, time.monotonic() + 30))
            obj_c_real = compute_obj(a_c * np.sqrt(2 * n_best))
            if obj_c_real < best_score:
                best_score = obj_c_real
                best_seq = a_c.copy()
                print(f"  Coord: obj={obj_c_real:.6f}")

        # Final LSE squeeze
        if time.monotonic() < deadline - 3:
            a_best = best_seq / np.sum(best_seq)
            t_b = min(deadline - time.monotonic() - 2, 10.0)
            final_dl = time.monotonic() + t_b
            ns = max(500, int(t_b / 0.003))
            a_f, obj_f_norm = lse_descent(a_best, n_steps=ns,
                                           beta_start=50.0, beta_end=120.0,
                                           momentum=0.9, step_scale=0.15,
                                           deadline=final_dl)
            obj_f_real = compute_obj(a_f * np.sqrt(2 * n_best))
            if obj_f_real < best_score:
                best_score = obj_f_real
                best_seq = a_f.copy()
                print(f"  Final LSE: obj={obj_f_real:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
