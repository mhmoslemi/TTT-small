    return A


def solve_lp(a, M, k_list):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for k in k_list, g >= 0."""
    n = len(a)
    A = build_A(a, k_list)
    c = -np.ones(n)
    b_ub = np.full(len(k_list), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_iteration(a, max_k=200):
    """One LP iteration. Returns (g, sum_g) or (None, None)."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))

    # Select top-k active constraints (those closest to M)
    n_constraints = 2 * n - 1
    if max_k < n_constraints:
        # Get indices of largest convolution coefficients
        top_idx = np.argsort(conv)[::-1][:max_k]
        k_list = [int(x) for x in top_idx]
    else:
        k_list = list(range(n_constraints))

    g, sum_g = solve_lp(a, M, k_list)
    return g, sum_g, M, conv


def lp_optimize(a, max_time=60.0, verbose=False):
    """Run LP-based optimization. Returns best (a, obj)."""
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()

    step_size = 0.3
    stall = 0
    max_stall = 30
    it = 0

    while time.monotonic() - start < max_time:
        g, sum_g, M, conv = lp_iteration(a, max_k=min(2 * len(a), 300))
        if g is None:
            break

        it += 1
        if sum_g <= 1.0 + 1e-6:
            stall += 1
            if stall > max_stall:
                # Try with all constraints
                g, sum_g, M, conv = lp_iteration(a, max_k=2 * len(a) - 1)
                if g is None or sum_g <= 1.0 + 1e-6:
                    break
            else:
                continue
        stall = 0

        g = g / sum_g
        # Adaptive step size: use line search
        best_t = step_size
        best_new_obj = compute_obj((1 - step_size) * a + step_size * g)
        for t in [step_size * 0.5, step_size, step_size * 1.5, step_size * 0.7]:
            t = min(1.0, max(0.01, t))
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_new_obj:
                best_new_obj = obj_try
                best_t = t

        a_new = (1 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose and it % 50 == 0:
                print(f"      LP it={it}: obj={best_obj:.6f}, sum_g={sum_g:.4f}")
        a = a_new

        # Slowly decrease step size
        step_size = max(0.05, step_size * 0.998)

    return best_a, best_obj


def subgradient_optimize(a, max_time=30.0, tau_start=20.0, tau_end=100.0):
    """Smoothed subgradient optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    step = 0.05
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = it / max(1, int(max_time * 5))
        tau = tau_start + (tau_end - tau_start) * min(1, frac)

        conv = compute_conv(a)
        max_conv = np.max(conv)

        # Softmax weights (smoothed max indicator)
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum

        # Gradient of max(conv) w.r.t. a
        # d/d(a[i]) conv[k] = a[k-i] if valid
        # grad[i] = sum_k weights[k] * a[k-i] (for valid k-i)
        # This is np.correlate(weights, a, 'full')[offset]
        c = np.correlate(weights, a, mode='full')
        grad = 2.0 * c[:n]

        # Center for sum constraint
        grad_centered = grad - np.mean(grad)

        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # Still move, but don't update best

        step *= 0.999

    return best_a, best_obj


def coordinate_descent(a, max_time=30.0):
    """Simple coordinate descent: try decreasing each coordinate."""
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    n = len(a)

    sweep = 0
    while time.monotonic() - start < max_time:
        improved = False
        for i in range(n):
            if time.monotonic() - start > max_time - 5:
                break
            if a[i] < 1e-8:
                continue
            # Try reducing a[i] by a small amount
            for frac in [0.5, 0.3, 0.1]:
                a_try = a.copy()
                a_try[i] *= frac
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        sweep += 1
        if not improved and sweep > 3:
            break

    return best_a, best_obj


def random_perturb(a, rng, scale=0.1):
    """Apply multiplicative random perturbation."""
    n = len(a)
    noise = rng.normal(0, scale, n)
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s


def structured_init(n, rng, style=0):
    """Generate structured initial sequence."""
    if style == 0:
        # Cosine window
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n + rng.uniform(0, 2*np.pi)))
    elif style == 1:
        # Triangular
        x = np.arange(n)
        center = n / 2
        a = 1 - np.abs(x - center) / center
    elif style == 2:
        # Gaussian
        x = np.arange(n)
        sigma = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((x - n/2) / sigma)**2)
    elif style == 3:
        # Bimodal
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n*0.25) / (n*0.15))**2) + np.exp(-0.5 * ((x - n*0.75) / (n*0.15))**2)
    elif style == 4:
        # Random
        a = rng.random(n)
    else:
        # Linear ramp with noise
        x = np.arange(n)
        a = (x + 1) / n + rng.random(n) * 0.3
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    return a / s


def run_for_length(n, seed, max_time=60.0, start_a=None, verbose=False):
    """Run combined optimization for a given sequence length."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    best_a = None
    best_obj = np.inf

    # Number of restarts based on time budget
    n_restarts = max(2, int(max_time / 25))

    for r in range(n_restarts):
        if time.monotonic() - start > max_time - 10:
            break

        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        # Get initial sequence
        if r == 0 and start_a is not None:
            a = start_a.copy()
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            else:
                a = structured_init(n, rng, style=rng.integers(0, 6))
        else:
            a = structured_init(n, rng, style=rng.integers(0, 6))

        # Phase 1: LP optimization
        t_lp = remaining * 0.6
        a_lp, obj_lp = lp_optimize(a, max_time=t_lp, verbose=verbose)

        # Phase 2: Subgradient polish
        remaining = max_time - (time.monotonic() - start)
        if remaining > 5:
            a_sg, obj_sg = subgradient_optimize(a_lp, max_time=remaining * 0.6)
            if obj_sg < obj_lp:
                a_lp, obj_lp = a_sg, obj_sg

        # Phase 3: Coordinate descent
        remaining = max_time - (time.monotonic() - start)
        if remaining > 5:
            a_cd, obj_cd = coordinate_descent(a_lp, max_time=remaining * 0.5)
            if obj_cd < obj_lp:
                a_lp, obj_lp = a_cd, obj_cd

        # Phase 4: Perturb and re-optimize if stuck
        remaining = max_time - (time.monotonic() - start)
        if remaining > 10 and r < n_restarts - 1:
            a_pert = random_perturb(a_lp, rng, scale=0.15)
            a_p2, obj_p2 = lp_optimize(a_pert, max_time=remaining * 0.7)
            if obj_p2 < obj_lp:
                a_lp, obj_lp = a_p2, obj_p2

        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"    n={n} restart {r}: obj={best_obj:.6f}")

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        parent = np.maximum(parent, 0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 1e-10:
                obj = compute_obj(parent)
                print(f"Parent (n={parent.size}): obj={obj:.6f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Combined LP + subgradient + coordinate descent optimization.
    Explores multiple sequence lengths with aggressive LP iterations.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    start_a_for_parent = None
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        start_a_for_parent = parent.copy()

    # Determine sequence lengths to explore
    n_parent = len(parent) if parent is not None else 250
    n_values = sorted(set([
        50, 75, 100, 125, 150, 175, 200, 250, 300, 350, 400, 500, 600, 750, 1000
    ]))
    # Add parent length and neighbors
    n_values = sorted(set(n_values + [n_parent,
                                       max(20, n_parent - 50),
                                       max(30, n_parent - 20),
                                       min(1200, n_parent + 50),
                                       min(1200, n_parent + 100)]))

    print(f"Exploring n values: {n_values}")
    print(f"Parent obj: {best_score:.6f}" if best_score < np.inf else "No parent")

    # Phase 1: Quick scan of lengths to find good candidates
    print("\n=== Phase 1: Quick scan ===")
    for n in n_values:
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(25.0, remaining * 0.05)
        alloc = max(10.0, alloc)

        start_a = start_a_for_parent if n == n_parent else None
        a_opt, obj = run_for_length(n, seed=seed + n * 31, max_time=alloc,
                                    start_a=start_a)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  UPDATE: n={n}, obj={best_score:.6f}")

    # Phase 2: Intensive optimization on best lengths
    print("\n=== Phase 2: Intensive optimization ===")
    # Focus on the best length and its neighbors
    n_best = len(best_seq) if best_seq is not None else n_parent
    focus_lengths = sorted(set([
        n_best, max(20, n_best - 50), max(30, n_best - 20),
        min(1200, n_best + 50), min(1200, n_best + 100)
    ]))

    for n in focus_lengths:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.3, 120.0)
        alloc = max(20.0, alloc)

        start_a = best_seq.copy() if n == n_best else None
        a_opt, obj = run_for_length(n, seed=seed + n * 7919 + 1000,
                                    max_time=alloc, start_a=start_a, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n
            print(f"  UPDATE: n={n}, obj={best_score:.6f}")

    # Phase 3: Final intensive polishing
    print("\n=== Phase 3: Final polishing ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_cur = best_seq.copy()
        n_cur = len(a_cur)
        polish_round = 0

        while time.monotonic() < deadline - 10:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # LP polish
            t_lp = remaining * 0.4
            a_lp, obj_lp = lp_optimize(a_cur, max_time=t_lp, verbose=True)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_cur = a_lp.copy()
                print(f"  Polish {polish_round} LP: obj={best_score:.6f}")

            # Subgradient polish
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_sg, obj_sg = subgradient_optimize(a_cur, max_time=remaining * 0.3)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    a_cur = a_sg.copy()
                    print(f"  Polish {polish_round} SG: obj={best_score:.6f}")

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 10 and n_cur <= 500:
                a_cd, obj_cd = coordinate_descent(a_cur, max_time=remaining * 0.3)
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy()
                    print(f"  Polish {polish_round} CD: obj={best_score:.6f}")

            # Perturbation
            remaining = deadline - time.monotonic()
            if remaining > 20:
                scales = [0.05, 0.1, 0.2]
                for scale in scales:
                    a_pert = random_perturb(a_cur, rng, scale=scale)
                    a_p, obj_p = lp_optimize(a_pert, max_time=min(15.0, remaining * 0.3))
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        print(f"  Polish {polish_round} perturb({scale}): obj={best_score:.6f}")
                    remaining = deadline - time.monotonic()
                    if remaining < 15:
                        break

    # Fallback
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Final validation
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-6:
            print(f"  Validation: computed={best_score:.6f}, validated={validation:.6f}")
        best_score = validation

    return result
