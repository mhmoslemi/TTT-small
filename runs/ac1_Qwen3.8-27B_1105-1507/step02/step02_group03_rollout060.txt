```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute score for sequence f (not necessarily normalized)."""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s

def grad_qk(f, k):
    """Compute gradient of q_k(f) = sum_i f[i]*f[k-i] with respect to f.
    Result: grad[m] = 2 * f[k-m] (for valid k-m in [0, n-1], else 0)."""
    n = len(f)
    grad = np.zeros(n)
    # For each m, grad[m] = 2 * f[k-m] if 0 <= k-m < n
    valid = (k >= 0) & (k < 2*n-1)
    if not valid:
        return grad
    # k-m in [0, n-1] means m in [k-(n-1), k]
    m_start = max(0, k - n + 1)
    m_end = min(n-1, k)
    for m in range(m_start, m_end + 1):
        grad[m] = 2.0 * f[k - m]
    return grad

def smooth_gradient(f, beta=50):
    """Compute the gradient of the log-sum-exp smooth approximation of max_k q_k(f)."""
    n = len(f)
    # Compute all q_k values
    conv = np.convolve(f, f)
    # Softmax weights
    shifted = beta * (conv - conv.max())  # for numerical stability
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient = sum_k weights[k] * grad_qk(f, k)
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-10:
            continue
        # grad_qk contribution: 2 * weights[k] * f[k-m] for valid m
        m_start = max(0, k - n + 1)
        m_end = min(n-1, k)
        grad[m_start:m_end+1] += 2.0 * weights[k] * f[k - np.arange(m_start, m_end+1)][::-1]
    
    return grad
```

```python
"""

LP-guided direction finding, and coordinate descent polishing.
Targets score < 1.5027 for sequences of length ~512.

import numpy as np
from scipy.optimize import linprog
import time


def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def score_of(f):
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def smooth_gradient(f, beta):
    """Compute gradient of log-sum-exp(beta * convolve(f,f)) approximation of max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    shifted = beta * (conv - conv.max())
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(n):
        jmax = min(n, 2 * n - 1 - m)
        if jmax > 0:
            grad[m] = 2.0 * np.dot(w[m:m + jmax], f[:jmax])
    return grad, conv


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

                cur_score = new_score
                if new_score < best_score:

def build_conv_matrix_from_f(f):
    """Build matrix A such that A @ g gives convolve(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_find_direction(f, M, time_limit=5.0):
    """Find direction g maximizing sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix_from_f(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-15:
        return None
    return g / g.sum()


def lp_refine(f, budget_s, t0=0.3, rng=None):
    """LP-guided iterative refinement."""

        rng = np.random.default_rng(0)

    n = len(f)

        if remaining < 10:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        lp_time = min(remaining * 0.1, max(2.0, remaining / 10.0))
        g = lp_find_direction(f, M, time_limit=lp_time)
        lp_calls += 1

        if g is None:
            break

        improved = False
        for step in [min(1.0, t * 1.3), t, t * 0.5, t * 0.25, t * 0.1]:
            f_new = (1.0 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:

            f_new = f_new / s
            new_score = score_of(f_new)
            if new_score < best_score - 1e-12:

                best_score = new_score

                f_pert = normalize(f_pert)
                if score_of(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = score_of(f)
                    t = 0.1
                else:
                    break

def coordinate_descent(f, n_sweeps=10, rng=None):
    """Coordinate descent: optimize one element at a time."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_score = score_of(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                # Try setting to small positive
                candidates = [1e-4, 1e-3, original + 1e-3]
            else:
                candidates = [0.0, original * 0.1, original * 0.3, original * 0.5,
                              original * 0.7, original * 0.85, original,
                              original * 1.15, original * 1.3, original * 1.5,
                              original * 2.0]
            # Add a few random candidates
            candidates.extend([original * exp for exp in rng.uniform(0.2, 3.0, 5)])

            best_val = original
            best_local = best_score

            for val in candidates:
                val = max(0.0, val)
                f[i] = val
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                conv = np.convolve(f, f)
                sc = 2.0 * n * conv.max() / (s * s)
                if sc < best_local:
                    best_local = sc
                    best_val = val

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score

        current_energy = np.sum(f ** 2)
        n_bins = len(mag)

        mag_new = np.full_like(mag, target_mag)

def diverse_init(n, rng):
    """Generate diverse initial sequence."""
    r = rng.random()
    if r < 0.2:
        # Asymmetric ramp
        x = np.arange(n) / n
        f = x ** rng.uniform(0.5, 2.0)
    elif r < 0.4:
        # Two bumps (asymmetric)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w1 = n * rng.uniform(0.05, 0.2)
        w2 = n * rng.uniform(0.05, 0.2)
        f = np.exp(-0.5 * (np.arange(n) - c1) ** 2 / w1 ** 2) + \
            np.exp(-0.5 * (np.arange(n) - c2) ** 2 / w2 ** 2)
    elif r < 0.6:
        # Cosine window (asymmetric)
        f = 0.5 * (1 + np.cos(np.pi * np.arange(n) / n * rng.uniform(0.5, 1.5)))
    elif r < 0.8:
        # Smooth random
        f = np.ones(n)
        for _ in range(8):
            center = int(rng.integers(0, n))
            width = int(rng.integers(3, max(4, n // 8)))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / max(1, width ** 2))
            f *= (1 + 0.8 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.01
    else:
        # Exponential decay (asymmetric)
        rate = rng.uniform(0.01, 0.1)
        f = np.exp(-rate * np.arange(n)) + np.exp(-rate * np.arange(n - 1, -1, -1) * 0.5)

    f = np.clip(f, 0.001, None)
    f = normalize(f)
    # Break symmetry
    f = f * (1 + 0.05 * rng.normal(0, 1, n))
    f = np.clip(f, 0.0001, None)
    f = normalize(f)
    return f


def pressure_descent(f, n_iter=500, eta=0.3, rng=None):
    """Pressure-based descent: reduce elements contributing most to peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = score_of(f)
    cur_score = best_score
    stall = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        # Subgradient at argmax positions
        # Use top few peaks for robustness
        sorted_idx = np.argsort(conv)[::-1]
        top_k = sorted_idx[:5]
        threshold = 0.95 * M

        grad = np.zeros(n)
        for k in top_k:
            if conv[k] < threshold:
                continue
            weight = (conv[k] - M) / (M - conv[top_k[-1]] + 1e-15) if len(top_k) > 1 else 1.0
            weight = max(0.0, weight)
            for m in range(n):
                idx = k - m
                if 0 <= idx < n:
                    grad[m] += 2.0 * weight * f[idx]

        if grad.max() > 0:
            grad /= grad.max()

        eta_eff = eta / (1.0 + 0.01 * it)
        f_new = f * np.exp(-eta_eff * grad)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        new_score = score_of(f_new)
        if new_score < cur_score - 1e-14:
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 150:
                f_pert = f * (1 + 0.1 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if score_of(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = score_of(f)
                    cur_score = best_score
                    stall = 0
                else:
                    break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines smooth gradient descent, LP refinement, pressure descent,
    spectral flattening, and coordinate descent across multiple strategies.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    best_overall = None
    best_score_overall = np.inf

    # Try parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            parent = normalize(parent)
            parent_score = score_of(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_overall = parent.copy()
            best_score_overall = parent_score

    # Determine lengths to try
    n_default = 512
    if parent is not None:
        lengths = [len(parent)]
        if len(parent) != 512:
            lengths.append(512)
        lengths.append(1024)
    else:
        lengths = [256, 512, 1024]

    print(f"Lengths to try: {lengths}, budget: {budget_s}s, target: 1.5027")

    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 90:
            break

        print(f"\n{'='*50}")
        print(f"n={n}, remaining={remaining:.0f}s, current best={best_score_overall:.6f}")

        # Determine time allocation for this length
        if len(lengths) == 3:
            t_alloc = min(remaining * 0.4, 400)
        elif len(lengths) == 2:
            t_alloc = min(remaining * 0.6, 600)
        else:
            t_alloc = remaining - 30

        # === Strategy 1: Smooth gradient descent ===
        t_grad = t_alloc * 0.35
        if t_grad > 20:
            f_init = None
            if parent is not None and len(parent) == n:
                f_init = parent * (1 + 0.1 * rng.normal(0, 1, n))
                f_init = np.clip(f_init, 0, None)
                f_init = normalize(f_init)
            else:
                f_init = diverse_init(n, rng)

            f1, s1, t_used = gradient_descent_smooth(f_init, t_grad,
                                                      beta_start=3.0, beta_end=200.0,
                                                      n_iter_max=15000, eta0=0.3,
                                                      trial_lr=True, rng=rng)
            print(f"  Gradient descent: score={s1:.6f} (time={t_used:.1f}s)")
            if s1 < best_score_overall:
                best_score_overall = s1
                best_overall = f1.copy()

        # === Strategy 2: Pressure descent + LP ===
        remaining = deadline - time.monotonic()
        if remaining > 60:
            f_init2 = diverse_init(n, rng)
            t_press = min(remaining * 0.2, 60)
            f2, s2 = pressure_descent(f_init2, n_iter=int(t_press * 30), eta=0.3, rng=rng)
            print(f"  Pressure descent: score={s2:.6f}")
            if s2 < best_score_overall:
                best_score_overall = s2
                best_overall = f2.copy()

            # LP refinement on best of (gradient result, pressure result)
            remaining = deadline - time.monotonic()
            if remaining > 30:
                t_lp = min(remaining * 0.3, 120)
                f_lp_start = best_overall.copy() if best_overall is not None else f2
                if len(f_lp_start) != n:
                    f_lp_start = f2
                f3, s3, t_used, n_lp = lp_refine(f_lp_start, t_lp, t0=0.4, rng=rng)
                print(f"  LP refine: score={s3:.6f} (LP calls={n_lp}, time={t_used:.1f}s)")
                if s3 < best_score_overall:
                    best_score_overall = s3
                    best_overall = f3.copy()

        # === Strategy 3: Spectral + gradient ===
        remaining = deadline - time.monotonic()
        if remaining > 40:
            f_init3 = diverse_init(n, rng)
            f4, s4 = spectral_flat(f_init3, n_iter=150, rng=rng)
            print(f"  Spectral flat: score={s4:.6f}")
            if s4 < best_score_overall:
                best_score_overall = s4
                best_overall = f4.copy()

            # Follow up with gradient
            remaining = deadline - time.monotonic()
            if remaining > 20:
                f5, s5, _ = gradient_descent_smooth(f4, min(remaining * 0.5, 60),
                                                     beta_start=50.0, beta_end=300.0,
                                                     n_iter_max=5000, eta0=0.1,
                                                     trial_lr=True, rng=rng)
                print(f"  Spectral+Grad: score={s5:.6f}")
                if s5 < best_score_overall:
                    best_score_overall = s5
                    best_overall = f5.copy()

        # === Strategy 4: Asymmetric init + full pipeline ===
        remaining = deadline - time.monotonic()
        if remaining > 60:
            # Deliberately asymmetric initialization
            x = np.arange(n) / n
            f_init4 = x ** 1.5 + 0.3 * np.exp(-5 * (x - 0.7) ** 2)
            f_init4 = np.clip(f_init4, 0.001, None)
            f_init4 = normalize(f_init4)

            f6, s6, _ = gradient_descent_smooth(f_init4, min(remaining * 0.3, 90),
                                                 beta_start=5.0, beta_end=200.0,
                                                 n_iter_max=8000, eta0=0.2,
                                                 trial_lr=True, rng=rng)
            print(f"  Asymmetric+Grad: score={s6:.6f}")
            if s6 < best_score_overall:
                best_score_overall = s6
                best_overall = f6.copy()

            remaining = deadline - time.monotonic()
            if remaining > 30:
                t_lp2 = min(remaining * 0.4, 90)
                f7, s7, _, _ = lp_refine(f6, t_lp2, t0=0.3, rng=rng)
                print(f"  Asym+LP: score={s7:.6f}")
                if s7 < best_score_overall:
                    best_score_overall = s7
                    best_overall = f7.copy()

    # === Final polishing phase ===
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"\n{'='*50}")
            print(f"Polishing phase, remaining={remaining:.0f}s, score={best_score_overall:.6f}")
            n_best = len(best_overall)

            # Coordinate descent
            t_cd = min(remaining * 0.4, 60)
            n_sweeps = max(3, int(t_cd * 0.5))
            f_cd, s_cd = coordinate_descent(best_overall, n_sweeps=n_sweeps, rng=rng)
            print(f"  After CD: score={s_cd:.6f}")
            if s_cd < best_score_overall:
                best_score_overall = s_cd
                best_overall = f_cd.copy()

            # Fine gradient descent
            remaining = deadline - time.monotonic()
            if remaining > 10:
                t_fine = min(remaining * 0.5, 60)
                f_fine, s_fine, _ = gradient_descent_smooth(best_overall, t_fine,
                                                             beta_start=100.0, beta_end=500.0,
                                                             n_iter_max=10000, eta0=0.05,
                                                             trial_lr=True, rng=rng)
                print(f"  Fine grad: score={s_fine:.6f}")
                if s_fine < best_score_overall:
                    best_score_overall = s_fine
                    best_overall = f_fine.copy()

            # Another CD pass
            remaining = deadline - time.monotonic()
            if remaining > 5:
                f_cd2, s_cd2 = coordinate_descent(best_overall, n_sweeps=5, rng=rng)
                print(f"  Final CD: score={s_cd2:.6f}")
                if s_cd2 < best_score_overall:
                    best_score_overall = s_cd2
                    best_overall = f_cd2.copy()

    # Final verification
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}, target=1.5027")
        if final_score < best_score_overall:
            best_score_overall = final_score
        return final_list

    # Fallback
    print("Fallback: returning uniform sequence")
    return [1.0] * 256
```
