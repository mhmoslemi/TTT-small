def fast_conv(a):

    n = len(a)

def compute_J(a):

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse matrix A where (A @ g)[k] = full_conv(a,g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    if not rows:
        return csr_matrix((2 * n - 1, n))
    return csr_matrix((np.array(vals), (np.array(rows), np.array(cols))),
                      shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max 1^T g s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.001)

def mass_transfer_search(a, max_iters, deadline, tracker, rng):
    """Local search transferring mass from high-gradient to low-gradient positions."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        top_k = np.argsort(b)[::-1][:max(1, n // 8)]

        grad = np.zeros(n)
        for k in top_k:
            for j in range(n):
                kc = k - j
                if 0 <= kc < n:
                    if 2 * j == k:
                        grad[j] += a[j]
                    else:
                        grad[j] += 2.0 * a[kc]

        active = a > 0.005 * a.max()
        if active.sum() < 4:
            active = np.ones(n, dtype=bool)

        grad_safe = np.where(active, grad, np.inf)
        n_sel = max(3, n // 12)
        reduce_idx = np.argsort(grad_safe)[:n_sel * 2]
        reduce_idx = reduce_idx[active[reduce_idx]]
        increase_idx = np.argsort(-grad_safe)[:n_sel * 2]
        increase_idx = increase_idx[active[increase_idx]]

        if len(reduce_idx) == 0 or len(increase_idx) == 0:
            break

        best_J = current_J
        best_a = None

        for _ in range(40):
            i = int(rng.choice(reduce_idx))
            j = int(rng.choice(increase_idx))
            if i == j:
                continue
            for frac in [0.01, 0.03, 0.06, 0.12]:
                delta = frac * a[i]
                a_trial = a.copy()
                a_trial[i] -= delta
                a_trial[j] += delta
                if a_trial[i] < 0:
                    continue
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-11:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    break

        if best_a is not None:
            a = best_a
            current_J = best_J
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                break

        if it % 30 == 0:
            tracker.update(a)

                a = resample_seq(a, new_n)
            else:
                a += rng.binomial(1, 0.3, n) * temperature * 0.2

            a = np.maximum(a, 0.001)
            a /= a.sum()

            iters = max(25, 80 - 2 * r)
            run_lp_iterations(a, iters, deadline, tracker)

            if tracker.best_score < score_before - 1e-10:
                print(f"    restart {r}: improved to {tracker.best_score:.8f}", flush=True)

            temperature *= 0.96

        tracker.report("Annealing")
        print(f"  Time: {time.monotonic() - t0:.1f}s", flush=True)

    # === Phase 4: L-BFGS-B ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 4: L-BFGS-B", flush=True)
        print(f"{'='*55}", flush=True)
        n = tracker.best.size

        def lbb_obj(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            b = fast_conv(ax / s)
            return float(b.max())

        res = minimize(lbb_obj, tracker.best.astype(np.float64), method='L-BFGS-B',
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 800, 'max_fun': 120000,
                                'ftol': 1e-18, 'gtol': 1e-14})
        if res.fun < 1e9:
            tracker.update(res.x)
            print(f"  L-BFGS-B obj: {res.fun:.10f}", flush=True)

        tracker.report("L-BFGS-B")
        print(f"  Time: {time.monotonic() - t0:.1f}s", flush=True)

    # === Phase 5: Final LP Polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        if n <= 256:
            t0 = time.monotonic()
            print(f"\n{'='*55}", flush=True)
            print(f"Phase 5: Final LP Polish", flush=True)
            print(f"{'='*55}", flush=True)
            run_lp_iterations(tracker.best, 400, deadline, tracker, adapt_M=False)
            tracker.report("Final LP")
            print(f"  Time: {time.monotonic() - t0:.1f}s", flush=True)

    # Final
    print(f"\n{'='*55}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f}  n={tracker.best.size}  "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'='*55}", flush=True)

    return [float(x) for x in tracker.best]
