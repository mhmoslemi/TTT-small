import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_max_grad(h, T):
    """Compute gradient of log-sum-exp(h/T) wrt h, scaled."""
    h_scaled = h / T
    h_max = np.max(h_scaled)
    exp_h = np.exp(h_scaled - h_max)
    softmax = exp_h / np.sum(exp_h)
    return softmax  # d LSE / d h = softmax(h/T) / T ... actually d LSE(h/T)/dh = softmax(h/T) / T
    # Wait: LSE(h/T) = log(sum(exp(h_i/T)))
    # d/dh_j LSE = (1/T) * exp(h_j/T) / sum(exp(h_i/T)) = (1/T) * softmax_j
    # So the gradient wrt h is softmax/T
    # And the gradient wrt a[i] is sum_k (softmax[k]/T) * dh[k]/da[i]

def build_A_matrix(a):
    """Build the matrix such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp_response(a, M):
    """Solve: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def smooth_max_optimize(a, max_iter=5000, T_start=1.0, T_end=0.01, verbose=False):
    """Optimize using log-sum-exp with annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        # Anneal temperature
        progress = it / max_iter
        T = T_start * (T_end / T_start) ** progress
        
        h = np.convolve(a, a)
        M = np.max(h)
        
        # Compute gradient of LSE(h/T) wrt a
        h_scaled = h / T
        h_max = np.max(h_scaled)
        exp_h = np.exp(h_scaled - h_max)
        softmax = exp_h / np.sum(exp_h)
        
        # Gradient: grad[i] = (1/T) * sum_k softmax[k] * dh[k]/da[i]
        # dh[k]/da[i] = 2*a[k-i] if 0 <= k-i < n
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if softmax[k] < 1e-15:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                grad[j] += 2.0 * a[k - j] * softmax[k]
        grad /= T
        
        # Multiplicative update
        lr = 0.1 * (1.0 - 0.5 * progress)
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-10)
        s = np.sum(a_new)
        a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 1000 == 0:
            print(f"    SmoothMax it={it}, T={T:.4f}, obj={best_obj:.6f}")
    
    return best_a, best_obj

def peak_reduce(a, max_steps=10000, verbose=False):
    """Reduce the peak of conv(a,a) by greedily reducing the most impactful elements."""
    n = len(a)
    a = a.copy().astype(np.float64)
    h = np.convolve(a, a)
    peak = np.max(h)
    best_a = a.copy()
    best_peak = peak
    best_obj = compute_obj(best_a)
    
    for step in range(max_steps):
        kstar = np.argmax(h)
        
        # Compute blame for each i
        # blame[i] = 2*a[kstar-i] if 0 <= kstar-i < n
        j_start = max(0, kstar - n + 1)
        j_end = min(n - 1, kstar)
        blame = np.zeros(n)
        blame[j_start:j_end+1] = 2.0 * a[kstar - j_end : kstar - j_start + 1][::-1]
        
        # Find the i with maximum blame (and a[i] > 0)
        max_blame_idx = np.argmax(blame * (a > 1e-8))
        max_blame = blame[max_blame_idx]
        
        if max_blame < 1e-12:
            break
        
        # Try reducing a[max_blame_idx]
        # The peak would decrease by approximately max_blame * delta / 2
        # But it might create a new peak elsewhere
        # Let's try a small reduction
        for frac in [0.01, 0.005, 0.001, 0.0005]:
            old_val = a[max_blame_idx]

            if new_peak < peak * (1 - 1e-5):  # peak decreased

                obj = compute_obj(a)

                    best_obj = obj
                    best_a = a.copy()

                a[max_blame_idx] = old_val
                for k in range(2*n-1):
                    ki = k - max_blame_idx
                    if 0 <= ki < n:
                        h[k] -= 2 * delta_a * a[ki]
        
        if verbose and step % 1000 == 0:
            print(f"    PeakReduce step={step}, peak={peak:.6f}, obj={best_obj:.6f}")
    
    return best_a, best_obj

Wait, the incremental update has a bug. When I change `a[i]` by `delta_a`, the change in `h[k]` is:
`h_new[k] - h_old[k] = (a_new[i] * a[k-i] + a[k-i] * a_new[i]) - (a_old[i] * a[k-i] + a[k-i] * a_old[i])`
(if both terms exist, i.e., `i != k-i`)
`= 2 * (a_new[i] - a_old[i]) * a[k-i] = 2 * delta_a * a[k-i]`

If `i = k-i` (i.e., `k = 2i`):
`h_new[2i] - h_old[2i] = a_new[i]^2 - a_old[i]^2 = (a_new[i] - a_old[i]) * (a_new[i] + a_old[i]) = delta_a * (2*a_old[i] + delta_a)`

For small `delta_a`, this is approximately `2 * delta_a * a_old[i] = 2 * delta_a * a[i]`, which is consistent with the general formula (since `a[k-i] = a[2i-i] = a[i]`).

So the incremental update `h[k] += 2 * delta_a * a[k-i]` is correct for small changes (and approximately correct for the diagonal too).

But there's a subtlety: when I update `a[i]`, I should use the NEW value of `a` for the `a[k-i]` factor... actually no, the change is:
`h_new[k] = sum_j a_new[j] * a_new[k-j]`
The change from changing only `a[i]` (from `a_old[i]` to `a_new[i]`):
- Terms involving `a[i]`: `a_new[i] * a[k-i]` (if `k-i != i`) + `a[k-i] * a_new[i]` (if `k-i != i`) = `2 * a_new[i] * a[k-i]`
  vs old: `2 * a_old[i] * a[k-i]`
  Change: `2 * (a_new[i] - a_old[i]) * a[k-i] = 2 * delta_a * a[k-i]` ✓
  
- If `k-i = i` (diagonal): `a_new[i]^2 - a_old[i]^2 = delta_a * (2*a_old[i] + delta_a)`

Uses smooth-max gradient descent, LP alternation, targeted peak reduction,

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def compute_lse_gradient(a, T):
    """Vectorized gradient of LSE(conv(a,a)/T) using FFT."""
    n = len(a)
    h = np.convolve(a, a)
    h_scaled = h / T
    h_max = np.max(h_scaled)
    exp_h = np.exp(h_scaled - h_max)
    softmax = exp_h / np.sum(exp_h)
    
    # grad[i] = (2/T) * sum_j a[j] * softmax[i+j]
    a_flip = a[::-1]
    conv = np.convolve(a_flip, softmax)
    grad = (2.0 / T) * conv[n-1 : n-1+n]
    return grad, h, softmax

def smooth_max_optimize(a, max_iter=3000, T_start=0.5, T_end=0.05, verbose=False):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)

def build_A_matrix(a):

        j_end = min(n - 1, k)
        A[k, j_start:j_end+1] = a[k - j_end : k - j_start + 1][::-1]
    return A

    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(a, max_iter=200, verbose=False):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        h = np.convolve(a, a)
        M = np.max(h)
        g = solve_lp_response(a, M)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

            best_obj = obj_g
            best_a = g.copy()

        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new / s
        
        if verbose and it % 50 == 0:
            print(f"      LP it={it}, obj={best_obj:.6f}")
    
    return best_a, best_obj

def subgradient_polish(a, n_steps=3000, verbose=False):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_obj = compute_obj(best_a)

        h = np.convolve(a, a)
        max_h = np.max(h)
        
        # Active set: top positions

        active = h >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = np.zeros(len(h), dtype=bool)
            active[int(np.argmax(h))] = True
            n_active = 1

Let me try a different approach. Define `b_shift[m] = b[m]` for `m = 0, ..., 2n-2`.

`e[i] = sum_{j=0}^{n-1} a[j] * b[i+j]`

This is the "cross-correlation" of `a` and `b` at lag `-i` (or equivalently, the "anti-correlation").

In terms of convolution: `e[i] = sum_j a[j] * b[i+j] = sum_j a[j] * b_rev[2n-2-(i+j)]` where `b_rev[m] = b[2n-2-m]`.

`= sum_j a[j] * b_rev[2n-2-i-j]`

This is `(a * b_rev)[2n-2-i]` (the standard convolution of a and b_rev, evaluated at position 2n-2-i).

Let me verify: `(a * b_rev)[m] = sum_j a[j] * b_rev[m-j] = sum_j a[j] * b[2n-2-(m-j)] = sum_j a[j] * b[2n-2-m+j]`

For `m = 2n-2-i`: `= sum_j a[j] * b[2n-2-(2n-2-i)+j] = sum_j a[j] * b[i+j]` ✓

So `e[i] = (a * b_rev)[2n-2-i]` where `b_rev[m] = b[2n-2-m]`.

Alternatively, `e[i] = conv(a, b[::-1])[2n-2-i]`.
