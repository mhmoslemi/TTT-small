"""
Optimized peak-shaving + LP + CD for minimizing 2n*max(conv)/(sum^2)
"""

import numpy as np
from scipy.optimize import linprog
import time

def normalize(f):
    s = f.sum()
    return f / s if s > 1e-30 else f

def score(f):
    n = len(f)
    b = np.convolve(f, f)
    s = f.sum()
    return 2.0 * n * b.max() / (s * s)

def peak_shave(f, n_iter, rng, eta0=0.3, eta_min=0.01, top_k=10, threshold=0.8):
    """
    Aggressive peak shaving with weighted pressure from top-k peaks.
    """
    f = f.copy().astype(float)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_s = score(f)
    
    for it in range(n_iter):
        b = np.convolve(f, f)
        M = b.max()
        
        # Find top-k peaks
        # Use threshold: peaks above threshold*M
        active = np.where(b >= threshold * M)[0]
        if len(active) > top_k:
            # Keep only the top-k
            top_idx = np.argsort(b)[::-1][:top_k]
            active = top_idx
        
        # Compute weighted pressure
        weights = b[active] / M  # All in [threshold, 1]
        # Or use exponential: weights = exp(alpha * (b[active] - M) / M)
        
        P = np.zeros(n)
        for idx, k in enumerate(active):
            # b[k] = sum_i f[i] * f[k-i]
            # Pressure on f[i] from peak k is f[k-i]
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                j = k - i
                if 0 <= j < n:
                    P[i] += weights[idx] * f[j]
        
        if P.max() < 1e-15:
            break
        
        # Adaptive eta
        eta = eta0 * (eta_min / eta0) ** (it / max(1, n_iter // 2))
        # Linear decay from eta0 to eta_min over first half, then stay at eta_min
        
        # Multiplicative update
        P_norm = P / P.max()
        f_new = f * np.exp(-eta * P_norm)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)
        
        s_new = score(f_new)
        if s_new < best_s:
            best_s = s_new
            best_f = f_new.copy()
            f = f_new
        else:
            # Try smaller step
            f_new2 = f * np.exp(-eta * 0.3 * P_norm)
            f_new2 = np.clip(f_new2, 1e-15, None)
            f_new2 = normalize(f_new2)
            s_new2 = score(f_new2)
            if s_new2 < best_s:
                best_s = s_new2
                best_f = f_new2.copy()
                f = f_new2
            else:
                # Stall - perturb

Let me think about one more thing: **the structure of the optimal solution**.

For the score to be low, we need the convolution to be "flat" (all values close to the average = 1/(2n-1) for normalized sum). This means the sequence should have "random-like" autocorrelation properties.

For non-negative sequences, this is like finding a "flat spectrum" non-negative signal. The DFT of such a signal would have flat magnitude (CAZAC-like), but the signal must be non-negative.

One approach: start with a random sequence, take its DFT, flatten the magnitude, and take the inverse DFT. The result might be negative in some places, so we clip and iterate. This is basically the spectral flattening approach.

Another approach: use the fact that the convolution is the "product" in the frequency domain (with a phase factor). Specifically:
DFT(convolve(f, f)) = DFT(f) * DFT(f) (with appropriate zero-padding for linear convolution)

So if we want the convolution to be flat, we want |DFT(f)|^2 to be flat, which means |DFT(f)| to be flat. This is the flat-magnitude spectrum condition.

But the constraint is that f >= 0, which in the frequency domain means that the inverse DFT of any flat-magnitude spectrum must be non-negative. This is a strong constraint.

The spectral flattening approach iterates between:
1. In frequency domain: set |DFT(f)| = constant (except DC)
2. In time domain: clip to non-negative and renormalize

This is similar to the Gerchberg-Saxton algorithm or the iterative Fourier transform algorithm (IFTA).

def spectral_flat(f, n_iter=500, rng=None):

    f = normalize(f)
    n_fft = 2 * n - 1  # For linear convolution, need at least 2n-1 points
    # Use next power of 2 for FFT efficiency
    n_fft_fft = 1 << (n_fft - 1).bit_length()
    
    best_f = f.copy()
    best_s = score(f)
    
    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: flat magnitude
        # Energy conservation: sum(f^2) = (1/n_fft_fft) * sum(|F|^2)
        # If we set |F[k]| = c for all k (k > 0), and |F[0]| = 1 (DC = sum(f) = 1)
        # Then sum(|F|^2) = 1 + (n_fft_fft//2) * c^2
        # And sum(f^2) = (1 + (n_fft_fft//2) * c^2) / n_fft_fft
        # We want to preserve sum(f^2), so:
        current_energy = np.sum(f**2)
        # n_fft_fft/2 = number of positive frequency bins (including DC)
        n_pos = n_fft_fft // 2 + 1  # Number of bins in rfft
        # sum(|F|^2) = |F[0]|^2 + sum_{k=1}^{n_pos-1} |F[k]|^2
        # = 1 + (n_pos - 1) * c^2
        # sum(f^2) = (1 + (n_pos-1)*c^2) / n_fft_fft
        # c^2 = (current_energy * n_fft_fft - 1) / (n_pos - 1)
        c2 = (current_energy * n_fft_fft - 1) / (n_pos - 1)
        if c2 < 0:
            c2 = 0
        c = np.sqrt(c2)
        
        mag_new = np.full(n_pos, c)
        mag_new[0] = 1.0  # DC
        
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft_fft)[:n]

        s = f_new.sum()
        if s < 1e-10:

        f_new = f_new / s

        if s_new < best_s:

            best_f = f_new.copy()
            f = f_new
        else:
            # Don't update f, but keep going (the phase is changing)
            pass
    
    return best_f, best_s

Peak-shaving + LP refinement + coordinate descent for minimizing
2n*max(convolve(f,f))/(sum(f))^2 with f >= 0.

import numpy as np
from scipy.optimize import linprog
import time

def norm(f):
    s = f.sum()
    return f / s if s > 1e-30 else f

def sc(f):
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    b = np.convolve(f, f)
    return 2.0 * n * b.max() / (s * s)

def peaks_pressure(f, top_k=10, thresh=0.7):
    """Compute pressure vector from top-k peaks."""
    n = len(f)
    b = np.convolve(f, f)
    M = b.max()
    if M < 1e-30:
        return np.zeros(n), M
    # Find peaks above threshold
    candidates = np.where(b >= thresh * M)[0]
    if len(candidates) > top_k:
        # Keep only top-k
        top_idx = np.argsort(b)[::-1][:top_k]
        candidates = top_idx
    # Compute weighted pressure
    w = b[candidates] / M
    P = np.zeros(n)
    for idx, k in enumerate(candidates):
        i_lo = max(0, k - n + 1)
        i_hi = min(n, k + 1)
        j_lo = k - (i_hi - 1)
        j_hi = k - i_lo + 1
        P[i_lo:i_hi] += w[idx] * f[j_lo:j_hi]
    return P, M

def peak_shave(f, n_iter=500, rng=None, eta0=0.5):

            best = f_new.copy()

        else:

            f_try = norm(f_try)
            s_try = sc(f_try)

                f = f_try

                        best = fp.copy()

def build_A(f):
    """Build convolution matrix: A @ g = convolve(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        A[k, i0:i1] = f[k - (i1 - 1):k - i0 + 1]
    return A

def lp_step(f, t_candidates=None, lp_timeout=5):
    """One LP step: find g maximizing sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    b = np.convolve(f, f)
    M = b.max()
    A = build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': lp_timeout})
        if not res.success or res.x is None:
            return None, None
        g = np.clip(res.x, 0, None)
        gs = g.sum()
        if gs < 1e-10:
            return None, None
        return g / gs, None
    except:
        return None, None

def lp_refine(f, time_budget, t0=0.5, rng=None):

    f = norm(f)
    n = len(f)
    best = f.copy()
    best_s = sc(f)
    t = t0

    while time.monotonic() - start < time_budget:

        if g is None:
            break
        improved = False
        for step in [min(1.0, t * 1.5), t, t * 0.5, t * 0.25, t * 0.1]:

    n = len(f)

            orig = f[i]

            best_ls = best_s

                if s < 1e-10:

                    continue
                s_val = sc(f)
                if s_val < best_ls:
                    best_ls = s_val
                    best_v = f[i]
            f[i] = best_v
            if best_ls < best_s:
                best_s = best_ls
        f = norm(f)
    
    return f, best_s

def spectral_flat(f, n_iter=200, rng=None):

    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = norm(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

"""
Multi-phase optimization: spectral flattening + adaptive peak shaving + LP refinement
+ coordinate descent, with aggressive time management and diverse restarts.

import numpy as np
from scipy.optimize import linprog
import time


def _norm(f):
    s = f.sum()
    return f / s if s > 1e-30 else f


def _sc(f):
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    b = np.convolve(f, f)
    return 2.0 * n * b.max() / (s * s)


def _pressure(f, top_k=8, thresh=0.75):
    n = len(f)
    b = np.convolve(f, f)
    M = b.max()
    if M < 1e-30:
        return np.zeros(n), M
    cands = np.where(b >= thresh * M)[0]
    if len(cands) > top_k:
        cands = np.argsort(b)[::-1][:top_k]
    w = b[cands] / M
    P = np.zeros(n)
    for idx, k in enumerate(cands):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        P[j_lo:j_hi + 1] += w[idx] * f[k - j_hi:k - j_lo + 1][::-1]
    return P, M


def _peak_shave(f, n_iter=500, rng=None, eta0=0.5, top_k=8, thresh_start=0.6, thresh_end=0.9):
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = _norm(f)
    n = len(f)
    best = f.copy()
    best_s = _sc(f)
    stall = 0

    for it in range(n_iter):
        frac = it / max(1, n_iter)
        thresh = thresh_start + (thresh_end - thresh_start) * frac
        P, M = _pressure(f, top_k=top_k, thresh=thresh)
        if P.max() < 1e-15:
            break
        eta = eta0 * max(0.05, 1.0 - 0.8 * frac)
        Pn = P / P.max()
        f_new = f * np.exp(-eta * Pn)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = _norm(f_new)
        s_new = _sc(f_new)
        if s_new < best_s - 1e-12:
            best_s = s_new
            best = f_new.copy()
            f = f_new
            stall = 0
        else:
            f_try = f * np.exp(-eta * 0.25 * Pn)
            f_try = np.clip(f_try, 1e-15, None)
            f_try = _norm(f_try)
            s_try = _sc(f_try)
            if s_try < best_s - 1e-12:
                best_s = s_try
                best = f_try.copy()
                f = f_try
                stall = 0
            else:
                stall += 1
                if stall > 40:
                    fp = f * (1 + 0.15 * rng.normal(size=n))
                    fp = np.clip(fp, 0, None)
                    fp = _norm(fp)
                    if _sc(fp) < best_s:
                        f = fp
                        best = fp.copy()
                        best_s = _sc(fp)
                        stall = 0
                    else:
                        break
    return best, best_s


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = f[k - j_hi:k - j_lo + 1][::-1]
    return A


def _lp_solve(f, timeout=5):
    n = len(f)
    b = np.convolve(f, f)
    M = b.max()
    A = _build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout})
        if not res.success or res.x is None:
            return None
        g = np.clip(res.x, 0, None)
        gs = g.sum()
        if gs < 1e-10:
            return None
        return g / gs
    except Exception:
        return None


def _lp_refine(f, time_budget, t0=0.4, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = _norm(f)
    n = len(f)
    best = f.copy()
    best_s = _sc(f)
    t = t0
    start = time.monotonic()
    n_lp_calls = 0

    while time.monotonic() - start < time_budget:
        g = _lp_solve(f, timeout=max(1, min(5, int(time_budget / 10))))
        n_lp_calls += 1
        if g is None:
            break
        improved = False
        for step in [min(1.0, t * 1.3), t, t * 0.5, t * 0.2, t * 0.05]:
            f_new = (1.0 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            s_new = _sc(f_new)
            if s_new < best_s - 1e-12:
                best_s = s_new
                best = f_new.copy()
                f = f_new
                t = min(1.0, t * 1.05)
                improved = True
                break
        if not improved:
            t *= 0.5
            if t < 0.003:
                fp = f + 0.05 * rng.normal(size=n)
                fp = np.clip(fp, 0, None)
                fp = _norm(fp)
                if _sc(fp) < best_s:
                    f = fp
                    best = fp.copy()
                    best_s = _sc(fp)
                    t = 0.1
                else:
                    break
    return best, best_s


def _cd_polish(f, n_sweeps=5, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = _norm(f)
    n = len(f)
    best_s = _sc(f)

    for _ in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_v = orig
            best_ls = best_s
            for factor in [0.4, 0.6, 0.8, 0.9, 1.0 / 0.9, 1.0 / 0.8, 1.0 / 0.6, 1.0 / 0.4]:
                f[i] = max(0.0, orig * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                sv = _sc(f)
                if sv < best_ls:
                    best_ls = sv
                    best_v = f[i]
            f[i] = best_v
            if best_ls < best_s:
                best_s = best_ls
        f = _norm(f)
    return f, best_s


def _spectral_flat(f, n_iter=300, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = _norm(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    best = f.copy()
    best_s = _sc(f)

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        E = np.sum(f ** 2)
        n_bins = len(F)
        c2 = (E * n_fft - 1.0) / (n_bins - 1)
        c = np.sqrt(max(0.0, c2))
        mag_new = np.full(n_bins, c)
        mag_new[0] = 1.0
        if it % 80 == 79:
            phase = phase + rng.normal(0, 0.03, n_bins)
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new /= s
        s_new = _sc(f_new)
        if s_new < best_s - 1e-12:
            best_s = s_new
            best = f_new.copy()
        f = f_new
    return best, best_s
