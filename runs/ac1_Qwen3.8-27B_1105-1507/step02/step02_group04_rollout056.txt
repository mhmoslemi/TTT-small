import numpy as np

def compute_conv(a):
    """Compute full convolution conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_max_weights(conv, tau):
    """Compute log-sum-exp weights for smooth max approximation."""
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= wsum
    return w


def grad_smoothed_max(a, tau):
    """Gradient of smoothed max of conv(a,a) w.r.t. a. Uses correl(w, a)."""
    n = len(a)
    conv = compute_conv(a)
    w = smooth_max_weights(conv, tau)
    # grad_i = 2 * sum_j a_j * w_{i+j} = 2 * correlate(w, a, 'full')[i]
    grad = 2.0 * np.correlate(w, a, 'full')[:n]
    return grad, conv, float(np.max(conv))


def grad_exact_max(a):
    """Exact subgradient at the argmax of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Subgradient: average over all argmax positions
    active = np.isclose(conv, max_val, atol=1e-10)
    w = np.zeros(len(conv))
    w[active] = 1.0 / active.sum()
    grad = 2.0 * np.correlate(w, a, 'full')[:n]
    return grad, conv, max_val


def project_nonneg_sum(a, target=1.0):
    """Project onto {x >= 0, sum(x) = target} via scaling."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.full(len(a), target / len(a))
    return a * (target / s)

def build_toeplitz(a):
    """Build (2n-1) x n Toeplitz matrix for convolution with a."""
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve(a, M, time_limit=5.0):
    """Solve: max sum(g) s.t. T(a)@g <= M, g >= 0."""
    n = len(a)
    T = build_toeplitz(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None

        # Choose method based on round
        if round_num % 4 == 1:
            # Gradient with medium tau
            a = gradient_descent(a, max_iter=int(remaining * 0.4 * 25),
                                 tau_start=10.0, tau_end=60.0, lr_init=0.08)
        elif round_num % 4 == 2:
            # LP
            a = lp_iterate(a, max_iter=min(20, int(remaining * 0.3 / 2)), t=0.4)
        elif round_num % 4 == 3:
            # Coordinate descent
            if n <= 500:
                a = coordinate_descent(a, max_sweeps=min(3, int(remaining / 5)))
            else:
                a = gradient_descent(a, max_iter=int(remaining * 0.3 * 20),
                                     tau_start=20.0, tau_end=80.0, lr_init=0.06)
        else:
            # Perturbed restart + gradient
            pert = a * np.exp(rng.normal(0, 0.03, n))
            pert = project_nonneg_sum(pert)
            obj_p = compute_obj(pert)
            obj_cur = compute_obj(a)
            if obj_p < obj_cur:
                a = pert
            a = gradient_descent(a, max_iter=int(remaining * 0.3 * 25),
                                 tau_start=8.0, tau_end=50.0, lr_init=0.1)

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"    Polish round {round_num}: obj={best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for optimal step function coefficients minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    
    Uses gradient descent, LP alternating, coordinate descent,
    and diverse initializations including Chebyshev windows.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Try parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(parent) if parent is not None else 200

    # Determine n values to explore
    n_candidates = set()
    for delta in [-100, -50, -20, 0, 20, 50, 100, 150]:
        n_try = n_parent + delta
        if 20 <= n_try <= 600:
            n_candidates.add(n_try)
    # Also try some round numbers
    for n_r in [50, 80, 100, 120, 150, 200, 250, 300, 400, 500]:
        if abs(n_r - n_parent) <= 200:
            n_candidates.add(n_r)
    n_values = sorted(n_candidates)

    print(f"Parent n={n_parent}, exploring n values: {n_values}")

    # Time allocation per n
    n_total = len(n_values)
    for idx, n in enumerate(n_values):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Allocate more time to n values near the parent
        dist = abs(n - n_parent)
        if dist <= 20:
            alloc = min(120.0, remaining * 0.25)
        elif dist <= 50:
            alloc = min(80.0, remaining * 0.18)
        elif dist <= 100:
            alloc = min(50.0, remaining * 0.12)
        else:
            alloc = min(30.0, remaining * 0.08)

        alloc = max(15.0, alloc)
        if time.monotonic() + alloc > deadline:
            alloc = max(15.0, deadline - time.monotonic() - 5)
        if alloc < 15:
            break

        print(f"\n--- n={n} ({dist} from parent), budget={alloc:.0f}s, {remaining:.0f}s left ---")

        inits = make_initializations(n, seed + n * 104729, parent)

        # Limit number of inits based on time
        max_inits = max(2, min(len(inits), int(alloc / 8)))
        inits = inits[:max_inits]

        for i_init, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            rem = deadline - time.monotonic()
            if rem < 15:
                break

            init_time = min(alloc / max_inits, rem - 5)
            if init_time < 10:
                break

            a_opt, obj_opt = run_optimization(n, a0, init_time, seed + i_init)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} init={i_init}: NEW BEST obj={best_score:.6f} "
                      f"(elapsed {time.monotonic()-start_time:.0f}s)")

    # Final intensive polishing
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n=== Final polish: n={len(best_seq)}, {remaining:.0f}s ===")
            a_pol, obj_pol = final_polish(best_seq, remaining)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    n_final = len(best_seq)
    print(f"\nFinal: n={n_final}, score={best_score:.6f}, "
          f"time={time.monotonic()-start_time:.1f}s")

    # Validate
    result = [float(x) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            print(f"  Note: evaluate_sequence gives {val:.6f} (vs computed {best_score:.6f})")
            best_score = val
    except Exception:
        pass

    return result
