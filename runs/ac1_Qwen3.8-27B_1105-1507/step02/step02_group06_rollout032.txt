```python
from scipy.optimize import linprog

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))

def compute_subgradient(f, epsilon=0.0):
    """Compute subgradient of max(conv) with respect to f.
    Uses all lags within epsilon of the max."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    
    # Find all active lags (within epsilon of max)
    if epsilon <= 0:
        active = np.where(conv >= M - 1e-10)[0]
    else:
        active = np.where(conv >= M - epsilon)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Equally weight active constraints
    weights = np.ones(len(active)) / len(active)
    
    grad = np.zeros(n)
    for k, w in zip(active, weights):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += w * 2.0 * f[k - j]
    
    return grad, M, conv

    n = len(f)

    f = f * (S / float(np.sum(f)))  # keep same sum

    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and no_improve < 200:

        M = float(np.max(conv))
        current_score = 2.0 * n * M / (S * S)

        if top_k is not None:

            weight_sum = 0
            for k in top_indices:

            if weight_sum < 1e-15:
                weight_sum = 1
            grad = np.zeros(n)
            for k in top_indices:
                w = max(conv[k] - 0.5 * M, 0) / weight_sum
                if w <= 0:

                j_max = min(k, n - 1)
```

At each iteration:

- Then, scale: f = f / sum(f)

- Find the largest t such that sum(max(f - t, 0)) = 1
- Then f_projected = max(f - t, 0)

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

This is the standard algorithm from Duchi et al. (2012).

```python
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(0, dtype=np.int32)
    cols = np.empty(0, dtype=np.int32)
    vals = np.empty(0, dtype=np.float64)
    
    # Count non-zeros
    nnz = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        nnz += (j_max - j_min + 1)
    
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    vals = np.empty(nnz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, time_budget=5.0):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    t_start = time.monotonic()
    if time.monotonic() - t_start > time_budget:
        return None
    
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def subgradient_step(f, lr, rel_tol=0.05, momentum=None, beta=0.9):
    """One subgradient step with all active constraints.
    f is assumed to have sum=1."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    
    # Active lags: within (1 - rel_tol) * M of the max
    threshold = M * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]
    weights = np.maximum(conv[active] - threshold, 0.0)
    W = float(np.sum(weights))
    if W < 1e-15:
        active = np.array([int(np.argmax(conv))])
        weights = np.array([1.0])
        W = 1.0
    weights = weights / W
    
    # Compute gradient: grad[j] = 2 * sum_i weights[i] * f[active[i] - j]
    grad = np.zeros(n)
    for i, k in enumerate(active):
        w = weights[i]
        if w < 1e-15:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        # f[k-j] for j in [j_min, j_max]
        # Vector: f[k-j_max : k-j_min+1] reversed
        seg = f[k - j_max : k - j_min + 1]
        grad[j_min : j_max + 1] += w * 2.0 * seg[::-1]
    
    if momentum is None:
        momentum = np.zeros(n)
    
    momentum = beta * momentum - lr * grad
    new_f = f + momentum
    new_f = project_simplex(new_f)
    
    return new_f, momentum, M, conv


def optimize_subgradient(f_init, deadline, lr_init=0.03, rel_tol=0.05,
                         beta=0.9, verbose_prefix="", check_interval=50):
    """Main subgradient optimization loop with adaptive step size."""
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, float('inf')
    f = f / S  # normalize to sum=1
    
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0
    it = 0
    last_improve_time = time.monotonic()
    
    while time.monotonic() < deadline:
        # Check for timeout on improvement
        if time.monotonic() - last_improve_time > 120.0:
            # Restart with perturbation
            f = best_f + 0.01 * np.random.default_rng(it).standard_normal(n)
            f = project_simplex(f)
            momentum = np.zeros(n)
            lr = lr_init
            no_improve = 0
            last_improve_time = time.monotonic()
            continue
        
        new_f, momentum, M, conv = subgradient_step(f, lr, rel_tol, momentum, beta)
        new_score = compute_score(new_f)
        it += 1
        
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            last_improve_time = time.monotonic()
            if verbose_prefix and it % check_interval == 0:
                print(f"    {verbose_prefix}it={it}: score={best_score:.6f}, lr={lr:.5f}")
        else:
            no_improve += 1
            lr *= 0.95
            momentum *= 0.5
            if lr < 1e-9 or no_improve > 500:
                # Try a smaller rel_tol for more precise gradient
                rel_tol = max(0.01, rel_tol * 0.7)
                no_improve = 0
                if rel_tol < 0.01:
                    break
    
    return best_f, best_score


def lp_polish(f_init, deadline, max_iters=50, verbose_prefix=""):
    """LP-based polishing with line search."""
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, float('inf')
    f = f / S
    
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        
        # Solve LP with slightly reduced M target
        M_target = M * 0.995
        g = solve_lp_direction(f, M_target, time_budget=10.0)
        if g is None:
            break
        
        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            break
        
        # Normalize g to have sum=1
        g = g / sum_g
        
        # Line search
        best_alpha = 0.0
        best_alpha_score = best_score
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
        for alpha in alphas:
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = project_simplex(candidate)
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_alpha_score:
                best_alpha_score = cand_score
                best_alpha = alpha
        
        it += 1
        if best_alpha > 0:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = project_simplex(f)
            best_f = f.copy()
            best_score = best_alpha_score
            if verbose_prefix:
                print(f"    {verbose_prefix}LP it={it}: score={best_score:.6f}, alpha={best_alpha:.3f}")
        else:
            break
    
    return best_f, best_score
```

Optimization of non-negative step function to minimize 2n*max(conv(a,a))/sum(a)^2.
Strategy: Adaptive subgradient descent with simplex projection, periodic LP polishing,
multiple initializations, and restart mechanism. Fixed length n=1024.
"""
import numpy as np
import time

def compute_score(f):
    """Compute the objective value directly."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.where(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = sum_j A[k,j]*g[j] = convolve(f,g)[k]."""
    n = len(f)

        j_max = min(k, n - 1)
        nnz += (j_max - j_min + 1)

    cols = np.empty(nnz, dtype=np.int64)

            cols[idx] = j

def do_subgradient_step(f, lr, rel_tol, momentum, beta, rng):
    """Perform one subgradient step. f has sum=1."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    
    # Active lags
    threshold = M * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    
    weights = np.maximum(conv[active] - threshold, 0.0)
    W = float(np.sum(weights))
    if W < 1e-15:
        weights = np.ones(len(active))
        W = float(len(active))
    weights = weights / W
    
    # Add small randomization to avoid symmetry issues
    if rng is not None and len(active) > 2:
        perturb = rng.standard_normal(len(active)) * 0.01
        weights = weights + perturb * np.max(weights)
        weights = np.maximum(weights, 0.0)
        W = float(np.sum(weights))
        if W > 0:
            weights = weights / W
    
    # Compute gradient
    grad = np.zeros(n)
    for i in range(len(active)):
        k = int(active[i])
        w = weights[i]
        if w < 1e-15:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        seg = f[k - j_max : k - j_min + 1][::-1]
        grad[j_min : j_max + 1] += w * 2.0 * seg
    
    momentum = beta * momentum - lr * grad
    new_f = f + momentum
    new_f = project_simplex(new_f)
    
    return new_f, momentum, M
