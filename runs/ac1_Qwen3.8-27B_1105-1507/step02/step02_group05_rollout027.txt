import numpy as np

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def cross_corr_n(w, a):
    """Compute r[m] = sum_k w[k]*a[k-m] for m=0..n-1 using FFT."""
    n = len(a)
    nw = len(w)
    N = 1
    while N < nw + n - 1:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    r = np.fft.irfft(np.conj(W) * A, N)
    return r[:n]


def smooth_obj_and_grad(x, beta):
    """Returns (objective, gradient) for the smoothed J_beta."""
    n = len(x)
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_conv(a)
    c = b / (s * s)
    
    max_c = c.max()
    exp_c = np.exp(beta * (c - max_c))
    w = exp_c / exp_c.sum()
    
    obj = 2.0 * n / beta * (max_c + np.log(exp_c.sum()))
    
    r = cross_corr_n(w, a)
    wb = np.dot(w, b)
    
    grad = (4.0 * n / (s * s)) * r - (4.0 * n / (s ** 3)) * wb
    return float(obj), grad.astype(np.float64)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    n = len(a)
    c_vec = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

                options={'time_limit': 30})
    if r.success and r.x is not None:

        if g.sum() > 1e-10:

    return None

def bfgs_smooth(a, beta_schedule, deadline, tracker):
    """Run L-BFGS-B on smoothed objective with increasing beta."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        def obj(x):
            val, _ = smooth_obj_and_grad(x, beta)
            return val
        
        def grad(x):
            _, g = smooth_obj_and_grad(x, beta)
            return g
        
        bounds = [(0.0, None)] * len(a)
        max_fun = min(3000, max(500, int(60 / (time.monotonic() - tracker.t_start + 1) * 1000)))
        
        res = minimize(obj, a, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': 150, 'max_fun': max_fun, 'ftol': 1e-18, 'gtol': 1e-12})
        if res.fun < 1e9:
            a = np.maximum(res.x, 0.0)
            tracker.update(a)
    
    return a

    return inits
