def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smoothed_gradient_multi(conv, a, tau, top_k=5):
    """Compute smoothed gradient focusing on top-k peaks."""
    n = len(a)
    # Find top-k peaks
    idx = np.argsort(conv)[-top_k:]
    # Weight by softmax over top-k
    top_vals = conv[idx]
    max_v = top_vals.max()
    w_top = np.exp((top_vals - max_v) / tau)
    w_top /= w_top.sum()
    
    # Also add a small uniform weight over all
    w_all = np.exp((conv - conv.max()) / (tau * 3))
    w_all /= w_all.sum()
    
    # Combine: 70% top-k, 30% all
    weights = 0.7 * np.zeros_like(conv) + 0.3 * w_all
    for i, idx_i in enumerate(idx):
        weights[idx_i] += 0.7 * w_top[i]
    
    # Compute gradient via correlation
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    return grad

def subgradient_step(a, tau, step, top_k=5):
    """One subgradient descent step with projection onto simplex."""
    n = len(a)
    conv = compute_conv(a)
    grad = smoothed_gradient_multi(conv, a, tau, top_k)
    # Center gradient
    grad_centered = grad - np.mean(grad)
    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a, np.max(conv)
    a_new = a_new / s
    return a_new, np.max(conv)

def build_A_sparse(a):
    """Build constraint matrix for LP. Returns in a form suitable for linprog."""
    n = len(a)
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max+1] = a[k-j_max : k-j_min+1][::-1]
    return A
