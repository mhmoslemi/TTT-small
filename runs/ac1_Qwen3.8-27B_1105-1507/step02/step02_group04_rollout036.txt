```python
    n = len(a)
    a = a.copy()
    a = a / np.sum(a)
    
    for it in range(max_iter):
        conv = compute_conv(a)
        M = np.max(conv)
        # Active set: indices where conv is close to max
        active = np.where(conv >= M - 1e-8)[0]
        
        if len(active) > n:
            # Keep only the top n
            active_sorted = active[np.argsort(conv[active])[::-1][:n]]
            active = active_sorted
        
        # Build the linear system
        m = len(active)
        # Unknowns: da (length n) and dM (scalar)
        # Equations: J[k,:] da - dM = M - conv[k] for k in active (m equations)
        #             1^T da = 1 - sum(a) (1 equation)
        # Total: m+1 equations, n+1 unknowns
        
        if m + 1 == n + 1:  # Square system
            A = np.zeros((m+1, n+1))
            b = np.zeros(m+1)
            for idx, k in enumerate(active):
                for i in range(n):
                    diff = k - i
                    if 0 <= diff < n:
                        A[idx, i] = 2 * a[diff]
                A[idx, n] = -1
                b[idx] = M - conv[k]
            A[m, :n] = 1.0
            A[m, n] = 0
            b[m] = 1 - np.sum(a)
            
            try:
                sol = np.linalg.solve(A, b)
            except np.linalg.LinAlgError:
                break
            da = sol[:n]
            dM = sol[n]
            
        elif m + 1 < n + 1:  # Underdetermined - use pseudo-inverse or add regularization
            # Use minimum norm solution
            A = np.zeros((m+1, n+1))
            b = np.zeros(m+1)
            for idx, k in enumerate(active):
                for i in range(n):
                    diff = k - i
                    if 0 <= diff < n:
                        A[idx, i] = 2 * a[diff]
                A[idx, n] = -1
                b[idx] = M - conv[k]
            A[m, :n] = 1.0
            b[m] = 1 - np.sum(a)
            
            try:
                sol, _, _, _ = np.linalg.lstsq(A, b, rcond=None)
            except:
                break
            da = sol[:n]
            dM = sol[n]
        else:  # m+1 > n+1: overdetermined

            b[m] = 1 - np.sum(a)

            except:
                break
            da = sol[:n]
            dM = sol[n]
        
        # Take a damped step
        alpha = 1.0
        for _ in range(50):
            a_new = a + alpha * da
            if np.any(a_new < 0):

                continue
            a_new = a_new / np.sum(a_new)
            conv_new = compute_conv(a_new)
            M_new = np.max(conv_new)
            if M_new <= M * (1 + 1e-10):
                break
            alpha *= 0.5
        
        a = a_new
        M = np.max(compute_conv(a))
        
        if alpha < 1e-8:
            break
    
    return a
```

conv[n-1] <= sqrt(sum a[j]²) * sqrt(sum a[n-1-j]²) = sum a[j]²

And sum a[j]² >= (sum a[j])²/n = 1/n.

So conv[n-1] <= sum a[j]², but conv[n-1] can be as small as 0 (e.g., if a is supported on {0, 1} and n-1 >= 2, then a[n-1-j] = 0 for j=0,1).

So the center of the convolution is NOT necessarily large for non-symmetric sequences!

This is the key insight: by making the sequence "lopsided" (concentrated on one end), we can make the center of the convolution small.

For example: a = [1, 0, 0, ..., 0]. conv = [1, 0, 0, ..., 0]. Max = 1. Objective = 2n*1/1 = 2n. Bad.

The issue is that for small support, the conv values are large relative to 1/n.

```python
                conv_new[k] += 2 * delta * a[diff]
    return conv_new


def coord_descent_1d(a, conv, i, max_evals=60):
    """Find optimal x = a[i] by 1D minimization of max(conv) with renormalization.
    Uses incremental conv updates. Returns (a_new, conv_new, improved)."""
    n = len(a)
    x_old = a[i]
    S = np.sum(a)  # current sum

    # We minimize: max(conv_new[k]) / (S + x - x_old)^2 * (S)^2
    # Since we renormalize, the objective is: max(conv_renorm) where conv_renorm = conv_after * (S/(S+x-x_old))^2
    # Equivalently: max(conv_after) / (S + x - x_old)^2

    # Bounds for x: [0, S + x_old) (can't exceed total sum)
    x_max = S * 2.0  # generous upper bound

    best_x = x_old
    best_val = np.max(conv) / (S * S)  # current normalized max

    # Quick check: what's the max if x=0?
    conv_zero = conv.copy()
    for k in range(max(0, i), min(2 * n - 1, i + n)):
        diff = k - i
        if 0 <= diff < n:
            if diff == i:
                conv_zero[k] -= x_old * x_old
            else:
                conv_zero[k] -= 2 * x_old * a[diff]
    S_zero = S - x_old
    if S_zero > 1e-10:
        val_zero = np.max(conv_zero) / (S_zero * S_zero)
        if val_zero < best_val:
            best_val = val_zero
            best_x = 0.0

    # Use golden section search on [0, x_max]
    if best_x == 0.0:
        # x=0 is already best, skip
        a[i] = 0.0
        return a, conv_zero, True

    # Try a few candidate values around the current
    candidates = [x_old * 0.5, x_old * 0.7, x_old * 1.3, x_old * 1.5, x_old * 0.3]
    candidates = [max(0, min(c, x_max)) for c in candidates]

    for x_try in candidates:
        if abs(x_try - x_old) < 1e-12:
            continue
        delta = x_try - x_old
        conv_try = conv.copy()
        for k in range(max(0, i), min(2 * n - 1, i + n)):
            diff = k - i
            if 0 <= diff < n:
                if diff == i:
                    conv_try[k] += delta * (x_try + x_old)
                else:
                    conv_try[k] += 2 * delta * a[diff]
        S_try = S + delta
        if S_try < 1e-10:
            continue
        val = np.max(conv_try) / (S_try * S_try)
        if val < best_val - 1e-14:
            best_val = val
            best_x = x_try

    # If we found improvement, do a finer search around best_x
    if best_x != x_old:
        # Narrow search
        lo = max(0, best_x - x_old * 0.5)
        hi = min(x_max, best_x + x_old * 0.5)
        if hi - lo < 1e-8:
            hi = best_x * 2 + 1e-8

        # Golden section
        gr = (np.sqrt(5) - 1) / 2
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)

        def eval_x(x):
            if x < 0:
                x = 0
            delta = x - x_old
            conv_try = conv.copy()
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                diff = k - i
                if 0 <= diff < n:
                    if diff == i:
                        conv_try[k] += delta * (x + x_old)
                    else:
                        conv_try[k] += 2 * delta * a[diff]
            S_try = S + delta
            if S_try < 1e-10:
                return 1e10
            return float(np.max(conv_try) / (S_try * S_try))

        f1 = eval_x(c1)
        f2 = eval_x(c2)
        for _ in range(20):
            if hi - lo < 1e-10:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = eval_x(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = eval_x(c2)

        x_opt = (lo + hi) / 2
        f_opt = eval_x(x_opt)
        if f_opt < best_val - 1e-14:
            best_val = f_opt
            best_x = x_opt

    if best_x != x_old:
        a[i] = best_x
        conv_new = conv.copy()
        delta = best_x - x_old
        for k in range(max(0, i), min(2 * n - 1, i + n)):
            diff = k - i
            if 0 <= diff < n:
                if diff == i:
                    conv_new[k] += delta * (best_x + x_old)
                else:
                    conv_new[k] += 2 * delta * a[diff]
        return a, conv_new, True
    return a, conv, False


def coord_descent_pass(a, max_sweeps=200, tol=1e-10, verbose=False):
    """Full coordinate descent: cycle through all coordinates."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = normalize(a)
    conv = compute_conv(a)
    best_max = np.max(conv)
    best_a = a.copy()

    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            a, conv, imp = coord_descent_1d(a, conv, i)
            if imp:
                improved = True
        # Renormalize after full sweep
        a = normalize(a)
        conv = compute_conv(a)  # Recompute to avoid drift
        cur_max = np.max(conv)
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            if verbose and sweep % 20 == 0:
                print(f"    CD sweep {sweep}: max={best_max:.10f}")
        if not improved and cur_max >= best_max - 1e-12:
            break

    return best_a, best_max


def subgradient_bb(a, max_iter=5000, verbose=False):
    """Subgradient method with Barzilai-Borwein step size."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = normalize(a)
    conv = compute_conv(a)
    best_max = np.max(conv)
    best_a = a.copy()

    a_prev = a.copy()
    grad_prev = None
    y_prev = None
    s_prev = None
    step = 0.1

    for it in range(max_iter):
        conv = compute_conv(a)
        M = np.max(conv)

        if M < best_max - 1e-13:
            best_max = M
            best_a = a.copy()

        # Find subgradient (average over near-active peaks)
        eps = max(1e-8 * M, 1e-12)
        active = np.where(conv >= M - eps)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Gradient of max: average of gradients at active points
        grad = np.zeros(n)
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                # grad[i] += 2*a[k-i] for i in range(i_start, i_end+1)
                grad[i_start:i_end + 1] += 2 * a[k - i_end:k - i_start + 1][::-1]
        grad /= len(active)

        # Center for sum constraint
        grad_c = grad - np.mean(grad_c if False else grad)
        grad_c = grad - np.mean(grad)

        # BB step size
        if grad_prev is not None and s_prev is not None:
            gs = np.dot(grad - grad_prev, s_prev)
            sy = np.dot(s_prev, s_prev)
            if sy > 1e-20:
                bb_step = sy / max(gs, 1e-20)
                if 1e-10 < bb_step < 10:
                    step = 0.5 * step + 0.5 * bb_step

        s_prev = -step * grad_c
        grad_prev = grad.copy()

        # Take step with projection
        a_new = a + s_prev
        a_new = np.maximum(a_new, 0)
        sum_new = np.sum(a_new)
        if sum_new < 1e-10:
            a_new = a.copy()
            sum_new = 1.0
        a_new = a_new / sum_new

        # Line search (backtracking)
        for ls in range(20):
            conv_new = compute_conv(a_new)
            if np.max(conv_new) <= M + 1e-10:
                break
            a_new = a + 0.5 * s_prev
            a_new = np.maximum(a_new, 0)
            sum_new = np.sum(a_new)
            if sum_new < 1e-10:
                break
            a_new = a_new / sum_new
            s_prev = -step * 0.5 * grad_c

        a = a_new

        if verbose and it % 500 == 0:
            print(f"    SBB it={it}: max={best_max:.10f}, step={step:.4f}")

    return best_a, best_max


def active_set_newton(a, max_iter=100, verbose=False):
    """Solve KKT system with active set Newton method."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = normalize(a)
    conv = compute_conv(a)
    best_max = np.max(conv)
    best_a = a.copy()

    for it in range(max_iter):
        conv = compute_conv(a)
        M = np.max(conv)

        if M < best_max - 1e-13:
            best_max = M
            best_a = a.copy()

        # Active set: top n values (or n+1 to have a square system with M)
        n_active = n  # We'll have n conv constraints + 1 sum = n+1 equations, n+1 unknowns
        k_active = np.argsort(conv)[::-1][:n_active]

        # Build system: J @ da - dM * 1 = M - conv[k_active]
        #              1^T @ da = 1 - sum(a)
        # Unknowns: [da (n), dM (1)] total n+1
        # Equations: n + 1 = n+1. Square!

        A = np.zeros((n_active + 1, n + 1))
        b = np.zeros(n_active + 1)

        for idx, k in enumerate(k_active):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                A[idx, i_start:i_end + 1] = 2 * a[k - i_end:k - i_start + 1][::-1]
            A[idx, n] = -1.0
            b[idx] = M - conv[k]

        A[n_active, :n] = 1.0
        b[n_active] = 1.0 - np.sum(a)

        try:
            sol = np.linalg.solve(A, b)
        except np.linalg.LinAlgError:
            try:
                sol, _, _, _ = np.linalg.lstsq(A, b, rcond=1e-10)
            except:
                break
            da = sol[:n]
            dM = sol[n]
        else:
            da = sol[:n]
            dM = sol[n]

        # Damped Newton step
        alpha = 1.0
        a_new = None
        for _ in range(40):
            a_trial = a + alpha * da
            if np.min(a_trial) < 0:
                alpha *= 0.5
                continue
            a_trial = normalize(a_trial)
            conv_trial = compute_conv(a_trial)
            M_trial = np.max(conv_trial)
            if M_trial <= M + 1e-12 * M:
                a_new = a_trial
                break
            alpha *= 0.5

        if a_new is None:
            # Try a small step
            alpha = 1e-4
            a_trial = a + alpha * da
            a_trial = np.maximum(a_trial, 0)
            a_trial = normalize(a_trial)
            conv_trial = compute_conv(a_trial)
            if np.max(conv_trial) <= M + 1e-10:
                a_new = a_trial
            else:
                break

        a = a_new
        new_M = np.max(compute_conv(a))

        if verbose and it % 10 == 0:
            print(f"    AS it={it}: max={best_max:.10f}, alpha={alpha:.2e}")

        if alpha < 1e-10:
            break

    return best_a, best_max


def lp_optimize_fast(a, max_iter=30, t=0.3):
    """LP-based optimization (faster version with time limit)."""
    n = len(a)
    if n > 350:
        return a, np.max(compute_conv(a))

    a = a.copy()
    a = np.maximum(a, 0)
    a = normalize(a)
    conv = compute_conv(a)
    best_max = np.max(conv)
    best_a = a.copy()

    for it in range(max_iter):
        M = np.max(compute_conv(a))

        # Build LP matrix efficiently
        n_vars = n
        n_constr = 2 * n - 1
        A_ub = np.zeros((n_constr, n_vars))
        for k in range(n_constr):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                A_ub[k, i_start:i_end + 1] = a[k - i_end:k - i_start + 1][::-1]

        c = -np.ones(n_vars)
        b_ub = np.full(n_constr, M)
        bounds = [(0.0, None)] * n_vars

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 10, 'presolve': True})
            if res.status == 0:
                g = res.x
                sum_g = np.sum(g)
                if sum_g > 1.0 + 1e-4:
                    g = g / sum_g
                    a_new = (1 - t) * a + t * g
                    a_new = np.maximum(a_new, 0)
                    a_new = normalize(a_new)
                    M_new = np.max(compute_conv(a_new))
                    if M_new < best_max:
                        best_max = M_new
                        best_a = a_new.copy()
                    a = a_new
        except Exception:
            break

    return best_a, best_max


def chirp_init(n, seed, variant=0):
    """Generate chirp-based initialization."""
    rng = np.random.default_rng(seed)
    i = np.arange(n)

    if variant == 0:
        # Quadratic phase (chirp)
        freq = rng.uniform(0.5, 3.0)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * i * i / n + rng.uniform(0, 2 * np.pi))
    elif variant == 1:
        # Modulated Gaussian
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        freq = rng.uniform(0.5, 3.0)
        a = np.exp(-0.5 * ((i - center) / width) ** 2) * (1 + 0.5 * np.cos(2 * np.pi * freq * i / n))
    elif variant == 2:
        # Asymmetric uniform
        a = np.linspace(rng.uniform(0.5, 1.5), rng.uniform(0.1, 0.5), n)
    elif variant == 3:
        # Two-peak
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.2)
        a = np.exp(-0.5 * ((i - c1) / w) ** 2) + np.exp(-0.5 * ((i - c2) / w) ** 2)
    else:
        # Random with smooth structure
        a = rng.random(n)
        # Smooth with a moving average
        k = max(1, n // 20)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def simulated_annealing(a, n_iterations=5000, T_start=0.05, T_end=0.001, seed=42):
    """Simulated annealing with coordinate perturbations."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = a.copy()
    a = normalize(a)
    conv = compute_conv(a)
    best_a = a.copy()
    best_max = np.max(conv)
    current_max = best_max
    current_obj = best_max  # Working with normalized (sum=1), so obj = 2n*max

    for it in range(n_iterations):
        T = T_start * (T_end / T_start) ** (it / n_iterations)

        # Perturb 1-3 random coordinates
        n_perturb = rng.integers(1, 4)
        indices = rng.choice(n, size=n_perturb, replace=False)
        a_trial = a.copy()
        for idx in indices:
            scale = np.exp(rng.normal(0, 0.3))
            a_trial[idx] *= scale
        a_trial = np.maximum(a_trial, 0)
        s = np.sum(a_trial)
        if s < 1e-10:
            continue
        a_trial = a_trial / s

        conv_trial = compute_conv(a_trial)
        new_max = np.max(conv_trial)
        delta = new_max - current_max

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            conv = conv_trial
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_trial.copy()

    return best_a, best_max


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj_val(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for the best sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses multiple strategies: chirp init, coordinate descent, active-set Newton,
    subgradient BB, LP, and simulated annealing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Try parent
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Collect candidate sequences to optimize
    candidates = []
    if parent is not None:
        candidates.append(('parent', parent.copy()))

    # Add chirp and random initializations for various n
    n_values = [50, 80, 120, 150, 200, 250, 300, 400, 500]
    if parent is not None:
        n_parent = len(parent)
        n_values = sorted(set(n_values + [n_parent, max(30, n_parent - 50), min(600, n_parent + 50)]))

    rng = np.random.default_rng(seed)
    for n in n_values:
        if time.monotonic() > deadline - 30:
            break
        for variant in range(3):
            a_init = chirp_init(n, seed + n * 100 + variant, variant)
            candidates.append((f'chirp_v{variant}_n{n}', a_init))

    # Add a few random inits for the parent's length (if exists)
    if parent is not None:
        n_p = len(parent)
        for r in range(3):
            a_rand = rng.random(n_p)
            a_rand = np.maximum(a_rand, 0)
            a_rand = normalize(a_rand)
            candidates.append((f'rand_n{n_p}_{r}', a_rand))

    print(f"Total candidates to optimize: {len(candidates)}")
    print(f"Time budget: {budget_s - (time.monotonic() - start_time):.0f}s")

    # Time allocation per candidate based on length
    for name, a_cand in candidates:
        if time.monotonic() > deadline - 10:
            break

        n = len(a_cand)
        remaining = deadline - time.monotonic()

        # Allocate time based on n
        if n <= 100:
            alloc = min(20.0, remaining * 0.05)
        elif n <= 200:
            alloc = min(40.0, remaining * 0.08)
        elif n <= 350:
            alloc = min(60.0, remaining * 0.1)
        else:
            alloc = min(80.0, remaining * 0.12)
        alloc = max(5.0, alloc)

        t_start = time.monotonic()
        a_cur = a_cand.copy()
        a_cur = np.maximum(a_cur, 0)
        a_cur = normalize(a_cur)
        cur_max = np.max(compute_conv(a_cur))
        if cur_max * 2 * n < best_score:
            best_score = 2 * n * cur_max
            best_seq = a_cur.copy()

        # Phase 1: Subgradient BB (fast, good for initial improvement)
        sg_time = min(alloc * 0.3, 15.0)
        max_iter_sg = int(sg_time * 500)
        a_sg, max_sg = subgradient_bb(a_cur, max_iter=max(100, max_iter_sg))
        obj_sg = 2 * n * max_sg
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  {name}: SBB improved to {best_score:.6f}")
        a_cur = a_sg
        if time.monotonic() - t_start > alloc:
            continue

        # Phase 2: Coordinate descent (exact, slow but thorough)
        elapsed = time.monotonic() - t_start
        cd_time = alloc - elapsed
        if cd_time > 2:
            max_sweeps = int(cd_time * 30)  # ~30 sweeps per second
            a_cd, max_cd = coord_descent_pass(a_cur, max_sweeps=max(10, max_sweeps))
            obj_cd = 2 * n * max_cd
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  {name}: CD improved to {best_score:.6f}")
            a_cur = a_cd
            if time.monotonic() - t_start > alloc:
                continue

        # Phase 3: Active-set Newton (fast convergence near optimum)
        elapsed = time.monotonic() - t_start
        as_time = alloc - elapsed
        if as_time > 1 and n <= 400:
            max_iter_as = int(as_time * 100)
            a_as, max_as = active_set_newton(a_cur, max_iter=max(10, max_iter_as))
            obj_as = 2 * n * max_as
            if obj_as < best_score:
                best_score = obj_as
                best_seq = a_as.copy()
                print(f"  {name}: AS improved to {best_score:.6f}")
            a_cur = a_as

        # Phase 4: Another CD pass on the Newton result
        elapsed = time.monotonic() - t_start
        if alloc - elapsed > 2 and n <= 350:
            a_cd2, max_cd2 = coord_descent_pass(a_cur, max_sweeps=int((alloc - elapsed) * 20))
            obj_cd2 = 2 * n * max_cd2
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  {name}: CD2 improved to {best_score:.6f}")
                a_cur = a_cd2

    # Final intensive polishing on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPolishing best (n={n_best}), {remaining:.0f}s remaining")

        a_cur = best_seq.copy()
        a_cur = normalize(a_cur)

        # Alternating: CD -> AS -> CD -> SA -> final CD
        polish_cycle = 0
        while time.monotonic() < deadline - 10:
            polish_cycle += 1
            rem = deadline - time.monotonic()
            if rem < 5:
                break

            # Coordinate descent
            max_sweeps = int(min(rem * 0.4, 500) * 20)
            a_cd, max_cd = coord_descent_pass(a_cur, max_sweeps=max(10, max_sweeps))
            obj = 2 * n_best * max_cd
            if obj < best_score:
                best_score = obj
                best_seq = a_cd.copy()
                print(f"  Polish {polish_cycle} CD: {best_score:.8f}")
            a_cur = a_cd

            if time.monotonic() > deadline - 10:
                break

            # Active-set Newton
            rem = deadline - time.monotonic()
            if rem > 5 and n_best <= 400:
                a_as, max_as = active_set_newton(a_cur, max_iter=int(min(rem * 0.3, 200) * 50))
                obj = 2 * n_best * max_as
                if obj < best_score:
                    best_score = obj
                    best_seq = a_as.copy()
                    print(f"  Polish {polish_cycle} AS: {best_score:.8f}")
                a_cur = a_as

            if time.monotonic() > deadline - 15:
                break

            # Simulated annealing (only for moderate n)
            rem = deadline - time.monotonic()
            if rem > 20 and n_best <= 300:
                n_sa = int(min(rem * 0.5, 5000))
                a_sa, max_sa = simulated_annealing(a_cur, n_iterations=n_sa, seed=seed + polish_cycle)
                obj = 2 * n_best * max_sa
                if obj < best_score:
                    best_score = obj
                    best_seq = a_sa.copy()
                    print(f"  Polish {polish_cycle} SA: {best_score:.8f}")
                    a_cur = a_sa

            if time.monotonic() > deadline - 10:
                break

            # LP step
            rem = deadline - time.monotonic()
            if rem > 15 and n_best <= 300:
                a_lp, max_lp = lp_optimize_fast(a_cur, max_iter=int(min(rem / 15, 20)))
                obj = 2 * n_best * max_lp
                if obj < best_score:
                    best_score = obj
                    best_seq = a_lp.copy()
                    print(f"  Polish {polish_cycle} LP: {best_score:.8f}")
                    a_cur = a_lp

    # One more try: if we have time, do a final global search with different n
    if best_seq is not None and time.monotonic() < deadline - 30:
        rem = deadline - time.monotonic()
        n_best = len(best_seq)
        # Try a few nearby lengths
        for dn in [-30, 30, -60, 60]:
            n_try = n_best + dn
            if n_try < 20 or n_try > 600:
                continue
            if time.monotonic() > deadline - 10:
                break
            a_try = chirp_init(n_try, seed + 9999 + dn, 0)
            a_try, _ = subgradient_bb(a_try, max_iter=2000)
            a_try, max_try = coord_descent_pass(a_try, max_sweeps=100)
            obj_try = 2 * n_try * max_try
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Final try n={n_try}: {best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj_val(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf and validation < best_score:
        print(f"  Validation gives better: {validation:.8f}")
        best_score = validation
    elif validation != best_score and validation < np.inf:
        print(f"  Validation: computed={best_score:.8f}, validated={validation:.8f}")
        best_score = validation

    return result
```
