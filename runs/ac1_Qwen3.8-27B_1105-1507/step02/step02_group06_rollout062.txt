"""
Hybrid Mirror Descent + LP optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-based fast convolution, adaptive mirror descent with top-k weighting,
LP-based direction finding (both max-sum and min-peak formulations),
adaptive line search, and periodic perturbation to escape local optima.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, hstack as sparse_hstack
from scipy.signal import fftconvolve


def fast_convolve(a, b):
    """FFT-based full convolution."""
    return fftconvolve(a, b, mode='full')


def compute_score(f):
    """Compute the objective value directly."""
    n = len(f)
    conv = fast_convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def compute_conv(f):
    """Compute full convolution of f with itself."""
    return fast_convolve(f, f)


def weighted_subgradient(f, conv, top_k=5):
    """Compute weighted subgradient of max_k conv_k with top-k weighting."""
    n = len(f)
    top_indices = np.argsort(conv)[-top_k:]
    # Weight by excess above median
    median_val = np.median(conv)
    weights = np.maximum(conv[top_indices] - median_val, 0.0)
    if weights.sum() < 1e-15:
        weights = np.ones(top_k) / top_k
    else:
        weights = weights / weights.sum()

    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            grad[js] += w * 2.0 * f[k - js]
    return grad


def mirror_descent(f, deadline, top_k=7, lr_init=0.5, verbose_prefix=""):
    """
    Mirror descent (multiplicative updates) for minimizing max convolution
    subject to sum(f)=1, f>=0.
    """
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-15:
        f = np.ones(len(f)) / len(f)
        S = 1.0
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)

    lr = lr_init
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() < deadline:
        conv = fast_convolve(f, f)
        M = float(np.max(conv))

        grad = weighted_subgradient(f, conv, top_k=top_k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Normalize gradient for scale-invariance
        grad = grad / (M + 1e-15)

        # Mirror descent update: f * exp(-lr * grad)
        update = np.exp(-lr * grad)
        f_new = f * update
        f_sum = f_new.sum()
        if f_sum < 1e-30:
            # Reset
            f = np.ones(n) / n
            lr *= 0.5
            no_improve += 1
            if no_improve > 200:
                break
            it += 1
            continue
        f_new = f_new / f_sum

        score = compute_score(f_new)
        it += 1

        if score < best_score:
            best_f = f_new.copy()
            best_score = score
            no_improve = 0
            if verbose_prefix and it % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    {verbose_prefix}MD iter {it}: score={best_score:.6f} lr={lr:.4f} t={elapsed:.1f}s")
        else:
            no_improve += 1
            if no_improve % 20000 == 0:
                lr *= 0.9
            if lr < 1e-7:
                break

        f = f_new
        # Very slow decay
        lr *= 0.99995

    return best_f, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_max_sum(f, M, A=None):

    n = len(f)

        A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    except Exception:
        pass
    return None


def solve_lp_min_peak(f, S, A=None):

    A_eq = np.zeros((1, n + 1))

    except Exception:

    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    step = step_init
    no_improve = 0
    it = 0
    t0 = time.monotonic()
    M_shrink = 0.0  # starts at 0, grows if LP succeeds

    while time.monotonic() < deadline:
        conv = fast_convolve(f, f)
        M = float(np.max(conv))
        if M <= 0:
            break

        t_start = time.monotonic()

        if use_min_peak:
            A = build_conv_matrix_sparse(f)
            g, t_opt = solve_lp_min_peak(f, S, A)
            if g is not None:
                # g already has sum = S (from the equality constraint)
                g = np.clip(g, 0.0, None)
                gs = g.sum()
                if gs < 1e-15:
                    g = None
                else:
                    g = g * (S / gs)
        else:
            A = build_conv_matrix_sparse(f)
            M_target = M * (1.0 - M_shrink)
            g = solve_lp_max_sum(f, M_target, A)
            if g is not None:
                sum_g = float(np.sum(g))
                if sum_g < 1e-15:
                    g = None
                else:
                    g = g * (S / sum_g)

        lp_time = time.monotonic() - t_start

        if g is None:
            no_improve += 1
            if not use_min_peak:
                M_shrink = max(0, M_shrink - 0.002)
            if no_improve > 15:
                break
            continue

        # Find best blend
        f_new, sc_new = best_blend_alpha(f, g, S, best_score)
        it += 1

        if sc_new < best_score:
            f = f_new
            best_f = f.copy()
            best_score = sc_new
            no_improve = 0
            if not use_min_peak:
                M_shrink = min(0.02, M_shrink + 0.0005)
            if it % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    {verbose_prefix}LP[{use_min_peak}] iter {it}: "
                      f"score={best_score:.6f} lp={lp_time:.2f}s t={elapsed:.1f}s")
        else:
            no_improve += 1
            step *= 0.85
            if not use_min_peak:
                M_shrink = max(0, M_shrink - 0.003)
            if no_improve > 20:
                break

    return best_f, best_score


def make_initializations(n, rng, parent=None):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    if parent is not None:
        if len(parent) == n:
            inits.append(('parent', parent.copy()))
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(('parent_interp', np.interp(x_new, x_old, parent)))

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Gaussian (various widths)
    center = (n - 1) / 2.0
    for sigma_factor in [2.0, 3.0, 4.0]:
        g = np.exp(-0.5 * ((idx - center) / (n / sigma_factor)) ** 2)
        inits.append((f'gauss_{sigma_factor}', g))

    # Bimodal
    bimodal = (np.exp(-0.5 * ((idx - n * 0.25) / (n / 8.0)) ** 2) +
               np.exp(-0.5 * ((idx - n * 0.75) / (n / 8.0)) ** 2))
    inits.append(('bimodal', bimodal))

    # Cosine window
    cos_w = np.clip((1 + np.cos(2 * np.pi * idx / n)) / 2, 0.01, None)
    inits.append(('cosine', cos_w))

    # Raised cosine
    rc = 0.5 * (1 + np.cos(np.pi * (2 * idx / (n - 1) - 1)))
    inits.append(('raised_cos', rc))

    # S-curve
    s_curve = 1.0 / (1.0 + np.exp(-5.0 * (idx / (n - 1) - 0.5)))
    inits.append(('sigmoid', s_curve))

    # Triangular
    tri = 1.0 - np.abs(2.0 * idx / (n - 1) - 1.0)
    tri = np.clip(tri, 0.01, None)
    inits.append(('triangular', tri))

    # Random
    inits.append(('random', rng.uniform(0.05, 1.0, n)))

    # Flat top with tapered edges
    edge = n // 8
    fflat = np.ones(n)
    fflat[:edge] = np.linspace(0.3, 1.0, edge)
    fflat[-edge:] = np.linspace(1.0, 0.3, edge)
    inits.append(('flat_top', fflat))

    return inits


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence minimizing the score."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(2024)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).copy()
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = p
            score = compute_score(p)
            if np.isfinite(score):
                best_score = score
                best_sequence = p.copy()
                print(f"Parent: n={p.size}, score={score:.6f}")
    except Exception as e:
        print(f"No parent available: {e}")

    n = 1024

    # Create initializations
    inits = make_initializations(n, rng, parent)

    # Phase 1: Quick mirror descent on each initialization (first 100s)
    print("\n=== Phase 1: Quick Mirror Descent (100s) ===")
    phase1_deadline = t0 + 100.0
    per_init_time = (phase1_deadline - t0) / len(inits)

    candidates = []
    for label, f_init in inits:
        if time.monotonic() > phase1_deadline - 3:
            break
        f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
        f_sum = f_init.sum()
        if f_sum < 1e-10:
            continue
        f_init = f_init / f_sum
        init_score = compute_score(f_init)

        # Quick mirror descent
        init_deadline = time.monotonic() + per_init_time
        f_opt, s_opt = mirror_descent(f_init, init_deadline, top_k=7, lr_init=0.4)

        if s_opt < init_score:
            init_score = s_opt
            f_init = f_opt

        if init_score < best_score:
            best_score = init_score
            best_sequence = f_init.copy()

        candidates.append((init_score, f_init.copy(), label))
        print(f"  {label}: {init_score:.6f}")

    # Sort candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\nBest after Phase 1: {best_score:.6f}")
    for sc, seq, lbl in candidates[:5]:
        print(f"  {lbl}: {sc:.6f}")

    # Phase 2: LP optimization on top candidates (next ~900s)
    print("\n=== Phase 2: LP Optimization ===")
    phase2_deadline = deadline - 250.0

    # Use the best candidate
    top_f = candidates[0][1].copy()
    top_s = candidates[0][0]

    # Try max-sum LP first
    t_start = time.monotonic()
    f_opt, s_opt = lp_optimize(top_f, min(phase2_deadline, t_start + 500),
                                step_init=0.15, use_min_peak=False,
                                verbose_prefix="[LP-maxsum]")
    if s_opt < best_score:
        best_score = s_opt
        best_sequence = f_opt.copy()
        print(f"  After max-sum LP: {best_score:.6f}")

    # Continue with min-peak LP if time remains
    remaining = phase2_deadline - time.monotonic()
    if remaining > 20:
        f_opt, s_opt = lp_optimize(best_sequence.copy(), time.monotonic() + remaining,
                                    step_init=0.1, use_min_peak=True,
                                    verbose_prefix="[LP-minpeak]")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  After min-peak LP: {best_score:.6f}")

    # Phase 3: Second LP pass with different starting point
    print(f"\n=== Phase 3: Second LP Pass (best={best_score:.6f}) ===")
    if len(candidates) > 1 and phase2_deadline - time.monotonic() > 30:
        second_f = candidates[1][1].copy()
        remaining = phase2_deadline - time.monotonic()
        f_opt, s_opt = lp_optimize(second_f, time.monotonic() + remaining,
                                    step_init=0.12, use_min_peak=False,
                                    verbose_prefix="[LP-2nd]")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  2nd LP passed: {best_score:.6f}")

    # Phase 4: Mirror descent polish (next 100s)
    print(f"\n=== Phase 4: MD Polish (best={best_score:.6f}) ===")
    phase4_deadline = deadline - 100.0
    if time.monotonic() < phase4_deadline:
        f_opt, s_opt = mirror_descent(best_sequence, phase4_deadline,
                                       top_k=11, lr_init=0.15,
                                       verbose_prefix="[MD-polish]")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  MD polish: {best_score:.6f}")

    # Phase 5: Final LP polish (last 100s)
    print(f"\n=== Phase 5: Final LP (best={best_score:.6f}) ===")
    if time.monotonic() < deadline - 10:
        f_opt, s_opt = lp_optimize(best_sequence, deadline,
                                    step_init=0.05, use_min_peak=False,
                                    verbose_prefix="[LP-final]")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  Final LP: {best_score:.6f}")

        # Quick MD at the very end
        remaining_dl = deadline - 2
        if time.monotonic() < remaining_dl - 5:
            f_opt, s_opt = mirror_descent(best_sequence, remaining_dl,
                                           top_k=15, lr_init=0.05)
            if s_opt < best_score:
                best_score = s_opt
                best_sequence = f_opt.copy()
                print(f"  Final MD: {best_score:.6f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(n)

    best_sequence = np.clip(best_sequence.astype(np.float64), 0.0, 1000.0)

    # Ensure sum is reasonable
    s = best_sequence.sum()
    if s < 0.01:
        best_sequence = np.ones(n)
        s = n

    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed! Using fallback.")
        best_sequence = np.ones(n)
        final_score = evaluate_sequence([float(x) for x in best_sequence])

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal_best={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
