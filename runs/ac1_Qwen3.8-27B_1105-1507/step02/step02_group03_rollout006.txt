import numpy as np
from scipy.optimize import linprog
import time

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s

def score(f):
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (f.sum() ** 2)

def build_conv_matrix(f):
    """A @ x = convolve(f, x, mode='full')"""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        for j in range(n):
            # conv(f,x)[k] = sum_j f[k-j] * x[j] where 0<=k-j<n
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    return A

def lp_water_filling(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0"""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    if not res.success:
        return None
    return np.clip(res.x, 0, None)

def iterated_lp(f, n_iters=100, t0=0.3, time_budget=60, rng=None):
    """Iterated LP water filling."""
    if rng is None:
        rng = np.random.default_rng(42)
    f = normalize(f.copy()).astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_s = score(f)
    t = t0
    start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        M = np.convolve(f, f).max()
        g = lp_water_filling(f, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Try blending
        improved = False
        for step in [t, t*0.5, t*2, 0.1]:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_s = score(f_new)
            if new_s < best_s:
                best_f = f_new.copy()
                best_s = new_s
                f = f_new
                t = min(1.0, t * 1.2)
                improved = True
                break
        
        if not improved:
            t *= 0.7
            if t < 0.01:
                # Perturb
                f_pert = f * (1 + 0.05 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if score(f_pert) < best_s:
                    f = f_pert
                    best_f = f.copy()
                    best_s = score(f)
                    t = 0.2
                else:
                    break
    
    return best_f, best_s

def subgradient_descent(f, n_iters=1000, lr0=0.01, time_budget=30, rng=None):
    """Projected subgradient descent on max-convolution."""
    if rng is None:
        rng = np.random.default_rng(42)
    f = normalize(f.copy()).astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_s = score(f)
    start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = np.convolve(f, f)
        k_star = np.argmax(conv)
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            gi = 0
            if 0 <= k_star - i < n:
                gi += f[k_star - i]
            if 0 <= i - k_star < n:
                gi += f[i - k_star]
            grad[i] = gi
        
        # Also consider near-max positions for robustness
        M = conv.max()
        for k in range(len(conv)):
            if conv[k] > 0.95 * M and k != k_star:
                for i in range(n):
                    gi = 0
                    if 0 <= k - i < n:
                        gi += f[k - i]
                    if 0 <= i - k < n:
                        gi += f[i - k]
                    grad[i] += 0.5 * gi
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        lr = lr0 / (1 + 0.001 * it)
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0, None)
        f_new = normalize(f_new)
        
        s = score(f_new)
        if s < best_s:
            best_f = f_new.copy()
            best_s = s
            f = f_new
        # else: keep f (or do a smaller step)
    
    return best_f, best_s

def ftl_approach(f, n_iters=500, time_budget=120, rng=None):

    if rng is None:

    best_f = f.copy()

            break

        M = conv.max()
        
        # Get dual solution (distribution over k's)
        # Solve: min M * sum(y) s.t. sum_k y_k * A_k >= 1, y >= 0
        # where A_k is the k-th row of the conv matrix
        # This is equivalent to: find y such that sum_k y_k * (f*f)[k] / sum_k y_k * M = 1

        # The "expected convolution" is: E_y[(f*f)[k]] = sum_k y_k * (f*f)[k]
        # The gradient of this w.r.t. f is: sum_k y_k * grad_k

        # Actually: sum_k y_k * f[k-i] = (y reflected) correlated with f at position i

Iterated LP water-filling with multiplicative weights, subgradient polishing,
and adaptive coordinate descent for minimizing peak autocorrelation of
non-negative sequences. Focuses on longer LP runs, better step-size
scheduling, and diverse initialization to escape local minima.

import numpy as np
from scipy.optimize import linprog
import time


def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (f.sum() ** 2)


def compute_score_fast(f):
    """Score assuming f is already normalized (sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build matrix A (shape (2n-1, n)) such that A @ x = np.convolve(f, x, mode='full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def build_conv_matrix_fast(f):
    """Faster construction using stride tricks."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        A[k, j_start:j_end] = f[k - j_start:k - j_end + 1]
    return A


def lp_water_filling(f, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_conv_matrix_fast(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def water_filling(f, time_budget, t0=0.4, rng=None, n_max_iters=500, lp_tl=5):
    """Iterated LP water filling with adaptive step size."""
    if rng is None:
        rng = np.random.default_rng(42)
    f = normalize(f.copy()).astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_s = compute_score_fast(f)
    t = t0
    start = time.monotonic()
    stall = 0

    for it in range(n_max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        g = lp_water_filling(f, M, time_limit=lp_tl)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        improved = False
        # Line search over step sizes
        for step in [min(1.0, t * 1.5), t, t * 0.5, t * 0.25, 0.05]:

                continue

            f_new = np.clip(f_new, 0, None)

                continue

                best_s = new_s

def multiplicative_weights(f, time_budget, eta0=0.3, rng=None):
    """Fast multiplicative weights: reduce elements contributing to convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(42)
    f = normalize(f.copy()).astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_s = compute_score_fast(f)
    start = time.monotonic()
    stall = 0

    it = 0
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        # Gradient: weighted sum of subgradients at active positions
        threshold = 0.85 * M
        active = np.where(conv > threshold)[0]
        if len(active) == 0:
            break

        # Compute gradient vector
        grad = np.zeros(n)
        for k in active:
            w = 1.0 / len(active)
            # First part: f[k-i] for valid i
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                # For i in [i_start, i_end), grad[i] += w * f[k-i]
                idx = k - np.arange(i_start, i_end)
                grad[i_start:i_end] += w * f[idx]
            # Second part: f[i-k] for valid i
            j_start = max(0, k)
            j_end = min(n, k + n)
            if j_start < j_end:
                idx = np.arange(j_start, j_end) - k
                grad[j_start:j_end] += w * f[idx]

        if grad.max() < 1e-15:
            break

        eta = eta0 / (1.0 + 0.005 * it)
        f_new = f * np.exp(-eta * grad)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        new_s = compute_score_fast(f_new)
        if new_s < best_s:
            best_f = f_new.copy()
            best_s = new_s
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Try accepting anyway (simulated annealing style)
                if rng.random() < 0.1:
                    f = f_new
                    stall = 0
                else:
                    break

        it += 1

    return best_f, best_s


def subgradient_polish(f, time_budget, lr0=0.02, rng=None):
    """Projected subgradient descent with multiple active peaks."""
    if rng is None:
        rng = np.random.default_rng(42)
    f = normalize(f.copy()).astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_s = compute_score_fast(f)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        # Multiple active peaks for robustness
        active = np.where(conv > 0.9 * M)[0]
        weights = np.full(len(active), 1.0 / len(active))

        grad = np.zeros(n)
        for idx_k in active:
            k = int(idx_k)
            w = weights[int(np.where(active == k)[0][0])] if len(active) > 1 else 1.0
            # Actually just use uniform
            w = 1.0 / len(active)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = k - np.arange(i_start, i_end)
                grad[i_start:i_end] += w * f[idx]
            j_start = max(0, k)
            j_end = min(n, k + n)
            if j_start < j_end:
                idx = np.arange(j_start, j_end) - k
                grad[j_start:j_end] += w * f[idx]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        lr = lr0 / (1.0 + 0.002 * it)
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0, None)
        sn = f_new.sum()
        if sn < 1e-10:
            break
        f_new = f_new / sn

        new_s = compute_score_fast(f_new)
        if new_s < best_s:
            best_f = f_new.copy()
            best_s = new_s
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break

        it += 1

    return best_f, best_s


def coordinate_descent(f, time_budget, rng=None):
    """Smart coordinate descent focusing on peak-contributing elements."""
    if rng is None:
        rng = np.random.default_rng(42)
    f = normalize(f.copy()).astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_s = compute_score_fast(f)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        j_star = int(np.argmax(conv))

        # Elements that most contribute to the peak: f[j_star - i] for valid i
        contrib = np.zeros(n)
        i_start = max(0, j_star - n + 1)
        i_end = min(n, j_star + 1)
        if i_start < i_end:
            idx = j_star - np.arange(i_start, i_end)
            contrib[i_start:i_end] = f[idx]

        # Process elements in order of contribution (highest first)
        order = np.argsort(-contrib)
        improved_this_sweep = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break
            original = f[i]
            if original < 1e-12:
                continue
            best_val = original
            best_local = best_s

            # Geometric search around current value
            for factor in [0.2, 0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 5.0]:
                f[i] = original * factor
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                sc = compute_score_fast(f / s)
                if sc < best_local:
                    best_local = sc
                    best_val = f[i]

            f[i] = best_val
            if best_val != original:
                improved_this_sweep = True
                f = normalize(f)
                if best_local < best_s:
                    best_s = best_local
                    best_f = f.copy()

        if not improved_this_sweep:
            # Try random perturbations
            for _ in range(5):
                i = int(rng.integers(0, n))
                original = f[i]
                f[i] = original * rng.uniform(0.3, 3.0)
                f = normalize(f)
                sc = compute_score_fast(f)
                if sc < best_s:
                    best_s = sc
                    best_f = f.copy()
                else:
                    f[i] = original
                    f = normalize(f)
            break

    return best_f, best_s


def spectral_init(n, rng):
    """Generate initial sequence with random phases in frequency domain."""
    n_fft = 2 * n
    F = np.zeros(n_fft // 2 + 1, dtype=complex)
    F[0] = np.sqrt(n)  # DC
    for k in range(1, n_fft // 2 + 1):
        F[k] = rng.uniform(0.5, 1.5) * np.exp(1j * rng.uniform(0, 2 * np.pi))
    f = np.real(np.fft.irfft(F, n_fft))[:n]
    f = np.clip(f, 0.05, None)
    f = normalize(f)
    return f


def gaussian_init(n, rng):
    """Gaussian bump initialization."""
    center = n * rng.uniform(0.3, 0.7)
    width = n * rng.uniform(0.15, 0.4)
    f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    f = np.clip(f, 0.01, None)
    f = f * (1 + 0.1 * rng.normal(0, 1, n))
    f = np.clip(f, 0.01, None)
    return normalize(f)


def cosine_init(n, rng):
    """Cosine window initialization."""
    f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    f = f * (1 + 0.05 * rng.normal(0, 1, n))
    f = np.clip(f, 0.01, None)
    return normalize(f)


def diverse_init(n, rng, init_type=None):
    if init_type is None:
        init_type = int(rng.integers(0, 3))
    if init_type == 0:
        return spectral_init(n, rng)
    elif init_type == 1:
        return gaussian_init(n, rng)
    else:
        return cosine_init(n, rng)
