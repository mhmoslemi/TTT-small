def tukey_window(n, r=0.3):
    """Generate a Tukey window."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def asp_window(n, alpha=1.0):
    """Asymmetric power window: w[i] = ((i+1)/n)^alpha * (1 - i/n)^alpha * 2 or similar."""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    w = np.power(x + 0.5, alpha)
    return w


def exp_decay_window(n, rate=1.0):
    """Exponentially decaying window."""
    x = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * x / n)
    return w / w.sum()


def random_smooth(n, seed=42):
    """Generate a random smooth non-negative sequence."""
    rng = np.random.default_rng(seed)
    x = rng.standard_normal(n)
    kernel_size = max(3, n // 50)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size / 3.0)) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(np.abs(x), kernel, mode='same')
    smooth = smooth + 0.1
    return smooth / smooth.sum()

    # Phase 2: LP refinement (30% of time)
    p2_deadline = t0 + total * 0.70
    if n <= 1024 and time.monotonic() < p2_deadline:
        f2, s2 = lp_refine(best_f, p2_deadline,
                           step_sizes=[0.03, 0.05, 0.1, 0.15, 0.2],
                           max_rounds=50,
                           verbose_prefix=verbose_prefix + "P2")
        if s2 < best_score:
            best_f, best_score = f2, s2

    # Phase 3: High-beta subgradient polish (20% of time)
    p3_deadline = t0 + total * 0.90
    if time.monotonic() < p3_deadline:
        f3, s3 = subgradient_optimize(best_f, p3_deadline,
                                       beta_start=100.0, beta_end=500.0,
                                       lr_start=0.01, lp_every=99999,
                                       use_lp=False,
                                       verbose_prefix=verbose_prefix + "P3")
        if s3 < best_score:
            best_f, best_score = f3, s3

    # Phase 4: Final LP (10% of time)
    if n <= 1024 and time.monotonic() < deadline - 2:
        f4, s4 = lp_refine(best_f, deadline - 2,
                           step_sizes=[0.02, 0.03, 0.05, 0.08],
                           max_rounds=20,
                           verbose_prefix=verbose_prefix + "P4")
        if s4 < best_score:
            best_f, best_score = f4, s4

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """Main search function: tries multiple lengths and initializations."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Try parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Define candidate configurations: (n, time_fraction, list_of_init_functions)
    # Focus on n=512 (parent) and n=1024 (potential improvement)
    total_budget = deadline - t0

    # Time allocation:
    # n=512 from parent: 35%
    # n=512 from other inits: 15%
    # n=1024: 25%
    # n=768: 10%
    # n=2048: 5% (quick check)
    # Perturbation restarts: 10%

    configs = []

    # n=512 from parent
    if parent is not None and len(parent) == 512:
        configs.append((512, 0.35, [('parent', parent.copy())]))

    # n=512 from scratch (diverse inits)
    n = 512
    inits_512 = []
    inits_512.append(('tukey_0.2', tukey_window(n, 0.2)))
    inits_512.append(('tukey_0.4', tukey_window(n, 0.4)))
    inits_512.append(('exp_decay', exp_decay_window(n, 1.5)))
    inits_512.append(('rand_smooth_1', random_smooth(n, seed=7)))
    if parent is not None and len(parent) != 512:
        from numpy import interp as np_interp
        pi = np_interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        inits_512.insert(0, ('parent_interp', np.clip(pi, 0, None)))
    configs.append((512, 0.15, inits_512))

    # n=1024
    n = 1024
    inits_1024 = []
    inits_1024.append(('tukey_0.3', tukey_window(n, 0.3)))
    inits_1024.append(('tukey_0.2', tukey_window(n, 0.2)))
    inits_1024.append(('exp_decay', exp_decay_window(n, 1.0)))
    inits_1024.append(('rand_smooth_1', random_smooth(n, seed=13)))
    inits_1024.append(('rand_smooth_2', random_smooth(n, seed=99)))
    if parent is not None:
        from numpy import interp as np_interp
        pi = np_interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        inits_1024.insert(0, ('parent_interp', np.clip(pi, 0, None)))
    configs.append((1024, 0.25, inits_1024))

    # n=768
    n = 768
    inits_768 = []
    inits_768.append(('tukey_0.3', tukey_window(n, 0.3)))
    inits_768.append(('exp_decay', exp_decay_window(n, 1.2)))
    inits_768.append(('rand_smooth_1', random_smooth(n, seed=23)))
    configs.append((768, 0.10, inits_768))

    # n=2048 (quick)
    n = 2048
    inits_2048 = [('tukey_0.3', tukey_window(n, 0.3)),
                  ('rand_smooth_1', random_smooth(n, seed=55))]
    configs.append((2048, 0.05, inits_2048))

    # Run each configuration
    for n, time_frac, inits in configs:
        if time.monotonic() > deadline - 30:
            print("Time running out, stopping.")
            break

        n_budget = total_budget * time_frac
        n_deadline = time.monotonic() + n_budget
        print(f"\n=== n={n}, budget={n_budget:.0f}s, {len(inits)} inits ===")

        if len(inits) == 0:
            continue

        # If we have a good score already, only pursue inits that might beat it
        per_init_time = n_budget / max(len(inits), 1) * 0.85

        for label, f_init in inits:
            if time.monotonic() > n_deadline:
                break

            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue
            f_init = f_init / S

            init_score = evaluate_sequence(f_init.tolist())
            if not np.isfinite(init_score):
                continue

            if init_score < best_score:
                best_score = init_score
                best_sequence = f_init.copy()
                print(f"  [{n} {label}] init score: {init_score:.6f}")

            # Decide whether to optimize this init
            # Only optimize if it's promising (within 5% of best) or it's the only init
            if init_score < best_score * 1.08 or len(inits) <= 2:
                init_deadline = min(n_deadline, time.monotonic() + per_init_time)
                if time.monotonic() < init_deadline - 5:
                    try:
                        bf, bs = combine_optimization(f_init, init_deadline,
                                                       verbose_prefix=f"[{n}/{label}] ")
                        if bs < best_score:
                            best_score = bs
                            best_sequence = bf.copy()
                            print(f"  [{n} {label}] improved to {bs:.6f}")
                        else:
                            print(f"  [{n} {label}] best was {bs:.6f}")
                    except Exception as e:
                        print(f"  [{n} {label}] error: {e}")
            else:
                print(f"  [{n} {label}] init {init_score:.6f}, skipping (not promising)")

    # Perturbation restarts on best
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Perturbation restarts (best: {best_score:.6f}, n={len(best_sequence)}) ===")
        restart_deadline = deadline - 10
        restart_budget = restart_deadline - time.monotonic()
        n_restarts = min(6, max(2, int(restart_budget / 40)))
        per_restart = restart_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > restart_deadline:
                break
            strength = 0.02 + 0.02 * r
            f_pert = perturb(best_sequence, rng, strength=strength)
            r_deadline = time.monotonic() + per_restart * 0.75

            try:
                bf, bs = combine_optimization(f_pert, r_deadline,
                                               verbose_prefix=f"[restart{r}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strength:.3f}): {bs:.6f} IMPROVED")
                else:
                    print(f"  Restart {r} (str={strength:.3f}): {bs:.6f} (no improvement)")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # Final polish on absolute best
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Final polish (n={len(best_sequence)}, score={best_score:.6f}) ===")
        try:
            # High-beta subgradient polish
            pf, ps = subgradient_optimize(best_sequence, deadline - 3,
                                           beta_start=200.0, beta_end=800.0,
                                           lr_start=0.005, lp_every=99999,
                                           use_lp=False,
                                           verbose_prefix="[polish] ")
            if ps < best_score:
                best_score = ps
                best_sequence = pf
                print(f"  Polish: {best_score:.6f}")

            # Quick LP if n is small enough
            if len(best_sequence) <= 1024:
                pf, ps = lp_refine(best_sequence, deadline - 2,
                                    step_sizes=[0.01, 0.02, 0.03, 0.05],
                                    max_rounds=10,
                                    verbose_prefix="[lp_final] ")
                if ps < best_score:
                    best_score = ps
                    best_sequence = pf
                    print(f"  Final LP: {best_score:.6f}")
        except Exception as e:
            print(f"  Final polish error: {e}")

    # Ensure valid output
    if best_sequence is None:
        best_sequence = tukey_window(512, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())
        print(f"  Fallback: {best_score:.6f}")

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
