import time

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def subgradient_of_max_conv(a):
    """Compute subgradient of max(conv(a,a)) with respect to a."""
    n = len(a)
    b = fast_conv(a)
    k_star = np.argmax(b)
    # d(b_k)/da_i = a_{k-i} for valid i
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = a[j]
    return grad, k_star, b.max()

    n = len(a)
    m = 2 * n - 1
    # Build convolution matrix
    A = np.zeros((m, n))

    return None

def line_search(a, g, best_score):

    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)

    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def fast_cross_conv(a, b):

    n = max(len(a), len(b))
    m = len(a) + len(b) - 1
    s = 1

def build_conv_matrix(a):
    """Build matrix where A @ x = convolve(a, x) (full linear)."""
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh+1] = a[k-jh:k-jl+1][::-1]
    return A

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    m = 2*n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(m, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        return res.x
    return None

        if M < 1e-15:
            break
        g = lp_solve(a, 1.0)
        if g is None:
            break
        g = g / g.sum()
        t, sc = line_search(a, g, best_score)
        if t > 1e-7 and sc < best_score - 1e-12:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            sv = a_new.sum()
            if sv < 1e-10:
                break
            a_new = a_new / sv
            actual = compute_score(a_new)
            if actual < best_score:
                best_score = actual
                best = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 40:
            break
        if best_score < 1.497:
            break

    return best, best_score


def make_init(n, rng, kind, parent=None):
    """Generate a valid non-negative initialization of length n."""
    t = np.arange(n) / n
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'cos1':
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t)
    elif kind == 'cos2':
        a = 1.0 + 0.3 * np.cos(4 * np.pi * t)
    elif kind == 'cos3':
        a = 1.0 + 0.25 * np.cos(6 * np.pi * t) + 0.15 * np.cos(4 * np.pi * t)
    elif kind == 'cos4':
        a = 1.0 + 0.2 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t) + 0.1 * np.cos(8 * np.pi * t)
    elif kind == 'bimodal':
        a = np.exp(-30 * (t - 0.25) ** 2) + np.exp(-30 * (t - 0.75) ** 2) + 0.05
    elif kind == 'trimodal':
        a = np.exp(-40 * (t - 0.2) ** 2) + np.exp(-40 * (t - 0.5) ** 2) + np.exp(-40 * (t - 0.8) ** 2) + 0.03
    elif kind == 'rand':
        a = rng.uniform(0.3, 1.0, n)
    elif kind == 'rand_sparse':
        a = rng.uniform(0.0, 1.0, n)
        a = np.maximum(a, 0.01)
    elif kind == 'spectral':
        a = rng.standard_normal(n)
        for it in range(50):
            A = np.fft.rfft(a)
            pw = np.abs(A) ** 2
            avg = pw.mean()
            pw_new = 0.4 * pw + 0.6 * avg
            mag = np.sqrt(np.maximum(pw_new, 1e-10))
            ph = A / (np.abs(A) + 1e-10)
            a = np.fft.irfft(mag * ph, n)
            a = np.maximum(a, 0.0) + 0.05
    elif kind == 'spectral2':
        a = rng.standard_normal(n)
        for it in range(30):
            A = np.fft.rfft(a)
            pw = np.abs(A) ** 2
            # More aggressive flattening: power-law
            pw_norm = pw / (pw.max() + 1e-10)
            pw_new = pw ** 0.5  # Compress dynamic range
            mag = np.sqrt(np.maximum(pw_new, 1e-10))
            ph = A / (np.abs(A) + 1e-10)
            a = np.fft.irfft(mag * ph, n)
            a = np.maximum(a, 0.0) + 0.05
    elif kind == 'parent':
        if parent is not None and parent.size > 1:
            if parent.size == n:
                a = parent.copy()
            else:
                t_old = np.linspace(0, 1, parent.size)
                t_new = np.linspace(0, 1, n)
                a = np.maximum(np.interp(t_new, t_old, parent), 0.0)
            if a.sum() < 1e-10:
                a = np.ones(n)
            return a / a.sum()
        a = np.ones(n)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def lse_refine(a0, deadline, betas=None):
    """
    Refine using log-sum-exp smoothed objective with Adam optimizer.
    Minimizes 2n * LSE_beta(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    if betas is None:
        betas = [30, 100, 500, 2000, 10000]

    n = len(a0)
    best = a0.copy()
    best_score = compute_score(a0)

    for beta in betas:
        if time.monotonic() > deadline:
            break

        x = best.copy()
        x = np.clip(x, 1e-8, None)
        x = x / x.sum()

        # Adam
        m_adv = np.zeros(n)
        v_adv = np.zeros(n)
        b1, b2, eps = 0.75, 0.95, 1e-8
        lr = 0.03
        step = 0
        best_local = x.copy()
        best_local_score = best_score
        no_improve = 0

        max_steps = 300
        for step in range(max_steps):
            if time.monotonic() > deadline:
                break
            step += 1

            x_pos = np.maximum(x, 0.0)
            sv = x_pos.sum()
            if sv < 0.01:
                break
            x_norm = x_pos / sv

            b = fast_conv(x_norm)
            b_max = b.max()
            shifted = beta * (b - b_max)
            shifted = np.clip(shifted, -50.0, 50.0)
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum < 1e-30:
                break
            w = w / w_sum

            # Gradient: grad[j] = sum_k w[k] * a[k-j] for valid k-j
            # = sum_{i=0}^{n-1} w[j+i] * a[i]
            # Compute via FFT cross-correlation
            size = 1
            while size < 3 * n:
                size <<= 1
            fa_ft = np.fft.rfft(x_norm, size)
            fw_ft = np.fft.rfft(w, size)
            # We need: grad[j] = sum_i x[i] * w[j+i]
            # = (x rev * w)[j + n - 1] via linear convolution
            x_rev = x_norm[::-1]
            fx = np.fft.rfft(x_rev, size)
            full = np.fft.irfft(fx * fw_ft, size)
            grad = full[n - 1:n - 1 + n]

            # Adam update
            m_adv = b1 * m_adv + (1 - b1) * grad
            v_adv = b2 * v_adv + (1 - b2) * grad ** 2
            m_hat = m_adv / (1 - b1 ** step)
            v_hat = v_adv / (1 - b2 ** step)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
            x = np.maximum(x, 0.0)
            sv2 = x.sum()
            if sv2 < 0.01:
                x = best_local.copy()
                break
            x = x / sv2

            sc = compute_score(x)
            if sc < best_local_score:
                best_local_score = sc
                best_local = x.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 80:
                break

        if best_local_score < best_score:
            best_score = best_local_score
            best = best_local.copy()

    return best, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-phase LP optimization with exact line search,
    spectral initialization, smooth refinement, and perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    total = float(budget_s)
    deadline = time.monotonic() + total - 20.0

    best = None
    best_score = float('inf')

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            sc = compute_score(p)
            if sc < best_score:
                best, best_score = p.copy(), sc
                print(f"Parent: J={sc:.6f} n={p.size}", flush=True)
    except (NameError, AttributeError):
        pass

    # === PHASE 1: Multi-start exploration (40% time) ===
    p1_frac = 0.40
    p1_deadline = time.monotonic() + (deadline - time.monotonic()) * p1_frac

    # Determine lengths to try
    if best is not None:
        lengths = [best.size]
    else:
        lengths = [512]
    for l in [1024, 256, 768]:
        if l not in lengths:
            lengths.append(l)

    kinds_list = ['parent', 'uniform', 'cos1', 'cos2', 'spectral', 'cos3', 'rand', 'spectral2', 'bimodal', 'cos4', 'trimodal', 'rand_sparse']

    print(f"\n=== Phase 1: Multi-start LP (deadline in {p1_deadline - time.monotonic():.0f}s) ===", flush=True)

    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        remaining = p1_deadline - time.monotonic()
        if remaining < 30:
            break

        # Estimate time per LP iteration
        est_iter = 0.04 if n <= 512 else 0.12 if n <= 768 else 0.25
        n_starts = max(2, min(len(kinds_list), int(remaining / (est_iter * 200))))

        print(f"  n={n}, trying {n_starts} initializations", flush=True)

        for i in range(n_starts):
            if time.monotonic() > p1_deadline:
                break
            kind = kinds_list[i % len(kinds_list)]
            a0 = make_init(n, rng, kind, best)
            if a0 is None or a0.sum() < 0.01:
                continue

            rem = p1_deadline - time.monotonic()
            max_it = max(100, min(500, int(rem / (est_iter * 3))))

            res, sc = lp_optimize(a0, max_it, p1_deadline)
            if sc < best_score:
                best, best_score = res.copy(), sc
                print(f"    NEW BEST P1: J={sc:.6f} (n={n}, kind={kind})", flush=True)

    print(f"  Phase 1 done. Best: J={best_score:.6f}", flush=True)

    # === PHASE 2: Long LP exploitation (35% time) ===
    p2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.45

    if best is not None:
        print(f"\n=== Phase 2: Long LP (n={best.size}, time={p2_deadline - time.monotonic():.0f}s) ===", flush=True)
        res, sc = lp_optimize(best, 100000, p2_deadline)
        if sc < best_score:
            best, best_score = res.copy(), sc
            print(f"    NEW BEST P2: J={sc:.6f}", flush=True)
        else:
            print(f"    P2 no improvement. Best: J={best_score:.6f}", flush=True)

    # === PHASE 3: Smooth refinement (10% time) ===
    p3_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.30

    if best is not None and time.monotonic() < p3_deadline:
        print(f"\n=== Phase 3: Smooth refinement ===", flush=True)
        res, sc = lse_refine(best, p3_deadline)
        if sc < best_score:
            best, best_score = res.copy(), sc
            print(f"    NEW BEST P3: J={sc:.6f}", flush=True)
        else:
            print(f"    P3 no improvement. Best: J={best_score:.6f}", flush=True)

    # === PHASE 4: Perturbation + LP restarts (remaining time) ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Phase 4: Perturbation search ===", flush=True)
        n = best.size
        est_iter = 0.04 if n <= 512 else 0.12
        scales = [0.03, 0.08, 0.12, 0.05, 0.15, 0.02, 0.2, 0.1, 0.04, 0.06]
        si = 0
        p4_best = best_score

        while time.monotonic() < deadline:
            scale = scales[si % len(scales)]
            si += 1
            noise = rng.normal(0, scale, n)
            ap = best + noise
            ap = np.clip(ap, 0, None)
            if ap.sum() < 0.01:
                continue
            ap = ap / ap.sum()

            rem = deadline - time.monotonic()
            if rem < 10:
                break
            max_it = max(50, min(2000, int(rem / (est_iter * 3 * (si % 5 + 1)))))

            res, sc = lp_optimize(ap, max_it, deadline)
            if sc < best_score:
                best, best_score = res.copy(), sc
                p4_best = best_score
                print(f"    NEW BEST P4: J={sc:.6f} (scale={scale:.3f}, iter={si})", flush=True)

        # If we improved in phase 4, do one more long polish
        if best_score < p4_best and time.monotonic() < deadline:
            rem = deadline - time.monotonic()
            if rem > 60:
                res, sc = lp_optimize(best, int(rem / est_iter), deadline)
                if sc < best_score:
                    best, best_score = res.copy(), sc
                    print(f"    POLISH: J={sc:.6f}", flush=True)

    # Final validation
    if best is None:
        best = np.ones(256) / 256
        best_score = compute_score(best)

    # Ensure values are valid for evaluate_sequence
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(256) / 256

    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: J={best_score:.6f} n={best.size}", flush=True)
    return [float(x) for x in best]
