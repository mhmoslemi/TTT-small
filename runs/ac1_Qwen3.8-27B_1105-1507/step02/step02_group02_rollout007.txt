import time

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)

    """Build matrix A where (A @ g_free) = conv(a, g_sym), g_sym symmetric extension of g_free."""
    n = len(a)
    m = 2 * n - 1
    half = n // 2 + 1
    A = np.zeros((m, half))

    half = n // 2 + 1

                  bounds=[(0, None)] * half, method='highs')

    # Symmetrize
    a = (a + a[::-1]) / 2
    a = a / a.sum()
    
    no_improve_count = 0
    last_best = best_score
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g, M = lp_best_response_sym(a)
        if g is None:
            break
        g = g / g.sum()
        
        # Backtracking line search
        best_t = 0.0
        best_a_t = a.copy()
        best_score_t = best_score
        t = 1.0
        for ls in range(10):
            a_candidate = (1 - t) * a + t * g
            a_candidate = np.clip(a_candidate, 0, None)
            a_candidate = (a_candidate + a_candidate[::-1]) / 2  # symmetrize
            s = a_candidate.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_candidate = a_candidate / s
            score = compute_score(a_candidate)
            if score < best_score_t:
                best_score_t = score
                best_t = t
                best_a_t = a_candidate.copy()
                if t >= 0.9:  # close to full step
                    break
            t *= 0.5
        
        if best_t > 0:
            a = best_a_t
            if best_score_t < best_score:
                best_score = best_score_t
                best = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            # Try a small fixed step
            t = 0.01
            a_candidate = (1 - t) * a + t * g
            a_candidate = np.clip(a_candidate, 0, None)
            a_candidate = (a_candidate + a_candidate[::-1]) / 2
            a_candidate = a_candidate / a_candidate.sum()
            score = compute_score(a_candidate)
            if score < best_score:
                best_score = score
                best = a_candidate.copy()
                no_improve_count = 0
            a = a_candidate
        
        # Check for convergence
        if it > 50 and no_improve_count > 100:
            break
        if it > 200 and (last_best - best_score) < 1e-10:
            break
        last_best = best_score
    
    return best, best_score

def smooth_grad(x, beta, n):
    """Compute gradient of smoothed objective 2n*LSE(beta*conv(x,x)) for normalized x."""
    x_pos = np.maximum(x, 1e-15)
    s = x_pos.sum()
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50, 50)
    p = np.exp(shifted)
    p = p / p.sum()
    
    # Gradient of LSE w.r.t. b is p * beta
    # Gradient of b w.r.t. x: db[k]/dx[j] = x_norm[k-j] + x_norm[j-k] (with bounds)
    # = (x_norm * x_norm)[j] in cross-correlation sense
    
    # grad_obj[j] = 2n * beta * sum_k p[k] * (x_norm[k-j] + x_norm[j-k] if valid)
    #             = 2n * beta * (p * x_norm)[j] where * is cross-correlation
    
    # Cross-correlation: (p ⋆ x)[j] = sum_k p[k] * x[k-j] (with bounds)
    # This is the same as conv(p_flipped, x) where p_flipped = p[::-1]
    # Or equivalently: fft_inv(fft(p) * conj(fft(x)))
    
    s_len = 1
    while s_len < 2 * n - 1:
        s_len <<= 1
    P = np.fft.rfft(p, s_len)
    X = np.fft.rfft(x_norm, s_len)
    
    # Cross-correlation of p and x: sum_k p[k]*x[k-j]
    # = ifft(conj(P) * X) ... let me verify
    # Actually: ifft(fft(p) * conj(fft(x)))[j] = sum_k p[k] * x[k-j] (circular)
    # For linear: we need zero-padding, which we have (s_len >= 2n-1)
    
    cross1 = np.fft.irfft(P * np.conj(X), s_len)[:n]  # sum_k p[k]*x[k-j]
    cross2 = np.fft.irfft(np.conj(P) * X, s_len)[:n]  # sum_k p[k]*x[j-k] (might be different)
    
    # Actually for real signals: cross1[j] = sum_k p[k]*x[k-j] and cross2[j] = sum_k p[k]*x[j-k]
    # cross2[j] = cross1[-j] = cross1[n-j] (circular)... but we want linear
    # Let me just compute: grad[j] = 2n * beta * (cross1[j] + cross2[j]) / s^2
    # Wait, the 1/s^2 comes from the normalization x_norm = x/s, and the chain rule
    
    # Let me be more careful. We're optimizing f(x) = 2n * LSE(beta * conv(x/s, x/s)) where s = sum(x)
    # = 2n / s^2 * LSE(beta * conv(x,x) / s^2)
    # = 2n / s^2 * LSE((beta/s^2) * conv(x,x))
    
    # Hmm, this is getting complicated. Let me just optimize over the normalized x directly.
    # i.e., we parameterize a = x / sum(x) and optimize over x (with the constraint sum(x)=1 built in by projection).
    
    # For now, let me compute the gradient of f(a) = 2n * LSE(beta * conv(a,a)) w.r.t. a
    # where a is already normalized.
    
    grad = 2.0 * n * beta * (cross1 + cross2)
    
    return grad / s  # chain rule for normalization... actually let me just return grad and handle normalization separately
