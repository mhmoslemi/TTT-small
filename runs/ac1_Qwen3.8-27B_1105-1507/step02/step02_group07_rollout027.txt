"""
Coordinate descent + subgradient + LP hybrid optimizer for minimizing
2n * max(conv(f,f)) / sum(f)^2 over non-negative sequences.
Uses exact 1D sub-optimization per coordinate, adaptive step sizes,
and multiple initialization strategies.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Compute the objective value without calling evaluate_sequence (faster)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_conv = np.max(conv)
    sum_f = np.sum(f)
    if sum_f < 0.01:
        return np.inf
    return float(2 * n * max_conv / (sum_f ** 2))


def coordinate_descent(f, deadline, max_passes=100, order='natural', verbose_prefix=""):
    """
    Coordinate descent: for each coordinate i, find optimal f[i] given others fixed.
    """
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S  # normalize to sum=1
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    
    if order == 'random':
        rng = np.random.default_rng(42)
    
    pass_num = 0
    no_improve = 0
    
    while time.monotonic() < deadline and pass_num < max_passes:
        pass_num += 1
        improved = False
        order = np.arange(n) if order == 'natural' else rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                    break
            
            fi = f[i]
            # Compute the 1D optimization
            # Conv values affected by f[i]: k from i to i+n-1 (where 0<=k-i<n)
            # conv[k] = A[k] + 2*f[k-i]*x for k != 2i
            # conv[2i] = A[2i] + x^2 if 2i < 2n-1
            
            # Compute current convolution
            conv = np.convolve(f, f)
            max_conv = np.max(conv)
            
            # Coefficients for affected k values
            # For k in [i, i+n-1] (clipped to [0, 2n-2]):
            k_start = i  # when k-i=0
            k_end = min(i + n - 1, 2 * n - 2)  # when k-i=n-1
            
            # A[k] = conv[k] - contribution from f[i]
            # Contribution: 2*f[k-i]*fi (or fi^2 if k=2i)
            ks = np.arange(k_start, k_end + 1)
            ks_valid = ks[(ks >= 0) & (ks < 2 * n - 1)]
            
            # For each valid k, compute the coefficient of x and x^2
            # b[k] = coefficient of x in conv[k](x)
            # c[k] = coefficient of x^2 in conv[k](x)
            # conv[k](x) = conv[k](fi) - 2*f[k-i]*fi + b[k]*x + c[k]*x^2
            # where b[k] = 2*f[k-i] (if k!=2i) or 0 (if k=2i)
            # c[k] = 1 (if k==2i) or 0
            # So conv[k](x) = A[k] + b[k]*x + c[k]*x^2
            # where A[k] = conv[k](fi) - b[k]*fi - c[k]*fi^2
            
            # Let's just compute directly:
            # conv[k](x) with f[i]=x:
            # = conv[k](fi) - contribution_old + contribution_new
            # contribution_old = 2*f[k-i]*fi (if k!=2i and valid) or fi^2 (if k==2i)
            # contribution_new = 2*f[k-i]*x (if k!=2i and valid) or x^2 (if k==2i)
            
            # Simpler approach: just evaluate the objective on a grid of x values
            
            # Determine search range for x
            x_max = max(3.0 * fi, 3.0 / n) if fi > 0 else 5.0 / n
            x_min = 0.0
            
            # Grid search
            grid = np.linspace(0, x_max, 30)
            # Also add some points near the current value
            if fi > 0:
                close_grid = np.linspace(max(0, fi * 0.5), min(x_max, fi * 1.5), 20)
                grid = np.unique(np.concatenate([grid, close_grid]))
            
            best_x = fi
            best_obj = compute_score(f)
            
            S_fixed = S - fi  # sum of all f[j] for j != i
            
            for x in grid:
                f[i] = x
                new_S = S_fixed + x
                if new_S < 1e-10:
                    continue
                # Compute score directly (avoiding full renormalization since it's scale-invariant)
                conv_new = np.convolve(f, f)
                max_c = np.max(conv_new)
                obj = 2.0 * n * max_c / (new_S ** 2)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x
            
            f[i] = best_x
            S = S_fixed + best_x
            f = f / S  # renormalize
            S = 1.0
            
            # Check if overall score improved
            cur_score = compute_score(f)
            if cur_score < best_score - 1e-10:
                best_score = cur_score
                best_f = f.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    
    return best_f, best_score

"""
Hybrid optimizer combining fast coordinate descent with exact 1D sub-optimization,
subgradient descent with exact max, LP direction finding, and multi-phase search.
Targets score < 1.5027 for the non-negative autocorrelation minimization problem.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Fast computation of 2n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    conv = np.convolve(f, f)
    max_conv = np.max(conv)
    sum_f = np.sum(f)
    if sum_f < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (sum_f ** 2))


def compute_score_fast(f, conv, sum_f):
    """Score from precomputed convolution and sum."""
    n = len(f)
    max_conv = np.max(conv)
    if sum_f < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (sum_f ** 2))


def coordinate_descent(f, deadline, max_passes=500, verbose_prefix="", n_grid=60, n_bisect=25):
    """
    Coordinate descent with exact 1D minimization per coordinate.
    Uses precomputed convolution and incremental updates.
    """
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, np.inf
    f = f / S
    S = 1.0
    n = len(f)
    conv = np.convolve(f, f)
    
    best_f = f.copy()
    best_score = compute_score_fast(f, conv, S)
    
    pass_num = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() < deadline and pass_num < max_passes:
        pass_num += 1
        improved = False
        total_improvement = 0.0
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline - 1:
                break
            
            fi = f[i]
            
            # Affected k range: k where 0 <= k-i < n
            k_start = max(0, i)  # k-i >= 0 => k >= i
            k_end = min(i + n - 1, 2 * n - 2)  # k-i <= n-1 => k <= i+n-1
            
            # Unaffected max
            if k_start > 0 or k_end < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_start:k_end + 1] = False
                if np.any(mask):
                    C_unaffected = float(np.max(conv[mask]))
                else:
                    C_unaffected = 0.0
            else:
                C_unaffected = 0.0
            
            # Compute A, b, c for affected k values
            affected_ks = np.arange(k_start, k_end + 1)
            n_aff = len(affected_ks)
            A = np.empty(n_aff)
            b = np.empty(n_aff)
            c = np.empty(n_aff)
            
            for idx, k in enumerate(affected_ks):
                ki = k - i
                if k == 2 * i:
                    c[idx] = 1.0
                    b[idx] = 0.0
                    A[idx] = conv[k] - fi * fi
                else:
                    b[idx] = 2.0 * f[ki]
                    c[idx] = 0.0
                    A[idx] = conv[k] - 2.0 * f[ki] * fi
            
            S_fixed = S - fi
            
            # Find optimal x via grid + bisection
            x_max = max(2.5 * fi + 1.0 / n, 3.0 / n) if fi > 0 else 5.0 / n

            grid = np.linspace(0.0, x_max, n_grid)
            if fi > 0:
                close = np.linspace(max(0, fi * 0.3), min(x_max, fi * 2.0), 20)
                grid = np.unique(np.concatenate([grid, close]))

            new_S_vals = S_fixed + grid
            valid = new_S_vals > 1e-10
            if not np.any(valid):
                continue
            
            # vals[idx_coord, idx_grid] = A[idx_coord] + b[idx_coord]*grid[idx_grid] + c[idx_coord]*grid[idx_grid]^2
            vals = A[:, None] + b[:, None] * grid[valid, None] + c[:, None] * grid[valid, None] ** 2
            max_vals = np.maximum(np.max(vals, axis=0), C_unaffected)
            objs = 2.0 * n * max_vals / (new_S_vals[valid] ** 2)
            
            best_idx = int(np.argmin(objs))
            best_x = grid[valid][best_idx]
            best_obj = objs[best_idx]
            
            # Bisection refinement
            if len(grid[valid]) >= 3:
                idx_b = best_idx
                lo = grid[valid][max(0, idx_b - 1)]

                for _ in range(n_bisect):
                    mid = 0.5 * (lo + hi)
                    vals_mid = A + b * mid + c * mid * mid
                    max_mid = max(float(np.max(vals_mid)), C_unaffected)
                    S_mid = S_fixed + mid
                    obj_mid = 2.0 * n * max_mid / (S_mid * S_mid)
                    
                    vals_lo = A + b * lo + c * lo * lo
                    max_lo = max(float(np.max(vals_lo)), C_unaffected)
                    S_lo = S_fixed + lo
                    obj_lo = 2.0 * n * max_lo / (S_lo * S_lo)
                    
                    if obj_mid <= obj_lo:
                        lo = mid
                    else:
                        hi = mid
                
                best_x = 0.5 * (lo + hi)
                S_best = S_fixed + best_x
                vals_best = A + b * best_x + c * best_x * best_x
                max_best = max(float(np.max(vals_best)), C_unaffected)
                best_obj = 2.0 * n * max_best / (S_best * S_best)
            
            if best_obj < best_score - 1e-14:
                # Update f[i] and convolution
                for idx, k in enumerate(affected_ks):
                    conv[k] = A[idx] + b[idx] * best_x + c[idx] * best_x * best_x
                delta = best_x - fi
                S += delta
                f[i] = best_x
                improved = True
                total_improvement += best_score - best_obj
                best_score = best_obj
                best_f = f.copy()
        
        # Renormalize periodically to avoid drift
        if pass_num % 10 == 0:
            S = float(np.sum(f))
            if S > 0:
                f = f / S
                S = 1.0
                conv = np.convolve(f, f)
                best_score = min(best_score, compute_score_fast(f, conv, S))
            
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    
    return best_f, best_score


def subgradient_descent(f, deadline, lr_init=0.005, decay=0.995, verbose_prefix=""):
    """
    Subgradient descent on the exact max of convolution.
    Uses Armijo line search for adaptive step size.
    """
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    conv = np.convolve(f, f)
    max_idx = int(np.argmax(conv))
    current_max = float(conv[max_idx])
    
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline and it < 100000:
        # Compute subgradient of max conv component
        # grad[i] = 2*f[max_idx - i] if 0 <= max_idx-i < n
        #          = 2*f[i] if max_idx == 2*i
        grad = np.zeros(n)
        for i in range(n):
            other = max_idx - i
            if 0 <= other < n:
                if other == i:
                    grad[i] = 2.0 * f[i]
                else:
                    grad[i] = 2.0 * f[other]
        
        # Projected gradient step (project onto simplex: f>=0, sum=1)
        new_f = f - lr * grad
        new_f = np.maximum(new_f, 0.0)
        S_new = float(np.sum(new_f))
        if S_new < 1e-10:
            break
        new_f = new_f / S_new
        
        # Score
        sc = compute_score(new_f)
        
        if sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            conv = np.convolve(f, f)
            max_idx = int(np.argmax(conv))
            lr = min(lr * 1.05, 0.05)  # slightly increase step if improving
            no_improve = 0
        else:
            no_improve += 1
            lr *= decay
            if lr < 1e-8:
                break
            if no_improve > 500:
                # Try to jitter
                f = new_f  # accept the step anyway to escape
                conv = np.convolve(f, f)
                max_idx = int(np.argmax(conv))
                no_improve = 0
                lr = lr_init * 0.1
        
        it += 1
        if it % 5000 == 0 and verbose_prefix:
            print(f"    {verbose_prefix}SG iter {it}: score={best_score:.6f}, lr={lr:.6f}")
    
    return best_f, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g) <= M."""
    n = len(f)
    if A_cache is None:
        A = build_conv_matrix_sparse(f)
    else:
        A = A_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_iterate(f, deadline, step_init=0.1, max_iters=100, verbose_prefix=""):
    """LP-based iterate with line search."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    best_f = f.copy()
    best_score = compute_score(f)
    
    A_cache = build_conv_matrix_sparse(f)
    it = 0
    no_improve = 0
    step = step_init
    
    while time.monotonic() < deadline and it < max_iters:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M <= 0:
            break
        
        g = lp_direction(f, M, A_cache)
        it += 1
        
        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue
        
        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue
        
        g = g / sum_g  # normalize g to sum=1
        
        # Line search
        best_t = step
        best_s = best_score
        for trial_t in [step, step * 1.5, step * 0.5, step * 2.0, step * 0.7]:
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = compute_score(new_f)
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t
        
        if best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, 1000.0)
            S = float(np.sum(f))
            f = f / S
            best_f = f.copy()
            best_score = best_s
            A_cache = build_conv_matrix_sparse(f)
            no_improve = 0
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.003 or no_improve > 30:
                break
    
    return best_f, best_score


def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def raised_cosine_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (i / m)))
            w[n - 1 - i] = w[i]
    return w


def optimal_taper_window(n, alpha=1.5):
    """Super-Gaussian taper: w[i] = exp(-alpha * (2*i/(n-1) - 1)^4) in edges."""
    x = np.linspace(-1, 1, n)
    w = np.exp(-alpha * np.abs(x) ** (2 * alpha))
    return w


def gaussian_window(n, sigma=0.3):
    x = np.linspace(-1, 1, n)
    w = np.exp(-0.5 * (x / sigma) ** 2)
    return w


def perturb(f, rng, strength=0.05):
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 2.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def multi_init_search(f_init, deadline, verbose_prefix=""):
    """Run coordinate descent + subgradient + LP in sequence."""
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, np.inf
    f_init = f_init / S
    
    best_f = f_init.copy()
    best_score = compute_score(f_init)
    t0 = time.monotonic()
    budget = deadline - t0
    
    # Phase 1: Coordinate descent (main workhorse)
    cd_deadline = t0 + budget * 0.55
    if budget > 10:
        f1, s1 = coordinate_descent(f_init, cd_deadline, max_passes=300, verbose_prefix=verbose_prefix + "CD1")
        if s1 < best_score:
            best_f, best_score = f1, s1
    
    # Phase 2: Subgradient descent
    sg_deadline = t0 + budget * 0.75
    if time.monotonic() < sg_deadline:

        if s2 < best_score:
            best_f, best_score = f2, s2
    
    # Phase 3: LP-based

        f3, s3 = lp_iterate(best_f, deadline - 5, step_init=0.08, max_iters=50, verbose_prefix=verbose_prefix + "LP")

        f4, s4 = coordinate_descent(best_f, deadline - 2, max_passes=200, n_grid=40, n_bisect=20, verbose_prefix=verbose_prefix + "CD2")

                                        max_passes=1000, n_grid=80, n_bisect=30,
                                        verbose_prefix="[final_CD]")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
                
            if time.monotonic() < deadline - 3:
                bf2, bs2 = coordinate_descent(best_sequence, deadline - 3,
                                              max_passes=500, n_grid=100, n_bisect=35,
                                              verbose_prefix="[final_CD2]")
                if bs2 < best_score:
                    best_score = bs2
                    best_sequence = bf2.copy()
                    
            print(f"  Final: {best_score:.6f}")
        except Exception as e:
            print(f"  Final CD error: {e}")
    
    if best_sequence is None:
        best_sequence = tukey_window(512, 0.3)
        best_score = compute_score(best_sequence)
    
    # Validate
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = compute_score(best_sequence)
    
    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    
    # Verify with actual evaluate_sequence
    try:
        verified = evaluate_sequence([float(x) for x in best_sequence])
        print(f"Verified score: {verified:.6f}")
        if verified < best_score:
            best_score = verified
    except Exception:
        pass
    
    return [float(x) for x in best_sequence]
