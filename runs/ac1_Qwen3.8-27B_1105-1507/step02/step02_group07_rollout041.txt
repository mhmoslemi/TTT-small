"""
Optimization of step function to minimize 2*n*max(conv(f,f))/sum(f)^2.
Uses coordinate descent with efficient 1D subproblem solving,
multiple window-based initializations, and multi-length search.
The key insight: conv(f,f)[k] = sum_j f[j]*f[k-j], and we want this
as flat as possible. The theoretical lower bound is 2n/(2n-1) ~ 1.
"""
import numpy as np
import time


def compute_score(f):
    """Fast score computation."""
    n = len(f)
    conv = np.convolve(f, f)
    max_b = np.max(conv)
    sum_a = np.sum(f)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def solve_1d_subproblem(conv, f, i, old_val):
    """
    Solve: min_x max_k conv_new[k] where conv_new is conv after setting f[i]=x.
    conv_new[k] = conv[k] + 2*(x-old)*f[k-i] (for valid k-i, k!=2i)
                 conv[2i] + x^2 - old^2 (for k=2i)
                 conv[k] (otherwise)
    Returns optimal x >= 0.
    """
    n = len(f)
    
    # Precompute the "static" part and the x-dependent coefficients
    # For each k, conv_new[k] = C[k] + B[k]*x + A[k]*x^2
    # A[k] = 1 if k == 2i, else 0
    # B[k] = 2*f[k-i] if 0<=k-i<n and k!=2i, else 0
    # C[k] = conv[k] - B[k]*old - A[k]*old^2  ... let me just use the delta form.
    # conv_new[k] = conv[k] + delta[k] where:
    #   delta[k] = 2*(x-old)*f[k-i] (0<=k-i<n, k!=2i)
    #   delta[k] = x^2 - old^2 (k=2i)
    #   delta[k] = 0 (otherwise)
    
    # For the 1D minimization, I'll evaluate phi(x) at several points.
    # phi(x) = max_k (conv[k] + delta_k(x))
    
    # Build arrays for efficient evaluation
    # valid_k: indices where 0<=k-i<n and k!=2i
    k_min = i  # k-i >= 0 => k >= i
    k_max = i + n - 1  # k-i <= n-1 => k <= i+n-1
    # But k must also be in [0, 2n-2]
    k_min = max(0, i)
    k_max = min(2 * n - 2, i + n - 1)
    
    if k_min <= k_max:
        valid_ks = np.arange(k_min, k_max + 1)
        # Exclude k = 2i
        if 2 * i >= k_min and 2 * i <= k_max:
            mask = valid_ks != 2 * i
            valid_ks = valid_ks[mask]
        if len(valid_ks) > 0:
            B_vals = 2.0 * f[valid_ks - i]  # 2*f[k-i] for valid k
            C_vals = conv[valid_ks]  # base values
        else:
            B_vals = np.array([])
            C_vals = np.array([])
    else:
        B_vals = np.array([])
        C_vals = np.array([])
    
    # k=2i term
    ki = 2 * i
    if 0 <= ki <= 2 * n - 2:
        C_ki = conv[ki] - old_val ** 2  # so that C_ki + x^2 = conv[ki] + x^2 - old^2
    else:
        C_ki = -np.inf
    
    # Other k (not in valid range and not k=2i): conv doesn't change
    if len(valid_ks) > 0:
        other_mask = np.ones(2 * n - 1, dtype=bool)
        other_mask[valid_ks] = False
        if 0 <= ki <= 2 * n - 2:
            other_mask[ki] = False
        M_other = np.max(conv[other_mask]) if np.any(other_mask) else -np.inf
    else:
        M_other = np.max(conv)
        if 0 <= ki <= 2 * n - 2:
            # Remove ki from the max
            conv_no_ki = conv.copy()
            conv_no_ki[ki] = -np.inf
            M_other = np.max(conv_no_ki)
    
    def phi(x):
        # conv_new[k] for valid ks: C_vals + B_vals * x
        # conv_new[ki]: C_ki + x^2
        # others: unchanged (= M_other)
        if len(B_vals) > 0:
            linear_max = np.max(C_vals + B_vals * x)
        else:
            linear_max = -np.inf
        quad_val = C_ki + x * x
        return max(linear_max, quad_val, M_other)
    
    # Golden section search on [0, x_hi]
    # Upper bound: current value gives phi(old_val). For large x, phi ~ x^2.
    x_hi = max(2.0 * old_val + 1e-6, 1e-6)
    # Expand if needed
    for _ in range(20):
        if phi(x_hi) > phi(x_hi * 0.5):
            break
        x_hi *= 2.0
    x_hi = min(x_hi, 10.0)  # cap
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x_lo = 0.0
    c = x_hi - gr * (x_hi - x_lo)
    d = x_lo + gr * (x_hi - x_lo)
    fc = phi(c)
    fd = phi(d)
    
    for _ in range(30):
        if fc < fd:
            x_hi = d
            d = c
            fd = fc
            c = x_hi - gr * (x_hi - x_lo)
            fc = phi(c)
        else:
            x_lo = c
            c = d
            fc = fd
            d = x_lo + gr * (x_hi - x_lo)
            fd = phi(d)
    
    x_opt = (x_lo + x_hi) / 2.0
    # Also check x=0
    if phi(0.0) < phi(x_opt):
        x_opt = 0.0
    return x_opt


def coordinate_descent(f, deadline, max_passes=50000):
    """Run coordinate descent to minimize the score."""
    n = len(f)
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    if S < 1e-10:
        return f, np.inf
    f = f / S
    
    conv = np.convolve(f, f)
    best_f = f.copy()
    best_score = compute_score(f)
    no_improve = 0
    
    for pass_num in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        pass_improved = False
        for i in range(n):
            old_val = f[i]
            if old_val < 1e-12:
                # Check if increasing from 0 helps
                # phi(0) = max(conv) (no change)
                # phi(eps) ≈ max(conv[k] + 2*eps*f[k-i]) for valid k
                # If all f[k-i] <= 0 for valid k, no improvement possible
                # Since f >= 0, f[k-i] >= 0, so increasing helps only if it reduces max
                # Skip for efficiency
                continue
            
            x_new = solve_1d_subproblem(conv, f, i, old_val)
            x_new = max(0.0, x_new)
            
            if abs(x_new - old_val) > 1e-12:
                # Update conv efficiently
                delta = x_new - old_val
                # Update conv entries
                k_start = max(0, i)
                k_end = min(2 * n - 1, i + n)
                for k in range(k_start, k_end):
                    if k == 2 * i:
                        conv[k] += x_new ** 2 - old_val ** 2
                    else:
                        conv[k] += 2.0 * delta * f[k - i]
                f[i] = x_new
                pass_improved = True
    
        # Renormalize
        S = np.sum(f)
        if S > 1e-10:
            f = f / S
            conv = np.convolve(f, f)
        
        sc = np.max(conv)
        score = 2.0 * n * sc / (np.sum(f) ** 2)
        if score < best_score - 1e-10:
            best_score = score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break
    
    return best_f, best_score


def softmax_smooth_max(conv, beta):
    """Compute smooth approximation of max."""
    max_c = np.max(conv)
    if beta * (np.max(conv) - max_c) < -700:
        return max_c
    exp_c = np.exp(beta * (conv - max_c))
    total = np.sum(exp_c)
    if total < 1e-300:
        return max_c
    return max_c + np.log(total) / beta


def softmax_gradient(f, beta):
    """Gradient of smooth max of conv(f,f)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_c = np.max(conv)
    exp_c = np.exp(beta * (conv - max_c))
    total = np.sum(exp_c)
    if total < 1e-300:
        return np.zeros(n)
    p = exp_c / total
    # grad[j] = 2 * sum_k p[k] * f[k-j] = 2 * (p correlated with f)
    # p has length 2n-1, f has length n
    # sum_k p[k]*f[k-j]: k ranges over [0,2n-2], k-j ranges over [0,n-1]
    # = sum_{k=j}^{j+2n-2} p[k]*f[k-j] but f[k-j]=0 outside [0,n-1]
    # = sum_{m=0}^{n-1} p[j+m]*f[m] (where m=k-j, need j+m <= 2n-2)
    # This is a cross-correlation
    # Use: for each j, grad[j] = 2 * dot(p[j:j+n], f) (with padding)
    # p has length 2n-1. For j in [0,n-1], p[j:j+n] has length n (since j+n-1 <= 2n-2).
    grad = 2.0 * np.array([np.dot(p[j:j+n], f) for j in range(n)])
    return grad


def adam_optimize(f, deadline, beta_start=50.0, beta_end=500.0, lr_start=0.02, max_iters=5000):
    """Adam optimization on smooth max of convolution."""
    n = len(f)
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    f = f / S
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline and it < max_iters:
        frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        grad = softmax_gradient(f, beta)
        
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        new_f = f - lr * update
        new_f = np.clip(new_f, 0.0, None)
        ns = np.sum(new_f)
        if ns < 1e-10:
            new_f = f
            ns = S
        new_f = new_f / ns
        
        sc = compute_score(new_f)
        if sc < best_score - 1e-12:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-8:
                    break
    
    return best_f, best_score


def make_hann(n):
    return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))


def make_blackman(n):
    w = np.ones(n)
    for i in range(n):
        w[i] = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    return np.maximum(w, 0)


def make_tukey(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2.0 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def make_flattop(n):
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    w = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k, ak in enumerate(a):
            s += ak * np.cos(np.pi * (2.0 * i / (n - 1) - 1 + 2 * k))
        w[i] = s
    return np.maximum(w, 0)


def make_spectral_init(n):
    """
    Initialize based on spectral considerations.
    We want conv(f,f) to be flat, which in freq domain means
    F(z)^2 ≈ c*(1+z+...+z^{2n-2}).
    So F(z) ≈ sqrt(c) * sqrt((1-z^{2n-1})/(1-z)).
    Approximate this and take inverse DFT.
    """
    N = 2 * n - 1
    # Ideally F[k] = sqrt(c * N) * delta_{k,0}, but that's only for uniform.
    # Instead, use a "Bartlett-like" spectrum that gives near-flat convolution.
    # The DFT of a triangular sequence is a squared sinc.
    # Let's try: f is proportional to the inverse DFT of |F_ideal|^2 where F_ideal is chosen.
    
    # Alternative: use the fact that for a "flat" convolution, we need
    # the autocorrelation (in the convolution sense) to be flat.
    # A good approximation is a super-Gaussian or a carefully designed window.
    
    # Use a raised-to-power cosine: f[i] = cos^p(pi * i / (n-1))
    # For p=1: cosine taper
    # For p=2: squared cosine (Bartlett-like)
    # Larger p gives more "rectangular" shape
    
    idx = np.arange(n, dtype=np.float64)
    t = idx / (n - 1)
    # Use a product of cosines (similar to Blackman but continuous)
    w = np.cos(np.pi * (t - 0.5)) ** 2  # squared cosine
    w = np.maximum(w, 1e-10)
    return w


def make_random_smooth(n, rng, smoothness=0.1):
    """Generate a random smooth positive sequence."""
    x = rng.standard_normal(n)
    # Smooth via convolution with Gaussian
    ksize = max(3, n // 8)
    kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) / (ksize / 3.0)) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(x, kernel, mode='same')
    # Make positive
    smooth = np.abs(smooth) + 0.1
    return smooth

                        print(f"  Perturb {r}: {bs:.6f}")
                except Exception as e:
                    print(f"  Perturb {r} error: {e}")
        
        # Final coordinate descent pass
        if time.monotonic() < deadline - 5:
            bf, bs = coordinate_descent(best_sequence, deadline - 5, max_passes=200000)
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
            print(f"  Final score: {best_score:.6f}")
    
    if best_sequence is None:
        best_sequence = make_hann(512)
        best_score = evaluate_sequence(best_sequence.tolist())
    
    # Ensure valid output
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())
    
    elapsed = time.monotonic() - t0
    final_score = evaluate_sequence(best_sequence.tolist())
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
