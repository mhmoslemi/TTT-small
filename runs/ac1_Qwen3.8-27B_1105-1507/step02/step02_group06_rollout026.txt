def adam_subgradient(f, S, deadline, top_k=5, lr=0.001, beta1=0.9, beta2=0.999):
    f = f.copy().astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    while time.monotonic() < deadline:
        t += 1
        conv = np.convolve(f, f)
        top_indices = np.argsort(conv)[-top_k:]
        
        grad = np.zeros(n)
        for k in top_indices:
            j_min = max(0, k - (n-1))
            j_max = min(k, n-1)
            for j in range(j_min, j_max+1):
                grad[j] += 2.0 * f[k-j]
        
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = project_onto_simplex(f_new, S)
        
        score = compute_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new.copy()
    
    return best_f, best_score

Wait, let me reconsider. (a*a)_k = sum over valid (j, k-j) of a_j * a_{k-j}. This is a bilinear form, not a quadratic form in the standard sense (it's not a^T M a with M symmetric).

Actually, (a*a)_k = sum_j a_j * a_{k-j}. If we write this as a vector operation, it's the inner product of a with a "shifted and reversed" version of a. Specifically, if we define (R_k a)_j = a_{k-j} (with out-of-bounds = 0), then (a*a)_k = a · (R_k a) = a^T R_k a.

R_k is a permutation-like matrix (shift and reverse). It's symmetric: (R_k a)_j = a_{k-j} and (R_k^T a)_j = a_{k-j} (same thing since R_k is symmetric for the full vector; for the truncated version, it's still symmetric).

The Hessian of a^T R_k a is R_k + R_k^T = 2*R_k (since R_k is symmetric). For this to be PSD, we need R_k to be PSD. But R_k is a permutation matrix (or a submatrix of one), which has eigenvalues on the unit circle, not all positive. So (a*a)_k is not convex in general.

This means the problem is non-convex, and we can't use convex optimization techniques directly.

However, the max of these functions is still "nice" in some sense - it's the pointwise max of quadratic forms.

Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses softmax-gradient with Nesterov momentum, periodic LP polishing,

import numpy as np

from scipy.sparse import csr_matrix, coo_matrix

def compute_score(f):
    """Fast score computation."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))

def project_nonneg_sum(f, S):
    """Project f onto {x >= 0, sum(x) = S} using bisection."""
    n = len(f)
    if S <= 0:
        return np.zeros(n)
    lo = float(np.min(f)) - 1.0
    hi = float(np.max(f)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(f - mid, 0.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.maximum(f - t, 0.0)

def softmax_gradient(f, temperature=0.5):
    """Compute softmax-weighted gradient of max_k(conv(f,f))."""
    n = len(f)
    conv = np.convolve(f, f)
    # Softmax weights
    shifted = conv - np.max(conv)
    weights = np.exp(shifted / temperature)
    weights /= weights.sum()
    
    # Compute gradient: grad_i = 2 * sum_k weights[k] * f[k-i]
    # This is 2 * (A @ weights) where A[k,i] = f[k-i] if valid
    # We can compute this using convolution!
    # (A @ weights)_i = sum_k weights[k] * f[k-i] = (weights * reverse(f))_i
    # Actually: sum_k weights[k] * f[k-i] = convolution of weights and reversed f
    # Let's just do it directly for clarity
    
    grad = np.zeros(n)
    f_rev = f[::-1]
    # sum_k weights[k] * f[k-i] for i = 0, ..., n-1
    # This is the (full) convolution of weights and f_rev, evaluated at appropriate positions
    # conv(weights, f_rev)[i + n - 1] = sum_k weights[k] * f_rev[i + n - 1 - k]
    # = sum_k weights[k] * f[n-1 - (i + n - 1 - k)] = sum_k weights[k] * f[k - i]
    # So grad = 2 * conv(weights, f_rev)[n-1 : n-1+n]
    
    conv_wf = np.convolve(weights, f_rev, mode='full')
    grad = 2.0 * conv_wf[n-1:n-1+n]
    
    return grad, conv

def topk_gradient(f, top_k=5):
    """Compute subgradient using top-k lags."""
    n = len(f)
    conv = np.convolve(f, f)
    top_indices = np.argsort(conv)[-top_k:]
    
    grad = np.zeros(n)
    for k in top_indices:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * f[k - j]
    
    return grad, conv

def run_nesterov(f, S, deadline, method='softmax', top_k=5, temperature=0.5,
                  lr_init=0.005, momentum_init=0.9, verbose=""):
    """Nesterov-accelerated optimization."""
    f = f.copy().astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    
    v = np.zeros(n)
    beta = momentum_init
    beta_const = momentum_init
    lr = lr_init
    y = f.copy()
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        # Compute gradient at y
        if method == 'softmax':
            grad, _ = softmax_gradient(y, temperature)
        else:
            grad, _ = topk_gradient(y, top_k)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        # Nesterov step
        v = beta * v - lr * grad
        f_new = f + v
        f_new = project_nonneg_sum(f_new, S)
        
        # Nesterov extrapolation
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = f_new + ((beta - 1.0) / beta_new) * (f_new - f)
        y = np.clip(y, 0.0, None)
        beta = beta_new
        
        score = compute_score(f_new)
        it += 1
        
        if score < best_score - 1e-9:
            best_score = score
            best_f = f_new.copy()
            f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.9
                beta = beta_const
                v *= 0.5
        
        if it % 500 == 0 and verbose:
            print(f"    {verbose}iter={it} best={best_score:.6f} lr={lr:.6f}")
        
        if lr < 1e-9:
            break
    
    return best_f, best_score

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = sum_j f[k-j]*g[j]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp(f, M, active_lags=None, max_active=100):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for active k."""
    n = len(f)
    if active_lags is None:
        conv_ff = np.convolve(f, f)
        active_lags = np.argsort(conv_ff)[-min(max_active, 2*n-1):]
    
    A = build_conv_matrix_sparse(f)
    # Use only active constraints
    A_sub = A[active_lags, :]
    b_ub = np.full(len(active_lags), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            return np.clip(result.x, 0, None)
    except Exception:
        pass
    return None

"""
Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses softmax-weighted gradient with Nesterov momentum, adaptive temperature,
periodic LP polishing with active-set method, random kicks to escape local minima,
and multi-start strategy at n=2048 with n=1024 backup.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Fast score computation."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def project_nonneg_sum(f, S):
    """Project f onto {x >= 0, sum(x) = S} using bisection."""
    n = len(f)
    if S <= 0:
        return np.zeros(n)
    lo = float(np.min(f)) - 1.0
    hi = float(np.max(f)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(f - mid, 0.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(f - t, 0.0)
    # Fix any numerical drift
    rsum = float(np.sum(res))
    if rsum > 1e-15:
        res = res * (S / rsum)
    return res


def softmax_gradient(f, temperature=0.5):
    """Compute softmax-weighted gradient of max_k(conv(f,f))."""
    n = len(f)
    conv = np.convolve(f, f)
    shifted = conv - np.max(conv)
    weights = np.exp(shifted / temperature)
    wsum = float(weights.sum())
    if wsum < 1e-30:
        weights = np.ones_like(conv) / len(conv)
    else:
        weights /= wsum

    # grad_i = 2 * sum_k weights[k] * f[k-i]
    f_rev = f[::-1]
    conv_wf = np.convolve(weights, f_rev, mode='full')
    grad = 2.0 * conv_wf[n - 1:n - 1 + n]
    return grad, conv


def topk_gradient(f, top_k=5):
    """Compute subgradient using top-k lags."""
    n = len(f)
    conv = np.convolve(f, f)
    top_indices = np.argsort(conv)[-top_k:]

    grad = np.zeros(n)
    for k in top_indices:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * f[k - j]

    return grad, conv


def run_nesterov(f, S, deadline, method='softmax', top_k=5,
                 temp_init=1.0, temp_min=0.1, temp_decay=0.9995,
                 lr_init=0.005, momentum_init=0.9,
                 kick_interval=500, kick_size=0.01, verbose=""):
    """Nesterov-accelerated optimization with adaptive temperature and kicks."""
    f = f.copy().astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)

    v = np.zeros(n)
    beta = momentum_init
    beta_const = momentum_init
    lr = lr_init
    y = f.copy()
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        # Compute gradient at y
        if method == 'softmax':
            grad, conv = softmax_gradient(y, temp)
        else:
            grad, conv = topk_gradient(y, top_k)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Nesterov step
        v = beta * v - lr * grad
        f_new = f + v
        f_new = project_nonneg_sum(f_new, S)

        # Nesterov extrapolation
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = f_new + ((beta - 1.0) / beta_new) * (f_new - f)
        y = np.clip(y, 0.0, None)
        beta = beta_new

        # Decay temperature
        temp = max(temp * temp_decay, temp_min)

        score = compute_score(f_new)
        it += 1

        if score < best_score - 1e-10:
            best_score = score
            best_f = f_new.copy()
            f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.92
                beta = beta_const
                v *= 0.3
            # Random kick to escape local minima
            if no_improve >= kick_interval:
                noise = np.random.randn(n) * kick_size
                f_kick = f + noise
                f_kick = project_nonneg_sum(f_kick, S)
                score_kick = compute_score(f_kick)
                if score_kick < best_score:
                    f = f_kick
                    best_f = f.copy()
                    best_score = score_kick
                no_improve = 0
                v = np.zeros(n)
                beta = beta_const
                lr = max(lr, lr_init * 0.3)

        if it % 1000 == 0 and verbose:
            print(f"    {verbose}iter={it} best={best_score:.6f} lr={lr:.6f} temp={temp:.4f}")

        if lr < 1e-9:
            break

    return best_f, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = sum_j f[k-j]*g[j]."""
    n = len(f)
    rows = np.empty(2 * n * n, dtype=np.int32)
    cols = np.empty(2 * n * n, dtype=np.int32)
    vals = np.empty(2 * n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp_active(f, M, active_lags):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for k in active_lags."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    A_sub = A[active_lags, :]
    b_ub = np.full(len(active_lags), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            return np.clip(result.x, 0.0, None)
    except Exception:
        pass
    return None


def lp_polish(f, S, deadline, verbose=""):
    """LP-based polishing with active-set constraint selection."""
    f = f.copy().astype(np.float64)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    step = 0.08
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if M <= 0:
            break

        # Active set: top 30 lags
        active = np.argsort(conv_ff)[-30:]

        g = solve_lp_active(f, M, active)
        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue

        g = g * (S / sum_g)

        # Check if g satisfies all constraints (grow active set if not)
        conv_fg = np.convolve(f, g)
        violated = np.where(conv_fg > M + 1e-8)[0]
        if len(violated) > 0:
            active = np.unique(np.concatenate([active, violated[:10]]))
            if len(active) > 100:
                active = np.argsort(conv_fg[active])[-80:]
            continue

        # Line search
        best_step_f = None
        best_step_score = best_score
        for alpha in [step, step * 0.5, step * 0.25, step * 1.5, step * 0.75]:
            alpha = min(alpha, 1.0)
            if alpha < 0.001:
                continue
            cand = (1.0 - alpha) * f + alpha * g
            cand = project_nonneg_sum(cand, S)
            sc = compute_score(cand)
            if sc < best_step_score:
                best_step_score = sc
                best_step_f = cand

        it += 1
        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            step = max(step * 0.995, 0.01)
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.004 or no_improve > 25:
                break

        if it % 5 == 0 and verbose:
            print(f"    {verbose}LP iter={it} score={best_score:.6f} step={step:.4f}")

    return best_f, best_score


def make_initializations(n, S, rng, parent=None):
    """Create diverse initializations for length n, normalized to sum S."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    u = np.ones(n) * S / n
    inits.append(('uniform', u))

    # Cosine taper
    cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    inits.append(('cosine', cos_t * S / cos_t.sum()))

    # Random
    r = rng.uniform(0, 1, n)
    inits.append(('rand1', r * S / r.sum()))
    r2 = rng.uniform(0.1, 1, n)
    inits.append(('rand2', r2 * S / r2.sum()))

    # Bimodal
    bim = (np.exp(-((idx - n * 0.25) ** 2) / (2 * (n * 0.12) ** 2)) +
           np.exp(-((idx - n * 0.75) ** 2) / (2 * (n * 0.12) ** 2)))
    inits.append(('bimodal', bim * S / bim.sum()))

    # High-frequency wave
    wave = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * idx / n)
    inits.append(('wave', wave * S / wave.sum()))

    # S-curve
    sc = 1.0 / (1.0 + np.exp(-6.0 * (idx / (n - 1) - 0.5)))
    inits.append(('sigmoid', sc * S / sc.sum()))

    # Parent interpolation
    if parent is not None:
        if len(parent) == n:
            p = parent * S / parent.sum()
            inits.append(('parent', p))
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            p = np.interp(x_new, x_old, parent)
            p = np.maximum(p, 0)
            if p.sum() > 1e-10:
                inits.append(('parent_interp', p * S / p.sum()))

    # Triangular
    tri = 1.0 - 2.0 * np.abs(idx / (n - 1) - 0.5)
    inits.append(('tri', tri * S / tri.sum()))

    return inits
