"""
LP-optimized sequence with line search, spectral initialization, and multi-length exploration.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating LP with optimal step sizes.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-accelerated autocorrelation."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix A where (a*g)_k = sum_j A[k,j]*g_j."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp(a, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. A@v ≤ M, g ≥ 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def line_search(a, g, n_evals=40):
    """Golden section search for optimal t in [0,1] minimizing f((1-t)a+tg)."""
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + 0.381966 * (hi - lo)
        m2 = lo + 0.618034 * (hi - lo)
        a1 = np.maximum((1 - m1) * a + m1 * g, 0)
        a2 = np.maximum((1 - m2) * a + m2 * g, 0)
        s1, s2 = np.sum(a1), np.sum(a2)
        if s1 < 1e-10 or s2 < 1e-10:
            break
        o1 = compute_obj(a1)
        o2 = compute_obj(a2)
        if o1 < o2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-6:
            break
    return (lo + hi) / 2.0


def lp_optimize(a, max_iter=300, time_limit=120.0, verbose=False):
    """Alternating LP with line search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp(a, M, time_limit=5.0)
        if g is None:
            break
        
        sum_g = float(np.sum(g))
        if sum_g < 1.0 + 1e-5:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0
        
        g = g / sum_g
        
        t = line_search(a, g)
        
        a_new = np.maximum((1 - t) * a + t * g, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            if verbose and it % 20 == 0:
                print(f"    LP it={it}: obj={best_obj:.8f}, t={t:.4f}")
        
        a = a_new
    
    return best_a, best_obj

def init_sequence(n, seed):
    """Generate an initial sequence of length n."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 6)
    x = np.arange(n)
    if choice == 0:
        a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1 + 1e-10) + rng.uniform(0, 2*np.pi)))
    elif choice == 1:
        center = n / 2.0
        width = n * rng.uniform(0.15, 0.5)
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif choice == 2:
        a = np.ones(n)
        a[n//4:3*n//4] = rng.uniform(0.3, 0.8)
    elif choice == 3:
        freq = rng.uniform(0.5, 3.0)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
    elif choice == 4:
        a = np.maximum(0, 1 - 2 * np.abs(x - n/2) / n)
        a = a ** rng.uniform(0.5, 3.0)
    else:
        a = rng.random(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * rng.uniform(0.3, 2.0) * x / n))
        a = a * (1 - 0.3 * rng.random(n))
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
