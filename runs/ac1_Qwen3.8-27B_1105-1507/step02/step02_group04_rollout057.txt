    start = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = np.sum(a)

    best_obj = compute_obj(a)
    if verbose:
        print(f"  Initial: obj={best_obj:.6f}")

    tau = 1.0
    stall = 0
    patience = 800
    last_lp_iter = 0
    last_cd_iter = 0
    iter_count = 0

    while time.monotonic() - start < max_time:
        conv = compute_conv_fft(a)
        current_obj = 2.0 * n * np.max(conv)

        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > patience:
            # Adaptive reset
            eta = max(0.01, eta * 0.5)
            tau = min(tau * 2.0, 500.0)
            # Add multiplicative noise
            rng = np.random.default_rng(hash((n, iter_count)) % (2**32))
            a = a * np.exp(rng.normal(0, 0.15, n))
            a = np.maximum(a, 1e-10)

    # Bimodal flat
    a = np.ones(n)
    a[n // 4: 3 * n // 4] *= rng.uniform(0.3, 0.7)
    a = np.maximum(a, 1e-6)
    inits.append(a / np.sum(a))

    return inits


def run_for_n(n, seed, max_time=60.0, verbose=False):
    """Run optimization for a given sequence length with multiple initializations."""
    rng = np.random.default_rng(seed)
    inits = generate_initializations(n, rng)

    best_a = None
    best_obj = np.inf

    # Evaluate all inits quickly
    for a_init in inits:
        obj = compute_obj(a_init)
        if obj < best_obj:
            best_obj = obj
            best_a = a_init.copy()

    # Optimize the top inits
    n_optimize = min(len(inits), max(2, int(max_time / 20)))
    sorted_inits = sorted(inits, key=lambda x: compute_obj(x))[:n_optimize]

    for idx, a_init in enumerate(sorted_inits):
        if time.monotonic() > time.monotonic() + max_time:
            break
        # Use a fresh time budget
        a_opt, obj = optimize_sequence(a_init, max_time=max(10, max_time / (n_optimize - idx)),
                                       verbose=False)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    n={n} init{idx}: obj={best_obj:.6f}")

    # Final coordinate descent polish
    if best_a is not None:
        best_a = coordinate_descent_pass(best_a, max_passes=10)
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < np.inf:
                print(f"Parent (n={parent.size}): obj={obj:.6f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for the best sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses mirror descent + LP + coordinate descent with multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Determine sequence lengths to explore
    n_values = [20, 30, 40, 50, 60, 80, 100, 120, 150, 180, 200, 250, 300, 350, 400, 500]

    if parent is not None:
        n_p = len(parent)
        n_values = sorted(set(n_values + [n_p, max(10, n_p - 30), min(600, n_p + 30)]))

    # First pass: quick evaluation
    print(f"Exploring n values: {n_values}")
    quick_results = []
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        # Quick: just evaluate initializations
        rng_n = np.random.default_rng(seed + n * 7919)
        inits = generate_initializations(n, rng_n)
        best_n_obj = np.inf
        best_n_a = None
        for a_init in inits:
            obj = compute_obj(a_init)
            if obj < best_n_obj:
                best_n_obj = obj
                best_n_a = a_init.copy()

        if best_n_a is not None:
            quick_results.append((n, best_n_a, best_n_obj))
            if best_n_obj < best_score:
                best_score = best_n_obj
                best_seq = best_n_a.copy()
                print(f"  Quick n={n}: obj={best_score:.6f}")

    # Sort by initial quality, best first
    quick_results.sort(key=lambda x: x[2])

    # Second pass: full optimization on the most promising lengths
    n_to_optimize = [r[0] for r in quick_results[:min(8, len(quick_results))]]
    for n in n_to_optimize:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation proportional to remaining
        if n <= 60:
            alloc = min(30.0, remaining * 0.15)
        elif n <= 120:
            alloc = min(50.0, remaining * 0.18)
        elif n <= 200:
            alloc = min(80.0, remaining * 0.2)
        elif n <= 300:
            alloc = min(120.0, remaining * 0.25)
        else:
            alloc = min(150.0, remaining * 0.3)

        alloc = max(15.0, alloc)
        print(f"Optimizing n={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        # Get the best init for this n from quick results
        init_a = None
        for r in quick_results:
            if r[0] == n:
                init_a = r[1].copy()
                break

        if init_a is None:
            continue

        a_opt, obj = run_for_n(n, seed + n * 7919, max_time=alloc, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  UPDATE: n={n}, obj={best_score:.6f}")

    # Final intensive polishing on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        a_cur = best_seq.copy()
        print(f"\nPolishing n={n_best}, {remaining:.0f}s remaining")

        polish_round = 0
        while time.monotonic() < deadline - 5:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break

            # Coordinate descent polish (most reliable local improvement)
            cd_time = min(20.0, remaining * 0.3)
            max_cd_passes = int(max(3, cd_time / 2))
            a_cur = coordinate_descent_pass(a_cur, max_passes=max_cd_passes)
            obj_cd = compute_obj(a_cur)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cur.copy()
                print(f"  Polish {polish_round} CD: obj={best_score:.6f}")

            # LP step
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 350:
                a_lp, improved = lp_step_with_linesearch(a_cur, t_max=0.3)
                if improved:
                    a_cur = a_lp
                    obj_lp = compute_obj(a_cur)
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = a_cur.copy()
                        print(f"  Polish {polish_round} LP: obj={best_score:.6f}")

            # Mirror descent burst
            remaining = deadline - time.monotonic()
            if remaining > 10:
                md_time = min(10.0, remaining * 0.2)
                a_cur, obj_md = optimize_sequence(a_cur, max_time=md_time)
                if obj_md < best_score:
                    best_score = obj_md
                    best_seq = a_cur.copy()
                    a_cur = a_cur.copy()
                    print(f"  Polish {polish_round} MD: obj={best_score:.6f}")

            # Small perturbation to escape local minima
            if remaining > 10 and polish_round % 3 == 0:
                noise_scale = 0.03
                a_pert = a_cur * np.exp(np.random.default_rng(polish_round).normal(0, noise_scale, n_best))
                a_pert = np.maximum(a_pert, 1e-10)
                a_pert = a_pert / np.sum(a_pert)
                # Quick coordinate descent on perturbed
                a_pert = coordinate_descent_pass(a_pert, max_passes=3)
                obj_p = compute_obj(a_pert)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_pert.copy()
                    a_cur = a_pert.copy()
                    print(f"  Polish {polish_round} perturb: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(50) / 50
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if np.isfinite(validation) and validation < best_score:
        print(f"  Validation: {validation:.6f} < computed {best_score:.6f}")
        best_score = validation

    return result
