"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A where (A @ x)_k = (a * x)_k for full convolution."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        # (a * x)_k = sum_j a_j * x_{k-j}
        # Valid j: max(0, k-(n-1)) <= j <= min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # a_j * x_{k-j}, so A[k, j] = a_j where x index is k-j
            # Actually: (A @ x)_k = sum_j A[k,j] * x_j
            # We want (a * x)_k = sum_j a_j * x_{k-j}
            # Let i = k-j, so j = k-i
            # (a * x)_k = sum_i a_{k-i} * x_i
            # So A[k, i] = a_{k-i} for valid i
            # i ranges from max(0, k-(n-1)) to min(n-1, k)
            for i in range(j_lo, j_hi + 1):
                A[k, i] = a[k - i]
    return A


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a)) / sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lp_min_peak(a):
    """
    Solve: min t s.t. conv(a, g) <= t, sum(g) = 1, g >= 0.
    Returns (g, t) where g is the optimal direction and t is the min peak.
    """
    n = len(a)
    m = 2 * n - 1
    A = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t => c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A @ g - t <= 0 => [A, -1] @ [g, t] <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1 => [1,...,1, 0] @ [g, t] = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        g = res.x[:n]
        t = res.x[n]
        return g, float(t)
    return None, None


def lp_max_sum(a, M):
    """
    Solve: max sum(g) s.t. conv(a, g) <= M, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    m = 2 * n - 1
    A = build_conv_matrix(a)
    
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success and res.x.sum() > 1e-10:
        return res.x / res.x.sum()
    return None


def smooth_objective(x, beta, n):
    """Log-sum-exp smoothed objective for normalized a."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / beta
    # J = 2n * max(c) / sum^2, but we normalized sum=1, so J = 2n * max(c)
    # The smooth approx: 2n * log_sum
    return float(2.0 * n * log_sum)


def smooth_refine(a, deadline, max_beta=50000):
    """Refine using log-sum-exp with increasing beta."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    
    betas = [50, 200, 1000, 5000, 20000]
    if max_beta > 20000:
        betas.append(50000)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = best.copy()
        x0 = np.clip(x0, 0.0001, None)
        x0 = x0 / x0.sum()
        
        try:
            res = minimize(
                smooth_objective, x0, args=(beta, n),
                method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': 200, 'maxfun': 5000,
                         'ftol': 1e-18, 'gtol': 1e-14}
            )
            if res.fun < 1e10:
                res_x = np.clip(res.x, 0, None)
                if res_x.sum() > 0.01:
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
        except Exception:
            pass
    
    return best, best_score

    best_local = a.copy()

    no_improve_count = 0
    step_sizes = [1.0, 0.7, 0.5, 0.3, 0.1]
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP: min t s.t. conv(a,g) <= t, sum(g)=1, g>=0
        g, t_opt = lp_min_peak(a)
        if g is None:
            break
        
        # Try multiple step sizes
        best_new_score = best_score
        best_new_a = a
        
        for alpha in step_sizes:
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            if score < best_new_score:
                best_new_score = score
                best_new_a = a_new.copy()
        
        if best_new_score < best_score:
            best_score = best_new_score
            best_local = best_new_a.copy()
            a = best_new_a
            no_improve_count = 0
        else:
            no_improve_count += 1
            # Try a smaller random step
            alpha_rand = 0.05 * np.exp(-no_improve_count * 0.5)
            a_new = (1.0 - alpha_rand) * a + alpha_rand * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                score = compute_score(a_new)
                if score < best_score:
                    best_score = score
                    best_local = a_new.copy()
                    a = a_new
                    no_improve_count = 0
            
            if no_improve_count > 30:
                break
        
        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: J={best_score:.6f}, t_opt={t_opt:.6f}", flush=True)
    
    return best_local, best_score

    inits.append(np.ones(n) / n)

def subgradient_step(a, alpha):
    """One subgradient descent step on the max autocorrelation."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    
    # Find all active lags
    active_mask = b >= M - 1e-10 * M
    active_lags = np.where(active_mask)[0]
    
    # Subgradient: average of gradients of active constraints
    # dc_k/da_m = a_{m+k} (if m+k < n) + a_{m-k} (if m >= k)
    # Note: k here is the lag in the full convolution, 0-indexed
    # Full conv: (a*a)_k = sum_j a_j * a_{k-j} for valid j
    # dc_k/da_m = a_{k-m} (if 0 <= k-m <= n-1) + a_{m-k} (if 0 <= m-k <= n-1)
    #             = a_{|k-m|} when m is within the valid range
    
    # Actually, let me be more careful.
    # (a*a)_k = sum_j a_j * a_{k-j} where j in [0,n-1] and k-j in [0,n-1]
    # = sum_j a_j * a_{k-j} for j in [max(0, k-n+1), min(n-1, k)]
    
    # dc_k/da_m:
    # a_m appears as a_j (j=m) if m is in the valid range: contributes a_{k-m}
    # a_m appears as a_{k-j} (k-j=m, j=k-m) if k-m is in [0,n-1]: contributes a_{k-m}
    # Wait, these are the same thing.
    
    # Let me think differently. 
    # c_k = sum_j a_j * a_{k-j}
    # dc_k/da_m = a_{k-m} * 1[valid] + a_{k-m} * 1[valid]  ... no
    # dc_k/da_m = (d/da_m) sum_j a_j * a_{k-j}
    # = (d/da_m) [term where j=m: a_m * a_{k-m}] + (d/da_m) [term where k-j=m, i.e., j=k-m: a_{k-m} * a_m]
    # But these are the same term! So it's just a_{k-m} (counted once)
    # UNLESS m = k-m, i.e., m = k/2, in which case it's 2*a_m * (d/da_m) ... no
    # Actually c_k = sum_j a_j * a_{k-j}. The term a_m * a_{k-m} appears once (when j=m)
    # and also when j=k-m we get a_{k-m} * a_m which is the same.
    # Wait no: when j = m, we get a_m * a_{k-m}.
    # When j = k-m, we get a_{k-m} * a_{k-(k-m)} = a_{k-m} * a_m.
    # These are the same term! So c_k = sum_j a_j * a_{k-j} counts each pair once.
    # dc_k/da_m = a_{k-m} if 0 <= k-m <= n-1, else 0.
    # Plus: if k-m = m (i.e., m = k/2), the term is a_m^2, and d(a_m^2)/da_m = 2*a_m.
    # But in the sum, the term j=m gives a_m * a_{k-m}. If k-m = m, this is a_m^2.
    # The derivative d(a_m * a_m)/da_m = 2*a_m.
    # But in the sum representation, it's just one term. So dc_k/da_m = a_{k-m} + (a_m if k-m=m else 0)
    # = a_{k-m} + a_m * 1[k=2m]
    
    # Hmm, this is getting complicated. Let me just use the matrix formulation.
    # A[k, j] = a_{k-j} if 0 <= k-j <= n-1, else 0
    # c = A @ a
    # dc_k/da = A[k, :]
    
    # So the subgradient of max(c) at a is:
    # s = (1/|S|) * sum_{k in S} A[k, :]
    # where S is the set of active lags.
    
    A = build_conv_matrix(a)
    s = A[active_lags, :].mean(axis=0)
    
    a_new = a - alpha * s
    a_new = np.clip(a_new, 0, None)
    ss = a_new.sum()
    if ss < 1e-10:
        return a
    a_new = a_new / ss
    return a_new
