```python
import time

def autocorr(a):
    """Compute full autocorrelation using FFT for efficiency."""
    n = len(a)
    size = 2 * n
    fa = np.fft.fft(a, size=size)
    ac = np.fft.ifft(fa * fa).real[:2*n-1]
    # The full convolution is: ac[-(n-1):][::-1] + ac
    # Actually, np.convolve(a,a) gives [a*n-1, ..., a*0, a*1, ..., a*n-1]
    # Let me just use np.convolve for correctness and speed (n<=1000 is fine)
    return np.convolve(a, a)

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    """Build constraint matrix A for LP: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A
```

```python
"""
Optimization for minimizing 2*n*max(autocorr)/(sum)^2 using:
1. LP alternating with adaptive constraints and momentum-based updates
2. FFT-accelerated subgradient with Nesterov momentum
3. Random perturbation restarts to escape local minima
4. Multi-length search (n from 200 to 800)

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. A*g <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None
```

```python
def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf

def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    """Build the Toeplitz matrix A where (A @ g)[k] = (a*g)[k] = sum_j a[m]*g[k-m] valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # (a*g)[k] = sum_j a[j] * g[k-j] where 0<=j<n and 0<=k-j<n
        # = sum_j a[j] * g[k-j]
        # For a fixed k, valid j range: max(0, k-n+1) <= j <= min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = a[k - j]
    return A

def momentum_subgradient(a_init, n_steps=50000, time_budget=600.0, verbose=False):
    """Subgradient method with heavy-ball momentum."""
    n = len(a_init)
    a = normalize(a_init.copy())
    v = np.zeros(n)  # momentum
    beta = 0.9  # momentum coefficient
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    
    step_base = 1.0 / np.sqrt(n)
    
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = np.convolve(a, a)
        max_val = conv.max()
        
        # Find active set (lagrange relaxation)
        tol = max_val * 0.01
        active = conv >= max_val - tol
        n_active = np.sum(active)
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            # grad_k[m] = 2*a[k-m] if 0<=k-m<n else 0
            m_lo = max(0, k - n + 1)

            grad[m_lo:m_hi+1] += 2.0 * a[idx]
        
        grad /= n_active
        
        # Adaptive step size (diminishing with periodic restart)
        progress = step / max(1, n_steps)
        step_size = step_base / (1 + 0.1 * progress)
        
        # Momentum update
        v = beta * v + grad
        a_new = a - step_size * v
        a = normalize(a_new)
        
        obj = compute_obj(a)

            best_obj = obj

        if step % 5000 == 0 and step > 0:
            v = np.zeros(n)
            beta = min(0.95, beta + 0.01)  # Increase momentum slightly
    
    return best_a, best_obj
```

```python
"""
Optimization minimizing 2*n*max(autocorr)/(sum)^2 using:
1. LP alternating with adaptive constraints and line search
2. Momentum-based subgradient with Polyak step size
3. Simulated annealing on LP constraint tightness
4. Random perturbation restarts to escape local minima
5. Multi-length search (n = parent, 300, 400, 500, 600, 800)

import numpy as np
from scipy.optimize import linprog
from scipy.special import i0
import time


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A @ g)[k] = sum_j a[j]*g[k-j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_for_g(a, M):
    """Solve LP: max 1^T g s.t. A_a @ g <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def kaiser_window(n, alpha):
    """Kaiser window."""
    j = np.arange(n, dtype=np.float64)
    x = 2.0 * j / (n - 1) - 1.0
    r = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    w = i0(alpha * r) / i0(alpha)
    return w


def make_chirp(n, m=1, amp=0.3):
    """1 + amp*cos(pi*m*j^2/n) clipped to non-negative."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(np.pi * m * j * j / float(n))
    return np.maximum(a, 0.1)


def make_quadratic_phase(n, depth=1.0):
    """Exponential quadratic: exp(-depth*(j/n - 0.5)^2 * 4)."""
    j = np.arange(n, dtype=np.float64)

def lp_anneal(a_init, max_iter=300, time_budget=60.0):
    """LP alternating with annealed constraint tightness and line search."""
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    n = len(a)

    # Anneal: start relaxed, get tighter

    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        # Determine relaxation based on iteration progress

        if progress < 0.1:

            relax = 1.1

def momentum_subgradient(a_init, time_budget=300.0, n_max_steps=2000000):
    """Subgradient with heavy-ball momentum and Polyak step size."""
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    v = np.zeros(n)
    beta = 0.9
    best_a = a.copy()
    best_obj = compute_obj(a)
    f_star_lower = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound
    start = time.monotonic()

    # For tracking momentum restarts
    steps_no_improve = 0
    improve_threshold = 0.0001

    for step in range(n_max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = np.convolve(a, a)
        max_val = float(np.max(conv))
        obj = 2.0 * n * max_val  # Since sum(a)=1

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            steps_no_improve = 0
        else:
            steps_no_improve += 1

        # Find active set with adaptive tolerance
        tol = max_val * (0.005 + 0.015 * min(1.0, step / 10000.0))
        active = conv >= max_val - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = np.zeros(len(conv), dtype=bool)
            active[np.argmax(conv)] = True
            n_active = 1

        # Compute subgradient via correlation
        b = np.zeros(2 * n - 1)
        active_indices = np.where(active)[0]
        b[active_indices] = 1.0
        sg = 2.0 / n_active * np.correlate(b, a, mode='valid')

        # Polyak step size (with dampening)
        obj_current = compute_obj(a)
        gap = obj_current - f_star_lower
        if gap < 0.001:
            gap = 0.001
        sg_norm_sq = float(np.dot(sg, sg))
        if sg_norm_sq < 1e-15:
            break
        step_size = gap / (sg_norm_sq + 1e-10)
        # Cap step size to avoid instability
        step_size = min(step_size, 0.5 / (1.0 + 0.01 * step))

        # Momentum update
        v = beta * v + sg
        a_new = a - step_size * v
        a = normalize(a_new)

        # Periodic momentum restart with perturbation
        if steps_no_improve > 20000:
            # Add small random perturbation
            rng_local = np.random.default_rng(step)
            a = normalize(a + 0.02 * rng_local.standard_normal(n))
            v = np.zeros(n)
            steps_no_improve = 0

        # Gradually increase momentum
        if step % 50000 == 0 and step > 0:
            beta = min(0.95, beta + 0.005)

    return best_a, best_obj


def subgradient_multi_restart(a_init, time_budget=300.0):
    """Run subgradient with multiple perturbation restarts."""
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    # Each segment gets a fraction of the budget
    n_segments = 5
    seg_time = time_budget / n_segments

    for seg in range(n_segments):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        a_cur, obj_cur = momentum_subgradient(a, time_budget=max(1.0, seg_time * 0.9))

            best_a = a_cur.copy()

        # Perturb for next segment
        sigma = 0.03 * (1.0 + 0.5 * seg)
        a = normalize(best_a + sigma * rng.standard_normal(n))

    return best_a, best_obj
```
