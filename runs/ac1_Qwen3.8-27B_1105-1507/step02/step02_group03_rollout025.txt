from scipy.optimize import linprog

def normalize(x):
    s = x.sum()
    if s < 1e-15:
        return x
    return x / s

def compute_score(f):
    """Score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()

def subgradient_pressure(f, n_peaks=5, alpha=1.0):
    """Compute pressure vector based on top peaks in convolution."""
    conv = np.convolve(f, f)
    n = len(f)
    
    # Find top peaks
    sorted_idx = np.argsort(conv)[::-1]
    top_peaks = sorted_idx[:n_peaks]
    
    # Weight by how much above average each peak is
    avg = conv.mean()
    weights = np.zeros(len(conv))
    for k in top_peaks:
        weights[k] = max(0, conv[k] - avg) ** alpha
    
    # Compute pressure: for each i, pressure[i] = sum_k weights[k] * f[k-i] (valid range)
    pressure = np.zeros(n)
    for i in range(n):
        for k in top_peaks:
            j = k - i
            if 0 <= j < n:
                pressure[i] += weights[k] * f[j]
    
    return pressure, conv

def lp_direction(f, M=None, rng=None, time_limit=2):
    """Solve LP to find best direction. Returns (g, active_set)."""
    n = len(f)
    conv = np.convolve(f, f)
    if M is None:
        M = conv.max()
    
    # Build A matrix: A @ b = convolve(f, b)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = f[k-j]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None, []
    
    if not res.success or res.x is None or res.x.sum() < 1e-10:
        return None, []
    
    # Find active constraints
    b = np.clip(res.x, 0, None)
    conv_fb = np.convolve(f, b)
    active = np.where(np.isclose(conv_fb, M, atol=M*1e-6 + 1e-15))[0]
    
    return b, active.tolist()

def block_sa(f, rng, n_iter=10000, T0=0.02, T_end=0.00001):

    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0

        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()
        r = rng.random()
        
        if r < 0.4:
            # Block multiplicative update
            width = max(1, int(rng.integers(1, max(2, n//8))))
            start = int(rng.integers(0, n - width))
            f_new[start:start+width] *= np.exp(rng.normal(0, 0.3))
        elif r < 0.6:
            # Smooth perturbation
            center = int(rng.integers(0, n))
            width = max(2, n // 16)
            kernel = np.exp(-0.5 * (np.arange(n) - center)**2 / width**2)
            f_new += rng.normal(0, 0.05) * kernel
        elif r < 0.8:
            # Pair swap
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        else:
            # Global rescale with noise
            f_new *= np.exp(rng.normal(0, 0.2, n))

        s = f_new.sum()

        new_score = compute_score(f_new)

            f = f_new
            cur_score = new_score
            if new_score < best_score:

                f = f_new
                improved = True

        if not improved:
            t *= 0.5
            if t < 0.003:
                f_pert = f + rng.normal(0, 0.008, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.15
                else:
                    break

    return best_f, best_score


def subgradient_pressure(f, n_peaks=5, power=1.0):
    conv = np.convolve(f, f)
    n = len(f)
    sorted_idx = np.argsort(conv)[::-1]
    top = sorted_idx[:n_peaks]
    avg = conv.mean()
    pressure = np.zeros(n)
    for k in top:
        w = max(0, conv[k] - avg) ** power
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                pressure[i] += w * f[j]
    return pressure, conv


def momentum_subgradient(f, n_iter=1000, eta0=0.05, momentum=0.9, n_peaks=5, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    v = np.zeros(n)
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        eta = eta0 / (1.0 + 0.005 * it)
        pressure, conv = subgradient_pressure(f, n_peaks=n_peaks, power=1.0)
        if pressure.max() > 1e-15:
            pressure = pressure / pressure.max()
        v = momentum * v + (1 - momentum) * pressure
        vn = np.linalg.norm(v)
        if vn > 1e-15:
            v = v / vn

        improved = False
        for step in [eta, eta * 0.5, eta * 2.0, eta * 0.25]:
            f_new = f - step * v
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            sc = compute_score(f_new)
            if sc < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = sc
                f = f_new
                v *= 0.3
                improved = True
                break
            elif sc < compute_score(f):
                f = f_new
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                f_pert = f * (1 + 0.05 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    v = np.zeros(n)
                    stall = 0
                else:
                    break

    return best_f, best_score


def spectral_shaping(f, n_iter=500, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        n_bins = len(mag)
        current_energy = np.sum(f ** 2)
        target_mag = np.sqrt(n_fft * current_energy / n_bins)
        blend = min(1.0, 0.05 + 0.1 * it / n_iter)
        mag_new = mag.copy()
        mag_new[0] = 1.0
        mag_new[1:] = (1 - blend) * mag[1:] + blend * target_mag
        if it % 100 == 99:
            phase = phase + rng.normal(0, 0.01, len(phase))
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        sc = compute_score(f_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break

    return best_f, best_score


def fast_coordinate_descent(f, n_sweeps=10, top_k=100, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        conv = np.convolve(f, f)
        M = conv.max()
        pressure = np.zeros(n)
        for k in range(2 * n - 1):
            if conv[k] > M * 0.95:
                w = (conv[k] - M * 0.95)
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        pressure[i] += w * f[j]

        top_indices = np.argsort(-pressure)[:top_k]

        for i in top_indices:
            if f[i] < 1e-10:
                continue
            original = f[i]
            best_val = original
            best_local = best_score
            for factor in [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.5, 0.3]:
                val = original * factor
                if val < 0:
                    continue
                f[i] = val
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                sc = compute_score(f / s)
                if sc < best_local:
                    best_local = sc
                    best_val = val
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = normalize(f)

    return f, best_score


def block_sa(f, rng, n_iter=5000, T0=0.01, T_end=0.00001):
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()
        r = rng.random()
        if r < 0.4:
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.2))
        elif r < 0.7:
            center = int(rng.integers(0, n))
            width = max(2, n // 20)
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
            f_new += rng.normal(0, 0.03) * kernel
        else:
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 3000 and it > n_iter // 2:
            break

    return best_f, best_score


def diverse_init(n, rng):
    r = rng.random()
    if r < 0.3:
        f = np.ones(n) * (1 + 0.03 * rng.normal(0, 1, n))
    elif r < 0.5:
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n)) + 0.1
    elif r < 0.7:
        f = np.ones(n)
        for _ in range(3):
            center = int(rng.integers(0, n))
            width = int(rng.integers(n // 10, n // 4))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / max(1, width ** 2))
            f *= (1 + 0.2 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05
    else:
        f = rng.random(n)
        k = max(1, n // 32)
        f = np.convolve(f, np.ones(k) / k, mode='same')
        f = f + 0.1
    f = np.clip(f, 0.01, None)
    f = normalize(f)
    return f
