LP water-filling, parametric initialization, and simulated annealing.

import numpy as np
from scipy.optimize import linprog

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def compute_conv_fft(a):
    """Compute autocorrelation via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    power = fa * np.conj(fa)
    conv = np.fft.irfft(power, N)[:2*n-1]
    return conv

def power_mean_and_grad(a, p):
    """Compute power mean of autocorrelation and its gradient."""
    n = len(a)
    conv = compute_conv_fft(a)
    
    # Clip small negative values from FFT numerical errors
    conv = np.maximum(conv, 0.0)
    
    if p == 1:
        pm = float(np.sum(conv))
        # gradient of sum(conv) = 2*a convolved... actually sum(conv) = S^2
        # so gradient is 2*S (constant). Not useful.
        return pm, 2.0 * np.sum(a) * np.ones(n)
    
    cp = conv ** p
    sum_cp = float(np.sum(cp))
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pm = sum_cp ** (1.0/p)
    
    w = conv ** (p - 1)
    factor = (p / sum_cp) * sum_cp ** (1.0/p)  # = p * sum_cp^(1/p - 1)
    # Actually: d/dx (sum x^p)^(1/p) = (1/p) * (sum x^p)^(1/p - 1) * p * x^(p-1) = (sum x^p)^(1/p-1) * x^(p-1)
    # So d(pm)/da[i] = (sum_cp)^(1/p - 1) * sum_k w[k] * dc[k]/da[i]
    
    factor = sum_cp ** (1.0/p - 1.0)
    
    # dc[k]/da[i] = 2*a[k-i] when 0 <= k-i < n
    # grad[i] = factor * sum_k w[k] * 2*a[k-i] (valid indices)
    # = 2*factor * sum_{j=0}^{n-1} w[i+j] * a[j] (where i+j < 2n-1)
    
    # Use FFT: g[i] = sum_j a[j] * w[i+j]
    # = ifft(conj(fft(a, N2)) * fft(w, N2)).real[i] where N2 >= n + (2n-1) - 1
    N2 = 1
    while N2 < n + (2*n - 1):
        N2 *= 2
    
    fa2 = np.fft.rfft(a, N2)
    fw2 = np.fft.rfft(w, N2)
    g = np.fft.irfft(np.conj(fa2) * fw2, N2)[:n]
    
    grad = 2.0 * factor * g
    return pm, grad

    """Adam optimization on power-mean objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    
    s_a = float(np.sum(a))
    if s_a < 1e-10:
        return a, np.inf
    a = a / s_a
    best_obj = compute_obj(a)

    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        # Linearly interpolate p from p_init to p_final
        frac = min(1.0, t / max(1, int(max_time / 0.01)))
        p = p_init + (p_final - p_init) * frac
        
        pm, grad = power_mean_and_grad(a, p)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        v_hat = v / (1 - beta2 ** t)

def lp_waterfill(a, max_iters=50, max_time=30.0):
    """LP-based water-filling iteration."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        # Build constraint matrix: A[k, j] = a[k-j] when valid
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            j_start = max(0, k - n + 1)
            j_end = min(n-1, k)
            for j in range(j_start, j_end+1):
                A[k, j] = a[k-j]
        
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': max_time - (time.monotonic()-start)})
        except Exception:
            break
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = float(np.sum(g))
        if g_sum < 1.0 + 1e-6:
            break
        
        g = g / g_sum
        
        # Try different step sizes
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.5/g_sum]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:

                a = a_new / s_new
