def compute_score(f):
    """Compute 2n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def normalize_simplex(f):
    """Normalize f to sum to 1 and be non-negative."""
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s < 1e-15:
        return np.ones_like(f) / len(f)
    return f / s


def subgradient_fft(f, percentile=50.0):
    """Compute weighted subgradient using FFT-based correlation.
    Returns (gradient, active_set_size, max_conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    threshold = float(np.percentile(conv, percentile))
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    excess = conv[active] - threshold
    total = float(np.sum(excess))
    if total < 1e-15:
        w = np.ones(len(active)) / len(active)
    else:
        w = excess / total
    # Build u vector of length 2n-1
    u = np.zeros(2 * n - 1)
    u[active] = w
    # grad[i] = 2 * sum_j u[i+j] * f[j] = 2 * correlate(u, f, 'valid')[i]
    grad = 2.0 * np.correlate(u, f, mode='valid')
    return grad, len(active), M

        f_prev = f.copy()

        L = 2.0 * float(n)

def simplex_project(z):
    """Project z onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    # Find tau such that sum(max(z - tau, 0)) = 1
    # Sort z in descending order
    z_sorted = np.sort(z)[::-1]
    # Find the largest rho such that z_sorted[rho] - sum(z_sorted[:rho+1])/(rho+1) > 0
    cumulative = np.cumsum(z_sorted)
    vals = (cumulative - 1.0) / np.arange(1, n + 1)
    positive_mask = z_sorted > vals
    if not np.any(positive_mask):
        return np.ones(n) / n
    rho = int(np.max(np.where(positive_mask)[0])) + 1
    tau = (cumulative[rho - 1] - 1.0) / rho
    return np.maximum(z - tau, 0.0)

            sum_g = float(np.sum(g))
            if sum_g > 1e-10:
                g = g / sum_g
                # Line search
                best_local_score = best_score
                best_local_f = None
                # Try a range of alphas
                alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
                for alpha in alphas:
                    candidate = (1.0 - alpha) * f + alpha * g
                    candidate = normalize_simplex(candidate)
                    sc = compute_score(candidate)
                    if np.isfinite(sc) and sc < best_local_score:
                        best_local_score = sc
                        best_local_f = candidate

                if best_local_f is not None:
                    f = best_local_f
                    best_f = f.copy()
                    best_score = best_local_score
                    no_improve = 0
                    if it % 100 == 0:
                        print(f"    {verbose_prefix}iter {it}: score={best_score:.6f} "
                              f"(n_cons={n_constraints}, lp_time={dt:.3f}s)")
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive constraint count
        if no_improve % 20 == 0 and no_improve > 0:
            if no_improve < 60:
                n_constraints = max(30, n_constraints // 2)
            else:
                n_constraints = min(300, n_constraints * 2)

        # Restart if stuck
        if no_improve > 80:
            print(f"    {verbose_prefix}Restart at iter {it}, score={best_score:.6f}")
            f = best_f.copy()
            # Add perturbation
            pert_size = min(0.05, 0.01 * no_improve / 20)
            f = normalize_simplex(f + rng.uniform(-pert_size, pert_size, n))
            no_improve = 0
            n_constraints = 60

        if it > 5000 and best_score < 1.505:
            # Very good, do fine polishing with more constraints
            n_constraints = min(500, n_constraints * 3)

    return best_f, best_score


def make_initializations(n, rng, parent=None):
    """Create diverse initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Parent
    if parent is not None:
        if len(parent) == n:
            inits.append(('parent', parent.copy()))
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(('parent_interp', np.interp(x_new, x_old, parent)))

    # Cosine window
    cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    inits.append(('cosine', cos_t))

    # Slightly tapered uniform
    taper = 1.0 - 0.3 * (np.abs(2.0 * idx / (n - 1) - 1.0) ** 2)
    inits.append(('tapered', taper))

    # Random
    r = rng.uniform(0.8, 1.2, size=n)
    inits.append(('random', r))

    # Two-scale: random at coarse level, interpolate
    coarse = rng.uniform(0.5, 1.5, size=n // 8)
    fine = np.interp(idx, np.linspace(0, n - 1, n // 8), coarse)
    inits.append(('two_scale', fine))

    return inits


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence minimizing the objective."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)
    n = 1024

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(2024)

    print(f"Starting optimization with budget {budget_s}s, n={n}")

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).copy()
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = p
            ps = compute_score(p)
            print(f"Parent: n={p.size}, score={ps:.6f}")
            if ps < best_score:
                best_score = ps
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent: {e}")

    # Phase 1: Quick exploration (first 120 seconds)
    print("\n=== Phase 1: Quick Exploration ===")
    exp_deadline = t0 + 120.0
    candidates = []

    all_inits = make_initializations(n, rng, parent)

    for label, f_init in all_inits:
        if time.monotonic() > exp_deadline - 20:
            break
        f_init = np.clip(f_init.astype(np.float64), 0.0, None)
        if np.sum(f_init) < 1e-10:
            continue
        f_init = normalize_simplex(f_init)
        init_score = compute_score(f_init)

        # Quick Nesterov run (up to 25 seconds each)
        qd = min(exp_deadline - 5, time.monotonic() + 25.0)
        f_opt, s_opt = nesterov_subgradient(f_init, qd, lr_init=0.05,
                                             top_percentile=70.0,
                                             verbose_prefix=f"  [{label}] ")
        if s_opt < init_score:
            init_score = s_opt
            f_init = f_opt

        print(f"  {label}: {init_score:.6f}")
        candidates.append((init_score, f_init.copy(), label))

        if init_score < best_score:
            best_score = init_score
            best_sequence = f_init.copy()

    # Sort and pick top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates:")
    for sc, seq, lbl in candidates[:5]:
        print(f"  {lbl}: {sc:.6f}")
    print(f"Best after exploration: {best_score:.6f}")

    # Phase 2: Focused optimization on best candidate (next ~1000 seconds)
    print(f"\n=== Phase 2: Focused Optimization (best={best_score:.6f}) ===")
    focus_deadline = t0 + (deadline - t0) * 0.75

    # Use multi-scale optimization on the best candidate
    f_best = candidates[0][1].copy()
    f_opt, s_opt = multi_scale_optimize(f_best, focus_deadline, verbose_prefix="  [multi] ")
    if s_opt < best_score:
        best_score = s_opt
        best_sequence = f_opt.copy()
        print(f"  Improved to: {best_score:.6f}")

    # Phase 3: Try second-best candidate if time remains
    if time.monotonic() < deadline - 100 and len(candidates) > 1:
        print(f"\n=== Phase 3: Second candidate ===")
        f-second = candidates[1][1].copy()
        d3 = min(deadline - 50, time.monotonic() + 200.0)
        f_opt2, s_opt2 = multi_scale_optimize(f_second, d3, verbose_prefix="  [2nd] ")
        if s_opt2 < best_score:
            best_score = s_opt2
            best_sequence = f_opt2.copy()
            print(f"  Second candidate improved to: {best_score:.6f}")

    # Phase 4: Final polish (last 150 seconds)
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}) ===")
        f_pol = best_sequence.copy()
        f_pol = normalize_simplex(f_pol)

        # Final LP with many constraints
        d4 = deadline - 5
        f_opt4, s_opt4 = lp_optimize(f_pol, d4, num_constraints=200,
                                      verbose_prefix="  [polish] ")
        if s_opt4 < best_score:
            best_score = s_opt4
            best_sequence = f_opt4.copy()

        # Quick subgradient polish
        if time.monotonic() < d4 - 5:
            f_opt5, s_opt5 = nesterov_subgradient(best_sequence, d4, lr_init=0.01,
                                                   top_percentile=90.0,
                                                   verbose_prefix="  [sub-polish] ")
            if s_opt5 < best_score:
                best_score = s_opt5
                best_sequence = f_opt5.copy()

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(n)
        best_score = compute_score(best_sequence)

    # Verify with official evaluator
    final_list = [float(x) for x in best_sequence]
    final_score = evaluate_sequence(final_list)
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed! Using fallback.")
        best_sequence = np.ones(n)
        final_list = [float(x) for x in best_sequence]
        final_score = evaluate_sequence(final_list)

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return final_list
