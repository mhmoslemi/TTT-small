5. Repeat

def active_set_step(a, M, max_active=50):
    """One step of the active set method.
    Linearize the top constraints and solve a QP.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = np.sum(a)
    
    # Find active constraints (top max_active positions sorted by conv value)
    k_max = max_active
    active = np.argsort(conv)[::-1][:k_max]
    
    # Build the linearized constraint matrix
    # For each active position k: conv[k] + sum_i 2*a[k-i]*da[i] <= M
    # (where a[k-i] = 0 if k-i out of range)
    A = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                A[idx, i] = 2.0 * a[ki]
    
    # Current conv values at active positions
    b = M - conv[active]  # We need: A @ da <= b
    
    # QP: minimize ||da||^2 subject to A @ da <= b, sum(da) = 0, a + da >= 0
    # Use scipy.optimize.minimize with method 'SLSQP' or trust-constr
    
    # Actually, let me use a simpler approach: projected gradient on the linearized problem.
    # Or just use scipy's minimize.
    
    from scipy.optimize import minimize
    
    def qp_obj(da):
        return np.dot(da, da)
    
    def qp_grad(da):
        return 2.0 * da
    
    constraints = [
        {'type': 'ineq', 'fun': lambda da: b - A @ da, 'jac': lambda da: -A},
        {'type': 'eq', 'fun': lambda da: np.sum(da), 'jac': lambda da: np.ones(n)}
    ]
    bounds = [(-a[i], None) for i in range(n)]  # a + da >= 0 => da >= -a
    
    result = minimize(qp_obj, np.zeros(n), jac=qp_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 100, 'ftol': 1e-10})
    
    if result.success:
        a_new = a + result.x
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new) * sum_a  # preserve sum
        return a_new
    return a

def compute_conv(a):
    """Compute linear self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_obj_grad(a, tau):
    """Compute log-sum-exp smoothed objective and its gradient."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Numerically stable log-sum-exp
    shifted = (conv - max_conv) / tau
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    
    if exp_sum < 1e-300:
        G = max_conv
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        G = max_conv + tau * np.log(exp_sum)
        weights = exp_vals / exp_sum
    
    sum_a = float(np.sum(a))
    obj = 2.0 * n * G / (sum_a * sum_a)
    
    # Gradient computation
    # G'(a)[i] = (2/tau) * sum_{m=0}^{n-1} weights[i+m] * a[m]
    corr = np.correlate(weights, a, mode='full')
    grad_G = (2.0 / tau) * corr[n-1:n-1+n]
    
    # Full gradient of obj
    grad = (2.0 * n / (sum_a ** 3)) * (grad_G * sum_a - 2.0 * G)
    
    return obj, grad

        return val

        _, g = smooth_obj_grad(x, tau)
        return g

                      bounds=bounds, options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol})
    return result.x
