    if s < 0.01:

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast vectorized subgradient with Nesterov acceleration
2. LP constraint-generation polish (reduced active set)
3. L-BFGS-B refinement
4. Multi-length exploration with resampling and random restarts
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize a to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def subgradient_direction(a, top_k=10):
    """Compute subgradient of max(b) with respect to a, using top-k active constraints."""
    n = len(a)
    b = fast_conv(a)
    k = min(top_k, len(b))
    # Get indices of top-k values
    if k == len(b):
        top_idx = np.arange(len(b))
    else:
        top_idx = np.argpartition(b, -k)[-k:]
    
    sg = np.zeros(n)
    for ki in top_idx:
        ki = int(ki)
        # sg[j] += 2 * a[ki-j] for valid j
        jl = max(0, ki - n + 1)
        jh = min(n - 1, ki)
        if jl <= jh:
            # a[ki-j] for j in [jl, jh] = a[ki-jh : ki-jl+1][::-1]
            sg[jl:jh+1] += 2.0 * a[ki - jh : ki - jl + 1][::-1]
    sg /= k
    return sg, b


def nesterov_subgradient(a_init, max_iters, deadline, tracker, top_k=10):
    """Nesterov-accelerated subgradient descent with adaptive step size."""
    a = normalize(a_init.copy())
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    v = np.zeros(n)  # momentum
    t_nest = 1.0
    lr = 0.1
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        t_new = (1 + np.sqrt(1 + 4 * t_nest**2)) / 2
        beta = (t_nest - 1) / t_new
        t_nest = t_new
        
        a_look = a + beta * v
        
        # Compute subgradient at lookahead point
        a_look = np.maximum(a_look, 0)
        s = a_look.sum()
        if s < 1e-10:
            a_look = a
        else:
            a_look = a_look / s
        
        sg, b = subgradient_direction(a_look, top_k)
        
        # Update
        v = a_look - a - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        J_new = compute_J(a_new)
        
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive step size
        if no_improve > 20:
            lr *= 0.7
            no_improve = 0
            # Reset momentum
            t_nest = 1.0
            v = np.zeros(n)
        
        if it % 50 == 0:
            tracker.update(a)
    
    if compute_J(best_a) < current_J:
        current_J = compute_J(best_a)
        tracker.update(best_a)
        a = best_a
    else:
        tracker.update(a)
    
    return a


def build_reduced_lp(a, M, threshold=0.85):
    """Build LP with only active constraints (b_k >= threshold * M)."""
    n = len(a)
    b = fast_conv(a)
    # Active set
    active = np.where(b >= threshold * M)[0]
    
    # Build constraint matrix: for each active k, row is [a[k-j] for valid j]
    rows = []
    cols = []
    vals = []
    for idx, ki in enumerate(active):
        ki = int(ki)
        jl = max(0, ki - n + 1)
        jh = min(n - 1, ki)
        for j in range(jl, jh + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[ki - j])
    
    from scipy.sparse import lil_matrix
    A = lil_matrix((len(active), n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    A = A.tocsr()
    
    return A, np.full(len(active), M), active


def lp_polish(a, max_iters, deadline, tracker):
    """LP-based polish with constraint generation."""
    a = normalize(a.copy())
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Build reduced LP
        A, b_ub, active = build_reduced_lp(a, M, threshold=0.85)
        
        c = -np.ones(n)
        bounds = [(0.0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not r.success or r.x is None:
            break
        
        g = np.maximum(r.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Choose step size
        best_t = 0.1
        best_J = current_J
        for t in [0.1, 0.2, 0.3, 0.5]:
            a_trial = normalize((1 - t) * a + t * g)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = t
        
        if best_t == 0.1:  # no improvement
            # Try smaller steps
            found = False
            for t in [0.05, 0.02, 0.01]:
                a_trial = normalize((1 - t) * a + t * g)
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
                    found = True
                    break
            if not found:
                break
        else:
            a = normalize((1 - best_t) * a + best_t * g)
            current_J = best_J
        
        if it % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.01)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

        inits.append(1.0 + 0.5 * np.cos(2 * np.pi * freq * t_arr + rng.uniform(0, 6.28)))
    
    # Window functions
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t_arr)))  # raised cosine
    inits.append((0.5 * (1 + np.cos(2 * np.pi * t_arr)))**2)  # squared
    inits.append(np.sin(np.pi * t_arr)**2)
    inits.append(np.exp(-4 * (t_arr - 0.5)**2) + 0.05)
    
    # Random smooth
    coeffs = rng.normal(0, 1, min(n, 20))
    inits.append(1.0 + 0.3 * np.sum(coeffs[i] * np.cos(2*np.pi*i*t_arr) for i in range(len(coeffs))))
    
    # Asymmetric
    inits.append(1.0 + 0.4 * (2*t_arr - 1))
    inits.append(np.exp(-t_arr**2 * 3) + 0.1)
    
    return [np.maximum(a, 0.01) for a in inits]

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated subgradient with adaptive step sizes
2. LP constraint-generation polish (reduced active set)
3. L-BFGS-B gradient refinement
4. Multi-length exploration with resampling and random restarts
5. Multigrid-style warm start (optimize short, resample long, re-optimize)
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize_arr(a):
    """Normalize a to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def subgradient_direction(a, top_k=10):
    """Compute subgradient of max(b) w.r.t. a using top-k active constraints."""
    n = len(a)
    b = fast_conv(a)
    k = min(top_k, len(b))
    if k == len(b):
        top_idx = np.arange(len(b))
    else:
        top_idx = np.argpartition(b, -k)[-k:]

    sg = np.zeros(n)
    for ki in top_idx:
        ki = int(ki)
        jl = max(0, ki - n + 1)
        jh = min(n - 1, ki)
        if jl <= jh:
            sg[jl:jh + 1] += 2.0 * a[ki - jh:ki - jl + 1][::-1]
    sg /= k
    return sg, b


def nesterov_subgradient(a_init, max_iters, deadline, tracker, top_k=8):
    """Nesterov-accelerated subgradient descent with adaptive step size."""
    a = normalize_arr(a_init.copy())
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    v = np.zeros(n)
    t_nest = 1.0
    lr = 0.15
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        t_nest = t_new

        a_look = a + beta * v
        a_look = np.maximum(a_look, 0.0)
        s = a_look.sum()
        if s < 1e-10:
            a_look = a.copy()
        else:
            a_look = a_look / s

        sg, b = subgradient_direction(a_look, top_k)

        v = a_look - a - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 30:
            lr *= 0.6
            no_improve = 0
            t_nest = 1.0
            v = np.zeros(n)
        elif it % 100 == 0 and no_improve == 0:
            lr = min(0.5, lr * 1.05)

        if it % 100 == 0:
            tracker.update(a)

    if best_J < current_J:
        current_J = best_J
        a = best_a
    tracker.update(a)
    return a


def build_reduced_constraints(a, M, threshold=0.80):
    """Build sparse constraint matrix for active constraints only."""
    n = len(a)
    b = fast_conv(a)
    active = np.where(b >= threshold * M)[0]

    if len(active) == 0:
        active = np.where(b >= 0.5 * M)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])

    rows = []
    cols = []
    vals = []
    for idx, ki in enumerate(active):
        ki = int(ki)
        jl = max(0, ki - n + 1)
        jh = min(n - 1, ki)
        for j in range(jl, jh + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[ki - j])

    A = lil_matrix((len(active), n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    A = A.tocsr()

    b_ub = np.full(len(active), M)
    return A, b_ub


def lp_polish(a, max_iters, deadline, tracker, threshold=0.80):
    """LP-based polish with constraint generation and adaptive steps."""
    a = normalize_arr(a.copy())
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        A, b_ub = build_reduced_constraints(a, M, threshold)

        c = -np.ones(n)
        bounds = [(0.0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        if not r.success or r.x is None:
            break

        g = np.maximum(r.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        best_t = 0.0
        best_J = current_J
        for t in [0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
            a_trial = normalize_arr((1 - t) * a + t * g)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = t

        if best_t == 0.0:
            stagnant += 1
            if stagnant > 5:
                break
        else:
            a = normalize_arr((1 - best_t) * a + best_t * g)
            current_J = best_J
            stagnant = 0

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.005)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

        inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t_arr + rng.uniform(0, 6.28)))

    inits.append((0.5 * (1.0 + np.cos(2.0 * np.pi * t_arr))) ** 2)
    inits.append(np.sin(np.pi * t_arr) ** 2)
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0))
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.1)
    inits.append((t_arr + 0.1) ** rng.uniform(-0.5, 0.5))

    # Smooth random
    n_harm = min(n, 15)
    coeffs = rng.normal(0, 0.2, n_harm)
    sig = np.zeros(n)
    for i in range(1, n_harm):
        sig += coeffs[i] * np.cos(2.0 * np.pi * i * t_arr)
    inits.append(1.0 + sig)

    return [np.maximum(a, 0.01) for a in inits]
