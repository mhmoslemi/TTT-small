"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Direct peak-minimizing LP (binary search on M)
2. Greedy peak reduction
3. LP mirror descent with geometric line search
4. Multiple lengths and diverse initializations

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(a):
    """Build matrix C where C @ g = conv(a, g). C is (2n-1) x n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return C

def lp_min_peak(a, M):
    """Check if there exists g >= 0, sum(g) = 1, such that conv(a,g)_k <= M for all k.
    Returns g if feasible, None otherwise."""
    n = len(a)
    c = np.zeros(n)  # minimize 0 (feasibility)
    C = build_conv_matrix(a)
    # conv(a, g)_k <= M for all k
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    # sum(g) = 1
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n
    
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        return np.maximum(r.x, 0)
    return None

def lp_min_peak_value(a):
    """Binary search for the minimum M such that there exists g >= 0, sum(g)=1, conv(a,g)_k <= M."""
    n = len(a)
    b = fast_conv(a)
    M_lo = b.min()
    M_hi = b.max()
    
    for _ in range(30):
        M_mid = (M_lo + M_hi) / 2
        g = lp_min_peak(a, M_mid)
        if g is not None:
            M_hi = M_mid
        else:
            M_lo = M_mid
    
    g = lp_min_peak(a, M_hi)
    if g is None:
        return None, M_hi
    return g, M_hi

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse matrix C where C @ g = conv(a, g). Shape: (2n-1, n)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return C


def lp_peak_min(a):
    """Find g >= 0, sum(g)=1 minimizing max_k conv(a,g)_k.
    Returns (g, min_peak_value)."""
    n = len(a)
    C = build_conv_matrix(a)
    num_c = 2 * n - 1
    
    # Variables: [g_0,...,g_{n-1}, t]
    # Minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: conv(a,g)_k - t <= 0  =>  [C, -I][g;t] <= 0
    I_mat = eye(num_c, format='csr')
    A_ub = hstack([C, -I_mat], format='csr')
    b_ub = np.zeros(num_c)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                bounds=bounds, method='highs', options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x[:n], 0.0)
        if g.sum() < 1e-15:
            return None, np.inf
        g /= g.sum()
        t = float(r.x[n])
        return g, t
    return None, np.inf


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def coordinate_descent_polish(a, max_iters, deadline, tracker):
    """Coordinate descent: iteratively adjust one element to reduce J."""

    a = a / a.sum()

For small t, the dominant term is (1-t)^2 * b_a + 2t(1-t) * b_ag ≈ b_a + 2t * b_ag (to first order in t). So minimizing the peak to first order is equivalent to minimizing max_k (b_a[k] + 2t * b_ag[k]), which is approximately minimizing max_k b_ag[k] (since b_a is fixed).

1. Use the new LP to find the direction g
2. Use a small step size t to avoid the quadratic term
3. As the iterations progress, the step size should be small enough that the first-order approximation holds

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def fast_conv_xy(a, y):
    """Cross-convolution of a and y via FFT."""
    n = len(a)
    m = len(y)
    s = 1
    while s < n + m - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fy = np.fft.rfft(y, s)
    return np.fft.irfft(fa * fy, s)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse matrix C where C @ g = conv(a, g). Shape: (2n-1, n)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return C


def lp_peak_min(a):
    """Find g >= 0, sum(g)=1 minimizing max_k conv(a,g)_k. Returns (g, min_peak)."""
    n = len(a)
    C = build_conv_matrix(a)
    num_c = 2 * n - 1

    c = np.zeros(n + 1)
    c[-1] = 1.0

    I_mat = eye(num_c, format='csr')
    A_ub = hstack([C, -I_mat], format='csr')
    b_ub = np.zeros(num_c)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                bounds=bounds, method='highs', options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x[:n], 0.0)
        gs = g.sum()
        if gs < 1e-15:
            return None, np.inf
        g = g / gs
        t = float(r.x[n])
        return g, t
    return None, np.inf


def lp_max_sum(a, M):
    """Old formulation: max sum(g) s.t. conv(a,g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    r = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def run_mirror_descent(a, max_iters, deadline, tracker, use_momentum=False, lp_mode='peak_min'):
    """Mirror descent with LP-based direction finding."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    momentum_v = np.zeros(n) if use_momentum else None
    beta_mom = 0.4
    stagnant = 0
    base_step = 0.15

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Find direction via LP
        if lp_mode == 'peak_min':
            g, _ = lp_peak_min(a)
        else:
            b = fast_conv(a)
            M = float(b.max())
            g = lp_max_sum(a, M)
            if g is not None:
                g = g / g.sum()

        if g is None:
            break

        direction = g - a

        if use_momentum:
            momentum_v = beta_mom * momentum_v + (1.0 - beta_mom) * direction
            step_dir = momentum_v
        else:
            step_dir = direction

        # Geometric line search
        decay = max(0.3, 1.0 - 0.5 * it / float(max(1, max_iters)))
        best_alpha = 0.0
        best_J = current_J
        max_alpha = base_step * (1.0 + 0.5 * decay)

        alphas = [max_alpha, max_alpha * 0.7, max_alpha * 0.5, max_alpha * 0.3,
                  max_alpha * 0.15, max_alpha * 0.05]

        for alpha in alphas:
            if alpha <= 0:
                continue
            a_trial = a + alpha * step_dir
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_alpha = alpha

        if best_alpha == 0.0:
            stagnant += 1
            if stagnant > 30:
                # Try random perturbation to escape
                a_trial = a + 0.005 * np.random.randn(n)
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J:
                        a = a_trial
                        current_J = J_trial
                        stagnant = 0
                    else:
                        break
                else:
                    break
            else:
                # Try tiny step
                a_trial = a + 0.005 * step_dir
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J:
                        a = a_trial
                        current_J = J_trial
                        stagnant = 0
                    elif stagnant > 10:
                        break
        else:
            stagnant = 0
            a = a + best_alpha * step_dir
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
                current_J = best_J

        if it % 30 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_exp_reweighting(a, max_iters, deadline, tracker):
    """Exponential reweighting: g_i = a_i * exp(-eta * (conv(a,a)_peak_contribution_i)/max)."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        peak_locs = np.where(b >= 0.95 * M)[0]

        # Compute "pressure" on each element
        pressure = np.zeros(n)
        for k in peak_locs:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                ki = k - j
                if 0 <= ki < n:
                    pressure[j] += a[j] * a[ki]

        max_pressure = pressure.max()
        if max_pressure < 1e-15:
            break

        # Exponential reweighting
        eta = 0.5 * (1.0 - 0.3 * it / float(max(1, max_iters)))
        weights = np.exp(-eta * pressure / max_pressure)
        g = a * weights
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Line search
        best_alpha = 0.0
        best_J = current_J
        for alpha in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_trial = (1 - alpha) * a + alpha * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_alpha = alpha

        if best_alpha == 0.0:
            break

        a = (1 - best_alpha) * a + best_alpha * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def peak_reduction_polish(a, deadline, tracker, max_rounds=100):
    """Targeted: reduce elements that contribute most to the peak."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        peak_k = int(np.argmax(b))

        # Find elements contributing to peak
        jl = max(0, peak_k - n + 1)
        jh = min(n - 1, peak_k)
        contributions = {}
        for j in range(jl, jh + 1):
            ki = peak_k - j
            if 0 <= ki < n:
                contributions[j] = a[j] * a[ki]

        if not contributions:
            break

        # Try reducing the top contributors
        sorted_contribs = sorted(contributions.items(), key=lambda x: x[1], reverse=True)
        improved = False

        for idx, (j, contrib) in enumerate(sorted_contribs[:5]):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.7, 0.5, 0.3, 1.3]:
                a[j] = max(old_val * factor, 1e-8)
                s = a.sum()
                if s < 1e-10:
                    a[j] = old_val
                    continue
                J_trial = compute_J(a / s)
                if J_trial < current_J - 1e-13:
                    a = a / s
                    current_J = J_trial
                    improved = True
                    break
            if improved:
                a[j] = a[j]  # keep the change
            else:
                a[j] = old_val

        if not improved:
            # Try increasing some non-contributing elements
            b = fast_conv(a)
            M = b.max()
            for j in range(n):
                if j not in contributions and a[j] < 0.5 * a.max():
                    old_val = a[j]
                    a[j] = old_val * 2.0
                    s = a.sum()
                    if s > 1e-10:
                        J_trial = compute_J(a / s)
                        if J_trial < current_J - 1e-13:
                            a = a / s
                            current_J = J_trial
                            improved = True
                            break
                    a[j] = old_val
            if not improved:
                break

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Mirror of parent
        inits.append(resample_seq(parent[::-1], n))

    t_arr = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.4, 1.0, n))

    # Cosine harmonics
    for freq in [1, 2, 3, 4, 5, 7]:
        inits.append(np.maximum(1.0 + 0.35 * np.cos(2 * np.pi * freq * t_arr + rng.uniform(0, 1)), 0.01))

    # Gaussian bumps
    inits.append(np.exp(-5.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.exp(-3.0 * (t_arr - 0.3) ** 2) + 0.05)
    inits.append(np.exp(-3.0 * (t_arr - 0.7) ** 2) + 0.05)
    inits.append(np.exp(-8.0 * (t_arr - 0.5) ** 2) + 0.02)

    # Asymmetric power laws
    inits.append((1.0 - t_arr + 0.1) ** 0.7 + 0.05)
    inits.append((t_arr + 0.1) ** 0.7 + 0.05)
    inits.append((1.0 - t_arr + 0.1) ** (-0.3) + 0.05)

    # Triangle / tent
    inits.append(1.0 - 2.0 * np.abs(t_arr - 0.5) + 0.1)
    inits.append(1.0 - 2.0 * np.abs(t_arr - 0.4) + 0.1)

    # Multi-peak
    inits.append(1.0 + 0.3 * np.cos(4 * np.pi * t_arr) * np.exp(-3 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(1.0 + 0.4 * np.cos(6 * np.pi * t_arr) * np.exp(-2 * (t_arr - 0.5) ** 2) + 0.1)

    # W-shaped (parabola)
    inits.append((t_arr - 0.5) ** 2 * 4.0 + 0.2)

    # Random sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    return inits

    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase6 FinalLP n={n_best} {'='*50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = max(30, int(remaining / 0.08))
        max_iters = min(max_iters, 500)
        a = run_mirror_descent(tracker.best, max_iters, deadline, tracker,
                               use_momentum=True, lp_mode='peak_min')
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
