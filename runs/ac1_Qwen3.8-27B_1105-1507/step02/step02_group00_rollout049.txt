```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP ping-pong alternation, smooth logsumexp gradient polishing,
and golden-section coordinate refinement across multiple sequence lengths.
Key improvement over parent: dual LP ping-pong, adaptive smooth polishing
with increasing beta, and better time allocation.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build matrix A where (A @ x)[k] = (a*x)[k] (full convolution)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # A[k, j] = a[k-j] for j in [j_start, j_end]
        js = np.arange(j_start, j_end + 1)
        A[k, js] = a[k - js]
    return A


def solve_lp(fixed, M):
    """Solve LP: max sum(x) s.t. (fixed*x)[k] <= M for all k, x >= 0."""
    n = len(fixed)
    A = build_conv_matrix(fixed)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def smooth_gradient(a, beta):
    """Compute gradient of (1/beta)*logsumexp(beta*conv(a,a)) w.r.t. a.
    Returns gradient vector of length n."""
    n = len(a)
    conv = np.convolve(a, a)  # length 2n-1
    # Softmax weights
    beta_conv = beta * conv
    max_bc = np.max(beta_conv)
    exp_bc = np.exp(beta_conv - max_bc)
    w = exp_bc / np.sum(exp_bc)  # length 2n-1
    
    # grad[m] = sum_j a[j] * w[m+j] for m=0..n-1
    # Build index matrix
    m_idx = np.arange(n)[:, None]  # (n,1)
    j_idx = np.arange(n)[None, :]  # (1,n)
    w_mat = w[m_idx + j_idx]  # (n,n), w_mat[m,j] = w[m+j]
    grad = np.sum(a[None, :] * w_mat, axis=1)
    return grad * 2.0  # factor of 2 from the derivative of convolution


def smooth_polish(a, n_steps=2000, beta_start=50.0, beta_end=500.0, verbose=False):
    """Smooth projected gradient using logsumexp approximation.
    Minimize max(conv(a,a)) subject to sum(a)=1, a>=0."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for step in range(n_steps):
        # Anneal beta
        frac = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac

        conv = np.convolve(a, a)
        max_conv = np.max(conv)

        if max_conv < 1e-15:
            break

        # Compute smooth gradient
        grad = smooth_gradient(a, beta)

        # Adaptive step size
        # Use a step that's proportional to the "spread" of a
        a_spread = np.max(a) - np.min(a)
        step_size = 0.3 * a_spread / (1.0 + 0.05 * step)

        # Gradient step
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

        if verbose and step % 200 == 0:
            print(f"    Smooth it={step}, beta={beta:.0f}, obj={best_obj:.6f}")

    return best_a, best_obj


def golden_section_1d(f, lo, hi, tol=1e-8, max_iter=50):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    a = hi - (hi - lo) / gr
    b = lo + (hi - lo) / gr
    fa = f(a)
    fb = f(b)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fa < fb:
            hi = b
            b = a
            fb = fa
            a = hi - (hi - lo) / gr
            fa = f(a)
        else:
            lo = a
            a = b
            fa = fb
            b = lo + (hi - lo) / gr
            fb = f(b)
    return (lo + hi) / 2.0


def coord_refine_golden(a, n_rounds=2, verbose=False):
    """Coordinate descent with golden section search on each coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            old_val = a[j]

            def f_1d(x):
                a[j] = max(0.0, x)
                s = np.sum(a)
                if s < 1e-10:
                    return 1e10
                return compute_obj(a / s)

            # Search range
            center = old_val
            width = max(center * 0.5, 0.01)
            lo = max(0.0, center - width)
            hi = center + width

            opt_x = golden_section_1d(f_1d, lo, hi, tol=1e-9, max_iter=30)
            a[j] = max(0.0, opt_x)

            s = np.sum(a)
            if s < 1e-10:
                a[j] = old_val
                continue
            a = a / s

            obj = compute_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                improved = True

        if not improved and rnd > 0:
            break

    return best_a, best_obj


def lp_ping_pong(a_init, max_iter=200, verbose=False, time_limit=None):
    """LP ping-pong: alternate between optimizing a (given g) and g (given a)."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start_time = time.monotonic()

    for it in range(max_iter):
        if time_limit is not None and time.monotonic() - start_time > time_limit:
            break

        # Step 1: Given a, find best g (max sum(g) s.t. (a*g) <= max(a*a))
        conv_a = np.convolve(a, a)
        M_a = float(np.max(conv_a))
        
        g = solve_lp(a, M_a)
        if g is None or np.sum(g) < 1e-10:
            break
        g = g / np.sum(g)
        obj_g = compute_obj(g)
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()

        # Step 2: Given g, find best a_new (max sum(a) s.t. (g*a) <= max(g*g))
        conv_g = np.convolve(g, g)
        M_g = float(np.max(conv_g))

        a_new = solve_lp(g, M_g)
        if a_new is None or np.sum(a_new) < 1e-10:
            break
        a_new = a_new / np.sum(a_new)
        obj_a_new = compute_obj(a_new)
        if obj_a_new < best_obj:
            best_obj = obj_a_new
            best_a = a_new.copy()

        # Blend to avoid oscillation
        t = 0.7
        a = (1.0 - t) * a + t * a_new
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-10:
            a = a_new
        else:
            a = a / s

        # Also check blend of a and g
        blend = 0.5 * a + 0.5 * g
        blend = np.maximum(blend, 0.0)
        s_blend = np.sum(blend)
        if s_blend > 1e-10:
            blend = blend / s_blend
            obj_blend = compute_obj(blend)
            if obj_blend < best_obj:
                best_obj = obj_blend
                best_a = blend.copy()

        if verbose and it % 20 == 0:
            print(f"    LP-PP it={it}: best_obj={best_obj:.6f}")

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    j = np.arange(n)
    if choice == 0:
        # Raised cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1)))
        a = a + 0.3 * rng.random(n)
    elif choice == 1:
        # Gaussian window
        sigma = rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((j - n / 2) / (sigma * n)) ** 2)
    elif choice == 2:
        # Cosine with random frequency
        freq = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * j / n + phase))
        a = a + 0.1 * rng.random(n)
    elif choice == 3:
        # Two-lobe
        mid = n // 2
        a = np.ones(n)
        a[:mid] = rng.uniform(0.5, 1.5)
        a[mid:] = rng.uniform(0.5, 1.5)
    elif choice == 4:
        # Linear taper
        taper = rng.uniform(0.1, 0.5)
        a = np.ones(n)
        a[:int(taper * n)] = np.linspace(0.3, 1.0, int(taper * n))
        a[-int(taper * n):] = np.linspace(1.0, 0.3, int(taper * n))
    elif choice == 5:
        # Quadratic
        a = 1.0 - 0.5 * ((2.0 * j / (n - 1) - 1.0)) ** 2
        a = np.maximum(a, 0.1)
    elif choice == 6:
        # Periodic
        period = int(rng.integers(max(3, n // 20), max(4, n // 5)))
        pattern = np.maximum(0.1, rng.random(period))
        a = np.tile(pattern, n // period + 1)[:n]
    else:
        # Random with smoothing
        a = rng.random(n)
        # Simple boxcar smooth
        kernel = np.ones(3) / 3.0
        a = np.convolve(a, kernel, mode='same')

    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: LP ping-pong + smooth logsumexp polishing + golden-section
    coordinate refinement. Tries multiple sequence lengths with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Sequence lengths to try
    n_values = [200, 250, 300, 350, 400]

    # Time allocation: ~70% LP, ~20% smooth, ~10% coord
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        print(f"\n=== N={n}: {remaining:.0f}s remaining, current best={best_score:.6f} ===")

        n_restarts = 6 if n <= 300 else 4

        for restart in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break

            # Choose initialization
            if restart == 0 and best_seq is not None and len(best_seq) == n:
                a_init = best_seq.copy()
            else:
                a_init = structured_init(n, rng)

            # LP ping-pong
            lp_time = remaining * 0.5 / max(1, n_restarts - restart)
            a, obj = lp_ping_pong(a_init, max_iter=300,
                                   time_limit=min(lp_time, 60.0))

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP-PP n={n} r={restart}: obj={obj:.6f}")

            # Smooth polishing
            if time.monotonic() < deadline - 60:
                time_left = deadline - time.monotonic()
                n_steps = int(min(3000, time_left / 0.05))
                if n_steps > 200:
                    a2, obj2 = smooth_polish(a, n_steps=n_steps,
                                              beta_start=50.0, beta_end=300.0)
                    if obj2 < best_score:
                        best_score = obj2
                        best_seq = a2.copy()
                        print(f"  Smooth n={n} r={restart}: obj={obj2:.6f}")

    # Phase 2: Final refinement on the best solution
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Phase 2: Refining n={n_best}, obj={best_score:.6f}, {remaining:.0f}s left ===")

        # More aggressive smooth polishing with high beta
        if n_best <= 400:
            n_steps = int(min(5000, remaining * 0.3 / 0.04))
            if n_steps > 500:
                a_sm, obj_sm = smooth_polish(best_seq, n_steps=n_steps,
                                              beta_start=100.0, beta_end=1000.0,
                                              verbose=True)
                if obj_sm < best_score:
                    best_score = obj_sm
                    best_seq = a_sm.copy()
                    print(f"  Final smooth: obj={obj_sm:.6f}")

        # One more LP ping-pong pass on the refined solution
        if time.monotonic() < deadline - 30 and n_best <= 400:
            lp_left = min(30.0, (deadline - time.monotonic()) * 0.2)
            a_lp, obj_lp = lp_ping_pong(best_seq, max_iter=200,
                                         time_limit=lp_left)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP-PP: obj={obj_lp:.6f}")

        # Coordinate refinement with golden section
        if time.monotonic() < deadline - 15 and n_best <= 350:
            time_left_coord = deadline - time.monotonic() - 5
            n_rounds = int(min(3, time_left_coord / (n_best * 0.01)))
            if n_rounds > 0:
                a_cr, obj_cr = coord_refine_golden(best_seq, n_rounds=n_rounds)
                if obj_cr < best_score:
                    best_score = obj_cr
                    best_seq = a_cr.copy()
                    print(f"  Final coord: obj={obj_cr:.6f}")

                # One more smooth pass after coord
                if time.monotonic() < deadline - 10:
                    n_steps2 = int(min(2000, (deadline - time.monotonic()) / 0.04))
                    if n_steps2 > 200:
                        a_ms, obj_ms = smooth_polish(best_seq, n_steps=n_steps2,
                                                      beta_start=200.0, beta_end=2000.0)
                        if obj_ms < best_score:
                            best_score = obj_ms
                            best_seq = a_ms.copy()
                            print(f"  Post-coord smooth: obj={obj_ms:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
```
