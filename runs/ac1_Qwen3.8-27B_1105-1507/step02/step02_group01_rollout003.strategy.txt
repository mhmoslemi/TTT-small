def build_A_base(a: np.ndarray) -> np.ndarray:
    """Build an A matrix for constraints for linear program."""
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)

        j_start = max(0, i - n + 1)
        j_end = min(n-1, i)

    return A
