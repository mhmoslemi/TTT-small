    # Let's use: np.correlate(weights, a, 'full')
    # np.correlate(v, a, 'full')[k] = sum_n v[n+k-(len(a)-1)] * a[n] ... this is confusing.
    #
    # Let me just use a direct approach with FFT for cross-correlation:
    # cross_corr[i] = sum_m a[m] * weights[m+i]
    # Define c = a (padded) and w = weights (padded)
    # cross_corr[i] is the i-th element of the cross-correlation of a with weights.
    # In signal processing: (a ⋆ w)[i] = sum_m a[m] * conj(w[m+i])
    # For real: = sum_m a[m] * w[m+i] ✓
    #
    # FFT computation of cross-correlation:
    # (a ⋆ w) = ifft(fft(a) * conj(fft(w)))
    # But we need the right indexing. Let's use:
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(weights, size)
    # Cross-correlation: ifft(fft(a) * conj(fft(w)))
    corr = np.fft.irfft(fa * np.conj(fw), size)
    # corr[k] = sum_m a[m] * weights[m - k] (circular, but we padded enough)
    # We want sum_m a[m] * weights[m + i] = corr[-i mod size] = corr[size - i] for i > 0
    # Actually: corr[k] = sum_m a[m] * w[(m-k) mod size]
    # We want: sum_m a[m] * w[m+i] = sum_m a[m] * w[(m-(-i)) mod size] = corr[-i mod size]
    # For i in [0, n-1]: corr[(size-i) % size] = corr[size-i] for i>0, corr[0] for i=0

    grad_a = np.zeros(n)
    for i in range(n):
        idx = (size - i) % size
        grad_a[i] = 2.0 * corr[idx]

    # Actually, let me verify with a simpler approach for correctness:
    # Use the Toeplitz-like structure directly for small n, FFT for large n
    # For now, let me use a verified method:
    # grad_a[i] = 2 * sum_{m=0}^{n-1} a[m] * weights[m+i] (weights indexed 0..2n-2, out of range = 0)
    # This is exactly the "cross-correlation" and can be computed as:
    # Let's build it via: for each m, a[m] contributes to grad_a[i] for i = 0,...,2n-2-m
    # grad_a[i] += 2 * a[m] * weights[m+i]

    # The FFT approach above should be correct. Let me double-check the indexing
    # by testing on a tiny example in the code. For now, trust it and verify at runtime.

    return f_val, grad_a


def project_simplex(a):
    """Project a onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(a)
    if np.sum(a) <= 0:
        return np.ones(n) / n
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(a - theta, 0)
    return w


def lbfgs_optimize(a_init, n, tau_start=10.0, tau_end=0.1, max_iter=500, time_limit=60.0):
    """
    Optimize using L-BFGS-B on the parameterization where a[n-1] = 1 - sum(a[0:n-1]).
    Uses adaptive tau (exponential decay).
    """
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    start_time = time.monotonic()
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # We optimize over a[0:n-1], with a[n-1] = 1 - sum(a[0:n-1])
    # Bounds: 0 <= a[i] <= 1 for i = 0,...,n-2

    def objective_and_grad(x):
        # x has length n-1, represents a[0:n-1]
        full_a = np.empty(n)
        full_a[:n-1] = x
        full_a[n-1] = 1.0 - np.sum(x)
        if full_a[n-1] < -1e-10:
            # Infeasible, return large value
            return 1e10, np.zeros(n-1)
        full_a = np.maximum(full_a, 0)
        # Renormalize to fix any numerical drift
        fs = np.sum(full_a)
        if fs < 1e-10:
            return 1e10, np.zeros(n-1)
        full_a = full_a / fs

        # Adaptive tau based on current value
        conv = compute_conv(full_a)
        cur_max = float(np.max(conv))
        # tau relative to the scale of the problem
        tau_cur = max(tau_end, tau_start * cur_max * 0.01)
        # Actually, use a fixed schedule based on iteration
        tau_cur = max(tau_end, tau_start * np.exp(-3.0 * _it_counter[0] / max(1, max_iter)))

        f_val, grad_a = obj_and_grad_from_a(full_a, tau_cur)

        # Gradient w.r.t. x (first n-1 components)
        # a[n-1] = 1 - sum(x), so d(a[n-1])/dx[i] = -1
        # grad_x[i] = grad_a[i] - grad_a[n-1]
        grad_x = grad_a[:n-1] - grad_a[n-1]

        _it_counter[0] += 1
        return f_val, grad_x

    _it_counter = [0]
    x0 = a[:n-1].copy()
    bounds = [(0.0, 1.0)] * (n - 1)

    try:
        res = minimize(objective_and_grad, x0, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                                'gtol': 1e-8, 'ftol': 1e-12,
                                'iprint': 0})
    except Exception as e:
        print(f"    L-BFGS error: {e}")
        return best_a, best_obj

    # Reconstruct
    if res.x is not None:
        full_a = np.empty(n)
        full_a[:n-1] = res.x
        full_a[n-1] = 1.0 - np.sum(res.x)
        full_a = np.maximum(full_a, 0)
        fs = np.sum(full_a)
        if fs > 1e-10:
            full_a = full_a / fs
            obj = 2.0 * n * float(np.max(compute_conv(full_a)))
            if obj < best_obj:
                best_obj = obj
                best_a = full_a.copy()

    return best_a, best_obj


def lbfgs_optimize_v2(a_init, n, time_limit=30.0):
    """
    L-BFGS optimization with multiple tau values (coarse to fine).
    Works directly on a with simplex projection.
    """
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    # Coarse to fine tau
    taun = [float(np.max(compute_conv(a)))]
    for tau_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
        if time.monotonic() - start > time_limit - 2:
            break
        tau = tau_mult * taun[0]
        if tau < 1e-6:
            tau = 1e-6

        def obj_grad(x):
            full_a = np.empty(n)
            full_a[:n-1] = x
            full_a[n-1] = 1.0 - np.sum(x)
            full_a = np.maximum(full_a, 0)
            fs = np.sum(full_a)
            if fs < 1e-10:
                return 1e10, np.zeros(n-1)
            full_a = full_a / fs
            conv = compute_conv(full_a)
            f_smooth = smooth_max_value(conv, tau)
            f_val = 2.0 * n * f_smooth
            weights = smooth_max_weights(conv, tau)
            # Compute gradient
            size = 1
            while size < 2 * n:
                size <<= 1
            fa = np.fft.rfft(full_a, size)
            fw = np.fft.rfft(weights, size)
            corr = np.fft.irfft(fa * np.conj(fw), size)
            grad_a = np.zeros(n)
            for i in range(n):
                idx = (size - i) % size
                grad_a[i] = 2.0 * corr[idx]
            grad_x = grad_a[:n-1] - grad_a[n-1]
            return f_val, grad_x

        x0 = a[:n-1].copy()
        bounds = [(0.0, 1.0)] * (n - 1)

        try:
            res = minimize(obj_grad, x0, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 200, 'maxfun': 400, 'gtol': 1e-10,
                                    'ftol': 1e-14, 'iprint': 0})
            if res.x is not None:
                full_a = np.empty(n)
                full_a[:n-1] = res.x
                full_a[n-1] = 1.0 - np.sum(res.x)
                full_a = np.maximum(full_a, 0)
                fs = np.sum(full_a)
                if fs > 1e-10:
                    full_a = full_a / fs
                    obj = 2.0 * n * float(np.max(compute_conv(full_a)))
                    if obj < best_obj:
                        best_obj = obj
                        best_a = full_a.copy()
                        a = full_a.copy()
        except Exception:
            pass

    return best_a, best_obj


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_step(a, M, max_time=15.0):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g, sum_g) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max_time})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_optimize(a, max_iter=30, t=0.3, time_limit=60.0):
    """Alternating LP optimization."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    stall = 0

    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = lp_step(a, M, max_time=min(20, time_limit - (time.monotonic() - start)))
        if g is None:
            break

        if sum_g <= 1.0 + 1e-6:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0

        g = g / sum_g
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def subgradient_step(a, eta, tau):
    """One subgradient descent step with smoothed max, maintaining sum=1 and a>=0."""
    n = len(a)
    conv = compute_conv(a)
    f_smooth = smooth_max_value(conv, tau)
    weights = smooth_max_weights(conv, tau)

    # Gradient of smoothed max
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(weights, size)
    corr = np.fft.irfft(fa * np.conj(fw), size)
    grad = np.zeros(n)
    for i in range(n):
        idx = (size - i) % size
        grad[i] = 2.0 * corr[idx]

    # Center the gradient (for sum constraint)
    grad_centered = grad - np.mean(grad)

    a_new = a - eta * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a
    a_new = a_new / s
    return a_new


def subgradient_optimize(a, max_iter=5000, tau_start=5.0, tau_end=50.0,
                          eta_start=0.1, time_limit=30.0):
    """Smoothed subgradient with adaptive step."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    stall = 0

    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break

        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac
        eta = eta_start * (1.0 + it) ** (-0.5)

        a = subgradient_step(a, eta, tau)
        obj = 2.0 * n * float(np.max(compute_conv(a)))

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                eta *= 0.5
                if eta < 1e-10:
                    break
                stall = 0

    return best_a, best_obj


def generate_initialization(n, rng, style=0):
    """Generate various initializations."""
    if style == 0:
        # Exponential decay (lopsided - good for this problem)
        rate = rng.uniform(0.5, 3.0)
        a = np.exp(-rate * np.arange(n) / n)
    elif style == 1:
        # Geometric
        r = rng.uniform(0.85, 0.99)
        a = r ** np.arange(n)
    elif style == 2:
        # Triangle (peaked in middle)
        center = n / 2.0
        width = rng.uniform(0.2, 0.8)
        a = np.exp(-((np.arange(n) - center) / (width * n)) ** 2)
    elif style == 3:
        # Random
        a = rng.random(n)
    elif style == 4:
        # Cosine weighted
        freq = rng.uniform(0.5, 3.0)
        a = 0.5 * (1 + np.cos(2 * np.pi * freq * np.arange(n) / n))
        a = np.maximum(a, 0)
    elif style == 5:
        # Two-hump
        c1 = rng.uniform(0.1, 0.4) * n
        c2 = rng.uniform(0.6, 0.9) * n
        w = rng.uniform(0.05, 0.15) * n
        a = np.exp(-((np.arange(n) - c1) / w) ** 2) + np.exp(-((np.arange(n) - c2) / w) ** 2)
    elif style == 6:
        # Linearly decreasing
        a = np.linspace(rng.uniform(0.5, 2.0), 0.01, n)
    elif style == 7:
        # Inverse (concentrated at start)
        a = 1.0 / (1.0 + np.arange(n) * rng.uniform(0.1, 1.0))
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def try_parent_sequence():
    """Try to use the parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 0.01:
                obj = 2.0 * len(parent) * float(np.max(compute_conv(parent))) / (s * s)
                print(f"Parent (n={parent.size}): obj={obj:.6f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_for_n(n, rng, time_budget=30.0, verbose=False):
    """Run full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(6, int(time_budget / 5)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 3:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Generate initialization
        style = init_idx % 8
        a = generate_initialization(n, rng, style)

        if verbose:
            init_obj = 2.0 * n * float(np.max(compute_conv(a)))
            print(f"      n={n} init{init_idx} (style={style}): obj={init_obj:.6f}")

        # Phase 1: L-BFGS optimization (coarse to fine)
        t1 = min(remaining * 0.4, 20.0)
        a_l, obj_l = lbfgs_optimize_v2(a, n, time_limit=t1)
        obj_l = 2.0 * n * float(np.max(compute_conv(a_l)))
        if obj_l < best_obj:
            best_obj = obj_l
            best_a = a_l.copy()

        # Phase 2: Subgradient (different geometry)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5 and (best_a is None or init_idx == 0):
            a_sg, obj_sg = subgradient_optimize(a_l, max_iter=3000,
                                                 tau_start=3.0, tau_end=30.0,
                                                 eta_start=0.08,
                                                 time_limit=min(remaining * 0.3, 15.0))
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                a_l = a_sg.copy()

        # Phase 3: LP polishing
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5 and n <= 500:
            lp_time = min(remaining * 0.5, 25.0)
            a_lp, obj_lp = lp_optimize(a_l, max_iter=30, t=0.3, time_limit=lp_time)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()

        # Phase 4: Final L-BFGS polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 3:
            a_f, obj_f = lbfgs_optimize_v2(best_a, n, time_limit=min(remaining * 0.6, 10.0))
            obj_f = 2.0 * n * float(np.max(compute_conv(a_f)))
            if obj_f < best_obj:
                best_obj = obj_f
                best_a = a_f.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = 2.0 * n * (1.0 / n)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-scale convex optimization with L-BFGS + subgradient + LP polishing.
    Explores sequence lengths from 30 to 600, focusing on lopsided initializations
    that exploit the non-palindromic structure for lower peaks.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"  Starting from parent: n={len(best_seq)}, score={best_score:.6f}")

    # Sequence lengths to explore
    n_values = [30, 40, 50, 60, 80, 100, 120, 150, 180, 200, 250, 300, 350, 400, 500]

    # Add parent's length and neighbors
    if parent is not None:
        n_p = len(parent)
        n_values = sorted(set(n_values + [n_p, max(20, n_p - 30), min(600, n_p + 30)]))

    print(f"\nExploring n values: {n_values}")
    print(f"Time budget: {float(budget_s):.0f}s, deadline in {deadline - start_time:.0f}s\n")

    # First pass: quick exploration
    for n in n_values:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Allocate time based on n (larger n gets more time in later passes)
        if n <= 60:
            alloc = min(15.0, remaining * 0.06)
        elif n <= 150:
            alloc = min(25.0, remaining * 0.08)
        elif n <= 300:
            alloc = min(40.0, remaining * 0.1)
        else:
            alloc = min(50.0, remaining * 0.12)

        alloc = max(8.0, alloc)
        print(f"  n={n}: {remaining:.0f}s left, allocating {alloc:.0f}s")

        a_opt, obj = optimize_for_n(n, rng, time_budget=alloc)
        # Validate
        test_list = [float(x) for x in a_opt]
        validated = evaluate_sequence(test_list)
        if validated < obj + 1e-6:
            obj = validated  # Use the validated score

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"    ** UPDATE: n={n}, score={best_score:.6f}")
        else:
            print(f"    n={n}: best={obj:.6f} (no update)")

    # Second pass: intensive optimization on the most promising n values
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        # Focus on n_best and nearby values
        focus_ns = sorted(set([n_best, max(20, n_best - 20), min(600, n_best + 20),
                                max(20, n_best - 50), min(600, n_best + 50)]))
        # Filter out already well-explored small ones
        focus_ns = [n for n in focus_ns if n >= 40]

        print(f"\n=== Intensive phase on n values: {focus_ns} ===")

        for n in focus_ns:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            alloc = min(remaining * 0.3, 120.0)
            alloc = max(20.0, alloc)
            print(f"  Intensive n={n}: {remaining:.0f}s left, allocating {alloc:.0f}s")

            a_opt, obj = optimize_for_n(n, rng, time_budget=alloc, verbose=True)
            test_list = [float(x) for x in a_opt]
            validated = evaluate_sequence(test_list)
            if validated < obj + 1e-6:
                obj = validated

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"    ** UPDATE: n={n}, score={best_score:.6f}")

    # Final polishing on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_final = len(best_seq)
        remaining = deadline - time.monotonic()
        print(f"\n=== Final polishing (n={n_final}), {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        a_cur = a_cur / np.sum(a_cur)

        # Round 1: L-BFGS with very fine tau
        a_cur, _ = lbfgs_optimize_v2(a_cur, n_final, time_limit=min(remaining * 0.3, 30.0))
        obj = 2.0 * n_final * float(np.max(compute_conv(a_cur)))
        if obj < best_score:
            best_score = obj
            best_seq = a_cur.copy()
            print(f"  Polish L-BFGS: {best_score:.6f}")

        remaining = deadline - time.monotonic()

        # Round 2: LP with small t (fine adjustments)
        if remaining > 20 and n_final <= 500:
            a_lp, obj_lp = lp_optimize(a_cur, max_iter=50, t=0.15,
                                        time_limit=min(remaining * 0.4, 40.0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_cur.copy() if False else a_lp.copy()
                print(f"  Polish LP: {best_score:.6f}")
            a_cur = a_lp.copy() if obj_lp < best_score else a_cur

        remaining = deadline - time.monotonic()

        # Round 3: Subgradient with high tau
        if remaining > 15:
            a_sg, obj_sg = subgradient_optimize(a_cur, max_iter=5000,
                                                 tau_start=20.0, tau_end=100.0,
                                                 eta_start=0.03,
                                                 time_limit=min(remaining * 0.4, 30.0))
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Polish SG: {best_score:.6f}")
            a_cur = a_sg.copy() if obj_sg < best_score else a_cur

        # Round 4: Another LP
        remaining = deadline - time.monotonic()
        if remaining > 15 and n_final <= 500:
            a_lp2, obj_lp2 = lp_optimize(a_cur, max_iter=40, t=0.2,
                                          time_limit=min(remaining * 0.5, 30.0))
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  Polish LP2: {best_score:.6f}")

        # Round 5: Final L-BFGS
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_fin, _ = lbfgs_optimize_v2(best_seq, n_final, time_limit=min(remaining * 0.7, 20.0))
            obj_fin = 2.0 * n_final * float(np.max(compute_conv(a_fin)))
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"  Polish final L-BFGS: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Final validation
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if np.isinf(validation):
        print("  WARNING: validation returned inf, using fallback")
        result = [1.0] * 50
        validation = evaluate_sequence(result)
    elif abs(validation - best_score) > 0.01:
        print(f"  Validation: computed={best_score:.6f}, validated={validation:.6f}")
        best_score = validation

    print(f"  Final validated score: {validation:.6f}")
    return result
