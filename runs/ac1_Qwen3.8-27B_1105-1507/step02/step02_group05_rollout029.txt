import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def logsumexp_objective(x, beta=50):
    """Smooth approximation of max(conv(x,x)) using log-sum-exp."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    # Log-sum-exp
    b_max = b.max()
    if b_max == 0:
        return 0.0
    log_sum = b_max + np.log(np.sum(np.exp(beta * (b - b_max))))
    return float(log_sum / beta)


def logsumexp_gradient(x, beta=50):
    """Gradient of log-sum-exp smoothed max(conv(x,x))."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return np.zeros_like(x)
    ax = ax / s
    b = fast_conv(ax)
    b_max = b.max()
    weights = np.exp(beta * (b - b_max))
    weights /= weights.sum()
    
    # Gradient of (a*a)_k with respect to a_i
    n = len(ax)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            # d(a*a)_k / da_j = a_{k-j} + a_{k-j} (with care for j = k-j)
            i = k - j
            if 0 <= i < n:
                if j == i:
                    grad[j] += w * 2 * ax[j]  # a_i^2 derivative is 2*a_i

def run_smoothed_lbfgs(a, deadline, tracker, beta=50, max_iter=300):
    """L-BFGS-B with log-sum-exp smoothing."""

            return 1e10

        if time.monotonic() > deadline:
            return np.zeros_like(x)
        return logsumexp_gradient(x, beta)

def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences, emphasizing non-symmetric shapes."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Non-symmetric perturbation of parent
        p = resample_seq(parent, n).copy()
        p = p + 0.2 * np.linspace(0, 1, n)
        inits.append(np.maximum(p, 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Decreasing ramp (non-symmetric)
    inits.append(np.linspace(1.0, 0.1, n))
    
    # Increasing ramp
    inits.append(np.linspace(0.1, 1.0, n))
    
    # Triangle centered (symmetric)
    inits.append(1.0 - np.abs(2.0 * t_arr - 1.0))
    
    # Triangle shifted left (non-symmetric)
    inits.append(1.0 - np.abs(t_arr - 0.3) * 2)
    
    # Triangle shifted right
    inits.append(1.0 - np.abs(t_arr - 0.7) * 2)
    
    # Exponential decay
    inits.append(np.exp(-2.0 * t_arr) + 0.05)
    
    # Exponential growth
    inits.append(np.exp(2.0 * t_arr) + 0.05)
    
    # Cosine variations
    for freq in [1, 2, 3]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))
    
    # Quadratic (convex)
    inits.append(t_arr ** 2 + 0.1)
    
    # Quadratic (concave)
    inits.append((1.0 - t_arr) ** 2 + 0.1)
    
    # Gaussian centered
    inits.append(np.exp(-5.0 * (t_arr - 0.5) ** 2) + 0.05)
    
    # Gaussian shifted
    inits.append(np.exp(-5.0 * (t_arr - 0.3) ** 2) + 0.05)
    inits.append(np.exp(-5.0 * (t_arr - 0.7) ** 2) + 0.05)
    
    # Random
    inits.append(rng.uniform(0.2, 1.0, n))
    
    # Sawtooth
    inits.append(np.mod(t_arr * 5.0, 1.0) + 0.1)
    
    return inits
