def compute_score(f):

def compute_score_fft(f):

        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M):
    """Solve: max sum(g) s.t. g >= 0, conv(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def multiplicative_update(f, eta=0.1, top_k=None, deadline=None):
    """Multiplicative (mirror descent) update to reduce peak autocorrelation."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    no_improve = 0
    it = 0
    
    while (deadline is None or time.monotonic() < deadline) and no_improve < 200:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M <= 0:
            break
        
        # Find active lags
        if top_k is not None:
            active = np.argsort(conv)[-top_k:]
        else:
            threshold = 0.9 * M
            active = np.where(conv >= threshold)[0]
        
        # Compute gradient (sum of shifts of f at active lags)
        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += f[k - j]
        
        # Multiplicative update
        f = f * np.exp(-eta * grad)
        f = np.maximum(f, 1e-15)  # Prevent exact zeros
        f = f * (S / np.sum(f))  # Renormalize
        
        score = compute_score(f)
        if score < best_score - 1e-10:
            best_f = f.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.9  # Reduce step size
        
        it += 1
        if it % 100 == 0:
            print(f"    mult_update iter {it}: score={best_score:.6f} eta={eta:.6f}")
    
    return best_f, best_score


def lp_descent(f, deadline, initial_step=0.1, verbose=False):
    """LP-based descent with line search."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    step = initial_step
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and no_improve < 100:
        conv_current = np.convolve(f, f)
        M = float(np.max(conv_current))
        
        # Solve LP
        t0 = time.monotonic()
        g = solve_lp(f, M)
        lp_time = time.monotonic() - t0
        
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            continue
        
        # Rescale g to match sum of f
        g = g * (S / sum_g)
        
        # Line search
        best_alpha = 0
        best_cand_score = best_score
        best_cand_f = None
        
        # Try several step sizes
        alphas = [step, step * 0.5, step * 0.25, step * 2, step * 0.75, step * 1.5]
        alphas = [min(a, 1.0) for a in alphas if a > 0.01]
        alphas = sorted(set(alphas))
        
        for alpha in alphas:
            cand = (1 - alpha) * f + alpha * g
            cand = np.maximum(cand, 0)
            cand_sum = float(np.sum(cand))
            if cand_sum < 1e-10:
                continue
            cand = cand * (S / cand_sum)
            cand_score = compute_score(cand)
            if cand_score < best_cand_score:
                best_cand_score = cand_score
                best_cand_f = cand
                best_alpha = alpha
        
        it += 1
        
        if best_cand_f is not None:
            f = best_cand_f
            best_f = f.copy()
            best_score = best_cand_score
            no_improve = 0
            # Slightly reduce step
            step = max(0.01, step * 0.99)
            if verbose and it % 50 == 0:
                print(f"    LP iter {it}: score={best_score:.6f} step={step:.4f} alpha={best_alpha:.4f}")
        else:
            no_improve += 1
            step *= 0.7
    
    return best_f, best_score


def logsumexp_optimize(f, deadline, beta_init=10.0, lr=0.01, verbose=False):
    """Optimize using log-sum-exp approximation of max."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    beta = beta_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and no_improve < 300:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        
        # Log-sum-exp weights
        shifted = conv - M  # Shift for numerical stability
        weights = np.exp(beta * shifted)
        weights = weights / np.sum(weights)
        
        # Gradient of log-sum-exp w.r.t. f
        grad = np.zeros(n)
        for k in range(len(conv)):
            if weights[k] < 1e-15:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += 2.0 * weights[k] * f[k - j]
        
        # Project gradient onto tangent space of {sum(x) = S}
        grad = grad - np.mean(grad)
        
        # Update
        new_f = f - lr * grad
        new_f = np.maximum(new_f, 0)
        new_f = new_f * (S / np.sum(new_f))
        
        score = compute_score(new_f)
        it += 1
        
        if score < best_score - 1e-10:
            best_f = new_f.copy()
            best_score = score
            no_improve = 0
            # Increase beta slowly
            beta *= 1.001
        else:
            no_improve += 1
            lr *= 0.95
            beta = max(5.0, beta * 0.99)
        
        if verbose and it % 200 == 0:
            print(f"    LSE iter {it}: score={best_score:.6f} beta={beta:.1f} lr={lr:.6f}")
    
    return best_f, best_score


def make_initializations(n, rng):
    inits = []
    idx = np.arange(n, dtype=np.float64)
    
    inits.append(('uniform', np.ones(n)))
    
    cos_t = (1.0 + np.cos(np.pi * idx / max(n - 1, 1))) / 2.0
    inits.append(('cosine', cos_t))
    
    inits.append(('ramp', np.linspace(1.0, 0.05, n)))
    inits.append(('ramp_inc', np.linspace(0.05, 1.0, n)))
    
    inits.append(('exp_decay', np.exp(-3.0 * idx / n)))
    inits.append(('exp_decay2', np.exp(-3.0 * (n - 1 - idx) / n)))
    
    center = (n - 1) / 2.0
    bell = np.exp(-0.5 * ((idx - center) / (n / 4.0)) ** 2)
    inits.append(('bell', bell))
    
    r = rng.uniform(0.1, 2.0, size=n)
    inits.append(('random', r))
    
    # Triangular
    tri = 1.0 - np.abs(idx - center) / center
    inits.append(('triangular', np.maximum(tri, 0.05)))
    
    # Flat top
    flat = np.ones(n)
    flat[:n//10] = 0.3
    flat[-n//10:] = 0.3
    inits.append(('flat_top', flat))
    
    return inits
