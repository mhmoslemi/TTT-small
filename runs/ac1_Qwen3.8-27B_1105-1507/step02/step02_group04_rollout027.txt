def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def project_simplex(v, s=1.0):
    """Project v onto {x : x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)

"""
High-quality alternating LP + subgradient optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Focuses on aggressive LP refinement of the
parent sequence, supplemented by diverse initializations and spectral polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_simplex(v, s=1.0):
    """Project v onto {x : x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(conv, a, tau):
    """Compute smoothed gradient of max(conv) using log-sum-exp weights."""
    n = len(a)
    max_c = np.max(conv)
    shifted = np.clip(tau * (conv - max_c), -500.0, 500.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    # grad_r = 2 * sum_{m=0}^{n-1} a[m] * w[m+r]
    grad = np.empty(n, dtype=np.float64)
    for r in range(n):
        grad[r] = 2.0 * np.dot(a, w[r:r + n])
    return grad


def subgradient_optimize(a, max_iter=10000, tau_start=10.0, tau_end=100.0,
                         step_init=0.05, momentum=0.85):
    """Smoothed subgradient with momentum and simplex projection."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a, 1.0)

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    velocity = np.zeros(n)
    step = step_init
    stall = 0
    patience = 500

    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        current_obj = compute_obj(a)

        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.7
                if step < 1e-10:
                    break
                noise = np.random.default_rng(it * 31 + 7).normal(0, 0.02, n)
                a = project_simplex(a + noise, 1.0)
                stall = 0
                continue

        grad = smoothed_grad(conv, a, tau)

        velocity = momentum * velocity - step * grad
        a_new = project_simplex(a + velocity, 1.0)

        obj_new = compute_obj(a_new)
        if obj_new > current_obj * 1.05 + 1e-6:
            velocity = velocity * 0.3
            a_new = a

        a = a_new
        step = step_init / (1.0 + 0.005 * it)

    return best_a, best_obj


def solve_lp_given_a(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns (g, sum_g) or (None, None)."""
    n = len(a)
    # (a*g)_k = sum_{m: 0<=k-m<n} a[k-m] * g[m]
    # A[k, m] = a[k-m] if 0 <= k-m <= n-1, else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        if m_min <= m_max:
            A[k, m_min:m_max + 1] = a[k - m_max:k - m_min + 1][::-1]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12,
                               'dual_stopping_tolerance': 1e-12})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_alternating(a, max_iter=200, t_start=0.5, t_end=0.05, verbose=False):
    """Alternating LP: fix a, optimize g via LP, mix."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a, 1.0)

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    stall = 0
    for it in range(max_iter):
        frac = it / max(1, max_iter)
        t = t_start + (t_end - t_start) * frac

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_given_a(a, M)
        if g is None:
            break

        if sum_g <= 1.0 + 1e-8:
            stall += 1
            if stall > 50:
                break
            continue
        stall = 0

        g = g / sum_g
        a_new = (1.0 - t) * a + t * g
        a_new = project_simplex(a_new, 1.0)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose and (it % 50 == 0 or it < 5):
                print(f"    LP it={it}: obj={best_obj:.10f}, sum_g={sum_g:.6f}, t={t:.4f}")
        a = a_new

    return best_a, best_obj

            if n_iter < 10:
                break
            a_cur, obj_cur = lp_alternating(a_cur, max_iter=n_iter,
                                             t_start=t_start, t_end=t_end, verbose=(round_num == 0))
            if obj_cur < best_score:
                best_score = obj_cur
                best_seq = a_cur.copy()
                print(f"  Round {round_num}: {best_score:.10f}")

        # Direct replacement LP
        if time.monotonic() < deadline - 60:
            a_cur, obj_cur = lp_alternating_direct(best_seq, max_iter=200, verbose=True)
            if obj_cur < best_score:
                best_score = obj_cur
                best_seq = a_cur.copy()
                print(f"  Direct LP: {best_score:.10f}")

        # Subgradient polish
        if time.monotonic() < deadline - 30:
            a_sg, _ = subgradient_optimize(best_seq, max_iter=15000,
                                            tau_start=30.0, tau_end=150.0, step_init=0.02)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  SG polish: {best_score:.10f}")

        print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")
        print(f"  Current best: {best_score:.10f}")

    # Phase 2: Explore other lengths
    print(f"\n=== Phase 2: Exploration ===")
    n_values = [80, 100, 120, 150, 180, 200, 250, 300, 350, 400]
    if parent is not None:
        n_parent = len(parent)
        n_values = sorted(set(n_values + [n_parent]))

    for n in n_values:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if parent is not None and n == len(parent):
            alloc = min(60.0, remaining * 0.12)
        elif n <= 150:
            alloc = min(40.0, remaining * 0.1)
        else:
            alloc = min(50.0, remaining * 0.1)
        alloc = max(10.0, alloc)

        print(f"  n={n}: alloc={alloc:.0f}s, remaining={remaining:.0f}s")
        a_opt, obj = run_for_n(n, seed=seed + n * 7919, max_time=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.10f}")

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = len(best_seq)
        print(f"\n=== Phase 3: Final polish (n={n_best}) ===")

        a_cur = best_seq.copy()
        round_num = 0
        while time.monotonic() < deadline - 8:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            # LP
            if n_best <= 500:
                n_iter = int(min(150, remaining // 0.08))
                if n_iter > 5:
                    a_lp, obj_lp = lp_alternating(a_cur, max_iter=n_iter,
                                                   t_start=0.3, t_end=0.03)
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = a_lp.copy()
                        a_cur = a_lp.copy()
                        print(f"  P3-{round_num} LP: {best_score:.10f}")

            # Direct LP
            if n_best <= 500:
                remaining = deadline - time.monotonic()
                if remaining > 10:
                    a_lp, obj_lp = lp_alternating_direct(a_cur, max_iter=int(min(100, remaining // 0.1)))
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = a_lp.copy()
                        a_cur = a_lp.copy()
                        print(f"  P3-{round_num} direct: {best_score:.10f}")

            # Subgradient with high tau (focus on peak)
            remaining = deadline - time.monotonic()
            if remaining > 5:
                a_sg, _ = subgradient_optimize(a_cur, max_iter=int(min(8000, remaining * 4)),
                                                tau_start=80.0, tau_end=300.0, step_init=0.01)
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    a_cur = a_sg.copy()
                    print(f"  P3-{round_num} SG: {best_score:.10f}")

            # Small perturbation to escape local minima
            remaining = deadline - time.monotonic()
            if remaining > 5:
                rng = np.random.default_rng(round_num * 9999 + seed)
                for scale in [0.02, 0.05, 0.1]:
                    noise = np.exp(rng.normal(0, scale, n_best))
                    a_pert = project_simplex(a_cur * noise, 1.0)
                    obj_p = compute_obj(a_pert)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"  P3-{round_num} perturb({scale}): {best_score:.10f}")
                        break

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.10f}, time={elapsed:.1f}s")

    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if np.isfinite(validation):
        if validation < best_score:
            print(f"  Validation improved: {validation:.10f}")
            best_score = validation
        elif abs(validation - best_score) > 1e-6:
            print(f"  Validation discrepancy: computed={best_score:.10f}, validated={validation:.10f}")
            best_score = validation

    return result
