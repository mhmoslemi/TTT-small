"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Compute objective: 2n * max(conv) / sum^2."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 1e-10:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)_k <= M for all k."""
    n = len(f)
    try:
        A = build_conv_matrix_sparse(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success and result.x is not None:
            return np.clip(result.x, 0, None)
    except Exception:
        pass
    return None


def subgradient_direction(f, conv, top_k=5, temperature=0.1):
    """Compute subgradient of max_k conv[k] with softmax-like weights near the peak."""
    n = len(f)
    M = float(np.max(conv))
    if M <= 0:
        return np.zeros(n)

    # Softmax weights: emphasize lags close to the maximum
    scaled = (conv - M) / (M * temperature + 1e-15)
    weights = np.exp(scaled)
    weights /= weights.sum()

    # Only use top-k for efficiency
    top_indices = np.argsort(conv)[-top_k:]
    w_mask = np.zeros(len(conv))
    w_mask[top_indices] = 1.0
    weights *= w_mask
    total_w = weights.sum()
    if total_w < 1e-15:
        return np.zeros(n)
    weights /= total_w

    grad = np.zeros(n)
    for k in top_indices:
        w = weights[k]
        if w < 1e-12:
            continue
        j_min = max(0, k)
        j_max = min(n - 1, k + n - 1)
        for j in range(j_min, j_max + 1):
            other = k - j
            if 0 <= other < n:
                grad[j] += weights[k] * 2.0 * f[other]
    return grad


def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    n = len(v)
    u = v.copy()
    if np.sum(u) < 1e-15:
        return np.ones(n) * (target_sum / n)
    # Sort in descending order
    sorted_u = np.sort(u)[::-1]
    cssv = np.cumsum(sorted_u)
    rho = np.nonzero(sorted_u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) < 1:
        return np.ones(n) * (target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(u - theta, 0.0)

            break
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 1e-15:
            break

        t_lp_start = time.monotonic()
        g = solve_lp_direction(f, M)
        lp_time = time.monotonic() - t_lp_start

        if g is None or np.sum(g) < 1e-15:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        sum_g = float(np.sum(g))
        g = g * (S / sum_g)

        # Line search
        best_step_score = best_score
        best_step_f = None
        alphas = [step, step * 0.5, step * 1.5, step * 0.25, step * 2.0]
        for alpha in alphas:
            alpha = min(alpha, 1.0)
            if alpha < 0.005:
                continue
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = project_simplex(candidate, S)
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate

        it += 1

        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            if verbose and it % 10 == 0:
                elapsed = time.monotonic() - t_start
                print(f"    {verbose_prefix}LP it={it} score={best_score:.6f} step={step:.4f} lp_t={lp_time:.2f}s")
            step = max(step * 0.99, 0.02)
        else:
            no_improve += 1
            step *= 0.8
            if step < 0.005 or no_improve > 30:
                break

    return best_f, best_score


def alternating_optimize(f, deadline, verbose=False, verbose_prefix=""):
    """Alternate LP and subgradient with adaptive time allocation."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    t_start = time.monotonic()
    use_lp = True
    lp_streak = 0
    sub_streak = 0

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            break

        if use_lp:
            # Give LP a chunk of time (one solve)
            sub_deadline = time.monotonic() + min(30.0, remaining * 0.3)
            f, s = lp_optimize(f, sub_deadline, step_init=0.1,
                               verbose=verbose, verbose_prefix=verbose_prefix + "LP:")
            if s < best_score:
                best_score = s
                best_f = f.copy()
            lp_streak += 1
            sub_streak = 0
            if lp_streak >= 5:
                use_lp = False
        else:
            sub_deadline = time.monotonic() + min(20.0, remaining * 0.3)
            f, s = fast_subgradient_optimize(f, sub_deadline, lr_init=0.005,
                                             top_k=7, temperature=0.05,
                                             verbose=verbose, verbose_prefix=verbose_prefix + "SUB:")
            if s < best_score:
                best_score = s
                best_f = f.copy()
            sub_streak += 1
            lp_streak = 0
            if sub_streak >= 3:
                use_lp = True

    return best_f, best_score


def make_initializations(n, rng):
    """Create diverse initializations exploiting structural insights."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform (baseline, score = 2)
    inits.append(('uniform', np.ones(n)))

    # U-shape: high at edges, low in center (reduces center convolution)
    u_shape = 0.3 + 0.7 * (2.0 * idx / (n - 1) - 1.0) ** 2
    inits.append(('u_shape', u_shape))

    # Inverted U (bell): high center, low edges
    bell = 0.2 + 0.8 * np.exp(-0.5 * ((idx - center) / (n / 6.0)) ** 2)
    inits.append(('bell', bell))

    # Cosine taper
    cos_t = 0.1 + 0.9 * (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    inits.append(('cosine', cos_t))

    # Power law decay from left
    for p in [0.5, 1.0, 2.0]:
        pw = (1.0 - idx / (n - 1)) ** p + 0.05
        inits.append((f'pow_{p}', pw))

    # Bimodal
    two_peak = (np.exp(-0.5 * ((idx - n * 0.25) / (n / 10.0)) ** 2) +
                np.exp(-0.5 * ((idx - n * 0.75) / (n / 10.0)) ** 2) + 0.05)
    inits.append(('bimodal', two_peak))

    # Triangular
    tri = 1.0 - np.abs(idx / (n - 1) - 0.5) * 2.0
    tri = tri * 0.7 + 0.3
    inits.append(('tri', tri))

    # Random with structure
    r = rng.uniform(0.1, 1.0, size=n)
    r += 0.3 * np.sin(2 * np.pi * idx / n)
    r = np.clip(r, 0.05, None)
    inits.append(('rand_sin', r))

    # Random smooth
    r2 = rng.uniform(0.1, 1.0, size=n)
    # Smooth with moving average
    kernel = np.ones(20) / 20.0
    r2 = np.convolve(r2, kernel, mode='same')
    r2 = np.clip(r2, 0.05, None)
    inits.append(('rand_smooth', r2))

    # "Anti-concentration": emphasize middle third
    mask = np.zeros(n)
    mask[n//4:3*n//4] = 1.0
    anti = 0.2 + 0.8 * mask * (1.0 - 0.3 * np.exp(-0.5 * ((idx - center) / (n/8.0))**2))
    inits.append(('mid_emph', anti))

    return inits


def interpolate_to_length(seq, new_n):
    """Interpolate sequence to a different length."""
    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence minimizing the evaluation function."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(42)

    best_sequence = None
    best_score = float('inf')

    # Load parent sequence if available
    parent_seq = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent_seq = parent
            score = compute_score(parent)
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"No parent available: {e}")

    # Phase 1: Quick exploration (~250s)
    print("\n=== Phase 1: Quick Exploration ===")
    expl_deadline = t0 + 250.0

    # Determine lengths to try
    parent_n = len(parent_seq) if parent_seq is not None else 1024
    lengths = sorted(set([parent_n, max(256, parent_n // 2), min(4096, parent_n * 2)]))

    candidates = []  # (score, seq, n, label)

    for n in lengths:
        if time.monotonic() > expl_deadline - 50:
            break
        print(f"\n--- Exploring n={n} ---")

        inits = make_initializations(n, rng)

        # Add parent-interpolated
        if parent_seq is not None:
            if n == len(parent_seq):
                inits.append(('parent', parent_seq.copy()))
            else:
                inits.append(('parent_interp', interpolate_to_length(parent_seq, n)))

        for label, f_init in inits:
            if time.monotonic() > expl_deadline - 20:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            if np.sum(f_init) < 1e-10:
                continue

            init_score = compute_score(f_init)
            if not np.isfinite(init_score):
                continue

            # Quick subgradient burst
            quick_dead = min(expl_deadline, time.monotonic() + 20.0)
            f_opt, s_opt = fast_subgradient_optimize(f_init, quick_dead,
                                                      lr_init=0.02, top_k=5,
                                                      temperature=0.15)
            score = min(init_score, s_opt)
            if s_opt < init_score:
                f_init = f_opt

            if score < best_score:
                best_score = score
                best_sequence = f_init.copy()

            candidates.append((score, f_init.copy(), n, label))
            print(f"  {label}: {init_score:.6f} -> {score:.6f}")

    # Sort and show top
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates:")
    for sc, seq, n, lbl in candidates[:8]:
        print(f"  n={n:4d} {lbl:15s}: {sc:.6f}")
    print(f"Best after exploration: {best_score:.6f}")

    # Phase 2: Focused optimization (~1000s)
    print(f"\n=== Phase 2: Focused Optimization (best={best_score:.6f}) ===")
    focus_deadline = deadline - 200.0

    # Pick top diverse candidates (different n or very different scores)
    seen = set()
    top_cands = []
    for sc, seq, n, lbl in candidates:
        key = n
        if key not in seen:
            top_cands.append((sc, seq, n, lbl))
            seen.add(key)
        if len(top_cands) >= 3:
            break

    # Allocate time proportionally
    total_focus_time = focus_deadline - time.monotonic()
    time_per_cand = total_focus_time / len(top_cands)

    for i, (sc, f_init, n, lbl) in enumerate(top_cands):
        if time.monotonic() > focus_deadline - 30:
            break
        alloc = min(time_per_cand, focus_deadline - time.monotonic() - 30)
        print(f"\n--- Focus opt: n={n} {lbl} (start={sc:.6f}, alloc={alloc:.0f}s) ---")

        f_opt, s_opt = alternating_optimize(f_init, time.monotonic() + alloc,
                                             verbose=True, verbose_prefix=f"[n={n} {lbl}] ")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # Phase 3: Final polish (~200s)
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        n_best = len(best_sequence)

        # LP polish
        f_pol, s_pol = lp_optimize(best_sequence, deadline, step_init=0.05,
                                    verbose=True, verbose_prefix="[polishLP] ")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()

        # Subgradient polish with very small steps
        f_pol, s_pol = fast_subgradient_optimize(best_sequence, deadline,
                                                  lr_init=0.002, top_k=10,
                                                  temperature=0.03,
                                                  verbose=True, verbose_prefix="[polishSUB] ")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()

        # Final micro-LP
        f_pol, s_pol = lp_optimize(best_sequence, deadline, step_init=0.02,
                                    verbose=True, verbose_prefix="[microLP] ")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = compute_score(best_sequence)

    # Final verification
    final_check = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_check):
        print("WARNING: Final sequence failed! Using fallback.")
        best_sequence = np.ones(512)
        final_check = evaluate_sequence([float(x) for x in best_sequence])

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_check:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
