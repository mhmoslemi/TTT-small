"""
Minimizes J(a) = 2n*max(conv(a,a))/sum(a)^2 using:
1. LP best-response with adaptive line search (primary driver)
2. Structured restarts using dual-guided directional perturbations
3. Multi-initialization strategy (gappy, cosine, parent-resampled)
4. Log-sum-exp smooth refinement with L-BFGS-B (final polish)
Key improvement: line search for mixing parameter + structured restarts
from binding-constraint analysis to escape local optima.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_solve(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    g = None
    dual = None
    if res.success and res.x.sum() > 1e-10:
        g = res.x
        try:
            if res.ineqlin is not None and hasattr(res.ineqlin, 'marginals'):
                dual = np.clip(res.ineqlin.marginals, 0, None)
        except Exception:
            pass
    return g, dual


def line_search(a, g, best_score):
    best_s = best_score
    best_t = 0.0
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        st = a_t.sum()
        if st < 0.01:
            continue
        a_t = a_t / st
        sc = compute_score(a_t)
        if sc < best_s:
            best_s = sc
            best_t = t
    return best_t, best_s


def structured_perturbed(a, dual, b, rng, strength=0.15):
    n = len(a)
    a_new = a.copy()
    if dual is not None:
        top_idx = np.argsort(dual)[::-1][:max(3, n // 8)]
        for k in top_idx:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            if jl <= jh:
                a_new[jl:jh + 1] *= (1.0 - 0.5 * strength)
    peak_idx = np.argsort(b)[::-1][:max(3, n // 10)]
    for k in peak_idx:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            a_new[jl:jh + 1] *= (1.0 - 0.3 * strength)
    noise = rng.normal(0, 0.03, n)
    a_new += noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 0.01:
        return a.copy()
    return a_new / s
