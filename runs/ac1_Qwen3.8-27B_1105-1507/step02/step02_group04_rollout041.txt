This is like taking a, reversing it, and shifting by (k* - (n-1)).

Let me verify: if k* = n-1 (the middle), then g[j] = 2*a[n-1-j], which is exactly 2*reverse(a).

So for the middle lag, the subgradient is 2*reverse(a). To reduce (a*a)[n-1] = sum a[i]*a[n-1-i], we should decrease a[i] where a[n-1-i] is large, i.e., we should decrease the "anti-correlated" parts.

This makes sense: to reduce the sum of products a[i]*a[n-1-i], we should reduce a[i] where a[n-1-i] is large.

OK, now let me also think about the LP approach. The LP is:
max sum(g) subject to (a*g)[k] <= M for all k, g >= 0.

where M = max conv(a,a).

This finds a direction g that, when mixed with a, doesn't increase the peak. If sum(g) > 1 (with sum(a) = 1), then the mixture (1-t)*a + t*g (normalized) should have a lower peak.

The LP constraint matrix is:
A[k, j] = a[k-j] if 0 <= k-j < n, else 0. (This is the same as the subgradient pattern, without the factor of 2.)

So A is a Toeplitz matrix where A[k,j] = a[k-j] (valid indices).

from scipy.optimize import linprog
import time

def fft_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj_and_peak(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf, 0
    conv = fft_conv(a)
    idx = int(np.argmax(conv))
    obj = 2.0 * n * conv[idx] / (s*s)
    return obj, idx

def subgradient(a, peak_idx):
    n = len(a)
    k = peak_idx
    g = np.zeros(n)
    # g[j] = 2*a[k-j] if 0 <= k-j < n
    valid_j_start = max(0, k - (n-1))
    valid_j_end = min(n-1, k)
    for j in range(valid_j_start, valid_j_end + 1):
        g[j] = 2.0 * a[k - j]
    return g

def project(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        return a
    return a / s

def sg_descent(a, max_iter=10000, step_init=0.1, verbose=False):
    """Subgradient descent with exact peak identification."""
    a = project(a.copy())
    best_a = a.copy()
    best_obj, _ = compute_obj_and_peak(a)
    
    step = step_init
    stall = 0
    patience = 500
    
    for it in range(max_iter):
        obj, peak = compute_obj_and_peak(a)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.5
                if step < 1e-10:
                    break
                # Add noise to escape
                a = a + np.random.normal(0, 0.01, len(a))
                a = project(a)
                continue
        
        g = subgradient(a, peak)
        g_centered = g - np.mean(g)
        a_new = a - step * g_centered
        a = project(a_new)
        
        if it > 100:
            step *= 0.999
        
    return best_a, best_obj

def build_toeplitz(a):
    """Build the constraint matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def lp_step(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def lp_optimize(a, max_iter=50, t=0.3, time_limit_per_lp=15):
    """Alternating LP optimization."""
    n = len(a)
    a = project(a.copy())
    best_a = a.copy()
    best_obj, _ = compute_obj_and_peak(a)
    stall = 0
    
    for it in range(max_iter):
        conv = fft_conv(a)
        M = float(np.max(conv))
        
        g = lp_step(a, M, time_limit=time_limit_per_lp)
        if g is None:
            break
        
        sum_g = np.sum(g)
        if sum_g <= 1.0 + 1e-3:
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0
        
        g_norm = g / sum_g
        a_new = (1.0 - t) * a + t * g_norm
        a = project(a_new)
        
        obj, _ = compute_obj_and_peak(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

import numpy as np
from scipy.optimize import linprog
import time


def fft_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_peak(a):
    """Compute objective and peak index."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf, 0
    conv = fft_conv(a)
    idx = int(np.argmax(conv))
    obj = 2.0 * n * conv[idx] / (s * s)
    return obj, idx


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def exact_subgrad(a, peak_idx):
    """Exact subgradient of max(conv(a,a)) at the peak lag.
    d(conv[peak])/da[j] = 2*a[peak-j] if 0 <= peak-j < n, else 0."""
    n = len(a)
    k = peak_idx
    g = np.zeros(n)
    j_start = max(0, k - (n - 1))
    j_end = min(n - 1, k)
    if j_start <= j_end:

    return g
