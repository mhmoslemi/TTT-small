"""
Optimized smooth gradient descent with Adam for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothing of the max convolution for differentiable optimization,
combined with LP-based direction finding for escaping local minima.
Primary innovation: efficient gradient via np.correlate, adaptive tau schedule,
Adam optimizer with per-coordinate adaptive learning rates.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_true_score(a):
    """Compute the true objective value."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def smooth_max_and_grad(a, tau):
    """
    Compute smooth approximation to max(conv(a,a)) and its gradient.
    Uses log-sum-exp for numerical stability.
    Gradient: grad[i] = 2 * sum_j p[i+j] * a[j] computed via np.correlate.
    """
    n = len(a)
    conv = np.convolve(a, a)

    # Numerically stable softmax weights
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted -= float(np.max(shifted))
    p = np.exp(shifted)
    p_sum = float(np.sum(p))
    p /= p_sum
    smooth_val = max_conv + np.log(p_sum) / tau

    # Efficient gradient computation
    grad = 2.0 * np.correlate(p, a, mode='valid')

    return smooth_val, grad, p


def adam_smooth_optimize(a_init, deadline, tau_start=1.0, tau_end=1000.0,
                         lr=0.02, beta1=0.9, beta2=0.999, eps=1e-8,
                         check_interval=30, verbose_prefix=""):
    """
    Adam optimization of smooth max(conv) with adaptive tau schedule.
    Normalizes to sum=1 after each step.
    """
    a = a_init.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return a, float('inf')
    a = a / S

    best_a = a.copy()
    best_score = compute_true_score(a)
    n = len(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t_start = time.monotonic()
    total_budget = deadline - t_start
    stuck_count = 0
    last_best_t = 0

    while time.monotonic() < deadline:
        elapsed_frac = (time.monotonic() - t_start) / max(total_budget, 1e-10)
        # Exponential tau ramp over first 60%, then hold
        if elapsed_frac < 0.6:
            tau = tau_start * (tau_end / tau_start) ** (elapsed_frac / 0.6)
        else:
            tau = tau_end

        smooth_val, grad, p = smooth_max_and_grad(a, tau)

        # Center gradient to maintain sum constraint
        grad_centered = grad - np.mean(grad)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_centered
        v = beta2 * v + (1.0 - beta2) * grad_centered ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.clip(a_new, 0.0, None)
        S_new = float(np.sum(a_new))
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        a = a_new

        if t % check_interval == 0:
            score = compute_true_score(a)
            if score < best_score - 1e-8:
                best_score = score
                best_a = a.copy()
                stuck_count = 0
                last_best_t = t
                if t % (check_interval * 20) == 0:
                    print(f"  {verbose_prefix}iter {t}: tau={tau:.1f}, score={score:.6f}")
            else:
                stuck_count += 1

            # If stuck for long, try a perturbation
            if stuck_count > 200 and t - last_best_t > 2000:
                noise = np.random.default_rng(t).normal(0, 0.01, n)
                a_pert = a + noise
                a_pert = np.clip(a_pert, 0.0, None)
                S_p = float(np.sum(a_pert))
                if S_p > 1e-10:
                    a = a_pert / S_p
                    m = np.zeros(n)
                    v = np.zeros(n)
                    stuck_count = 0
                last_best_t = t

    score = compute_true_score(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    f = np.asarray(f)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(f[k - j])
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.array(vals_list, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    try:
        A = build_conv_matrix_sparse(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_direction_optimize(a_init, deadline, step_init=0.15, max_lp_fails=20,
                          verbose_prefix=""):
    """LP-based direction finding optimization."""
    a = a_init.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return a, float('inf')
    a = a / S

    best_a = a.copy()
    best_score = compute_true_score(a)
    n = len(a)
    step = step_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv_f = np.convolve(a, a)
        M = float(np.max(conv_f))

        g = solve_lp_direction(a, M)
        if g is None:
            no_improve += 1
            if no_improve > max_lp_fails:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            if no_improve > max_lp_fails:
                break
            continue

        g = g / sum_g  # normalize to sum=1

        it += 1
        # Try multiple step sizes
        best_step_score = best_score
        best_step_a = None
        for alpha in [step, step * 0.5, step * 1.5, step * 0.25, 0.05, 0.1]:
            alpha = min(alpha, 1.0)
            candidate = (1.0 - alpha) * a + alpha * g
            candidate = np.clip(candidate, 0.0, None)
            S_c = float(np.sum(candidate))
            if S_c < 1e-10:
                continue
            candidate = candidate / S_c
            cand_score = compute_true_score(candidate)
            if cand_score < best_step_score:
                best_step_score = cand_score
                best_step_a = candidate

        if best_step_a is not None:
            a = best_step_a
            best_a = a.copy()
            best_score = best_step_score
            no_improve = 0
            step *= 0.999
            step = max(step, 0.005)
            if it % 50 == 0:
                print(f"  {verbose_prefix}LP iter {it}: score={best_score:.6f} (step={step:.4f})")
        else:
            no_improve += 1
            step *= 0.8
            if step < 0.002 or no_improve > 50:
                break

    return best_a, best_score


def subgradient_polish(a_init, deadline, lr_init=0.01, top_k=5, momentum_decay=0.9,
                       verbose_prefix=""):
    """Subgradient descent with momentum for final polishing."""
    a = a_init.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return a, float('inf')
    a = a / S

    best_a = a.copy()
    best_score = compute_true_score(a)
    n = len(a)
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < 100000:
        conv = np.convolve(a, a)
        M = float(np.max(conv))

        # Top-k subgradient
        top_indices = np.argsort(conv)[-top_k:]
        mean_conv = float(np.sum(conv)) / (2 * n - 1)
        weights = np.maximum(conv[top_indices] - mean_conv, 0)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-15:
            weights = np.ones(top_k) / top_k
        else:
            weights = weights / w_sum

        grad = np.zeros(n)
        for k, w in zip(top_indices, weights):
            if w <= 0:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += w * 2.0 * a[k - j]

        grad_centered = grad - np.mean(grad)
        norm_g = np.linalg.norm(grad_centered)
        if norm_g < 1e-15:
            break

        grad_centered = grad_centered / (norm_g + 1e-15)
        momentum = momentum_decay * momentum - lr * grad_centered
        a_new = a + momentum
        a_new = np.clip(a_new, 0.0, None)
        S_new = float(np.sum(a_new))
        if S_new < 1e-10:
            lr *= 0.5
            no_improve += 1
            it += 1
            if no_improve > 100:
                break
            continue
        a_new = a_new / S_new
        a = a_new
        it += 1

        score = compute_true_score(a)
        if score < best_score - 1e-9:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.98
            if lr < 1e-6 or no_improve > 300:
                break

    return best_a, best_score


def make_initializations(n, rng, parent=None):
    """Create diverse initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # If parent available, use it
    if parent is not None:
        if len(parent) == n:
            inits.append(('parent', parent.copy()))
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(('parent_interp', np.interp(x_new, x_old, parent)))

    # Cosine taper
    cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    inits.append(('cosine', cos_t))

    # Bell (narrow)
    bell_n = np.exp(-0.5 * ((idx - center) / (n / 6.0)) ** 2)
    inits.append(('bell_narrow', bell_n))

    # Bell (wide)
    bell_w = np.exp(-0.5 * ((idx - center) / (n / 3.0)) ** 2)
    inits.append(('bell_wide', bell_w))

    # Bimodal
    two_peak = (np.exp(-0.5 * ((idx - n * 0.25) / (n / 10.0)) ** 2) +
                np.exp(-0.5 * ((idx - n * 0.75) / (n / 10.0)) ** 2))
    inits.append(('bimodal', two_peak))

    # Random (uniform)
    inits.append(('rand_uniform', rng.uniform(0.5, 1.5, size=n)))

    # Random (log-normal)
    inits.append(('rand_lognorm', np.exp(rng.normal(0, 0.3, size=n))))

    # Random (binary-like, smoothed)
    binary = rng.choice([0.0, 1.0], size=n)
    # Smooth it
    kernel = np.ones(5) / 5.0
    smoothed = np.convolve(binary, kernel, mode='same') + 0.1
    inits.append(('rand_bin_smooth', smoothed))

    # Quadratic (concave)
    quad = 1.0 - (2.0 * idx / (n - 1) - 1.0) ** 2
    quad = np.clip(quad, 0.05, None)
    inits.append(('quadratic', quad))

    # Sine-based (designed for flat autocorrelation)
    sine = 0.5 + 0.5 * np.sin(2 * np.pi * idx / n + np.pi / 4)
    inits.append(('sine', sine))

    # Power law
    power = (idx + 1) ** 0.3
    inits.append(('power', power / np.max(power)))

    return inits

            break
        print(f"\n--- Deep optimization: {lbl} (start={sc:.6f}) ---")

        a_opt, s_opt = mixed_optimize(f_init, focus_deadline,
                                       verbose_prefix=f"[{lbl}] ")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = a_opt.copy()
            print(f"  Improved to: {best_score:.6f}")

    # Phase 3: Final polish and random restarts (last 200s)
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")

    if best_sequence is not None and time.monotonic() < deadline - 15:
        # First: try Adam with different hyperparameters on best
        remaining = deadline - time.monotonic() - 30
        adam_polish_deadline = time.monotonic() + remaining * 0.4
        a_opt, s_opt = adam_smooth_optimize(best_sequence, adam_polish_deadline,
                                             tau_start=10.0, tau_end=2000.0,
                                             lr=0.005, check_interval=20,
                                             verbose_prefix="[AdamPolish] ")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = a_opt.copy()
            print(f"  Adam polish: {best_score:.6f}")

        # LP polish
        if time.monotonic() < deadline - 30:
            lp_polish_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.3
            a_opt, s_opt = lp_direction_optimize(best_sequence, lp_polish_deadline,
                                                  step_init=0.05, max_lp_fails=15,
                                                  verbose_prefix="[LPPolish] ")
            if s_opt < best_score:
                best_score = s_opt
                best_sequence = a_opt.copy()
                print(f"  LP polish: {best_score:.6f}")

        # Subgradient polish
        if time.monotonic() < deadline - 15:
            a_opt, s_opt = subgradient_polish(best_sequence, deadline - 5,
                                              lr_init=0.003, top_k=10,
                                              verbose_prefix="[SubPolish] ")
            if s_opt < best_score:
                best_score = s_opt
                best_sequence = a_opt.copy()
                print(f"  Sub polish: {best_score:.6f}")

        # Random restarts: try perturbing best and re-optimizing
        if time.monotonic() < deadline - 20:
            print("  Trying random restarts...")
            restart_budget = deadline - time.monotonic() - 5
            n_restarts = min(3, max(1, int(restart_budget / 30)))
            for r in range(n_restarts):
                if time.monotonic() > deadline - 10:
                    break
                noise_scale = 0.05 * (r + 1)
                noise = np.random.default_rng(1000 + r).normal(0, noise_scale, n)
                a_pert = best_sequence + noise
                a_pert = np.clip(a_pert, 0.0, None)
                S_p = float(np.sum(a_pert))
                if S_p < 1e-10:
                    continue
                a_pert = a_pert / S_p

                restart_deadline = time.monotonic() + (deadline - time.monotonic() - 5) / (n_restarts - r)
                a_opt, s_opt = adam_smooth_optimize(a_pert, restart_deadline,
                                                     tau_start=5.0, tau_end=1000.0,
                                                     lr=0.01, check_interval=20,
                                                     verbose_prefix=f"[Restart{r}] ")
                if s_opt < best_score:
                    best_score = s_opt
                    best_sequence = a_opt.copy()
                    print(f"  Restart {r} improved to: {best_score:.6f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(n)
        best_score = float('inf')

    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed! Using uniform fallback.")
        best_sequence = np.ones(n)
        final_score = evaluate_sequence([float(x) for x in best_sequence])

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
