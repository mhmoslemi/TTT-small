1. FFT-accelerated smoothed gradient descent (log-sum-exp, annealed beta)
2. LP best-response for strong local convergence  
3. First-order projected gradient with adaptive step
4. Multi-start perturbation and length adaptation

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M, deadline=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 30})
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        return res.x
    return None


def smoothed_grad(a, beta):
    """Compute gradient of beta-smoothed max(conv(a,a)) objective."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    bc = beta * (c - c_max)
    bc = np.clip(bc, -50.0, 50.0)
    exp_bc = np.exp(bc)
    p = exp_bc / exp_bc.sum()
    
    # grad_j = sum_k p_k * a_{k-j} for valid k-j
    n_p = len(p)  # 2n-1
    s = 1
    while s < n + n_p - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fp = np.fft.rfft(p, s)
    corr = np.fft.irfft(fa * np.conj(fp), s)
    
    grad = np.zeros(n)
    grad[0] = corr[0]
    if n > 1:
        grad[1:] = corr[s - 1:s - n:-1]
    
    return grad, c_max


def smoothed_grad_descent(a, deadline, beta_schedule=None):

    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    
    if beta_schedule is None:
        beta_schedule = [10, 50, 200, 1000, 5000, 20000, 100000]
    
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        phase_deadline = min(deadline, time.monotonic() + max(2, (deadline - time.monotonic()) / len(beta_schedule)))
        
        # Step size: start moderate, decrease
        step = 0.1 / (1 + 0.1 * beta * 0.01)
        step = max(0.001, min(0.5, step))
        
        it = 0
        while time.monotonic() < phase_deadline:

            grad, c_max = smoothed_grad(a, beta)
            
            # Normalize gradient (it's the gradient of the unnormalized objective)
            # We want to minimize on the simplex, so project the gradient
            # Remove the mean to stay on the simplex
            grad = grad - grad.mean()
            
            # Step
            a_new = a - step * grad
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 0.01:

                continue
            a_new = a_new / s

                step = min(0.5, step * 1.1)
        
        print(f"    beta={beta}, {it} iters, best={best_score:.6f}", flush=True)

def resample(a, new_n):
    """Resample sequence to new length."""
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, new_n)
    a_new = np.maximum(np.interp(t_new, t_old, a), 0)
    if a_new.sum() > 0.01:
        a_new = a_new / a_new.sum()
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

1. FFT-accelerated smoothed gradient descent with annealed log-sum-exp (beta from 10 to 100000)
2. LP best-response for strong local convergence

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x). Shape (2n-1, n)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        return res.x
    return None


def smoothed_subgrad(a, beta):
    """
    Compute subgradient of f_beta(a) = 2n * smoothed_max(conv(a,a)).
    Returns (grad, c_max) where grad_j = sum_k p_k * a_{k-j} (valid k-j).
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    bc = beta * (c - c_max)
    bc = np.clip(bc, -50.0, 50.0)
    exp_bc = np.exp(bc)
    p = exp_bc / exp_bc.sum()

    # grad_j = sum_k p_k * a_{k-j} for 0<=k-j<n
    # Use FFT-based cross-correlation: grad_j = (a cross-correlate p)(-j)
    n_p = len(p)  # 2n-1
    s = 1
    while s < n + n_p - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fp = np.fft.rfft(p, s)
    corr = np.fft.irfft(fa * np.conj(fp), s)

    grad = np.zeros(n)
    grad[0] = corr[0]
    if n > 1:
        grad[1:] = corr[s - 1:s - n:-1]

    return grad, c_max

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, new_n)
    a_new = np.maximum(np.interp(t_new, t_old, a), 0)
    if a_new.sum() > 0.01:
        a_new = a_new / a_new.sum()
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

    if a_sym.sum() > 0.01:

    else:
        a_sym = np.ones(n) / n
    return a_sym

        remaining = ph2_deadline - time.monotonic()
        est_per_iter = 0.025 if n <= 512 else 0.04 if n <= 768 else 0.08
        max_iters = max(100, int(remaining * 0.8 / est_per_iter))
        max_iters = min(max_iters, 10000)

        a_result, score = lp_iteration(best, max_iters, ph2_deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  P2: J={score:.6f}", flush=True)

        # Try symmetric version
        if time.monotonic() < ph2_deadline:
            a_sym = symmetrize(best)
            remaining = ph2_deadline - time.monotonic()
            max_iters2 = max(50, int(remaining * 0.5 / est_per_iter))
            a_res2, sc2 = lp_iteration(a_sym, max_iters2, ph2_deadline)
            if sc2 < best_score:
                best = a_res2.copy()
                best_score = sc2
                print(f"  P2 sym: J={sc2:.6f}", flush=True)

        # Try at adjacent length
        for n2 in [max(128, n - 128), min(1024, n + 128)]:
            if time.monotonic() > ph2_deadline:
                break
            if n2 == n or n2 < 64:
                continue
            if n2 >= 768 and (ph2_deadline - time.monotonic()) < 45:
                continue
            a2 = resample(best, n2)
            est2 = 0.025 if n2 <= 512 else 0.04
            max_iters3 = max(50, int((ph2_deadline - time.monotonic()) * 0.6 / est2))
            a_res3, sc3 = lp_iteration(a2, max_iters3, ph2_deadline)
            if sc3 < best_score:
                best = a_res3.copy()
                best_score = sc3
                print(f"  P2 n={n2}: J={sc3:.6f}", flush=True)

    # === PHASE 3: Perturbation + combined refinement ===
    ph3_deadline = deadline
    print(f"\n=== Phase 3: Perturbation Search ===", flush=True)

    if best is not None:
        n = best.size
        remaining = ph3_deadline - time.monotonic()
        n_tries = max(3, min(20, int(remaining / 25)))

        noise_scales = [0.02, 0.04, 0.01, 0.06, 0.015, 0.03, 0.005, 0.05]

        for trial in range(n_tries):
            if time.monotonic() > ph3_deadline:
                break
            scale = noise_scales[trial % len(noise_scales)]
            noise = rng.normal(0, scale, n)
            a_pert = best + noise
            a_pert = np.clip(a_pert, 0, None)
            if a_pert.sum() < 0.01:
                continue
            a_pert = a_pert / a_pert.sum()

            # Smoothed gradient (quick)
            a_res, sc = smoothed_grad_descent(a_pert, ph3_deadline,
                                              beta_schedule=[500, 5000, 50000])
            if sc < best_score:
                best = a_res.copy()
                best_score = sc
                print(f"  P3 t={trial} grad: J={sc:.6f}", flush=True)

            # LP (quick)
            if time.monotonic() < ph3_deadline:
                est = 0.025 if n <= 512 else 0.04
                max_it = max(30, int((ph3_deadline - time.monotonic()) / max(1, n_tries - trial) / est))
                max_it = min(max_it, 500)
                a_res2, sc2 = lp_iteration(a_pert, max_it, ph3_deadline)
                if sc2 < best_score:
                    best = a_res2.copy()
                    best_score = sc2
                    print(f"  P3 t={trial} LP: J={sc2:.6f}", flush=True)

    # === PHASE 4: Final deep polish (alternate grad + LP) ===
    print(f"\n=== Phase 4: Final Polish ===", flush=True)

    if best is not None and time.monotonic() < deadline:
        n = best.size
        a = best.copy()

        for rnd in range(8):
            if time.monotonic() > deadline:
                break

            # High-beta smoothed gradient
            a, sc = smoothed_grad_descent(a, deadline,
                                          beta_schedule=[10000, 50000, 200000])
            if sc < best_score:
                best = a.copy()
                best_score = sc

            # LP polish
            if time.monotonic() < deadline:
                est = 0.025 if n <= 512 else 0.04
                max_it = max(50, int((deadline - time.monotonic()) * 0.5 / est))
                max_it = min(max_it, 3000)
                a, sc2 = lp_iteration(best, max_it, deadline)
                if sc2 < best_score:
                    best = a.copy()
                    best_score = sc2
                    print(f"  P4 rnd={rnd}: J={best_score:.6f}", flush=True)
                else:
                    # If no improvement, break early
                    if rnd > 3:
                        break

    # Final validation
    if best is None:
        best = np.ones(256) / 256

    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
