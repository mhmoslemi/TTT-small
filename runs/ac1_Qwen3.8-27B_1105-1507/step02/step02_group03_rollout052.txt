import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    return 2.0 * n * conv.max() / (s * s)

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def build_conv_matrix(f):
    """A @ g = convolve(f, g, mode='full')"""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    return A

def lp_iteration(f, M, time_limit=2.0):
    """Solve max sum(g) s.t. A_f @ g <= M, g >= 0"""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.clip(res.x, 0, None), res.fun
    return None, None

def lp_optimize(f, time_budget, t0=0.5):
    """Iterative LP optimization"""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    start_time = time.monotonic()
    
    t = t0
    stall = 0
    
    while time.monotonic() - start_time < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        g, _ = lp_iteration(f, M, time_limit=2.0)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1.0 + 1e-10:
            # Can't improve
            break
        
        g = normalize(g)
        
        # Line search
        improved = False
        for step in [min(1.0, t), t*0.7, t*0.5, t*0.25, t*0.1]:
            f_new = (1-step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            score = compute_score(f_new)
            if score < best_score - 1e-12:
                best_score = score
                best_f = f_new.copy()
                f = f_new
                t = min(1.0, t * 1.2)
                improved = True
                stall = 0
                break
        
        if not improved:
            t *= 0.5
            stall += 1
            if stall > 20:
                break
    
    return best_f, best_score

"""
LP-based iterative optimization with spectral initialization and simulated annealing polishing.
Focuses on the two-sided LP method: fix f, solve for g that maximizes sum(g) 
subject to conv(f,g) <= M, then blend. Multiple initializations and sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def build_conv_matrix(f):
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    # For each output position k, A[k, i] = f[k-i] if 0 <= k-i < n
    for i in range(n):
        # f[i] contributes to positions k = i, i+1, ..., i+n-1 in the convolution
        # conv[k] includes f[i] * f[k-i], so A[k, i] = f[k-i]
        # For fixed i, k ranges from i to i+n-1
        # A[i:i+n, i] = f[0], f[1], ..., f[n-1]
        A[i:i+n, i] = f
    return A

def lp_solve(f, M, time_limit=3.0):
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success and res.x is not None:
        return np.clip(res.x, 0, None)
    return None

def lp_optimize(f, time_budget, verbose=True):
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    start_time = time.monotonic()
    
    t = 0.5  # blend parameter
    stall = 0
    iter_count = 0
    
    while time.monotonic() - start_time < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        g = lp_solve(f, M, time_limit=min(3.0, max(1.0, time_budget/100)))
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum <= 1.0 + 1e-9:
            # At local optimum
            stall += 1
            if stall > 10:
                # Try perturbation
                rng_local = np.random.default_rng(hash(f.tobytes()) % 2**32)
                f_pert = f * (1 + 0.1 * rng_local.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall = 0
                    t = 0.3
                else:
                    break
            continue
        else:
            stall = 0
        
        g = normalize(g)
        
        # Line search
        improved = False
        steps = [min(1.0, t), t*0.7, t*0.5, t*0.3, t*0.15, 0.05]
        for step in steps:

            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:

            f_new = f_new / s
            score = compute_score(f_new)
            if score < best_score - 1e-12:
                best_score = score
                best_f = f_new.copy()
                f = f_new

                improved = True

                if verbose and iter_count % 50 == 0:
                    elapsed = time.monotonic() - start_time
                    remaining = time_budget - elapsed
                    print(f"    LP iter {iter_count}: score={best_score:.6f}, t={t:.3f}, elapsed={elapsed:.0f}s")
                break

def spectral_flat(f, n_iter=200, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2*n - 1).bit_length()
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        
        n_bins = len(mag)
        current_energy = np.sum(f**2)
        target_mag = np.sqrt(current_energy * n_fft / n_bins)
        
        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0  # DC = sum(f) = 1
        
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        
        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
        f = f_new
    
    return best_f, best_score

def diverse_init(n, rng):
    r = rng.random()
    if r < 0.2:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center)**2 / width**2)
    elif r < 0.4:
        f = 0.5 * (1 + np.cos(2*np.pi*np.arange(n)/n))
    elif r < 0.6:
        f = np.abs(np.sin(np.pi*np.arange(n)/n * rng.uniform(1.5, 3.0)))
        f += 0.1
    elif r < 0.8:
        f = np.ones(n)
        for _ in range(8):
            c = int(rng.integers(0, n))
            w = max(2, int(n//16))
            kernel = np.exp(-0.5*(np.arange(n)-c)**2/w**2)
            f *= (1 + 0.5*rng.standard_normal()*kernel)
        f = np.abs(f) + 0.05
    else:
        f = np.ones(n) + 0.3 * rng.normal(0, 1, n)
    
    f = np.clip(f, 0.01, None)
    f = normalize(f)
    f = f * (1 + 0.03 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f

def pressure_update(f, n_iter=300, eta=0.15, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Subgradient: for the argmax position j*, 
        # gradient of conv[j*] w.r.t. f[i] is f[j*-i]
        j_star = np.argmax(conv)
        grad = np.zeros(n)
        for i in range(n):
            j = j_star - i
            if 0 <= j < n:
                grad[i] = f[j]
        
        # Also consider other peaks
        # Sort by value and take top few
        top_k = min(5, 2*n-1)
        top_indices = np.argsort(conv)[-top_k:]
        for j_star2 in top_indices:
            if j_star2 == j_star:
                continue
            weight = conv[j_star2] / M
            for i in range(n):
                j = j_star2 - i
                if 0 <= j < n:
                    grad[i] += weight * f[j]
        
        if grad.max() > 0:
            grad = grad / grad.max()
        
        eta_eff = eta * 0.95**(it // 50)
        f_new = f * np.exp(-eta_eff * grad)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)
        
        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                f_pert = f * (1 + 0.1 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall = 0
                else:
                    break
    
    return best_f, best_score

def simulated_annealing(f, rng, n_iter=5000, T0=0.03):
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    
    for it in range(n_iter):
        T = T0 * 0.9995**it
        f_new = f.copy()
        
        r = rng.random()
        if r < 0.25:
            width = max(1, int(rng.integers(1, max(2, n//16))))
            start = int(rng.integers(0, n-width))
            f_new[start:start+width] *= np.exp(rng.normal(0, 0.5))
        elif r < 0.45:
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.05))
        elif r < 0.65:
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.85:
            center = int(rng.integers(0, n))
            width = max(2, n//32)
            kernel = np.exp(-0.5*(np.arange(n)-center)**2/width**2)
            f_new += rng.normal(0, 0.1)*kernel
        else:
            f_new *= np.exp(rng.normal(0, 0.3, n))
        
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)
        
        if new_score < cur_score or rng.random() < np.exp(-(new_score-cur_score)/max(T, 1e-15)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
    
    return best_f, best_score

def coordinate_descent(f, n_sweeps=5, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-10:
                continue
            best_val = original
            best_local = best_score
            for factor in [0.3, 0.5, 0.7, 0.85, 1.0, 1.18, 1.43, 2.0, 3.33]:
                f[i] = max(0.0, original * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s*s)
                if score < best_local:
                    best_local = score
                    best_val = max(0.0, original * factor)
            f[i] = best_val
            best_score = min(best_score, best_local)
        f = normalize(f)
    
    return f, best_score

"""
LP-based iterative optimization for minimizing 2*n*max(conv)/sum^2.
Uses two-sided LP method (fix f, optimize g via LP), spectral initialization,
pressure updates, simulated annealing, and coordinate descent.
Multiple sequence lengths and diverse initializations to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute 2*n*max(convolve(f,f))/sum(f)^2."""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def normalize(f):
    """Normalize to sum=1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def lp_solve(f, M, time_limit=3.0):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None

def spectral_flat(f, n_iter=200, rng=None):
    """FFT-based spectral flattening."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_score = compute_score(f)

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        n_bins = len(mag)
        current_energy = np.sum(f ** 2)
        target_mag = np.sqrt(current_energy * n_fft / n_bins)

        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
        f = f_new

    return best_f, best_score


def pressure_update(f, n_iter=300, eta=0.15, rng=None):
    """Subgradient pressure-based update."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        # Subgradient from top peaks
        top_k = min(8, 2 * n - 1)
        top_indices = np.argsort(conv)[-top_k:]
        grad = np.zeros(n)
        for j_star in top_indices:
            weight = conv[j_star] / M
            lo = max(0, j_star - n + 1)
            hi = min(n, j_star + 1)
            for i in range(lo, hi):
                j = j_star - i
                if 0 <= j < n:
                    grad[i] += weight * f[j]

        if grad.max() > 0:
            grad = grad / grad.max()

        eta_eff = eta * (0.97 ** (it // 30))
        f_new = f * np.exp(-eta_eff * grad)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 60:

                f_pert = np.clip(f_pert, 0, None)

                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall = 0
                else:
                    break

    return best_f, best_score


def diverse_init(n, rng):
    """Generate diverse initialization."""
    r = rng.random()
    if r < 0.2:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.4:
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.6:
        f = np.abs(np.sin(np.pi * np.arange(n) / n * rng.uniform(1.5, 3.0)))
        f += 0.1
    elif r < 0.8:
        f = np.ones(n)
        for _ in range(10):
            c = int(rng.integers(0, n))
            w = max(2, int(n // 16))
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
            f *= (1 + 0.5 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05
    else:
        f = 1.0 + 0.3 * rng.normal(0, 1, n)

    f = np.clip(f, 0.01, None)
    f = normalize(f)
    f = f * (1 + 0.03 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def simulated_annealing(f, rng, n_iter=5000, T0=0.025):
    """Simulated annealing with diverse moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (1.0 - 0.9999 * it / n_iter)
        if T < 1e-8:
            T = 1e-8
        f_new = f.copy()

        r = rng.random()
        if r < 0.25:
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.5))
        elif r < 0.45:
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.04))
        elif r < 0.6:
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.8:
            center = int(rng.integers(0, n))
            width = max(2, n // 32)
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
            f_new += rng.normal(0, 0.08) * kernel
        else:
            f_new *= np.exp(rng.normal(0, 0.25, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 3000 and it > n_iter // 2:
            break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=5, rng=None):
    """Coordinate descent polishing."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-10:
                continue
            best_val = original
            best_local = best_score
            for factor in [0.3, 0.5, 0.7, 0.85, 1.0, 1.18, 1.43, 2.0, 3.33]:
                f[i] = max(0.0, original * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = max(0.0, original * factor)
            f[i] = best_val
            best_score = min(best_score, best_local)
        f = normalize(f)

    return f, best_score
