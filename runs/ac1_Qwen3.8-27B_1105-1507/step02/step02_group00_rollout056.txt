```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0 or np.sum(a) < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = a[k-j]
    return A

def solve_lp_min_t(a):
    """Solve: min t s.t. conv(a, g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t)."""
    n = len(a)
    A = build_conv_matrix(a)  # shape (2n-1, n)
    # Variables: g[0], ..., g[n-1], t
    # Objective: minimize t -> c = [0, ..., 0, 1]
    c = np.zeros(n+1)
    c[n] = 1.0
    
    # Inequality constraints: A @ g - t <= 0 for each k
    # => A @ [g; t] <= 0 where the augmented matrix has -1 in the t column
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = res.x[:n]
        t = res.x[n]
        return g, t
    return None, np.inf
```

"""
Improved LP-BCD (Block Coordinate Descent) with min-t LP formulation,
diverse initializations, multi-scale n values, and subgradient/coordinate polishing
to minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def solve_lp_min_t(A, n):
    """Solve min t s.t. A@g <= t*1, sum(g)=1, g>=0, t>=0. Returns (g, t)."""
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], res.x[n]
    return None, np.inf


def solve_lp_max_sum(A, n, M):
    """Solve max sum(g) s.t. A@g <= M*1, g>=0. Returns g."""
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0 and res.x is not None and np.sum(res.x) > 1e-10:
        return res.x
    return None


def lp_bcd(a_init, max_iter=400, alpha=0.5, verbose=False, deadline=None,
           use_min_t=True, use_max_sum=False):
    """LP Block Coordinate Descent."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = build_conv_matrix(a)
        g = None

        if use_min_t:
            g, t_lp = solve_lp_min_t(A, n)
            if g is None:
                break
            g = g / np.sum(g) if np.sum(g) > 1e-10 else None
            if g is None:
                break

        if use_max_sum and (g is None or it % 3 == 0):
            conv_a = np.convolve(a, a)
            M = float(np.max(conv_a))
            g2 = solve_lp_max_sum(A, n, M)
            if g2 is not None:
                g2 = g2 / np.sum(g2)
                # Prefer the one with lower resulting peak
                if g is None:
                    g = g2
                else:
                    test1 = (1 - alpha) * a + alpha * g
                    test2 = (1 - alpha) * a + alpha * g2
                    p1 = float(np.max(np.convolve(test1, test1)))
                    p2 = float(np.max(np.convolve(test2, test2)))
                    if p2 < p1:
                        g = g2

        if g is None:
            break

        # Adaptive alpha: start aggressive, decay
        alpha_t = alpha * (0.97 ** (it // 30))
        alpha_t = max(0.1, alpha_t)

        a_new = (1.0 - alpha_t) * a + alpha_t * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g.copy()
        else:
            a = a_new / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if verbose and it % 50 == 0:
            print(f"    BCD it={it}: obj={obj:.6f}")

    return best_a, best_obj


def subgradient_polish(a, n_steps=2000, verbose=False, deadline=None):
    """Subgradient polishing for min max(conv(a,a)) s.t. sum(a)=1, a>=0."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        h = np.convolve(a, a)
        max_h = float(np.max(h))

        # Active set with relative tolerance (widen over time)
        tol = max_h * (0.005 + 0.05 * (step / max(1, n_steps)))
        active = h >= max_h - tol
        if not np.any(active):
            active = h >= max_h - 1e-12

        # Subgradient
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            grad[j_lo:j_hi + 1] += 2.0 * a[k - j_hi:k - j_lo + 1][::-1]
        grad /= len(active_idx)

        # Step size (Polyağun-Shapiro)
        step_size = 0.3 / (1.0 + 0.02 * step)

        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def coord_refine(a, n_rounds=5, verbose=False, deadline=None):
    """Coordinate descent refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    multipliers = [0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            old_val = a[j]
            best_x = old_val
            best_local = compute_obj(a)
            for mult in multipliers:
                x = old_val * mult
                a[j] = max(0.0, x)
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = max(0.0, x)
            a[j] = best_x
            if best_x != old_val:
                improved = True

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if not improved and rnd > 0:
            break

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        # Cosine
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0.01, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif choice == 1:
        # U-shaped (high at edges)
        x = np.linspace(-1, 1, n)
        a = 1.0 + rng.uniform(0.5, 2.0) * x * x
    elif choice = 2:
        # V-shaped (high in middle)
        x = np.linspace(-1, 1, n)
        a = 1.0 + rng.uniform(0.5, 2.0) * (1 - x * x)
    elif choice == 3:
        # Periodic
        period = int(rng.integers(3, max(4, n // 10)))
        pattern = rng.random(period) + 0.1
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 4:
        # Two-block
        split = int(n * rng.uniform(0.3, 0.7))
        a = np.ones(n)
        a[:split] = rng.uniform(0.5, 2.0)
        a[split:] = rng.uniform(0.5, 2.0)
    elif choice == 5:
        # Exponential decay from edges
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a = np.exp(-rng.uniform(0.1, 1.0) * d / n)
    elif choice == 6:
        # Random smooth (Gaussian process-like)
        a = np.ones(n)
        for _ in range(3):
            center = rng.integers(0, n)
            width = rng.uniform(n * 0.1, n * 0.5)
            amp = rng.uniform(-0.5, 0.5)
            a += amp * np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a = np.maximum(a, 0.01)
    else:
        a = rng.random(n) + 0.05

    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n) + 0.05
        s = np.sum(a)
    return a / s

"""

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def solve_lp_min_t(A, n):
    """Solve min t s.t. A@g <= t*1, sum(g)=1, g>=0, t>=0. Returns (g, t)."""
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0 and res.x is not None:
        return res.x[:n], res.x[n]
    return None, np.inf


def solve_lp_max_sum(A, n, M):

                    print(f"  BCD3: obj={obj5:.6f}")

        # Final subgradient
        if time.monotonic() < phase3_end - 3:
            sg_steps = int(min(1500, (phase3_end - time.monotonic()) / 0.012))
            if sg_steps > 100:
                a6, obj6 = subgradient_polish(best_seq, n_steps=sg_steps,
                                               deadline=phase3_end)
                if obj6 < best_score:
                    best_score = obj6
                    best_seq = a6.copy()
                    print(f"  SG3: obj={obj6:.6f}")

        # Final coord
        if time.monotonic() < phase3_end - 2 and n_best <= 400:
            a7, obj7 = coord_refine(best_seq, n_rounds=3, deadline=phase3_end)
            if obj7 < best_score:
                best_score = obj7
                best_seq = a7.copy()
                print(f"  Coord2: obj={obj7:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    best_seq = np.clip(best_seq, 0, 1000.0)
    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
