"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Nesterov accelerated proximal gradient with active-set subgradients
2. Multi-scale optimization with diverse initializations
3. Periodic stochastic perturbations for escaping local minima
4. LP-based best-response polish
5. Coordinate descent refinement
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def compute_active_subgradient(a, rel_tol=0.1):
    """
    Compute subgradient of max_k (a*a)_k using active set average.
    Returns (gradient, b_max).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    threshold = b_max * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]
    grad = np.zeros(n)
    for k in active:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            grad[idx] += 2.0 * a[k - idx]
    grad /= len(active)
    return grad, b_max

def nesterov_solve(a, max_iters=2000, deadline=None, tracker=None, 
                   lr_init=0.3, momentum=0.9, perturb_every=200, perturb_sigma=0.03):
    """
    Nesterov accelerated gradient descent for minimizing max_k (a*a)_k
    subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    a_prev = a.copy()
    lr = lr_init
    stuck_count = 0
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation
        a_extrap = a + momentum * (a - a_prev)
        a_extrap = np.maximum(a_extrap, 0)
        s = a_extrap.sum()
        if s < 1e-12:
            a_extrap = a.copy()
        else:
            a_extrap = a_extrap / s
        
        # Subgradient step
        grad, _ = compute_active_subgradient(a_extrap)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-12:
            lr *= 0.5
            continue
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            stuck_count = 0
            # Slowly increase lr if improving
            lr = min(lr * 1.01, 1.0)
        else:
            stuck_count += 1
            if J_new > best_J * 1.005:
                lr *= 0.85
        
        # Periodic perturbation
        if stuck_count >= perturb_every:
            a_new = a_new + perturb_sigma * np.random.randn(n)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            lr = max(lr, lr_init * 0.5)
            stuck_count = 0
            # Try to recover
            J_new = compute_J(a_new)
            if J_new < best_J * 1.1:
                pass  # accept perturbation
            else:
                a_new = best_a.copy()  # revert
        
        a_prev = a
        a = a_new
        
        if tracker is not None and it % 100 == 0:
            tracker.update(best_a)
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            indices = np.arange(jl, jh + 1)
            vals.extend(a[k - indices].tolist())
    from scipy.sparse import lil_matrix
    A = lil_matrix((2 * n - 1, n))
    A[rows, cols] = vals
    return A.tocsr()

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_polish(a, max_iters=50, deadline=None, tracker=None):
    """LP mirror descent polish."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Try a few step sizes
        best_trial = None
        best_trial_J = best_J
        for t in [0.3, 0.5, 0.7, 1.0]:
            trial = (1 - t) * a + t * g
            trial = np.maximum(trial, 0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial = trial / s
            J_trial = compute_J(trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_trial = trial
        
        if best_trial is None:
            break
        
        a = best_trial
        if best_trial_J < best_J:
            best_J = best_trial_J
            best_a = a.copy()
            if tracker is not None:
                tracker.update(a)
        else:
            break
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a

def coordinate_descent(a, max_sweeps=5, deadline=None):
    """Simple coordinate descent: optimize one element at a time."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            # Try modifying a[j]
            # The objective depends on a[j] through all conv[k] where k-j is valid
            # Simple approach: try a few values of a[j]
            old_val = a[j]
            rest_sum = s - old_val
            # The optimal a[j] given all other elements: 
            # We want to minimize max_k conv_k, and conv_k = sum_i a_i * a_{k-i}
            # The terms involving a[j] are: a[j]*a[k-j] for valid k-j
            # So conv_k gets contribution a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j] (if j != k-j)
            # or a[j]^2 (if j == k-j)
            # This is complex to optimize exactly. Use a simple grid.
            candidates = [old_val * 0.5, old_val * 0.7, old_val, old_val * 1.3, old_val * 1.5,
                         rest_sum * 0.01, rest_sum * 0.05]
            best_val = old_val
            best_J = compute_J(a)
            for cv in candidates:
                cv = max(0, min(cv, 1000))
                a[j] = cv
                a_new_sum = a.sum()
                if a_new_sum < 1e-10:
                    a[j] = old_val
                    continue
                # Don't normalize here, just evaluate
                J_cand = compute_J(a)
                if J_cand < best_J - 1e-10:
                    best_J = J_cand
                    best_val = cv
            a[j] = best_val
            if best_val != old_val:
                improved = True
        if not improved:
            break
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Random
    inits.append(rng.uniform(0.2, 1.0, n))
    
    # Cosine patterns
    for freq in [1, 2, 3, 4, 5, 7, 10]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Gaussian bump
    inits.append(np.exp(-5.0 * (t - 0.5) ** 2) + 0.05)
    
    # Two bumps
    inits.append(np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.05)
    
    # Triangle wave
    inits.append(np.abs(2.0 * (t - np.floor(t + 0.5))) + 0.1)
    
    # Power law
    inits.append(np.maximum((t + 0.01) ** rng.uniform(-1, 1), 0.01))
    
    # Sparse spikes
    n_spikes = max(3, n // 16)
    pos = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[pos] = 1.0
    # Smooth slightly
    from numpy import convolve as np_conv
    kernel = np.ones(3) / 3.0
    inits.append(np.maximum(np_conv(sparse, kernel, mode='same'), 0.01))
    
    # Symmetric random
    half = rng.uniform(0.2, 1.0, n // 2 + 1)
    sym = np.concatenate([half, half[-2::-1]]) if n % 2 == 0 else np.concatenate([half, half[-1::-1]])
    inits.append(np.maximum(sym[:n], 0.01))
    
    return inits

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smoothed max subgradient with momentum (Nesterov)
2. Multi-scale, multi-init exploration
3. LP best-response polish
4. Stochastic restarts
Key advantage over LP-only: 100x faster per-iteration, enabling many more iterations.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_subgradient(a, eps=10.0):
    """
    Compute gradient of smooth max: (1/eps)*log(sum exp(eps*(a*a)_k)).
    Returns (gradient, smooth_max_value).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    # Active set: lags with significant softmax weight
    threshold = b_max - 12.0 / eps
    active = np.where(b >= threshold)[0]
    
    b_act = b[active]
    w = np.exp(eps * (b_act - b_max))
    w_sum = w.sum()
    if w_sum < 1e-15:
        # Fallback: use only argmax
        active = np.array([np.argmax(b)])
        w = np.array([1.0])
        w_sum = 1.0
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for i, k in enumerate(active):
        wi = w[i]
        if wi < 1e-10:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += wi * 2.0 * a[k - idx]
    
    # Smooth max value
    b_shifted = b - b_max
    smooth_val = b_max + np.log(np.sum(np.exp(eps * b_shifted))) / eps
    
    return grad, float(smooth_val)


def nesterov_smooth(a, max_iters=3000, deadline=None, tracker=None,

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            idx = np.arange(jl, jh + 1)
            vals.extend(a[k - idx].tolist())
    A = lil_matrix((2 * n - 1, n))
    A[rows, cols] = vals
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_polish(a, max_iters=50, deadline=None, tracker=None):
    """LP mirror descent polish."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        best_trial = None
        best_trial_J = best_J
        for t in [0.3, 0.5, 0.7, 1.0]:
            trial = (1 - t) * a + t * g
            trial = np.maximum(trial, 0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial = trial / s
            J_trial = compute_J(trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_trial = trial
        
        if best_trial is None:
            break
        
        a = best_trial
        if best_trial_J < best_J:
            best_J = best_trial_J
            best_a = a.copy()
            if tracker is not None:
                tracker.update(a)
        else:
            break
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a
