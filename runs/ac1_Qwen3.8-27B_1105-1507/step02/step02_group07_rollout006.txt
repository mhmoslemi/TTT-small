def build_conv_matrix(f):
    """Build matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((np.array(vals), (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))), shape=(2*n-1, n))

def solve_lp(f, M, A):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M."""

    return None

def softmax_grad(f, beta):
    """Compute gradient of soft-max of conv(f,f)."""
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:

    p = exp_C / total
    grad = 2.0 * np.correlate(p, f, mode='valid')
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val

def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M, A):
    """Solve: max sum(g) s.t. g >= 0, A@g <= M."""
    n = len(f)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M), bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

        p = exp_C / total
        grad = 2.0 * np.correlate(p, f, mode='valid')
        total_grad += grad
    total_grad /= len(betas)
    return total_grad

            else:
                # Try smaller reduction
                factor = 1.0 - 0.02 * reduce_weights
                new_f2 = f * factor
                new_f2 = np.clip(new_f2, 0.0, 1000.0)
                ns2 = float(np.sum(new_f2))
                if ns2 < 1e-10:
                    break
                new_f2 = new_f2 / ns2
                sc2 = evaluate_sequence(new_f2.tolist())
                if np.isfinite(sc2) and sc2 < best_score:
                    f = new_f2
                    best_f = f.copy()
                    best_score = sc2
                else:
                    break

    return best_f, best_score


def skewed_window(n, skew=0.15, sharpness=2.0):
    """Generate a skewed super-Gaussian window."""
    x = np.linspace(0, 1, n)
    center = 0.5 + skew * 0.3
    left_width = 0.35 * (1.0 - skew * 0.5)
    right_width = 0.35 * (1.0 + skew * 0.5)
    w = np.zeros(n)
    for i in range(n):
        if x[i] <= center:
            w[i] = np.exp(-((x[i] - center) / left_width) ** sharpness)
        else:
            w[i] = np.exp(-((x[i] - center) / right_width) ** sharpness)
    return w + 1e-6


def diverse_inits(n, seed=0):
    """Generate diverse initializations including asymmetric ones."""
    rng = np.random.default_rng(seed)
    inits = []

    # Skewed windows
    for skew in [-0.3, -0.15, 0.0, 0.15, 0.3]:
        for sharp in [1.5, 2.0, 3.0]:
            inits.append(skewed_window(n, skew, sharp))

    # Tukey-like
    for r in [0.2, 0.35, 0.5]:
        w = np.ones(n)
        m = int(r * n / 4)
        if m > 0:
            idx = np.arange(m) / m
            w[:m] = 0.5 * (1 + np.cos(np.pi * (idx - 0.5 + 1)))
            w[n-m:] = w[:m]
        inits.append(w)

    # Random smooth
    for s in [1, 2]:
        rrng = np.random.default_rng(seed + s)
        x = rrng.standard_normal(n)
        ksize = max(3, n // 20)
        kernel = np.exp(-0.5 * (np.arange(-ksize, ksize+1) / (ksize//2 + 1)) ** 2)
        smooth = np.abs(np.convolve(x, kernel, 'same')) + 0.01
        inits.append(smooth)

    return inits


def propose_candidate(budget_s=1500):
    """Main search: multi-phase optimization with diverse strategies."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0, 1000)
        S = float(np.sum(parent))
        if S > 0.01:
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent: n={parent.size}, score={score:.6f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_sequence is None:
        n_default = 512
        best_sequence = skewed_window(n_default, 0.1, 2.0)
        best_score = evaluate_sequence(best_sequence.tolist())
        print(f"Default init: n={n_default}, score={best_score:.6f}")

    n = len(best_sequence)

    # === Phase 1: LP optimization on best (30% time) ===
    print(f"\n[Phase 1] LP optimization (n={n}, best={best_score:.6f})")
    p1_deadline = t0 + (deadline - t0) * 0.30
    try:
        f1, s1 = lp_optimize(best_sequence, p1_deadline, max_rounds=40)
        if s1 < best_score:
            best_score = s1
            best_sequence = f1
            print(f"  Improved to {s1:.6f}")
        else:
            print(f"  No improvement (still {best_score:.6f})")
    except Exception as e:
        print(f"  Error: {e}")

    # === Phase 2: Multi-beta softmax (25% time) ===
    print(f"\n[Phase 2] Multi-beta softmax (best={best_score:.6f})")
    p2_deadline = t0 + (deadline - t0) * 0.55
    try:
        f2, s2 = softmax_optimize(best_sequence, p2_deadline,
                                   beta_start=5, beta_end=400, lr_start=0.025)
        if s2 < best_score:
            best_score = s2
            best_sequence = f2
            print(f"  Improved to {s2:.6f}")
        else:
            print(f"  No improvement (still {best_score:.6f})")
    except Exception as e:
        print(f"  Error: {e}")

    # === Phase 3: Targeted peak reduction (10% time) ===
    print(f"\n[Phase 3] Targeted peak reduction (best={best_score:.6f})")
    p3_deadline = t0 + (deadline - t0) * 0.65
    try:
        f3, s3 = targeted_peak_reduction(best_sequence, p3_deadline, max_iters=300)
        if s3 < best_score:
            best_score = s3
            best_sequence = f3
            print(f"  Improved to {s3:.6f}")
    except Exception as e:
        print(f"  Error: {e}")

    # === Phase 4: LP again (10% time) ===
    print(f"\n[Phase 4] LP round 2 (best={best_score:.6f})")
    p4_deadline = t0 + (deadline - t0) * 0.75
    try:
        f4, s4 = lp_optimize(best_sequence, p4_deadline, max_rounds=25)
        if s4 < best_score:
            best_score = s4
            best_sequence = f4
            print(f"  Improved to {s4:.6f}")
    except Exception as e:
        print(f"  Error: {e}")

    # === Phase 5: Try diverse initializations (10% time) ===
    print(f"\n[Phase 5] Diverse inits (best={best_score:.6f}, n={n})")
    p5_deadline = t0 + (deadline - t0) * 0.85
    inits = diverse_inits(n, seed=7)
    # Quick screen
    screened = []
    for fi in inits:
        fi = np.clip(fi.astype(np.float64), 0, 1000)
        S = float(np.sum(fi))
        if S < 0.01:
            continue
        fi = fi / S
        sc = evaluate_sequence(fi.tolist())
        if np.isfinite(sc):
            screened.append((sc, fi))
    screened.sort(key=lambda x: x[0])

    # Optimize top 3 from screening
    top_k = min(3, len(screened))
    if top_k > 0:
        per_init_time = (p5_deadline - t0 - (deadline - t0) * 0.85) / top_k
        for rank in range(top_k):
            if time.monotonic() > p5_deadline:
                break
            sc_init, fi = screened[rank]
            if sc_init < best_score:
                best_score = sc_init
                best_sequence = fi.copy()
                print(f"  Init {rank}: {sc_init:.6f} (new best)")

            d = min(p5_deadline, time.monotonic() + max(10, per_init_time))
            if time.monotonic() < d - 3:
                try:
                    fb, sb = lp_optimize(fi, d, max_rounds=15)
                    if sb < best_score:
                        best_score = sb
                        best_sequence = fb
                        print(f"  Init {rank} optimized: {sb:.6f}")
                except Exception:
                    pass

    # === Phase 6: Final softmax polish (7% time) ===
    print(f"\n[Phase 6] Final polish (best={best_score:.6f})")
    try:
        f6, s6 = softmax_optimize(best_sequence, deadline - 8,
                                   beta_start=100, beta_end=800, lr_start=0.01)
        if s6 < best_score:
            best_score = s6
            best_sequence = f6
            print(f"  Improved to {s6:.6f}")
    except Exception as e:
        print(f"  Error: {e}")

    # === Phase 7: Final LP (3% time) ===
    if time.monotonic() < deadline - 5:
        try:
            f7, s7 = lp_optimize(best_sequence, deadline - 5, max_rounds=10)
            if s7 < best_score:
                best_score = s7
                best_sequence = f7
                print(f"  Final LP: {s7:.6f}")
        except Exception:
            pass

    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_sequence)}, score={best_score:.6f}, time={time.monotonic()-t0:.1f}s")
    print(f"{'='*50}")

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = skewed_window(512, 0.1, 2.0)
        best_score = evaluate_sequence(best_sequence.tolist())

    return [float(x) for x in best_sequence]
