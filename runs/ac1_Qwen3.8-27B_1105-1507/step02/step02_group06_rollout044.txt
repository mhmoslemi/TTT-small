Uses a combination of LP direction finding, weighted subgradient descent,
and multiplicative (mirror) updates with multiple sequence lengths.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Compute the objective value directly."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_conv_matrix_sparse(f):
    """Build sparse matrix for convolution with f."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def weighted_subgradient(f, threshold_frac=0.95):
    """Compute weighted subgradient using all lags above threshold."""
    n = len(f)
    conv = np.convolve(f, f)
    M = np.max(conv)
    threshold = threshold_frac * M
    
    # Weights: excess above threshold
    weights = np.maximum(conv - threshold, 0)
    total_w = np.sum(weights)
    if total_w < 1e-15:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
        total_w = 1.0
    
    # Gradient of sum_k weights[k] * conv[k] with respect to f
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] <= 0:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += weights[k] * 2.0 * f[k - j]
    
    return grad, M


def softmax_subgradient(f, temperature=1.0):
    """Compute subgradient using softmax approximation of max."""
    n = len(f)
    conv = np.convolve(f, f)
    
    # Softmax weights
    t = temperature
    shifted = t * (conv - np.max(conv))
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += weights[k] * 2.0 * f[k - j]
    
    return grad


def optical_subgradient(f, top_k=10):
    """Use top-k lags with equal weight (simpler but effective)."""
    n = len(f)
    conv = np.convolve(f, f)
    top_indices = np.argsort(conv)[-top_k:]
    
    grad = np.zeros(n)
    for k in top_indices:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):

def multiplicative_update(f, grad, lr):
    """Multiplicative (mirror) descent step."""
    # grad is the subgradient of the objective (we want to decrease it)
    # a_j <- a_j * exp(-lr * grad_j)
    # Normalize grad for stability
    max_grad = np.max(np.abs(grad))
    if max_grad < 1e-15:
        return f.copy()
    grad_norm = grad / max_grad
    update = 1.0 - lr * grad_norm
    update = np.clip(update, 1e-6, None)
    new_f = f * update
    return np.clip(new_f, 0.0, None)

                         lr_decay=0.999, verbose_prefix="", max_iter=100000):

"""
Optimized search for minimum peak-to-average autocorrelation sequences.
Combines LP direction finding, multiplicative mirror descent, and adaptive
subgradient methods with multiple sequence lengths and restarts.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Compute the objective value directly (fast, no type checks)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@x)[k] = convolve(f, x reversed at k)."""
    n = len(f)
    # For full convolution: (f*g)[k] = sum_j f[j]*g[k-j]
    # We want A such that (A@g)[k] = sum_j f[j]*g[k-j] for k=0..2n-2
    # (A@g)[k] = sum_j A[k,j] * g[j] where A[k,j] = f[k-j] if 0<=k-j<n
    total_entries = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        total_entries += (j_max - j_min + 1)
    
    rows = np.empty(total_entries, dtype=np.int32)
    cols = np.empty(total_entries, dtype=np.int32)
    vals = np.empty(total_entries, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f, g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_subgradient_topk(f, top_k=5):
    """Compute subgradient using top-k lags of convolution."""
    n = len(f)
    conv = np.convolve(f, f)
    top_indices = np.argsort(conv)[-top_k:]
    
    grad = np.zeros(n)
    for k in top_indices:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * f[k - j]
    
    return grad


def compute_weighted_subgradient(f, frac=0.92):
    """Compute weighted subgradient using lags above a fraction of max."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    threshold = frac * M
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = conv[k] - threshold
        if w <= 0:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += w * 2.0 * f[k - j]
    
    return grad


def softmax_subgradient(f, beta=10.0):
    """Compute subgradient using softmax weighting of all lags."""
    n = len(f)
    conv = np.convolve(f, f)
    
    # Softmax weights (numerically stable)
    shifted = beta * (conv - np.max(conv))
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-12:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += weights[k] * 2.0 * f[k - j]
    
    return grad


def normalize_to_sum(f, target_sum):
    """Scale f to have the given sum, maintaining non-negativity."""
    s = float(np.sum(f))
    if s < 1e-15:
        return f
    return f * (target_sum / s)


def optimize_lp_heavy(f, deadline, step_init=0.1, max_iterations=100,
                      verbose_prefix=""):
    """Intensive LP-based optimization. Each iteration solves an LP."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    S = float(np.sum(f))
    step = step_init
    no_improve = 0
    
    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break
        
        t_start = time.monotonic()
        g = solve_lp_direction(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        lp_time = time.monotonic() - t_start
        
        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue
        
        g = g * (S / sum_g)
        
        # Line search with multiple candidates
        best_cand = None
        best_cand_score = best_score
        for alpha in [step, step * 0.5, step * 0.8, step * 1.2]:
            alpha = min(alpha, 1.0)
            if alpha < 0.005:
                continue
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0.0, None)
            candidate = normalize_to_sum(candidate, S)
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_cand_score:
                best_cand_score = cand_score
                best_cand = candidate
        
        if best_cand is not None:
            f = best_cand
            best_f = f.copy()
            best_score = best_cand_score
            no_improve = 0
            step = max(step * 0.98, 0.01)
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.005 or no_improve > 30:
                break
        
        if verbose_prefix and best_score < 1.52:
            print(f"  {verbose_prefix}LP iter: score={best_score:.6f} "
                  f"(step={step:.4f}, t={lp_time:.1f}s)")
    
    return best_f, best_score


def optimize_adaptive_grad(f, deadline, lr_init=0.02, top_k=5,
                           use_weighted=False, verbose_prefix=""):
    """Adaptive subgradient descent with restarts."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    S = float(np.sum(f))
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < 500000:
        if use_weighted:
            grad = compute_weighted_subgradient(f, frac=0.90)
        else:
            grad = compute_subgradient_topk(f, top_k=top_k)
        
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break
        grad = grad / norm_grad
        
        # Try the update
        new_f = f - lr * grad
        new_f = np.clip(new_f, 0.0, None)
        new_f = normalize_to_sum(new_f, S)
        
        new_score = compute_score(new_f)
        it += 1
        
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.97
            if no_improve > 5:
                # Small restart: add tiny random perturbation
                if np.random.random() < 0.1:
                    pert = np.random.default_rng(it).normal(0, lr * 0.1, len(f))
                    f = np.clip(f + pert, 0.0, None)
                    f = normalize_to_sum(f, S)
                no_improve = 0
            if lr < 1e-9:
                break
    
    return best_f, best_score


def optimize_multiplicative(f, deadline, lr_init=0.03, verbose_prefix=""):
    """Multiplicative (mirror) descent for non-negative variables."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    S = float(np.sum(f))
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < 100000:
        grad = compute_weighted_subgradient(f, frac=0.93)
        max_g = float(np.max(grad))
        if max_g < 1e-15:
            break
        
        # Multiplicative update: a_j *= (1 - lr * g_j / max_g)
        ratio = 1.0 - lr * grad / max_g
        ratio = np.clip(ratio, 1e-8, None)
        new_f = f * ratio
        new_f = np.clip(new_f, 0.0, None)
        new_f = normalize_to_sum(new_f, S)
        
        new_score = compute_score(new_f)
        it += 1
        
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.98
            if lr < 1e-8 or no_improve > 500:
                break
    
    return best_f, best_score


def interpolate_to_length(x, new_n):
    """Interpolate array x to new length new_n."""
    old_n = len(x)
    if old_n == new_n:
        return x.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, x)


def make_structured_init(n, rng):
    """Create diverse structured initializations."""
    inits = {}
    idx = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n)
    
    # Cosine
    inits['cosine'] = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    
    # Bell
    center = (n - 1) / 2.0
    inits['bell'] = np.exp(-0.5 * ((idx - center) / (n / 4.0)) ** 2)
    
    # Random
    inits['random'] = rng.uniform(0.3, 1.7, size=n)
    
    # S-curve
    inits['sigmoid'] = 1.0 / (1.0 + np.exp(-6.0 * (idx / (n - 1) - 0.5)))
    
    # Quadratic
    quad = 1.0 - (2.0 * idx / (n - 1) - 1.0) ** 2
    inits['quad'] = np.clip(quad, 0.1, None)
    
    # Two-sided exponential
    inits['exp'] = np.exp(-1.5 * np.abs(idx - center) / (n / 3.0))
    
    return inits


def full_optimize(f, deadline, n, rng, verbose_prefix=""):
    """Full multi-phase optimization on a single sequence."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    
    # Phase A: Quick subgradient warmup (20% of time)
    phase_a_end = time.monotonic() + (deadline - time.monotonic()) * 0.20
    f_a, s_a = optimize_adaptive_grad(f, phase_a_end, lr_init=0.015,
                                       top_k=5, verbose_prefix=verbose_prefix + "A:")
    if s_a < best_score:
        best_score = s_a
        best_f = f_a.copy()
    
    # Phase B: LP optimization (35% of remaining time)
    remaining = deadline - time.monotonic()
    phase_b_end = time.monotonic() + remaining * 0.40
    f_b, s_b = optimize_lp_heavy(f_a, phase_b_end, step_init=0.08,
                                  verbose_prefix=verbose_prefix + "B:")
    if s_b < best_score:
        best_score = s_b
        best_f = f_b.copy()
    
    # Phase C: Multiplicative descent (20% of remaining)
    remaining = deadline - time.monotonic()
    phase_c_end = time.monotonic() + remaining * 0.30
    f_c, s_c = optimize_multiplicative(f_b, phase_c_end, lr_init=0.02,
                                        verbose_prefix=verbose_prefix + "C:")
    if s_c < best_score:
        best_score = s_c
        best_f = f_c.copy()
    
    # Phase D: Final subgradient polish with finer steps (remaining time)
    f_d, s_d = optimize_adaptive_grad(f_c, deadline, lr_init=0.005,
                                       top_k=3, use_weighted=True,
                                       verbose_prefix=verbose_prefix + "D:")
    if s_d < best_score:
        best_score = s_d
        best_f = f_d.copy()
    
    # Phase E: Final LP polish
    f_e, s_e = optimize_lp_heavy(f_d, deadline, step_init=0.03,
                                  verbose_prefix=verbose_prefix + "E:")
    if s_e < best_score:
        best_score = s_e
        best_f = f_e.copy()
    
    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)
    
    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)
    
    # Try to load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except Exception as e:
        print(f"No parent available: {e}")
    
    # Determine lengths to try
    parent_n = len(parent) if parent is not None else 1024
    lengths_to_try = [1024, 2048]
    if parent_n not in lengths_to_try and parent_n <= 2048:
        lengths_to_try.insert(0, parent_n)
    
    if parent_n > 2048:
        lengths_to_try = [parent_n, 1024, 2048]
    
    print(f"\nLengths to try: {lengths_to_try}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - t0:.0f}s")
    
    results = []  # (score, sequence, n, label)
    
    # Phase 1: Quick exploration to find promising candidates
    print("\n" + "=" * 60)
    print("PHASE 1: Quick Exploration")
    print("=" * 60)
    
    exploration_deadline = t0 + min(180.0, (deadline - t0) * 0.15)
    
    for n in lengths_to_try:
        if time.monotonic() > exploration_deadline - 20:
            break
        print(f"\n--- Exploring n={n} ---")
        
        inits = make_structured_init(n, rng)
        
        # Add parent-based inits
        if parent is not None:
            if len(parent) == n:
                inits['parent'] = parent.copy()
            else:
                inits['parent_interp'] = interpolate_to_length(parent, n)
        
        # Quick optimization of each init (15s each)
        for label, f_init in inits.items():
            if time.monotonic() > exploration_deadline - 10:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            if np.sum(f_init) < 1e-10:
                continue
            
            init_score = compute_score(f_init)
            if not np.isfinite(init_score):
                continue
            
            # Very quick optimization
            quick_deadline = min(exploration_deadline, time.monotonic() + 12.0)
            f_opt, s_opt = optimize_adaptive_grad(f_init, quick_deadline,
                                                   lr_init=0.01, top_k=5)
            if s_opt < init_score:
                init_score = s_opt
                f_init = f_opt
            
            print(f"  {label}: {init_score:.6f}")
            
            if s_opt < best_score:
                best_score = s_opt
                best_sequence = f_init.copy()
            
            results.append((s_opt, f_init.copy(), n, label))
    
    # Sort by score
    results.sort(key=lambda x: x[0])
    print(f"\nTop candidates after exploration:")
    for sc, seq, n, lbl in results[:5]:
        print(f"  n={n} {lbl}: {sc:.6f}")
    print(f"Best so far: {best_score:.6f}")
    
    # Phase 2: Focused optimization on top candidates
    print("\n" + "=" * 60)
    print("PHASE 2: Focused Optimization")
    print("=" * 60)
    
    # Select top candidates ensuring diversity in length
    seen_lengths = set()
    top_candidates = []
    for sc, seq, n, lbl in results:
        if n not in seen_lengths:
            top_candidates.append((sc, seq, n, lbl))
            seen_lengths.add(n)
        if len(top_candidates) >= 3:
            break
    
    # Also ensure best sequence is included
    if best_sequence is not None:
        best_n = len(best_sequence)
        if best_n not in seen_lengths:
            top_candidates.insert(0, (best_score, best_sequence.copy(), best_n, "best"))
    
    focus_deadline = deadline - 120.0  # Leave 120s for final polish
    
    for i, (sc, f_init, n, lbl) in enumerate(top_candidates):
        if time.monotonic() > focus_deadline - 30:
            break
        
        # Allocate time: more time for earlier candidates
        remaining = focus_deadline - time.monotonic()
        frac = 0.5 if i == 0 else (0.3 if i == 1 else 0.2)
        phase_deadline = time.monotonic() + remaining * frac
        
        print(f"\n--- Focused opt #{i}: n={n} {lbl} (start={sc:.6f}, "
              f"~{phase_deadline - time.monotonic():.0f}s) ---")
        
        f_opt, s_opt = full_optimize(f_init, phase_deadline, n, rng,
                                      verbose_prefix=f"[{i}]")
        
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  NEW BEST: {best_score:.6f}")
        else:
            print(f"  Best from this: {s_opt:.6f}")
    
    # Phase 3: Final polish on the best sequence
    print("\n" + "=" * 60)
    print(f"PHASE 3: Final Polish (current best={best_score:.6f})")
    print("=" * 60)
    
    if best_sequence is not None and time.monotonic() < deadline - 10:
        n_best = len(best_sequence)
        print(f"Polishing n={n_best}...")
        
        # Fine subgradient
        f_pol, s_pol = optimize_adaptive_grad(best_sequence, deadline,
                                               lr_init=0.003, top_k=3,
                                               use_weighted=True,
                                               verbose_prefix="[polish-grad] ")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()
            print(f"  After grad polish: {best_score:.6f}")
        
        # Final LP
        f_pol, s_pol = optimize_lp_heavy(best_sequence, deadline,
                                          step_init=0.02,
                                          verbose_prefix="[polish-LP] ")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()
            print(f"  After LP polish: {best_score:.6f}")
        
        # One more very fine subgradient
        f_pol, s_pol = optimize_adaptive_grad(best_sequence, deadline,
                                               lr_init=0.001, top_k=2,
                                               use_weighted=True)
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()
            print(f"  Final: {best_score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(1024)
        best_score = evaluate_sequence(best_sequence.tolist())
    
    # Ensure validity
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence invalid! Using fallback.")
        best_sequence = np.ones(512)
        final_score = evaluate_sequence(best_sequence.tolist())
    
    elapsed = time.monotonic() - t0
    print(f"\n{'=' * 60}")
    print(f"FINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")
    print(f"{'=' * 60}")
    
    return [float(x) for x in best_sequence]
