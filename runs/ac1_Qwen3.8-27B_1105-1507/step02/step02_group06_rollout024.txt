Optimized search for minimum 2n*max(conv(a,a))/sum(a)^2 using enhanced LP-based
direction finding with adaptive M reduction, spectral initialization, and
multi-phase optimization strategy.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = sum_j f[j]*g[k-j] = (f*g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M, n):
    """Solve: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                                  'primal_feasibility_tolerance': 1e-9})
        if result.success:
            return result.x
    except Exception as e:
        pass
    return None

def spectral_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    # Create a random phase with constant magnitude
    freqs = np.fft.rfftfreq(n)
    magnitude = np.ones(len(freqs))
    phase = rng.uniform(0, 2*np.pi, len(freqs))
    spectrum = magnitude * np.exp(1j * phase)
    # Make it Hermitian-symmetric for real output
    F = np.zeros(n, dtype=complex)
    F[:len(freqs)] = spectrum
    F[len(freqs):] = np.conj(F[1:len(freqs)][::-1]) if n % 2 == 0 else np.conj(F[1:len(freqs)-1][::-1])
    
    a = np.real(np.fft.ifft(F))
    a = np.clip(a, 0, None)
    if np.sum(a) < 1e-10:
        return np.ones(n)
    return a / np.sum(a)

def compute_score(f):
    """Compute 2n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def smooth_obj_grad(a, tau, n):
