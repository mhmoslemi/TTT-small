"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based alternating optimization with line search, smooth gradient
(log-sum-exp) polishing, and coordinate refinement. Tries multiple sequence
lengths and diverse initializations focused on the parent construction.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix: A[k,j] = a[j] if k-j is valid index, else 0.
    So that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # A[k, j] = a[k - j] for j in [j_start, j_end]
        idx = k - np.arange(j_start, j_end + 1)
        A[k, j_start:j_end + 1] = a[idx]
    return A


def solve_lp_min_max(a):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M_opt)."""
    n = len(a)
    A_conv = build_A(a)
    # Variables: g[0..n-1], M (total n+1 variables)
    # Objective: minimize M -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    # Inequality: A_conv @ g - M <= 0 -> [A_conv, -1] @ [g, M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], res.x[n]
    return None, None


def lp_alternate_with_linesearch(a_init, max_iter=200, verbose=False):
    """LP alternating optimization with line search on blend parameter."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(max_iter):
        g, M_opt = solve_lp_min_max(a)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        current_max = float(np.max(np.convolve(a, a)))

        # Line search over blend parameter
        best_t = 0.0
        best_trial_obj = best_obj
        for t in [0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 1.0]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = np.sum(a_trial)
            if st < 1e-10:
                continue
            a_trial = a_trial / st
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_trial_obj:
                best_trial_obj = obj_trial
                best_t = t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            s = np.sum(a_new)
            if s > 1e-10:
                a = a_new / s
            else:
                a = g
        else:
            a = g

        obj_a = compute_obj(a)
        if obj_a < best_obj:
            best_obj = obj_a
            best_a = a.copy()

        # Also check g directly
        obj_g = compute_obj(g)
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()

        if verbose and (it % 30 == 0):
            print(f"    LP it={it}: best_obj={best_obj:.6f}, M_opt={M_opt:.6f}")

        # Convergence check
        if it > 10 and abs(current_max - M_opt) / max(current_max, 1e-10) < 1e-6:
            break

    return best_a, best_obj


def smooth_gradient_polish(a, n_steps=5000, epsilon=50.0, verbose=False):
    """Polish using smooth gradient (log-sum-exp) on the max of convolution.
    Minimize f_eps(a) = (1/eps)*log(sum_k exp(eps*c_k)) s.t. sum(a)=1, a>=0."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    eta0 = 0.5
    alpha = 0.3

    for step in range(n_steps):
        # Compute convolution
        c = np.convolve(a, a)
        max_c = float(np.max(c))

        # Compute softmax weights (numerically stable)
        eps_c = epsilon * (c - max_c)
        exp_vals = np.exp(eps_c)
        p = exp_vals / np.sum(exp_vals)

        # Compute gradient: grad[j] = sum_k p_k * dc_k/da_j
        # dc_k/da_j = a[k-j]*I(0<=k-j<n) + a[j-k]*I(0<=j-k<n)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-12:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                # a[k-j] is valid (0 <= k-j <= n-1 by construction)
                grad[j] += pk * a[k - j]
                # a[j-k] is valid when 0 <= j-k < n
                if j - k >= 0 and j - k < n:
                    grad[j] += pk * a[j - k]

        # Adaptive step size
        eta = eta0 / (1.0 + alpha * step * 0.01)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

        if verbose and step % 500 == 0:
            print(f"    SG step={step}: obj={compute_obj(a):.6f}, best={best_obj:.6f}")

    return best_a, best_obj

        conv = np.convolve(p, a, mode='full')

        grad = corr[:n] + conv[:n]

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

        # Decay step size
        if step % 500 == 0 and step > 0:
            eta *= 0.85

    return best_a, best_obj


def coord_refine_fine(a, n_rounds=5, verbose=False):
    """Coordinate descent with fine local search around current value."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            old_val = a[j]
            if old_val < 1e-10:
                # Try small positive values
                candidates = [0.0, 0.001, 0.005, 0.01, 0.02]
            else:
                # Try fine multipliers around current value
                candidates = [old_val * m for m in [0.5, 0.7, 0.85, 0.95, 1.0, 1.05, 1.1, 1.3, 1.5]]
                candidates.append(0.0)

            best_x = old_val
            best_local = compute_obj(a)

            for x in candidates:
                a[j] = max(0.0, x)
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = max(0.0, x)

            if best_x != old_val:
                a[j] = best_x
                improved = True

            # Renormalize
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

        # Update best
        cur_obj = compute_obj(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        if not improved and rnd > 0:
            break

        if verbose:
            print(f"    Coord round {rnd}: obj={best_obj:.6f}")

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 7))
    j = np.arange(n)
    if choice == 0:
        # Raised cosine (high in middle)
        a = 0.5 * (1 + np.cos(2 * np.pi * (j - n / 2) / n))
    elif choice == 1:
        # Two-lobe (high at edges)
        a = 0.5 * (1 - np.cos(2 * np.pi * (j - n / 2) / n))
    elif choice == 2:
        # Gaussian centered
        sigma = rng.uniform(0.2, 0.4) * n
        a = np.exp(-0.5 * ((j - n / 2) / sigma) ** 2)
    elif choice == 3:
        # Two Gaussians at edges
        sigma = rng.uniform(0.1, 0.3) * n
        a = np.exp(-0.5 * (j / sigma) ** 2) + np.exp(-0.5 * ((j - n + 1) / sigma) ** 2)
    elif choice == 4:
        # Periodic with random period
        period = int(rng.integers(max(4, n // 20), max(5, n // 4)))
        freq = rng.integers(1, max(2, period // 2))
        a = np.abs(np.cos(2 * np.pi * freq * j / period)) + 0.1
    elif choice == 5:
        # Quadratic (parabola)
        a = 1.0 - (2 * (j - n / 2) / n) ** 2
        a = np.maximum(a, 0.01)
    else:
        # Smooth random
        a = rng.random(n)
        # Smooth with a moving average
        k = max(1, n // 10)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: LP alternation + smooth gradient + coordinate refinement.
    Starts from parent and explores nearby n values.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    def remaining():
        return deadline - time.monotonic()

    # Phase 1: Polish the parent (or best known) sequence
    if best_seq is not None and remaining() > 60:
        n_p = len(best_seq)
        print(f"\nPhase 1: Polishing parent n={n_p}, {remaining():.0f}s remaining")

        # LP alternation
        if remaining() > 30:
            max_iter = int(min(300, remaining() / 0.05))
            a_lp, obj_lp = lp_alternate_with_linesearch(best_seq, max_iter=max_iter, verbose=True)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  After LP: obj={best_score:.6f}")

        # Smooth gradient polish
        if remaining() > 30:
            n_steps = int(min(15000, remaining() / 0.005))
            if n_steps > 500:
                a_sg, obj_sg = fast_smooth_gradient(best_seq, n_steps=n_steps, epsilon=40.0)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    print(f"  After SG: obj={best_score:.6f}")

        # Coordinate refinement
        if remaining() > 20 and len(best_seq) <= 500:
            a_cr, obj_cr = coord_refine_fine(best_seq, n_rounds=5, verbose=True)
            if obj_cr < best_score:
                best_score = obj_cr
                best_seq = a_cr.copy()
                print(f"  After Coord: obj={best_score:.6f}")

        # Another round of LP + SG
        if remaining() > 30:
            max_iter = int(min(200, remaining() / 0.05))
            a_lp2, obj_lp2 = lp_alternate_with_linesearch(best_seq, max_iter=max_iter)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  After LP2: obj={best_score:.6f}")

            if remaining() > 20:
                n_steps = int(min(10000, remaining() / 0.005))
                if n_steps > 500:
                    a_sg2, obj_sg2 = fast_smooth_gradient(best_seq, n_steps=n_steps, epsilon=80.0)
                    if obj_sg2 < best_score:
                        best_score = obj_sg2
                        best_seq = a_sg2.copy()
                        print(f"  After SG2: obj={best_score:.6f}")

    # Phase 2: Try other n values
    n_values = [250, 280, 300, 320, 350, 400, 500]
    # Remove the parent n to avoid redundant work
    if best_seq is not None:
        n_p = len(best_seq)
        n_values = [n for n in n_values if abs(n - n_p) > 20]
        # Add n_p back with priority
        n_values = [n_p] + n_values

    for n in n_values:
        if remaining() < 30:
            break
        if n == len(best_seq) if best_seq is not None else False:
            continue  # Skip if already optimized

        print(f"\nPhase 2: Trying n={n}, {remaining():.0f}s remaining, best={best_score:.6f}")

        n_restarts = 3 if n <= 350 else 2
        for restart in range(n_restarts):
            if remaining() < 15:
                break

            if restart == 0 and parent is not None and len(parent) == n:
                a_init = parent.copy()
            else:
                a_init = structured_init(n, rng)

            # LP alternation
            max_iter = int(min(100, remaining() / 0.1))
            if max_iter > 20:
                a_lp, obj_lp = lp_alternate_with_linesearch(a_init, max_iter=max_iter)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  LP n={n} r={restart}: obj={best_score:.6f}")

            # Quick smooth gradient
            if remaining() > 15:
                n_steps = int(min(3000, remaining() / 0.01))
                if n_steps > 200:
                    a_sg, obj_sg = fast_smooth_gradient(a_lp if 'a_lp' in dir() else a_init,
                                                       n_steps=n_steps, epsilon=50.0)
                    if obj_sg < best_score:
                        best_score = obj_sg
                        best_seq = a_sg.copy()
                        print(f"  SG n={n} r={restart}: obj={best_score:.6f}")

    # Phase 3: Final intensive refinement of the best solution
    if best_seq is not None and remaining() > 20:
        n_best = len(best_seq)
        print(f"\nPhase 3: Final refinement n={n_best}, obj={best_score:.6f}, {remaining():.0f}s left")

        # Smooth gradient with high epsilon (sharp)
        if remaining() > 15:
            n_steps = int(min(20000, remaining() / 0.003))
            if n_steps > 1000:
                a_f, obj_f = fast_smooth_gradient(best_seq, n_steps=n_steps, epsilon=100.0)
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    print(f"  Final SG: obj={best_score:.6f}")

        # Coordinate refinement
        if remaining() > 15 and n_best <= 600:
            a_c, obj_c = coord_refine_fine(best_seq, n_rounds=10, verbose=True)
            if obj_c < best_score:
                best_score = obj_c
                best_seq = a_c.copy()
                print(f"  Final Coord: obj={best_score:.6f}")

        # One more LP pass
        if remaining() > 15 and n_best <= 500:
            max_iter = int(min(100, remaining() / 0.08))
            if max_iter > 20:
                a_l, obj_l = lp_alternate_with_linesearch(best_seq, max_iter=max_iter)
                if obj_l < best_score:
                    best_score = obj_l
                    best_seq = a_l.copy()
                    print(f"  Final LP: obj={best_score:.6f}")

        # Very fine coordinate polish
        if remaining() > 10 and n_best <= 400:
            a_f2, obj_f2 = coord_refine_fine(best_seq, n_rounds=5)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Ultra Coord: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
