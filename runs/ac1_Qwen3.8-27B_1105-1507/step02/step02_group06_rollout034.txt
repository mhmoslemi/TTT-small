import time

def compute_convolution(f):
    return np.convolve(f, f)

    return log_sum_exp

    n = len(f)
    conv = np.convolve(f, f)
    S = np.sum(f)
    
    # Softmax weights
    conv_centered = conv - np.max(conv)
    p = np.exp(conv_centered / tau)
    p /= np.sum(p)

    grad = (2.0 * n / (S * S)) * lse_grad - (4.0 * n * lse_val / (S * S * S)) * f
    
    return grad, lse_val

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 1e-15:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def compute_smoothed_grad(f, tau):
    """Compute gradient of 2n * LSE(conv(f,f), tau) / (sum(f))^2."""
    n = len(f)
    conv = np.convolve(f, f)
    S = float(np.sum(f))
    
    # Softmax weights (numerically stable)
    conv_centered = conv - np.max(conv)
    p = np.exp(conv_centered / tau)
    p_sum = float(np.sum(p))
    p /= p_sum
    
    # d LSE / d f[i] = 2 * sum_m f[m] * p[i+m]
    # = 2 * conv(f[::-1], p)[i + n - 1]
    f_rev = f[::-1]
    cross = np.convolve(f_rev, p)
    lse_grad = 2.0 * cross[n-1 : 2*n-1]
    
    # LSE value
    lse_val = float(np.max(conv) + tau * np.log(p_sum))
    
    # Full gradient
    grad = (2.0 * n / (S * S)) * lse_grad - (4.0 * n * lse_val / (S * S * S)) * f
    
    return grad, lse_val, M if 'M' in dir() else float(np.max(conv))

Combines annealed smoothed gradient descent (log-sum-exp with adaptive tau),

    n = len(f)
    conv = np.convolve(f, f)

    conv_max = float(np.max(conv))
    exp_vals = np.exp((conv - conv_max) / tau)
    exp_sum = float(np.sum(exp_vals))

    # d LSE / d f[i] = 2 * (correlation of f with p centered)
    cross = np.convolve(f[::-1], p)

def annealed_smoothed_descent(f_init, deadline, verbose_prefix=""):
    """Annealed smoothed gradient descent with Nesterov momentum."""
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-15:
        return f, float('inf')
    f = f / S
    best_f = f.copy()
    best_score = 2.0 * n * float(np.max(np.convolve(f, f)))
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.05
    beta_decay = 0.9997
    it = 0
    no_improve = 0
    epoch = 0
    t_start = time.monotonic()

    while time.monotonic() < deadline and it < 300000:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        conv_mean = float(np.sum(conv)) / (2 * n - 1)
        spread = M - conv_mean
        if spread < 1e-12:
            break
        # Adaptive tau: wider in early epochs, narrower later
        tau = spread * max(0.05, 0.5 - 0.005 * epoch)
        tau = max(tau, 1e-8)

        grad, lse_val, S_cur = smoothed_gradient(f, tau)
        grad -= float(np.mean(grad))
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-14:
            break
        grad_normed = grad / grad_norm

        momentum = beta * momentum + lr * grad_normed
        f_new = f + momentum
        f_new = np.maximum(f_new, 0.0)
        S_new = float(np.sum(f_new))
        if S_new < 1e-15:
            lr *= 0.5
            momentum *= 0.5
            it += 1
            no_improve += 1
            if no_improve > 300:
                break
            continue
        f_new = f_new / S_new

        conv_new = np.convolve(f_new, f_new)
        score = 2.0 * n * float(np.max(conv_new))
        it += 1
        epoch += 1

        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.85
                momentum *= 0.5

        if no_improve > 400:
            # Random restart perturbation
            f = best_f + 0.05 * np.random.default_rng(
                time.monotonic_ns() % 10000).standard_normal(n)

            S_r = float(np.sum(f))
            if S_r > 1e-15:
                f = f / S_r
            momentum = np.zeros(n)
            lr = 0.03
            no_improve = 0

        beta = max(0.3, beta * beta_decay)

        if it % 5000 == 0:
            print(f"    {verbose_prefix}iter={it} score={best_score:.6f} "
                  f"lr={lr:.5f} beta={beta:.3f} tau={tau:.5f} "

def coordinate_descent_refine(f_init, deadline, verbose_prefix=""):
    """Coordinate descent with randomized order for local refinement."""
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-15:
        return f, float('inf')
    f = f / S
    best_f = f.copy()
    best_score = 2.0 * n * float(np.max(np.convolve(f, f)))
    rng = np.random.default_rng(99)
    t_start = time.monotonic()

    while time.monotonic() < deadline:
        improved_this_round = False
        order = rng.permutation(n)
        for i in order:
            old_val = f[i]
            S_rest = S - old_val
            base_conv = np.convolve(np.delete(f, i), np.delete(f, i))
            f_k = np.delete(f, i)
            candidates = [0.0, old_val * 0.5, old_val * 0.8, old_val,
                          old_val * 1.2, old_val * 1.5, old_val * 2.0,
                          S / n, 0.1 * S / n, 0.01 * S / n]
            best_cv = old_val
            best_cs = 2.0 * n * float(np.max(np.convolve(f, f)))
            for cv in candidates:
                if cv < 0:
                    cv = 0.0
                new_S = S_rest + cv
                if new_S < 1e-15:
                    continue
                new_f = np.empty(n)
                new_f[:i] = f[:i]
                new_f[i] = cv
                new_f[i + 1:] = f[i + 1:]
                new_f = new_f / new_S
                new_conv = np.convolve(new_f, new_f)
                new_M = float(np.max(new_conv))
                new_score = 2.0 * n * new_M
                if new_score < best_cs:
                    best_cs = new_score
                    best_cv = cv
            if best_cv != old_val:
                f[i] = best_cv
                S = S_rest + best_cv
                f = f / S
                improved_this_round = True

        current_score = 2.0 * n * float(np.max(np.convolve(f, f)))
        if current_score < best_score:
            best_score = current_score
            best_f = f.copy()
        if not improved_this_round:
            break
        if time.monotonic() - t_start > 10:
            print(f"    {verbose_prefix}CD: score={best_score:.6f} "
                  f"time={time.monotonic()-t_start:.1f}s")

    return best_f, best_score


def lp_refine(f_init, deadline, verbose_prefix=""):
    """LP-based direction finding with line search."""
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-15:
        return f, float('inf')
    f = f / S
    best_f = f.copy()
    best_score = 2.0 * n * float(np.max(np.convolve(f, f)))
    step = 0.08
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        g = solve_lp_direction(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue
        g = g * (S / sum_g)
        best_step_score = best_score
        best_step_f = None
        for alpha in [step, step * 0.5, step * 1.5, step * 0.25]:
            alpha = min(alpha, 1.0)
            if alpha < 0.001:
                continue
            candidate = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            cand_sum = float(np.sum(candidate))
            if cand_sum < 1e-15:
                continue
            candidate = candidate / cand_sum
            cand_score = 2.0 * n * float(np.max(np.convolve(candidate, candidate)))
            if cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate
        it += 1
        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            step *= 0.99
            step = max(step, 0.005)
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.002 or no_improve > 40:
                break
        if it % 10 == 0:
            print(f"    {verbose_prefix}LP iter={it} score={best_score:.6f} "
                  f"step={step:.4f}")

    return best_f, best_score


def simulated_annealing(f_init, deadline, verbose_prefix=""):
    """Simulated annealing with coordinate perturbations."""
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-15:
        return f, float('inf')
    f = f / S
    best_f = f.copy()
    best_score = 2.0 * n * float(np.max(np.convolve(f, f)))
    current_score = best_score
    rng = np.random.default_rng(77)
    t_start = time.monotonic()
    total_time = max(deadline - t_start, 1.0)
    it = 0

    while time.monotonic() < deadline and it < 500000:
        elapsed_frac = (time.monotonic() - t_start) / total_time
        T = max(0.001, 1.0 * (1.0 - elapsed_frac) ** 2)
        n_perturb = rng.integers(1, max(2, n // 64))
        idx = rng.integers(0, n, size=n_perturb)
        f_new = f.copy()
        for i in idx:
            factor = np.exp(rng.standard_normal() * T * 2.0)
            f_new[i] = max(0.0, f[i] * factor)
        S_new = float(np.sum(f_new))
        if S_new < 1e-15:
            it += 1
            continue
        f_new = f_new / S_new
        new_score = 2.0 * n * float(np.max(np.convolve(f_new, f_new)))
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            f = f_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
        it += 1
        if it % 20000 == 0:
            print(f"    {verbose_prefix}SA iter={it} best={best_score:.6f} "
                  f"T={T:.4f} time={time.monotonic()-t_start:.1f}s")

    return best_f, best_score
