    val_sequence = [max(0.0, x) for x in val_sequence]
    val_sequence = [min(1000.0, x) for x in val_sequence]
    convex = np.convolve(val_sequence, val_sequence, mode='full')
    max_conv = float(np.max(convex))
    sum_val = float(np.sum(val_sequence))
    if sum_val < 0.01:
        return np.inf
    return 2.0 * n * max_conv / (sum_val ** 2)

def objective(x):
    x = np.minimum(np.maximum(x, 0.0), 1000.0)
    fval = evaluate_sequence(x)
    return fval

def combined_optimization(n, seed=42):
    """
    Combined optimization using multiple methods including gradient-based and simplex.
    Returns a sequence of length n.
    """
    # Initialization choices
    initial_type = np.random.randint(0, 5)
    if initial_type == 0:
        a = 0.5 + 0.5 * np.roll(0.5 * np.ones(n), int(seed * 100) % n, axis=-1) * (1 + np.cos(2 * np.pi * np.arange(n) / n + seed))
    elif initial_type == 1:
        a = np.ones(n)
        a[int(seed * 100) % n//3: int(seed * 200) % n//3] = seed + 0.1 * np.random.randn(len(a[int(seed * 100) % n//3: int(seed * 200) % n//3])) * (0.2 + 0.1 * np.random.rand()) * 100
    else:
        a = seed + 0.4 * np.random.randn(n) * (0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n) /n + seed))
    
    # Normalize to sum 1.0 to control scale
    a = a / np.sum(a)
    valley = minimize(objective, a, method='Nelder-Mead', options={'maxfev': 1500, 'xtol': 1e-8})
    candidate = valley.x
    candidate = candidate / np.sum(candidate)
    return candidate, evaluate_sequence(candidate), np.sum(candidate) >= 0.01
