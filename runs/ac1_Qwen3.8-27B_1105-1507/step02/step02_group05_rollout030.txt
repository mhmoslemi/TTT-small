def smooth_max_gradient(b, a, T):
    """Compute gradient of T*log(sum(exp(b_k/T))) with respect to a."""
    n = len(a)
    # Softmax over b
    b_scaled = b / T
    b_scaled -= b_scaled.max()
    p = np.exp(b_scaled)
    p /= p.sum()
    
    # Gradient: grad[m] = sum_k p[k] * (a[k-m] + a[m-k])
    n_b = len(b)  # 2n-1
    grad = np.zeros(n)
    for m in range(n):
        s = 0.0
        for k in range(n_b):
            k_minus_m = k - m
            m_minus_k = m - k
            if 0 <= k_minus_m < n:
                s += p[k] * a[k_minus_m]
            if 0 <= m_minus_k < n:
                s += p[k] * a[m_minus_k]
        grad[m] = s
    return grad

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def smooth_max_grad(b, a, T):
    """Gradient of T*log(sum(exp(b_k/T))) w.r.t. a"""
    n = len(a)
    # Compute softmax p over b
    b_shifted = b - b.max()
    p = np.exp(b_shifted / T)
    p /= p.sum()
    
    # Compute gradient using convolution
    W = 2*n - 1
    rev_p = p[::-1]
    # cross1[m] = sum_k p[k] * a[k-m] for m=0..n-1
    ca = np.convolve(rev_p, a, mode='full')
    cross1 = ca[n-1:W][::-1]
    
    # cross2[m] = sum_k p[k] * a[m-k] for m=0..n-1
    pa = np.convolve(p, a, mode='full')
    cross2 = pa[:n]
    
    return cross1 + cross2

def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def projected_grad_descent(a, max_iters, deadline, tracker):
    """PGD with smooth max for the convolution objective."""
    n = len(a)
    a = a / a.sum()
    
    T = 2.0 / n  # Initial temperature
    T_min = 0.5 / n
    T_decay = 0.995
    lr = 1.0 / n  # Initial learning rate
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        grad = smooth_max_grad(b, a, T)
        
        # Backtracking line search
        step = lr
        a_new = None
        for _ in range(20):
            candidate = proj_simplex(a - step * grad)
            b_cand = fast_conv(candidate)
            if b_cand.max() <= M + 1e-15:
                a_new = candidate
                break
            step *= 0.5
        
        if a_new is None:
            break
        
        a = a_new
        
        if it % 5 == 0:
            T = max(T_min, T * T_decay)
            tracker.update(a)
    
    tracker.update(a)
    return a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Continuation + Projected Gradient Descent with smooth-max (log-sum-exp) on convolution
2. Nesterov momentum for accelerated convergence
3. Coordinate descent polish (1D golden-section search per coordinate)
4. LP best-response mirror descent for final refinement

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(b, a, T):
    """Gradient of T*log(sum(exp(b_k/T))) w.r.t. a using FFT-based convolution."""
    n = len(a)
    W = 2 * n - 1
    # Softmax over b with temperature T
    b_shifted = b - b.max()
    p = np.exp(b_shifted / T)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(W) / W
    else:
        p /= p_sum

    # grad[m] = sum_k p[k]*(a[k-m] + a[m-k]) for valid indices
    # term1[m] = sum_k p[k]*a[k-m]  (cross-correlation)
    # term2[m] = sum_k p[k]*a[m-k]  (regular convolution)
    rev_p = p[::-1]
    ca = np.convolve(rev_p, a, mode='full')
    term1 = ca[n - 1:W][::-1]

    pa = np.convolve(p, a, mode='full')
    term2 = pa[:n]

    return term1 + term2


def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_A_sparse_n(a):
    """Build sparse convolution matrix A where (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    W = 2 * n - 1
    A = lil_matrix((W, n))
    for k in range(W):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse_n(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 5})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.3, 1.0, n) / n)
    inits.append(np.maximum(rng.normal(0, 1, n), 0) + 0.01)

    # Cosine profiles
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t - np.pi), 0.01))

    # Gaussian bump
    inits.append(np.exp(-5.0 * (t - 0.5) ** 2) + 0.05)
    # Two bumps
    inits.append(np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2) + 0.05)
    # Triangle
    inits.append(1.0 - np.abs(2 * t - 1.0) + 0.1)

    # Sparse spikes
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    kernel = np.ones(5) / 5.0
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.02)

    # Quadratic residue pattern (for appropriate n)
    p = n
    if n >= 7:
        # Find a prime near n
        def is_prime(num):
            if num < 2:
                return False
            for i in range(2, int(num ** 0.5) + 1):
                if num % i == 0:
                    return False
            return True
        prime = n
        for d in range(0, 10):
            if is_prime(n - d) and (n - d) % 4 == 1:
                prime = n - d
                break
            if is_prime(n + d) and (n + d) % 4 == 1:
                prime = n + d
                break
        if prime >= 7 and prime <= n + 5:
            residues = set()
            for x in range(1, prime):
                residues.add((x * x) % prime)
            qr = np.zeros(n)
            for r in residues:
                if r < n:
                    qr[r] = 1.0
            # Smooth slightly
            if n > 8:
                kernel2 = np.ones(3) / 3.0
                qr = np.convolve(qr, kernel2, mode='same')
            if qr.sum() > 0:
                inits.append(qr + 0.01)

    # Power law
    inits.append((t + 0.05) ** rng.uniform(-0.5, 1.0) + 0.01)

    # Sinc-like
    inits.append(np.maximum(np.abs(np.sin(np.pi * t)) / (np.pi * t + 0.1) + 0.1, 0.01))

    for init in inits:
        init = np.maximum(np.asarray(init, dtype=np.float64), 0.001)
        s = init.sum()
        if s > 0:
            init /= s
        inits[-1] = init  # replace with normalized
    return inits


def pgd_continuation(a, max_iters, deadline, tracker):
    """Projected gradient descent with smooth-max continuation and momentum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-6)
    a /= a.sum()

    T = 4.0 / n
    T_min = 0.3 / n
    T_decay = 0.997
    lr = 0.5 / n
    momentum = 0.7
    v = np.zeros(n)

    best_local = a.copy()
    best_local_M = fast_conv(a).max()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        if M < best_local_M - 1e-14:
            best_local_M = M
            best_local = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        grad = smooth_max_grad(b, a, T)
        # Center gradient (remove mean to preserve sum approximately)
        grad_c = grad - grad.mean()

        # Nesterov momentum
        v = momentum * v + grad_c
        a_new = proj_simplex(a - lr * v)

        # Backtracking: check if we improved
        b_new = fast_conv(a_new)
        M_new = b_new.max()
        if M_new >= M * (1 - 1e-12):
            # Halve step
            v = v * 0.5
            a_new = proj_simplex(a - lr * 0.5 * v)
            b_new = fast_conv(a_new)
            M_new = b_new.max()
            if M_new >= M:
                v = np.zeros(n)
                # Anneal: add noise to escape
                if stall_count > 50:
                    noise = rng_global.normal(0, 0.05, n)
                    a_new = np.maximum(a + noise, 0.001)
                    a_new = proj_simplex(a_new)
                    stall_count = 0
                elif it > 20:
                    break
        else:
            a = a_new

        # Schedule
        T = max(T_min, T * T_decay)
        lr = max(0.01 / n, lr * 0.9995)

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(best_local)
    return best_local


def coordinate_descent(a, max_passes, deadline, tracker):
    """Coordinate descent: optimize each coordinate via golden section search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            # Search range for a[j]
            lo = 0.0
            hi = old_val * 3.0 + 1.0 / n
            # Golden section
            gr = (np.sqrt(5) - 1) / 2
            best_t = old_val
            best_M = fast_conv(a).max()

            for _ in range(30):
                m1 = lo + gr * (hi - lo)
                m2 = hi - gr * (hi - lo)

                a[j] = m1
                a_j = np.maximum(a, 0)
                s_j = a_j.sum()
                if s_j < 1e-10:
                    a[j] = old_val
                    break
                a_j /= s_j
                M1 = fast_conv(a_j).max()

                a[j] = m2
                a_j = np.maximum(a, 0)
                s_j = a_j.sum()
                if s_j < 1e-10:
                    a[j] = old_val
                    break
                a_j /= s_j
                M2 = fast_conv(a_j).max()

                if M1 < M2:
                    hi = m2
                else:
                    lo = m1

                a_j = np.maximum(a, 0)
                s_j = a_j.sum()
                if s_j > 1e-10:
                    a_j /= s_j
                    M_mid = fast_conv(a_j).max()
                    if M_mid < best_M:
                        best_M = M_mid
                        best_t = a[j]
                        improved = True

            a[j] = best_t
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-10:
                a[j] = old_val
                continue
            a /= s

        if not improved:
            break
        if p % 2 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP-based mirror descent (best response)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive step
        base_t = max(0.05, 0.4 * (1.0 - it / float(2 * max_iters)))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5)]

        best_a = None
        best_J = current_J
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            t_small = 0.03
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def sa_restart(a, temperature, n_iters, deadline, tracker):

                a = coordinate_descent(a, 5, deadline, tracker)
            tracker.report("Phase5")
            print(f"  Phase5 done in {time.monotonic()-t0:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128
        tracker.best_score = compute_J(tracker.best)
        print("FALLBACK: using uniform sequence", flush=True)

    tracker.report("FINAL")
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
