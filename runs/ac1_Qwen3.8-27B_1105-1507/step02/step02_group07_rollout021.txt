def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):

            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            step_size = min(step_size * 1.1, 0.5)

            # Rebuild A_cache periodically
            if it % 10 == 0:
                A_cache = build_conv_matrix_sparse(f)

            if it % 30 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.7f} (t={best_t:.3f})")
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.003 or no_improve > 50:
                break

    return best_f, best_score


def compute_soft_max_gradient(f, beta):
    """Compute gradient of soft-max approximation to max(conv(f,f))."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    if max_C <= 0:
        return np.zeros(n), 0.0
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0
    p = exp_C / total
    corr = np.correlate(p, f, mode='valid')
    grad = 2.0 * corr
    soft_max_val = float(max_C + np.log(total) / beta)
    return grad, soft_max_val


def softmax_gradient_optimize(f, deadline, beta_start=20.0, beta_end=500.0,
                              lr_start=0.02, max_iters=5000, verbose_prefix=""):
    """Optimize using soft-max gradient with Adam and cosine LR schedule."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    no_improve = 0
    it = 0
    lr = lr_start

    while time.monotonic() < deadline and it < max_iters:
        it_frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** it_frac

        # Cosine annealing for learning rate
        lr_cos = lr * 0.5 * (1 + np.cos(np.pi * min(1.0, it / max_iters)))

        grad, soft_val = compute_soft_max_gradient(f, beta)

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)

        # Try a few step sizes
        found_improvement = False
        for lr_trial in [lr_cos, lr_cos * 0.5, lr_cos * 2.0]:
            new_f = f - lr_trial * update
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_score:
                f = new_f
                best_f = f.copy()
                best_score = sc
                no_improve = 0
                found_improvement = True
                break

        if not found_improvement:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break
        else:
            if it % 200 == 0:
                print(f"    {verbose_prefix}SM iter {it}: score={best_score:.7f} (beta={beta:.1f})")

    return best_f, best_score


def tilt_window(n, alpha=0.3, base='hann'):
    """Create a tilted window: base window * linear tilt."""
    idx = np.arange(n, dtype=np.float64)
    if base == 'hann':
        w = 0.5 * (1 + np.cos(2 * np.pi * idx / max(n - 1, 1)))
    elif base == 'tukey':
        w = np.ones(n)
        m = int(np.floor(0.3 * n / 2))
        if m > 0:
            for i in range(m):
                w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
                w[n - 1 - i] = w[i]
    else:
        w = np.ones(n)
    # Apply tilt
    tilt = 1.0 + alpha * (2.0 * idx / max(n - 1, 1) - 1.0)
    return w * np.clip(tilt, 0.1, None)


def parabolic_tilt(n, alpha=0.5):
    """Create a sequence with parabolic tilt for asymmetry."""
    idx = np.arange(n, dtype=np.float64)
    t = 2.0 * idx / max(n - 1, 1) - 1.0  # in [-1, 1]
    # Base: parabolic (1 - t^2)
    base = 1.0 - t ** 2
    # Tilt: linear
    tilt = 1.0 + alpha * t
    return np.maximum(base * tilt, 0.01)


def skewed_gaussian(n, center_frac=0.4, width_frac=0.25):
    """Create a skewed Gaussian centered at center_frac."""
    idx = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    width = width_frac * (n - 1)
    w = np.exp(-0.5 * ((idx - center) / width) ** 2)
    return w + 0.01


def perturb_directional(f, peak_lag, rng, strength=0.05):
    """Perturb f in a direction that specifically targets reducing the peak."""
    n = len(f)
    f_new = f.copy()
    # Reduce f at positions that contribute most to conv[peak_lag]
    for i in range(n):
        j = peak_lag - i
        if 0 <= j < n:
            f_new[i] *= (1.0 - strength * 0.5 * f[j] / (np.sum(f) + 1e-10))
    # Add some random perturbation
    noise = 1.0 + strength * 0.3 * rng.standard_normal(n)
    f_new *= np.clip(noise, 0.5, 1.5)
    f_new = np.clip(f_new, 0.0, 1000.0)
    S = float(np.sum(f_new))
    if S < 1e-10:
        return f.copy()
    return f_new / S


def perturb(f, rng, strength=0.05):
    """Add multiplicative noise and renormalize."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 2.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def full_optimize(f_init, deadline, phase1_frac=0.35, phase2_frac=0.45,
                  verbose_prefix=""):
    """Run multi-phase optimization."""
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    f_init = f_init / S

    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())
    t0 = time.monotonic()
    total_budget = deadline - t0

    # Phase 1: LP-based
    if time.monotonic() < deadline - 10:
        p1_deadline = t0 + total_budget * phase1_frac
        f1, s1 = lp_iterate_improved(f_init, p1_deadline, step_size_init=0.12,
                                      max_iters=200, verbose_prefix=verbose_prefix + "P1")
        if s1 < best_score:
            best_f, best_score = f1, s1

    # Phase 2: Soft-max gradient
    if time.monotonic() < deadline - 10:
        p2_deadline = t0 + total_budget * (phase1_frac + phase2_frac)
        f2, s2 = softmax_gradient_optimize(best_f, p2_deadline, beta_start=20.0,
                                            beta_end=400.0, lr_start=0.025,
                                            max_iters=4000, verbose_prefix=verbose_prefix + "P2")
        if s2 < best_score:
            best_f, best_score = f2, s2

    # Phase 3: LP polish
    if time.monotonic() < deadline - 5:
        f3, s3 = lp_iterate_improved(best_f, deadline - 5, step_size_init=0.04,
                                      max_iters=50, verbose_prefix=verbose_prefix + "P3")
        if s3 < best_score:
            best_f, best_score = f3, s3

    # Phase 4: Final high-beta gradient
    if time.monotonic() < deadline - 3:
        f4, s4 = softmax_gradient_optimize(best_f, deadline - 3, beta_start=200.0,
                                            beta_end=800.0, lr_start=0.005,
                                            max_iters=2000, verbose_prefix=verbose_prefix + "P4")
        if s4 < best_score:
            best_f, best_score = f4, s4

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
    Uses improved LP direction finding, adaptive soft-max gradient with cosine
    annealing, asymmetric tilt initializations, and multi-length exploration.
    Key insight: asymmetric sequences achieve score < 2 (symmetric ones cannot).
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.7f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    total_budget = deadline - t0

    # === Phase A: Refine n=512 from parent (largest time allocation) ===
    n_512_budget = total_budget * 0.40
    n_512_deadline = t0 + n_512_budget
    print(f"\n=== Phase A: n=512 refinement (budget {n_512_budget:.0f}s) ===")

    if best_sequence is not None and len(best_sequence) == 512:
        # Start from parent directly
        f_start = best_sequence.copy()
    else:
        # Interpolate or create
        if best_sequence is not None:
            from numpy import interp
            f_start = interp(np.linspace(0, 1, 512),
                             np.linspace(0, 1, len(best_sequence)),
                             best_sequence)
        else:
            f_start = tilt_window(512, alpha=0.3)
        f_start = np.clip(f_start, 0.0, 1000.0)
        S = float(np.sum(f_start))
        f_start = f_start / S

    try:
        bf, bs = full_optimize(f_start, n_512_deadline,
                                phase1_frac=0.4, phase2_frac=0.4,
                                verbose_prefix="[512] ")
        if bs < best_score:
            best_score = bs
            best_sequence = bf.copy()
            print(f"  n=512 refined: {bs:.7f}")
    except Exception as e:
        print(f"  n=512 error: {e}")

    # === Phase B: Try tilted/asymmetric initializations on n=512 ===
    n_512b_budget = total_budget * 0.15
    n_512b_deadline = min(deadline - 60, t0 + n_512_budget + n_512b_budget)
    print(f"\n=== Phase B: n=512 alternate inits (budget {n_512b_deadline - time.monotonic():.0f}s) ===")

    alt_inits_512 = [
        ('tilt_04', tilt_window(512, alpha=0.4, base='hann')),
        ('tilt_06', tilt_window(512, alpha=0.6, base='tukey')),
        ('parabolic', parabolic_tilt(512, alpha=0.4)),
        ('skew_gauss', skewed_gaussian(512, center_frac=0.35, width_frac=0.3)),
        ('skew_gauss2', skewed_gaussian(512, center_frac=0.45, width_frac=0.2)),
    ]

    # Also add perturbed parent if available
    if best_sequence is not None and len(best_sequence) == 512:
        for strength in [0.03, 0.08]:
            fp = perturb(best_sequence, rng, strength=strength)
            alt_inits_512.append((f'pert_{strength}', fp))
        # Directional perturbation
        conv_b = np.convolve(best_sequence, best_sequence)
        peak_lag = int(np.argmax(conv_b))
        alt_inits_512.append(('dir_pert', perturb_directional(best_sequence, peak_lag, rng, 0.04)))

    per_alt_time = max(10.0, (n_512b_deadline - time.monotonic()) / max(len(alt_inits_512), 1) * 0.7)

    for label, f_init in alt_inits_512:
        if time.monotonic() > n_512b_deadline:
            break
        f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
        S = float(np.sum(f_init))
        if S < 0.01:
            continue
        f_init = f_init / S
        init_sc = evaluate_sequence(f_init.tolist())
        if not np.isfinite(init_sc):
            continue
        if init_sc < best_score:
            best_score = init_sc
            best_sequence = f_init.copy()

        # Only optimize if promising
        if init_sc < best_score * 1.02:
            alt_deadline = min(n_512b_deadline, time.monotonic() + per_alt_time)
            if time.monotonic() < alt_deadline - 5:
                try:
                    bf, bs = full_optimize(f_init, alt_deadline,
                                            phase1_frac=0.5, phase2_frac=0.3,
                                            verbose_prefix=f"[512b {label}] ")
                    if bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        print(f"  {label}: improved to {bs:.7f}")
                except Exception as e:
                    print(f"  {label} error: {e}")

    # === Phase C: Try n=1024 ===
    n_1024_budget = total_budget * 0.25
    n_1024_start = max(time.monotonic(), t0 + n_512_budget + n_512b_budget)
    n_1024_deadline = min(deadline - 40, n_1024_start + n_1024_budget)
    print(f"\n=== Phase C: n=1024 (budget {n_1024_deadline - time.monotonic():.0f}s) ===")

    if time.monotonic() < n_1024_deadline - 20:
        # Interpolate best 512 to 1024
        if best_sequence is not None:
            from numpy import interp
            f_1024 = interp(np.linspace(0, 1, 1024),
                            np.linspace(0, 1, len(best_sequence)),
                            best_sequence)
        else:
            f_1024 = tilt_window(1024, alpha=0.3)
        f_1024 = np.clip(f_1024, 0.0, 1000.0)
        S = float(np.sum(f_1024))
        f_1024 = f_1024 / S

        try:
            bf, bs = full_optimize(f_1024, n_1024_deadline,
                                    phase1_frac=0.35, phase2_frac=0.45,
                                    verbose_prefix="[1024] ")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
                print(f"  n=1024: {bs:.7f}")
        except Exception as e:
            print(f"  n=1024 error: {e}")

    # === Phase D: Try n=768 (intermediate length) ===
    n_768_deadline = min(deadline - 20, time.monotonic() + total_budget * 0.10)
    print(f"\n=== Phase D: n=768 (budget {n_768_deadline - time.monotonic():.0f}s) ===")

    if time.monotonic() < n_768_deadline - 15:
        if best_sequence is not None:
            from numpy import interp
            f_768 = interp(np.linspace(0, 1, 768),
                          np.linspace(0, 1, len(best_sequence)),
                          best_sequence)
        else:
            f_768 = tilt_window(768, alpha=0.3)
        f_768 = np.clip(f_768, 0.0, 1000.0)
        S = float(np.sum(f_768))
        f_768 = f_768 / S

        try:
            bf, bs = full_optimize(f_768, n_768_deadline,
                                    phase1_frac=0.4, phase2_frac=0.4,
                                    verbose_prefix="[768] ")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
                print(f"  n=768: {bs:.7f}")
        except Exception as e:
            print(f"  n=768 error: {e}")

    # === Phase E: Perturbation restarts on best found ===
    print(f"\n=== Phase E: Restarts (current best: {best_score:.7f}, n={len(best_sequence) if best_sequence is not None else 0}) ===")

    if best_sequence is not None and time.monotonic() < deadline - 25:
        restart_budget = deadline - time.monotonic() - 15
        n_restarts = min(6, max(2, int(restart_budget / 40)))
        per_restart = restart_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break

            strengths = [0.02, 0.04, 0.06, 0.08, 0.12, 0.15]
            strength = strengths[r % len(strengths)]

            if r % 3 == 2:
                # Directional perturbation
                conv_b = np.convolve(best_sequence, best_sequence)
                peak_lag = int(np.argmax(conv_b))
                f_pert = perturb_directional(best_sequence, peak_lag, rng, strength)
            else:
                f_pert = perturb(best_sequence, rng, strength=strength)

            f_pert_deadline = time.monotonic() + per_restart * 0.8
            try:
                bf, bs = full_optimize(f_pert, f_pert_deadline,
                                        phase1_frac=0.4, phase2_frac=0.4,
                                        verbose_prefix=f"[rst{r}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strength:.3f}): {bs:.7f}")
                else:
                    print(f"  Restart {r} (str={strength:.3f}): no improvement ({bs:.7f})")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # === Phase F: Final intensive polish ===
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Phase F: Final polish (n={len(best_sequence)}, score={best_score:.7f}) ===")
        try:
            # Very high beta gradient
            bf, bs = softmax_gradient_optimize(best_sequence, deadline - 5,
                                                beta_start=300.0, beta_end=1000.0,
                                                lr_start=0.003, max_iters=3000,
                                                verbose_prefix="[final_SM] ")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()

            # Final LP
            if time.monotonic() < deadline - 2:
                bf, bs = lp_iterate_improved(best_sequence, deadline - 2,
                                              step_size_init=0.02, max_iters=30,
                                              verbose_prefix="[final_LP] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()

            print(f"  Final: {best_score:.7f}")
        except Exception as e:
            print(f"  Final polish error: {e}")

    # Ensure validity
    if best_sequence is None:
        best_sequence = tilt_window(512, alpha=0.3)
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.7f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
