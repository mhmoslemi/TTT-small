```python
def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_cross_corr_matrix(a):
    """Build the matrix A such that (A @ b)[k] = (a*b)[k] for cross-correlation."""
    n_a = len(a)
    n_max = 2 * n_a - 1
    A = np.zeros((n_max, n_a), dtype=np.float64)
    for k in range(n_max):
        for j in range(n_a):
            # (a*b)[k] = sum_j a[j] * b[k-j]

            idx = k - j
            if 0 <= idx < n_a:
                A[k, j] = a[idx]
    return A
```

(b*b)[k] = sum_{j≠i, j≠k-i} b[j]*b[k-j]

sum_{j≠i} b[j]*a[k-j] = (b*b)[k] (no x term since k-i is out of range)
(a*a)[k] = (b*b)[k] + 0 = (b*b)[k]

Let me summarize:
- If k = 2i (and 0 ≤ 2i < 2n-1): (a*a)[k] = c[k] + x^2
- If k ≠ 2i: (a*a)[k] = c[k] + 2x*b[k-i] (where b[m]=0 if m out of [0,n-1])

Let me verify with the example: n=3, a=[p,q,r], i=1, b=[p,0,r].
c = b*b = [p^2, 0, 2pr, 0, r^2]

k=0: c[0] + 2x*b[0-1] = p^2 + 2x*b[-1] = p^2 + 0 = p^2 ✓
k=1: c[1] + 2x*b[1-1] = 0 + 2x*b[0] = 2x*p = 2pq ✓
k=2: c[2] + x^2 = 2pr + q^2 ✓ (since k=2i=2)
k=3: c[3] + 2x*b[3-1] = 0 + 2x*b[2] = 2x*r = 2qr ✓
k=4: c[4] + 2x*b[4-1] = r^2 + 2x*b[3] = r^2 + 0 = r^2 ✓

So the formula is:
- If k = 2i: (a*a)[k] = c[k] + x^2
- Else: (a*a)[k] = c[k] + 2*x*b[k-i] (with b[m]=0 for m out of range)

Now the objective for coordinate i:
f(x) = 2n * max_k d_k(x) / (S+x)^2

where d_k(x) = c[k] + x^2 if k=2i, else c[k] + 2x*b[k-i], and S = sum(b).

To minimize f(x) over x >= 0:
- f(x) is continuous
- Each d_k(x) is either x^2 + c[2i] (quadratic) or 2x*b[k-i] + c[k] (linear)
- The max of these is convex (max of convex functions)
- Dividing by (S+x)^2 (which is convex and increasing) makes it more complex

Actually, f(x) = max_k [d_k(x) / (S+x)^2] (since 2n is a constant factor).

Each g_k(x) = d_k(x) / (S+x)^2:
- For linear d_k: g_k(x) = (A*x + B) / (S+x)^2 where A = 2*b[k-i], B = c[k]
  - g_k'(x) = [A(S+x)^2 - (Ax+B)*2(S+x)] / (S+x)^4 = [A(S+x) - 2(Ax+B)] / (S+x)^3 = [AS - Ax - 2B] / (S+x)^3
  - Critical point: x = (AS - 2B) / A (if A > 0)
  
- For quadratic d_k (k=2i): g_k(x) = (x^2 + B) / (S+x)^2 where B = c[2i]
  - g_k'(x) = [2x(S+x)^2 - (x^2+B)*2(S+x)] / (S+x)^4 = [2x(S+x) - 2(x^2+B)] / (S+x)^3 = [2xS + 2x^2 - 2x^2 - 2B] / (S+x)^3 = 2(xS - B) / (S+x)^3
  - Critical point: x = B/S (if B > 0)

So each piece has at most one critical point (a minimum, since the function goes to 0 as x→∞ and is positive).

The function f(x) = max_k g_k(x) is not necessarily convex, but it's the upper envelope of unimodal functions. We can find its minimum by:
1. Evaluating at x=0
2. Evaluating at all critical points of individual g_k
3. Evaluating at intersection points (where two g_k's are equal and both are the maximum)

For a practical approach, I'll just evaluate f(x) at many points and pick the best. Since this is a 1D problem, I can use fine grid search or golden section search.

Actually, since f(x) → 0 as x → ∞ (all g_k → 0), the minimum is not at infinity. And f(0) = 2n * max(c) / S^2 is the current objective (approximately). So the minimum is in [0, some reasonable upper bound].

For the coordinate descent, I'll:
1. For each coordinate i (in some order), find the best x by evaluating f(x) on a grid [0, 5*S/n] (or similar)
2. Update a[i] = best_x
3. Repeat for multiple passes

This should be effective because each coordinate update is exact (or near-exact).

```python
"""
Step function coefficient optimization using cyclic coordinate descent,
LP-based alternation, and subgradient methods to minimize
2n*max(a*a)/(sum a)^2. Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_cross_corr_matrix(a):
    """Build matrix A such that A @ b = a * b (full cross-correlation)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A


def solve_lp(a, M=None):
    """Solve LP: max sum(b) s.t. A(a) @ b <= M, b >= 0."""
    n = len(a)
    if M is None:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
    A = build_cross_corr_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None


def coordinate_descent(a, max_passes=20, time_limit=None):
    """Cyclic coordinate descent on the objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    if time_limit is None:
        start = time.monotonic()
    else:
        start = 0  # will be handled by caller
    
    for p in range(max_passes):
        if time_limit is not None and time.monotonic() - start > time_limit:
            break
        
        for i in range(n):
            if time_limit is not None and time.monotonic() - start > time_limit:
                    break
            
            # Save current value
            x_old = a[i]
            a[i] = 0.0
            b = a.copy()  # b has b[i] = 0
            S = float(np.sum(b))
            
            if S < 1e-12:
                a[i] = x_old
                continue
            
            # Precompute c = b * b
            c = np.convolve(b, b)
            
            # For each k, compute d_k(x) = c[k] + 2*x*b[k-i] (or c[k] + x^2 if k=2i)
            # We want to minimize f(x) = max_k d_k(x) / (S+x)^2 over x >= 0
            
            # Quick approach: evaluate on a grid
            # Upper bound for x: the current value scaled, or S/n * some factor
            x_max = max(5.0 * x_old, 5.0 * S / n, 1e-6)
            
            # Grid search
            num_points = 50
            x_grid = np.linspace(0.0, x_max, num_points)
            
            best_x = 0.0
            best_f = np.inf
            
            for x in x_grid:
                # Compute max_k d_k(x)
                # d_k(x) = c[k] + 2*x*b[k-i] for k != 2i
                # d_{2i}(x) = c[2i] + x^2 for k = 2i (if 0 <= 2i < 2n-1)
                
                w = np.zeros(2 * n - 1)
                # For k, b[k-i] is non-zero when 0 <= k-i < n, i.e., i <= k <= i+n-1
                k_start = max(0, i)
                k_end = min(2 * n - 2, i + n - 1)
                if k_start <= k_end:
                    w[k_start:k_end+1] = 2.0 * x * b[k_start-i:k_end-i+1]
                
                # Add x^2 term at k=2i
                if 0 <= 2 * i < 2 * n - 1:
                    w[2 * i] += x * x
                
                max_d = float(np.max(c + w))
                f_val = max_d / (S + x) ** 2
                
                if f_val < best_f:
                    best_f = f_val
                    best_x = x
            
            # Refine around best_x with finer grid
            if best_x > 0:
                x_fine = np.linspace(max(0, best_x * 0.5), best_x * 1.5, 50)
                for x in x_fine:
                    w = np.zeros(2 * n - 1)
                    k_start = max(0, i)
                    k_end = min(2 * n - 2, i + n - 1)
                    if k_start <= k_end:
                        w[k_start:k_end+1] = 2.0 * x * b[k_start-i:k_end-i+1]
                    if 0 <= 2 * i < 2 * n - 1:
                        w[2 * i] += x * x
                    max_d = float(np.max(c + w))
                    f_val = max_d / (S + x) ** 2
                    if f_val < best_f:
                        best_f = f_val
                        best_x = x
            elif best_x == 0:
                # Try small values
                x_fine = np.linspace(0.0, x_max * 0.1, 50)
                for x in x_fine:
                    w = np.zeros(2 * n - 1)
                    k_start = max(0, i)
                    k_end = min(2 * n - 2, i + n - 1)
                    if k_start <= k_end:
                        w[k_start:k_end+1] = 2.0 * x * b[k_start-i:k_end-i+1]
                    if 0 <= 2 * i < 2 * n - 1:
                        w[2 * i] += x * x
                    max_d = float(np.max(c + w))
                    f_val = max_d / (S + x) ** 2
                    if f_val < best_f:
                        best_f = f_val
                        best_x = x
            
            a[i] = best_x
        
        # After each pass, check objective
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj
```

```python
def coordinate_descent_fast(a, max_passes=20, time_limit=None, num_points=100):
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    
    for p in range(max_passes):
        if time.monotonic() - start > time_limit:
            break
        
        for i in range(n):
            if time.monotonic() - start > time_limit:
                break
            
            x_old = a[i]
            a[i] = 0.0
            b = a.copy()
            S = float(np.sum(b))
            
            if S < 1e-12:
                a[i] = x_old
                continue
            
            c = np.convolve(b, b)  # (2n-1,)
            
            # b_shifted[k] = b[k-i] for valid k
            b_shifted = np.zeros(2 * n - 1)
            b_shifted[i:i+n] = b  # positions i, i+1, ..., i+n-1 get b[0], b[1], ...
            
            # e_2i: unit vector at position 2i
            e_2i = np.zeros(2 * n - 1)
            if 0 <= 2 * i < 2 * n - 1:
                e_2i[2 * i] = 1.0
            
            # Grid search (vectorized)
            x_max = max(10.0 * x_old, 10.0 * S / n, 1e-6)
            x_grid = np.linspace(0.0, x_max, num_points)
            
            # For each x in x_grid:
            # total[k] = c[k] + 2*x*b_shifted[k] + x^2*e_2i[k]
            # = c[k] + x*(2*b_shifted[k]) + x^2*e_2i[k]
            
            # Shape: (P, 2n-1)
            # outer(x_grid, 2*b_shifted) gives (P, 2n-1) with [p,k] = x_grid[p] * 2*b_shifted[k]
            # outer(x_grid**2, e_2i) gives (P, 2n-1) with [p,k] = x_grid[p]^2 * e_2i[k]
            
            linear = np.outer(x_grid, 2.0 * b_shifted)  # (P, 2n-1)
            quad = np.outer(x_grid**2, e_2i)  # (P, 2n-1)
            total = c[None, :] + linear + quad  # (P, 2n-1)
            max_total = np.max(total, axis=1)  # (P,)
            f_vals = max_total / (S + x_grid)**2  # (P,)
            
            best_idx = int(np.argmin(f_vals))
            best_x = x_grid[best_idx]
            
            # Refine with finer grid around best
            margin = max(x_max * 0.05, abs(x_grid[1] - x_grid[0]))
            x_fine = np.linspace(max(0, best_x - margin), best_x + margin, num_points)
            linear_f = np.outer(x_fine, 2.0 * b_shifted)
            quad_f = np.outer(x_fine**2, e_2i)
            total_f = c[None, :] + linear_f + quad_f
            max_total_f = np.max(total_f, axis=1)
            f_vals_f = max_total_f / (S + x_fine)**2
            best_idx_f = int(np.argmin(f_vals_f))
            best_x = x_fine[best_idx_f]
            
            a[i] = best_x
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj
```

But wait, I can optimize further. Instead of recomputing c = b*b from scratch each time, I can update it incrementally. When I change a[i] from x_old to x_new, the change in a*a is:
(a_new * a_new)[k] - (a_old * a_old)[k] = (x_new - x_old) * (a[ k-i] + a[i-k]) + (x_new^2 - x_old^2) * delta(k, 2i)

This would make each coordinate update O(n) instead of O(n^2). Let me implement this:

```python
def coordinate_descent_incr(a, max_passes=50, time_limit=None, num_points=80):
    n = len(a)
    a = a.copy().astype(np.float64)
    conv = np.convolve(a, a).copy()  # maintain running convolution
    best_obj = 2.0 * n * float(np.max(conv)) / (float(np.sum(a))**2)
    best_a = a.copy()
    start = time.monotonic()
    S = float(np.sum(a))
    
    for p in range(max_passes):
        if time_limit is not None and time.monotonic() - start > time_limit:
            break
        
        improved = False
        for i in range(n):
            if time_limit is not None and time.monotonic() - start > time_limit:
                break
            
            x_old = a[i]
            b = a.copy()
            b[i] = 0.0
            S_b = S - x_old  # sum of b
            
            if S_b < 1e-12:
                continue
            
            # c = b * b = conv - (contribution of a[i] to conv)
            # conv[k] = c[k] + 2*x_old*b[k-i] + x_old^2*e_2i[k]
            # So c[k] = conv[k] - 2*x_old*b[k-i] - x_old^2*e_2i[k]
            
            b_shifted = np.zeros(2 * n - 1)
            b_shifted[i:i+n] = b
            e_2i = np.zeros(2 * n - 1)
            if 0 <= 2 * i < 2 * n - 1:
                e_2i[2 * i] = 1.0
            
            c = conv - 2.0 * x_old * b_shifted - x_old**2 * e_2i
            # Clamp to avoid numerical issues (should be >= 0)
            c = np.maximum(c, 0.0)
            
            # Grid search
            x_max = max(10.0 * x_old, 10.0 * S_b / n, 1e-6)
            x_grid = np.linspace(0.0, x_max, num_points)
            
            linear = np.outer(x_grid, 2.0 * b_shifted)
            quad = np.outer(x_grid**2, e_2i)
            total = c[None, :] + linear + quad
            max_total = np.max(total, axis=1)
            f_vals = max_total / (S_b + x_grid)**2
            
            best_idx = int(np.argmin(f_vals))
            best_x = x_grid[best_idx]
            
            # Refine
            if num_points > 4:
                step = x_max / num_points
                x_fine = np.linspace(max(0, best_x - step), best_x + step, num_points)
                linear_f = np.outer(x_fine, 2.0 * b_shifted)
                quad_f = np.outer(x_fine**2, e_2i)
                total_f = c[None, :] + linear_f + quad_f
                max_total_f = np.max(total_f, axis=1)
                f_vals_f = max_total_f / (S_b + x_fine)**2
                best_idx_f = int(np.argmin(f_vals_f))
                best_x = x_fine[best_idx_f]
            
            # Update
            a[i] = best_x
            S = S_b + best_x
            
            # Update conv incrementally
            # conv_new[k] = c[k] + 2*best_x*b_shifted[k] + best_x^2*e_2i[k]
```

```python
"""
Step function coefficient optimization using fast incremental coordinate descent,
LP-based alternation, and structured initialization to minimize
2n*max(a*a)/(sum a)^2. Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_cross_corr_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return A


def solve_lp(a, M=None):
    n = len(a)
    if M is None:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
    A = build_cross_corr_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None


def coord_descent(a, max_passes=100, time_limit=60.0, num_grid=60):
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-15:
        return a, np.inf
    a = a / S
    S = 1.0
    conv = np.convolve(a, a).copy()
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    tol = 1e-10

    for p in range(max_passes):
        if time.monotonic() - start > time_limit:
            break
        any_change = False
        for i in range(n):
            if time.monotonic() - start > time_limit:
                break
            x_old = a[i]
            S_b = S - x_old
            if S_b < 1e-12:
                continue

            b_shifted = np.zeros(2 * n - 1)
            b_shifted[i:i + n] = a
            pos_2i = 2 * i
            if 0 <= pos_2i < 2 * n - 1:
                b_shifted[pos_2i] = 0.0

            c = conv - 2.0 * x_old * b_shifted
            if 0 <= pos_2i < 2 * n - 1:
                c[pos_2i] -= x_old * x_old
            c = np.maximum(c, 0.0)

            x_max = max(20.0 * S_b / n, 5.0 * x_old + 1e-8)
            x_grid = np.linspace(0.0, x_max, num_grid)

            lin = np.outer(x_grid, 2.0 * b_shifted)
            if 0 <= pos_2i < 2 * n - 1:
                lin[:, pos_2i] += x_grid * x_grid
            total = c[None, :] + lin
            max_total = np.max(total, axis=1)
            f_vals = max_total / (S_b + x_grid) ** 2

            best_idx = int(np.argmin(f_vals))
            best_x = x_grid[best_idx]

            step = x_max / max(1, num_grid - 1)
            lo = max(0.0, best_x - 3 * step)
            hi = best_x + 3 * step
            x_fine = np.linspace(lo, hi, num_grid)
            lin_f = np.outer(x_fine, 2.0 * b_shifted)
            if 0 <= pos_2i < 2 * n - 1:
                lin_f[:, pos_2i] += x_fine * x_fine
            total_f = c[None, :] + lin_f
            max_total_f = np.max(total_f, axis=1)
            f_vals_f = max_total_f / (S_b + x_fine) ** 2
            best_idx_f = int(np.argmin(f_vals_f))
            best_x = x_fine[best_idx_f]

            if abs(best_x - x_old) > tol:
                a[i] = best_x
                S = S_b + best_x
                conv = c + 2.0 * best_x * b_shifted
                if 0 <= pos_2i < 2 * n - 1:
                    conv[pos_2i] += best_x * best_x
                any_change = True

        obj = 2.0 * n * float(np.max(conv))
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        if not any_change:
            break

    return best_a, best_obj


def lp_alternation(a, max_iters=80, time_limit=60.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-15:
        return a, np.inf
    a = a / S
    conv = np.convolve(a, a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        M = float(np.max(conv))
        b_sol, b_sum = solve_lp(a, M)
        if b_sol is None:
            break
        if b_sum <= 1.0 + 1e-8:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0
        b_norm = b_sol / b_sum
        t = 0.4 * (0.7 ** (it / 20.0))
        t = max(0.05, t)
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        conv_new = np.convolve(a_new, a_new)
        obj_new = 2.0 * n * float(np.max(conv_new))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new

    return best_a, best_obj


def subgradient_descent(a, max_iters=500, time_limit=30.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-15:
        return a, np.inf
    a = a / S
    conv = np.convolve(a, a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    lr = 0.05
    decay = 0.995

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        conv = np.convolve(a, a)
        max_val = float(np.max(conv))
        peak_idx = np.where(conv >= max_val - 1e-12)[0]

        grad = np.zeros(n)
        for k in peak_idx:
            for j in range(n):
                idx1 = k - j
                if 0 <= idx1 < n:
                    grad[j] += a[idx1]
                idx2 = j - k
                if 0 <= idx2 < n:
                    grad[j] += a[idx2]

        grad = grad / len(peak_idx)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = 2.0 * n * float(np.max(np.convolve(a_new, a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        lr *= decay
        if lr < 1e-6:
            lr = 0.02

    return best_a, best_obj


def structured_init(n, rng):
    choice = int(rng.integers(0, 5))
    if choice == 0:
        c = 1.0 / np.sqrt(2.0 * n - 1.0)
        a = np.ones(n) * 0.3
        a[0] = c
        a[1] = c
        a[-1] = c
        a[-2] = c
        if n > 6:
            a[2] = c * 0.7
            a[-3] = c * 0.7
        s = np.sum(a)
        return a / s
    elif choice == 1:
        center = (n - 1) / 2.0
        width = n * rng.uniform(0.3, 0.7)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a += 0.05
        return a / np.sum(a)
    elif choice == 2:
        a = np.zeros(n)
        a[0:n // 3] = 1.0
        a[2 * n // 3:n] = 1.0
        a += 0.1 * rng.random(n)
        return a / np.sum(a)
    elif choice == 3:
        period = max(2, int(rng.integers(n // 20, n // 3 + 1)))
        pattern = np.maximum(0, rng.random(period) - 0.15)
        a = np.tile(pattern, n // period + 1)[:n]
        a += 0.05 * rng.random(n)
        return a / np.sum(a)
    else:
        a = np.maximum(0, rng.normal(0.5, 0.4, n))
        a[0] += 0.5
        a[-1] += 0.5
        s = np.sum(a)
        return a / s if s > 1e-10 else np.ones(n) / n

            p_norm = parent / np.sum(parent)
            p_opt, p_obj = full_optimize(p_norm, min(120.0, float(budget_s) * 0.15))
            if p_obj < best_score:
                best_score = p_obj
                best_seq = p_opt.copy()
                print(f"Parent optimized (n={len(p_opt)}): obj={p_obj:.6f}")
    except Exception as e:
        print(f"Parent optimization error: {e}")

    # Try different n values with multiple initializations
    n_values = [200, 250, 300, 350, 400, 450, 500]
    print(f"\nSearching over n values: {n_values}")

    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        num_starts = max(2, min(3, int(remaining / 80)))
        time_per_start = min(35.0, remaining / max(1, num_starts) - 5)
        if time_per_start < 10:
            time_per_start = 10

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_starts} starts, {time_per_start:.0f}s each")

        for r in range(num_starts):
            if time.monotonic() > deadline - 30:
                break
            a_init = structured_init(n, rng)
            a_opt, obj_opt = full_optimize(a_init, time_per_start)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r}: obj={obj_opt:.6f}")

    # Final refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s remaining, n={len(best_seq)}")

        # Multiple refinement passes
        for ref_pass in range(3):
            if time.monotonic() > deadline - 10:
                break
            ref_time = min(remaining * 0.4, 40.0)
            if ref_time < 5:
                break
            a_ref, obj_ref = full_optimize(best_seq.copy(), ref_time)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  Refinement pass {ref_pass}: obj={best_score:.6f}")
            remaining = deadline - time.monotonic()

        # Coordinate descent with many passes on the final best
        if time.monotonic() < deadline - 10:
            cd_final_time = deadline - time.monotonic() - 5
            if cd_final_time > 5:
                a_fd, obj_fd = coord_descent(best_seq, max_passes=500, time_limit=cd_final_time)
                if obj_fd < best_score:
                    best_score = obj_fd
                    best_seq = a_fd.copy()
                    print(f"  Final CD: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
```
