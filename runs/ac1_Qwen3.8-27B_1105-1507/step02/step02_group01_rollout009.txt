"""
Optimized step function coefficient search using improved LP-based alternating
with optimal blending, adaptive M-tightening, soft-max gradient descent with
backtracking, and stochastic block perturbation with annealing.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective value for sequence a."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_cross_corr_matrix(a):
    """Build matrix A such that (A @ g)[k] = sum_i a[i]*g[k-i] (cross-correlation)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        # A[k, j] = a[k - j] for valid j
        for j in range(j_lo, j_hi + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. cross_corr(a,g) <= M, 0 <= g <= U. Returns g or None."""
    n = len(a)
    A = build_cross_corr_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    U = 2.0 * float(np.max(a)) + 1.0 / n
    bounds = [(0.0, U)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None


def optimal_blend(a, g, sum_a, sum_g):
    """Find optimal t in [0,1] minimizing F((1-t)a + t*g)."""
    n = len(a)

    def F_of_t(t):
        h = (1.0 - t) * a + t * g
        s_h = (1.0 - t) * sum_a + t * sum_g
        if s_h < 1e-10:
            return np.inf
        conv_h = np.convolve(h, h)
        return 2.0 * n * float(np.max(conv_h)) / (s_h * s_h)

    # Golden section search on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = F_of_t(c)
    fd = F_of_t(d)
    for _ in range(40):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = F_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = F_of_t(d)
    t_opt = (lo + hi) / 2.0
    return t_opt, F_of_t(t_opt)


def lp_optimize(a, time_budget, verbose=False):
    """LP-based alternating optimization with optimal blending and adaptive M."""
    n = len(a)
    a = a.copy().astype(np.float64)
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return a, np.inf
    a = a / sum_a
    sum_a = 1.0

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    M_target_mult = 1.0  # We'll try to use M * M_target_mult as bound

    iteration = 0
    while time.monotonic() - start < time_budget:
        conv = np.convolve(a, a)
        M = float(np.max(conv))

        # Try with current M first
        g, g_sum = solve_lp_direction(a, M)
        if g is None:

                    g = g_t
                    g_sum = g_sum_t
                    break
            if g_sum <= sum_a + 1e-8:
                stall_count += 1
                if stall_count > 15:
                    break

        stall_count = 0
        iteration += 1

        h = h / s_h

            best_obj = obj_h
            a = h
        else:

            print(f"    LP iter {iteration}: obj={best_obj:.6f}, t={best_t:.3f}, time={elapsed:.1f}s")

    return best_a, best_obj


def soft_max_grad(a, T):
    """Compute soft-max value and gradient for max(conv(a,a))."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    sm_val = max_c + T * np.log(sum_exp)

    # Gradient of sm_val w.r.t. a[i]
    # d(sm_val)/d(a[i]) = sum_k p[k] * d(conv[k])/d(a[i]) = sum_k p[k] * 2*a[k-i] (if valid)
    # Using the matrix: grad[i] = 2 * (p_cross @ a)[i] where p_cross[k,j] = p[k] * [valid]
    # More efficiently: grad = 2 * A_p @ a where A_p is cross-corr matrix with p
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-10:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            grad[j] += 2.0 * p[k] * a[k - j]
    return sm_val, grad


def soft_max_gd(a, time_budget, T_init=None):
    """Gradient descent on soft-max approximation with backtracking."""
    n = len(a)
    a = a.copy().astype(np.float64)
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return a, np.inf
    a = a / sum_a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.05 * float(np.max(conv))

    T = T_init
    lr = 1.0
    best_lr = lr
    last_improve_iter = 0
    no_improve = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        sm_val, grad = soft_max_grad(a, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Projected gradient step (project onto simplex)
        lr_try = lr
        improve = False
        for _ in range(10):
            a_new = a - lr_try * grad
            a_new = np.clip(a_new, 0.0, None)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                lr_try *= 0.5
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj or obj_new < compute_obj(a):
                a = a_new
                if obj_new < best_obj:
                    best_a = a_new.copy()
                    best_obj = obj_new
                    last_improve_iter = it
                    no_improve = 0
                    best_lr = lr_try
                improve = True
                break
            else:
                lr_try *= 0.5

        if improve:
            # Gradually decrease temperature
            if T > 0.0005:
                T = max(0.0005, T * 0.998)
            # Occasionally increase lr
            if it % 50 == 0 and lr < best_lr * 2:
                lr = min(best_lr * 1.5, 5.0)
        else:
            lr *= 0.7
            no_improve += 1
            if no_improve > 100:

                    a = a / s
                no_improve = 0

def block_perturb_anneal(a, time_budget, rng):
    """Stochastic block perturbation with simulated annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return a, np.inf
    a = a / sum_a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    deadline = start + time_budget

    T_anneal = 0.005
    T_min = 0.0001
    cooldown = 0.9995

    it = 0
    while time.monotonic() < deadline:
        # Random block size
        width = max(1, int(n * rng.uniform(0.02, 0.15)))
        pos = int(rng.integers(0, n - width + 1))

        # Perturbation
        candidate = a.copy()
        sigma = 0.02 + 0.05 * T_anneal / 0.005
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = float(np.sum(candidate))
        if s < 1e-10:
            it += 1
            continue
        candidate = candidate / s
        obj_cand = compute_obj(candidate)

        delta = obj_cand - compute_obj(a)
        if delta < 0 or rng.random() < np.exp(-delta / max(T_anneal, 1e-8)):
            a = candidate
            if obj_cand < best_obj:
                best_a = candidate.copy()
                best_obj = obj_cand
                T_anneal = max(T_min, T_anneal * 0.99)
            else:
                T_anneal = max(T_min, T_anneal * cooldown)
        else:
            T_anneal = max(T_min, T_anneal * cooldown)

        it += 1

    return best_a, best_obj


def coord_descent_polish(a, time_budget, rng):
    """Coordinate descent: try small changes to individual coordinates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return a, np.inf
    a = a / sum_a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    deadline = start + time_budget

    # Precompute: for each coordinate i, which conv entries are affected
    # When we change a[i] and renormalize, ALL conv entries change (due to renorm)
    # So just try perturbations

    steps = [0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001,
             0.02, -0.02, 0.05, -0.05]

    it = 0
    while time.monotonic() < deadline:
        i = int(rng.integers(0, n))
        # Try transferring mass from i to j
        j = int(rng.integers(0, n))
        if i == j:
            continue
        for step in steps:
            candidate = a.copy()
            candidate[i] += step
            candidate[j] -= step
            if candidate[i] < 0 or candidate[j] < 0:
                continue
            s = float(np.sum(candidate))
            if abs(s - 1.0) > 1e-6:
                continue  # Sum should be preserved
            obj_c = compute_obj(candidate)
            if obj_c < best_obj:
                best_a = candidate.copy()
                best_obj = obj_c
                a = candidate
                break
        it += 1
        if it % 1000 == 0:
            print(f"      CD iter {it}: obj={best_obj:.6f}")

    return best_a, best_obj


def edge_aware_init(n, rng):
    """Initialize with mass concentrated at edges."""
    c = 1.0 / np.sqrt(2.0 * n - 1.0)
    a = np.zeros(n)
    a[0] = c
    a[1] = c
    a[-1] = c
    a[-2] = c
    if n > 4:
        remaining = 1.0 - 4.0 * c
        if remaining < 0:
            remaining = 0.0
        interior = np.ones(n - 4) * (remaining / max(1, n - 4))
        noise = rng.normal(0, 0.2, n - 4)
        interior = np.maximum(0, interior * np.exp(noise))
        s = np.sum(interior)
        if s > 1e-10:
            interior = interior * remaining / s
        a[2:n - 2] = interior
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def structured_init(n, rng):

        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.1, 0.25)
        a = np.exp(-0.5 * ((np.arange(n) - c1) / w) ** 2) + np.exp(-0.5 * ((np.arange(n) - c2) / w) ** 2)
        a += 0.05 * np.ones(n)
        s = np.sum(a)
        return a / s
    elif choice == 4:
        # Triangle (peaked at edges)
        a = np.abs(np.arange(n) - n / 2.0) / (n / 2.0)
        a = 1.0 - a  # Inverted: peaked at edges
        a += 0.1
        s = np.sum(a)
        return a / s
    elif choice == 5:
        # Random non-negative
        a = np.maximum(0, rng.normal(0.5, 0.4, n))
        s = np.sum(a)
        return a / s if s > 1e-10 else np.ones(n) / n
    elif choice == 6:
        # Exponential decay from both ends
        decay = rng.uniform(0.5, 2.0)
        a = np.exp(-decay * np.abs(np.arange(n) - n / 2.0) / (n / 2.0))
        a += 0.1
        s = np.sum(a)
        return a / s
    else:
        # Periodic pattern
        period = max(2, int(n * rng.uniform(0.02, 0.1)))
        pattern = np.maximum(0.1, rng.random(period))
        a = np.tile(pattern, n // period + 1)[:n]
        s = np.sum(a)
        return a / s


def full_pipeline(a_init, time_budget, rng, verbose=False):
    """Run the full optimization pipeline on a given initialization."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    if verbose:
        print(f"    Init: obj={best_obj:.6f}")
    start = time.monotonic()

    # Phase 1: LP optimization
    lp_time = time_budget * 0.45
    if time_budget > 5:
        a, obj = lp_optimize(a, min(lp_time, time_budget * 0.45), verbose)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
            if verbose:
                print(f"    After LP: obj={best_obj:.6f}")

    # Phase 2: Soft-max GD
    remaining = (start + time_budget) - time.monotonic()
    if remaining > 3:
        gd_time = min(remaining * 0.3, time_budget * 0.25)
        a, obj = soft_max_gd(a, gd_time)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
            if verbose:
                print(f"    After GD: obj={best_obj:.6f}")

    # Phase 3: Second LP round
    remaining = (start + time_budget) - time.monotonic()
    if remaining > 5:
        lp_time2 = min(remaining * 0.3, time_budget * 0.15)
        a, obj = lp_optimize(a, lp_time2)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
            if verbose:
                print(f"    After LP2: obj={best_obj:.6f}")

    # Phase 4: Block perturbation with annealing
    remaining = (start + time_budget) - time.monotonic()
    if remaining > 3:
        bp_time = min(remaining * 0.5, time_budget * 0.15)
        a, obj = block_perturb_anneal(a, bp_time, rng)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
            if verbose:
                print(f"    After BP: obj={best_obj:.6f}")

    # Phase 5: Coordinate descent polish
    remaining = (start + time_budget) - time.monotonic()
    if remaining > 3:
        cd_time = min(remaining * 0.6, time_budget * 0.1)
        a, obj = coord_descent_polish(a, cd_time, rng)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
            if verbose:
                print(f"    After CD: obj={best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: structured init -> LP alternating -> soft-max GD
    -> block perturbation -> coordinate descent. Tries multiple sequence lengths
    and initializations with random restarts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf

    # Try parent construction
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Optimize the parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_time = min(80.0, float(budget_s) * 0.15)
            if parent.size <= 500:
                p_opt, p_obj = full_pipeline(parent.copy(), parent_time, rng, verbose=True)
                if p_obj < best_score:
                    best_score = p_obj
                    best_seq = p_opt.copy()
                    print(f"Parent optimized: obj={p_obj:.6f}")
    except Exception as e:
        print(f"Parent optimization failed: {e}")

    # Main search over different n values
    # Focus on n values that balance LP speed and objective quality
    n_values = [200, 250, 300, 350, 400, 500, 600, 750]

    # Shuffle slightly to add diversity
    n_values = n_values[:]  # Keep fixed order for predictability

    for n in n_values:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        num_remaining = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining)
        time_per_n = min(time_per_n, 120.0)

        num_restarts = max(2, min(4, int(time_per_n / 30)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts, {time_per_restart:.1f}s each")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 50:
                break
            a_init = structured_init(n, rng)
            a_opt, obj_opt = full_pipeline(a_init, time_per_restart, rng, verbose=(r == 0))

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r}: obj={obj_opt:.6f}")

    # Final refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s remaining")

        # Run full pipeline again on best
        a_ref, obj_ref = full_pipeline(best_seq.copy(), min(remaining * 0.5, 80.0), rng, verbose=True)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After final pipeline: obj={best_score:.6f}")

        # Extra block perturbation
        if time.monotonic() < deadline - 5:
            bp_time = deadline - time.monotonic() - 4.0
            if bp_time > 5:
                a_bp, obj_bp = block_perturb_anneal(best_seq, bp_time, rng)
                if obj_bp < best_score:
                    best_score = obj_bp
                    best_seq = a_bp.copy()
                    print(f"  After final BP: obj={best_score:.6f}")

        # Extra coordinate descent
        if time.monotonic() < deadline - 3:
            cd_time = deadline - time.monotonic() - 2.0
            if cd_time > 3:
                a_cd, obj_cd = coord_descent_polish(best_seq, cd_time, rng)
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    print(f"  After final CD: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
