    n = len(a)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based peak-minimization best-response iterations
2. Coordinate descent with exact 1D subproblem solving
3. Multi-start with diverse initializations and multiple sequence lengths
4. Smooth log-sum-exp refinement for final polish

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build Toeplitz-like matrix A where (A @ x)_k = conv(a, x)_k."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh+1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_min_peak(a):
    """
    Solve LP: min t s.t. conv(a,g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt).
    """
    n = len(a)
    m = 2 * n - 1
    A = build_conv_matrix(a)
    
    # Variables: x = [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A@g - t <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x[:n].sum() > 1e-10:
        g = np.maximum(res.x[:n], 0)
        g = g / g.sum()
        return g, float(res.x[-1])
    return None, float('inf')


def coord_descent_sweep(a, deadline, n_sweeps=10):
    """
    Coordinate descent: for each coordinate, find optimal 1D value.
    """
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            a_i = a[i]
            S = a.sum()
            
            # Compute base conv with a[i] = 0
            a[i] = 0.0
            b_base = fast_conv(a)
            a[i] = a_i
            
            # For new value x at position i:
            # b_k(x) = b_base_k + 2*x*a[k-i] (if 0<=k-i<n) + x^2 (if k=2i)
            # S(x) = S - a_i + x
            # Minimize max_k b_k(x) / S(x)^2
            
            # Quick check: try a few values
            candidates = [0.0, a_i * 0.5, a_i, a_i * 1.5, a_i * 2.0]
            # Also try the value that equalizes the top 2 lags
            offsets = np.arange(2 * n - 1) - i
            valid = (offsets >= 0) & (offsets < n)
            c_vals = np.zeros(2 * n - 1)
            c_vals[valid] = a[offsets[valid]]
            
            # The top 2 lags by current b value
            top2 = np.argsort(b_base + 2 * a_i * c_vals)[-2:]
            if len(top2) >= 2 and c_vals[top2[0]] != c_vals[top2[1]]:
                x_eq = (b_base[top2[1]] - b_base[top2[0]]) / (2 * (c_vals[top2[0]] - c_vals[top2[1]]))
                if x_eq > 0:
                    candidates.append(x_eq)
            
            best_x = a_i
            best_val = compute_score(a)
            
            for x in candidates:
                x = max(0.0, min(x, 1000.0))
                a[i] = x
                sc = compute_score(a)
                if sc < best_val:
                    best_val = sc
                    best_x = x
            a[i] = best_x
            
            if best_val < best_score:
                best_score = best_val
                best = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best, best_score


def lp_iteration(a, max_iters, deadline):
    """Run LP min-peak best-response iterations."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.0001, None)
    a = a / a.sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g, t_opt = lp_min_peak(a)
        if g is None:
            break
        
        # Line search on mixing parameter
        best_alpha = 0.0
        best_sc = best_score
        
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_new = (1 - alpha) * a + alpha * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_alpha = alpha
            if alpha == 1.0 and sc <= best_score:
                break
        
        if best_alpha > 0:
            a_new = (1 - best_alpha) * a + best_alpha * g
            a_new = np.clip(a_new, 0.0, None)
            a_new = a_new / a_new.sum()
            a = a_new
            if best_sc < best_score:
                best_score = best_sc
                best_local = a_new.copy()
        else:
            # No improvement, try smaller steps
            broke = True
            for alpha in [0.05, 0.02, 0.01]:
                a_new = (1 - alpha) * a + alpha * g
                a_new = np.clip(a_new, 0.0, None)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new = a_new / s
                sc = compute_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_local = a_new.copy()
                    a = a_new
                    broke = False
                    break
            if broke:
                # Check convergence
                if it > 20:
                    break
                else:
                    continue
    
    return best_local, best_score


def smooth_objective(x, beta, n):
    """Log-sum-exp smoothed objective for normalized non-negative x."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / beta
    return float(2.0 * n * log_sum)


def smooth_refine(a, deadline):
    """Refine using smooth log-sum-exp with L-BFGS-B, beta annealing."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    
    betas = [50, 200, 1000, 5000]
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = best.copy()
        x0 = np.clip(x0, 0.001, None)
        x0 = x0 / x0.sum()
        
        try:
            res = minimize(smooth_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 100, 'maxfun': 1500,
                                    'ftol': 1e-18, 'gtol': 1e-18})
            if res.fun < 1e10:
                res_x = np.clip(res.x, 0, None)
                if res_x.sum() > 0.01:
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
        except Exception:
            pass
    
    return best, best_score

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based peak-minimization best-response (directly minimizes achievable peak)
2. Coordinate descent with exact 1D subproblem solving
3. Multi-start with diverse initializations and multiple sequence lengths
4. Smooth log-sum-exp refinement with L-BFGS-B for final polish
5. Gaussian-mixture parameterization for fast initial exploration
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix A where (A @ x)_k = conv(a, x)_k for k=0..2n-2."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_min_peak(a):
    """
    Solve LP: min t s.t. conv(a,g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g_normalized, t_opt).
    """
    n = len(a)
    m = 2 * n - 1
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x[:n].sum() > 1e-10:
        g = np.maximum(res.x[:n], 0)
        g = g / g.sum()
        return g, float(res.x[-1])
    return None, float('inf')


def coord_descent_sweep(a, deadline, n_sweeps=10, rng=None):
    """
    Coordinate descent: for each coordinate, find optimal 1D value via golden section.
    """
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    a = a.copy()

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        improved = False
        order = list(range(n))
        if rng is not None:
            rng.shuffle(order)

        for i in order:
            if time.monotonic() > deadline:
                break

            a_i = a[i]
            S = a.sum()

            # Compute base conv with a[i] = 0
            a[i] = 0.0
            b_base = fast_conv(a)
            a[i] = a_i

            # For new value x: b_k(x) = b_base_k + 2*x*a[(k-i) mod n valid] + x^2*(k==2i)
            offsets = np.arange(2 * n - 1) - i
            valid = (offsets >= 0) & (offsets < n)
            c_vals = np.zeros(2 * n - 1)
            c_vals[valid] = a[offsets[valid]]
            quad_mask = (offsets == i)  # k - i == i means k == 2i

            def obj_1d(x):
                Sx = S - a_i + x
                if Sx <= 1e-10:
                    return 1e10
                b_x = b_base + 2.0 * x * c_vals
                b_x[quad_mask] += x * x
                return b_x.max() / (Sx * Sx)

            # Golden section search on [0, hi]
            x_lo = 0.0
            x_hi = max(3.0 * a_i + 1e-6, S * 0.1)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            f_c = obj_1d(c_pt)
            f_d = obj_1d(d_pt)

            for _ in range(40):
                if f_c < f_d:
                    x_hi = d_pt
                    d_pt = c_pt
                    f_d = f_c
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    f_c = obj_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    f_c = f_d
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    f_d = obj_1d(d_pt)

            x_opt = (x_lo + x_hi) / 2.0
            a[i] = max(0.0, x_opt)

            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best = a.copy()
                improved = True

        if not improved and sweep > 0:
            break

    return best, best_score


def lp_iteration(a, max_iters, deadline, rng=None):
    """Run LP min-peak best-response iterations with line search."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 1e-6, None)
    a = a / a.sum()

    stall_count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g, t_opt = lp_min_peak(a)
        if g is None:
            break

        # Line search on mixing parameter alpha
        best_alpha = 0.0
        best_sc = best_score

        alphas = [0.3, 0.5, 0.7, 1.0, 0.1, 0.2, 0.9]
        for alpha in alphas:
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_alpha = alpha
                if 1.0 - sc / best_score > 0.05:
                    break  # big improvement, take it

        if best_alpha > 0:
            a_new = (1.0 - best_alpha) * a + best_alpha * g
            a_new = np.clip(a_new, 0.0, None)
            a_new = a_new / a_new.sum()
            a = a_new
            if best_sc < best_score:
                best_score = best_sc
                best_local = a_new.copy()
                stall_count = 0
            else:
                stall_count += 1
        else:
            # Try extrapolation
            for beta in [1.2, 1.5, 2.0]:
                a_new = a + beta * (g - a)
                a_new = np.clip(a_new, 0.0, None)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new = a_new / s
                sc = compute_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_local = a_new.copy()
                    a = a_new
                    stall_count = 0
                    break
            else:
                stall_count += 1
                if stall_count > 15:
                    break
                # Perturb to escape
                if rng is not None:
                    a = a + rng.normal(0, 0.01, len(a))
                    a = np.clip(a, 1e-6, None)
                    a = a / a.sum()

    return best_local, best_score


def smooth_objective(x, beta, n):
    """Log-sum-exp smoothed objective."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / beta
    return float(2.0 * n * log_sum)


def smooth_refine(a, deadline):
    """Refine using smooth log-sum-exp with L-BFGS-B, beta annealing."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)

    betas = [30, 100, 500, 2000]
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = best.copy()
        x0 = np.clip(x0, 0.001, None)
        x0 = x0 / x0.sum()

        try:
            res = minimize(smooth_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 80, 'maxfun': 1000,
                                    'ftol': 1e-18, 'gtol': 1e-18})
            if res.fun < 1e10:
                res_x = np.clip(res.x, 0, None)
                if res_x.sum() > 0.01:
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
        except Exception:
            pass

    return best, best_score


def gaussian_mixture_init(n, params, dtype=np.float64):
    """Create sequence from Gaussian mixture parameters: list of (weight, center, width)."""
    t = np.linspace(0, 1, n)
    a = np.zeros(n)
    for w, c, s in params:
        a += w * np.exp(-0.5 * ((t - c) / s) ** 2)
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    return a


def optimize_gaussian_mixture(n, n_bumps, deadline, rng):

                    c = 0.1 + 0.8 * (1.0 / (1.0 + np.exp(-res.x[3 * b + 1] * 5)))
                    s = np.exp(res.x[3 * b + 2])
                    trial_params.append((w, c, s))
                best_a = gaussian_mixture_init(n, trial_params)
                best_score = float(res.fun)
        except Exception:
            pass

    if best_a is not None:
        return best_a, best_score
    return None, float('inf')


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n)

    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())

    inits.append(np.ones(n) / n)

    for freq, amp in [(1, 0.3), (2, 0.25), (3, 0.2), (1, 0.5), (2, 0.4), (4, 0.15)]:
        c = 1.0 + amp * np.cos(2 * np.pi * freq * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    c = np.exp(-4 * (t - 0.5) ** 2)
    inits.append(c / c.sum())

    c = np.exp(-30 * (t - 0.25) ** 2) + np.exp(-30 * (t - 0.75) ** 2)
    inits.append(c / c.sum())

    c = 1.0 + 0.5 * (2 * t - 1) ** 2
    inits.append(c / c.sum())

    inits.append(rng.uniform(0.2, 1.0, n) / n)

    x = rng.uniform(0, 1, min(n, 8))
    c = np.maximum(np.interp(t, np.linspace(0, 1, len(x)), x), 0.01)
    inits.append(c / c.sum())

    # S-curve
    c = 1.0 / (1.0 + np.exp(-8 * (t - 0.5)))
    inits.append(c / c.sum())

    # Power law
    c = t ** 0.5 + (1 - t) ** 0.5
    inits.append(c / c.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: LP peak-minimization + coordinate descent + smooth refinement.
    Uses multiple lengths, diverse initializations, and perturbation search.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 10.0

    best = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            sc = compute_score(p)
            if sc < best_score:
                best = p.copy()
                best_score = sc
                print(f"Parent: J={sc:.6f} (n={p.size})", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 0: Gaussian mixture exploration (fast) ===
    if best is None or best_score > 1.6:
        print(f"\n=== Phase 0: Gaussian Mixture Init ===", flush=True)
        t0 = time.monotonic()
        parent_len = best.size if best is not None else 512
        for n in [parent_len, 384, 640]:
            if time.monotonic() > t0 + 60:
                break
            gm_a, gm_sc = optimize_gaussian_mixture(n, 3, t0 + 120, rng)
            if gm_a is not None and gm_sc < best_score:
                # Refine with a few LP steps
                gm_a_nm = gm_a / gm_a.sum()
                a_r, sc_r = lp_iteration(gm_a_nm, 200, min(t0 + 120, deadline))
                a_r, sc_r = coord_descent_sweep(a_r, min(t0 + 120, deadline), 5, rng)
                if sc_r < best_score:
                    best = a_r.copy()
                    best_score = sc_r
                    print(f"  GM init: J={sc_r:.6f} (n={n})", flush=True)
        print(f"  Phase 0 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    if best is None:
        best = np.ones(512) / 512
        best_score = compute_score(best)
        print(f"  Default init: J={best_score:.6f}", flush=True)

    parent_len = best.size

    # === Phase 1: LP best-response + CD ===
    print(f"\n=== Phase 1: LP Best-Response + CD ===", flush=True)
    t0 = time.monotonic()
    lp_budget = total_budget * 0.45
    lp_deadline = min(time.monotonic() + lp_budget, deadline)

    lengths = [parent_len]
    for l in [256, 384, 512, 640, 768, 896, 1024, 1280]:
        if l not in lengths and l <= 1280:
            lengths.append(l)
    lengths.sort(key=lambda l: abs(l - parent_len))

    for n in lengths:
        if time.monotonic() > lp_deadline:
            break
        remaining = lp_deadline - time.monotonic()
        if remaining < 45:
            break

        print(f"\n  LP+CD: n={n}, time_left={remaining:.0f}s", flush=True)
        t_n0 = time.monotonic()

        inits = generate_initializations(n, rng, best)
        max_inits = min(len(inits), max(3, int(remaining / 50)))

        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > lp_deadline:
                break

            a0 = np.clip(a0, 1e-6, None)
            a0 = a0 / a0.sum()

            remaining = lp_deadline - time.monotonic()
            est_time = 0.04 if n <= 512 else 0.08 if n <= 768 else 0.12
            n_remaining_inits = max(1, max_inits - i)
            max_iters = max(50, int(remaining * 0.6 / (est_time * n_remaining_inits)))
            max_iters = min(max_iters, 5000)

            a_result, score = lp_iteration(a0, max_iters, lp_deadline, rng)

            # Follow up with CD
            if time.monotonic() < lp_deadline - 10:
                a_result, score = coord_descent_sweep(a_result, lp_deadline, 5, rng)

            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"    NEW BEST: J={score:.6f} (n={n}, init={i}, {time.monotonic()-t_n0:.1f}s)", flush=True)

        print(f"  n={n} done in {time.monotonic()-t_n0:.1f}s", flush=True)

    print(f"\n  Phase 1 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 2: Smooth refinement ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Phase 2: Smooth Refinement ===", flush=True)
        t0 = time.monotonic()
        smooth_deadline = min(time.monotonic() + total_budget * 0.2, deadline - 60)

        a_result, score = smooth_refine(best, smooth_deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (smooth): J={score:.6f}", flush=True)

        # CD after smooth
        if time.monotonic() < smooth_deadline:
            a_result, score = coord_descent_sweep(best, smooth_deadline, 5, rng)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST (CD after smooth): J={score:.6f}", flush=True)

        # Try neighboring lengths
        for n2 in [max(128, best.size - 128), max(128, best.size + 128)]:
            if time.monotonic() > smooth_deadline:
                break
            if n2 == best.size:
                continue
            t_old = np.linspace(0, 1, best.size)
            t_new = np.linspace(0, 1, n2)
            a2 = np.maximum(np.interp(t_new, t_old, best), 0)
            if a2.sum() > 0.01:
                a2 = a2 / a2.sum()
                a_result2, score2 = smooth_refine(a2, smooth_deadline)
                if time.monotonic() < smooth_deadline:
                    a_result2, score2 = coord_descent_sweep(a_result2, smooth_deadline, 3, rng)
                if score2 < best_score:
                    best = a_result2.copy()
                    best_score = score2
                    print(f"  NEW BEST (smooth n={n2}): J={score2:.6f}", flush=True)

        print(f"  Phase 2 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 3: Multi-perturbation search ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Phase 3: Perturbation Search ===", flush=True)
        t0 = time.monotonic()
        n = best.size
        remaining = deadline - time.monotonic()
        n_tries = max(5, min(40, int(remaining / 20)))
        noise_scales = [0.02, 0.05, 0.03, 0.08, 0.01, 0.1, 0.04, 0.06]

        for trial in range(n_tries):
            if time.monotonic() > deadline:
                break

            scale = noise_scales[trial % len(noise_scales)]
            noise = rng.normal(0, scale, n)
            a_pert = best + noise
            a_pert = np.clip(a_pert, 0, None)
            if a_pert.sum() < 0.01:
                continue
            a_pert = a_pert / a_pert.sum()

            remaining = deadline - time.monotonic()
            est_time = 0.04 if n <= 512 else 0.08
            n_left = max(1, n_tries - trial)
            max_iters = max(30, int(remaining / n_left / est_time / 3))
            max_iters = min(max_iters, 2000)

            a_result, score = lp_iteration(a_pert, max_iters, deadline, rng)
            if time.monotonic() < deadline:
                a_result, score = coord_descent_sweep(a_result, deadline, 3, rng)

            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST (pert {trial}): J={score:.6f}", flush=True)

        print(f"  Phase 3 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === Phase 4: Final intensive polish ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Phase 4: Final Polish ===", flush=True)
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        n = best.size
        est_time = 0.04 if n <= 512 else 0.08

        # Extended LP
        max_iters = max(200, int(remaining * 0.4 / est_time))
        max_iters = min(max_iters, 10000)
        a_result, score = lp_iteration(best, max_iters, deadline, rng)
        if score < best_score:
            best = a_result.copy()
            best_score = score

        # Extended CD
        if time.monotonic() < deadline:
            a_result, score = coord_descent_sweep(best, deadline, 15, rng)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  CD polish: J={score:.6f}", flush=True)

        # Smooth polish
        if time.monotonic() < deadline - 5:
            a_result, score = smooth_refine(best, deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  Smooth polish: J={score:.6f}", flush=True)

            # One more CD
            if time.monotonic() < deadline - 2:
                a_result, score = coord_descent_sweep(best, deadline, 5, rng)
                if score < best_score:
                    best = a_result.copy()
                    best_score = score
                    print(f"  Final CD: J={score:.6f}", flush=True)

        print(f"  Phase 4 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    if best is None:
        best = np.ones(256) / 256
        best_score = compute_score(best)

    # Final validation with actual evaluator
    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
