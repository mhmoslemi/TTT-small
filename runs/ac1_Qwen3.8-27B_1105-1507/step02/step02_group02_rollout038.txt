The step size $\eta$ controls how aggressive this shift is. Too large: we might zero out some coordinates. Too small: slow convergence.

def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def cross_corr_c_a(c, a):

    Compute g_j = sum_l a_l * c_{l+j} for j = 0, ..., n-1.
    where c has length 2n-1 and a has length n.
    Uses FFT.

    mc = len(c)  # 2n-1

    # = IFFT(conj(FFT(a_L)) * FFT(c_L)) ... hmm let me think

    # Via FFT: IFFT(FFT(a_L) * conj(FFT(c_L))) ... no
    # Standard: xcorr(x, y)_k = sum_n x_{n+k} conj(y_n)
    # So xcorr(c, a)_j = sum_l c_{l+j} conj(a_l) = sum_l c_{l+j} * a_l (all real)

    # For our purposes (with zero-padding), this equals sum_l c[l+j] * a[l] for valid j.
    # FFT: (c ⋆ a)[j] = IFFT(conj(FFT(c_L)) * FFT(a_L))[j]

    result = np.fft.irfft(np.conj(fcc) * fac, L)

def exp_gradient_step(a, epsilon_frac, eta):
    """
    One exponentiated gradient step.
    a: normalized sequence (sums to 1)
    epsilon_frac: fraction of max to define active set
    eta: step size
    Returns new normalized a.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    if b_max < 1e-15:
        return a.copy()
    
    # Active set
    threshold = b_max * (1.0 - epsilon_frac)
    S = b >= threshold
    if S.sum() == 0:
        S = np.zeros(2*n-1, dtype=bool)
        S[np.argmax(b)] = True
    
    # Compute gradient g_j = (1/|S|) * sum_{k in S} a_{k-j} * 1[0 <= k-j < n]
    c = S.astype(np.float64) / S.sum()
    g = cross_corr_c_a(c, a)
    
    # Exponentiated update
    # a_j_new = a_j * exp(-eta * g_j) / sum(...)
    log_a = np.log(np.maximum(a, 1e-300))
    update = log_a - eta * g
    # Normalize
    update -= update.max()  # for numerical stability
    new_a = np.exp(update)
    new_sum = new_a.sum()
    if new_sum < 1e-15:
        return a.copy()
    new_a /= new_sum
    
    return new_a

def build_conv_matrix(a):
    """Build matrix A where (A @ x) gives full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1

def lp_best_response(a, M):

    return None

def coordinate_descent(a, deadline, n_rounds=50):
    """Simple coordinate descent: try small changes to each coordinate."""
    a = a.copy()

            # Try increasing and decreasing a[j]

    return a, best_score

    # Uniform
    inits.append(np.ones(n) / n)

    # Fourier modes
    for k in [1, 2, 3, 4, 6, 8]:
        amp = 0.3 / np.sqrt(k)
        c = 1.0 + amp * np.cos(2 * np.pi * k * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Gaussian center
    c = np.exp(-((t_arr - 0.5) / 0.3) ** 2)
    inits.append(c / c.sum())

    c = np.exp(-((t_arr - 0.3) / 0.1) ** 2) + np.exp(-((t_arr - 0.7) / 0.1) ** 2)

    c = 1.0 - np.abs(t_arr - 0.5)
    c = np.maximum(c, 0.05)
    inits.append(c / c.sum())

    c = rng.uniform(0.3, 1.0, n)
    inits.append(c / c.sum())

    # Power law
    c = (1.0 + t_arr) ** 2
    inits.append(c / c.sum())

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast exponentiated gradient (mirror descent in log-space) via FFT
2. Active set annealing (broad to narrow) with adaptive step size
3. LP polishing with line search for final refinement
4. Multi-scale, multi-start search with perturbation restarts
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def cross_corr_c_a(c, a):
    """
    Compute g_j = sum_l a_l * c_{l+j} for j = 0, ..., n-1.
    c has length 2n-1, a has length n.
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    mc = len(c)
    L = 1
    while L < n + mc - 1:
        L <<= 1
    fac = np.fft.rfft(a, L)
    fcc = np.fft.rfft(c, L)
    result = np.fft.irfft(np.conj(fcc) * fac, L)
    return result[:n]


def softmax(theta):
    """Numerically stable softmax."""
    theta_shifted = theta - theta.max()
    exp_vals = np.exp(theta_shifted)
    return exp_vals / exp_vals.sum()


def exp_gradient_step(theta, epsilon_frac, eta):
    """
    One exponentiated gradient step in log-space.
    theta: log of the sequence (unnormalized)
    Returns (new_theta, score).
    """
    n = len(theta)
    a = softmax(theta)
    b = fast_conv(a)
    b_max = b.max()
    score = float(2.0 * n * b_max)

    if b_max < 1e-15:
        return theta, score

    threshold = b_max * (1.0 - epsilon_frac)
    S = b >= threshold
    if S.sum() == 0:
        S = np.zeros(2 * n - 1, dtype=bool)
        S[int(np.argmax(b))] = True

    c = S.astype(np.float64) / S.sum()
    g = cross_corr_c_a(c, a)

    theta_new = theta - eta * g
    # Prevent overflow: rescale
    if theta_new.max() > 500:
        theta_new = theta_new - theta_new.max() + 1.0
    return theta_new, score


def exp_gradient_optimize(a0, deadline, label="", verbose=False):
    """
    Run exponentiated gradient optimization until deadline.
    a0: initial sequence (non-negative, will be converted to log-space)
    Returns (best_sequence, best_score).
    """

    a = a / a.sum()
    n = len(a)

    theta = np.log(a)
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score

    it = 0
    no_improve = 0
    rng_local = np.random.default_rng(12345 + n)

    while time.monotonic() < deadline:
        # Annealing schedule
        progress = min(1.0, it / 8000.0)
        # Epsilon: start broad (0.4), sharpen to 0.01
        epsilon_frac = max(0.01, 0.4 * (1.0 - progress) ** 1.5 + 0.01 * progress)
        # Step size: start moderate, decay
        eta = 1.0 / (1.0 + 0.03 * it)

        theta, new_score = exp_gradient_step(theta, epsilon_frac, eta)
        a = softmax(theta)

        if new_score < current_score - 1e-10:
            current_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        else:
            no_improve += 1
            if no_improve > 300:
                # Perturb in log-space to escape local optimum
                noise = rng_local.normal(0, 0.3, n)
                theta = theta + noise
                a = softmax(theta)
                current_score = compute_score(a)
                no_improve = 0

        it += 1

    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix A where (A @ x) gives full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def lp_polish(a, deadline, max_iters=500):
    """LP polishing with line search over step sizes."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()

        # Line search over multiple step sizes
        best_a_new = a
        best_s = best_score
        for t in [0.3, 0.5, 0.7, 0.9, 1.1]:
            if t > 1.0:
                a_new = (1.0 - t) * a + t * g
            else:
                a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            if score < best_s:
                best_s = score
                best_a_new = a_new

        if best_s < best_score - 1e-12:
            best_score = best_s
            best_local = best_a_new.copy()
            a = best_a_new
        else:
            # Try over-relaxation
            for t in [1.2, 1.5]:
                a_new = (1.0 - t) * a + t * g
                a_new = np.clip(a_new, 0.0, None)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new = a_new / s
                score = compute_score(a_new)
                if score < best_score:
                    best_score = score
                    best_local = a_new.copy()
                    a = a_new
                    break
            else:
                break

    return best_local, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    t_arr = np.linspace(0, 1, n)

    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        p = np.maximum(np.interp(t_arr, t_old, parent), 0)
        if p.sum() > 0.01:
            inits.append(p / p.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Fourier modes with varying amplitudes
    for k, amp in [(1, 0.35), (2, 0.3), (3, 0.25), (4, 0.2), (6, 0.15), (8, 0.12)]:
        c = 1.0 + amp * np.cos(2 * np.pi * k * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Sine modes
    for k in [1, 2, 3]:
        c = 1.0 + 0.3 * np.sin(2 * np.pi * k * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Gaussian center
    c = np.exp(-((t_arr - 0.5) / 0.25) ** 2)
    inits.append(c / c.sum())

    # Bimodal
    c = np.exp(-((t_arr - 0.25) / 0.08) ** 2) + np.exp(-((t_arr - 0.75) / 0.08) ** 2)
    inits.append(c / c.sum())

    # Trimodal
    c = (np.exp(-((t_arr - 0.2) / 0.06) ** 2) +
         np.exp(-((t_arr - 0.5) / 0.06) ** 2) +
         np.exp(-((t_arr - 0.8) / 0.06) ** 2))
    inits.append(c / c.sum())

    # Triangular
    c = 1.0 - np.abs(t_arr - 0.5)
    c = np.maximum(c, 0.05)
    inits.append(c / c.sum())

    # U-shaped (inverted triangle)
    c = 0.1 + np.abs(t_arr - 0.5)
    inits.append(c / c.sum())

    # Random smooth (low-pass filtered random)
    c = rng.uniform(0, 1, n)
    # Simple smoothing
    kernel = np.ones(5) / 5
    c_smooth = np.convolve(c, kernel, mode='same')
    c_smooth = np.maximum(c_smooth, 0.01)
    inits.append(c_smooth / c_smooth.sum())

    # Power law
    c = (1.0 + t_arr) ** 3
    inits.append(c / c.sum())

    c = (1.0 - t_arr) ** 3
    inits.append(c / c.sum())

    # W-shaped
    c = 0.5 + 0.5 * np.abs(np.sin(2 * np.pi * 2 * t_arr))
    inits.append(c / c.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: exponentiated gradient + LP polish + perturbation.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 25.0  # safety margin

    best = None
    best_score = float('inf')

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            sc = compute_score(p)
            if sc < best_score:
                best = p.copy()
                best_score = sc
                print(f"Parent: J={sc:.6f} (n={p.size})", flush=True)
    except (NameError, AttributeError):
        pass

    # === PHASE 1: Exponentiated gradient (60% of budget) ===
    exp_budget = total_budget * 0.60
    exp_deadline = time.monotonic() + exp_budget

    lengths = [512, 256, 768, 1024, 384, 640]
    if best is not None:
        parent_len = best.size
        lengths = [parent_len] + [l for l in lengths if l != parent_len]

    print(f"\n=== Phase 1: Exponentiated Gradient ===", flush=True)
    t0 = time.monotonic()

    for n in lengths:
        remaining = exp_deadline - time.monotonic()
        if remaining < 20:
            break
        if n >= 1024 and remaining < 100:
            continue
        if n >= 768 and remaining < 50:
            continue

        print(f"  n={n}, time remaining={remaining:.0f}s", flush=True)

        inits = generate_initializations(n, rng, best)

        # Allocate time per initialization
        time_per_init = min(25.0, max(5.0, remaining * 0.85 / len(inits)))

        for i, a0 in enumerate(inits):
            if time.monotonic() > exp_deadline:
                break
            init_deadline = min(time.monotonic() + time_per_init, exp_deadline)
            if init_deadline - time.monotonic() < 3:
                break

            a_result, score = exp_gradient_optimize(
                a0, init_deadline,
                label=f"n={n},i={i}",
                verbose=(i == 0)
            )
            if score < best_score:
                best_score = score
                best = a_result.copy()
                print(f"    NEW BEST: J={best_score:.6f} (n={n}, init={i}, "
                      f"t={time.monotonic()-t0:.0f}s)", flush=True)

    print(f"  Phase 1 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 2: LP Polish (20% of budget) ===
    lp_budget = total_budget * 0.20
    lp_deadline = min(time.monotonic() + lp_budget, deadline)

    if best is not None and time.monotonic() < lp_deadline:
        print(f"\n=== Phase 2: LP Polish ===", flush=True)
        t0 = time.monotonic()

        a_result, score = lp_polish(best, lp_deadline, max_iters=300)
        if score < best_score:
            best_score = score
            best = a_result.copy()
            print(f"  NEW BEST (LP): J={best_score:.6f}", flush=True)

        print(f"  Phase 2 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 3: Perturbation search (15% of budget) ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Phase 3: Perturbation Search ===", flush=True)
        t0 = time.monotonic()
        n = best.size
        remaining = deadline - time.monotonic()

        n_tries = max(5, int(remaining / 12))
        noise_scales = [0.05, 0.1, 0.03, 0.15, 0.02, 0.2, 0.08, 0.12, 0.06, 0.25]

        for trial in range(n_tries):
            if time.monotonic() > deadline:
                break
            scale = noise_scales[trial % len(noise_scales)]
            noise = rng.normal(0, scale, n)
            a_pert = best + noise
            a_pert = np.clip(a_pert, 1e-10, None)
            a_pert = a_pert / a_pert.sum()

            try_deadline = min(time.monotonic() + 10, deadline)
            a_result, score = exp_gradient_optimize(
                a_pert, try_deadline,
                label=f"pert{trial}"
            )
            if score < best_score:
                best_score = score
                best = a_result.copy()
                print(f"  NEW BEST (pert {trial}): J={best_score:.6f}", flush=True)

        print(f"  Phase 3 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 4: Final polish (remaining time) ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Phase 4: Final Polish ===", flush=True)
        t0 = time.monotonic()

        # LP polish
        a_result, score = lp_polish(best, min(time.monotonic() + (deadline - time.monotonic()) * 0.5, deadline),
                                     max_iters=200)
        if score < best_score:
            best_score = score
            best = a_result.copy()
            print(f"  After LP: J={best_score:.6f}", flush=True)

        # Exp-gradient polish
        if time.monotonic() < deadline:
            a_result, score = exp_gradient_optimize(best, deadline, label="final")
            if score < best_score:
                best_score = score
                best = a_result.copy()
            print(f"  After exp-grad: J={best_score:.6f}", flush=True)

        print(f"  Phase 4 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # Final validation
    if best is None:
        best = np.ones(256) / 256
        best_score = compute_score(best)

    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score):
        if final_score < best_score:
            best_score = final_score
        print(f"\nFINAL (evaluator): J={final_score:.6f}, n={best.size}", flush=True)
    else:
        print(f"\nFINAL: J={best_score:.6f}, n={best.size} (evaluator returned inf)", flush=True)

    return [float(x) for x in best]
