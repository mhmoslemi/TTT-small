def subgradient_step(f, top_k=10):
    """Compute one subgradient step direction (normalized)."""
    n = len(f)
    conv = np.convolve(f, f)
    top_idx = np.argsort(conv)[-top_k:]

    for k in top_idx:
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * f[k - j]

    if norm < 1e-15:

def compute_score(f):
    """Fast computation of 2n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = (f*g)[k] for full convolution."""
    n = len(f)
    # A is (2n-1) x n, A[k,j] = f[k-j] if 0<=k-j<=n-1
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)

            cols[idx] = j

                grad = weighted_grad(f, dual, n)
            else:
                top_idx = np.argsort(np.convolve(f, f))[-15:]
                grad = get_subgradient(f, top_idx)
                gn = np.linalg.norm(grad)
                if gn > 1e-15:
                    grad /= gn
            for lr in [0.03, 0.015, 0.007]:
                cand = f - lr * grad
                cand = np.clip(cand, 0.0, None)
                cs = float(np.sum(cand))
                if cs > 1e-10:
                    cand = cand * (S / cs)
                    s = compute_score(cand)
                    if s < best_score:
                        f = cand
                        best_score = s
                        best_f = f.copy()
                        no_improve_streak = 0
                        break
            if verbose and lp_count % 5 == 0:
                print(f"    LP #{lp_count}: no LP step, subgrad score={best_score:.6f} t={lp_time:.2f}s")

        if no_improve_streak > 50:
            break

    return best_f, best_score


def optimize_nesterov(f, deadline, top_k=15, verbose=True):
    """Nesterov-accelerated subgradient descent."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    v = np.zeros(n)
    lr = 0.04
    it = 0
    no_improve = 0
    beta = 0.8

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        top_idx = np.argsort(conv)[-top_k:]
        grad = get_subgradient(f, top_idx)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        # Nesterov momentum
        v = beta * v - lr * grad
        beta = min(0.95, 0.8 + 0.1 * (it / 1000.0))
        cand = f + v
        cand = np.clip(cand, 0.0, None)
        cs = float(np.sum(cand))
        if cs < 1e-10:
            lr *= 0.5
            no_improve += 1
            if no_improve > 300:
                break
            it += 1
            continue
        cand = cand * (S / cs)
        s = compute_score(cand)
        it += 1

        if np.isfinite(s) and s < best_score:
            f = cand
            best_score = s
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.98
            v *= 0.5
            if lr < 1e-9 or no_improve > 500:
                break

        if verbose and it % 2000 == 0:
            print(f"    Nesterov it={it}: score={best_score:.6f} lr={lr:.6f}")

    return best_f, best_score


def optimize_multiscale_subgrad(f, deadline, verbose=True):
    """Subgradient with multiple top_k values and adaptive learning rate."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    v = np.zeros(n)
    it = 0
    no_improve = 0
    top_k_cycle = [8, 12, 20, 30]
    ci = 0

    while time.monotonic() < deadline:
        top_k = top_k_cycle[ci % len(top_k_cycle)]
        ci += 1
        conv = np.convolve(f, f)
        top_idx = np.argsort(conv)[-top_k:]
        grad = get_subgradient(f, top_idx)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        # Adaptive lr based on no_improve
        lr = 0.03 / (1.0 + 0.01 * no_improve)
        v = 0.85 * v - lr * grad
        cand = f + v
        cand = np.clip(cand, 0.0, None)
        cs = float(np.sum(cand))
        if cs < 1e-10:
            v = np.zeros(n)
            no_improve += 1
            if no_improve > 500:
                break
            it += 1
            continue
        cand = cand * (S / cs)
        s = compute_score(cand)
        it += 1

        if np.isfinite(s) and s < best_score:
            f = cand
            best_score = s
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.5
            if no_improve > 800:
                break

        if verbose and it % 3000 == 0:
            print(f"    MultiSG it={it}: score={best_score:.6f} top_k={top_k}")

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry: search for best sequence minimizing the objective."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)

    n = 1024
    best_score = float('inf')
    best_seq = None

    # Load and prepare parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            if len(parent) != n:
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n)
                parent = np.interp(x_new, x_old, parent)
            sc = compute_score(parent)
            if np.isfinite(sc):
                best_score = sc
                best_seq = parent.copy()
                print(f"Parent loaded: n={n}, score={sc:.6f}")
            off_sc = evaluate_sequence(parent.tolist())
            print(f"Parent official: {off_sc:.6f}")
    except (NameError, Exception) as e:
        print(f"Parent unavailable: {e}")
        parent = np.ones(n)

    f_start = best_seq.copy() if best_seq is not None else np.ones(n)

    # Phase 1: Quick Nesterov warmup (80s)
    print("\n=== Phase 1: Nesterov Warmup (80s) ===")
    d1 = t0 + 80.0
    f_opt, s_opt = optimize_nesterov(f_start, d1, top_k=12, verbose=True)
    if s_opt < best_score:
        best_score = s_opt
        best_seq = f_opt.copy()
    print(f"After Phase 1: {best_score:.6f}")

    # Phase 2: Main LP optimization (800s)
    print("\n=== Phase 2: LP Optimization (800s) ===")
    d2 = t0 + 880.0
    f_opt, s_opt = optimize_lp_main(best_seq.copy(), d2, verbose=True)
    if s_opt < best_score:
        best_score = s_opt
        best_seq = f_opt.copy()
    print(f"After Phase 2: {best_score:.6f}")

    # Phase 3: Multi-scale subgradient refinement (250s)
    print("\n=== Phase 3: Multi-scale Subgradient (250s) ===")
    d3 = t0 + 1130.0
    f_opt, s_opt = optimize_multiscale_subgrad(best_seq.copy(), d3, verbose=True)
    if s_opt < best_score:
        best_score = s_opt
        best_seq = f_opt.copy()
    print(f"After Phase 3: {best_score:.6f}")

    # Phase 4: Final LP polish (250s)
    print("\n=== Phase 4: Final LP Polish (250s) ===")
    d4 = deadline
    f_opt, s_opt = optimize_lp_main(best_seq.copy(), d4, verbose=True)
    if s_opt < best_score:
        best_score = s_opt
        best_seq = f_opt.copy()
    print(f"After Phase 4: {best_score:.6f}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(n)

    # Ensure valid output
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_seq])
    if not np.isfinite(final_score):
        print("WARNING: Failed evaluation, using fallback.")
        best_seq = np.ones(n)
        final_score = evaluate_sequence(best_seq.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_seq)}, internal={best_score:.6f}, official={final_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_seq]
