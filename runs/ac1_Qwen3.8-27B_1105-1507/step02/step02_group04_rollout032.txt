def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def golden_section_min(f, a=0, b=1, tol=1e-5, max_iter=50):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2

def adam_subgradient(a, max_iter=5000, K=5, tau_start=5.0, tau_end=50.0,
                     lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                     line_search=True, patience=500):
    """Adam-adapted smoothed subgradient descent with top-K peak focusing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s  # normalize to sum = 1
    
    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    
    stall = 0
    
    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac
        
        conv = compute_conv(a)
        current_max = float(np.max(conv))
        
        if current_max < best_max - 1e-12:
            best_max = current_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                break
        
        # Compute top-K gradient
        grad = top_k_gradient(conv, a, K=K, tau=tau)
        
        # Center gradient (for simplex constraint)
        grad_centered = grad - np.mean(grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered**2
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        
        if line_search:
            # Try a few step sizes and pick the best
            best_step = lr
            best_obj_val = current_max
            for step_mult in [0.25, 0.5, 1.0, 2.0]:
                step = lr * step_mult
                a_trial = a - step * m_hat / (np.sqrt(v_hat) + eps)
                a_trial = np.maximum(a_trial, 0)
                s_trial = np.sum(a_trial)
                if s_trial < 1e-10:
                    continue
                a_trial = a_trial / s_trial
                trial_max = float(np.max(compute_conv(a_trial)))
                if trial_max < best_obj_val:
                    best_obj_val = trial_max
                    best_step = step
            
            a = a - best_step * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s < 1e-10:
                break
            a = a / s
        else:
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s < 1e-10:
                break
            a = a / s
    
    return best_a, best_max

def coordinate_descent(a, max_iter=200, tol=1e-8):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))  # with sum=1
    
    for it in range(max_iter):
        improved = False
        for i in range(n):
            # Optimize a_i while keeping others fixed
            # After optimization, renormalize
            a_orig = a[i]
            
            # Define the objective as a function of a_i
            # If we set a[i] = x, then sum = s - a_orig + x
            # conv changes: for each k, conv[k] changes
            # The change in conv[k] when a[i] changes from a_orig to x:
            # delta_i * a[k-i] for each valid k-i (appears twice if k != 2i, once with delta^2 if k=2i)
            
            # For efficiency, precompute the "base" convolution without a[i]
            # conv_base[k] = conv[k] - 2*a_orig*a[k-i] (if k != 2i) or conv[k] - a_orig^2 (if k == 2i)
            # Actually: conv[k] = sum_j a[j]*a[k-j]. The terms involving a[i] are:
            #   a[i]*a[k-i] (when i is the first factor) + a[k-i]*a[i] (when i is the second factor)
            #   = 2*a[i]*a[k-i] if k != 2i
            #   = a[i]^2 if k == 2i
            # So conv_base[k] = conv[k] - 2*a_orig*a[k-i] (k != 2i) or conv[k] - a_orig^2 (k == 2i)
            # And conv_new[k] = conv_base[k] + 2*x*a[k-i] (k != 2i) or conv_base[k] + x^2 (k == 2i)
            #   where a[k-i] is the current value (for k != 2i, a[k-i] != a[i] so it's fixed)
            
            conv = compute_conv(a)
            conv_base = conv.copy()
            for k in range(2*n-1):
                j = k - i
                if 0 <= j < n:
                    if j == i:
                        conv_base[k] = conv[k] - a_orig * a_orig
                    else:
                        conv_base[k] = conv[k] - 2.0 * a_orig * a[j]
            
            # Now conv_new[k] = conv_base[k] + 2*x*a[k-i] (j != i) or conv_base[k] + x^2 (j == i)
            # We want to minimize max_k conv_new[k] / (sum_new)^2
            # where sum_new = s - a_orig + x
            
            def obj_of_x(x):
                if x < 0:
                    return np.inf
                sum_new = s - a_orig + x
                if sum_new < 1e-10:
                    return np.inf
                conv_new = conv_base.copy()
                for k in range(2*n-1):
                    j = k - i
                    if 0 <= j < n:
                        if j == i:
                            conv_new[k] += x * x
                        else:
                            conv_new[k] += 2.0 * x * a[j]
                return 2.0 * n * np.max(conv_new) / (sum_new * sum_new)
            
            # Minimize over x in [0, 10*a_orig + 0.1] (reasonable range)
            x_upper = max(10 * a_orig + 0.1, 2.0 / n)
            res = minimize_scalar(obj_of_x, bounds=(0, x_upper), method='bounded',
                                  options={'xatol': 1e-6})
            
            if res.success and res.fun < best_obj - tol:
                a[i] = res.x
                best_obj = res.fun
                improved = True
        
        if not improved:
            break
    
    return a, best_obj

def lp_step(a, M, n_constraints=None):
    """LP step: find g that maximizes sum subject to (a*g)_k <= M, g >= 0.
    Only enforce top constraints for speed."""
    n = len(a)
    conv = compute_conv(a)
    M = max(M, float(np.max(conv)))
    
    if n_constraints is None:
        n_constraints = min(2*n-1, max(50, n))  # Use subset of constraints
    
    # Select the most important constraints (largest conv values)
    top_idx = np.argsort(conv)[-n_constraints:]
    
    # Build constraint matrix for selected constraints
    A = np.zeros((n_constraints, n))
    for row, k in enumerate(top_idx):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[row, j_min:j_max+1] = a[k - j_max:k - j_min + 1][::-1]
    
    c = -np.ones(n)
    b_ub = np.full(n_constraints, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def run_optimization(n, seed, max_time=60.0, verbose=False):
    """Run optimization for a given sequence length."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    
    best_a = None
    best_obj = np.inf
    
    # Generate initializations
    n_inits = max(3, int(min(8, max_time / 15)))
    inits = []
    
    for idx in range(n_inits):
        choice = idx % 6
        if choice == 0:
            # Flat with cosine modulation
            f = rng.integers(1, max(2, n//10))
            amp = rng.uniform(0.2, 0.8)
            a = 1.0 + amp * np.cos(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 2*np.pi))
        elif choice == 1:
            # Flat
            a = np.ones(n)
        elif choice == 2:
            # Random
            a = rng.random(n)
        elif choice == 3:
            # Exponential decay from center
            center = (n-1) / 2.0
            rate = rng.uniform(0.5, 3.0)
            a = np.exp(-rate * np.abs(np.arange(n) - center) / center)
        elif choice == 4:
            # Bipartite (high-low pattern)
            period = rng.integers(3, max(4, n//20))
            a = np.where(np.arange(n) % period < period//2, 
                        rng.uniform(0.7, 1.5), rng.uniform(0.1, 0.4))
        elif choice == 5:
            # Multi-frequency modulation
            a = np.ones(n)
            for _ in range(rng.integers(1, 4)):
                f = rng.integers(1, max(2, n//5))
                amp = rng.uniform(0.1, 0.4)
                a += amp * np.cos(2*np.pi*f*np.arange(n)/n + rng.uniform(0, 2*np.pi))
        
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
        inits.append(a)
    
    # Try each initialization
    for a_init in inits:

"""
Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam-adapted smoothed subgradient with top-K peak focusing,
line search, coordinate descent polish, LP steps, and diverse initializations.
Focuses optimization around the parent length (250) with exploration of neighbors.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def top_k_gradient(conv, a, K=5, tau=10.0):
    """Compute gradient of max(conv) using top-K peaks with softmax weighting."""
    n = len(a)
    K = min(K, len(conv))
    top_k_idx = np.argsort(conv)[-K:]
    top_k_vals = conv[top_k_idx]
    shifted = tau * (top_k_vals - top_k_vals.max())
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.ones(K) / K
    else:
        weights /= w_sum

    w_full = np.zeros(2 * n - 1)
    w_full[top_k_idx] = weights

    # grad[i] = 2 * sum_k w_full[k] * a[k-i]
    # = 2 * (w_full * a_rev)[n-1+i] where a_rev = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    a_rev = a[::-1]
    conv_full = np.fft.irfft(
        np.fft.rfft(w_full, size) * np.fft.rfft(a_rev, size), size)
    g = 2.0 * conv_full[n - 1:n - 1 + n]
    return g


def adam_subgradient(a, max_iter=5000, K=5, tau_start=5.0, tau_end=60.0,
                     lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                     line_search=False, patience=400):
    """Adam-adapted smoothed subgradient descent with top-K peak focusing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    m = np.zeros(n)
    v = np.zeros(n)
    stall = 0

    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        current_max = float(np.max(conv))

        if current_max < best_max - 1e-12:
            best_max = current_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                break

        grad = top_k_gradient(conv, a, K=K, tau=tau)
        grad_centered = grad - np.mean(grad)

        m = beta1 * m + (1.0 - beta1) * grad_centered
        v = beta2 * v + (1.0 - beta2) * grad_centered ** 2
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        if line_search:
            best_step = lr
            best_val = current_max
            for step_mult in [0.25, 0.5, 1.0, 2.0]:
                step = lr * step_mult
                a_trial = a - step * m_hat / (np.sqrt(v_hat) + eps)
                a_trial = np.maximum(a_trial, 0)
                s_trial = np.sum(a_trial)
                if s_trial < 1e-10:
                    continue
                a_trial = a_trial / s_trial
                trial_max = float(np.max(compute_conv(a_trial)))
                if trial_max < best_val:
                    best_val = trial_max
                    best_step = step
            a = a - best_step * m_hat / (np.sqrt(v_hat) + eps)
        else:
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s

    return best_a, best_max


def coordinate_descent(a, max_passes=50):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)

    for pass_idx in range(max_passes):
        improved = False
        for i in range(n):
            a_orig = a[i]

            def obj_of_x(x_val):
                if x_val < 0:
                    return np.inf
                a_trial = a.copy()
                a_trial[i] = x_val
                return compute_obj(a_trial)

            x_upper = max(10.0 * a_orig + 0.01, 4.0 / n)
            try:
                res = minimize_scalar(obj_of_x, bounds=(0.0, x_upper),
                                      method='bounded',
                                      options={'xatol': 1e-7, 'maxiter': 30})
                if res.success and res.fun < best_obj - 1e-12:
                    a[i] = res.x
                    best_obj = res.fun
                    improved = True
            except Exception:
                pass

        if not improved:
            break

    return a, best_obj


def lp_step(a, M, n_constraints=None):
    """LP step: find g maximizing sum subject to (a*g)_k <= M, g >= 0."""
    n = len(a)
    if n_constraints is None:
        n_constraints = min(2 * n - 1, max(50, n))

    conv = compute_conv(a)
    M = max(M, float(np.max(conv)))
    top_idx = np.argsort(conv)[-n_constraints:]

    A = np.zeros((n_constraints, n))
    for row, k in enumerate(top_idx):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[row, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]

    c = -np.ones(n)
    b_ub = np.full(n_constraints, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def generate_init(n, rng, idx):
    """Generate diverse initializations."""
    choice = idx % 8
    if choice == 0:
        f = rng.integers(1, max(2, n // 10))
        amp = rng.uniform(0.2, 0.8)
        a = 1.0 + amp * np.cos(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    elif choice == 1:
        a = np.ones(n)
    elif choice == 2:
        a = rng.random(n)
    elif choice == 3:
        center = (n - 1) / 2.0
        rate = rng.uniform(0.5, 3.0)
        a = np.exp(-rate * np.abs(np.arange(n) - center) / max(center, 1))
    elif choice == 4:
        period = rng.integers(3, max(4, n // 15))
        a = np.where(np.arange(n) % period < period // 2,
                     rng.uniform(0.7, 1.5), rng.uniform(0.05, 0.3))
    elif choice == 5:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.integers(1, max(2, n // 5))
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        # Two-tone with random boundary
        split = rng.integers(n // 4, 3 * n // 4)
        a = np.concatenate([rng.uniform(0.5, 1.5, split), rng.uniform(0.1, 0.6, n - split)])
    else:

        a = rng.uniform(0.01, 0.1, n)
        n_peaks = rng.integers(3, max(4, n // 10))
        peak_idx = rng.choice(n, size=n_peaks, replace=False)
        a[peak_idx] = rng.uniform(0.5, 1.5, n_peaks)

    a = np.maximum(a, 0.001)
    a = a / np.sum(a)
    return a

                                        K=K, tau_start=tau_s, tau_end=tau_e,
                                        lr=lr, line_search=ls, patience=patience)
            obj = compute_obj(a_cur)
            if obj < best_score:
                best_score = obj
                best_seq = a_cur.copy()
                print(f"  Round {round_idx} (K={K}, lr={lr}): obj={best_score:.6f}")
            else:
                a_cur = best_seq.copy()

            # LP step
            remaining = deadline - time.monotonic()
            if remaining > 12 and n_best <= 350:
                a_lp = best_seq.copy()
                for _ in range(8):
                    if time.monotonic() > deadline - 10:
                        break
                    conv = compute_conv(a_lp)
                    M = float(np.max(conv))
                    g = lp_step(a_lp, M)
                    if g is None:
                        break
                    sum_g = np.sum(g)
                    if sum_g < 1e-10:
                        break
                    g = g / sum_g
                    a_new = 0.75 * a_lp + 0.25 * g
                    a_new = np.maximum(a_new, 0)
                    a_new = a_new / np.sum(a_new)
                    obj_new = compute_obj(a_new)
                    if obj_new < best_score - 1e-9:
                        best_score = obj_new
                        best_seq = a_new.copy()
                        a_lp = a_new.copy()
                    else:
                        break

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 8 and n_best <= 300:
                a_cd, obj_cd = coordinate_descent(best_seq,
                                                  max_passes=int(min(25, remaining * 2)))
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    print(f"  Round {round_idx} CD: obj={best_score:.6f}")

            # Perturbation + quick re-optimize
            remaining = deadline - time.monotonic()
            if remaining > 15:
                noise_scale = 0.02 + 0.01 * round_idx
                a_pert = best_seq * np.exp(rng.normal(0, noise_scale, n_best))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / np.sum(a_pert)
                a_pert, _ = adam_subgradient(a_pert, max_iter=int(remaining * 2),
                                             K=5, tau_start=10, tau_end=50,
                                             lr=0.03, patience=100)
                obj_p = compute_obj(a_pert)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_pert.copy()
                    print(f"  Round {round_idx} perturb: obj={best_score:.6f}")

    # Phase 3: Final coordinate descent polish
    if best_seq is not None and time.monotonic() < deadline - 5 and len(best_seq) <= 300:
        remaining = deadline - time.monotonic()
        print(f"\nFinal CD polish, {remaining:.0f}s remaining")
        a_final, obj_final = coordinate_descent(best_seq,
                                                max_passes=int(min(50, remaining * 3)))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.6f}")

        # One more Adam pass after CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_pol, _ = adam_subgradient(best_seq, max_iter=int(remaining * 4),
                                        K=3, tau_start=20, tau_end=80,
                                        lr=0.01, line_search=True, patience=200)
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Final Adam: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation != best_score and validation < np.inf:
        print(f"  Validation: computed={best_score:.6f}, validated={validation:.6f}")
        best_score = validation

    return result
