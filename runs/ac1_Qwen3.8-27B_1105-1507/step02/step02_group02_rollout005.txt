import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A where (A @ x)[k] = sum_i a[k-i]*x[i] (circular-free)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def compute_score(a):
    """Compute J(a) directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lp_best_response(a, M, prev_g=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def peel_redistribute(a, n_peels=5, eta=0.05):
    """Peel mass from peak-contributing positions and add to low positions."""
    n = len(a)
    b = fast_conv(a)
    b_mean = b.mean()
    a = a.copy()
    
    for _ in range(n_peels):
        # Find peak positions
        peak_idx = np.argmax(b)
        if b[peak_idx] <= b_mean * 1.01:
            break
        
        # Find contributing pair for peak_idx
        # b[peak_idx] = sum_i a[i]*a[peak_idx - i]
        # Find i that maximizes a[i]*a[peak_idx - i]
        jl = max(0, peak_idx - n + 1)
        jh = min(n - 1, peak_idx)
        if jl > jh:
            break
        contrib = a[jl:jh+1] * a[peak_idx - jh:peak_idx - jl + 1][::-1]
        i_best = jl + np.argmax(contrib)
        j_best = peak_idx - i_best
        
        if i_best == j_best:
            # Diagonal, just reduce
            reduce_amt = eta * a[i_best]
            a[i_best] -= reduce_amt
        else:
            reduce_amt = eta * 0.5 * (a[i_best] + a[j_best])
            a[i_best] -= 0.5 * reduce_amt
            a[j_best] -= 0.5 * reduce_amt
        
        # Add to low positions
        a = np.maximum(a, 0)
        if a.sum() < 0.01:
            break
        # Add mass to positions with lowest contribution to b
        low_idx = np.argmin(a)
        a[low_idx] += reduce_amt
        
        # Update b
        b = fast_conv(a)
    
    return np.maximum(a, 0)


def spectral_smooth(a, sigma=1.0):
    """Smooth in frequency domain to reduce high-frequency components."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    freqs = np.fft.rfftfreq(s)
    gauss = np.exp(-2 * (np.pi * sigma * freqs) ** 2)
    fa *= gauss
    a_smooth = np.fft.irfft(fa, s)[:n]
    return np.maximum(a_smooth, 0)


def smooth_objective(x, beta, n):
    """Log-sum-exp smoothed objective."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / beta
    return float(2.0 * n * log_sum / (1.0))  # Already normalized

def smooth_refine(a, deadline):
    """Refine using smooth log-sum-exp with L-BFGS-B."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)

                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,

            if res.fun < 1e10:

                    score = compute_score(res_x)
                    if score < best_score:

                        best_score = score
        except Exception:
            pass

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A where (A @ x)[k] = sum_i a[k-i]*x[i] for valid i."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def peel_redistribute(a, n_peels=5, eta=0.05):
    """Peel mass from peak-contributing positions and redistribute to low positions."""
    n = len(a)
    b = fast_conv(a)
    a = a.copy()

    for _ in range(n_peels):
        peak_idx = int(np.argmax(b))
        b_mean = b.mean()
        if b[peak_idx] <= b_mean * 1.02:
            break

        jl = max(0, peak_idx - n + 1)
        jh = min(n - 1, peak_idx)
        if jl > jh:
            break
        contrib = a[jl:jh + 1] * a[peak_idx - jh:peak_idx - jl + 1][::-1]
        i_best = jl + int(np.argmax(contrib))
        j_best = peak_idx - i_best

        if i_best == j_best:
            reduce_amt = eta * a[i_best]
            a[i_best] -= reduce_amt
        else:
            reduce_amt = eta * 0.5 * (a[i_best] + a[j_best])
            a[i_best] -= 0.5 * reduce_amt
            a[j_best] -= 0.5 * reduce_amt

        a = np.maximum(a, 0)
        if a.sum() < 0.01:
            break

        low_idx = int(np.argmin(a))
        a[low_idx] += reduce_amt
        b = fast_conv(a)

    return np.maximum(a, 0)


def spectral_smooth(a, sigma=1.0):
    """Smooth in frequency domain to reduce high-frequency components."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:

    fa = np.fft.rfft(a, s)
    freqs = np.fft.rfftfreq(s)
    gauss = np.exp(-2 * (np.pi * sigma * freqs) ** 2)
    fa *= gauss
    a_smooth = np.fft.irfft(fa, s)[:n]
    return np.maximum(a_smooth, 0)

            max_iters = min(max_iters, 4000)

            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"    NEW BEST: J={score:.6f} (n={n}, init={i})", flush=True)

        print(f"  n={n} done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 2: Stochastic subgradient (fast, many steps) ===
    if best is not None and time.monotonic() < deadline:
        smooth_budget_phase2 = max(30.0, (deadline - time.monotonic()) * 0.10)
        sg_deadline = time.monotonic() + smooth_budget_phase2
        t0 = time.monotonic()
        print(f"\n=== Phase 2: Stochastic Subgradient ===", flush=True)

        n = best.size
        n_steps = int(smooth_budget_phase2 * 20000)  # Very fast per step
        n_steps = min(n_steps, 500000)
        a_result, score = stochastic_subgradient(best, n_steps, 0.01, sg_deadline, rng)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (SGD): J={score:.6f}", flush=True)

        print(f"  SGD done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 3: Smooth L-BFGS-B refinement ===
    smooth_budget = max(60.0, (deadline - time.monotonic()) * 0.20)
    smooth_deadline = min(time.monotonic() + smooth_budget, deadline)

    if best is not None and time.monotonic() < smooth_deadline:
        t0 = time.monotonic()
        print(f"\n=== Phase 3: Smooth Refinement ===", flush=True)

        a_result, score = smooth_refine(best, smooth_deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (smooth): J={score:.6f} (n={best.size})", flush=True)

        # Try at 2x length
        n2 = min(2048, best.size * 2)
        if n2 != best.size and time.monotonic() < smooth_deadline:
            t_old = np.linspace(0, 1, best.size)
            t_new = np.linspace(0, 1, n2)
            a2 = np.maximum(np.interp(t_new, t_old, best), 0)
            if a2.sum() > 0.01:
                a2 = a2 / a2.sum()
                a_result2, score2 = smooth_refine(a2, smooth_deadline)
                if score2 < best_score:
                    best = a_result2.copy()
                    best_score = score2
                    print(f"  NEW BEST (smooth n={n2}): J={score2:.6f}", flush=True)

        print(f"  Smooth done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 4: Perturbation + LP refinement ===
    if best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n=== Phase 4: Perturbation Search ===", flush=True)
        n = best.size
        remaining = deadline - time.monotonic()

        n_tries = max(5, min(60, int(remaining / 12)))
        noise_scales = [0.03, 0.05, 0.08, 0.02, 0.12, 0.04, 0.06]

        est_per_iter = 0.022 if n <= 512 else 0.035 if n <= 768 else 0.06

        for trial in range(n_tries):
            if time.monotonic() > deadline:
                break
            scale = noise_scales[trial % len(noise_scales)]
            noise = rng.normal(0, scale, n)
            a_pert = best + noise
            a_pert = np.clip(a_pert, 0, None)
            if a_pert.sum() < 0.01:
                continue
            a_pert = a_pert / a_pert.sum()

            remaining = deadline - time.monotonic()
            max_iters = max(30, int(remaining / (n_tries - trial + 1) / est_per_iter))
            max_iters = min(max_iters, 1000)

            a_result, score = lp_iteration(a_pert, max_iters, deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST (pert {trial}): J={score:.6f}", flush=True)

        print(f"  Perturb done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 5: Final long polish ===
    if best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n=== Phase 5: Final Polish ===", flush=True)
        a = best.copy()
        remaining = deadline - time.monotonic()
        n = a.size
        est_per_iter = 0.022 if n <= 512 else 0.035 if n <= 768 else 0.06
        max_iters = max(100, int(remaining * 0.7 / est_per_iter))

        a_result, score = lp_iteration(a, max_iters, deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score

        # One more smooth pass
        if time.monotonic() < deadline:
            a_result, score = smooth_refine(best, deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score

        # Final peel-redistribute
        if time.monotonic() < deadline:
            a_peel = peel_redistribute(best, n_peels=10, eta=0.02)
            if a_peel.sum() > 0.01:
                a_peel = a_peel / a_peel.sum()
                s_peel = compute_score(a_peel)
                if s_peel < best_score:
                    best = a_peel.copy()
                    best_score = s_peel

        print(f"  Polish done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # Final validation using the actual evaluator
    if best is None:
        best = np.ones(256) / 256
        best_score = compute_score(best)

    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
