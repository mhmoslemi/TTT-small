```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Log-sum-exp smoothed L-BFGS-B with continuation (adaptive tau increasing)
2. Multi-start with diverse initializations at multiple sequence lengths
3. LP-based mirror descent refinement
4. Coordinate descent final polish
Key innovation: efficient O(n log n) gradient via FFT-based cross-correlation for
the log-sum-exp smoothed max, enabling fast L-BFGS-B optimization.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute exact J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_obj_grad(x, tau):
    """
    Log-sum-exp smoothed J(a) with analytic gradient.
    J_tau(a) = 2n * (1/tau) * log(sum_k exp(tau * b_k)) / (sum(a))^2
    where b = a * a (autocorrelation).
    """
    n = len(x)
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)

    b = fast_conv(a)

    # Numerically stable log-sum-exp
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_shifted = np.exp(shifted)
    exp_sum = exp_shifted.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    log_sum_exp = b_max + np.log(exp_sum) / tau

    # Objective
    obj = 2.0 * n * log_sum_exp / (s * s)

    # Softmax weights
    w = exp_shifted / exp_sum

    # Compute s_i = sum_j w[i+j] * a[j] via FFT cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    corr = np.fft.irfft(fw * np.conj(fa), N)
    s_vals = corr[:n]

    # Gradient: 4n*s_i/S^2 - 4n*LSE*a_i/S^3
    grad = (4.0 * n * s_vals) / (s * s) - (4.0 * n * log_sum_exp * a) / (s * s * s)

    return obj, grad


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals.extend(seg.tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def run_lbfgs_continuation(a, tau_schedule, deadline, tracker, max_iter_per_tau=200):
    """
    Run L-BFGS-B with log-sum-exp continuation.
    tau_schedule: list of tau values to use in sequence.
    """
    a = a.astype(np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-10)

    bounds = [(0.0, None)] * n

    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break

        # Objective wrapper (top-level compatible via caching)
        tau_holder = [tau]

        def _obj(x):
            val, _ = lse_obj_grad(x, tau_holder[0])
            return val

        def _grad(x):
            _, g = lse_obj_grad(x, tau_holder[0])
            return g

        res = minimize(_obj, a, jac=_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter_per_tau,
                                'max_fun': max_iter_per_tau * 3,
                                'ftol': 1e-16,
                                'gtol': 1e-14})

        if res.fun < 1e9:
            a_new = np.maximum(res.x, 0.0)
            if a_new.sum() > 1e-10:
                a = a_new
                # Track progress
                if tau == tau_schedule[0] or tau % 3 == 0:
                    tracker.update(a)

    tracker.update(a)
    return a


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step sizes
        base_t = max(0.03, 0.4 * (1.0 - it / float(2 * max_iters)))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5)]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # Try very small step
            t_small = 0.01
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break

        if it % 30 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    # Uniform
    inits.append(np.ones(n))

    # Gaussian centered
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.exp(-8.0 * (t_arr - 0.5) ** 2) + 0.02)
    inits.append(np.exp(-2.0 * (t_arr - 0.5) ** 2) + 0.1)

    # Cosine variations
    for freq in [1, 2, 3, 4]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr))
    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * t_arr + rng.uniform(0, 6.28)))

    # Triangular
    inits.append(1.0 - 2.0 * np.abs(t_arr - 0.5))

    # Parabolic
    inits.append(1.0 + 0.5 * (2.0 * t_arr - 1.0) ** 2)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.1)
    inits.append(np.abs(np.sin(2.0 * np.pi * t_arr)) + 0.1)

    # Random smooth
    for _ in range(3):
        coeffs = rng.normal(0, 1, min(10, n // 4 + 1))
        freqs = rng.permutation(min(10, n // 4 + 1))
        sig = np.zeros(n)
        for i in range(len(coeffs)):
            sig += coeffs[i] * np.cos(2.0 * np.pi * (freqs[i] + 1) * t_arr)
        inits.append(np.maximum(1.0 + 0.3 * sig / (1.0 + np.abs(sig).max()), 0.01))

    # Two-peak
    inits.append(np.exp(-6.0 * (t_arr - 0.25) ** 2) + np.exp(-6.0 * (t_arr - 0.75) ** 2) + 0.05)

    # Sparse
    sparse = np.zeros(n)
    n_spikes = max(4, n // 20)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.03)

    for init in inits:
        init = np.maximum(init, 0.01).astype(np.float64)
        init = init / init.sum()

    return inits


def coordinate_descent_polish(a, max_iters, deadline, tracker):
    """Simple coordinate descent: try increasing/decreasing each element."""
    a = a.astype(np.float64).copy()
    n = len(a)
    J_current = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            a_save = a[i]
            for factor in [0.8, 0.9, 1.1, 1.2]:
                a[i] = a_save * factor
                if a[i] < 0:
                    a[i] = 0.0
                J_trial = compute_J(a)
                if J_trial < J_current - 1e-12:
                    J_current = J_trial
                    improved = True
                    break
            a[i] = a[i] if a[i] > 0 else a_save
        if not improved:
            break
    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: L-BFGS-B with log-sum-exp continuation at multiple lengths
    Phase 2: LP mirror descent refinement
    Phase 3: Coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # ============ Phase 1: L-BFGS-B with continuation ============
    print(f"\n{'='*60}", flush=True)
    print("Phase 1: L-BFGS-B with log-sum-exp continuation", flush=True)
    print(f"{'='*60}", flush=True)

    # Lengths to try, with time allocation
    length_configs = [
        (128, [3, 6, 12, 24, 48, 96, 192, 384], 150),
        (256, [3, 6, 12, 24, 48, 96, 192, 384], 200),
        (512, [3, 6, 12, 24, 48, 96, 192], 250),
        (1024, [3, 6, 12, 24, 48, 96], 300),
    ]

    # If parent is available, also try its exact length
    if parent is not None and parent.size not in [c[0] for c in length_configs]:
        length_configs.insert(0, (parent.size, [3, 6, 12, 24, 48, 96, 192, 384, 768], 150))

    # Limit total number of initializations based on time
    total_t0 = time.monotonic()

    for n, tau_sched, time_budget in length_configs:
        if time.monotonic() > deadline:
            break
        phase_t0 = time.monotonic()
        print(f"\n  --- n={n}, tau_schedule={tau_sched}, budget={time_budget}s ---", flush=True)

        inits = generate_initializations(n, rng, parent)

        # Limit initializations based on time budget
        max_inits = min(len(inits), max(2, int(time_budget // 30)))
        if max_inits < len(inits):
            # Prioritize: parent first, then diverse
            inits = inits[:max_inits]

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            if time.monotonic() - phase_t0 > time_budget:
                break

            t0 = time.monotonic()
            a_result = run_lbfgs_continuation(a0, tau_sched, deadline, tracker,
                                              max_iter_per_tau=150)
            elapsed = time.monotonic() - t0
            J_val = compute_J(a_result)
            print(f"    init {idx}: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)

        phase_elapsed = time.monotonic() - phase_t0
        tracker.report(f"Phase1 n={n} ({phase_elapsed:.1f}s)")

    # ============ Phase 2: LP Mirror Descent refinement ============
    if tracker.best is not None and time.monotonic() < deadline - 60:
        print(f"\n{'='*60}", flush=True)
        print("Phase 2: LP Mirror Descent refinement", flush=True)
        print(f"{'='*60}", flush=True)

        n = tracker.best.size
        t0 = time.monotonic()

        # Run LP from best and from perturbed versions
        for pert_level in [0.0, 0.05, 0.1, 0.15, 0.25]:
            if time.monotonic() > deadline - 30:
                break
            a_start = tracker.best.copy().astype(np.float64)
            if pert_level > 0:
                a_start = a_start + rng.normal(0, pert_level, n)
                a_start = np.maximum(a_start, 0.001)
                a_start = a_start / a_start.sum()

            iters = max(30, min(150, int(60 * (1.0 - pert_level))))
            a_result = run_lp_mirror_descent(a_start, iters, deadline, tracker)
            J_val = compute_J(a_result)
            print(f"  pert={pert_level:.2f}: J={J_val:.8f}", flush=True)

        tracker.report(f"Phase 2 ({time.monotonic()-t0:.1f}s)")

    # ============ Phase 3: More L-BFGS-B rounds from best ============
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print("Phase 3: Additional L-BFGS-B refinement", flush=True)
        print(f"{'='*60}", flush=True)

        n = tracker.best.size
        t0 = time.monotonic()

        # Try different tau ranges focused on sharpening
        tau_sharp = [50, 100, 200, 400, 800]
        a_current = tracker.best.copy()
        a_result = run_lbfgs_continuation(a_current, tau_sharp, deadline, tracker,
                                          max_iter_per_tau=300)
        J_val = compute_J(a_result)
        print(f"  sharpened: J={J_val:.8f}", flush=True)

        # Also try from slightly perturbed best
        a_pert = tracker.best.copy() + rng.normal(0, 0.02, n)
        a_pert = np.maximum(a_pert, 0.0)
        if a_pert.sum() > 0.01:
            a_pert = a_pert / a_pert.sum()
            tau_mid = [10, 30, 60, 120]
            a_result2 = run_lbfgs_continuation(a_pert, tau_mid, deadline, tracker,
                                               max_iter_per_tau=200)
            J_val2 = compute_J(a_result2)
            print(f"  perturbed: J={J_val2:.8f}", flush=True)

        tracker.report(f"Phase 3 ({time.monotonic()-t0:.1f}s)")

    # ============ Phase 4: Coordinate descent polish ============
    if tracker.best is not None and time.monotonic() < deadline - 10:
        print(f"\n{'='*60}", flush=True)
        print("Phase 4: Coordinate descent polish", flush=True)
        print(f"{'='*60}", flush=True)

        n = tracker.best.size
        # Only do CD for small n (it's O(n^2) per iteration)
        if n <= 512:
            max_cd_iters = min(50, int((deadline - time.monotonic()) / 5))
            a_result = coordinate_descent_polish(tracker.best, max_cd_iters,
                                                 deadline, tracker)
        else:
            # For large n, do limited CD on a subset
            print(f"  Skipping CD for large n={n}, trying limited perturbation instead", flush=True)
            for _ in range(20):
                if time.monotonic() > deadline - 10:
                    break
                a_pert = tracker.best.copy()
                # Randomly perturb a few coordinates
                n_pert = min(20, n // 10)
                idx = rng.choice(n, size=n_pert, replace=False)
                for i in idx:
                    a_pert[i] *= rng.uniform(0.7, 1.3)
                a_pert = np.maximum(a_pert, 0.0)
                if a_pert.sum() > 0.01:
                    a_pert = a_pert / a_pert.sum()
                    tracker.update(a_pert)

        tracker.report("Phase 4")

    # ============ Phase 5: Final LP polish ============
    if tracker.best is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*60}", flush=True)
        print("Phase 5: Final LP polish", flush=True)
        print(f"{'='*60}", flush=True)

        n = tracker.best.size
        if n <= 256:
            iters = min(100, int((deadline - time.monotonic()) / 2))
            a_result = run_lp_mirror_descent(tracker.best, max(20, iters), deadline, tracker)
        elif n <= 512:
            iters = min(50, int((deadline - time.monotonic()) / 3))
            a_result = run_lp_mirror_descent(tracker.best, max(10, iters), deadline, tracker)

        tracker.report("Phase 5 (final)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
```
