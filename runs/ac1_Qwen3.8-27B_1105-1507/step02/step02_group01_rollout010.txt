        # Relaxation
        a_padded[:n] = (1 - relax) * a_padded[:n] + relax * a_new
        a = a_padded[:n].copy()
        
        # Track best
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build the convolution matrix A where (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    return A


def build_conv_matrix_fast(a):
    """Build the convolution matrix more efficiently."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    # For row k, column j: A[k,j] = a[k-j] if 0 <= k-j < n
    for j in range(n):
        # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k < j+n
        k_start = j
        k_end = j + n - 1
        A[k_start:k_end+1, j] = a[:n]
    return A


def solve_lp_for_g(a, M, max_sum=None):
    """
    Solve LP: maximize sum(g) subject to conv(a,g) <= M, g >= 0.
    Returns optimal g and its sum.
    """
    n = len(a)
    A = build_conv_matrix_fast(a)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        g = res.x
        g_sum = float(np.sum(g))
        return g, g_sum
    return None, None


def lp_alternate(a, max_time, max_iters=200, t_start=0.5, t_min=0.1):

        if time.monotonic() - start > max_time:
            break

        M = float(np.max(conv))

        if g is None:
            break
        
        if g_sum <= 1.0 + 1e-6:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0
        
        g = g / g_sum  # normalize to sum 1
        
        t = max(t_min, t_start * (1.0 - it / max(float(max_iters), 1)))

iteration to flatten self-convolution, LP alternating refinement, and

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def gs_iterate(a, n, n_iter, relax, seed=None):
    rng = np.random.default_rng(seed)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    target_mag = 1.0 / np.sqrt(2 * n - 1)
    a_ref = a.copy().astype(np.float64)
    best_a = a_ref.copy()
    best_flatness = float(np.max(np.convolve(a_ref, a_ref)) * (2 * n - 1))

    for it in range(n_iter):
        a_pad = np.zeros(N)
        a_pad[:n] = a_ref
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        phase = np.angle(A)
        A_new = np.where(mag > 1e-15, target_mag * np.exp(1j * phase), 0.0)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-12:
            break
        a_new = a_new / s
        a_ref = (1.0 - relax) * a_ref + relax * a_new
        conv = np.convolve(a_ref, a_ref)
        flatness = float(np.max(conv)) * (2 * n - 1)
        if flatness < best_flatness:

    n = len(a)
    best_a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        width = max(1, int(n * rng.uniform(0.01, 0.12)))
        pos = int(rng.integers(0, n - width + 1))
        sigma = 0.02 + 0.04 * rng.random()
        a_new = best_a.copy()
        a_new[pos:pos+width] *= np.exp(rng.normal(0, sigma, width))
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj

    return best_a, best_obj


def coordinate_refine(a, max_time, rng):
    n = len(a)
    best_a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    start = time.monotonic()
    factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.9, 1.1, 0.95, 1.05]

    while time.monotonic() - start < max_time:
        improved = False
        idx = list(range(n))
        rng.shuffle(idx)
        for i in idx:
            if time.monotonic() - start > max_time:
                break
            for f in factors:
                a_new = best_a.copy()
                a_new[i] *= f
                if a_new[i] < 0:
                    continue
                s = np.sum(a_new)
                if s < 1e-10:
                    continue
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_a = a_new.copy()
                    best_obj = obj
                    improved = True
                    break
        if not improved:
            break

    return best_a, best_obj


def gs_init(n, rng, n_iter=200):
    a = np.maximum(0, rng.normal(1.0, 0.3, n))
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a_final, _ = gs_iterate(a, n, n_iter=n_iter, relax=0.5, seed=int(rng.integers(0, 2**31)))
    return a_final


def hybrid_optimize(a_init, time_budget, rng):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: GS (aggressive)
    alloc1 = min(20.0, time_budget * 0.3)
    n_iter1 = max(100, int(alloc1 / 0.08))
    a1, obj1 = gs_iterate(a, n, n_iter=n_iter1, relax=0.5, seed=int(rng.integers(0, 2**31)))
    if obj1 < best_obj:
        best_a = a1.copy()
        best_obj = obj1

    # Phase 2: LP alternating
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.6:
        alloc2 = min(25.0, time_budget * 0.6 - elapsed)
        a2, obj2 = lp_alternate(best_a, max(1.0, alloc2))
        if obj2 < best_obj:
            best_a = a2.copy()
            best_obj = obj2

    # Phase 3: GS (fine, smaller relaxation)
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.75:
        alloc3 = min(15.0, time_budget * 0.75 - elapsed)
        n_iter3 = max(50, int(alloc3 / 0.08))
        a3, obj3 = gs_iterate(best_a, n, n_iter=n_iter3, relax=0.3, seed=int(rng.integers(0, 2**31)))
        if obj3 < best_obj:
            best_a = a3.copy()
            best_obj = obj3

    # Phase 4: LP again
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.88:
        alloc4 = min(15.0, time_budget * 0.88 - elapsed)
        a4, obj4 = lp_alternate(best_a, max(1.0, alloc4))
        if obj4 < best_obj:
            best_a = a4.copy()
            best_obj = obj4

    # Phase 5: Coordinate refine
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.94:
        alloc5 = min(8.0, time_budget * 0.94 - elapsed)
        a5, obj5 = coordinate_refine(best_a, max(0.5, alloc5), rng)
        if obj5 < best_obj:
            best_a = a5.copy()
            best_obj = obj5

    # Phase 6: Block perturbation
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.99:
        alloc6 = time_budget * 0.99 - elapsed
        a6, obj6 = block_perturb_refine(best_a, max(0.5, alloc6), rng)
        if obj6 < best_obj:
            best_a = a6.copy()
            best_obj = obj6

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 6.0

    best_seq = None
    best_score = np.inf

    # Try parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Optimize parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.size <= 500:
            time_remaining = deadline - time.monotonic()
            alloc = min(50.0, time_remaining * 0.12)
            p_opt, p_obj = hybrid_optimize(parent.copy(), alloc, rng)
            if p_obj < best_score:
                best_score = p_obj
                best_seq = p_opt.copy()
                print(f"Parent optimized: obj={p_obj:.6f}")
    except Exception as e:
        print(f"Parent opt error: {e}")

    # Main search over n values
    n_values = [150, 200, 250, 300, 350, 400, 500]

    for n in n_values:
        if time.monotonic() > deadline - 25:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        num_remaining = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining)
        time_per_n = min(time_per_n, 130.0)

        num_restarts = max(2, min(3, int(time_per_n / 35)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts, ~{time_per_restart:.0f}s each")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 25:
                break
            a_init = gs_init(n, rng, n_iter=150)
            a_opt, obj_opt = hybrid_optimize(a_init, time_per_restart, rng)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r}: obj={obj_opt:.6f}")

    # Final refinement
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s")

        for round_num in range(4):
            if time.monotonic() > deadline - 3:
                break
            alloc = min(remaining * 0.3, 35.0)
            a_ref, obj_ref = hybrid_optimize(best_seq.copy(), alloc, rng)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  Round {round_num}: obj={best_score:.6f} ** IMPROVED **")
            else:
                print(f"  Round {round_num}: obj={obj_ref:.6f}")
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break

        if time.monotonic() < deadline - 2:
            bp_time = deadline - time.monotonic() - 1.0
            a_bp, obj_bp = block_perturb_refine(best_seq, max(1.0, bp_time), rng)
            if obj_bp < best_score:
                best_score = obj_bp
                best_seq = a_bp.copy()
                print(f"  After BP: obj={best_score:.6f} ** IMPROVED **")

    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
