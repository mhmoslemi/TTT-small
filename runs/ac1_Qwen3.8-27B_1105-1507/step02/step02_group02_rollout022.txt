import numpy as np
from scipy.optimize import linprog, minimize

def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def spectral_init(n, seed, n_iter=150):
    """Generate initial sequence via Gerchberg-Saxton spectral shaping."""
    rng = np.random.default_rng(seed)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    a = rng.uniform(0.3, 1.5, n)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iter):
        a_p = np.zeros(N)
        a_p[:n] = a
        A = np.fft.rfft(a_p)
        S = a.sum()
        target = S / np.sqrt(2.0 * n)
        mags = np.abs(A)
        phases = A / np.maximum(mags, 1e-30)
        new_A = np.zeros_like(A)
        new_A[0] = S
        new_A[1:] = target * phases[1:]
        a_new = np.fft.irfft(new_A, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 0.01:
            break
        a_new = a_new / s_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_sc


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        return np.maximum(res.x, 0)
    return None

def smooth_obj(x, beta, n):
    """Log-sum-exp smoothed objective."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_n = x_pos / s
    b = fast_conv(x_n)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max + np.log(np.sum(np.exp(shifted))) / beta
    return float(2.0 * n * log_sum)

    for beta in betas:

        x0 = np.clip(best, 0.001, None)
        x0 = x0 / x0.sum()
        
        try:

            if res.x is not None and res.fun < 1e10:

                    if sc < best_score:

        except Exception:
