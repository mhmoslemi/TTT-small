"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with backtracking line search and adaptive M
2. Adam-style smoothed gradient descent with aggressive beta annealing
3. Multi-start perturbation search with diverse noise structures

Key improvements: analytical gradients for log-sum-exp, better step-size
adaptiveness, and more aggressive exploration of the solution space.

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        return res.x
    return None


def smooth_obj_grad(a, beta):
    """
    Log-sum-exp smoothed objective and gradient for normalized a.
    Objective: 2*n*(1/beta)*log(sum(exp(beta*c_k))) where c = conv(a,a).
    Returns (objective_value, gradient_vector).
    """
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    log_sum_val = c_max + np.log(exp_shifted.sum()) / beta
    obj = 2.0 * n * log_sum_val

    # Gradient: d(obj)/da_j = 4*n * sum_k p_k * a_{k-j}
    # where p_k = exp(beta*c_k) / sum(exp(beta*c_k))
    p = exp_shifted / exp_shifted.sum()

    # Compute cross-correlation: grad[j] = sum_k p[k] * a[k-j] for valid k-j
    # Using direct computation (O(n^2), fine for n<=1024)
    grad = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(j + n, 2 * n - 1)
        if k_lo < k_hi:
            grad[j] = np.dot(p[k_lo:k_hi], a[0:k_hi - k_lo])

    grad = 4.0 * n * grad
    return obj, grad


def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.argmax(cond)
    theta = cssv[k] / (k + 1)
    w = np.maximum(v - theta, 0)
    return w / w.sum()


def adam_smooth_descent(a, deadline, max_steps=50000):
    """
    Adam optimizer on the simplex with log-sum-exp smoothed objective.
    Uses aggressive beta annealing schedule.
    """
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = compute_score(a)

    beta0 = 50.0
    beta1 = 50000.0
    lr = 0.05
    beta1_decay = (beta1 / beta0) ** (1.0 / max_steps)

    step_count = 0
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        beta = beta0 * (beta1_decay ** step)
        obj, grad = smooth_obj_grad(a, beta)

        t += 1
        b1, b2 = 0.9, 0.999
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new)

        step_count += 1
        if step_count % 200 == 0 or step < 5:
            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                if step < 5 or score < 1.51:
                    print(f"    Adam step {step}: J={score:.6f}, beta={beta:.0f}", flush=True)
        a = a_new

    # Final check
    score = compute_score(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def lp_iteration_v2(a, max_iters, deadline, m_scale=1.0):
    """
    LP best-response with backtracking line search.
    m_scale: multiplier on M to give LP more/less freedom.
    """
    best_local = a.copy()
    best_score = compute_score(a)
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max()) * m_scale
        g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 1.0
        a_old_score = compute_score(a)
        improved = False
        while t >= 0.01:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            s = a_new.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_local = a_new.copy()
                a = a_new
                improved = True
                break
            t *= 0.5

        if not improved:
            # Try very small step
            t = 0.01
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                score = compute_score(a_new)
                if score < best_score:
                    best_score = score
                    best_local = a_new.copy()
                a = a_new
            if t < 0.01:
                break

    return best_local, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t_arr = np.linspace(0, 1, n)

    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t_arr, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(('parent', resampled / resampled.sum()))

    inits.append(('uniform', np.ones(n) / n))

    # Various structured inits
    for freq, amp in [(1, 0.3), (2, 0.3), (3, 0.2), (1, 0.5), (2, 0.5), (4, 0.15)]:
        c = 1.0 + amp * np.cos(2 * np.pi * freq * t_arr)
        c = np.maximum(c, 0.01)
        inits.append((f'cos{freq}_{amp}', c / c.sum()))

    # Sin-based
    for freq in [1, 2, 3]:
        c = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t_arr)
        c = np.maximum(c, 0.01)
        inits.append((f'sin{freq}', c / c.sum()))

    # Asymmetric (key insight: asymmetry helps reduce center peak)
    c = 1.0 + 0.4 * (t_arr - 0.5)
    c = np.maximum(c, 0.01)
    inits.append(('asym_lin', c / c.sum()))

    c = 1.0 + 0.6 * (t_arr - 0.5) ** 2
    inits.append(('quad', c / c.sum()))

    # Bimodal
    c = 1.0 + 0.5 * np.cos(4 * np.pi * t_arr)
    c = np.maximum(c, 0.01)
    inits.append(('bimodal', c / c.sum()))

    # Random
    r = rng.uniform(0.2, 1.0, n)
    inits.append(('random', r / r.sum()))

    # Sparse-like
    c = np.ones(n) * 0.1
    c[rng.choice(n, size=n // 4, replace=False)] = 1.0
    inits.append(('sparse', c / c.sum()))

    return inits

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)

    a_new = np.maximum(np.interp(t_new, t_old, a), 0)
    if a_new.sum() > 0.01:

    return a_new

                print(f"  New best (Adam) at n={n2}: J={sc:.6f}", flush=True)

    print(f"  Phase 5 done: best={best_score:.6f}", flush=True)

    # === PHASE 6: Final intensive optimization ===
    print(f"\n=== PHASE 6: Final Optimization ===", flush=True)
    final_deadline = deadline - 5.0
    remaining = final_deadline - time.monotonic()

    if remaining > 10:
        # Alternate LP and Adam
        for round in range(20):
            if time.monotonic() > final_deadline:
                break

            # LP step
            est_per_iter = 0.04 if n <= 512 else 0.08
            max_iters = max(50, int(remaining * 0.3 / est_per_iter))
            a_res, sc = lp_iteration_v2(best, max_iters, final_deadline, m_scale=1.0)
            if sc < best_score:
                best_score = sc
                best = a_res.copy()
                print(f"  Final round {round}: LP J={sc:.6f}", flush=True)

            # Adam step
            if time.monotonic() < final_deadline:
                a_res, sc = adam_smooth_descent(best, final_deadline, max_steps=20000)
                if sc < best_score:
                    best_score = sc
                    best = a_res.copy()
                    print(f"  Final round {round}: Adam J={sc:.6f}", flush=True)

            remaining = final_deadline - time.monotonic()
            if remaining < 5:
                break

    # Final validation
    if best is None:
        best = np.ones(256) / 256

    best = np.clip(best, 0.0, 1000.0)
    best_score = compute_score(best)

    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: J={best_score:.6f}, n={len(best)}", flush=True)
    return [float(x) for x in best]
