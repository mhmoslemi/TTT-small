from scipy.optimize import linprog
from scipy.sparse import csr_matrix

    n = len(f)

    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:

    rows = []
    cols = []
    vals = []

        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])

"""
LP-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming direction-finding with adaptive line search,
subgradient kick-starts, and coordinate descent polishing.
Key insight: optimal sequences must be asymmetric to beat the symmetric bound of 2.
Focuses computational budget on n=1024 with a secondary n=2048 attempt.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Compute 2n*max_conv/sum^2 directly."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp(f, M):
    """Solve: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def subgradient_step(f, lr, top_k=5):
    """One subgradient step to reduce the peak convolution."""
    n = len(f)
    conv = np.convolve(f, f)
    top_indices = np.argsort(conv)[-top_k:]
    weights = np.maximum(conv[top_indices] - conv.mean(), 0)
    if weights.sum() < 1e-15:
        weights = np.ones(top_k)
    w_norm = weights / weights.sum()

    grad = np.zeros(n)
    for k, w in zip(top_indices, w_norm):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * w * f[k - j]

    S = float(np.sum(f))
    f_new = f - lr * grad
    f_new = np.clip(f_new, 0.0, None)
    new_S = float(np.sum(f_new))
    if new_S < 1e-10:
        return f.copy()
    return f_new * (S / new_S)

    f = f.copy().astype(np.float64)

                if sc < best_cand_score:
                    best_cand_score = sc
                    best_cand_val = cv
            f[i] = best_cand_val

        it += 1
        new_score = compute_score(f)
        if new_score < best_score:
            best_score = new_score
            best_f = f.copy()
            if verbose_prefix and it % 10 == 0:
                print(f"    {verbose_prefix}CD pass {it}: score={best_score:.6f}")
        else:
            # Reset to best and break if no improvement for 3 passes
            if it % 3 == 0:
                break

    return best_f, best_score


def make_ramp_init(n, alpha=1.0):
    """Create an asymmetric ramp initialization."""
    idx = np.arange(n, dtype=np.float64)
    # Exponential ramp (heavier on one side)
    return np.exp(alpha * idx / n)


def make_bimodal_asym(n):
    """Asymmetric bimodal initialization."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-((idx - 0.3 * n) / (0.2 * n)) ** 2) + \
           0.5 * np.exp(-((idx - 0.7 * n) / (0.15 * n)) ** 2)


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence minimizing the objective."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 3.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(123)

    # Try to load parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            sc = evaluate_sequence(parent.tolist())
            if np.isfinite(sc):
                best_score = sc
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={sc:.6f}")
    except (NameError, Exception) as e:
        print(f"No parent available: {e}")

    # Define candidate (n, init) pairs to try
    n_primary = 1024
    n_secondary = 2048

    candidate_inits = []

    # Primary length with multiple inits
    if parent is not None and len(parent) == n_primary:
        candidate_inits.append((n_primary, 'parent', parent.copy()))
    elif parent is not None:
        # Interpolate parent to n_primary
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n_primary)
        parent_interp = np.interp(x_new, x_old, parent)
        candidate_inits.append((n_primary, 'parent_interp', parent_interp))

    # Additional initializations for primary length
    candidate_inits.append((n_primary, 'exp_ramp', make_ramp_init(n_primary, alpha=2.0)))
    candidate_inits.append((n_primary, 'exp_ramp2', make_ramp_init(n_primary, alpha=1.0)))
    candidate_inits.append((n_primary, 'bimodal_asym', make_bimodal_asym(n_primary)))
    candidate_inits.append((n_primary, 'linear_ramp', np.linspace(0.1, 2.0, n_primary)))
    # Random init
    candidate_inits.append((n_primary, 'random', rng.uniform(0.1, 1.0, n_primary)))

    # Secondary length
    if parent is not None:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n_secondary)
        candidate_inits.append((n_secondary, 'parent_interp2', np.interp(x_new, x_old, parent)))
    candidate_inits.append((n_secondary, 'exp_ramp_2k', make_ramp_init(n_secondary, alpha=2.0)))

    # Evaluate initializations quickly
    scored_inits = []
    for n, label, f_init in candidate_inits:
        f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
        if np.sum(f_init) < 1e-10:
            continue
        sc = compute_score(f_init)
        if np.isfinite(sc):
            scored_inits.append((sc, n, label, f_init.copy()))
            if sc < best_score:
                best_score = sc
                best_sequence = f_init.copy()
            print(f"  Init n={n} {label}: score={sc:.6f}")

    # Sort by score
    scored_inits.sort(key=lambda x: x[0])
    print(f"\nBest init: {scored_inits[0][2]} (n={scored_inits[0][1]}): {scored_inits[0][0]:.6f}")
    print(f"Overall best so far: {best_score:.6f}")

    # Phase 1: Quick subgradient warmup on top 3 inits (60 seconds)
    print("\n=== Phase 1: Subgradient Warmup ===")
    warmup_deadline = t0 + 60.0
    warmed_up = []
    for sc, n, label, f_init in scored_inits[:3]:
        if time.monotonic() > warmup_deadline - 5:
            break
        # Quick subgradient optimization
        f_warm = f_init.copy()
        S_warm = float(np.sum(f_warm))
        lr = 0.005
        for _ in range(200):
            if time.monotonic() > warmup_deadline - 5:
                break
            f_warm = subgradient_step(f_warm, lr, top_k=5)
            f_warm = np.clip(f_warm, 0.0, None)
            new_S = float(np.sum(f_warm))
            if new_S < 1e-10:
                break
            f_warm = f_warm * (S_warm / new_S)
            lr *= 0.995

        sc_warm = compute_score(f_warm)
        warmed_up.append((sc_warm, n, label, f_warm.copy()))
        if sc_warm < best_score:
            best_score = sc_warm
            best_sequence = f_warm.copy()
        print(f"  Warmup {label}: {sc:.6f} -> {sc_warm:.6f}")

    # Sort warmed up by score
    warmed_up.sort(key=lambda x: x[0])

    # Phase 2: Main LP optimization on best candidate (majority of time)
    print("\n=== Phase 2: Main LP Optimization ===")
    # Allocate time: 70% to best, 20% to second, 10% to third
    remaining = deadline - time.monotonic()
    lp_allocations = [
        (remaining * 0.55, warmed_up[0]),
    ]
    if len(warmed_up) > 1 and remaining * 0.35 > 30:
        lp_allocations.append((remaining * 0.25, warmed_up[1]))
    if len(warmed_up) > 2 and remaining * 0.15 > 20:
        lp_allocations.append((remaining * 0.10, warmed_up[2]))

    for alloc_time, (sc, n, label, f_init) in lp_allocations:
        if time.monotonic() > deadline - 30:
            break
        lp_deadline = min(t0 + t0 + alloc_time, deadline - 60)
        # Make sure we don't overshoot
        lp_deadline = min(lp_deadline, deadline - 60)
        if time.monotonic() > lp_deadline - 10:
            continue

        print(f"\n  LP optimize: n={n} {label} (start={sc:.6f}, time≈{alloc_time:.0f}s)")
        f_opt, s_opt = optimize_lp(f_init, lp_deadline, step_init=0.12,
                                    verbose_prefix=f"[{label}]")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  Improved to: {best_score:.6f}")

    # Phase 3: Coordinate descent polish (remaining time)
    print(f"\n=== Phase 3: CD Polish (best={best_score:.6f}) ===")
    if best_sequence is not None and time.monotonic() < deadline - 15:
        # Polish the best sequence
        n_best = len(best_sequence)
        cd_deadline = deadline - 5
        # Split: 60% for best, 40% for trying a fresh n=1024 or 2048
        cd1_deadline = min(time.monotonic() + (cd_deadline - time.monotonic()) * 0.5,
                           deadline - 20)
        f_pol, s_pol = coordinate_descent_polish(best_sequence, cd1_deadline,
                                                  verbose_prefix="[CD-best]")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()
            print(f"  CD polish improved to: {best_score:.6f}")

        # Try LP on a fresh init at the best length
        if time.monotonic() < deadline - 30:
            fresh_init = make_ramp_init(n_best, alpha=2.5)
            fresh_init += rng.uniform(0, 0.1, n_best)
            fresh_sc = compute_score(fresh_init)
            if fresh_sc < best_score * 1.1:
                print(f"  Fresh init (ramp+noise): {fresh_sc:.6f}")
                f_opt, s_opt = optimize_lp(fresh_init, deadline - 10, step_init=0.1)
                if s_opt < best_score:
                    best_score = s_opt
                    best_sequence = f_opt.copy()

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(1024)
        best_score = compute_score(best_sequence)

    # Clip and verify
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed! Using fallback.")
        best_sequence = np.ones(512)
        final_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
